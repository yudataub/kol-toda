👑 חִידוֹנֵי סֵפֶר שְׁמוּאֵל

51 חִידוֹנִים לְפִי תָּכְנִית הַלִּמּוּדִים שֶׁל מִשְׂרַד הַחִנּוּךְ (תַּנַ״ךְ חֶמֶ״ד), לְכִתּוֹת ה׳–ו׳. בְּכָל חִידוֹן כַּרְטִיסֵי לִמּוּד עִם אַהֲבַת ה׳ 💛 וְשָׁלוֹשׁ רָמוֹת: תְּמוּנוֹת, שְׁאֵלוֹת לַחֲכָמִים, וּבָרָמָה הַשְּׁלִישִׁית מִשְׂחָקִים – בְּכָל פַּעַם מִשְׂחָק אַחֵר

🙏 שְׁמוּאֵל א׳ – חַנָּה וּשְׁמוּאֵל

1. אֶלְקָנָה וּמִשְׁפַּחְתּוֹפֶּרֶק א׳, פְּסוּקִים א׳–ח׳
2. תְּפִלַּת חַנָּהפֶּרֶק א׳, פְּסוּקִים ט׳–כ״ח
3. שִׁירַת חַנָּהפֶּרֶק ב׳, פְּסוּקִים א׳–י״א
4. בְּנֵי עֵלִי וְהַכֹּהֵן הַנֶּאֱמָןפֶּרֶק ב׳, פְּסוּקִים י״ב–ל״ו
5. ׳דַּבֵּר כִּי שֹׁמֵעַ עַבְדֶּךָ׳ – ה׳ קוֹרֵא לִשְׁמוּאֵלפֶּרֶק ג׳, פְּסוּקִים א׳–כ״א

📦 שְׁמוּאֵל א׳ – אֲרוֹן ה׳

6. הַמִּלְחָמָה בְּאֶבֶן הָעֵזֶר – לֹא הָאָרוֹן מוֹשִׁיעַפֶּרֶק ד׳, פְּסוּקִים א׳–י״א
7. חֻרְבַּן שִׁילֹה – ׳גָּלָה כָבוֹד מִיִּשְׂרָאֵל׳פֶּרֶק ד׳, פְּסוּקִים י״ב–כ״ב
8. אֲרוֹן ה׳ בְּאֶרֶץ פְּלִשְׁתִּים – ׳וַתַּעַל שַׁוְעַת הָעִיר הַשָּׁמָיִם׳פֶּרֶק ה׳
9. הָאָרוֹן שָׁב – ׳מִי יוּכַל לַעֲמֹד לִפְנֵי ה׳ הָאֱלֹהִים הַקָּדוֹשׁ הַזֶּה׳פֶּרֶק ו׳, וּפֶרֶק ז׳ פְּסוּקִים א׳–ב׳
10. מַעֲמַד הַתְּשׁוּבָה בַּמִּצְפָּהסֵפֶר שְׁמוּאֵל א׳, פֶּרֶק ז׳, פְּסוּקִים ג׳–י״ז

👑 שְׁמוּאֵל א׳ – הַמְלָכַת שָׁאוּל

11. שִׂימָה לָּנוּ מֶלֶךְסֵפֶר שְׁמוּאֵל א׳, פֶּרֶק ח׳, פְּסוּקִים א׳–ט׳
12. מִשְׁפַּט הַמֶּלֶךְסֵפֶר שְׁמוּאֵל א׳, פֶּרֶק ח׳, פְּסוּקִים י׳–כ״ב
13. שָׁאוּל וְהָאֲתוֹנוֹתפֶּרֶק ט׳, פְּסוּקִים א׳–כ״א
14. שָׁאוּל נָגִיד עַל עַמּוֹפֶּרֶק ט׳, פְּסוּקִים כ״ב–כ״ז; פֶּרֶק י׳, פְּסוּקִים א׳–ט״ז
15. יְחִי הַמֶּלֶךְפֶּרֶק י׳, פְּסוּקִים י״ז–כ״ז
16. נָחָשׁ הָעַמּוֹנִי וְיָבֵשׁ גִּלְעָדפֶּרֶק י״א
17. נְאוּם שְׁמוּאֵלפֶּרֶק י״ב

⚔️ שְׁמוּאֵל א׳ – מִלְחֲמוֹת שָׁאוּל

18. שָׁאוּל בַּגִּלְגָּלפֶּרֶק י״ג, פְּסוּקִים א׳–י״ד
19. מִכְמָשׂ: יְהוֹנָתָן וְנֹשֵׂא כֵלָיו – ׳אֵין לַה׳ מַעְצוֹר לְהוֹשִׁיעַ׳פֶּרֶק י״ג, פְּסוּקִים ט״ו–כ״ג; פֶּרֶק י״ד, פְּסוּקִים א׳–ט״ז
20. הַשְּׁבוּעָה וְיַעְרַת הַדְּבָשׁ – ׳כִּי עִם אֱלֹהִים עָשָׂה הַיּוֹם הַזֶּה׳פֶּרֶק י״ד, פְּסוּקִים י״ז–מ״ו
21. צַו הַמִּלְחָמָה בַּעֲמָלֵק – ׳שְׁמַע לְקוֹל דִּבְרֵי ה׳׳פֶּרֶק ט״ו, פְּסוּקִים א׳–ט׳
22. שְׁמֹעַ מִזֶּבַח טוֹב – שְׁמוּאֵל וְשָׁאוּל בַּגִּלְגָּלפֶּרֶק ט״ו, פְּסוּקִים י׳–ל״ה

🎵 שְׁמוּאֵל א׳ – דָּוִד בְּבֵית שָׁאוּל

23. מְשִׁיחַת דָּוִד וְדָוִד הַמְנַגֵּן – ׳וַה׳ יִרְאֶה לַלֵּבָב׳פֶּרֶק ט״ז
24. אֶתְגַּר גָּלְיָת – ׳כִּי חֵרֵף מַעַרְכוֹת אֱלֹהִים חַיִּים׳פֶּרֶק י״ז, פְּסוּקִים א׳–ל״א
25. דָּוִד וְגָלְיָתפֶּרֶק י״ז, פְּסוּקִים ל״ב–נ״ח
26. בְּרִית יְהוֹנָתָן וְדָוִדפֶּרֶק י״ח, פְּסוּקִים א׳–ט׳
28. דָּוִד נִמְלָט מִשָּׁאוּל – ׳וַיִּבְרַח וַיִּמָּלֵט׳פֶּרֶק י״ט

🏜️ שְׁמוּאֵל א׳ – נְדוּדֵי דָּוִד

29. הַחֵץ וְהָאֶבֶן הָאָזֶל – בְּרִית דָּוִד וִיהוֹנָתָןפֶּרֶק כ׳
30. לֶחֶם הַפָּנִים בְּנֹב וּמְעָרַת עֲדֻלָּםפֶּרֶק כ״א; פֶּרֶק כ״ב, פְּסוּקִים א׳–ה׳
31. כֹּהֲנֵי נֹב: ׳אָנֹכִי סַבֹּתִי׳ – דָּוִד לוֹקֵחַ אַחֲרָיוּתפֶּרֶק כ״ב, פְּסוּקִים ו׳–כ״ג
32. דָּוִד מַצִּיל אֶת קְעִילָה: ׳הַגֶּד נָא לְעַבְדֶּךָ׳פֶּרֶק כ״ג, פְּסוּקִים א׳–כ״ח
34. נָבָל וַאֲבִיגַיִל – ׳בָּרוּךְ טַעְמֵךְ׳פֶּרֶק כ״ה
35. הַחֲנִית וְצַפַּחַת הַמַּיִם – ׳מִי שָׁלַח יָדוֹ בִּמְשִׁיחַ ה׳ וְנִקָּה׳פֶּרֶק כ״ו
36. דָּוִד בִּשְׂדֵה פְלִשְׁתִּים – גַּת וְצִקְלַגפֶּרֶק כ״ז

⛰️ שְׁמוּאֵל א׳ – הַגִּלְבֹּעַ וְסִכּוּם

37. שָׁאוּל בְּעֵין דּוֹר – ׳תָּמִים תִּהְיֶה עִם ה׳ אֱלֹהֶיךָ׳פֶּרֶק כ״ח
38. דָּוִד אֵינוֹ יוֹצֵא לַמִּלְחָמָה – ׳הָשֵׁב אֶת הָאִישׁ׳פֶּרֶק כ״ט

👑 שְׁמוּאֵל ב׳ – דָּוִד מוֹלֵךְ

1. הַקִּינָה עַל שָׁאוּל וִיהוֹנָתָןפֶּרֶק א׳, פְּסוּקִים א׳–כ״ז
2. דָּוִד מֶלֶךְ בְּחֶבְרוֹןפֶּרֶק ב׳, פְּסוּקִים א׳–י״א
4. אַבְנֵר פּוֹנֶה אֶל דָּוִדפֶּרֶק ג׳, פְּסוּקִים א׳–כ״א
5. שַׂר וְגָדוֹל נָפַלפֶּרֶק ג׳, פְּסוּקִים כ״ב–ל״ט
7. מֶלֶךְ עַל כָּל יִשְׂרָאֵל בִּירוּשָׁלַיִם – ׳וַה׳ אֱלֹהֵי צְבָאוֹת עִמּוֹ׳פֶּרֶק ה׳
8. אֲרוֹן ה׳ בַּדֶּרֶךְ – ׳אֵיךְ יָבוֹא אֵלַי אֲרוֹן ה׳׳פֶּרֶק ו׳, פְּסוּקִים א׳–י״א
9. מְפַזֵּז וּמְכַרְכֵּר לִפְנֵי ה׳ – ׳לִפְנֵי ה׳ אֲשֶׁר בָּחַר בִּי׳פֶּרֶק ו׳, פְּסוּקִים י״ב–כ״ב

🏛️ שְׁמוּאֵל ב׳ – בִּסּוּס הַמַּלְכוּת

10. בַּיִת לְדָוִד וּבַיִת לַה׳פֶּרֶק ז׳
11. מִשְׁפָּט וּצְדָקָה לְכָל עַמּוֹפֶּרֶק ח׳
13. חֲזַק וְנִתְחַזַּק – חֶסֶד דָּוִד וּמִלְחֶמֶת עַמּוֹן וַאֲרַםפֶּרֶק י׳

💔 שְׁמוּאֵל ב׳ – חֵטְא וּתְשׁוּבָה

16. אַהֲבָה הַתְּלוּיָה בְדָבָרפֶּרֶק י״ג, פְּסוּקִים א׳–כ״ב
17. בַּעַל חָצוֹר וּבְרִיחָה לִגְשׁוּרפֶּרֶק י״ג, פְּסוּקִים כ״ג–ל״ט

🌳 שְׁמוּאֵל ב׳ – מֶרֶד אַבְשָׁלוֹם

19. וַיְגַנֵּב אַבְשָׁלוֹם אֶת לֵב אַנְשֵׁי יִשְׂרָאֵלפֶּרֶק ט״ו, פְּסוּקִים א׳–י״ב
22. עֲצַת אֲחִיתֹפֶל וַעֲצַת חוּשַׁיפֶּרֶק ט״ז, פְּסוּקִים ט״ו–י״ט, כ״ג; פֶּרֶק י״ז, פְּסוּקִים א׳–י״ד
25. הֲשָׁלוֹם לַנַּעַר לְאַבְשָׁלוֹם? – אֲחִימַעַץ וְהַכּוּשִׁי רָצִים אֶל דָּוִדפֶּרֶק י״ח, פְּסוּקִים י״ט–ל״ב

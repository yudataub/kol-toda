מַה יָּפוּ פְעָמַיִךְ

רָמָה א׳ – נֶאֱמָנוּת וַעֲלִיָּה

עֲלִיַּת הָרֶגֶל וּנְאֱמָנוּת בַּגָּלוּת: הָאֻמּוֹת מְפַתּוֹת, וְיִשְׂרָאֵל עוֹנִים ׳אֲנִי לְדוֹדִי וְעָלַי תְּשׁוּקָתוֹ׳ וּמַשְׁכִּימִים לְבָתֵּי הַמִּדְרָשׁ

🎯 לַחִידוֹן

💛 אַהֲבַת הַשֵּׁם: שׁוּם כָּבוֹד שֶׁבָּעוֹלָם לֹא שָׁוֶה אֶת הַקֶּשֶׁר שֶׁלִּי עִם ה׳ 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ז׳, פָּסוּק א׳

רָמָה א׳ – נֶאֱמָנוּת וַעֲלִיָּה

📣 ׳שׁוּבִי שׁוּבִי הַשּׁוּלַמִּית׳

בַּגָּלוּת הָאֻמּוֹת קוֹרְאוֹת לְיִשְׂרָאֵל. רַשִׁ״י: ׳אוֹמְרִים אֵלַי: ״שׁוּבִי שׁוּבִי מֵאַחַר הַמָּקוֹם״׳ – כְּלוֹמַר, חִזְרוּ מִלֶּכֶת אַחֲרֵי ה׳. ׳הַשּׁוּלַמִּית׳ הִיא ׳הַשְּׁלֵמָה בֶאֱמוּנָתֵךְ עִמּוֹ׳. הֵם מַבְטִיחִים מְשִׂרוֹת וּגְדֻלָּה: ׳וְנֶחֱזֶה בָּךְ – נַצִּיב מִמֵּךְ נְצִיבִים וְשִׁלְטוֹנִים׳, וְיִשְׂרָאֵל עוֹנִים: ׳מַה תֶּחֱזוּ בַּשּׁוּלַמִּית׳?

הָאֻמּוֹת הִבְטִיחוּ לְיִשְׂרָאֵל כָּבוֹד אִם יַעַזְבוּ אֶת ה׳, וְיִשְׂרָאֵל סֵרְבוּ.

1

💛 אַהֲבַת הַשֵּׁם: כָּל צַעַד שֶׁאֲנִי עוֹשֶׂה לְכִוּוּן ה׳ יָפֶה בְּעֵינָיו 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ז׳, פָּסוּק ב׳

רָמָה א׳ – נֶאֱמָנוּת וַעֲלִיָּה

👣 ׳מַה יָּפוּ פְעָמַיִךְ בַּנְּעָלִים׳

רַשִׁ״י: ׳מַה יָּפוּ פְעָמַיִךְ בַּנְּעָלִים׳ הוּא ׳בַּעֲלִיַּת הָרְגָלִים, בַּת נְדִיבִים׳. שָׁלוֹשׁ פְּעָמִים בַּשָּׁנָה – פֶּסַח, שָׁבוּעוֹת וְסֻכּוֹת – יִשְׂרָאֵל עוֹלִים בְּרֶגֶל לִירוּשָׁלַיִם לְבֵית הַמִּקְדָּשׁ. הַפְּסִיעוֹת שֶׁלָּהֶם נָאוֹת כִּי הֵן הוֹלְכוֹת אֶל ה׳.

בְּכָל שָׁלוֹשׁ רְגָלִים עוֹלִים לִירוּשָׁלַיִם, וְהַהֲלִיכָה אֶל בֵּית ה׳ יָפָה בְּעֵינֵי ה׳.

2

💛 אַהֲבַת הַשֵּׁם: הַמִּצְווֹת שֶׁלִּי קְשׁוּרוֹת לְבִרְיַת הָעוֹלָם כֻּלּוֹ 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ז׳, פָּסוּק ב׳

רָמָה א׳ – נֶאֱמָנוּת וַעֲלִיָּה

💧 ׳חַמּוּקֵי... כְּמוֹ חֲלָאִים׳ – הַשִּׁיתִין

רַשִׁ״י מַזְכִּיר שֶׁרַבּוֹתֵינוּ דָרְשׁוּ אֶת הַפָּסוּק ׳עַל נִקְבֵי הַשִּׁיתִין שֶׁל נְסָכִים שֶׁנַּעֲשׂוּ מִשֵּׁשֶׁת יְמֵי בְרֵאשִׁית׳. אֵלּוּ חוֹרִים בַּמִּזְבֵּחַ שֶׁדֶּרֶכָּם יָרְדוּ הַיַּיִן וְהַמַּיִם שֶׁנִּסְּכוּ. וְ׳מַעֲשֵׂה יְדֵי אָמָּן׳ הֵם ׳יְדֵי הַקָּדוֹשׁ בָּרוּךְ הוּא׳.

בַּמִּזְבֵּחַ הָיוּ חוֹרִים לַנְּסָכִים שֶׁנִּבְרְאוּ מִשֵּׁשֶׁת יְמֵי בְרֵאשִׁית.

3

💛 אַהֲבַת הַשֵּׁם: הַחָכְמָה שֶׁל הַתּוֹרָה יְכוֹלָה לְהַפְתִּיעַ גַּם אֶת מִי שֶׁאֵינוֹ מְאַמִּין 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ז׳, פָּסוּק ה׳

רָמָה א׳ – נֶאֱמָנוּת וַעֲלִיָּה

🔭 ׳עֵינַיִךְ בְּרֵכוֹת בְּחֶשְׁבּוֹן׳

רַשִׁ״י: ׳חֲכָמַיִךְ, כְּשֶׁהֵם יוֹשְׁבִים בְּשַׁעֲרֵי יְרוּשָׁלַיִם... וַעֲסוּקִים בְּחֶשְׁבּוֹן תְּקוּפוֹת וּמַזָּלוֹת, חָכְמָתָם וּבִינָתָם לְעֵינֵי הָעַמִּים׳. הַחֲכָמִים שֶׁקּוֹבְעִים אֶת הַלּוּחַ מְחַשְּׁבִים אֶת הַשָּׁמַיִם, וְהָאֻמּוֹת מִסְתַּכְּלוֹת בְּהִתְפַּעֲלוּת. וְ׳צוֹפֶה פְּנֵי דַמָּשֶׂק׳ – יְרוּשָׁלַיִם עֲתִידָה לְהִתְרַחֵב.

חֲכָמֵי יִשְׂרָאֵל בַּשַּׁעַר מְחַשְּׁבִים אֶת הַלּוּחַ, וְהָעַמִּים רוֹאִים אֶת חָכְמָתָם.

4

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי רָץ לְקַיֵּם מִצְוָה, ה׳ מִתְקַשֵּׁר אֵלַי 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ז׳, פָּסוּק ו׳

רָמָה א׳ – נֶאֱמָנוּת וַעֲלִיָּה

🧿 ׳רֹאשֵׁךְ עָלַיִךְ כַּכַּרְמֶל׳

רַשִׁ״י מְבָאֵר: ׳אֵלּוּ תְּפִלִּין שֶׁבָּרֹאשׁ שֶׁנֶּאֱמַר בָּהֶן: ״וְרָאוּ כָּל עַמֵּי הָאָרֶץ כִּי שֵׁם ה׳ נִקְרָא עָלֶיךָ וְיָרְאוּ מִמֶּךָ״׳. וְ׳מֶלֶךְ אָסוּר בָּרְהָטִים׳ – ה׳ ׳נִקְשָׁר בְּאַהֲבָה, בְּמִצְוֹת, וּבְרִיצוֹת שֶׁאַתֶּם רָצִים לְפָנָיו׳. כְּשֶׁאָנוּ רָצִים לַמִּצְוָה, ה׳ נִקְשָׁר אֵלֵינוּ.

הַתְּפִלִּין הֵם סִימָן לְכָל הָעוֹלָם שֶׁשֵּׁם ה׳ נִקְרָא עָלֵינוּ.

5

💛 אַהֲבַת הַשֵּׁם: ה׳ גֵּאֶה בִּי כְּשֶׁאֲנִי עוֹמֵד זָקוּף לִכְבוֹדוֹ 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ז׳, פְּסוּקִים ח׳–ט׳

רָמָה א׳ – נֶאֱמָנוּת וַעֲלִיָּה

🌴 ׳זֹאת קוֹמָתֵךְ דָּמְתָה לְתָמָר׳

רַשִׁ״י: ׳רָאִינוּ נוֹי קוֹמָתֵךְ בִּימֵי נְבוּכַדְנֶצַּר, שֶׁכָּל הָאֻמּוֹת הָיוּ כּוֹרְעוֹת וְנוֹפְלוֹת... וְאַתָּה עָמַדְתָּ בְּקוֹמָה זְקוּפָה כַּתָּמָר הַזֶּה׳. ׳דָּנִיֵּאל, חֲנַנְיָה, מִישָׁאֵל, וַעֲזַרְיָה׳ הֵם הָאַשְׁכֹּלוֹת: ׳הִשְׁפִּיעוּ לְהָנִיק וּלְלַמֵּד אֶת הַכֹּל שֶׁאֵין יִרְאָה כְּיִרְאַתְכֶם׳. ה׳ גַּם ׳מִתְפָּאֵר בֵּין חֲיָלוֹת שֶׁל מַעְלָה בָכֶם׳.

בִּימֵי נְבוּכַדְנֶצַּר כֻּלָּם כָּרְעוּ, וְיִשְׂרָאֵל עָמְדוּ זְקוּפִים כְּמוֹ תָּמָר.

6

💛 אַהֲבַת הַשֵּׁם: ה׳ מִתְגַּעְגֵּעַ אֵלַי כְּמוֹ שֶׁאֲנִי מִתְגַּעְגֵּעַ אֵלָיו 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ז׳, פָּסוּק י״א

רָמָה א׳ – נֶאֱמָנוּת וַעֲלִיָּה

💞 ׳אֲנִי לְדוֹדִי וְעָלַי תְּשׁוּקָתוֹ׳

רַשִׁ״י: ׳זְהִירָה אֲנִי לְהָשִׁיב לָהֶם שֶׁאֶעֱמֹד בֶּאֱמוּנָתִי, שֶׁיְּהֵא חִכִּי הוֹלֵךְ לִפְנֵי דוֹדִי לְאַהֲבַת מִישׁוֹר, שֶׁהוּא מִן הַלֵּב, וְלֹא בִרְמִיָּה וַעֲקִיבָה׳. הַתְּשׁוּבָה הִיא: ׳אֲנִי לְדוֹדִי וְגַם הוּא מִשְׁתּוֹקֵק לִי׳. ה׳ לֹא רַק אָהוּב – הוּא מִתְגַּעְגֵּעַ אֵלֵינוּ גַּם הוּא.

יִשְׂרָאֵל עוֹנִים לָאֻמּוֹת: אֲנִי שֶׁל ה׳ וְה׳ מִשְׁתּוֹקֵק אֵלַי.

7

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַשְׁכִּים לְהִתְפַּלֵּל וְלִלְמֹד, כִּי ה׳ מַמְתִּין לִי שָׁם 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ז׳, פְּסוּקִים י״ב–י״ג

רָמָה א׳ – נֶאֱמָנוּת וַעֲלִיָּה

🍇 ׳נַשְׁכִּימָה לַכְּרָמִים׳

רַשִׁ״י: כְּנֶסֶת יִשְׂרָאֵל אוֹמֶרֶת לַה׳: ׳אַל תְּדִינֵנִי כְּיוֹשְׁבֵי כְרַכִּים... אֶלָּא כְּיוֹשְׁבֵי פְרָזוֹת שֶׁהֵם בַּעֲלֵי אֻמָּנוּת וְעוֹסְקִין בַּתּוֹרָה מִתּוֹךְ הַדְּחָק׳. ׳נַשְׁכִּימָה לַכְּרָמִים׳ הֵם ׳בָּתֵּי כְנֵסִיּוֹת וּבָתֵּי מִדְרָשׁוֹת׳, וְ׳בַּעֲלֵי מִקְרָא׳ וּ׳בַּעֲלֵי מִשְׁנָה׳ וּ׳בַּעֲלֵי תַלְמוּד׳ הֵם הַגֶּפֶן, הַסְּמָדַר וְהָרִמּוֹנִים.

יִשְׂרָאֵל מַשְׁכִּימִים לְבָתֵּי הַכְּנֶסֶת וְהַמִּדְרָשׁ, וְה׳ רוֹאֶה כַּמָּה הֵם גְּדֵלִים בְּתוֹרָה.

8
💡

רָמָה א׳ – נֶאֱמָנוּת וַעֲלִיָּה

מָה לוֹקְחִים מִרָמָה א׳?

הָאֻמּוֹת מְפַתּוֹת וּמַבְטִיחוֹת כָּבוֹד, וְיִשְׂרָאֵל נֶאֱמָנִים: עוֹלִים לָרֶגֶל, מְחַשְּׁבִים אֶת הַלּוּחַ, עוֹמְדִים זְקוּפִים וּמַשְׁכִּימִים לְבָתֵּי הַמִּדְרָשׁ.

9
❤️
אֲנִי לְדוֹדִי וְעָלַי תְּשׁוּקָתוֹ

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • הָאֻמּוֹת הִצִּיעוּ כָּבוֹד וּמְשִׂרוֹת, וְיִשְׂרָאֵל סֵרְבוּ. מָתַי אָמַרְתִּי ׳לֹא׳ לְמַשֶּׁהוּ מְפַתֶּה כִּי רָצִיתִי לִשְׁמֹר עַל הַקֶּשֶׁר עִם ה׳?
  • שְׁלוֹשָׁה רְגָלִים הֵם זְמַן ׳לִגְדֹּל׳ בְּשִׂמְחָה. מָה אֲנִי הוֹלֵךְ לְחַדֵּשׁ בְּחַג הַקָּרוֹב כְּדֵי לִשְׂמֹחַ בּוֹ?
  • רַשִׁ״י אוֹמֵר שֶׁ׳מִקְרָא, מִשְׁנָה וְתַלְמוּד׳ הֵם שָׁלָבִים בַּצְּמִיחָה. בְּאֵיזֶה שָׁלָב אֲנִי עַכְשָׁיו, וּמָה הַצַּעַד הַבָּא?

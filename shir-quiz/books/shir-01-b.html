מָשְׁכֵנִי אַחֲרֶיךָ נָּרוּצָה

🕊️ שִׁיר הַשִּׁירִים · מָשְׁכֵנִי אַחֲרֶיךָ נָּרוּצָה · רָמָה ב׳

מָשְׁכֵנִי אַחֲרֶיךָ נָּרוּצָה

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

שִׁיר הַשִּׁירִים לְפִי רַשִׁ״י מַתְחִיל בְּחִבָּה רִאשׁוֹנָה: הוֹלְכִים אַחֲרֵי ה׳ בַּמִּדְבָּר, רוֹאִים אֶת הַנֵּס בַּיָּם וְזוֹכְרִים שֶׁהוּא אוֹהֵב אוֹתָנוּ

📖 פֶּרֶק א׳

📜 הַסֵּפֶר הַמָּלֵא
📚

📜 הַפָּסוּקשִׁיר הַשִּׁירִים אֲשֶׁר לִשְׁלֹמֹה

💛 אַהֲבַת הַשֵּׁם: הַמֶּלֶךְ שֶׁבַּשִּׁיר הוּא ה׳, וְהַשִּׁיר הוּא שִׁיר שֶׁל שָׁלוֹם וְאַהֲבָה 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק א׳, פָּסוּק א׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

📚 רַשִׁ״י: ׳שִׁיר שֶׁהוּא עַל כָּל הַשִּׁירִים׳

רַשִׁ״י פּוֹתֵחַ: ׳שָׁנוּ רַבּוֹתֵינוּ: ״כָּל שְׁלֹמֹה הָאֲמוּרִים בְּשִׁיר הַשִּׁירִים קֹדֶשׁ, מֶלֶךְ שֶׁהַשָּׁלוֹם שֶׁלּוֹ״׳. כְּלוֹמַר, כָּל פַּעַם שֶׁכָּתוּב ׳הַמֶּלֶךְ׳ אוֹ ׳שְׁלֹמֹה׳, הַכַּוָּנָה לַקָּדוֹשׁ בָּרוּךְ הוּא, ׳מֶלֶךְ שֶׁהַשָּׁלוֹם שֶׁלּוֹ׳.

רַשִׁ״י: כָּל ׳שְׁלֹמֹה׳ בַּשִּׁיר הוּא שֵׁם לַקָּדוֹשׁ בָּרוּךְ הוּא, ׳מֶלֶךְ שֶׁהַשָּׁלוֹם שֶׁלּוֹ׳.

1
🍷

📜 הַפָּסוּק…דֹּדֶיךָ מִיָּיִן

💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה הִיא הַמַּתָּנָה הֲכִי מְתוּקָה שֶׁה׳ נָתַן לִי 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק א׳, פָּסוּק ב׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🍷 ׳כִּי טוֹבִים דֹּדֶיךָ מִיָּיִן׳

רַשִׁ״י מְבָאֵר: ׳טוֹבִים לִי דּוֹדֶיךָ מִכָּל מִשְׁתֵּה יַיִן וּמִכָּל עֹנֶג וְשִׂמְחָה׳. וְהַנִּמְשָׁל: ה׳ ׳נָתַן לָהֶם תּוֹרָתוֹ וְדִבֵּר עִמָּהֶם פָּנִים אֶל פָּנִים׳, וְהַחִבָּה הַזֹּאת עֲרֵבָה לָהֶם יוֹתֵר מִכָּל שַׁעֲשׁוּעַ.

רַשִׁ״י: אַהֲבַת ה׳ וְהַתּוֹרָה עֲרֵבוֹת מִכָּל עֹנֶג וְשִׂמְחָה בָּעוֹלָם.

2
🕊️

📜 הַפָּסוּקלְרֵיחַ שְׁמָנֶיךָ טוֹבִים שֶׁמֶן

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי מְסַפֵּר עַל נִסֵּי ה׳, אֲנִי עוֹזֵר לְרֵיחוֹ לְהִתְפַּשֵּׁט 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק א׳, פָּסוּק ג׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🕊️ ׳שֶׁמֶן תּוּרַק שְׁמֶךָ׳

רַשִׁ״י מַסְבִּיר אֶת הַמָּשָׁל: שֶׁמֶן טוֹב בִּצְלוֹחִית סְגוּרָה אֵין רֵיחוֹ נִדָּף, אֲבָל ׳פּוֹתְחָהּ וּמֵרִיק שַׁמְנָהּ לִכְלִי אַחֵר, רֵיחוֹ נוֹדֵף׳. וְכָךְ שֵׁם ה׳ נִשְׁמַע בְּכָל הָעוֹלָם ׳בַּעֲשׂוֹתְךָ נוֹרָאוֹת בְּמִצְרָיִם׳. הַנִּסִּים פָּתְחוּ אֶת הַצְּלוֹחִית.

רַשִׁ״י: הַנִּסִּים בְּמִצְרַיִם הֵם כְּמוֹ שֶׁמֶן שֶׁנִּשְׁפָּךְ – רֵיחוֹ שֶׁל שֵׁם ה׳ הִגִּיעַ לְכָל הָעוֹלָם.

3
⚖️

📜 הַפָּסוּקשְׁחוֹרָה אֲנִי

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אוֹתִי עִם הַחֻלְשׁוֹת שֶׁלִּי, וְגַם עִם הַכֹּחוֹת שֶׁלִּי 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק א׳, פָּסוּק ה׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

⚖️ ׳שְׁחוֹרָה אֲנִי וְנָאוָה׳

כְּנֶסֶת יִשְׂרָאֵל מִתְאָרֶת עַצְמָהּ בִּשְׁנֵי צְבָעִים. רַשִׁ״י: ׳דֻּגְמָא הִיא זוֹ: אוֹמֶרֶת כְּנֶסֶת יִשְׂרָאֵל לָאֻמּוֹת: ״שְׁחוֹרָה אֲנִי בְמַעֲשַׂי וְנָאָה אֲנִי בְמַעֲשֵׂה אֲבוֹתַי״׳. וְהִיא קוֹרֵאת לָאֻמּוֹת ׳בְּנוֹת יְרוּשָׁלַיִם׳, ׳עַל שֵׁם שֶׁהִיא עֲתִידָה לֵעָשׂוֹת מֶטְרוֹפּוֹלִין לְכֻלָּן׳ – עִיר רָאשִׁית לְכָל הָעַמִּים.

רַשִׁ״י: יֵשׁ בָּנוּ טָעוּיוֹת וְיֵשׁ בָּנוּ זְכוּיוֹת, וִירוּשָׁלַיִם תִּהְיֶה עִיר רָאשִׁית לְכָל הָעַמִּים.

4
🌳

📜 הַפָּסוּק…בְּנֵי אִמִּי נִחֲרוּ־בִי שָׂמֻנִי

💛 אַהֲבַת הַשֵּׁם: הַחֵטְא הוּא כְּמוֹ שֶׁמֶשׁ חֲזָקָה, וְה׳ הוּא הַצֵּל שֶׁמַּחֲזִיר אוֹתִי לְעַצְמִי 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק א׳, פָּסוּק ו׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🌳 ׳שֶׁשְּׁזָפַתְנִי הַשָּׁמֶשׁ׳

רַשִׁ״י: ׳אֵין שַׁחֲרוּתִי וְכִעוּרִי מִמְּעֵי אִמִּי, אֶלָּא עַל יְדֵי שְׁזִיפַת הַשֶּׁמֶשׁ, שֶׁאוֹתוֹ שַׁחֲרוּת נוֹחַ לְהִתְלַבֵּן כְּשֶׁיַּעֲמֹד בַּצֵּל׳. הַנִּמְשָׁל: הַחֵטְא אֵינוֹ חֵלֶק מִמֶּנּוּ מִתְּחִלָּה; מַשֶּׁהוּ מִבַּחוּץ ׳שָׁזַף׳ אוֹתָנוּ, וְקַל לַחְזֹר לְהִיוֹת נְקִיִּים כְּשֶׁחוֹזְרִים לְצֵל ה׳.

רַשִׁ״י: מִי שֶׁטָּעָה יָכוֹל לַחְזֹר – הַשַּׁחֲרוּת ׳נוֹחַ לְהִתְלַבֵּן׳ בַּצֵּל.

5
🌅

📜 הַפָּסוּקלְסֻסָתִי בְּרִכְבֵי פַרְעֹה דִּמִּיתִיךְ רַעְיָתִי

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי שׁוֹתֵק וּבוֹטֵחַ בַּה׳, הוּא נִלְחָם בִּשְׁבִילִי 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק א׳, פָּסוּק ט׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🌅 ׳דִּמִּיתִיךְ רַעְיָתִי׳

רַשִׁ״י מַסְבִּיר שֶׁ׳דִּמִּיתִיךְ׳ הוּא מִלְּשׁוֹן שְׁתִיקָה: ׳שָׁם דִּמִּיתִיךְ רַעְיָתִי, שִׁתַּקְתִּיךְ מִצַּעֲקָתֵךְ, שֶׁנֶּאֱמַר ״וְאַתֶּם תַּחֲרִשׁוּן״׳. וְיֵשׁ פֵּרוּשׁ נוֹסָף: ׳שָׁם הֶרְאֵיתִי לַכֹּל שֶׁרַעְיָתִי אַתְּ׳ – בַּיָּם סוּף כָּל הָעוֹלָם רָאָה שֶׁיִּשְׂרָאֵל אֲהוּבִים עַל ה׳.

רַשִׁ״י: בַּיָּם סוּף ה׳ הֶרְאָה לַכֹּל שֶׁיִּשְׂרָאֵל הֵם ׳רַעְיָתִי׳ – הָאֲהוּבִים שֶׁלּוֹ.

6
🏆

📜 הַפָּסוּקהִנָּךְ יָפָה רַעְיָתִי

💛 אַהֲבַת הַשֵּׁם: ה׳ לֹא נֶעֱלָב לָנֶצַח; הוּא מְחַזֵּק אוֹתִי וְאוֹמֵר לִי: אַתָּה יָקָר לִי 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק א׳, פָּסוּק ט״ו (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🏆 ׳הִנָּךְ יָפָה׳ בְּ׳נַעֲשֶׂה׳ וּבְ׳נִשְׁמָע׳

אַחֲרֵי הַחֵטְא ה׳ מְרַצֶּה אֶת עַמּוֹ. רַשִׁ״י: ׳מָחַלְתִּי לָךְ עַל עֲוֹנֵךְ, וַהֲרֵי אַתְּ יָפָה בְ״נַעֲשֶׂה״ וְהִנָּךְ יָפָה בְ״נִשְׁמָע״׳. וְעוֹד: בְּחֵטְא הָעֵגֶל ׳וַיֵּאָסְפוּ אֵלָיו כָּל בְּנֵי לֵוִי׳ וְלֹא טָעוּ, וְאַחַר כָּךְ נֶאֱמַר עַל מְלֶאכֶת הַמִּשְׁכָּן: ׳וַיְבָרֶךְ אֹתָם משֶׁה׳. ה׳ מְחַזֵּק אֶת עַמּוֹ בִּדְבָרִים טוֹבִים.

רַשִׁ״י: ה׳ מְשַׁבֵּחַ אֶת יִשְׂרָאֵל עַל ׳נַעֲשֶׂה וְנִשְׁמָע׳ וְעַל בִּנְיַן הַמִּשְׁכָּן, וּמְחַזֵּק אוֹתָם אַחֲרֵי הַחֵטְא.

7
💡

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

מָה לוֹקְחִים מִרָמָה ב׳?

רַשִׁ״י מְגַלֶּה שֶׁהַמֶּלֶךְ שֶׁבַּשִּׁיר הוּא ה׳, וְשֶׁגַּם בְּשָׁעָה שֶׁיֵּשׁ בָּנוּ שַׁחֲרוּת – ׳נוֹחַ לְהִתְלַבֵּן׳, וְה׳ מְשַׁבֵּחַ אוֹתָנוּ עַל ׳נַעֲשֶׂה וְנִשְׁמָע׳.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • מַתַּי הָלַכְתִּי אַחֲרֵי מִישֶׁהוּ שֶׁסָּמַכְתִּי עָלָיו לְגַמְרֵי, כְּמוֹ עַם יִשְׂרָאֵל בַּמִּדְבָּר? מָה הִרְגַּשְׁתִּי?
  • רַשִׁ״י אוֹמֵר שֶׁגַּם אַחֲרֵי חֵטְא הָעֵגֶל ה׳ מָחַל וְאָמַר ׳הִנָּךְ יָפָה׳. מָה אֲנִי יָכוֹל לוֹמַר לְחָבֵר שֶׁטָּעָה כְּדֵי לְחַזֵּק אוֹתוֹ?
  • אֵיזֶה נֵס אוֹ טוֹבָה מֵה׳ אֲנִי רוֹצֶה לְסַפֵּר הַיּוֹם, כְּדֵי שֶׁרֵיחַ הַשֶּׁמֶן יִתְפַּשֵּׁט?

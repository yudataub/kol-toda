ה׳ מְקַלֵּס אֶת עַמּוֹ, בּוֹחֵר בְּבֵית עוֹלָמִים בְּהַר הַמּוֹרִיָּה, וְהוֹלֵךְ אִתָּנוּ גַּם בַּגָּלוּת – ׳בְּצָרָתֵךְ לִי צָר׳

רָמָה א׳ – ה׳ אִתָּנוּ

🕊️ ׳הִנָּךְ יָפָה׳ – ה׳ מְקַלֵּס אֶת עַמּוֹ

בְּפֶרֶק ד׳ ה׳ מְשַׁבֵּחַ אֶת יִשְׂרָאֵל. רַשִׁ״י: ׳קִלְּסָן וְרִצָּן וְעָרְבוּ עָלָיו קָרְבְּנוֹתֵיהֶם׳. ׳עֵינַיִךְ יוֹנִים׳ מְלַמֵּד שֶׁיִּשְׂרָאֵל דְּבוּקִים בַּקָּדוֹשׁ בָּרוּךְ הוּא כְּמוֹ יוֹנָה הַדְּבֵקָה בְּבֶן זוּגָהּ: ׳כָּךְ אַתְּ נָתַתְּ שֶׁכֶם לִסְבֹּל עֻלִּי וּמוֹרָאִי׳.

ה׳ שָׂמֵחַ בְּיִשְׂרָאֵל, מְקַבֵּל אֶת קָרְבְּנוֹתֵיהֶם, וְאוֹהֵב אֶת הַדְּבֵקוּת שֶׁלָּהֶם.

1

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹאֶה בִּי מִצְווֹת שֶׁאֲנִי לֹא תָּמִיד שָׂם לֵב אֲלֵיהֶן 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ד׳, פָּסוּק ג׳

רָמָה א׳ – ה׳ אִתָּנוּ

🧵 ׳כְּחוּט הַשָּׁנִי׳ – עֲמִידָה בַּהַבְטָחָה

רַשִׁ״י מְבָאֵר שֶׁיִּשְׂרָאֵל ׳נָאוֹת לְהַבְטִיחַ וְלִשְׁמֹר הַבְטָחָתָם, כְּמוֹ שֶׁעָשׂוּ הַמְרַגְּלִים לְרָחָב׳ שֶׁאָמְרוּ לָהּ ׳אֶת תִּקְוַת חוּט הַשָּׁנִי׳. וְעוֹד: ׳רֵיקָנִים שֶׁבֵּיךְ מְלֵאִים מִצְוֹת כְּרִמּוֹן׳ – גַּם מִי שֶׁנִּרְאֶה ׳רֵיק׳ מָלֵא מִצְוֹת.

יִשְׂרָאֵל שׁוֹמְרִים עַל הַבְטָחָה, וְאֲפִלּוּ ׳הָרֵיקִים׳ שֶׁבָּהֶם מְלֵאִים מִצְוֹת.

2

💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה הִיא הַמִּבְצָר שֶׁלִּי 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ד׳, פָּסוּק ד׳

רָמָה א׳ – ה׳ אִתָּנוּ

🏰 ׳כְּמִגְדַּל דָּוִד׳

רַשִׁ״י: ׳כְּמִגְדַּל דָּוִד׳ הוּא ׳מְצוּדַת צִיּוֹן שֶׁהוּא מְקוֹם חֹזֶק וּמִבְצָר׳, וְהַדֻּגְמָא הִיא ׳לִשְׁכַּת הַגָּזִית, שֶׁהָיְתָה חָזְקָן וּמִבְצָרָן שֶׁל יִשְׂרָאֵל׳. מִשָּׁם יוֹצֵאת הוֹרָאָה, ׳שֶׁהַתּוֹרָה מָגֵן לְיִשְׂרָאֵל׳. הַמָּגִנִּים הַתְּלוּיִים הֵם הַתַּלְמִידִים.

בֵּית הַדִּין בְּלִשְׁכַּת הַגָּזִית הוּא הַמִּבְצָר שֶׁל יִשְׂרָאֵל, כִּי הַתּוֹרָה מָגֵן.

3

💛 אַהֲבַת הַשֵּׁם: מִי שֶׁאוֹהֵב אֶת ה׳ אוֹהֵב גַּם אֶת הַבְּרִיּוֹת שֶׁנִּבְרְאוּ בְּצַלְמוֹ 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ד׳, פָּסוּק ה׳

רָמָה א׳ – ה׳ אִתָּנוּ

🪨 ׳תְּאוֹמֵי צְבִיָּה׳ – שְׁנֵי הַלּוּחוֹת

רַשִׁ״י מְפָרֵשׁ שֶׁהַפָּסוּק מְדַבֵּר עַל ׳מֹשֶׁה וְאַהֲרֹן׳. וְדָבָר אַחֵר: עַל ׳הַלּוּחוֹת, תְּאוֹמֵי צְבִיָּה, שֶׁהֵם מְכֻוָּנוֹת בְּמִדָּה אַחַת׳, ׳חֲמִשָּׁה דִּבְּרוֹת עַל זוֹ וַחֲמִשָּׁה עַל זוֹ, דִּבּוּר כְּנֶגֶד דִּבּוּר׳. לְמָשָׁל: ׳אָנֹכִי׳ כְּנֶגֶד ׳לֹא תִרְצַח׳, ׳שֶׁהָרוֹצֵחַ מְמַעֵט אֶת הַדְּמוּת שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא׳.

חֲמֵשֶׁת הַדִּבְּרוֹת הָרִאשׁוֹנִים (בֵּין אָדָם לַמָּקוֹם) וַחֲמֵשֶׁת הַשְּׁנִיִּים (בֵּין אָדָם לַחֲבֵרוֹ) מְכֻוָּנִים זֶה כְּנֶגֶד זֶה.

4

💛 אַהֲבַת הַשֵּׁם: ה׳ מוֹצֵא לִי תָּמִיד מָקוֹם חָדָשׁ לְהַתְחִיל בּוֹ 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ד׳, פְּסוּקִים ו׳–ז׳

רָמָה א׳ – ה׳ אִתָּנוּ

⛰️ ׳אֵלֶךְ לִי אֶל הַר הַמּוֹר׳

רַשִׁ״י מְבָאֵר מָה קָרָה כְּשֶׁחָטְאוּ בְּנֵי עֵלִי בְּשִׁילֹה: ה׳ אוֹמֵר, ׳אֶסְתַּלֵּק מֵעֲלֵיכֶם וְאֶטּשׁ מִשְׁכָּן הַזֶּה וְאֶבְחַר לִי בְהַר הַמּוֹרִיָּה, בְּבֵית עוֹלָמִים׳. וְשָׁם, בְּבֵית הַמִּקְדָּשׁ, ׳כֻּלָּךְ יָפָה... וּמוּם אֵין בָּךְ׳, ׳וְאֶרְצֶה שָׁם כָּל קָרְבְּנוֹתַיִךְ׳.

אַחֲרֵי שִׁילֹה ה׳ בָּחַר בְּהַר הַמּוֹרִיָּה לִבְנוֹת בֵּית קָבוּעַ – בֵּית עוֹלָמִים.

5

💛 אַהֲבַת הַשֵּׁם: ה׳ אִתִּי בְּכָל מָקוֹם שֶׁאֲנִי הוֹלֵךְ 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ד׳, פָּסוּק ח׳

רָמָה א׳ – ה׳ אִתָּנוּ

🚶 ׳אִתִּי מִלְּבָנוֹן תָּבוֹאִי׳

רַשִׁ״י מְבָאֵר: ׳כְּשֶׁתִּגְלוּ מִלְּבָנוֹן זֶה, אִתִּי תִגְלוּ, כִּי אֲנִי אֶגְלֶה עִמָּכֶם׳. ה׳ הוֹלֵךְ אִתָּנוּ בַּגָּלוּת: ׳וְאַף כָּל יְמֵי הַגּוֹלָה, בְּצָרָתֵךְ לִי צָר׳. וּכְשֶׁנָּשׁוּב – ׳אֲנִי אָשׁוּב עִמָּכֶם׳.

בַּגָּלוּת ה׳ לֹא עוֹזֵב אֶת עַמּוֹ – הוּא הוֹלֵךְ אִתָּם וְחוֹזֵר אִתָּם.

6

💛 אַהֲבַת הַשֵּׁם: הַצִּיצִית שֶׁלִּי מַזְכִּירָה לִי שֶׁאֲנִי שֶׁל ה׳ 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ד׳, פָּסוּק י״א

רָמָה א׳ – ה׳ אִתָּנוּ

🧵 ׳רֵיחַ שַׂלְמֹתַיִךְ׳ – תּוֹרָה וּמִצְווֹת

רַשִׁ״י: ׳נֹפֶת תִּטֹּפְנָה... טַעֲמֵי תוֹרָה׳, וְ׳רֵיחַ שַׂלְמֹתַיִךְ׳ הֵם ׳מִצְוֹת הֲגוּנוֹת הַנּוֹהֲגוֹת בְּשַׂלְמוֹתַיִךְ: צִיצִית, תְּכֵלֶת, בִּגְדֵי כְהֻנָּה, אִסּוּר שַׁעַטְנֵז׳. אֶת הַתּוֹרָה אוֹמְרִים כְּמוֹ דְּבַשׁ מָתוֹק, וְאֶת הַבְּגָדִים עוֹטְפִים בְּמִצְווֹת.

הַתּוֹרָה מְתוּקָה כְּמוֹ דְּבַשׁ, וּבִגְדֵי הַמִּצְוָה כְּמוֹ צִיצִית מְצַיְּנִים אֶת יִשְׂרָאֵל.

7

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְחַכֶּה לְקִבּוּץ הַגָּלֻיּוֹת, וְה׳ מַחֲזִיר אוֹתִי הַבַּיְתָה 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ד׳, פָּסוּק ט״ז

רָמָה א׳ – ה׳ אִתָּנוּ

🌍 ׳עוּרִי צָפוֹן וּבוֹאִי תֵימָן׳

רַשִׁ״י מְבָאֵר: ה׳ מְצַוֶּה אֶת הָרוּחוֹת לָשֵׂאת אֶת הָרֵיחַ הַטּוֹב שֶׁל יִשְׂרָאֵל לְרָחוֹק, ׳עַל שֵׁם שֶׁהַגָּלֻיּוֹת מִתְקַבְּצוֹת, וּמִכָּל הַגּוֹיִם מְבִיאִים אוֹתָם מִנְחָה לִירוּשָׁלַיִם׳. וּבִימֵי הַבִּנְיָן יִהְיוּ יִשְׂרָאֵל נִקְבָּצִים שָׁם לְמוֹעֲדִים וְלָרְגָלִים, ׳וְיִשְׂרָאֵל מְשִׁיבִין: ״יָבֹא דוֹדִי לְגַנּוֹ״׳.

בֶּעָתִיד כָּל הַגָּלֻיּוֹת יִתְקַבְּצוּ לִירוּשָׁלַיִם, וְיִשְׂרָאֵל יִקְרְאוּ לַה׳: בּוֹא לְבֵיתְךָ.

8

מָה לוֹקְחִים מִרָמָה א׳?

ה׳ מְקַלֵּס אֶת עַמּוֹ וְשׁוֹמֵר עַל הַתּוֹרָה, בּוֹחֵר בְּבֵית עוֹלָמִים וְהוֹלֵךְ אִתָּנוּ בַּגָּלוּת.

  • ה׳ הוֹלֵךְ אִתָּנוּ ׳בְּכֹל אֲשֶׁר תֵּצְאוּ וְתָבוֹאוּ׳. מָתַי הִרְגַּשְׁתִּי שֶׁאֲנִי לֹא לְבַד, אֲפִלּוּ כְּשֶׁהָיָה לִי קָשֶׁה?
  • רַשִׁ״י אוֹמֵר שֶׁהַתַּלְמִידִים הֵם ׳חִצִּים׳ וְהַתּוֹרָה הִיא הַמָּגֵן. אֵיךְ אֲנִי יָכוֹל לִהְיוֹת חֵלֶק מִשִּׁיר שֶׁל לוֹמְדֵי תּוֹרָה?

אֲנִי יְשֵׁנָה וְלִבִּי עֵר

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

ה׳ דּוֹפֵק בְּאַהֲבָה עַל דֶּלֶת הַלֵּב דֶּרֶךְ הַנְּבִיאִים; גַּם כְּשֶׁיִּשְׂרָאֵל ׳יְשֵׁנִים׳ וְהַמִּקְדָּשׁ נֶחֱרַב, הַלֵּב נִשְׁאָר עֵר – ׳חוֹלַת אַהֲבָה אָנִי׳

📖 פֶּרֶק ה׳

📜 הַפָּסוּקבָּאתִי לְגַנִּי

💛 אַהֲבַת הַשֵּׁם: ה׳ שָׂמֵחַ בְּכָל דָּבָר קָטָן שֶׁאֲנִי מֵבִיא לוֹ בְּאַהֲבָה 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ה׳, פָּסוּק א׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🍯 ׳אָכַלְתִּי יַעְרִי עִם דִּבְשִׁי׳

רַשִׁ״י: ׳אָרִיתִי׳ הוּא ׳לָקַטְתִּי׳, ׳וְנֶאֱמַר עַל שֵׁם הַקְּטֹרֶת, שֶׁהִקְטִירוּ קְטֹרֶת יָחִיד הַנְּשִׂיאִים עַל מִזְבֵּחַ הַחִיצוֹן וְנִתְקַבְּלָה, וְהוּא דָבָר שֶׁאֵינוֹ נוֹהֵג לְדוֹרוֹת׳. ׳יַעְרִי עִם דִּבְשִׁי׳ – יֵשׁ דְּבַשׁ שֶׁגָּדֵל בְּקָנִים, ׳וַאֲנִי מֵרֹב חִבָּה אָכַלְתִּי׳ אֶת הַקָּנֶה עִם הַדְּבַשׁ. כֹּה חָבִיב עָלָיו הַקָּרְבָּן.

רַשִׁ״י: מֵרֹב חִבָּה ה׳ אָכַל ׳אֶת הַיַּעַר עִם הַדְּבַשׁ׳ – קִבֵּל אֶת קָרְבְּנוֹת הַנְּשִׂיאִים בִּשְׂמָחָה.

📜 הַפָּסוּקאֲנִי יְשֵׁנָה וְלִבִּי עֵר

💛 אַהֲבַת הַשֵּׁם: בִּזְמַן שֶׁטּוֹב לִי אֲנִי זוֹכֵר לְהוֹדוֹת לַה׳ וְלֹא נִרְדָּם 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ה׳, פָּסוּק ב׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🌙 ׳שֶׁשָּׁקַט בְּבַיִת רִאשׁוֹן׳

רַשִׁ״י מְבָאֵר שֶׁהַפָּסוּק מְדַבֵּר עַל הַבַּיִת הָרִאשׁוֹן: כְּשֶׁיִּשְׂרָאֵל הָיוּ ׳שְׁלֵוָה וּשְׁקֵטָה׳, הֵם ׳נוֹאַשְׁתִּי מֵעֲבֹד הַקָּדוֹשׁ בָּרוּךְ הוּא כִּישֵׁנָה וְנִרְדָּמֶת׳. וְהַקָּדוֹשׁ בָּרוּךְ הוּא, ׳צוּר לְבָבִי וְחֶלְקִי׳, ׳עֵר לְשָׁמְרֵנִי וּלְהֵיטִיב לִי׳. כְּשֶׁטּוֹב לָנוּ, קַל לְהִרְדַּם; וה׳ אֵינוֹ נִרְדָּם.

רַשִׁ״י: דַּוְקָא בִּזְמַן שֶׁלְּוָה וְשֶׁקֶט הָעָם נִרְדָּם מֵעֲבוֹדַת ה׳, וה׳ נִשְׁאָר עֵר.

💧

📜 הַפָּסוּק…קוֹל דּוֹדִי דוֹפֵק פִּתְחִי־לִי

💛 אַהֲבַת הַשֵּׁם: ה׳ בָּא אֵלַי בְּרַחֲמִים וְלֹא בְּכַעַס 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ה׳, פָּסוּק ב׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

💧 ׳שֶׁרֹאשִׁי נִמְלָא טָל׳

רַשִׁ״י מְבָאֵר אֶת לִבּוֹ שֶׁל ה׳: ׳שֶׁרֹאשִׁי נִמְלָא טָל׳ – ׳שֶׁאֲנִי מָלֵא רָצוֹן וְנַחַת רוּחַ אַבְרָהָם אָבִיךְ, שֶׁעָרְבוּ עָלַי מַעֲשָׂיו כַּטַּל׳. אֲבָל גַּם ׳בְּיָדִי הַרְבֵּה קְבוּצוֹת מִינֵי פֻרְעָנִיּוֹת לִפָּרַע מֵעוֹזְבַי׳. הוּא דּוֹפֵק בְּטַל, בְּרַחֲמִים, וְרוֹצֶה שֶׁנִּפְתַּח לוֹ כְּדֵי שֶׁלֹּא יִצְטָרֵךְ לְהַפְעִיל אֶת הַפֻּרְעָנוּת.

רַשִׁ״י: ה׳ בָּא בְּטַל שֶׁל רַחֲמִים בִּזְכוּת אַבְרָהָם, כְּדֵי שֶׁלֹּא נִצְטָרֵךְ לְפֻרְעָנוּת.

🧥

📜 הַפָּסוּקדּוֹדִי שָׁלַח יָדוֹ מִן־הַחֹר וּמֵעַי

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁהַהֶרְגֵּל חָזָק, אֲנִי לֹא מְוַתֵּר עַל לַחְזֹר 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ה׳, פָּסוּק ד׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🧥 ׳לֹא אוּכַל לָשׁוּב׳

רַשִׁ״י: ׳פָּשַׁטְתִּי אֶת כֻּתָּנְתִּי׳ הוּא ׳כְּבָר לִמַּדְתִּי לְעַצְמִי דְרָכִים אֲחֵרִים; לֹא אוּכַל לָשׁוּב אֵלֶיךָ עוֹד, כְּעִנְיָן שֶׁנֶּאֱמַר: ״וּמִן אָז חָדַלְנוּ לְקַטֵּר לִמְלֶכֶת הַשָּׁמַיִם״׳, ׳שֶׁהָיוּ הַדְּרָכִים הָאֵלֶּה יְשָׁרִים בְּעֵינֵיהֶם׳. הַהִתְנַצְּלוּת הִיא שֶׁהַהֶרְגֵּל הִתְבַּסֵּס. אֲבָל ה׳ ׳שָׁלַח יָדוֹ׳ – לֹא וִתֵּר עֲלֵיהֶם.

רַשִׁ״י: כְּשֶׁאָדָם רָגִיל בְּדֶרֶךְ לֹא טוֹבָה, הוּא אוֹמֵר ׳לֹא אוּכַל לָשׁוּב׳; אֲבָל ה׳ מַמְשִׁיךְ לִפְנוֹת אֵלָיו.

📜

📜 הַפָּסוּקפָּתַחְתִּי אֲנִי לְדוֹדִי וְדוֹדִי חָמַק עָבָר

💛 אַהֲבַת הַשֵּׁם: תְּשׁוּבָה לְעוֹלָם לֹא הוֹלֶכֶת לְאִבּוּד 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ה׳, פָּסוּק ו׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

📜 ׳וְדוֹדִי חָמַק עָבָר׳ – מָה קָרָה אַחֲרֵי חִזְקִיָּהוּ

רַשִׁ״י: ׳לֹא בִטֵּל גְּזֵרָתוֹ׳. אֶל חִזְקִיָּהוּ נֶאֱמַר ׳הִנֵּה יָמִים בָּאִים וְנִשָּׂא כָּל אֲשֶׁר בְּבֵיתֶךָ׳ וְ׳מִבָּנֶיךָ... יִקָּחוּ׳ – ׳אֵלּוּ דָנִיֵּאל, חֲנַנְיָה, מִישָׁאֵל וַעֲזַרְיָה׳. גַּם בִּימֵי יֹאשִׁיָּהוּ, דֶּרֶךְ חֻלְדָּה הַנְּבִיאָה, נֶאֱמַר: ׳הִנְנִי מֵבִיא רָעָה׳. אֲבָל הַנְּבִיאִים הוֹסִיפוּ לִקְרֹא לַתְּשׁוּבָה כְּדֵי שֶׁהַגָּלוּת תִּהְיֶה קַלָּה יוֹתֵר.

רַשִׁ״י: הַגְּזֵרָה נִשְׁאֲרָה, אֲבָל הַתְּשׁוּבָה שֶׁל חִזְקִיָּהוּ וְיֹאשִׁיָּהוּ הֵקֵלָה אוֹתָהּ.

🗡️

📜 הַפָּסוּקמְצָאֻנִי הַשֹּׁמְרִים הַסֹּבְבִים בָּעִיר הִכּוּנִי

💛 אַהֲבַת הַשֵּׁם: בֵּית הַמִּקְדָּשׁ הוּא הַקִּשּׁוּט הַיָּקָר בְּיוֹתֵר שֶׁלָּנוּ 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ה׳, פָּסוּק ז׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🗡️ ׳בֵּית הַמִּקְדָּשׁ׳ – ׳רְדִידִי׳

רַשִׁ״י: ׳מְצָאֻנִי הַשֹּׁמְרִים הַסֹּבְבִים בָּעִיר׳ – ׳נְבוּכַדְנֶאצַּר וְחֵילוֹתָיו׳, ׳לִנְקוֹם נִקְמָתוֹ שֶׁל מָקוֹם׳. ׳נָשְׂאוּ אֶת רְדִידִי׳ – ׳בֵּית הַמִּקְדָּשׁ׳. הַמִּקְדָּשׁ נִקְרָא ׳רְדִידִי׳ – בֶּגֶד שֶׁעוֹטְפִים בּוֹ וּמְקַשְּׁטִים – כִּי הוּא הַיָּפֶה בְּיוֹתֵר שֶׁיֵּשׁ לְעַם יִשְׂרָאֵל.

רַשִׁ״י: בֵּית הַמִּקְדָּשׁ הוּא ׳רְדִידִי׳ – הַקִּשּׁוּט הַיָּקָר בְּיוֹתֵר שֶׁל יִשְׂרָאֵל.

📜 הַפָּסוּק…לוֹ שֶׁחוֹלַת אַהֲבָה אָנִי

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִחְיוֹת כָּךְ שֶׁגַּם אֲחֵרִים יָעִידוּ שֶׁאֲנִי אוֹהֵב אֶת ה׳ 💛

📖 שִׁיר הַשִּׁירִים, פֶּרֶק ה׳, פָּסוּק ח׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🔑 ׳שֶׁחוֹלַת אַהֲבָה אָנִי׳

יִשְׂרָאֵל אוֹמְרִים לָאֻמּוֹת: ׳הִשְׁבַּעְתִּי אֶתְכֶם... אִם תִּמְצְאוּ אֶת דּוֹדִי מַה תַּגִּידוּ לוֹ׳? רַשִׁ״י: ׳שֶׁתָּעִידוּ עָלַי שֶׁבִּשְׁבִיל אַהֲבָתוֹ חָלִיתִי בְּיִסּוּרִים קָשִׁים בֵּינֵיכֶם. יָבֹא נְבוּכַדְנֶאצַּר וְיָעִיד, יָבֹא אֱלִיפַז וְצוֹפַר וְכָל נְבִיאֵי הָאֻמּוֹת, וְיָעִידוּ עָלַי שֶׁקִּיַּמְתִּי אֶת הַתּוֹרָה׳.

רַשִׁ״י: אֲפִלּוּ הָאֻמּוֹת יָעִידוּ בְּיוֹם הַדִּין שֶׁיִּשְׂרָאֵל שָׁמְרוּ אֶת הַתּוֹרָה מֵאַהֲבָה.

💡

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

מָה לוֹקְחִים מִרָמָה ב׳?

רַשִׁ״י מְבָאֵר אֶת הַטַּל שֶׁבָּא בְּרַחֲמִים, אֶת ׳לֹא אוּכַל לָשׁוּב׳ וְאֶת הַגְּזֵרָה שֶׁהוּקְלָה בִּזְכוּת הַתְּשׁוּבָה.

אֲנִי יְשֵׁנָה וְלִבִּי עֵר

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • ה׳ דּוֹפֵק עַל הַדֶּלֶת בְּטַל שֶׁל רַחֲמִים. מָתַי הִרְגַּשְׁתִּי שֶׁמַּשֶּׁהוּ ׳דוֹפֵק׳ עַל הַלֵּב שֶׁלִּי וּמְבַקֵּשׁ שֶׁאֶפְתַּח?
  • רַשִׁ״י אוֹמֵר שֶׁיִּשְׂרָאֵל חָשְׁבוּ ׳לֹא אוּכַל לָשׁוּב׳. עַל אֵיזֶה דָּבָר אֲנִי חוֹשֵׁב שֶׁכְּבָר מְאֻחָּר מִדַּי לְתַקֵּן – וְאוּלַי לֹא?
  • בִּזְמַן שֶׁטּוֹב לִי קַל לְהִרְדַּם. מָה אֲנִי יָכוֹל לַעֲשׂוֹת כְּדֵי שֶׁהַלֵּב שֶׁלִּי יִשָּׁאֵר עֵר גַּם כְּשֶׁטּוֹב?

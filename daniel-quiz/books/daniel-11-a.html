בְּסוֹף הַסֵּפֶר דָּנִיֵּאל לוֹמֵד שֶׁבִּימֵי צָרָה יֵשׁ יוֹדְעֵי אֱלֹהִים שֶׁמִּתְחַזְּקִים, מַשְׂכִּילִים שֶׁמְּלַמְּדִים, וְה׳ שׁוֹמֵר עַל הַכְּתוּבִים בַּסֵּפֶר

🗝️ הַחָזוֹן הָאַחֲרוֹן

בַּפְּרָקִים הָאַחֲרוֹנִים שֶׁל הַסֵּפֶר דָּנִיֵּאל שׁוֹמֵעַ ׳אֶת הָרָשׁוּם בִּכְתָב אֱמֶת׳ – חָזוֹן עַל מַלְכֻיּוֹת וּמִלְחָמוֹת בִּימֵי הַגָּלוּת. אֲנַחְנוּ לֹא נִכְנָסִים לְפִרְטֵי הַמִּלְחָמוֹת, אֶלָּא לַדְּבָרִים שֶׁנּוֹגְעִים לְלִבֵּנוּ: מָה עוֹשִׂים כְּשֶׁקָּשֶׁה.

הַחָזוֹן מְלַמֵּד אוֹתָנוּ מָה עוֹשִׂים בִּימֵי צָרָה.

📖 דָּנִיֵּאל, פֶּרֶק י״א, פָּסוּק ל״ב

🎭 מְחַנְּפִים

הַכָּתוּב אוֹמֵר: ׳וּמַרְשִׁיעֵי בְרִית יַחֲנִיף בַּחֲלַקּוֹת׳ – מִי שֶׁעוֹזֵב אֶת הַבְּרִית, הַמֶּלֶךְ מְחַנֵּף אוֹתוֹ בִּדְבָרִים חֲלָקִים. יֵשׁ אֲנָשִׁים שֶׁנִּסְחָפִים אַחֲרֵי הַחֲנֻפָּה וְאַחֲרֵי הַכָּבוֹד.

דְּבָרִים חֲלָקִים יְכוֹלִים לְהַרְחִיק אוֹתָנוּ מֵה׳.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְהַכִּיר אוֹתְךָ, ה׳, וּלְהִתְחַזֵּק 💛

📖 דָּנִיֵּאל, פֶּרֶק י״א, פָּסוּק ל״ב

🛡️ יֹדְעֵי אֱלֹהָיו יַחֲזִקוּ

מוּל הַמְּחַנְּפִים: ׳וְעַם יֹדְעֵי אֱלֹהָיו יַחֲזִקוּ וְעָשׂוּ׳ – מִי שֶׁמַּכִּיר אֶת אֱלֹהָיו מִתְחַזֵּק וְעוֹשֶׂה. הוּא לֹא נִסְחָף, כִּי הוּא יוֹדֵעַ מִי הוּא וּמִי ה׳.

מִי שֶׁמַּכִּיר אֶת ה׳ – מִתְחַזֵּק וְעוֹשֶׂה.

🎓

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹלֵחַ לָנוּ מוֹרִים וּמַדְרִיכִים בְּאַהֲבָה 💛

📖 דָּנִיֵּאל, פֶּרֶק י״א, פָּסוּק ל״ג

🎓 מַשְׂכִּילֵי עָם

׳וּמַשְׂכִּילֵי עָם יָבִינוּ לָרַבִּים׳ – חֲכָמֵי הָעָם מְבִינִים אֶת הָרַבִּים וּמְלַמְּדִים אוֹתָם. בִּזְמַן שֶׁהָעָם לֹא יוֹדֵעַ מָה לַעֲשׂוֹת, הַחֲכָמִים מְלַמְּדִים וּמַדְרִיכִים.

חֲכָמִים מְלַמְּדִים אֶת הָרַבִּים בִּשְׁעַת קֹשִׁי.

🤝

💛 אַהֲבַת הַשֵּׁם: הָעֵזֶר הַקָּטָן שֶׁלְּךָ, ה׳, גָּדוֹל בְּעֵינַי 💛

📖 דָּנִיֵּאל, פֶּרֶק י״א, פָּסוּק ל״ד

🤝 עֵזֶר מְעָט

אֲפִלּוּ כְּשֶׁהַצַּדִּיקִים נִכְשָׁלִים, ׳וּבְהִכָּשְׁלָם יֵעָזְרוּ עֵזֶר מְעָט׳ – הֵם מְקַבְּלִים עֵזֶר, אַף אִם קָטָן. ה׳ לֹא נוֹטֵשׁ אוֹתָם, וְאֵין הֵם נִכְלִים בְּיַד הָאוֹיֵב.

ה׳ עוֹזֵר גַּם כְּשֶׁהָעֵזֶר נִרְאֶה קָטָן.

💛 אַהֲבַת הַשֵּׁם: ה׳ מְצָרֵף אוֹתִי כְּדֵי שֶׁאֶהְיֶה זָהָב טָהוֹר 💛

🔥 יִצָּרְפוּ וְיִתְלַבְּנוּ

הַכָּתוּב אוֹמֵר שֶׁמִּן הַמַּשְׂכִּילִים יִכָּשְׁלוּ ׳לִצְרוֹף בָּהֶם וּלְבָרֵר וְלַלְבֵּן עַד עֵת קֵץ כִּי עוֹד לַמּוֹעֵד׳. הַקֹּשִׁי כְּמוֹ מַתֶּכֶת שֶׁמְּצָרְפִים בָּאֵשׁ: הִיא נִקִּית וְיָפָה יוֹתֵר.

הַקֹּשִׁי מְצָרֵף אוֹתָנוּ וּמְנַקֶּה.

💛 אַהֲבַת הַשֵּׁם: ה׳ מַכִּיר אוֹתִי בְּשֵׁם, וְכוֹתֵב אוֹתִי בַּסֵּפֶר שֶׁלּוֹ 💛

📖 דָּנִיֵּאל, פֶּרֶק י״ב, פְּסוּקִים א׳, ד׳

רָמָה א׳ – חֲכָמִים וְיוֹדְעֵי אֱלֹהִים

📕 כָּל הַנִּמְצָא כָּתוּב בַּסֵּפֶר

הַמַּלְאָךְ אָמַר: ׳וּבָעֵת הַהִיא יִמָּלֵט עַמְּךָ כָּל הַנִּמְצָא כָּתוּב בַּסֵּפֶר׳ – יִנָּצְלוּ כָּל אֵלּוּ שֶׁכְּתוּבִים בַּסֵּפֶר. וְנֶאֱמַר לְדָנִיֵּאל: ׳וְאַתָּה דָנִיֵּאל סְתֹם הַדְּבָרִים וַחֲתֹם הַסֵּפֶר עַד עֵת קֵץ׳.

ה׳ שׁוֹמֵר עַל כָּל מִי שֶׁכָּתוּב בְּסִפְרוֹ.

⚗️ יִתְבָּרְרוּ וְיִתְלַבְּנוּ

הַכָּתוּב: ׳יִתְבָּרֲרוּ וְיִתְלַבְּנוּ וְיִצָּרְפוּ רַבִּים, וְהִרְשִׁיעוּ רְשָׁעִים וְלֹא יָבִינוּ כָּל רְשָׁעִים, וְהַמַּשְׂכִּילִים יָבִינוּ׳. וּמוֹסִיף: ׳אַשְׁרֵי הַמְחַכֶּה׳ – אַשְׁרֵי מִי שֶׁמְּחַכֶּה לַגְּאֻלָּה.

אַשְׁרֵי מִי שֶׁמְּחַכֶּה.

מָה לוֹקְחִים מִרָמָה א׳?

בְּסוֹף הַסֵּפֶר דָּנִיֵּאל רוֹאֶה שֶׁבִּזְמַן קָשֶׁה יֵשׁ אֲנָשִׁים שֶׁמִּתְחַזְּקִים וּמְלַמְּדִים, וְה׳ שׁוֹמֵר עֲלֵיהֶם. הַגְּאֻלָּה עוֹד תָּבוֹא, וְה׳ אוֹהֵב אֶת מִי שֶׁמְּחַכֶּה.

🤔 חוֹשְׁבִים עַל זֶה

  • הַפָּסוּק אוֹמֵר ׳וְעַם יֹדְעֵי אֱלֹהָיו יַחֲזִקוּ וְעָשׂוּ׳. מָה אֲנִי יָכוֹל לַעֲשׂוֹת הַיּוֹם כְּדֵי לְהַכִּיר אֶת ה׳ יוֹתֵר – וְלֹא רַק לָדַעַת עָלָיו?
  • ׳וּמַשְׂכִּילֵי עָם יָבִינוּ לָרַבִּים׳ – חֲכָמִים שֶׁמְּלַמְּדִים אֲחֵרִים. מִי לִמֵּד אוֹתִי מַשֶּׁהוּ חָשׁוּב, וְלְמִי אֲנִי יָכוֹל לְלַמֵּד?
  • הַפָּסוּק מְדַבֵּר עַל מִי שֶׁמְּחַכֶּה לַגְּאֻלָּה גַּם כְּשֶׁהִיא מִתְעַכֶּבֶת. אֵיךְ אֲנִי מַמְשִׁיךְ לְקַוּוֹת כְּשֶׁדָּבָר חָשׁוּב לֹא קוֹרֶה?

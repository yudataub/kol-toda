הֵן אִיתַי אֱלָהַנָא

רָמָה ב׳ – עִיּוּן בַּנֶּאֱמָנוּת

חֲנַנְיָה מִישָׁאֵל וַעֲזַרְיָה לֹא מִשְׁתַּחֲוִים לְצֶלֶם הַמֶּלֶךְ, נִכְנָסִים לְכִבְשַׁן הָאֵשׁ – וְה׳ מַצִּיל אוֹתָם

📖 פֶּרֶק ג׳

🎯 לַחִידוֹן
🗣️

📜 הַפָּסוּקעֲנוֹ שַׁדְרַךְ מֵישַׁךְ וַעֲבֵד נְגוֹ וְאָמְרִין לְמַלְכָּא נְבוּכַדְנֶצַּר לָא־חַשְׁחִין אֲנַחְנָה עַל־דְּנָה פִּתְגָם לַהֲתָבוּתָךְ

📖 דָּנִיֵּאל, פֶּרֶק ג׳, פָּסוּק ט״ז (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּנֶּאֱמָנוּת

🗣️ רַשִׁ״י: ׳לָא חַשְׁחִין אֲנַחְנָא׳

הַשְּׁלוֹשָׁה אָמְרוּ ׳לָא חַשְׁחִין אֲנַחְנָה עַל דְּנָה פִּתְגָם לַהֲתָבוּתָךְ׳. רַשִׁ״י מְבָאֵר: ׳לֹא חוֹשְׁשִׁין אָנוּ עַל זֹאת לָקַחַת עֵצָה מָה לְהָשִׁיבְךָ, כִּי הַמַּעֲנֶה מוּכָן וְשָׁגוּר בְּפִינוּ׳. הַתְּשׁוּבָה לֹא הִצְטָרְכָה מַחְשָׁבָה, כִּי כְּבָר נֶחְרְתָה בַּלֵּב.

תְּשׁוּבָה שֶׁל אֱמוּנָה אֵינָהּ צְרִיכָה זְמַן הִתְלַבְּטוּת.

📜 הַפָּסוּקעֲנוֹ שַׁדְרַךְ מֵישַׁךְ וַעֲבֵד נְגוֹ וְאָמְרִין לְמַלְכָּא נְבוּכַדְנֶצַּר לָא־חַשְׁחִין אֲנַחְנָה עַל־דְּנָה פִּתְגָם לַהֲתָבוּתָךְ

💛 אַהֲבַת הַשֵּׁם: ה׳ הוּא הַמֶּלֶךְ הָאֲמִתִּי שֶׁאוֹהֵב אוֹתָנוּ 💛

📖 דָּנִיֵּאל, פֶּרֶק ג׳, פָּסוּק ט״ז (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּנֶּאֱמָנוּת

👑 רַשִׁ״י: ׳לְמַלְכָּא נְבוּכַדְנֶצַּר׳

הֵם אָמְרוּ ׳לְמַלְכָּא נְבוּכַדְנֶצַּר׳ – וְרַשִׁ״י שׁוֹאֵל: ׳לָמָּה הוּזְכַּר שְׁמוֹ?׳ וְעוֹנֶה: ׳אִם לְקַבֵּל עָלֵינוּ גְּזֵרָתְךָ לָתֵת מִסִּין וְאַרְנוֹנִיּוֹת – מֶלֶךְ אַתָּה. אִם לִכְפֹּר בַּהַקָּדוֹשׁ בָּרוּךְ הוּא – נְבוּכַדְנֶצַּר שְׁפַל אֲנָשִׁים בְּעֵינֵינוּ׳. מִכָּאן: מְכַבְּדִים אֶת הַמֶּלֶךְ, אֲבָל לֹא נִכְנָעִים בְּעִנְיְנֵי אֱמוּנָה.

בַּכֹּל נִכְנָעִים לַמֶּלֶךְ – חוּץ מֵאֱמוּנָה.

🛟

📜 הַפָּסוּקהֵן אִיתַי אֱלָהַנָא דִּי־אֲנַחְנָא פָלְחִין יָכִל לְשֵׁיזָבוּתַנָא מִן־אַתּוּן נוּרָא יָקִדְתָּא וּמִן־יְדָךְ מַלְכָּא יְשֵׁיזִב

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹטֵחַ בְּךָ, ה׳ – אֵין דָּבָר שֶׁקָּשֶׁה לְךָ 💛

📖 דָּנִיֵּאל, פֶּרֶק ג׳, פָּסוּק י״ז (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנֶּאֱמָנוּת

🛟 רַשִׁ״י: ׳יָכִל לְשֵׁיזָבוּתַנָא׳

׳יָכִל לְשֵׁיזָבוּתַנָא מִן אַתּוּן נוּרָא׳ – רַשִׁ״י: ׳יָכוֹל לְהַצִּילֵנוּ מִכָּל צָרָה׳. וּמוֹסִיף: ׳וּמִן יְדָךְ מַלְכָּא יְשֵׁיזִב – וּמִיָּדְךָ יַצִּיל׳. מְצוּדַת דָּוִד: ׳וּבְכָל אֹפֶן שֶׁתִּרְצֶה לְהָמִית אוֹתָנוּ יָכוֹל הוּא לְהַצִּיל מִכֻּלָּם׳. הַשְּׁלוֹשָׁה הָאֱמִינוּ שֶׁלְּה׳ אֵין מַגְבִּילִים.

ה׳ יָכוֹל לְהַצִּיל מִכָּל צָרָה, בְּכָל אֹפֶן.

3
❤️

📜 הַפָּסוּקוְהֵן לָא יְדִיעַ לֶהֱוֵא־לָךְ מַלְכָּא דִּי לֵאלָהָךְ אִיתַנָא פָלְחִין וּלְצֶלֶם דַּהֲבָא דִּי הֲקֵימְתָּ לָא נִסְגֻּד

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹהֵב אוֹתְךָ, ה׳, בְּלִי שֶׁתַּצִּיל – וְגַם כְּשֶׁתַּצִּיל 💛

📖 דָּנִיֵּאל, פֶּרֶק ג׳, פָּסוּק י״ח (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנֶּאֱמָנוּת

❤️ רַשִׁ״י: ׳וְהֵן לָא׳

הַמִּלִּים הָחֲשׁוּבוֹת: ׳וְהֵן לָא יְדִיעַ לֶהֱוֵא לָךְ׳. רַשִׁ״י מְבָאֵר: ׳וְאִם לֹא יַחְפֹּץ לְהַצִּיל גּוּפֵנוּ – יָדִיעַ יְהֵא לְךָ׳. מְצוּדַת דָּוִד: ׳אַף אִם לֹא יַחְפֹּץ לְהַצִּיל – עַל כָּל זֹאת לֹא נַעֲבֹד לֵאלֹהֶיךָ׳. הָאַהֲבָה לַה׳ אֵינָהּ תְּלוּיָה בְּנֵס.

אוֹהֲבִים אֶת ה׳ גַּם כְּשֶׁלֹּא רוֹאִים נֵס.

🌡️

📜 הַפָּסוּקבֵּאדַיִן נְבוּכַדְנֶצַּר הִתְמְלִי חֱמָא וּצְלֵם אַנְפּוֹהִי אֶשְׁתַּנִּי עַל־שַׁדְרַךְ מֵישַׁךְ וַעֲבֵד נְגוֹ עָנֵה וְאָמַר לְמֵזֵא לְאַתּוּנָא חַד־שִׁבְעָה עַל דִּי חֲזֵה לְמֵזְיֵהּ

💛 אַהֲבַת הַשֵּׁם: ה׳ גָּדוֹל מִכָּל נִסָּיוֹן 💛

📖 דָּנִיֵּאל, פֶּרֶק ג׳, פָּסוּק י״ט (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנֶּאֱמָנוּת

🌡️ רַשִׁ״י: ׳חַד שִׁבְעָה׳

הַמֶּלֶךְ צִוָּה ׳לְמֵזֵא לְאַתּוּנָא חַד שִׁבְעָה עַל דִּי חֲזֵה לְמֵזְיֵהּ׳. רַשִׁ״י מְבָאֵר: ׳עַל אַחַת שֶׁבַע פְּעָמִים שֶׁשִּׁעֲרוּהוּ עַד כְּדֵי הַסָּקוֹ׳. וּמְצוּדַת דָּוִד: ׳לְהוֹסִיף וּלְהַסִּיק בְּפַעַם אַחַת כְּשִׁעוּר שִׁבְעָה הַסָּקוֹת׳. כְּלוֹמַר: הַמֶּלֶךְ רָצָה שֶׁהַנֵּס, אִם יִהְיֶה, יִהְיֶה בִּלְתִּי אֶפְשָׁרִי לְהַכְחִישׁ.

כְּשֶׁמַּגְדִּילִים אֶת הַנִּסָּיוֹן, מַגְדִּילִים גַּם אֶת הַנֵּס.

5
🚶

📜 הַפָּסוּקעָנֵה וְאָמַר הָא־אֲנָה חָזֵה גֻּבְרִין אַרְבְּעָה שְׁרַיִן מַהְלְכִין בְּגוֹא־נוּרָא וַחֲבָל לָא־אִיתַי בְּהוֹן וְרֵוֵהּ דִּי רְבִיעָאָה דָּמֵה לְבַר־אֱלָהִין

💛 אַהֲבַת הַשֵּׁם: ה׳ מְשַׁחְרֵר אוֹתִי מִכָּל מַה שֶּׁקּוֹשֵׁר אוֹתִי 💛

📖 דָּנִיֵּאל, פֶּרֶק ג׳, פָּסוּק כ״ה (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּנֶּאֱמָנוּת

🚶 רַשִׁ״י: ׳שְׁרַיִן מַהְלְכִין׳

הַמֶּלֶךְ רָאָה בִּפְנִים ׳שְׁרַיִן מַהְלְכִין בְּגוֹא נוּרָא׳. רַשִׁ״י מְבָאֵר: ׳מֻתָּרִין מֵאֱסוּרֵיהֶן וְהוֹלְכִים בְּתוֹךְ הָאֵשׁ׳. כְּלוֹמַר: הַחֲבָלִים שֶׁקָּשְׁרוּ אוֹתָם נִשְׂרְפוּ, וְהֵם נִשְׁאֲרוּ חָפְשִׁיִּים. וְעַל הָרְבִיעִי כָּתַב רַשִׁ״י שֶׁהוּא ׳דּוֹמֶה לְמַלְאָךְ׳ שֶׁרָאָה נְבוּכַדְנֶצַּר בִּימֵי סַנְחֵרִיב.

הַנֵּס הוּא לֹא רַק שֶׁלֹּא נִשְׂרְפוּ – אֶלָּא שֶׁנִּשְׁתַּחְרְרוּ.

🌿

📜 הַפָּסוּקעָנֵה נְבוּכַדְנֶצַּר וְאָמַר בְּרִיךְ אֱלָהֲהוֹן דִּי־שַׁדְרַךְ מֵישַׁךְ וַעֲבֵד נְגוֹ דִּי־שְׁלַח מַלְאֲכֵהּ וְשֵׁיזִב לְעַבְדוֹהִי דִּי הִתְרְחִצוּ עֲלוֹהִי

💛 אַהֲבַת הַשֵּׁם: הַבִּטָּחוֹן בְּךָ, ה׳, שׁוֹמֵר עָלַי בְּכָל מָקוֹם 💛

📖 דָּנִיֵּאל, פֶּרֶק ג׳, פְּסוּקִים כ״ז–כ״ח (רַשִׁ״י וּמְצוּדַת דָּוִד)

🌿 רַשִׁ״י: ׳וּשְׂעַר רֵאשְׁהוֹן לָא הִתְחָרַךְ׳

הַנֵּס הָיָה שָׁלֵם: רַשִׁ״י: ׳לֹא שָׁלְטָה הָאֵשׁ בְּגוּפָם, וּשְׂעַר רֹאשָׁם לֹא נִחְרַךְ׳. וּמְצוּדַת דָּוִד: ׳וְאַף רֵיחַ אֵשׁ לֹא עָבְרָה בָּהֶם׳. בַּסּוֹף נְבוּכַדְנֶצַּר אָמַר: ׳בְּרִיךְ אֱלָהֲהוֹן... דִּי הִתְרְחִצוּ עֲלוֹהִי׳ – רַשִׁ״י: ׳אֲשֶׁר בָּטְחוּ עָלָיו׳.

מִי שֶׁבּוֹטֵחַ בַּה׳ – גַּם הָאֵשׁ לֹא עוֹשָׂה בּוֹ רֶשֶׁם.

מָה לוֹקְחִים מִרָמָה ב׳?

בְּדִבְרֵי רַשִׁ״י וּמְצוּדַת דָּוִד רוֹאִים שֶׁהַשְּׁלוֹשָׁה לֹא סָמְכוּ עַל נֵס. הֵם אָמְרוּ: ׳אִם ה׳ יַצִּיל – טוֹב. וְאִם לֹא – עֲדַיִן לֹא נִשְׁתַּחֲוֶה׳. זוֹ אַהֲבָה שֶׁאֵינָהּ תְּלוּיָה בִּתְשׁוּעָה.

🤔 חוֹשְׁבִים עַל זֶה

  • הַשְּׁלוֹשָׁה אָמְרוּ: ׳וְהֵן לָא – יְדִיעַ לֶהֱוֵא לָךְ מַלְכָּא׳. אֲנִי יוֹדֵעַ מָה הַדָּבָר שֶׁאֲנִי לֹא מוּכָן לְוַתֵּר עָלָיו גַּם אִם יִהְיֶה קָשֶׁה?
  • הֵם לֹא הִתְוַכְּחוּ וְלֹא צָעֲקוּ, וְהַתְּשׁוּבָה הָיְתָה ׳מוּכָנָה וּשְׁגוּרָה בְּפִיהֶם׳. מָתַי אֲנִי מַרְגִּישׁ שֶׁאֲנִי יוֹדֵעַ בְּדִיּוּק מָה אֲנִי עוֹמֵד עָלָיו?
  • בַּסּוֹף גַּם נְבוּכַדְנֶצַּר בֵּרֵךְ אֶת ה׳. מָתַי רָאִיתִי מִישֶׁהוּ שֶׁלֹּא הֵבִין, וְהַנֶּאֱמָנוּת שֶׁלִּי הוֹדִיעָה לוֹ מָה זֶה אֱמוּנָה?

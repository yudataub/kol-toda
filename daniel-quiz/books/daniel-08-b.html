וְקַבְּלוּן מַלְכוּתָא קַדִּישֵׁי עֶלְיוֹנִין

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

דָּנִיֵּאל רוֹאֶה בַּלַּיְלָה אֵיךְ הָעוֹלָם נִשְׁפָּט וְהַמַּלְכֻיּוֹת עוֹבְרוֹת – וּבַסּוֹף הַמַּלְכוּת נִתֶּנֶת לְעַם קְדוֹשֵׁי עֶלְיוֹן

📖 פְּרָקִים ז׳–ח׳

📜 הַסֵּפֶר הַמָּלֵא
🪑

📜 הַפָּסוּקחָזֵה הֲוֵית עַד דִּי כָרְסָוָן רְמִיו וְעַתִּיק יוֹמִין יְתִב

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹפֵט בְּרַחֲמִים וְצְדָקָה, כִּי הוּא אוֹהֵב אוֹתָנוּ 💛

📖 דָּנִיֵּאל, פֶּרֶק ז׳, פָּסוּק ט׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

🪑 רַשִׁ״י: ׳כָּרְסָוָן רְמִיו׳

׳כָּרְסָוָן רְמִיו׳ – רַשִׁ״י מְבָאֵר: ׳כִּסְאוֹת הוּטְּלוּ וְנִתְקְנוּ לֵישֵׁב בְּמִשְׁפָּט, אֶחָד לְדִין וְאֶחָד לִצְדָקָה לְיִשְׂרָאֵל׳. שְׁנֵי כִּסְאוֹת – אֶחָד לְדִין וְאֶחָד לִצְדָקָה. ה׳ בּוֹחֵר לְהַעֲמִיד אֶת כִּסֵּא הַצְּדָקָה לְיִשְׂרָאֵל לְצַד כִּסֵּא הַדִּין.

לְצַד הַדִּין יֵשׁ גַּם כִּסֵּא שֶׁל צְדָקָה וְרַחֲמִים.

📜 הַפָּסוּק…לְבוּשֵׁהּ כִּתְלַג חִוָּר

💛 אַהֲבַת הַשֵּׁם: ה׳ מְלַבֵּן אֶת חֲטָאַי בְּאַהֲבָה 💛

📖 דָּנִיֵּאל, פֶּרֶק ז׳, פָּסוּק ט׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

⚪ רַשִׁ״י: ׳כִּתְלַג חִוָּר׳

הַלְּבוּשׁ הָיָה ׳כִּתְלַג חִוָּר׳ – כִּשֶּׁלֶג לָבָן. רַשִׁ״י מְבָאֵר: ׳לְלַבֵּן עֲוֹנוֹת עַמּוֹ׳. הַלֹּבֶן מְסַמֵּל נִקָּיוֹן – ה׳ מְלַבֵּן אֶת הָעֲוֹנוֹת, כְּמוֹ שֶׁהַשֶּׁלֶג מְכַסֶּה אֶת הַכֹּל. וְהַפָּסוּק מְדַבֵּר עַל הַדִּין, אַךְ אֲפִלּוּ הַדִּין כּוֹלֵל נִקּוּי.

לוֹבֶן הַשֶּׁלֶג מְסַמֵּל נִקּוּי הָעֲוֹנוֹת.

🧼

📜 הַפָּסוּק…וּשְׂעַר רֵאשֵׁהּ כַּעֲמַר נְקֵא

💛 אַהֲבַת הַשֵּׁם: ה׳ צַדִּיק וְאוֹהֵב טוֹב 💛

📖 דָּנִיֵּאל, פֶּרֶק ז׳, פָּסוּק ט׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

🧼 רַשִׁ״י: ׳כַּעֲמַר נְקֵא׳

׳וּשְׂעַר רֵאשֵׁהּ כַּעֲמַר נְקֵא׳ – רַשִׁ״י: ׳מְנַקֶּה עַצְמוֹ מִזְּכֻיּוֹת שֶׁיֵּשׁ לָעַכּוּ״ם לְפָנָיו, וּמְשַׁלֵּם לָהֶם כָּל שְׂכָרָם בָּעוֹלָם הַזֶּה בְּוַיִּקְרָא רַבָּה׳. כְּלוֹמַר: ה׳ מַעֲנִיק אֲפִלּוּ לְעַמִּים שֶׁעָשׂוּ טוֹב אֶת שְׂכָרָם בָּעוֹלָם הַזֶּה, וְהַדִּין יִהְיֶה נָקִי.

ה׳ אֵינוֹ מְקַפֵּחַ שְׂכַר אֶף אֶחָד.

3
👤

📜 הַפָּסוּק…וַאֲרוּ עִם־עֲנָנֵי שְׁמַיָּא כְּבַר אֱנָשׁ אָתֵה הֲוָה

💛 אַהֲבַת הַשֵּׁם: ה׳ יָבִיא לָנוּ גְּאֻלָּה שְׁלֵמָה 💛

📖 דָּנִיֵּאל, פֶּרֶק ז׳, פָּסוּק י״ג (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

👤 רַשִׁ״י: ׳כְּבַר אֱנָשׁ אָתֵה הֲוָה׳

רַשִׁ״י מְבָאֵר: ׳כְּבַר אֱנָשׁ אָתֵה – הוּא מֶלֶךְ הַמָּשִׁיחַ׳, ׳וְעַד עַתִּיק יוֹמַיָּא מְטָה – שֶׁהָיָה יוֹשֵׁב בְּמִשְׁפָּט וְדָן אֶת הָאֻמּוֹת׳. וּמְצוּדַת דָּוִד: ׳הָיָה בָּא כִּדְמוּת בֶּן אָדָם, וְעַל מֶלֶךְ הַמָּשִׁיחַ יְרַמֵּז׳. הַמָּשִׁיחַ בָּא מִן הָעֲנָנִים – בְּאוֹפֶן נִסִּי.

מֶלֶךְ הַמָּשִׁיחַ בָּא כְּנֵס מִמַּעַל.

🌍

📜 הַפָּסוּקוּמַלְכוּתָה וְשָׁלְטָנָא וּרְבוּתָא דִּי מַלְכְוָת תְּחוֹת כָּל־שְׁמַיָּא יְהִיבַת לְעַם קַדִּישֵׁי עֶלְיוֹנִין

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אֶת עַמּוֹ וּמוֹשִׁיב אוֹתוֹ בִּכְבוֹד 💛

📖 דָּנִיֵּאל, פֶּרֶק ז׳, פָּסוּק כ״ז (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

🌍 רַשִׁ״י וּמְצוּדַת דָּוִד: ׳לְעַם קַדִּישֵׁי עֶלְיוֹנִין׳

הַפָּסוּק אוֹמֵר: ׳וּמַלְכוּתָה וְשָׁלְטָנָא וּרְבוּתָא... יְהִיבַת לְעַם קַדִּישֵׁי עֶלְיוֹנִין׳. רַשִׁ״י: ׳וְיִשְׁתַּמְעוּן – יִהְיוּ נִשְׁמָעִים לְמִצְוָתוֹ לַעֲשׂוֹתוֹ׳. מְצוּדַת דָּוִד: ׳הַמַּלְכוּת וְהַמֶּמְשָׁלָה וְהַגְּדוּלָה שֶׁל מַלְכֻיּוֹת הָעַכּוּ״ם... יִהְיֶה נָתוּן לְעַם קְדוֹשֵׁי עֶלְיוֹנִים, וְהֵם יִשְׂרָאֵל׳.

הָעוֹלָם כֻּלּוֹ יִשְׁמַע לְמִצְוַת ה׳.

📜 הַפָּסוּקוָאֶשְׁמַע קוֹל־אָדָם בֵּין אוּלָי וַיִּקְרָא וַיֹּאמַר גַּבְרִיאֵל הָבֵן לְהַלָּז אֶת־הַמַּרְאֶה

💛 אַהֲבַת הַשֵּׁם: ה׳ מְחַבֵּב אוֹתִי וּמְכַבֵּד אוֹתִי 💛

📖 דָּנִיֵּאל, פֶּרֶק ח׳, פָּסוּק ט״ז (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

✨ רַשִׁ״י: ׳הָבֵן לְהַלָּז׳

גַּבְרִיאֵל נִקְרָא: ׳הָבֵן לְהַלָּז אֶת הַמַּרְאֶה׳. רַשִׁ״י מְבָאֵר: ׳פֵּרֵשׁ לָזֶה אֶת הֶחָזוֹן׳, וְעַל ׳לְהַלָּז׳ – ׳לְשׁוֹן חֲשִׁיבוּת, כְּמוֹ שֶׁקָּרְאוּ חֲמוּדוֹת, כָּל מָקוֹם שֶׁנֶּאֱמַר הַלָּז בַּעַל צוּרָה הוּא׳. כְּלוֹמַר, הַלָּשׁוֹן הַזּוֹ מַבִּיעָה כָּבוֹד וְחִבָּה כְּלַפֵּי דָּנִיֵּאל.

ה׳ מַתְחַבֵּב עָלַי וּמְכַבֵּד אוֹתִי.

⏳

📜 הַפָּסוּק…כִּי לְמוֹעֵד קֵץ

💛 אַהֲבַת הַשֵּׁם: אֲנִי יוֹדֵעַ שֶׁה׳ לֹא יִכְעַס לָנֶצַח, כִּי הוּא אוֹהֵב אוֹתִי 💛

📖 דָּנִיֵּאל, פֶּרֶק ח׳, פָּסוּק י״ט (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

⏳ מְצוּדַת דָּוִד: ׳כִּי לְמוֹעֵד קֵץ׳

גַּבְרִיאֵל מַרְגִּיעַ: ׳הִנְנִי מוֹדִיעֲךָ אֵת אֲשֶׁר יִהְיֶה בְּאַחֲרִית הַזָּעַם כִּי לְמוֹעֵד קֵץ׳. מְצוּדַת דָּוִד מְבָאֵר: ׳בְּסוֹף יְמֵי הַזָּעַם, הֵם יְמֵי הַגָּלוּת. לְמוֹעֵד זְמַן הַגָּלוּת יִהְיֶה קֵץ וְלֹא לְעוֹלָם יִקְצֹף׳.

הַגָּלוּת קָצוּב, וְה׳ לֹא כּוֹעֵס לָנֶצַח.

💡

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

מָה לוֹקְחִים מִרָמָה ב׳?

רַשִׁ״י וּמְצוּדַת דָּוִד מְבָאֲרִים שֶׁהֶחָזוֹן הוּא חָזוֹן שֶׁל נֶחָמָה: ה׳ יוֹשֵׁב בְּדִין, מְנַקֶּה עֲוֹנוֹת, וּמוֹסֵר אֶת הַמַּלְכוּת לְיִשְׂרָאֵל. כָּל הַחָזוֹן ׳כִּי לְמוֹעֵד קֵץ׳.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • דָּנִיֵּאל חָלַם חָזוֹן גָּדוֹל וְנִבְהַל, אֲבָל ׳מִלְּתָא בְּלִבִּי נִטְרֵת׳ – שָׁמַר אֶת הַדְּבָרִים בְּלִבּוֹ. אֵיזֶה דָּבָר גָּדוֹל אֲנִי שׁוֹמֵר בְּלִבִּי וּמַמְתִּין לַהֲבָנָתוֹ?
  • אַחֲרֵי הַחָזוֹן דָּנִיֵּאל חָלָה, וְאָז ׳וָאָקוּם וָאֶעֱשֶׂה אֶת מְלֶאכֶת הַמֶּלֶךְ׳. מָה עוֹזֵר לִי לָקוּם וְלַעֲשׂוֹת אֶת מַה שֶּׁצָּרִיךְ גַּם כְּשֶׁאֲנִי מֻתְרַע?
  • גַּבְרִיאֵל אוֹמֵר ׳כִּי לְמוֹעֵד קֵץ׳ – לָכֹל יֵשׁ זְמַן סִיּוּם, וְה׳ לֹא כּוֹעֵס לָנֶצַח. אֵיךְ זֶה נוֹתֵן לִי כֹּחַ בִּזְמַן קָשֶׁה?

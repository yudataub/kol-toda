מְנֵא מְנֵא תְּקֵל וּפַרְסִין

בֵּלְשַׁאצַּר שׁוֹתֶה בִּכְלֵי בֵּית הַמִּקְדָּשׁ, כְּתָב מִסְתּוֹרִי עַל הַקִּיר, וְדָנִיֵּאל שֶׁאֵינוֹ מְפַחֵד לוֹמַר אֶת הָאֱמֶת לַמֶּלֶךְ

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹאֶה כָּל מַה שֶּׁקּוֹרֶה בְּבָתֵּינוּ 💛

📖 דָּנִיֵּאל, פֶּרֶק ה׳, פָּסוּק א׳

רָמָה א׳ – הַמִּשְׁתֶּה וְהַכְּתָב

🍷 מִשְׁתֶּה הַמֶּלֶךְ

בֵּלְשַׁאצַּר מַלְכָּא עָשָׂה ׳לְחֶם רַב לְרַבְרְבָנוֹהִי אֲלַף׳ – סְעוּדָה גְּדוֹלָה לְאֶלֶף שָׂרִים, ׳וְלָקֳבֵל אַלְפָּא חַמְרָא שָׁתֵה׳. הוּא שָׁתָה יַיִן מוּל כֻּלָּם, וְהָיָה שִׂמְחָה גְּדוֹלָה בְּאַרְמוֹן הַמֶּלֶךְ.

מִשְׁתֶּה גָּדוֹל בְּאַרְמוֹן הַמֶּלֶךְ.

1
🏺

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אֶת בֵּיתוֹ וּמְכַבֵּד אֶת כֵּלָיו 💛

📖 דָּנִיֵּאל, פֶּרֶק ה׳, פְּסוּקִים ב׳–ד׳

רָמָה א׳ – הַמִּשְׁתֶּה וְהַכְּתָב

🏺 כְּלֵי בֵּית הַמִּקְדָּשׁ

בֵּלְשַׁאצַּר בִּקֵּשׁ ׳בִּטְעֵם חַמְרָא׳ – בְּעֵצַת הַיַּיִן – לְהָבִיא אֶת כְּלֵי הַזָּהָב וְהַכֶּסֶף שֶׁהוֹצִיא נְבוּכַדְנֶצַּר ׳מִן הֵיכְלָא דִּי בִירוּשְׁלֶם׳. הֵם שָׁתוּ בָּהֶם יַיִן, וְשִׁבְּחוּ אֱלֹהֵי זָהָב וְכֶסֶף.

הוּא בִּזָּה אֶת כְּלֵי הַקֹּדֶשׁ.

2
😨

💛 אַהֲבַת הַשֵּׁם: ה׳ יָכוֹל לְדַבֵּר אֵלֵינוּ בְּכָל דֶּרֶךְ 💛

📖 דָּנִיֵּאל, פֶּרֶק ה׳, פְּסוּקִים ה׳–ו׳

רָמָה א׳ – הַמִּשְׁתֶּה וְהַכְּתָב

😨 כְּתָב עַל הַקִּיר

פִּתְאוֹם, ׳בַּהּ שַׁעֲתָה נְפַקָה אֶצְבְּעָן דִּי יַד אֱנָשׁ וְכָתְבָן׳ – יָצְאוּ אֶצְבְּעוֹת כְּיַד אָדָם וְכָתְבוּ עַל טִיחַ הַקִּיר, מוּל הַמְּנוֹרָה. הַמֶּלֶךְ רָאָה, ׳זִיוֺהִי שְׁנוֹהִי׳ – פָּנָיו הִשְׁתַּנּוּ, ׳וְאַרְכֻבָּתֵהּ דָּא לְדָא נָקְשָׁן׳ – בִּרְכָּיו הִתְנַגְּשׁוּ מִפַּחַד.

אֲפִלּוּ מֶלֶךְ מִתְפַּחֵד כְּשֶׁה׳ מוֹדִיעַ.

3
📣

💛 אַהֲבַת הַשֵּׁם: רַק ה׳ יוֹדֵעַ סוֹד כָּל הַסְּתָרִים 💛

📖 דָּנִיֵּאל, פֶּרֶק ה׳, פְּסוּקִים ז׳–ט׳

רָמָה א׳ – הַמִּשְׁתֶּה וְהַכְּתָב

📣 אַף חָכָם לֹא יָכוֹל

הַמֶּלֶךְ קָרָא לַחַרְטֻמִּים: ׳כָּל אֱנָשׁ דִּי יִקְרֵה כְּתָבָה דְנָה וּפִשְׁרֵהּ יְחַוִּנַּנִי – אַרְגְּוָנָא יִלְבַּשׁ וְהַמְנִיכָא דִי דַהֲבָא עַל צַוְּארֵהּ וְתַלְתִּי בְמַלְכוּתָא יִשְׁלַט׳. כְּלוֹמַר, מִי שֶׁיִּקְרָא יִלְבַּשׁ אַרְגָּמָן וְשַׁרְשֶׁרֶת זָהָב וְיִהְיֶה שְׁלִישִׁי בְּמַלְכוּת. אַךְ ׳לָא כָהֲלִין כְּתָבָא לְמִקְרֵא׳ – אֵין מִי שֶׁיָּכוֹל לִקְרֹא.

הַחַרְטֻמִּים לֹא יְכוֹלִים לִקְרֹא אֶת הַכְּתָב.

4
🗣️

💛 אַהֲבַת הַשֵּׁם: ה׳ זוֹכֵר אוֹתִי גַּם כְּשֶׁכֻּלָּם שׁוֹכְחִים 💛

📖 דָּנִיֵּאל, פֶּרֶק ה׳, פְּסוּקִים י׳–י״ב

רָמָה א׳ – הַמִּשְׁתֶּה וְהַכְּתָב

🗣️ הַמַּלְכָּה זוֹכֶרֶת אֶת דָּנִיֵּאל

הַמַּלְכָּה נִכְנְסָה לְבֵית הַמִּשְׁתֶּה וְאָמְרָה: ׳אִיתַי גְּבַר בְּמַלְכוּתָךְ דִּי רוּחַ אֱלָהִין קַדִּישִׁין בֵּהּ... וּבְיוֹמֵי אֲבוּךְ נַהִירוּ וְשָׂכְלְתָנוּ הִשְׁתְּכַחַת בֵּהּ׳. הִיא הִזְכִּירָה אֶת דָּנִיֵּאל, אֲשֶׁר נְבוּכַדְנֶצַּר הֶעֱמִיד כְּרַב חַרְטֻמִּים.

דָּנִיֵּאל כְּבָר נִשְׁכַּח בַּאֲרמוֹן – אַךְ לֹא אֵצֶל ה׳.

5
🙅

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב מִי שֶׁאוֹמֵר אֱמֶת בְּלִי פַּחַד 💛

📖 דָּנִיֵּאל, פֶּרֶק ה׳, פְּסוּקִים י״ג–כ״ב

רָמָה א׳ – הַמִּשְׁתֶּה וְהַכְּתָב

🙅 מַתְּנָתָךְ לָךְ לֶהֶוְיָן

הַמֶּלֶךְ הִבְטִיחַ לְדָנִיֵּאל אַרְגָּמָן וְזָהָב, וְדָנִיֵּאל עָנָה: ׳מַתְּנָתָךְ לָךְ לֶהֶוְיָן וּנְבָזְבְּיָתָךְ לְאָחֳרָן הַב, בְּרַם כְּתָבָא אֶקְרֵא לְמַלְכָּא׳ – הַמַּתָּנוֹת שֶׁלְּךָ יִהְיוּ לְךָ, אֲבָל אֶת הַכְּתָב אֶקְרָא. הוּא גַּם הֶזְכִּיר לַמֶּלֶךְ מָה קָרָה לִנְבוּכַדְנֶצַּר שֶׁהִתְגָּאָה.

דָּנִיֵּאל אוֹמֵר אֱמֶת בְּלִי לָקַחַת מַתָּנוֹת.

6
📜

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב צֶדֶק וּמִשְׁפָּט 💛

📖 דָּנִיֵּאל, פֶּרֶק ה׳, פְּסוּקִים כ״ד–כ״ח

רָמָה א׳ – הַמִּשְׁתֶּה וְהַכְּתָב

📜 מְנֵא מְנֵא תְּקֵל וּפַרְסִין

הַכְּתָב: ׳מְנֵא מְנֵא תְּקֵל וּפַרְסִין׳. דָּנִיֵּאל פָּתַר: ׳מְנֵא – מְנָה אֱלָהָא מַלְכוּתָךְ וְהַשְׁלְמַהּ׳ – ה׳ מָנָה אֶת הַמַּלְכוּת וְסִיֵּם אוֹתָהּ. ׳תְּקֵל – תְּקִילְתָּה בְמֹאזַנְיָא וְהִשְׁתְּכַחַתְּ חַסִּיר׳ – נִשְׁקַלְתָּ וְנִמְצֵאתָ חָסֵר. ׳פְּרֵס – פְּרִיסַת מַלְכוּתָךְ וִיהִיבַת לְמָדַי וּפָרָס׳.

ה׳ שׁוֹקֵל אֶת הַמַּעֲשִׂים בְּצֶדֶק.

7
🌄

💛 אַהֲבַת הַשֵּׁם: ה׳ מֵבִיא גְּאֻלָּה לְעַמּוֹ בַּזְּמַן הַנָּכוֹן 💛

📖 דָּנִיֵּאל, פֶּרֶק ה׳, פְּסוּקִים כ״ט–ל׳

רָמָה א׳ – הַמִּשְׁתֶּה וְהַכְּתָב

🌄 הַמַּלְכוּת עוֹבֶרֶת

הַמֶּלֶךְ קִיֵּם אֶת הַבְטָחָתוֹ וְהִלְבִּישׁ אֶת דָּנִיֵּאל אַרְגָּמָן וְשַׁרְשֶׁרֶת זָהָב. אַךְ בַּלַּיְלָה הַהוּא נִגְמְרָה מַלְכוּת בֵּלְשַׁאצַּר, וְהַמַּלְכוּת עָבְרָה לְדָרְיָוֶשׁ הַמָּדִי. כְּלֵי הַמִּקְדָּשׁ עוֹד יַחְזְרוּ לִירוּשָׁלַיִם.

גַּאֲוָה הוֹפֶכֶת בִּן־לַיְלָה לְאַיִן.

8
💡

רָמָה א׳ – הַמִּשְׁתֶּה וְהַכְּתָב

מָה לוֹקְחִים מִרָמָה א׳?

בֵּלְשַׁאצַּר שָׁכַח אֶת ה׳ וְהִשְׁתַּמֵּשׁ בִּכְלֵי הַמִּקְדָּשׁ לְבִלּוּי. דָּנִיֵּאל לֹא פָּחַד לוֹמַר לוֹ אֶת הָאֱמֶת – כִּי הוּא אוֹהֵב אֶת ה׳ וְאוֹהֵב גַּם אֶת הַמֶּלֶךְ שֶׁיָּשׁוּב.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • דָּנִיֵּאל לֹא הִסְכִּים לָקַחַת מַתָּנוֹת: ׳מַתְּנָתָךְ לָךְ לֶהֶוְיָן׳. מָתַי הָיָה לִי קַל לֹא לָקַחַת מַשֶּׁהוּ, וְאֵיךְ הִרְגַּשְׁתִּי אַחַר כָּךְ?
  • בֵּלְשַׁאצַּר ״הָיָה יוֹדֵעַ״ מַה קָּרָה לִנְבוּכַדְנֶצַּר – ׳וְלָא הַשְׁפֵּלְתְּ לִבְבָךְ כָּל קֳבֵל דִּי כָל דְּנָה יְדַעְתָּ׳. אֵיךְ אֲנִי לוֹמֵד מִטָּעֻיּוֹת שֶׁל אֲחֵרִים וְלֹא מַחְכִּים לַהֲרִיסָה שֶׁלִּי?
  • הַכֵּלִים שֶׁל בֵּית ה׳ הָיוּ קְדוֹשִׁים. אֵיזֶה דְּבָרִים קְדוֹשִׁים יֵשׁ לִי – סֵפֶר, טַלִּית, שֵׁם ה׳ – וְאֵיךְ אֲנִי מִתְיַחֵס אֲלֵיהֶם בְּכָבוֹד?

רָמָה א׳ – הָאָרֶץ וְהָעִיר

🗺️ ׳וּנְחַלְתֶּם אוֹתָהּ אִישׁ כְּאָחִיו׳

בְּסוֹף סֵפֶר יְחֶזְקֵאל ה׳ מְדַבֵּר עַל חֲלֻקַּת הָאָרֶץ בֵּין שְׁנֵים עָשָׂר שִׁבְטֵי יִשְׂרָאֵל: ׳כֹּה אָמַר אֲדֹנָי ה׳ גֵּה גְבוּל אֲשֶׁר תִּתְנַחֲלוּ אֶת הָאָרֶץ לִשְׁנֵי עָשָׂר שִׁבְטֵי יִשְׂרָאֵל׳. וְהַכְּלָל: ׳וּנְחַלְתֶּם אוֹתָהּ אִישׁ כְּאָחִיו אֲשֶׁר נָשָׂאתִי אֶת יָדִי לְתִתָּהּ לַאֲבֹתֵיכֶם׳. כֻּלָּם שָׁווֹת: אָח כְּאָחִיו.

הָאָרֶץ תֵּחָלֵק בֵּין כָּל הַשְּׁבָטִים בְּשָׁוָה – ׳אִישׁ כְּאָחִיו׳.

1
🧭

💛 אַהֲבַת הַשֵּׁם: לְכָל אֶחָד יֵשׁ מָקוֹם אֵצֶל ה׳, וְאֶצְל כֻּלָּנוּ 💛

📖 יְחֶזְקֵאל, פֶּרֶק מ״ח, פְּסוּקִים א׳–ז׳

רָמָה א׳ – הָאָרֶץ וְהָעִיר

🧭 ׳וְאֵלֶּה שְׁמוֹת הַשְּׁבָטִים׳

פֶּרֶק מ״ח מַתְחִיל: ׳וְאֵלֶּה שְׁמוֹת הַשְּׁבָטִים׳, וְהוּא מוֹנֶה אוֹתָם מִצָּפוֹן לַדָּרוֹם: דָּן, אָשֵׁר, נַפְתָּלִי, מְנַשֶּׁה, אֶפְרַיִם, רְאוּבֵן, יְהוּדָה. כָּל שֵׁבֶט מְקַבֵּל רְצוּעָה אַחַת מִן הַמִּזְרָח עַד הַיָּם. ׳דָּן אֶחָד... אָשֵׁר אֶחָד... נַפְתָּלִי אֶחָד׳ – אֶחָד לְכָל שֵׁבֶט, כְּמוֹ שׁוּרוֹת בְּכֶרֶם.

כָּל שֵׁבֶט מְקַבֵּל רְצוּעָה אַחַת, מִן הַצָּפוֹן לַדָּרוֹם.

2
🕍

💛 אַהֲבַת הַשֵּׁם: ה׳ שָׂם אֶת הַמִּקְדָּשׁ בְּלֵב הָאָרֶץ, כִּי הוּא רוֹצֶה לִהְיוֹת בְּלֵב הָעָם וּבְלִבִּי 💛

📖 יְחֶזְקֵאל, פֶּרֶק מ״ח, פְּסוּקִים ח׳–י׳

רָמָה א׳ – הָאָרֶץ וְהָעִיר

🕍 ׳וְהָיָה מִקְדַּשׁ ה׳ בְּתוֹכוֹ׳

בֵּין יְהוּדָה לְבִנְיָמִין יֵשׁ אֵזוֹר מְיֻחָד: ׳תְּרוּמַת הַקֹּדֶשׁ׳. ׳הַתְּרוּמָה אֲשֶׁר תָּרִימוּ לַה׳׳ – חֵלֶק מִן הָאָרֶץ שֶׁנִּתָּן לַה׳. וּבְאֶמְצַע הָאֵזוֹר הַקָּדוֹשׁ: ׳וְהָיָה מִקְדַּשׁ ה׳ בְּתוֹכוֹ׳. הַמִּקְדָּשׁ נִמְצָא בְּלֵב הָאָרֶץ, בָּאֶמְצַע – לֹא בַּצַּד.

הַמִּקְדָּשׁ עוֹמֵד בְּאֶמְצַע הָאֵזוֹר הַקָּדוֹשׁ, בְּלֵב הָאָרֶץ.

3
🕯️

💛 אַהֲבַת הַשֵּׁם: יֵשׁ דְּבָרִים שֶׁאֵינָם לִמְכִירָה – הַקְּדֻשָּׁה שֶׁלִּי שֶׁל ה׳ 💛

📖 יְחֶזְקֵאל, פֶּרֶק מ״ח, פְּסוּקִים י״א–י״ד

רָמָה א׳ – הָאָרֶץ וְהָעִיר

🕯️ ׳כִּי קֹדֶשׁ לַה׳׳

אֶת הַחֵלֶק הַקָּדוֹשׁ מְקַבְּלִים הַכֹּהֲנִים וְהַלְוִיִּם: ׳לַכֹּהֲנִים הַמְקֻדָּשׁ מִבְּנֵי צָדוֹק אֲשֶׁר שָׁמְרוּ מִשְׁמַרְתִּי׳. וְה׳ מוֹסִיף: ׳וְלֹא יִמְכְּרוּ מִמֶּנּוּ... כִּי קֹדֶשׁ לַה׳׳. הַחֵלֶק הַזֶּה אֵינוֹ לִמְכִירָה – הוּא שֶׁל ה׳. הַכֹּהֲנִים שׁוֹמְרִים עָלָיו וּמְשָׁרְתִים בּוֹ.

הַכֹּהֲנִים וְהַלְוִיִּם שׁוֹמְרִים עַל הַחֵלֶק הַקָּדוֹשׁ, וְהוּא אֵינוֹ נִמְכָּר – ׳כִּי קֹדֶשׁ לַה׳׳.

🤲

💛 אַהֲבַת הַשֵּׁם: ה׳ מְקַבֵּל בְּאַהֲבָה כָּל אֶחָד שֶׁרוֹצֶה לִהְיוֹת בְּתוֹךְ עַמּוֹ 💛

📖 יְחֶזְקֵאל, פֶּרֶק מ״ז, פְּסוּקִים כ״ב–כ״ג

רָמָה א׳ – הָאָרֶץ וְהָעִיר

🤲 ׳וְהָיוּ לָכֶם כְּאֶזְרָח׳

ה׳ אוֹמֵר דָּבָר מֵרַגֵּשׁ גַּם עַל מִי שֶׁלֹּא נוֹלַד בְּעַם יִשְׂרָאֵל: ׳וּלְהַגֵּרִים הַגָּרִים בְּתוֹכְכֶם אֲשֶׁר הוֹלִדוּ בָנִים בְּתוֹכְכֶם וְהָיוּ לָכֶם כְּאֶזְרָח בִּבְנֵי יִשְׂרָאֵל׳. ׳וְהָיָה בַשֵּׁבֶט אֲשֶׁר גָּר הַגֵּר אִתּוֹ שָׁם תִּתְּנוּ נַחֲלָתוֹ׳. גַּם הַגֵּר מְקַבֵּל נַחֲלָה בְּתוֹךְ הַשֵּׁבֶט שֶׁהוּא גָּר בּוֹ.

גַּם גֵּר שֶׁמִּצְטָרֵף לְעַם יִשְׂרָאֵל מְקַבֵּל נַחֲלָה כְּמוֹ כָּל אֶזְרָח.

🏙️

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁכֻּלָּנוּ נִבְנֶה אֶת הָעִיר יַחַד 💛

📖 יְחֶזְקֵאל, פֶּרֶק מ״ח, פְּסוּקִים ט״ו, י״ח–י״ט

רָמָה א׳ – הָאָרֶץ וְהָעִיר

🏙️ ׳וְהָעֹבֵד הָעִיר יַעַבְדוּהוּ מִכֹּל שִׁבְטֵי יִשְׂרָאֵל׳

בְּלֵב הָאֵזוֹר הַקָּדוֹשׁ ה׳ שָׂם עִיר: ׳חֹל הוּא לָעִיר לְמוֹשָׁב וּלְמִגְרָשׁ וְהָיְתָה הָעִיר בְּתוֹכוֹ׳. הָעִיר נוֹתֶנֶת לֶחֶם לְכֻלָּם: ׳וְהָיְתָה תְבוּאָתוֹ לְלֶחֶם לְעֹבְדֵי הָעִיר׳. ׳וְהָעֹבֵד הָעִיר יַעַבְדוּהוּ מִכֹּל שִׁבְטֵי יִשְׂרָאֵל׳ – כָּל הַשְּׁבָטִים מִשְׁתַּתְּפִים בַּעֲבוֹדַת הָעִיר, יַחַד.

בָּעִיר, כָּל הַשְּׁבָטִים עוֹבְדִים יַחַד וְכֻלָּם חֵלֶק מִמֶּנָּה.

🚪

💛 אַהֲבַת הַשֵּׁם: לְכָל שֵׁבֶט – וְלִי – יֵשׁ שַׁעַר פָּתוּחַ אֶל הָעִיר שֶׁל ה׳ 💛

📖 יְחֶזְקֵאל, פֶּרֶק מ״ח, פְּסוּקִים ל׳–ל״ד

רָמָה א׳ – הָאָרֶץ וְהָעִיר

🚪 ׳שַׁעֲרֵי הָעִיר עַל שְׁמוֹת שִׁבְטֵי יִשְׂרָאֵל׳

לָעִיר יֵשׁ שְׁנֵים עָשָׂר שְׁעָרִים, וְכָל שַׁעַר נִקְרָא עַל שֵׁם שֵׁבֶט: ׳וְשַׁעֲרֵי הָעִיר עַל שְׁמוֹת שִׁבְטֵי יִשְׂרָאֵל׳. בַּצָּפוֹן: ׳שַׁעַר רְאוּבֵן אֶחָד שַׁעַר יְהוּדָה אֶחָד שַׁעַר לֵוִי אֶחָד׳. בַּמִּזְרָח: יוֹסֵף, בִּנְיָמִין וְדָן. בַּדָּרוֹם: שִׁמְעוֹן, יִשָּׂשכָר וּזְבוּלֻן. בַּמַּעֲרָב: גָּד, אָשֵׁר וְנַפְתָּלִי. שְׁלוֹשָׁה שְׁעָרִים בְּכָל צַד.

בָּעִיר שְׁנֵים עָשָׂר שְׁעָרִים, וְכָל שֵׁבֶט נִכָּנָס דֶּרֶךְ הַשַּׁעַר שֶׁלּוֹ.

💛

💛 אַהֲבַת הַשֵּׁם: ה׳ שָׁם, בְּתוֹךְ הָעִיר וּבְתוֹךְ הַלֵּב שֶׁלִּי 💛

📖 יְחֶזְקֵאל, פֶּרֶק מ״ח, פָּסוּק ל״ה

רָמָה א׳ – הָאָרֶץ וְהָעִיר

💛 ׳וְשֵׁם הָעִיר מִיּוֹם ה׳ שָׁמָּה׳

הַפָּסוּק הָאַחֲרוֹן בְּסֵפֶר יְחֶזְקֵאל: ׳סָבִיב שְׁמֹנָה עָשָׂר אָלֶף וְשֵׁם הָעִיר מִיּוֹם ה׳ שָׁמָּה׳. מֵאוֹתוֹ יוֹם וְהָלְאָה שֵׁם הָעִיר הוּא ׳ה׳ שָׁמָּה׳ – ה׳ נִמְצָא שָׁם, בְּתוֹךְ הָעִיר. הַסֵּפֶר שֶׁהִתְחִיל בָּעֶצֶב שֶׁל הַגָּלוּת, וְה׳ בַּגּוֹלָה עִם עַמּוֹ, מִסְתַּיֵּם בְּכָךְ שֶׁה׳ שׁוֹכֵן בְּתוֹךְ עִירוֹ וּבְתוֹךְ עַמּוֹ.

הַסֵּפֶר נִגְמָר בְּשֵׁם הָעִיר: ׳ה׳ שָׁמָּה׳ – ה׳ שׁוֹכֵן בְּתוֹכָהּ.

מָה לוֹקְחִים מִרָמָה א׳?

סֵפֶר יְחֶזְקֵאל שֶׁהִתְחִיל בַּגּוֹלָה מִסְתַּיֵּם בְּתִקְוָה: הָאָרֶץ נֶחְלֶקֶת ׳אִישׁ כְּאָחִיו׳, יֵשׁ שַׁעַר לְכָל שֵׁבֶט, וְשֵׁם הָעִיר הוּא ׳ה׳ שָׁמָּה׳. ה׳ אוֹהֵב אוֹתָנוּ וְרוֹצֶה לִשְׁכֹּן בְּתוֹכֵנוּ.

🤔 חוֹשְׁבִים עַל זֶה

  • ׳אִישׁ כְּאָחִיו׳ – כֻּלָּם שָׁוִים בַּחֲלֻקַּת הָאָרֶץ. מָתַי אֲנִי מְקַנֵּא בַּחֲבֵרִים שֶׁלִּי, וְאֵיךְ אֲנִי יָכוֹל לִשְׂמֹחַ בְּטוֹבָתָם?
  • גַּם הַגֵּר מְקַבֵּל נַחֲלָה, וְגַם הָעוֹבֵד הַפָּשׁוּט מְקַבֵּל חֵלֶק. אֵיךְ אֲנִי מְקַבֵּל מִישֶׁהוּ חָדָשׁ לַקְּבוּצָה שֶׁלִּי?
  • ׳ה׳ שָׁמָּה׳ – ה׳ שׁוֹכֵן בְּתוֹךְ הָעִיר וְהָעָם. עַל מָה אֲנִי רוֹצֶה לְהוֹדוֹת לַה׳ וּלְבַקֵּשׁ מִמֶּנּוּ בִּתְפִלָּה עַל יְרוּשָׁלַיִם?

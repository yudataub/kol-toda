רָמָה א׳ – הַדֶּרֶךְ מִמִּצְרַיִם

זִקְנֵי הַגּוֹלָה בָּאִים לִדְרֹשׁ אֶת ה׳, וְה׳ מְסַפֵּר לָהֶם אֶת כָּל הַדֶּרֶךְ מִמִּצְרַיִם – וּמַבְטִיחַ: לֹא נִהְיֶה כַּגּוֹיִם, ה׳ יִקְשֹׁר אוֹתָנוּ אֵלָיו וְיִקַבֵּץ אוֹתָנוּ

📖 יְחֶזְקֵאל, פֶּרֶק י״ט, פְּסוּקִים א׳–ט׳

רָמָה א׳ – הַדֶּרֶךְ מִמִּצְרַיִם

🌿 קִינָה עַל נְשִׂיאֵי יִשְׂרָאֵל

יְחֶזְקֵאל מְסַפֵּר מָשָׁל קָצָר וְעָצוּב: ׳מָה אִמְּךָ לְבִיָּא בֵּין אֲרָיוֹת רָבָצָה׳ – לְבִיאָה חֲשׁוּבָה שֶׁגִּדְּלָה גּוּרִים בֵּין הָאֲרָיוֹת. הַמָּשָׁל הוּא עַל בֵּית הַמַּלְכוּת שֶׁל יְהוּדָה. הַגּוּרִים הֵם הַמְּלָכִים הַצְּעִירִים, וְהַקִּינָה אוֹמֶרֶת שֶׁהֵם לֹא הָלְכוּ בַּדֶּרֶךְ הַטּוֹבָה.

הַמָּשָׁל עַל הַלְּבִיאָה וְהַגּוּרִים הוּא קִינָה עַל מְלָכִים שֶׁלֹּא הָלְכוּ בְּדֶרֶךְ ה׳.

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹאֶה כַּמָּה הָיִינוּ יָפִים וְחֲזָקִים, וְהוּא רוֹצֶה לִשְׁתֹּל אוֹתָנוּ מֵחָדָשׁ 💛

📖 יְחֶזְקֵאל, פֶּרֶק י״ט, פְּסוּקִים י׳–י״ד

רָמָה א׳ – הַדֶּרֶךְ מִמִּצְרַיִם

🌱 הַגֶּפֶן שֶׁנִּשְׁתְּלָה

בְּסוֹף הַקִּינָה בָּא מָשָׁל שֵׁנִי: ׳אִמְּךָ כַגֶּפֶן... עַל מַיִם שְׁתוּלָה׳. הַגֶּפֶן הָיְתָה יָפָה וּפוֹרִיָּה, גְּדוֹלָה וְחֲזָקָה – אֲבָל נֶעֶקְרָה וְהוּשְׁלְכָה לָאָרֶץ. ׳וְקִינָה הִיא וְהָיְתָה לְקִינָה׳.

הַגֶּפֶן שֶׁהָיְתָה פּוֹרִיָּה וְנֶעֶקְרָה הִיא מָשָׁל לְמַה שֶּׁקָּרָה לְבֵית דָּוִד.

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹמֵעַ אוֹתִי כְּשֶׁאֲנִי בָּא לִשְׁאֹל וּלְבַקֵּשׁ 💛

📖 יְחֶזְקֵאל, פֶּרֶק כ׳, פְּסוּקִים א׳–ד׳

רָמָה א׳ – הַדֶּרֶךְ מִמִּצְרַיִם

🧓 הַזְּקֵנִים בָּאִים לִדְרֹשׁ

בַּשָּׁנָה הַשְּׁבִיעִית לַגּוֹלָה, בַּחֹדֶשׁ הַחֲמִישִׁי בָּעֲשִׂירִי לַחֹדֶשׁ, בָּאוּ זִקְנֵי יִשְׂרָאֵל לִדְרֹשׁ אֶת ה׳ וְיָשְׁבוּ לִפְנֵי יְחֶזְקֵאל. ה׳ אוֹמֵר לַנָּבִיא: ׳הֲלִדְרֹשׁ אֹתִי אַתֶּם בָּאִים?׳ – וּמְצַוֶּה עָלָיו לְהוֹדִיעַ לָהֶם אֶת כָּל הַדֶּרֶךְ שֶׁל אֲבוֹתֵיהֶם.

הַזְּקֵנִים בָּאוּ לִשְׁאֹל אֶת ה׳, וה׳ עוֹנֶה לָהֶם בְּסִפּוּר מִן הַהַתְחָלָה.

3

💛 אַהֲבַת הַשֵּׁם: ה׳ בָּחַר בִּי וְאוֹהֵב אוֹתִי מֵהַיּוֹם הָרִאשׁוֹן 💛

📖 יְחֶזְקֵאל, פֶּרֶק כ׳, פְּסוּקִים ה׳–ט׳

רָמָה א׳ – הַדֶּרֶךְ מִמִּצְרַיִם

💛 ׳בְּיוֹם בָּחֳרִי בְיִשְׂרָאֵל׳

ה׳ מַתְחִיל מִמִּצְרַיִם: ׳בְּיוֹם בָּחֳרִי בְיִשְׂרָאֵל וָאֶשָּׂא יָדִי לְזֶרַע בֵּית יַעֲקֹב׳. הוּא הוֹצִיא אוֹתָם אֶל אֶרֶץ ׳זָבַת חָלָב וּדְבַשׁ, צְבִי הִיא לְכָל הָאֲרָצוֹת׳, וּבִקֵּשׁ מֵהֶם דָּבָר אֶחָד: ׳אִישׁ שִׁקּוּצֵי עֵינָיו הַשְׁלִיכוּ׳. כְּשֶׁהֵם לֹא שָׁמְעוּ, ה׳ אָמַר ׳וָאַעַשׂ לְמַעַן שְׁמִי׳ – וְלֹא הִפְקִיר אוֹתָם.

ה׳ בָּחַר בָּנוּ, הוֹצִיא אוֹתָנוּ מִמִּצְרַיִם, וְלֹא הִרְפָּה מֵאִתָּנוּ.

4

💛 אַהֲבַת הַשֵּׁם: בְּכָל שַׁבָּת אֲנִי מַרְגִּישׁ שֶׁה׳ נָתַן לִי יוֹם מִיֻּחָד שֶׁל קִרְבָה 💛

📖 יְחֶזְקֵאל, פֶּרֶק כ׳, פְּסוּקִים י״א–י״ב

רָמָה א׳ – הַדֶּרֶךְ מִמִּצְרַיִם

🕯️ ׳אוֹת בֵּינִי וּבֵינֵיהֶם׳

בַּמִּדְבָּר ה׳ נָתַן לָהֶם ׳אֶת חֻקּוֹתַי וְאֶת מִשְׁפָּטַי... אֲשֶׁר יַעֲשֶׂה אוֹתָם הָאָדָם וָחַי בָּהֶם׳ – הַתּוֹרָה הִיא חַיִּים. וְעוֹד: ׳וְגַם אֶת שַׁבְּתוֹתַי נָתַתִּי לָהֶם לִהְיוֹת לְאוֹת בֵּינִי וּבֵינֵיהֶם לָדַעַת כִּי אֲנִי ה׳ מְקַדְּשָׁם׳. הַשַּׁבָּת הִיא מַתָּנָה וְגַם סִימָן שֶׁל קִרְבָה.

הַתּוֹרָה נִתְּנָה כְּדֵי שֶׁנִּחְיֶה בָּהּ, וְהַשַּׁבָּת הִיא אוֹת שֶׁל קִרְבָה בֵּינֵינוּ לְבֵין ה׳.

💛 אַהֲבַת הַשֵּׁם: ה׳ חָס עָלַי וְנוֹתֵן לִי הִזְדַּמְּנוּת חֲדָשָׁה 💛

📖 יְחֶזְקֵאל, פֶּרֶק כ׳, פְּסוּקִים י״ג–י״ט

רָמָה א׳ – הַדֶּרֶךְ מִמִּצְרַיִם

🤲 ׳וַתָּחָס עֵינִי עֲלֵיהֶם׳

בַּמִּדְבָּר הָעָם הִתְרַחֵק, וְה׳ אָמַר שֶׁיִּרְצֶה לְהַעֲנִישׁ – אֲבָל שוּב: ׳וָאַעַשׂ לְמַעַן שְׁמִי׳. ׳וַתָּחָס עֵינִי עֲלֵיהֶם מִשַּׁחֲתָם וְלֹא עָשִׂיתִי אוֹתָם כָּלָה בַּמִּדְבָּר׳. ה׳ פּוֹנֶה אֶל הַבָּנִים: ׳בְּחֻקּוֹתַי לֵכוּ וְאֶת מִשְׁפָּטַי שִׁמְרוּ וַעֲשׂוּ אוֹתָם׳.

ה׳ ׳חָס׳ עַל הָעָם וְלֹא הִרְפָּה, וְהִמְשִׁיךְ לְהַזְמִין אֶת הַבָּנִים לַדֶּרֶךְ הַטּוֹבָה.

6

💛 אַהֲבַת הַשֵּׁם: ה׳ לֹא מַרְפֶּה מִמֶּנִּי וְלֹא נוֹתֵן לִי לְהִסְתַּלֵּק מִמֶּנּוּ 💛

📖 יְחֶזְקֵאל, פֶּרֶק כ׳, פָּסוּק ל״ב

רָמָה א׳ – הַדֶּרֶךְ מִמִּצְרַיִם

🚫 ׳הָיוֹ לֹא תִהְיֶה׳

בַּגּוֹלָה חָשְׁבוּ: ׳נִהְיֶה כַגּוֹיִם׳. ה׳ עוֹנֶה בְּתַקִּיפוּת וּבְאַהֲבָה: ׳וְהָעֹלָה עַל רוּחֲכֶם הָיוֹ לֹא תִהְיֶה׳ – הַמַּחֲשָׁבָה הַזֹּאת לֹא תִּתְקַיֵּם. ה׳ לֹא יַנִּיחַ לָנוּ לְהַפְסִיק לִהְיוֹת הָעָם שֶׁלּוֹ.

׳הָיוֹ לֹא תִהְיֶה׳ – יִשְׂרָאֵל לֹא יִהְיֶה כַּגּוֹיִם, כִּי ה׳ לֹא עוֹזֵב אוֹתָנוּ.

7

💛 אַהֲבַת הַשֵּׁם: ה׳ מְחַכֶּה לִי בְּבַיִת, וְהוּא שָׂמֵחַ כְּשֶׁאֲנִי חוֹזֵר 💛

📖 יְחֶזְקֵאל, פֶּרֶק כ׳, פְּסוּקִים ל״ד, מ׳–מ״ד

רָמָה א׳ – הַדֶּרֶךְ מִמִּצְרַיִם

🏔️ ׳בְּרֵיחַ נִיחֹחַ אֶרְצֶה אֶתְכֶם׳

הַסִּפּוּר נִגְמָר בַּהַבְטָחָה: ׳וְהוֹצֵאתִי אֶתְכֶם מִן הָעַמִּים וְקִבַּצְתִּי אֶתְכֶם מִן הָאֲרָצוֹת׳ וִיבִיאוּ אוֹתָם אֶל הַר קָדְשׁוֹ. ׳בְּרֵיחַ נִיחֹחַ אֶרְצֶה אֶתְכֶם׳ – ה׳ יִהְיֶה מְרֻצֶּה מֵהֶם. וְהַמִּשְׁפָּט הַחוֹזֵר בַּסֵּפֶר: ׳וִידַעְתֶּם כִּי אֲנִי ה׳׳.

ה׳ יְקַבֵּץ אוֹתָנוּ מִכָּל הָאֲרָצוֹת וְיִרְצֶה בָּנוּ – ׳וִידַעְתֶּם כִּי אֲנִי ה׳׳.

מָה לוֹקְחִים מִרָמָה א׳?

יְחֶזְקֵאל מֵבִיא לַגּוֹלָה אֶת הַסִּפּוּר הָאָרֹךְ שֶׁל הַקֶּשֶׁר בֵּינֵינוּ לְבֵין ה׳: הוּא בָּחַר בָּנוּ, הוֹצִיא אוֹתָנוּ, נָתַן לָנוּ תּוֹרָה וְשַׁבָּת – וּבְכָל פַּעַם שֶׁטָּעִינוּ, לֹא עָזַב. וְהַהַבְטָחָה בַּסּוֹף: נִהְיֶה שׁוּב עִמּוֹ, בַּהַר הַקֹּדֶשׁ.

🤔 חוֹשְׁבִים עַל זֶה

  • ה׳ נָתַן לָנוּ אֶת הַשַּׁבָּת ׳לְאוֹת בֵּינִי וּבֵינֵיהֶם׳. מָה בַּשַּׁבָּת שֶׁלִּי גּוֹרֵם לִי לְהַרְגִּישׁ שֶׁאֲנִי קָרוֹב לַה׳?
  • בְּכָל פַּעַם שֶׁהָעָם טָעָה, ה׳ אָמַר ׳וָאַעַשׂ לְמַעַן שְׁמִי׳ וְלֹא עָזַב אוֹתָם. אֵיךְ אֲנִי מַרְגִּישׁ כְּשֶׁאֲנִי יוֹדֵעַ שֶׁה׳ לֹא מַרְפֶּה מִמֶּנִּי?
  • הָעָם חָשַׁב ׳נִהְיֶה כַּגּוֹיִם׳, וְה׳ אָמַר ׳לֹא תִהְיֶה׳. מָה מַחְזִיק אוֹתִי יְהוּדִי גַּם כְּשֶׁסְּבִיבִי הַכֹּל שׁוֹנֶה?

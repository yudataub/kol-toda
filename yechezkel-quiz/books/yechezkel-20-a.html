יְחֶזְקֵאל שׁוֹמֵעַ מֵה׳ מָה קָרָה לָעָם: ׳בֵּית יִשְׂרָאֵל יֹשְׁבִים עַל אַדְמָתָם וַיְטַמְּאוּ אוֹתָהּ בְּדַרְכָּם וּבַעֲלִילוֹתָם׳. הֵם יָשְׁבוּ בָּאָרֶץ הַקְּדוֹשָׁה וְלֹא שָׁמְרוּ עָלֶיהָ. לָכֵן ה׳ ׳אָפִיץ אֹתָם בַּגּוֹיִם וַיִּזָּרוּ בָּאֲרָצוֹת׳. בְּקִצּוּר: הָעָם גָּלָה, וְהִתְפַּזֵּר בֵּין הָעַמִּים.

הָעָם יָצָא לַגָּלוּת כִּי לֹא שָׁמַר עַל הָאָרֶץ הַקְּדוֹשָׁה.

בַּגּוֹלָה קָרָה דָּבָר עָצוּב: הַגּוֹיִם הִסְתַּכְּלוּ עַל בְּנֵי יִשְׂרָאֵל וְאָמְרוּ: ׳עַם ה׳ אֵלֶּה וּמֵאַרְצוֹ יָצָאוּ׳. כְּלוֹמַר, ׳אִם ה׳ הוּא אֱלֹהֵיהֶם – לָמָּה הוּא לֹא הִצִּיל אוֹתָם?׳ כָּךְ ׳חִלְּלוּ אֶת שֵׁם קָדְשִׁי׳. ה׳ ׳חָמַל עַל שֵׁם קָדְשִׁי׳, וְהוּא רוֹצֶה לְתַקֵּן אֶת זֶה.

הַגָּלוּת גָּרְמָה לָעַמִּים לִטְעוֹת וְלַחְשֹׁב שֶׁה׳ עָזַב אֶת עַמּוֹ.

ה׳ אוֹמֵר לְבֵית יִשְׂרָאֵל דָּבָר מַפְתִּיעַ: ׳לֹא לְמַעַנְכֶם אֲנִי עֹשֶׂה בֵּית יִשְׂרָאֵל כִּי אִם לְשֵׁם קָדְשִׁי׳. הַגְּאֻלָּה תָּבוֹא כְּדֵי שֶׁשֵּׁם ה׳ יִתְקַדֵּשׁ: ׳וְקִדַּשְׁתִּי אֶת שְׁמִי הַגָּדוֹל הַמְחֻלָּל בַּגּוֹיִם... וְיָדְעוּ הַגּוֹיִם כִּי אֲנִי ה׳... בְּהִקָּדְשִׁי בָכֶם לְעֵינֵיהֶם׳. כְּשֶׁיִּרְאוּ אוֹתָנוּ חוֹזְרִים הַבַּיְתָה – הַגּוֹיִם יָבִינוּ שֶׁה׳ אִתָּנוּ.

הַגְּאֻלָּה תָּבוֹא כְּדֵי שֶׁשֵּׁם ה׳ יִתְקַדֵּשׁ בְּעֵינֵי כָּל הָעוֹלָם.

הִנֵּה הַהַבְטָחָה: ׳וְלָקַחְתִּי אֶתְכֶם מִן הַגּוֹיִם וְקִבַּצְתִּי אֶתְכֶם מִכָּל הָאֲרָצוֹת וְהֵבֵאתִי אֶתְכֶם אֶל אַדְמַתְכֶם׳. שְׁלוֹשָׁה פְּעָלִים: ׳וְלָקַחְתִּי׳, ׳וְקִבַּצְתִּי׳, ׳וְהֵבֵאתִי׳. ה׳ בְּעַצְמוֹ לוֹקֵחַ אוֹתָנוּ, אוֹסֵף אוֹתָנוּ מִכָּל הָעוֹלָם, וּמֵבִיא אוֹתָנוּ לְאַדְמָתֵנוּ.

ה׳ לוֹקֵחַ, מְקַבֵּץ וּמֵבִיא אֶת עַמּוֹ מִכָּל הָאֲרָצוֹת אֶל אַדְמָתוֹ.

ה׳ בְּעַצְמוֹ מְטַהֵר אֶת הָעָם בְּמַיִם טְהוֹרִים.

הָעוֹלָם שֶׁל הַלֵּב מִשְׁתַּנֶּה: ׳וְנָתַתִּי לָכֶם לֵב חָדָשׁ וְרוּחַ חֲדָשָׁה אֶתֵּן בְּקִרְבְּכֶם וַהֲסִרֹתִי אֶת לֵב הָאֶבֶן מִבְּשַׂרְכֶם וְנָתַתִּי לָכֶם לֵב בָּשָׂר׳. לֵב אֶבֶן הוּא לֵב קָשֶׁה וּסָגוּר. לֵב בָּשָׂר הוּא לֵב רַךְ שֶׁמַּרְגִּישׁ, שֶׁמִּתְרַגֵּשׁ וְשֶׁרוֹצֶה לְהִתְקָרֵב לַה׳. ה׳ מַבְטִיחַ לִתֵּן לָנוּ אֶת הַלֵּב הַזֶּה בְּמַתָּנָה.

ה׳ מַחֲלִיף אֶת לֵב הָאֶבֶן בְּלֵב בָּשָׂר: לֵב חַי שֶׁמַּרְגִּישׁ.

כְּשֶׁהַלֵּב מִשְׁתַּנֶּה, הַכֹּל מִסְתַּדֵּר: ׳וִישַׁבְתֶּם בָּאָרֶץ אֲשֶׁר נָתַתִּי לַאֲבֹתֵיכֶם וִהְיִיתֶם לִי לְעָם וְאָנֹכִי אֶהְיֶה לָכֶם לֵאלֹהִים׳. וְגַם הַפַּרְנָסָה בָּאָה: ׳וְקָרָאתִי אֶל הַדָּגָן וְהִרְבֵּיתִי אֹתוֹ וְלֹא אֶתֵּן עֲלֵיכֶם רָעָב׳. ׳לֹא תִקְחוּ עוֹד חֶרְפַּת רָעָב בַּגּוֹיִם׳.

ה׳ יִהְיֶה לָנוּ לֵאלֹהִים וְאָנוּ נִהְיֶה לוֹ לְעָם, בְּאֶרֶץ שֶׁפַע וּבְלִי רָעָב.

הָאָרֶץ הַשּׁוֹמֵמָה נַעֲשֵׂית כְּגַן עֵדֶן, וְהַגּוֹיִם יֵדְעוּ שֶׁה׳ עָשָׂה זֹאת.

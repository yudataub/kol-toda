ה׳ שָׁמָּה

רָמָה ב׳ – עִיּוּן בַּחֲלֻקַּת הָאָרֶץ

הָאָרֶץ נֶחְלֶקֶת בֵּין הַשְּׁבָטִים ׳אִישׁ כְּאָחִיו׳, יֵשׁ שַׁעַר לְכָל שֵׁבֶט – וְשֵׁם הָעִיר הוּא ׳ה׳ שָׁמָּה׳, כִּי ה׳ שׁוֹכֵן בְּתוֹכָהּ

📖 פֶּרֶק מ״ח

🎯 לַחִידוֹן
⚖️

📜 הַפָּסוּקוּנְחַלְתֶּם אוֹתָהּ אִישׁ כְּאָחִיו אֲשֶׁר

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁבֵּינֵינוּ לֹא תִּהְיֶה קִנְאָה, וְשֶׁכֻּלָּנוּ נִהְיֶה אַחִים בֶּאֱמֶת 💛

📖 יְחֶזְקֵאל, פֶּרֶק מ״ז, פָּסוּק י״ד (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּחֲלֻקַּת הָאָרֶץ

⚖️ רַשִׁ״י: ׳לֹא כָּרִאשׁוֹנָה אִישׁ לְפִי פְּקוּדָיו׳

רַשִׁ״י מְבָאֵר: ׳אִישׁ כְּאָחִיו – לֹא כָּרִאשׁוֹנָה אִישׁ לְפִי פְּקוּדָיו, אֶלָּא עַכְשָׁיו כָּל הַשְּׁבָטִים שָׁוִין׳. וּמְצוּדַת דָּוִד: ׳לֹא כַּחֲלֻקָּה הָרִאשׁוֹנָה שֶׁבִּימֵי יְהוֹשֻׁעַ, שֶׁלָּקְחוּ אִישׁ לְפִי פְּקוּדָיו׳ – כְּלוֹמַר לְפִי מִסְפַּר הָאֲנָשִׁים. בַּחֲלֻקָּה הָרִאשׁוֹנָה שֵׁבֶט גָּדוֹל קִבֵּל יוֹתֵר, וְעַכְשָׁיו כֻּלָּם שָׁוִין. ה׳ לֹא רוֹצֶה שֶׁיִּהְיֶה קִנְאָה בֵּין הַשְּׁבָטִים.

בַּחֲלֻקָּה הָרִאשׁוֹנָה – לְפִי מִסְפַּר הָאֲנָשִׁים; בַּחֲלֻקָּה הָעֲתִידִית – כֻּלָּם שָׁוִין, ׳אִישׁ כְּאָחִיו׳.

1
🌿

📜 הַפָּסוּק…אֲשֶׁר תִּתְנַחֲלוּ אֶת הָאָרֶץ לִשְׁנֵי עָשָׂר שִׁבְטֵי יִשְׂרָאֵל יוֹסֵף חֲבָלִים

💛 אַהֲבַת הַשֵּׁם: לְכָל שֵׁבֶט יֵשׁ תַּפְקִיד, וְלִי יֵשׁ מָקוֹם וְתַפְקִיד אֵצֶל ה׳ 💛

📖 יְחֶזְקֵאל, פֶּרֶק מ״ז, פָּסוּק י״ג (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּחֲלֻקַּת הָאָרֶץ

🌿 רַשִׁ״י: ׳יוֹסֵף חֲבָלִים׳

רַשִׁ״י מְבָאֵר: ׳יוֹסֵף חֲבָלִים – יוֹסֵף יִטּוֹל שְׁנֵי חֲלָקִים, אֶחָד לִמְנַשֶּׁה וְאֶחָד לְאֶפְרַיִם׳. וּמְצוּדַת דָּוִד מַסְבִּיר אֵיךְ הֵם שְׁנֵים עָשָׂר שְׁבָטִים: ׳אִם שֵׁבֶט לֵוִי לֹא יִנְחֲלוּ נַחֲלָה׳, אָמַר: ׳הִנֵּה יוֹסֵף יִקַּח שְׁנֵי חֲלָקִים, אֶחָד לִמְנַשֶּׁה וְאֶחָד לְאֶפְרַיִם׳. כְּלוֹמַר: לֵוִי מְקַבֵּל עֲבוֹדַת ה׳ וְלֹא נַחֲלָה, וְיוֹסֵף – אֶחָד שֶׁמְּקַבֵּל שְׁנַיִם בִּשְׁנֵי בָּנָיו.

יֵשׁ שְׁנֵים עָשָׂר חֲלָקִים: לֵוִי לֹא נוֹחֵל, וְיוֹסֵף מְקַבֵּל שְׁנַיִם (אֶפְרַיִם וּמְנַשֶּׁה).

2
🍇

📜 הַפָּסוּקוְאֵלֶּה שְׁמוֹת הַשְּׁבָטִים מִקְצֵה

💛 אַהֲבַת הַשֵּׁם: ה׳ יוֹדֵעַ בְּדִיּוּק אֵיפֹה הַמָּקוֹם שֶׁלִּי, וְשׁוֹמֵר לִי אוֹתוֹ 💛

📖 יְחֶזְקֵאל, פֶּרֶק מ״ח, פָּסוּק א׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּחֲלֻקַּת הָאָרֶץ

🍇 רַשִׁ״י: ׳כְּמִין שׁוּרַת הַכֶּרֶם׳

רַשִׁ״י מְבָאֵר אֶת הַחֲלֻקָּה (עַל ׳אִישׁ כְּאָחִיו׳, פֶּרֶק מ״ז פָּסוּק י״ד): ׳וּכְמִין שׁוּרַת הַכֶּרֶם כָּל חֵלֶק וְחֵלֶק מִן הַגְּבוּל הַמִּזְרָחִי עַד יָם אוֹקְיָנוֹס לַמַּעֲרָב... וְאֵין שְׁנֵי שְׁבָטִים בִּרְצוּעָה אַחַת׳. כְּלוֹמַר: כְּמוֹ כֶּרֶם עִם שׁוּרוֹת מְסֻדָּרוֹת, כָּל שׁוּרָה לְשֵׁבֶט אֶחָד. וְעַל הַפָּסוּק ׳וְאֵלֶּה שְׁמוֹת הַשְּׁבָטִים׳ רַשִׁ״י: ׳סֵדֶר חֲלָקִים שֶׁיְּחַלְּקוּהָ בֵּינֵיהֶם, מְקוֹם כָּל אֶחָד וְאֶחָד׳. כָּל אֶחָד יוֹדֵעַ בְּדִיּוּק אֵיפֹה מְקוֹמוֹ.

הָאָרֶץ מְחֻלֶּקֶת כְּמוֹ כֶּרֶם: שׁוּרוֹת מְסֻדָּרוֹת, לְכָל שֵׁבֶט שׁוּרָה.

🏛️

📜 הַפָּסוּקוְהָיָה מִקְדַּשׁ ה׳ בְּתוֹכוֹ

💛 אַהֲבַת הַשֵּׁם: ה׳ שָׂם אֶת בֵּיתוֹ בְּאֶמְצַע, וְכֻלָּנוּ סְבִיבוֹ 💛

📖 יְחֶזְקֵאל, פֶּרֶק מ״ח, פָּסוּק י׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּחֲלֻקַּת הָאָרֶץ

🏛️ רַשִׁ״י: ׳הוּא הַר הַבַּיִת׳

רַשִׁ״י מְבָאֵר אֶת הַמִּלִּים ׳וְהָיָה מִקְדַּשׁ ה׳ בְּתוֹכוֹ׳: ׳הוּא הַר הַבַּיִת... בְּאֶמְצַע רְצוּעָה זוֹ, וְכָל סְבִיבוֹתָיו לַכֹּהֲנִים׳. וְעַל ׳לַכֹּהֲנִים הַמְקֻדָּשׁ׳: ׳הַחֵלֶק הַזֶּה הַמְקֻדָּשׁ יִהְיֶה לַכֹּהֲנִים׳. הַמִּקְדָּשׁ אֵינוֹ לְבַד – הוּא מוּקָּף בְּכֹהֲנִים שֶׁשּׁוֹמְרִים עָלָיו, וּמוּקָּף בִּלְוִיִּם וּבְשִׁבְטֵי יִשְׂרָאֵל, וְכֻלָּם סְבִיבוֹ.

הַמִּקְדָּשׁ בְּאֶמְצַע, מוּקָּף כֹּהֲנִים, לְוִיִּם וְכָל הַשְּׁבָטִים.

🤝

📜 הַפָּסוּקוְהָעֹבֵד הָעִיר יַעַבְדוּהוּ מִכֹּל שִׁבְטֵי יִשְׂרָאֵל

💛 אַהֲבַת הַשֵּׁם: כָּל אֶחָד חָשׁוּב לַה׳, גַּם מִי שֶׁעוֹשֶׂה עֲבוֹדָה פְּשׁוּטָה 💛

📖 יְחֶזְקֵאל, פֶּרֶק מ״ח, פְּסוּקִים י״ח–י״ט (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּחֲלֻקַּת הָאָרֶץ

🤝 רַשִׁ״י: ׳שֶׁנָּתְנוּ לָהֶם זֶה חֵלֶק בָּאָרֶץ׳

רַשִׁ״י מְבָאֵר אֶת הַפָּסוּק ׳וְהָעֹבֵד הָעִיר יַעַבְדוּהוּ מִכֹּל שִׁבְטֵי יִשְׂרָאֵל׳: ׳שֶׁנָּתְנוּ לָהֶם זֶה חֵלֶק בָּאָרֶץ׳. וְעַל ׳לְעֹבְדֵי הָעִיר׳ הוּא מֵבִיא בְּשֵׁם רַבִּי מְנַחֵם: הֵם ׳הַגִּבְעוֹנִים הַנְּתוּנִים חוֹטְבֵי עֵצִים וְשׁוֹאֲבֵי מַיִם׳. כְּלוֹמַר: כָּל מִי שֶׁעוֹבֵד אֶת הָעִיר – גַּם מִי שֶׁעוֹשֶׂה אֶת הַמְּלָאכוֹת הַפְּשׁוּטוֹת – מְקַבֵּל חֵלֶק בָּאָרֶץ. אֵין אֶחָד שֶׁנִּשְׁאָר בְּלִי מָקוֹם.

גַּם מִי שֶׁעוֹבֵד עֲבוֹדוֹת פְּשׁוּטוֹת בָּעִיר מְקַבֵּל חֵלֶק בָּאָרֶץ.

🔱

📜 הַפָּסוּקוְשַׁעֲרֵי הָעִיר עַל שְׁמוֹת שִׁבְטֵי יִשְׂרָאֵל שְׁעָרִים

💛 אַהֲבַת הַשֵּׁם: הַשַּׁעַר שֶׁלִּי פָּתוּחַ, וְלֹא סוֹגְרִים אוֹתוֹ בִּפְנֵי אַף אֶחָד 💛

📖 יְחֶזְקֵאל, פֶּרֶק מ״ח, פְּסוּקִים ל׳–ל״א (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּחֲלֻקַּת הָאָרֶץ

🔱 מְצוּדַת דָּוִד: שְׁעָרִים עַל שֵׁם הַשְּׁבָטִים

רַשִׁ״י מְבָאֵר: ׳תּוֹצְאוֹת הָעִיר – שְׁעָרֶיהָ לְאַרְבַּע רוּחוֹתֶיהָ׳. וּמְצוּדַת דָּוִד עַל ׳עַל שְׁמוֹת שִׁבְטֵי יִשְׂרָאֵל׳: ׳יִקָּרְאוּ עַל שְׁמוֹת הַשְּׁבָטִים, וְזֶהוּ לְאוֹת שֶׁיִּהְיֶה לְכָל הַשְּׁבָטִים חֵלֶק בָּעִיר זֶה כָּזֶה׳. וְעַל ׳שַׁעַר רְאוּבֵן אֶחָד׳: ׳אֶחָד יִהְיֶה נִקְרָא שַׁעַר רְאוּבֵן וְכֵן כֻּלָּם׳. הַשְּׁעָרִים הֵם הוֹדָעָה: לְכָל שֵׁבֶט יֵשׁ בַּיִת בָּעִיר ה׳.

שְׁעָרִים עַל שֵׁם הַשְּׁבָטִים: אוֹת שֶׁלְּכָל שֵׁבֶט יֵשׁ חֵלֶק בָּעִיר.

🏷️

📜 הַפָּסוּק…עָשָׂר אָלֶף וְשֵׁם הָעִיר מִיּוֹם ה׳

💛 אַהֲבַת הַשֵּׁם: הַשֵּׁם הַיָּפֶה הַזֶּה שֶׁל יְרוּשָׁלַיִם הוּא בַּיִת שֶׁל ה׳ וְשֶׁלָּנוּ 💛

📖 יְחֶזְקֵאל, פֶּרֶק מ״ח, פָּסוּק ל״ה (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּחֲלֻקַּת הָאָרֶץ

🏷️ רַשִׁ״י: ׳וְשֵׁם הָעִיר׳ – אֵיזֶה שֵׁם?

רַשִׁ״י מְבָאֵר אֶת הַפָּסוּק: ׳תִּרְגֵּם יוֹנָתָן: וּשְׁמָא דְקַרְתָּא דִּמְתָפָּרַשׁ מִיּוֹמָא דִּי יִשְׁרֵי ה׳ שְׁכִינְתֵּיהּ תַּמָּן, כְּלוֹמַר: לֹא יִקָּרֵא לָהּ שֵׁם אַחֵר, אֶלָּא שֵׁם הָרִאשׁוֹן שֶׁהָיָה לָהּ מִיּוֹם שֶׁעָקַד אַבְרָהָם אֶת יִצְחָק בְּנוֹ שָׁם, שֶׁקָּרָא לוֹ ׳יִרְאֶה׳, וְהִיא הָיָה שְׁמָהּ שָׁלֵם – הֲרֵי יְרוּשָׁלַיִם׳. וּמְצוּדַת דָּוִד: ׳מִיּוֹם הַהוּא וָהָלְאָה יִהְיֶה שֵׁם הָעִיר ה׳ שָׁמָּה, עַד עוֹלָם לֹא תָּסִיר עוֹד שְׁכִינָתוֹ מִן הָעִיר׳.

הַשֵּׁם הָעֲתִידִי הוּא הַשֵּׁם הָרִאשׁוֹן: יְרוּשָׁלַיִם = ׳יִרְאֶה׳ + ׳שָׁלֵם׳, וְה׳ שָׁם לָנֶצַח.

רָמָה ב׳ – עִיּוּן בַּחֲלֻקַּת הָאָרֶץ

מָה לוֹקְחִים מִרָמָה ב׳?

רַשִׁ״י וּמְצוּדַת דָּוִד מְגַלִּים שֶׁהַחֲלֻקָּה הָעֲתִידִית שָׁוָה לְכֻלָּם, שֶׁגַּם עוֹבֵד הָעִיר מְקַבֵּל חֵלֶק, וְשֶׁשֵּׁם הָעִיר הוּא הַשֵּׁם הָרִאשׁוֹן שֶׁלָּהּ – שֵׁם שֶׁל חִבּוּר וְשֶׁל קֵרוּב לַה׳.

וְשֵׁם הָעִיר מִיּוֹם ה׳ שָׁמָּה

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳אִישׁ כְּאָחִיו׳ – כֻּלָּם שָׁוִים בַּחֲלֻקַּת הָאָרֶץ. מָתַי אֲנִי מְקַנֵּא בַּחֲבֵרִים שֶׁלִּי, וְאֵיךְ אֲנִי יָכוֹל לִשְׂמֹחַ בְּטוֹבָתָם?
  • גַּם הַגֵּר מְקַבֵּל נַחֲלָה, וְגַם הָעוֹבֵד הַפָּשׁוּט מְקַבֵּל חֵלֶק. אֵיךְ אֲנִי מְקַבֵּל מִישֶׁהוּ חָדָשׁ לַקְּבוּצָה שֶׁלִּי?
  • ׳ה׳ שָׁמָּה׳ – ה׳ שׁוֹכֵן בְּתוֹךְ הָעִיר וְהָעָם. עַל מָה אֲנִי רוֹצֶה לְהוֹדוֹת לַה׳ וּלְבַקֵּשׁ מִמֶּנּוּ בִּתְפִלָּה עַל יְרוּשָׁלַיִם?

ה׳ מוֹכִיחַ אֶת הָרוֹעִים שֶׁדָּאֲגוּ לְעַצְמָם, וּבָא בְּעַצְמוֹ לְחַפֵּשׂ אֶת הַצֹּאן הָאֲבוּדָה – כִּי הוּא אוֹהֵב כָּל אֶחָד מֵעַמּוֹ

רָמָה א׳ – הָרוֹעֶה הַטּוֹב

🐑 ׳הוֹי רֹעֵי יִשְׂרָאֵל׳

יְחֶזְקֵאל בֶּן בּוּזִי הַכֹּהֵן, בַּגּוֹלָה בְּבָבֶל, שׁוֹמֵעַ מֵה׳ נְבוּאָה עַל הַמַּנְהִיגִים. הֵם נִקְרְאוּ ׳רוֹעִים׳, כִּי תַּפְקִידָם לִשְׁמֹר עַל הָעָם כְּמוֹ שֶׁרוֹעֶה שׁוֹמֵר עַל הַצֹּאן. אֲבָל הֵם דָּאֲגוּ רַק לְעַצְמָם: ׳אֶת הַחֵלֶב תֹּאכֵלוּ וְאֶת הַצֶּמֶר תִּלְבָּשׁוּ... הַצֹּאן לֹא תִרְעוּ׳. ה׳ שׁוֹאֵל: ׳הֲלוֹא הַצֹּאן יִרְעוּ הָרֹעִים׳ – הֲרֵי תַּפְקִידוֹ שֶׁל רוֹעֶה לִדְאֹג לַצֹּאן!

מַנְהִיג אֲמִתִּי דּוֹאֵג לַצֹּאן, לֹא לְעַצְמוֹ.

רָמָה א׳ – הָרוֹעֶה הַטּוֹב

🩹 ׳אֶת הַחוֹלָה לֹא רִפֵּאתֶם׳

הַנָּבִיא מוֹנֶה מָה הָיָה צָרִיךְ הָרוֹעֶה לַעֲשׂוֹת: ׳אֶת הַנַּחְלוֹת לֹא חִזַּקְתֶּם וְאֶת הַחוֹלָה לֹא רִפֵּאתֶם וְלַנִּשְׁבֶּרֶת לֹא חֲבַשְׁתֶּם וְאֶת הַנִּדַּחַת לֹא הֲשֵׁבֹתֶם וְאֶת הָאֹבֶדֶת לֹא בִקַּשְׁתֶּם׳. חָמֵשׁ פְּעָמִים ׳לֹא׳! הַכִּבְשָׂה הַחַלָּשָׁה, הַחוֹלָה, הַפְּצוּעָה, זוֹ שֶׁהִתְרַחֲקָה וְזוֹ שֶׁהִלְכָה לְאִבּוּד – אִישׁ לֹא טִפֵּל בָּהֶן.

חָמֵשׁ פְּעָמִים ׳לֹא׳: הָרוֹעִים לֹא טִפְּלוּ בַּחַלָּשִׁים וּבָאֲבוּדִים.

⛰️ ׳וַתְּפוּצֶינָה מִבְּלִי רֹעֶה׳

בְּלִי רוֹעֶה, הַצֹּאן מִתְפַּזֶּרֶת: ׳יִשְׁגּוּ צֹאנִי בְּכָל הֶהָרִים וְעַל כָּל גִּבְעָה רָמָה וְעַל כָּל פְּנֵי הָאָרֶץ נָפֹצוּ צֹאנִי׳. וְהָעֶצֶב הַגָּדוֹל בְּסוֹף הַפָּסוּק: ׳וְאֵין דּוֹרֵשׁ וְאֵין מְבַקֵּשׁ׳ – אַף אֶחָד לֹא שׁוֹאֵל וְלֹא מְחַפֵּשׂ אֶת הַצֹּאן. זֶה מָה שֶׁקָּרָה לְעַם יִשְׂרָאֵל כְּשֶׁגָּלָה לְבָבֶל וְלַמְּקוֹמוֹת הָרְחוֹקִים.

הַצֹּאן הִתְפַּזְּרָה, וְלֹא הָיָה מִי שֶׁיְּחַפֵּשׂ אוֹתָהּ.

רָמָה א׳ – הָרוֹעֶה הַטּוֹב

🤲 ׳הִנְנִי אָנִי וְדָרַשְׁתִּי אֶת צֹאנִי׳

וְאָז ה׳ אוֹמֵר דָּבָר מַפְתִּיעַ: אִם הָרוֹעִים לֹא עָשׂוּ – ׳הִנְנִי אָנִי וְדָרַשְׁתִּי אֶת צֹאנִי וּבִקַּרְתִּים׳. ה׳ עַצְמוֹ בָּא! ׳כְּבַקָּרַת רֹעֶה עֶדְרוֹ בְּיוֹם הֱיוֹתוֹ בְתוֹךְ צֹאנוֹ נִפְרָשׁוֹת כֵּן אֲבַקֵּר אֶת צֹאנִי׳, וְיַצִּיל אוֹתָם ׳מִכָּל הַמְּקוֹמוֹת אֲשֶׁר נָפֹצוּ שָׁם בְּיוֹם עָנָן וַעֲרָפֶל׳ – גַּם מִן הַיָּמִים הַקָּשִׁים וְהַחֲשׁוּכִים.

ה׳ בָּא בְּעַצְמוֹ לְחַפֵּשׂ אֶת הַצֹּאן שֶׁנֶּעֶזְבָה.

🌄

💛 אַהֲבַת הַשֵּׁם: הַבַּיִת שֶׁלָּנוּ מְחַכֶּה לָנוּ, וְה׳ בְּעַצְמוֹ מוֹבִיל אוֹתָנוּ אֵלָיו 💛

📖 יְחֶזְקֵאל, פֶּרֶק ל״ד, פְּסוּקִים י״ג–י״ד

רָמָה א׳ – הָרוֹעֶה הַטּוֹב

🌄 ׳וְהוֹצֵאתִים מִן הָעַמִּים׳

ה׳ מַבְטִיחַ לֶאֱסֹף אֶת עַמּוֹ: ׳וְהוֹצֵאתִים מִן הָעַמִּים וְקִבַּצְתִּים מִן הָאֲרָצוֹת וַהֲבִיאֹתִים אֶל אַדְמָתָם׳. שָׁם יִרְעוּ ׳בְּמִרְעֶה טוֹב׳, בְּהָרֵי מְרוֹם יִשְׂרָאֵל, ׳בְּנָוֶה טוֹב׳. כָּל הַנְּבוּאָה הַזּוֹ בָּאָה בִּזְמַן שֶׁיְּחֶזְקֵאל וַאֲחֵיהֶם יוֹשְׁבִים בַּגּוֹלָה – וְה׳ אוֹמֵר לָהֶם: תַּחְזְרוּ הַבַּיְתָה.

ה׳ יְקַבֵּץ אֶת עַמּוֹ מִכָּל הָאֲרָצוֹת וְיָשִׁיב אוֹתוֹ לְאַדְמָתוֹ.

📖 יְחֶזְקֵאל, פֶּרֶק ל״ד, פְּסוּקִים ט״ו–ט״ז

רָמָה א׳ – הָרוֹעֶה הַטּוֹב

💛 ׳אֶת הָאֹבֶדֶת אֲבַקֵּשׁ׳

זֶה לֵב הַנְּבוּאָה: ׳אֲנִי אֶרְעֶה צֹאנִי וַאֲנִי אַרְבִּיצֵם... אֶת הָאֹבֶדֶת אֲבַקֵּשׁ וְאֶת הַנִּדַּחַת אָשִׁיב וְלַנִּשְׁבֶּרֶת אֶחֱבֹשׁ וְאֶת הַחוֹלָה אֲחַזֵּק׳. כָּל מָה שֶׁהָרוֹעִים לֹא עָשׂוּ – ה׳ עוֹשֶׂה. הַחֲמִשָּׁה ׳לֹא׳ הֵפְכוּ לַחֲמִשָּׁה ׳אֲנִי אֶעֱשֶׂה׳: לְבַקֵּשׁ, לְהָשִׁיב, לַחֲבֹשׁ, לְחַזֵּק וְלִרְעוֹת.

ה׳ מְבַקֵּשׁ אֶת הָאֹבֶדֶת, מַשִׁיב אֶת הַנִּדַּחַת, חוֹבֵשׁ לַנִּשְׁבֶּרֶת וּמְחַזֵּק אֶת הַחוֹלָה.

👑

💛 אַהֲבַת הַשֵּׁם: ה׳ מֵכִין לָנוּ מַנְהִיג טוֹב, כִּי הוּא רוֹצֶה שֶׁיִּהְיֶה לָנוּ טוֹב 💛

רָמָה א׳ – הָרוֹעֶה הַטּוֹב

👑 ׳וַהֲקִמֹתִי עֲלֵיהֶם רֹעֶה אֶחָד׳

אַחֲרֵי שֶׁה׳ בְּעַצְמוֹ מְטַפֵּל בַּצֹּאן, הוּא מַבְטִיחַ רוֹעֶה טוֹב: ׳וַהֲקִמֹתִי עֲלֵיהֶם רֹעֶה אֶחָד וְרָעָה אֶתְהֶן אֵת עַבְדִּי דָוִיד הוּא יִרְעֶה אֹתָם וְהוּא יִהְיֶה לָהֶן לְרֹעֶה׳. דָּוִד הָיָה בְּעַצְמוֹ רוֹעֶה צֹאן לִפְנֵי שֶׁהָיָה מֶלֶךְ, וְהוּא יֹאמַר בִּתְהִלִּים: ׳ה׳ רֹעִי׳. ׳וַאֲנִי ה׳ אֶהְיֶה לָהֶם לֵאלֹהִים׳.

ה׳ יָקִים רוֹעֶה אֶחָד מִזֶּרַע דָּוִד, שֶׁיִּדְאַג לַכֹּל.

בְּסוֹף הַפֶּרֶק בָּאָה הַתְּמוּנָה הַיָּפָה: ׳וְכָרַתִּי לָהֶם בְּרִית שָׁלוֹם... וְיָשְׁבוּ בַמִּדְבָּר לָבֶטַח וְיָשְׁנוּ בַּיְּעָרִים׳. ׳וְהוֹרַדְתִּי הַגֶּשֶׁם בְּעִתּוֹ גִּשְׁמֵי בְרָכָה יִהְיוּ׳, וְהָעֵץ יִתֵּן פִּרְיוֹ וְהָאָרֶץ יְבוּלָהּ. וְאָז ה׳ אוֹמֵר לַצֹּאן שֶׁלּוֹ: ׳וְאַתֵּן צֹאנִי צֹאן מַרְעִיתִי אָדָם אַתֶּם אֲנִי אֱלֹהֵיכֶם׳.

שָׁלוֹם, גְּשָׁמִים בְּעִתָּם וּבְטָחוֹן – וּבְסוֹף ה׳ אוֹמֵר: ׳אָדָם אַתֶּם׳.

מָה לוֹקְחִים מִרָמָה א׳?

יְחֶזְקֵאל שׁוֹמֵעַ בַּגּוֹלָה אֶת אַהֲבַת ה׳ לַצֹּאן שֶׁלּוֹ: כְּשֶׁהָרוֹעִים הִזְנִיחוּ, ה׳ בָּא בְּעַצְמוֹ – ׳אֶת הָאֹבֶדֶת אֲבַקֵּשׁ וְאֶת הַנִּדַּחַת אָשִׁיב׳. ה׳ לֹא מוֹתֵר עַל אַף אֶחָד מֵעַמּוֹ.

🤔 חוֹשְׁבִים עַל זֶה

  • הַכְּבָשָׂה שֶׁנִּדְחֲקָה בְּצַד הָעֵדֶר – מִי בַּכִּתָּה אוֹ בַּשְּׁכוּנָה שֶׁלִּי מַרְגִּישׁ כָּךְ? מָה אֲנִי יָכוֹל לַעֲשׂוֹת הַשָּׁבוּעַ כְּדֵי לְקָרֵב אוֹתוֹ?

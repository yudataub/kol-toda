רָמָה ג׳ – מַעֲמִיקִים

ה׳ מוֹכִיחַ אֶת הַר שֵׂעִיר שֶׁשָּׂמַח לְאֵיד, וְאוֹמֵר לְהָרֵי יִשְׂרָאֵל: תְּנוּ פֵּרוֹת, כִּי הָעָם שֶׁלִּי קָרוֹב לָשׁוּב

🎯 לַחִידוֹן

📜 הַפָּסוּקוְאַתֶּם הָרֵי יִשְׂרָאֵל עַנְפְּכֶם תִּתֵּנוּ וּפֶרְיְכֶם תִּשְׂאוּ לְעַמִּי יִשְׂרָאֵל כִּי קֵרְבוּ לָבוֹא

💛 אַהֲבַת הַשֵּׁם: ה׳ מַרְאֶה לָנוּ אֶת הַגְּאֻלָּה בְּעֵינַיִם – בְּכָל פְּרִי שֶׁאֲנִי אוֹכֵל מִן הָאָרֶץ 💛

📖 יְחֶזְקֵאל, פֶּרֶק ל״ו, פָּסוּק ח׳ (סַנְהֶדְרִין צ״ח ע״א)

רָמָה ג׳ – מַעֲמִיקִים

🍇 ׳אֵין לְךָ קֵץ מְגֻלֶּה מִזֶּה׳

הַגְּמָרָא (סַנְהֶדְרִין צ״ח ע״א) אוֹמֶרֶת: ׳וְאָמַר רַבִּי אַבָּא: אֵין לְךָ קֵץ מְגֻלֶּה מִזֶּה, שֶׁנֶּאֱמַר: וְאַתֶּם הָרֵי יִשְׂרָאֵל עַנְפְּכֶם תִּתֵּנוּ וּפֶרְיְכֶם תִּשְׂאוּ לְעַמִּי יִשְׂרָאֵל׳. כְּלוֹמַר: כְּשֶׁאֶרֶץ יִשְׂרָאֵל תַּתְחִיל לָתֵת פֵּרוֹת בְּשֶׁפַע – זֶה הַסִּימָן הֲכִי בָּרוּר שֶׁהַגְּאֻלָּה קְרוֹבָה. וְרַב בְּאֶרֶץ יִשְׂרָאֵל לִמֵּד: אֵין צֹרֶךְ לְחַכּוֹת לְאוֹת בַּשָּׁמַיִם. מִי שֶׁרוֹאֶה אֶת הָאָרֶץ פּוֹרַחַת, שֶׁהָיְתָה שֹׁמֵמָה, רוֹאֶה אֶת הַנְּבוּאָה מִתְקַיֶּמֶת.

רַבִּי אַבָּא: אֵין סִימָן בָּרוּר לַגְּאֻלָּה יוֹתֵר מֵאֶרֶץ יִשְׂרָאֵל שֶׁנּוֹתֶנֶת פֵּרוֹת בְּשֶׁפַע.

רָמָה ג׳ – מַעֲמִיקִים

🌅 הָאָרֶץ שֶׁחִכְּתָה אַלְפַּיִם שָׁנָה

רַב בְּאֶרֶץ יִשְׂרָאֵל סִפֵּר שֶׁתַּיָּר אָמֵרִיקָנִי שֶׁבִּקֵּר בָּאָרֶץ לִפְנֵי כְּמֵאָה וַחֲמִשִּׁים שָׁנָה כָּתַב שֶׁהִיא שׁוֹמֵמָה: אֲדָמָה רֵיקָה, כִּמְעַט בְּלִי עֵצִים וְכִמְעַט בְּלִי אֲנָשִׁים. כָּךְ הַתּוֹרָה אָמְרָה: ׳וַהֲשִׁמֹּתִי אֲנִי אֶת הָאָרֶץ׳, וְהָאָרֶץ לֹא נָתְנָה פֵּרוֹת לַזָּרִים. אֲבָל מִשֶּׁעַם יִשְׂרָאֵל חָזַר, הָאָרֶץ פּוֹרַחַת וְנוֹתֶנֶת פֵּרוֹת רַבִּים. הַנְּבוּאָה ׳עַנְפְּכֶם תִּתֵּנוּ׳ מִתְקַיֶּמֶת לְעֵינֵינוּ: הַהָרִים הַשּׁוֹמֵמִים הֵם עַכְשָׁיו גַּנִּים וּכְרָמִים.

הָאָרֶץ שֶׁהָיְתָה שׁוֹמֵמָה אַלְפַּיִם שָׁנָה פּוֹרַחַת מִיּוֹם שֶׁבָּנֶיהָ חָזְרוּ.

רָמָה ג׳ – מַעֲמִיקִים

📖 מָה יְחֶזְקֵאל מְלַמֵּד אוֹתִי הַיּוֹם

יְחֶזְקֵאל מְלַמֵּד שְׁנֵי דְּבָרִים. רִאשׁוֹן: לֹא שׂוֹמְחִים לְאֵידוֹ שֶׁל אַחֵר וְלֹא מְדַבְּרִים עָלָיו – כִּי ה׳ שׁוֹמֵעַ. שֵׁנִי: כְּשֶׁאֲנִי אוֹכֵל פְּרִי מֵאֶרֶץ יִשְׂרָאֵל וּמְבָרֵךְ עָלָיו, אֲנִי רוֹאֶה נְבוּאָה מִתְקַיֶּמֶת. אֶפְשָׁר לְבָרֵךְ בְּכַוָּנָה וּלְהוֹדוֹת לַה׳: ׳הָאָרֶץ נוֹתֶנֶת פֵּרוֹת – הַגְּאֻלָּה בַּדֶּרֶךְ׳.

לֹא שׂוֹמְחִים לְאֵיד, וּמוֹדִים לַה׳ עַל כָּל פְּרִי שֶׁהָאָרֶץ נוֹתֶנֶת.

רָמָה ג׳ – מַעֲמִיקִים

מָה לוֹקְחִים מִרָמָה ג׳?

הַנְּבוּאָה ׳עַנְפְּכֶם תִּתֵּנוּ׳ מִתְקַיֶּמֶת בְּדוֹרֵנוּ: הַהָרִים שֶׁהָיוּ שְׁמָמָה נוֹתְנִים פֵּרוֹת, וַאֲנַחְנוּ רוֹאִים אֶת זֶה בְּעֵינֵינוּ. כְּשֶׁנֹּאכַל פְּרִי מֵהָאָרֶץ, נוֹדֶה לַה׳ עַל הַגְּאֻלָּה שֶׁמַּתְחִילָה.

סוֹף רָמָה ג׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳הֶאָח׳ – שִׂמְחָה לְאֵיד. מָתַי נִסִּיתִי לִשְׂמֹחַ כְּשֶׁחָבֵר נִכְשַׁל, וְאֵיךְ אֲנִי רוֹצֶה לְהָגִיב בַּפַּעַם הַבָּאָה?
  • ׳הִנְנִי אֲלֵיכֶם וּפָנִיתִי אֲלֵיכֶם׳ – ה׳ פּוֹנֶה אֶל הָאָרֶץ בְּאַהֲבָה. מַה אֲנִי מַרְגִּישׁ כְּשֶׁאֲנִי יוֹדֵעַ שֶׁה׳ ׳פּוֹנֶה׳ גַּם אֵלַי?
  • ׳עַנְפְּכֶם תִּתֵּנוּ׳ מִתְקַיֵּם בְּדוֹרֵנוּ. בַּפַּעַם הַבָּאָה שֶׁאֲבָרֵךְ עַל פְּרִי מִן הָאָרֶץ, עַל מָה אֲנִי רוֹצֶה לְהוֹדוֹת לַה׳?

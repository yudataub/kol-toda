📜 סֵפֶר יְחֶזְקֵאל · וְאֶת הַנִּדַּחַת אָשִׁיב · רָמָה ב׳

וְאֶת הַנִּדַּחַת אָשִׁיב

רָמָה ב׳ – עִיּוּן בְּמָשָׁל הַצֹּאן

ה׳ מוֹכִיחַ אֶת הָרוֹעִים שֶׁדָּאֲגוּ לְעַצְמָם, וּבָא בְּעַצְמוֹ לְחַפֵּשׂ אֶת הַצֹּאן הָאֲבוּדָה – כִּי הוּא אוֹהֵב כָּל אֶחָד מֵעַמּוֹ

📖 פֶּרֶק ל״ד

🎯 לַחִידוֹן

📜 הַפָּסוּקהוֹי רֹעֵי יִשְׂרָאֵל אֲשֶׁר הָיוּ רֹעִים אוֹתָם הֲלוֹא הַצֹּאן יִרְעוּ הָרֹעִים

💛 אַהֲבַת הַשֵּׁם: ה׳ מְחַפֵּשׂ מַנְהִיגִים שֶׁאוֹהֲבִים אֶת הָעָם כְּמוֹ שֶׁהוּא אוֹהֵב אוֹתוֹ 💛

📖 יְחֶזְקֵאל, פֶּרֶק ל״ד, פָּסוּק ב׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בְּמָשָׁל הַצֹּאן

🏛️ רַשִׁ״י: מִי הֵם הָרוֹעִים?

מִי הֵם ׳רֹעֵי יִשְׂרָאֵל׳? רַשִׁ״י מְבָאֵר: ׳עַל פַּרְנָסֵי יִשְׂרָאֵל׳. וּמַה עָשׂוּ? ׳אֶת עַצְמָם הָיוּ מְעַנְּגִין מִמָּמוֹן חֲבֵרֵיהֶם אֲשֶׁר תַּחְתֵּיהֶם׳. וּמְצוּדַת דָּוִד מוֹסִיף: ׳הֵם הַשָּׂרִים מַנְהִיגֵי הָעָם כָּרוֹעִים הַמַּנְהִיגִים אֶת הַצֹּאן׳. וְהַמִּשְׁפָּט הָאַחֲרוֹן בַּפָּסוּק הוּא שְׁאֵלָה: ׳הֲלוֹא הַצֹּאן יִרְעוּ הָרֹעִים׳, וְרַשִׁ״י: ׳הֲלֹא מִשְׁפַּט הָרוֹעִים לִרְעוֹת אֶת הַצֹּאן וְלֹא לִרְעוֹת אֶת עַצְמָן׳.

תַּפְקִיד הַמַּנְהִיג הוּא לִדְאֹג לַצִּבּוּר, לֹא לְהִתְעַנֵּג מִמָּמוֹנוֹ.

1

📜 הַפָּסוּקאֶת הַנַּחְלוֹת לֹא חִזַּקְתֶּם וְאֶת הַחוֹלָה לֹא רִפֵּאתֶם וְלַנִּשְׁבֶּרֶת לֹא חֲבַשְׁתֶּם

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב דַּוְקָא אֶת הַחַלָּשִׁים וְדוֹאֵג לָהֶם 💛

📖 יְחֶזְקֵאל, פֶּרֶק ל״ד, פָּסוּק ד׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בְּמָשָׁל הַצֹּאן

🩺 רַשִׁ״י: ׳הַנַּחְלוֹת׳ ו׳לֹא חֲבַשְׁתֶּם׳

מָה הַמִּלִּים הַקָּשׁוֹת בְּפָסוּק ד׳? רַשִׁ״י מְבָאֵר: ׳אֶת הַנַּחְלוֹת – אֶת הַכְּחוּשׁוֹת מֵאֵין כֹּחַ, שֶׁצְּרִיכוֹת חִזּוּק לִתְמֹךְ בָּהֶן בַּיָּד׳. וְ׳לֹא חֲבַשְׁתֶּם׳ – ׳לִקְשֹׁר שִׁבְרֵי הָעֶצֶם זֶה עִם זֶה כְּמִשְׁפַּט הָרוֹפְאִים׳. וּמְצוּדַת דָּוִד מַסְבִּיר אֶת הַנִּמְשָׁל: הַמַּנְהִיגִים ׳לֹא חִזְּקוּ אֶת הָעֲנִיִּים׳ לָקַחַת אֶת מִשְׁפָּטָם מִיַּד הָעוֹשְׁקִים.

הַנִּמְשָׁל: מַנְהִיג צָרִיךְ לַעֲזֹר לָעֲנִיִּים וְלַחַלָּשִׁים לְקַבֵּל אֶת הַצֶּדֶק שֶׁלָּהֶם.

2

📜 הַפָּסוּקיִשְׁגּוּ צֹאנִי בְּכָל הֶהָרִים וְעַל כָּל גִּבְעָה רָמָה… וְאֵין דּוֹרֵשׁ וְאֵין מְבַקֵּשׁ

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁה׳ מוֹכִיחַ אוֹתִי, הוּא רוֹצֶה שֶׁאֶחֱזֹר אֵלָיו 💛

📖 יְחֶזְקֵאל, פֶּרֶק ל״ד, פָּסוּק ו׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בְּמָשָׁל הַצֹּאן

🧭 ׳וְאֵין דּוֹרֵשׁ וְאֵין מְבַקֵּשׁ׳

מָה פֵּרוּשׁ הַמִּלִּים ׳יִשְׁגּוּ צֹאנִי׳? רַשִׁ״י מְבָאֵר: ׳הוֹלְכוֹת טוֹעוֹת... נַעֲשׂוּ הֶפְקֵר לַעֲשׂוֹת אִישׁ אִישׁ כִּרְצוֹנוֹ, וְלֹא הוֹכַחְתֶּם אוֹתָם׳. וּמְצוּדַת דָּוִד עַל ׳וְאֵין מְבַקֵּשׁ׳: ׳אֵין מִי שֶׁמַּחֲזִירָם לְמוּטָב לְמַעַן יִהְיוּ נוֹשָׁעִים׳. כְּלוֹמַר: הַמַּנְהִיג לֹא רַק שׁוֹמֵר עַל הַגּוּף – הוּא גַּם מוֹכִיחַ וּמַחֲזִיר בְּאַהֲבָה.

רוֹעֶה טוֹב לֹא מַשְׁאִיר אֶת הַצֹּאן הֶפְקֵר: הוּא מוֹכִיחַ וּמַחֲזִיר לַדֶּרֶךְ.

3

📜 הַפָּסוּקכִּי כֹּה אָמַר אֲדֹנָי ה׳ הִנְנִי אָנִי וְדָרַשְׁתִּי אֶת צֹאנִי וּבִקַּרְתִּים

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁקָּשֶׁה לִי וְהַכֹּל מְעֻנָּן, ה׳ בָּא לְבַקֵּר אוֹתִי 💛

📖 יְחֶזְקֵאל, פֶּרֶק ל״ד, פְּסוּקִים י״א–י״ב (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּמָשָׁל הַצֹּאן

🏥 רַשִׁ״י: ׳וּבִקַּרְתִּים׳ – בִּקּוּר חוֹלִים

ה׳ אוֹמֵר: ׳הִנְנִי אָנִי וְדָרַשְׁתִּי אֶת צֹאנִי וּבִקַּרְתִּים׳. מָה פֵּרוּשׁ ׳וּבִקַּרְתִּים׳? רַשִׁ״י מְבָאֵר: ׳לְשׁוֹן בִּקּוּר חוֹלִים׳! ה׳ כְּמוֹ מְבַקֵּר חוֹלֶה בַּבַּיִת. וְעַל ׳בְּיוֹם עָנָן וַעֲרָפֶל׳ רַשִׁ״י אוֹמֵר: ׳כִּי כֵּן דֶּרֶךְ הַצֹּאן לְהִתְפַּזֵּר בְּיוֹם הַמְעֻנָּן׳. כְּשֶׁהַשָּׁמַיִם אֲפֹרִים וְלֹא רוֹאִים טוֹב – שָׁם הָרוֹעֶה הוֹלֵךְ בֵּין הַכְּבָשׂוֹת וְאוֹסֵף אוֹתָן.

׳וּבִקַּרְתִּים׳ – ה׳ בָּא לְבַקֵּר אוֹתָנוּ כְּמוֹ שֶׁמְּבַקְּרִים חוֹלֶה, וּבִמְיֻחָד בַּיָּמִים הַקָּשִׁים.

4

📜 הַפָּסוּקיַעַן בְּצַד וּבְכָתֵף תֶּהְדֹּפוּ וּבְקַרְנֵיכֶם תְּנַגְּחוּ כָּל הַנַּחְלוֹת עַד אֲשֶׁר הֲפִיצוֹתֶם אוֹתָנָה אֶל הַחוּצָה

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹאֶה כְּשֶׁחָזָק דּוֹחֵף חָלָשׁ, וְהוּא עוֹמֵד לְצַד הַחָלָשׁ 💛

📖 יְחֶזְקֵאל, פֶּרֶק ל״ד, פְּסוּקִים י״ז–כ״א (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּמָשָׁל הַצֹּאן

🐏 ׳בְּצַד וּבְכָתֵף תֶּהְדֹּפוּ׳

ה׳ מְדַבֵּר גַּם אֶל הַכְּבָשׂוֹת: ׳הִנְנִי שֹׁפֵט בֵּין שֶׂה לָשֶׂה׳. רַשִׁ״י מְבָאֵר: ׳בֵּין בַּעֲלֵי אֱגְרוֹפִין לַחַלָּשִׁים׳. הַחֲזָקוֹת רוֹעוֹת בַּמִּרְעֶה הַטּוֹב, אֲבָל גַּם ׳יֶתֶר מִרְעֵיכֶם תִּרְמְסוּ בְּרַגְלֵיכֶם׳ וְגַם הַמַּיִם הַצְּלוּלִים הֵן עוֹכְרוֹת, וְאָז הַחַלָּשׁוֹת שׁוֹתוֹת ׳מִרְפַּשׂ רַגְלֵיכֶם׳. ׳בְּצַד וּבְכָתֵף תֶּהְדֹּפוּ וּבְקַרְנֵיכֶם תְּנַגְּחוּ כָּל הַנַּחְלוֹת׳.

ה׳ שׁוֹפֵט לֹא רַק אֶת הָרוֹעִים – גַּם אֶת הַחֲזָקִים שֶׁדּוֹחֲפִים אֶת הַחַלָּשִׁים.

5

📜 הַפָּסוּקאֶת הָאֹבֶדֶת אֲבַקֵּשׁ וְאֶת הַנִּדַּחַת אָשִׁיב

💛 אַהֲבַת הַשֵּׁם: ה׳ מוֹכִיחַ כִּי הוּא רוֹצֶה לְתַקֵּן, וְלֹא כִּי הוּא רוֹצֶה לְהַעֲנִישׁ 💛

📖 יְחֶזְקֵאל, פֶּרֶק ל״ד, פְּסוּקִים ד׳, ט״ו–ט״ז (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בְּמָשָׁל הַצֹּאן

⚖️ חָמֵשׁ פְּעָמִים ׳לֹא׳ מוּל חָמֵשׁ פְּעָמִים ׳אֲנִי׳

אֵיךְ הַנְּבוּאָה בְּנוּיָה? בְּפָסוּק ד׳ הָרוֹעִים: ׳לֹא חִזַּקְתֶּם, לֹא רִפֵּאתֶם, לֹא חֲבַשְׁתֶּם, לֹא הֲשֵׁבֹתֶם, לֹא בִקַּשְׁתֶּם׳. וּבְפָסוּק ט״ז ה׳ עוֹנֶה כְּנֶגֶד כָּל אֶחָד: ׳אֲבַקֵּשׁ, אָשִׁיב, אֶחֱבֹשׁ, אֲחַזֵּק׳. מְצוּדַת דָּוִד מְבָאֵר שֶׁ׳אֲנִי אֶרְעֶה צֹאנִי׳ מַשְׁמָעוֹ: ׳אֲנִי בְּעַצְמִי אַשְׁגִּיחַ בְּתִקְנָתָם׳. הַתּוֹכֵחָה הִיא לְשֵׁם נֶחָמָה – ה׳ מְתַקֵּן אֶת מָה שֶׁהֻזַּק.

הַתּוֹכֵחָה הִיא הַהַתְחָלָה, וְהַנֶּחָמָה: מָה שֶׁהָרוֹעִים הִזְנִיחוּ – ה׳ מְתַקֵּן בְּעַצְמוֹ.

📜 הַפָּסוּקוְאַתֵּן צֹאנִי צֹאן מַרְעִיתִי אָדָם אַתֶּם אֲנִי

💛 אַהֲבַת הַשֵּׁם: אֲנִי חָשׁוּב לַה׳, וְהוּא קָרוֹב אֵלַי יוֹתֵר מִמַּה שֶׁנִּדְמֶה לִי 💛

📖 יְחֶזְקֵאל, פֶּרֶק ל״ד, פָּסוּק ל״א (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בְּמָשָׁל הַצֹּאן

🕊️ רַשִׁ״י: ׳אָדָם אַתֶּם׳

בַּסּוֹף ה׳ אוֹמֵר: ׳וְאַתֵּן צֹאנִי צֹאן מַרְעִיתִי אָדָם אַתֶּם׳. רַשִׁ״י מְבָאֵר: ׳אֵינְכֶם כִּבְהֵמוֹת בְּעֵינַי׳. וּמְצוּדַת דָּוִד: ׳חֲשׁוּבִים וּכְדֶרֶךְ בֶּן אָדָם, וַאֲנִי מְיַחֵד שְׁמִי עֲלֵיכֶם לְהִקָּרֵא אֱלֹהֵיכֶם, וְלֹא הָיִיתִי מְיַחֵד שְׁמִי עֲלֵיכֶם אִם הֱיִיתֶם בְּעֵינַי כִּבְהֵמוֹת׳. כְּלוֹמַר: אֲפִלּוּ בַּמָּשָׁל שֶׁל צֹאן, ה׳ מַזְכִּיר שֶׁאָנוּ בְּנֵי אָדָם יְקָרִים לוֹ.

ה׳ קוֹרֵא לְעַמּוֹ ׳אָדָם׳ – חָשׁוּב וִיקָר, וְלֹא סְתָם צֹאן.

מָה לוֹקְחִים מִרָמָה ב׳?

רַשִׁ״י וּמְצוּדַת דָּוִד מְגַלִּים כַּמָּה אַהֲבָה יֵשׁ בַּמָּשָׁל: ׳וּבִקַּרְתִּים׳ הוּא בִּקּוּר חוֹלִים, ׳אָדָם אַתֶּם׳ הוּא כָּבוֹד – וְה׳ שׁוֹפֵט גַּם אֶת הַחֲזָקִים שֶׁדּוֹחֲפִים אֶת הַחַלָּשִׁים.

8
❤️
אֶת הָאֹבֶדֶת אֲבַקֵּשׁ וְאֶת הַנִּדַּחַת אָשִׁיב

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • הַכְּבָשָׂה שֶׁנִּדְחֲקָה בְּצַד הָעֵדֶר – מִי בַּכִּתָּה אוֹ בַּשְּׁכוּנָה שֶׁלִּי מַרְגִּישׁ כָּךְ? מָה אֲנִי יָכוֹל לַעֲשׂוֹת הַשָּׁבוּעַ כְּדֵי לְקָרֵב אוֹתוֹ?
  • ׳וּבִקַּרְתִּים׳ הוּא בִּקּוּר חוֹלִים. אֵיךְ אֲנִי מַרְגִּישׁ כְּשֶׁאֲנִי יוֹדֵעַ שֶׁה׳ בָּא לְבַקֵּר אוֹתִי דַּוְקָא כְּשֶׁקָּשֶׁה לִי?
  • ׳אָדָם אַתֶּם׳ – ה׳ קוֹרֵא לָנוּ ׳אָדָם׳. מָה פֵּרוּשׁ לִהְיוֹת ׳אָדָם׳ אֲמִתִּי, וְעַל מָה אֲנִי רוֹצֶה לְהוֹדוֹת לַה׳ שֶׁהוּא רוֹאֶה אוֹתִי?

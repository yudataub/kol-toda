יַעַן גָּבַהּ לִבְּךָ

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

עַמּוֹן וְצוֹר שָׂמְחוּ כְּשֶׁיְּרוּשָׁלַיִם נָפְלָה, וּמֶלֶךְ צוֹר חָשַׁב שֶׁהוּא כְּאֵל – וְהַנָּבִיא מְלַמֵּד: לֹא שׂוֹמְחִים לְאֵיד, וְגַאֲוָה הִיא הָרִאשׁוֹנָה לִפֹּל, וְיִשְׂרָאֵל יֵשְׁבוּ לָבֶטַח

📖 פְּרָקִים כ״ה–כ״ח

🎯 לַחִידוֹן

📜 הַפָּסוּק… דְּבַר אֲדֹנָי ה׳ כֹּה אָמַר אֲדֹנָי ה׳ יַעַן אָמְרֵךְ הֶאָח

💛 אַהֲבַת הַשֵּׁם: הַמִּקְדָּשׁ יָקָר לַה׳, וְלָכֵן הוּא כּוֹאֵב כְּשֶׁשָּׂמֵחִים עָלָיו לְרָעָה 💛

📖 יְחֶזְקֵאל, פֶּרֶק כ״ה, פָּסוּק ג׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

📣 רַשִׁ״י: מַה פֵּרוּשׁ ׳הֶאָח׳ וְ׳נִחָל׳?

רַשִׁ״י מְבָאֵר: ׳הֶאָח – לְשׁוֹן שִׂמְחָה׳, ו׳כִּי נָחַל׳: ׳כִּי נִנְחַל וְהָיָה מוֹרָשָׁה לַאֲחֵרִים׳. וּמְצוּדַת דָּוִד: ׳בַּעֲבוּר שֶׁשָּׂמַחְתָּ עַל מִקְדָּשִׁי אֲשֶׁר נִתְחַלֵּל לְהִמָּסֵר בְּיַד הָעַכּוּ״ם׳. כְּלוֹמַר, עַמּוֹן שָׂמַח שֶׁהַמִּקְדָּשׁ עָבַר לְיָדַיִם אֲחֵרוֹת.

רַשִׁ״י: ׳הֶאָח׳ הוּא קוֹל שִׂמְחָה – וְהוּא שִׂמְחָה עַל מִקְדָּשׁ שֶׁנִּמְסַר לַאֲחֵרִים.

1

📜 הַפָּסוּקכִּי כֹה אָמַר אֲדֹנָי ה׳ יַעַן מַחְאֲךָ יָד

💛 אַהֲבַת הַשֵּׁם: הַלֵּב שֶׁלִּי קוֹבֵעַ אֶת הַמַּשְׁמָעוּת שֶׁל מַה שֶּׁאֲנִי עוֹשֶׂה 💛

📖 יְחֶזְקֵאל, פֶּרֶק כ״ה, פָּסוּק ו׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🖐️ רַשִׁ״י: אוֹתָהּ תְּנוּעָה לְשִׂמְחָה וּלְאֵבֶל

׳מַחְאֲךָ יָד וְרַקְעֲךָ בְּרָגֶל׳ – רַשִׁ״י מְבָאֵר: הֵם עָשׂוּ אֶת זֶה ׳לְשֵׁם שִׂמְחָה׳. וְהוּא מַעֲיר שֶׁבְּרֹאשׁ הַסֵּפֶר (בְּפֶרֶק ו׳) ה׳ אָמַר לִיחֶזְקֵאל ׳הַכֵּה בְכַפְּךָ וּרְקַע בְּרַגְלְךָ׳ לְעִנְיַן צַעַר – ׳יֵשׁ דְּבָרִים שֶׁעוֹשִׂים אוֹתָם לְאֵבֶל וּלְשִׂמְחָה, אֲבָל אֵין עִנְיָנָם שָׁוֶה׳. אוֹתָהּ תְּנוּעָה, אֲבָל שְׁתֵּי כַּוָּנוֹת שׁוֹנוֹת לְגַמְרֵי.

רַשִׁ״י: אוֹתָהּ תְּנוּעָה יְכוֹלָה לְבַטֵּא צַעַר אוֹ שִׂמְחָה – הַכַּוָּנָה שֶׁבַּלֵּב קוֹבַעַת.

2

📜 הַפָּסוּקכֹּה אָמַר אֲדֹנָי ה׳ יַעַן אֲמֹר מוֹאָב וְשֵׂעִיר הִנֵּה כְּכָל הַגּוֹיִם בֵּית יְהוּדָה

💛 אַהֲבַת הַשֵּׁם: ה׳ מַשְׁגִּיחַ עָלַי בְּאַהֲבָה, גַּם כְּשֶׁאֲחֵרִים אוֹמְרִים אַחֶרֶת 💛

📖 יְחֶזְקֵאל, פֶּרֶק כ״ה, פָּסוּק ח׳ (מְצוּדַת דָּוִד וְרַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🪞 מְצוּדַת דָּוִד: מַה אָמַר מוֹאָב?

מוֹאָב וְשֵׂעִיר אָמְרוּ ׳הִנֵּה כְּכָל הַגּוֹיִם בֵּית יְהוּדָה׳. מְצוּדַת דָּוִד מְבָאֵר: ׳כְּמוֹ שֶׁכָּל הָעוֹבְדֵי כּוֹכָבִים הָיוּ נֶעֱזָבִים לְהַנְהָגַת מַעַרְכוֹת הַשָּׁמַיִם, כֵּן גַּם בֵּית יְהוּדָה, וְאֵין דְּבֵקָה בָהֶם הַשְׁגָּחַת ה׳ יוֹתֵר מִבְּכָל הָעוֹבְדֵי כּוֹכָבִים׳. וְרַשִׁ״י: ׳בָּטְלָה אַהֲבַת הַקָּדוֹשׁ בָּרוּךְ הוּא מֵהֶם׳. הַהַשְׁוָאָה: מוֹאָב חוֹשֵׁב שֶׁיְּהוּדָה כְּכָל הָעַמִּים; אֲבָל לְפִי הַנָּבִיא ה׳ אוֹהֵב וּמַשְׁגִּיחַ.

מוֹאָב אָמַר שֶׁה׳ לֹא מַשְׁגִּיחַ עַל יְהוּדָה – וה׳ עוֹנֶה שֶׁהוּא אוֹהֵב אוֹתָם.

3

📜 הַפָּסוּקבֶּן אָדָם יַעַן אֲשֶׁר אָמְרָה צֹּר עַל יְרוּשָׁלִַם הֶאָח נִשְׁבְּרָה דַּלְתוֹת הָעַמִּים

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי מַרְוִיחַ מִצָּרָה שֶׁל מִישֶׁהוּ אַחֵר, זֶה לֹא רֶוַח אֲמִתִּי 💛

📖 יְחֶזְקֵאל, פֶּרֶק כ״ו, פָּסוּק ב׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🚪 רַשִׁ״י: ׳נִשְׁבְּרָה דַּלְתוֹת הָעַמִּים׳

׳נִשְׁבְּרָה דַּלְתוֹת הָעַמִּים׳ – רַשִׁ״י מְבָאֵר: ׳עִיר שֶׁכָּל הָעַכּוּ״ם הָיוּ נִכְנָסִים לָהּ לִסְחוֹרָה, מֵעַתָּה תִּשָּׁבֵר׳, וְ׳נָסֵבָּה אֵלַי׳: ׳מֵעַתָּה הוּסַבָּה סְחוֹרוֹת הָעַמִּים אֵלַי׳. וְ׳אִמָּלְאָה הֶחֳרָבָה׳: ׳אֶתְמַלֵּא אֲנִי מִן הָעִיר יְרוּשָׁלַיִם הַחֲרֵבָה׳. צוֹר חָשְׁבָה לְהַרְוִיחַ מִצָּרָתָהּ שֶׁל אַחֶרֶת.

רַשִׁ״י: צוֹר שָׂמְחָה כִּי חָשְׁבָה שֶׁתִּתְמַלֵּא בִּסְחוֹרוֹת שֶׁל יְרוּשָׁלַיִם.

4

📜 הַפָּסוּק… יָם רֹכֶלֶת הָעַמִּים אֶל אִיִּים רַבִּים כֹּה

💛 אַהֲבַת הַשֵּׁם: מִי שֶׁעָנָו לֹא מַכְבִּיד עַל עַצְמוֹ בְּגַאֲוָה, וְה׳ שׁוֹמֵר עָלָיו 💛

📖 יְחֶזְקֵאל, פֶּרֶק כ״ז, פְּסוּקִים ג׳, ה׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

⛵ רַשִׁ״י: סְפִינָה שֶׁטָּבְעָה

׳צוֹר אַתְּ אָמַרְתְּ אֲנִי כְּלִילַת יֹפִי׳ – רַשִׁ״י מְבָאֵר: ׳עַד עַכְשָׁיו הַכֹּל אוֹמְרִים עַל יְרוּשָׁלַיִם כְּלִילַת יֹפִי... עַכְשָׁיו אַתְּ מִתְגָּאָה לוֹמַר אֲנִי כְּלִילַת הַיֹּפִי׳. וְעַל ׳בְּרוֹשִׁים מִשְּׂנִיר׳: ׳לְפִי שֶׁהָיְתָה בְּנוּיָה בַּיָּם וּבַיָּם אָבְדָה, הִמְשִׁילָהּ הַנָּבִיא לִסְפִינָה מְשֻׁבַּחַת שֶׁהִכְבִּידוּ מַשָּׂאָהּ בְּתוֹכָהּ וְטָבְעָה בְּרוּחַ קָדִים׳. כְּשֶׁמִּתְגָּאִים וּמַעֲמִיסִים עַל עַצְמֵנוּ יוֹתֵר מִדַּי – אֶפְשָׁר לִטְבֹּעַ.

רַשִׁ״י: סְפִינָה שֶׁהִכְבִּידוּ מַשָּׂאָהּ טָבְעָה – מָשָׁל לָעִיר שֶׁהִתְגָּאֲתָה בְּיָפְיָהּ.

5

📜 הַפָּסוּקבְּרֹב חָכְמָתְךָ בִּרְכֻלָּתְךָ הִרְבִּיתָ חֵילֶךָ וַיִּגְבַּהּ לְבָבְךָ בְּחֵילֶךָ

💛 אַהֲבַת הַשֵּׁם: כָּל מַה שֶּׁיֵּשׁ לִי הוּא מַתָּנָה מִה׳, וַאֲנִי מוֹדֶה עָלָיו 💛

📖 יְחֶזְקֵאל, פֶּרֶק כ״ח, פָּסוּק ה׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

💰 מְצוּדַת דָּוִד: ׳הַכֹּל בָּא מִיַּד ה׳׳

׳בְּחָכְמָתְךָ וּבִתְבוּנָתְךָ עָשִׂיתָ לְּךָ חָיִל׳ – רַשִׁ״י מְבָאֵר: ׳קָנִיתָ לְךָ נְכָסִים׳. וּמְצוּדַת דָּוִד: ׳וְכִי בְּרֹב חָכְמָתְךָ... הִרְבֵּיתָ עָשְׁרְךָ עַד שֶׁגָּבַהּ לִבְּךָ... הֲלֹא הַכֹּל בָּא מִיַּד ה׳ וּמָה לְךָ לְהִתְגָּאוֹת בָּזֶה׳. הַלֶּקַח: הַחָכְמָה וְהָעֹשֶׁר הֵם מַתָּנָה, וְלָכֵן אֵין בָּהּ סִבָּה לְהִתְגָּאוֹת.

מְצוּדַת דָּוִד: הַכֹּל בָּא מִיַּד ה׳ – אֵין סִבָּה לְהִתְגָּאוֹת.

📜 הַפָּסוּקוְלֹא יִהְיֶה עוֹד לְבֵית יִשְׂרָאֵל סִלּוֹן מַמְאִיר וְקוֹץ מַכְאִב מִכֹּל

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹמֵר עָלֵינוּ שֶׁלֹּא נִהְיֶה נִרְדָּפִים 💛

📖 יְחֶזְקֵאל, פֶּרֶק כ״ח, פָּסוּק כ״ד (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🌿 רַשִׁ״י: ׳סִלּוֹן מַמְאִיר׳

׳וְלֹא יִהְיֶה עוֹד לְבֵית יִשְׂרָאֵל סִלּוֹן מַמְאִיר וְקוֹץ מַכְאִב׳ – רַשִׁ״י מְבָאֵר: ׳סִלּוֹן – מִין קִמְשׁוֹנִים וְחוֹחִים׳, וְ׳מַמְאִיר – עִנְיַן הַכְאָבָה׳. וְעַל כָּל הָעַמִּים: ׳מֵאַחַר שֶׁיֹּאבְדוּ כֻּלָּן לֹא יִהְיֶה סִלּוֹן וְקוֹץ לְיִשְׂרָאֵל בְּכָל סְבִיבוֹתָם׳. הַשְּׁכֵנִים שֶׁהָיוּ כְּמוֹ קוֹצִים יִפְסְקוּ לַהֲכִאִיב, וְיִשְׂרָאֵל יִתְיַשְּׁבוּ לָבֶטַח.

רַשִׁ״י: ׳סִלּוֹן׳ הוּא קוֹץ – הַשְּׁכֵנִים הַמַּכְאִיבִים לֹא יִהְיוּ עוֹד.

מָה לוֹקְחִים מִרָמָה ב׳?

רַשִׁ״י וּמְצוּדַת דָּוִד מְבָאֲרִים אֶת מִלַּת הַשִּׂמְחָה הָרָעָה ׳הֶאָח׳, אֶת הַסְּפִינָה שֶׁטָּבְעָה מִכֹּבֶד הַמַּשָּׂא, וְאֶת הַהַבְטָחָה לְיִשְׂרָאֵל.

  • עַמּוֹן אָמַר ׳הֶאָח׳ כְּשֶׁהַמִּקְדָּשׁ נִפְגַּע. מָתַי בָּאתִי לְהִשְׂתַּמֵּחַ כְּשֶׁחָבֵר נִכְשַׁל, וּמָה עָשִׂיתִי אַחַר כָּךְ?
  • מֶלֶךְ צוֹר הָיָה עָשִׁיר וְחָכָם, וְגָבַהּ לִבּוֹ. אֵיךְ אֲנִי מַרְגִּישׁ כְּשֶׁיֵּשׁ לִי הַצְלָחָה – וְאֵיךְ אֲנִי נִשְׁאָר עָנָו?
  • ה׳ מַבְטִיחַ שֶׁיִּשְׂרָאֵל ׳יֵשְׁבוּ לָבֶטַח וּבָנוּ בָתִּים וְנָטְעוּ כְרָמִים׳. אֵיפֹה אֲנִי רוֹאֶה אֶת זֶה מִתְקַיֵּם?

בֶּן אָדָם עֲמֹד עַל רַגְלֶיךָ

רָמָה ב׳ – עִיּוּן בַּשְּׁלִיחוּת

יְחֶזְקֵאל הַכֹּהֵן, שֶׁגָּלָה לְבָבֶל, שׁוֹמֵעַ אֶת דְּבַר ה׳ דַּוְקָא בַּגּוֹלָה וְיוֹצֵא לִשְׁלִיחוּת – ה׳ עִם עַמּוֹ גַּם שָׁם

📖 פְּרָקִים א׳–ב׳

🎯 לַחִידוֹן

📜 הַפָּסוּקהָיֹה הָיָה דְבַר ה׳ אֶל יְחֶזְקֵאל בֶּן בּוּזִי הַכֹּהֵן

💛 אַהֲבַת הַשֵּׁם: ה׳ לֹא מַפְסִיק לְדַבֵּר אֵלֵינוּ כְּשֶׁאָנוּ עוֹבְרִים מָקוֹם 💛

📖 יְחֶזְקֵאל, פֶּרֶק א׳, פָּסוּק ג׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּשְּׁלִיחוּת

🏞️ רַשִׁ״י: ׳הָיֹה הָיָה׳

הַכָּתוּב אוֹמֵר ׳הָיֹה הָיָה דְבַר ה׳׳ – לָמָּה הַמִּלָּה כְּפוּלָה? רַשִׁ״י מְבָאֵר: ׳הָיָה כְּבָר׳, ׳לְפִי שֶׁאֵין הַשְּׁכִינָה שׁוֹרָה עַל נָבִיא בְּחוּצָה לָאָרֶץ אֶלָּא עַל יְדֵי שֶׁשָּׁרְתָה עָלָיו תְּחִלָּה בָּאָרֶץ׳. כְּלוֹמַר, ה׳ דִּבֵּר אֶל יְחֶזְקֵאל עוֹד קֹדֶם שֶׁגָּלָה, וְהַדִּבּוּר הִמְשִׁיךְ אִתּוֹ לְבָבֶל.

רַשִׁ״י: ה׳ דִּבֵּר אֶל יְחֶזְקֵאל קֹדֶם בָּאָרֶץ, וְהַדִּבּוּר הִמְשִׁיךְ אִתּוֹ לַגּוֹלָה.

1

📜 הַפָּסוּקבַּחֲמִשָּׁה לַחֹדֶשׁ הִיא הַשָּׁנָה הַחֲמִישִׁית לְגָלוּת הַמֶּלֶךְ יוֹיָכִין

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹלֵחַ נָבִיא בְּדִיּוּק בַּזְּמַן שֶׁאֲנַחְנוּ צְרִיכִים אוֹתוֹ 💛

📖 יְחֶזְקֵאל, פֶּרֶק א׳, פָּסוּק ב׳

רָמָה ב׳ – עִיּוּן בַּשְּׁלִיחוּת

📅 רֶקַע: חָמֵשׁ שָׁנִים בַּגּוֹלָה

הַכָּתוּב מוֹנֶה זְמַן: ׳הִיא הַשָּׁנָה הַחֲמִישִׁית לְגָלוּת הַמֶּלֶךְ יוֹיָכִין׳. הַמֶּלֶךְ יְהוֹיָכִין כְּבָר בְּבָבֶל, וּבִירוּשָׁלַיִם מָלַךְ אַחֲרָיו צִדְקִיָּהוּ, שֶׁהִמְלִיךְ עָלֶיהָ נְבוּכַדְנֶצַּר – כָּךְ מְבָאֵר רַשִׁ״י בְּפֶרֶק י״א. יְרוּשָׁלַיִם עוֹמֶדֶת עֲדַיִן, וְהַגּוֹלִים בְּבָבֶל מְקַוִּים לַחֲזֹר. בְּתוֹךְ הַזְּמַן הַזֶּה הַנָּבִיא מַתְחִיל לְדַבֵּר.

חָמֵשׁ שָׁנִים אַחֲרֵי גָּלוּת יְהוֹיָכִין, כְּשֶׁיְּרוּשָׁלַיִם עוֹמֶדֶת עֲדַיִן – יְחֶזְקֵאל מַתְחִיל לְנַבֵּא.

2

📜 הַפָּסוּקוַיֹּאמֶר אֵלָי בֶּן אָדָם עֲמֹד עַל רַגְלֶיךָ וַאֲדַבֵּר אֹתָךְ

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אוֹתִי כְּשֶׁאֲנִי עָנָו וְקָרוֹב אֵלָיו 💛

📖 יְחֶזְקֵאל, פֶּרֶק ב׳, פָּסוּק א׳ (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּשְּׁלִיחוּת

🌤️ מְצוּדַת דָּוִד: לָמָּה ׳בֶּן אָדָם׳

מְצוּדַת דָּוִד שׁוֹאֵל מַדּוּעַ ה׳ קוֹרֵא לִיחֶזְקֵאל ׳בֶּן אָדָם׳, וּמְבָאֵר: ׳קְרָאוֹ בֶּן אָדָם כְּדֵי שֶׁלֹּא יִתְגָּאֶה׳, אַחֲרֵי הַמַּרְאֶה הַגָּדוֹל שֶׁרָאָה. הַנָּבִיא רָאָה דָּבָר גָּדוֹל מְאֹד, וּבְכָל זֹאת ה׳ מַזְכִּיר לוֹ שֶׁהוּא עֲדַיִן אָדָם. הַמִּלָּה הַזֹּאת שׁוֹמֶרֶת עָלָיו מִגַּאֲוָה.

מְצוּדַת דָּוִד: ׳בֶּן אָדָם׳ – כְּדֵי שֶׁיְּחֶזְקֵאל לֹא יִתְגָּאֶה.

3

📜 הַפָּסוּק…שׁוֹלֵחַ אֲנִי אוֹתְךָ אֶל בְּנֵי יִשְׂרָאֵל אֶל גּוֹיִם הַמּוֹרְדִים

💛 אַהֲבַת הַשֵּׁם: ה׳ זוֹכֵר כָּל שֵׁבֶט וְכָל אֶחָד מִמֶּנּוּ 💛

📖 יְחֶזְקֵאל, פֶּרֶק ב׳, פָּסוּק ג׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּשְּׁלִיחוּת

🤝 רַשִׁ״י: ׳אֶל גּוֹיִם הַמּוֹרְדִים׳

ה׳ שׁוֹלֵחַ אֶת יְחֶזְקֵאל ׳אֶל גּוֹיִם הַמּוֹרְדִים׳. מַדּוּעַ ׳גּוֹיִם׳, בִּלְשׁוֹן רַבִּים? רַשִׁ״י מְבָאֵר: ׳שְׁנֵי שְׁבָטִים קְרוּיִין שְׁנֵי גּוֹיִם׳. מְצוּדַת דָּוִד מוֹסִיף: הֵם יְהוּדָה וּבִנְיָמִין, ׳וְכָל אֶחָד קָרוּי עַם לְעַצְמוֹ׳. הַנָּבִיא מְדַבֵּר אֶל שְׁנֵי הַשְּׁבָטִים שֶׁנִּשְׁאֲרוּ.

רַשִׁ״י: ׳גּוֹיִם׳ – שְׁנֵי שְׁבָטִים, יְהוּדָה וּבִנְיָמִין.

4

📜 הַפָּסוּקוְהַבָּנִים קְשֵׁי פָנִים וְחִזְקֵי לֵב

💛 אַהֲבַת הַשֵּׁם: ה׳ לֹא מוֹתֵר עַל אַף אֶחָד, גַּם לֹא עַל מִי שֶׁקָּשֶׁה לוֹ לִשְׁמֹעַ 💛

📖 יְחֶזְקֵאל, פֶּרֶק ב׳, פָּסוּק ד׳ (מְצוּדַת צִיּוֹן)

רָמָה ב׳ – עִיּוּן בַּשְּׁלִיחוּת

🪨 מְצוּדַת צִיּוֹן: ׳קְשֵׁי פָנִים׳

הַכָּתוּב אוֹמֵר עַל הַבָּנִים: ׳קְשֵׁי פָנִים וְחִזְקֵי לֵב׳. מְצוּדַת צִיּוֹן מְבָאֵר אֶת ׳קְשֵׁי פָנִים׳: ׳הוּא הָפוּךְ מִן בּוֹשֶׁת הַפָּנִים׳. אָדָם עִם פָּנִים רַכּוֹת מִתְבַּיֵּשׁ וּמְשַׁנֶּה אֶת דַּרְכּוֹ, וְכָאן הַפָּנִים קָשׁוֹת וְלֹא מִתְבַּיְּשׁוֹת. ׳מֶרִי׳, כְּמוֹ ׳מְמַרִּים הֱיִיתֶם׳, הוּא עִנְיַן מֶרֶד. וְה׳ בְּכָל זֹאת שׁוֹלֵחַ נָבִיא, כִּי הוּא רוֹצֶה שֶׁיַּחְזְרוּ.

׳קְשֵׁי פָנִים׳ – אֲנָשִׁים שֶׁלֹּא מִתְבַּיְּשִׁים, וּבְכָל זֹאת ה׳ שׁוֹלֵחַ לָהֶם נָבִיא.

5

📜 הַפָּסוּקוְאַתָּה בֶן אָדָם אַל תִּירָא מֵהֶם

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁאֲנָשִׁים מַעֲצִיבִים אוֹתִי, ה׳ שׁוֹמֵר עָלַי וְאוֹמֵר לִי: אַל תִּירָא 💛

📖 יְחֶזְקֵאל, פֶּרֶק ב׳, פָּסוּק ו׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּשְּׁלִיחוּת

🌿 רַשִׁ״י: ׳אֶל עַקְרַבִּים אַתָּה יוֹשֵׁב׳

ה׳ אוֹמֵר: ׳אַל תִּירָא מֵהֶם׳. רַשִׁ״י מְבָאֵר אֶת ׳סָרָבִים וְסַלּוֹנִים׳: ׳לְשׁוֹן קוֹצִים׳. וְעַל ׳אֶל עַקְרַבִּים אַתָּה יוֹשֵׁב׳ הוּא מְבָאֵר: ׳הָעָם אֲשֶׁר אַתָּה מוֹכִיחַ יִלְאוּךָ וִיקְנִיטוּךָ כְּאִלּוּ אַתָּה יוֹשֵׁב עַל עַקְרַבִּים׳. הַמִּלִּים הָאֵלֶּה כּוֹאֲבוֹת, אֲבָל ה׳ אוֹמֵר לוֹ שָׁלוֹשׁ פְּעָמִים לֹא לִפְחֹד.

רַשִׁ״י: אֲנָשִׁים מַקְנִיטִים אוֹתְךָ כְּמוֹ קוֹצִים וְעַקְרַבִּים – ׳אַל תִּירָא׳.

6

📜 הַפָּסוּקוְהֵמָּה אִם יִשְׁמְעוּ וְאִם יֶחְדָּלוּ כִּי בֵּית מְרִי הֵמָּה וְיָדְעוּ כִּי נָבִיא הָיָה בְתוֹכָם

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁה׳ מַזְהִיר אוֹתִי, הוּא עוֹשֶׂה זֹאת כִּי אִכְפַּת לוֹ מִמֶּנִּי 💛

📖 יְחֶזְקֵאל, פֶּרֶק ב׳, פָּסוּק ה׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּשְּׁלִיחוּת

💛 רַשִׁ״י: ׳אֲנִי חָפֵץ שֶׁיֵּדְעוּ׳

רַשִׁ״י מְבָאֵר אֶת ׳וְיָדְעוּ כִּי נָבִיא הָיָה בְתוֹכָם׳: ׳אֶלָּא שֶׁאֲנִי חָפֵץ שֶׁיֵּדְעוּ לְעֵת פֻּרְעָנוּתָם כִּי נָבִיא הָיָה בְּתוֹכָם שֶׁהוֹכִיחָם׳. מְצוּדַת דָּוִד מוֹסִיף: אָז יִהְיֶה תּוֹעֶלֶת, כִּי ׳לֹא יְהַרְהֲרוּ אַחַר מִדַּת הַדִּין׳. ה׳ לֹא רוֹצֶה שֶׁהָעָם יַחְשֹׁב שֶׁהוּא עָזַב אוֹתָם, אֶלָּא שֶׁיֵּדְעוּ שֶׁהוּא הִזְהִיר מֵרֹאשׁ מֵאַהֲבָה.

רַשִׁ״י: ה׳ חָפֵץ שֶׁהָעָם יֵדַע שֶׁהוּא הִזְהִיר אוֹתָם בְּאַהֲבָה לִפְנֵי הַכֹּל.

7

רָמָה ב׳ – עִיּוּן בַּשְּׁלִיחוּת

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מַרְאִים אֶת הָאַהֲבָה שֶׁבַּפְּסוּקִים: הַדִּבּוּר הִמְשִׁיךְ אֶת הַנָּבִיא לַגּוֹלָה, ה׳ שׁוֹמֵר עָלָיו מִגַּאֲוָה בַּמִּלָּה ׳בֶּן אָדָם׳, וְהוּא חָפֵץ שֶׁהָעָם יֵדַע שֶׁהוּא הֻזְהַר בְּאַהֲבָה.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • ה׳ אָמַר לִיחֶזְקֵאל ׳עֲמֹד עַל רַגְלֶיךָ׳ וְגַם נָתַן לוֹ כֹּחַ לַעֲמֹד. מָתַי הִרְגַּשְׁתִּי שֶׁה׳ נָתַן לִי כֹּחַ לַעֲשׂוֹת דָּבָר שֶׁהָיָה קָשֶׁה לִי?
  • יְחֶזְקֵאל שָׁמַע אֶת דְּבַר ה׳ ׳שָׁם׳, בַּגּוֹלָה. אֵיפֹה אֲנִי מַרְגִּישׁ אֶת ה׳ קָרוֹב אֵלַי – בְּבֵית הַכְּנֶסֶת, בַּטֶּבַע, בַּבַּיִת – וְעַל מָה אֲנִי רוֹצֶה לְהוֹדוֹת לוֹ?
  • ׳אִם יִשְׁמְעוּ וְאִם יֶחְדָּלוּ׳ – גַּם כְּשֶׁלֹּא בָּטוּחַ שֶׁיַּקְשִׁיבוּ, ה׳ שׁוֹלֵחַ לְדַבֵּר. אֵיזֶה דָּבָר טוֹב אֲנִי יָכוֹל לוֹמַר הַשָּׁבוּעַ לְמִישֶׁהוּ, גַּם אִם לֹא אֶרְאֶה מִיָּד אִם הוּא הִקְשִׁיב?

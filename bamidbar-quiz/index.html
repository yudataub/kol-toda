🏕️ חִידוֹנֵי חֻמַּשׁ בַּמִּדְבָּר

63 חִידוֹנִים לְפִי תָּכְנִית הַלִּמּוּדִים שֶׁל מִשְׂרַד הַחִנּוּךְ (תַּנַ״ךְ חֶמֶ״ד), לְכִתָּה ה׳. בְּכָל חִידוֹן כַּרְטִיסֵי לִמּוּד עִם אַהֲבַת ה׳ 💛 וְשָׁלוֹשׁ רָמוֹת: תְּמוּנוֹת, שְׁאֵלוֹת לַחֲכָמִים, וּבָרָמָה הַשְּׁלִישִׁית מִשְׂחָקִים – בְּכָל פַּעַם מִשְׂחָק אַחֵר

🏕️ סִדּוּר הַמַּחֲנוֹת וְהַהֲכָנוֹת לַמַּסָּע

1. פְּתִיחַת הַסֵּפֶר וְהַצִּוּוּי עַל הַמִּפְקָדפֶּרֶק א׳, פְּסוּקִים א׳–ט״ז
2. מִסְפְּרֵי הַשְּׁבָטִים וְשֵׁבֶט לֵוִיפֶּרֶק א׳, פְּסוּקִים י״ז–נ״ד
3. סֵדֶר הַדְּגָלִים וְהַמַּחֲנֶהפֶּרֶק ב׳, פְּסוּקִים א׳–ל״ד
4. בְּנֵי אַהֲרֹן וּמִשְׁפְּחוֹת הַלְוִיִּםפֶּרֶק ג׳, פְּסוּקִים א׳–ל״ט
5. הַלְוִיִּם תַּחַת הַבְּכוֹרוֹתפֶּרֶק ג׳, פְּסוּקִים מ׳–נ״א
6. מַשָּׂא הַמִּשְׁכָּן בַּמַּסָּעוֹתפֶּרֶק ד׳, פְּסוּקִים א׳–מ״ט
7. שִׁלּוּחַ הַטְּמֵאִים – ׳אֲשֶׁר אֲנִי שֹׁכֵן בְּתוֹכָם׳פֶּרֶק ה׳, פְּסוּקִים א׳–י׳
8. שְׁלוֹם בַּיִת – ׳וְיָשֵׂם לְךָ שָׁלוֹם׳פֶּרֶק ה׳, פְּסוּקִים י״א–ל״א; פֶּרֶק ו׳, פְּסוּקִים כ״ב–כ״ו
9. דִּינֵי הַנָּזִיר – ׳קָדֹשׁ יִהְיֶה׳פֶּרֶק ו׳, פְּסוּקִים א׳–כ״א
10. בִּרְכַּת כֹּהֲנִים – ׳יְבָרֶכְךָ ה׳ וְיִשְׁמְרֶךָ׳פֶּרֶק ו׳, פְּסוּקִים כ״ב–כ״ז
11. עֲגָלוֹת וּבָקָר לַמִּשְׁכָּן – מַתְּנַת הַנְּשִׂיאִיםפֶּרֶק ז׳, פְּסוּקִים א׳–י״א
12. קָרְבְּנוֹת הַנְּשִׂיאִים – שְׁנֵים עָשָׂר יָמִים שֶׁל חֲנֻכָּהפֶּרֶק ז׳, פְּסוּקִים י״ב–פ״ט
13. הַדְלָקַת הַמְּנוֹרָה וְהַקְדָּשַׁת הַלְוִיִּם – ׳אַשְׁרֵיכֶם׳פֶּרֶק ח׳, פְּסוּקִים א׳–כ״ו
14. פֶּסַח שֵׁנִי – ׳לָמָּה נִגָּרַע׳פֶּרֶק ט׳, פְּסוּקִים א׳–י״ד
15. עַמּוּד הֶעָנָן וְעַמּוּד הָאֵשׁ – ׳עַל פִּי ה׳ יַחֲנוּ׳פֶּרֶק ט׳, פְּסוּקִים ט״ו–כ״ג
16. חֲצוֹצְרוֹת הַכֶּסֶף – ׳וְנִזְכַּרְתֶּם לִפְנֵי ה׳ אֱלֹהֵיכֶם׳פֶּרֶק י׳, פְּסוּקִים א׳–י׳
17. יְצִיאָה לַמַּסָּע – ׳עַל פִּי ה׳ בְּיַד מֹשֶׁה׳פֶּרֶק י׳, פְּסוּקִים י״א–ל״ו
18. סִכּוּם: סִדּוּר הַמַּחֲנוֹת – ׳עַל פִּי ה׳ יַחֲנוּ וְעַל פִּי ה׳ יִסָּעוּ׳פֶּרֶק א׳–י׳ (חֲזָרָה)

🐦 הַמַּסָּעוֹת, הַתְּלוּנוֹת וּפָרָה אֲדֻמָּה

19. הַמִּתְאוֹנְנִים וְתַאֲוַת הַבָּשָׂר – ׳אֶל הַמָּן עֵינֵינוּ׳פֶּרֶק י״א, פְּסוּקִים א׳–ט״ו
20. שִׁבְעִים הַזְּקֵנִים וְהַשְּׂלָו – ׳וְלֹא תִשָּׂא אַתָּה לְבַדֶּךָ׳פֶּרֶק י״א, פְּסוּקִים ט״ז–ל״ה
21. מֹשֶׁה הֶעָנָו וּתְפִלָּתוֹ – ׳אֵל נָא רְפָא נָא לָהּ׳פֶּרֶק י״ב, פְּסוּקִים א׳–ט״ז
22. שִׁלּוּחַ הַמְרַגְּלִים – ׳שְׁלַח לְךָ אֲנָשִׁים׳פֶּרֶק י״ג, פְּסוּקִים א׳–כ׳
23. הַמְרַגְּלִים בָּאָרֶץ – ׳עָלֹה נַעֲלֶה׳פֶּרֶק י״ג, פְּסוּקִים כ״א–ל״ג
24. כָּלֵב וִיהוֹשֻׁעַ מוּל הָעָם – ׳טוֹבָה הָאָרֶץ מְאֹד מְאֹד׳פֶּרֶק י״ד, פְּסוּקִים א׳–כ״ה
25. גְּזֵרַת אַרְבָּעִים שָׁנָה וְהַמַּעְפִּילִיםפֶּרֶק י״ד, פְּסוּקִים כ״ו–מ״ה
26. ׳כִּי תָבֹאוּ׳ – נְסָכִים וְחַלָּהפֶּרֶק ט״ו, פְּסוּקִים א׳–כ״א
27. הַמְקֹשֵׁשׁ וְצִיצִית – ׳וּרְאִיתֶם אֹתוֹ וּזְכַרְתֶּם׳פֶּרֶק ט״ו, פְּסוּקִים ל״ב–מ״א
28. מַחֲלֹקֶת קֹרַח – ׳רַב לָכֶם׳פֶּרֶק ט״ז, פְּסוּקִים א׳–י״א
29. דָּתָן וַאֲבִירָם – ׳לֹא נַעֲלֶה׳פֶּרֶק ט״ז, פְּסוּקִים י״ב–כ״ב
30. הָאוֹת שֶׁל מֹשֶׁה – ׳סוּרוּ נָא׳פֶּרֶק ט״ז, פְּסוּקִים כ״ג–ל״ה
31. אַהֲרֹן עוֹצֵר אֶת הַמַּגֵּפָה – ׳וַתֵּעָצַר הַמַּגֵּפָה׳פֶּרֶק י״ז, פְּסוּקִים א׳–ט״ו
32. מַעֲשֵׂה הַמַּטּוֹת – ׳וַיָּצֵץ צִיץ וַיִּגְמֹל שְׁקֵדִים׳פֶּרֶק י״ז, פְּסוּקִים ט״ז–כ״ח
33. מַתְּנוֹת כְּהֻנָּה וּלְוִיָּה – ׳אֲנִי חֶלְקְךָ וְנַחֲלָתְךָ׳פֶּרֶק י״ח, פְּסוּקִים א׳–ל״ב
34. פָּרָה אֲדֻמָּה – ׳זֹאת חֻקַּת הַתּוֹרָה׳פֶּרֶק י״ט, פְּסוּקִים א׳–כ״ב
35. סִכּוּם: הַמַּסָּעוֹת וְהַתְּלוּנּוֹת – ׳סָלַחְתִּי כִּדְבָרֶךָ׳פְּרָקִים י״א–י״ט (חֲזָרָה)

🫏 הַמַּסָּע בְּעֵבֶר הַיַּרְדֵּן

36. מֵי מְרִיבָה – ׳וְדִבַּרְתֶּם אֶל הַסֶּלַע׳פֶּרֶק כ׳, פְּסוּקִים א׳–י״ג
37. מַעֲקַף אֶרֶץ אֱדוֹם – ׳דֶּרֶךְ הַמֶּלֶךְ נֵלֵךְ׳פֶּרֶק כ׳, פְּסוּקִים י״ד–כ״א
38. מוֹת אַהֲרֹן – ׳כֹּל בֵּית יִשְׂרָאֵל׳פֶּרֶק כ׳, פְּסוּקִים כ״ב–כ״ט
39. עֲרָד וְנָחָשׁ הַנְּחֹשֶׁת – ׳וְרָאָה אֹתוֹ וָחָי׳פֶּרֶק כ״א, פְּסוּקִים א׳–ט׳
40. סִיחוֹן וְעוֹג – מֵאַרְנוֹן עַד הַבָּשָׁןפֶּרֶק כ״א, פְּסוּקִים י׳–ל״ה
41. בָּלָק שׁוֹלֵחַ אֶל בִּלְעָם – ׳כִּי בָרוּךְ הוּא׳פֶּרֶק כ״ב, פְּסוּקִים א׳–כ״א
42. בִּלְעָם וְהָאָתוֹן – ׳וַתֵּרֶא הָאָתוֹן׳פֶּרֶק כ״ב, פְּסוּקִים כ״ב–ל״ה
43. בִּלְעָם וּבָלָק – שְׁנֵי הַנִּסְיוֹנוֹת הָרִאשׁוֹנִיםפֶּרֶק כ״ב, פָּסוּק ל״ו – פֶּרֶק כ״ג, פָּסוּק כ״ו
44. מַה טֹּבוּ אֹהָלֶיךָ – הַנִּסָּיוֹן הַשְּׁלִישִׁיפֶּרֶק כ״ג, פָּסוּק כ״ז – פֶּרֶק כ״ד, פָּסוּק כ״ה
45. בַּשִּׁטִּים – חֵטְא בַּעַל פְּעוֹר וְהַתִּקּוּןפֶּרֶק כ״ה, פְּסוּקִים א׳–ט׳
46. פִּינְחָס – ׳בְּרִיתִי שָׁלוֹם׳פֶּרֶק כ״ה, פְּסוּקִים י׳–י״ח
47. הַמַּסָּע לְעֵבֶר הַיַּרְדֵּן – מִקָּדֵשׁ עַד עַרְבוֹת מוֹאָבפְּרָקִים כ׳–כ״ב (סִכּוּם הַמַּסָּע)

🗺️ לִקְרַאת הַכִּבּוּשׁ וְהַנַּחֲלָה

48. מִפְקַד הַשָּׁנָה הָאַרְבָּעִים – ׳שְׂאוּ אֶת רֹאשׁ׳פֶּרֶק כ״ו, פְּסוּקִים א׳–נ״א
49. חֲלֻקַּת הָאָרֶץ בַּגּוֹרָל – ׳עַל פִּי הַגּוֹרָל׳פֶּרֶק כ״ו, פְּסוּקִים נ״ב–ס״ה
50. בְּנוֹת צְלָפְחָד – ׳תְּנָה לָּנוּ אֲחֻזָּה׳פֶּרֶק כ״ז, פְּסוּקִים א׳–י״א
51. מִנּוּי יְהוֹשֻׁעַ – ׳אִישׁ אֲשֶׁר רוּחַ בּוֹ׳פֶּרֶק כ״ז, פְּסוּקִים י״ב–כ״ג
52. קָרְבַּן הַתָּמִיד וְהַמּוּסָפִיםפֶּרֶק כ״ח, פְּסוּקִים א׳–ל״א
53. מוּסְפֵי הַמּוֹעֲדִיםפֶּרֶק כ״ט, פְּסוּקִים א׳–ל״ט
54. הֲפָרַת נְדָרִיםפֶּרֶק ל׳, פְּסוּקִים א׳–י״ז
55. נִקְמַת מִדְיָן – ׳נִקְמַת ה׳׳פֶּרֶק ל״א, פְּסוּקִים א׳–כ״ד
56. חֲלוּקַת הַשָּׁלָל – ׳חֶצְיוֹ לָאֵלּוּ וְחֶצְיוֹ לָאֵלּוּ׳פֶּרֶק ל״א, פְּסוּקִים כ״ה–נ״ד
57. בַּקָּשַׁת בְּנֵי גָד וּבְנֵי רְאוּבֵן – ׳חֲלוּצִים לִפְנֵי בְּנֵי יִשְׂרָאֵל׳פֶּרֶק ל״ב, פְּסוּקִים א׳–כ״ז
58. חֲלוּצִים לִפְנֵי ה׳ – הַתְּנָאִים וַחֲצִי שֵׁבֶט מְנַשֶּׁהפֶּרֶק ל״ב, פְּסוּקִים כ״ח–מ״ב
59. אֵלֶּה מַסְעֵי בְנֵי יִשְׂרָאֵל – מִמִּצְרַיִם עַד עַרְבוֹת מוֹאָבפֶּרֶק ל״ג, פְּסוּקִים א׳–מ״ט
60. גְּבוּלוֹת הָאָרֶץ – ׳זֹאת תִּהְיֶה לָכֶם הָאָרֶץ לִגְבֻלֹתֶיהָ׳פֶּרֶק ל״ג פְּסוּקִים נ׳–נ״ד, וּפֶּרֶק ל״ד, פְּסוּקִים א׳–כ״ט
61. עָרֵי הַלְוִיִּם וְעָרֵי מִקְלָט – ׳אֲנִי ה׳ שֹׁכֵן בְּתוֹךְ בְּנֵי יִשְׂרָאֵל׳פֶּרֶק ל״ה
62. ׳לֹא תִסֹּב נַחֲלָה׳ – בְּנוֹת צְלָפְחָד וְסוֹף הַסֵּפֶרפֶּרֶק ל״ו, פְּסוּקִים א׳–י״ג
63. סִכּוּם סֵפֶר בַּמִּדְבָּר – ׳עַל פִּי ה׳ יַחֲנוּ וְעַל פִּי ה׳ יִסָּעוּ׳חֻמַּשׁ בַּמִּדְבָּר – כָּל הַסֵּפֶר

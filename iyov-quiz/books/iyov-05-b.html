חַיִּים וָחֶסֶד עָשִׂיתָ עִמָּדִי

רָמָה ב׳ – עִיּוּן בַּיְצִירָה וּבַתִּקְוָה

אִיּוֹב מוֹדֶה לְה׳ עַל הַחַיִּים, צוֹפַר שׁוֹאֵל אִם אֶפְשָׁר לָדַעַת הַכֹּל, וְאִיּוֹב מוֹצֵא תִּקְוָה בָּעֵץ

📖 פְּרָקִים י׳–י״ד

🎯 לַחִידוֹן
🥣

📜 הַפָּסוּקהֲלֹא כֶחָלָב תַּתִּיכֵנִי וְכַגְּבִנָּה תַּקְפִּיאֵנִי עוֹר וּבָשָׂר תַּלְבִּישֵׁנִי

💛 אַהֲבַת הַשֵּׁם: ה׳ חָשַׁב עַל כָּל פְּרָט בַּגּוּף שֶׁלִּי 💛

📖 אִיּוֹב, פֶּרֶק י׳, פְּסוּקִים י׳–י״א (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּיְצִירָה וּבַתִּקְוָה

🥣 מְצוּדַת דָּוִד: ׳הֲלֹא כֶחָלָב׳

מְצוּדַת דָּוִד מְבָאֵר: ״בְּעֵת הַיְצִירָה הִתַּכְתָּ אוֹתִי כֶּחָלָב לִבְרֹר הַצָּלוּל, וְאַחַר כָּךְ הִקְפֵּאתָ אוֹתִי כַּגְּבִנָּה – הִיא הַתְחָלַת הַיְצִירָה״. אַחַר כָּךְ ״הִלְבַּשְׁתָּ אוֹתִי עוֹר וּבָשָׂר לְשָׁמְרָם מֵהַנֶּזֶק״, וְ״בְּעֲצָמוֹת וְגִידִים... הַמְקַשְּׁרִים יַחַד אֶת הָעֲצָמוֹת״. אִיּוֹב מְדַבֵּר עַל הַגּוּף כְּעַל מַעֲשֵׂה אָמָּן.

הַגּוּף שֶׁלִּי בָּנוּי בְּחָכְמָה, כְּמוֹ אָמָּן שֶׁבּוֹנֶה כְּלִי יָקָר.

1
💎

📜 הַפָּסוּקחַיִּים וָחֶסֶד עָשִׂיתָ עִמָּדִי וּפְקֻדָּתְךָ שָׁמְרָה רוּחִי

💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לִי חַיִּים בְּלִי שֶׁעָשִׂיתִי כְּלוּם 💛

📖 אִיּוֹב, פֶּרֶק י׳, פָּסוּק י״ב (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּיְצִירָה וּבַתִּקְוָה

💎 מְצוּדַת דָּוִד: ׳חַיִּים וָחֶסֶד׳

מְצוּדַת דָּוִד מְבָאֵר: ״עָשִׂיתָ עִמָּדִי הַחַיִּים לָתֵת בִּי רוּחַ הַחַיּוּת, וְלֹא בִּשְׁבִיל גְּמוּל מָה, כִּי עֲדַיִן לֹא עָשִׂיתִי מְאוּמָה, אֲבָל בְּחֶסֶד עָשִׂיתָ זֹאת״. וְרַשִׁ״י: ״פְּקֻדָּתְךָ – פְּקֻדַּת מִשְׁמַרְתְּךָ שֶׁמָּנִיתָ לִי שׁוֹמְרִים״. הַחַיִּים נִתְּנוּ לִי חִנָּם, וְה׳ גַּם שׁוֹמֵר עָלַי.

הַחַיִּים הֵם מַתָּנָה חִנָּם, וְהַשְּׁמִירָה גַּם כֵּן.

2
🧭

📜 הַפָּסוּקהַחֵקֶר אֱלוֹהַ תִּמְצָא אִם עַד תַּכְלִית שַׁדַּי תִּמְצָא

💛 אַהֲבַת הַשֵּׁם: ה׳ יוֹדֵעַ מָה טוֹב, וַאֲנִי סוֹמֵךְ עָלָיו 💛

📖 אִיּוֹב, פֶּרֶק י״א, פָּסוּק ז׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּיְצִירָה וּבַתִּקְוָה

🧭 רַשִׁ״י: ׳הַחֵקֶר אֱלוֹהַּ תִּמְצָא׳

רַשִׁ״י מְבָאֵר אֶת דִּבְרֵי צוֹפַר: ״שֶׁאַתָּה סָבוּר שֶׁקִּיַּמְתָּ הַכֹּל״. מְצוּדַת דָּוִד: ״מָה תִּתְרַעֵם עַל רָעוֹת הַצַּדִּיק וְטוֹבוֹת הָרָשָׁע? וְכִי אַתָּה תִּמְצָא חֵקֶר מִשְׁפַּט ה׳?״. צוֹפַר צוֹדֵק שֶׁאָדָם לֹא מֵבִין אֶת כָּל הַנְהָגַת ה׳, אֲבָל מִכָּךְ הוּא מַסִּיק שֶׁאִיּוֹב צָרִיךְ לָשׁוּב – וְה׳ יֹאמַר בַּסּוֹף שֶׁהַחֲבֵרִים לֹא דִּבְּרוּ נְכוֹנָה.

אֲנִי לֹא יוֹדֵעַ אֶת כָּל הַחֶשְׁבּוֹן שֶׁל ה׳, וְזֶה בְּסֵדֶר.

3
🏔️

📜 הַפָּסוּקגָּבְהֵי שָׁמַיִם מַה תִּפְעָל עֲמֻקָּה מִשְּׁאוֹל מַה תֵּדָע

💛 אַהֲבַת הַשֵּׁם: ה׳ גָּדוֹל מִמֶּנִּי, וַאֲנִי שָׂמֵחַ בְּכָךְ 💛

📖 אִיּוֹב, פֶּרֶק י״א, פְּסוּקִים ח׳–ט׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּיְצִירָה וּבַתִּקְוָה

🏔️ רַשִׁ״י: ׳גָּבְהֵי שָׁמַיִם׳

רַשִׁ״י מְבָאֵר: ״דָּבָר שֶׁהוּא גָּבוֹהַּ כְּגָבְהֵי שָׁמַיִם, מָה תִּפְעַל עַתָּה? וַעֲמֻקָּה הִיא הַחָכְמָה מִשְּׁאוֹל״. מְצוּדַת דָּוִד: סֵדֶר הַגְּמוּל הוּא ״אֲרֻכָּה מֵאֶרֶץ מִדָּהּ וּרְחָבָה מִנִּי יָם״ – אִי אֶפְשָׁר לְהַשִּׂיג אוֹתוֹ בְּחָכְמַת הַלֵּב. הַתְּמוּנָה: הַשָּׁמַיִם גְּבוֹהִים וְהַשְּׁאוֹל עָמֹק – וְאָדָם קָטָן.

מִי שֶׁמֵּבִין כַּמָּה ה׳ גָּדוֹל, מִתְמַלֵּא עֲנָוָה.

4
🦅

📜 הַפָּסוּקמִי לֹא יָדַע בְּכָל אֵלֶּה כִּי יַד ה׳ עָשְׂתָה זֹּאת אֲשֶׁר בְּיָדוֹ נֶפֶשׁ כָּל חָי וְרוּחַ כָּל בְּשַׂר אִישׁ

💛 אַהֲבַת הַשֵּׁם: ה׳ כָּתַב בַּטֶּבַע אֶת הָאַהֲבָה שֶׁלּוֹ אֵלֵינוּ 💛

📖 אִיּוֹב, פֶּרֶק י״ב, פְּסוּקִים ז׳, ט׳ (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּיְצִירָה וּבַתִּקְוָה

🦅 מְצוּדַת דָּוִד: ׳שְׁאַל נָא בְהֵמוֹת׳

מְצוּדַת דָּוִד מְבָאֵר: ״שְׁאַל אֶל הַבְּהֵמוֹת, וְכָל אַחַת תְּלַמֵּד אוֹתְךָ... מִשְּׁלֵמוּת יְצִירַת הַיְּצוּרִים וּמֵהַחָכְמָה אֲשֶׁר בָּהֶם״. וְעַל ׳כִּי יַד ה׳ עָשְׂתָה זֹאת׳: ״הֲלֹא יַד ה׳ עָשְׂתָה כָּל זֹאת, וּמֵהֶם הֲלֹא יֻקַּח רְאָיָה עַל עֹמֶק חָכְמָתוֹ שֶׁאֵין חֵקֶר לָהּ״. הַחַיּוֹת הֵן סֵפֶר שֶׁל חָכְמַת ה׳.

הַטֶּבַע הוּא סֵפֶר שֶׁמְּלַמֵּד עַל ה׳.

5
🙌

📜 הַפָּסוּקהֵן יִקְטְלֵנִי לוֹ אֲיַחֵל אַךְ דְּרָכַי אֶל פָּנָיו אוֹכִיחַ גַּם הוּא לִי לִישׁוּעָה

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אֶת מִי שֶׁמְּדַבֵּר אֵלָיו בְּאֱמֶת 💛

📖 אִיּוֹב, פֶּרֶק י״ג, פְּסוּקִים ט״ו–ט״ז (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּיְצִירָה וּבַתִּקְוָה

🙌 רַשִׁ״י: ׳לוֹ אֲיַחֵל׳

בַּכְּתִיב כָּתוּב ׳לֹא אֲיַחֵל׳, וּבַקְּרִי ׳לוֹ אֲיַחֵל׳. רַשִׁ״י מְבָאֵר כַּקְּרִי: ״הֲלֹא אִם יִקְטְלֵנִי אֵינִי נִפְרָד מִמֶּנּוּ, וְתָמִיד לוֹ אֲיַחֵל, עַל כֵּן אֵין מֶרֶד וָפֶשַׁע בְּדִבְרַי״. וְעַל ׳כִּי לֹא לְפָנָיו חָנֵף יָבוֹא׳ הוּא כּוֹתֵב: ״וְזֶה שֶׁנֶּאֱמַר לָהֶם לְקַמָּן (מ״ב) כִּי לֹא דִבַּרְתֶּם אֵלַי נְכוֹנָה״. אִיּוֹב לֹא מַחֲנִיף, הוּא דּוֹרֵשׁ אֶמֶת.

הַמְּאַמֵּן אֶת ה׳ שׁוֹאֵל שְׁאֵלוֹת בְּאֱמֶת, וְלֹא מַחֲנִיף.

6
🌿

📜 הַפָּסוּקכִּי יֵשׁ לָעֵץ תִּקְוָה אִם יִכָּרֵת וְעוֹד יַחֲלִיף וְיֹנַקְתּוֹ לֹא תֶחְדָּל מֵרֵיחַ מַיִם יַפְרִחַ

💛 אַהֲבַת הַשֵּׁם: ה׳ מְחַדֵּשׁ אוֹתִי כְּמוֹ עֵץ 💛

📖 אִיּוֹב, פֶּרֶק י״ד, פְּסוּקִים ז׳, ט׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּיְצִירָה וּבַתִּקְוָה

🌿 מְצוּדַת דָּוִד: ׳מֵרֵיחַ מַיִם׳

מְצוּדַת דָּוִד מְבָאֵר: הָעֵץ ״אִם יִזְקִין בָּאָרֶץ שָׁרְשׁוֹ... כְּשֶׁיָּבוֹא עָלָיו מְעַט מַיִם וּמִיָּד כְּשֶׁיָּרִיחַ אֶת הַמַּיִם יַחֲזֹר וְיִפְרַח וְיַגְדִּיל עָנָף כְּאִלּוּ נְטָעוֹ זֶה מִקָּרוֹב״. רַשִׁ״י: ׳כִּי יֵשׁ לָעֵץ תִּקְוָה׳ – ״שֶׁיַּחֲלֹף״. אִיּוֹב לוֹמֵד מֵהָעֵץ: יֵשׁ תִּקְוָה לְהִתְחַדֵּשׁ.

גַּם מֵעֵץ יָשָׁן אֶפְשָׁר לְהַצְמִיחַ עָנָף חָדָשׁ.

💡

רָמָה ב׳ – עִיּוּן בַּיְצִירָה וּבַתִּקְוָה

מָה לוֹקְחִים מִרָמָה ב׳?

רַשִׁ״י וּמְצוּדַת דָּוִד מְפָרְשִׁים אֶת הַיְצִירָה, אֶת הַחַיִּים כְּחֶסֶד חִנָּם, וְאֶת הַתִּקְוָה שֶׁל הָעֵץ.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • אִיּוֹב אוֹמֵר לְה׳ ׳חַיִּים וָחֶסֶד עָשִׂיתָ עִמָּדִי׳. עַל אֵלּוּ חַיִּים וַחֲסָדִים אֲנִי רוֹצֶה לְהוֹדוֹת הַיּוֹם?
  • אִיּוֹב מְקַוֶּה לְה׳ גַּם כְּשֶׁקָּשֶׁה. מָתַי אֲנִי הִמְשַׁכְתִּי לְקַוּוֹת אַף עַל פִּי שֶׁהָיָה קָשֶׁה?
  • ׳שְׁאַל נָא בְהֵמוֹת וְתֹרֶךָּ׳ – מָה אֲנִי יָכוֹל לִלְמֹד מִחַיָּה אוֹ מֵעוֹף שֶׁרָאִיתִי?

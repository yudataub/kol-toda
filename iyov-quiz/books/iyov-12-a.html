וַה׳ שָׁב אֶת שְׁבוּת אִיּוֹב

רָמָה א׳ – הַסּוֹף הַטּוֹב

הַחֲבֵרִים מִתְכַּפְּרִים, אִיּוֹב מִתְפַּלֵּל בַּעֲבוּרָם, וְה׳ מַחֲזִיר לוֹ הַכֹּל וּמוֹסִיף – הַסּוֹף טוֹב

📖 אִיּוֹב, פֶּרֶק מ״ב, פָּסוּק ז׳

רָמָה א׳ – הַסּוֹף הַטּוֹב

🌅 ׳לֹא דִבַּרְתֶּם אֵלַי נְכוֹנָה׳

אַחֲרֵי שֶׁה׳ דִּבֵּר אֶל אִיּוֹב, הוּא פּוֹנֶה אֶל אֱלִיפַז: ׳חָרָה אַפִּי בְךָ וּבִשְׁנֵי רֵעֶיךָ, כִּי לֹא דִבַּרְתֶּם אֵלַי נְכוֹנָה כְּעַבְדִּי אִיּוֹב׳. ה׳ קוֹרֵא לְאִיּוֹב ׳עַבְדִּי׳ – וּמַעֲרִיךְ אוֹתוֹ.

ה׳ קָרָא לְאִיּוֹב ׳עַבְדִּי׳ – וְזֹאת הַמַּעֲלָה הַגְּדוֹלָה.

🐏

💛 אַהֲבַת הַשֵּׁם: ה׳ מְלַמֵּד אוֹתִי לָסְלֹחַ וּלְהִתְפַּלֵּל עַל אֲחֵרִים 💛

📖 אִיּוֹב, פֶּרֶק מ״ב, פָּסוּק ח׳

רָמָה א׳ – הַסּוֹף הַטּוֹב

🐏 ׳וְאִיּוֹב עַבְדִּי יִתְפַּלֵּל עֲלֵיכֶם׳

ה׳ אוֹמֵר לַחֲבֵרִים: ׳קְחוּ לָכֶם שִׁבְעָה פָרִים וְשִׁבְעָה אֵילִים וּלְכוּ אֶל עַבְדִּי אִיּוֹב וְהַעֲלִיתֶם עוֹלָה בַּעַדְכֶם, וְאִיּוֹב עַבְדִּי יִתְפַּלֵּל עֲלֵיכֶם׳. הֵם צְרִיכִים לְהַקְרִיב קָרְבָּן, וְאִיּוֹב – הַנִּפְגָּע – הוּא זֶה שֶׁיִּתְפַּלֵּל עֲלֵיהֶם.

מִי שֶׁנִּפְגַּע מִתְפַּלֵּל עַל מִי שֶׁפָּגַע בּוֹ.

🚶

💛 אַהֲבַת הַשֵּׁם: ה׳ מְקַבֵּל אֶת מִי שֶׁשָּׁב מִטָּעוּתוֹ 💛

📖 אִיּוֹב, פֶּרֶק מ״ב, פָּסוּק ט׳

רָמָה א׳ – הַסּוֹף הַטּוֹב

🚶 הַחֲבֵרִים הוֹלְכִים

שְׁלוֹשֶׁת הָרֵעִים לֹא מִתְוַכְּחִים. ׳וַיֵּלְכוּ אֱלִיפַז הַתֵּימָנִי וּבִלְדַּד הַשּׁוּחִי צֹפַר הַנַּעֲמָתִי, וַיַּעֲשׂוּ כַּאֲשֶׁר דִּבֶּר אֲלֵיהֶם ה׳׳. אַחֲרֵי כָּל הַוִּכּוּחִים הֵם מַקְשִׁיבִים לְה׳. ׳וַיִּשָּׂא ה׳ אֶת פְּנֵי אִיּוֹב׳ – ה׳ קִבֵּל אֶת הַתְּפִלָּה שֶׁלּוֹ.

הַחֲבֵרִים יָדְעוּ לְהוֹדוֹת וְלַעֲשׂוֹת כְּפִי שֶׁה׳ אָמַר.

🙏

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אוֹתִי כְּשֶׁאֲנִי מִתְפַּלֵּל בִּשְׁבִיל אֲחֵרִים 💛

📖 אִיּוֹב, פֶּרֶק מ״ב, פָּסוּק י׳

רָמָה א׳ – הַסּוֹף הַטּוֹב

🙏 ׳בְּהִתְפַּלְלוֹ בְּעַד רֵעֵהוּ׳

וְעַכְשָׁיו הַפָּסוּק הַמֶּרְכָּזִי: ׳וַה׳ שָׁב אֶת שְׁבוּת אִיּוֹב בְּהִתְפַּלְלוֹ בְּעַד רֵעֵהוּ, וַיֹּסֶף ה׳ אֶת כָּל אֲשֶׁר לְאִיּוֹב לְמִשְׁנֶה׳. כְּשֶׁאִיּוֹב הִתְפַּלֵּל עַל הַחֲבֵרִים, ה׳ הֶחֱזִיר לוֹ אֶת הַכֹּל – וְהוֹסִיף כְּפוּל.

כְּשֶׁאֲנִי מִתְפַּלֵּל בִּשְׁבִיל חָבֵר, גַּם הַמַּצָּב שֶׁלִּי מִשְׁתַּפֵּר.

🎁

💛 אַהֲבַת הַשֵּׁם: ה׳ מַחֲזִיר גַּם אֶת הַחֲבֵרִים 💛

📖 אִיּוֹב, פֶּרֶק מ״ב, פָּסוּק י״א

רָמָה א׳ – הַסּוֹף הַטּוֹב

🎁 הַמִּשְׁפָּחָה וְהַחֲבֵרִים חוֹזְרִים

׳וַיָּבֹאוּ אֵלָיו כָּל אֶחָיו וְכָל אַחְיוֹתָיו וְכָל יֹדְעָיו לְפָנִים, וַיֹּאכְלוּ עִמּוֹ לֶחֶם בְּבֵיתוֹ, וַיָּנֻדוּ לוֹ וַיְנַחֲמוּ אֹתוֹ׳. כָּל אֶחָד נָתַן לוֹ מַתָּנָה: ׳קְשִׂיטָה אַחַת וְנֶזֶם זָהָב אֶחָד׳. הָאוֹהֲבִים חָזְרוּ, וְהֵם סוֹעֲדִים וּמְנַחֲמִים.

כְּשֶׁחוֹזֵר הַטּוֹב – חֲבֵרִים חוֹזְרִים, וְאִיּוֹב מְקַבֵּל אוֹתָם.

5
🐑

💛 אַהֲבַת הַשֵּׁם: תּוֹדָה לְךָ ה׳ עַל הַבְּרָכָה שֶׁאַתָּה מַכִּין לִי 💛

📖 אִיּוֹב, פֶּרֶק מ״ב, פָּסוּק י״ב

רָמָה א׳ – הַסּוֹף הַטּוֹב

🐑 ׳וַה׳ בֵּרַךְ אֶת אַחֲרִית אִיּוֹב׳

׳וַה׳ בֵּרַךְ אֶת אַחֲרִית אִיּוֹב מֵרֵאשִׁתוֹ, וַיְהִי לוֹ אַרְבָּעָה עָשָׂר אֶלֶף צֹאן וְשֵׁשֶׁת אֲלָפִים גְּמַלִּים וְאֶלֶף צֶמֶד בָּקָר וְאֶלֶף אֲתוֹנוֹת׳. פִּי שְׁנַיִם מִמָּה שֶׁהָיָה לוֹ בַּהַתְחָלָה (פֶּרֶק א׳).

הַסּוֹף שֶׁל אִיּוֹב טוֹב מִן הַהַתְחָלָה.

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי מִשְׁפָּחָה וְשִׂמְחָה 💛

📖 אִיּוֹב, פֶּרֶק מ״ב, פָּסוּק י״ג

רָמָה א׳ – הַסּוֹף הַטּוֹב

👨‍👦‍👦 הַמִּשְׁפָּחָה חוֹזֶרֶת

׳וַיְהִי לוֹ שִׁבְעָנָה בָנִים וְשָׁלוֹשׁ בָּנוֹת׳ – שׁוּב שִׁבְעָה בָּנִים וּשְׁלוֹשׁ בָּנוֹת. הַמִּשְׁפָּחָה שֶׁאָבְדָה חוֹזֶרֶת, וְהַבַּיִת שֶׁל אִיּוֹב מָלֵא בְּשִׂמְחָה.

ה׳ יָכוֹל לְהָשִׁיב שִׂמְחָה אֲפִלּוּ אַחֲרֵי עֶצֶב גָּדוֹל.

🌳

💛 אַהֲבַת הַשֵּׁם: תּוֹדָה לְךָ ה׳ עַל חַיִּים אֲרוּכִּים וּטוֹבִים 💛

📖 אִיּוֹב, פֶּרֶק מ״ב, פְּסוּקִים ט״ז–י״ז

רָמָה א׳ – הַסּוֹף הַטּוֹב

🌳 ׳זָקֵן וּשְׂבַע יָמִים׳

׳וַיְחִי אִיּוֹב אַחֲרֵי זֹאת מֵאָה וְאַרְבָּעִים שָׁנָה, וַיִּרְאֶה אֶת בָּנָיו וְאֶת בְּנֵי בָנָיו אַרְבָּעָה דֹּרוֹת. וַיָּמָת אִיּוֹב זָקֵן וּשְׂבַע יָמִים׳. הוּא חַי חַיִּים אֲרוּכִּים, רָאָה אַרְבָּעָה דּוֹרוֹת, וְהָיָה שָׂבֵעַ מֵחַיָּיו.

הַסּוֹף הַטּוֹב: זָקֵן וְשָׂבֵעַ יָמִים.

💡

רָמָה א׳ – הַסּוֹף הַטּוֹב

מָה לוֹקְחִים מִרָמָה א׳?

ה׳ מַעֲרִיךְ אֶת אִיּוֹב, הַחֲבֵרִים מִתְפַּיְּסִים, וְאִיּוֹב מִתְפַּלֵּל עֲלֵיהֶם. ה׳ מֵשִׁיב לוֹ אֶת הַכֹּל פִּי שְׁנַיִם, וְהוּא חַי זָקֵן וְשָׂבֵעַ. ה׳ אוֹהֵב אוֹתוֹ.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • אִיּוֹב הִתְפַּלֵּל עַל הַחֲבֵרִים שֶׁפָּגְעוּ בּוֹ. עַל מִי אֲנִי יָכוֹל לְהִתְפַּלֵּל, אֲפִלּוּ אִם הוּא פָּגַע בִּי?
  • ׳הַמִּתְפַּלֵּל עַל חֲבֵרוֹ נַעֲנֶה תְּחִלָּה׳ – אֵיזוֹ בַּקָּשָׁה אֲנִי רוֹצֶה לְבַקֵּשׁ בִּשְׁבִיל חָבֵר הַיּוֹם?
  • הַסּוֹף שֶׁל אִיּוֹב טוֹב מֵהַהַתְחָלָה. מָתַי רָאִיתִי שֶׁסּוֹף קָשֶׁה נֶהְפַּךְ לְטוֹב?

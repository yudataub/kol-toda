🕯️ חִידוֹנֵי חֻמַּשׁ וַיִּקְרָא

44 חִידוֹנִים לְפִי תָּכְנִית הַלִּמּוּדִים שֶׁל מִשְׂרַד הַחִנּוּךְ (תַּנַ״ךְ חֶמֶ״ד), לְכִתָּה ד׳. בְּכָל חִידוֹן כַּרְטִיסֵי לִמּוּד עִם אַהֲבַת ה׳ 💛 וְשָׁלוֹשׁ רָמוֹת: תְּמוּנוֹת, שְׁאֵלוֹת לַחֲכָמִים, וּבָרָמָה הַשְּׁלִישִׁית מִשְׂחָקִים – בְּכָל פַּעַם מִשְׂחָק אַחֵר

🔥 הַקָּרְבָּנוֹת

1. וַיִּקְרָא אֶל מֹשֶׁה – קָרְבַּן עוֹלָהפֶּרֶק א׳, פְּסוּקִים א׳–י״ז
2. נֶפֶשׁ כִּי תַקְרִיב – קָרְבַּן מִנְחָהפֶּרֶק ב׳, פְּסוּקִים א׳–ט״ז
3. זֶבַח שְׁלָמִיםפֶּרֶק ג׳, פְּסוּקִים א׳–י״ז
4. קָרְבַּן חַטָּאת – אַרְבָּעָה חוֹטְאִיםפֶּרֶק ד׳, פְּסוּקִים א׳–ל״ה
5. אָשָׁם וְעוֹלֶה וְיוֹרֵדפֶּרֶק ה׳, פְּסוּקִים א׳–כ״ו
6. תּוֹרַת הָעוֹלָה, הַמִּנְחָה וְהַחַטָּאתפֶּרֶק ו׳, פְּסוּקִים א׳–כ״ג

👑 הַמִּלּוּאִים וְהַיּוֹם הַשְּׁמִינִי

7. תּוֹרַת הָאָשָׁם וְזֶבַח הַשְּׁלָמִים – תּוֹדָה לַה׳פֶּרֶק ז׳, פְּסוּקִים א׳–ל״ח
8. שִׁבְעַת יְמֵי הַמִּלּוּאִים – חֲנִיכַת אַהֲרֹן וּבָנָיופֶּרֶק ח׳, פְּסוּקִים א׳–ל״ו
9. הַיּוֹם הַשְּׁמִינִי – חֲנֻכַּת הַמִּשְׁכָּןפֶּרֶק ט׳, פְּסוּקִים א׳–כ״ד
10. ׳בִּקְרֹבַי אֶקָּדֵשׁ׳ – ׳וַיִּדֹּם אַהֲרֹן׳פֶּרֶק י׳, פְּסוּקִים א׳–כ׳

💧 טָהֳרָה וְכַפָּרָה

11. בְּהֵמוֹת וְדָגִים – סִימָנֵי טָהֳרָהפֶּרֶק י״א, פְּסוּקִים א׳–י״ב
12. עוֹפוֹת וּשְׁרָצִים – ׳וִהְיִיתֶם קְדֹשִׁים׳פֶּרֶק י״א, פְּסוּקִים י״ג–מ״ז
13. הַמְּצֹרָע – ׳בָּדָד יֵשֵׁב׳פֶּרֶק י״ב, פְּסוּקִים א׳–ח׳; פֶּרֶק י״ג, פְּסוּקִים א׳–ג׳, מ״ה–מ״ו
14. טָהֳרַת הַמְּצֹרָע – ׳שְׁתֵּי צִפֳּרִים׳פֶּרֶק י״ד, פְּסוּקִים א׳–י״א, ל״ג–ל״ו, נ״ד–נ״ז
15. כֹּהֵן גָּדוֹל בְּקֹדֶשׁ הַקֳּדָשִׁיםפֶּרֶק ט״ז, פְּסוּקִים א׳–י״ט
16. יוֹם הַכִּפּוּרִים – שָׂעִיר לַעֲזָאזֵלפֶּרֶק ט״ז, פְּסוּקִים ז׳–י׳, כ׳–ל״ד
17. אִסּוּר אֲכִילַת דָּם וְכִסּוּי הַדָּםפֶּרֶק י״ז, פְּסוּקִים א׳–ט״ז
18. לֹא תַעֲשׂוּ כְּמַעֲשֵׂה אֶרֶץ מִצְרַיִםפֶּרֶק י״ח, פְּסוּקִים א׳–ה׳, כ״ד–ל׳; פֶּרֶק כ׳, פְּסוּקִים ו׳–ח׳, כ״ב–כ״ו

📜 לוֹמְדִים רַשִׁ״י

19. רַשִׁ״י הַפַּרְשָׁן – מֵאִיר אֶת הַתּוֹרָהפֶּרֶק א׳, פְּסוּקִים א׳–ב׳, ט׳; פֶּרֶק ט׳, פָּסוּק כ״ב (רַשִׁ״י)
20. כְּתַב רַשִׁ״י וְדִבּוּר הַמַּתְחִיל – אֵיזוֹ מִלָּה מְבָאֵר רַשִׁ״י?פֶּרֶק א׳, פָּסוּק ג׳; פֶּרֶק ב׳, פָּסוּק א׳; פֶּרֶק ג׳, פָּסוּק א׳; פֶּרֶק ט׳, פְּסוּקִים ד׳, כ״ב–כ״ג (רַשִׁ״י)
21. שְׁאֵלָה וּתְשׁוּבָה בְּרַשִׁ״י – ׳לָמָּה נֶאֱמַר?׳פֶּרֶק א׳, פְּסוּקִים א׳–ב׳; פֶּרֶק ב׳, פָּסוּק א׳; פֶּרֶק ו׳, פְּסוּקִים ג׳–ד׳; פֶּרֶק ט׳, פָּסוּק כ״ג (רַשִׁ״י)

✨ קְדֹשִׁים תִּהְיוּ

22. קְדֹשִׁים תִּהְיוּ – ׳כִּי קָדוֹשׁ אֲנִי׳פֶּרֶק י״ט, פְּסוּקִים א׳–ג׳
23. מַתְּנוֹת עֲנִיִּים – ׳לֶעָנִי וְלַגֵּר תַּעֲזֹב אֹתָם׳פֶּרֶק י״ט, פְּסוּקִים ד׳–י׳
24. לֹא תִּגְנֹבוּ – צֶדֶק וְיֹשֶׁרפֶּרֶק י״ט, פְּסוּקִים י״א–י״ג
25. לִפְנֵי עִוֵּר לֹא תִתֵּן מִכְשׁוֹלפֶּרֶק י״ט, פְּסוּקִים י״ד–ט״ו
26. לֹא תֵלֵךְ רָכִיל – וְלֹא תַעֲמֹד עַל דַּם רֵעֶךָפֶּרֶק י״ט, פְּסוּקִים ט״ז–י״ז
27. וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָפֶּרֶק י״ט, פָּסוּק י״ח
28. כִּלְאַיִם, עָרְלָה וְחֻקּוֹת הַגּוֹיִם – ׳אֶת חֻקֹּתַי תִּשְׁמֹרוּ׳פֶּרֶק י״ט, פְּסוּקִים י״ט–ל״א
29. מִפְּנֵי שֵׂיבָה תָּקוּם – זָקֵן, גֵּר וּמֹאזְנֵי צֶדֶקפֶּרֶק י״ט, פְּסוּקִים ל״ב–ל״ז

🕯️ הַכֹּהֲנִים וּמוֹעֲדֵי ה׳

30. קְדֻשַּׁת הַכֹּהֲנִים – ׳קְדֹשִׁים יִהְיוּ לֵאלֹהֵיהֶם׳פֶּרֶק כ״א, פְּסוּקִים א׳–ט״ו
31. קָדְשֵׁי הַמִּקְדָּשׁ – ׳וְנִקְדַּשְׁתִּי בְּתוֹךְ בְּנֵי יִשְׂרָאֵל׳פֶּרֶק כ״א, פְּסוּקִים ט״ז–כ״ד; פֶּרֶק כ״ב, פְּסוּקִים א׳–ל״ג
32. מוֹעֲדֵי ה׳ – שַׁבָּת וּפֶסַחפֶּרֶק כ״ג, פְּסוּקִים א׳–ח׳
33. סְפִירַת הָעֹמֶר וְחַג הַשָּׁבוּעוֹתפֶּרֶק כ״ג, פְּסוּקִים ט׳–כ״ב
34. הַחֹדֶשׁ הַשְּׁבִיעִי – רֹאשׁ הַשָּׁנָה, יוֹם הַכִּפּוּרִים וְסֻכּוֹתפֶּרֶק כ״ג, פְּסוּקִים כ״ג–מ״ד
35. מוֹעֲדֵי יִשְׂרָאֵל – תְּעוּדַת זֶהוּתפֶּרֶק כ״ג
36. שֶׁמֶן הַמָּאוֹר וְלֶחֶם הַפָּנִיםפֶּרֶק כ״ד, פְּסוּקִים א׳–ט׳
37. הַמְקַלֵּל וְדִין הַמַּזִּיק – ׳מִשְׁפָּט אֶחָד יִהְיֶה לָכֶם׳פֶּרֶק כ״ד, פְּסוּקִים י׳–כ״ג

🌾 בְּהַר וּבְחֻקֹּתַי

38. שַׁבַּת הָאָרֶץ – ׳שַׁבָּת לַה׳׳פֶּרֶק כ״ה, פְּסוּקִים א׳–ז׳, י״ח–כ״ב
39. שְׁנַת הַיּוֹבֵל – ׳וּקְרָאתֶם דְּרוֹר בָּאָרֶץ׳פֶּרֶק כ״ה, פְּסוּקִים ח׳–י״ז, י״ד–כ״ד
40. וְכִי יָמוּךְ אָחִיךָ – עֲרֵבוּת וְחֶסֶדפֶּרֶק כ״ה, פְּסוּקִים כ״ה–נ״ה
41. אִם בְּחֻקֹּתַי תֵּלֵכוּ – בִּרְכַּת הַשָּׁלוֹםפֶּרֶק כ״ו, פְּסוּקִים ג׳–י״ג
42. הַתּוֹכֵחָה וְהַנֶּחָמָה – ׳וְזָכַרְתִּי אֶת בְּרִיתִי׳פֶּרֶק כ״ו, פְּסוּקִים י״ד–מ״ו

🏆 עֲרָכִין וְסִכּוּם הַחֻמָּשׁ

43. עֲרָכִין וְהֶקְדֵּשׁוֹת – ׳אֵלֶּה הַמִּצְוֺת׳פֶּרֶק כ״ז, פְּסוּקִים א׳–ל״ד
44. סִכּוּם חֻמַּשׁ וַיִּקְרָא – ׳קְדֹשִׁים תִּהְיוּ׳פְּרָקִים א׳–כ״ז

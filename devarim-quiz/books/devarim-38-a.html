לָעַמִּים יֵשׁ קוֹסְמִים וּמְעוֹנְנִים – ׳וְאַתָּה לֹא כֵן נָתַן לְךָ ה׳ אֱלֹהֶיךָ׳. ׳נָבִיא מִקִּרְבְּךָ מֵאַחֶיךָ כָּמֹנִי יָקִים לְךָ ה׳ אֱלֹהֶיךָ, אֵלָיו תִּשְׁמָעוּן׳. וּמָה יֹאמַר הַנָּבִיא? לֹא ׳מָחָר יִהְיֶה כָּךְ וְכָךְ׳, אֶלָּא: שׁוּבוּ בִּתְשׁוּבָה, שִׁמְרוּ אֶת הַתּוֹרָה וְהַמִּצְווֹת – כְּדֵי שֶׁתַּאֲרִיכוּ יָמִים בָּאָרֶץ. אֶת דִּבְרֵי הַנְּבִיאִים אֲנַחְנוּ קוֹרְאִים בַּהַפְטָרוֹת עַד הַיּוֹם.

הַנָּבִיא מְכַוֵּן אוֹתָנוּ בַּדֶּרֶךְ הַיְּשָׁרָה – לְתוֹרָה וְלִתְשׁוּבָה.

מָה לוֹקְחִים מִרָמָה א׳?

הַגּוֹיִם בְּאֶרֶץ כְּנַעַן פָּחֲדוּ מִן הֶעָתִיד, וְחִפְּשׂוּ סִימָנִים בְּכָל מָקוֹם – בְּמַקֵּל, בְּעוֹנָה וּבְחָתוּל שֶׁעָבַר. לָנוּ ה׳ נָתַן דֶּרֶךְ אַחֶרֶת: כֹּהֵן שֶׁמְּשָׁרֵת אוֹתוֹ, נָבִיא שֶׁמְּדַבֵּר אֶת דְּבָרָיו, וְלֵב תָּמִים שֶׁסּוֹמֵךְ עָלָיו. מִי שֶׁהוֹלֵךְ עִם ה׳ בִּתְמִימוּת לֹא צָרִיךְ לְפַחֵד מִמָּחָר – כִּי ה׳ כְּבָר נִמְצָא שָׁם.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי דָּאַגְתִּי מְאוֹד מִמָּה שֶׁיִּהְיֶה מָחָר – וְאֵיךְ אֲנִי יָכוֹל לִסְמֹךְ עַל ה׳ בַּפַּעַם הַבָּאָה?
  • מָה פֵּרוּשׁ בִּשְׁבִילִי ׳לִהְיוֹת תָּמִים עִם ה׳׳ – בְּמִבְחָן, בְּמִשְׂחָק, בְּיוֹם רָגִיל?
  • אֵיךְ אֲנִי יָכוֹל לְהַכִּיר טוֹבָה לְמִי שֶׁמְּלַמֵּד אוֹתִי תּוֹרָה, כְּמוֹ שֶׁעַם יִשְׂרָאֵל נוֹתֵן לַכֹּהֲנִים?

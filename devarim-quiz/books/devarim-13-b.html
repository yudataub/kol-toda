ה׳ קִבֵּל אֶת הַבַּקָּשָׁה וְשִׁבַּח אוֹתָהּ: ׳הֵיטִיבוּ כָּל אֲשֶׁר דִּבֵּרוּ. מִי יִתֵּן וְהָיָה לְבָבָם זֶה לָהֶם לְיִרְאָה אֹתִי וְלִשְׁמֹר אֶת כָּל מִצְוֹתַי כָּל הַיָּמִים׳.

מִכְתָּב רִשְׁמִי שֶׁמַּגִּיעַ בַּדֹּאַר מְחַיֵּב בְּדִיּוּק כְּמוֹ מִכְתָּב שֶׁמּוֹסְרִים בַּיָּד – וְכָךְ גַּם הַתּוֹרָה: אַף שֶׁקִּבַּלְנוּ אוֹתָהּ דֶּרֶךְ מֹשֶׁה, אֲנַחְנוּ מְחֻיָּבִים בָּהּ בְּדִיּוּק כְּאִלּוּ שָׁמַעְנוּ מִפִּי ה׳.

ה׳ שִׁבַּח אֶת בַּקָּשַׁת הָעָם – וְהַתּוֹרָה מְחַיֶּבֶת גַּם דֶּרֶךְ שָׁלִיחַ.

8
💡

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

מָה לוֹקְחִים מִרָמָה ב׳?

שְׁתֵּי הַדִּבְּרוֹת הָאֵלֶּה, שַׁבָּת וְכִבּוּד הוֹרִים, נִתְּנוּ כְּבָר בְּמָרָה וְנִכְתְּבוּ בַּלּוּחַ שֶׁל ה׳ – כִּי הֵן בּוֹנוֹת אֶת הַבַּיִת הַיְּהוּדִי. בְּשַׁבָּת אֲנַחְנוּ זוֹכְרִים שֶׁה׳ בָּרָא אֶת הָעוֹלָם וְגָאַל אוֹתָנוּ, וּבְכִבּוּד הוֹרִים אֲנַחְנוּ מוֹדִים לְמִי שֶׁהֵם שֻׁתָּפָיו. וְה׳ אוֹמֵר עָלֵינוּ ׳מִי יִתֵּן׳ – הוּא מְחַכֶּה שֶׁנִּבְחַר בַּטּוֹב, ׳לְמַעַן יִיטַב לָהֶם וְלִבְנֵיהֶם לְעֹלָם׳, בָּאָרֶץ שֶׁהוּא נוֹתֵן לָנוּ.

🗣️ הַתִּקּוּן שֶׁלָּנוּ: לְדַבֵּר בְּשִׁבְחָהּ שֶׁל הָאָרֶץ

לָמָּה מֹשֶׁה מַזְכִּיר אֶת חֵטְא הַמְרַגְּלִים דַּוְקָא עַכְשָׁו? כִּי עַם יִשְׂרָאֵל שׁוּב עוֹמֵד עַל סַף הָאָרֶץ, וְחָשׁוּב שֶׁיִּכָּנְסוּ אֵלֶיהָ מִתּוֹךְ רְאִיַּת הַטּוֹב שֶׁבָּהּ, וְלֹא מִתּוֹךְ חִפּוּשׂ פְּגָמִים.

וְאֵיךְ מְתַקְּנִים אֶת הַחֵטְא גַּם הַיּוֹם? מְדַבְּרִים בְּשִׁבְחָהּ שֶׁל אֶרֶץ יִשְׂרָאֵל: כָּל אֶחָד מְסַפֵּר עַל מָקוֹם אוֹ דָּבָר שֶׁהוּא אוֹהֵב בָּאָרֶץ.

דִּבּוּר טוֹב עַל הָאָרֶץ מְתַקֵּן אֶת דִּבַּת הַמְּרַגְּלִים.

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּרַגְּלִים רָאוּ אֶת אוֹתָהּ אָרֶץ שֶׁרָאוּ כָּלֵב וִיהוֹשֻׁעַ – אֲבָל הַלֵּב שֶׁלָּהֶם רָאָה אַחֶרֶת. ה׳ אָהַב אֶת עַמּוֹ, נָשָׂא אוֹתוֹ כְּבֵן, וְהָלַךְ לְפָנָיו בָּאֵשׁ וּבֶעָנָן. כְּשֶׁאֲנַחְנוּ רוֹאִים בְּעֵינַיִם שֶׁל אֱמוּנָה וּמְדַבְּרִים בְּשִׁבְחָהּ שֶׁל אֶרֶץ יִשְׂרָאֵל, אֲנַחְנוּ מְמַלְּאִים אַחֲרֵי ה׳ כְּמוֹ כָּלֵב, וְזוֹכִים לָאָרֶץ הַטּוֹבָה.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי הַפַּחַד גָּרַם לִי לִרְאוֹת דָּבָר גָּדוֹל וּמַפְחִיד יוֹתֵר מִמָּה שֶׁהוּא בֶּאֱמֶת?
  • אֵיזֶה דָּבָר טוֹב בְּאֶרֶץ יִשְׂרָאֵל אֲנִי יָכוֹל לְסַפֵּר הַיּוֹם לַחֲבֵרִים שֶׁלִּי?
  • מָתַי הִרְגַּשְׁתִּי שֶׁה׳ נוֹשֵׂא אוֹתִי, כְּמוֹ אַבָּא שֶׁנּוֹשֵׂא אֶת בְּנוֹ?

פָּרָשַׁת כִּי תֵצֵא מְלַמֶּדֶת שֶׁאֵין מָקוֹם שֶׁאֵין בּוֹ מִצְוָה – לֹא הַבַּיִת, לֹא הָרְחוֹב וְלֹא שְׂדֵה הַקְּרָב. ה׳ מַכִּיר אֶת הַלֵּב שֶׁלָּנוּ וְעוֹזֵר לָנוּ לְנַצֵּחַ אֶת הַיֵּצֶר. וּכְשֶׁחַיָּלֵינוּ יוֹצְאִים לְהָגֵן עַל אֶרֶץ יִשְׂרָאֵל, ה׳ מִתְהַלֵּךְ בְּתוֹךְ הַמַּחֲנֶה שֶׁלָּהֶם.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי הַיֵּצֶר שֶׁלִּי חָזָק בִּמְיֻחָד – וּמָה יָכוֹל לַעֲזֹר לִי לַעֲצֹר?
  • אֵיזוֹ מִצְוָה אֲנִי יָכוֹל לְקַיֵּם הַיּוֹם בָּרְחוֹב אוֹ בַּשָּׂדֶה, וְלֹא רַק בְּבֵית הַכְּנֶסֶת?
  • אֵיךְ מַעֲשֶׂה טוֹב אֶחָד שֶׁלִּי יָכוֹל לְהוֹבִיל לְעוֹד מַעֲשֶׂה טוֹב?

רַשִׁ״י מְבָאֵר אֶת ׳וְהַלֵּוִי וְהַגֵּר וְהַיָּתוֹם וְהָאַלְמָנָה׳: אַרְבָּעָה שֶׁלִּי כְּנֶגֶד אַרְבָּעָה שֶׁלְּךָ – בִּנְךָ וּבִתֶּךָ וְעַבְדְּךָ וַאֲמָתֶךָ. אִם אַתָּה מְשַׂמֵּחַ אֶת שֶׁלִּי, אֲנִי מְשַׂמֵּחַ אֶת שֶׁלְּךָ. כְּאִלּוּ ה׳ אוֹמֵר: לַלֵּוִי אֵין נַחֲלָה וְאֵין תְּבוּאָה, הַגֵּר רָחוֹק מִמִּשְׁפַּחְתּוֹ, לַיָּתוֹם וְלָאַלְמָנָה אֵין מִי שֶׁיְּשַׂמַּח אוֹתָם – הֵם ׳שֶׁלִּי׳. תַּזְמִין אוֹתָם לַשֻּׁלְחָן שֶׁלְּךָ, וַאֲנִי אֶדְאַג לְשִׂמְחַת הַמִּשְׁפָּחָה שֶׁלְּךָ.

מִי שֶׁמְּשַׂמֵּחַ אֶת הָאַרְבָּעָה שֶׁל ה׳ – ה׳ מְשַׂמֵּחַ אֶת הָאַרְבָּעָה שֶׁלּוֹ.

רָמָה א׳ – עוֹלִים לִירוּשָׁלַיִם

מָה לוֹקְחִים מִרָמָה א׳?

שָׁלֹשׁ פְּעָמִים בַּשָּׁנָה כָּל עַם יִשְׂרָאֵל נִפְגָּשׁ בִּירוּשָׁלַיִם, לִפְנֵי ה׳. מֵבִיאִים קָרְבָּנוֹת, אוֹכְלִים מַצָּה וְזוֹכְרִים אֶת יְצִיאַת מִצְרַיִם, וּשְׂמֵחִים בַּיְּבוּל שֶׁל אֶרֶץ יִשְׂרָאֵל. אֲבָל הַשִּׂמְחָה שְׁלֵמָה רַק כְּשֶׁגַּם מִי שֶׁאֵין לוֹ יוֹשֵׁב אִתָּנוּ לַשֻּׁלְחָן – כִּי אֶת הָאֲנָשִׁים הָאֵלֶּה ה׳ קוֹרֵא ׳שֶׁלִּי׳.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • אֶת מִי אֲנִי יָכוֹל לְהַזְמִין לַשֻּׁלְחָן שֶׁלָּנוּ בֶּחָג הַבָּא, כְּדֵי שֶׁגַּם הוּא יִשְׂמַח?
  • מָתַי הִרְגַּשְׁתִּי ׳מִחוּץ לָעִנְיָנִים׳ – וּמָה זֶה מְלַמֵּד אוֹתִי עַל יְלָדִים אֲחֵרִים?
  • עַל אֵיזוֹ בְּרָכָה שֶׁקִּבַּלְתִּי הַשָּׁנָה אֲנִי רוֹצֶה לְהוֹדוֹת לְה׳ בֶּחָג?

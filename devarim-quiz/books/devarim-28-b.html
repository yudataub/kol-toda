📖 מַה מֻּתָּר וּמָה אָסוּר

הַתּוֹרָה בּוֹנָה אֶת הַפָּרָשָׁה בְּסֵדֶר: בַּבְּהֵמוֹת – פְּסוּקִים ד׳–ו׳ מְלַמְּדִים מָה מֻתָּר, וּפְסוּקִים ז׳–ח׳ – מָה אָסוּר.

בַּדָּגִים – פָּסוּק ט׳ הַמֻּתָּר וּפָסוּק י׳ הָאָסוּר.

הָאֲסוּרִים שֶׁהַתּוֹרָה מְפָרֶטֶת בַּבְּהֵמוֹת הֵם אֵלֶּה שֶׁיֵּשׁ לָהֶם רַק סִימָן אֶחָד, כִּי הֵם עֲלוּלִים לְבַלְבֵּל.

קֹדֶם הַמֻּתָּר, אַחַר כָּךְ הָאָסוּר – וְהַמְּבַלְבְּלִים בְּשֵׁמוֹת.

וְעוֹד לָמַדְנוּ מֵרַשִׁ״י: גַּם הַחַיָּה, כְּמוֹ אַיָּל וּצְבִי, בִּכְלַל ׳בְּהֵמָה׳.

בַּבְּהֵמוֹת – רְשִׁימַת הַטְּהוֹרוֹת. בָּעוֹפוֹת – רְשִׁימַת הַטְּמֵאִים.

שֶׁרֶץ הָעוֹף = נְמוּכִים וְרוֹחֲשִׁים: זְבוּבִים, צְרָעִים, חֲגָבִים טְמֵאִים.

🦌 לֹא תֹאכְלוּ כָל נְבֵלָה

׳לֹא תֹאכְלוּ כָל נְבֵלָה׳ – נְבֵלָה הִיא בְּהֵמָה שֶׁמֵּתָה בְּלִי שְׁחִיטָה כְּשֵׁרָה. גַּם בְּהֵמָה טְהוֹרָה, אִם מֵתָה מֵעַצְמָהּ אוֹ שֶׁיָּרוּ בָּהּ בְּצַיִד – אֲסוּרָה. אֶת הַנְּבֵלָה מֻתָּר לָתֵת לְגֵר תּוֹשָׁב אוֹ לִמְכֹּר לְנָכְרִי.

רַשִׁ״י: גֵּר תּוֹשָׁב הוּא מִי שֶׁקִּבֵּל עָלָיו שֶׁלֹּא לַעֲבֹד עֲבוֹדָה זָרָה, וְאוֹכֵל נְבֵלוֹת.

בָּשָׂר מֻתָּר רַק אַחֲרֵי שְׁחִיטָה כְּשֵׁרָה.

🏷️ עַם קָדוֹשׁ – גַּם בַּסּוּפֶּר

הַפָּרָשָׁה מַתְחִילָה וּמִסְתַּיֶּמֶת בְּ׳עַם קָדוֹשׁ׳ – הַמַּאֲכָלוֹת מַבְדִּילִים אֶת הַיְּהוּדִי מִסְּבִיבָתוֹ, וּבִמְיֻחָד בְּחוּץ לָאָרֶץ, כְּשֶׁיְּהוּדִים חַיִּים בֵּין נָכְרִים וּמַקְפִּידִים עַל אֹכֶל כָּשֵׁר. הַיּוֹם מַכְנִיסִים לְהַרְבֵּה מוּצָרִים – גַּם לְלֶחֶם וּלְפַסְטָה – שֻׁמָּנִים, מְיַצְּבִים וְחֹמָרִים שֶׁלִּפְעָמִים בָּאִים מִבְּהֵמוֹת טְמֵאוֹת. לָכֵן כִּמְעַט כָּל מוּצָר צָרִיךְ הֶכְשֵׁר: בַּסּוּפֶּר, בַּמִּסְעָדָה, בַּמָּטוֹס, בָּאָרֶץ וּבְחוּץ לָאָרֶץ.

בּוֹדְקִים הֶכְשֵׁר – כִּי אֲנַחְנוּ עַם קָדוֹשׁ בְּכָל מָקוֹם.

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

מָה לוֹקְחִים מִרָמָה ב׳?

יְהוּדִים בְּכָל הַדּוֹרוֹת, בְּכָל אֶרֶץ וּבְכָל מַצָּב, שָׁמְרוּ עַל כַּשְׁרוּת – וְזֶה הָיָה אֶחָד הַסִּימָנִים הַבּוֹלְטִים שֶׁל עַם קָדוֹשׁ. הַכַּשְׁרוּת מְלַמֶּדֶת אוֹתָנוּ לִבְדֹּק, לְהִתְאַפֵּק וּלְהַקְשִׁיב לְה׳ גַּם בִּדְבָרִים קְטַנִּים שֶׁל כָּל יוֹם. וְכָל אֲרוּחָה כְּשֵׁרָה הוֹפֶכֶת אֶת הַשֻּׁלְחָן שֶׁלָּנוּ לְמָקוֹם שֶׁל קְדֻשָּׁה וְאַהֲבַת ה׳.

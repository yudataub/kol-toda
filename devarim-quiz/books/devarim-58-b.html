שִׁירַת הַאֲזִינוּ כְּתוּבָה בְּסֵפֶר הַתּוֹרָה בְּצוּרָה מְיֻחֶדֶת, לֹא בְּשׁוּרוֹת מְלֵאוֹת.

יֵשׁ שְׁתֵּי שִׁירוֹת שֶׁכְּתוּבוֹת כָּךְ: שִׁירַת הַיָּם (בְּפָרָשַׁת בְּשַׁלַּח) וְשִׁירַת הַאֲזִינוּ.

וְיֵשׁ הֶבְדֵּל: שִׁירַת הַאֲזִינוּ כְּתוּבָה ׳אָרִיחַ עַל גַּבֵּי אָרִיחַ׳ – שְׁנֵי טוּרִים יְשָׁרִים.

שִׁירַת הַיָּם כְּתוּבָה ׳אָרִיחַ עַל גַּבֵּי לְבֵנָה, וּלְבֵנָה עַל גַּבֵּי אָרִיחַ׳ – כְּמוֹ קִיר לְבֵנִים.

הַאֲזִינוּ – אָרִיחַ עַל גַּבֵּי אָרִיחַ. שִׁירַת הַיָּם – אָרִיחַ עַל גַּבֵּי לְבֵנָה.

ה׳ אוֹמֵר לְמֹשֶׁה: הָעָם יִכָּנֵס לָאָרֶץ, ׳וְאָכַל וְשָׂבַע וְדָשֵׁן, וּפָנָה אֶל אֱלֹהִים אֲחֵרִים׳.

׳הֶסְתֵּר פָּנִים׳ – ה׳ מַשְׁגִּיחַ עָלֵינוּ בְּאֹפֶן נִסְתָּר, בְּלִי נִסִּים גְּלוּיִים.

וְכֵן בַּשִּׁירָה: ׳וַיֹּאמֶר: אַסְתִּירָה פָנַי מֵהֶם, אֶרְאֶה מָה אַחֲרִיתָם׳.

הֶסְתֵּר פָּנִים = ה׳ כְּאִלּוּ מַסְתִּיר אֶת הַהַשְׁגָּחָה הַגְּלוּיָה.

הַהֶסְתֵּר לֹא נִמְשָׁךְ לְעוֹלָם. ׳כִּי יָדִין ה׳ עַמּוֹ וְעַל עֲבָדָיו יִתְנֶחָם, כִּי יִרְאֶה כִּי אָזְלַת יָד וְאֶפֶס עָצוּר וְעָזוּב׳.

כְּשֶׁה׳ רוֹאֶה שֶׁאֵין לְיִשְׂרָאֵל כֹּחַ – מִתְמַלֵּא רַחֲמִים וּמוֹשִׁיעַ.

וְאַף כְּשֶׁהָיָה רָאוּי לִדִין קָשֶׁה – ׳אָמַרְתִּי אַפְאֵיהֶם׳ – ה׳ אֵינוֹ עוֹשֶׂה כָּךְ, כְּדֵי שֶׁהָאוֹיְבִים לֹא יֹאמְרוּ ׳יָדֵינוּ רָמָה וְלֹא ה׳ פָּעַל כָּל זֹאת׳.

ה׳ מַעֲנִישׁ – אֲבָל לְעוֹלָם לֹא עוֹזֵב, וּבַסּוֹף מְרַחֵם וּמוֹשִׁיעַ.

🧑‍⚖️ הַשִּׁירָה – עֵדָה

לָמָּה צָרִיךְ לָשִׂים אֶת הַשִּׁירָה בַּפֶּה? ׳וְהָיָה כִּי תִמְצֶאןָ אֹתוֹ רָעוֹת רַבּוֹת וְצָרוֹת, וְעָנְתָה הַשִּׁירָה הַזֹּאת לְפָנָיו לְעֵד, כִּי לֹא תִשָּׁכַח מִפִּי זַרְעוֹ׳.

כְּשֶׁיָּבוֹאוּ זְמַנִּים קָשִׁים, יִזְכְּרוּ אֶת הַשִּׁירָה וְיָבִינוּ שֶׁה׳ יָדַע זֹאת מֵרֹאשׁ – וּשֶׁהוּא הִבְטִיחַ שֶׁבַּסּוֹף יְרַחֵם.

וְשׁוּב ה׳ מְחַזֵּק אֶת יְהוֹשֻׁעַ: ׳חֲזַק וֶאֱמָץ... וְאָנֹכִי אֶהְיֶה עִמָּךְ׳.

הַשִּׁירָה לֹא תִּשָּׁכַח מִפִּי הַזֶּרַע – וְתָעִיד בַּזְּמַנִּים הַקָּשִׁים.

🎶 הַזִּי״ו לָ״ךְ

בְּכָל יוֹם הָיוּ הַלְוִיִּם שָׁרִים שִׁיר בְּבֵית הַמִּקְדָּשׁ.

בְּשַׁבָּת, עִם קָרְבַּן הַתָּמִיד – ׳מִזְמוֹר שִׁיר לְיוֹם הַשַּׁבָּת׳.

וְעִם קָרְבַּן הַמּוּסָף – שִׁירַת הַאֲזִינוּ, מְחֻלֶּקֶת לְשִׁשָּׁה חֲלָקִים, חֵלֶק אֶחָד בְּכָל שַׁבָּת.

רָאשֵׁי הַתֵּבוֹת שֶׁל הַחֲלָקִים הֵם ׳הַזִּי״ו לָ״ךְ׳ (׳הַאֲזִינוּ׳, ׳זְכֹר יְמוֹת עוֹלָם׳...), וְזוֹ גַּם הַחֲלֻקָּה שֶׁל הָעֲלִיּוֹת בְּפָרָשַׁת הַאֲזִינוּ עַד הַיּוֹם.

הַלְוִיִּם שָׁרוּ אֶת הַאֲזִינוּ בְּמוּסַף שַׁבָּת – שִׁשָּׁה חֲלָקִים.

הַשִּׁירָה מִסְתַּיֶּמֶת בְּתִקְוָה: ׳הַרְנִינוּ גוֹיִם עַמּוֹ... וְכִפֶּר אַדְמָתוֹ עַמּוֹ׳.

רַשִׁ״י: לְאוֹתוֹ הַזְּמַן יְשַׁבְּחוּ הָאֻמּוֹת אֶת יִשְׂרָאֵל – רְאוּ מַה שִּׁבְחָהּ שֶׁל אֻמָּה זוֹ, שֶׁדָּבְקוּ בַּקָּדוֹשׁ בָּרוּךְ הוּא בְּכָל הַתְּלָאוֹת וְלֹא עֲזָבוּהוּ.

וְאַחֲרֵי הַשִּׁירָה מֹשֶׁה אוֹמֵר: ׳כִּי לֹא דָבָר רֵק הוּא מִכֶּם, כִּי הוּא חַיֵּיכֶם׳ – רַשִׁ״י: לֹא לְחִנָּם אַתֶּם יְגֵעִים בָּהּ, כִּי הַרְבֵּה שָׂכָר תָּלוּי בָּהּ.

בַּסּוֹף – כָּל הָעַמִּים יְשַׁבְּחוּ אֶת יִשְׂרָאֵל שֶׁלֹּא עָזַב אֶת ה׳.

מָה לוֹקְחִים מִרָמָה ב׳?

שִׁירַת הַאֲזִינוּ הִיא הַבְטָחָה שֶׁלֹּא נִשְׁבֶּרֶת: גַּם כְּשֶׁעַם יִשְׂרָאֵל עוֹזֵב אֶת ה׳, ה׳ לֹא עוֹזֵב אוֹתוֹ. הוּא מַסְתִּיר פָּנִים – וּמִתְנַחֵם, מַעֲנִישׁ – וּמוֹשִׁיעַ. לָכֵן שָׁרוּ אוֹתָהּ הַלְוִיִּם בַּמִּקְדָּשׁ, וְלָכֵן נָהֲגוּ לְלַמֵּד אוֹתָהּ בְּעַל פֶּה – כְּדֵי שֶׁבְּכָל דּוֹר נֵדַע: ה׳ שׁוֹמֵר עָלֵינוּ כְּנֶשֶׁר עַל גּוֹזָלָיו, וְלֹא יַעַזְבֵנוּ לְעוֹלָם.

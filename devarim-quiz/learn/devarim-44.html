וְשָׁם אוֹמֵר רַבִּי מֵאִיר: בְּשָׁעָה שֶׁאָדָם מִצְטַעֵר, הַשְּׁכִינָה מִצְטַעֶרֶת עִמּוֹ. וְאִם ה׳ מִצְטַעֵר עַל דָּמָם שֶׁל רְשָׁעִים שֶׁנִּשְׁפַּךְ – קַל וָחֹמֶר עַל דָּמָם שֶׁל צַדִּיקִים.
זוֹ אַהֲבַת ה׳ לִבְרִיּוֹתָיו: הוּא כּוֹאֵב עִם כָּל אָדָם, תָּמִיד.
💡 מָתַי מֻתָּר לְהַמְתִּין – וּמִנְהַג יְרוּשָׁלַיִם
אֵין אִסּוּר כְּשֶׁהַשְּׁהִיָּה הִיא לְצֹרֶךְ הַמֵּת: לְהָכִין תַּכְרִיכִים, אוֹ לְהַמְתִּין לִמְלַוִּים שֶׁיָּבוֹאוּ לְכַבְּדוֹ.
וּבִירוּשָׁלַיִם יֵשׁ מִנְהָג מְיֻחָד: אֵין מַלִּינִים בָּהּ אֶת הַמֵּת כְּלָל, וְקוֹבְרִים אוֹתוֹ גַּם בְּשָׁעוֹת מְאֻחָרוֹת בַּלַּיְלָה.

וְעוֹד מְלַמֵּד הַפָּסוּק דְּבַר תּוֹרָה קָטָן בְּדִקְדּוּק: בִּפְסוּקִים כ״ב–כ״ג מוֹפִיעָה הַמִּלָּה ׳כִּי׳ בִּשְׁלוֹשָׁה פֵּרוּשִׁים שׁוֹנִים:
׳וְכִי יִהְיֶה בְאִישׁ חֵטְא׳ – כַּאֲשֶׁר (הַמִּקְרֶה).
׳כִּי קָבוֹר תִּקְבְּרֶנּוּ׳ – אֶלָּא (הַדִּין).
׳כִּי קִלְלַת אֱלֹהִים תָּלוּי׳ – מִשּׁוּם שֶׁ (הַטַּעַם).

🤔 חִידַת הִתְמַצְּאוּת
מִי בַּתַּנַ״ךְ קִיֵּם אֶת הַחֹק הַזֶּה?
יְהוֹשֻׁעַ – בְּמֶלֶךְ הָעַי וּבְמַלְכֵי הַדָּרוֹם (יְהוֹשֻׁעַ פְּרָקִים ח׳, י׳), שֶׁהוֹרִיד אוֹתָם מִן הָעֵץ עִם בּוֹא הַשֶּׁמֶשׁ.
וּבְאֹפֶן אַחֵר – אַנְשֵׁי יָבֵשׁ גִּלְעָד, שֶׁהָלְכוּ וְלָקְחוּ אֶת גּוּפוֹת שָׁאוּל וּבָנָיו מֵחוֹמַת בֵּית שְׁאָן כְּדֵי לְהָבִיאָם לִקְבוּרָה (שְׁמוּאֵל א׳ ל״א).
💛 אַהֲבַת הַשֵּׁם
ה׳ שׁוֹמֵר עַל כְּבוֹדוֹ שֶׁל כָּל אָדָם, גַּם שֶׁל מִי שֶׁנִּכְשַׁל. אִם כָּךְ – כַּמָּה יָקָר אֲנִי בְּעֵינָיו, וְכַמָּה אֲנִי צָרִיךְ לְכַבֵּד כָּל אָדָם שֶׁאֲנִי פּוֹגֵשׁ, כִּי בְּכֻלָּם יֵשׁ צֶלֶם אֱלֹהִים 💛

📜מִלּוֹן קָטָן לַפֶּרֶק – מִלִּים מֵרַשִׁ״י

לִפְנֵי שֶׁסּוֹגְרִים אֶת הַפֶּרֶק, נֶאֱסֹף יַחַד אֶת הַמִּלִּים שֶׁרַשִׁ״י בֵּאֵר לָנוּ. כְּשֶׁיּוֹדְעִים אֶת פֵּרוּשׁ הַמִּלִּים – הַפְּסוּקִים נִפְתָּחִים כְּמוֹ דֶּלֶת.

💡 מִלּוֹן
׳סוֹרֵר׳ – סָר מִן הַדֶּרֶךְ.
׳וּמוֹרֶה׳ – מְסָרֵב בְּדִבְרֵי אָבִיו, לְשׁוֹן מַמְרִים.
׳פִּי שְׁנַיִם׳ – כְּנֶגֶד שְׁנֵי אַחִים.
׳בְּכֹל אֲשֶׁר יִמָּצֵא לוֹ׳ – לֹא בְּמָה שֶׁרָאוּי לָבוֹא לְאַחַר מִיתַת הָאָב.
׳כִּי קִלְלַת אֱלֹהִים תָּלוּי׳ – זִלְזוּלוֹ שֶׁל מֶלֶךְ.
׳לְבַכֵּר׳ – לְהַעֲדִיף.
׳רֵאשִׁית אֹנוֹ׳ – רֵאשִׁית כֹּחוֹ.
׳זוֹלֵל וְסֹבֵא׳ – אוֹכֵל וְשׁוֹתֶה בְּלִי גְּבוּל.

שִׂימוּ לֵב מָה מְאַחֵד אֶת שְׁלוֹשֶׁת הַנּוֹשְׂאִים שֶׁלָּמַדְנוּ הַיּוֹם: כְּבוֹד הָאָדָם.
כְּבוֹד הַבֵּן הַבְּכוֹר, שֶׁמְּקַבֵּל אֶת מָה שֶׁמַּגִּיעַ לוֹ גַּם אִם אָבִיו אוֹהֵב אַחֵר יוֹתֵר; כְּבוֹד הַהוֹרִים, שֶׁפָּרָשָׁה שְׁלֵמָה נִכְתְּבָה כְּדֵי לְלַמְּדוֹ; וּכְבוֹדוֹ שֶׁל אָדָם שֶׁנִּפְטַר, שֶׁאָסוּר לְהַשְׁאִירוֹ בְּבִזָּיוֹן.

וּבְכָל אֵלֶּה מְסֻתֶּרֶת אוֹתָהּ אַהֲבָה גְּדוֹלָה: ה׳ בָּרָא אֶת הָאָדָם בְּצַלְמוֹ, קָרָא לָנוּ ׳בָּנִים אַתֶּם לַה׳ אֱלֹהֵיכֶם׳, וְלָכֵן הוּא שׁוֹמֵר עָלֵינוּ מִכָּל צַד – בַּמָּמוֹן, בַּמִּשְׁפָּחָה וּבַכָּבוֹד.

💛 אַהֲבַת הַשֵּׁם
כְּשֶׁאֲנִי לוֹמֵד אֶת פֵּרוּשׁ הַמִּלִּים, אֲנִי מַרְגִּישׁ שֶׁאֲנִי מַכִּיר אֶת ה׳ קְצָת יוֹתֵר טוֹב. כָּל מִלָּה בַּתּוֹרָה הִיא חַלּוֹן קָטָן שֶׁדַּרְכּוֹ אֶפְשָׁר לִרְאוֹת כַּמָּה הוּא אוֹהֵב אוֹתָנוּ 💛

✨ לְסִכּוּם

  • בְּפָרָשָׁה הֲלָכָתִית הַתּוֹרָה מְצִיגָה קֹדֶם מִקְרֶה וְאַחַר כָּךְ דִּין, כְּמוֹ בְּבֵית דִּין. כָּאן הַמִּקְרֶה: אָב שֶׁהַבְּכוֹר שֶׁלּוֹ נוֹלַד לָאִשָּׁה הָאֲהוּבָה פָּחוֹת; הַדִּין: ׳לֹא יוּכַל לְבַכֵּר׳.
  • הַמִּלָּה הַמַּנְחָה הִיא ב־כ־ר – בְּכוֹר, לְבַכֵּר, מִשְׁפַּט הַבְּכוֹרָה. ׳לְבַכֵּר׳ פֵּרוּשׁוֹ לְהַעֲדִיף.
  • הַבְּכוֹר נוֹטֵל פִּי שְׁנַיִם – רַשִׁ״י: כְּנֶגֶד שְׁנֵי אַחִים. מְחַלְּקִים לְמִסְפַּר הַבָּנִים וְעוֹד אֶחָד, וְרַק בַּנְּכָסִים שֶׁמְּצוּיִים אֵצֶל הָאָב בִּשְׁעַת מוֹתוֹ.
  • בִּימֵי קֶדֶם הָיָה מָצוּי שֶׁלְּאִישׁ שְׁתֵּי נָשִׁים; לִפְנֵי כְּאֶלֶף שָׁנָה תִּקֵּן רַבֵּנוּ גֵּרְשׁוֹם חֵרֶם הָאוֹסֵר זֹאת.
  • בֵּן סוֹרֵר וּמוֹרֶה ׳לֹא הָיָה וְלֹא עָתִיד לִהְיוֹת׳ – תְּנָאָיו נְדִירִים, וְהוּא נִדּוֹן עַל שֵׁם סוֹפוֹ. וְלָמָּה נִכְתַּב? ׳דְּרֹשׁ וְקַבֵּל שָׂכָר׳ – לְלַמְּדֵנוּ כַּמָּה חֲשׁוּבָה מִצְוַת כִּבּוּד הוֹרִים.
  • אָסוּר לְהָלִין אֶת הַמֵּת: ׳קָבוֹר תִּקְבְּרֶנּוּ בַּיּוֹם הַהוּא׳, כִּי הָאָדָם נִבְרָא בְּצֶלֶם – וּמָשָׁל הַתְּאוֹמִים שֶׁל רַשִׁ״י מְלַמֵּד שֶׁבִּזְיוֹנוֹ הוּא זִלְזוּלוֹ שֶׁל מֶלֶךְ. ה׳ אוֹהֵב כָּל אָדָם וּמִצְטַעֵר בְּצַעֲרוֹ.

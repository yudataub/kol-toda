מִצְוַת מַעֲקֶה

הַחֹמֶר הַמָּלֵא: ׳וְלֹא תָשִׂים דָּמִים בְּבֵיתֶךָ׳ – זְהִירוּת, אַחְרָיוּת, וּמִצְוָה גּוֹרֶרֶת מִצְוָה (כב, ח)
🎯 לַחִידוֹן🏠 כָּל הַחִידוֹנִים
הַשִּׁעוּר הַזֶּה הוּא עַל פָּסוּק אֶחָד בִּלְבַד – וְיֵשׁ בּוֹ עוֹלָם שָׁלֵם. בַּיּוֹם הָרִאשׁוֹן לָמַדְנוּ אַרְבָּעָה פְּסוּקִים (הֲשָׁבַת אֲבֵדָה), אֶתְמוֹל שְׁנַיִם (שִׁלּוּחַ הַקֵּן), וְהַיּוֹם נִתְעַכֵּב עַל פָּסוּק אֶחָד: ׳כִּי תִבְנֶה בַּיִת חָדָשׁ וְעָשִׂיתָ מַעֲקֶה לְגַגֶּךָ׳.
נִלְמַד מָה זֶה מַעֲקֶה, לָמָּה יֵשׁ כָּאן שְׁנֵי דִּינִים – מִצְוַת עֲשֵׂה וּמִצְוַת לֹא תַעֲשֶׂה – וְאֵיךְ מִן הַגַּג שֶׁל פַּעַם לוֹמְדִים עִקָּרוֹן גָּדוֹל שֶׁנּוֹגֵעַ לַבַּיִת שֶׁלָּנוּ הַיּוֹם: לִבְרֵכָה, לְשֶׁקַע חַשְׁמַל, לְסֻלָּם רָעוּעַ וּלְעָנָף שֶׁעוֹמֵד לִפֹּל.
וּבְעִקָּר נִרְאֶה כַּמָּה ה׳ אוֹהֵב כָּל אָדָם: הוּא כָּל כָּךְ שׁוֹמֵר עַל הַחַיִּים, עַד שֶׁהוּא מְצַוֶּה אוֹתִי לִבְנוֹת גָּדֵר קְטַנָּה עַל הַגַּג – כְּדֵי שֶׁאֲפִלּוּ אִישׁ אֶחָד לֹא יִפָּגַע 💛
בַּשִּׁעוּר: ⚠️ תַּמְרוּרֵי סַכָּנָה – פְּתִיחָה 🏠 כִּי תִבְנֶה בַּיִת חָדָשׁ – מִקְרֶה וְדִין 🧱 מָה זֶה מַעֲקֶה? גָּדֵר סָבִיב לַגַּג ✅ שְׁנֵי דִּינִים בְּפָסוּק אֶחָד – וְלֹא תָשִׂים דָּמִים 📕 הָרַמְבַּ״ם: כָּל מִכְשׁוֹל שֶׁיֵּשׁ בּוֹ סַכָּנָה – מְסִירִים 🧭 דִּבֵּר הַכָּתוּב בַּהֹוֶה – מִדֻּגְמָה לְעִקָּרוֹן 🔌 דֻּגְמָאוֹת מִן הַחַיִּים – וְגַם כֶּלֶב רַע וְסֻלָּם רָעוּעַ 🔁 שְׁלוֹשֶׁת אֲלָפִים שָׁנָה – וְהַתּוֹרָה מַעֲשִׂית הַיּוֹם 🎲 ׳כִּי יִפֹּל הַנֹּפֵל׳ – רָאוּי זֶה לִפֹּל 🔗 סְמִיכוּת פָּרָשִׁיּוֹת – מִצְוָה גּוֹרֶרֶת מִצְוָה

⚠️תַּמְרוּרֵי סַכָּנָה – פְּתִיחָה

לִפְנֵי שֶׁפּוֹתְחִים חֻמָּשׁ, נִזְכֹּר שְׁלָטִים שֶׁכֻּלָּנוּ מַכִּירִים:
סַכָּנַת הַחְלָקָה – בְּמָקוֹם רָטֹב.
מֶתַח גָּבוֹהַּ – עַל אֲרוֹן חַשְׁמַל.
חֹמֶר רָעִיל – דָּבָר שֶׁצָּרִיךְ לְהִתְרַחֵק מִמֶּנּוּ.
מִשְׁטָח חַם – בְּמִטְבָּח אוֹ בְּמוֹסְדוֹת צִבּוּר.
מַחְסוֹם שֶׁיּוֹרֵד – עָדִיף לֹא לַעֲבֹר תַּחְתָּיו.

מָה מְשֻׁתָּף לְכֻלָּם? כֻּלָּם אוֹמְרִים מִלָּה אַחַת: ׳הִזָּהֵר!׳.
גַּם הַתּוֹרָה נוֹתֶנֶת לָנוּ מִצְוָה שֶׁל זְהִירוּת – מִצְוַת מַעֲקֶה – וְהִיא כְּתוּבָה בְּפָסוּק אֶחָד קָצָר.

וְהִנֵּה הַהֶבְדֵּל הַגָּדוֹל: שֶׁלֶט אַזְהָרָה אוֹמֵר לִי ׳תִּשְׁמֹר עַל עַצְמְךָ׳. הַתּוֹרָה מוֹסִיפָה מַשֶּׁהוּ גָּדוֹל יוֹתֵר: תִּשְׁמֹר שֶׁאֲחֵרִים לֹא יִפָּגְעוּ מִמְּךָ. ה׳ אוֹהֵב כָּל אָדָם, וְלָכֵן הוּא מַפְקִיד בְּיָדַי אֶת הַשְּׁמִירָה עַל הַשֵּׁנִי.

💛 אַהֲבַת הַשֵּׁם
ה׳ בָּרָא אוֹתִי וְגַם אֶת הַשָּׁכֵן שֶׁלִּי, וְהוּא אוֹהֵב אֶת שְׁנֵינוּ. כְּשֶׁאֲנִי מֵסִיר מִכְשׁוֹל קָטָן מֵהַבַּיִת, אֲנִי שֻׁתָּף לַה׳ בִּשְׁמִירָה עַל הַחַיִּים שֶׁהוּא נָתַן 💛

🏠כִּי תִבְנֶה בַּיִת חָדָשׁ – מִקְרֶה וְדִין

📖 פֶּרֶק כ״ב, פָּסוּק ח׳ ←
ח כִּי תִבְנֶה בַּיִת חָדָשׁ וְעָשִׂיתָ מַעֲקֶה לְגַגֶּךָ וְלֹא תָשִׂים דָּמִים בְּבֵיתֶךָ כִּי יִפֹּל הַנֹּפֵל מִמֶּנּוּ

הַפָּסוּק: ׳כִּי תִבְנֶה בַּיִת חָדָשׁ וְעָשִׂיתָ מַעֲקֶה לְגַגֶּךָ, וְלֹא תָשִׂים דָּמִים בְּבֵיתֶךָ כִּי יִפֹּל הַנֹּפֵל מִמֶּנּוּ׳.

כְּמוֹ בְּמִצְווֹת שֶׁלָּמַדְנוּ קֹדֶם, גַּם כָּאן אֶפְשָׁר לְחַלֵּק אֶת הַפָּסוּק לִשְׁנֵי חֲלָקִים:
׳כִּי תִבְנֶה בַּיִת חָדָשׁ׳ – זֶה הַמִּקְרֶה: מָתַי הַמִּצְוָה שַׁיֶּכֶת.
׳וְעָשִׂיתָ מַעֲקֶה לְגַגֶּךָ׳ – זֶה הַדִּין: מָה צָרִיךְ לַעֲשׂוֹת.

וּמָה עִם הַמְשֵׁךְ הַפָּסוּק, ׳וְלֹא תָשִׂים דָּמִים בְּבֵיתֶךָ כִּי יִפֹּל הַנֹּפֵל מִמֶּנּוּ׳? גַּם הוּא שַׁיָּךְ לַדִּין – וְנַרְאֶה מִיָּד שֶׁיֵּשׁ כָּאן שְׁנֵי דִּינִים נִפְרָדִים.

🤔 לַמּוֹרֶה: מְשִׂימַת סִמּוּן
אֶפְשָׁר לְהַצִּיג אֶת הַפָּסוּק עַל הַלּוּחַ וּלְבַקֵּשׁ מֵהַתַּלְמִידִים לְסַמֵּן בִּשְׁנֵי צְבָעִים: אֵיפֹה הַמִּקְרֶה וְאֵיפֹה הַדִּין.
אַחַר כָּךְ לִשְׁאֹל: הַאִם ׳וְלֹא תָשִׂים דָּמִים׳ הוּא חֵלֶק מֵהַמִּקְרֶה אוֹ מֵהַדִּין? מִכָּאן מַגִּיעִים לְגִלּוּי שֶׁיֵּשׁ בַּפָּסוּק שְׁנֵי דִּינִים.

שִׂימוּ לֵב לַמִּלָּה ׳חָדָשׁ׳: הַבַּיִת רַק נִבְנָה, הַכֹּל שָׂמֵחַ וְנָקִי – וּכְבָר בָּרֶגַע הָרִאשׁוֹן הַתּוֹרָה מְבַקֶּשֶׁת לַחְשֹׁב עַל הַבְּטִיחוּת. כָּךְ אוֹהֵב אָב: עוֹד לִפְנֵי שֶׁקָּרָה מַשֶּׁהוּ, הוּא כְּבָר דּוֹאֵג.

💛 אַהֲבַת הַשֵּׁם
הַשִּׂמְחָה שֶׁל בַּיִת חָדָשׁ הִיא מַתָּנָה מֵה׳, וְהוּא מְלַמֵּד אוֹתִי אֵיךְ לִשְׁמֹר שֶׁהַשִּׂמְחָה תִּשָּׁאֵר שְׁלֵמָה 💛

׳מַעֲקֶה׳ – גָּדֵר סָבִיב לַגַּג. וְאוּנְקְלוֹס תִּרְגֵּם ׳תִּיקָא׳, כְּעֵין תִּיק שֶׁמְּשַׁמֵּר מָה שֶׁבְּתוֹכוֹ.

רַשִׁ״י מְבָאֵר בִּשְׁתֵּי מִלִּים: מַעֲקֶה הוּא גָּדֵר סְבִיב הַגַּג.
וְאוּנְקְלוֹס, שֶׁתִּרְגֵּם אֶת הַתּוֹרָה לַאֲרָמִית, בָּחַר מִלָּה יָפָה: ׳תִּיקָא׳ – תִּיק. כְּמוֹ תִּיק שֶׁשּׁוֹמֵר עַל מָה שֶׁבְּתוֹכוֹ, כָּךְ הַמַּעֲקֶה שׁוֹמֵר עַל הָאֲנָשִׁים שֶׁעַל הַגַּג.

💡 לָמָּה דַּוְקָא גַּג?
בִּימֵי הַתּוֹרָה הַגַּגּוֹת הָיוּ שְׁטוּחִים, וְלֹא מְשֻׁפָּעִים כְּמוֹ גַּג רְעָפִים. הָיוּ עוֹלִים אֲלֵיהֶם לָנוּחַ, לְיַבֵּשׁ פֵּרוֹת וְלִישֹׁן בַּקַּיִץ – מַמָּשׁ כְּמוֹ חֶדֶר נוֹסָף.
גַּג שֶׁמִּשְׁתַּמְּשִׁים בּוֹ בְּלִי גָּדֵר הוּא מָקוֹם מְסֻכָּן, וְלָכֵן הַתּוֹרָה מְצַוָּה לְהַקִּיף אוֹתוֹ בְּמַעֲקֶה.

הַמִּצְוָה שַׁיֶּכֶת לְבָתֵּי מְגוּרִים וּלְכָל גַּג שֶׁמִּשְׁתַּמְּשִׁים בּוֹ, וְלָכֵן גַּם הַיּוֹם עוֹשִׂים מַעֲקֶה לְמִרְפֶּסֶת וְלַגַּג. וּלְדַעַת הָרַמְבַּ״ם אַף מְבָרְכִים עַל הַמִּצְוָה – בְּרָכָה עַל גָּדֵר! כִּי שְׁמִירָה עַל חַיֵּי אָדָם הִיא מִצְוָה גְּדוֹלָה וּשְׂמֵחָה.

בַּפָּסוּק הַקָּצָר הַזֶּה יֵשׁ שְׁנֵי דִּינִים:
מִצְוַת עֲשֵׂה – ׳וְעָשִׂיתָ מַעֲקֶה לְגַגֶּךָ׳: חוֹבָה לִבְנוֹת אֶת הַגָּדֵר.

💡 וְגַם: ׳לְמַעַן יִיטַב לָךְ׳
בְּמִצְוַת שִׁלּוּחַ הַקֵּן כָּתוּב ׳לְמַעַן יִיטַב לָךְ וְהַאֲרַכְתָּ יָמִים׳, וְרַשִׁ״י מְבָאֵר שָׁם: אִם מִצְוָה קַלָּה שֶׁאֵין בָּהּ חֶסְרוֹן כִּיס זָכְתָה לְשָׂכָר כָּזֶה – קַל וָחֹמֶר לְשָׂכָר שֶׁל מִצְווֹת חֲמוּרוֹת.
וְהִנֵּה דַּוְקָא אַחֲרֶיהָ בָּאָה מִצְוַת מַעֲקֶה: הַשָּׂכָר עַל מִצְוָה אַחַת – הוּא הַהִזְדַּמְּנוּת לְקַיֵּם אֶת הַבָּאָה.

אֲנַחְנוּ רְגִילִים לַחְשֹׁב שֶׁשָּׂכָר הוּא מַתָּנָה. הַתּוֹרָה מְלַמֶּדֶת שֶׁהַשָּׂכָר הֲכִי יָפֶה שֶׁה׳ נוֹתֵן הוּא עוֹד מִצְוָה – עוֹד הִזְדַּמְּנוּת לְהִתְקָרֵב אֵלָיו. כְּמוֹ אַבָּא שֶׁמַּזְמִין אֶת הַיֶּלֶד לַעֲזֹר לוֹ שׁוּב, כִּי הוּא אוֹהֵב לִהְיוֹת אִתּוֹ.

💛 אַהֲבַת הַשֵּׁם
ה׳ מְסַדֵּר לִי אֶת הַחַיִּים כָּךְ שֶׁכָּל מַעֲשֶׂה טוֹב יִפְתַּח לִי דֶּלֶת לְמַעֲשֶׂה טוֹב נוֹסָף. אֲנִי אוֹהֵב אוֹתוֹ, וְהוּא מוֹשֵׁךְ אוֹתִי עוֹד וָעוֹד קָרוֹב אֵלָיו 💛

✨ לְסִכּוּם

  • ׳כִּי תִבְנֶה בַּיִת חָדָשׁ׳ – זֶה הַמִּקְרֶה; ׳וְעָשִׂיתָ מַעֲקֶה לְגַגֶּךָ׳ – זֶה הַדִּין. מַעֲקֶה לְפִי רַשִׁ״י הוּא גָּדֵר סָבִיב לַגַּג, וְאוּנְקְלוֹס תִּרְגֵּם ׳תִּיקָא׳ – כְּעֵין תִּיק שֶׁשּׁוֹמֵר מָה שֶׁבְּתוֹכוֹ.
  • בַּפָּסוּק שְׁנֵי דִּינִים: מִצְוַת עֲשֵׂה – לַעֲשׂוֹת מַעֲקֶה; מִצְוַת לֹא תַעֲשֶׂה – ׳וְלֹא תָשִׂים דָּמִים בְּבֵיתֶךָ׳. הַבּוֹנֶה אַחְרַאי עַל כָּל מִכְשׁוֹל, וְכָל עוֹד לֹא תִּקֵּן – עוֹבֵר בְּלָאו.
  • הָרַמְבַּ״ם (הִלְכוֹת רוֹצֵחַ וּשְׁמִירַת הַנֶּפֶשׁ): כָּל מִכְשׁוֹל שֶׁיֵּשׁ בּוֹ סַכָּנַת נְפָשׁוֹת – מִצְוַת עֲשֵׂה לַהֲסִירוֹ וּלְהִזָּהֵר ׳יָפֶה יָפֶה׳, מִן הַפָּסוּק ׳הִשָּׁמֶר לְךָ וּשְׁמֹר נַפְשְׁךָ מְאֹד׳.
  • סֵפֶר הַחִנּוּךְ: לְהָסִיר אֶת הַמִּכְשׁוֹלִים מִכָּל מִשְׁכְּנוֹתֵינוּ – ׳לְגַגֶּךָ׳ הוּא ׳דִּבֵּר הַכָּתוּב בַּהֹוֶה׳: דֻּגְמָה מְצוּיָה שֶׁמִּמֶּנָּה לוֹמְדִים עִקָּרוֹן.
  • בַּגְּמָרָא: אָסוּר לְגַדֵּל כֶּלֶב רַע וּלְהַעֲמִיד סֻלָּם רָעוּעַ בַּבַּיִת – מִן הַפָּסוּק ׳וְלֹא תָשִׂים דָּמִים בְּבֵיתֶךָ׳. וְכֵן בְּרֵכָה, חֹמֶר רָעִיל, שֶׁקַע פָּתוּחַ, עָנָף רָעוּעַ וּבְלָמִים שְׁחוּקִים.
  • רַשִׁ״י עַל ׳כִּי יִפֹּל הַנֹּפֵל׳: רָאוּי זֶה לִפֹּל – וְאַף עַל פִּי כֵן לֹא תִתְגַּלְגֵּל מִיתָתוֹ עַל יָדְךָ, שֶׁמְּגַלְגְּלִין זְכוּת עַל יְדֵי זַכַּאי.
  • רַשִׁ״י עַל ׳כִּי תִבְנֶה׳ – סְמִיכוּת פָּרָשִׁיּוֹת וְהָעֵרֶךְ ׳מִצְוָה גּוֹרֶרֶת מִצְוָה׳: שִׁלּוּחַ הַקֵּן ← בַּיִת חָדָשׁ וּמַעֲקֶה ← כֶּרֶם, שָׂדֶה וּבְגָדִים נָאִים. וְהַכֹּל מֵאַהֲבַת ה׳ אֵלֵינוּ, שֶׁשּׁוֹמֵר עַל חַיֵּינוּ וּמְקָרֵב אוֹתָנוּ אֵלָיו 💛

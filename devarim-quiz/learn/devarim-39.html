בַּשִּׁעוּר הַזֶּה אֲנַחְנוּ פּוֹגְשִׁים שָׁלֹשׁ מִצְווֹת גְּדוֹלוֹת שֶׁכֻּלָּן שׁוֹמְרוֹת עַל הָאָדָם: עָרֵי מִקְלָט – שֶׁשּׁוֹמְרוֹת עַל הַחַיִּים; אִסּוּר הֲסָגַת גְּבוּל – שֶׁשּׁוֹמֵר עַל הַנַּחֲלָה; וְדִינֵי עֵדוּת – שֶׁשּׁוֹמְרִים עַל הָאֱמֶת בְּבֵית הַדִּין.
נִלְמַד מָה זֶה בִּכְלָל ׳מִקְלָט׳, אֵילוּ שֵׁשׁ עָרִים הָיוּ וְהֵיכָן, אֵיךְ שִׁלְּטוּ אֶת הַדְּרָכִים, מָה קוֹרֶה לְמִי שֶׁפָּגַע בְּלִי כַּוָּנָה, וּמָה הַדִּין שֶׁל ׳עֵדִים זוֹמְמִים׳.
וּמֵאֲחוֹרֵי כָּל זֶה מִתְגַּלָּה דָּבָר אֶחָד: כַּמָּה ה׳ אוֹהֵב כָּל אָדָם – אֶת חַיָּיו, אֶת הָאֲדָמָה שֶׁנָּתַן לוֹ, וְאֶת כְּבוֹדוֹ. גַּם לְמִי שֶׁנִּכְשַׁל בְּלִי כַּוָּנָה ה׳ מֵכִין מָקוֹם בָּטוּחַ, כִּי הוּא לֹא מְוַתֵּר עַל אַף אֶחָד מֵאִתָּנוּ.
בַּשִּׁעוּר: 🛡️ מָה זֶה מִקְלָט? – מָקוֹם שֶׁשּׁוֹמֵר 🗺️ שֵׁשׁ עָרֵי מִקְלָט – שָׁלֹשׁ וְשָׁלֹשׁ 💛 ׳מִצְוָה שֶׁאֶפְשָׁר לְקַיְּמָהּ – אֲקַיְּמֶנָּה׳ 🪧 ׳תָּכִין לְךָ הַדֶּרֶךְ׳ – שְׁלָטִים וּמֶרְחַקִּים 🪓 ׳בִּבְלִי דַעַת׳ – הַמִּקְרֶה שֶׁל הַגַּרְזֶן 📚 עִיר שֶׁל תּוֹרָה, תִּקּוּן וּתְשׁוּבָה ⚠️ שׁוֹגֵג וּמֵזִיד – וּ׳מִצְוָה קַלָּה׳ שֶׁנִּהְיֵית חֲמוּרָה 🧭 ׳וְיָסַפְתָּ לְךָ עוֹד שָׁלֹשׁ׳ – תֵּשַׁע עָרִים 🚧 ׳לֹא תַסִּיג גְּבוּל׳ – שְׁמִירָה עַל הַנַּחֲלָה 🔁 שְׁנֵי לָאוִין – וְדַוְקָא בְּאֶרֶץ יִשְׂרָאֵל 👥 עַל פִּי שְׁנֵי עֵדִים – וּמַהוּ ׳בִּנְיַן אָב׳ 🕵️ עֵדִים זוֹמְמִים – ׳כַּאֲשֶׁר זָמַם׳ 🍇 כֶּרֶם נָבוֹת – כְּשֶׁכָּל הַמִּצְווֹת נִשְׁבְּרוּ

🛡️מָה זֶה מִקְלָט? – מָקוֹם שֶׁשּׁוֹמֵר

נַתְחִיל מֵהַמִּלָּה עַצְמָהּ. ׳מִקְלָט׳ בָּא מִן הַשֹּׁרֶשׁ ׳לִקְלֹט׳ – לִתְפֹּס וְלִשְׁמֹר. אוֹמְרִים ׳קָלַטְתִּי אֶת הַכַּדּוּר׳, וְאוֹמְרִים גַּם ׳אַתָּה קוֹלֵט?׳ – כְּלוֹמַר, תָּפַסְתָּ אֶת הָעִנְיָן.
וּכְשֶׁאוֹמְרִים ׳לָרֶדֶת לַמִּקְלָט׳ – הַכַּוָּנָה לְמָקוֹם בָּטוּחַ: מָקוֹם מוּגָן, שֶׁהַסַּכָּנָה שֶׁבַּחוּץ אֵינָהּ יְכוֹלָה לִפְגֹּעַ בְּמִי שֶׁנִּמְצָא בִּפְנִים.

וּמָה זֹאת עִיר מִקְלָט? עִיר שֶׁקּוֹלֶטֶת וְשׁוֹמֶרֶת עַל אָדָם מְסֻיָּם. עַל מִי? עַל מִי שֶׁגָּרַם לְמוֹתוֹ שֶׁל אָדָם בִּשְׁגָגָה – בְּלִי כַּוָּנָה כְּלָל.
בְּנֵי הַמִּשְׁפָּחָה שֶׁל הַנֶּהֱרָג כּוֹעֲסִים מְאוֹד וְרוֹצִים לִנְקֹם, וְהַתּוֹרָה קוֹרֵאת לָהֶם ׳גּוֹאֵל הַדָּם׳. מֵהָרֶגַע שֶׁהַבּוֹרֵחַ נִכְנָס אֶל הָעִיר – הָעִיר קוֹלֶטֶת אוֹתוֹ, וְאָסוּר לִפְגֹּעַ בּוֹ.

אֶת דִּינֵי עָרֵי הַמִּקְלָט לָמַדְנוּ כְּבָר בְּסֵפֶר בְּמִדְבַּר, בַּפְּרָשׁוֹת הָאַחֲרוֹנוֹת שֶׁל הַחֻמָּשׁ. וְכָאן, בְּחֻמַּשׁ דְּבָרִים, מֹשֶׁה חוֹזֵר עֲלֵיהֶם – כִּי דְּבָרִים הוּא ׳מִשְׁנֵה תּוֹרָה׳, הַחֻמָּשׁ שֶׁבּוֹ מֹשֶׁה שׁוֹנֶה וְחוֹזֵר עַל מָה שֶׁכְּבָר נֶאֱמַר.

💡 לֹא ׳עֹנֶשׁ׳ וְלֹא ׳פְּרָס׳ – מָקוֹם בָּטוּחַ
שִׂימוּ לֵב: עִיר הַמִּקְלָט אֵינָהּ בֵּית סֹהַר, וְגַם אֵינָהּ פְּרָס.
הִיא מָקוֹם שֶׁמֵּגֵן עַל מִי שֶׁנִּכְשַׁל בְּלִי כַּוָּנָה, עַד שֶׁבֵּית הַדִּין יִבְדֹּק אֶת הַמִּקְרֶה וְיִפְסֹק אֶת דִּינוֹ.

וְכָאן כְּבָר רוֹאִים אֶת אַהֲבַת ה׳: גַּם אָדָם שֶׁקָּרָה עַל יָדוֹ אָסוֹן נוֹרָא, וְכָל הָעוֹלָם כּוֹעֵס עָלָיו – ה׳ אֵינוֹ מַפְקִיר אוֹתוֹ. הוּא מֵכִין לוֹ עִיר שְׁלֵמָה, מֵרֹאשׁ, כְּדֵי שֶׁיִּהְיֶה לוֹ לְאָן לָרוּץ.

בְּסוֹף הַנְּאוּם הַגָּדוֹל שֶׁבְּפֶרֶק ד׳, אַחֲרֵי שֶׁמֹּשֶׁה מְצַוֶּה ׳וְשָׁמַרְתָּ אֶת חֻקָּיו וְאֶת מִצְוֺתָיו׳, כָּתוּב: ׳אָז יַבְדִּיל מֹשֶׁה שָׁלֹשׁ עָרִים בְּעֵבֶר הַיַּרְדֵּן מִזְרְחָה שָׁמֶשׁ׳.
וְהַתּוֹרָה מְפָרֶטֶת אוֹתָן: ׳אֶת בֶּצֶר בַּמִּדְבָּר בְּאֶרֶץ הַמִּישֹׁר לָרֻאוּבֵנִי, וְאֶת רָאמֹת בַּגִּלְעָד לַגָּדִי, וְאֶת גּוֹלָן בַּבָּשָׁן לַמְנַשִּׁי׳. שְׁלוֹשָׁה שְׁבָטִים קִבְּלוּ נַחֲלָה בְּעֵבֶר הַיַּרְדֵּן הַמִּזְרָחִי, וּבְכָל נַחֲלָה – עִיר מִקְלָט אַחַת.

לְנָבוֹת הַיִּזְרְעֵאלִי הָיָה כֶּרֶם – נַחֲלַת אֲבוֹתָיו – סָמוּךְ לְאַרְמוֹנוֹ שֶׁל אַחְאָב מֶלֶךְ יִשְׂרָאֵל.
אַחְאָב חָמַד אֶת הַכֶּרֶם וּבִקֵּשׁ לִקְנוֹת אוֹתוֹ. נָבוֹת סֵרֵב, וְאָמַר: ׳חָלִילָה לִּי מֵה׳ מִתִּתִּי אֶת נַחֲלַת אֲבֹתַי לָךְ׳.
אִיזֶבֶל אִשְׁתּוֹ שָׂכְרָה עֵדֵי שֶׁקֶר שֶׁהֵעִידוּ נֶגֶד נָבוֹת. עַל פִּי עֵדוּתָם נִגְזַר עָלָיו דִּין קָשֶׁה, וְאַחַר כָּךְ לָקַח אַחְאָב אֶת הַכֶּרֶם.

בּוֹאוּ נִסְפֹּר: נַחֲלָה שֶׁנִּגְזְלָה – כְּמוֹ ׳לֹא תַסִּיג גְּבוּל׳. עֵדוּת שֶׁקֶר – כְּמוֹ עֵדִים זוֹמְמִים. וּשְׁפִיכוּת דָּמִים – הַנּוֹשֵׂא שֶׁל תְּחִלַּת הַפֶּרֶק שֶׁלָּנוּ.
הַכֹּל הִתְחִיל בְּחֶמְדָּה אַחַת קְטַנָּה שֶׁל מֶלֶךְ, וְנִגְמַר בְּאָסוֹן.

וְכָךְ מִתְקַיֵּם מָה שֶׁרַשִׁ״י לִמֵּד אוֹתָנוּ: ׳עָבַר אָדָם עַל מִצְוָה קַלָּה – סוֹפוֹ לַעֲבֹר עַל מִצְוָה חֲמוּרָה׳. וְכָךְ גַּם דִּבְרֵי אַבְן עֶזְרָא: הֲסָגַת גְּבוּל מְבִיאָה לִידֵי רִיב, מַכּוֹת וּרְצִיחָה.
אִלּוּ הִקְפִּידוּ אַחְאָב וְאִיזֶבֶל עַל הַמִּצְווֹת הָאֵלֶּה – לֹא הָיָה קוֹרֶה הַמַּעֲשֶׂה הַנּוֹרָא הַזֶּה כְּלָל.

🤔 מָה אֶפְשָׁר לִלְמֹד מִנָּבוֹת?
נָבוֹת לֹא אָמַר ׳אֲנִי לֹא רוֹצֶה׳ אֶלָּא ׳חָלִילָה לִּי מֵה׳׳.
בִּשְׁבִילוֹ נַחֲלַת אֲבוֹתָיו הָיְתָה מַתָּנָה מֵה׳, וְלֹא סְחוֹרָה לִמְכִירָה. גַּם מוּל מֶלֶךְ – הוּא נִשְׁאַר נֶאֱמָן לַה׳ וְלָאָרֶץ.
💛 אַהֲבַת הַשֵּׁם
נָבוֹת אָהַב אֶת חֶלְקוֹ בְּאֶרֶץ יִשְׂרָאֵל כְּמוֹ מַתָּנָה מֵאַבָּא. גַּם אֲנִי רוֹצֶה לֶאֱהֹב כָּךְ אֶת הָאָרֶץ שֶׁה׳ נָתַן לָנוּ, וְלִשְׁמֹר עָלֶיהָ בְּאַהֲבָה וּבֶאֱמֶת 💛

✨ לְסִכּוּם

  • ׳מִקְלָט׳ מִלְּשׁוֹן ׳לִקְלֹט׳ – מָקוֹם בָּטוּחַ שֶׁקּוֹלֵט וְשׁוֹמֵר עַל מִי שֶׁפָּגַע בִּשְׁגָגָה, מִפְּנֵי ׳גּוֹאֵל הַדָּם׳. הַדִּינִים נִלְמְדוּ כְּבָר בְּסֵפֶר בְּמִדְבַּר, וּבְדְבָרִים מֹשֶׁה חוֹזֵר עֲלֵיהֶם.
  • שֵׁשׁ עָרִים: בֶּצֶר, רָאמֹת וְגוֹלָן בְּעֵבֶר הַיַּרְדֵּן הַמִּזְרָחִי; קֶדֶשׁ, שְׁכֶם וְחֶבְרוֹן בְּאֶרֶץ כְּנַעַן. וְלֶעָתִיד לָבֹא – עוֹד שָׁלֹשׁ, הֲרֵי תֵּשַׁע.
  • מֹשֶׁה הִבְדִּיל אֶת הֶעָרִים שֶׁבְּעֵבֶר הַיַּרְדֵּן אַף שֶׁעוֹד לֹא קָלְטוּ: ׳מִצְוָה שֶׁאֶפְשָׁר לְקַיְּמָהּ – אֲקַיְּמֶנָּה׳.
  • ׳תָּכִין לְךָ הַדֶּרֶךְ׳ – שְׁלָטִים ׳מִקְלָט מִקְלָט׳ עַל פָּרָשַׁת דְּרָכִים, וְהָאָרֶץ מְחֻלֶּקֶת לִשְׁלוֹשָׁה חֲלָקִים שָׁוִים, כְּדֵי שֶׁהַדֶּרֶךְ תִּהְיֶה קְצָרָה.
  • בּוֹרֵחַ רַק מִי שֶׁפָּגַע בִּבְלִי דַעַת וּבְלִי שִׂנְאָה – כְּמוֹ הַבַּרְזֶל שֶׁנָּשַׁל מִן הַגַּרְזֶן. הַמֵּזִיד – זִקְנֵי עִירוֹ לוֹקְחִים אוֹתוֹ מִשָּׁם, כִּי ׳עָבַר אָדָם עַל מִצְוָה קַלָּה – סוֹפוֹ לַעֲבֹר עַל מִצְוָה חֲמוּרָה׳.
  • ׳לֹא תַסִּיג גְּבוּל׳ – לְהָזִיז אֶת סִימַן הַגְּבוּל לְתוֹךְ שְׂדֵה חֲבֵרוֹ. בְּאֶרֶץ יִשְׂרָאֵל עוֹבֵר בִּשְׁנֵי לָאוִין (גַּם ׳לֹא תִגְזֹל׳), כִּי הַנַּחֲלָה הִיא מַתְּנַת ה׳.
  • עַל פִּי שְׁנֵי עֵדִים יָקוּם דָּבָר; ׳עֵד אֶחָד׳ הוּא בִּנְיַן אָב – כָּל ׳עֵד׳ שֶׁבַּתּוֹרָה שְׁנַיִם. וְעֵדִים זוֹמְמִים, שֶׁהוּכַח שֶׁלֹּא הָיוּ בַּמָּקוֹם – ׳וַעֲשִׂיתֶם לוֹ כַּאֲשֶׁר זָמַם׳.
  • בְּכֶרֶם נָבוֹת נִפְגְּשׁוּ כָּל הָאִסּוּרִים: נַחֲלָה שֶׁנִּגְזְלָה, עֵדוּת שֶׁקֶר וּשְׁפִיכוּת דָּמִים. ה׳ נָתַן אֶת הַמִּצְווֹת הָאֵלֶּה מֵאַהֲבָה – לִשְׁמֹר עַל הַחַיִּים, עַל הַנַּחֲלָה וְעַל הָאֱמֶת.

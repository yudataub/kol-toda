עַם קָדוֹשׁ

הַחֹמֶר הַמָּלֵא: ׳בָּנִים אַתֶּם לַה׳ אֱלֹהֵיכֶם׳ – בְּנֵי מֶלֶךְ שׁוֹמְרִים עַל הַגּוּף וְעַל הַכָּבוֹד (פָּרָשַׁת רְאֵה, פֶּרֶק י״ד)
🎯 לַחִידוֹן🏠 כָּל הַחִידוֹנִים
שְׁנֵי פְּסוּקִים קְצָרִים, וּבָהֶם אַחַד הַמִּשְׁפָּטִים הַמְּרַגְּשִׁים בְּכָל הַתּוֹרָה: ׳בָּנִים אַתֶּם לַה׳ אֱלֹהֵיכֶם׳.
כָּל יְהוּדִי הוּא בֵּן שֶׁל מֶלֶךְ מַלְכֵי הַמְּלָכִים – נָסִיךְ אֲמִתִּי. וּמִתּוֹךְ זֶה בָּאָה מִצְוָה: לֹא לִפְגֹּעַ בַּגּוּף, לֹא לִשְׂרֹט וְלֹא לַעֲשׂוֹת קָרְחָה עַל הַמֵּת, כְּדֶרֶךְ שֶׁעָשׂוּ הָעַמִּים שֶׁבְּאֶרֶץ כְּנַעַן.
נִלְמַד מָה מְבָאֵר רַשִׁ״י, מָה פֵּרוּשׁ ׳קָדוֹשׁ׳ וּמָה פֵּרוּשׁ ׳סְגֻלָּה׳, וְנָבִין שֶׁהַמִּצְוָה הַזּוֹ כֻּלָּהּ אַהֲבָה: ה׳ אוֹהֵב אוֹתָנוּ כְּמוֹ אַבָּא, וְלָכֵן הוּא שׁוֹמֵר עָלֵינוּ שֶׁנִּהְיֶה נָאִים, שְׁלֵמִים וּמְכֻבָּדִים 💛
בַּשִּׁעוּר: 💂 מִשְׁמַר הַמַּלְכָּה – הַבְּגָדִים מְסַפְּרִים מִי אֲנַחְנוּ 🧩 פְּתִיחָה, צִוּוּי וְנִימּוּק – מִבְנֶה הַפְּסוּקִים 👑 בָּנִים אַתֶּם לַה׳ – נְסִיכִים אֲמִתִּיִּים 🚫 לֹא תִתְגֹּדְדוּ וְלֹא תָשִׂימוּ קָרְחָה – מָה אָסוּר? ✨ בָּנָיו שֶׁל מָקוֹם – ׳וְאַתֶּם רְאוּיִין לִהְיוֹת נָאִים׳ 🕍 מָה זֶה ׳קָדוֹשׁ׳? – מָקוֹם, זְמַן, חֵפֶץ... וְעַם 💎 עַם סְגֻלָּה – וּקְדֻשָּׁה מִשְּׁנֵי מְקוֹרוֹת 🌧️ גַּם בְּצַעַר – יֵשׁ לִי אַבָּא 📜 שְׁנֵי הַפְּסוּקִים בְּעַל פֶּה

💂מִשְׁמַר הַמַּלְכָּה – הַבְּגָדִים מְסַפְּרִים מִי אֲנַחְנוּ

נַתְחִיל דַּוְקָא בְּלוֹנְדוֹן. לְיַד אַרְמוֹן בָּקִינְגְהָם עוֹמְדִים חַיָּלֵי הַמִּשְׁמָר הַמְּפֻרְסָמִים: מַדִּים מְיֻחָדִים, עֲמִידָה זְקוּפָה, וּפָנִים שֶׁאֵינָן זָזוֹת. תַּיָּרִים מְנַסִּים לְהַצְחִיק אוֹתָם – וְהֵם אֵינָם מְגִיבִים.
פַּעַם בְּיוֹם, בְּשָׁעָה קְבוּעָה, מִתְחַלֵּף הַמִּשְׁמָר בְּטֶקֶס מַרְשִׁים. כָּל מִי שֶׁרוֹאֶה אוֹתָם מֵבִין מִיָּד: הַחַיָּלִים הָאֵלֶּה שַׁיָּכִים לַמֶּלֶךְ.

🤔 לַמּוֹרֶה: שֵׁשׁ תְּמוּנוֹת
אֶפְשָׁר לְהַרְאוֹת לַתַּלְמִידִים שֵׁשׁ תְּמוּנוֹת שֶׁל אֲנָשִׁים לְפִי לְבוּשָׁם – חֲתֻנָּה, אַנְשֵׁי עֲסָקִים, בַּחוּרֵי יְשִׁיבָה, שׁוֹטְרִים, עוֹרְכֵי דִּין וְחַיָּלִים – וְלִשְׁאֹל: מִנַּיִן יָדַעְתֶּם מִי כָּל אֶחָד?
הַתְּשׁוּבָה: לֹא סִפְּרוּ לָנוּ – הַבְּגָדִים סִפְּרוּ. הַלְּבוּשׁ וְהַהוֹפָעָה אוֹמְרִים מַשֶּׁהוּ עַל מִי שֶׁעוֹמֵד מֵאֲחוֹרֵיהֶם.

וְזֶה בְּדִיּוּק הָעִנְיָן שֶׁל שְׁנֵי הַפְּסוּקִים שֶׁלָּנוּ. אִם הַהוֹפָעָה שֶׁל חַיָּל מְסַפֶּרֶת שֶׁהוּא שֶׁל הַמֶּלֶךְ – מָה מְסַפֶּרֶת הַהוֹפָעָה שֶׁל יְהוּדִי?
יְהוּדִי מְיַצֵּג אֶת הַקָּדוֹשׁ בָּרוּךְ הוּא. לָכֵן יֵשׁ מַשְׁמָעוּת לַדֶּרֶךְ שֶׁבָּהּ הוּא נִרְאֶה, מְדַבֵּר וּמִתְנַהֵג.

💛 אַהֲבַת הַשֵּׁם
ה׳ לֹא הִסְתִּיר אוֹתָנוּ בְּצַד. הוּא בָּחַר שֶׁדַּוְקָא אֲנַחְנוּ נְסַפֵּר לְכָל הָעוֹלָם מִיהוּ הַמֶּלֶךְ. כְּשֶׁאֲנִי נִרְאֶה יָפֶה וּמְכֻבָּד, אֲנִי מַרְגִּישׁ שֶׁאֲנִי שֶׁל ה׳ 💛

🧩פְּתִיחָה, צִוּוּי וְנִימּוּק – מִבְנֶה הַפְּסוּקִים

📖 פֶּרֶק י״ד, פְּסוּקִים א׳–ב׳ ←
א בָּנִים אַתֶּם לַה׳ אֱלֹהֵיכֶם לֹא תִתְגֹּדְדוּ וְלֹא תָשִׂימוּ קָרְחָה בֵּין עֵינֵיכֶם לָמֵת ב כִּי עַם קָדוֹשׁ אַתָּה לַה׳ אֱלֹהֶיךָ וּבְךָ בָּחַר ה׳ לִהְיוֹת לוֹ לְעַם סְגֻלָּה מִכֹּל הָעַמִּים אֲשֶׁר עַל פְּנֵי הָאֲדָמָה

לִפְנֵי שֶׁמְּבִינִים אֶת הַמִּלִּים, אֶפְשָׁר כְּבָר לִרְאוֹת אֶת הַמִּבְנֶה. שְׁנֵי הַפְּסוּקִים מִתְחַלְּקִים בְּבֵרוּר לִשְׁלוֹשָׁה חֲלָקִים:
הַפְּתִיחָה: ׳בָּנִים אַתֶּם לַה׳ אֱלֹהֵיכֶם׳.
הַצִּוּוּי (וְיֵשׁ בּוֹ שְׁנַיִם): ׳לֹא תִתְגֹּדְדוּ וְלֹא תָשִׂימוּ קָרְחָה בֵּין עֵינֵיכֶם לָמֵת׳.
הַנִּימּוּק – הַסִּבָּה: ׳כִּי עַם קָדוֹשׁ אַתָּה לַה׳ אֱלֹהֶיךָ וּבְךָ בָּחַר ה׳ לִהְיוֹת לוֹ לְעַם סְגֻלָּה מִכֹּל הָעַמִּים אֲשֶׁר עַל פְּנֵי הָאֲדָמָה׳.

בִּפְתִיחָה שֶׁל פָּרָשָׁה יֵשׁ בְּדֶרֶךְ כְּלָל מֶסֶר חָשׁוּב – אֲמִירָה שֶׁפּוֹתַחַת אֶת הַלֵּב לִפְנֵי הַמִּצְוָה. וְכָאן הַפְּתִיחָה הִיא הַמִּשְׁפָּט הַיָּפֶה: אַתֶּם בָּנִים שֶׁל ה׳.
וְהַנִּימּוּק בַּסּוֹף מַסְבִּיר לָמָּה – כִּי אַתָּה עַם קָדוֹשׁ, וּבְךָ בָּחַר ה׳.

🤔 פְּעִילוּת בַּכִּתָּה
לִקְרֹא אֶת שְׁנֵי הַפְּסוּקִים וּלְסַמֵּן בִּשְׁלוֹשָׁה צְבָעִים: פְּתִיחָה, צִוּוּי, נִימּוּק.
שְׁאֵלָה לְדִיּוּן: מָה הָיָה חָסֵר לָנוּ אִם הַתּוֹרָה הָיְתָה כּוֹתֶבֶת רַק אֶת הַצִּוּוּי, בְּלִי הַפְּתִיחָה וּבְלִי הַנִּימּוּק?
💛 אַהֲבַת הַשֵּׁם
ה׳ לֹא רַק מְצַוֶּה – הוּא גַּם מַסְבִּיר, וְגַם מַזְכִּיר לִי כַּמָּה אֲנִי יָקָר לוֹ. אַבָּא שֶׁאוֹהֵב מְדַבֵּר כָּךְ עִם הַיְּלָדִים שֶׁלּוֹ 💛

👑בָּנִים אַתֶּם לַה׳ – נְסִיכִים אֲמִתִּיִּים

📖 פֶּרֶק י״ד, פָּסוּק א׳ ←
א בָּנִים אַתֶּם לַה׳ אֱלֹהֵיכֶם לֹא תִתְגֹּדְדוּ וְלֹא תָשִׂימוּ קָרְחָה בֵּין עֵינֵיכֶם לָמֵת

הַפָּסוּק פּוֹתֵחַ בְּמִלִּים שֶׁכְּדַאי לִזְכֹּר לְכָל הַחַיִּים: ׳בָּנִים אַתֶּם לַה׳ אֱלֹהֵיכֶם׳.
לֹא ׳עֲבָדִים אַתֶּם׳ וְלֹא ׳חַיָּלִים אַתֶּם׳ – בָּנִים. וּמִי הוּא הָאַבָּא? מֶלֶךְ מַלְכֵי הַמְּלָכִים. וּמִי שֶׁהוּא בֵּן שֶׁל מֶלֶךְ – הוּא נָסִיךְ אֲמִתִּי.

אִם קָרְאוּ לָכֶם פַּעַם הַהוֹרִים אוֹ הַחֲבֵרִים ׳נָסִיךְ׳ אוֹ ׳נְסִיכָה׳ – דְּעוּ שֶׁזֶּה בֶּאֱמֶת נָכוֹן, וְכָתוּב בַּתּוֹרָה.
אֲבָל לִהְיוֹת בֶּן מֶלֶךְ זוֹ גַּם אַחְרָיוּת: נָסִיךְ יוֹדֵעַ מִי הַהוֹרִים שֶׁלּוֹ, וְהוּא מְיַצֵּג אוֹתָם וְאֶת כָּל הַמִּשְׁפָּחָה בְּכָל מָקוֹם שֶׁהוּא הוֹלֵךְ.

🤔 מַד הַנָּסִיךְ
לִפְנֵי שֶׁבּוֹחֲרִים אֵיךְ לְהִתְלַבֵּשׁ, אֵיךְ לְדַבֵּר וְאֵיךְ לְהִתְנַהֵג, אֶפְשָׁר לִשְׁאֹל שְׁאֵלָה אַחַת קְטַנָּה:
הַאִם זֶה מַתְאִים לְבֵן שֶׁל מֶלֶךְ?
אֶפְשָׁר לְדַמְיֵן תְּמוּנוֹת וּלְמַיֵּן: זֶה מַתְאִים, זֶה פָּחוֹת, וְזֶה בִּכְלָל לֹא. אֶת הַדִּיּוּן הַזֶּה כְּדַאי לַעֲשׂוֹת יַחַד עִם הַמּוֹרוֹת וְהַמּוֹרִים.
💛 אַהֲבַת הַשֵּׁם
אֲנִי לֹא סְתָם יֶלֶד בָּעוֹלָם: אֲנִי בֵּן שֶׁל ה׳. בְּכָל בֹּקֶר כְּשֶׁאֲנִי קָם, אֲנִי יָכוֹל לְהַזְכִּיר לְעַצְמִי שֶׁמֶּלֶךְ הָעוֹלָם קוֹרֵא לִי ׳בְּנִי׳ 💛

🚫לֹא תִתְגֹּדְדוּ וְלֹא תָשִׂימוּ קָרְחָה – מָה אָסוּר?

📖 פֶּרֶק י״ד, פָּסוּק א׳ ←
א בָּנִים אַתֶּם לַה׳ אֱלֹהֵיכֶם לֹא תִתְגֹּדְדוּ וְלֹא תָשִׂימוּ קָרְחָה בֵּין עֵינֵיכֶם לָמֵת

הָעַמִּים שֶׁגָּרוּ בְּאֶרֶץ כְּנַעַן, הָאֱמוֹרִיִּים, נָהֲגוּ מִנְהָג קָשֶׁה: כְּשֶׁמֵּת לָהֶם מֵת, הֵם הָיוּ שׂוֹרְטִים וּפוֹצְעִים אֶת הַגּוּף בְּכֵלִים, וְגַם תּוֹלְשִׁים שְׂעָרוֹת מִן הָרֹאשׁ, כְּדֵי לְהַרְאוֹת אֶת הָאֵבֶל שֶׁלָּהֶם.
וְהַתּוֹרָה אוֹמֶרֶת לָנוּ: אֲנַחְנוּ לֹא נוֹהֲגִים כָּךְ.

📜 רַשִׁ״י
׳לֹא תִתְגֹּדְדוּ׳ – לֹא תִּתְּנוּ גְּדִידָה וְשֶׂרֶט בִּבְשַׂרְכֶם עַל מֵת, כְּדֶרֶךְ שֶׁהָאֱמוֹרִיִּים עוֹשִׂין.

׳גְּדִידָה׳ וְ׳שֶׂרֶט׳ הֵן פְּצִיעָה וּשְׂרִיטָה בַּבָּשָׂר. הַצִּוּוּי הָרִאשׁוֹן: אָסוּר לִפְגֹּעַ בַּגּוּף מִתּוֹךְ צַעַר עַל הַמֵּת.

📜 רַשִׁ״י
׳לֹא תִתְגֹּדְדוּ׳ – ...לְפִי שֶׁאַתֶּם בָּנָיו שֶׁל מָקוֹם וְאַתֶּם רְאוּיִין לִהְיוֹת נָאִים, וְלֹא גְּדוּדִים וּמְקֹרָחִים.

׳מָקוֹם׳ הוּא כִּנּוּי לַה׳ (כִּי הוּא מְקוֹמוֹ שֶׁל עוֹלָם). וְ׳נָאִים׳ – יָפִים וּמְכֻבָּדִים.
כְּלוֹמַר: בֵּן שֶׁל מֶלֶךְ רָאוּי לְהֵרָאוֹת יָפֶה וּמְכֻבָּד, וְלֹא פָּצוּעַ, שָׂרוּט וּמְקֹרָח בְּכָל מִינֵי צוּרוֹת מְשֻׁנּוֹת.

וּמִכָּאן לוֹמְדִים כְּלָל גָּדוֹל: יֵשׁ עֵרֶךְ לַכָּבוֹד שֶׁאָדָם נוֹתֵן לְגוּפוֹ וּלְהוֹפָעָתוֹ.

💡 הַנְּקֻדָּה שֶׁל הַתָּכְנִית

זוֹ הַתְּשׁוּבָה לַשְּׁאֵלָה ׳מָה מַסְבִּיר רַשִׁ״י עַל ׳לֹא תִתְגֹּדְדוּ׳׳.
💛 אַהֲבַת הַשֵּׁם

🕍מָה זֶה ׳קָדוֹשׁ׳? – מָקוֹם, זְמַן, חֵפֶץ... וְעַם

ב כִּי עַם קָדוֹשׁ אַתָּה לַה׳ אֱלֹהֶיךָ וּבְךָ בָּחַר ה׳ לִהְיוֹת לוֹ לְעַם סְגֻלָּה מִכֹּל הָעַמִּים אֲשֶׁר עַל פְּנֵי הָאֲדָמָה

נַעֲצֹר רֶגַע עַל הַמִּלָּה שֶׁבַּנִּימּוּק: ׳קָדוֹשׁ׳. מָה עוֹלֶה לָכֶם בָּרֹאשׁ כְּשֶׁשּׁוֹמְעִים אוֹתָהּ?
אוּלַי מְקוֹמוֹת קְדוֹשִׁים – בֵּית הַמִּקְדָּשׁ, הַכֹּתֶל הַמַּעֲרָבִי, בֵּית הַכְּנֶסֶת.
אוּלַי תַּשְׁמִישֵׁי קְדֻשָּׁה – תְּפִלִּין, טַלִּית, סֵפֶר תּוֹרָה.
אוּלַי זְמַן קָדוֹשׁ – הַשַּׁבָּת וְנֵרוֹת הַשַּׁבָּת.

וּמָה הַמְּשֻׁתָּף לְכֻלָּם? ׳קָדוֹשׁ׳ פֵּרוּשׁוֹ מְיֻחָד וְנִבְדָּל – דָּבָר שֶׁיֵּשׁ לוֹ הִתְנַהֲגוּת מְיֻחֶדֶת מִשֶּׁלּוֹ.
בְּבֵית הַכְּנֶסֶת לֹא מִתְנַהֲגִים כְּמוֹ בְּמָקוֹם רָגִיל. בְּשַׁבָּת מְיַחֲדִים אֶת הַיּוֹם בְּמַאֲכָל, בִּלְבוּשׁ וּבְהִתְנַהֲגוּת.

וְעַכְשָׁיו הַחִדּוּשׁ הַגָּדוֹל: גַּם עַם יָכוֹל לִהְיוֹת קָדוֹשׁ.

💎עַם סְגֻלָּה – וּקְדֻשָּׁה מִשְּׁנֵי מְקוֹרוֹת

׳וּבְךָ בָּחַר ה׳ לִהְיוֹת לוֹ לְעַם סְגֻלָּה מִכֹּל הָעַמִּים אֲשֶׁר עַל פְּנֵי הָאֲדָמָה׳.
סְגֻלָּה הִיא אוֹצָר יָקָר וְאָהוּב, שֶׁשּׁוֹמְרִים עָלָיו בְּמָקוֹם מְיֻחָד. כְּמוֹ תַּכְשִׁיט שֶׁעוֹבֵר בַּמִּשְׁפָּחָה דּוֹרוֹת רַבִּים.

💡 שְׁתֵּי דַּרְכֵי אֵבֶל
הָאֱמוֹרִיִּים הֶרְאוּ אֶת הַצַּעַר שֶׁלָּהֶם בִּפְצִיעָה וּבְקָרְחָה – כִּי הֵם לֹא הֶאֱמִינוּ שֶׁיֵּשׁ מִי שֶׁשּׁוֹמֵעַ אוֹתָם.
עַם יִשְׂרָאֵל מַרְאֶה אֶת הַצַּעַר בִּתְפִלָּה, בְּדִמְעָה וּבְזִכָּרוֹן טוֹב – כִּי אֲנַחְנוּ יוֹדְעִים שֶׁה׳ שׁוֹמֵעַ.
💛 אַהֲבַת הַשֵּׁם
גַּם כְּשֶׁקָּשֶׁה לִי וַאֲנִי בּוֹכֶה, אֲנִי יָכוֹל לְהַגִּיד בַּלֵּב: ׳אַבָּא, אֲנִי הַבֵּן שֶׁלְּךָ׳. ה׳ לֹא עוֹזֵב אוֹתִי אַף פַּעַם 💛

📜שְׁנֵי הַפְּסוּקִים בְּעַל פֶּה

📖 פֶּרֶק י״ד, פְּסוּקִים א׳–ב׳ ←
א בָּנִים אַתֶּם לַה׳ אֱלֹהֵיכֶם לֹא תִתְגֹּדְדוּ וְלֹא תָשִׂימוּ קָרְחָה בֵּין עֵינֵיכֶם לָמֵת ב כִּי עַם קָדוֹשׁ אַתָּה לַה׳ אֱלֹהֶיךָ וּבְךָ בָּחַר ה׳ לִהְיוֹת לוֹ לְעַם סְגֻלָּה מִכֹּל הָעַמִּים אֲשֶׁר עַל פְּנֵי הָאֲדָמָה

אֶת שְׁנֵי הַפְּסוּקִים הָאֵלֶּה לוֹמְדִים בְּעַל פֶּה. הֵם קְצָרִים, וְהֵם נוֹתְנִים כֹּחַ לְכָל הַחַיִּים:
׳בָּנִים אַתֶּם לַה׳ אֱלֹהֵיכֶם, לֹא תִתְגֹּדְדוּ וְלֹא תָשִׂימוּ קָרְחָה בֵּין עֵינֵיכֶם לָמֵת.
כִּי עַם קָדוֹשׁ אַתָּה לַה׳ אֱלֹהֶיךָ, וּבְךָ בָּחַר ה׳ לִהְיוֹת לוֹ לְעַם סְגֻלָּה מִכֹּל הָעַמִּים אֲשֶׁר עַל פְּנֵי הָאֲדָמָה׳.

💡 עֵצָה לְשִׁנּוּן
מְחַלְּקִים לִשְׁלוֹשָׁה: בָּנִים אַתֶּם (הַפְּתִיחָה), לֹא תִתְגֹּדְדוּ (הַצִּוּוּי), כִּי עַם קָדוֹשׁ (הַנִּימּוּק).
מִי שֶׁזּוֹכֵר אֶת הַמִּבְנֶה – זוֹכֵר אֶת הַפְּסוּקִים.
💡 רַשִׁ״י בִּקְצָרָה
׳לֹא תִתְגֹּדְדוּ׳ – לֹא תִּתְּנוּ גְּדִידָה וְשֶׂרֶט בִּבְשַׂרְכֶם עַל מֵת, כְּדֶרֶךְ הָאֱמוֹרִיִּים.
׳בֵּין עֵינֵיכֶם׳ – אֵצֶל הַפַּדַּחַת.
הַסִּבָּה – אַתֶּם בָּנָיו שֶׁל מָקוֹם, וּרְאוּיִין לִהְיוֹת נָאִים.
׳כִּי עַם קָדוֹשׁ אַתָּה׳ – קְדֻשַּׁת עַצְמְךָ מֵאֲבוֹתֶיךָ, וְעוֹד – וּבְךָ בָּחַר ה׳.
💛 אַהֲבַת הַשֵּׁם
כְּשֶׁאֲנִי אוֹמֵר אֶת הַפְּסוּקִים הָאֵלֶּה בְּעַל פֶּה, הַמִּלִּים שֶׁל ה׳ נִכְנָסוֹת לְתוֹךְ הַלֵּב שֶׁלִּי וְנִשְׁאָרוֹת שָׁם לָתָמִיד 💛

✨ לְסִכּוּם

  • ׳בָּנִים אַתֶּם לַה׳ אֱלֹהֵיכֶם׳ – כָּל יְהוּדִי הוּא בֵּן שֶׁל מֶלֶךְ מַלְכֵי הַמְּלָכִים, נָסִיךְ אֲמִתִּי. וְזוֹ גַּם אַחְרָיוּת: נָסִיךְ מְיַצֵּג אֶת הַמִּשְׁפָּחָה שֶׁלּוֹ.
  • שְׁנֵי הַפְּסוּקִים מִתְחַלְּקִים לִשְׁלוֹשָׁה: פְּתִיחָה (בָּנִים אַתֶּם), צִוּוּי (לֹא תִתְגֹּדְדוּ וְלֹא תָשִׂימוּ קָרְחָה), וְנִימּוּק (כִּי עַם קָדוֹשׁ אַתָּה).
  • רַשִׁ״י: ׳לֹא תִתְגֹּדְדוּ׳ – לֹא תִּתְּנוּ גְּדִידָה וְשֶׂרֶט בַּבָּשָׂר עַל מֵת, כְּדֶרֶךְ הָאֱמוֹרִיִּים; ׳בֵּין עֵינֵיכֶם׳ – אֵצֶל הַפַּדַּחַת.
  • הַטַּעַם: ׳לְפִי שֶׁאַתֶּם בָּנָיו שֶׁל מָקוֹם וְאַתֶּם רְאוּיִין לִהְיוֹת נָאִים׳ – וּמִכָּאן שֶׁיֵּשׁ עֵרֶךְ לַכָּבוֹד שֶׁאָדָם נוֹתֵן לְגוּפוֹ וּלְהוֹפָעָתוֹ.
  • ׳קָדוֹשׁ׳ – מְיֻחָד וְנִבְדָּל: יֵשׁ מָקוֹם קָדוֹשׁ, זְמַן קָדוֹשׁ, חֲפָצִים קְדוֹשִׁים – וְגַם עַם קָדוֹשׁ, שֶׁיֵּשׁ לוֹ חֻקִּים מִשֶּׁלּוֹ, שׁוֹנִים מֵחֻקּוֹת הַגּוֹיִם.
  • ׳עַם סְגֻלָּה׳ – אוֹצָר יָקָר וְאָהוּב. וּלְפִי רַשִׁ״י הַקְּדֻשָּׁה בָּאָה מִשְּׁנֵי מְקוֹרוֹת: בִּירֻשָּׁה מֵאֲבוֹתֵינוּ, וְעוֹד – שֶׁה׳ עַצְמוֹ בָּחַר בָּנוּ. ה׳ אוֹהֵב אוֹתָנוּ, וְלָכֵן הוּא שׁוֹמֵר עָלֵינוּ שְׁלֵמִים וּמְכֻבָּדִים 💛

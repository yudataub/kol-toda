הַחֻמָּשׁ הַחֲמִישִׁי

הַחֹמֶר הַמָּלֵא: דְּבָרִים וּמִשְׁנֵה תּוֹרָה – מֹשֶׁה רַבֵּנוּ מְלַמֵּד אֶת הַדּוֹר הַנִּכְנָס לָאָרֶץ
🎯 לַחִידוֹן🏠 כָּל הַחִידוֹנִים
הַיּוֹם אֲנַחְנוּ פּוֹתְחִים אֶת חֻמַּשׁ דְּבָרִים – הַחֻמָּשׁ הַחֲמִישִׁי וְהָאַחֲרוֹן בַּתּוֹרָה. זֶה הֶמְשֵׁךְ שֶׁל מַסָּע אָרֹךְ שֶׁהִתְחִיל בְּכִתָּה א׳, כְּשֶׁפָּתַחְנוּ חֻמָּשׁ בַּפַּעַם הָרִאשׁוֹנָה.
נִלְמַד לָמָּה קוֹרְאִים לַחֻמָּשׁ ׳דְּבָרִים׳ וְלָמָּה קוֹרְאִים לוֹ גַּם ׳מִשְׁנֵה תּוֹרָה׳, מָתַי וְאֵיפֹה מֹשֶׁה אָמַר אֶת דְּבָרָיו, מָה בָּנוּי בַּסֵּפֶר הַזֶּה, וְאֵילוּ אֶתְגָּרִים חִכּוּ לְעַם יִשְׂרָאֵל בָּאָרֶץ.
וּבְעִקָּר נַרְגִּישׁ כַּמָּה ה׳ אוֹהֵב אוֹתָנוּ: הוּא לֹא הִסְתַּפֵּק בְּכָךְ שֶׁנִּשְׁמַע אֶת הַתּוֹרָה פַּעַם אַחַת, אֶלָּא שָׁלַח אֶת מֹשֶׁה רַבֵּנוּ לַחֲזֹר עָלֶיהָ, לְבָאֵר אוֹתָהּ וְלָתֵת לָנוּ צֵידָה לַדֶּרֶךְ לְכָל הַדּוֹרוֹת 💛
בַּשִּׁעוּר: 📚 חֲמִשָּׁה חֻמְּשֵׁי תּוֹרָה – הַמַּסָּע מִכִּתָּה א׳ עַד כִּתָּה ו׳ 🎉 סִיּוּם חֲמִשָּׁה חֻמְּשֵׁי תּוֹרָה 🗣️ ׳אֵלֶּה הַדְּבָרִים׳ – הַשֵּׁם הָרִאשׁוֹן, וּמֹשֶׁה רַבֵּנוּ הַמְּלַמֵּד 🔁 ׳מִשְׁנֵה תּוֹרָה׳ – הַשֵּׁם הַשֵּׁנִי שֶׁל הַחֻמָּשׁ 🌄 בְּעַרְבוֹת מוֹאָב – ׳הוֹאִיל מֹשֶׁה בֵּאֵר אֶת הַתּוֹרָה׳ 📘 לֹא לְהִתְבַּלְבֵּל: ׳מִשְׁנֵה תּוֹרָה׳ שֶׁל הָרַמְבַּ״ם 🧱 מִבְנֵה הַסֵּפֶר וְתָכְנִית הַשָּׁנָה 📜 הָרַמְבַּ״ן: מָה עִנְיַן הַסֵּפֶר – וְהָאֶתְגָּרִים שֶׁבָּאָרֶץ 🧭 עִקְּרֵי הָאֱמוּנָה שֶׁבַּחֵלֶק הָרִאשׁוֹן

📚חֲמִשָּׁה חֻמְּשֵׁי תּוֹרָה – הַמַּסָּע מִכִּתָּה א׳ עַד כִּתָּה ו׳

הַתּוֹרָה שֶׁנָּתַן ה׳ לְמֹשֶׁה רַבֵּנוּ בְּנוּיָה מֵחֲמִשָּׁה חֻמָּשִׁים: בְּרֵאשִׁית, שְׁמוֹת, וַיִּקְרָא, בְּמִדְבַּר וּדְבָרִים.
הַשָּׁנָה אֲנַחְנוּ פּוֹתְחִים אֶת הַחֲמִישִׁי וְהָאַחֲרוֹן – חֻמַּשׁ דְּבָרִים. וּכְשֶׁמְּסַיְּמִים אוֹתוֹ, מְסַיְּמִים אֶת כָּל הַתּוֹרָה כֻּלָּהּ.

נִזְכֹּר רֶגַע אֵיךְ הִגַּעְנוּ עַד כָּאן. בְּכִתָּה א׳ פָּתַחְנוּ חֻמָּשׁ בַּפַּעַם הָרִאשׁוֹנָה וְקָרָאנוּ: ׳בְּרֵאשִׁית בָּרָא אֱלֹהִים׳.
בְּכִתּוֹת א׳–ב׳ לָמַדְנוּ אֶת חֻמַּשׁ בְּרֵאשִׁית: בְּרִיאַת הָעוֹלָם וְסִפּוּרֵי הָאָבוֹת – אַבְרָהָם, יִצְחָק, יַעֲקֹב וְיוֹסֵף.
בְּכִתָּה ג׳ – חֻמַּשׁ שְׁמוֹת: שִׁעְבּוּד מִצְרַיִם, הַמַּכּוֹת, יְצִיאַת מִצְרַיִם, קְרִיעַת יַם סוּף, מַתַּן תּוֹרָה וּבִנְיַן הַמִּשְׁכָּן.
בְּכִתָּה ד׳ – חֻמַּשׁ וַיִּקְרָא: הַקָּרְבָּנוֹת וְהַהִתְקָרְבוּת אֶל ה׳, הַכַּפָּרָה וְהַטָּהֳרָה, וּמִצְוַת ׳קְדֹשִׁים תִּהְיוּ׳.
בְּכִתָּה ה׳ – חֻמַּשׁ בְּמִדְבַּר: סִדּוּר הַמַּחֲנֶה, מַסַּע הַמַּחֲנֶה וְחֵטְא הַמְרַגְּלִים.
וּבְכִתָּה ו׳ – דְּבָרִים.

בְּחֻמַּשׁ בְּמִדְבַּר נִגְזַר עַל דּוֹר הַמִּדְבָּר שֶׁלֹּא יִכָּנֵס לָאָרֶץ, וּבְמֶשֶׁךְ אַרְבָּעִים שָׁנָה גָּדַל דּוֹר חָדָשׁ – דּוֹר הַבָּנִים. הֵם אֵלֶּה שֶׁעוֹמְדִים עַכְשָׁיו, בְּסוֹף חֻמַּשׁ בְּמִדְבַּר וּבִתְחִלַּת חֻמַּשׁ דְּבָרִים, מוּכָנִים לְהִכָּנֵס לְאֶרֶץ יִשְׂרָאֵל.
שִׂימוּ לֵב כַּמָּה ה׳ אוֹהֵב אוֹתָנוּ: הוּא לֹא נָתַן לָנוּ דַּף אֶחָד שֶׁל הוֹרָאוֹת, אֶלָּא חֲמִשָּׁה חֻמָּשִׁים שְׁלֵמִים – סִפּוּרִים, מִצְווֹת וְהַדְרָכוֹת – כְּדֵי שֶׁנֵּדַע אֵיךְ לִחְיוֹת וְאֵיךְ לִהְיוֹת קְרוֹבִים אֵלָיו.

💡 חֲמִשָּׁה חֻמָּשִׁים בִּקְצָרָה
בְּרֵאשִׁית – בְּרִיאַת הָעוֹלָם וְהָאָבוֹת.
שְׁמוֹת – יְצִיאַת מִצְרַיִם, מַתַּן תּוֹרָה וְהַמִּשְׁכָּן.
וַיִּקְרָא – הַקָּרְבָּנוֹת וְהַטָּהֳרָה.
בְּמִדְבַּר – הַמַּחֲנֶה, הַמַּסָּע וְחֵטְא הַמְרַגְּלִים.
דְּבָרִים – נְאוּם מֹשֶׁה לִפְנֵי הַכְּנִיסָה לָאָרֶץ.
💛 אַהֲבַת הַשֵּׁם
כָּל שָׁנָה קִבַּלְתִּי מֵה׳ עוֹד חֵלֶק מִן הַתּוֹרָה שֶׁלּוֹ, כְּמוֹ מַתָּנָה חֲדָשָׁה. הַשָּׁנָה אֲנִי מְקַבֵּל אֶת הַחֵלֶק הָאַחֲרוֹן – וְזוֹ זְכוּת גְּדוֹלָה 💛

🎉סִיּוּם חֲמִשָּׁה חֻמְּשֵׁי תּוֹרָה

בְּסוֹף הַשָּׁנָה, כְּשֶׁנְּסַיֵּם אֶת חֻמַּשׁ דְּבָרִים, לֹא נְסַיֵּם רַק חֻמָּשׁ אֶחָד. נְסַיֵּם אֶת כָּל חֲמִשָּׁה חֻמְּשֵׁי תּוֹרָה – מִ׳בְּרֵאשִׁית בָּרָא׳ וְעַד הַפָּסוּק הָאַחֲרוֹן שֶׁל הַתּוֹרָה.
לָכֵן עוֹשִׂים מְסִבַּת סִיּוּם גְּדוֹלָה. אֵין הַרְבֵּה פְּעָמִים בַּחַיִּים שֶׁיֶּלֶד מְסַיֵּם לִלְמֹד אֶת כָּל הַתּוֹרָה כֻּלָּהּ.

לְמִידַת הַתּוֹרָה הִיא הַלִּמּוּד הֶחָשׁוּב בְּיוֹתֵר בְּבֵית הַסֵּפֶר הַיְּסוֹדִי. לוֹמְדִים חֶשְׁבּוֹן, לְשׁוֹן וּמַדָּעִים – אֲבָל הַתּוֹרָה הִיא דִּבְרֵי ה׳ בְּעַצְמוֹ, וְהִיא זוֹ שֶׁמְּלַמֶּדֶת אוֹתָנוּ מִי אֲנַחְנוּ וּמָה ה׳ רוֹצֶה מֵאִתָּנוּ.

📚 יְלָדִים יְהוּדִים בְּכָל הַדּוֹרוֹת
בְּכָל הַדּוֹרוֹת, בְּכָל הַמְּקוֹמוֹת וּבְכָל הַמַּצָּבִים – יְלָדִים יְהוּדִים לָמְדוּ תּוֹרָה. גַּם כְּשֶׁהָיָה קָשֶׁה, גַּם כְּשֶׁנָּדְדוּ מִמָּקוֹם לְמָקוֹם, גַּם כְּשֶׁלֹּא הָיוּ סְפָרִים לְכָל אֶחָד.
הַמְּשִׂימָה הַזּוֹ עוֹבֶרֶת אֵלֵינוּ: לֹא מְוַתְּרִים עַל לִמּוּד תּוֹרָה, בְּשׁוּם מַצָּב.

וְיֵשׁ בַּסִּיּוּם גַּם שִׂמְחָה שֶׁל אַהֲבָה: כְּשֶׁאֲנִי מְסַיֵּם אֶת הַתּוֹרָה, אֲנִי כְּמוֹ מִי שֶׁקִּבֵּל מִכְתָּב אָרֹךְ מֵאַבָּא שֶׁבַּשָּׁמַיִם וְקָרָא אוֹתוֹ עַד הַסּוֹף – וְעַכְשָׁיו רוֹצֶה לְהַתְחִיל שׁוּב מֵהַהַתְחָלָה.

💛 אַהֲבַת הַשֵּׁם
ה׳ שָׂמֵחַ בְּכָל פָּסוּק שֶׁאֲנִי לוֹמֵד, וְלֹא רַק בַּסִּיּוּם. כָּל יוֹם שֶׁל לִמּוּד הוּא מַתָּנָה שֶׁאֲנִי נוֹתֵן לוֹ – וְהוּא נוֹתֵן לִי 💛

🗣️׳אֵלֶּה הַדְּבָרִים׳ – הַשֵּׁם הָרִאשׁוֹן, וּמֹשֶׁה רַבֵּנוּ הַמְּלַמֵּד

📖 פֶּרֶק א׳, פָּסוּק א׳ ←
א אֵלֶּה הַדְּבָרִים אֲשֶׁר דִּבֶּר מֹשֶׁה אֶל כָּל יִשְׂרָאֵל בְּעֵבֶר הַיַּרְדֵּן בַּמִּדְבָּר בָּעֲרָבָה מוֹל סוּף בֵּין פָּארָן וּבֵין תֹּפֶל וְלָבָן וַחֲצֵרֹת וְדִי זָהָב

הַחֻמָּשׁ נִפְתָּח בַּמִּלִּים ׳אֵלֶּה הַדְּבָרִים אֲשֶׁר דִּבֶּר מֹשֶׁה אֶל כָּל יִשְׂרָאֵל׳.
כְּמוֹ חֻמָּשִׁים אֲחֵרִים, גַּם הַחֻמָּשׁ הַזֶּה נִקְרָא עַל שֵׁם אַחַת הַמִּלִּים הָרִאשׁוֹנוֹת שֶׁבּוֹ – הַמִּלָּה הַשְּׁנִיָּה: ׳הַדְּבָרִים׳. כָּל הַחֻמָּשׁ הוּא דְּבָרִים שֶׁמֹּשֶׁה מְדַבֵּר אֶל בְּנֵי יִשְׂרָאֵל לִפְנֵי הַכְּנִיסָה לָאָרֶץ.

וְכָאן שְׁאֵלָה יָפָה: מָה הֵם הַדְּבָרִים הַגְּדוֹלִים שֶׁעָשָׂה מֹשֶׁה רַבֵּנוּ לְמַעַן עַם יִשְׂרָאֵל?
הוּא הָיָה הַשָּׁלִיחַ שֶׁל ה׳ בְּעֶשֶׂר הַמַּכּוֹת, בִּיצִיאַת מִצְרַיִם וּבִקְרִיעַת יַם סוּף; הוּא הוֹרִיד אֶת הַלּוּחוֹת מִן הָהָר; הוּא בָּנָה אֶת הַמִּשְׁכָּן; הוּא הִתְפַּלֵּל עַל הָעָם וְהוֹרִיד לָהֶם אֶת הַמָּן.
וּבְכָל זֹאת – חֻמָּשׁ שָׁלֵם מְסַפֵּר עַל כָּךְ שֶׁמֹּשֶׁה דִּבֵּר דְּבָרִים.

הַתְּשׁוּבָה: מֹשֶׁה אֵינוֹ סְתָם מֶלֶךְ שֶׁשּׁוֹלֵט בָּעָם. הוּא מַנְהִיג שֶׁמְּחַנֵּךְ וּמְלַמֵּד, וְלָכֵן אֲנַחְנוּ קוֹרְאִים לוֹ ׳מֹשֶׁה רַבֵּנוּ׳ – הָרַב שֶׁלָּנוּ.

ה׳ נָתַן לִי לֵב שֶׁיָּכוֹל לִבְחֹר. כְּשֶׁאֲנִי בּוֹחֵר לִלְמֹד, לְהִתְפַּלֵּל וְלַעֲשׂוֹת טוֹב – אֲנִי אוֹמֵר לוֹ בְּלִי מִלִּים: אֲנִי אוֹהֵב אוֹתְךָ 💛

✨ לְסִכּוּם

  • הַתּוֹרָה בְּנוּיָה מֵחֲמִשָּׁה חֻמָּשִׁים, וּדְבָרִים הוּא הַחֲמִישִׁי וְהָאַחֲרוֹן. כְּשֶׁנְּסַיֵּם אוֹתוֹ – נְסַיֵּם אֶת כָּל הַתּוֹרָה, וְנַעֲשֶׂה מְסִבַּת סִיּוּם חֲמִשָּׁה חֻמְּשֵׁי תּוֹרָה.
  • ׳דְּבָרִים׳ – עַל שֵׁם הַמִּלָּה הַשְּׁנִיָּה, ׳אֵלֶּה הַדְּבָרִים׳. ׳מִשְׁנֵה תּוֹרָה׳ – עַל שֵׁם הַתֹּכֶן: מֹשֶׁה חוֹזֵר וְשׁוֹנֶה אֶת הַתּוֹרָה, כְּמוֹ ׳לֶחֶם מִשְׁנֶה׳.
  • מֹשֶׁה אֵינוֹ רַק מַנְהִיג – הוּא מְחַנֵּךְ וּמְלַמֵּד, וְלָכֵן ׳מֹשֶׁה רַבֵּנוּ׳. רַשִׁ״י: ׳הוֹאִיל׳ – הִתְחִיל; ׳בֵּאֵר אֶת הַתּוֹרָה׳ – בְּשִׁבְעִים לָשׁוֹן פֵּרְשָׁהּ לָהֶם.
  • הַנְּאוּם נֶאֱמָר בְּעַרְבוֹת מוֹאָב, בַּשָּׁנָה הָאַרְבָּעִים, רֶגַע לִפְנֵי הַכְּנִיסָה לָאָרֶץ. מֹשֶׁה אֵינוֹ עוֹבֵר אֶת הַיַּרְדֵּן, וּדְבָרָיו הֵם צֵידָה לַדֶּרֶךְ לְכָל הַדּוֹרוֹת.
  • הַסֵּפֶר בָּנוּי מִשְּׁלוֹשָׁה חֲלָקִים: מוּסָר וְתוֹכֵחָה (א׳–י״א), מִצְווֹת (י״ב–כ״ו), בְּרִית וּבְרָכוֹת (כ״ז–ל״ד). הָרַמְבַּ״ן: מֹשֶׁה מְבָאֵר לַדּוֹר הַנִּכְנָס לָאָרֶץ אֶת רֹב הַמִּצְווֹת הַצְּרִיכוֹת לְיִשְׂרָאֵל.
  • בָּאָרֶץ מְחַכִּים אֶתְגָּרִים: מִלְחָמָה, תַּרְבּוּת זָרָה, חַקְלָאוּת וְשֶׁפַע, וְהִתְיַשְּׁבוּת. לָכֵן מֹשֶׁה מְחַזֵּק עִקְּרֵי אֱמוּנָה – וְכָל זֶה מִפְּנֵי שֶׁה׳ אוֹהֵב אוֹתָנוּ וְרוֹצֶה בְּטוֹבָתֵנוּ.

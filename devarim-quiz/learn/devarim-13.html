בַּשִּׁעוּר הַקּוֹדֵם לָמַדְנוּ אֶת עֲשֶׂרֶת הַדִּבְּרוֹת וּמִיַּנּוּ אוֹתָן לִשְׁנֵי לוּחוֹת: הָרִאשׁוֹן – מִצְווֹת בֵּין אָדָם לַמָּקוֹם, וְהַשֵּׁנִי – מִצְווֹת בֵּין אָדָם לַחֲבֵרוֹ.
יֵשׁ גַּם מִיּוּן נוֹסָף: חָמֵשׁ הַדִּבְּרוֹת שֶׁבַּלּוּחַ הַשֵּׁנִי כֻּלָּן מִצְווֹת לֹא תַעֲשֶׂה – לֹא תִּרְצָח, לֹא תִּנְאָף, לֹא תִּגְנֹב, לֹא תַעֲנֶה בְרֵעֲךָ עֵד שָׁוְא, וְלֹא תַחְמֹד וְלֹא תִתְאַוֶּה. לְעֻמָּתָן, בַּלּוּחַ הָרִאשׁוֹן יֵשׁ גַּם מִצְווֹת עֲשֵׂה וְגַם מִצְווֹת לֹא תַעֲשֶׂה.
הַיּוֹם נִתְמַקֵּד בִּשְׁתַּיִם מֵהַדִּבְּרוֹת שֶׁבַּלּוּחַ הָרִאשׁוֹן: הַשַּׁבָּת וְכִבּוּד הוֹרִים.

שִׁשָּׁה יָמִים בַּשָּׁבוּעַ אָדָם עוֹבֵד וְעוֹשֶׂה אֶת כָּל מְלַאכְתּוֹ, וְיוֹם אֶחָד – הַיּוֹם הַשְּׁבִיעִי – הוּא יוֹם קָדוֹשׁ לַה׳.

מִתַּחַת לַקַּו, הַפָּסוּק הָאַחֲרוֹן, הוּא הַנִּמּוּק – לָמָּה: ׳וְזָכַרְתָּ כִּי עֶבֶד הָיִיתָ בְּאֶרֶץ מִצְרַיִם׳.
בְּהֶמְשֵׁךְ נַשְׁוֶה אֶת הַדִּבְּרָה לְסֵפֶר שְׁמוֹת, וְנְגַלֶּה שִׁנּוּי גַּם בַּצִּוּוּי וְגַם בַּנִּמּוּק.

🔍שְׁמוֹת מוּל דְּבָרִים – לָמָּה יֵשׁ שִׁנּוּיִים?

📚 מָשָׁל: הַתַּקָּנוֹן שֶׁל בֵּית הַסֵּפֶר

לְתַלְמִידֵי כִּתָּה א׳ בִּתְחִלַּת הַשָּׁנָה יַסְבִּירוּ בְּדֶרֶךְ אַחַת; לְתַלְמִיד חָדָשׁ שֶׁמִּצְטָרֵף לְכִתָּה גְּבוֹהָה, כְּשֶׁכָּל בֵּית הַסֵּפֶר כְּבָר מַכִּיר אֶת הַכְּלָלִים – בְּדֶרֶךְ אַחֶרֶת; וּלְמִי שֶׁכְּבָר עָבַר עֲלֵיהֶם – רַק יַזְכִּירוּ בְּקִצּוּר.

כְּשֶׁמַּנִּיחִים זֶה לְצַד זֶה אֶת הַדִּבְּרוֹת שֶׁבְּסֵפֶר שְׁמוֹת (פֶּרֶק כ׳) וְאֶת הַדִּבְּרוֹת שֶׁבְּחֻמַּשׁ דְּבָרִים (פֶּרֶק ה׳), מְגַלִּים:

בְּאַרְבַּע יֵשׁ שִׁנּוּיִים: שְׁמִירַת שַׁבָּת, כִּבּוּד הוֹרִים, ׳לֹא תַעֲנֶה בְרֵעֲךָ׳ וְ׳לֹא תַחְמֹד׳.

💡 טַבְלַת הַהַשְׁוָאָה שֶׁלָּנוּ

שַׁבָּת – הַטַּעַם: בִּשְׁמוֹת ׳כִּי שֵׁשֶׁת יָמִים עָשָׂה ה׳ אֶת הַשָּׁמַיִם וְאֶת הָאָרֶץ׳; בִּדְבָרִים ׳לְמַעַן יָנוּחַ עַבְדְּךָ וַאֲמָתְךָ כָּמוֹךָ, וְזָכַרְתָּ כִּי עֶבֶד הָיִיתָ בְּאֶרֶץ מִצְרַיִם׳.
כִּבּוּד הוֹרִים: בִּשְׁמוֹת ׳לְמַעַן יַאֲרִכוּן יָמֶיךָ׳; בִּדְבָרִים נוֹסְפוּ ׳כַּאֲשֶׁר צִוְּךָ ה׳ אֱלֹהֶיךָ׳ וְגַם ׳וּלְמַעַן יִיטַב לָךְ׳.

אִם רַשִׁ״י לִמֵּד אוֹתָנוּ שֶׁהַכֹּל נֶאֱמַר ׳בְּדִבּוּר אֶחָד׳, וְהַצִּוּוּי שֶׁל ה׳ לֹא הִשְׁתַּנָּה – לָמָּה הַנֻּסָּח שׁוֹנֶה? שְׁתֵּי סִבּוֹת:

הַזְּמַן שׁוֹנֶה: הָעָם עוֹמֵד לְהִכָּנֵס לְאֶרֶץ יִשְׂרָאֵל, וּלְכָל מִשְׁפָּחָה תִּהְיֶה שָׁם נַחֲלָה, בַּיִת וְשָׂדֶה – מְצִיאוּת חֲדָשָׁה שֶׁל מָקוֹם יַצִּיב וּבָטוּחַ.

🤔 מָשָׁל מִבֵּית הַסֵּפֶר

מָתַי הִיא תּוּכַל לְדַבֵּר אִתּוֹ בְּצוּרָה בְּרוּרָה וּמְפֹרֶטֶת יוֹתֵר – בְּשִׂיחָה אִישִׁית, אוֹ כְּשֶׁכָּל הַכִּתָּה שׁוֹמַעַת?
בְּשִׂיחָה אִישִׁית. וְכָךְ גַּם כָּאן: בְּחֻמַּשׁ דְּבָרִים ה׳ מְדַבֵּר ׳בְּאַרְבַּע עֵינַיִם׳ עִם עַם יִשְׂרָאֵל, וְלָכֵן הַשָּׂכָר מְפֹרָט יוֹתֵר.

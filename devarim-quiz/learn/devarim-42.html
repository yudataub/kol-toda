עֶגְלָה עֲרוּפָה

הַחֹמֶר הַמָּלֵא: כָּל אָדָם חָשׁוּב – וְכָל הַמַּנְהִיגִים לוֹקְחִים אַחְרָיוּת (דְּבָרִים כ״א, א–ט)
🎯 לַחִידוֹן🏠 כָּל הַחִידוֹנִים
הִגַּעְנוּ לַשִּׁעוּר הָאַחֲרוֹן בְּפָרָשַׁת שֹׁפְטִים, וּלְפָנֵינוּ תַּעֲלוּמָה: נִמְצָא אָדָם חָלָל בַּשָּׂדֶה, וְאִישׁ אֵינוֹ יוֹדֵעַ מִי פָּגַע בּוֹ.
מָה עוֹשִׂים? הַתּוֹרָה מְצַוָּה מִצְוָה מְיֻחֶדֶת מְאוֹד: זִקְנֵי הַסַּנְהֶדְרִין יוֹצְאִים מִירוּשָׁלַיִם וּמוֹדְדִים אֶת הַמֶּרְחָק אֶל הֶעָרִים, זִקְנֵי הָעִיר הַקְּרוֹבָה עוֹרְפִים עֶגְלָה בְּנַחַל אֵיתָן וְרוֹחֲצִים אֶת יְדֵיהֶם, וְהַכֹּהֲנִים מִתְפַּלְּלִים ׳כַּפֵּר לְעַמְּךָ יִשְׂרָאֵל׳.
בַּשִּׁעוּר הַזֶּה נִלְמַד כַּמָּה יְקָרִים חַיָּיו שֶׁל כָּל אָדָם בְּעֵינֵי ה׳ – עַד כְּדֵי כָּךְ שֶׁמּוֹתוֹ שֶׁל אָדָם אֶחָד אֵינוֹ ׳עוֹבֵר לְסֵדֶר הַיּוֹם׳, וְכָל הַהַנְהָגָה שֶׁל הָעָם מִתְעַסֶּקֶת בּוֹ וְלוֹקַחַת אַחְרָיוּת.
בַּשִּׁעוּר: 👑 הַמַּנְהִיגִים שֶׁל פָּרָשַׁת שֹׁפְטִים 🔎 הַתַּעֲלוּמָה – ׳לֹא נוֹדַע מִי הִכָּהוּ׳ 🏛️ וְיָצְאוּ זְקֵנֶיךָ וְשֹׁפְטֶיךָ – וּמוֹדְדִים בְּמוֹ יְדֵיהֶם 🏘️ הָעִיר הַקְּרוֹבָה – וְהָעֶגְלָה שֶׁלֹּא מָשְׁכָה בְּעֹל 🏞️ נַחַל אֵיתָן – וְעָרְפוּ שָׁם אֶת הָעֶגְלָה 🙌 יָדֵינוּ לֹא שָׁפְכוּ – בְּלֹא מְזוֹנוֹת וּבְלֹא לְוָיָה 🕊️ וְנִגְּשׁוּ הַכֹּהֲנִים – ׳כַּפֵּר לְעַמְּךָ יִשְׂרָאֵל׳ 👥 מִי עוֹשֶׂה מָה? שְׁלוֹשָׁה תַּפְקִידִים בַּפָּרָשָׁה ⚖️ וְאַתָּה תְּבַעֵר הַדָּם הַנָּקִי – הַיָּשָׁר בְּעֵינֵי ה׳

👑הַמַּנְהִיגִים שֶׁל פָּרָשַׁת שֹׁפְטִים

לִפְנֵי שֶׁנִּכָּנֵס לַתַּעֲלוּמָה, נִזְכֹּר דָּבָר כְּלָלִי שֶׁלָּמַדְנוּ: פָּרָשַׁת שֹׁפְטִים הִיא הַפָּרָשָׁה שֶׁל הַמַּנְהִיגִים – הָאֲנָשִׁים וְהַתַּפְקִידִים שֶׁמַּנְהִיגִים אֶת עַם יִשְׂרָאֵל.

שׁוֹפְטִים וְשׁוֹטְרִים – בֵּית הַדִּין פּוֹסֵק אֶת הַדִּין, וְהַשּׁוֹטְרִים דּוֹאֲגִים שֶׁהַדִּין יִתְקַיֵּם בַּצִּבּוּר.
הַכֹּהֵן – אֵין לוֹ נַחֲלָה בָּאָרֶץ, וְאָנוּ מְצֻוִּים לָתֵת לוֹ מַתָּנוֹת וְלִתְמֹךְ בּוֹ.
הַמֶּלֶךְ – כּוֹתֵב לְעַצְמוֹ סֵפֶר תּוֹרָה, כְּדֵי שֶׁיִּשְׁמֹר יִרְאַת שָׁמַיִם וְלֹא יִתְגָּאֶה עַל הָעָם.
הַנָּבִיא – רַק אֵלָיו פּוֹנִים כְּשֶׁרוֹצִים לָדַעַת מָה ה׳ רוֹצֶה, וְלֹא אֶל מְנַחֲשִׁים וְקוֹסְמִים.
הַזְּקֵנִים – לֹא דְּמוּת אַחַת אֶלָּא מוֹסָד שָׁלֵם: בֵּית הַדִּין הַגָּדוֹל שֶׁבִּירוּשָׁלַיִם, הַנִּקְרָא סַנְהֶדְרִין. אֵלָיו עוֹלִים כְּשֶׁ׳כִּי יִפָּלֵא מִמְּךָ דָבָר לַמִּשְׁפָּט׳.

וְהִנֵּה דָּבָר מַפְתִּיעַ: בַּפָּרָשָׁה הָאַחֲרוֹנָה שֶׁל שֹׁפְטִים, פָּרָשַׁת עֶגְלָה עֲרוּפָה, מִשְׁתַּתְּפִים הַשּׁוֹפְטִים, הַכֹּהֲנִים וְהַזְּקֵנִים – כִּמְעַט כָּל הַמַּנְהִיגִים יַחַד. כַּמָּה גָּדוֹל חַיָּב לִהְיוֹת הָעִנְיָן, שֶׁכֻּלָּם נִדְרָשִׁים לְטַפֵּל בּוֹ!

וְלָמָּה בִּכְלָל ה׳ נָתַן לָנוּ מַנְהִיגִים? כִּי הוּא אוֹהֵב אוֹתָנוּ וְרוֹצֶה שֶׁיִּהְיֶה לָנוּ מִי שֶׁיִּשְׁמֹר עָלֵינוּ, מִי שֶׁיַּדְרִיךְ וּמִי שֶׁיִּשְׂא בָּאַחְרָיוּת. אַבָּא טוֹב אֵינוֹ מַשְׁאִיר אֶת יְלָדָיו לְבַד.

💛 אַהֲבַת הַשֵּׁם
ה׳ סִדֵּר לָנוּ עַם שֶׁיֵּשׁ בּוֹ סֵדֶר: שׁוֹפְטִים, כֹּהֲנִים, מֶלֶךְ וּנְבִיאִים. כְּשֶׁאֲנִי קוֹרֵא אֶת הַפָּרָשָׁה אֲנִי מַרְגִּישׁ שֶׁה׳ לֹא שָׁכַח אַף אֶחָד – גַּם לֹא אוֹתִי 💛

🔎הַתַּעֲלוּמָה – ׳לֹא נוֹדַע מִי הִכָּהוּ׳

📖 פֶּרֶק כ״א, פָּסוּק א׳ ←
א כִּי יִמָּצֵא חָלָל בָּאֲדָמָה אֲשֶׁר ה׳ אֱלֹהֶיךָ נֹתֵן לְךָ לְרִשְׁתָּהּ נֹפֵל בַּשָּׂדֶה לֹא נוֹדַע מִי הִכָּהוּ

כָּךְ פּוֹתַחַת הַתּוֹרָה: ׳כִּי יִמָּצֵא חָלָל בָּאֲדָמָה אֲשֶׁר ה׳ אֱלֹהֶיךָ נֹתֵן לְךָ לְרִשְׁתָּהּ, נֹפֵל בַּשָּׂדֶה, לֹא נוֹדַע מִי הִכָּהוּ׳.
אָדָם נִמְצָא מֵת בְּשֶׁטַח פָּתוּחַ, מִחוּץ לָעִיר. הוּא אֵינוֹ ׳נָפַל מֵחֲמַת חֻלְשָׁה׳ – הוּא חָלָל, מִישֶׁהוּ פָּגַע בּוֹ. אֲבָל מִי? אִישׁ אֵינוֹ יוֹדֵעַ.

זוֹ תַּעֲלוּמָה: אֵין עֵדִים, אֵין מִי שֶׁרָאָה, וְאֵין מִי שֶׁמְּסַפֵּר. בְּבֵית דִּין רָגִיל אִי אֶפְשָׁר לַעֲשׂוֹת כְּלוּם בְּלִי עֵדִים.
וְדַוְקָא כָּאן הַתּוֹרָה מְלַמֶּדֶת מָה כֵּן עוֹשִׂים, וְלֹא מַנִּיחָה לָעָם לְהַגִּיד ׳אֵין מָה לַעֲשׂוֹת׳.

שִׂימוּ לֵב: הַתּוֹרָה אֵינָהּ מְסַפֶּרֶת לָנוּ מִי הָיָה הָאִישׁ, מֵאֵיזוֹ מִשְׁפָּחָה וּמֵאֵיזֶה שֵׁבֶט. אוּלַי הוּא עוֹבֵר אֹרַח שֶׁאֵין לוֹ כָּאן אַף אֶחָד.
וּבְכָל זֹאת כָּל הָעָם מִתְעוֹרֵר בִּגְלָלוֹ – כִּי לִפְנֵי ה׳ כָּל אָדָם חָשׁוּב, גַּם מִי שֶׁאֵין לוֹ מִי שֶׁיִּדְאַג לוֹ.

💛 אַהֲבַת הַשֵּׁם
כָּל אָדָם נִבְרָא בְּצֶלֶם אֱלֹהִים, וְלָכֵן ה׳ אֵינוֹ מְוַתֵּר עַל אַף אֶחָד. גַּם אִם אִישׁ לֹא יוֹדֵעַ אֶת שְׁמִי – ה׳ יוֹדֵעַ, וְאוֹהֵב אוֹתִי 💛

🏛️וְיָצְאוּ זְקֵנֶיךָ וְשֹׁפְטֶיךָ – וּמוֹדְדִים בְּמוֹ יְדֵיהֶם

ב וְיָצְאוּ זְקֵנֶיךָ וְשֹׁפְטֶיךָ וּמָדְדוּ אֶל הֶעָרִים אֲשֶׁר סְבִיבֹת הֶחָלָל

׳וְיָצְאוּ זְקֵנֶיךָ וְשֹׁפְטֶיךָ וּמָדְדוּ אֶל הֶעָרִים אֲשֶׁר סְבִיבֹת הֶחָלָל׳. מִי הֵם הַזְּקֵנִים הָאֵלֶּה?

📜 רַשִׁ״י
׳וְיָצְאוּ זְקֵנֶיךָ׳ – מְיֻחָדִים שֶׁבִּזְקֵנֶיךָ, אֵלּוּ סַנְהֶדְרֵי גְדוֹלָה.

לֹא סְתָם זְקֵנִים, וְלֹא פְּקִידִים קְטַנִּים: בֵּית הַדִּין הַגָּדוֹל שֶׁבִּירוּשָׁלַיִם – שִׁבְעִים וְאֶחָד הַחֲכָמִים הַגְּדוֹלִים בְּיוֹתֵר בְּעַם יִשְׂרָאֵל – יוֹצְאִים מִמְּקוֹמָם וְהוֹלְכִים אֶל הַשָּׂדֶה.

וּמָה הֵם עוֹשִׂים שָׁם? אוּלַי הֵם בָּאִים לַחְקֹר וְלִפְתֹּר אֶת הַתַּעֲלוּמָה? לֹא. הֵם עוֹשִׂים דָּבָר אַחֵר לְגַמְרֵי: הֵם מוֹדְדִים.

📜 רַשִׁ״י
׳וּמָדְדוּ׳ – מִמָּקוֹם שֶׁהֶחָלָל שׁוֹכֵב.
׳אֶל הֶעָרִים אֲשֶׁר סְבִיבֹת הֶחָלָל׳ – לְכָל צַד, לֵידַע אֵיזוֹ קְרוֹבָה.

כְּלוֹמַר: מַתְחִילִים לִמְדֹּד בְּדִיּוּק מִן הַמָּקוֹם שֶׁבּוֹ הוּא שׁוֹכֵב, וּמוֹדְדִים לְכָל הַכִּוּוּנִים – אֶל כָּל עִיר שֶׁבַּסְּבִיבָה – כְּדֵי לָדַעַת אֵיזוֹ עִיר הִיא הַקְּרוֹבָה בְּיוֹתֵר. לוֹקְחִים חֶבֶל מִדָּה, וּמוֹדְדִים.

🤔 תָּאֲרוּ לְעַצְמְכֶם
בְּעִיר קְטַנָּה בְּאֶרֶץ יִשְׂרָאֵל, לְיַד הַשָּׂדֶה, מוֹפִיעָה פִּתְאוֹם מִשְׁלַחַת הַסַּנְהֶדְרִין מִירוּשָׁלַיִם.
הַחֲכָמִים הַגְּדוֹלִים בְּיוֹתֵר אֵינָם שׁוֹלְחִים שְׁלִיחַ – הֵם בָּאִים בְּעַצְמָם, וּבְמוֹ יְדֵיהֶם מוֹתְחִים אֶת חֶבֶל הַמִּדָּה.
מִי שֶׁרוֹאֶה אֶת זֶה מֵבִין מִיָּד: כָּאן קָרָה דָּבָר נוֹרָא, וְאִי אֶפְשָׁר לְהִתְעַלֵּם מִמֶּנּוּ.

לָמָּה ה׳ צִוָּה דַּוְקָא כָּךְ? כְּדֵי שֶׁכֻּלָּם יִרְאוּ בְּעֵינַיִם שֶׁחַיֵּי אָדָם אֵינָם עִנְיָן קָטָן. ה׳ אוֹהֵב אֶת בְּרוּאָיו, וְלָכֵן הוּא מַזִּיז בִּשְׁבִיל אָדָם אֶחָד אֶת כָּל הַהַנְהָגָה.

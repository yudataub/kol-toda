אַרְבָּעָה פְּסוּקִים קְטַנִּים, וּבְתוֹכָם שְׁתֵּי מִצְווֹת גְּדוֹלוֹת: מִצְוַת הֲשָׁבַת אֲבֵדָה וּמִצְוַת טְעִינָה.
הַתּוֹרָה מְלַמֶּדֶת אוֹתָנוּ לֹא רַק מָה אָסוּר לַעֲשׂוֹת לָרְכוּשׁ שֶׁל אַחֵר – לֹא לְהַזִּיק, לֹא לִגְנֹב, לֹא לָקַחַת בְּלִי רְשׁוּת – אֶלָּא גַּם מָה חַיָּבִים לַעֲשׂוֹת בִּשְׁבִילוֹ: לַעֲצֹר, לְהִתְאַמֵּץ, לִשְׁמֹר וּלְהַחֲזִיר.
נִלְמַד אֶת פֵּרוּשֵׁי רַשִׁ״י, נַכִּיר אֶת ׳אֶבֶן הַטּוֹעַן׳ שֶׁהָיְתָה בִּירוּשָׁלַיִם, נִשְׁמַע אֶת סִפּוּר הַתַּרְנְגוֹלוֹת שֶׁל רַבִּי חֲנִינָא בֶּן דּוֹסָא, וְנָבִין לָמָּה הַמִּלָּה ׳אָחִיךָ׳ חוֹזֶרֶת כָּאן שׁוּב וָשׁוּב.
וּבְעִקָּר נַרְגִּישׁ: ה׳ אוֹהֵב כָּל יְהוּדִי, וְאִכְפַּת לוֹ גַּם מֵהָאֲבֵדָה הַקְּטַנָּה שֶׁלּוֹ – וְהוּא מְבַקֵּשׁ מֵאִתָּנוּ לִהְיוֹת הַשְּׁלִיחִים שֶׁלּוֹ 💛
בַּשִּׁעוּר: ✂️ אַרְבָּעָה פְּסוּקִים – שְׁנֵי נוֹשְׂאִים 🐑 שְׁתֵּי מִצְווֹת בְּפָסוּק אֶחָד 🙈 ׳וְהִתְעַלַּמְתָּ׳ – כּוֹבֵשׁ עַיִן 🏠 וַאֲסַפְתּוֹ אֶל תּוֹךְ בֵּיתֶךָ – שְׁמִירָה, הַכְרָזָה וְסִימָנִים 🧥 וְכֵן תַּעֲשֶׂה לְכָל אֲבֵדַת אָחִיךָ – וְהַתַּרְנְגוֹלוֹת שֶׁל רַבִּי חֲנִינָא 🫏 הָקֵם תָּקִים עִמּוֹ – מִצְוַת טְעִינָה 👬 ׳אָחִיךָ׳ – לֹא לְהִתְעַלְּמוּת, כֵּן לְאַחֲוָה 👕 הַפָּסוּק הַבָּא: ׳לֹא יִלְבַּשׁ גֶּבֶר שִׂמְלַת אִשָּׁה׳ 📜 מִלִּים מֵרַשִׁ״י – חֲזָרָה קְצָרָה

לִפְנֵי שֶׁמַּתְחִילִים, שְׁאֵלָה: אֵילוּ דְּבָרִים אֲנַחְנוּ צְרִיכִים לְהִזָּהֵר שֶׁלֹּא לַעֲשׂוֹת לָרְכוּשׁ שֶׁל חֲבֵרֵנוּ?
לֹא לְהַזִּיק, לֹא לִגְנֹב, לֹא לִשְׁאֹל בְּלִי רְשׁוּת, לֹא לְקַלְקֵל.
וְעַכְשָׁו שְׁאֵלָה הֲפוּכָה: אֵילוּ דְּבָרִים אֲנַחְנוּ צְרִיכִים כֵּן לַעֲשׂוֹת בִּשְׁבִיל הָרְכוּשׁ שֶׁל חֲבֵרֵנוּ? עַל זֶה בְּדִיּוּק מְדַבְּרִים הַפְּסוּקִים שֶׁלָּנוּ: לִשְׁמֹר עָלָיו, לְהַצִּיל אוֹתוֹ, וּלְהַחֲזִיר אוֹתוֹ לַבְּעָלִים.

לְפָנֵינוּ אַרְבָּעָה פְּסוּקִים, וְהֵם נֶחְלָקִים לִשְׁתֵּי יְחִידוֹת:

הַמִּשְׁפָּחָה רָצָה בָּרְחוֹב אַחֲרֵי מַשָּׂאִית הָאַשְׁפָּה וְעָצְרָה אוֹתָהּ. הַנֶּהָג רוֹקֵן אֶת כָּל הַמַּשָּׂאִית בְּאֶמְצַע הָרְחוֹב, וּמֵאוֹת מִתְנַדְּבִים בָּאוּ לְחַפֵּשׂ – וְהַטַּבַּעַת נִמְצְאָה.
כְּשֶׁאָנוּ מְאַבְּדִים מַשֶּׁהוּ יָקָר, אֲנַחְנוּ רוֹצִים שֶׁכָּל הָעוֹלָם יַעֲצֹר בִּשְׁבִילֵנוּ. הַתּוֹרָה מְבַקֶּשֶׁת: תַּעֲשֶׂה כָּךְ גַּם בִּשְׁבִיל אַחֵר.

מִתּוֹךְ הַמִּצְוָה הַזּוֹ אֶפְשָׁר לְהַרְגִּישׁ כַּמָּה ה׳ אוֹהֵב כָּל יְהוּדִי. לַה׳ אִכְפַּת מֵהַכִּיס שֶׁל אָדָם פָּשׁוּט, מֵהַשֶּׂה שֶׁבָּרַח לוֹ וּמֵהָאַרְנָק שֶׁנָּפַל לוֹ, וְהוּא כּוֹתֵב עַל כָּךְ אַרְבָּעָה פְּסוּקִים שְׁלֵמִים בַּתּוֹרָה. וְאוֹתָנוּ הוּא מַזְמִין לִהְיוֹת הַשְּׁלִיחִים שֶׁלּוֹ שֶׁמַּחֲזִירִים לַחֲבֵרֵנוּ אֶת מָה שֶׁאָבַד לוֹ.

הַתּוֹרָה רוֹצָה שֶׁכָּל אֶחָד יִשְׁמֹר עַל הַלְּבוּשׁ הַמַּתְאִים לוֹ, וְלֹא יַעֲשֶׂה אֶת עַצְמוֹ כְּאִלּוּ הוּא מִישֶׁהוּ אַחֵר.

בְּכָל פַּעַם שֶׁאֲנַחְנוּ פּוֹתְחִים חֻמָּשׁ עִם רַשִׁ״י, אֲנַחְנוּ מְגַלִּים כַּמָּה פְּרָטִים אוֹהֲבִים מִסְתַּתְּרִים בְּכָל מִלָּה שֶׁל הַתּוֹרָה. זוֹ מַתָּנָה שֶׁה׳ נָתַן לָנוּ: תּוֹרָה שֶׁאֶפְשָׁר לְהַעֲמִיק בָּהּ וְלֶאֱהֹב אוֹתָהּ עוֹד וְעוֹד.

  • אַרְבָּעָה פְּסוּקִים, שְׁנֵי נוֹשְׂאִים: פְּסוּקִים א׳–ג׳ הֲשָׁבַת אֲבֵדָה, פָּסוּק ד׳ מִצְוַת טְעִינָה.
  • בְּפָסוּק א׳ שְׁתֵּי מִצְווֹת: לֹא תַעֲשֶׂה – ׳לֹא תִרְאֶה... וְהִתְעַלַּמְתָּ׳, וְעֲשֵׂה – ׳הָשֵׁב תְּשִׁיבֵם לְאָחִיךָ׳. ׳נִדָּחִים׳ – טוֹעִים בַּדֶּרֶךְ.
  • ׳וְהִתְעַלַּמְתָּ׳ לְפִי רַשִׁ״י – כּוֹבֵשׁ עַיִן כְּאִלּוּ אֵינוֹ רוֹאֵהוּ. בַּפְּשָׁט אָסוּר לְהִתְעַלֵּם; וּבַדְּרָשׁ – פְּעָמִים שֶׁאַתָּה מִתְעַלֵּם: כֹּהֵן בְּבֵית הַקְּבָרוֹת, זָקֵן שֶׁאֵינוֹ לְפִי כְּבוֹדוֹ, וּמִי שֶׁמְּלַאכְתּוֹ מְרֻבָּה.
  • אִם אֵינְךָ יוֹדֵעַ שֶׁל מִי הָאֲבֵדָה – וַאֲסַפְתּוֹ אֶל תּוֹךְ בֵּיתֶךָ: שׁוֹמְרִים, מַכְרִיזִים, וּמַחֲזִירִים לְמִי שֶׁנּוֹתֵן סִימָנִים (׳דָּרְשֵׁהוּ שֶׁלֹּא יְהֵא רַמַּאי׳). וְ׳וַהֲשֵׁבֹתוֹ לוֹ׳ – שֶׁתְּהֵא בּוֹ הֲשָׁבָה.
  • פָּסוּק ג׳ מַרְחִיב לְכָל אֲבֵדָה – חֲמוֹר, שִׂמְלָה, אַרְנָק וּמַפְתֵּחַ; וְהַסְּפוֹרְנוֹ מוֹסִיף: גַּם דְּבָרִים שֶׁאֵין רָגִיל לְאַבֵּד. וְרַבִּי חֲנִינָא בֶּן דּוֹסָא הֶחֱזִיר עִזִּים תְּמוּרַת תַּרְנְגוֹלוֹת שֶׁשָּׁמַר.
  • ׳הָקֵם תָּקִים׳ – טְעִינָה, לְהַטְעִין מַשָּׂא שֶׁנָּפַל; ׳עִמּוֹ׳ – עִם הַבְּעָלִים. בִּשְׁמוֹת כ״ג – פְּרִיקָה. הַמִּלָּה ׳אָחִיךָ׳ חוֹזֶרֶת בְּכָל הַפְּסוּקִים: לֹא לְהִתְעַלְּמוּת – כֵּן לְאַחֲוָה, כִּי ה׳ אוֹהֵב אוֹתָנוּ וְרוֹצֶה שֶׁנִּהְיֶה מִשְׁפָּחָה אַחַת.

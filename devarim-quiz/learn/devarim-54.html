צַעַר – עַל הַכְּאֵב עַצְמוֹ. חֲכָמִים נָתְנוּ אֲפִלּוּ דֶּרֶךְ לְהַעֲרִיךְ כַּמָּה אָדָם הָיָה מוּכָן לְשַׁלֵּם כְּדֵי לֹא לְקַבֵּל מַכָּה כָּזוֹ.
רִפּוּי – עַל הַתְּרוּפוֹת, הַתַּחְבּוֹשׁוֹת וְהַטִּפּוּלִים.
שֶׁבֶת – עַל הַזְּמַן שֶׁהַנִּזָּק יָשַׁב בְּבֵיתוֹ וְלֹא יָכוֹל הָיָה לַעֲבֹד. גַּם הַיּוֹם כֻּלָּם מְבִינִים שֶׁיְּמֵי מַחֲלָה הֵם הֶפְסֵד כַּסְפִּי.

אַרְבָּעָה הַתַּשְׁלוּמִים הָאֵלֶּה כְּתוּבִים כְּבָר בְּפָרָשַׁת מִשְׁפָּטִים (שְׁמוֹת פֶּרֶק כ״א). חֻמַּשׁ דְּבָרִים בָּא לְהַשְׁלִים אֶת הַתְּמוּנָה – וְאֶת הַתַּשְׁלוּם הַחֲמִישִׁי נִלְמַד בַּסָּעִיף הַבָּא.

📜 רַשִׁ״י
׳וְקַצֹּתָה אֶת כַּפָּהּ׳ – מָמוֹן, דְּמֵי בָּשְׁתּוֹ, הַכֹּל לְפִי הַמְבַיֵּשׁ וְהַמִּתְבַּיֵּשׁ. אוֹ אֵינוֹ אֶלָּא יָדָהּ מַמָּשׁ? נֶאֱמַר כָּאן ׳לֹא תָחוֹס׳, וְנֶאֱמַר לְהַלָּן בְּעֵדִים זוֹמְמִין ׳לֹא תָחוֹס׳ – מָה לְהַלָּן מָמוֹן, אַף כָּאן מָמוֹן.

רַשִׁ״י מְלַמֵּד שְׁנֵי דְּבָרִים: הָרִאשׁוֹן – הַתַּשְׁלוּם הוּא כֶּסֶף וְלֹא עֹנֶשׁ בַּגּוּף, וְאֶת זֶה לוֹמְדִים מִגְּזֵרָה שָׁוָה: הַמִּלִּים ׳לֹא תָחוֹס׳ מוֹפִיעוֹת גַּם אֵצֶל עֵדִים זוֹמְמִין, וְשָׁם מְדֻבָּר בְּמָמוֹן.
וְהַשֵּׁנִי – ׳הַכֹּל לְפִי הַמְבַיֵּשׁ וְהַמִּתְבַּיֵּשׁ׳: גֹּדֶל הַתַּשְׁלוּם תָּלוּי בְּמִי שֶׁבִּיֵּשׁ וּבְמִי שֶׁהִתְבַּיֵּשׁ. אִם אָדָם פָּשׁוּט בִּיֵּשׁ אָדָם גָּדוֹל וְנִכְבָּד לְעֵינֵי רַבִּים – הַבּוּשָׁה גְּדוֹלָה יוֹתֵר, וְהַתַּשְׁלוּם גָּדוֹל יוֹתֵר.

וְיָפֶה לִרְאוֹת אֵיךְ כָּל הַשִּׁעוּר מִתְחַבֵּר לְחוּט אֶחָד: זְהִירוּת בִּכְבוֹד הַזּוּלַת (בֹּשֶׁת), זְהִירוּת בַּמָּמוֹן שֶׁל הַזּוּלַת (מִשְׁקָלוֹת), זְהִירוּת בַּלָּשׁוֹן (מִרְיָם) וּזְהִירוּת בָּאֱמוּנָה (עֲמָלֵק).

💛 אַהֲבַת הַשֵּׁם
הַתּוֹרָה מְבַקֶּשֶׁת מִמֶּנִּי לִשְׁמֹר עַל הַפֶּה, עַל הַיָּדַיִם, עַל הַכִּיס וְעַל הַלֵּב. ה׳ רוֹצֶה אוֹתִי שָׁלֵם מִכָּל הַצְּדָדִים – כִּי כָּךְ אוֹהֲבִים 💛

✨ לְסִכּוּם

  • הַחוֹבֵל בַּחֲבֵרוֹ חַיָּב חֲמִשָּׁה תַּשְׁלוּמִים: נֶזֶק, צַעַר, רִפּוּי וְשֶׁבֶת – בְּפָרָשַׁת מִשְׁפָּטִים – וְבֹשֶׁת, שֶׁמִּתְחַדֵּשׁ בְּחֻמַּשׁ דְּבָרִים.
  • רַשִׁ״י: ׳מָמוֹן, דְּמֵי בָּשְׁתּוֹ, הַכֹּל לְפִי הַמְבַיֵּשׁ וְהַמִּתְבַּיֵּשׁ׳. הַתּוֹרָה רוֹאָה בַּבּוּשָׁה נֵזֶק אֲמִתִּי – וּמִכָּאן חֲשִׁיבוּת הַזְּהִירוּת בִּכְבוֹד הַזּוּלַת.
  • אָסוּר לְהַחְזִיק אֶבֶן וָאֶבֶן (מִשְׁקָל) אוֹ אֵיפָה וְאֵיפָה (נֶפַח) – אֲפִלּוּ בַּכִּיס וּבַבַּיִת. ׳אֶבֶן שְׁלֵמָה וָצֶדֶק יִהְיֶה לָּךְ׳, וְהַשָּׂכָר: אֲרִיכוּת יָמִים עַל הָאֲדָמָה.
  • רַשִׁ״י: ׳אִם עָשִׂיתָ כֵּן – לֹא יִהְיֶה לְךָ כְּלוּם׳, וּכְנֶגְדּוֹ ׳אִם עָשִׂיתָ כֵּן – יִהְיֶה לְךָ הַרְבֵּה׳. וְגַם הַיּוֹם: בְּלִי פִּרְסוּם מַטְעֶה וּבְלִי פֵּרוֹת רָעִים מִתַּחַת לַיָּפִים.
  • פָּרָשַׁת עֲמָלֵק בָּאָה מִיָּד אַחֲרֵי הַמִּשְׁקָלוֹת: ׳אִם שִׁקַּרְתָּ בַּמִּדּוֹת וּבַמִּשְׁקָלוֹת – הֱוֵי דּוֹאֵג מִגֵּרוּי הָאוֹיֵב׳.
  • ׳קָרְךָ׳ – לְשׁוֹן מִקְרֶה (עֲמָלֵק בָּא לַעֲקֹר אֶת הָאֱמוּנָה בְּהַשְׁגָּחָה), וְגַם לְשׁוֹן קֹר – מְשַׁל הָאַמְבַּטְיָה הָרוֹתַחַת: הוּא נִכְוָה, אֲבָל גָּרַם לָעַמִּים לְהַפְסִיק לְפַחֵד.
  • שְׁנֵי דְּבָרִים עוֹזְרִים לִזְכֹּר: סְמִיכוּת הַזְּמַן וְחֲשִׁיבוּת הָאֵרוּעַ. פְּסוּקִים י״ז–י״ח מְלַמְּדִים אֶת הַחֲשִׁיבוּת, וּפָסוּק י״ט מַזְהִיר ׳לֹא תִּשְׁכָּח׳.
  • מִצְוַת מְחִיַּת עֲמָלֵק – בְּאֶרֶץ יִשְׂרָאֵל, אַחֲרֵי הַמִּלְחָמוֹת. וְהַיּוֹם: שַׁבַּת זָכוֹר, רַעֲשָׁנִים בְּפוּרִים, וְחִזּוּק הָאֱמוּנָה בַּה׳ וּבְהַשְׁגָּחָתוֹ – הַהֵפֶךְ מִן ה׳מִקְרֶה׳.

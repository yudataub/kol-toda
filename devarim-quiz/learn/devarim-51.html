📜 רַשִׁ״י
׳הִשָּׁמֶר בְּנֶגַע הַצָּרַעַת׳ – שֶׁלֹּא תִתְלֹשׁ סִימָנֵי טֻמְאָה וְלֹא תָקֹץ אֶת הַבַּהֶרֶת.
׳כְּכֹל אֲשֶׁר יוֹרוּ אֶתְכֶם׳ – אִם לְהַסְגִּיר, אִם לְהַחְלִיט, אִם לְטַהֵר.

כְּלוֹמַר: אָסוּר לְהַעֲלִים אֶת סִימָנֵי הַנֶּגַע כְּדֵי ׳לְהִתְחַמֵּק׳ מִן הַדִּין – צָרִיךְ לָלֶכֶת אֶל הַכֹּהֵן וְלַעֲשׂוֹת כְּכָל שֶׁיּוֹרֶה.

וּמִיָּד אַחַר כָּךְ בָּא הַפָּסוּק הַמְּפֻרְסָם: ׳זָכוֹר אֵת אֲשֶׁר עָשָׂה ה׳ אֱלֹהֶיךָ לְמִרְיָם בַּדֶּרֶךְ בְּצֵאתְכֶם מִמִּצְרָיִם׳.

📜 רַשִׁ״י
׳זָכוֹר אֵת אֲשֶׁר עָשָׂה ה׳ אֱלֹהֶיךָ לְמִרְיָם׳ – אִם בָּאתָ לְהִזָּהֵר שֶׁלֹּא תִלְקֶה בְּצָרַעַת, אַל תְּסַפֵּר לְשׁוֹן הָרָע; זְכֹר הֶעָשׂוּי לְמִרְיָם, שֶׁדִּבְּרָה בְּאָחִיהָ וְלָקְתָה בִּנְגָעִים.

מִרְיָם הַנְּבִיאָה הָיְתָה צַדֶּקֶת גְּדוֹלָה, וְהִיא דִּבְּרָה עַל אָחִיהָ מֹשֶׁה – וְאֵינָהּ אֶלָּא אֲחוֹתוֹ הָאוֹהֶבֶת, שֶׁשָּׁמְרָה עָלָיו כְּשֶׁהָיָה תִּינוֹק בַּיְאוֹר. וּבְכָל זֹאת לָקְתָה בְּצָרַעַת.
זֶהוּ עֵרֶךְ שְׁמִירַת הַלָּשׁוֹן: הַתּוֹרָה מְבַקֶּשֶׁת מִכָּל אֶחָד מֵאִתָּנוּ לִזְכֹּר אֶת מִרְיָם לִפְנֵי שֶׁהוּא פּוֹתֵחַ אֶת הַפֶּה עַל חֲבֵרוֹ.

💡 וּמָה הַקֶּשֶׁר לְדִינֵי הַמַּשְׁכּוֹן?
כָּל הַפָּרָשָׁה הַזּוֹ מְלַמֶּדֶת דָּבָר אֶחָד: רְגִישׁוּת לַזּוּלַת.
לֹא לְהִתְפָּרֵץ לַבַּיִת, לֹא לְעַכֵּב שָׂכָר, לֹא לָקַחַת אֶת הָרֵחַיִם – וְלֹא לְדַבֵּר לְשׁוֹן הָרָע.
אֶפְשָׁר לִפְגֹּעַ בְּאָדָם בַּיָּד, וְאֶפְשָׁר לִפְגֹּעַ בּוֹ בַּפֶּה. הַתּוֹרָה אוֹסֶרֶת אֶת שְׁנֵיהֶם.
💛 אַהֲבַת הַשֵּׁם
ה׳ נָתַן לִי פֶּה כְּדֵי לְבָרֵךְ, לִלְמֹד תּוֹרָה וּלְשַׂמֵּחַ חֲבֵרִים. כְּשֶׁאֲנִי שׁוֹמֵר אֶת לְשׁוֹנִי, אֲנִי מַחֲזִיר לוֹ אֶת הַמַּתָּנָה בְּדִיּוּק כְּמוֹ שֶׁהוּא אוֹהֵב 💛

✨ לְסִכּוּם

  • מַשְׁכּוֹן (בַּתּוֹרָה: עֲבוֹט) הוּא חֵפֶץ שֶׁהַלּוֶֹה נוֹתֵן לַמַּלְוֶה כְּבִטָּחוֹן, וְחוֹזֵר אֵלָיו כְּשֶׁהַחוֹב מְשֻׁלָּם. ׳נֹשֶׁה׳ הוּא מִי שֶׁבָּא לִגְבּוֹת אֶת הַחוֹב.
  • ׳לֹא תָבֹא אֶל בֵּיתוֹ... בַּחוּץ תַּעֲמֹד׳ – גַּם כְּשֶׁהַצֶּדֶק אִתְּךָ, אָסוּר לְהִתְפָּרֵץ וְלִבְחֹר מַשְׁכּוֹן בְּכֹחַ. הָעוֹלָם אֵינוֹ הֶפְקֵר, וְיֵשׁ בֵּית דִּין.
  • ׳וְאִם אִישׁ עָנִי הוּא לֹא תִשְׁכַּב בַּעֲבֹטוֹ׳ – כְּסוּת לַיְלָה מַחְזִירִים בָּעֶרֶב, וּכְסוּת יוֹם בַּבֹּקֶר. הֶעָנִי שׁוֹכֵב בְּשַׂלְמָתוֹ וּמְבָרֵךְ, וְגַם אִם אֵינוֹ מְבָרֵךְ – ׳וּלְךָ תִּהְיֶה צְדָקָה׳.
  • ׳בְּיוֹמוֹ תִתֵּן שְׂכָרוֹ׳ – אָסוּר לְעַכֵּב שְׂכַר פּוֹעֵל אֲפִלּוּ לַיְלָה אֶחָד. גּוֹזֵל לוֹקֵחַ מָה שֶׁבִּרְשׁוּת חֲבֵרוֹ; עוֹשֵׁק מְעַכֵּב אֶצְלוֹ מָה שֶׁשַּׁיָּךְ לַשֵּׁנִי. רַשִׁ״י: הָעוֹשֵׁק עָנִי עוֹבֵר בִּשְׁנֵי לָאוִין.
  • ׳וְאֵלָיו הוּא נֹשֵׂא אֶת נַפְשׁוֹ׳ – הַפּוֹעֵל מִסְתַּכֵּן בִּשְׁבִיל שְׂכָרוֹ, וְאָסוּר לְהִתְעַלֵּם מִכָּךְ. וְאִם יִקְרָא עָלֶיךָ אֶל ה׳ – מְמַהֲרִין לְהִפָּרַע.
  • עוֹד בַּפֶּרֶק: אֵין מְמַשְׁכְּנִים רֵחַיִם וָרֶכֶב, אָסוּר גּוֹנֵב נֶפֶשׁ, ׳אִישׁ בְּחֶטְאוֹ׳ – אֵין מְקַבְּלִים עֵדוּת קְרוֹבִים – וְזָכוֹר לְמִרְיָם: אַל תְּסַפֵּר לְשׁוֹן הָרָע. הַכֹּל מִצְוָה אַחַת גְּדוֹלָה: לִהְיוֹת רָגִישׁ לַזּוּלַת, כְּמוֹ שֶׁה׳ רָגִישׁ אֵלֵינוּ.

הַמַּסָּע בְּעֵבֶר הַיַּרְדֵּן הַמִּזְרָחִי

הַחֹמֶר הַמָּלֵא: אֱדוֹם, מוֹאָב וְעַמּוֹן – מַה שֶּׁשֶּׁלָּהֶם אֵינוֹ שֶׁלְּךָ (דְּבָרִים ב׳, א–כ״ג)
🎯 לַחִידוֹן🏠 כָּל הַחִידוֹנִים
בַּשִּׁעוּר הַזֶּה אֲנַחְנוּ פּוֹתְחִים אֶת פֶּרֶק ב׳, וְעוֹלִים עַל הַמַּפָּה: בְּנֵי יִשְׂרָאֵל צוֹעֲדִים בְּעֵבֶר הַיַּרְדֵּן הַמִּזְרָחִי, בַּדֶּרֶךְ אֶל אֶרֶץ יִשְׂרָאֵל, וּפוֹגְשִׁים בַּדֶּרֶךְ שָׁלֹשׁ אֻמּוֹת: אֱדוֹם, מוֹאָב וְעַמּוֹן.
בְּכָל פַּעַם שֶׁהֵם מִתְקָרְבִים לְאַחַת מֵהֶן, ה׳ מַזְהִיר: ׳אַל תִּתְגָּרוּ בָם׳. לָמָּה? כִּי הָאֲדָמָה הַזּוֹ הִיא יְרֻשָּׁה שֶׁה׳ נָתַן לָהֶם – וּמַה שֶּׁשֶּׁלָּהֶם אֵינוֹ שֶׁלְּךָ.
נִלְמַד עַל הַהַקָּפָה שֶׁל הַר שֵׂעִיר, עַל הַחוֹבָה לִקְנוֹת אֹכֶל וּמַיִם בְּכֶסֶף, עַל הַכָּרַת הַטּוֹב שֶׁמַּגִּיעָה לְזַרְעוֹ שֶׁל עֵשָׂו וְשֶׁל לוֹט, וְעַל הַמִּלָּה שֶׁחוֹזֶרֶת שׁוּב וָשׁוּב בַּפְּסוּקִים.
וּבְעִקָּר נִרְאֶה אֵיךְ ה׳ אוֹהֵב אוֹתָנוּ: הוּא מְלַוֶּה אוֹתָנוּ אַרְבָּעִים שָׁנָה בַּמִּדְבָּר וְלֹא מַחְסִיר לָנוּ דָּבָר, וְהוּא מְלַמֵּד אוֹתָנוּ לִהְיוֹת עַם יָשָׁר, שֶׁאֵינוֹ לוֹקֵחַ בְּכֹחַ מַה שֶּׁאֵינוֹ שֶׁלּוֹ.
בַּשִּׁעוּר: 🔁 מִשְׁנֵה תּוֹרָה – חוֹזְרִים אֶל פָּרָשַׁת חֻקַּת ⛰️ ׳וַנָּסָב אֶת הַר שֵׂעִיר יָמִים רַבִּים׳ – וְעַכְשָׁו: פְּנוּ צָפֹנָה 🏔️ אֱדוֹם – בְּנֵי עֵשָׂו: ׳אַל תִּתְגָּרוּ בָם׳ 💰 אֹכֶל וּמַיִם בְּכֶסֶף – וְהַכָּרַת הַטּוֹב 🏜️ מוֹאָב – בְּנֵי לוֹט: ׳אַל תָּצַר וְאַל תִּתְגָּר׳ ⏳ נַחַל זֶרֶד – שְׁלֹשִׁים וּשְׁמוֹנֶה שָׁנָה בְּפָסוּק אֶחָד ⛺ עַמּוֹן – וְהַהֶבְדֵּל הַקָּטָן בֵּין מוֹאָב לְעַמּוֹן 🔑 יְרֻשָּׁה – עֵץ הַמִּשְׁפָּחָה וּשְׂכָרוֹ שֶׁל לוֹט 🗺️ לֹא ׳הַמַּעֲבָר׳ אֶלָּא ׳הַמַּעֲקָף׳ – מַה שֶּׁשֶּׁלָּהֶם אֵינוֹ שֶׁלְּךָ

🔁מִשְׁנֵה תּוֹרָה – חוֹזְרִים אֶל פָּרָשַׁת חֻקַּת

בַּשִּׁעוּרִים הַקּוֹדְמִים לָמַדְנוּ אֶת פֶּרֶק א׳, וְעַכְשָׁו אֲנַחְנוּ עוֹבְרִים לְפֶרֶק ב׳. גַּם כָּאן מֹשֶׁה עוֹשֶׂה אֶת מַה שֶּׁהוּא עוֹשֶׂה בְּכָל חֻמַּשׁ דְּבָרִים, שֶׁנִּקְרָא ׳מִשְׁנֵה תּוֹרָה׳: הוּא חוֹזֵר עַל אֵרוּעִים שֶׁכְּבָר כְּתוּבִים בַּחֻמָּשִׁים הַקּוֹדְמִים.

הַפַּעַם הוּא חוֹזֵר עַל הַמַּסָּע בְּעֵבֶר הַיַּרְדֵּן הַמִּזְרָחִי – הַדֶּרֶךְ שֶׁבָּהּ הָלְכוּ בְּנֵי יִשְׂרָאֵל מַמָּשׁ לִפְנֵי הַכְּנִיסָה לָאָרֶץ.
בְּאֵיזֶה חֻמָּשׁ לָמַדְנוּ עַל הַמַּסָּע הַזֶּה? בְּחֻמַּשׁ בַּמִּדְבָּר, בְּפָרָשַׁת חֻקַּת. שָׁם מְסֻפָּר מַה קָּרָה בְּפֹעַל, וְכָאן מֹשֶׁה מַזְכִּיר אֶת הַדְּבָרִים שׁוּב.

מָתַי זֶה קָרָה? בַּשָּׁנָה הָאַרְבָּעִים לִיצִיאַת מִצְרַיִם – כְּלוֹמַר, רַק לִפְנֵי חֳדָשִׁים סְפוּרִים!
וְזֶה מְיֻחָד: אֶת מַעֲמַד הַר סִינַי אוֹ אֶת חֵטְא הַמְרַגְּלִים רֹב הַשּׁוֹמְעִים לֹא זָכְרוּ, כִּי הָיוּ יְלָדִים קְטַנִּים אוֹ עוֹד לֹא נוֹלְדוּ. אֲבָל אֶת הַמַּסָּע הַזֶּה הַבָּנִים חָווּ בְּעַצְמָם – הֵם הָלְכוּ בַּדֶּרֶךְ הַזּוֹ בְּרַגְלֵיהֶם.

💡 לַמּוֹרֶה: לְאַתֵּר אֶת הַמָּקוֹר בְּבַמִּדְבָּר
כְּדַאי לִפְתֹּחַ חֻמַּשׁ בַּמִּדְבָּר בְּפָרָשַׁת חֻקַּת (פֶּרֶק כ׳), וְלִמְצֹא יַחַד עִם הַתַּלְמִידִים אֶת סִפּוּר הַשְּׁלִיחִים אֶל מֶלֶךְ אֱדוֹם.
כָּךְ מִתְבָּרֵר שֶׁפֶּרֶק ב׳ בִּדְבָרִים אֵינוֹ סִפּוּר חָדָשׁ, אֶלָּא מַבָּט נוֹסָף עַל אוֹתוֹ מַסָּע עַצְמוֹ.

וְלָמָּה מֹשֶׁה חוֹזֵר עַל מַשֶּׁהוּ כָּל כָּךְ טָרִי? כִּי הוּא רוֹצֶה שֶׁהַדּוֹר הַנִּכְנָס לָאָרֶץ לֹא יִרְאֶה בַּמַּסָּע סְתָם הֲלִיכָה אֲרֻכָּה בְּשֶׁמֶשׁ, אֶלָּא דֶּרֶךְ שֶׁה׳ הוֹבִיל בָּהּ. כְּשֶׁיּוֹדְעִים מִי מַנְהִיג אוֹתָנוּ – גַּם הַדֶּרֶךְ הַקָּשָׁה נִרְאֵית אַחֶרֶת, וְהַלֵּב מִתְמַלֵּא אַהֲבָה וְהוֹדָיָה.

הַמַּעֲקָף הָאָרֹךְ הַזֶּה הֶאֱרִיךְ אֶת הַדֶּרֶךְ בְּמֵאוֹת קִילוֹמֶטְרִים, בְּשֶׁטַח הָרִים קָשֶׁה וְחַם מְאוֹד. וּבְכָל זֹאת הָלְכוּ בּוֹ, בְּלִי לְהִתְלוֹנֵן עַל הַמֶּרְחָק.
לָכֵן הַכּוֹתֶרֶת הַנְּכוֹנָה לַשִּׁעוּר אֵינָהּ ׳הַמַּעֲבָר בְּאַרְצוֹת אֱדוֹם, מוֹאָב וְעַמּוֹן׳, אֶלָּא ׳הַמַּעֲקָף׳ – כִּי בְּאַרְצוֹת הָאֵלֶּה כְּלָל לֹא עָבְרוּ.

וְזֶה הַלֶּקַח הַגָּדוֹל: אִלּוּ בְּנֵי יִשְׂרָאֵל הָיוּ כּוֹבְשִׁים כָּל עַם שֶׁמְּעַכֵּב אוֹתָם בַּדֶּרֶךְ, הֵם הָיוּ מַתְחִילִים לַחְשֹׁב ׳כֹּחִי וְעֹצֶם יָדִי׳.
אֲבָל ה׳ הוּא שֶׁקּוֹבֵעַ מָה מֻתָּר וּמָה אָסוּר: אֶרֶץ יִשְׂרָאֵל שַׁיֶּכֶת לָנוּ, וּמַה שֶּׁאֵינוֹ שֶׁלָּנוּ – אֵינוֹ שֶׁלָּנוּ, גַּם אִם יֵשׁ לָנוּ כֹּחַ לָקַחַת אוֹתוֹ.

וְלֶקַח נוֹסָף, עָדִין וְיָפֶה: עֵשָׂו וְלוֹט לֹא הָיוּ צַדִּיקִים גְּדוֹלִים. וּבְכָל זֹאת אִי אֶפְשָׁר לִמְחֹק אֶת הַמַּעֲשִׂים הַטּוֹבִים שֶׁעָשׂוּ – עֵשָׂו שֶׁכִּבֵּד אֶת אָבִיו, וְלוֹט שֶׁהָיָה נֶאֱמָן לְאַבְרָהָם.
אֵין אָדָם שֶׁכֻּלּוֹ רַע, וְהַתּוֹרָה מְלַמֶּדֶת אוֹתָנוּ לִרְאוֹת אֶת הַטּוֹב שֶׁיֵּשׁ בְּכָל אֶחָד.

מַתְחִילִים פָּרָשָׁה חֲדָשָׁה! אַחֲרֵי דְּבָרִים, וָאֶתְחַנַּן, עֵקֶב וּרְאֵה – הִגַּעְנוּ לַפָּרָשָׁה הַחֲמִישִׁית: פָּרָשַׁת שֹׁפְטִים.
הַפָּרָשָׁה נִפְתַּחַת בְּשָׁלֹשׁ מִילִים גְּדוֹלוֹת: ׳שֹׁפְטִים וְשֹׁטְרִים תִּתֶּן לְךָ בְּכָל שְׁעָרֶיךָ׳. בְּנֵי יִשְׂרָאֵל עוֹמְדִים לְהִכָּנֵס לָאָרֶץ, לְהִתְפַּזֵּר בְּכָל רַחֲבֵי אֶרֶץ יִשְׂרָאֵל וְלִבְנוֹת עָרִים – וּבְכָל עִיר וָעִיר צָרִיךְ שֶׁיִּהְיֶה בֵּית דִּין שֶׁשּׁוֹפֵט מִשְׁפַּט צֶדֶק.
נִלְמַד מַה הַהֶבְדֵּל בֵּין שׁוֹפֵט לְשׁוֹטֵר, אֶת שָׁלֹשׁ הָאַזְהָרוֹת שֶׁל הַדַּיָּן, אֶת חֲמִשָּׁה פֵּרוּשֵׁי רַשִׁ״י שֶׁבְּלִעֲדֵיהֶם אִי אֶפְשָׁר לְהָבִין אֶת הַפָּסוּק, וְסִפּוּרִים מַדְהִימִים עַל חֲכָמִים שֶׁנִּזְהֲרוּ מֵאֲבַק שֹׁחַד.
וּבְעִקָּר נִרְאֶה כַּמָּה ה׳ אוֹהֵב אוֹתָנוּ: הוּא לֹא הִשְׁאִיר אוֹתָנוּ בְּלִי צֶדֶק. הוּא נָתַן לָנוּ מִשְׁפָּט יָשָׁר כְּדֵי שֶׁנּוּכַל לִחְיוֹת בְּשֶׁקֶט וּבְבִטָּחוֹן בָּאָרֶץ שֶׁנָּתַן לָנוּ 💛
בַּשִּׁעוּר: ⚖️ פָּרָשַׁת שֹׁפְטִים – שׁוֹפֵט וְשׁוֹטֵר, שְׁנֵי תַּפְקִידִים 🏰 ׳בְּכָל שְׁעָרֶיךָ׳ – בֵּית דִּין בְּכָל עִיר וָעִיר 🎯 לְמִי מְדַבֵּר כָּל פָּסוּק? שְׁלוֹשָׁה נִמְעָנִים 📏 ׳לֹא תַטֶּה מִשְׁפָּט׳ – כְּמַשְׁמָעוֹ 🙂 ׳לֹא תַכִּיר פָּנִים׳ – אַף בִּשְׁעַת הַטְּעָנוֹת 💰 ׳וְלֹא תִקַּח שֹׁחַד׳ – אֲפִלּוּ לִשְׁפֹּט צֶדֶק 📚 חֲכָמִים שֶׁנִּזְהֲרוּ מֵאֲבַק שֹׁחַד 🏃 ׳צֶדֶק צֶדֶק תִּרְדֹּף׳ – וְהַשָּׂכָר: לִחְיוֹת בָּאָרֶץ 🏛️ אַרְבַּע הָרָשֻׁיּוֹת שֶׁל פָּרָשַׁת שֹׁפְטִים 🌳 אֲשֵׁרָה וּמַצֵּבָה – מִיָּד אַחֲרֵי הַשּׁוֹפְטִים 🎶 עַל פֶּה וּלְזִכָּרוֹן – הַפְּסוּקִים וְחֲמֵשֶׁת פֵּרוּשֵׁי רַשִׁ״י

כְּבָר לְמַדְתֶּם אֶת הַכְּלָל: שֵׁם הַפָּרָשָׁה נִמְצָא בַּפָּסוּק הָרִאשׁוֹן שֶׁלָּהּ. וְכָאן הַפָּסוּק פּוֹתֵחַ: ׳שֹׁפְטִים וְשֹׁטְרִים תִּתֶּן לְךָ בְּכָל שְׁעָרֶיךָ׳ – וּמִכָּאן הַשֵּׁם ׳פָּרָשַׁת שֹׁפְטִים׳.
וְשִׂימוּ לֵב: הַתּוֹרָה מוֹנָה שְׁנֵי תַּפְקִידִים – שׁוֹפְטִים וְשׁוֹטְרִים. זֶה לֹא אוֹתוֹ אָדָם וְלֹא אוֹתוֹ תַּפְקִיד.

📜 רַשִׁ״י
׳שֹׁפְטִים וְשֹׁטְרִים׳ – שׁוֹפְטִים, דַּיָּנִים הַפּוֹסְקִים אֶת הַדִּין; וְשׁוֹטְרִים, הָרוֹדִין אֶת הָעָם אַחַר מִצְוָתָם, שֶׁמַּכִּין וְכוֹפְתִין בְּמַקֵּל וּבִרְצוּעָה עַד שֶׁיְּקַבֵּל עָלָיו אֶת דִּין הַשּׁוֹפֵט.

כְּלוֹמַר: הַשּׁוֹפֵט הוּא הַדַּיָּן שֶׁיּוֹשֵׁב בְּבֵית הַדִּין, שׁוֹמֵעַ אֶת שְׁנֵי הַצְּדָדִים וּפוֹסֵק מַהוּ דִּין הָאֱמֶת. הַשּׁוֹטֵר הוּא הָאוֹכֵף: הוּא דּוֹאֵג שֶׁמַּה שֶּׁהַשּׁוֹפֵט פָּסַק בֶּאֱמֶת יִתְקַיֵּם – בָּרְחוֹב, בַּשּׁוּק וּבְכָל מָקוֹם.
הָעָם צָרִיךְ אֶת שְׁנֵיהֶם: גַּם אֲנָשִׁים שֶׁיּוֹרוּ אֶת דִּין הָאֱמֶת, וְגַם אֲנָשִׁים שֶׁיִּדְאֲגוּ שֶׁהַדִּין הַזֶּה אָכֵן נִשְׁמָר.

בְּסוֹף הַפָּסוּק מוֹפִיעָה הַמַּטָּרָה: ׳וְשָׁפְטוּ אֶת הָעָם מִשְׁפַּט צֶדֶק׳.

📜 רַשִׁ״י
׳וְשָׁפְטוּ אֶת הָעָם וְגוֹ׳׳ – מַנֵּה דַּיָּנִין מֻמְחִים וְצַדִּיקִים לִשְׁפּוֹט צֶדֶק.

דַּיָּן צָרִיךְ שְׁנֵי דְּבָרִים: מֻמְחִיּוּת – שֶׁיֵּדַע אֶת הַהֲלָכָה וְאֶת הַדִּין, וְצִדְקוּת – שֶׁיִּהְיֶה אִישׁ יָשָׁר שֶׁאוֹהֵב אֶת הָאֱמֶת. חָכָם שֶׁאֵינוֹ יָשָׁר, אוֹ יָשָׁר שֶׁאֵינוֹ יוֹדֵעַ – אֵינָם יְכוֹלִים לִשְׁפֹּט מִשְׁפַּט צֶדֶק.

💛 אַהֲבַת הַשֵּׁם
ה׳ אוֹהֵב כָּל אָדָם מֵעַמּוֹ, וְלָכֵן הוּא לֹא רוֹצֶה שֶׁיִּהְיֶה אֲפִלּוּ יֶלֶד אֶחָד שֶׁנֶּעֱשָׂה לוֹ עָוֶל בְּלִי שֶׁיִּהְיֶה מִי שֶׁיַּעֲזֹר לוֹ. לָכֵן הוּא צִוָּה עָלֵינוּ לְמַנּוֹת שׁוֹפְטִים וְשׁוֹטְרִים – כְּמוֹ אַבָּא שֶׁדּוֹאֵג שֶׁבַּבַּיִת שֶׁלּוֹ יִהְיֶה טוֹב לְכֻלָּם 💛

מֹשֶׁה אוֹמֵר אֶת הַנְּאוּם הַזֶּה בְּעֵבֶר הַיַּרְדֵּן. בְּנֵי יִשְׂרָאֵל עֲתִידִים בְּקָרוֹב מְאוֹד לַעֲבֹר אֶת הַיַּרְדֵּן, לְהִתְפַּזֵּר בְּכָל רַחֲבֵי הָאָרֶץ וְלִבְנוֹת עָרִים – בְּכָל שֵׁבֶט וָשֵׁבֶט: בְּקָדֵשׁ, בְּחֶבְרוֹן, בִּבְאֵר שֶׁבַע וְעוֹד וְעוֹד.

בִּימֵי קֶדֶם כָּל עִיר הָיְתָה מֻקֶּפֶת חוֹמָה, וּבְכָל חוֹמָה הָיָה שַׁעַר הָעִיר. לָכֵן כְּשֶׁכָּתוּב בַּתּוֹרָה ׳בְּכָל שְׁעָרֶיךָ׳ – הַכַּוָּנָה בְּכָל הֶעָרִים שֶׁלְּךָ.

׳בְּכָל שְׁעָרֶיךָ׳ – בְּכָל עִיר וָעִיר.
׳לִשְׁבָטֶיךָ׳ – מוּסָב עַל ׳תִּתֶּן לְךָ׳: שׁוֹפְטִים וְשׁוֹטְרִים תִּתֶּן לְךָ לִשְׁבָטֶיךָ בְּכָל שְׁעָרֶיךָ.
׳לִשְׁבָטֶיךָ׳ – מְלַמֵּד שֶׁמּוֹשִׁיבִין דַּיָּנִין לְכָל שֵׁבֶט וָשֵׁבֶט וּבְכָל עִיר וָעִיר.

כְּלוֹמַר: שְׁתֵּי חֲלוּקוֹת. בֵּית דִּין לְכָל שֵׁבֶט, וְגַם בֵּית דִּין בְּכָל עִיר. אִם נִכָּנֵס לְעִיר שֶׁל בְּנֵי יִשְׂרָאֵל בְּאֶרֶץ יִשְׂרָאֵל – נִמְצָא בָּהּ בֵּית דִּין. זֶה הַסִּימָן הַיָּקָר שֶׁל עַם יִשְׂרָאֵל בְּאַרְצוֹ.

💡 לָמָּה דַּוְקָא בַּשַּׁעַר?
בְּשַׁעַר הָעִיר הָיְתָה רְחָבָה גְּדוֹלָה, וְדַרְכָּהּ כָּל אֶחָד יָצָא וּבָא. בֵּית הַדִּין יָשַׁב בַּשַּׁעַר – בְּמָקוֹם הֲכִי מֶרְכָּזִי – כְּדֵי שֶׁכָּל מִי שֶׁנִּכְנָס לָעִיר יִשְׁמַע מִיָּד מָה הַכְּלָלִים הַנּוֹהֲגִים בָּהּ.
יֵשׁ בָּזֶה גַּם מַשְׁמָעוּת מַעֲשִׂית (קַל לְהַגִּיעַ) וְגַם סִמְלִית: הַצֶּדֶק עוֹמֵד בַּכְּנִיסָה לָעִיר, לְעֵינֵי כֻּלָּם.
🤔 חִידָה: מָה קָרָה בְּשַׁעַר הָעִיר?
בַּתַּנַ״ךְ אֵרוּעִים רַבִּים קָרוּ בַּשַּׁעַר:
אַבְרָהָם קוֹנֶה אֶת מְעָרַת הַמַּכְפֵּלָה (בְּרֵאשִׁית כ״ג).
בֹּעַז עוֹלֶה הַשַּׁעַר בְּעִנְיַן רוּת (רוּת ד׳).
כִּנּוּס אַנְשֵׁי שְׁכֶם (בְּרֵאשִׁית ל״ד).
מַלְכֵי יִשְׂרָאֵל וִיהוּדָה יוֹשְׁבִים בַּשַּׁעַר לִפְנֵי הַמִּלְחָמָה בַּאֲרָם (מְלָכִים א׳ כ״ב).
וְעוֹד: לוֹט יוֹשֵׁב בְּשַׁעַר סְדוֹם, וּמָרְדְּכַי יוֹשֵׁב בְּשַׁעַר הַמֶּלֶךְ.

ה׳ נָתַן לָנוּ אֶרֶץ, וְהוּא רוֹצֶה שֶׁבְּכָל עִיר בָּהּ יִהְיֶה צֶדֶק – מֵהַגָּדוֹלָה וְעַד הַקְּטַנָּה. כְּשֶׁאֲנִי מְטַיֵּל בְּעִיר בְּאֶרֶץ יִשְׂרָאֵל אֲנִי נִזְכָּר: ה׳ הִבְטִיחַ אֶת הָאָרֶץ הַזּוֹ לַאֲבוֹתַי, וְנָתַן לָהּ גַּם לֵב שֶׁל צֶדֶק 💛

׳לֹא תַטֶּה מִשְׁפָּט, לֹא תַכִּיר פָּנִים, וְלֹא תִקַּח שֹׁחַד, כִּי הַשֹּׁחַד יְעַוֵּר עֵינֵי חֲכָמִים וִיסַלֵּף דִּבְרֵי צַדִּיקִם׳ (פָּסוּק י״ט).

🤔 עֵצָה לְשִׁנּוּן
פָּסוּק י״ט בָּנוּי יָפֶה לְזִכָּרוֹן: שָׁלֹשׁ אַזְהָרוֹת קְצָרוֹת (לֹא תַטֶּה, לֹא תַכִּיר, לֹא תִקַּח), וְאַחֲרֵיהֶן נִמּוּק אֶחָד (כִּי הַשֹּׁחַד יְעַוֵּר...).
אֶפְשָׁר לוֹמַר אוֹתוֹ בְּקֶצֶב, אוֹ לְצַיֵּר שְׁלוֹשָׁה סְמָלִים: מֹאזְנַיִם, שְׁתֵּי פָּנִים, וּמַטְבֵּעַ.

וְהִנֵּה חֲמֵשֶׁת פֵּרוּשֵׁי רַשִׁ״י שֶׁל הַפָּסוּק, בִּקְצָרָה:
׳לֹא תַטֶּה מִשְׁפָּט׳ – כְּמַשְׁמָעוֹ.
׳לֹא תַכִּיר פָּנִים׳ – אַף בִּשְׁעַת הַטְּעָנוֹת.
׳וְלֹא תִקַּח שֹׁחַד׳ – אֲפִלּוּ לִשְׁפּוֹט צֶדֶק.
׳כִּי הַשֹּׁחַד יְעַוֵּר׳ – אִי אֶפְשָׁר שֶׁלֹּא יַטֶּה אֶת לִבּוֹ אֶצְלוֹ.
׳דִּבְרֵי צַדִּיקִם׳ – דְּבָרִים הַמְּצֻדָּקִים, מִשְׁפְּטֵי אֱמֶת.
וּבַפָּסוּק הַבָּא: ׳צֶדֶק צֶדֶק תִּרְדֹּף׳ – הַלֵּךְ אַחַר בֵּית דִּין יָפֶה.

💡 כְּלָל בַּכִּתָּה – לְמָה הוּא נָחוּץ?
חִשְׁבוּ עַל כְּלָל אֶחָד שֶׁקַּיָּם בַּכִּתָּה אוֹ בְּבֵית הַסֵּפֶר, וְנַמְּקוּ לָמָּה הוּא נָחוּץ.
לְמָשָׁל: אָסוּר לְדַבֵּר בְּאֶמְצַע הַשִּׁעוּר – כִּי הַדִּבּוּרִים מַפְרִיעִים לַמּוֹרֶה לְלַמֵּד וְלַתַּלְמִידִים לִלְמֹד. בְּלִי הַכְּלָל הַזֶּה אִי אֶפְשָׁר לִלְמֹד.
וְכָךְ גַּם בְּעַם שָׁלֵם: בְּלִי שׁוֹפְטִים שֶׁפּוֹסְקִים וּבְלִי שׁוֹטְרִים שֶׁדּוֹאֲגִים שֶׁהַדִּין יִתְקַיֵּם – הָיְתָה אַנְדְּרָלָמוּסְיָה. חֲזַ״ל אָמְרוּ: ׳אִלְמָלֵא מוֹרָאָהּ שֶׁל מַלְכוּת, אִישׁ אֶת רֵעֵהוּ חַיִּים בָּלָעוּ׳.
💛 אַהֲבַת הַשֵּׁם
כְּשֶׁאֲנִי אוֹמֵר אֶת הַפְּסוּקִים הָאֵלֶּה בְּעַל פֶּה, הֵם נַעֲשִׂים שֶׁלִּי – חֲקוּקִים בַּלֵּב. וְזוֹ אַהֲבָה: ה׳ נָתַן לִי מִילִים שֶׁאוּכַל לָשֵׂאת אִתִּי לְכָל מָקוֹם, וּבְכָל פַּעַם שֶׁאֲנִי אוֹמֵר אוֹתָן אֲנִי קָרוֹב אֵלָיו 💛

✨ לְסִכּוּם

  • פָּרָשַׁת שֹׁפְטִים נִקְרֵאת עַל שֵׁם הַפָּסוּק הָרִאשׁוֹן. רַשִׁ״י: שׁוֹפְטִים – דַּיָּנִים הַפּוֹסְקִים אֶת הַדִּין; שׁוֹטְרִים – הָרוֹדִין אֶת הָעָם אַחַר מִצְוָתָם, שֶׁדּוֹאֲגִים שֶׁהַדִּין יִתְקַיֵּם.
  • ׳בְּכָל שְׁעָרֶיךָ׳ – בְּכָל עִיר וָעִיר, וְ׳לִשְׁבָטֶיךָ׳ – לְכָל שֵׁבֶט וָשֵׁבֶט. בֵּית הַדִּין יָשַׁב בַּשַּׁעַר, בַּמָּקוֹם הַמֶּרְכָּזִי, כְּדֵי שֶׁכָּל הַנִּכְנָס יֵדַע אֶת כְּלָלֵי הָעִיר.
  • שְׁלוֹשָׁה נִמְעָנִים: פָּסוּק י״ח – הַנְהָגַת הָעָם (לְמַנּוֹת דַּיָּנִים מֻמְחִים וְצַדִּיקִים); פָּסוּק י״ט – הַשּׁוֹפֵט (שָׁלֹשׁ אַזְהָרוֹת); פָּסוּק כ׳ – כֻּלָּם (׳צֶדֶק צֶדֶק תִּרְדֹּף׳).
  • חֲמִשָּׁה רַשִׁ״י: ׳לֹא תַטֶּה מִשְׁפָּט׳ – כְּמַשְׁמָעוֹ; ׳לֹא תַכִּיר פָּנִים׳ – אַף בִּשְׁעַת הַטְּעָנוֹת, שֶׁלֹּא מִסְתַּתְּמִין טְעָנוֹתָיו; ׳וְלֹא תִקַּח שֹׁחַד׳ – אֲפִלּוּ לִשְׁפּוֹט צֶדֶק; ׳כִּי הַשֹּׁחַד יְעַוֵּר׳ – אִי אֶפְשָׁר שֶׁלֹּא יַטֶּה אֶת לִבּוֹ; ׳דִּבְרֵי צַדִּיקִם׳ – מִשְׁפְּטֵי אֱמֶת.
  • סִפּוּרֵי הַחֲכָמִים מְלַמְּדִים שֶׁאֲפִלּוּ אֲבַק שֹׁחַד פּוֹסֵל: שְׁמוּאֵל (יָד בַּגֶּשֶׁר), מָר עוּקְבָא (כִּסּוּי לִכְלוּךְ), וְרַבִּי יִשְׁמָעֵאל בְּרַבִּי יוֹסֵי (פֵּרוֹת יוֹם מֻקְדָּם). וְהָרַב מָרְדְּכַי אֵלִיָּהוּ קָרָא בְּכָל יוֹם פֶּתֶק שֶׁל תְּפִלָּה לִשְׁפֹּט בְּצֶדֶק.
  • ׳צֶדֶק צֶדֶק תִּרְדֹּף׳ – הַלֵּךְ אַחַר בֵּית דִּין יָפֶה, וְהַשָּׂכָר: ׳לְמַעַן תִּחְיֶה וְיָרַשְׁתָּ אֶת הָאָרֶץ׳. יְרֻשַּׁת הָאָרֶץ תְּלוּיָה בְּמִשְׁפַּט צֶדֶק, וּכְדִבְרֵי יְשַׁעְיָהוּ: ׳צִיּוֹן בְּמִשְׁפָּט תִּפָּדֶה׳.
  • אַחַר כָּךְ: אִסּוּר אֲשֵׁרָה (אַף מִשְּׁעַת נְטִיעָה, וְאַף נְטִיעַת אִילָן בְּהַר הַבַּיִת) וְאִסּוּר מַצֵּבָה – שֶׁהָיְתָה אֲהוּבָה בִּימֵי הָאָבוֹת, וְנִשְׂנְאָה מִשֶּׁעֲשָׂאוּהָ הַכְּנַעֲנִים חֹק לַעֲבוֹדָה זָרָה.
  • וּבַפָּרָשָׁה אַרְבַּע רָשֻׁיּוֹת: שׁוֹפְטִים וְשׁוֹטְרִים, כֹּהֲנִים וּבֵית דִּין גָּדוֹל, מֶלֶךְ וְנָבִיא. וְכָל זֶה מֵאַהֲבַת ה׳ אוֹתָנוּ – שֶׁרוֹצֶה שֶׁנֵּשֵׁב בְּאַרְצֵנוּ בְּצֶדֶק, בְּשָׁלוֹם וּבְבִטְחָה.

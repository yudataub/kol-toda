בַּשִּׁעוּר הַזֶּה נִלְמַד שְׁלוֹשָׁה פְּסוּקִים בִּלְבַד – דְּבָרִים י״א, י׳–י״ב – אֲבָל יֵשׁ בָּהֶם אֶחָד הַדְּבָרִים הַגְּדוֹלִים בְּיוֹתֵר שֶׁהַתּוֹרָה מְלַמֶּדֶת עַל אַרְצֵנוּ.
הַתּוֹרָה מַשְׁוָה בֵּין אֶרֶץ מִצְרַיִם, שֶׁמִּמֶּנָּה יָצְאוּ בְּנֵי יִשְׂרָאֵל, לְבֵין אֶרֶץ יִשְׂרָאֵל, שֶׁאֵלֶיהָ הֵם הוֹלְכִים. בְּמִצְרַיִם זוֹרֵם הַנִּילוּס וּמַשְׁקֶה אֶת הַשָּׂדוֹת בְּקַלּוּת; אֶרֶץ יִשְׂרָאֵל הִיא אֶרֶץ הָרִים וּבְקָעוֹת, וְהִיא תְּלוּיָה בַּגֶּשֶׁם.
וְדַוְקָא בָּזֶה נִמְצֵאת הַמַּעֲלָה הַגְּדוֹלָה: ׳תָּמִיד עֵינֵי ה׳ אֱלֹהֶיךָ בָּהּ׳. זוֹ הָאָרֶץ הַיְּחִידָה בָּעוֹלָם שֶׁה׳ מַבִּיט בָּהּ כָּל הַזְּמַן, מֵרֵאשִׁית הַשָּׁנָה וְעַד אַחֲרִית שָׁנָה – אֶרֶץ שֶׁל קֶשֶׁר, שֶׁל תְּפִלָּה וְשֶׁל אַהֲבָה בֵּינֵינוּ לְבֵין ה׳ 💛
בַּשִּׁעוּר: 🧳 מָה נַּגִּיד לַתַּיָּר? – הַשְּׁאֵלָה שֶׁפּוֹתַחַת אֶת הַשִּׁעוּר 👀 וִידַעְתֶּם הַיּוֹם – הַדּוֹר שֶׁרָאָה בְּעֵינָיו 🏞️ לֹא כְּאֶרֶץ מִצְרַיִם – הַנִּילוּס וְהַהַשְׁקָיָה בָּרֶגֶל 🏛️ רַשִׁ״י: ׳אֶלָּא טוֹבָה הֵימֶנָּה׳ – חֶבְרוֹן, צֹעַן וְרַעְמְסֵס ⛰️ אֶרֶץ הָרִים וּבְקָעוֹת – לִמְטַר הַשָּׁמַיִם תִּשְׁתֶּה מָּיִם 😴 רַשִׁ״י: ׳אַתָּה יָשֵׁן עַל מִטָּתְךָ׳ – יִתְרוֹן הַגֶּשֶׁם 👁️ תָּמִיד עֵינֵי ה׳ בָּהּ – אֶרֶץ שֶׁה׳ דּוֹרֵשׁ אוֹתָהּ 🌊 מִפְלַס הַכִּנֶּרֶת וּתְפִלּוֹת הַגְּשָׁמִים 📊 טַבְלַת הַהַשְׁוָאָה – וְסִכּוּם שְׁלוֹשֶׁת הַפְּסוּקִים

🧳מָה נַּגִּיד לַתַּיָּר? – הַשְּׁאֵלָה שֶׁפּוֹתַחַת אֶת הַשִּׁעוּר

דַּמְיְנוּ לְעַצְמְכֶם: אַתֶּם מְטַיְּלִים בִּשְׁבִילֵי אַרְצֵנוּ, וּפִתְאוֹם אַתֶּם פּוֹגְשִׁים תַּיָּר שֶׁהִגִּיעַ מֵאֶרֶץ רְחוֹקָה. הוּא מְסַפֵּר בְּהִתְרַגְּשׁוּת עַל הַמְּקוֹמוֹת שֶׁבָּהֶם הָיָה: ׳אֵיזֶה נוֹפִים! אֵיזֶה נְהָרוֹת גְּדוֹלִים! כַּמָּה מַיִם יֵשׁ שָׁם!׳.
וְהוּא שׁוֹאֵל אֶתְכֶם: בַּמֶּה בְּעֶצֶם אֶרֶץ יִשְׂרָאֵל מְיֻחֶדֶת מִכָּל הָאֲרָצוֹת?

אֶפְשָׁר כַּמּוּבָן לְסַפֵּר לוֹ עַל הַנּוֹפִים הַנִּפְלָאִים שֶׁיֵּשׁ אֶצְלֵנוּ, מֵהַחֶרְמוֹן בַּצָּפוֹן וְעַד הָעֲרָבָה בַּדָּרוֹם. אֲבָל זוֹ עֲדַיִן לֹא תְּשׁוּבָה מְלֵאָה: גַּם בַּאֲרָצוֹת אֲחֵרוֹת יֵשׁ הָרִים יָפִים, יְעָרוֹת וּנְהָרוֹת אֲדִירִים.

הַתּוֹרָה מְגַלָּה לָנוּ מַעֲלָה אַחֶרֶת לְגַמְרֵי, שֶׁאֵין כָּמוֹהָ בְּשׁוּם מָקוֹם בָּעוֹלָם. בִּשְׁלוֹשָׁה פְּסוּקִים בִּלְבַד הִיא מַשְׁוָה בֵּין אֶרֶץ יִשְׂרָאֵל לְאֶרֶץ מִצְרַיִם, וּמְלַמֶּדֶת שֶׁאֶרֶץ יִשְׂרָאֵל הִיא אֶרֶץ הַהַשְׁגָּחָה: הָאָרֶץ שֶׁה׳ מַבִּיט בָּהּ תָּמִיד, וְשֶׁהַיּוֹשְׁבִים בָּהּ קְשׁוּרִים אֵלָיו בְּכָל יוֹם וּבְכָל טִפָּה שֶׁל גֶּשֶׁם.

אֶפְשָׁר לִפְתֹּחַ בְּהַמְחָזָה: תַּלְמִיד אֶחָד הוּא הַתַּיָּר מֵאֶרֶץ שׁוֹפַעַת מַיִם, וְתַלְמִיד אַחֵר הוּא הַמַּדְרִיךְ הַיִּשְׂרְאֵלִי.
אַחֲרֵי שֶׁכָּל הַתְּשׁוּבוֹת הָרְגִילוֹת נֶאֶמְרוּ (נוֹפִים, אֹכֶל, הִיסְטוֹרְיָה), פּוֹתְחִים חֻמָּשׁ – וּמְגַלִּים אֶת הַתְּשׁוּבָה שֶׁל הַתּוֹרָה.

׳וִידַעְתֶּם הַיּוֹם׳ – תְּנוּ לֵב לָדַעַת וּלְהָבִין וּלְקַבֵּל תּוֹכַחְתִּי.
׳כִּי לֹא אֶת בְּנֵיכֶם׳ – אֲנִי מְדַבֵּר עַכְשָׁו, שֶׁיּוּכְלוּ לוֹמַר: אָנוּ לֹא יָדַעְנוּ וְלֹא רָאִינוּ בְּכָל זֶה.

מֹשֶׁה אוֹמֵר: אֲנִי לֹא מְדַבֵּר עַכְשָׁו אֶל הַיְּלָדִים הַקְּטַנִּים, שֶׁיְּכוֹלִים לוֹמַר ׳לֹא הָיִינוּ שָׁם, לֹא רָאִינוּ׳. אֲנִי מְדַבֵּר אֲלֵיכֶם, שֶׁרְאִיתֶם הַכֹּל בְּעֵינֵיכֶם.

וּמָה בְּדִיּוּק הֵם רָאוּ?

אֶת קְרִיעַת יַם סוּף – ׳אֲשֶׁר הֵצִיף אֶת מֵי יַם סוּף עַל פְּנֵיהֶם בְּרָדְפָם אַחֲרֵיכֶם׳.
אֶת הַנִּסִּים בַּמִּדְבָּר – ׳וַאֲשֶׁר עָשָׂה לָכֶם בַּמִּדְבָּר עַד בֹּאֲכֶם עַד הַמָּקוֹם הַזֶּה׳.

מַעֲשֵׂר שֵׁנִי

הַחֹמֶר הַמָּלֵא: פֵּרוֹת שֶׁעוֹלִים לִירוּשָׁלַיִם – וְלֵב שֶׁלּוֹמֵד לִירֹא אֶת ה׳ (פָּרָשַׁת רְאֵה)
🎯 לַחִידוֹן🏠 כָּל הַחִידוֹנִים
פָּרָשַׁת רְאֵה מְלֵאָה בְּמִצְווֹת, וּבַשִּׁעוּר הַזֶּה נִלְמַד מִצְוָה אַחַת מִקְּבוּצַת הַמִּצְווֹת הַתְּלוּיוֹת בָּאָרֶץ – מִצְווֹת שֶׁמְּקַיְּמִים אוֹתָן רַק בְּאֶרֶץ יִשְׂרָאֵל, וְהֵן קְשׁוּרוֹת לָאֲדָמָה, לַיְּבוּל וְלַפֵּרוֹת.
נַכִּיר אֶת הַסֵּדֶר: תְּרוּמָה לַכֹּהֵן, מַעֲשֵׂר רִאשׁוֹן לַלֵּוִי, וְאַחַר כָּךְ מַעֲשֵׂר שֵׁנִי – מַעֲשֵׂר מְיֻחָד, שֶׁהַבְּעָלִים עַצְמָם אוֹכְלִים אוֹתוֹ, אֲבָל רַק בִּירוּשָׁלַיִם.
וּבְעִקָּר נִרְאֶה כַּמָּה ה׳ אוֹהֵב אוֹתָנוּ: הוּא נָתַן לָנוּ אֶרֶץ שֶׁמַּצְמִיחָה פֵּרוֹת, וּבִקֵּשׁ שֶׁפַּעַם בְּשָׁנָה נַעֲלֶה אִתָּם אֶל בֵּיתוֹ – נֹאכַל שָׁם בְּשִׂמְחָה, נִלְמַד תּוֹרָה, וְנִתְחַבֵּר אֵלָיו 💛
בַּשִּׁעוּר: 🌍 הַמִּצְווֹת הַתְּלוּיוֹת בָּאָרֶץ 🎁 תְּרוּמָה לַכֹּהֵן וּמַעֲשֵׂר רִאשׁוֹן לַלֵּוִי 🌾 ׳עַשֵּׂר תְּעַשֵּׂר׳ – מָה עִנְיַן זֶה אֵצֶל זֶה? 🍇 מַעֲשֵׂר שֵׁנִי – הַמַּעֲשֵׂר שֶׁל הַבְּעָלִים 🏙️ בַּמָּקוֹם אֲשֶׁר יִבְחַר – וּ׳לְמַעַן תִּלְמַד לְיִרְאָה׳ 📅 מַחְזוֹר שֶׁל שֶׁבַע שָׁנִים 🪙 ׳וְנָתַתָּה בַּכֶּסֶף׳ – כְּשֶׁהַדֶּרֶךְ רְחוֹקָה 🛒 קוֹנִים בִּירוּשָׁלַיִם – וּשְׂמֵחִים 🧺 מַעֲשַׂר עָנִי וּבִעוּר מַעַשְׂרוֹת

🌍הַמִּצְווֹת הַתְּלוּיוֹת בָּאָרֶץ

אֶפְשָׁר לְמַיֵּן אֶת הַמִּצְווֹת לִקְבוּצוֹת. יֵשׁ קְבוּצָה שֶׁנִּקְרֵאת ׳הַמִּצְווֹת הַתְּלוּיוֹת בָּאָרֶץ׳: מִצְווֹת שֶׁמְּקַיְּמִים אוֹתָן רַק בְּאֶרֶץ יִשְׂרָאֵל. מִי שֶׁגָּר בְּחוּץ לָאָרֶץ אֵינוֹ חַיָּב בָּהֶן.
כָּל הַמִּצְווֹת הָאֵלֶּה קְשׁוּרוֹת לָאֲדָמָה, לַיְּבוּל וְלַפֵּרוֹת – כִּי הָאָרֶץ הִיא אַרְצוֹ שֶׁל ה׳, וְהוּא נָתַן לָנוּ אוֹתָהּ בְּמַתָּנָה.

הַדֻּגְמָה הַמֻּכֶּרֶת בְּיוֹתֵר הִיא מִצְוַת הַשְּׁמִטָּה: אַחַת לְשֶׁבַע שָׁנִים שׁוֹמְטִים אֶת עֲבוֹדַת הָאֲדָמָה – לֹא חוֹרְשִׁים, לֹא זוֹרְעִים, וּמַפְקִירִים אֶת הַשָּׂדוֹת לְכָל אָדָם.

וּמָה קוֹרֶה בְּשֵׁשׁ הַשָּׁנִים הָאֲחֵרוֹת, שֶׁאֵינָן שְׁמִטָּה? בְּכָל אַחַת מֵהֶן מַפְרִישִׁים תְּרוּמוֹת וּמַעַשְׂרוֹת: מִי שֶׁגָּדְלוּ לוֹ פֵּרוֹת בַּשָּׂדֶה, מַפְרִישׁ מֵהֶם חֵלֶק וְנוֹתֵן אוֹתוֹ לְמִי שֶׁצָּרִיךְ – לְמִי שֶׁאֵין לוֹ אֲדָמָה וְאֵין לוֹ נַחֲלָה.

💡 לָמָּה דַּוְקָא בָּאָרֶץ?
אֶרֶץ יִשְׂרָאֵל הִיא הָאָרֶץ שֶׁה׳ בָּחַר בָּהּ. בָּהּ נִמְצָא הַמִּקְדָּשׁ, בָּהּ נִמְצָאִים הַכֹּהֲנִים וְהַלְוִיִּם, וּבָהּ הַפֵּרוֹת עַצְמָם קְדוֹשִׁים.
לָכֵן דַּוְקָא כָּאן יֵשׁ מִצְווֹת שֶׁל אֲדָמָה וְשֶׁל יְבוּל.
💛 אַהֲבַת הַשֵּׁם
כְּשֶׁאֲנִי מַפְרִישׁ מֵהַפֵּרוֹת שֶׁלִּי, אֲנִי נִזְכָּר שֶׁהֵם בִּכְלָל לֹא ׳שֶׁלִּי׳ לְבַד – ה׳ נָתַן אֶת הַגֶּשֶׁם, אֶת הַשֶּׁמֶשׁ וְאֶת הָאֲדָמָה. הַמִּצְווֹת הַתְּלוּיוֹת בָּאָרֶץ הֵן כְּמוֹ לְחִיצַת יָד בֵּינִי לְבֵין ה׳ עַל הָאָרֶץ שֶׁהוּא נָתַן לִי 💛

🎁תְּרוּמָה לַכֹּהֵן וּמַעֲשֵׂר רִאשׁוֹן לַלֵּוִי

📖 פֶּרֶק י״ד, פָּסוּק כ״ז ←
כז וְהַלֵּוִי אֲשֶׁר בִּשְׁעָרֶיךָ לֹא תַעַזְבֶנּוּ כִּי אֵין לוֹ חֵלֶק וְנַחֲלָה עִמָּךְ

בַּשָּׁלָב הָרִאשׁוֹן, מִן הַיְּבוּל נוֹתְנִים לַכֹּהֲנִים וְלַלְוִיִּם. הַכֹּהֲנִים עוֹבְדִים בְּבֵית הַמִּקְדָּשׁ, הַלְוִיִּם מְלַמְּדִים אֶת הָעָם תּוֹרָה – וְלִשְׁנֵיהֶם אֵין נַחֲלָה, אֵין שָׂדֶה מִשֶּׁלָּהֶם.

הַמַּתָּנָה הָרִאשׁוֹנָה הוֹלֶכֶת לַכֹּהֵן, וְהִיא נִקְרֵאת ׳תְּרוּמָה׳.
אַחֲרֶיהָ נוֹתְנִים לַלֵּוִי מַעֲשֵׂר – וְהַמִּלָּה ׳מַעֲשֵׂר׳ מַזְכִּירָה אֶת הַמִּסְפָּר עֶשֶׂר: אֶחָד מִכָּל עֲשָׂרָה. מִתּוֹךְ עֲשָׂרָה תַּפּוּחִים – תַּפּוּחַ אֶחָד לַלֵּוִי. זֶה נִקְרָא מַעֲשֵׂר רִאשׁוֹן.

📖 הַפָּסוּק
׳וְהַלֵּוִי אֲשֶׁר בִּשְׁעָרֶיךָ לֹא תַעַזְבֶנּוּ, כִּי אֵין לוֹ חֵלֶק וְנַחֲלָה עִמָּךְ׳ (י״ד, כ״ז)
📜 רַשִׁ״י
׳וְהַלֵּוִי וְגוֹ׳ לֹא תַעַזְבֶנּוּ׳ – מִלִּתֵּן לוֹ מַעֲשֵׂר רִאשׁוֹן.
׳כִּי אֵין לוֹ חֵלֶק וְנַחֲלָה עִמָּךְ׳ – יָצְאוּ לֶקֶט שִׁכְחָה וּפֵאָה וְהֶפְקֵר, שֶׁאַף הוּא יֵשׁ לוֹ חֵלֶק עִמְּךָ בָּהֶן כָּמוֹךָ, וְאֵינָן חַיָּבִין בְּמַעֲשֵׂר.

כְּלוֹמַר: הַתּוֹרָה מַזְהִירָה שֶׁלֹּא ׳לַעֲזֹב׳ אֶת הַלֵּוִי – אָסוּר לִשְׁכֹּחַ אוֹתוֹ וְלֹא לָתֵת לוֹ אֶת הַמַּעֲשֵׂר הָרִאשׁוֹן.
וּמוֹסִיף רַשִׁ״י: מִמָּה שֶׁאֵין לוֹ חֵלֶק וְנַחֲלָה לוֹמְדִים שֶׁבִּדְבָרִים שֶׁגַּם הַלֵּוִי יָכוֹל לָקַחַת מֵהֶם בְּעַצְמוֹ, כְּמוֹ לֶקֶט שִׁכְחָה וּפֵאָה וּפֵרוֹת שֶׁהֻפְקְרוּ – אֵין חִיּוּב מַעֲשֵׂר כְּלָל.

💡 מִי מְקַבֵּל מָה?
תְּרוּמָה – לַכֹּהֵן.
מַעֲשֵׂר רִאשׁוֹן – לַלֵּוִי.
מַעֲשֵׂר שֵׁנִי – הַבְּעָלִים אוֹכְלִים בִּירוּשָׁלַיִם.
מַעֲשַׂר עָנִי – לַגֵּר, לַיָּתוֹם וְלָאַלְמָנָה.
וּמַעֲשֵׂר הוּא תָּמִיד אֶחָד מִכָּל עֲשָׂרָה.

ה׳ דּוֹאֵג לְכָל אֶחָד מִבָּנָיו: גַּם לְמִי שֶׁאֵין לוֹ שָׂדֶה וְאֵין לוֹ נַחֲלָה. הוּא נָתַן לִי יְבוּל וּבִקֵּשׁ שֶׁאֶזְכֹּר גַּם אֶת הַשְּׁכֵנִים שֶׁלִּי – וְכָךְ הוּא מְלַמֵּד אוֹתִי לֶאֱהֹב כְּמוֹ שֶׁהוּא אוֹהֵב 💛

🌾׳עַשֵּׂר תְּעַשֵּׂר׳ – מָה עִנְיַן זֶה אֵצֶל זֶה?

📖 פֶּרֶק י״ד, פְּסוּקִים כ״א–כ״ב ←
כא לֹא תֹאכְלוּ כָל נְבֵלָה לַגֵּר אֲשֶׁר בִּשְׁעָרֶיךָ תִּתְּנֶנָּה וַאֲכָלָהּ אוֹ מָכֹר לְנָכְרִי כִּי עַם קָדוֹשׁ אַתָּה לַה׳ אֱלֹהֶיךָ לֹא תְבַשֵּׁל גְּדִי בַּחֲלֵב אִמּוֹ כב עַשֵּׂר תְּעַשֵּׂר אֵת כָּל תְּבוּאַת זַרְעֶךָ הַיֹּצֵא הַשָּׂדֶה שָׁנָה שָׁנָה

הַפָּסוּק שֶׁלִּפְנֵי הַמִּצְוָה שֶׁלָּנוּ מִסְתַּיֵּם בְּמִלִּים: ׳לֹא תְבַשֵּׁל גְּדִי בַּחֲלֵב אִמּוֹ׳ – אִסּוּר בָּשָׂר בְּחָלָב. וּמִיָּד אַחֲרָיו בָּא: ׳עַשֵּׂר תְּעַשֵּׂר אֵת כָּל תְּבוּאַת זַרְעֶךָ׳.
רַשִׁ״י שׁוֹאֵל: מָה הַקֶּשֶׁר בֵּין הַשְּׁנַיִם?

📜 רַשִׁ״י
׳עַשֵּׂר תְּעַשֵּׂר׳ – מָה עִנְיַן זֶה אֵצֶל זֶה? אָמַר לָהֶם הַקָּדוֹשׁ בָּרוּךְ הוּא לְיִשְׂרָאֵל: לֹא תִּגְרְמוּ לִי לְבַשֵּׁל גְּדָיִים שֶׁל תְּבוּאָה עַד שֶׁהֵן בִּמְעֵי אִמּוֹתֵיהֶן, שֶׁאִם אֵין אַתֶּם מְעַשְּׂרִים מַעַשְׂרוֹת כָּרָאוּי, כְּשֶׁהוּא סָמוּךְ לְהִתְבַּשֵּׁל אֲנִי מוֹצִיא רוּחַ קָדִים וְהִיא מְשַׁדַּפְתָּן, שֶׁנֶּאֱמַר ׳וּשְׁדֵפָה לִפְנֵי קָמָה׳. וְכֵן לְעִנְיַן בִּכּוּרִים.

רַשִׁ״י מְצַיֵּר תְּמוּנָה: הַשִּׁבֹּלֶת הִיא כְּמוֹ ׳אֵם׳, וְהַגַּרְגֵּר שֶׁגָּדֵל בְּתוֹכָהּ הוּא כְּמוֹ ׳גְּדִי׳ קָטָן שֶׁעוֹד לֹא נוֹלַד.
אִם לֹא מְעַשְּׂרִים כָּרָאוּי, בְּדִיּוּק כְּשֶׁהַתְּבוּאָה קְרוֹבָה לְהַבְשִׁיל יוֹצֵאת רוּחַ קָדִים – רוּחַ חַמָּה וְיָבֵשָׁה מִן הַמִּדְבָּר – וּמְיַבֶּשֶׁת אֶת הַגַּרְגְּרִים בְּתוֹךְ הַשִּׁבֹּלֶת.

ה׳ מְבַקֵּשׁ שֶׁהֶעָנִי, הַגֵּר, הַיָּתוֹם וְהָאַלְמָנָה יֹאכְלוּ וְיִשְׂבְּעוּ – לֹא רַק שֶׁיְּקַבְּלוּ מַשֶּׁהוּ. וּמִי שֶׁנּוֹתֵן כָּךְ, ה׳ מַבְטִיחַ לוֹ בְּרָכָה ׳בְּכָל מַעֲשֵׂה יָדְךָ׳. כַּמָּה אוֹהֵב אוֹתָנוּ אָב שֶׁשָּׂמֵחַ לְבָרֵךְ אֶת מִי שֶׁמְּשַׂמֵּחַ אֲחֵרִים 💛

✨ לְסִכּוּם

  • הַמִּצְווֹת הַתְּלוּיוֹת בָּאָרֶץ מְקֻיָּמוֹת רַק בְּאֶרֶץ יִשְׂרָאֵל: שְׁמִטָּה אַחַת לְשֶׁבַע שָׁנִים, וּבִשְׁאָר הַשָּׁנִים תְּרוּמוֹת וּמַעַשְׂרוֹת.
  • הַסֵּדֶר: תְּרוּמָה לַכֹּהֵן, אַחֲרֶיהָ מַעֲשֵׂר רִאשׁוֹן לַלֵּוִי – אֶחָד מִכָּל עֲשָׂרָה – כִּי אֵין לָהֶם חֵלֶק וְנַחֲלָה.
  • מַעֲשֵׂר שֵׁנִי מְיֻחָד: הַבְּעָלִים זוֹכִים בּוֹ, אֲבָל אוֹכְלִים אוֹתוֹ רַק בַּמָּקוֹם אֲשֶׁר יִבְחַר ה׳ – בִּירוּשָׁלַיִם, ׳לְמַעַן תִּלְמַד לְיִרְאָה אֶת ה׳ אֱלֹהֶיךָ כָּל הַיָּמִים׳.
  • מִי שֶׁגָּר רָחוֹק פּוֹדֶה אֶת הַפֵּרוֹת בְּכֶסֶף – ׳וְצַרְתָּ הַכֶּסֶף בְּיָדְךָ׳ – וּבִירוּשָׁלַיִם קוֹנֶה בּוֹ אֹכֶל שֶׁגָּדֵל מִן הָאָרֶץ וְרָאוּי לְמַאֲכַל אָדָם, וְאוֹכֵל וְשָׂמֵחַ.
  • מַחְזוֹר הַשָּׁנִים: א׳, ב׳, ד׳, ה׳ – מַעֲשֵׂר שֵׁנִי; ג׳ וְו׳ – מַעֲשַׂר עָנִי; ז׳ – שְׁמִטָּה. בְּסוֹף שָׁלֹשׁ שָׁנִים – בִּעוּר מַעַשְׂרוֹת, וְלֶעָנִי נוֹתְנִים ׳כְּדֵי שְׂבִיעָה׳.
  • וְכָל זֶה מֵאַהֲבָה: ה׳ נָתַן לָנוּ אֶרֶץ וּפֵרוֹת, הִזְמִין אוֹתָנוּ לַעֲלוֹת אִתָּם אֶל בֵּיתוֹ בִּירוּשָׁלַיִם, וְלִמֵּד אוֹתָנוּ לְשַׂמֵּחַ גַּם אֶת הַלֵּוִי, הַגֵּר, הַיָּתוֹם וְהָאַלְמָנָה 💛

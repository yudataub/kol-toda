כֹּהֵן מָשׁוּחַ מִלְחָמָה

הַחֹמֶר הַמָּלֵא: ׳שְׁמַע יִשְׂרָאֵל, אַל יֵרַךְ לְבַבְכֶם׳ – אֱמוּנָה לִפְנֵי הַקְּרָב (דְּבָרִים כ׳, א–ט)
🎯 לַחִידוֹן🏠 כָּל הַחִידוֹנִים
בְּנֵי יִשְׂרָאֵל עוֹמְדִים בְּעֵבֶר הַיַּרְדֵּן, עוֹד מְעַט יַעַבְרוּ אֶת הַיַּרְדֵּן אֶל אֶרֶץ כְּנַעַן – וְשָׁם מְחַכִּים לָהֶם עַמִּים וּצְבָאוֹת. לָכֵן הַתּוֹרָה מְלַמֶּדֶת כָּאן דִּינֵי מִלְחָמָה.
אֲבָל הַדָּבָר הָרִאשׁוֹן שֶׁהַתּוֹרָה מְטַפֶּלֶת בּוֹ אֵינוֹ הַנֶּשֶׁק וְאֵינוֹ הַתַּכְנִית הַצְּבָאִית, אֶלָּא הַלֵּב שֶׁל הַלּוֹחֲמִים: ׳לֹא תִירָא מֵהֶם, כִּי ה׳ אֱלֹהֶיךָ עִמָּךְ׳. לִפְנֵי הַקְּרָב נִגָּשׁ כֹּהֵן מָשׁוּחַ מִלְחָמָה וּמְדַבֵּר אֶל הָעָם דִּבְרֵי אֱמוּנָה וְחִזּוּק.
נִלְמַד אֶת אַרְבַּע הָאַזְהָרוֹת שֶׁל הַכֹּהֵן, אֶת ׳נִצְחוֹנוֹ שֶׁל מָקוֹם׳, אֶת אַרְבָּעָה הָאֲנָשִׁים שֶׁחוֹזְרִים הַבַּיְתָה – וּבְעִקָּר נִרְאֶה כַּמָּה ה׳ אוֹהֵב אוֹתָנוּ: הוּא הוֹלֵךְ אִתָּנוּ בְּעַצְמוֹ, וְהוּא שָׂם לֵב גַּם לַצַּעַר הַקָּטָן שֶׁל חַיָּל אֶחָד.
בַּשִּׁעוּר: 🔎 מָה הַנּוֹשֵׂא? מְרַפְרְפִים וּמְחַפְּשִׂים ⚖️ ׳כִּי תֵצֵא לַמִּלְחָמָה עַל אֹיְבֶיךָ׳ – וְלֹא תִירָא 🧑‍🦳 ׳וְנִגַּשׁ הַכֹּהֵן׳ – וְ׳שְׁמַע יִשְׂרָאֵל׳ 🛡️ אַרְבַּע אַזְהָרוֹת – נֶגֶד אַרְבָּעָה קוֹלוֹת 🪨 ׳בְּנִצְחוֹנוֹ שֶׁל מָקוֹם׳ – דָּוִד וְגָלְיָת ⚔️ מִלְחֶמֶת מִצְוָה וּמִלְחֶמֶת רְשׁוּת 🏠 שְׁלֹשָׁה שֶׁחוֹזְרִים הַבַּיְתָה 😨 ׳הַיָּרֵא וְרַךְ הַלֵּבָב׳ – שְׁנֵי פֵּרוּשִׁים 🚩 ׳וּפָקְדוּ שָׂרֵי צְבָאוֹת׳ – הַסֵּדֶר, וְהַמַּכַּבִּים

🔎מָה הַנּוֹשֵׂא? מְרַפְרְפִים וּמְחַפְּשִׂים

📖 פֶּרֶק כ׳, פָּסוּק א׳ ←
א כִּי תֵצֵא לַמִּלְחָמָה עַל אֹיְבֶיךָ וְרָאִיתָ סוּס וָרֶכֶב עַם רַב מִמְּךָ לֹא תִירָא מֵהֶם כִּי ה׳ אֱלֹהֶיךָ עִמָּךְ הַמַּעַלְךָ מֵאֶרֶץ מִצְרָיִם

לִפְנֵי שֶׁנִּכְנָסִים לַפְּסוּקִים, כְּדַאי לָדַעַת שִׁיטָה פְּשׁוּטָה: אֶפְשָׁר לָדַעַת מָה הַנּוֹשֵׂא שֶׁל פָּרָשָׁה עוֹד לִפְנֵי שֶׁקּוֹרְאִים אוֹתָהּ לָעֹמֶק.
אֵיךְ? מִסְתַּכְּלִים בַּפָּסוּק הָרִאשׁוֹן, וּמְחַפְּשִׂים מִלָּה שֶׁחוֹזֶרֶת שׁוּב וָשׁוּב.

בַּפֶּרֶק הַקּוֹדֵם לָמַדְנוּ עַל עָרֵי הַמִּקְלָט וְעַל אִסּוּר הַשָּׂגַת גְּבוּל, וְאַחֲרֵיהֶם בָּאָה פָּרָשָׁה שֶׁלֹּא לָמַדְנוּ. אִם בְּכָל זֹאת רוֹצִים לָדַעַת עַל מָה הִיא – מְרַפְרְפִים, וְרוֹאִים שֶׁחוֹזְרוֹת בָּהּ הַמִּלִּים ׳עֵד׳ וְ׳עֵדִים׳. מִיָּד מְבִינִים: הִיא עוֹסֶקֶת בְּבֵית הַדִּין וּבְעֵדוּת.

וְעַכְשָׁו פֶּרֶק כ׳: ׳כִּי תֵצֵא לַמִּלְחָמָה עַל אֹיְבֶיךָ׳, ׳וְהָיָה כְּקָרָבְכֶם אֶל הַמִּלְחָמָה׳, ׳אַתֶּם קְרֵבִים הַיּוֹם לַמִּלְחָמָה׳, ׳לְהִלָּחֵם לָכֶם׳, ׳שָׂרֵי צְבָאוֹת׳.
הַנּוֹשֵׂא בָּרוּר כַּשֶּׁמֶשׁ: דִּינֵי מִלְחָמָה. בְּנֵי יִשְׂרָאֵל עוֹמְדִים לַעֲבֹר אֶת הַיַּרְדֵּן, וְשָׁם מְחַכִּים לָהֶם צְבָאוֹת שֶׁל עַמֵּי כְּנַעַן.

🤔 לַמּוֹרֶה: מָה עוֹלֶה בָּרֹאשׁ כְּשֶׁאוֹמְרִים ׳מִלְחָמָה׳?
כְּדַאי לִפְתֹּחַ בְּסִיעוּר מוֹחוֹת קָצָר. הַתַּלְמִידִים יַצִּיעוּ: אוֹיְבִים, נֶשֶׁק, חַיָּלִים, צָבָא, סַכָּנָה, שְׁבוּיִים, שָׁלָל, שֶׁטַח, פַּחַד.
וּמִישֶׁהוּ יַגִּיד גַּם אֱמוּנָה – וְזֶה בְּדִיּוּק הַנּוֹשֵׂא שֶׁל הַפָּרָשָׁה שֶׁלָּנוּ: לְחַזֵּק אֶת הָרוּחַ וְאֶת הָאֱמוּנָה לִקְרַאת הַקְּרָב.

שִׂימוּ לֵב: הַתּוֹרָה אֵינָהּ מַתְחִילָה בִּכְלֵי הַנֶּשֶׁק וְלֹא בַּתָּכְנִית הַצְּבָאִית, אֶלָּא בָּעוֹלָם הַפְּנִימִי שֶׁל הַלּוֹחֵם. ה׳ אוֹהֵב אֶת בָּנָיו, וְלָכֵן הַדָּבָר הָרִאשׁוֹן שֶׁהוּא דּוֹאֵג לוֹ הוּא הַלֵּב שֶׁלָּהֶם.

💛 אַהֲבַת הַשֵּׁם
ה׳ לֹא שׁוֹלֵחַ אוֹתָנוּ לְבַד לְשׁוּם מָקוֹם. גַּם כְּשֶׁאֲנִי נִכְנָס לְמָקוֹם מַפְחִיד – מִבְחָן, חֹשֶׁךְ, דָּבָר חָדָשׁ – הַתּוֹרָה מְלַמֶּדֶת אוֹתִי שֶׁה׳ הוֹלֵךְ אִתִּי 💛

לָמָּה דִּינֵי הַמִּלְחָמָה בָּאִים דַּוְקָא כָּאן, מִיָּד אַחֲרֵי דִּינֵי הַשּׁוֹפְטִים וְהָעֵדִים?

📜 רַשִׁ״י
׳כִּי תֵצֵא לַמִּלְחָמָה׳ – סָמַךְ הַכָּתוּב יְצִיאַת מִלְחָמָה לְכָאן, לוֹמַר לְךָ שֶׁאֵין מְחֻסַּר אֵבֶר יוֹצֵא לַמִּלְחָמָה. דָּבָר אַחֵר, לוֹמַר לְךָ: אִם עָשִׂיתָ מִשְׁפַּט צֶדֶק, אַתָּה מֻבְטָח שֶׁאִם תֵּצֵא לַמִּלְחָמָה אַתָּה נוֹצֵחַ. וְכֵן דָּוִד הוּא אוֹמֵר: ׳עָשִׂיתִי מִשְׁפָּט וָצֶדֶק, בַּל תַּנִּיחֵנִי לְעֹשְׁקָי׳.

זֶה חִדּוּשׁ גָּדוֹל: הַנִּצָּחוֹן בַּקְּרָב תָּלוּי בְּמָה שֶׁקּוֹרֶה בְּבֵית הַדִּין וּבָרְחוֹב. עַם שֶׁשּׁוֹפֵט בְּצֶדֶק וּמִתְנַהֵג בְּיֹשֶׁר – ה׳ עוֹמֵד לְצִדּוֹ.

📜 רַשִׁ״י
׳עַל אֹיְבֶךָ׳ – יִהְיוּ בְּעֵינֶיךָ כְּאוֹיְבִים, אַל תְּרַחֵם עֲלֵיהֶם, כִּי לֹא יְרַחֲמוּ עָלֶיךָ.
📜 רַשִׁ״י: ׳עַל אֹיְבֵיכֶם׳ (פָּסוּק ג׳)
׳עַל אֹיְבֵיכֶם׳ – אֵין אֵלּוּ אֲחֵיכֶם, שֶׁאִם תִּפְּלוּ בְּיָדָם אֵינָם מְרַחֲמִים עֲלֵיכֶם. אֵין זוֹ כְּמִלְחֶמֶת יְהוּדָה עִם יִשְׂרָאֵל, שֶׁנֶּאֱמַר: ׳וַיָּקֻמוּ הָאֲנָשִׁים אֲשֶׁר נִקְּבוּ בְשֵׁמוֹת וַיַּחֲזִיקוּ בַשִּׁבְיָה... וַיַּלְבִּישׁוּם וַיַּנְעִלוּם וַיַּאֲכִלוּם וַיַּשְׁקוּם... וַיְבִיאוּם יְרֵחוֹ עִיר הַתְּמָרִים אֵצֶל אֲחֵיהֶם׳ – אֶלָּא עַל אוֹיְבֵיכֶם אַתֶּם הוֹלְכִים, לְפִיכָךְ הִתְחַזְּקוּ לַמִּלְחָמָה.

רַשִׁ״י מַזְכִּיר סִפּוּר מִדִּבְרֵי הַיָּמִים: כְּשֶׁמַּלְכוּת יִשְׂרָאֵל לָקְחָה שְׁבוּיִים מִיהוּדָה, קָמוּ אֲנָשִׁים וְהֶחֱזִירוּ אוֹתָם, הִלְבִּישׁוּ אוֹתָם וְהֶאֱכִילוּ אוֹתָם – כִּי הֵם אַחִים.
אֲבָל הַמִּלְחָמָה שֶׁל פָּרָשָׁתֵנוּ אֵינָהּ מִלְחָמָה שֶׁל אַחִים. מוּל אוֹיֵב אֲמִתִּי צָרִיךְ לְהִתְחַזֵּק, וְלֹא לְהִתְרַכֵּךְ.

וְעַכְשָׁו הַלֵּב: ׳וְרָאִיתָ סוּס וָרֶכֶב עַם רַב מִמְּךָ – לֹא תִירָא מֵהֶם, כִּי ה׳ אֱלֹהֶיךָ עִמָּךְ הַמַּעַלְךָ מֵאֶרֶץ מִצְרָיִם׳.
בָּעֵינַיִם רוֹאִים סוּסִים, מֶרְכָּבוֹת וְהַרְבֵּה חַיָּלִים. אֲבָל יֵשׁ מַשֶּׁהוּ שֶׁלֹּא רוֹאִים בָּעַיִן, וְהַלֵּב מַרְגִּישׁ אוֹתוֹ: ה׳ עִמָּךְ.

זֶהוּ מִשְׁפָּט שֶׁל אַהֲבָה: ה׳ אוֹמֵר לָנוּ – אַתֶּם מוֹדְדִים בָּעֵינַיִם שֶׁלָּכֶם וּמְפַחֲדִים, וַאֲנִי מַבִּיט אֲחֶרֶת. אוֹתוֹ ה׳ שֶׁהֶעֱלָה אֶתְכֶם מִמִּצְרַיִם, עָשָׂה נִסִּים בַּמִּדְבָּר וְהֵבִיא אֶתְכֶם עַד כָּאן – הוּא זֶה שֶׁמְּלַוֶּה אֶתְכֶם גַּם עַכְשָׁו.

  • אַרְבַּע אַזְהָרוֹת כְּנֶגֶד אַרְבָּעָה קוֹלוֹת שֶׁל הָאוֹיֵב: צַהֲלַת סוּסִים, הַגָּפַת תְּרֵיסִין, קוֹל קְרָנוֹת וְקוֹל צְוָחָה. וְהַטַּעַם הַגָּדוֹל: הֵם בָּאִים בְּנִצְחוֹנוֹ שֶׁל בָּשָׂר וָדָם, וְאָנוּ בְּנִצְחוֹנוֹ שֶׁל מָקוֹם – ׳הַהֹלֵךְ עִמָּכֶם׳, זֶה מַחֲנֵה הָאָרוֹן.
  • בְּמִלְחֶמֶת רְשׁוּת חוֹזְרִים הַבַּיְתָה אַרְבָּעָה: בָּנָה בַּיִת וְלֹא חֲנָכוֹ, נָטַע כֶּרֶם וְלֹא חִלְּלוֹ, אֵרַשׂ אִשָּׁה וְלֹא לְקָחָהּ – כִּי הֵם טְרוּדִים וְיֵשׁ בָּזֶה עַגְמַת נֶפֶשׁ; וְהַיָּרֵא וְרַךְ הַלֵּבָב – כִּי פַּחַד מַדְבִּיק.
  • ׳הַיָּרֵא וְרַךְ הַלֵּבָב׳ – רַבִּי עֲקִיבָא: כְּמַשְׁמָעוֹ; רַבִּי יוֹסֵי הַגְּלִילִי: הַיָּרֵא מֵעֲבֵרוֹת שֶׁבְּיָדוֹ, וְהַהַכְרָזוֹת הָאֲחֵרוֹת בָּאוֹת לְכַסּוֹת עָלָיו. אַחֲרֵי כֵן ׳וּפָקְדוּ שָׂרֵי צְבָאוֹת׳ – וְאָסוּר לִבְרֹחַ, שֶׁתְּחִלַּת נְפִילָה נִיסָה.

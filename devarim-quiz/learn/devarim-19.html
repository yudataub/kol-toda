בַּשִּׁעוּר הַזֶּה נִלְמַד עַל סַכָּנָה מְיֻחֶדֶת: לֹא סַכָּנָה שֶׁל מִלְחָמָה אוֹ שֶׁל רָעָב, אֶלָּא הַסַּכָּנָה שֶׁבַּהַצְלָחָה – כְּשֶׁהַכֹּל הוֹלֵךְ טוֹב, יֵשׁ בָּתִּים, כֶּסֶף וְשֶׁפַע, וְהַלֵּב מִתְחִיל לוֹמַר ׳כֹּחִי וְעֹצֶם יָדִי עָשָׂה לִי אֶת הַחַיִל הַזֶּה׳.
נִלְמַד אֶת הַתַּהֲלִיךְ שֶׁמְּתָאֲרִים הַפְּסוּקִים, אֶת זִכְרוֹן הַנִּסִּים בַּמִּדְבָּר, וְאֶת הַפִּתְרוֹן הַגָּדוֹל: ׳וְזָכַרְתָּ אֶת ה׳ אֱלֹהֶיךָ כִּי הוּא הַנֹּתֵן לְךָ כֹּחַ׳. וּנְגַלֶּה שֶׁדַּוְקָא בַּזִּכָּרוֹן הַזֶּה מִסְתַּתֶּרֶת אַהֲבָה גְּדוֹלָה: ה׳ נָתַן לְכָל אֶחָד מֵאִתָּנוּ כֹּחוֹת וְכִשְׁרוֹנוֹת, כְּמוֹ מַתָּנָה, כְּדֵי שֶׁנַּצְלִיחַ וְנֵיטִיב לָעוֹלָם.
בַּשִּׁעוּר: ⏳ הִשָּׁמֶר לְךָ – מָתַי הַסַּכָּנָה הַזֹּאת? 🏡 וְכֹל אֲשֶׁר לְךָ יִרְבֶּה – הַשֶּׁפַע הַגָּדוֹל 🎈 וְרָם לְבָבֶךָ – הַלֵּב שֶׁמִּתְנַפֵּחַ 🐍 זִכְרוֹן הַמִּדְבָּר – נָחָשׁ, עַקְרָב, מַיִם מִסֶּלַע וּמָן 💪 ׳כֹּחִי וְעֹצֶם יָדִי עָשָׂה לִי אֶת הַחַיִל הַזֶּה׳ 🎁 וְזָכַרְתָּ – ׳הוּא הַנֹּתֵן לְךָ כֹּחַ לַעֲשׂוֹת חָיִל׳ 🌟 לֹא לְהִתְבַּיֵּשׁ בַּכִּשָּׁרוֹן – מָה הִיא עֲנָוָה אֲמִתִּית ⚠️ הַעִדֹתִי בָכֶם הַיּוֹם – ׳עֵקֶב לֹא תִשְׁמְעוּן׳ 🤝 לְמַעַן הָקִים אֶת בְּרִיתוֹ – לֹא בְצִדְקָתְךָ

פָּרָשַׁת עֵקֶב הִיא חֵלֶק מֵהַנְּאוּם הַגָּדוֹל שֶׁל מֹשֶׁה, לִפְנֵי הַכְּנִיסָה לָאָרֶץ. מֹשֶׁה לֹא יַעֲבֹר אֶת הַיַּרְדֵּן, וְהוּא יוֹדֵעַ שֶׁמֵּעֵבֶר לַיַּרְדֵּן הַמְּצִיאוּת תִּשְׁתַּנֶּה מִן הַקָּצֶה אֶל הַקָּצֶה: לֹא עוֹד מָן מִן הַשָּׁמַיִם וְעָנָן שֶׁמּוֹלִיךְ בַּדֶּרֶךְ, אֶלָּא שָׂדוֹת, בָּתִּים, מִלְחָמוֹת וְעָרִים.
לָכֵן הוּא טוֹרֵחַ לְהַזְהִיר אֶת הָעָם מֵרֹאשׁ מִכָּל מָה שֶׁהֵם עֲתִידִים לִפְגֹּשׁ. כְּמוֹ אַבָּא שֶׁמְּלַוֶּה אֶת בְּנוֹ לִפְנֵי נְסִיעָה אֲרֻכָּה וּמַסְבִּיר לוֹ בְּדִיּוּק מֵאֵיפֹה לְהִזָּהֵר – מִפְּנֵי שֶׁהוּא אוֹהֵב אוֹתוֹ.

🤔 הַצִּיר שֶׁלָּנוּ: לְכָל אַזְהָרָה יֵשׁ זְמַן
בְּכָל פַּעַם שֶׁמֹּשֶׁה אוֹמֵר ׳הִשָּׁמֶר לְךָ׳, כְּדַאי לִבְדֹּק: מָתַי בְּדִיּוּק הַסַּכָּנָה הַזֹּאת אֲמוּרָה לִקְרוֹת?
׳כִּי תֹאמַר בִּלְבָבְךָ רַבִּים הַגּוֹיִם הָאֵלֶּה מִמֶּנִּי, אֵיכָה אוּכַל לְהוֹרִישָׁם׳ – זֶהוּ פַּחַד מִן הָאוֹיְבִים, וְהוּא שַׁיָּךְ לִפְנֵי הַמִּלְחָמוֹת וּבִשְׁעַת הַמִּלְחָמוֹת.
הָאַזְהָרָה שֶׁלֹּא לַחְמֹד אֶת הַכֶּסֶף וְהַזָּהָב שֶׁעַל פִּסְלֵי הָעַמִּים – שַׁיֶּכֶת לִשְׁלַב הַמִּלְחָמָה עַצְמָהּ, כְּשֶׁלּוֹקְחִים שָׁלָל.
וּמָה עִם הָאַזְהָרָה שֶׁלָּנוּ?

לִפְנֵי הָאַזְהָרָה בָּא פָּסוּק מָתוֹק: ׳וְאָכַלְתָּ וְשָׂבָעְתָּ וּבֵרַכְתָּ אֶת ה׳ אֱלֹהֶיךָ עַל הָאָרֶץ הַטֹּבָה אֲשֶׁר נָתַן לָךְ׳ – מִכָּאן לוֹמְדִים אֶת מִצְוַת בִּרְכַּת הַמָּזוֹן. אוֹכְלִים, שְׂבֵעִים, וְאוֹמְרִים תּוֹדָה.
וּמִיָּד אַחֲרֵי זֶה: ׳הִשָּׁמֶר לְךָ פֶּן תִּשְׁכַּח אֶת ה׳ אֱלֹהֶיךָ לְבִלְתִּי שְׁמֹר מִצְוֹתָיו וּמִשְׁפָּטָיו וְחֻקֹּתָיו׳. שְׁתֵּי אַזְהָרוֹת בְּפָסוּק אֶחָד: לֹא לִשְׁכֹּחַ אֶת ה׳, וְלֹא לְהַפְסִיק לִשְׁמֹר אֶת הַמִּצְווֹת. הַשְּׁנַיִם קְשׁוּרִים זֶה בָּזֶה: מִי שֶׁשּׁוֹכֵחַ מִי נָתַן לוֹ הַכֹּל – מַפְסִיק גַּם לַעֲשׂוֹת אֶת רְצוֹנוֹ.

וּמָתַי תִּקְרֶה הַסַּכָּנָה הַזֹּאת? הַפָּסוּק הַבָּא כְּבָר יְגַלֶּה: ׳פֶּן תֹּאכַל וְשָׂבָעְתָּ וּבָתִּים טוֹבִים תִּבְנֶה וְיָשָׁבְתָּ׳ – זֶה שְׁלַב הַהִתְיַשְּׁבוּת, הַרְבֵּה אַחֲרֵי הַכִּבּוּשׁ. עוֹד יַעַבְרוּ שָׁנִים שֶׁל מִלְחָמוֹת, וְרַק אַחַר כָּךְ יַגִּיעוּ הַשֶּׁקֶט וְהַשֶּׁפַע – וְאָז תַּגִּיעַ הַסַּכָּנָה.

׳פֶּן תֹּאכַל וְשָׂבָעְתָּ, וּבָתִּים טוֹבִים תִּבְנֶה וְיָשָׁבְתָּ׳. אוֹכֶל יֵשׁ בְּשֶׁפַע, בָּתִּים נִבְנִים, וְהָעָם מִתְיַשֵּׁב בָּאָרֶץ וּבוֹנֶה עָרִים.

׳וּבְקָרְךָ וְצֹאנְךָ יִרְבְּיֻן, וְכֶסֶף וְזָהָב יִרְבֶּה לָּךְ, וְכֹל אֲשֶׁר לְךָ יִרְבֶּה׳. הַבָּקָר וְהַצֹּאן מִתְרַבִּים, הַכֶּסֶף וְהַזָּהָב מִצְטַבְּרִים, וּבְקִצּוּר: כָּל מָה שֶׁעוֹשִׂים מַצְלִיחַ.

זֶה לֹא קָרָה רַק בִּימֵי הַתַּנַ״ךְ. הַמַּצָּב הַזֶּה יָכוֹל לִקְרוֹת בְּכָל דּוֹר וּבְכָל מַצָּב: עַם שֶׁמַּצְלִיחַ סוֹלֵל כְּבִישִׁים וּבוֹנֶה עָרִים, יֵשׁ לוֹ צָבָא חָזָק עִם מְטוֹסִים וְטַנְקִים, הַכַּלְכָּלָה פּוֹרַחַת, וְהַטֶּכְנוֹלוֹגְיָה מִתְקַדֶּמֶת. כָּל מָה שֶׁהוּא נוֹגֵעַ בּוֹ – מַצְלִיחַ.
וְגַם אֵצֶל יֶלֶד זֶה קוֹרֶה בְּקָטָן: הַצִּיּוּנִים מְצֻיָּנִים, הַחֲבֵרִים אוֹהֲבִים, מַצְלִיחִים בַּסְּפּוֹרְט וּבַנְּגִינָה – וְהַכֹּל נִרְאֶה פָּשׁוּט וּמוּבָן מֵאֵלָיו.

💡 שְׂאֵלָה לַכִּתָּה
׳כָּל מָה שֶׁתַּעֲשֶׂה – יַצְלִיחַ׳. תְּנוּ דֻּגְמָאוֹת מֵהַיּוֹם: בְּמָה עַם יָכוֹל לְהַצְלִיחַ? וּבְמָה יָכוֹל לְהַצְלִיחַ יֶלֶד?
וְאַחַר כָּךְ נִשְׁאַל אֶת הַשְּׁאֵלָה הַחֲשׁוּבָה: מָה קוֹרֶה בַּלֵּב שֶׁל מִי שֶׁהַכֹּל מַצְלִיחַ לוֹ?

שִׂימוּ לֵב: הַתּוֹרָה אֵינָהּ אוֹמֶרֶת שֶׁהַשֶּׁפַע הוּא רַע, חַס וְשָׁלוֹם. הַשֶּׁפַע הוּא מַתָּנָה שֶׁל ה׳, וְהוּא בְּעַצְמוֹ הִבְטִיחַ אוֹתוֹ. ה׳ אוֹהֵב לָתֵת לָנוּ טוֹב – בָּתִּים, לֶחֶם, שִׂמְחָה וְהַצְלָחָה. הָאַזְהָרָה הִיא רַק עַל מָה שֶׁהַלֵּב עוֹשֶׂה עִם הַמַּתָּנָה הַזֹּאת.

יד וְרָם לְבָבֶךָ וְשָׁכַחְתָּ אֶת ה׳ אֱלֹהֶיךָ הַמּוֹצִיאֲךָ מֵאֶרֶץ מִצְרַיִם מִבֵּית עֲבָדִים

וְאַחֲרֵי כָּל הַשֶּׁפַע מַגִּיעָה הַסַּכָּנָה: ׳וְרָם לְבָבֶךָ, וְשָׁכַחְתָּ אֶת ה׳ אֱלֹהֶיךָ הַמּוֹצִיאֲךָ מֵאֶרֶץ מִצְרַיִם מִבֵּית עֲבָדִים׳.

מָה זֶה ׳וְרָם לְבָבֶךָ׳? זוֹ תְּחוּשָׁה שֶׁל גַּאֲוָה: הַלֵּב מִתְנַפֵּחַ וּמִתְרוֹמֵם, וְאוֹמֵר לְעַצְמוֹ ׳אֲנִי הִצְלַחְתִּי, אֲנִי עָשִׂיתִי׳. וּמִיָּד אַחֲרֵי הַגַּאֲוָה בָּאָה הַשִּׁכְחָה: ׳וְשָׁכַחְתָּ אֶת ה׳ אֱלֹהֶיךָ׳.
כַּמָּה מַהֵר אֶפְשָׁר לִשְׁכֹּחַ! עוֹד לֹא עָבְרוּ שָׁנִים רַבּוֹת, וּכְבָר שׁוֹכְחִים מִי הוֹצִיא אוֹתָנוּ מִמִּצְרַיִם, מִבֵּית עֲבָדִים – וְאֶת כָּל הַנִּסִּים שֶׁעָשָׂה לָנוּ בַּמִּדְבָּר.

💡 סִדְרוּ אֶת שַׁלְבֵי הַתַּהֲלִיךְ
הַפְּסוּקִים מְתָאֲרִים תַּהֲלִיךְ מְסֻדָּר, שָׁלָב אַחַר שָׁלָב:
1. יִהְיֶה לָאָדָם טוֹב – אֹכֶל, בָּתִּים, בָּקָר וְצֹאן, כֶּסֶף וְזָהָב (פְּסוּקִים י״ב–י״ג).
2. ׳וְרָם לְבָבֶךָ׳ – מִתְפַּתַּחַת אֶצְלוֹ מִדַּת הַגַּאֲוָה (פָּסוּק י״ד).
3. ׳וְשָׁכַחְתָּ אֶת ה׳ אֱלֹהֶיךָ׳ – הוּא לֹא מַכִּיר לַה׳ טוֹבָה עַל כָּל מָה שֶׁנָּתַן לוֹ.
4. ׳כֹּחִי וְעֹצֶם יָדִי עָשָׂה לִי אֶת הַחַיִל הַזֶּה׳ (פָּסוּק י״ז).
וְהַפִּתְרוֹן: ׳וְזָכַרְתָּ אֶת ה׳ אֱלֹהֶיךָ כִּי הוּא הַנֹּתֵן לְךָ כֹּחַ לַעֲשׂוֹת חָיִל׳ (פָּסוּק י״ח).

לָמָּה דַּוְקָא הַשֶּׁפַע מֵבִיא לְשִׁכְחָה? כִּי כְּשֶׁחָסֵר לָנוּ מַשֶּׁהוּ – אֲנַחְנוּ מִתְפַּלְּלִים וּמְבַקְּשִׁים, וְזוֹכְרִים שֶׁאֲנַחְנוּ צְרִיכִים אֶת ה׳. בַּמִּדְבָּר הָעָם הָיָה זָקוּק לַה׳ לְכָל פְּרוּסַת לֶחֶם וּלְכָל כּוֹס מַיִם. אֲבָל כְּשֶׁהַמְּקָרֵר מָלֵא וְהַבַּיִת בָּנוּי, קַל לִשְׁכֹּחַ.
וְדַוְקָא מִשּׁוּם כָּךְ מֹשֶׁה מַזְהִיר מֵרֹאשׁ: הַזִּכָּרוֹן צָרִיךְ עֲבוֹדָה.

🐍זִכְרוֹן הַמִּדְבָּר – נָחָשׁ, עַקְרָב, מַיִם מִסֶּלַע וּמָן

📖 פֶּרֶק ח׳, פְּסוּקִים ט״ו–ט״ז ←
טו הַמּוֹלִיכֲךָ בַּמִּדְבָּר הַגָּדֹל וְהַנּוֹרָא נָחָשׁ שָׂרָף וְעַקְרָב וְצִמָּאוֹן אֲשֶׁר אֵין מָיִם הַמּוֹצִיא לְךָ מַיִם מִצּוּר הַחַלָּמִישׁ טז הַמַּאֲכִלְךָ מָן בַּמִּדְבָּר אֲשֶׁר לֹא יָדְעוּן אֲבֹתֶיךָ לְמַעַן עַנֹּתְךָ וּלְמַעַן נַסֹּתֶךָ לְהֵיטִבְךָ בְּאַחֲרִיתֶךָ

כְּדֵי שֶׁלֹּא נִשְׁכַּח, מֹשֶׁה מוֹנֶה אֶת הַטּוֹבוֹת שֶׁה׳ עָשָׂה לָנוּ בַּמִּדְבָּר. הַזִּכָּרוֹן הַזֶּה עוֹד טָרִי לְגַמְרֵי – הֵם יָצְאוּ מֵהַמִּדְבָּר רַק עַכְשָׁו!

׳הַמּוֹלִיכֲךָ בַּמִּדְבָּר הַגָּדֹל וְהַנּוֹרָא, נָחָשׁ שָׂרָף וְעַקְרָב וְצִמָּאוֹן אֲשֶׁר אֵין מָיִם׳.
בַּמִּדְבָּר יֵשׁ מֶרְחַקִּים עֲצוּמִים, וְאָדָם יָכוֹל לִתְעוֹת בּוֹ – וְה׳ הוֹלִיךְ אֶת בְּנֵי יִשְׂרָאֵל בַּדֶּרֶךְ.
בַּמִּדְבָּר יֵשׁ נְחָשִׁים אַרְסִיִּים וְעַקְרַבִּים – וְה׳ שָׁמַר עֲלֵיהֶם מִכֻּלָּם.
בַּמִּדְבָּר יֵשׁ צִמָּאוֹן וְאֵין מַיִם – וְה׳ נָתַן לָהֶם מַיִם.

׳הַמּוֹצִיא לְךָ מַיִם מִצּוּר הַחַלָּמִישׁ, הַמַּאֲכִלְךָ מָן בַּמִּדְבָּר אֲשֶׁר לֹא יָדְעוּן אֲבֹתֶיךָ׳.
׳צוּר הַחַלָּמִישׁ׳ הוּא סֶלַע קָשֶׁה מְאוֹד – וּמִמֶּנּוּ בְּדִיּוּק יָצְאוּ מַיִם. וְהַמָּן הָיָה אֹכֶל שֶׁיָּרַד מִן הַשָּׁמַיִם, אֹכֶל שֶׁאֲפִילּוּ הָאָבוֹת לֹא הִכִּירוּ.
כָּל יוֹם, אַרְבָּעִים שָׁנָה, קָמוּ בַּבֹּקֶר וְקִבְּלוּ אֲרוּחָה מִיַּד ה׳.

וְלָמָּה בִּכְלָל הָיוּ צְרִיכִים לַעֲבֹר מִדְבָּר קָשֶׁה כָּזֶה? הַפָּסוּק עוֹנֶה: ׳לְמַעַן עַנֹּתְךָ וּלְמַעַן נַסֹּתֶךָ, לְהֵיטִבְךָ בְּאַחֲרִיתֶךָ׳.
הַקֹּשִׁי לֹא הָיָה עֹנֶשׁ, אֶלָּא דֶּרֶךְ: ה׳ רָצָה שֶׁנִּלְמַד לִסְמֹךְ עָלָיו, כְּדֵי לְהֵיטִיב לָנוּ בַּסּוֹף. כְּמוֹ אִמָּא שֶׁמַּמְרִיצָה אֶת הַיֶּלֶד לִלְמֹד וּלְהִתְאַמֵּץ – לֹא כְּדֵי לְהַקְשׁוֹת עָלָיו, אֶלָּא כְּדֵי שֶׁיִּהְיֶה לוֹ טוֹב.

יז וְאָמַרְתָּ בִּלְבָבֶךָ כֹּחִי וְעֹצֶם יָדִי עָשָׂה לִי אֶת הַחַיִל הַזֶּה

וְאַחֲרֵי כָּל הַטּוֹבוֹת הָאֵלֶּה מַגִּיעַ הַמִּשְׁפָּט הַמְּסֻכָּן בְּיוֹתֵר: ׳וְאָמַרְתָּ בִּלְבָבֶךָ: כֹּחִי וְעֹצֶם יָדִי עָשָׂה לִי אֶת הַחַיִל הַזֶּה׳.
׳עֹצֶם׳ – מִלְּשׁוֹן עָצְמָה, כֹּחַ גָּדוֹל. ׳הַחַיִל׳ – הָעֹשֶׁר וְהַהַצְלָחָה. כְּלוֹמַר: ׳יֵשׁ לִי עָרִים, יֵשׁ לִי בָּתִּים, יֵשׁ לִי כֶּסֶף וְזָהָב – וּמִי עָשָׂה אֶת כָּל זֶה? אֲנִי! אֲנִי בָּנִיתִי, זֶה הַכֹּחַ שֶׁלִּי׳.

שִׂימוּ לֵב לְמִלָּה אַחַת בַּפָּסוּק: ׳וְאָמַרְתָּ בִּלְבָבֶךָ׳. הַגַּאֲוָה מַתְחִילָה בִּפְנִים, בַּלֵּב, וְלֹא תָּמִיד אוֹמְרִים אוֹתָהּ בְּקוֹל. וְדַוְקָא שָׁם, בַּלֵּב, צָרִיךְ לְהִזָּהֵר.

📚 מַצְּבוֹת הַנִּצָּחוֹן שֶׁל הַמְּלָכִים

מֹשֶׁה אוֹמֵר: עַם שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא לֹא מִתְנַהֵג כָּךְ. כְּשֶׁנְּנַצֵּחַ וְנַצְלִיחַ וְנִבְנֶה עָרִים – לֹא נֹאמַר ׳זֶה אֲנִי׳.

הַתְּשׁוּבָה: הַצְלָחָה גְּדוֹלָה יְכוֹלָה לִגְרֹם תְּחוּשַׁת גַּאֲוָה – הַרְגָּשָׁה שֶׁבְּכֹחִי וּבְכִשְׁרוֹנִי שִׁנִּיתִי אֶת הַמְּצִיאוּת. בְּדִיּוּק עַל הַהַרְגָּשָׁה הַזֹּאת מְדַבֵּר הַפָּסוּק שֶׁלָּנוּ.

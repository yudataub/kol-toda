הַחֹמֶר הַמָּלֵא: הַלְוָאָה שֶׁל אַחִים – בְּחֶסֶד, בְּלִי נֶשֶׁךְ (פָּרָשַׁת כִּי תֵצֵא, פֶּרֶק כ״ג)
🎯 לַחִידוֹן🏠 כָּל הַחִידוֹנִים
בַּשִּׁעוּר הַזֶּה נִלְמַד מִצְוָה שֶׁנּוֹגַעַת לַחַיִּים מַמָּשׁ כָּל יוֹם: אִסּוּר רִבִּית. כְּשֶׁאָח שֶׁלִּי זָקוּק לְכֶסֶף – אֲנִי מַלְוֶה לוֹ בְּחֶסֶד, וְלֹא מַרְוִיחַ מִן הַמְּצוּקָה שֶׁלּוֹ.
נָבִין מָה זוֹ הַלְוָאָה, מָה הֵם ׳קֶרֶן׳ וְ׳רִבִּית׳, לָמָּה הַתּוֹרָה קוֹרֵאת לָרִבִּית ׳נֶשֶׁךְ׳, וּמָה מְחַדֵּשׁ חֻמַּשׁ דְּבָרִים עַל מָה שֶׁכָּתוּב בְּחֻמַּשׁ וַיִּקְרָא.
נִלְמַד גַּם אֶת שְׁאָר הַפֶּרֶק: עַמּוֹן וּמוֹאָב שֶׁהָיוּ כְּפוּיֵי טוֹבָה, אֱדוֹם וּמִצְרַיִם שֶׁמֵּהֶם לוֹמְדִים הַכָּרַת הַטּוֹב, קְדֻשַּׁת הַמַּחֲנֶה בַּמִּלְחָמָה, וְדִינֵי נְדָרִים – ׳מוֹצָא שְׂפָתֶיךָ תִּשְׁמֹר׳.
וּבְכָל זֶה נִרְאֶה דָּבָר אֶחָד גָּדוֹל: ה׳ אוֹהֵב אוֹתָנוּ, וְהוּא רוֹצֶה שֶׁנִּהְיֶה עַם שֶׁל אַחִים – טוֹבִים זֶה לָזֶה, נְקִיִּים וְנֶאֱמָנִים לְמִלָּה שֶׁלָּנוּ.
בַּשִּׁעוּר: 💰 מָה זוֹ הַלְוָאָה? קֶרֶן וְרִבִּית 🐍 נֶשֶׁךְ – לָמָּה הַתּוֹרָה קוֹרֵאת לָרִבִּית כָּךְ? 🚫 ׳לֹא תַשִּׁיךְ לְאָחִיךָ׳ – אַזְהָרָה כְּפוּלָה 🥛 נֶשֶׁךְ אֹכֶל – רִבִּית לֹא רַק בְּכֶסֶף 👨‍👦 ׳לְאָחִיךָ׳ – אַחֲוָה וְחֶסֶד, וּבְרָכָה בְּכָל מִשְׁלַח יָד 🍞 עַמּוֹן וּמוֹאָב – כְּפוּיֵי טוֹבָה; אֱדוֹם וּמִצְרַיִם – הַכָּרַת הַטּוֹב ⛺ וְהָיָה מַחֲנֶיךָ קָדוֹשׁ – קְדֻשַּׁת מַחֲנֵה יִשְׂרָאֵל 🗣️ מוֹצָא שְׂפָתֶיךָ תִּשְׁמֹר – נְדָרִים

💰מָה זוֹ הַלְוָאָה? קֶרֶן וְרִבִּית

לִפְנֵי שֶׁמְּבִינִים מָה זוֹ רִבִּית, צָרִיךְ לְהָבִין מָה זוֹ הַלְוָאָה.
אָדָם זָקוּק לְכֶסֶף וְאֵין לוֹ. הוּא נִגָּשׁ לְחָבֵר וְאוֹמֵר: ׳תַּלְוֶה לִי בְּבַקָּשָׁה מֵאָה שְׁקָלִים׳. הוּא מְקַבֵּל אֶת הַכֶּסֶף, מִשְׁתַּמֵּשׁ בּוֹ, וְאַחֲרֵי חֹדֶשׁ אוֹ חֳדָשַׁיִם – כְּפִי שֶׁקָּבְעוּ – הוּא מַחֲזִיר בְּדִיּוּק אֶת אוֹתוֹ סְכוּם.
מִי שֶׁנּוֹתֵן אֶת הַכֶּסֶף נִקְרָא ׳מַלְוֶה׳, וּמִי שֶׁמְּקַבֵּל אוֹתוֹ נִקְרָא ׳לוֶֹה׳.

🤔 שְׁאֵלָה לַכִּתָּה: מָתַי אֲנָשִׁים לוֹקְחִים הַלְוָאָה?
חִשְׁבוּ עַל שְׁנֵי מַצָּבִים אוֹ שְׁלוֹשָׁה.
כְּשֶׁאֵין כֶּסֶף אֲפִלּוּ לְאֹכֶל – אָדָם נִגָּשׁ וּמְבַקֵּשׁ הַלְוָאָה.
כְּשֶׁמִּתְחַתְּנִים – וְאֵין כַּרְגַּע אֶת כָּל הוֹצְאוֹת הַחֲתֻנָּה.
כְּשֶׁבּוֹנִים בַּיִת – הוֹצָאָה גְּדוֹלָה שֶׁאֶפְשָׁר לְהַחֲזִיר רַק לְאַט לְאַט.
יֵשׁ עוֹד הַרְבֵּה מַצָּבִים. הַמְּשֻׁתָּף לְכֻלָּם: אָדָם צָרִיךְ עֶזְרָה בְּדִיּוּק עַכְשָׁו.

וּמָה הִיא רִבִּית? הַמַּלְוֶה נוֹתֵן לַלּוֶֹה מֵאָה שְׁקָלִים, וְאוֹמֵר לוֹ: ׳כְּשֶׁתַּחֲזִיר לִי – תַּחֲזִיר מֵאָה וַחֲמִשָּׁה עָשָׂר׳.
הַמֵּאָה שֶׁהוּא נָתַן הֵם הַ׳קֶּרֶן׳ – הַסְּכוּם הַמְּקוֹרִי שֶׁל הַהַלְוָאָה.
וַחֲמִשָּׁה עָשָׂר הַשְּׁקָלִים הַנּוֹסָפִים, שֶׁמְּשַׁלְּמִים בִּשְׁבִיל עֶצֶם הַהַלְוָאָה, הֵם הָ׳רִבִּית׳.

💡 וְכָאן בָּאָה הַתּוֹרָה
הַתּוֹרָה אוֹמֶרֶת: ׳לֹא תַשִּׁיךְ לְאָחִיךָ׳ – אָסוּר לָקַחַת רִבִּית מֵאָח, מִיְּהוּדִי.
הַלְוָאָה כֵּן, בְּשִׂמְחָה; תּוֹסֶפֶת תַּשְׁלוּם עַל הַהַלְוָאָה – לֹא.

לָמָּה? כִּי ה׳ אוֹהֵב אֶת הֶעָנִי וְאֶת מִי שֶׁקָּשֶׁה לוֹ, וְהוּא לֹא רוֹצֶה שֶׁדַּוְקָא בָּרֶגַע שֶׁבּוֹ אָדָם נִמְצָא בְּמְצוּקָה – מִישֶׁהוּ יַרְוִיחַ מִמֶּנּוּ. מִי שֶׁמַּלְוֶה בְּלִי רִבִּית, עוֹשֶׂה בְּדִיּוּק אֶת מָה שֶׁה׳ עוֹשֶׂה אִתָּנוּ: נוֹתֵן בְּחִנָּם, מֵאַהֲבָה.

ה׳ נוֹתֵן לִי כָּל יוֹם אֹכֶל, בְּרִיאוּת וּמִשְׁפָּחָה – וְאֵינוֹ מְבַקֵּשׁ עַל זֶה שׁוּם תַּשְׁלוּם. כְּשֶׁאֲנִי מַלְוֶה לְחָבֵר בְּלִי לְבַקֵּשׁ תּוֹסֶפֶת, אֲנִי נִהְיֶה קְצָת דּוֹמֶה לְאַבָּא שֶׁבַּשָּׁמַיִם 💛

קִרְאוּ אֶת שְׁנֵי הַפְּסוּקִים, וְחַפְּשׂוּ מִלָּה אַחַת שֶׁחוֹזֶרֶת שׁוּב וָשׁוּב בְּצוּרוֹת שׁוֹנוֹת.
מָצָאתֶם? זֶה הַשֹּׁרֶשׁ נ־שׁ־ךְ, וְהוּא מוֹפִיעַ כָּאן שֶׁבַע פְּעָמִים: תַּשִּׁיךְ, נֶשֶׁךְ, נֶשֶׁךְ, נֶשֶׁךְ, יִשָּׁךְ, תַּשִּׁיךְ, תַּשִּׁיךְ.
׳נֶשֶׁךְ׳ הִיא הַמִּלָּה שֶׁל הַתּוֹרָה לְרִבִּית.

📜 רַשִׁ״י (שְׁמוֹת כ״ב, כ״ד): לָמָּה ׳נֶשֶׁךְ׳?
׳נֶשֶׁךְ׳ – רִבִּית שֶׁהוּא כִּנְשִׁיכַת נָחָשׁ, שֶׁנּוֹשֵׁךְ חַבּוּרָה קְטַנָּה בְּרַגְלוֹ וְאֵינוֹ מַרְגִּישׁ, וּפִתְאוֹם הוּא מְבַצְבֵּץ וְנוֹפֵחַ עַד קָדְקֳדוֹ. כָּךְ רִבִּית – אֵינוֹ מַרְגִּישׁ וְאֵינוֹ נִכָּר, עַד שֶׁהָרִבִּית עוֹלָה וּמְחַסַּרְתּוֹ מָמוֹן הַרְבֵּה.
📚 שְׁתֵּי תְּמוּנוֹת: מִי נִפְגַּע יוֹתֵר?
בַּתְּמוּנָה הָאַחַת – נֶהָג מֵרוֹצִים שֶׁהִתְהַפְּכָה לוֹ הַמְּכוֹנִית. כֻּלּוֹ חַבּוּרוֹת וּשְׁבָרִים, וְכָל אֶחָד רוֹאֶה מִיָּד שֶׁקָּרָה לוֹ מַשֶּׁהוּ.
בַּתְּמוּנָה הַשְּׁנִיָּה – אִישׁ מְבֻגָּר שֶׁיָּצָא לַגִּנָּה, הִרְגִּישׁ שִׁפְשׁוּף קַל בָּרֶגֶל, וְלֹא שָׂם לֵב בִּכְלָל שֶׁנָּחָשׁ הִכִּישׁ אוֹתוֹ. הוּא הִמְשִׁיךְ בְּיוֹמוֹ כְּרָגִיל – וּבֵינְתַיִם הָאֶרֶס הִתְפַּשֵּׁט בְּגוּפוֹ, עַד שֶׁהָיָה מְאֻחָר מִדַּי.
מָה שֶׁלֹּא מַרְגִּישִׁים – מְסֻכָּן יוֹתֵר. בְּדִיּוּק כָּךְ הִיא הָרִבִּית.

רִבִּית דְּרִבִּית: בַּהַתְחָלָה הַתּוֹסֶפֶת נִרְאֵית קְטַנָּה וְלֹא חֲשׁוּבָה. אֲבָל אִם לֹא מְשַׁלְּמִים אוֹתָהּ בַּזְּמַן, גַּם הָרִבִּית עַצְמָהּ מַתְחִילָה לִצְבֹּר רִבִּית, וְאַחַר כָּךְ גַּם הִיא – וְכָךְ הַחוֹב גָּדֵל וְגָדֵל.
אַחֲרֵי כַּמָּה שָׁנִים הַחוֹב יָכוֹל לִהְיוֹת גָּדוֹל אֲפִלּוּ מֵהַהַלְוָאָה עַצְמָהּ. זֶהוּ הַ׳נֶּשֶׁךְ׳: בַּהַתְחָלָה אֵין מַרְגִּישִׁים, וּבַסּוֹף הַחוֹב תּוֹפֵחַ וְחוֹנֵק אֶת הֶעָנִי.

💡 לַמּוֹרֶה
אֶפְשָׁר לְהַרְאוֹת בַּכִּתָּה אֵיךְ חוֹב קָטָן גָּדֵל: מַתְחִילִים מִמֵּאָה, מוֹסִיפִים עֲשָׂרָה אֲחוּזִים בְּכָל שָׁנָה, וּמְחַשְּׁבִים חָמֵשׁ שָׁנִים.
הַמַּסְקָנָה שֶׁל הַכַּלְכָּלָה הַנְּבוֹנָה: חוֹסְכִים לִזְמַן אָרֹךְ, וְלוֹוִים לִזְמַן קָצָר.

וְכָאן רוֹאִים אֶת אַהֲבַת ה׳: הוּא לֹא חִכָּה שֶׁנִּלְמַד זֹאת לְבַד מֵהַנִּסָּיוֹן הַמַּר. הוּא הִזְהִיר אוֹתָנוּ מֵרֹאשׁ, בְּמִלָּה אַחַת חֲזָקָה – ׳נֶשֶׁךְ׳ – כְּמוֹ אַבָּא שֶׁצּוֹעֵק לַיֶּלֶד: זְהִירוּת, יֵשׁ שָׁם נָחָשׁ!

הַקְּרִיאָה לְשָׁלוֹם וּבַל תַּשְׁחִית

הַחֹמֶר הַמָּלֵא: גַּם בַּמִּלְחָמָה – קֹדֶם שָׁלוֹם, וְשׁוֹמְרִים עַל עֵץ הַפְּרִי (פָּרָשַׁת שׁוֹפְטִים, פֶּרֶק כ׳)
🎯 לַחִידוֹן🏠 כָּל הַחִידוֹנִים
בַּשִּׁעוּר הַקּוֹדֵם לָמַדְנוּ אֶת הַחֵלֶק הָרִאשׁוֹן שֶׁל פֶּרֶק כ׳, שֶׁעוֹסֵק בָּרוּחַ שֶׁל הַלּוֹחֲמִים: לְחַזֵּק אוֹתָם וּלְשַׁחְרֵר אֶת מִי שֶׁמְּפַחֵד.
עַכְשָׁיו נִלְמַד אֶת הַחֵלֶק הַשֵּׁנִי שֶׁל הַפֶּרֶק. אֵיךְ יוֹדְעִים בַּמֶּה הוּא עוֹסֵק? בַּפַּטֶּנְט שֶׁלָּנוּ: מִסְתַּכְּלִים עַל הַפָּסוּק הָרִאשׁוֹן – ׳כִּי תִקְרַב אֶל עִיר לְהִלָּחֵם עָלֶיהָ׳ – וְעַל הָאַחֲרוֹן – ׳עֹשָׂה עִמְּךָ מִלְחָמָה׳. גַּם כָּאן מְדֻבָּר בְּמִלְחָמָה, אֲבָל הַפַּעַם בַּהֲלָכוֹת שֶׁהַתּוֹרָה מְצַוָּה בִּזְמַן מִלְחָמָה.
שְׁנֵי דְּבָרִים נִלְמַד: אֶת הַקְּרִיאָה לְשָׁלוֹם לִפְנֵי כָּל מִלְחָמָה, וְאֶת הָאִסּוּר לְהַשְׁחִית עֵץ פְּרִי בִּזְמַן מָצוֹר.
וּבְעִקָּר נִרְאֶה כַּמָּה ה׳ אוֹהֵב אֶת עוֹלָמוֹ: הוּא מְלַמֵּד אוֹתָנוּ לְבַקֵּשׁ שָׁלוֹם עוֹד לִפְנֵי שֶׁמְּרִימִים חֶרֶב, וְלִשְׁמֹר אֲפִלּוּ עַל עֵץ אֶחָד שֶׁעוֹשֶׂה פְּרִי.
בַּשִּׁעוּר: 📣 וְקָרָאתָ אֵלֶיהָ לְשָׁלוֹם – פּוֹתְחִים בְּשָׁלוֹם 🏳️ אִם שָׁלוֹם תַּעַנְךָ – הָעִיר נִכְנַעַת 🛡️ וְאִם לֹא תַשְׁלִים עִמָּךְ – וּנְתָנָהּ ה׳ בְּיָדֶךָ 🗺️ הֶעָרִים הָרְחֹקוֹת – מִלְחֶמֶת רְשׁוּת וּמִלְחֶמֶת מִצְוָה 🌳 לֹא תַשְׁחִית אֶת עֵצָהּ – עֵץ הַפְּרִי בַּמָּצוֹר 🍎 כִּי מִמֶּנּוּ תֹאכֵל – וְכִי הָעֵץ הוּא אָדָם? 🚫 בַּל תַּשְׁחִית – הָאִסּוּר הַגָּדוֹל, בְּכָל דָּבָר

📣וְקָרָאתָ אֵלֶיהָ לְשָׁלוֹם – פּוֹתְחִים בְּשָׁלוֹם

📖 פֶּרֶק כ׳, פָּסוּק י׳ ←
י כִּי תִקְרַב אֶל עִיר לְהִלָּחֵם עָלֶיהָ וְקָרָאתָ אֵלֶיהָ לְשָׁלוֹם

הַהֲלָכָה הָרִאשׁוֹנָה בַּחֵלֶק הַזֶּה: ׳כִּי תִקְרַב אֶל עִיר לְהִלָּחֵם עָלֶיהָ וְקָרָאתָ אֵלֶיהָ לְשָׁלוֹם׳.
כַּאֲשֶׁר הַצָּבָא מִתְקָרֵב אֶל עִיר שֶׁיּוֹצְאִים לְהִלָּחֵם בָּהּ – לִפְנֵי שֶׁנִּכְנָסִים לַלְּחִימָה עַצְמָהּ, מַצִּיעִים לְאַנְשֵׁי הָעִיר לְסַיֵּם אֶת הָעִנְיָן בְּלִי מִלְחָמָה.

מָה אוֹמְרִים לָהֶם? אַנְשֵׁי הָעִיר, אֲנַחְנוּ מַצִּיעִים לָכֶם לְהִמָּנַע מִמִּלְחָמָה: אִם תִּפְתְּחוּ אֶת שַׁעֲרֵי הָעִיר וְתֵצְאוּ אֵלֵינוּ – לֹא תִּהְיֶה כָּאן מִלְחָמָה כְּלָל.
שִׂימוּ לֵב כַּמָּה זֶה מַפְלִיא: הַתּוֹרָה מְדַבֶּרֶת כָּאן אֶל צָבָא שֶׁכְּבָר יָצָא לַדֶּרֶךְ, וְהַמִּצְוָה הָרִאשׁוֹנָה שֶׁהוּא מְקַבֵּל הִיא לְנַסּוֹת לֹא לְהִלָּחֵם.

📖 הַפָּסוּק הַקּוֹדֵם
לִפְנֵי כֵן נֶאֱמַר: ׳וְהָיָה כְּכַלֹּת הַשֹּׁטְרִים לְדַבֵּר אֶל הָעָם וּפָקְדוּ שָׂרֵי צְבָאוֹת בְּרֹאשׁ הָעָם׳ – שָׁם נִגְמָר הַחֵלֶק הָרִאשׁוֹן, שֶׁל חִזּוּק הַלּוֹחֲמִים, וּמִכָּאן מַתְחִיל הַחֵלֶק הַשֵּׁנִי: הֲלָכוֹת הַמִּלְחָמָה עַצְמָהּ.

לָמָּה ה׳ צִוָּה דַּוְקָא כָּךְ? כִּי ה׳ אוֹהֵב אֶת הַבְּרִיּוֹת שֶׁבָּרָא, וְאֵינוֹ חָפֵץ בְּמִלְחָמָה. גַּם כְּשֶׁיֵּשׁ הֶכְרֵחַ לָצֵאת לַקְּרָב, הוּא מְבַקֵּשׁ מֵאִתָּנוּ: נַסּוּ קֹדֶם בְּדֶרֶךְ שֶׁל שָׁלוֹם.
וְזֶה מְלַמֵּד גַּם אוֹתִי: כְּשֶׁיֵּשׁ רִיב, הַדָּבָר הָרִאשׁוֹן שֶׁה׳ רוֹצֶה מִמֶּנִּי הוּא לְנַסּוֹת לְהַגִּיעַ לְשָׁלוֹם.

💛 אַהֲבַת הַשֵּׁם
ה׳ מְלַמֵּד אוֹתִי שֶׁאֲפִלּוּ בְּרֶגַע הֲכִי קָשֶׁה – קֹדֶם כֹּל מְנַסִּים שָׁלוֹם. אִם כָּךְ הוּא מְצַוֶּה צָבָא שָׁלֵם, כַּמָּה הוּא שָׂמֵחַ כְּשֶׁאֲנִי עוֹשֶׂה שָׁלוֹם עִם חָבֵר בַּכִּתָּה 💛

🏳️אִם שָׁלוֹם תַּעַנְךָ – הָעִיר נִכְנַעַת

📖 פֶּרֶק כ׳, פָּסוּק י״א ←
יא וְהָיָה אִם שָׁלוֹם תַּעַנְךָ וּפָתְחָה לָךְ וְהָיָה כָּל הָעָם הַנִּמְצָא בָהּ יִהְיוּ לְךָ לָמַס וַעֲבָדוּךָ

הַכֹּל תָּלוּי עַכְשָׁיו בִּתְשׁוּבַת אַנְשֵׁי הָעִיר. ׳וְהָיָה אִם שָׁלוֹם תַּעַנְךָ וּפָתְחָה לָךְ׳ – אִם הֵם עוֹנִים ׳שָׁלוֹם׳.
בֶּעָבָר הָיוּ הֶעָרִים מֻקָּפוֹת חוֹמָה, וּבַחוֹמָה הָיוּ שְׁעָרִים גְּדוֹלִים שֶׁנִּסְגְּרוּ וְנִתְבַּצְּרוּ. (גַּם הַיּוֹם אֶפְשָׁר לִרְאוֹת שַׁעַר כָּזֶה בְּחוֹמוֹת יְרוּשָׁלַיִם, שַׁעַר יָפוֹ.)
׳וּפָתְחָה לָךְ׳ – הֵם פּוֹתְחִים אֶת שַׁעֲרֵי הָעִיר, מְנִיפִים דֶּגֶל לָבָן שֶׁל כְּנִיעָה, וְאָז אֵין נִלְחָמִים בָּהֶם.

׳וְהָיָה כָּל הָעָם הַנִּמְצָא בָהּ יִהְיוּ לְךָ לָמַס וַעֲבָדוּךָ׳. הָעִיר נוֹהֶגֶת כְּמוֹ כָּל עִיר שֶׁנִּכְנַעַת: מְשַׁלֶּמֶת מַס וּמְקַבֶּלֶת עָלֶיהָ אֶת שִׁלְטוֹן יִשְׂרָאֵל – וְהַחַיִּים נִשְׁמָרִים.

📜 רַשִׁ״י
׳לָמַס וַעֲבָדוּךָ׳ – עַד שֶׁיְּקַבְּלוּ עֲלֵיהֶם מִסִּים וְשִׁעְבּוּד.
📜 רַשִׁ״י
׳כָּל הָעָם הַנִּמְצָא בָהּ׳ – אֲפִלּוּ אַתָּה מוֹצֵא בָּהּ מִשִּׁבְעָה אֻמּוֹת שֶׁנִּצְטַוֵּיתָ לְהַחֲרִימָם, אַתָּה רַשַּׁאי לְקַיְּמָם.

הַקְּרִיאָה לְשָׁלוֹם אֵינָהּ ׳הַצָּגָה׳. אִם הָעִיר נִכְנַעַת – אָסוּר לְהִלָּחֵם בָּהּ, וְאֵין נוֹגְעִים בְּאַנְשֶׁיהָ. ה׳ שׂוֹנֵא שְׁפִיכוּת דָּמִים, וּמִי שֶׁמּוּכָן לְהַפְסִיק אֶת הַמִּלְחָמָה – מְקַבְּלִים אוֹתוֹ.

💛 אַהֲבַת הַשֵּׁם
ה׳ שָׂמֵחַ בְּכָל חַיִּים שֶׁנִּצָּלִים. גַּם בָּעִיר רְחוֹקָה, שֶׁל עַם אַחֵר – אִם הֵם מוּכָנִים לְשָׁלוֹם, ה׳ מְצַוֶּה אוֹתָנוּ לְהַנִּיחַ לָהֶם. כַּמָּה טוֹב לִהְיוֹת עַם שֶׁל אֱלֹהִים כָּזֶה 💛

🛡️וְאִם לֹא תַשְׁלִים עִמָּךְ – וּנְתָנָהּ ה׳ בְּיָדֶךָ

📖 פֶּרֶק כ׳, פְּסוּקִים י״ב–י״ד ←
יב וְאִם לֹא תַשְׁלִים עִמָּךְ וְעָשְׂתָה עִמְּךָ מִלְחָמָה וְצַרְתָּ עָלֶיהָּ יג וּנְתָנָהּ ה׳ אֱלֹהֶיךָ בְּיָדֶךָ וְהִכִּיתָ אֶת כָּל זְכוּרָהּ לְפִי חָרֶב יד רַק הַנָּשִׁים וְהַטַּף וְהַבְּהֵמָה וְכֹל אֲשֶׁר יִהְיֶה בָעִיר כָּל שְׁלָלָהּ תָּבֹז לָךְ וְאָכַלְתָּ אֶת שְׁלַל אֹיְבֶיךָ אֲשֶׁר נָתַן ה׳ אֱלֹהֶיךָ לָךְ

אֲבָל יִתָּכֵן שֶׁאַנְשֵׁי הָעִיר יַעֲנוּ: לֹא, אֲנַחְנוּ בּוֹחֲרִים לְהִלָּחֵם בָּכֶם. עַל זֶה אוֹמֶרֶת הַתּוֹרָה: ׳וְאִם לֹא תַשְׁלִים עִמָּךְ וְעָשְׂתָה עִמְּךָ מִלְחָמָה וְצַרְתָּ עָלֶיהָ׳.
אָז מֻתָּר לְהַמְשִׁיךְ בְּדֶרֶךְ הַמִּלְחָמָה וְלָצוּר עַל הָעִיר. נָתַתָּ לָהֶם הִזְדַּמְּנוּת, וְהֵם לֹא נִצְּלוּ אוֹתָהּ.

📜 רַשִׁ״י
׳וְאִם לֹא תַשְׁלִים עִמָּךְ וְעָשְׂתָה עִמְּךָ מִלְחָמָה׳ – הַכָּתוּב מְבַשֶּׂרְךָ שֶׁאִם לֹא תַשְׁלִים עִמְּךָ, סוֹפָהּ לְהִלָּחֵם בְּךָ אִם תַּנִּיחֶנָּה וְתֵלֵךְ.
׳וְצַרְתָּ עָלֶיהָ׳ – אַף לְהַרְעִיבָהּ וּלְהַצְמִיאָהּ.

רַשִׁ״י מְלַמֵּד: אֵין כָּאן נְקָמָה סְתָם. עִיר שֶׁדָּחֲתָה אֶת הַשָּׁלוֹם וּבָחֲרָה בַּמִּלְחָמָה – אִם תַּעֲזֹב אוֹתָהּ וְתֵלֵךְ, בַּסּוֹף הִיא תִּלָּחֵם בְּךָ. לָכֵן מֻתָּר לְהַמְשִׁיךְ.

וְאָז בָּא הַפָּסוּק הַמַּפְתִּיעַ: ׳וּנְתָנָהּ ה׳ אֱלֹהֶיךָ בְּיָדֶךָ׳.

׳כִּי הָאָדָם עֵץ הַשָּׂדֶה׳ – הֲרֵי ׳כִּי׳ מְשַׁמֵּשׁ בִּלְשׁוֹן ׳דִּלְמָא׳: שֶׁמָּא הָאָדָם עֵץ הַשָּׂדֶה לְהִכָּנֵס בְּתוֹךְ הַמָּצוֹר מִפָּנֶיךָ לְהִתְיַסֵּר בְּיִסּוּרֵי רָעָב וְצָמָא כְּאַנְשֵׁי הָעִיר? לָמָּה תַשְׁחִיתֶנּוּ?

רַשִׁ״י, וְכֵן רֹב הַמְּפָרְשִׁים, מְבָאֵר אֶת הַמִּלִּים הָאֵלֶּה כִּשְׁאֵלָה: וְכִי הָעֵץ הוּא אָדָם, שֶׁיִּכָּנֵס אִתְּךָ לַמָּצוֹר וְיִסְבֹּל רָעָב וְצָמָא כְּמוֹ אַנְשֵׁי הָעִיר? הָעֵץ אֵינוֹ הָאוֹיֵב שֶׁלְּךָ! אַתָּה נִלְחָם בַּאֲנָשִׁים, לֹא בָּעֵצִים. אִם כֵּן – לָמָּה תַּשְׁחִית אוֹתוֹ?

💡 הַשִּׁיר הַמְּפֻרְסָם
אֶת הַמִּלִּים ׳כִּי הָאָדָם עֵץ הַשָּׂדֶה׳ אַתֶּם בְּוַדַּאי מַכִּירִים מִשִּׁיר יָדוּעַ, שֶׁמַּשְׁוֶה בֵּין הָאָדָם לָעֵץ. הַמִּלִּים לְקוּחוֹת מִן הַפָּסוּק שֶׁלָּנוּ – אֲבָל פְּשׁוּטוֹ שֶׁל מִקְרָא לְפִי רַשִׁ״י הוּא שְׁאֵלָה: וְכִי הָעֵץ הוּא אָדָם?

וּמָה עוֹשִׂים כְּשֶׁבֶּאֱמֶת צְרִיכִים עֵצִים לַמָּצוֹר? אוֹמֶרֶת הַתּוֹרָה: ׳רַק עֵץ אֲשֶׁר תֵּדַע כִּי לֹא עֵץ מַאֲכָל הוּא אֹתוֹ תַשְׁחִית וְכָרָתָּ וּבָנִיתָ מָצוֹר עַל הָעִיר׳.
כּוֹרְתִים עֲצֵי סְרָק – עֵצִים שֶׁאֵין בָּהֶם פְּרִי, כְּמוֹ אֹרֶן אוֹ אַלּוֹן – וּבוֹנִים מֵהֶם אֶת מִתְקְנֵי הַמָּצוֹר. הַצֹּרֶךְ הַצְּבָאִי נֶעֱנֶה, וְעֵץ הַפְּרִי נִשְׁאָר עַל מְקוֹמוֹ.

📖 פֶּרֶק כ׳, פְּסוּקִים י״ט–כ׳ ←

חֲזַ״ל לָמְדוּ מִן הַפְּסוּקִים הָאֵלֶּה דָּבָר גָּדוֹל: הָאִסּוּר אֵינוֹ רַק עַל עֵץ פְּרִי בִּשְׁעַת מִלְחָמָה. זֶהוּ הָאִסּוּר הַגָּדוֹל שֶׁל ׳בַּל תַּשְׁחִית׳ – אָסוּר לְקַלְקֵל שׁוּם דָּבָר בְּלִי צֹרֶךְ.
הַמִּצְוָה שֶׁלֹּא לַעֲקֹר עֵץ פְּרִי חָלָה בְּכָל מָקוֹם וּבְכָל זְמַן – גַּם הַיּוֹם, גַּם בְּלִי מִלְחָמָה (אֶלָּא בִּתְנָאִים מְסֻיָּמִים).

אִם כֵּן, לָמָּה נֶאֱמַר הָאִסּוּר דַּוְקָא אֵצֶל הַלּוֹחֲמִים בַּמָּצוֹר?

💡 הָרַמְבַּ״ן מְבָאֵר
הָאִסּוּר נֶאֱמַר בִּמְיֻחָד לַלּוֹחֲמִים, כִּי זוֹ דֻּגְמָה מְצוּיָה: בִּסְעָרַת הַקְּרָב וּבִימֵי מָצוֹר אֲרֻכִּים יֵשׁ כַּעַס גָּדוֹל עַל הָעִיר שֶׁצָּרִים עָלֶיהָ, וְהַלּוֹחֲמִים עֲלוּלִים לְהַשְׁחִית אֶת הַסְּבִיבָה מִתּוֹךְ כַּעַס, נַקְמָנוּת אוֹ חֹסֶר מַעַשׂ.
הַתּוֹרָה מַזְהִירָה בַּדֻּגְמָה הַשְּׁכִיחָה הַזּוֹ – אֲבָל הָאִסּוּר עַצְמוֹ הוּא עַל כָּל דָּבָר.

וְכָךְ כּוֹתֵב בַּעַל סֵפֶר הַחִנּוּךְ עַל הַמִּצְוָה שֶׁבַּפָּרָשָׁה שֶׁלָּנוּ: ׳כְּמוֹ כֵן נִכְנָס תַּחַת זֶה הַלָּאו שֶׁלֹּא לַעֲשׂוֹת שׁוּם הֶפְסֵד, כְּגוֹן לִשְׂרֹף אוֹ לִקְרֹעַ בֶּגֶד אוֹ לְשַׁבֵּר כְּלִי לְבַטָּלָה׳ – כְּלוֹמַר בְּלִי שׁוּם צֹרֶךְ.

וּמָה הַמַּטָּרָה? ׳שֶׁהוּא כְּדֵי לְלַמֵּד נַפְשֵׁנוּ לֶאֱהֹב הַטּוֹב וְהַתּוֹעֶלֶת וּלְהִדָּבֵק בּוֹ׳.
הִנֵּה הַלֵּב שֶׁל הַמִּצְוָה: ה׳ רוֹצֶה שֶׁנִּהְיֶה אֲנָשִׁים שֶׁאוֹהֲבִים אֶת הַטּוֹב. מִי שֶׁמִּתְרַגֵּל לִשְׁמֹר – לוֹמֵד לֶאֱהֹב; וּמִי שֶׁמִּתְרַגֵּל לְקַלְקֵל – נַפְשׁוֹ מִתְקַלְקֶלֶת.

📚 סֵפֶר הַחִנּוּךְ: חֲסִידִים וּרְשָׁעִים
׳וְזֶהוּ דֶּרֶךְ הַחֲסִידִים וְאַנְשֵׁי מַעֲשֶׂה, אוֹהֲבִים שָׁלוֹם וּשְׂמֵחִים בְּטוֹב הַבְּרִיּוֹת וּמְקָרְבִים אוֹתָן לַתּוֹרָה, וְלֹא יְאַבְּדוּ אֲפִלּוּ גַּרְגֵּר שֶׁל חַרְדָּל בָּעוֹלָם׳.
׳וְלֹא כֵן הָרְשָׁעִים, אֲחֵיהֶם שֶׁל מַזִּיקִין, שְׂמֵחִים בְּהַשְׁחָתַת עוֹלָם וְהֵמָּה הַמַּשְׁחִיתִים׳.
גַּרְגֵּר חַרְדָּל הוּא זֶרַע זָעִיר – וְאֵצֶל הַצַּדִּיק גַּם הוּא יָקָר, כִּי ה׳ בָּרָא אוֹתוֹ.

וְעַכְשָׁיו נָבִין מָה בֶּאֱמֶת הוֹלֵךְ כָּאן: אַתָּה חוֹשֵׁב שֶׁאַתָּה מַשְׁחִית רַק אֹכֶל? רַק קִיר, רַק תִּיק אוֹ מַכְשִׁיר? אַתָּה מַשְׁחִית אֶת הַנֶּפֶשׁ.
גְּרָפִיטִי עַל קִיר, בֶּגֶד שֶׁנִּקְרַע סְתָם, חֵפֶץ טוֹב שֶׁנִּזְרָק, אֹכֶל שֶׁהוּשְׁלַךְ לַפַּח – כָּל אֵלֶּה הַשְׁחָתָה.
אָדָם יְרֵא שָׁמַיִם, שֶׁהוֹלֵךְ בְּדֶרֶךְ הַתּוֹרָה וּמְקַבֵּל עָלָיו אֶת הָאִסּוּר הַזֶּה, אוֹהֵב אֶת בִּנְיַן הָעוֹלָם וְלֹא אֶת הַשְׁחָתָתוֹ: תָּמִיד לִבְנוֹת, תָּמִיד לְתַקֵּן, וְלֹא לְקַלְקֵל.

✨ לְסִכּוּם

  • לִפְנֵי מִלְחֶמֶת רְשׁוּת יֵשׁ שְׁנֵי שְׁלַבִּים: רִאשׁוֹן – קְרִיאָה לְשָׁלוֹם, וְאִם הָעִיר נִכְנַעַת וּמְשַׁלֶּמֶת מַס אָסוּר לְהִלָּחֵם בָּהּ; שֵׁנִי – אִם לֹא הִשְׁלִימָה, מֻתָּר לְהִלָּחֵם וְלָצוּר עָלֶיהָ.

  • ׳וּנְתָנָהּ ה׳ אֱלֹהֶיךָ בְּיָדֶךָ׳ – רַשִׁ״י: אִם עָשִׂיתָ כָּל הָאָמוּר בָּעִנְיָן, סוֹף שֶׁה׳ נוֹתְנָהּ בְּיָדֶךָ. הַנִּצָּחוֹן מַתָּנָה מֵה׳.
  • רַשִׁ״י עַל ׳כִּי תִקְרַב אֶל עִיר׳: בְּמִלְחֶמֶת הָרְשׁוּת הַכָּתוּב מְדַבֵּר – בֶּעָרִים הָרְחוֹקוֹת. בְּעַמֵּי כְּנַעַן שֶׁבָּאָרֶץ, מִלְחֶמֶת מִצְוָה, הַדִּין אַחֵר.
  • בַּמָּצוֹר אָסוּר לְהַשְׁחִית עֵץ פְּרִי – ׳כִּי מִמֶּנּוּ תֹאכֵל׳. רַשִׁ״י: ׳כִּי הָאָדָם עֵץ הַשָּׂדֶה׳ – שֶׁמָּא הָעֵץ הוּא אָדָם? וּכְשֶׁצְּרִיכִים עֵצִים – כּוֹרְתִים עֲצֵי סְרָק.

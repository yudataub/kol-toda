בַּשִּׁעוּר הַזֶּה אֲנַחְנוּ נִכְנָסִים אֶל הַמִּטְבָּח הַיְּהוּדִי. הַתּוֹרָה מְלַמֶּדֶת אוֹתָנוּ מָה מֻתָּר לְהַכְנִיס לַפֶּה וּמָה אָסוּר: אֵילוּ בְּהֵמוֹת, אֵילוּ דָּגִים וְאֵילוּ עוֹפוֹת.
נִלְמַד אֶת סִימָנֵי הַטָּהֳרָה – פַּרְסָה שְׁסוּעָה וּמַעֲלַת גֵּרָה בַּבְּהֵמָה, סְנַפִּיר וְקַשְׂקֶשֶׂת בַּדָּג – נַכִּיר אֶת בַּעֲלֵי הַחַיִּים שֶׁ׳מְּבַלְבְּלִים׳ כְּמוֹ הַגָּמָל וְהַחֲזִיר, וְנִרְאֶה שֶׁבָּעוֹפוֹת אֵין סִימָנִים אֶלָּא רְשִׁימָה וּמָסֹרֶת.
בַּסּוֹף נִלְמַד עוֹד שְׁלוֹשָׁה אִסּוּרִים: שֶׁרֶץ הָעוֹף, נְבֵלָה וּבָשָׂר בְּחָלָב.
וּמֵעַל הַכֹּל עוֹמֶדֶת שׁוּרָה אַחַת: ׳כִּי עַם קָדוֹשׁ אַתָּה לַה׳ אֱלֹהֶיךָ׳. ה׳ בָּחַר בָּנוּ וְאוֹהֵב אוֹתָנוּ, וְלָכֵן הוּא שׁוֹמֵר עָלֵינוּ שֶׁנִּהְיֶה קְדוֹשִׁים – גַּם בַּדְּבָרִים הַפְּשׁוּטִים שֶׁבַּחַיִּים, גַּם בַּצַּלַּחַת שֶׁעַל הַשֻּׁלְחָן 💛
בַּשִּׁעוּר: 🍽️ לֹא תֹאכַל כָּל תּוֹעֵבָה – הַפְּתִיחָה לַנּוֹשֵׂא 🐄 סִימָנֵי בְּהֵמָה טְהוֹרָה – פַּרְסָה שְׁסוּעָה וּמַעֲלַת גֵּרָה ⚖️ ׳פּוֹרֵט אֶת הַמּוּעָט׳ – וְהַחַיָּה בִּכְלַל בְּהֵמָה 🐪 אֵלֶּה שֶׁמְּבַלְבְּלִים – גָּמָל, אַרְנֶבֶת, שָׁפָן וַחֲזִיר 🐟 דָּגִים – סְנַפִּיר וְקַשְׂקֶשֶׂת 🦅 עוֹפוֹת – רְשִׁימָה, מָסֹרֶת וְשֶׁרֶץ הָעוֹף 🦁 הָרַמְבַּ״ן: אַכְזָרִיּוּת וַעֲדִינוּת 🥛 עוֹד שְׁנֵי אִסּוּרִים – נְבֵלָה וּבָשָׂר בְּחָלָב 🏷️ עַם קָדוֹשׁ – גַּם בַּסּוּפֶּר וְגַם בְּחוּץ לָאָרֶץ

׳כָּל תּוֹעֵבָה׳ – כָּל שֶׁתִּעַבְתִּי לְךָ. ...בִּשֵּׁל בָּשָׂר בְּחָלָב, הֲרֵי דָּבָר שֶׁתִּעַבְתִּי לְךָ, וְהִזְהִיר כָּאן עַל אֲכִילָתוֹ.

כְּלוֹמַר: ׳תּוֹעֵבָה׳ אֵינָהּ רַק מִלָּה כְּלָלִית. רַשִׁ״י מְבָאֵר שֶׁהַפָּסוּק מְכַוֵּן אֶל דְּבָרִים שֶׁה׳ אָסַר, וּמוֹסִיף אַזְהָרָה גַּם עַל אֲכִילָתָם – כְּמוֹ בָּשָׂר שֶׁבֻּשַּׁל בְּחָלָב.

וְכָךְ בָּנוּי הַפֶּרֶק, בְּסֵדֶר בָּרוּר:
בְּהֵמוֹת – פְּסוּקִים ד׳–ח׳.

עוֹפוֹת – פְּסוּקִים י״א–כ׳.
אִסּוּרִים נוֹסָפִים – פָּסוּק כ״א.

וְיֵשׁ כָּאן סֵדֶר נוֹסָף, יָפֶה מְאוֹד. בַּבְּהֵמוֹת: פְּסוּקִים ד׳–ו׳ מְלַמְּדִים מָה מֻתָּר, וּפְסוּקִים ז׳–ח׳ – מָה אָסוּר. גַּם בַּדָּגִים: פָּסוּק ט׳ הַמֻּתָּר, פָּסוּק י׳ הָאָסוּר.
וְשִׂימוּ לֵב: הָאֲסוּרִים שֶׁהַתּוֹרָה טוֹרַחַת לְפָרֵט בַּבְּהֵמוֹת הֵם דַּוְקָא אֵלֶּה שֶׁיֵּשׁ לָהֶם רַק סִימָן אֶחָד – כִּי הֵם עֲלוּלִים לְבַלְבֵּל אוֹתָנוּ.

לִפְנֵי הַכֹּל, בְּפָסוּק ב׳, הַתּוֹרָה אוֹמֶרֶת אֶת הַטַּעַם: ׳כִּי עַם קָדוֹשׁ אַתָּה לַה׳ אֱלֹהֶיךָ, וּבְךָ בָּחַר ה׳ לִהְיוֹת לוֹ לְעַם סְגֻלָּה מִכֹּל הָעַמִּים׳.
ה׳ בָּחַר בָּנוּ מִתּוֹךְ אַהֲבָה, וְכָל מָה שֶׁהוּא מְבַקֵּשׁ מֵאִתָּנוּ נוֹבֵעַ מֵהָאַהֲבָה הַזּוֹ: הוּא רוֹצֶה שֶׁנִּהְיֶה שֶׁלּוֹ, קְרוֹבִים אֵלָיו וּקְדוֹשִׁים כְּמוֹ אוֹצָר יָקָר.

וְיֵשׁ גַּם סִימָן פְּנִימִי: ׳מַעֲלַת גֵּרָה׳. לְרֹב בַּעֲלֵי הַחַיִּים יֵשׁ קֵבָה אַחַת, אֲבָל לַפָּרָה יֵשׁ כַּמָּה קֵבוֹת: הָאֹכֶל נִכְנָס לַקֵּבָה, הַפָּרָה מַעֲלָה אוֹתוֹ בַּחֲזָרָה אֶל הַפֶּה וְלוֹעֶסֶת אוֹתוֹ שׁוּב, וְרַק אָז הוּא עוֹבֵר הָלְאָה לְעִכּוּל נוֹסָף. זוֹ הַעֲלָאַת הַגֵּרָה.

✨ לְסִכּוּם

  • הַפָּסוּק ׳לֹא תֹאכַל כָּל תּוֹעֵבָה׳ פּוֹתֵחַ אֶת הַנּוֹשֵׂא, וְאַחֲרָיו הַתּוֹרָה מְפָרֶטֶת בְּסֵדֶר: בְּהֵמוֹת, דָּגִים, עוֹפוֹת וְאִסּוּרִים נוֹסָפִים – וּבְכָל סוּג גַּם הַמֻּתָּר וְגַם הָאָסוּר.
  • סִימָנֵי בְּהֵמָה טְהוֹרָה: פַּרְסָה שְׁסוּעָה (סִימָן חִיצוֹנִי) וּמַעֲלַת גֵּרָה (סִימָן פְּנִימִי). צְרִיכִים אֶת כָּל הַסִּימָנִים – לְגָמָל, לְאַרְנֶבֶת וּלְשָׁפָן יֵשׁ רַק גֵּרָה, וְלַחֲזִיר רַק פַּרְסָה.
  • סִימָנֵי דָּג טָהוֹר: סְנַפִּיר וְקַשְׂקֶשֶׂת. לַכְּרִישׁ יֵשׁ סְנַפִּיר בְּלִי קַשְׂקֶשֶׂת – וְהוּא אָסוּר.
  • בָּעוֹפוֹת אֵין סִימָנִים אֶלָּא רְשִׁימָה שֶׁל אֲסוּרִים, וְאוֹכְלִים רַק עוֹף שֶׁיֵּשׁ עָלָיו מָסֹרֶת. רַשִׁ״י: בַּבְּהֵמוֹת הַתּוֹרָה מוֹנָה אֶת הַטְּהוֹרוֹת וּבָעוֹפוֹת אֶת הַטְּמֵאִים, כִּי ׳פּוֹרֵט אֶת הַמּוּעָט׳.
  • רַשִׁ״י עַל ׳שֶׁרֶץ הָעוֹף׳: הֵם הַנְּמוּכִים הָרוֹחֲשִׁים עַל הָאָרֶץ – זְבוּבִים, צְרָעוֹת וַחֲגָבִים טְמֵאִים. וְעוֹד שְׁנֵי אִסּוּרִים בְּפָסוּק כ״א: נְבֵלָה – בְּהֵמָה בְּלִי שְׁחִיטָה כְּשֵׁרָה, וּבָשָׂר בְּחָלָב.
  • הַכֹּל מִשּׁוּם ׳כִּי עַם קָדוֹשׁ אַתָּה לַה׳ אֱלֹהֶיךָ׳: ה׳ בָּחַר בָּנוּ מֵאַהֲבָה, הַכְּשֵׁרוּת מַבְדִּילָה אוֹתָנוּ מִסְּבִיבָתֵנוּ, וּבְכָל הֶכְשֵׁר שֶׁאֲנַחְנוּ בּוֹדְקִים אֲנַחְנוּ בּוֹחֲרִים בַּה׳ בַּחֲזָרָה.

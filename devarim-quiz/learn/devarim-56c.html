הַבְּרָכוֹת וְהַקְּלָלוֹת

הַחֹמֶר הַמָּלֵא: ׳וּבָאוּ עָלֶיךָ כָּל הַבְּרָכוֹת הָאֵלֶּה׳, בְּרִית עַרְבוֹת מוֹאָב וַעֲבוֹדַת ה׳ בְּשִׂמְחָה (פָּרָשַׁת כִּי תָבוֹא, פֶּרֶק כ״ח)
🎯 לַחִידוֹן🏠 כָּל הַחִידוֹנִים
לִפְנֵי שֶׁעַם יִשְׂרָאֵל נִכְנָס לָאָרֶץ, מֹשֶׁה רַבֵּנוּ מַעֲמִיד לְפָנָיו שְׁתֵּי דְּרָכִים: דֶּרֶךְ הַבְּרָכָה וְדֶרֶךְ הַקְּלָלָה.
אִם נִשְׁמַע בְּקוֹל ה׳ – תָּבוֹא עָלֵינוּ בְּרָכָה בְּכָל מָקוֹם וּבְכָל דָּבָר: בָּעִיר וּבַשָּׂדֶה, בַּיְּלָדִים, בַּשָּׂדוֹת, בַּגְּשָׁמִים וּבַבִּטָּחוֹן. וְאִם חַס וְשָׁלוֹם נִתְרַחֵק – הַטּוֹב הַזֶּה יֶחְסַר.
בַּשִּׁעוּר הַזֶּה נִלְמַד אֶת הַבְּרָכוֹת בִּפְרוֹטְרוֹט, נַכִּיר אֶת מִבְנֵה הַפֶּרֶק, נִלְמַד אֶת הַקְּלָלוֹת רַק בִּכְלָלִיּוּת, וְנַגִּיעַ אֶל הַמִּשְׁפָּט הַגָּדוֹל שֶׁבְּאֶמְצַע הַפֶּרֶק: ׳תַּחַת אֲשֶׁר לֹא עָבַדְתָּ אֶת ה׳ אֱלֹהֶיךָ בְּשִׂמְחָה וּבְטוּב לֵבָב׳.
וּבְעִקָּר נִרְאֶה אֵיךְ כָּל הַפֶּרֶק הַזֶּה הוּא דִּבּוּר שֶׁל אַבָּא אוֹהֵב: ה׳ רוֹצֶה לְהַשְׁפִּיעַ עָלֵינוּ טוֹב, וְהוּא מְסַפֵּר לָנוּ מֵרֹאשׁ אֵיךְ לְקַבֵּל אוֹתוֹ.
בַּשִּׁעוּר: 👂 ׳אִם שָׁמוֹעַ תִּשְׁמַע׳ – הַתְּנַאי שֶׁל הַבְּרָכָה 🧩 מִבְנֵה הַפֶּרֶק – כְּלָל, פְּרָט וּכְלָל 🧺 בְּרָכוֹת בְּכָל מָקוֹם – בָּעִיר, בַּשָּׂדֶה, בַּסַּל וּבַבַּיִת 👑 בְּרָכוֹת לְכָל הָעָם – בִּטָּחוֹן, גֶּשֶׁם וּכְבוֹד 👤 מְמַיְּנִים אֶת הַבְּרָכוֹת – וּמִלּוֹן קָטָן מֵרַשִׁ״י 🌗 הַקְּלָלוֹת – הַצַּד הַהָפוּךְ שֶׁל הַבְּרָכָה 😊 ׳בְּשִׂמְחָה וּבְטוּב לֵבָב׳ – הַסִּבָּה שֶׁבְּאֶמְצַע הַפֶּרֶק 📖 אֵיךְ קוֹרְאִים אֶת הַפָּרָשָׁה – וּמָה הַנֶּחָמָה 🤝 בְּרִית עַרְבוֹת מוֹאָב – ׳אֵלֶּה דִבְרֵי הַבְּרִית׳

׳לִכְרֹת אֶת בְּנֵי יִשְׂרָאֵל׳ – שֶׁיְּקַבְּלוּ עֲלֵיהֶם אֶת הַתּוֹרָה בְּאָלָה וּבִשְׁבוּעָה.
׳מִלְּבַד הַבְּרִית׳ – קְלָלוֹת שֶׁבְּתוֹרַת כֹּהֲנִים שֶׁנֶּאֶמְרוּ בְּסִינַי.

הֵיכָן עוֹמֵד הָעָם? בְּעַרְבוֹת מוֹאָב – מִמִּזְרָח לַיַּרְדֵּן, מוּל יְרִיחוֹ. כְּדַאי לְהַרְאוֹת זֹאת עַל הַמַּפָּה: הָעָם עוֹמֵד מַמָּשׁ עַל סַף הָאָרֶץ, וְהַיַּרְדֵּן לְפָנָיו.

💡 אִם כְּבָר הָיְתָה בְּרִית בְּחֹרֵב – לָמָּה בְּרִית נוֹסֶפֶת?
שְׁתֵּי תְּשׁוּבוֹת:
א. צָרִיךְ שֶׁתִּהְיֶה הַמְשָׁכָה שֶׁל בְּרִית סִינַי בְּאֶרֶץ יִשְׂרָאֵל. וְדַוְקָא מֹשֶׁה, הַמְתַוֵּךְ שֶׁל בְּרִית סִינַי, צָרִיךְ לִכְרֹת אוֹתָהּ – כְּדֵי שֶׁלֹּא יֹאמְרוּ שֶׁעִם הַחְלָפַת הַמַּנְהִיג הִשְׁתַּנּוּ תְּנָאֵי הַבְּרִית.
ב. לְפִי הַגְּמָרָא (שַׁבָּת פ״ח.), בְּרִית סִינַי נַעֲשְׂתָה מִתּוֹךְ הִתְרַגְּשׁוּת וְאֵימָה גְּדוֹלָה, וּבְרִית עַרְבוֹת מוֹאָב נַעֲשֵׂית מִתּוֹךְ דַּעַת מְלֵאָה וּצְלוּלָה, אַחֲרֵי אַרְבָּעִים שָׁנָה שֶׁל לִמּוּד וְהִתְנַסּוּת בַּמִּדְבָּר.

הַבְּרִית הַשְּׁנִיָּה מְחַזֶּקֶת אֶת הָרִאשׁוֹנָה וְנוֹתֶנֶת לָהּ מַשְׁמָעוּת נוֹסֶפֶת: לֹא רַק עַם שֶׁקִּבֵּל תּוֹרָה בַּמִּדְבָּר, אֶלָּא עַם ה׳ שֶׁחַי לְפִי הַתּוֹרָה בְּאַרְצוֹ.
וְהַבְּרָכוֹת וְהַקְּלָלוֹת שֶׁל הַפֶּרֶק הֵן ׳דִּבְרֵי הַבְּרִית׳ עַצְמָהּ: מִי שֶׁכּוֹרֵת בְּרִית מְקַבֵּל עָלָיו הִתְחַיְּבוּת, וּמַכִּיר גַּם בַּטּוֹב שֶׁיָּבוֹא וְגַם בַּחֶסְרוֹן שֶׁיִּהְיֶה אִם יַעֲזֹב.

🤔 כֹּחַ שֶׁל מִלִּים
לָמָּה כָּל כָּךְ מְשַׂמֵּחַ לְקַבֵּל בְּרָכָה, וְכָל כָּךְ לֹא נָעִים לִשְׁמֹעַ קְלָלָה?
כִּי לַמִּלִּים יֵשׁ כֹּחַ. בְּרָכָה שֶׁמְּקַבְּלִים מֵאַבָּא, מֵאִמָּא אוֹ מֵהָרַב הוֹלֶכֶת אִתָּנוּ וּמְחַזֶּקֶת אֶת הַבִּטָּחוֹן בַּיְּכוֹלוֹת שֶׁלָּנוּ. וְכָל שֶׁכֵּן בְּרָכָה שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא בְּעַצְמוֹ.
לָכֵן כְּדַאי לְהַרְגִּיל אֶת הַפֶּה לְבָרֵךְ וּלְדַבֵּר טוֹב.
💛 אַהֲבַת הַשֵּׁם
ה׳ כָּרַת אִתִּי בְּרִית – אִתִּי, בְּעַרְבוֹת מוֹאָב וּבְכָל דּוֹר. גַּם כְּשֶׁאֲנִי קָטָן וְטוֹעֶה, הַקֶּשֶׁר אֵינוֹ נִפְסָק, כִּי בְּרִית הִיא לְעוֹלָם. זוֹ הַמַּתָּנָה הֲכִי גְּדוֹלָה שֶׁיֵּשׁ לְעַם יִשְׂרָאֵל 💛

✨ לְסִכּוּם

  • הַבְּרָכוֹת פּוֹתְחוֹת בִּתְנַאי: ׳אִם שָׁמוֹעַ תִּשְׁמַע בְּקוֹל ה׳׳ – אָז ׳וּבָאוּ עָלֶיךָ כָּל הַבְּרָכוֹת הָאֵלֶּה וְהִשִּׂיגֻךָ׳: הַבְּרָכוֹת רוֹדְפוֹת אַחֲרֵינוּ בְּעַצְמָן.
  • הַפֶּרֶק מִתְחַלֵּק לִשְׁנַיִם: בְּרָכוֹת א׳–י״ד, קְלָלוֹת ט״ו–ס״ח. וְהַבְּרָכוֹת בְּנוּיוֹת כְּלָל (א׳–ב׳), פְּרָט (ג׳–י״ב) וּכְלָל (י״ג–י״ד).
  • בִּרְכוֹת הַיָּחִיד: בָּעִיר וּבַשָּׂדֶה, פְּרִי הַבֶּטֶן וְהָאֲדָמָה, טַנְאֲךָ וּמִשְׁאַרְתֶּךָ, וּ׳בְבֹאֶךָ וּבְצֵאתֶךָ׳ – יְצִיאָה מִן הָעוֹלָם בְּלֹא חֵטְא.
  • בִּרְכוֹת הָעָם: אוֹיְבִים נִגָּפִים וּבוֹרְחִים בְּשִׁבְעָה דְרָכִים, ׳עַם קָדוֹשׁ׳, ׳אוֹצָרוֹ הַטּוֹב׳ – מְטַר אַרְצְךָ בְּעִתּוֹ – וְ׳לְרֹאשׁ וְלֹא לְזָנָב׳.
  • בַּפְּתִיחָה לַקְּלָלוֹת נוֹסְפָה הַמִּלָּה ׳וְחֻקֹּתָיו׳, וּפְסוּקֵי הַקְּלָלוֹת רַבִּים יוֹתֵר – כִּי מֹשֶׁה רוֹצֶה שֶׁהָעָם יִקַּח אַחֲרָיוּת לִפְנֵי הַכְּנִיסָה לָאָרֶץ. ׳קְלָלָה׳ הִיא מָה שֶׁקּוֹרֶה כְּשֶׁמִּתְרַחֲקִים מִן ה׳.
  • ׳תַּחַת אֲשֶׁר לֹא עָבַדְתָּ אֶת ה׳ אֱלֹהֶיךָ בְּשִׂמְחָה וּבְטוּב לֵבָב מֵרֹב כֹּל׳ – ה׳ רוֹצֶה מִצְווֹת מִתּוֹךְ שִׂמְחָה, וּמִכָּאן ׳עִבְדוּ אֶת ה׳ בְּשִׂמְחָה׳.
  • ׳אֵין מַפְסִיקִין בַּקְּלָלוֹת׳, וְקוֹרְאִים אֶת הַפָּרָשָׁה לִפְנֵי רֹאשׁ הַשָּׁנָה – ׳תִּכְלֶה שָׁנָה וְקִלְלוֹתֶיהָ׳. וּמִיָּד אַחֲרֶיהָ ׳אַתֶּם נִצָּבִים׳: יֵשׁ לָכֶם עֲמִידָה!
  • ׳אֵלֶּה דִבְרֵי הַבְּרִית׳ – בְּרִית עַרְבוֹת מוֹאָב, מוּל יְרִיחוֹ, מִתּוֹךְ דַּעַת מְלֵאָה אַחֲרֵי אַרְבָּעִים שָׁנָה, מְחַזֶּקֶת אֶת בְּרִית חֹרֵב – קֶשֶׁר שֶׁל אַהֲבָה בֵּין ה׳ לְעַמּוֹ, לְעוֹלָם.

וְהָיָה אִם שָׁמֹעַ

הַחֹמֶר הַמָּלֵא: הַפָּרָשָׁה הַשְּׁנִיָּה שֶׁל קְרִיאַת שְׁמַע – שָׂכָר וָעֹנֶשׁ בְּאֶרֶץ הַגְּשָׁמִים (פֶּרֶק י״א)
🎯 לַחִידוֹן🏠 כָּל הַחִידוֹנִים
בַּשִּׁעוּר הַקּוֹדֵם לָמַדְנוּ שֶׁאֶרֶץ יִשְׂרָאֵל הִיא אֶרֶץ הָרִים וּבְקָעוֹת לִמְטַר הַשָּׁמַיִם תִּשְׁתֶּה מָּיִם – אֶרֶץ שֶׁתְּלוּיָה בַּגֶּשֶׁם שֶׁה׳ נוֹתֵן.
עַכְשָׁו הַתּוֹרָה מַמְשִׁיכָה וְאוֹמֶרֶת: הַדָּבָר הַזֶּה נָכוֹן לְכָל עַם יִשְׂרָאֵל יַחַד. אִם נִשְׁמַע אֶל הַמִּצְווֹת – יֵרְדוּ גְּשָׁמִים בְּעִתָּם, וְנֵשֵׁב בָּאָרֶץ לְאֹרֶךְ יָמִים. וְאִם חָלִילָה נָסוּר מִן הַדֶּרֶךְ – יֵעָצְרוּ הַשָּׁמַיִם.
זֶה הָעִקָּרוֹן הַגָּדוֹל שֶׁל שָׂכָר וָעֹנֶשׁ, וְהוּא כָּתוּב בַּפָּרָשָׁה שֶׁכֻּלָּנוּ מַכִּירִים: ׳וְהָיָה אִם שָׁמֹעַ׳ – הַפָּרָשָׁה הַשְּׁנִיָּה שֶׁל קְרִיאַת שְׁמַע, שֶׁאוֹתָהּ אֲנַחְנוּ אוֹמְרִים בְּכָל בֹּקֶר וּבְכָל עֶרֶב, וְכוֹתְבִים בַּתְּפִלִּין וּבַמְּזוּזָה.
וּבְעִקָּר נִרְאֶה כַּמָּה ה׳ קָרוֹב אֵלֵינוּ: הוּא לֹא נָתַן לָנוּ אֶרֶץ שֶׁמַּסְתַּדֶּרֶת לְבַד, אֶלָּא אֶרֶץ שֶׁבָּהּ עֵינָיו נְתוּנוֹת תָּמִיד, וְאֵלֶיהָ הוּא מַבִּיט בְּכָל יוֹם 💛
בַּשִּׁעוּר: 🌊 מִפְלַס הַכִּנֶּרֶת – אֶרֶץ שֶׁתְּלוּיָה בַּגֶּשֶׁם 📖 הַפָּרָשָׁה הַשְּׁנִיָּה שֶׁל קְרִיאַת שְׁמַע 👑 לָמָּה ׳שְׁמַע׳ קוֹדֶמֶת? עֹל מַלְכוּת שָׁמַיִם וְעֹל מִצְווֹת 🔀 ׳אִם׳ – תְּנַאי, וְתוֹרָה שֶׁהִיא חֲדָשָׁה בְּכָל יוֹם 🙏 לְאַהֲבָה אֶת ה׳ וּלְעָבְדוֹ בְּכָל לְבַבְכֶם – עֲבוֹדָה שֶׁבַּלֵּב 🌦️ הַשָּׂכָר: מָטָר בְּעִתּוֹ, יוֹרֶה וּמַלְקוֹשׁ ⚠️ הִשָּׁמְרוּ לָכֶם – וְעָצַר אֶת הַשָּׁמַיִם 🪧 צִיּוּנִים בַּדֶּרֶךְ, לִמּוּד לַבָּנִים וְאֹרֶךְ יָמִים 🎶 פְּסוּקִים י״ג–י״ז בְּעַל פֶּה, וְסִכּוּם הַפֵּרוּשִׁים

יב אֶרֶץ אֲשֶׁר ה׳ אֱלֹהֶיךָ דֹּרֵשׁ אֹתָהּ תָּמִיד עֵינֵי ה׳ אֱלֹהֶיךָ בָּהּ מֵרֵשִׁית הַשָּׁנָה וְעַד אַחֲרִית שָׁנָה

בְּיוֹם שֶׁל גֶּשֶׁם חָזָק פּוֹתְחִים אֶת הַחֲדָשׁוֹת וְשׁוֹמְעִים טַבְלָה שְׁלֵמָה שֶׁל מְדִידַת כַּמֻּיּוֹת הַמִּשְׁקָעִים: כַּמָּה מִילִימֶטְרִים יָרְדוּ בַּצָּפוֹן, כַּמָּה בַּמֶּרְכָּז וְכַמָּה בַּדָּרוֹם.
וְאַחַר כָּךְ כֻּלָּם מְחַכִּים לִשְׁמֹעַ אֶת הַמִּסְפָּר הֶחָשׁוּב בְּיוֹתֵר: מָה מִפְלַס הַכִּנֶּרֶת – כַּמָּה עָלוּ פְּנֵי הַמַּיִם בַּיָּם הַגָּדוֹל שֶׁלָּנוּ.

יֵשׁ אוֹמְרִים שֶׁהַכִּנֶּרֶת הִיא ׳הַבָּרוֹמֶטֶר הַלְּאֻמִּי׳. בָּרוֹמֶטֶר הוּא מַכְשִׁיר מְדִידָה, וְכָאן הַכַּוָּנָה: כְּשֶׁמּוֹדְדִים אֶת הַכִּנֶּרֶת, מוֹדְדִים בְּעֶצֶם אֶת מַצַּב הָרוּחַ שֶׁל כָּל הָאָרֶץ.
כִּנֶּרֶת מְלֵאָה – כֻּלָּנוּ שְׂמֵחִים. כִּנֶּרֶת הוֹלֶכֶת וּפוֹחֶתֶת וַחֲסֵרָה – כֻּלָּנוּ דּוֹאֲגִים. הָעִנְיָן שֶׁל הַמַּיִם מַעֲסִיק אֶת מְדִינַת יִשְׂרָאֵל כָּל הַחֹרֶף.

וְלָמָּה זֶה כָּךְ? כִּי כָּךְ לִמְּדָה אוֹתָנוּ הַתּוֹרָה בַּשִּׁעוּר הַקּוֹדֵם: אֶרֶץ יִשְׂרָאֵל תְּלוּיָה בַּגֶּשֶׁם. הִיא אֶרֶץ הָרִים וּבְקָעוֹת, בְּלִי נְהַר נִילוּס שֶׁמַּשְׁקֶה אוֹתָהּ, וְהַמַּיִם שֶׁלָּהּ יוֹרְדִים מִן הַשָּׁמַיִם.
וְעַל כָּךְ נֶאֱמַר: ׳אֶרֶץ אֲשֶׁר ה׳ אֱלֹהֶיךָ דֹּרֵשׁ אֹתָהּ תָּמִיד, עֵינֵי ה׳ אֱלֹהֶיךָ בָּהּ מֵרֵשִׁית הַשָּׁנָה וְעַד אַחֲרִית שָׁנָה׳.

📜 רַשִׁ״י
׳אֲשֶׁר ה׳ אֱלֹהֶיךָ דֹּרֵשׁ אֹתָהּ׳ – וַהֲלֹא אַף כָּל הָאֲרָצוֹת הוּא דּוֹרֵשׁ, אֶלָּא כִּבְיָכוֹל אֵינוֹ דּוֹרֵשׁ אֶלָּא אוֹתָהּ, וְעַל יְדֵי אוֹתָהּ דְּרִישָׁה שֶׁדּוֹרְשָׁהּ – דּוֹרֵשׁ אֶת כָּל הָאֲרָצוֹת עִמָּהּ.
׳תָּמִיד עֵינֵי ה׳ אֱלֹהֶיךָ בָּהּ׳ – לִרְאוֹת מָה הִיא צְרִיכָה.
׳מֵרֵשִׁית הַשָּׁנָה׳ – מֵרֹאשׁ הַשָּׁנָה נִדּוֹן מָה יְהֵא בְּסוֹפָהּ.

כְּלוֹמַר: ה׳ מַשְׁגִּיחַ עַל כָּל הָעוֹלָם, אֲבָל אֶת אֶרֶץ יִשְׂרָאֵל הוּא ׳דּוֹרֵשׁ׳ – מְחַפֵּשׂ וּמַבִּיט – בְּמִיוּחָד, וּבִזְכוּת הַהַשְׁגָּחָה עַל הָאָרֶץ שֶׁלָּנוּ מְקַבֵּל כָּל הָעוֹלָם אֶת שֶׁלּוֹ.
אֲנַחְנוּ אֵינֶנּוּ יוֹדְעִים אֶת הַחֶשְׁבּוֹנוֹת שֶׁל רִבּוֹנוֹ שֶׁל עוֹלָם, וְאָסוּר לָנוּ לִשְׁפֹּט מִי אָשֵׁם בִּבְצֹרֶת. אֲבָל אֶת הָעִקָּרוֹן הַתּוֹרָה אוֹמֶרֶת בְּפֵרוּשׁ: הָאָרֶץ הַזֹּאת צְמֵאָה לַמַּיִם, וְהָאָדָם שֶׁיּוֹשֵׁב בָּהּ מִתְפַּלֵּל וְשָׂם לֵב לְמַעֲשָׂיו.

ה׳ יָכוֹל הָיָה לָתֵת לָנוּ אֶרֶץ שֶׁל נְהָרוֹת, שֶׁאֵין צָרִיךְ לְבַקֵּשׁ בָּהּ כְּלוּם. הוּא בָּחַר לָתֵת לָנוּ דַּוְקָא אֶרֶץ שֶׁמְּרִימָה אֶת הָעֵינַיִם לַשָּׁמַיִם – כִּי הוּא אוֹהֵב אוֹתָנוּ וְרוֹצֶה שֶׁנִּהְיֶה קְרוֹבִים אֵלָיו כָּל הַשָּׁנָה 💛

יח וְשַׂמְתֶּם אֶת דְּבָרַי אֵלֶּה עַל לְבַבְכֶם וְעַל נַפְשְׁכֶם וּקְשַׁרְתֶּם אֹתָם לְאוֹת עַל יֶדְכֶם וְהָיוּ לְטוֹטָפֹת בֵּין עֵינֵיכֶם יט וְלִמַּדְתֶּם אֹתָם אֶת בְּנֵיכֶם לְדַבֵּר בָּם בְּשִׁבְתְּךָ בְּבֵיתֶךָ וּבְלֶכְתְּךָ בַדֶּרֶךְ וּבְשָׁכְבְּךָ וּבְקוּמֶךָ כ וּכְתַבְתָּם עַל מְזוּזוֹת בֵּיתֶךָ וּבִשְׁעָרֶיךָ

תִּקְרְאוּ אֶת הַמִּלִּים ׳וְהָיָה אִם שָׁמֹעַ תִּשְׁמְעוּ אֶל מִצְוֹתַי׳ – וּתְגַלּוּ שֶׁהֵן מֻכָּרוֹת לָכֶם מְאוֹד. לָמָּה? כִּי זוֹ הַפָּרָשָׁה הַשְּׁנִיָּה שֶׁל קְרִיאַת שְׁמַע, וְהִיא כְּתוּבָה בַּסִּדּוּר.
קֹדֶם אוֹמְרִים ׳שְׁמַע יִשְׂרָאֵל׳ וְ׳וְאָהַבְתָּ׳ (פֶּרֶק ו׳), וּמִיָּד אַחֲרֵיהֶן ׳וְהָיָה אִם שָׁמֹעַ׳ (פֶּרֶק י״א). אֲנַחְנוּ אוֹמְרִים אוֹתָהּ בְּכָל בֹּקֶר וּבְכָל עֶרֶב – בְּשַׁחֲרִית וּבְעַרְבִית, בִּימוֹת הַחֹל, בְּשַׁבָּת וּבֶחָגִים.

וְלֹא רַק בַּתְּפִלָּה. שִׂימוּ לֵב לַפְּסוּקִים שֶׁבְּסוֹף הַפָּרָשָׁה: ׳וְשַׂמְתֶּם אֶת דְּבָרַי אֵלֶּה עַל לְבַבְכֶם וְעַל נַפְשְׁכֶם, וּקְשַׁרְתֶּם אֹתָם לְאוֹת עַל יֶדְכֶם וְהָיוּ לְטוֹטָפֹת בֵּין עֵינֵיכֶם׳.
כְּבָר רָאִינוּ פְּסוּקִים דּוֹמִים בְּפָרָשַׁת שְׁמַע. מִכָּאן שֶׁגַּם פָּרָשַׁת ׳וְהָיָה אִם שָׁמֹעַ׳ כְּתוּבָה בַּתְּפִלִּין, בְּשֶׁל הַיָּד וּבְשֶׁל הָרֹאשׁ.

וְעוֹד כָּתוּב: ׳וּכְתַבְתָּם עַל מְזוּזוֹת בֵּיתֶךָ וּבִשְׁעָרֶיךָ׳. הַסּוֹפֵר שֶׁכּוֹתֵב קְלָף שֶׁל מְזוּזָה אֵינוֹ כּוֹתֵב עָלָיו רַק אֶת פָּרָשַׁת ׳שְׁמַע׳ – הוּא כּוֹתֵב שְׁתֵּי פָּרָשׁוֹת: ׳שְׁמַע׳ וְ׳וְהָיָה אִם שָׁמֹעַ׳.

💡 עַל הַקְּלָף שֶׁל הַמְּזוּזָה
אִם תִּסְתַּכְּלוּ בִּקְלָף שֶׁל מְזוּזָה לִפְנֵי שֶׁקּוֹבְעִים אוֹתוֹ, תִּרְאוּ שׁוּרוֹת שֶׁכְּתוּבוֹת בִּכְתָב אַשּׁוּרִי יָפֶה.
הַחֵלֶק הָרִאשׁוֹן הוּא ׳שְׁמַע יִשְׂרָאֵל׳, וְהַחֵלֶק הַשֵּׁנִי הוּא ׳וְהָיָה אִם שָׁמֹעַ׳. שְׁתֵּי הַפָּרָשׁוֹת שׁוֹמְרוֹת עַל פֶּתַח הַבַּיִת שֶׁלָּנוּ.

׳הִשָּׁמְרוּ לָכֶם׳ – אֵין אָדָם מוֹרֵד אֶלָּא מִתּוֹךְ שְׂבִיעָה.
׳לְדַבֵּר בָּם׳ – מִשָּׁעָה שֶׁהַבֵּן יוֹדֵעַ לְדַבֵּר.

וְאַל תִּשְׁכְּחוּ אֶת רַשִׁ״י הַמְּפֻרְסָם שֶׁל הַפָּרָשָׁה, שֶׁחַיָּבִים לָדַעַת: ׳וּלְעָבְדוֹ בְּכָל לְבַבְכֶם׳ – עֲבוֹדָה שֶׁהִיא בַּלֵּב, וְזוֹ הִיא תְּפִלָּה.

בַּפַּעַם הַבָּאָה שֶׁתַּגִּיעוּ בַּסִּדּוּר לַמִּלִּים ׳וְהָיָה אִם שָׁמֹעַ׳ – עִצְרוּ רֶגַע.

✨ לְסִכּוּם

  • ׳וְהָיָה אִם שָׁמֹעַ׳ הִיא הַפָּרָשָׁה הַשְּׁנִיָּה שֶׁל קְרִיאַת שְׁמַע: אוֹמְרִים אוֹתָהּ בֹּקֶר וָעֶרֶב, כּוֹתְבִים אוֹתָהּ בַּתְּפִלִּין, וְעַל קְלָף הַמְּזוּזָה הִיא כְּתוּבָה יַחַד עִם ׳שְׁמַע׳.
  • ׳שְׁמַע׳ קוֹדֶמֶת לָהּ כְּדֵי שֶׁנְּקַבֵּל תְּחִלָּה עֹל מַלְכוּת שָׁמַיִם וְאַחַר כָּךְ עֹל מִצְווֹת – קֹדֶם הַמֶּלֶךְ, אַחַר כָּךְ הַמִּצְווֹת.
  • הַמִּלָּה ׳אִם׳ פּוֹתַחַת תְּנַאי: אִם נִשְׁמַע – ׳וְנָתַתִּי מְטַר אַרְצְכֶם בְּעִתּוֹ, יוֹרֶה וּמַלְקוֹשׁ׳, דָּגָן תִּירוֹשׁ וְיִצְהָר, עֵשֶׂב לַבְּהֵמוֹת וְשֹׂבַע. וְאִם נָסוּר – ׳וְעָצַר אֶת הַשָּׁמַיִם׳ וְגָלוּת.
  • רַשִׁ״י הַמְּפֻרְסָם: ׳וּלְעָבְדוֹ בְּכָל לְבַבְכֶם׳ – עֲבוֹדָה שֶׁבַּלֵּב, זוֹ תְּפִלָּה. וְ׳לְאַהֲבָה׳ – כָּל מָה שֶׁתַּעֲשׂוּ, עֲשׂוּ מֵאַהֲבָה.
  • ׳אִם שָׁמֹעַ בַּיָּשָׁן – תִּשְׁמְעוּ בֶּחָדָשׁ׳, וְהַתּוֹרָה צְרִיכָה לִהְיוֹת חֲדָשָׁה בְּכָל יוֹם. וְ׳הִשָּׁמְרוּ לָכֶם׳ – אֵין אָדָם מוֹרֵד אֶלָּא מִתּוֹךְ שְׂבִיעָה.
  • גַּם בַּגָּלוּת נִשְׁאָרִים מְצֻיָּנִים בַּמִּצְווֹת, מְלַמְּדִים אֶת הַבָּנִים תּוֹרָה מִשָּׁעָה שֶׁהֵם יוֹדְעִים לְדַבֵּר, וּמִן הַמִּלִּים ׳לָתֵת לָהֶם׳ לוֹמְדִים תְּחִיַּת הַמֵּתִים מִן הַתּוֹרָה – כִּי ה׳ אוֹהֵב אוֹתָנוּ לָנֶצַח.

בַּשִּׁעוּר הַקּוֹדֵם לָמַדְנוּ עַל בַּקָּשַׁת מֹשֶׁה לְהִכָּנֵס לָאָרֶץ. עַכְשָׁו מֹשֶׁה מַמְשִׁיךְ וּמְלַמֵּד אֶת בְּנֵי יִשְׂרָאֵל עַל מָה הֵם צְרִיכִים לְהַקְפִּיד כְּשֶׁיִּכָּנְסוּ לְאֶרֶץ יִשְׂרָאֵל.
נִלְמַד אַרְבָּעָה פְּסוּקִים קְטַנִּים שֶׁיֵּשׁ בָּהֶם דָּבָר גָּדוֹל: ׳לֹא תֹסִפוּ עַל הַדָּבָר אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם וְלֹא תִגְרְעוּ מִמֶּנּוּ׳. אֶת מִצְווֹת הַתּוֹרָה עוֹשִׂים בְּדִיּוּק: לֹא מוֹסִיפִים וְלֹא מַחְסִירִים.
נִרְאֶה מָה רַשִׁ״י מְבָאֵר וּמָה הָרַמְבַּ״ן מְבָאֵר, נִלְמַד מָתַי דַּוְקָא כֵּן מֻתָּר לְהוֹסִיף, וּנְגַלֶּה שֶׁכָּל זֶה הוּא מֵאַהֲבָה: ה׳ נָתַן לָנוּ תּוֹרָה שְׁלֵמָה וּמֻשְׁלֶמֶת, כְּמוֹ מַתָּנָה מְדֻיֶּקֶת שֶׁהוּכְנָה בְּדִיּוּק בִּשְׁבִילֵנוּ 💛
בַּשִּׁעוּר: 🍬 בְּדִיּוּק – לֹא יוֹתֵר וְלֹא פָּחוֹת 💧 הַטִּפְטוּף – כַּמָּה מַיִם בְּדִיּוּק 👂 וְעַתָּה יִשְׂרָאֵל שְׁמַע – לְמַעַן תִּחְיוּ ⚖️ הַפָּסוּק עַצְמוֹ: לֹא תֹסִפוּ וְלֹא תִגְרְעוּ 📦 רַשִׁ״י: שָׁלֹשׁ דֻּגְמָאוֹת לָאִסּוּר 📚 הָרַמְבַּ״ן: תַּרְיַ״ג מִצְווֹת – לֹא 614 וְלֹא 612 📖 מָתַי כֵּן מֻתָּר לְהוֹסִיף? – לִמּוּד תּוֹרָה וּגְזֵרוֹת חֲכָמִים 💎 תּוֹרַת ה׳ תְּמִימָה – טַעַם אֶחָד לִשְׁנֵי הָאִסּוּרִים 🔗 בַּעַל פְּעוֹר – וְאַתֶּם הַדְּבֵקִים בַּה׳

🍬בְּדִיּוּק – לֹא יוֹתֵר וְלֹא פָּחוֹת

נַתְחִיל בְּחִידָה: בַּצִּנְצֶנֶת יֵשׁ סֻכָּרִיּוֹת. כַּמָּה? נַסּוּ לְנַחֵשׁ! הַתְּשׁוּבָה: 96 סֻכָּרִיּוֹת.
אֶחָד נִחֵשׁ 93, וְאֶחָד נִחֵשׁ 99. מִי יוֹתֵר צוֹדֵק? אַף אֶחָד! כְּשֶׁצְּרִיכִים לָדַעַת בְּדִיּוּק, לֹא מְשַׁנֶּה אִם נִחַשְׁתָּ פָּחוֹת אוֹ יוֹתֵר – לֹא קָלַעְתָּ.

📚 מָשָׁל: הַמִּתְכּוֹן שֶׁל הָעוּגָה
בְּהַרְבֵּה מַאֲכָלִים יֵשׁ מִתְכּוֹן מְדֻיָּק. אִם כָּתוּב 180 מִילִילִיטֶר שֶׁמֶן, וְשַׂמְתֶּם 230 אוֹ 150 – הָעוּגָה לֹא תֵּצֵא בַּמִּרְקָם הַנָּכוֹן.
לֹא מְשַׁנֶּה אִם הוֹסַפְתֶּם אוֹ הֶחְסַרְתֶּם: פָּשׁוּט לֹא עֲשִׂיתֶם בְּדִיּוּק.
וְכָךְ גַּם הַבַּנָּאִים שֶׁבּוֹנִים בָּתִּים: הֵם מְעַרְבְּבִים מֶלֶט, חָצָץ, חוֹל וּמַיִם לְפִי מִתְכּוֹן מְדֻיָּק. יוֹתֵר מַיִם אוֹ פָּחוֹת מַיִם – הַבַּיִת לֹא יַחֲזִיק.

יֵשׁ דְּבָרִים שֶׁפָּשׁוּט צָרִיךְ לַעֲשׂוֹת בְּדִיּוּק, וְהַהַצְלָחָה תְּלוּיָה בַּדִּיּוּק הַזֶּה.

וְכָךְ גַּם בְּאַהֲבָה: אַבָּא שֶׁמֵּכִין לִבְנוֹ תְּרוּפָה סוֹפֵר אֶת הַטִּפּוֹת בְּדִיּוּק, כִּי הוּא אוֹהֵב אוֹתוֹ וְרוֹצֶה בְּטוֹבָתוֹ. כָּךְ ה׳ נָתַן לָנוּ מִצְווֹת מְדֻיָּקוֹת – כָּל פְּרָט בָּהֶן הוּא לְטוֹבָתֵנוּ.

ה׳ לֹא נָתַן לָנוּ מִצְווֹת ׳בְּעֵרֶךְ׳. כָּל מִסְפָּר וְכָל פְּרָט נִבְחֲרוּ בְּדִיּוּק בִּשְׁבִילִי, כְּמוֹ מַתָּנָה שֶׁהוּכְנָה בְּאַהֲבָה בְּדִיּוּק בְּמִדָּה שֶׁלִּי 💛

💧הַטִּפְטוּף – כַּמָּה מַיִם בְּדִיּוּק

פַּעַם הִשְׁקוּ אֶת הַגִּנּוֹת בְּצִנּוֹר הַשְׁקָיָה אוֹ בְּמַמְטֵרוֹת. לִפְעָמִים הַצְּמָחִים קִבְּלוּ יוֹתֵר מַיִם מִמָּה שֶׁהֵם צְרִיכִים, וְלִפְעָמִים פָּחוֹת.

וְאָז הִמְצִיאוּ בְּאֶרֶץ יִשְׂרָאֵל אֶת הַטִּפְטָפוֹת: צִנּוֹר דַּק שֶׁמְּטַפְטֵף לְכָל צֶמַח בְּדִיּוּק אֶת כַּמּוּת הַמַּיִם שֶׁהוּא צָרִיךְ – וְכָךְ הוּא גָּדֵל הֵיטֵב.
הַהַמְצָאָה הַזּוֹ, שֶׁנּוֹלְדָה כָּאן בָּאָרֶץ, עָזְרָה לְחַקְלָאִים בְּכָל הָעוֹלָם.

זוֹ בְּדִיּוּק הַנִּמְשָׁל שֶׁל הַמִּצְווֹת: לִשְׁמֹר אֶת הַמִּצְווֹת בְּדִיּוּק – לֹא פָּחוֹת וְלֹא יוֹתֵר – זֶה מָה שֶׁמַּצְמִיחַ אוֹתָנוּ.

👂וְעַתָּה יִשְׂרָאֵל שְׁמַע – לְמַעַן תִּחְיוּ

מֹשֶׁה פּוֹתֵחַ: ׳וְעַתָּה יִשְׂרָאֵל שְׁמַע אֶל הַחֻקִּים וְאֶל הַמִּשְׁפָּטִים אֲשֶׁר אָנֹכִי מְלַמֵּד אֶתְכֶם לַעֲשׂוֹת׳.
שִׂימוּ לֵב: לֹא מַסְפִּיק לִשְׁמֹעַ וְלָדַעַת – לוֹמְדִים כְּדֵי לַעֲשׂוֹת.

וּמָה הַשָּׂכָר? ׳לְמַעַן תִּחְיוּ וּבָאתֶם וִירִשְׁתֶּם אֶת הָאָרֶץ אֲשֶׁר ה׳ אֱלֹהֵי אֲבֹתֵיכֶם נֹתֵן לָכֶם׳.
מִי שֶׁשּׁוֹמֵר אֶת הַמִּצְווֹת – זוֹכֶה לְחַיִּים וְזוֹכֶה לְאֶרֶץ יִשְׂרָאֵל. הַתּוֹרָה וְהָאָרֶץ קְשׁוּרוֹת זוֹ בְּזוֹ: הָאָרֶץ הִיא הַמָּקוֹם שֶׁבּוֹ מְקַיְּמִים אֶת הַתּוֹרָה בִּשְׁלֵמוּת.

בְּהֶמְשֵׁךְ אוֹמֵר מֹשֶׁה: ׳רְאֵה לִמַּדְתִּי אֶתְכֶם חֻקִּים וּמִשְׁפָּטִים כַּאֲשֶׁר צִוַּנִי ה׳ אֱלֹהָי לַעֲשׂוֹת כֵּן בְּקֶרֶב הָאָרֶץ אֲשֶׁר אַתֶּם בָּאִים שָׁמָּה לְרִשְׁתָּהּ׳.
מֹשֶׁה מַדְגִּישׁ: אֲנִי לִמַּדְתִּי אֶתְכֶם בְּדִיּוּק מָה שֶׁה׳ צִוָּה אוֹתִי – לֹא הוֹסַפְתִּי וְלֹא גָּרַעְתִּי. כָּךְ הוּא נוֹתֵן לָנוּ דֻּגְמָה אִישִׁית.

💡 זוֹ הֲלָכָה, לֹא רַק מוּסָר

וְעוֹד נְקֻדָּה בְּ׳לֹא תִגְרְעוּ׳: לִפְעָמִים אָדָם חוֹשֵׁב, ׳הַמִּצְוָה הַזֹּאת הָיְתָה חֲשׁוּבָה פַּעַם, אֲבָל הַיּוֹם הַטַּעַם שֶׁלָּהּ כְּבָר לֹא קַיָּם׳. זוֹ טָעוּת!
גַּם מִצְוָה שֶׁנִּדְמֶה לָנוּ שֶׁהַטַּעַם שֶׁלָּהּ נִגְמַר – נִשְׁאֶרֶת בִּמְלוֹאָהּ. דְּבֵקִים בְּתוֹרַת ה׳ הַתְּמִימָה בְּכָל דּוֹר וָדוֹר.

💛 אַהֲבַת הַשֵּׁם
הַתּוֹרָה שֶׁלִּי הִיא אוֹתָהּ תּוֹרָה שֶׁקִּבֵּל מֹשֶׁה בְּסִינַי – שְׁלֵמָה, בְּלִי שִׁנּוּי. זֶה מְמַלֵּא אוֹתִי גַּאֲוָה וְאַהֲבָה: אֲנִי מַחְזִיק בַּיָּד מַתָּנָה שֶׁה׳ נָתַן, בְּדִיּוּק כְּמוֹ שֶׁנְּתָנָהּ 💛

🔗בַּעַל פְּעוֹר – וְאַתֶּם הַדְּבֵקִים בַּה׳

📖 פֶּרֶק ד׳, פְּסוּקִים ג׳–ד׳ ←
ג עֵינֵיכֶם הָרֹאֹת אֵת אֲשֶׁר עָשָׂה ה׳ בְּבַעַל פְּעוֹר כִּי כָל הָאִישׁ אֲשֶׁר הָלַךְ אַחֲרֵי בַעַל פְּעוֹר הִשְׁמִידוֹ ה׳ אֱלֹהֶיךָ מִקִּרְבֶּךָ ד וְאַתֶּם הַדְּבֵקִים בַּה׳ אֱלֹהֵיכֶם חַיִּים כֻּלְּכֶם הַיּוֹם

מֹשֶׁה מַזְכִּיר לָעָם מָה קָרָה מַמָּשׁ לִפְנֵי כֵן, בְּסוֹף הַמַּסָּע בַּמִּדְבָּר: ׳עֵינֵיכֶם הָרֹאֹת אֵת אֲשֶׁר עָשָׂה ה׳ בְּבַעַל פְּעוֹר׳.
בַּעַל פְּעוֹר הָיָה עֲבוֹדָה זָרָה שֶׁל מוֹאָב, וַאֲנָשִׁים מִיִּשְׂרָאֵל נִמְשְׁכוּ אַחֲרֶיהָ. אוֹמֵר הַפָּסוּק: ׳כִּי כָל הָאִישׁ אֲשֶׁר הָלַךְ אַחֲרֵי בַעַל פְּעוֹר הִשְׁמִידוֹ ה׳ אֱלֹהֶיךָ מִקִּרְבֶּךָ׳.

וּמוּלוֹ הַפָּסוּק הַנִּפְלָא: ׳וְאַתֶּם הַדְּבֵקִים בַּה׳ אֱלֹהֵיכֶם – חַיִּים כֻּלְּכֶם הַיּוֹם׳.
מֹשֶׁה אוֹמֵר: רְאִיתֶם בְּעֵינֵיכֶם מָה קוֹרֶה לְמִי שֶׁעוֹזֵב אֶת ה׳, וּמָה קוֹרֶה לְמִי שֶׁנִּדְבָּק בּוֹ. אֵין צֹרֶךְ בְּהוֹכָחוֹת – זֶה קָרָה מוּל הָעֵינַיִם.

וְזֶה מִתְחַבֵּר יָפֶה לְכָל הַשִּׁעוּר: הַדְּבֵקוּת בַּה׳ הִיא לִשְׁמֹר אֶת תּוֹרָתוֹ בִּשְׁלֵמוּת – לֹא לְהוֹסִיף וְלֹא לִגְרֹעַ. מִי שֶׁדָּבֵק בַּה׳ – חַי.

🤔 מָה זֶה ׳דְּבֵקִים׳?
דָּבֵק הוּא מַשֶּׁהוּ שֶׁמְּחַבֵּר שְׁנֵי דְּבָרִים כָּל כָּךְ חָזָק, עַד שֶׁקָּשֶׁה לְהַפְרִיד בֵּינֵיהֶם.
כָּךְ אֲנַחְנוּ אֲמוּרִים לִהְיוֹת עִם ה׳: מְחֻבָּרִים אֵלָיו תָּמִיד – בַּתְּפִלָּה, בַּלִּמּוּד, בַּמִּצְווֹת וּבַמַּחֲשָׁבָה שֶׁהוּא אִתִּי כָּל הַיּוֹם.

׳וְאַתֶּם הַדְּבֵקִים בַּה׳ אֱלֹהֵיכֶם חַיִּים כֻּלְּכֶם הַיּוֹם׳ – ה׳ רוֹצֶה אוֹתִי קָרוֹב אֵלָיו, דָּבוּק בּוֹ, חַי וְשָׂמֵחַ. כָּל מִצְוָה שֶׁאֲנִי עוֹשֶׂה בְּדִיּוּק הִיא עוֹד נְקֻדַּת חִבּוּר בֵּינִי לְבֵינוֹ 💛

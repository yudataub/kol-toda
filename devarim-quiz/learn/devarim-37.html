פָּרָשַׁת הַמֶּלֶךְ

הַחֹמֶר הַמָּלֵא: מֶלֶךְ שֶׁסֵּפֶר הַתּוֹרָה הוֹלֵךְ אִתּוֹ – וְלִבּוֹ לֹא רָם מֵאֶחָיו (פָּרָשַׁת שֹׁפְטִים, פֶּרֶק י״ז)
🎯 לַחִידוֹן🏠 כָּל הַחִידוֹנִים
בַּשִּׁעוּרִים הָאַחֲרוֹנִים לָמַדְנוּ עַל הַשּׁוֹפְטִים וְעַל הַדַּיָּנִים – אֲנָשִׁים שֶׁיֵּשׁ לָהֶם תַּפְקִיד צִבּוּרִי. עַכְשָׁו הַתּוֹרָה מְדַבֶּרֶת עַל מַנְהִיג נוֹסָף, וְהוּא הָרִאשׁוֹן שֶׁבְּכֻלָּם: הַמֶּלֶךְ.
כְּשֶׁאוֹמְרִים ׳מֶלֶךְ׳ חוֹשְׁבִים מִיָּד עַל אַרְמוֹן, כֶּתֶר, כָּבוֹד, צָבָא וְעֹשֶׁר. נִלְמַד מָה הַתּוֹרָה אוֹמֶרֶת: מָתַי מֻתָּר לְמַנּוֹת מֶלֶךְ, בְּאֵילוּ תְּנָאִים, אֵילוּ שְׁלֹשָׁה דְּבָרִים אָסוּר לוֹ לְהַרְבּוֹת, וְאֵיזוֹ מִצְוָה מְיֻחֶדֶת נִתְּנָה רַק לוֹ.
וּבְעִקָּר נִרְאֶה דָּבָר נִפְלָא: ה׳ אוֹהֵב אֶת עַמּוֹ, וְלָכֵן הוּא דּוֹאֵג שֶׁגַּם הַמַּנְהִיג הֶחָזָק בְּיוֹתֵר יִהְיֶה צָמוּד לַתּוֹרָה וְיִזְכֹּר שֶׁהוּא אָח שֶׁל כָּל יְהוּדִי.
בַּשִּׁעוּר: 👑 מָה מַזְכִּירָה הַמִּלָּה ׳מֶלֶךְ׳? 🏞️ כִּי תָבֹא אֶל הָאָרֶץ – מָתַי תַּעֲלֶה הַבַּקָּשָׁה? ✅ שׂוֹם תָּשִׂים – אֲבָל בִּתְנָאִים 🐎 ׳רַק לֹא יַרְבֶּה לּוֹ סוּסִים׳ 🪙 נָשִׁים, כֶּסֶף וְזָהָב – ׳וְלֹא יָסוּר לְבָבוֹ׳ ⚖️ שָׁלֹשׁ ׳לֹא תַעֲשֶׂה׳ – וּמָה הַמַּטָּרָה שֶׁל כָּל אַחַת 📜 שְׁנֵי סִפְרֵי תּוֹרָה – וְאֶחָד הוֹלֵךְ אִתּוֹ 🙇 ׳לְבִלְתִּי רוּם לְבָבוֹ מֵאֶחָיו׳ – אִסּוּר הַגַּאֲוָה 👨‍👦 ׳הוּא וּבָנָיו בְּקֶרֶב יִשְׂרָאֵל׳ – וּמְלָכִים בַּתַּנַ״ךְ

👑מָה מַזְכִּירָה הַמִּלָּה ׳מֶלֶךְ׳?

📖 פֶּרֶק י״ז, פָּסוּק י״ג ←
יג וְכָל הָעָם יִשְׁמְעוּ וְיִרָאוּ וְלֹא יְזִידוּן עוֹד

בַּפְּסוּקִים הַקּוֹדְמִים לָמַדְנוּ עַל הַשּׁוֹפְטִים, עַל הַדַּיָּנִים וְעַל בֵּית הַדִּין הַגָּדוֹל. כֻּלָּם אֲנָשִׁים שֶׁיֵּשׁ לָהֶם תַּפְקִיד שֶׁל מַנְהִיגוּת. עַכְשָׁו הַתּוֹרָה עוֹבֶרֶת לְמַנְהִיג נוֹסָף – הַמֶּלֶךְ.

🤔 לַמּוֹרֶה: פְּתִיחָה – לָמָּה רוֹצִים לִהְיוֹת מֶלֶךְ?
אֶפְשָׁר לִפְתֹּחַ בְּקֶטַע מֻכָּר: גּוּר אֲרָיוֹת קָטָן שֶׁחוֹלֵם לִהְיוֹת מֶלֶךְ, וְשָׁר עַל הַיּוֹם שֶׁבּוֹ כֻּלָּם יִשְׁמְעוּ לוֹ.
וְלִשְׁאֹל אֶת הַתַּלְמִידִים: מַדּוּעַ הוּא כָּל כָּךְ מִשְׁתּוֹקֵק לִמְלֹךְ? הַאִם גַּם אַתֶּם הֱיִיתֶם רוֹצִים לִהְיוֹת מְלָכִים? וּמָה הֱיִיתֶם עוֹשִׂים אִלּוּ הֱיִיתֶם מְלָכִים בְּעַם יִשְׂרָאֵל?
אֶת הַתְּשׁוּבוֹת כּוֹתְבִים עַל הַלּוּחַ, וּבְסוֹף הַשִּׁעוּר מַשְׁוִים אוֹתָן לְמָה שֶׁהַתּוֹרָה בֶּאֱמֶת מְצַוָּה.

כְּשֶׁשּׁוֹאֲלִים יְלָדִים אֵיזוֹ מִלָּה קוֹפֶצֶת לָרֹאשׁ כְּשֶׁאוֹמְרִים ׳מֶלֶךְ׳, הַתְּשׁוּבוֹת דּוֹמוֹת: אַרְמוֹן יָפֶה, כֶּתֶר, כָּבוֹד, שָׂרִים, צָבָא וְעֹשֶׁר.
אֲבָל הַתּוֹרָה מְצַיֶּרֶת תְּמוּנָה אַחֶרֶת לְגַמְרֵי. מֶלֶךְ בְּיִשְׂרָאֵל הוּא אָדָם שֶׁצָּרִיךְ לִהְיוֹת צָמוּד לַתּוֹרָה יוֹתֵר מִכָּל אָדָם אַחֵר, שֶׁלֹּא יִתְגָּאֶה עַל הָעָם, וְשֶׁכָּל מַלְכוּתוֹ תִּהְיֶה לְמַעַן ה׳ וּלְמַעַן הָעָם.

וְכָךְ מִסְתַּבֵּר שֶׁהַכֶּתֶר הָאֲמִתִּי שֶׁל מֶלֶךְ יִשְׂרָאֵל אֵינוֹ עָשׂוּי זָהָב, אֶלָּא מִלִּים שֶׁל תּוֹרָה. ה׳ אוֹהֵב אֶת עַמּוֹ, וְלָכֵן הוּא רוֹצֶה שֶׁמִּי שֶׁמַּנְהִיג אוֹתוֹ יִהְיֶה קָרוֹב אֵלָיו.

💛 אַהֲבַת הַשֵּׁם
ה׳ הוּא מֶלֶךְ מַלְכֵי הַמְּלָכִים, וְגַם הַמֶּלֶךְ הַגָּדוֹל בְּיוֹתֵר הוּא רַק עֶבֶד שֶׁלּוֹ. כְּשֶׁאֲנִי מִתְפַּלֵּל וְאוֹמֵר ׳מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט׳, אֲנִי זוֹכֵר שֶׁיֵּשׁ לִי מֶלֶךְ שֶׁאוֹהֵב אוֹתִי בֶּאֱמֶת 💛

🏞️כִּי תָבֹא אֶל הָאָרֶץ – מָתַי תַּעֲלֶה הַבַּקָּשָׁה?

📖 פֶּרֶק י״ז, פָּסוּק י״ד ←
יד כִּי תָבֹא אֶל הָאָרֶץ אֲשֶׁר ה׳ אֱלֹהֶיךָ נֹתֵן לָךְ וִירִשְׁתָּהּ וְיָשַׁבְתָּה בָּהּ וְאָמַרְתָּ אָשִׂימָה עָלַי מֶלֶךְ כְּכָל הַגּוֹיִם אֲשֶׁר סְבִיבֹתָי

הַרְבֵּה פְּעָמִים שָׁאַלְנוּ: כְּשֶׁהַתּוֹרָה מְדַבֶּרֶת עַל מַשֶּׁהוּ שֶׁיִּקְרֶה בְּאֶרֶץ יִשְׂרָאֵל – בְּאֵיזֶה שָׁלָב מְדֻבָּר? בִּזְמַן הַמִּלְחָמוֹת, בִּזְמַן הַהִתְיַשְּׁבוּת, אוֹ אַחֲרֶיהָ?
כָּאן הַפָּסוּק עוֹנֶה בְּעַצְמוֹ: ׳כִּי תָבֹא אֶל הָאָרֶץ... וִירִשְׁתָּהּ וְיָשַׁבְתָּה בָּהּ, וְאָמַרְתָּ אָשִׂימָה עָלַי מֶלֶךְ׳. קֹדֶם כּוֹבְשִׁים אֶת הָאָרֶץ, אַחַר כָּךְ כָּל שֵׁבֶט מִתְיַשֵּׁב בַּחֵלֶק שֶׁלּוֹ – וְרַק אָז תַּעֲלֶה הַבַּקָּשָׁה לְמֶלֶךְ.

׳אָשִׂימָה עָלַי מֶלֶךְ כְּכָל הַגּוֹיִם אֲשֶׁר סְבִיבֹתָי׳. בְּנֵי יִשְׂרָאֵל יִסְתַּכְּלוּ מִסָּבִיב וְיִרְאוּ: לֶאֱדוֹם יֵשׁ מֶלֶךְ, לְמוֹאָב יֵשׁ מֶלֶךְ, לְעַמּוֹן, לַאֲרַם צוֹבָא וְלַפְּלִשְׁתִּים יֵשׁ מְלָכִים – גַּם אֲנַחְנוּ רוֹצִים.

וְלָמָּה בֶּאֱמֶת רוֹצִים מֶלֶךְ? כִּי מֶלֶךְ מְאַרְגֵּן צָבָא, וְהַצָּבָא נוֹתֵן בִּטָּחוֹן. הוּא גַּם מְסַדֵּר אֶת הַמְּדִינָה: בּוֹנֶה עָרִים, מֵקִים חוֹמוֹת, מְנַהֵל מִסְחָר מְסֻדָּר עִם עַמִּים אֲחֵרִים וְשׁוֹמֵר עַל הַתּוֹשָׁבִים. עַם שֶׁיֵּשׁ לוֹ שִׁלְטוֹן – הוּא עַם חָזָק וּמְבֻסָּס יוֹתֵר.

💡 רְשׁוּת אוֹ חוֹבָה?
הַתּוֹרָה לֹא כּוֹתֶבֶת ׳תָּשִׂים מֶלֶךְ׳ אֶלָּא ׳וְאָמַרְתָּ אָשִׂימָה עָלַי מֶלֶךְ׳. בַּנִּסּוּחַ הַזֶּה דָּנוּ חֲזַ״ל בְּמַסֶּכֶת סַנְהֶדְרִין, וְנֶחְלְקוּ: הַאִם מִנּוּי מֶלֶךְ הוּא רְשׁוּת – רַק אִם יִשְׂרָאֵל יְבַקְּשׁוּ – אוֹ חוֹבָה?
הָרַמְבַּ״ן כּוֹתֵב שֶׁזּוֹ מִצְוַת עֲשֵׂה: ׳כְּמוֹ וֶאֱמוֹר אָשִׂימָה עָלַי מֶלֶךְ׳ – מִצְוָה שֶׁיָּבוֹאוּ לִפְנֵי הַכֹּהֲנִים הַלְוִיִּם וְאֶל הַשּׁוֹפֵט וְיֹאמְרוּ לָהֶם: רְצוֹנֵנוּ שֶׁנָּשִׂים עָלֵינוּ מֶלֶךְ.
📜 רַשִׁ״י
׳וְהָיָה כְשִׁבְתּוֹ׳ – אִם עָשָׂה כֵן, כְּדַאי הוּא שֶׁתִּתְקַיֵּם מַלְכוּתוֹ.

הַמַּלְכוּת אֵינָהּ מַתָּנָה אוֹטוֹמָטִית. מֶלֶךְ שֶׁשּׁוֹמֵר אֶת מָה שֶׁהַתּוֹרָה מְצַוָּה אוֹתוֹ – רָאוּי שֶׁמַּלְכוּתוֹ תַּחֲזִיק מַעֲמָד.

💛 אַהֲבַת הַשֵּׁם
ה׳ נָתַן לָנוּ אֶרֶץ, וְגַם דָּאַג שֶׁיִּהְיֶה בָּהּ סֵדֶר וְשִׁלְטוֹן שֶׁיִּשְׁמֹר עָלֵינוּ. הוּא לֹא עָזַב אוֹתָנוּ לְבַד אַחֲרֵי הַכִּבּוּשׁ – הוּא מְלַוֶּה אוֹתָנוּ גַּם בַּחַיִּים הַיּוֹם־יוֹמִיִּים בָּאָרֶץ 💛

✅שׂוֹם תָּשִׂים – אֲבָל בִּתְנָאִים

📖 פֶּרֶק י״ז, פָּסוּק ט״ו ←
טו שׂוֹם תָּשִׂים עָלֶיךָ מֶלֶךְ אֲשֶׁר יִבְחַר ה׳ אֱלֹהֶיךָ בּוֹ מִקֶּרֶב אַחֶיךָ תָּשִׂים עָלֶיךָ מֶלֶךְ לֹא תוּכַל לָתֵת עָלֶיךָ אִישׁ נָכְרִי אֲשֶׁר לֹא אָחִיךָ הוּא

הַתּוֹרָה לֹא פּוֹסֶלֶת אֶת הַבַּקָּשָׁה. הִיא אוֹמֶרֶת: ׳שׂוֹם תָּשִׂים עָלֶיךָ מֶלֶךְ׳ – אֶפְשָׁר לְמַנּוֹת מֶלֶךְ. אֲבָל כָּל הַפְּסוּקִים שֶׁאַחֲרֵי כֵן מְלַמְּדִים אֶת הַתְּנָאִים הַמְּיֻחָדִים שֶׁבָּהֶם מֶלֶךְ יִשְׂרָאֵל הוּא מֶלֶךְ כָּשֵׁר.

שְׁנֵי תְּנָאִים כְּבָר בַּפָּסוּק הַזֶּה:

׳הוּא וּבָנָיו׳ – מַגִּיד שֶׁאִם בְּנוֹ הָגוּן לַמַּלְכוּת, הוּא קוֹדֵם לְכָל אָדָם.

׳מִכְּלַל הֵן אַתָּה שׁוֹמֵעַ לָאו׳ – מִתּוֹךְ הַהַבְטָחָה הַחִיּוּבִית לוֹמְדִים גַּם אֶת הַהֵפֶךְ: מֶלֶךְ שֶׁלֹּא יִשְׁמֹר, מַלְכוּתוֹ לֹא תַּאֲרִיךְ יָמִים. וְשִׂימוּ לֵב: הַמַּלְכוּת עוֹבֶרֶת לַבֵּן רַק אִם הוּא הָגוּן לַמַּלְכוּת – לֹא מַסְפִּיק לִהְיוֹת בֶּן שֶׁל מֶלֶךְ.

📚 שְׁלוֹשָׁה סוּגֵי מְלָכִים
בַּתַּנַ״ךְ אָנוּ פּוֹגְשִׁים מְלָכִים שֶׁהָלְכוּ נֶגֶד הָעָם וְהֵצִיקוּ לוֹ – הַדֻּגְמָה הַמְּפֻרְסֶמֶת הִיא אֲבִימֶלֶךְ.
יֵשׁ מְלָכִים שֶׁהִסִּיטוּ אֶת הָעָם לַעֲבוֹדָה זָרָה וְנִלְחֲמוּ בִּנְבִיאֵי יִשְׂרָאֵל – כְּמוֹ אַחְאָב.
וְיֵשׁ מְלָכִים צַדִּיקִים, שֶׁלִּבָּם הָיָה שָׁלֵם עִם ה׳ וְכָל מַלְכוּתָם הָיְתָה לְמַעַן הָעָם – כְּמוֹ דָּוִד הַמֶּלֶךְ. דָּוִד, שֶׁהָיָה מֶלֶךְ אַדִּיר, כָּתַב בִּתְהִלִּים: ׳ה׳, לֹא גָבַהּ לִבִּי וְלֹא רָמוּ עֵינַי׳.
💡 סִכּוּם רַשִׁ״י וְהַמְּפָרְשִׁים
סוּסִים – ׳שֶׁלֹּא יָשִׁיב אֶת הָעָם מִצְרַיְמָה׳.
נָשִׁים – ׳וְלֹא יָסוּר לְבָבוֹ׳.
כֶּסֶף וְזָהָב – ׳כְּדֵי לָתֵת לְאַכְסַנְיָא׳.
׳מִשְׁנֵה הַתּוֹרָה׳ – שְׁתֵּי סִפְרֵי תּוֹרָה.
׳לְבִלְתִּי רוּם לְבָבוֹ׳ – אִסּוּר הַגַּאֲוָה (הָרַמְבַּ״ן).
׳וּלְבִלְתִּי סוּר מִן הַמִּצְוָה׳ – אֲפִלּוּ מִצְוָה קַלָּה שֶׁל נָבִיא, כְּמוֹ שָׁאוּל.
🤔 לַמּוֹרֶה: הַתַּקָּנוֹן שֶׁל הַכִּתָּה
אֶפְשָׁר לְבַקֵּשׁ מֵהַתַּלְמִידִים לְנַסֵּחַ בְּ׳מוֹעֶצֶת הַכִּתָּה׳ כְּלָלִים לְמֶלֶךְ, וְאַחַר כָּךְ לְהַשְׁווֹת: מָה דּוֹמֶה וּמָה שׁוֹנֶה בֵּינָם לְבֵין מָה שֶׁהַתּוֹרָה מְצַוָּה אֶת מֶלֶךְ יִשְׂרָאֵל?
עַל פִּי רֹב הַכְּלָלִים שֶׁלָּנוּ עוֹסְקִים בְּמָה שֶׁהַמֶּלֶךְ יַעֲשֶׂה לָעָם, וְהַתּוֹרָה עוֹסֶקֶת גַּם בְּמָה שֶׁקּוֹרֶה בְּתוֹךְ לִבּוֹ שֶׁל הַמֶּלֶךְ.

זֶה הַחִדּוּשׁ הַגָּדוֹל שֶׁל פָּרָשַׁת הַמֶּלֶךְ: מֶלֶךְ יִשְׂרָאֵל, בַּאֲשֶׁר הוּא מֶלֶךְ, כָּפוּף לְתוֹרַת ה׳, וְאָסוּר לוֹ לְהִשְׁתַּמֵּשׁ בְּכֹחוֹ כְּדֵי לְהִשְׂתָּרֵר עַל אֶחָיו.

💛 אַהֲבַת הַשֵּׁם
כָּל הַמִּצְווֹת הָאֵלֶּה הֵן מַתָּנָה שֶׁל אַהֲבָה: ה׳ רוֹצֶה שֶׁיִּהְיֶה לָנוּ מַנְהִיג שֶׁאוֹהֵב אוֹתָנוּ כְּמוֹ אָח, שֶׁהַתּוֹרָה בְּיָדוֹ וְה׳ בְּלִבּוֹ. וַאֲנַחְנוּ מִתְפַּלְּלִים בְּכָל יוֹם שֶׁה׳ יָשִׁיב לָנוּ מַלְכוּת כָּזוֹ בְּאֶרֶץ יִשְׂרָאֵל 💛

✨ לְסִכּוּם

  • ׳וְאָמַרְתָּ אָשִׂימָה עָלַי מֶלֶךְ׳ – הַבַּקָּשָׁה תַּעֲלֶה אַחֲרֵי הַכִּבּוּשׁ וְהַהִתְיַשְּׁבוּת בָּאָרֶץ. חֲכָמִים נֶחְלְקוּ אִם מִנּוּי מֶלֶךְ הוּא רְשׁוּת אוֹ חוֹבָה, וְהָרַמְבַּ״ן כּוֹתֵב שֶׁזּוֹ מִצְוַת עֲשֵׂה.
  • הַתְּנָאִים: הַנָּבִיא מְאַשֵּׁר (׳אֲשֶׁר יִבְחַר ה׳ בּוֹ׳), וְהַמֶּלֶךְ חַיָּב לִהְיוֹת מִקֶּרֶב אַחֶיךָ – מֵעַם יִשְׂרָאֵל.
  • שְׁלוֹשָׁה ׳לֹא יַרְבֶּה לּוֹ׳: סוּסִים (שֶׁלֹּא יָשִׁיב אֶת הָעָם מִצְרַיְמָה, וְשֶׁיִּבְטַח בַּה׳), נָשִׁים (׳וְלֹא יָסוּר לְבָבוֹ׳), כֶּסֶף וְזָהָב (רַק כְּדֵי לָתֵת לְאַכְסַנְיָא, וְשֶׁלֹּא יַכְבִּיד עַל הָעָם).
  • מִצְוַת עֲשֵׂה אַחַת: ׳וְכָתַב לוֹ אֶת מִשְׁנֵה הַתּוֹרָה׳ – שְׁנֵי סִפְרֵי תּוֹרָה, אֶחָד בְּבֵית גְּנָזָיו וְאֶחָד שֶׁנִּכְנָס וְיוֹצֵא עִמּוֹ לְכָל מָקוֹם.
  • הַטַּעַם: ׳לְמַעַן יִלְמַד לְיִרְאָה אֶת ה׳׳, ׳לְבִלְתִּי רוּם לְבָבוֹ מֵאֶחָיו׳ וְ׳לְבִלְתִּי סוּר מִן הַמִּצְוָה׳ – אֲפִלּוּ מִצְוָה קַלָּה שֶׁל נָבִיא, כְּמוֹ שֶׁלָּמַדְנוּ מִשָּׁאוּל.
  • הָרַמְבַּ״ן: קַל וָחֹמֶר – אִם הַמֶּלֶךְ מֻזְהָר שֶׁלִּבּוֹ יִהְיֶה שָׁפָל כְּכָל אֶחָיו, קַל וָחֹמֶר כָּל אָדָם. כִּי לַה׳ לְבַדּוֹ הַגְּדֻלָּה וְהָרוֹמְמוּת.
  • הַשָּׂכָר: ׳לְמַעַן יַאֲרִיךְ יָמִים עַל מַמְלַכְתּוֹ הוּא וּבָנָיו׳ – וְכָל זֶה מֵאַהֲבַת ה׳ אֵלֵינוּ: שֶׁיִּהְיֶה לָנוּ מַנְהִיג שֶׁהַתּוֹרָה בְּיָדוֹ וְהָעָם בְּלִבּוֹ.

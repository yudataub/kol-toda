שְׁמִטַּת כְּסָפִים

הַחֹמֶר הַמָּלֵא: יָד שֶׁנִּפְתַּחַת לָאָח, וּבְרָכָה שֶׁאֵין בָּהּ אֶבְיוֹן – פָּרָשַׁת רְאֵה, פֶּרֶק ט״ו
🎯 לַחִידוֹן🏠 כָּל הַחִידוֹנִים
כֻּלָּנוּ מַכִּירִים אֶת שְׁנַת הַשְּׁמִטָּה שֶׁל הַשָּׂדֶה: אַחַת לְשֶׁבַע שָׁנִים הָאֲדָמָה נָחָה וְהַפֵּרוֹת הֶפְקֵר לַכֹּל. אֲבָל בְּפָרָשַׁת רְאֵה מְחַכָּה לָנוּ הַפְתָּעָה: יֵשׁ שְׁמִטָּה גַּם לַכֶּסֶף.
בְּסוֹף שְׁנַת הַשְּׁמִטָּה כָּל הַחוֹבוֹת נִשְׁמָטִים: מִי שֶׁהִלְוָה כֶּסֶף לַחֲבֵרוֹ – פּוֹתֵחַ אֶת הַיָּד וּמְוַתֵּר. נִלְמַד אֶת הַפְּסוּקִים, אֶת הַמִּלִּים הַקָּשׁוֹת, אֶת רַשִׁ״י, אֶת הַבְּרָכָה הַגְּדוֹלָה ׳אֶפֶס כִּי לֹא יִהְיֶה בְּךָ אֶבְיוֹן׳, וְאֶת תַּקָּנַת הַפְּרוֹזְבּוּל שֶׁל הִלֵּל הַזָּקֵן – שֶׁנּוֹהֶגֶת עַד הַיּוֹם.
וּבְעִקָּר נִרְאֶה כַּמָּה ה׳ אוֹהֵב כָּל יְהוּדִי: הוּא לֹא מַסְכִּים שֶׁאָח שֶׁלָּנוּ יִשָּׁאֵר שָׁבוּי בְּחוֹבוֹת, וְהוּא מְלַמֵּד אוֹתָנוּ לִהְיוֹת כְּמוֹתוֹ – נְדִיבִים, רַחֲמָנִים וּבַעֲלֵי עַיִן טוֹבָה 💛
בַּשִּׁעוּר: 🔁 מִשְׁנֵה תּוֹרָה – וּמִצְוָה שֶׁכְּבָר פָּגַשְׁנוּ ⏳ מִקֵּץ שֶׁבַע שָׁנִים – הַמִּצְוָה שֶׁבַּסּוֹף ✋ שָׁמוֹט – פּוֹתְחִים אֶת הַיָּד 🤝 רֵעֵהוּ וְאָחִיו – וְאֶת הַנָּכְרִי תִּגֹּשׂ 🌟 אֶפֶס כִּי לֹא יִהְיֶה בְּךָ אֶבְיוֹן – בְּרָכָה וּתְנַאי ⚖️ לֹא יִהְיֶה – לֹא יֶחְדַּל: שְׁאֵלָה וּתְשׁוּבָה בְּרַשִׁ״י 💭 הִשָּׁמֶר לְךָ – לֵב, עַיִן וְיָד 📜 הִלֵּל הַזָּקֵן וְהַפְּרוֹזְבּוּל 🌾 מִצְוָה קָשָׁה – וְחִנּוּךְ לַלֵּב

🔁מִשְׁנֵה תּוֹרָה – וּמִצְוָה שֶׁכְּבָר פָּגַשְׁנוּ

לְחֻמַּשׁ דְּבָרִים יֵשׁ שֵׁם נוֹסָף: ׳מִשְׁנֵה תּוֹרָה׳. ׳מִשְׁנֶה׳ – מִלְּשׁוֹן ׳שֵׁנִי׳: יֵשׁ מִצְווֹת שֶׁכְּבָר נֶאֶמְרוּ בַּחֻמָּשִׁים הַקּוֹדְמִים, וּמֹשֶׁה רַבֵּנוּ חוֹזֵר עֲלֵיהֶן פַּעַם שְׁנִיָּה – וּמוֹסִיף.
גַּם הַמִּצְוָה שֶׁלְּפָנֵינוּ אֵינָהּ חֲדָשָׁה לְגַמְרֵי: אֶת מִצְוַת הַשְּׁמִטָּה לָמַדְנוּ כְּבָר בְּחֻמַּשׁ וַיִּקְרָא, בְּפָרָשַׁת בְּהַר.

כְּשֶׁאוֹמְרִים ׳שְׁמִטָּה׳, אֵיזוֹ תְּמוּנָה עוֹלָה בָּרֹאשׁ? כִּמְעַט תָּמִיד – שָׂדֶה אוֹ כֶּרֶם.
וְכָךְ הוּא בֶּאֱמֶת: אַחַת לְשֶׁבַע שָׁנִים לֹא חוֹרְשִׁים וְלֹא זוֹרְעִים אֶת הָאֲדָמָה בְּאֶרֶץ יִשְׂרָאֵל, וּמַפְקִירִים אֶת הַפֵּרוֹת – כָּל אֶחָד, וְגַם הֶעָנִי, יָכוֹל לְהִכָּנֵס וְלֶאֱכֹל. הַפֵּרוֹת קְדוֹשִׁים בִּקְדֻשַּׁת שְׁבִיעִית, וְצָרִיךְ לִנְהֹג בָּהֶם בִּזְהִירוּת וּבְכָבוֹד.

אֲבָל כָּאן, בְּפָרָשַׁת רְאֵה, מִתְגַּלֶּה שֶׁיֵּשׁ לַשְּׁמִטָּה צַד נוֹסָף לְגַמְרֵי. לֹא שָׂדֶה וְלֹא פֵּרוֹת, אֶלָּא – כֶּסֶף. מָה הַקֶּשֶׁר בֵּין שְׁמִטָּה לְכֶסֶף? עַל זֶה נִלְמַד בַּשִּׁעוּר הַזֶּה.

💛 אַהֲבַת הַשֵּׁם
אוֹתוֹ ה׳ שֶׁנָּתַן לָנוּ אֶרֶץ וְשָׂדוֹת, גַּם בִּקֵּשׁ שֶׁפַּעַם בְּשֶׁבַע שָׁנִים נַרְפֶּה אֶת הַיָּד וְנִזְכֹּר שֶׁהַכֹּל שֶׁלּוֹ. כְּשֶׁאֲנִי מְוַתֵּר, אֲנִי לֹא מַפְסִיד – אֲנִי מִתְקָרֵב אֶל ה׳ שֶׁנּוֹתֵן לִי אֶת הַכֹּל 💛

⏳מִקֵּץ שֶׁבַע שָׁנִים – הַמִּצְוָה שֶׁבַּסּוֹף

לִשְׁנַת הַשְּׁמִטָּה יֵשׁ הַתְחָלָה וְיֵשׁ סוֹף. הִיא נִכְנֶסֶת בְּרֹאשׁ הַשָּׁנָה, בְּא׳ בְּתִשְׁרֵי, וּמִסְתַּיֶּמֶת בְּכ״ט בֶּאֱלוּל.
רֹב מִצְווֹת הַשְּׁמִטָּה חָלוֹת מִתְּחִלַּת הַשָּׁנָה: מֵאוֹתוֹ רֶגַע אָסוּר לַעֲשׂוֹת מְלָאכוֹת בַּשָּׂדֶה וּבַכֶּרֶם, וּמֵאוֹתוֹ רֶגַע הַפֵּרוֹת הֶפְקֵר לַכֹּל.

אֲבָל הַפָּסוּק שֶׁלָּנוּ פּוֹתֵחַ בְּמִלָּה מְיֻחֶדֶת: ׳מִקֵּץ שֶׁבַע שָׁנִים תַּעֲשֶׂה שְׁמִטָּה׳. ׳מִקֵּץ׳ – בַּסּוֹף.
יֵשׁ אִם כֵּן מִצְוָה אַחַת שֶׁמְּקַיְּמִים דַּוְקָא בְּסוֹף הַשָּׁנָה הַשְּׁבִיעִית, וְהִיא הַתּוֹסֶפֶת הַגְּדוֹלָה שֶׁל חֻמַּשׁ דְּבָרִים: שְׁמִטַּת כְּסָפִים.

📜 רַשִׁ״י
׳מִקֵּץ שֶׁבַע שָׁנִים׳ – יָכוֹל שֶׁבַע שָׁנִים לְכָל מַלְוֶה וּמַלְוֶה? תַּלְמוּד לוֹמַר ׳קָרְבָה שְׁנַת הַשֶּׁבַע׳. וְאִם אַתָּה אוֹמֵר שֶׁבַע שָׁנִים לְכָל מַלְוֶה וּמַלְוֶה, לְהַלְוָאַת כָּל אֶחָד וְאֶחָד – הֵיאַךְ הִיא קְרֵבָה? הָא לָמַדְתָּ: שֶׁבַע שָׁנִים לְמִנְיַן הַשְּׁמִטּוֹת.

רַשִׁ״י שׁוֹאֵל שְׁאֵלָה נֶהְדֶּרֶת: אוּלַי ׳מִקֵּץ שֶׁבַע שָׁנִים׳ פֵּרוּשׁוֹ שֶׁכָּל הַלְוָאָה נִשְׁמֶטֶת שֶׁבַע שָׁנִים מֵהַיּוֹם שֶׁבּוֹ הִלְווּ אוֹתָהּ?
וְהוּא מוֹכִיחַ מִן הַכָּתוּב ׳קָרְבָה שְׁנַת הַשֶּׁבַע׳ (פָּסוּק ט׳): אִם לְכָל הַלְוָאָה הָיָה מִנְיָן מִשֶּׁלָּהּ, אִי אֶפְשָׁר הָיָה לוֹמַר שֶׁהַשָּׁנָה ׳קְרֵבָה׳ לְכֻלָּם יַחַד. מִכָּאן שֶׁהַמִּנְיָן הוּא לְמִנְיַן הַשְּׁמִטּוֹת – שָׁנָה אַחַת לְכֻלָּם, לְכָל עַם יִשְׂרָאֵל.

יֵשׁ בָּעוֹלָם שָׁנָה אַחַת שֶׁכָּל עַם יִשְׂרָאֵל מְחַכֶּה לָהּ יַחַד. ה׳ עָשָׂה אוֹתָנוּ מִשְׁפָּחָה אַחַת גְּדוֹלָה, שֶׁגַּם הַשָּׁנִים שֶׁלָּהּ מְשֻׁתָּפוֹת 💛

׳וְזֶה דְּבַר הַשְּׁמִטָּה: שָׁמוֹט כָּל בַּעַל מַשֵּׁה יָדוֹ אֲשֶׁר יַשֶּׁה בְּרֵעֵהוּ, לֹא יִגֹּשׂ אֶת רֵעֵהוּ וְאֶת אָחִיו, כִּי קָרָא שְׁמִטָּה לַה׳׳.

מָה זֶה ׳שָׁמוֹט׳? תְּאַר לְךָ שֶׁאַתָּה מַחֲזִיק מַשֶּׁהוּ חָזָק בְּאֶגְרוֹף, וְאָז פּוֹתֵחַ אֶת הַיָּד – וְהַדָּבָר נוֹפֵל. הוּא כְּבָר אֵינוֹ אֶצְלְךָ. זֶה ׳שָׁמוֹט׳: לַעֲזֹב, לְהַנִּיחַ, לְוַתֵּר.
׳בַּעַל מַשֵּׁה׳ – מִי שֶׁהִלְוָה כֶּסֶף לַחֲבֵרוֹ וּמַגִּיעַ לוֹ חוֹב.

׳שָׁמוֹט כָּל בַּעַל מַשֵּׁה יָדוֹ׳ – שְׁמוֹט אֶת יָדוֹ שֶׁל כָּל בַּעַל מַשֵּׁה.

לְפִי רַשִׁ״י, הַתּוֹרָה מְצַוָּה עַל הַיָּד הַזּוֹ שֶׁל הַמַּלְוֶה: הַיָּד שֶׁהָיְתָה מוּשֶׁטֶת לִגְבּוֹת – נִפְתַּחַת וּמְרַפָּה.
וְכָךְ נֶאֱמָר גַּם ׳לֹא יִגֹּשׂ אֶת רֵעֵהוּ׳: כְּשֶׁמַּגִּיעַ סוֹף שְׁנַת הַשְּׁמִטָּה, הַמַּלְוֶה אֵינוֹ יָכוֹל לָבוֹא אֶל הַלּוֶֹה וְלוֹמַר לוֹ ׳תַּחֲזִיר לִי׳. הַחוֹב נִשְׁמָט – הוּא מִתְבַּטֵּל, וּמִי שֶׁהָיָה חַיָּב כֶּסֶף כְּבָר אֵינוֹ חַיָּב לְהַחֲזִיר.

📚 רְאוּבֵן וְשִׁמְעוֹן – וְהַחוֹזֶה שֶׁבֵּינֵיהֶם
רְאוּבֵן יָרַד מִנְּכָסָיו, וּפָנָה לְשִׁמְעוֹן בְּבַקָּשַׁת הַלְוָאָה. מָה תִּהְיֶה תְּשׁוּבָתוֹ שֶׁל שִׁמְעוֹן? הַאִם יַסְכִּים לְהַלְווֹת בְּכָל מִקְרֶה?
נְנַסֶּה לִכְתֹּב בֵּינֵיהֶם הֶסְכֵּם. מָה חַיָּב לְהוֹפִיעַ בּוֹ? שֵׁם הַלּוֶֹה וְשֵׁם הַמַּלְוֶה, סְכוּם הַהַלְוָאָה וּתְנָאֵי הַהַחְזָרָה – מָתַי וְאֵיךְ מַחֲזִירִים.
וְעַכְשָׁיו, מִתּוֹךְ הַפְּסוּקִים, נְגַלֶּה מָתַי שִׁמְעוֹן יִצְטָרֵךְ לְוַתֵּר עַל כַּסְפּוֹ לְטוֹבַת רְאוּבֵן: בְּסוֹף שְׁנַת הַשְּׁמִטָּה – אָז מִתְבַּטְּלִים הַחוֹבוֹת.

הַתּוֹרָה מַכִּירָה אֶת לִבּוֹ שֶׁל הָאָדָם. הִיא יוֹדַעַת בְּדִיּוּק מָה עָלוּל לִקְרוֹת כְּשֶׁשְּׁנַת הַשְּׁמִטָּה מִתְקָרֶבֶת, וְלָכֵן הִיא מַזְהִירָה:
׳הִשָּׁמֶר לְךָ פֶּן יִהְיֶה דָבָר עִם לְבָבְךָ בְלִיַּעַל לֵאמֹר: קָרְבָה שְׁנַת הַשֶּׁבַע, שְׁנַת הַשְּׁמִטָּה, וְרָעָה עֵינְךָ בְּאָחִיךָ הָאֶבְיוֹן וְלֹא תִתֶּן לוֹ, וְקָרָא עָלֶיךָ אֶל ה׳ וְהָיָה בְךָ חֵטְא׳.

מָה הַמִּקְרֶה? הֶעָנִי מְבַקֵּשׁ הַלְוָאָה, וְהֶעָשִׁיר חוֹשֵׁב בְּלִבּוֹ: ׳אִם אַלְוֶה עַכְשָׁו – הַשְּׁמִטָּה תְּבַטֵּל אֶת הַחוֹב וְהַכֶּסֶף שֶׁלִּי יֵלֵךְ׳. לָכֵן הוּא מְסָרֵב.
הַתּוֹרָה קוֹרֵאת לַמַּחֲשָׁבָה הַזּוֹ ׳בְּלִיַּעַל׳. ׳בְּלִיַּעַל׳ – בְּלִי עֹל: אָדָם שֶׁאֵינוֹ מוּכָן לְקַבֵּל עָלָיו עֹל וְאַחְרָיוּת. אֵיזֶה עֹל? עֹל הַצִּבּוּר – הָאַחְרָיוּת לִדְאֹג לַזּוּלַת, לָצֵאת מֵהַמַּבָּט עַל עַצְמִי וּלְהַבִּיט עַל הָאַחֵר.

וְשִׂימוּ לֵב לָאֵיבָר שֶׁמּוֹפִיעַ בַּפָּסוּק: עַיִן. ׳וְרָעָה עֵינְךָ בְּאָחִיךָ הָאֶבְיוֹן׳ – זוֹ עַיִן רָעָה, עַיִן שֶׁמִּצְטַעֶרֶת עַל מָה שֶׁנּוֹתְנִים לָאַחֵר. הַתּוֹרָה רוֹצָה מֵאִתָּנוּ עַיִן טוֹבָה.

📜 רַשִׁ״י
׳וְקָרָא עָלֶיךָ׳ – יָכוֹל מִצְוָה? תַּלְמוּד לוֹמַר ׳וְלֹא יִקְרָא׳.
׳וְהָיָה בְךָ חֵטְא׳ – מִכָּל מָקוֹם, אֲפִלּוּ לֹא יִקְרָא. אִם כֵּן, לָמָּה נֶאֱמַר ׳וְקָרָא עָלֶיךָ׳? מְמַהֵר אֲנִי לִפָּרַע עַל יְדֵי הַקּוֹרֵא יוֹתֵר מִמִּי שֶׁאֵינוֹ קוֹרֵא.

רַשִׁ״י מְלַמֵּד שְׁנֵי דְּבָרִים: הֶעָנִי אֵינוֹ חַיָּב לִצְעֹק אֶל ה׳ – זוֹ אֵינָהּ מִצְוָה; וְגַם אִם לֹא יִצְעַק, מִי שֶׁסֵּרֵב לוֹ נוֹשֵׂא חֵטְא.
אֲבָל כְּשֶׁאָדָם כּוֹאֵב וְצוֹעֵק אֶל ה׳ – ה׳ שׁוֹמֵעַ מִיָּד. אֵין קוֹל שֶׁקָּרוֹב אֶל ה׳ כְּמוֹ קוֹלוֹ שֶׁל אָדָם שָׁבוּר.

💡 הַמְשֵׁךְ הַפָּרָשָׁה – לַשִּׁעוּר הַבָּא
מִיָּד אַחַר כָּךְ מְצַוָּה הַתּוֹרָה: ׳לֹא תְאַמֵּץ אֶת לְבָבְךָ וְלֹא תִקְפֹּץ אֶת יָדְךָ׳, ׳כִּי פָתֹחַ תִּפְתַּח אֶת יָדְךָ לוֹ וְהַעֲבֵט תַּעֲבִיטֶנּוּ׳, ׳נָתוֹן תִּתֵּן לוֹ וְלֹא יֵרַע לְבָבְךָ׳.
שְׁתֵּי תְּנוּעוֹת שֶׁל יָד: יָד קְפוּצָה כְּאֶגְרוֹף מוּל יָד פְּתוּחָה. וּשְׁנֵי מַצָּבִים שֶׁל לֵב: לֵב קָשֶׁה מוּל לֵב רַךְ.
וְהַבְטָחָה: ׳כִּי בִּגְלַל הַדָּבָר הַזֶּה יְבָרֶכְךָ ה׳ אֱלֹהֶיךָ בְּכָל מַעֲשֶׂךָ וּבְכֹל מִשְׁלַח יָדֶךָ׳. אֶת כָּל זֶה נִלְמַד בְּהַרְחָבָה בַּשִּׁעוּר הַבָּא.

׳פָּתֹחַ תִּפְתַּח׳ – אֲפִלּוּ כַּמָּה פְּעָמִים.
׳כִּי פָתֹחַ תִּפְתַּח׳ – הֲרֵי ׳כִּי׳ מְשַׁמֵּשׁ בִּלְשׁוֹן ׳אֶלָּא׳.
׳וְהַעֲבֵט תַּעֲבִיטֶנּוּ׳ – אִם לֹא רָצָה בְּמַתָּנָה, תֵּן לוֹ בְּהַלְוָאָה.
💛 אַהֲבַת הַשֵּׁם
ה׳ לֹא מְבַקֵּשׁ מִמֶּנִּי רַק יָד פְּתוּחָה – הוּא מְבַקֵּשׁ גַּם לֵב שֶׁלֹּא יֵרַע בִּשְׁעַת הַנְּתִינָה. הוּא רוֹצֶה שֶׁאֶהְיֶה דּוֹמֶה לוֹ: לָתֵת בְּשִׂמְחָה, בְּעַיִן טוֹבָה, כְּמוֹ שֶׁהוּא נוֹתֵן לִי בְּכָל בֹּקֶר מֵחָדָשׁ 💛

הַדָּבָר שֶׁהַתּוֹרָה הִזְהִירָה מִפָּנָיו – קָרָה בֶּאֱמֶת. שָׁנִים רַבּוֹת אַחֲרֵי מֹשֶׁה רַבֵּנוּ, בִּימֵי בַּיִת שֵׁנִי, רָאָה הִלֵּל הַזָּקֵן שֶׁהָעָם נִמְנְעוּ מִלְּהַלְווֹת זֶה אֶת זֶה לִקְרַאת הַשְּׁמִטָּה, כִּי חָשְׁשׁוּ שֶׁכַּסְפָּם יִשָּׁמֵט – וְכָךְ הֵם עָבְרוּ עַל מָה שֶׁכָּתוּב מְפֹרָשׁ בַּתּוֹרָה: ׳הִשָּׁמֶר לְךָ׳.

מֵרֶגַע זֶה הַחוֹבוֹת אֵינָם ׳חוֹבוֹת פְּרָטִיִּים׳ אֶלָּא שֶׁל בֵּית הַדִּין, וְלָכֵן אֵינָם נִשְׁמָטִים; הַמַּלְוֶה גּוֹבֶה אוֹתָם כִּשְׁלִיחַ בֵּית הַדִּין גַּם אַחֲרֵי הַשְּׁמִטָּה.

הִלֵּל הַזָּקֵן תִּקֵּן פְּרוֹזְבּוּל, וְהוּא נוֹהֵג גַּם הַיּוֹם.

חַכְמֵי יִשְׂרָאֵל אוֹהֲבִים אֶת עַם יִשְׂרָאֵל בְּדִיּוּק כְּמוֹ שֶׁה׳ אוֹהֵב אוֹתָנוּ. הִלֵּל הַזָּקֵן חִפֵּשׂ דֶּרֶךְ שֶׁגַּם הָעָנִי יוּכַל לִלְווֹת וְגַם הַמַּלְוֶה לֹא יַפְסִיד – כִּי זֶה רְצוֹן ה׳: שֶׁלְּכֻלָּם יִהְיֶה טוֹב 💛

🌾מִצְוָה קָשָׁה – וְחִנּוּךְ לַלֵּב

שְׁמִטַּת כְּסָפִים דּוֹמָה מְאוֹד לִשְׁמִטַּת הַקַּרְקָעוֹת: גַּם הִיא נִסָּיוֹן גָּדוֹל וּמִצְוָה מְיֻחֶדֶת, וְגַם הִיא דּוֹרֶשֶׁת הַרְבֵּה אֱמוּנָה. לְוַתֵּר עַל חוֹב שֶׁהָיָה מַגִּיעַ לִי – זֶה קָשֶׁה מְאוֹד, וְזֶה דּוֹרֵשׁ הַרְבֵּה מִן הַלֵּב.

וְחָשׁוּב לְהַדְגִּישׁ: הַתּוֹרָה לֹא נָתְנָה אֶת הַמִּצְוָה הַזּוֹ כְּדֵי שֶׁמִּי שֶׁיֵּשׁ לוֹ כֶּסֶף יְנַצֵּל אוֹתָהּ וְלֹא יַחֲזִיר חוֹבוֹת, אֶלָּא כְּדֵי לַעֲזֹר לָעֲנִיִּים שֶׁנִּמְצָאִים בִּמְצוּקָה וְשָׁקְעוּ בַּחוֹבוֹת.
בְּמִי ׳מִתְחַשֶּׁבֶת׳ הַתּוֹרָה בַּמִּצְוָה הַזּוֹ? בְּעִקָּר בָּעֲנִיִּים – הִיא רוֹצָה לְצַמְצֵם אֶת הַפְּעָרִים בֵּין הָעֲשִׁירִים לָעֲנִיִּים, כְּמוֹ שֶׁבִּשְׁנַת הַשְּׁמִטָּה שֶׁל הַשָּׂדֶה כָּל אֶחָד נִכְנָס וְאוֹכֵל מִן הַפֵּרוֹת.

💡 סֵפֶר הַחִנּוּךְ: לְמָה הַמִּצְוָה מְחַנֶּכֶת?
סֵפֶר הַחִנּוּךְ כּוֹתֵב שֶׁהַמִּצְוָה בָּאָה ׳לְלַמֵּד נַפְשֵׁנוּ בַּמִּדּוֹת הַמְעֻלּוֹת: מִדַּת הַנְּדִיבוּת וְעַיִן טוֹבָה, וְלִקְבֹּעַ בִּלְבָבֵנוּ הַבִּטָּחוֹן הַגָּדוֹל בַּשֵּׁם בָּרוּךְ הוּא׳.
כְּלוֹמַר: מִי שֶׁשּׁוֹמֵט אֶת הַחוֹב מִתְאַמֵּן לִהְיוֹת נָדִיב, וְתוֹךְ כָּךְ הוּא לוֹמֵד שֶׁהַפַּרְנָסָה בָּאָה מֵה׳ – וְלֹא מִן הַחוֹב הַזֶּה דַּוְקָא.

אִם נַעֲשֶׂה רְשִׁימָה שֶׁל הַמִּדּוֹת שֶׁהַמִּצְוָה הַזּוֹ מְלַמֶּדֶת, נְגַלֶּה אוֹצָר שָׁלֵם:
רַחֲמָנוּת – לְהַרְגִּישׁ אֶת הַצַּעַר שֶׁל הָאַחֵר.
נְדִיבוּת – הַהֵפֶךְ מִקַּמְצָנוּת.
וַתְּרָנוּת וְעַיִן טוֹבָה – לִשְׂמֹחַ בְּטוֹבַת הַזּוּלַת.
אֱמוּנָה – הַכָּרָה שֶׁכָּל מָה שֶׁיֵּשׁ לִי הוּא מֵאֵת ה׳, שֶׁ׳הוּא הַנֹּתֵן לְךָ כֹּחַ לַעֲשׂוֹת חָיִל׳.

וּמִי שֶׁמְּקַיֵּם אֶת מִצְווֹת ה׳ – ה׳ מַמְשִׁיךְ לָתֵת לוֹ בְּרָכָה. זֶה בְּדִיּוּק מָה שֶׁהִבְטִיחָה הַתּוֹרָה: ׳כִּי בָרֵךְ יְבָרֶכְךָ ה׳ בָּאָרֶץ׳. הַכֶּסֶף שֶׁנִּשְׁמַט אֵינוֹ הֶפְסֵד – הוּא פִּקָּדוֹן בְּיַד ה׳.

✨ לְסִכּוּם

  • חֻמַּשׁ דְּבָרִים הוּא ׳מִשְׁנֵה תּוֹרָה׳: אֶת שְׁמִטַּת הַקַּרְקָעוֹת לָמַדְנוּ בְּפָרָשַׁת בְּהַר, וְכָאן מִתְגַּלֶּה הַצַּד הַנּוֹסָף – שְׁמִטַּת כְּסָפִים.
  • ׳מִקֵּץ שֶׁבַע שָׁנִים׳ – בְּסוֹף שְׁנַת הַשְּׁמִטָּה, בְּכ״ט בֶּאֱלוּל, נִשְׁמָטִים הַחוֹבוֹת. רַשִׁ״י מוֹכִיחַ מִ׳קָרְבָה שְׁנַת הַשֶּׁבַע׳ שֶׁהַמִּנְיָן הוּא לְמִנְיַן הַשְּׁמִטּוֹת – שָׁנָה אַחַת לְכֻלָּם.
  • ׳שָׁמוֹט׳ – לִפְתֹּחַ אֶת הַיָּד וּלְוַתֵּר; ׳בַּעַל מַשֵּׁה׳ – הַמַּלְוֶה; ׳לֹא יִגֹּשׂ׳ – לֹא יִגְבֶּה. אֶת הַנָּכְרִי מֻתָּר לִתְבֹּעַ, וְזוֹ מִצְוַת עֲשֵׂה.
  • הַתּוֹרָה קוֹרֵאת לַלּוֶֹה ׳רֵעֵהוּ׳ וְ׳אָחִיו׳ – פַּעֲמַיִם כָּל אֶחָד – כְּמוֹ ׳וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ׳, כְּדֵי שֶׁנַּרְגִּישׁ אַחְרָיוּת וְנִשְׁמֹט בְּאַהֲבָה.
  • ׳אֶפֶס כִּי לֹא יִהְיֶה בְּךָ אֶבְיוֹן׳ – בִּרְכַּת ה׳, בִּתְנַאי ׳רַק אִם שָׁמוֹעַ תִּשְׁמַע׳. וְהַסְּתִירָה לְ׳לֹא יֶחְדַּל אֶבְיוֹן׳ נִפְתֶּרֶת בְּרַשִׁ״י: עוֹשִׂים רְצוֹנוֹ שֶׁל מָקוֹם – אֶבְיוֹנִים בַּאֲחֵרִים וְלֹא בָּכֶם.
  • ׳הִשָּׁמֶר לְךָ׳ – אָסוּר לְסָרֵב לְהַלְווֹת לִפְנֵי הַשְּׁמִטָּה; ׳בְּלִיַּעַל׳ – בְּלִי עֹל. הִלֵּל הַזָּקֵן תִּקֵּן פְּרוֹזְבּוּל לְטוֹבַת הַמַּלְוִים וְהָעֲנִיִּים כְּאֶחָד, וְהוּא נוֹהֵג עַד הַיּוֹם.

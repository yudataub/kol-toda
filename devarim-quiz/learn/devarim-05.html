מַעֲשֵׂה הַמְרַגְּלִים

הַחֹמֶר הַמָּלֵא: עֵינַיִם שֶׁל פַּחַד וְעֵינַיִם שֶׁל אֱמוּנָה – ׳ה׳ אֱלֹהֵיכֶם הַהֹלֵךְ לִפְנֵיכֶם׳ (דְּבָרִים א׳, י״ט–מ״ו)

בַּשִּׁעוּר הַזֶּה מֹשֶׁה רַבֵּנוּ חוֹזֵר וּמְסַפֵּר לְעַם יִשְׂרָאֵל סִפּוּר שֶׁקָּרָה שְׁלוֹשִׁים וּשְׁמוֹנֶה שָׁנִים לִפְנֵי כֵן: מַעֲשֵׂה הַמְרַגְּלִים.
נִלְמַד אֵיךְ הִגִּיעוּ בְּנֵי יִשְׂרָאֵל עַד שַׁעַר הָאָרֶץ, מִי בִּקֵּשׁ לִשְׁלֹחַ מְרַגְּלִים, מָה הֵם אָמְרוּ כְּשֶׁחָזְרוּ, לָמָּה הָעָם בָּכָה וְסֵרֵב לַעֲלוֹת, וּמָה עָנָה לָהֶם מֹשֶׁה: ׳לֹא תַעַרְצוּן וְלֹא תִירְאוּן מֵהֶם׳.
וּבְעִקָּר נִלְמַד דָּבָר אֶחָד גָּדוֹל: יֵשׁ רְאִיָּה שֶׁל פַּחַד וְיֵשׁ רְאִיָּה שֶׁל אֱמוּנָה. מִי שֶׁזּוֹכֵר שֶׁה׳ אוֹהֵב אוֹתוֹ וְהוֹלֵךְ לְפָנָיו – רוֹאֶה אֶת הָעוֹלָם אַחֶרֶת לְגַמְרֵי 💛
בַּשִּׁעוּר: 👁️ פְּתִיחָה: הָעֵינַיִם, הַמֹּחַ וְהַלֵּב – וְסִפּוּר שֶׁחוֹזֵר מִדּוֹר לְדוֹר 🏜️ מֵחֹרֵב עַד קָדֵשׁ בַּרְנֵעַ – ׳עֲלֵה רֵשׁ׳ 🙋 ׳וַתִּקְרְבוּן אֵלַי כֻּלְּכֶם׳ – מִי בִּקֵּשׁ לִשְׁלֹחַ מְרַגְּלִים? 🍇 ׳טוֹבָה הָאָרֶץ׳ – מָה הֵבִיאוּ וּמָה אָמְרוּ 😢 ׳וְלֹא אֲבִיתֶם לַעֲלֹת׳ – עֵינַיִם שֶׁל פַּחַד 🤲 ׳לֹא תַעַרְצוּן׳ – ה׳ הַהֹלֵךְ לִפְנֵיכֶם ⚖️ מִי יִכָּנֵס לָאָרֶץ? שָׂכָר, עֹנֶשׁ – וְתוֹצָאָה טִבְעִית 🐝 הַמַּעְפִּילִים – תִּקּוּן בַּדֶּרֶךְ הַלֹּא נְכוֹנָה 🗣️ הַתִּקּוּן שֶׁלָּנוּ: לְדַבֵּר בְּשִׁבְחָהּ שֶׁל הָאָרֶץ

👁️פְּתִיחָה: הָעֵינַיִם, הַמֹּחַ וְהַלֵּב – וְסִפּוּר שֶׁחוֹזֵר מִדּוֹר לְדוֹר

לֹא תָּמִיד אֲנַחְנוּ רוֹאִים נָכוֹן. יֵשׁ תְּמוּנוֹת שֶׁבָּהֶן שְׁתֵּי דְּמֻיּוֹת הֵן בְּדִיּוּק בְּאוֹתוֹ גֹּדֶל, וְאָנוּ בְּטוּחִים שֶׁאַחַת גְּדוֹלָה וְאַחַת קְטַנָּה. יֵשׁ צוּרוֹת שֶׁנִּרְאוֹת כְּאִלּוּ הֵן זָזוֹת, וְהֵן אֵינָן זָזוֹת כְּלָל. וְיֵשׁ קַוִּים שֶׁנִּרְאִים עֲקֻמִּים, וּכְשֶׁמּוֹדְדִים אוֹתָם מְגַלִּים שֶׁהֵם יְשָׁרִים לְגַמְרֵי.
זֶה נִקְרָא תַּעְתּוּעַ רְאִיָּה: הַמֹּחַ מְקַבֵּל אֶת הַסִּימָנִים מִן הָעֵינַיִם, ׳מְתַקֵּן׳ אוֹתָם בְּעַצְמוֹ – וּמַרְאֶה לָנוּ מַשֶּׁהוּ שֶׁאֵינוֹ נָכוֹן.

וְאִם הַמֹּחַ יָכוֹל לְבַלְבֵּל אֶת הָעֵינַיִם – גַּם הַלֵּב יָכוֹל. אָדָם שֶׁלִּבּוֹ מָלֵא פַּחַד מִסְתַּכֵּל עַל הָעוֹלָם וְרוֹאֶה רַק קְשָׁיִים וְסַכָּנוֹת; וְאָדָם שֶׁלִּבּוֹ מָלֵא אֱמוּנָה מִסְתַּכֵּל עַל אוֹתוֹ דָּבָר עַצְמוֹ – וְרוֹאֶה שֶׁה׳ אִתּוֹ.
לֵב מְפַחֵד רוֹאֶה אַחֶרֶת מִלֵּב מַאֲמִין. עַל זֶה בְּדִיּוּק מְדַבֵּר מַעֲשֵׂה הַמְרַגְּלִים.

חֻמַּשׁ דְּבָרִים נִקְרָא ׳מִשְׁנֵה תּוֹרָה׳, כִּי מֹשֶׁה חוֹזֵר בּוֹ עַל דְּבָרִים שֶׁכְּבָר נֶאֶמְרוּ. אֶת מַעֲשֵׂה הַמְרַגְּלִים כְּבָר לָמַדְנוּ בְּחֻמַּשׁ בְּמִדְבָּר, בְּפָרָשַׁת שְׁלַח (פְּרָקִים י״ג–י״ד).
הַמַּעֲשֶׂה קָרָה בַּשָּׁנָה הַשְּׁנִיָּה לְיצִיאַת מִצְרַיִם, וּבִגְלָלוֹ נִגְזְרוּ עַל הָעָם אַרְבָּעִים שְׁנוֹת נְדוּדִים בַּמִּדְבָּר. עַכְשָׁיו, בַּשָּׁנָה הָאַרְבָּעִים – שְׁלוֹשִׁים וּשְׁמוֹנֶה שָׁנִים אַחֲרֵי הַמַּעֲשֶׂה – מֹשֶׁה עוֹמֵד בְּעֵבֶר הַיַּרְדֵּן וּמְסַפֵּר אוֹתוֹ שׁוּב, לַדּוֹר הֶחָדָשׁ שֶׁעוֹמֵד לְהִכָּנֵס לָאָרֶץ.

📚 סִפּוּר שֶׁמְּסַפְּרִים שׁוּב וָשׁוּב
בְּכָל מִשְׁפָּחָה יֵשׁ סִפּוּר שֶׁמְּסַפְּרִים אוֹתוֹ שׁוּב וָשׁוּב: אֵיךְ סַבָּא נִצַּל, אֵיךְ סַבְתָּא עָלְתָה לָאָרֶץ, אֵיךְ הִתְגַּבְּרוּ עַל קֹשִׁי גָּדוֹל.
מָתַי מְסַפְּרִים? בְּלֵיל הַסֵּדֶר, בַּשַּׁבָּת, בְּשִׂמְחָה מִשְׁפַּחְתִּית – עַד שֶׁהַסִּפּוּר הוֹפֵךְ לְחֵלֶק מֵהַזִּכָּרוֹן הַמִּשְׁפַּחְתִּי.
וּמָה הַמַּטָּרָה? שֶׁהַדּוֹרוֹת הַבָּאִים יִלְמְדוּ מֵהַהִתְמוֹדְדוּת שֶׁל הַדּוֹרוֹת שֶׁעָבְרוּ וְיַפְנִימוּ אֶת הָעֲרָכִים שֶׁחֲשׁוּבִים בַּמִּשְׁפָּחָה.
כָּךְ בְּדִיּוּק עוֹשֶׂה מֹשֶׁה: מַעֲשֵׂה הַמְרַגְּלִים הוּא סִפּוּר שֶׁל כָּל עַם יִשְׂרָאֵל, וְהוּא מְסַפֵּר אוֹתוֹ שׁוּב כְּדֵי שֶׁנִּלְמַד מִמֶּנּוּ – עַד הַיּוֹם.

גַּם אֲנַחְנוּ חֵלֶק מֵהַמִּשְׁפָּחָה הַזּוֹ. יְצִיאַת מִצְרַיִם, מַעֲמַד הַר סִינַי וּמַעֲשֵׂה הַמְרַגְּלִים הֵם מוֹרֶשֶׁת שֶׁעוֹבֶרֶת בְּעַם יִשְׂרָאֵל מִדּוֹר לְדוֹר, וּבְכָל פַּעַם שֶׁאֲנַחְנוּ לוֹמְדִים אוֹתָם אֲנַחְנוּ מִתְחַבְּרִים אֶל כָּל הַדּוֹרוֹת שֶׁלְּפָנֵינוּ.

לִפְנֵי הַסִּפּוּר, מֹשֶׁה מַזְכִּיר אֶת הַדֶּרֶךְ: ׳וַנִּסַּע מֵחֹרֵב וַנֵּלֶךְ אֵת כָּל הַמִּדְבָּר הַגָּדוֹל וְהַנּוֹרָא הַהוּא׳. חֹרֵב הוּא הַר סִינַי, וּמִשָּׁם יָצְאוּ בְּנֵי יִשְׂרָאֵל לְדַרְכָּם אֶל הָאָרֶץ.

📜 רַשִׁ״י
׳הַמִּדְבָּר הַגָּדוֹל וְהַנּוֹרָא׳ – שֶׁהָיוּ בּוֹ נְחָשִׁים כְּקוֹרוֹת וְעַקְרַבִּים כְּקַשָּׁתוֹת.

קוֹרָה הִיא קוֹרַת עֵץ עָבָה שֶׁל גַּג, וְקֶשֶׁת הִיא כְּלִי נֶשֶׁק. כְּלוֹמַר: נְחָשִׁים עֲנָקִיִּים וְעַקְרַבִּים גְּדוֹלִים הָיוּ שָׁם – מִדְבָּר מְסֻכָּן בֶּאֱמֶת.
וְדַוְקָא מִכָּאן רוֹאִים אֶת גֹּדֶל הַשְּׁמִירָה: עַם שָׁלֵם, עִם נָשִׁים וִילָדִים, עָבַר אֶת הַמִּדְבָּר הַנּוֹרָא הַזֶּה בְּשָׁלוֹם – כִּי ה׳ שָׁמַר עֲלֵיהֶם בְּכָל צַעַד.

וְכָךְ הִגִּיעוּ ׳עַד קָדֵשׁ בַּרְנֵעַ׳, מָקוֹם שֶׁנִּמְצָא מַמָּשׁ בַּכְּנִיסָה לְאֶרֶץ יִשְׂרָאֵל. מֹשֶׁה אָמַר לָהֶם: ׳בָּאתֶם עַד הַר הָאֱמֹרִי אֲשֶׁר ה׳ אֱלֹהֵינוּ נֹתֵן לָנוּ׳ – הָאָרֶץ כְּבָר כָּאן, מוּל הָעֵינַיִם.

׳רְאֵה נָתַן ה׳ אֱלֹהֶיךָ לְפָנֶיךָ אֶת הָאָרֶץ, עֲלֵה רֵשׁ... אַל תִּירָא וְאַל תֵּחָת׳.
׳רֵשׁ׳ – רְשׁוּ אֶת הָאָרֶץ, כִּבְשׁוּ אוֹתָהּ וְקַבְּלוּ אוֹתָהּ לִנְחָלָה.
׳אַל תִּירָא וְאַל תֵּחָת׳ – שְׁתֵּי מִלִּים שֶׁפֵּרוּשָׁן אֶחָד: אַל תְּפַחֲדוּ.

💡 לָמָּה הַתּוֹרָה אוֹמֶרֶת פַּעֲמַיִם ׳אַל תְּפַחֲדוּ׳?
כִּי זֶה בֶּאֱמֶת לֹא פָּשׁוּט. לְהִכָּנֵס לְמִלְחָמָה הוּא דָּבָר מַפְחִיד.
וּכְשֶׁה׳ יוֹדֵעַ שֶׁקָּשֶׁה לָנוּ, הוּא מְחַזֵּק אוֹתָנוּ שׁוּב וָשׁוּב – כְּמוֹ אַבָּא שֶׁמַּנִּיחַ יָד עַל הַכָּתֵף וְאוֹמֵר: אַל תִּדְאַג, אֲנִי כָּאן.

׳לֹא תַעַרְצוּן׳ – לְשׁוֹן שְׁבִירָה.
׳יִלָּחֵם לָכֶם׳ – בִּשְׁבִילְכֶם.
׳אֲשֶׁר דָּרַךְ בָּהּ׳ – חֶבְרוֹן.
׳הִתְאַנַּף׳ – נִתְמַלֵּא רֹגֶז.
׳לֹא תַעֲלוּ׳ – לֹא עֲלִיָּה תְּהֵא לָכֶם אֶלָּא יְרִידָה.
׳כַּאֲשֶׁר תַּעֲשֶׂינָה הַדְּבֹרִים׳ – כְּשֶׁהִיא מַכָּה, מִיָּד מֵתָה.

וְכָךְ נִסְגָּר הַמַּעְגָּל שֶׁבּוֹ פָּתַחְנוּ: הָעֵינַיִם רוֹאוֹת, אֲבָל הַלֵּב מַחְלִיט מָה רוֹאִים. הַמְרַגְּלִים וְהָעָם רָאוּ חוֹמוֹת וַעֲנָקִים; יְהוֹשֻׁעַ וְכָלֵב רָאוּ אֶרֶץ טוֹבָה וְאֶת ה׳ שֶׁהוֹלֵךְ לְפָנֵינוּ.
גַּם לָנוּ יֵשׁ בְּחִירָה כָּזֹאת בְּכָל יוֹם: לְחַפֵּשׂ מָה חָסֵר, אוֹ לְהוֹדוֹת עַל מָה שֶׁיֵּשׁ.

💛 אַהֲבַת הַשֵּׁם
כְּשֶׁאֲנִי מְדַבֵּר בְּשִׁבְחָהּ שֶׁל אֶרֶץ יִשְׂרָאֵל, אֲנִי כְּמוֹ יֶלֶד שֶׁמְּשַׁבֵּחַ אֶת הַמַּתָּנָה שֶׁקִּבֵּל מֵאַבָּא – וְאַבָּא שָׂמֵחַ. ה׳ בָּחַר בָּאָרֶץ הַזּוֹ בִּשְׁבִילֵנוּ מֵאַהֲבָה, וַאֲנִי אוֹהֵב אוֹתָהּ בַּחֲזָרָה 💛

✨ לְסִכּוּם

  • לִפְעָמִים הָעֵינַיִם מְטֻעְטָעוֹת – וְגַם הַלֵּב: לֵב מְפַחֵד רוֹאֶה אַחֶרֶת מִלֵּב מַאֲמִין. זֶהוּ הַלֶּקַח הַגָּדוֹל שֶׁל מַעֲשֵׂה הַמְרַגְּלִים.
  • הַמַּעֲשֶׂה קָרָה בַּשָּׁנָה הַשְּׁנִיָּה וְנִלְמַד בְּפָרָשַׁת שְׁלַח, וּמֹשֶׁה חוֹזֵר עָלָיו בַּשָּׁנָה הָאַרְבָּעִים – כְּמוֹ סִפּוּר מִשְׁפַּחְתִּי שֶׁמְּסַפְּרִים שׁוּב וָשׁוּב כְּדֵי שֶׁהַדּוֹרוֹת יִלְמְדוּ.
  • הָעָם בִּקֵּשׁ לִשְׁלֹחַ מְרַגְּלִים (׳וַתִּקְרְבוּן אֵלַי כֻּלְּכֶם׳ – בְּעִרְבּוּבְיָא), מֹשֶׁה לָקַח שְׁנֵים עָשָׂר אֲנָשִׁים – אִישׁ לְכָל שֵׁבֶט, בְּלִי שֵׁבֶט לֵוִי – וְהִסְכִּים מִתּוֹךְ תִּקְוָה שֶׁיַּחְזְרוּ בָּהֶם.
  • ׳טוֹבָה הָאָרֶץ׳ אָמְרוּ יְהוֹשֻׁעַ וְכָלֵב, וְהָעָם הוּא שֶׁמֵּרָה, רִגֵּן וְאָמַר ׳בְּשִׂנְאַת ה׳ אֹתָנוּ׳ – וְרַשִׁ״י מֵשִׁיב: ׳וְהוּא הָיָה אוֹהֵב אֶתְכֶם׳.
  • תְּשׁוּבַת מֹשֶׁה לְכָל הַדּוֹרוֹת: ׳לֹא תַעַרְצוּן וְלֹא תִירְאוּן מֵהֶם, ה׳ אֱלֹהֵיכֶם הַהֹלֵךְ לִפְנֵיכֶם הוּא יִלָּחֵם לָכֶם׳ – וְהוּא נוֹשֵׂא אוֹתָנוּ ׳כַּאֲשֶׁר יִשָּׂא אִישׁ אֶת בְּנוֹ׳.
  • שָׂכָר וָעֹנֶשׁ הֵם תּוֹצָאָה טִבְעִית: דּוֹר הַמִּדְבָּר לֹא רָאָה אֶת הַטּוֹב וְלֹא זָכָה בָּאָרֶץ; כָּלֵב קִבֵּל אֶת חֶבְרוֹן, יְהוֹשֻׁעַ יַנְחִיל אֶת הָאָרֶץ, מֹשֶׁה נוֹשֵׂא בְּאַחְרָיוּת הַמַּנְהִיג, וְהַטַּף – ׳הֵמָּה יָבֹאוּ שָׁמָּה׳.
  • הַמַּעְפִּילִים נִסּוּ לְתַקֵּן נֶגֶד דְּבַר ה׳ (׳כִּי אֵינֶנִּי בְּקִרְבְּכֶם׳) וְנִכְשְׁלוּ. הַתִּקּוּן הָאֲמִתִּי הוּא לִשְׁמֹעַ בְּקוֹל ה׳ – וּבְיָמֵינוּ: לְדַבֵּר בְּשִׁבְחָהּ שֶׁל אֶרֶץ יִשְׂרָאֵל.

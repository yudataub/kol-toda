הַבְּרָכָה וְהַקְּלָלָה

בַּשִּׁעוּר הַזֶּה אֲנַחְנוּ פּוֹתְחִים אֶת פָּרָשַׁת רְאֵה, וּמַגִּיעִים לְאַחַד הָרַעְיוֹנוֹת הַגְּדוֹלִים בְּיוֹתֵר בַּתּוֹרָה: ה׳ שָׂם לְפָנֵינוּ בְּרָכָה וּקְלָלָה, וְנוֹתֵן לָנוּ לִבְחֹר בְּעַצְמֵנוּ בְּאֵיזוֹ דֶּרֶךְ לָלֶכֶת.
נִלְמַד מָה זוֹ בְּרָכָה וּמָה זוֹ קְלָלָה בִּלְשׁוֹן הַתּוֹרָה, בַּמֶּה הֵן תְּלוּיוֹת, וְאֵיךְ בְּנֵי יִשְׂרָאֵל עָמְדוּ עַל הַר גְּרִזִים וְהַר עֵיבָל וְקִבְּלוּ עֲלֵיהֶם אֶת הַתּוֹרָה בַּכְּנִיסָה לָאָרֶץ.
וּבְעִקָּר נִרְאֶה כַּמָּה ה׳ אוֹהֵב אוֹתָנוּ: הוּא לֹא כּוֹפֶה עָלֵינוּ לִהְיוֹת טוֹבִים, אֶלָּא נוֹתֵן בְּיָדֵינוּ אֶת הַהֶגֶה – כִּי הוּא מַאֲמִין בָּנוּ, וְרוֹצֶה שֶׁנִּבְחַר בּוֹ מֵאַהֲבָה 💛
בַּשִּׁעוּר: 👀 פָּרָשַׁת רְאֵה – ׳רְאֵה׳: הִסְתַּכֵּל וְהִתְבּוֹנֵן 🔵 מָה זוֹ בְּרָכָה וּמָה זוֹ קְלָלָה? 🔀 בַּמֶּה זֶה תָּלוּי? – אִם... אָז... ⛰️ הַר גְּרִזִים וְהַר עֵיבָל – סִימָנִים בַּדֶּרֶךְ 📣 הַמַּעֲמָד הַגָּדוֹל – ׳בָּרוּךְ׳, ׳אָרוּר׳, וְכָל הָעָם אוֹמֵר אָמֵן 📜 ׳כְּתַרְגּוּמוֹ׳ – אֵיךְ ׳נוֹתְנִים בְּרָכָה עַל הַר׳? 🚪 לָמָּה דַּוְקָא עַכְשָׁו? – רֶגַע לִפְנֵי הַכְּנִיסָה לָאָרֶץ 🛤️ כּוֹתֶרֶת שְׁנִיָּה: הַבְּחִירָה הַחָפְשִׁית 👑 חִזְקִיָּהוּ הַמֶּלֶךְ – אֶפְשָׁר לְשַׁנּוֹת!

דֶּרֶךְ אֲרֻכָּה כְּבָר עָשִׂינוּ בְּחֻמַּשׁ דְּבָרִים, וַאֲנַחְנוּ מִתְקַדְּמִים יָפֶה: לָמַדְנוּ אֶת פָּרָשַׁת דְּבָרִים, אֶת פָּרָשַׁת וָאֶתְחַנַּן וְאֶת פָּרָשַׁת עֵקֶב – שָׁלֹשׁ פָּרָשׁוֹת סִיַּמְנוּ, וְעַכְשָׁו אֲנַחְנוּ מַתְחִילִים אֶת הָרְבִיעִית: פָּרָשַׁת רְאֵה.

לָמָּה הִיא נִקְרֵאת כָּךְ? כְּמוֹ בְּכָל הַפָּרָשׁוֹת – עַל שֵׁם הַמִּלָּה הָרִאשׁוֹנָה שֶׁבָּהּ: ׳רְאֵה אָנֹכִי נֹתֵן לִפְנֵיכֶם הַיּוֹם׳.
וְ׳רְאֵה׳ אֵינוֹ רַק לִפְקֹחַ עֵינַיִם. הוּא אוֹמֵר: הִסְתַּכֵּל וְהִתְבּוֹנֵן. יֵשׁ כָּאן מַשֶּׁהוּ גָּדוֹל שֶׁצָּרִיךְ לִרְאוֹת גַּם בַּלֵּב, וְלֹא רַק בָּעַיִן.

💡 ׳רְאֵה׳ – בִּלְשׁוֹן יָחִיד
שִׂימוּ לֵב לְדָבָר מְיֻחָד: מֹשֶׁה עוֹמֵד לִפְנֵי כָּל עַם יִשְׂרָאֵל, אֲבָל הוּא פּוֹתֵחַ בִּלְשׁוֹן יָחִיד – ׳רְאֵה׳, וְלֹא ׳רְאוּ׳.
כְּאִלּוּ ה׳ פּוֹנֶה אֶל כָּל יֶלֶד וְאֶל כָּל יַלְדָּה בִּנְפָרָד וְאוֹמֵר: הַבְּחִירָה הִיא שֶׁלְּךָ, וְאַתָּה חָשׁוּב לִי.

וּמִיָּד אַחַר כָּךְ הַפָּסוּק מַמְשִׁיךְ בִּלְשׁוֹן רַבִּים: ׳נֹתֵן לִפְנֵיכֶם׳. כָּךְ הַתּוֹרָה מְלַמֶּדֶת שֶׁהַבְּחִירָה הִיא גַּם שֶׁל כָּל הָעָם יַחַד, וְגַם שֶׁל כָּל אֶחָד לְבַדּוֹ.
ה׳ אוֹהֵב אֶת עַם יִשְׂרָאֵל כֻּלּוֹ, וְאוֹהֵב גַּם כָּל אֶחָד מֵאִתָּנוּ בִּפְנֵי עַצְמוֹ – וְלָכֵן הוּא מְדַבֵּר אֵלֵינוּ בִּשְׁנֵי הַלְּשׁוֹנוֹת.

💛 אַהֲבַת הַשֵּׁם
ה׳ אֵינוֹ מְדַבֵּר אֵלֵינוּ מֵרָחוֹק. הוּא פּוֹנֶה אֵלַי בִּלְשׁוֹן יָחִיד, כְּמוֹ אַבָּא שֶׁמִּתְכּוֹפֵף אֶל הַיֶּלֶד שֶׁלּוֹ וְאוֹמֵר לוֹ בְּשֶׁקֶט: תִּסְתַּכֵּל, יַקִּירִי – אֲנִי נוֹתֵן לְךָ אֶת הַטּוֹב 💛

🔵מָה זוֹ בְּרָכָה וּמָה זוֹ קְלָלָה?

כו רְאֵה אָנֹכִי נֹתֵן לִפְנֵיכֶם הַיּוֹם בְּרָכָה וּקְלָלָה כז אֶת הַבְּרָכָה אֲשֶׁר תִּשְׁמְעוּ אֶל מִצְוֺת ה׳ אֱלֹהֵיכֶם אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם הַיּוֹם כח וְהַקְּלָלָה אִם לֹא תִשְׁמְעוּ אֶל מִצְוֺת ה׳ אֱלֹהֵיכֶם וְסַרְתֶּם מִן הַדֶּרֶךְ אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם הַיּוֹם לָלֶכֶת אַחֲרֵי אֱלֹהִים אֲחֵרִים אֲשֶׁר לֹא יְדַעְתֶּם

׳רְאֵה אָנֹכִי נֹתֵן לִפְנֵיכֶם הַיּוֹם בְּרָכָה וּקְלָלָה׳. ה׳ שָׂם לְפָנֵינוּ שְׁתֵּי דְּרָכִים, וּבְסוֹף כָּל דֶּרֶךְ יֵשׁ תּוֹצָאָה אַחֶרֶת.

מָה זוֹ בְּרָכָה? אֶת זֶה כְּבָר לָמַדְנוּ בַּפָּרָשׁוֹת הָאַחֲרוֹנוֹת. בְּרָכָה הִיא כָּל הַטּוֹב: שֶׁפַע וּפַרְנָסָה, גֶּשֶׁם בְּעִתּוֹ, שָׂדוֹת שֶׁנּוֹתְנִים פְּרִי, בִּטָּחוֹן בָּאָרֶץ וְנִצָּחוֹן עַל הָאוֹיְבִים.

וּמָה זוֹ קְלָלָה? כָּאן צָרִיךְ לְהִזָּהֵר מִטָּעוּת. בִּלְשׁוֹן הַתּוֹרָה קְלָלָה אֵינָהּ מִלִּים גַּסּוֹת וְלֹא דִּבּוּר מְכֹעָר, כְּמוֹ שֶׁאוֹמְרִים הַיּוֹם.
קְלָלָה הִיא הַהֵפֶךְ מִבְּרָכָה: דְּבָרִים רָעִים שֶׁבָּאִים עַל הָאָדָם – הֶפְסֵד לָאוֹיְבִים, גָּלוּת מִן הָאָרֶץ, עֹנִי וְאֵין פַּרְנָסָה, וְאֵין גֶּשֶׁם.
כְּשֶׁאָדָם ׳מְקַלֵּל׳ בִּלְשׁוֹן הַתּוֹרָה וַחֲזַ״ל, הַכַּוָּנָה הִיא שֶׁהוּא מְאַחֵל שֶׁיָּבוֹאוּ עַל חֲבֵרוֹ דְּבָרִים רָעִים.

🤔 פְּעִילוּת: צוֹבְעִים אֶת הַפְּסוּקִים
אֶת הַמִּלָּה ׳בְּרָכָה׳ צוֹבְעִים בְּכָחֹל – צֶבַע נָעִים וְרָגוּעַ.
אֶת הַמִּלָּה ׳קְלָלָה׳ צוֹבְעִים בְּאָדֹם – צֶבַע שֶׁל אַזְהָרָה.
עַכְשָׁו חַפְּשׂוּ בַּפְּסוּקִים עוֹד מְקוֹמוֹת שֶׁבָּהֶם מוֹפִיעוֹת הַמִּלִּים הָאֵלֶּה. תִּמְצְאוּ אוֹתָן בְּפָסוּק כ״ז (׳אֶת הַבְּרָכָה׳), בְּפָסוּק כ״ח (׳וְהַקְּלָלָה׳) וּבְפָסוּק כ״ט (׳הַבְּרָכָה... הַקְּלָלָה׳).
וּכְשֶׁרוֹאִים אֵילוּ מִלִּים חוֹזְרוֹת – כְּבָר יוֹדְעִים אֵיזוֹ כּוֹתֶרֶת לָתֵת לַפְּסוּקִים: ׳הַבְּרָכָה וְהַקְּלָלָה׳.

׳רְאֵה אָנֹכִי נֹתֵן לִפְנֵיכֶם הַיּוֹם בְּרָכָה וּקְלָלָה׳ – הָאֲמוּרוֹת בְּהַר גְּרִזִים וּבְהַר עֵיבָל.

כְּלוֹמַר: הַבְּרָכָה וְהַקְּלָלָה שֶׁמֹּשֶׁה מַזְכִּיר כָּאן אֵינָן רַק רַעְיוֹן – הֵן אוֹתָן בְּרָכוֹת וּקְלָלוֹת שֶׁבְּנֵי יִשְׂרָאֵל יֹאמְרוּ בְּעַצְמָם, בְּמַעֲמָד גָּדוֹל, עַל שְׁנֵי הֶהָרִים (וְעַל כָּךְ נִלְמַד בַּהֲמֶשֶׁךְ).

׳וְהַקְּלָלָה – אִם לֹא תִשְׁמְעוּ אֶל מִצְוֹת ה׳ אֱלֹהֵיכֶם, וְסַרְתֶּם מִן הַדֶּרֶךְ... לָלֶכֶת אַחֲרֵי אֱלֹהִים אֲחֵרִים׳.

עַכְשָׁו הוּא עוֹמֵד רָטֹב, מְלֻכְלָךְ וְלֹא נָעִים לוֹ. הַאִם הוּא יָכוֹל לִכְעֹס עַל אִמָּא? וַדַּאי שֶׁלֹּא – הוּא בָּחַר, וְזוֹ הַתּוֹצָאָה שֶׁל הַבְּחִירָה שֶׁלּוֹ.
וְאִמָּא? הִיא אֵינָהּ צְרִיכָה לִצְעֹק, אֶלָּא לְהַסְבִּיר בְּנַחַת: רָאִיתָ מָה קוֹרֶה? וּמָה יִלְמַד הַיֶּלֶד לַפַּעַם הַבָּאָה? שֶׁכְּדַאי לְהַקְשִׁיב – כִּי מִי שֶׁאוֹהֵב אוֹתוֹ בֶּאֱמֶת גַּם יוֹדֵעַ מָה טוֹב בִּשְׁבִילוֹ.
כָּךְ בְּדִיּוּק הַפְּסוּקִים שֶׁלָּנוּ: ה׳ מַרְאֶה לָנוּ אֶת שְׁתֵּי הַדְּרָכִים מֵרֹאשׁ, וּבַפַּעַם הַבָּאָה אֶפְשָׁר לִבְחֹר אַחֶרֶת.

וְכָאן מִתְגַּלָּה אַהֲבָה עֲמֻקָּה: ה׳ יָכוֹל הָיָה לִבְרֹא אוֹתָנוּ כְּמוֹ רוֹבּוֹטִים שֶׁעוֹשִׂים רַק טוֹב. אֲבָל הוּא רָצָה שֶׁנִּהְיֶה בְּנֵי חוֹרִין, שֶׁנִּבְחַר בּוֹ מֵעַצְמֵנוּ.
כִּי אַהֲבָה שֶׁכּוֹפִים אוֹתָהּ אֵינָהּ אַהֲבָה. רַק כְּשֶׁיֵּשׁ בְּחִירָה – הַמִּצְוָה שֶׁלִּי הִיא מַתָּנָה שֶׁאֲנִי נוֹתֵן לַה׳.

ה׳ נָתַן לִי אֶת הַהֶגֶה בַּיָּדַיִם, כִּי הוּא מַאֲמִין בִּי. וּכְשֶׁאֲנִי בּוֹחֵר לַעֲשׂוֹת אֶת רְצוֹנוֹ – לֹא כִּי מֻכְרָחִים, אֶלָּא כִּי אֲנִי אוֹהֵב אוֹתוֹ – זוֹ הַמַּתָּנָה הֲכִי יָפָה שֶׁאֲנִי יָכוֹל לָתֵת לוֹ 💛

👑חִזְקִיָּהוּ הַמֶּלֶךְ – אֶפְשָׁר לְשַׁנּוֹת!

אִם הַכֹּל בְּיָדֵינוּ, מָה קוֹרֶה כְּשֶׁנִּרְאֶה שֶׁכְּבָר הֻחְלַט מֵעָלֵינוּ? עַל כָּךְ מְלַמֵּד סִפּוּרוֹ שֶׁל חִזְקִיָּהוּ הַמֶּלֶךְ.

📚 הַמֶּלֶךְ שֶׁלֹּא הִתְיָאֵשׁ
חִזְקִיָּהוּ, מֶלֶךְ יְהוּדָה, חָלָה. בָּא אֵלָיו יְשַׁעְיָהוּ הַנָּבִיא וְאָמַר לוֹ שֶׁכְּבָר נִגְזְרָה עָלָיו גְּזֵרָה שֶׁיָּמוּת.
חִזְקִיָּהוּ לֹא הִתְיָאֵשׁ וְלֹא אָמַר ׳אֵין מָה לַעֲשׂוֹת׳. הוּא הֵפֵךְ אֶת פָּנָיו אֶל הַקִּיר, הִתְפַּלֵּל וּבִקֵּשׁ רַחֲמִים.
וְה׳ שָׁמַע אֶת תְּפִלָּתוֹ: הַגְּזֵרָה בֻּטְּלָה, חִזְקִיָּהוּ הִתְרַפֵּא, וְנוֹסְפוּ לוֹ שָׁנִים שֶׁל חַיִּים.

מָה הַמֶּסֶר? לַקָּדוֹשׁ בָּרוּךְ הוּא יֵשׁ תָּכְנִיּוֹת לְכָל אֶחָד וְאַחַת מֵאִתָּנוּ – אֲבָל אֶפְשָׁר לְשַׁנּוֹת אוֹתָן בְּמַעֲשִׂים טוֹבִים וּבִתְפִלָּה.
גַּם כְּשֶׁנִּרְאֶה שֶׁהַכֹּל סָגוּר – שַׁעֲרֵי תְּפִלָּה לֹא נִנְעָלִים. וְזֶה בְּעַצְמוֹ סִימָן לְאַהֲבָתוֹ: אַבָּא שֶׁמַּקְשִׁיב לַבַּקָּשׁוֹת שֶׁל הַיְּלָדִים שֶׁלּוֹ, וּמוּכָן לְשַׁנּוֹת בִּשְׁבִילָם.

💡 בְּחִירָה גַּם בַּמָּקוֹם הֲכִי קָשֶׁה
לַמּוֹרֶה: בְּכִתּוֹת הַגְּבוֹהוֹת אֶפְשָׁר לְסַפֵּר, בַּעֲדִינוּת, עַל אֲנָשִׁים שֶׁהָיוּ בְּמָקוֹם קָשֶׁה מְאוֹד וְלֹא הָיְתָה לָהֶם שְׁלִיטָה עַל מָה שֶׁקּוֹרֶה סְבִיבָם.
וְאַף עַל פִּי כֵן נִשְׁאֲרָה בְּיָדָם בְּחִירָה אַחַת: אֵיךְ לְהִתְנַהֵג – לְעוֹדֵד אֲחֵרִים, לְחַלֵּק אֶת פְּרוּסַת הַלֶּחֶם הָאַחֲרוֹנָה, לִשְׁמֹר עַל חֵרוּת פְּנִימִית וְעַל אֱמוּנָה.
הַבְּחִירוֹת הָאֵלֶּה נָתְנוּ לַחַיִּים שֶׁלָּהֶם מַשְׁמָעוּת.
🤔 פְּעִילוּת סִכּוּם: כַּמָּה בְּחִירוֹת בָּחַרְתִּי הַיּוֹם?
מִי יָכוֹל לְסַפֵּר כַּמָּה בְּחִירוֹת בָּחַר מֵהָרֶגַע שֶׁקָּם בַּבֹּקֶר וְעַד עַכְשָׁו?
מָתַי לָקוּם, מָה לִלְבֹּשׁ, לְבָרֵךְ אוֹ לִשְׁכֹּחַ, לַעֲנוֹת יָפֶה אוֹ לִכְעֹס, לְשַׁתֵּף חָבֵר אוֹ לְהִתְעַלֵּם...
עַכְשָׁו בַּחֲרוּ בְּחִירָה מַשְׁמָעוּתִית אַחַת שֶׁבְּחַרְתֶּם הַיּוֹם, וְשַׁאֲלוּ: מָה הָיוּ הָאֶפְשָׁרֻיּוֹת? מָה בָּחַרְתִּי? מָה יָצָא מִזֶּה? וּמָה אֶבְחַר בַּפַּעַם הַבָּאָה?

וְכָךְ מִתְחַבֵּר הַכֹּל: ׳רְאֵה אָנֹכִי נֹתֵן לִפְנֵיכֶם הַיּוֹם׳ – הַיּוֹם, בְּכָל יוֹם מֵחָדָשׁ. בְּכָל בֹּקֶר ה׳ שָׂם לְפָנַי מֵחָדָשׁ אֶת הַדֶּרֶךְ הַטּוֹבָה, וּמְחַכֶּה שֶׁאֶבְחַר בָּהּ.

ה׳ לֹא נוֹעֵל שׁוּם דֶּלֶת לִפְנֵי יֶלֶד שֶׁרוֹצֶה לִהְיוֹת טוֹב. גַּם אִם טָעִיתִי אֶתְמוֹל – הַיּוֹם אֲנִי יָכוֹל לְהִתְפַּלֵּל, לְהִשְׁתַּנּוֹת וְלַחֲזֹר אֵלָיו, וְהוּא מְחַכֶּה לִי בְּאַהֲבָה 💛

✨ לְסִכּוּם

  • פָּרָשַׁת רְאֵה נִקְרֵאת עַל שֵׁם הַמִּלָּה הָרִאשׁוֹנָה – ׳רְאֵה׳: הִסְתַּכֵּל וְהִתְבּוֹנֵן. מֹשֶׁה מְדַבֵּר אֶל כָּל הָעָם, אֲבָל בִּלְשׁוֹן יָחִיד.
  • בְּרָכָה הִיא כָּל הַטּוֹב – שֶׁפַע, פַּרְנָסָה, גֶּשֶׁם וּבִטָּחוֹן. קְלָלָה אֵינָהּ מִלִּים גַּסּוֹת, אֶלָּא הַהֵפֶךְ מִבְּרָכָה: עֹנִי, גָּלוּת וְהֶפְסֵד.
  • הַבְּרָכָה בָּאָה ׳עַל מְנָת אֲשֶׁר תִּשְׁמְעוּ׳, וְהַקְּלָלָה – אִם יָסוּרוּ מִן הַדֶּרֶךְ. וְכָל הָעוֹבֵד עֲבוֹדָה זָרָה סָר מִכָּל הַדֶּרֶךְ, כְּכוֹפֵר בְּכָל הַתּוֹרָה כֻּלָּהּ.
  • הַמַּעֲמָד הִתְקַיֵּם אַחֲרֵי מַעֲבַר הַיַּרְדֵּן עַל הַר גְּרִזִים וְהַר עֵיבָל, לְיַד שְׁכֶם: שִׁשָּׁה שְׁבָטִים עַל כָּל הַר, הָאָרוֹן בָּאֶמְצַע, ׳בָּרוּךְ׳ וְ׳אָרוּר׳ – וְכָל הָעָם עָנָה ׳אָמֵן׳.
  • ׳וְנָתַתָּה אֶת הַבְּרָכָה׳ – כְּתַרְגּוּמוֹ: ׳יָת מְבָרְכַיָּא׳, אֶת הַמְבָרְכִים; וְ׳עַל הַר גְּרִזִים׳ – כְּלַפֵּי הָהָר. אוּנְקְלוֹס תִּרְגֵּם אֶת הַתּוֹרָה לַאֲרָמִית וְגַם הִסְבִּיר אוֹתָהּ.
  • הַכּוֹתֶרֶת הַשְּׁנִיָּה הִיא הַבְּחִירָה הַחָפְשִׁית: הָרַמְבַּ״ם – ׳רְשׁוּת כָּל אָדָם נְתוּנָה לוֹ׳; וְחִזְקִיָּהוּ – אֶפְשָׁר לְשַׁנּוֹת בִּתְפִלָּה וּבְמַעֲשִׂים. ה׳ נָתַן לָנוּ לִבְחֹר, כִּי הוּא רוֹצֶה שֶׁנִּבְחַר בּוֹ מֵאַהֲבָה.

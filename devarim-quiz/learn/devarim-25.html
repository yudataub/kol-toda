אַרְבָּעִים שָׁנָה עָמַד הַמִּשְׁכָּן בְּמֶרְכַּז הַמַּחֲנֶה, וְכָל הַחַיִּים שֶׁל עַם יִשְׂרָאֵל סָבְבוּ אוֹתוֹ. עַכְשָׁיו, עֶרֶב הַכְּנִיסָה לָאָרֶץ, הַכֹּל עוֹמֵד לְהִשְׁתַּנּוֹת: הָעָם יִתְפַּזֵּר לִנְחָלוֹת וּלְעָרִים רְחוֹקוֹת זוֹ מִזּוֹ.
וְאָז נִשְׁאֶלֶת הַשְּׁאֵלָה הַגְּדוֹלָה: אֵיפֹה יַעַבְדוּ אֶת ה׳? מֹשֶׁה עוֹנֶה: לֹא עַל הֶהָרִים וְלֹא תַּחַת כָּל עֵץ רַעֲנָן, אֶלָּא ׳אֶל הַמָּקוֹם אֲשֶׁר יִבְחַר ה׳׳ – מָקוֹם אֶחָד, שֶׁכָּל הָעָם עוֹלֶה אֵלָיו.
נִלְמַד עַל עֲקִירַת הָעֲבוֹדָה הַזָּרָה מִן הָאָרֶץ, עַל הֲלָכוֹת גְּנִיזָה, עַל הַהֶבְדֵּל בֵּין הַמִּשְׁכָּן לַמִּקְדָּשׁ, עַל הֶתֵּר בְּשַׂר תַּאֲוָה וְאִסּוּר הַדָּם, וְעַל הַזְּהִירוּת מִסַּקְרָנוּת. וּבְעִקָּר נִרְאֶה כַּמָּה ה׳ רוֹצֶה לִהְיוֹת קָרוֹב אֵלֵינוּ – וְכַמָּה אֲנַחְנוּ מִתְגַּעְגְּעִים אֵלָיו.
בַּשִּׁעוּר: ⛺ הַמִּשְׁכָּן – לֵב הַמַּחֲנֶה 🌳 אַבֵּד תְּאַבְּדוּן – עֲקִירַת הָעֲבוֹדָה הַזָּרָה מִן הָאָרֶץ 📄 לֹא תַעֲשׂוּן כֵּן – שְׁלוֹשָׁה פֵּרוּשִׁים בְּרַשִׁ״י, וַהֲלָכוֹת גְּנִיזָה 🕍 כִּי אִם אֶל הַמָּקוֹם אֲשֶׁר יִבְחַר ה׳ – ׳לְשִׁכְנוֹ תִדְרְשׁוּ׳ 🐑 שָׁמָּה תָבִיאוּ – וּשְׂמַחְתֶּם לִפְנֵי ה׳ 🏛️ מִקְדָּשׁ אֶחָד בִּלְבַד – מִשְׁכָּן, מִקְדָּשׁ וְצִיר הַזְּמַן 🥩 בְּכָל אַוַּת נַפְשְׁךָ – הֶתֵּר בְּשַׂר תַּאֲוָה וְאִסּוּר הַדָּם 🔍 וּפֶן תִּדְרֹשׁ – הַסַּכָּנָה שֶׁבַּסַּקְרָנוּת 💛 בַּדֶּרֶךְ אֶל הַמָּקוֹם – מִלּוֹן רַשִׁ״י וְגַעְגּוּעַ

📖 פֶּרֶק י״ב, פָּסוּק א׳ ←
א אֵלֶּה הַחֻקִּים וְהַמִּשְׁפָּטִים אֲשֶׁר תִּשְׁמְרוּן לַעֲשׂוֹת בָּאָרֶץ אֲשֶׁר נָתַן ה׳ אֱלֹהֵי אֲבֹתֶיךָ לְךָ לְרִשְׁתָּהּ כָּל הַיָּמִים אֲשֶׁר אַתֶּם חַיִּים עַל הָאֲדָמָה

אַרְבָּעִים שָׁנָה נָדְדוּ בְּנֵי יִשְׂרָאֵל בַּמִּדְבָּר, וּבְכָל מָקוֹם שֶׁחָנוּ בּוֹ עָמַד הַמִּשְׁכָּן בְּמֶרְכַּז הַמַּחֲנֶה. שְׁנֵים עָשָׂר הַשְּׁבָטִים חָנוּ סְבִיבוֹ, שְׁלוֹשָׁה בְּכָל צַד, וְהַמִּשְׁכָּן בָּאֶמְצַע – כְּמוֹ לֵב בְּתוֹךְ הַגּוּף.
כְּשֶׁנָּסְעוּ – פֵּרְקוּ אוֹתוֹ וְנָשְׂאוּ אוֹתוֹ, וּכְשֶׁחָנוּ – הַדָּבָר הָרִאשׁוֹן שֶׁהֵקִימוּ הָיָה הַמִּשְׁכָּן.

נְנַסֶּה לַחְשֹׁב: אֵילוּ דְּבָרִים בַּחַיִּים שֶׁל יְהוּדִי בַּמִּדְבָּר הָיוּ קְשׁוּרִים לַמִּשְׁכָּן?
קָרְבְּנוֹת נְדָבָה: קָרָה לְאָדָם מַשֶּׁהוּ טוֹב וְהוּא רָצָה לְהוֹדוֹת – הֵבִיא עוֹלָה אוֹ שְׁלָמִים.
קָרְבְּנוֹת כַּפָּרָה: חָטָא – הֵבִיא אֶת קָרְבַּן הַכַּפָּרָה שֶׁלּוֹ.
חֲצוֹצְרוֹת: תָּקְעוּ בָּהֶן – וְהַנְּשִׂיאִים נוֹעֲדוּ אֶל פֶּתַח אֹהֶל מוֹעֵד.
יוֹם הַכִּפּוּרִים: כִּפְּרוּ שָׁם עַל כָּל מַחֲנֵה יִשְׂרָאֵל.
דְּבַר ה׳: שָׁם דִּבֵּר ה׳ עִם מֹשֶׁה, וּמִשָּׁם עָבְרוּ הַמִּצְווֹת אֶל כָּל הָעָם.

וְעוֹד וְעוֹד: הַתְּרוּמוֹת וּמַחֲצִית הַשֶּׁקֶל, עִנְיְנֵי טָהֳרָה, עֲבוֹדַת הַכֹּהֲנִים וְהַלְוִיִּים, הַתְּפִלּוֹת וְהַנְּאוּמִים שֶׁל מֹשֶׁה.
הַמִּשְׁכָּן הָיָה הַלֵּב שֶׁל הַמַּחֲנֶה – כֻּלָּם הָיוּ קְרוֹבִים אֵלָיו, וְכֻלָּם הִרְגִּישׁוּ לְיָדוֹ אֶת הַקְּדֻשָּׁה וְאֶת הַשְּׁכִינָה.

וְלָמָּה בִּקֵּשׁ ה׳ מִשְׁכָּן דַּוְקָא בָּאֶמְצַע? כִּי הוּא רָצָה לִהְיוֹת קָרוֹב אֵלֵינוּ. כָּל יֶלֶד שֶׁיָּצָא מֵאֹהֶל מִשְׁפַּחְתּוֹ רָאָה אֶת עַנְנֵי הַכָּבוֹד מֵעַל הַמִּשְׁכָּן וְיָדַע: ה׳ גָּר אִתָּנוּ כָּאן, בְּתוֹךְ הַמַּחֲנֶה.

וְעַכְשָׁיו, בְּעֵבֶר הַיַּרְדֵּן, פּוֹתֵחַ מֹשֶׁה חֵלֶק חָדָשׁ בַּנְּאוּם: ׳אֵלֶּה הַחֻקִּים וְהַמִּשְׁפָּטִים אֲשֶׁר תִּשְׁמְרוּן לַעֲשׂוֹת בָּאָרֶץ אֲשֶׁר נָתַן ה׳ אֱלֹהֵי אֲבֹתֶיךָ לְךָ לְרִשְׁתָּהּ׳.
אֵלֶּה מִצְווֹת שֶׁשַּׁיָּכוֹת לַחַיִּים הַחֲדָשִׁים – לַחַיִּים בְּאֶרֶץ יִשְׂרָאֵל.

כִּי בָּאָרֶץ הַכֹּל יִשְׁתַּנֶּה: לֹא עוֹד מַחֲנֶה אֶחָד מְסֻדָּר סְבִיב הַמִּשְׁכָּן, אֶלָּא עָם שֶׁמִּתְפַּזֵּר לִנְחָלוֹת – שֵׁבֶט נַפְתָּלִי בַּצָּפוֹן, שִׁמְעוֹן בַּדָּרוֹם, זְבוּלוּן לְיַד הַיָּם.
אָדָם שֶׁגָּר בַּגָּלִיל אוֹ בַּנֶּגֶב רָחוֹק מְאוֹד מִן הַמִּשְׁכָּן. וְאָז אֵיפֹה יַעַבְדוּ אֶת ה׳? זוֹ הַשְּׁאֵלָה שֶׁפֶּרֶק י״ב כֻּלּוֹ עוֹנֶה עָלֶיהָ.

🤔 לַמּוֹרֶה: לְהַמְחִישׁ אֶת הַשִּׁנּוּי
אֶפְשָׁר לְצַיֵּר עַל הַלּוּחַ שְׁנֵי שִׂרְטוּטִים: בָּרִאשׁוֹן – מַחֲנֵה הַמִּדְבָּר, הַמִּשְׁכָּן בָּאֶמְצַע וְהַדְּגָלִים סְבִיבוֹ. בַּשֵּׁנִי – מַפַּת אֶרֶץ יִשְׂרָאֵל עִם נַחֲלוֹת הַשְּׁבָטִים, מִדָּן וְעַד בְּאֵר שֶׁבַע.
וְלִשְׁאֹל: מָה הִשְׁתַּנָּה? הַמֶּרְחָק. וְלַשִּׁנּוּי הַזֶּה יֵשׁ מַשְׁמָעוּת גַּשְׁמִית (דֶּרֶךְ אֲרֻכָּה) וְגַם רוּחָנִית (הַקְּדֻשָּׁה כְּבָר אֵינָהּ מִחוּץ לָאֹהֶל).
💛 אַהֲבַת הַשֵּׁם
בַּמִּדְבָּר ה׳ שָׁם אֶת בֵּיתוֹ בְּדִיּוּק בָּאֶמְצַע, כְּדֵי שֶׁכָּל אֶחָד יַרְגִּישׁ שֶׁהוּא קָרוֹב אֵלָיו. גַּם הַיּוֹם ה׳ רוֹצֶה לִהְיוֹת בַּמֶּרְכָּז שֶׁל הַחַיִּים שֶׁלִּי – וְלֹא בַּפִּנָּה 💛

🌳אַבֵּד תְּאַבְּדוּן – עֲקִירַת הָעֲבוֹדָה הַזָּרָה מִן הָאָרֶץ

📖 פֶּרֶק י״ב, פְּסוּקִים ב׳–ג׳ ←
ב אַבֵּד תְּאַבְּדוּן אֶת כָּל הַמְּקֹמוֹת אֲשֶׁר עָבְדוּ שָׁם הַגּוֹיִם אֲשֶׁר אַתֶּם יֹרְשִׁים אֹתָם אֶת אֱלֹהֵיהֶם עַל הֶהָרִים הָרָמִים וְעַל הַגְּבָעוֹת וְתַחַת כָּל עֵץ רַעֲנָן ג וְנִתַּצְתֶּם אֶת מִזְבּחֹתָם וְשִׁבַּרְתֶּם אֶת מַצֵּבֹתָם וַאֲשֵׁרֵיהֶם תִּשְׂרְפוּן בָּאֵשׁ וּפְסִילֵי אֱלֹהֵיהֶם תְּגַדֵּעוּן וְאִבַּדְתֶּם אֶת שְׁמָם מִן הַמָּקוֹם הַהוּא

בְּנֵי יִשְׂרָאֵל נִכְנָסִים לָאָרֶץ – וּפוֹגְשִׁים שָׁם קֹשִׁי גָּדוֹל. בִּמְקוֹם הַמִּשְׁכָּן שֶׁבְּלֵב הַמַּחֲנֶה הֵם רוֹאִים גּוֹיִים, וּבִמְקוֹם עֲנַן הַקְּטֹרֶת הֵם רוֹאִים פְּסָלִים, עֵצִים שֶׁעוֹבְדִים אוֹתָם וְתַרְבּוּת טְמֵאָה.
אֶפְשָׁר לְשַׁעֵר מָה הִרְגִּישׁוּ בַּלֵּב: כְּאֵב גָּדוֹל, וְאוּלַי גַּם פַּחַד – אֵיךְ נִחְיֶה מוּל כָּל זֶה?

שְׁנֵי דְּבָרִים צְרִיכִים בְּנֵי יִשְׂרָאֵל לַעֲשׂוֹת עִם גּוֹיֵי הָאָרֶץ:
לְהִלָּחֵם בָּהֶם וְלִכְבֹּשׁ מִיָּדָם אֶת הָאָרֶץ.
לְהִזָּהֵר מְאֹד שֶׁלֹּא לִלְמֹד מִמַּעֲשֵׂיהֶם הָרָעִים.

וְעַל אֵיזֶה עָווֹן שֶׁלָּהֶם הַתּוֹרָה בּוֹחֶרֶת לְסַפֵּר לָנוּ? עַל עֲבוֹדָה זָרָה.
וּמָה רַע בָּהּ? זֶה לְהַחְלִיף אֶת ה׳, שֶׁבָּרָא אוֹתָנוּ וְאוֹהֵב אוֹתָנוּ, בְּאֵל זָר: לִבְנוֹת פֶּסֶל מֵעֵץ אוֹ מֵאֶבֶן, לְהַאֲמִין שֶׁהוּא הָאֱלֹהִים וְלִכְרֹעַ לְפָנָיו. אֵין בּוֹ שׁוּם כֹּחַ וְאֵין בּוֹ שׁוּם אַהֲבָה – וְהוּא לוֹקֵחַ אֶת הַלֵּב מֵה׳ הָאֲמִתִּי.

פָּסוּק ב׳ מְסַפֵּר אֵיפֹה עָבְדוּ הַגּוֹיִם: ׳עַל הֶהָרִים הָרָמִים וְעַל הַגְּבָעוֹת וְתַחַת כָּל עֵץ רַעֲנָן׳.

וְיוֹדְעִים מַהוּ הַסֵּפֶר הַנִּמְכָּר בְּיוֹתֵר בְּכָל הַזְּמַנִּים? הַתַּנַ״ךְ שֶׁלָּנוּ. כָּל הָעוֹלָם רוֹצֶה לִקְרֹא אוֹתוֹ וּלְהַכִּיר אוֹתוֹ.
וְגַם בַּפֶּרֶק שֶׁלָּנוּ מְסֻפָּר עַל חֵפֶץ שֶׁהָיָה בִּזְמַנּוֹ הַגָּדוֹל בָּעוֹלָם.

׳כִּי רַק עוֹג מֶלֶךְ הַבָּשָׁן נִשְׁאַר מִיֶּתֶר הָרְפָאִים, הִנֵּה עַרְשׂוֹ עֶרֶשׂ בַּרְזֶל, הֲלֹה הִוא בְּרַבַּת בְּנֵי עַמּוֹן, תֵּשַׁע אַמּוֹת אָרְכָּהּ וְאַרְבַּע אַמּוֹת רָחְבָּהּ בְּאַמַּת אִישׁ׳.
׳עֶרֶשׂ׳ הִיא מִטָּה – כְּמוֹ ׳שִׁיר עֶרֶשׂ׳ שֶׁשָּׁרִים לִפְנֵי הַשֵּׁנָה. הָרְפָאִים הָיוּ עַם קַדְמוֹן שֶׁל עֲנָקִים, וְעוֹג הָיָה הָאַחֲרוֹן שֶׁנִּשְׁאַר מֵהֶם. מִטָּתוֹ הָיְתָה עֲשׂוּיָה בַּרְזֶל, וְהִיא נִשְׁמְרָה בְּרַבַּת בְּנֵי עַמּוֹן, הָעִיר הַגְּדוֹלָה שֶׁל עַמּוֹן.

וּמָה הֵן ׳תֵּשַׁע אַמּוֹת׳? אַמָּה הִיא מִדָּה שֶׁמָּדְדוּ בָּהּ בִּימֵי קֶדֶם: מִן הַמַּרְפֵּק וְעַד קְצֵה הָאֶצְבַּע. לְפִי הָאַמָּה שֶׁל אָדָם רָגִיל, תֵּשַׁע אַמּוֹת הֵן קָרוֹב לַחֲמִשָּׁה מֶטֶר אֹרֶךְ, וְאַרְבַּע אַמּוֹת – יוֹתֵר מִשְּׁנֵי מֶטֶר רֹחַב. אֵיזוֹ מִטָּה עֲנָקִית!
וְרַשִׁ״י מוֹסִיף שֶׁ׳בְּאַמַּת אִישׁ׳ פֵּרוּשׁוֹ בָּאַמָּה שֶׁל עוֹג עַצְמוֹ – לְפִי גֹּדֶל הַיָּד שֶׁלּוֹ.

וְאִם כָּךְ – כַּמָּה גָּדוֹל הַנֵּס שֶׁיִּשְׂרָאֵל נִצְּחוּ אוֹתוֹ! הַמִּדּוֹת שֶׁל הַמִּטָּה הֵן הַדֶּרֶךְ שֶׁל הַתּוֹרָה לְסַפֵּר עַל גְּדֻלַּת ה׳.

וְעוֹד דָּבָר יָפֶה: עַל הָאָרֶץ הַזֹּאת כָּתוּב ׳הַהוּא יִקָּרֵא אֶרֶץ רְפָאִים׳, וְרַשִׁ״י מְבָאֵר: ׳הִיא אוֹתָהּ שֶׁנָּתַתִּי לְאַבְרָהָם׳.

אַחֲרֵי שְׁתֵּי הַמִּלְחָמוֹת יֵשׁ לְעַם יִשְׂרָאֵל שֶׁטַח גָּדוֹל בְּעֵבֶר הַיַּרְדֵּן הַמִּזְרָחִי: הַגִּלְעָד (אַרְצוֹ שֶׁל סִיחֹן), מִמִּזְרָח לַיַּרְדֵּן וּלְיָם הַמֶּלַח, וּהַבָּשָׁן (אַרְצוֹ שֶׁל עוֹג) – צָפוֹנָה מִמֶּנּוּ.
אֶת הָאָרֶץ הַזֹּאת נָתַן מֹשֶׁה לִשְׁנֵי שְׁבָטִים וָחֵצִי: רְאוּבֵן, גָּד וַחֲצִי שֵׁבֶט הַמְנַשֶּׁה.

גָּד – בָּאֶמְצַע, מֵחֶשְׁבּוֹן וְעַד נַחַל יַבֹּק.
חֲצִי שֵׁבֶט הַמְנַשֶּׁה – הַצְּפוֹנִי בְּיוֹתֵר, מִן הַיַּבֹּק וְעַד הַחֶרְמוֹן: ׳וְיֶתֶר הַגִּלְעָד וְכָל הַבָּשָׁן מַמְלֶכֶת עוֹג׳.

׳וְאֶת הָאָרֶץ הַזֹּאת יָרַשְׁנוּ בָּעֵת הַהִוא׳ – הָאֲמוּרָה לְמַעְלָה: מִנַּחַל אַרְנוֹן וְעַד הַר חֶרְמוֹן.

הָעִקָּר הוּא הַתְּמוּנָה: הַגִּלְעָד וְהַבָּשָׁן נִמְצָאִים מִמִּזְרָח לַיַּרְדֵּן, וּבָהֶם הִתְנַחֲלוּ שְׁנֵי שְׁבָטִים וָחֵצִי – רְאוּבֵן בַּדָּרוֹם, גָּד בָּאֶמְצַע וַחֲצִי מְנַשֶּׁה בַּצָּפוֹן.

לָמָּה חָשׁוּב שֶׁלְּכָל שֵׁבֶט תִּהְיֶה נַחֲלָה מְדֻיֶּקֶת, עִם גְּבוּלוֹת? כִּי ה׳ אוֹהֵב כָּל שֵׁבֶט וְכָל מִשְׁפָּחָה בְּעַם יִשְׂרָאֵל, וְרוֹצֶה שֶׁלְּכָל אֶחָד יִהְיֶה בַּיִת מִשֶּׁלּוֹ בָּאָרֶץ – מָקוֹם לִחְיוֹת בּוֹ חַיִּים שֶׁל תּוֹרָה.

שְׁנֵי הַפְּסוּקִים הָאַחֲרוֹנִים שֶׁל פָּרָשַׁת דְּבָרִים הֵם דִּבְרֵי חִזּוּק שֶׁמֹּשֶׁה אוֹמֵר לְיהוֹשֻׁעַ, הַמַּנְהִיג שֶׁיַּכְנִיס אֶת הָעָם לָאָרֶץ:

מִי הֵם ׳שְׁנֵי הַמְּלָכִים הָאֵלֶּה׳? סִיחֹן וְעוֹג, שְׁנֵי מַלְכֵי הָאֱמֹרִי הָאַדִּירִים.
מֹשֶׁה אוֹמֵר לִיהוֹשֻׁעַ: אַתָּה לֹא שָׁמַעְתָּ עַל זֶה מֵאֲחֵרִים – רָאִיתָ בְּעֵינֶיךָ אֵיךְ ה׳ הִפִּיל אוֹתָם. וּמִי שֶׁעָשָׂה זֹאת בְּעֵבֶר הַיַּרְדֵּן הַמִּזְרָחִי, יַעֲשֶׂה זֹאת גַּם בְּאֶרֶץ כְּנַעַן.

׳לֹא תִירָאוּם כִּי ה׳ אֱלֹהֵיכֶם הוּא הַנִּלְחָם לָכֶם׳.
הַנִּצָּחוֹן שֶׁל אֶתְמוֹל נוֹתֵן כֹּחַ לַמִּלְחָמָה שֶׁל מָחָר. יְהוֹשֻׁעַ וְכָל יִשְׂרָאֵל צְרִיכִים לִשְׁאֹב עִדּוּד וֶאֱמוּנָה מִמָּה שֶׁכְּבָר רָאוּ, וְלֹא לְפַחֵד מִמָּה שֶׁעוֹד לֹא קָרָה.

אָדָם שֶׁמְּפַחֵד מִמִּבְחָן, מִנְּאוּם אוֹ מִמַּשֶּׁהוּ חָדָשׁ – כְּדַאי לוֹ לְהִזָּכֵר בַּפְּעָמִים שֶׁה׳ כְּבָר עָזַר לוֹ.
׳עֵינֶיךָ הָרֹאֹת׳ – תִּזְכֹּר מָה כְּבָר עָבַרְתָּ בְּעֶזְרַת ה׳, וְתֵדַע שֶׁהוּא אִתְּךָ גַּם עַכְשָׁו.
💡 וּמָה הַפָּסוּק הַבָּא?
מִיָּד אַחֲרֵי כֵן מַתְחִילָה פָּרָשַׁת וָאֶתְחַנַּן: ׳וָאֶתְחַנַּן אֶל ה׳ בָּעֵת הַהִוא לֵאמֹר׳ – תְּפִלָּתוֹ שֶׁל מֹשֶׁה שֶׁיִּזְכֶּה לְהִכָּנֵס לָאָרֶץ.
רַשִׁ״י מְבָאֵר שֶׁ׳חִנּוּן׳ הוּא לְשׁוֹן מַתְּנַת חִנָּם: גַּם צַדִּיקִים כְּמוֹ מֹשֶׁה אֵינָם מְבַקְשִׁים בִּזְכוּת מַעֲשֵׂיהֶם, אֶלָּא כְּמַתָּנָה מֵה׳.
וְעוֹד מְבָאֵר רַשִׁ״י: ׳בָּעֵת הַהִוא׳ – לְאַחַר שֶׁכָּבַשׁ אֶת אֶרֶץ סִיחֹן וְעוֹג, חָשַׁב מֹשֶׁה שֶׁאוּלַי הֻתַּר הַנֶּדֶר.

׳כִּי ה׳ אֱלֹהֵיכֶם הוּא הַנִּלְחָם לָכֶם׳ – ה׳ לֹא רַק צוֹפֶה מִן הַצַּד, הוּא נִלְחָם בִּשְׁבִילֵנוּ. מִי שֶׁיּוֹדֵעַ שֶׁאוֹהֵב גָּדוֹל כָּזֶה עוֹמֵד לְצִדּוֹ – כְּבָר פָּחוֹת מְפַחֵד 💛

📜לַחֲזָרָה: מִסְפָּרִים, שֵׁמוֹת וְרַשִׁ״י בַּפֶּרֶק

שִׁשִּׁים עִיר בְּחֶבֶל אַרְגֹּב, מְבֻצָּרוֹת בְּחוֹמָה גְּבוֹהָה – וּמִלְּבַדָּן עָרֵי הַפְּרָזִי הַרְבֵּה מְאֹד.
מִטַּת הַבַּרְזֶל: תֵּשַׁע אַמּוֹת אֹרֶךְ וְאַרְבַּע אַמּוֹת רֹחַב.
אַרְבָּעָה שֵׁמוֹת לַחֶרְמוֹן: חֶרְמוֹן, שִׂרְיֹן (בְּפִי הַצִּידֹנִים), שְׂנִיר (בְּפִי הָאֱמֹרִי) וְשִׂיאֹן.
שְׁנֵי שְׁבָטִים וָחֵצִי בְּעֵבֶר הַיַּרְדֵּן הַמִּזְרָחִי, מִדָּרוֹם לְצָפוֹן: רְאוּבֵן, גָּד וַחֲצִי שֵׁבֶט הַמְנַשֶּׁה – בֵּין נַחַל אַרְנֹן, נַחַל יַבֹּק וְהַר הַחֶרְמוֹן.

׳אַל תִּירָא אֹתוֹ׳ – מֹשֶׁה חָשַׁשׁ מִזְּכוּת שֶׁעוֹג שִׁמֵּשׁ אֶת אַבְרָהָם.
׳מֵעָרֵי הַפְּרָזִי׳ – פְּרוּזוֹת וּפְתוּחוֹת בְּלֹא חוֹמָה.
׳מִיַּד׳ – מֵרְשׁוּת.
׳מִיֶּתֶר הָרְפָאִים׳ – עוֹג הוּא ׳הַפָּלִיט׳ שֶׁנִּמְלַט מִן הַמִּלְחָמָה.
׳בְּאַמַּת אִישׁ׳ – בְּאַמַּת עוֹג.
׳שְׂנִיר׳ – שֶׁלֶג.

׳לִפְנֵי אֲחֵיכֶם׳ – לְפִי שֶׁהָיוּ גִּבּוֹרִים, וְאוֹיְבִים נוֹפְלִים לִפְנֵיהֶם.

כְּשֶׁמִּסְתַּכְּלִים עַל כָּל הַפֶּרֶק יַחַד רוֹאִים שְׁלוֹשָׁה שְׁלַבִּים: קֹדֶם ה׳ נוֹתֵן נִצָּחוֹן, אַחַר כָּךְ מְחַלְּקִים אֶת הָאָרֶץ בְּצֶדֶק בֵּין הַשְּׁבָטִים, וּבַסּוֹף מְחַזְּקִים זֶה אֶת זֶה לִקְרַאת הַהֶמְשֵׁךְ.
וְזֶהוּ בְּדִיּוּק מָה שֶׁה׳ עוֹשֶׂה אִתָּנוּ: נוֹתֵן, מְסַדֵּר, וּמְחַזֵּק – כִּי הוּא אוֹהֵב אוֹתָנוּ וְרוֹצֶה שֶׁנֵּשֵׁב בְּשִׂמְחָה בָּאָרֶץ שֶׁלּוֹ.

💛 אַהֲבַת הַשֵּׁם
פָּרָשַׁת דְּבָרִים מִסְתַּיֶּמֶת בְּמִלִּים שֶׁל אַהֲבָה וּבִטָּחוֹן: ׳ה׳ אֱלֹהֵיכֶם הוּא הַנִּלְחָם לָכֶם׳. אֲנִי זוֹכֵר שֶׁאֲנִי שַׁיָּךְ לְעַם שֶׁה׳ בָּחַר בּוֹ, לְאֶרֶץ שֶׁכָּל הָעוֹלָם מִתְפָּאֵר בָּהּ, וּלְתוֹרָה שֶׁהִיא הַסֵּפֶר הֲכִי יָקָר בָּעוֹלָם 💛

✨ לְסִכּוּם

  • אַחֲרֵי הַנִּצָּחוֹן עַל סִיחֹן פָּנוּ בְּנֵי יִשְׂרָאֵל צָפוֹנָה (׳וַנַּעַל׳ – כָּל צַד צָפוֹן הוּא עֲלִיָּה), וְעוֹג מֶלֶךְ הַבָּשָׁן יָצָא לִקְרָאתָם לַמִּלְחָמָה בְּאֶדְרֶעִי. הַבָּשָׁן הוּא אֵזוֹר רָמַת הַגּוֹלָן שֶׁל הַיּוֹם.
  • ה׳ אָמַר ׳אַל תִּירָא אֹתוֹ׳, כִּי מֹשֶׁה חָשַׁשׁ מִזְּכוּת הַמִּצְוָה שֶׁל עוֹג – שֶׁבָּא כְּ׳פָלִיט׳ וְסִפֵּר לְאַבְרָהָם עַל לוֹט. פַּחַד רוּחָנִי, לֹא פַּחַד מִצָּבָא.
  • יִשְׂרָאֵל לָכְדוּ שִׁשִּׁים עִיר מְבֻצָּרוֹת בְּחֶבֶל אַרְגֹּב, וְעוֹד עָרֵי פְּרָזִי – פְּתוּחוֹת בְּלֹא חוֹמָה.
  • עוֹג נִשְׁאַר ׳מִיֶּתֶר הָרְפָאִים׳, וּמִטָּתוֹ – עֶרֶשׂ בַּרְזֶל, תֵּשַׁע אַמּוֹת עַל אַרְבַּע – מְלַמֶּדֶת כַּמָּה אַדִּיר הָיָה, וְכַמָּה גָּדוֹל הַנֵּס שֶׁנִּצְּחוּהוּ.
  • לְהַר הַחֶרְמוֹן אַרְבָּעָה שֵׁמוֹת – חֶרְמוֹן, שִׂרְיֹן, שְׂנִיר (שֶׁלֶג) וְשִׂיאֹן – לְהַגִּיד שֶׁבַח אֶרֶץ יִשְׂרָאֵל, שֶׁאַרְבַּע מַלְכֻיּוֹת הִתְפָּאֲרוּ בָּהּ.
  • מֹשֶׁה נָתַן אֶת הַגִּלְעָד וְהַבָּשָׁן לִרְאוּבֵן (דָּרוֹם), גָּד (אֶמְצַע) וַחֲצִי שֵׁבֶט הַמְנַשֶּׁה (צָפוֹן), וְצִוָּה אוֹתָם לַעֲבֹר חֲלוּצִים לִפְנֵי אֲחֵיהֶם וְלַחֲזֹר רַק אַחֲרֵי שֶׁכָּל הַשְּׁבָטִים יִנְחֲלוּ.
  • פָּרָשַׁת דְּבָרִים מִסְתַּיֶּמֶת בְּחִזּוּק לִיהוֹשֻׁעַ: ׳עֵינֶיךָ הָרֹאֹת׳ – מִי שֶׁזּוֹכֵר מָה ה׳ עָשָׂה, אֵינוֹ מְפַחֵד מִמָּה שֶׁעוֹד יָבוֹא, ׳כִּי ה׳ אֱלֹהֵיכֶם הוּא הַנִּלְחָם לָכֶם׳.

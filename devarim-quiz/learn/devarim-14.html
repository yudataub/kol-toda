בַּשִּׁעוּר הַזֶּה אֲנַחְנוּ לוֹמְדִים אֶת הַפְּסוּקִים הַמְּפֻרְסָמִים בְּיוֹתֵר בַּתּוֹרָה: פָּרָשַׁת שְׁמַע (דְּבָרִים ו׳, ד׳–ט׳). אֵין יְהוּדִי שֶׁאֵינוֹ מַכִּיר אוֹתָם, וְאֵין יְהוּדִי שֶׁאֵינָם מְרַגְּשִׁים אוֹתוֹ.
נִלְמַד פָּסוּק אַחַר פָּסוּק, וְנִמְצָא בָּהֶם שֵׁשׁ מִצְווֹת: יִחוּד ה׳, אַהֲבַת ה׳, תַּלְמוּד תּוֹרָה, קְרִיאַת שְׁמַע, תְּפִלִּין וּמְזוּזָה. נָבִין אֶת פֵּרוּשֵׁי רַשִׁ״י עַל ׳וְאָהַבְתָּ׳, עַל ׳וּבְכָל נַפְשְׁךָ׳ וְעַל ׳וּבְכָל מְאֹדֶךָ׳, וְנִשְׁמַע סִפּוּרִים עַל יְהוּדִים שֶׁאָמְרוּ ׳שְׁמַע יִשְׂרָאֵל׳ בָּרְגָעִים הַגְּדוֹלִים בְּיוֹתֵר.
וּבְעִקָּר נַרְגִּישׁ: הַפְּסוּקִים הָאֵלֶּה הֵם לֵב הָאֱמוּנָה שֶׁלָּנוּ, וְהַדֶּרֶךְ שֶׁבָּהּ אֲנַחְנוּ אוֹמְרִים לַה׳ פַּעֲמַיִם בְּכָל יוֹם: אֲנַחְנוּ שֶׁלְּךָ, וַאֲנַחְנוּ אוֹהֲבִים אוֹתְךָ 💛
בַּשִּׁעוּר: 👂 שְׁמַע יִשְׂרָאֵל – לֵב הָאֱמוּנָה 👑 קַבָּלַת עֹל מַלְכוּת שָׁמַיִם – מָתַי אוֹמְרִים ׳שְׁמַע יִשְׂרָאֵל׳ ❤️ וְאָהַבְתָּ – מֵאַהֲבָה וְלֹא מִיִּרְאָה 🕯️ בְּכָל נַפְשְׁךָ – מְסִירוּת נֶפֶשׁ 🪙 בְּכָל מְאֹדֶךָ – בְּמָמוֹן וּבְכָל מִדָּה וּמִדָּה 📖 וְהָיוּ הַדְּבָרִים הָאֵלֶּה – וְשִׁנַּנְתָּם לְבָנֶיךָ 🌙 בְּשָׁכְבְּךָ וּבְקוּמֶךָ – מִצְוַת קְרִיאַת שְׁמַע 📦 תְּפִלִּין וּמְזוּזָה – מֵהַגּוּף עַד שַׁעַר הָעִיר 🌍 ׳שְׁמַע יִשְׂרָאֵל׳ בְּרֶגַע הָאֱמֶת – סִפּוּרִים

👂שְׁמַע יִשְׂרָאֵל – לֵב הָאֱמוּנָה

📖 פֶּרֶק ו׳, פָּסוּק ד׳ ←
ד שְׁמַע יִשְׂרָאֵל ה׳ אֱלֹהֵינוּ ה׳ אֶחָד

מֹשֶׁה רַבֵּנוּ עוֹמֵד לִפְנֵי עַם יִשְׂרָאֵל, מְעַט לִפְנֵי הַכְּנִיסָה לָאָרֶץ, וְאוֹמֵר שִׁשָּׁה פְּסוּקִים קְצָרִים שֶׁהָפְכוּ לְלֵב הָאֱמוּנָה שֶׁל כָּל יְהוּדִי.
הַפָּסוּק הָרִאשׁוֹן פּוֹנֶה אֶל עַם יִשְׂרָאֵל – ׳שְׁמַע יִשְׂרָאֵל׳. מֹשֶׁה קוֹרֵא לָעָם כֻּלּוֹ בְּשֵׁם, כְּמוֹ אָב שֶׁקּוֹרֵא לְבָנָיו לִפְנֵי שֶׁהוּא אוֹמֵר לָהֶם אֶת הַדָּבָר הֶחָשׁוּב בְּיוֹתֵר.

מָה נֶאֱמַר בַּפָּסוּק? ׳ה׳ אֱלֹהֵינוּ׳ – ה׳ הוּא הָאֱלֹהִים שֶׁלָּנוּ, וַאֲנַחְנוּ שׁוֹמְעִים בְּקוֹלוֹ.
׳ה׳ אֶחָד׳ – אֵין עוֹד מִלְבַדּוֹ, לֹא בַּשָּׁמַיִם וְלֹא בָּאָרֶץ. בַּדּוֹר שֶׁנִּכְנַס לָאָרֶץ הָיוּ עַמִּים שֶׁהֶאֱמִינוּ בֵּאלִילִים רַבִּים, וְעַם יִשְׂרָאֵל מַכְרִיז: אֵין אֱלֹהִים אַחֵר.
זוֹ הַמִּצְוָה הָרִאשׁוֹנָה שֶׁבַּפָּרָשָׁה: לְהַאֲמִין בְּיִחוּד ה׳.

📜 רַשִׁ״י
׳ה׳ אֱלֹהֵינוּ ה׳ אֶחָד׳ – ה׳ שֶׁהוּא אֱלֹהֵינוּ עַתָּה וְלֹא אֱלֹהֵי הָאֻמּוֹת, הוּא עָתִיד לִהְיוֹת ה׳ אֶחָד, שֶׁנֶּאֱמַר: ׳כִּי אָז אֶהְפֹּךְ אֶל עַמִּים שָׂפָה בְרוּרָה לִקְרֹא כֻלָּם בְּשֵׁם ה׳׳, וְנֶאֱמַר: ׳בַּיּוֹם הַהוּא יִהְיֶה ה׳ אֶחָד וּשְׁמוֹ אֶחָד׳.

כְּלוֹמַר: הַיּוֹם רַק עַם יִשְׂרָאֵל מַכִּיר בַּה׳ וְקוֹרֵא לוֹ ׳אֱלֹהֵינוּ׳. אֲבָל יָבוֹא יוֹם, אוֹמְרִים הַנְּבִיאִים צְפַנְיָה וּזְכַרְיָה, שֶׁכָּל הָעַמִּים בָּעוֹלָם יַכִּירוּ בַּה׳ וְיִקְרְאוּ בִּשְׁמוֹ. כְּשֶׁאֲנַחְנוּ אוֹמְרִים ׳ה׳ אֶחָד׳ אֲנַחְנוּ גַּם מַאֲמִינִים וְגַם מְחַכִּים לַיּוֹם הַגָּדוֹל הַזֶּה.

💛 אַהֲבַת הַשֵּׁם
כְּשֶׁאֲנִי אוֹמֵר ׳שְׁמַע יִשְׂרָאֵל׳ אֲנִי לֹא לְבַד: כָּל עַם יִשְׂרָאֵל, בְּכָל הָעוֹלָם וּבְכָל הַדּוֹרוֹת, אוֹמֵר אֶת אוֹתָן הַמִּלִּים בְּדִיּוּק. וַאֲנִי יוֹדֵעַ שֶׁה׳ אֶחָד שׁוֹמֵעַ אֶת כֻּלָּנוּ, וְגַם אוֹתִי 💛

👑קַבָּלַת עֹל מַלְכוּת שָׁמַיִם – מָתַי אוֹמְרִים ׳שְׁמַע יִשְׂרָאֵל׳

כְּשֶׁאוֹמְרִים ׳שְׁמַע יִשְׂרָאֵל׳ אֲנַחְנוּ מְקַבְּלִים אֶת ה׳ כְּמֶלֶךְ עָלֵינוּ. לַדָּבָר הַזֶּה קוֹרְאִים ׳קַבָּלַת עֹל מַלְכוּת שָׁמַיִם׳: כְּמוֹ מִי שֶׁמַּכְרִיז שֶׁהוּא שֶׁל הַמֶּלֶךְ, וְשֶׁהוּא מוּכָן לַעֲשׂוֹת אֶת רְצוֹנוֹ.

מִן הַתּוֹרָה אוֹמְרִים אֶת פָּרָשַׁת שְׁמַע פַּעֲמַיִם בְּכָל יוֹם: בַּבֹּקֶר בִּתְפִלַּת שַׁחֲרִית, וּבָעֶרֶב בִּתְפִלַּת עַרְבִית. זוֹ מִצְוָה מִן הַתּוֹרָה, מִן הַמִּלִּים ׳בְּשָׁכְבְּךָ וּבְקוּמֶךָ׳.

💡 מִנְהָגִים שֶׁל עַם יִשְׂרָאֵל
מִלְּבַד הַמִּצְוָה שֶׁל בֹּקֶר וָעֶרֶב, נָהַג עַם יִשְׂרָאֵל לוֹמַר אֶת הַפָּסוּק גַּם בִּרְגָעִים מְיֻחָדִים:
בְּסוֹף יוֹם הַכִּפּוּרִים, בִּתְפִלַּת נְעִילָה, כְּשֶׁכָּל הַקָּהָל עוֹמֵד וְאוֹמֵר יַחַד ׳שְׁמַע יִשְׂרָאֵל׳.
בִּפְתִיחַת הַהֵיכָל (אֲרוֹן הַקֹּדֶשׁ) בְּשַׁבָּת וּבְחַג.
בַּלַּיְלָה שֶׁלִּפְנֵי בְּרִית מִילָה, לְיַד עֲרִיסַת הַתִּינוֹק.
לִפְנֵי הַשֵּׁנָה בַּלַּיְלָה – קְרִיאַת שְׁמַע שֶׁעַל הַמִּטָּה.
בִּשְׁעַת סַכָּנָה וְלִפְנֵי פְּטִירָה.
וּבַעֲצָרוֹת גְּדוֹלוֹת שֶׁל תְּפִלָּה, כְּמוֹ בַּכֹּתֶל הַמַּעֲרָבִי, שֶׁמִּסְתַּיְּמוֹת בְּקַבָּלַת עֹל מַלְכוּת שָׁמַיִם.

וְיֵשׁ עוֹד שְׁנֵי מְקוֹמוֹת שֶׁבָּהֶם פָּרָשַׁת שְׁמַע נִמְצֵאת אִתָּנוּ כָּל הַיּוֹם: בַּתְּפִלִּין שֶׁעַל הַזְּרוֹעַ וְעַל הָרֹאשׁ, וּבַמְּזוּזָה שֶׁעַל פֶּתַח הַבַּיִת. בִּשְׁנֵיהֶם כְּתוּבָה פָּרָשַׁת שְׁמַע עַל קְלָף.

🤔 לַמּוֹרֶה: מִצְוָה אוֹ מִנְהָג?
אֶפְשָׁר לְהַצִּיג לַתַּלְמִידִים אִיּוּרִים – זְרִיחָה, בֵּית כְּנֶסֶת בָּעֶרֶב, מַחֲזוֹר שֶׁל יוֹם כִּפּוּר, עֲרִיסַת תִּינוֹק, הֲנָחַת תְּפִלִּין, חֲזִית בַּיִת עִם מְזוּזָה, מִטָּה – וּלְבַקֵּשׁ לְמַיֵּן:
מִן הַתּוֹרָה: הַבֹּקֶר וְהָעֶרֶב (׳בְּשָׁכְבְּךָ וּבְקוּמֶךָ׳).
מִנְהָג יָפֶה שֶׁל יִשְׂרָאֵל: נְעִילָה, פְּתִיחַת הַהֵיכָל, לֵיל הַבְּרִית, לִפְנֵי הַשֵּׁנָה, בִּשְׁעַת סַכָּנָה.
וְהַתְּפִלִּין וְהַמְּזוּזָה – מִצְווֹת בִּפְנֵי עַצְמָן, שֶׁבְּתוֹכָן כְּתוּבָה הַפָּרָשָׁה.

וְעַכְשָׁיו הַפָּסוּק שֶׁהוּא הַשִּׂיא שֶׁל הַפֶּרֶק: ׳וְאָכַלְתָּ וְשָׂבַעְתָּ וּבֵרַכְתָּ אֶת ה׳ אֱלֹהֶיךָ עַל הָאָרֶץ הַטֹּבָה אֲשֶׁר נָתַן לָךְ׳.
בַּמִּדְבָּר אָכַלְנוּ מָן, וְיָדַעְנוּ בְּוַדַּאי שֶׁהוּא מֵה׳ – הֲרֵי הוּא יָרַד מִן הַשָּׁמַיִם! אֲבָל בָּאָרֶץ הָאֹכֶל צוֹמֵחַ מִן הָאֲדָמָה, וְאָדָם עָלוּל לַחְשֹׁב שֶׁהַכֹּל מֵעַצְמוֹ. לָכֵן צָרִיךְ לִזְכֹּר לְהוֹדוֹת.

מָה מִתְחַיֵּב מִכָּל הַשֶּׁפַע הַזֶּה? לְהוֹדוֹת לַה׳. מִן הַפָּסוּק הַזֶּה לָמְדוּ חֲזַ״ל אֶת חוֹבַת בִּרְכַּת הַמָּזוֹן מִן הַתּוֹרָה – לְבָרֵךְ אַחֲרֵי סְעוּדָה שֶׁל לֶחֶם.

עַל מַאֲכָל שֶׁנַּעֲשָׂה מֵחֲמֵשֶׁת מִינֵי דָּגָן, עַל יַיִן, וְעַל הַפֵּרוֹת שֶׁנִּשְׁתַּבְּחָה בָּהֶם אֶרֶץ יִשְׂרָאֵל – עֲנָבִים, תְּאֵנִים, רִמּוֹנִים, זֵיתִים וּתְמָרִים – מְבָרְכִים בְּרָכָה מְיֻחֶדֶת: בְּרָכָה ׳מֵעֵין שָׁלֹשׁ׳, שֶׁהִיא כְּמוֹ בִּרְכַּת הַמָּזוֹן בְּקִצּוּר.
גַּם בָּהּ אֲנַחְנוּ מַזְכִּירִים אֶת אֶרֶץ יִשְׂרָאֵל וּמוֹדִים עַל פֵּרוֹתֶיהָ.

אֶת אַרְבַּעַת הַפְּסוּקִים הָאֵלֶּה לוֹמְדִים בְּעַל פֶּה:
׳כִּי ה׳ אֱלֹהֶיךָ מְבִיאֲךָ אֶל אֶרֶץ טוֹבָה, אֶרֶץ נַחֲלֵי מָיִם עֲיָנֹת וּתְהֹמֹת יֹצְאִים בַּבִּקְעָה וּבָהָר.
אֶרֶץ חִטָּה וּשְׂעֹרָה וְגֶפֶן וּתְאֵנָה וְרִמּוֹן, אֶרֶץ זֵית שֶׁמֶן וּדְבָשׁ.
אֶרֶץ אֲשֶׁר לֹא בְמִסְכֵּנֻת תֹּאכַל בָּהּ לֶחֶם, לֹא תֶחְסַר כֹּל בָּהּ, אֶרֶץ אֲשֶׁר אֲבָנֶיהָ בַרְזֶל וּמֵהֲרָרֶיהָ תַּחְצֹב נְחֹשֶׁת.
וְאָכַלְתָּ וְשָׂבַעְתָּ וּבֵרַכְתָּ אֶת ה׳ אֱלֹהֶיךָ עַל הָאָרֶץ הַטֹּבָה אֲשֶׁר נָתַן לָךְ׳.

וּמִיָּד אַחַר כָּךְ בָּאָה אַזְהָרָה שֶׁתַּמְשִׁיךְ בַּשִּׁעוּר הַבָּא: ׳הִשָּׁמֶר לְךָ פֶּן תִּשְׁכַּח אֶת ה׳ אֱלֹהֶיךָ לְבִלְתִּי שְׁמֹר מִצְוֺתָיו׳. הַשֶּׁפַע הוּא מַתָּנָה גְּדוֹלָה – וְצָרִיךְ לְהִזָּהֵר שֶׁהוּא לֹא יַשְׁכִּיחַ מֵאִתָּנוּ מִי נָתַן אוֹתוֹ.

✨ לְסִכּוּם

  • ׳כָּל הַמִּצְוָה׳ – אִם הִתְחַלְתָּ בְּמִצְוָה, גְּמֹר אוֹתָהּ, שֶׁהִיא נִקְרֵאת עַל שֵׁם הַגּוֹמְרָהּ: מֹשֶׁה נָשָׂא אֶת עַצְמוֹת יוֹסֵף, וּבְנֵי יִשְׂרָאֵל קָבְרוּ אוֹתָן – וְהַמִּצְוָה נִקְרֵאת עַל שְׁמָם.
  • בַּמִּדְבָּר הָאֹכֶל בָּא בְּנִסִּים – מָן, בְּגָדִים שֶׁלֹּא בָּלוּ וְרַגְלַיִם שֶׁלֹּא נָפְחוּ; וּבָאָרֶץ הוּא בָּא בְּשֶׁפַע טִבְעִי. בִּשְׁנֵי הַמְּקוֹמוֹת – אוֹתוֹ ה׳ שֶׁדּוֹאֵג לָנוּ.
  • הַמַּסָּע בַּמִּדְבָּר הָיָה גַּם נֵס וְגַם נִסָּיוֹן: ׳לְמַעַן עַנֹּתְךָ לְנַסֹּתְךָ׳. לְפִי הָרַמְבַּ״ן, הַמָּן שֶׁיָּרַד לְיוֹם אֶחָד לִמֵּד אֶת הָעָם לִסְמֹךְ עַל ה׳ – וּמִמֶּנּוּ נוֹדַע שֶׁיִּשְׁמְרוּ מִצְווֹת לְעוֹלָם.
  • ׳לֹא עַל הַלֶּחֶם לְבַדּוֹ יִחְיֶה הָאָדָם, כִּי עַל כָּל מוֹצָא פִי ה׳׳ – הַמָּזוֹן וְהַבִּטָּחוֹן שֶׁל הָאָדָם בָּאִים מֵה׳.
  • ׳כַּאֲשֶׁר יְיַסֵּר אִישׁ אֶת בְּנוֹ׳ – הַיִּסּוּרִים בָּאִים מֵאַהֲבָה, כְּמוֹ חִנּוּךְ שֶׁל אַבָּא: ׳וְאֹהֲבוֹ שִׁחֲרוֹ מוּסָר׳.
  • אֶרֶץ יִשְׂרָאֵל נִשְׁתַּבְּחָה בְּמַיִם, בְּשִׁבְעַת הַמִּינִים, בְּפַרְנָסָה בְּכָבוֹד וּבְמַחְצָבִים – חֲמִשָּׁה מִינֵי טוֹב בְּחֶלְקָה אַחַת. וְעַל הַכֹּל: ׳וְאָכַלְתָּ וְשָׂבַעְתָּ וּבֵרַכְתָּ׳ – מְקוֹר בִּרְכַּת הַמָּזוֹן מִן הַתּוֹרָה.

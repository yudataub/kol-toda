הָאֲבָנִים, הַמִּזְבֵּחַ וְהַר עֵיבָל

הַחֹמֶר הַמָּלֵא: כּוֹתְבִים אֶת הַתּוֹרָה עַל אֲבָנִים, וְכָל הָעָם עוֹנֶה ׳אָמֵן׳ (פָּרָשַׁת כִּי תָבוֹא, פֶּרֶק כ״ז)
🎯 לַחִידוֹן🏠 כָּל הַחִידוֹנִים
עַם יִשְׂרָאֵל עוֹמֵד מַמָּשׁ לִפְנֵי הַכְּנִיסָה לְאֶרֶץ יִשְׂרָאֵל. עוֹד מְעַט יַעַבְרוּ אֶת הַיַּרְדֵּן – בְּלִי מֹשֶׁה רַבֵּנוּ, שֶׁיִּשָּׁאֵר מֵעֵבֶר לַיַּרְדֵּן.
וְאָז מְצַוֶּה מֹשֶׁה עַל שְׁנֵי מַעֲמָדִים גְּדוֹלִים: לְהָקִים אֲבָנִים גְּדוֹלוֹת וְלִכְתֹּב עֲלֵיהֶן אֶת דִּבְרֵי הַתּוֹרָה, וְלִבְנוֹת מִזְבֵּחַ בְּהַר עֵיבָל וְלִשְׂמֹחַ שָׁם לִפְנֵי ה׳. וְאַחַר כָּךְ – מַעֲמַד הַבְּרָכָה וְהַקְּלָלָה: שִׁשָּׁה שְׁבָטִים עַל הַר גְּרִזִים, שִׁשָּׁה עַל הַר עֵיבָל, וְכָל הָעָם עוֹנֶה ׳אָמֵן׳.
נִלְמַד מָה הָיָה כָּתוּב עַל הָאֲבָנִים, מָה זֶה ׳שִׂיד׳ וּמָה זֶה ׳בַּאֵר הֵיטֵב׳, אֵיפֹה נִמְצָאִים שְׁנֵי הֶהָרִים, מָה אָמְרוּ הַלְוִיִּם – וּמָה גִּלּוּ הַחוֹפְרִים בְּהַר עֵיבָל לִפְנֵי כְּאַרְבָּעִים שָׁנָה.
וּבְעִקָּר נִרְאֶה כַּמָּה ה׳ אוֹהֵב אוֹתָנוּ: הוּא נָתַן לָנוּ אֶרֶץ, וְנָתַן לָנוּ תּוֹרָה, וְרוֹצֶה שֶׁנִּזְכֹּר אֶת הַקֶּשֶׁר בֵּינֵיהֶן בְּשִׂמְחָה וְלֹא בְּפַחַד 💛
בַּשִּׁעוּר: 📢 שָׁמֹר אֶת כָּל הַמִּצְוָה – לָמָּה מֹשֶׁה חוֹזֵר וּמַזְהִיר? 🪨 אֲבָנִים גְּדוֹלוֹת, שִׂיד – וְכָל דִּבְרֵי הַתּוֹרָה 🌍 בַּאֵר הֵיטֵב – בְּשִׁבְעִים לָשׁוֹן ⛰️ הַר עֵיבָל וְהַר גְּרִזִים – בְּלֵב הָאָרֶץ 🔥 מִזְבֵּחַ אֲבָנִים שְׁלֵמוֹת – וְשָׂמַחְתָּ לִפְנֵי ה׳ 📅 הַיּוֹם הַזֶּה נִהְיֵיתָ לְעָם – בְּרִית מִתְחַדֶּשֶׁת 👥 שֵׁשׁ וְשֵׁשׁ – סֵדֶר הַמַּעֲמָד בִּשְׁנֵי הֶהָרִים ❓ אֵיפֹה הַבְּרָכוֹת? – הַקֻּשְׁיָה שֶׁל רַשִׁ״י ⭕ מַעְגְּלֵי הַחַיִּים – וּדְבָרִים שֶׁבַּסֵּתֶר

📢שָׁמֹר אֶת כָּל הַמִּצְוָה – לָמָּה מֹשֶׁה חוֹזֵר וּמַזְהִיר?

א וַיְצַו מֹשֶׁה וְזִקְנֵי יִשְׂרָאֵל אֶת הָעָם לֵאמֹר שָׁמֹר אֶת כָּל הַמִּצְוָה אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם הַיּוֹם

׳וַיְצַו מֹשֶׁה וְזִקְנֵי יִשְׂרָאֵל אֶת הָעָם לֵאמֹר: שָׁמֹר אֶת כָּל הַמִּצְוָה אֲשֶׁר אָנֹכִי מְצַוֶּה אֶתְכֶם הַיּוֹם׳.
מֹשֶׁה כְּבָר אָמַר אֶת הַדְּבָרִים הָאֵלֶּה פְּעָמִים רַבּוֹת בְּחֻמַּשׁ דְּבָרִים. אָז לָמָּה שׁוּב?

כִּי עַכְשָׁיו זֶה כְּבָר מַמָּשׁ לִפְנֵי הַכְּנִיסָה לָאָרֶץ. עוֹד מְעַט יַעַבְרוּ אֶת הַיַּרְדֵּן – וּמֹשֶׁה לֹא יִהְיֶה אִתָּם. שָׁם הֵם יְקַיְּמוּ אֶת הַמִּצְווֹת בְּלִי מֹשֶׁה, בְּלִי הַמַּנְהִיג שֶׁהוֹצִיא אוֹתָם מִמִּצְרַיִם וְלִמֵּד אוֹתָם תּוֹרָה אַרְבָּעִים שָׁנָה.

📜 רַשִׁ״י
׳שָׁמֹר אֶת כָּל הַמִּצְוָה׳ – לְשׁוֹן הֹוֶה.

כְּלוֹמַר: הַמִּלָּה ׳שָׁמֹר׳ אֵינָהּ רַק צִוּוּי לְפַעַם אַחַת, אֶלָּא דָּבָר שֶׁקּוֹרֶה כָּל הַזְּמַן, תָּמִיד. שְׁמִירַת הַמִּצְווֹת הִיא לֹא מְשִׂימָה שֶׁגּוֹמְרִים – הִיא דֶּרֶךְ חַיִּים שֶׁמִּתְחַדֶּשֶׁת בְּכָל יוֹם.

🤔 בַּמֶּה יִהְיֶה שׁוֹנֶה קִיּוּם הַמִּצְווֹת בָּאָרֶץ?
בַּמִּדְבָּר הַחַיִּים הָיוּ נִסִּיִּים: מָן מִן הַשָּׁמַיִם, בְּאֵר מַיִם, עַנְנֵי כָּבוֹד – וְלֹא הָיוּ שָׂדוֹת, בָּתִּים וּשְׁכֵנִים.
בָּאָרֶץ כָּל מִשְׁפָּחָה תְּקַבֵּל נַחֲלָה: בַּיִת, שָׂדֶה וְכֶרֶם. יִהְיוּ מִסְחָר, בָּתֵּי דִּין וּשְׁכֵנִים – וְיֵשׁ מִצְווֹת רַבּוֹת שֶׁתְּלוּיוֹת בָּאָרֶץ: תְּרוּמוֹת וּמַעַשְׂרוֹת, שְׁמִטָּה, בִּכּוּרִים, לֶקֶט שִׁכְחָה וּפֵאָה.
הַחֲשָׁשׁ שֶׁל מֹשֶׁה: שֶׁהָעָם יִהְיֶה עָסוּק בָּעֲבוֹדָה וּבָרְכוּשׁ וְיִשְׁכַּח אֶת נוֹתֵן הָאָרֶץ. הַחֲשָׁשׁ שֶׁל הָעָם: אֵיךְ נִסְתַּדֵּר בְּלִי מֹשֶׁה?

אָז מָה מֹשֶׁה עוֹשֶׂה כְּדֵי לְחַזֵּק אֶת הָעָם? הוּא לֹא מִסְתַּפֵּק בְּמִלִּים. הוּא מְצַוֶּה עַל מַעֲמָדִים גְּדוֹלִים וּמְרַגְּשִׁים שֶׁיֵּעָשׂוּ מִיָּד עִם הַכְּנִיסָה לָאָרֶץ: אֲבָנִים גְּדוֹלוֹת שֶׁכְּתוּבָה עֲלֵיהֶן הַתּוֹרָה, מִזְבֵּחַ, קָרְבָּנוֹת וְשִׂמְחָה, וּמַעֲמַד בְּרָכָה וּקְלָלָה. דָּבָר שֶׁרוֹאִים וְשׁוֹמְעִים וְחָוִים – נִשְׁאָר בַּלֵּב.

💛 אַהֲבַת הַשֵּׁם
מֹשֶׁה חוֹזֵר וּמַזְהִיר, כִּי הוּא אוֹהֵב אֶת עַם יִשְׂרָאֵל וְרוֹצֶה שֶׁיִּהְיֶה לָהֶם טוֹב בָּאָרֶץ. וְגַם אֲנִי, כְּשֶׁאַבָּא וְאִמָּא מַזְכִּירִים לִי שׁוּב וָשׁוּב דָּבָר חָשׁוּב – זֶה סִימָן שֶׁהֵם אוֹהֲבִים אוֹתִי. כָּךְ גַּם ה׳, שֶׁמַּזְכִּיר לָנוּ אֶת הַתּוֹרָה מֵאַהֲבָה 💛

🪨אֲבָנִים גְּדוֹלוֹת, שִׂיד – וְכָל דִּבְרֵי הַתּוֹרָה

📖 פֶּרֶק כ״ז, פְּסוּקִים ב׳–ג׳ ←
ב וְהָיָה בַּיּוֹם אֲשֶׁר תַּעַבְרוּ אֶת הַיַּרְדֵּן אֶל הָאָרֶץ אֲשֶׁר ה׳ אֱלֹהֶיךָ נֹתֵן לָךְ וַהֲקֵמֹתָ לְךָ אֲבָנִים גְּדֹלוֹת וְשַׂדְתָּ אֹתָם בַּשִּׂיד ג וְכָתַבְתָּ עֲלֵיהֶן אֶת כָּל דִּבְרֵי הַתּוֹרָה הַזֹּאת בְּעָבְרֶךָ לְמַעַן אֲשֶׁר תָּבֹא אֶל הָאָרֶץ אֲ‍שֶׁר ה׳ אֱלֹהֶיךָ נֹתֵן לְךָ אֶרֶץ זָבַת חָלָב וּדְבַשׁ כַּאֲשֶׁר דִּבֶּר ה׳ אֱלֹהֵי אֲבֹתֶיךָ לָךְ

׳וְהָיָה בַּיּוֹם אֲשֶׁר תַּעַבְרוּ אֶת הַיַּרְדֵּן אֶל הָאָרֶץ אֲשֶׁר ה׳ אֱלֹהֶיךָ נֹתֵן לָךְ, וַהֲקֵמֹתָ לְךָ אֲבָנִים גְּדֹלוֹת וְשַׂדְתָּ אֹתָם בַּשִּׂיד׳.
מִיָּד בַּיּוֹם שֶׁעוֹבְרִים אֶת הַיַּרְדֵּן – לֹא מְחַכִּים, לֹא דּוֹחִים – מְקִימִים אֲבָנִים גְּדוֹלוֹת.

שִׂיד הוּא חֹמֶר לָבָן שֶׁמּוֹרְחִים עַל אֶבֶן אוֹ עַל קִיר (כְּמוֹ ׳סִיּוּד׳ שֶׁל קִירוֹת הַיּוֹם). הוּא עוֹשֶׂה מִשְׁטָח לָבָן וְחָלָק, נוֹחַ לִכְתִיבָה, וְהוּא גַּם מְחַזֵּק וְשׁוֹמֵר עַל הָאֶבֶן מִפְּנֵי הַגֶּשֶׁם וְהָרוּחַ.
וְיֵשׁ בָּזֶה גַּם סֵמֶל: כְּמוֹ שֶׁהַשִּׂיד שׁוֹמֵר עַל הָאֲבָנִים, כָּךְ אֲנַחְנוּ שׁוֹמְרִים עַל דִּבְרֵי הַתּוֹרָה הַכְּתוּבִים עֲלֵיהֶן.

׳וְכָתַבְתָּ עֲלֵיהֶן אֶת כָּל דִּבְרֵי הַתּוֹרָה הַזֹּאת, בְּעָבְרֶךָ, לְמַעַן אֲשֶׁר תָּבֹא אֶל הָאָרֶץ... אֶרֶץ זָבַת חָלָב וּדְבַשׁ׳.
שִׂימוּ לֵב אֵיךְ הַפָּסוּק מְחַבֵּר שְׁנֵי דְּבָרִים: כּוֹתְבִים אֶת הַתּוֹרָה בִּשְׁעַת הַכְּנִיסָה לָאָרֶץ. לָמָּה? כְּדֵי לְלַמֵּד שֶׁהָאָרֶץ וְהַתּוֹרָה קְשׁוּרוֹת זוֹ בָּזוֹ: ה׳ נוֹתֵן לָנוּ אֶרֶץ טוֹבָה, וַאֲנַחְנוּ חַיִּים בָּהּ לְפִי תּוֹרָתוֹ.

💡 לָמָּה דַּוְקָא עַל אֲבָנִים גְּדוֹלוֹת?
אֶבֶן גְּדוֹלָה אִי אֶפְשָׁר לְהַחְבִּיא בַּכִּיס. הִיא עוֹמֶדֶת בַּחוּץ, רוֹאִים אוֹתָהּ מֵרָחוֹק, וְהִיא נִשְׁאֶרֶת שָׁם שָׁנִים רַבּוֹת.
וְעוֹד: דָּבָר כָּתוּב קַל יוֹתֵר לִזְכֹּר מִדָּבָר שֶׁרַק אָמְרוּ. הַכְּתִיבָה הוֹפֶכֶת אֶת הַהִתְחַיְּבוּת לְמַשֶּׁהוּ שֶׁאֶפְשָׁר לִרְאוֹת בָּעֵינַיִם – וְכָל מִי שֶׁעוֹבֵר שָׁם נִזְכָּר בָּהּ מֵחָדָשׁ.
💡 מָה בְּדִיּוּק נִכְתַּב עַל הָאֲבָנִים?
הַמְּפָרְשִׁים הִצִּיעוּ אֶפְשָׁרֻיּוֹת שׁוֹנוֹת: יֵשׁ אוֹמְרִים שֶׁנִּכְתְּבוּ כָּל הַמִּצְווֹת, יֵשׁ אוֹמְרִים הַתּוֹרָה כֻּלָּהּ, וְיֵשׁ אוֹמְרִים שֶׁנִּכְתְּבוּ עֲשֶׂרֶת הַדִּבְּרוֹת.
וְהָעִקָּר אֵינוֹ בַּפְּרָטִים: הָעִקָּר הוּא הַמַּעֲשֶׂה עַצְמוֹ – שֶׁדִּבְרֵי ה׳ מֻצָּגִים לְעֵינֵי כֻּלָּם, וּמַזְכִּירִים לָעָם אֶת הַקֶּשֶׁר שֶׁלּוֹ לַה׳.

כְּלוֹמַר: הָאֲמִירָה הָאַחֲרוֹנָה אֵינָהּ עוֹד פְּרָט, אֶלָּא קַבָּלָה עַל כָּל הַתּוֹרָה כֻּלָּהּ. וְכָךְ מִתְחַבֵּר הַכֹּל: הָאֲבָנִים שֶׁעֲלֵיהֶן כְּתוּבָה הַתּוֹרָה, וְהַמַּעֲמָד שֶׁבּוֹ הָעָם עוֹנֶה אָמֵן – שְׁנֵיהֶם מַטְרָתָם אַחַת: לְחַזֵּק אֶת הַקֶּשֶׁר בֵּין עַם יִשְׂרָאֵל לַה׳ בְּאַרְצוֹ.

💛 אַהֲבַת הַשֵּׁם
ה׳ מְצַוֶּה אוֹתָנוּ לִשְׁמֹר עַל הַחַלָּשׁ, עַל הַגֵּר, עַל הַיָּתוֹם וְעַל הָאַלְמָנָה, וְלֹא לִפְגֹּעַ בְּאִישׁ גַּם בַּסֵּתֶר – כִּי הוּא אוֹהֵב כָּל אֶחָד מִבָּנָיו. כְּשֶׁאֲנִי נִזְהָר בִּכְבוֹד חֲבֵרִי וּבִכְבוֹד הוֹרַי, אֲנִי עוֹשֶׂה נַחַת רוּחַ לְאָבִי שֶׁבַּשָּׁמַיִם 💛

✨ לְסִכּוּם

  • לִפְנֵי הַכְּנִיסָה לָאָרֶץ מֹשֶׁה מַזְהִיר שׁוּב: ׳שָׁמֹר אֶת כָּל הַמִּצְוָה׳ – כִּי בָּאָרֶץ יְקַיְּמוּ אֶת הַמִּצְווֹת בְּלִי מֹשֶׁה, וְיִהְיוּ שָׁם שָׂדוֹת, בָּתִּים וּמִצְווֹת חֲדָשׁוֹת הַתְּלוּיוֹת בָּאָרֶץ.
  • מִיָּד בְּמַעֲבַר הַיַּרְדֵּן מְקִימִים אֲבָנִים גְּדוֹלוֹת, מְסַיְּדִים אוֹתָן בְּשִׂיד, וְכוֹתְבִים עֲלֵיהֶן אֶת דִּבְרֵי הַתּוֹרָה – כְּדֵי שֶׁכֻּלָּם יִרְאוּ מֵרָחוֹק שֶׁהָאָרֶץ וְהַתּוֹרָה קְשׁוּרוֹת זוֹ בָּזוֹ.
  • ׳בַּאֵר הֵיטֵב׳ – לְפִי רַשִׁ״י בְּשִׁבְעִים לָשׁוֹן, כְּדֵי שֶׁכָּל הָעַמִּים יוּכְלוּ לִקְרֹא.
  • בְּהַר עֵיבָל, מִצְּפוֹן לִשְׁכֶם וּמוּל הַר גְּרִזִים, בּוֹנִים גַּם מִזְבֵּחַ מֵאֲבָנִים שְׁלֵמוֹת שֶׁלֹּא הוּנַף עֲלֵיהֶן בַּרְזֶל, מַקְרִיבִים עוֹלוֹת וּשְׁלָמִים – וְשָׂמְחִים לִפְנֵי ה׳. יְהוֹשֻׁעַ קִיֵּם זֹאת, וּבַחֲפִירוֹת בְּהַר עֵיבָל נִמְצָא ׳מִזְבַּח יְהוֹשֻׁעַ׳.
  • ׳הַיּוֹם הַזֶּה נִהְיֵיתָ לְעָם׳ – רַשִׁ״י: בְּכָל יוֹם יִהְיוּ בְּעֵינֶיךָ כְּאִלּוּ הַיּוֹם בָּאתָ עִמּוֹ בִּבְרִית.
  • בַּמַּעֲמָד: שִׁשָּׁה שְׁבָטִים עַל הַר גְּרִזִים לִבְרָכָה, שִׁשָּׁה עַל הַר עֵיבָל עַל הַקְּלָלָה, וְהַכֹּהֲנִים וְהַלְוִיִּם וְהָאָרוֹן לְמַטָּה בָּאֶמְצַע. הַלְוִיִּם קוֹרְאִים בְּקוֹל רָם, וְכָל הָעָם עוֹנֶה ׳אָמֵן׳ – אֱמֶת, אֲנִי מְקַבֵּל עָלַי.
  • בַּפְּסוּקִים כְּתוּבוֹת רַק הַקְּלָלוֹת, וְרַשִׁ״י מְגַלֶּה שֶׁהַבְּרָכוֹת נֶאֶמְרוּ קֹדֶם בְּאוֹתָהּ לָשׁוֹן בְּדֶרֶךְ חִיּוּב. ה׳ אֵינוֹ רוֹצֶה בַּקְּלָלָה – הוּא פּוֹתֵחַ תָּמִיד בִּבְרָכָה, מֵאַהֲבָה.

׳כִּי תָבֹא בְּכֶרֶם רֵעֶךָ׳ – בְּפוֹעֵל הַכָּתוּב מְדַבֵּר.

כְּלוֹמַר: הַתּוֹרָה מְדַבֶּרֶת עַל פּוֹעֵל שֶׁבַּעַל הַבַּיִת שָׂכַר אוֹתוֹ כְּדֵי לִבְצֹר אֶת הָעֲנָבִים. הוּא נִכְנָס לַכֶּרֶם בִּרְשׁוּת הַבְּעָלִים וְעוֹבֵד שָׁם, וְלֹא סְתָם עוֹבֵר אֹרַח שֶׁנִּכְנַס לִקְטֹף.
פּוֹעֵל בַּכֶּרֶם מַתְחִיל אֶת יוֹמוֹ מֻקְדָּם מְאוֹד, לִפְנֵי שֶׁהַשֶּׁמֶשׁ חַמָּה, וְעוֹבֵד שָׁעוֹת אֲרֻכּוֹת בֵּין הַגְּפָנִים. יָדָיו מְלֵאוֹת אֶשְׁכּוֹלוֹת נִפְלָאִים כָּל הַיּוֹם – בָּרוּר שֶׁהוּא רוֹצֶה לִטְעֹם!

וְכָאן מִתְגַּלָּה אַהֲבַת ה׳: ה׳ רוֹאֶה אֶת הַפּוֹעֵל הֶעָיֵף וְהֶרָעֵב, וְנוֹתֵן לוֹ רְשׁוּת מִן הַתּוֹרָה לֶאֱכֹל מִן הָעֲנָבִים שֶׁבְּיָדָיו. הַתּוֹרָה דּוֹאֶגֶת גַּם לַקָּטָן וְגַם לָעוֹבֵד, וְלֹא רַק לְבַעַל הָרְכוּשׁ.

📜׳מִפִּי הַשְּׁמוּעָה לָמְדוּ׳ – תּוֹרָה שֶׁבְּעַל פֶּה

הָרַמְבַּ״ם כּוֹתֵב בְּהִלְכוֹת שְׂכִירוּת: ׳מִפִּי הַשְּׁמוּעָה לָמְדוּ שֶׁאֵין הַכָּתוּב מְדַבֵּר אֶלָּא בְּשָׂכִיר. וְכִי אִלּוּ לֹא שְׂכָרוֹ, מִי הִתִּיר לוֹ שֶׁיָּבוֹא בְּכֶרֶם רֵעֵהוּ אוֹ בְּקָמָה שֶׁלּוֹ שֶׁלֹּא מִדַּעְתּוֹ?׳

מָה זוֹ ׳שְׁמוּעָה׳? אֵלֶּה דִּבְרֵי הַחֲכָמִים שֶׁנִּמְסְרוּ מֵחָכָם לְחָכָם, מִדּוֹר לְדוֹר – הַתּוֹרָה שֶׁבְּעַל פֶּה.
מִכָּאן לִמּוּד גָּדוֹל: אִי אֶפְשָׁר לִקְרֹא אֶת הַתּוֹרָה לְבַד, בְּלִי הַמָּסֹרֶת שֶׁל חֲכָמֵינוּ. בְּלִי הַתּוֹרָה שֶׁבְּעַל פֶּה הָיִינוּ טוֹעִים וְחוֹשְׁבִים שֶׁמֻּתָּר לְהִכָּנֵס לַכֶּרֶם שֶׁל כָּל אֶחָד.

וְכָךְ מִתְבָּרֵר הַפֵּרוּשׁ הַנָּכוֹן: ׳כִּי תָבֹא׳ – כְּשֶׁאַתָּה בָּא בִּרְשׁוּת הַבְּעָלִים, כִּי הֵם שָׂכְרוּ אוֹתְךָ לַעֲבוֹדָה. אָז, וְרַק אָז, מֻתָּר לְךָ לֶאֱכֹל – וּבַמִּדָּה שֶׁהַתּוֹרָה קָבְעָה.

ה׳ נָתַן לָנוּ תּוֹרָה שֶׁבִּכְתָב וְתוֹרָה שֶׁבְּעַל פֶּה יַחַד. הַתּוֹרָה שֶׁבִּכְתָב הִיא כְּמוֹ רָאשֵׁי פְּרָקִים, וְהַתּוֹרָה שֶׁבְּעַל פֶּה מְבָאֶרֶת אוֹתָם.
זוֹ הַסִּבָּה שֶׁאֲנַחְנוּ לוֹמְדִים תּוֹרָה אֵצֶל רַבָּנִים וּמוֹרִים, וְלֹא רַק קוֹרְאִים לְבַד.

אוֹכְלִים רַק מָה שֶׁלּוֹקְחִים בַּיָּד, עַכְשָׁו – כְּדֵי שְׂבִיעַת הַנֶּפֶשׁ, וְלֹא יוֹתֵר.

⚖️סֵפֶר הַחִנּוּךְ: שְׁתֵּי מִצְווֹת – עַל בַּעַל הַבַּיִת וְעַל הַפּוֹעֵל

סֵפֶר הַחִנּוּךְ – חִבּוּר שֶׁכָּתַב חָכָם בִּסְפָרַד, וּבוֹ סָדַר אֶת כָּל תַּרְיַ״ג הַמִּצְווֹת לְפִי פָּרָשׁוֹת הַשָּׁבוּעַ – מוֹנֶה כָּאן שְׁתֵּי מִצְווֹת נִפְרָדוֹת:

מִצְוָה שְׁנִיָּה – עַל הַפּוֹעֵל: שֶׁלֹּא יִקַּח מִן הָאֹכֶל יוֹתֵר מִכְּדֵי אֲכִילָתוֹ.

הוּא יֹאמַר לְעַצְמוֹ: לֹא נִכְנַסְתִּי לַמְּקָרֵר שֶׁלּוֹ – זֶה רַק שָׂדֶה! הַתּוֹרָה אוֹמֶרֶת: גַּם בַּשָּׂדֶה – בְּיֹשֶׁר.

💡 כְּלָל גָּדוֹל
בְּכָל מָקוֹם שֶׁהַכִּשָּׁלוֹן מָצוּי – שֶׁאֲנָשִׁים נוֹטִים לְהַתִּיר לְעַצְמָם – הַתּוֹרָה מוֹסִיפָה אַזְהָרָה מְיֻחֶדֶת.
זֶה לֹא כְּדֵי לְהַכְבִּיד עָלֵינוּ, אֶלָּא כְּדֵי לִשְׁמֹר עָלֵינוּ מִפְּנֵי הַנְּטִיָּה הַטִּבְעִית שֶׁלָּנוּ.

שְׁתֵּי הַמִּצְווֹת יַחַד מְצַיְּרוֹת תְּמוּנָה שְׁלֵמָה שֶׁל אַהֲבָה: ה׳ מְבַקֵּשׁ מִבַּעַל הַבַּיִת לְהַשְׁפִּיעַ בְּנֶדֶב לֵב, וּמְבַקֵּשׁ מֵהַפּוֹעֵל לְקַבֵּל בְּעָנְוָה. כָּךְ שְׁנֵיהֶם נִשְׁאָרִים חֲבֵרִים, וּבִרְכַּת ה׳ שׁוֹרָה עֲלֵיהֶם.

🌾קָלִי בַּשָּׂדֶה שֶׁל בֹּעַז – נֶפֶשׁ יָפָה בְּפֹעַל

אֶת הַמְּלִילוֹת אֶפְשָׁר לֶאֱכֹל חַיּוֹת, אֲבָל טוֹב יוֹתֵר לֶאֱכֹל אוֹתָן קְלוּיוֹת – וְזֶה נִקְרָא ׳קָלִי׳: גַּרְגִּירֵי חִטָּה קְלוּיִים בָּאֵשׁ.

📚 חִידָה: מִי נָתַן ׳קָלִי׳?
בִּמְגִלַּת רוּת מְסֻפָּר עַל בֹּעַז, בַּעַל הַשָּׂדֶה מִבֵּית לֶחֶם, שֶׁהִזְמִין אֶת רוּת לֶאֱכֹל עִם הַקּוֹצְרִים: ׳וַיִּצְבָּט לָהּ קָלִי׳ – הוּא הוֹשִׁיט לָהּ בְּיָדוֹ חִטָּה קְלוּיָה.
רוּת לֹא הָיְתָה אֲפִלּוּ פּוֹעֶלֶת שֶׁלּוֹ – הִיא בָּאָה לְלַקֵּט – וּבֹעַז דָּאַג שֶׁתִּשְׂבַּע.

זוֹ בְּדִיּוּק הַ׳נֶפֶשׁ יָפָה׳ שֶׁעָלֶיהָ מְדַבֵּר סֵפֶר הַחִנּוּךְ: בַּעַל בַּיִת שֶׁדּוֹאֵג שֶׁכָּל מִי שֶׁנִּמְצָא בַּשָּׂדֶה שֶׁלּוֹ יִשְׂבַּע וְיַרְגִּישׁ טוֹב.
וְרָאוּ מָה יָצָא מִזֶּה: מִבֹּעַז וְרוּת נוֹלְדָה שׁוֹשֶׁלֶת דָּוִד הַמֶּלֶךְ. הַנְּדִיבוּת מְבִיאָה בְּרָכָה, כְּמוֹ שֶׁהִבְטִיחָה הַתּוֹרָה.

וְשִׂימוּ לֵב: כָּל הַסִּפּוּר הַזֶּה קָרָה בְּאֶרֶץ יִשְׂרָאֵל, בַּשָּׂדוֹת שֶׁל בֵּית לֶחֶם. הַמִּצְווֹת הָאֵלֶּה – כֶּרֶם, קָמָה, פּוֹעֲלִים וְקוֹצְרִים – חַיּוֹת וְנוֹשְׁמוֹת בָּאָרֶץ שֶׁה׳ נָתַן לָנוּ.

💛 אַהֲבַת הַשֵּׁם
אֲנִי אוֹהֵב לַחְשֹׁב שֶׁבְּכָל שָׂדֶה וּבְכָל כֶּרֶם בְּאֶרֶץ יִשְׂרָאֵל יֵשׁ מִצְווֹת מִן הַתּוֹרָה. הָאָרֶץ הַזּוֹ הִיא הַמָּקוֹם שֶׁבּוֹ אֲנִי יָכוֹל לְהַרְאוֹת לַה׳ אֶת אַהֲבָתִי – דַּוְקָא בִּדְבָרִים פְּשׁוּטִים כְּמוֹ עֵנָב וְגַרְגִּיר חִטָּה 💛

📏׳שֵׂכֶל יָשָׁר׳ – לִהְיוֹת הָגוּן בְּכָל מָקוֹם

שְׁתֵּי הַמִּצְווֹת שֶׁלָּמַדְנוּ מְלַמְּדוֹת אוֹתָנוּ עִקָּרוֹן אֶחָד גָּדוֹל: לְהִתְנַהֵג בְּמִדָּה.
בַּעַל הַבַּיִת נוֹתֵן בִּנְדִיבוּת וְלֹא מְדַקְדֵּק, וְהַפּוֹעֵל מְקַבֵּל וְלֹא מְנַצֵּל וְלֹא מַגְזִים. זֶה מָה שֶׁנִּקְרָא ׳שֵׂכֶל יָשָׁר׳ – לִהְיוֹת יָשָׁר, כְּלוֹמַר הָגוּן, לִנְהֹג בְּהִגָּיוֹן וּבַפְּרוֹפּוֹרְצְיָה הַנְּכוֹנָה.

וְהַכְּלָל הַזֶּה נָכוֹן בְּכָל מָקוֹם: כְּשֶׁמְּקַבְּלִים רְשׁוּת לְהִשְׁתַּמֵּשׁ בְּמַשֶּׁהוּ – בְּמָלוֹן, בַּאֲרוּחָה, אֵצֶל חֲבֵרִים, בְּצַעֲצוּעַ שֶׁהִשְׁאִילוּ לָנוּ – לֹא מִתְנַפְּלִים, אֶלָּא מִתְנַהֲגִים בְּאִפּוּק.

וְזִכְרוּ: הַדִּין חָל עַל כָּל יְבוּל, וְלֹא רַק עַל עֲנָבִים וְחִטָּה. וְהָעִקָּרוֹן שֶׁל הֲגִינוּת חָל עַל כָּל מַעֲבִיד וְכָל עוֹבֵד – גַּם הַיּוֹם, גַּם בְּמִשְׂרָד וּבַחֲנוּת, וְלֹא רַק בַּכֶּרֶם.

🤔 לְדִיּוּן בַּכִּתָּה
אֶפְשָׁר לְהַרְחִיב וּלְדַבֵּר עַל מִקְרִים מֵהַחַיִּים: נַעַר שֶׁעוֹבֵד בְּחֻפְשַׁת הַקַּיִץ – מָה מַגִּיעַ לוֹ וּמָה לֹא; שִׁמּוּשׁ מֻפְרָז בִּרְכוּשׁ שֶׁל הַמַּעֲבִיד; הַשְׁאָלַת חֵפֶץ מֵחָבֵר.
הַשְּׁאֵלָה שֶׁמְּלַוָּה אוֹתָנוּ: הַאִם אֲנִי נוֹהֵג בִּפְרוֹפּוֹרְצְיָה?

  • בַּשָּׂדֶה: ׳וְקָטַפְתָּ מְלִילֹת בְּיָדֶךָ׳ – גַּרְגִּירֵי חִטָּה שֶׁמּוֹלְלִים בַּיָּד; ׳וְחֶרְמֵשׁ לֹא תָנִיף׳ – אָסוּר לִקְצֹר לְעַצְמוֹ בִּכְלִי. וְהוּא הַדִּין בְּכָל יְבוּל.
  • רַשִׁ״י: ׳כְּנַפְשְׁךָ׳ – כַּמָּה שֶׁתִּרְצֶה; ׳שָׂבְעֶךָ׳ – וְלֹא אֲכִילָה גַּסָּה. וְרַשִׁ״י מוֹסִיף: הַהֶתֵּר רַק בִּשְׁעַת הַבָּצִיר, וְלֹא לְמִי שֶׁשְּׂכָרוֹ לַעֲדֹר וּלְקַשְׁקֵשׁ.
  • הָרַמְבַּ״ם: ׳מִפִּי הַשְּׁמוּעָה לָמְדוּ שֶׁאֵין הַכָּתוּב מְדַבֵּר אֶלָּא בְּשָׂכִיר׳ – בְּלִי הַתּוֹרָה שֶׁבְּעַל פֶּה אִי אֶפְשָׁר לְהָבִין אֶת הַפָּסוּק. וְעוֹד: ׳מְלַמְּדִין אֶת הָאָדָם שֶׁלֹּא יִהְיֶה רַעַבְתָן׳.
  • סֵפֶר הַחִנּוּךְ: שְׁתֵּי מִצְווֹת – עַל בַּעַל הַבַּיִת לָתֵת בְּ׳נֶפֶשׁ יָפָה וְרָצוֹן טוֹב׳, וְעַל הַפּוֹעֵל שֶׁלֹּא יִקַּח יוֹתֵר מִכְּדֵי אֲכִילָתוֹ – כִּי גַּם בַּשָּׂדֶה צְרִיכִים יֹשֶׁר.
  • הַמַּסָּר הַגָּדוֹל: הֲגִינוּת, יֹשֶׁר וְאִפּוּק. ה׳ נוֹתֵן בְּיָד רְחָבָה וּמְחַנֵּךְ אוֹתָנוּ לִהְיוֹת אֲנָשִׁים טוֹבִים – וְזוֹ אַהֲבָתוֹ אֵלֵינוּ.

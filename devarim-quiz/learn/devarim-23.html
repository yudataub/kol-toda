׳וְעַתָּה יִשְׂרָאֵל׳ – אַף עַל פִּי שֶׁעֲשִׂיתֶם כָּל זֹאת, עוֹדֶנּוּ רַחֲמָיו וְחִבָּתוֹ עֲלֵיכֶם.
׳כִּי אִם לְיִרְאָה׳ – רַבּוֹתֵינוּ דָּרְשׁוּ מִכָּאן: הַכֹּל בִּידֵי שָׁמַיִם חוּץ מִיִּרְאַת שָׁמַיִם.

רַשִׁ״י מְגַלֶּה כָּאן אַהֲבָה עֲצוּמָה: גַּם אַחֲרֵי חֵטְא הָעֵגֶל, ה׳ עֲדַיִן אוֹהֵב אֶת יִשְׂרָאֵל וּמְבַקֵּשׁ מֵהֶם רַק שֶׁיִּירְאוּ אוֹתוֹ וְיֹאהֲבוּ אוֹתוֹ.
וְלָמָּה דַּוְקָא יִרְאַת שָׁמַיִם? כִּי אֶת זֶה רַק אֲנַחְנוּ יְכוֹלִים לָתֵת לוֹ – זוֹ הַבְּחִירָה שֶׁלָּנוּ, הַמַּתָּנָה שֶׁהַלֵּב שֶׁלָּנוּ נוֹתֵן לַה׳. וְאַף הִיא ׳לְטוֹב לָךְ׳, כְּדֵי שֶׁנְּקַבֵּל שָׂכָר.

וְלָמַדְנוּ גַּם עַל אַהֲבַת הַגֵּר: ׳עֹשֶׂה מִשְׁפַּט יָתוֹם וְאַלְמָנָה וְאֹהֵב גֵּר לָתֵת לוֹ לֶחֶם וְשִׂמְלָה׳, ׳וַאֲהַבְתֶּם אֶת הַגֵּר כִּי גֵרִים הֱיִיתֶם בְּאֶרֶץ מִצְרָיִם׳.

📜 רַשִׁ״י
׳עֹשֶׂה מִשְׁפַּט יָתוֹם וְאַלְמָנָה׳ – הֲרֵי גְּבוּרָה, וְאֵצֶל גְּבוּרָתוֹ אַתָּה מוֹצֵא עַנְוְתָנוּתוֹ.
׳כִּי גֵרִים הֱיִיתֶם׳ – מוּם שֶׁבְּךָ אַל תֹּאמַר לַחֲבֵרְךָ.

בְּפֶרֶק י״א חָזַרְנוּ לְאֶרֶץ יִשְׂרָאֵל, וְהַפַּעַם בְּהַשְׁוָאָה לְאֶרֶץ מִצְרַיִם: בְּמִצְרַיִם הָיוּ צְרִיכִים לְהָבִיא מַיִם מִן הַנִּילוּס בָּרֶגֶל וְלַעֲמֹל הַרְבֵּה, וְהַשָּׂדֶה הַנָּמוּךְ שׁוֹתֶה וְהַגָּבוֹהַּ לֹא.
אֲבָל אֶרֶץ יִשְׂרָאֵל הִיא ׳אֶרֶץ הָרִים וּבְקָעֹת, לִמְטַר הַשָּׁמַיִם תִּשְׁתֶּה מָּיִם׳.

׳וְהִשְׁקִיתָ בְרַגְלְךָ׳ – אֶרֶץ מִצְרַיִם הָיְתָה צְרִיכָה לְהָבִיא מַיִם מִנִּילוּס בְּרַגְלְךָ... אַתָּה יָשֵׁן עַל מִטָּתְךָ, וְהַקָּדוֹשׁ בָּרוּךְ הוּא מַשְׁקֶה נָמוּךְ וְגָבוֹהַּ כְּאַחַת.
׳אֲשֶׁר ה׳ אֱלֹהֶיךָ דֹּרֵשׁ אֹתָהּ׳ – כִּבְיָכוֹל אֵינוֹ דּוֹרֵשׁ אֶלָּא אוֹתָהּ, וְעַל יְדֵי אוֹתָהּ דְּרִישָׁה דּוֹרֵשׁ אֶת כָּל הָאֲרָצוֹת עִמָּהּ.

׳עֵינֵי ה׳ אֱלֹהֶיךָ בָּהּ מֵרֵשִׁית הַשָּׁנָה וְעַד אַחֲרִית שָׁנָה׳ – ה׳ מַבִּיט בְּאֶרֶץ יִשְׂרָאֵל כָּל הַשָּׁנָה, רוֹאֶה מָה הִיא צְרִיכָה וְדוֹאֵג לָהּ. אֵין אֶרֶץ אֲהוּבָה מִזּוֹ!

וּבְפָרָשַׁת ׳וְהָיָה אִם שָׁמֹעַ׳ לָמַדְנוּ אֶת הָעִקָּרוֹן שֶׁל שָׂכָר וָעֹנֶשׁ: אִם נִשְׁמַע לַמִּצְווֹת – ׳וְנָתַתִּי מְטַר אַרְצְכֶם בְּעִתּוֹ יוֹרֶה וּמַלְקוֹשׁ׳; וְאִם נִפְרֹשׁ מִן הַתּוֹרָה – ׳וְעָצַר אֶת הַשָּׁמַיִם וְלֹא יִהְיֶה מָטָר׳.
הַיְּשִׁיבָה בָּאָרֶץ תְּלוּיָה בְּקִיּוּם הַמִּצְווֹת.

📜 רַשִׁ״י
׳יוֹרֶה׳ – הִיא רְבִיעָה הַנּוֹפֶלֶת לְאַחַר הַזְּרִיעָה, שֶׁמַּרְוָה אֶת הָאָרֶץ וְאֶת הַזְּרָעִים.
׳וּמַלְקוֹשׁ׳ – רְבִיעָה הַיּוֹרֶדֶת סָמוּךְ לַקָּצִיר, לְמַלֹּאות הַתְּבוּאָה בְּקַשֶּׁיהָ.
׳לְאַהֲבָה אֶת ה׳׳ – שֶׁלֹּא תֹּאמַר הֲרֵי אֲנִי לוֹמֵד בִּשְׁבִיל שֶׁאֶהְיֶה עָשִׁיר... אֶלָּא כָּל מָה שֶׁתַּעֲשׂוּ עֲשׂוּ מֵאַהֲבָה.
׳וּלְעָבְדוֹ בְּכָל לְבַבְכֶם׳ – עֲבוֹדָה שֶׁהִיא בַּלֵּב, וְזוֹ הִיא תְּפִלָּה.

וּבְסוֹף הַפָּרָשִׁיָּה: ׳וְשַׂמְתֶּם אֶת דְּבָרַי אֵלֶּה עַל לְבַבְכֶם׳ – תְּפִלִּין וּמְזוּזָה, וְ׳וְלִמַּדְתֶּם אֹתָם אֶת בְּנֵיכֶם לְדַבֵּר בָּם׳ – רַשִׁ״י מְלַמֵּד שֶׁמִשָּׁעָה שֶׁהַבֵּן יוֹדֵעַ לְדַבֵּר, אָבִיו מְלַמְּדוֹ תּוֹרָה: ׳תּוֹרָה צִוָּה לָנוּ מֹשֶׁה׳.

💛 אַהֲבַת הַשֵּׁם
ה׳ מַשְׁקֶה אֶת אֶרֶץ יִשְׂרָאֵל בְּעַצְמוֹ, וְעֵינָיו בָּהּ כָּל הַשָּׁנָה. כְּשֶׁיּוֹרֵד גֶּשֶׁם בַּחֹרֶף, אֲנִי יוֹדֵעַ שֶׁזֶּה ה׳ מִתְבּוֹנֵן בָּאָרֶץ שֶׁלּוֹ וּבַיְּלָדִים שֶׁלּוֹ, וְדוֹאֵג לָנוּ 💛

🔺מְשֻׁלַּשׁ הַקְּסָמִים – עַם, תּוֹרָה וְאֶרֶץ

אִם נִסְתַּכֵּל עַל כָּל פָּרָשַׁת עֵקֶב מִלְמַעְלָה, נְגַלֶּה שְׁלוֹשָׁה דְּבָרִים שֶׁחוֹזְרִים שׁוּב וָשׁוּב:

עַם יִשְׂרָאֵל – הָעָם שֶׁנִּלְחָם בַּגּוֹיִם הָרַבִּים (׳רַבִּים הַגּוֹיִם הָאֵלֶּה מִמֶּנִּי׳), מִתְיַשֵּׁב, בּוֹנֶה בָּתִּים וְעָרִים, וְצָרִיךְ לְהִזָּהֵר מִ׳וְרָם לְבָבֶךָ׳.
אֶרֶץ יִשְׂרָאֵל – הָאָרֶץ שֶׁעוֹמְדִים לְהִכָּנֵס אֵלֶיהָ: שִׁבְחָהּ, מַעֲלוֹתֶיהָ, שִׁבְעַת הַמִּינִים, וְהַגֶּשֶׁם שֶׁה׳ נוֹתֵן בָּהּ.
תּוֹרַת יִשְׂרָאֵל – לִשְׁמֹר אֶת הַתּוֹרָה וְאֶת הַמִּצְווֹת, לֶאֱהֹב אֶת ה׳ וְלָלֶכֶת בִּדְרָכָיו.

וְזֶהוּ ׳מְשֻׁלַּשׁ הַקְּסָמִים׳: עַם יִשְׂרָאֵל שֶׁשּׁוֹמֵר אֶת הַתּוֹרָה – יוֹרֵשׁ אֶת אֶרֶץ יִשְׂרָאֵל.
כְּשֶׁשְּׁלוֹשֶׁת הַדְּבָרִים מְחֻבָּרִים יַחַד, הַמַּעְגָּל שָׁלֵם. זֶהוּ הַמֶּסֶר הַגָּדוֹל שֶׁל פָּרָשַׁת עֵקֶב.

💡 רַשִׁ״י בִּקְצָרָה – לַחֲזָרָה
׳שָׁמֹר תִּשְׁמְרוּן׳ – שֶׁלֹּא יִשְׁתַּכַּח תַּלְמוּדוֹ.
׳לָלֶכֶת בְּכָל דְּרָכָיו׳ – הוּא רַחוּם וְאַתָּה רַחוּם.
׳וּלְדָבְקָה בוֹ׳ – הִדָּבֵק בַּחֲכָמִים.
׳וְהוֹרִישׁ ה׳׳ – עֲשִׂיתֶם מָה שֶׁעֲלֵיכֶם, אַף אֲנִי אֶעֱשֶׂה מָה שֶׁעָלַי.
׳וַעֲצֻמִים מִכֶּם׳ – אַתֶּם גִּבּוֹרִים וְהֵם גִּבּוֹרִים מִכֶּם.
׳אִישׁ׳ – אֲפִלּוּ כְּעוֹג מֶלֶךְ הַבָּשָׁן.
׳פַּחְדְּכֶם׳ – עַל הַקְּרוֹבִים, ׳וּמוֹרַאֲכֶם׳ – עַל הָרְחוֹקִים.
׳עֵקֶב׳ – מִצְווֹת שֶׁאָדָם דָּשׁ בַּעֲקֵבָיו.
🤔 מְשִׂימַת הַמֶּדַלְיָה: ׳וְאָכַלְתָּ וְשָׂבָעְתָּ׳
בְּפָרָשַׁת עֵקֶב פָּגַשְׁנוּ פַּעֲמַיִם אֶת הַפָּסוּק ׳וְאָכַלְתָּ וְשָׂבָעְתָּ׳: בְּפֶרֶק ח׳ וּבְפֶרֶק י״א (בְּ׳וְהָיָה אִם שָׁמֹעַ׳). וּפָגַשְׁנוּ גַּם מַאֲכָלִים רַבִּים: חִטָּה וּשְׂעוֹרָה, גֶּפֶן, תְּאֵנָה וְרִמּוֹן, זֵית שֶׁמֶן וּדְבָשׁ, דָּגָן תִּירוֹשׁ וְיִצְהָר.
הַמְּשִׂימָה: לְהַעֲלוֹת תְּמוּנָה שֶׁל אֹכֶל – וּלְהַסְבִּיר לְאֵיזֶה פָּסוּק בַּפָּרָשָׁה הוּא קָשׁוּר. הַהֶסְבֵּר הוּא הָעִיקָּר, כִּי כָּךְ אֲנַחְנוּ מְחַבְּרִים בֵּין מָה שֶׁאֲנַחְנוּ עוֹשִׂים בֶּאֱמֶת לְבֵין מָה שֶׁלָּמַדְנוּ בַּתּוֹרָה.

וְשִׂימוּ לֵב לַדָּבָר הַיָּפֶה: גַּם כְּשֶׁהַתּוֹרָה מְדַבֶּרֶת עַל מִלְחָמוֹת וְעַל גְּבוּלוֹת, הַלֵּב שֶׁל הַכֹּל הוּא ׳לְאַהֲבָה אֶת ה׳ אֱלֹהֵיכֶם׳. הָאָרֶץ אֵינָהּ רַק פִּסַּת קַרְקַע – הִיא הַבַּיִת שֶׁבּוֹ עַם יִשְׂרָאֵל חַי עִם ה׳.

💛 אַהֲבַת הַשֵּׁם
סִיַּמְתִּי פָּרָשָׁה שְׁלֵמָה בַּתּוֹרָה! וּבְכָל פָּסוּק בָּהּ ה׳ אוֹמֵר לִי אֶת אוֹתוֹ דָּבָר: אֲנִי אוֹהֵב אוֹתְךָ, נָתַתִּי לְךָ אֶרֶץ טוֹבָה, וַאֲנִי רוֹצֶה שֶׁתִּהְיֶה קָרוֹב אֵלַי. וַאֲנִי עוֹנֶה לוֹ: גַּם אֲנִי אוֹהֵב אוֹתְךָ 💛

✨ לְסִכּוּם

  • פְּסוּקֵי הַסִּיּוּם: ׳כִּי אִם שָׁמֹר תִּשְׁמְרוּן אֶת כָּל הַמִּצְוָה הַזֹּאת... לְאַהֲבָה אֶת ה׳, לָלֶכֶת בְּכָל דְּרָכָיו וּלְדָבְקָה בוֹ׳ – זֶהוּ הַתְּנַאי לַהַבְטָחָה.
  • רַשִׁ״י: ׳שָׁמֹר תִּשְׁמְרוּן׳ – לְהִזָּהֵר בְּתַלְמוּדוֹ שֶׁלֹּא יִשְׁתַּכַּח; ׳לָלֶכֶת בְּכָל דְּרָכָיו׳ – הוּא רַחוּם וְאַתָּה תְּהֵא רַחוּם; ׳וּלְדָבְקָה בוֹ׳ – הִדָּבֵק בַּתַּלְמִידִים וּבַחֲכָמִים.
  • ׳וְהוֹרִישׁ ה׳׳ – עֲשִׂיתֶם מָה שֶׁעֲלֵיכֶם, אַף אֲנִי אֶעֱשֶׂה מָה שֶׁעָלַי. וּ׳וַעֲצֻמִים מִכֶּם׳ – אַתֶּם גִּבּוֹרִים מִשְּׁאָר הָאֻמּוֹת, וְהֵם גִּבּוֹרִים מִכֶּם.
  • ׳כָּל הַמָּקוֹם אֲשֶׁר תִּדְרֹךְ כַּף רַגְלְכֶם בּוֹ לָכֶם יִהְיֶה׳ – מִן הַמִּדְבָּר בַּדָּרוֹם וְעַד הַלְּבָנוֹן בַּצָּפוֹן, מִנְּהַר פְּרָת בַּמִּזְרָח וְעַד הַיָּם הָאַחֲרוֹן בַּמַּעֲרָב.
  • ׳לֹא יִתְיַצֵּב אִישׁ׳ – אֲפִלּוּ כְּעוֹג מֶלֶךְ הַבָּשָׁן. ׳פַּחְדְּכֶם׳ עַל הַקְּרוֹבִים (בֶּהָלָה פִּתְאוֹמִית) וּ׳מוֹרַאֲכֶם׳ עַל הָרְחוֹקִים (דְּאָגָה מִיָּמִים רַבִּים), וְהַכֹּל ׳כַּאֲשֶׁר דִּבֶּר לָכֶם׳.
  • הַפָּרָשָׁה נִקְרֵאת ׳עֵקֶב׳ עַל שֵׁם הַמִּצְווֹת הַקַּלּוֹת שֶׁאָדָם דָּשׁ בַּעֲקֵבָיו – וּמִי שֶׁשּׁוֹמֵר גַּם עֲלֵיהֶן, זוֹכֶה לִדְרֹךְ בְּכָל הָאָרֶץ.
  • הַמֶּסֶר הַגָּדוֹל: עַם יִשְׂרָאֵל שֶׁשּׁוֹמֵר אֶת תּוֹרַת יִשְׂרָאֵל – יוֹרֵשׁ אֶת אֶרֶץ יִשְׂרָאֵל. וְהַכֹּל מִתּוֹךְ אַהֲבָה: ׳לְאַהֲבָה אֶת ה׳ אֱלֹהֵיכֶם׳.

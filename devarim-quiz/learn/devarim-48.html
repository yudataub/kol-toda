אִסּוּרֵי כִּלְאַיִם

הַחֹמֶר הַמָּלֵא: כִּלְאֵי הַכֶּרֶם, כִּלְאֵי בְּהֵמָה, שַׁעַטְנֵז – וּמִצְוַת צִיצִית שֶׁסְּמוּכָה לָהֶם
🎯 לַחִידוֹן🏠 כָּל הַחִידוֹנִים
בַּפָּרָשָׁה שֶׁלָּנוּ, בִּשְׁלוֹשָׁה פְּסוּקִים קְצָרִים בִּלְבַד, מִסְתַּתְּרוֹת שָׁלוֹשׁ מִצְווֹת – וְכֻלָּן עוֹסְקוֹת בְּנוֹשֵׂא אֶחָד: כִּלְאַיִם, עֵרְבּוּב שֶׁאָסוּר לַעֲשׂוֹת בֵּין שְׁנֵי מִינִים.
אַחַת בַּכֶּרֶם, אַחַת בַּבְּהֵמוֹת וְאַחַת בַּבְּגָדִים. נִלְמַד מָה בְּדִיּוּק אָסוּר, מָה מְחַדֶּשֶׁת פָּרָשָׁתֵנוּ עַל מָה שֶׁכְּתוּב בְּסֵפֶר וַיִּקְרָא, וְאֵילּוּ טְעָמִים יָפִים נֶאֶמְרוּ בַּמִּצְווֹת הָאֵלֶּה – בֵּינֵיהֶם צַעַר בַּעֲלֵי חַיִּים.
וּבַסּוֹף נִפְגֹּשׁ אֶת הַפָּסוּק שֶׁבָּא מִיָּד אַחֲרֵיהֶם – ׳גְּדִלִים תַּעֲשֶׂה לָּךְ׳, מִצְוַת צִיצִית – וְנָבִין לָמָּה דַּוְקָא הִיא נִסְמְכָה לְאִסּוּרֵי הַכִּלְאַיִם.
וּבְכָל הַדֶּרֶךְ נִרְאֶה כַּמָּה ה׳ אוֹהֵב אוֹתָנוּ: הוּא מְלַוֶּה אוֹתָנוּ בַּשָּׂדֶה, בַּגִּנָּה, בָּאֻרְוָה וַאֲפִלּוּ בַּאֲרוֹן הַבְּגָדִים – וּבְכָל מָקוֹם נוֹתֵן לָנוּ מִצְוָה, כְּדֵי שֶׁנִּהְיֶה קְרוֹבִים אֵלָיו.
בַּשִּׁעוּר: 🔀 שָׁלוֹשׁ מִצְווֹת – וְכֻלָּן כִּלְאַיִם 🍇 לֹא תִזְרַע כַּרְמְךָ כִּלְאָיִם – כִּלְאֵי הַכֶּרֶם 📗 מִשְׁנֵה תּוֹרָה – מָה מוֹסִיפָה פָּרָשָׁתֵנוּ? 🥕 עֲרוּגוֹת בַּגִּנָּה – כָּל מִין בִּמְקוֹמוֹ 🐂 לֹא תַחֲרֹשׁ בְּשׁוֹר וּבַחֲמֹר – כִּלְאֵי בְּהֵמָה 💛 לָמָּה אָסוּר? – צַעַר בַּעֲלֵי חַיִּים 🧶 שַׁעַטְנֵז – צֶמֶר וּפִשְׁתִּים יַחְדָּו 🔬 מַעְבְּדוֹת שַׁעַטְנֵז – אֵיךְ בּוֹדְקִים בֶּגֶד? 🪢 גְּדִלִים תַּעֲשֶׂה לָּךְ – מִצְוַת צִיצִית

בַּיָּמִים הָאַחֲרוֹנִים לָמַדְנוּ מִצְוָה אַחַר מִצְוָה: הֲשָׁבַת אֲבֵדָה, שִׁלּוּחַ הַקֵּן וּמַעֲקֶה. הַתּוֹרָה מַמְשִׁיכָה כָּאן בְּרֶצֶף, וְאֵין זֶה בְּמִקְרֶה – רַשִׁ״י מַסְבִּיר לָמָּה הַפָּרָשִׁיּוֹת הָאֵלֶּה נִסְמְכוּ זוֹ לָזוֹ.

📜 רַשִׁ״י
׳כִּי תִבְנֶה בַּיִת חָדָשׁ׳ – אִם קִיַּמְתָּ מִצְוַת שִׁלּוּחַ הַקֵּן, סוֹפְךָ לִבְנוֹת בַּיִת חָדָשׁ וּתְקַיֵּם מִצְוַת מַעֲקֶה, שֶׁמִּצְוָה גּוֹרֶרֶת מִצְוָה, וְתַגִּיעַ לְכֶרֶם וְשָׂדֶה וְלִבְגָדִים נָאִים – לְכָךְ נִסְמְכוּ פָּרָשִׁיּוֹת הַלָּלוּ.

וְהִנֵּה בְּדִיּוּק זֶה הַסֵּדֶר שֶׁל הַפְּסוּקִים שֶׁלָּנוּ: כֶּרֶם, שָׂדֶה וּבְהֵמוֹת וּבְגָדִים. מִי שֶׁמְּקַיֵּם מִצְוָה אַחַת, ה׳ מְזַמֵּן לוֹ עוֹד מִצְוָה וְעוֹד מִצְוָה – כְּמוֹ אַבָּא שֶׁשָּׂמֵחַ לָתֵת לַיֶּלֶד שֶׁלּוֹ עוֹד וָעוֹד מַתָּנוֹת.

🤔 לַמּוֹרֶה: פּוֹתְחִים בַּפְּסוּקִים
קוֹרְאִים יַחַד אֶת פְּסוּקִים ט׳–י״א, וְשׁוֹאֲלִים: כַּמָּה מִצְווֹת יֵשׁ כָּאן? וְאַחַר כָּךְ: תְּנוּ כּוֹתֶרֶת קְצָרָה לְכָל מִצְוָה, בְּמִלָּה אוֹ שְׁתַּיִם.
הַתְּשׁוּבָה: שָׁלוֹשׁ מִצְווֹת – כֶּרֶם, בְּהֵמוֹת, בְּגָדִים.

הַמִּלָּה כִּלְאַיִם פֵּרוּשָׁהּ עֵרְבּוּב שֶׁאָסוּר לַעֲשׂוֹת – עֵרְבּוּב שֶׁל שְׁנֵי מִינִים. בִּשְׁלוֹשֶׁת הַפְּסוּקִים שֶׁלָּנוּ הָעֵרְבּוּב נֶאֱסָר בִּשְׁלוֹשָׁה תְּחוּמִים שׁוֹנִים, וּלְכִלְאַיִם שֶׁבַּבְּגָדִים יֵשׁ אֲפִלּוּ שֵׁם מְיֻחָד: שַׁעַטְנֵז.

וְכָךְ זֶה מִסְתַּדֵּר בַּפְּסוּקִים:
פָּסוּק ט׳ – בַּזְּרִיעָה: כִּלְאֵי הַכֶּרֶם, ׳לֹא תִזְרַע כַּרְמְךָ כִּלְאָיִם׳.
פָּסוּק י׳ – בַּחֲרִישָׁה: כִּלְאֵי בְּהֵמָה, ׳לֹא תַחֲרֹשׁ בְּשׁוֹר וּבַחֲמֹר יַחְדָּו׳.
פָּסוּק י״א – בַּלְּבִישָׁה: כִּלְאֵי בְּגָדִים, ׳לֹא תִלְבַּשׁ שַׁעַטְנֵז צֶמֶר וּפִשְׁתִּים יַחְדָּו׳.

🍇לֹא תִזְרַע כַּרְמְךָ כִּלְאָיִם – כִּלְאֵי הַכֶּרֶם

ט לֹא תִזְרַע כַּרְמְךָ כִּלְאָיִם פֶּן תִּקְדַּשׁ הַמְלֵאָה הַזֶּרַע אֲשֶׁר תִּזְרָע וּתְבוּאַת הַכָּרֶם

כֶּרֶם הוּא שָׂדֶה שֶׁל גְּפָנִים. הַתּוֹרָה אוֹסֶרֶת לִזְרֹעַ בְּתוֹךְ הַכֶּרֶם אוֹ לְיָדוֹ מִינִים אֲחֵרִים. מָה בְּדִיּוּק? רַשִׁ״י מְפָרֵט.

📜 רַשִׁ״י
׳כִּלְאַיִם׳ – חִטָּה וּשְׂעוֹרָה וְחַרְצָן בְּמַפֹּלֶת יָד.

חַרְצָן הוּא הַגַּרְעִין שֶׁל הָעֲנָבִים, שֶׁמִּמֶּנּוּ צוֹמַחַת הַגֶּפֶן. ׳בְּמַפֹּלֶת יָד׳ פֵּרוּשׁוֹ שֶׁהָאָדָם הִפִּיל אֶת שְׁלוֹשֶׁת הַזְּרָעִים מִיָּדוֹ בִּזְרִיעָה אַחַת.
זֶהוּ הָאִסּוּר מִן הַתּוֹרָה: שְׁנֵי מִינֵי תְּבוּאָה יַחַד עִם הַגֶּפֶן. וְחֲזַ״ל הוֹסִיפוּ וְאָסְרוּ גַּם מִינִים אֲחֵרִים, כְּמוֹ יְרָקוֹת, מִתַּחַת לַגֶּפֶן אוֹ לְיָדָהּ.

💡 בַּשָּׂדֶה מִמָּשׁ
לִפְעָמִים רוֹאִים גְּפָנִים שֶׁמְּנַצְּלִים אֶת הַמָּקוֹם שֶׁתַּחְתֵּיהֶן וְזוֹרְעִים שָׁם מְלָפְפוֹנִים אוֹ יְרָקוֹת אֲחֵרִים.
אָסוּר לַעֲשׂוֹת כָּךְ. זוֹ בְּדִיּוּק הַזְּרִיעָה שֶׁהַתּוֹרָה אוֹסֶרֶת בְּפֶרֶק כ״ב.

וּמָה קוֹרֶה לְמִי שֶׁעָבַר וְזָרַע? הַפָּסוּק מַזְהִיר: ׳פֶּן תִּקְדַּשׁ הַמְלֵאָה הַזֶּרַע אֲשֶׁר תִּזְרָע וּתְבוּאַת הַכָּרֶם׳.

📜 רַשִׁ״י
׳פֶּן תִּקְדַּשׁ׳ – כְּתַרְגּוּמוֹ ׳תִּסְתָּאָב׳. כָּל דָּבָר הַנִּתְעָב עַל הָאָדָם, בֵּין לְשֶׁבַח כְּגוֹן הֶקְדֵּשׁ, בֵּין לִגְנַאי כְּגוֹן אִסּוּר – נוֹפֵל בּוֹ לְשׁוֹן ׳קֹדֶשׁ׳.
׳הַמְלֵאָה׳ – זֶה מִלּוּי וְתוֹסֶפֶת שֶׁהַזֶּרַע מוֹסִיף.

כְּלוֹמַר: הַמִּלָּה ׳תִּקְדַּשׁ׳ אֵינָהּ קְדֻשָּׁה שֶׁל מִצְוָה, אֶלָּא ׳תֵּאָסֵר׳ – אוֹנְקְלוֹס מְתַרְגֵּם ׳תִּסְתָּאָב׳. וְ׳הַמְלֵאָה׳ הִיא כָּל מָה שֶׁיִּתְמַלֵּא וְיִגְדַּל מִן הַזְּרִיעָה הַזּוֹ.
וְהַדִּין: אִם זָרַע וְגָדַל – כָּל מָה שֶׁגָּדֵל אָסוּר, וַאֲפִלּוּ בַּהֲנָאָה. אָסוּר לֶאֱכֹל, אָסוּר לִמְכֹּר, אָסוּר לֵהָנוֹת מִמֶּנּוּ.

💛 אַהֲבַת הַשֵּׁם
ה׳ שָׂם לָנוּ גְּבוּל דַּוְקָא בַּמָּקוֹם הֲכִי יָפֶה – בַּכֶּרֶם הַמְּלֵא עֲנָבִים. הוּא לֹא מְקַנֵּא בָּנוּ: הוּא רוֹצֶה שֶׁגַּם הָאֲדָמָה שֶׁלָּנוּ וְגַם הָאֹכֶל שֶׁלָּנוּ יִהְיוּ מְחֻבָּרִים אֵלָיו 💛

📗מִשְׁנֵה תּוֹרָה – מָה מוֹסִיפָה פָּרָשָׁתֵנוּ?

סֵפֶר דְּבָרִים נִקְרָא ׳מִשְׁנֵה תּוֹרָה׳: מֹשֶׁה חוֹזֵר בּוֹ עַל דְּבָרִים שֶׁכְּבָר נֶאֶמְרוּ בַּחֻמָּשִׁים הַקּוֹדְמִים – וּמוֹסִיף עֲלֵיהֶם.
גַּם אִסּוּרֵי הַכִּלְאַיִם כְּבָר נִזְכְּרוּ, בְּסֵפֶר וַיִּקְרָא בְּפָרָשַׁת קְדוֹשִׁים. לָכֵן חֲזַ״ל שָׁאֲלוּ: אִם כָּךְ, מָה הֶחָדָשׁ כָּאן?

💡 וַיִּקְרָא מוּל דְּבָרִים
בְּוַיִּקְרָא (י״ט, י״ט): ׳בְּהֶמְתְּךָ לֹא תַרְבִּיעַ כִּלְאַיִם, שָׂדְךָ לֹא תִזְרַע כִּלְאָיִם, וּבֶגֶד כִּלְאַיִם שַׁעַטְנֵז לֹא יַעֲלֶה עָלֶיךָ׳.
בִּדְבָרִים (כ״ב, ט׳): ׳לֹא תִזְרַע כַּרְמְךָ כִּלְאָיִם פֶּן תִּקְדַּשׁ...׳.
שִׂימוּ לֵב לַהֶבְדֵּל: שָׁם ׳שָׂדְךָ׳, וְכָאן ׳כַּרְמְךָ׳ – וְנוֹסְפוּ הַמִּלִּים ׳פֶּן תִּקְדַּשׁ׳.

הַמִּלִּים ׳פֶּן תִּקְדַּשׁ׳ הֵן הַחִדּוּשׁ. מֵהֶן לָמַדְנוּ שֶׁיֵּשׁ שְׁנֵי סוּגִים:
כִּלְאֵי זְרָעִים רְגִילִים – זְרִיעָה שֶׁל שְׁנֵי מִינִים יַחַד בַּשָּׂדֶה. הַזְּרִיעָה אֲסוּרָה, אֲבָל אִם עָבַר וְזָרַע – הַגִּדּוּלִים אֵינָם אֲסוּרִים בַּהֲנָאָה.
כִּלְאֵי הַכֶּרֶם – זְרִיעָה בַּכֶּרֶם. כָּאן, אִם עָבַר וְזָרַע, הַיְּבוּל נֶאֱסָר בַּהֲנָאָה.

לָכֵן כִּלְאֵי הַכֶּרֶם הֵם הַחֲמוּרִים שֶׁבַּכִּלְאַיִם שֶׁל גִּדּוּלֵי הָאֲדָמָה. וְזוֹ גַּם אַחַת הַסִּבּוֹת שֶׁצָּרִיךְ הֶכְשֵׁר עַל פֵּרוֹת וִירָקוֹת: כְּדֵי לָדַעַת שֶׁהֵם גָּדְלוּ בְּכַשְׁרוּת, בְּלִי אִסּוּר כִּלְאַיִם.

🥕עֲרוּגוֹת בַּגִּנָּה – כָּל מִין בִּמְקוֹמוֹ

בּוֹאוּ נֵצֵא לְגִנַּת יְרָקוֹת: גֶּזֶר, קוֹלוֹרַבִּי, עַגְבָנִיּוֹת שֶׁרִי, פִּלְפֵּל. אֵין כָּאן גֶּפֶן, וְלָכֵן אֵין כָּאן כִּלְאֵי הַכֶּרֶם – אֲבָל עֲדַיִן צָרִיךְ לְהִזָּהֵר.
אֵיךְ? מַפְרִידִים בֵּין הַמִּינִים: כָּל יֶרֶק בָּעֲרוּגָה שֶׁלּוֹ, הַגֶּזֶר לְחוּד, הַקּוֹלוֹרַבִּי לְחוּד וְהָעַגְבָנִיָּה לְחוּד.

זֶהוּ אִסּוּר כִּלְאֵי זְרָעִים, שֶׁכָּתוּב בְּסֵפֶר וַיִּקְרָא פֶּרֶק י״ט. מִן הַתּוֹרָה הָאִסּוּר הוּא בִּזְרִיעַת מִינֵי תְּבוּאָה יַחַד, וּבִירָקוֹת – חֲזַ״ל הֶחְמִירוּ וְדָרְשׁוּ לְהַפְרִיד גַּם בֵּינֵיהֶם.

💡 שָׁלוֹשׁ מַדְרֵגוֹת בְּגִדּוּלֵי הָאֲדָמָה
כִּלְאֵי הַכֶּרֶם – הַחֲמוּרִים: הַגִּדּוּלִים נֶאֱסָרִים בַּהֲנָאָה.
כִּלְאֵי זְרָעִים בַּשָּׂדֶה – אָסוּר לִזְרֹעַ, אֲבָל בְּדִיעֲבַד הַגִּדּוּלִים מֻתָּרִים.
יְרָקוֹת בַּגִּנָּה – הַהַפְרָדָה בֵּינֵיהֶם מִדִּבְרֵי חֲזַ״ל.

יֵשׁ בָּזֶה דָּבָר מַרְגִּישׁ: גַּם כְּשֶׁאֲנַחְנוּ שׁוֹתְלִים שְׁתִיל קָטָן בַּגִּנָּה, ה׳ נִמְצָא אִתָּנוּ. הוּא נָתַן לָנוּ מִצְוָה גַּם שָׁם, כְּדֵי שֶׁנִּזְכֹּר מִי בָּרָא אֶת הָאֲדָמָה, אֶת הַזֶּרַע וְאֶת הַגֶּשֶׁם.

📖 פֶּרֶק כ״ב, פָּסוּק י׳ ←
י לֹא תַחֲרֹשׁ בְּשׁוֹר וּבַחֲמֹר יַחְדָּו

הַמִּצְוָה הַשְּׁנִיָּה נוֹגַעַת לַבְּהֵמוֹת. כְּדֵי לְהָבִין אוֹתָהּ, בּוֹאוּ נִתְבּוֹנֵן בַּחֲרִישָׁה: קוֹשְׁרִים שְׁתֵּי בְּהֵמוֹת לְעֹל אֶחָד (בַּתַּנַ״ךְ קוֹרְאִים לָזֶה ׳צֶמֶד בָּקָר׳), וּבְאֶמְצַע מְחֻבֶּרֶת הַמַּחֲרֵשָׁה. הַקָּצֶה הֶחָד שֶׁלָּהּ הוֹפֵךְ אֶת הָאֲדָמָה, יוֹצֵר תְּלָמִים, וְכָךְ הַזְּרָעִים יְכוֹלִים לְהִקָּלֵט.

וְהַתּוֹרָה אוֹמֶרֶת: מֻתָּר לַחֲרֹשׁ בִּשְׁנֵי שְׁוָרִים, וּמֻתָּר בִּשְׁנֵי חֲמוֹרִים – אֲבָל לֹא בְּשׁוֹר וַחֲמוֹר יַחַד.

׳לֹא תַחֲרֹשׁ בְּשׁוֹר וּבַחֲמוֹר׳ – הוּא הַדִּין לְכָל שְׁנֵי מִינִים שֶׁבָּעוֹלָם, וְהוּא הַדִּין לְהַנְהִיגָם יַחַד קְשׁוּרִים זוּגוֹת בְּהוֹלָכַת שׁוּם מַשָּׂא.

💡 מָה הַחִדּוּשׁ מוּל סֵפֶר וַיִּקְרָא?
בְּוַיִּקְרָא כָּתוּב ׳בְּהֶמְתְּךָ לֹא תַרְבִּיעַ כִּלְאַיִם׳ – אָסוּר לְהוֹלִיד מִין חָדָשׁ מִשְּׁתֵּי בְּהֵמוֹת שׁוֹנוֹת.

💛לָמָּה אָסוּר? – צַעַר בַּעֲלֵי חַיִּים

📜 הָאַבְן עֶזְרָא: אֵין כֹּחַ הַחֲמוֹר כְּכֹחַ הַשּׁוֹר

💡 הַחִזְקוּנִי: זֶה אוֹכֵל וְזֶה מִתְעַנֶּה

צֶמֶר – מֵעוֹלַם הַחַי: גּוֹזְזִים אֶת צֶמֶר הַכְּבָשִׂים, סוֹרְקִים אוֹתוֹ, מְנַפְּצִים וְטוֹוִים מִמֶּנּוּ חוּטִים.
פִּשְׁתִּים – מֵעוֹלַם הַצּוֹמֵחַ: קוֹצְרִים אֶת צֶמַח הַפִּשְׁתָּה, מְנַפְּצִים, מְיַבְּשִׁים וְטוֹוִים מִמֶּנּוּ חוּטִים.

וְהַדִּין: בֶּגֶד שֶׁאָרוּג גַּם מֵחוּטֵי צֶמֶר וְגַם מֵחוּטֵי פִּשְׁתִּים – אָסוּר לִלְבֹּשׁ.

📜 רַשִׁ״י
׳שַׁעַטְנֵז׳ – לְשׁוֹן עֵרוּב. וְרַבּוֹתֵינוּ פֵּרְשׁוּ: שׁוּעַ טָווּי וְנוּז.

💛 אַהֲבַת הַשֵּׁם

🔬מַעְבְּדוֹת שַׁעַטְנֵז – אֵיךְ בּוֹדְקִים בֶּגֶד?

הַיּוֹם מְיַצְּרִים בְּגָדִים מֵהַרְבֵּה חֳמָרִים: כֻּתְנָה, מֶשִׁי – שֶׁאוֹתוֹ מְפִיקִים מִן הַחוּט שֶׁל תּוֹלַעַת הַמֶּשִׁי – וְגַם חֳמָרִים מְלָאכוּתִיִּים, סִינְתֵּטִיִּים.
בַּחֳמָרִים הָאֵלֶּה אֵין שַׁעַטְנֵז כְּלָל, כִּי הָאִסּוּר הוּא רַק בְּצֶמֶר וּפִשְׁתִּים יַחַד.

אֲבָל יֵשׁ בְּגָדִים שֶׁכֵּן צָרִיךְ לִבְדֹּק, וּבְעִקָּר חֲלִיפוֹת: לִפְעָמִים מְעֹרָב בָּהֶן צֶמֶר מִן הַחַי יַחַד עִם פִּשְׁתָּן.
אֵיךְ יוֹדְעִים? בּוֹדְקִים בַּפֶּתֶק שֶׁעַל הַבֶּגֶד – כִּמְעַט בְּכָל בֶּגֶד כָּתוּב מִמָּה הוּא עָשׂוּי. וְיֵשׁ גַּם מַעְבָּדוֹת לִבְדִיקַת שַׁעַטְנֵז: מְבִיאִים לְשָׁם אֶת הַחֲלִיפָה, וְהַבּוֹדֵק מִסְתַּכֵּל בַּחוּטִים וְקוֹבֵעַ אִם יֵשׁ בָּהּ צֶמֶר וּפִשְׁתִּים יַחַד.

מִי הָיָה מַאֲמִין? הוֹלְכִים לַסּוּפֶּר לִקְנוֹת פֵּרוֹת – וְנִזְכָּרִים בְּכִלְאֵי הַכֶּרֶם וּבַהֶכְשֵׁר. הוֹלְכִים לִקְנוֹת בֶּגֶד חָדָשׁ – וְנִזְכָּרִים בְּשַׁעַטְנֵז. הַתּוֹרָה נִמְצֵאת בְּכָל פִּנָּה בַּחַיִּים.

💛 אַהֲבַת הַשֵּׁם
ה׳ נָתַן לִי מִצְווֹת גַּם בַּחֲנוּת הַבְּגָדִים וְגַם בַּסּוּפֶּר, כְּדֵי שֶׁלֹּא יִהְיֶה אַף רֶגַע בַּיּוֹם שֶׁבּוֹ אֲנִי רָחוֹק מִמֶּנּוּ. כָּל בְּדִיקָה קְטַנָּה הִיא דֶּרֶךְ לוֹמַר לַה׳: אֲנִי אוֹהֵב אוֹתְךָ 💛

🪢גְּדִלִים תַּעֲשֶׂה לָּךְ – מִצְוַת צִיצִית

📖 פֶּרֶק כ״ב, פָּסוּק י״ב ←

׳גְּדִלִים׳ הֵם מַעֲשֵׂה שַׁרְשְׁרוֹת שֶׁל חוּטִים – זוֹ מִצְוַת צִיצִית, שֶׁאוֹתָהּ כְּבָר לָמַדְנוּ בְּפָרָשַׁת שְׁלַח.

📜 רַשִׁ״י
׳גְּדִלִים תַּעֲשֶׂה לָּךְ׳ – אַף מִן הַכִּלְאַיִם, לְכָךְ סְמָכָן הַכָּתוּב.

מָה הַכַּוָּנָה? חוּט הַתְּכֵלֶת שֶׁבַּצִּיצִית הוּא חוּט צֶמֶר צָבוּעַ. וּמָה יִהְיֶה אִם הַטַּלִּית עַצְמָהּ עֲשׂוּיָה פִּשְׁתִּים? לִכְאוֹרָה זֶה שַׁעַטְנֵז!
לָכֵן הִסְמִיךְ הַכָּתוּב אֶת מִצְוַת צִיצִית לְאִסּוּר הַכִּלְאַיִם – לְלַמֵּד שֶׁבַּצִּיצִית מֻתָּר: מְקַיְּמִים אֶת הַמִּצְוָה אַף מִן הַכִּלְאַיִם.

💡 חִידַת שַׁעַטְנֵז
אֵיזֶה עוֹד בֶּגֶד, מִלְּבַד צִיצִית, עָשׂוּי מִצֶּמֶר וּפִשְׁתִּים – וּמִצְוָה לִלְבֹּשׁ אוֹתוֹ?
הַתְּשׁוּבָה: בִּגְדֵי הַכֹּהֵן הַגָּדוֹל – הָאֵפוֹד, הַחֹשֶׁן וְהָאַבְנֵט. הֵם עֲשׂוּיִים ׳תְּכֵלֶת וְאַרְגָּמָן וְתוֹלַעַת שָׁנִי׳ – צֶמֶר צָבוּעַ בִּצְבָעִים שׁוֹנִים – וְגַם ׳שֵׁשׁ מָשְׁזָר׳, שֶׁהוּא פִּשְׁתָּן.
וּלְדַעַת הָרַמְבַּ״ם גַּם הָאַבְנֵט שֶׁל כֹּהֵן הֶדְיוֹט הוּא כִּלְאַיִם, וְמֻתָּר לִלְבֹּשׁ אוֹתוֹ רַק בִּשְׁעַת הָעֲבוֹדָה בַּמִּקְדָּשׁ.
💡 מִלּוֹן קָטָן מֵרַשִׁ״י
׳כִּלְאַיִם׳ – חִטָּה וּשְׂעוֹרָה וְחַרְצָן.
׳פֶּן תִּקְדַּשׁ׳ – תִּסְתָּאָב, כְּלוֹמַר תֵּאָסֵר.
׳הַמְלֵאָה׳ – מִלּוּי וְתוֹסֶפֶת שֶׁהַזֶּרַע מוֹסִיף.
׳לֹא תַחֲרֹשׁ בְּשׁוֹר וּבַחֲמוֹר׳ – הוּא הַדִּין לְכָל שְׁנֵי מִינִים.
׳שַׁעַטְנֵז׳ – לְשׁוֹן עֵרוּב; וְרַבּוֹתֵינוּ: שׁוּעַ טָווּי וְנוּז.
׳גְּדִלִים׳ – אַף מִן הַכִּלְאַיִם, לְכָךְ סְמָכָן הַכָּתוּב.

הִנֵּה הַיֹּפִי שֶׁבַּסְּמִיכוּת: קֹדֶם ה׳ אוֹסֵר עָלֵינוּ עֵרְבּוּב, וּמִיָּד אַחַר כָּךְ הוּא נוֹתֵן לָנוּ מִצְוָה שֶׁבָּהּ הָעֵרְבּוּב דַּוְקָא מֻתָּר – כִּי הִיא כֻּלָּהּ קְדֻשָּׁה, וּכְדֵי לִזְכֹּר אֶת ה׳ וְאֶת מִצְווֹתָיו.

🤔 לַמּוֹרֶה: הֶמְשֵׁךְ הַפֶּרֶק

✨ לְסִכּוּם

  • כִּלְאַיִם הוּא עֵרְבּוּב אָסוּר בֵּין שְׁנֵי מִינִים, וּבְפָרָשָׁתֵנוּ יֵשׁ שָׁלוֹשׁ מִצְווֹת: בַּזְּרִיעָה (פָּסוּק ט׳), בַּחֲרִישָׁה (פָּסוּק י׳) וּבַלְּבִישָׁה (פָּסוּק י״א).
  • כִּלְאֵי הַכֶּרֶם: ׳לֹא תִזְרַע כַּרְמְךָ כִּלְאָיִם׳ – חִטָּה וּשְׂעוֹרָה וְחַרְצָן בְּמַפֹּלֶת יָד. וְאִם עָבַר וְזָרַע – ׳פֶּן תִּקְדַּשׁ׳: הַיְּבוּל אָסוּר אֲפִלּוּ בַּהֲנָאָה.
  • סֵפֶר דְּבָרִים הוּא ׳מִשְׁנֵה תּוֹרָה׳: בְּוַיִּקְרָא נֶאֶמְרוּ שְׁלוֹשֶׁת הַכִּלְאַיִם, וְכָאן הַחִדּוּשׁ – בַּכֶּרֶם ׳פֶּן תִּקְדַּשׁ׳, וּבַבְּהֵמוֹת אִסּוּר מְלָאכָה בִּשְׁנֵי מִינִים.
  • ׳לֹא תַחֲרֹשׁ בְּשׁוֹר וּבַחֲמֹר יַחְדָּו׳ – וְלֹא רַק בָּהֶם: הוּא הַדִּין לְכָל שְׁנֵי מִינִים, וְגַם לִנְשִׂיאַת מַשָּׂא. הַטַּעַם: ה׳ חָמַל עַל כָּל מַעֲשָׂיו – צַעַר בַּעֲלֵי חַיִּים.

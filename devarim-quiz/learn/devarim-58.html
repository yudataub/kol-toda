אֲנַחְנוּ כְּבָר מַמָּשׁ לִפְנֵי סִיּוּם הַתּוֹרָה! בַּתּוֹרָה יֵשׁ חֲמִשִּׁים וְאַרְבַּע פָּרָשׁוֹת, וּפָרָשַׁת הַאֲזִינוּ הִיא הַפָּרָשָׁה שֶׁלִּפְנֵי הָאַחֲרוֹנָה.
בַּשִּׁעוּר הַזֶּה נִלְמַד אֶת שִׁירַת הַאֲזִינוּ – שִׁירָה שֶׁה׳ צִוָּה לִכְתֹּב וּלְלַמֵּד, ׳שִׂימָהּ בְּפִיהֶם׳, כְּדֵי שֶׁלֹּא תִּשָּׁכַח לְעוֹלָם. נָבִין מָה זוֹ שִׁירָה, מָה הֵן תַּקְבִּילוֹת וְדִמּוּיִים, אֵיךְ הִיא כְּתוּבָה בְּסֵפֶר הַתּוֹרָה, וּמָה הִיא מְסַפֶּרֶת.
וּבְעִקָּר נִלְמַד אֶת הַסּוֹד הַגָּדוֹל שֶׁלָּהּ: גַּם כְּשֶׁעַם יִשְׂרָאֵל שׁוֹכֵחַ אֶת ה׳, ה׳ לְעוֹלָם אֵינוֹ עוֹזֵב אֶת עַמּוֹ – בַּסּוֹף הוּא תָּמִיד מְרַחֵם וּמוֹשִׁיעַ 💛
בַּשִּׁעוּר: 📚 הַפָּרָשָׁה שֶׁלִּפְנֵי הָאַחֲרוֹנָה 👄 ׳כִּתְבוּ לָכֶם אֶת הַשִּׁירָה הַזֹּאת... שִׂימָהּ בְּפִיהֶם׳ 🌫️ לָמָּה צְרִיכָה הַשִּׁירָה לִהְיוֹת עֵדָה? – הֶסְתֵּר פָּנִים 🧱 אָרִיחַ עַל גַּבֵּי אָרִיחַ – אֵיךְ הַשִּׁירָה כְּתוּבָה בַּתּוֹרָה 🎵 מָה זוֹ שִׁירָה? תַּקְבִּילוֹת וְדִמּוּיִים 👁️ ׳כְּאִישׁוֹן עֵינוֹ׳ – אֵיךְ ה׳ שָׁמַר עָלֵינוּ בַּמִּדְבָּר 🍯 ׳וַיִּשְׁמַן יְשֻׁרוּן וַיִּבְעָט׳ – כְּשֶׁשּׁוֹכְחִים מִי נָתַן 💞 ׳וְעַל עֲבָדָיו יִתְנֶחָם׳ – ה׳ לְעוֹלָם לֹא יַעֲזֹב 🎶 הַזִּי״ו לָ״ךְ – הַשִּׁירָה בְּבֵית הַמִּקְדָּשׁ 🌅 ׳הַרְנִינוּ גוֹיִם עַמּוֹ׳ – הַסִּיּוּם הַטּוֹב

📚הַפָּרָשָׁה שֶׁלִּפְנֵי הָאַחֲרוֹנָה

כַּמָּה פָּרָשׁוֹת יֵשׁ בַּתּוֹרָה? בַּתּוֹרָה יֵשׁ חֲמִשִּׁים וְאַרְבַּע פָּרָשׁוֹת, מִפָּרָשַׁת בְּרֵאשִׁית וְעַד הַפָּרָשָׁה הָאַחֲרוֹנָה – ׳וְזֹאת הַבְּרָכָה׳.
פָּרָשַׁת הַאֲזִינוּ הִיא הַפָּרָשָׁה הַחֲמִשִּׁים וְשָׁלֹשׁ – כְּלוֹמַר הַפָּרָשָׁה שֶׁלִּפְנֵי הָאַחֲרוֹנָה. אֲנַחְנוּ כָּל כָּךְ קְרוֹבִים לְסִיּוּם הַתּוֹרָה!

וְזוֹ פָּרָשָׁה מְיֻחֶדֶת מְאוֹד: כֻּלָּהּ שִׁירָה. לֹא סִפּוּר וְלֹא רְשִׁימַת מִצְווֹת, אֶלָּא שִׁירָה שְׁלֵמָה שֶׁה׳ נָתַן לְמֹשֶׁה כְּדֵי שֶׁיְּלַמֵּד אוֹתָהּ אֶת עַם יִשְׂרָאֵל.

פָּרָשַׁת הַאֲזִינוּ (פֶּרֶק ל״ב): הַשִּׁירָה עַצְמָהּ, מִ׳הַאֲזִינוּ הַשָּׁמַיִם׳ וְעַד ׳וְכִפֶּר אַדְמָתוֹ עַמּוֹ׳.
וְאַחֲרֶיהָ נִשְׁאֶרֶת רַק פָּרָשָׁה אַחַת: ׳וְזֹאת הַבְּרָכָה׳.

כְּשֶׁמְּסַיְּמִים חֹמֶשׁ אוֹמְרִים ׳חֲזַק חֲזַק וְנִתְחַזֵּק׳. אֲנַחְנוּ מַמָּשׁ לִפְנֵי זֶה – וְכָל פָּסוּק שֶׁאֲנַחְנוּ לוֹמְדִים הוּא עוֹד מַתָּנָה שֶׁה׳ נָתַן לָנוּ בְּאַהֲבָה.

ה׳ נָתַן לָנוּ תּוֹרָה שְׁלֵמָה, חֲמִשִּׁים וְאַרְבַּע פָּרָשׁוֹת, וְכָל אַחַת מֵהֶן הִיא מִכְתָּב שֶׁלּוֹ אֵלַי. כְּשֶׁאֲנִי מַגִּיעַ לַפָּרָשׁוֹת הָאַחֲרוֹנוֹת, אֲנִי מַרְגִּישׁ כַּמָּה הוּא רָצָה לְדַבֵּר אִתִּי 💛

בְּיוֹם הָאַחֲרוֹן שֶׁל מֹשֶׁה ה׳ נִגְלָה אֵלָיו בָּאֹהֶל בְּעַמּוּד עָנָן, וְאָמַר לוֹ: ׳וְעַתָּה כִּתְבוּ לָכֶם אֶת הַשִּׁירָה הַזֹּאת וְלַמְּדָהּ אֶת בְּנֵי יִשְׂרָאֵל, שִׂימָהּ בְּפִיהֶם׳.
זוֹ הַנְּבוּאָה הָאַחֲרוֹנָה שֶׁל ה׳ לְמֹשֶׁה, וְזוֹ גַּם הַמִּצְוָה הָאַחֲרוֹנָה בַּתּוֹרָה: לִכְתֹּב אֶת הַשִּׁירָה. חֲזַ״ל לוֹמְדִים מִכָּאן שֶׁמִּצְוָה עַל כָּל אֶחָד לִכְתֹּב לְעַצְמוֹ סֵפֶר תּוֹרָה.

📜 רַשִׁ״י
׳אֶת הַשִּׁירָה הַזֹּאת׳ – ׳הַאֲזִינוּ הַשָּׁמַיִם׳ עַד ׳וְכִפֶּר אַדְמָתוֹ עַמּוֹ׳.

כְּלוֹמַר: ׳הַשִּׁירָה הַזֹּאת׳ הִיא בְּדִיּוּק פָּרָשַׁת הַאֲזִינוּ – מֵהַפָּסוּק הָרִאשׁוֹן וְעַד הַפָּסוּק הָאַחֲרוֹן שֶׁל הַשִּׁירָה.

שִׂימוּ לֵב לִשְׁתֵּי הַמִּלִּים: ׳כִּתְבוּ׳ וְגַם ׳שִׂימָהּ בְּפִיהֶם׳. לִכְתֹּב – זֶה בִּכְתָב. אֲבָל ה׳ רָצָה יוֹתֵר: שֶׁהַשִּׁירָה תִּהְיֶה בַּפֶּה, בְּעַל פֶּה.
מֹשֶׁה לִמֵּד אֶת בְּנֵי דּוֹרוֹ, הֵם לִמְּדוּ אֶת יַלְדֵיהֶם, וְהַיְּלָדִים אֶת יַלְדֵיהֶם – עַד שֶׁהַשִּׁירָה שְׁמוּרָה בְּפִי הָעָם לְדוֹרוֹת. יֵשׁ קְהִלּוֹת שֶׁנָּהֲגוּ לְלַמֵּד אֶת שִׁירַת הַאֲזִינוּ בְּעַל פֶּה, וְיֵשׁ מָסֹרֶת עַל סְגֻלָּה שֶׁבַּדָּבָר.

📚 סִפּוּר: הַמְּנַהֵל שֶׁנִּכְנַס לַכִּתָּה
לִפְנֵי יוֹם כִּפּוּר נִכְנַס מְנַהֵל בֵּית הַסֵּפֶר לַכִּתָּה וְאָמַר: אֶת שִׁירַת הַאֲזִינוּ צָרִיךְ לָשִׂים מַמָּשׁ בַּפֶּה! אַתֶּם יוֹצְאִים לְחֻפְשַׁת סֻכּוֹת – וּכְשֶׁתַּחְזְרוּ אַחֲרֵי שִׂמְחַת תּוֹרָה אֲנִי בּוֹחֵן כָּל יֶלֶד וְיֶלֶד.
הַיְּלָדִים נִבְהֲלוּ, וְלָמְדוּ. וְעַד הַיּוֹם יוֹדֵעַ אוֹתוֹ תַּלְמִיד אֶת הַשִּׁירָה מִלָּה בְּמִלָּה.
זֶה בְּדִיּוּק ׳שִׂימָהּ בְּפִיהֶם׳: מָה שֶׁלּוֹמְדִים בְּעַל פֶּה – נִשְׁאָר אִתָּנוּ לְכָל הַחַיִּים.

׳וַיִּכְתֹּב מֹשֶׁה אֶת הַשִּׁירָה הַזֹּאת בַּיּוֹם הַהוּא וַיְלַמְּדָהּ אֶת בְּנֵי יִשְׂרָאֵל׳ – בְּיוֹמוֹ הָאַחֲרוֹן, מֹשֶׁה לֹא נָח. הוּא כָּתַב וְלִמֵּד, כְּדֵי שֶׁהָעָם שֶׁאוֹתוֹ אָהַב יִשָּׁאֵר קָשׁוּר לַה׳ גַּם אַחֲרָיו.

לִפְנֵי הַצִּוּוּי לִכְתֹּב אֶת הַשִּׁירָה, ה׳ אוֹמֵר לְמֹשֶׁה מָה עָתִיד לִקְרוֹת. הָעָם יִכָּנֵס לָאָרֶץ – וְשָׁם מְחַכֶּה נִסָּיוֹן.
׳כִּי אֲבִיאֶנּוּ אֶל הָאֲדָמָה... זָבַת חָלָב וּדְבַשׁ, וְאָכַל וְשָׂבַע וְדָשֵׁן, וּפָנָה אֶל אֱלֹהִים אֲחֵרִים׳. כְּשֶׁיִּהְיֶה שֶׁפַע וְטוֹב, הֵם עֲלוּלִים לִשְׁכֹּחַ מִי נָתַן אוֹתוֹ.

מָה יַעֲשֶׂה עַם יִשְׂרָאֵל? ׳וְקָם הָעָם הַזֶּה וְזָנָה אַחֲרֵי אֱלֹהֵי נֵכַר הָאָרֶץ... וַעֲזָבַנִי וְהֵפֵר אֶת בְּרִיתִי׳ – יֵלֵךְ אַחֲרֵי אֱלֹהִים אֲחֵרִים וְיָפֵר אֶת הַבְּרִית שֶׁכָּרַת עִם ה׳.
וּמָה תִּהְיֶה הַתְּגוּבָה? ׳וְחָרָה אַפִּי בוֹ בַּיּוֹם הַהוּא וַעֲזַבְתִּים וְהִסְתַּרְתִּי פָנַי מֵהֶם, וְהָיָה לֶאֱכֹל, וּמְצָאֻהוּ רָעוֹת רַבּוֹת וְצָרוֹת׳.

📜 רַשִׁ״י
׳וְהִסְתַּרְתִּי פָנַי׳ – כְּמוֹ שֶׁאֵינִי רוֹאֶה בְּצָרָתָם.

׳הֶסְתֵּר פָּנִים׳ הוּא הַבִּטּוּי שֶׁמּוֹפִיעַ כָּאן פַּעֲמַיִם: ה׳ מִתְרַחֵק כִּבְיָכוֹל, וּמַשְׁגִּיחַ עָלֵינוּ בְּאֹפֶן נִסְתָּר, בְּלִי נִסִּים גְּלוּיִים.
אַחֲרֵי הַצָּרוֹת הָעָם מַתְחִיל לְהַרְהֵר בְּלִבּוֹ: ׳הֲלֹא עַל כִּי אֵין אֱלֹהַי בְּקִרְבִּי מְצָאוּנִי הָרָעוֹת הָאֵלֶּה׳ – הֵם מְבִינִים מַדּוּעַ בָּאוּ הַצָּרוֹת.
וּבְכָל זֹאת: ׳וְאָנֹכִי הַסְתֵּר אַסְתִּיר פָּנַי בַּיּוֹם הַהוּא׳ – עוֹד הֶסְתֵּר.

💡 לָמָּה הֶסְתֵּר פָּנִים פַּעֲמַיִם?
׳דַּעַת מִקְרָא׳ מְבָאֵר: מִדָּה כְּנֶגֶד מִדָּה – הֵם פָּנוּ אֶל אֱלֹהִים אֲחֵרִים, וְלָכֵן הַהֶסְתֵּר גָּדֵל.
הָרַמְבַּ״ן מְבָאֵר אַחֶרֶת: הָעָם הִבִּיעַ חֲרָטָה, אֲבָל עוֹד לֹא הִתְוַדָּה וִדּוּי שָׁלֵם. הַהֶסְתֵּר הַשֵּׁנִי אֵינוֹ בָּא לְהַעֲנִישׁ, אֶלָּא לְכַוֵּן אוֹתָם לִתְשׁוּבָה שְׁלֵמָה – שֶׁיָּשׁוּבוּ אֶל ה׳ בֶּאֱמֶת.

וְכָאן מַגִּיעַ הַתַּפְקִיד שֶׁל הַשִּׁירָה: ׳וְהָיָה כִּי תִמְצֶאןָ אֹתוֹ רָעוֹת רַבּוֹת וְצָרוֹת, וְעָנְתָה הַשִּׁירָה הַזֹּאת לְפָנָיו לְעֵד, כִּי לֹא תִשָּׁכַח מִפִּי זַרְעוֹ׳.

📜 רַשִׁ״י
׳וְעָנְתָה הַשִּׁירָה הַזֹּאת לְפָנָיו לְעֵד׳ – שֶׁהִתְרֵיתִי בּוֹ בְּתוֹכָהּ עַל כָּל הַמּוֹצָאוֹת אוֹתָן.
׳כִּי לֹא תִשָּׁכַח מִפִּי זַרְעוֹ׳ – הֲרֵי זוֹ הַבְטָחָה לְיִשְׂרָאֵל, שֶׁאֵין תּוֹרָה מִשְׁתַּכַּחַת מִזַּרְעָם לְגַמְרֵי.

כְּשֶׁיָּבוֹאוּ הַזְּמַנִּים הַקָּשִׁים, יִזְכְּרוּ בְּנֵי יִשְׂרָאֵל אֶת הַשִּׁירָה וְיָבִינוּ: ה׳ יָדַע אֶת זֶה מֵרֹאשׁ, הוּא לֹא עָזַב אוֹתָנוּ – וְהוּא הִבְטִיחַ שֶׁבַּסּוֹף יְרַחֵם. וְהַהַבְטָחָה הַשְּׁנִיָּה מַפְלִיאָה עוֹד יוֹתֵר: הַתּוֹרָה לֹא תִּשָּׁכַח מֵעַם יִשְׂרָאֵל לְעוֹלָם.

📚 סִפּוּר: הַשָּׁעוֹן לְמַזְכֶּרֶת
שְׁנֵי חֲבֵרִים, יָרוֹן וְגִיל, נִפְרָדִים. אֶחָד עוֹבֵר לְעִיר אַחֶרֶת, וְהֵם רוֹצִים שֶׁהַקֶּשֶׁר לֹא יִתְנַתֵּק. אֶחָד מֵהֶם נוֹתֵן לַחֲבֵרוֹ שָׁעוֹן לְמַזְכֶּרֶת.
הַאִם זֶה רַעְיוֹן טוֹב? הַשָּׁעוֹן מַזְכִּיר, אֲבָל הוּא לֹא מְדַבֵּר. אֶפְשָׁר לְהַצִּיעַ עוֹד רַעְיוֹנוֹת: לִכְתֹּב מִכְתָּבִים, לְהִתְקַשֵּׁר, לְהִפָּגֵשׁ, לִלְמֹד יַחַד.
ה׳ בָּחַר בְּמַתָּנָה אַחֶרֶת לְגַמְרֵי: שִׁירָה שֶׁנִּמְצֵאת בַּפֶּה. הִיא לֹא נִשְׁכַּחַת, וְהִיא מַזְכִּירָה לָנוּ בְּכָל רֶגַע קָשֶׁה שֶׁה׳ אִתָּנוּ.
💛 אַהֲבַת הַשֵּׁם
גַּם כְּשֶׁנִּדְמֶה שֶׁה׳ מַסְתִּיר פָּנִים, הוּא עֲדַיִן כָּאן – רַק בְּלִי שֶׁאֲנִי רוֹאֶה. וְהַשִּׁירָה הִיא הַלַּחַשׁ שֶׁלּוֹ בְּאֹזְנִי: ׳אֲנִי לֹא עוֹזֵב אוֹתְךָ, גַּם כְּשֶׁקָּשֶׁה׳ 💛

כד וַיְהִי כְּכַלּוֹת מֹשֶׁה לִכְתֹּב אֶת דִּבְרֵי הַתּוֹרָה הַזֹּאת עַל סֵפֶר עַד תֻּמָּם

פִּתְחוּ סֵפֶר תּוֹרָה בְּפָרָשַׁת הַאֲזִינוּ, וְתִרְאוּ מִיָּד שֶׁמַּשֶּׁהוּ שׁוֹנֶה. הַשִּׁירָה אֵינָהּ כְּתוּבָה בְּשׁוּרוֹת מְלֵאוֹת וּרְצוּפוֹת כְּמוֹ שְׁאָר הַתּוֹרָה, אֶלָּא בְּמִבְנֶה מְיֻחָד, עִם רְוָחִים גְּדוֹלִים.

בַּתּוֹרָה יֵשׁ שְׁתֵּי שִׁירוֹת שֶׁכְּתוּבוֹת בְּצוּרָה מְיֻחֶדֶת: שִׁירַת הַיָּם (בְּפָרָשַׁת בְּשַׁלַּח) וְשִׁירַת הַאֲזִינוּ – וְיֵשׁ הֶבְדֵּל בֵּינֵיהֶן:
שִׁירַת הַאֲזִינוּ כְּתוּבָה ׳אָרִיחַ עַל גַּבֵּי אָרִיחַ׳ – שְׁנֵי טוּרִים יְשָׁרִים, אָרִיחַ מֵעַל אָרִיחַ וּבֵינֵיהֶם רֶוַח.
שִׁירַת הַיָּם כְּתוּבָה ׳אָרִיחַ עַל גַּבֵּי לְבֵנָה וּלְבֵנָה עַל גַּבֵּי אָרִיחַ׳ – כְּמוֹ קִיר לְבֵנִים, שֶׁבּוֹ כָּל שׁוּרָה מְכַסָּה אֶת הָרֶוַח שֶׁל הַשּׁוּרָה שֶׁמִּתַּחְתֶּיהָ.

💡 מִלּוֹן קָטָן
אָרִיחַ – לְבֵנָה קְצָרָה (חֲצִי לְבֵנָה).
לְבֵנָה – לְבֵנָה שְׁלֵמָה, אֲרֻכָּה.
כְּשֶׁבּוֹנִים קִיר, מְסַדְּרִים אוֹתָם לְסֵרוּגִין – וְכָךְ נִרְאֵית שִׁירַת הַיָּם.
🤔 לַמּוֹרֶה: לִבְנוֹת אֶת הַשִּׁירוֹת
כְּדַאי לְהַרְאוֹת לַתַּלְמִידִים סֵפֶר תּוֹרָה (אוֹ צִלּוּם) וּלְבַקֵּשׁ שֶׁיְּזַהוּ אֶת הַמִּבְנֶה הַמְּיֻחָד שֶׁל הַשִּׁירוֹת.
אֶפְשָׁר גַּם לִבְנוֹת מִלֶּגוֹ אֶת שְׁנֵי הַסּוּגִים: טוּרִים יְשָׁרִים לְהַאֲזִינוּ, וְקִיר לְבֵנִים לְשִׁירַת הַיָּם.

לָמָּה בִּכְלָל כּוֹתְבִים שִׁירָה אַחֶרֶת? כִּי שִׁירָה הִיא לֹא סְתָם טֶקְסְט. הַצּוּרָה הַמְּיֻחֶדֶת אוֹמֶרֶת לָעַיִן: עֲצֹר כָּאן, זֶה מְיֻחָד – כָּאן ה׳ מְדַבֵּר אֵלֶיךָ בְּשִׁיר.

💛 אַהֲבַת הַשֵּׁם
אֲפִלּוּ הַצּוּרָה שֶׁבָּהּ ה׳ צִוָּה לִכְתֹּב אֶת הַמִּלִּים שֶׁלּוֹ מְלֵאָה אַהֲבָה וְיֹפִי. אַבָּא שֶׁכּוֹתֵב לַיֶּלֶד שֶׁלּוֹ מִכְתָּב מְיֻחָד – מְעַצֵּב אוֹתוֹ יָפֶה 💛

וּבַפָּסוּק הַשְּׁלִישִׁי: ׳כִּי שֵׁם ה׳ אֶקְרָא, הָבוּ גֹדֶל לֵאלֹהֵינוּ׳. רַשִׁ״י מְבָאֵר: כְּשֶׁאֶקְרָא וְאַזְכִּיר אֶת שֵׁם ה׳ – אַתֶּם תְּנוּ גֹּדֶל לֵאלֹהֵינוּ וּבָרְכוּ שְׁמוֹ. מִכָּאן לָמְדוּ שֶׁעוֹנִין ׳בָּרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ׳ אַחֲרֵי בְּרָכָה.

בְּאֶמְצַע כָּל זֶה ה׳ מְחַזֵּק אֶת יְהוֹשֻׁעַ: ׳חֲזַק וֶאֱמָץ כִּי אַתָּה תָּבִיא אֶת בְּנֵי יִשְׂרָאֵל אֶל הָאָרֶץ... וְאָנֹכִי אֶהְיֶה עִמָּךְ׳.
אֶת הַמִּלִּים ׳חֲזַק וֶאֱמָץ׳ כְּבָר פָּגַשְׁנוּ בִּתְחִלַּת הַפֶּרֶק, כְּשֶׁמֹּשֶׁה אָמַר אוֹתָן לִיהוֹשֻׁעַ.
לָמָּה שׁוּב? כִּי עַכְשָׁיו, אַחֲרֵי שֶׁנֶּאֱמַר שֶׁיִּשְׂרָאֵל עֲתִידִים לַחְטֹא, יְהוֹשֻׁעַ צָרִיךְ חִזּוּק נוֹסָף – שֶׁיַּמְשִׁיךְ לְהוֹבִיל גַּם כְּשֶׁיִּהְיֶה קָשֶׁה. וְהַהַבְטָחָה הִיא הַחִזּוּק: ׳וְאָנֹכִי אֶהְיֶה עִמָּךְ׳.
💡 וְאֵיפֹה מַנִּיחִים אֶת סֵפֶר הַתּוֹרָה?
בַּפָּרָשָׁה יֵשׁ שְׁנֵי מַעֲשִׂים: א. כְּתִיבַת שִׁירַת הַאֲזִינוּ וְהוֹרָאָתָהּ – ׳שִׂימָהּ בְּפִיהֶם׳.
ב. הַנָּחַת הַתּוֹרָה הַכְּתוּבָה: ׳לָקֹחַ אֵת סֵפֶר הַתּוֹרָה הַזֶּה וְשַׂמְתֶּם אֹתוֹ מִצַּד אֲרוֹן בְּרִית ה׳ אֱלֹהֵיכֶם, וְהָיָה שָׁם בְּךָ לְעֵד׳.
הַתּוֹרָה שְׁמוּרָה בְּאָרוֹן – וְגַם שְׁמוּרָה בַּפֶּה. שְׁנֵי עֵדִים שֶׁלֹּא נִשְׁכָּחִים.
💛 אַהֲבַת הַשֵּׁם
בַּסּוֹף הַשִּׁירָה ה׳ מְפַיֵּס אֶת עַמּוֹ וְאֶת אַרְצוֹ, וְכָל הָעַמִּים מְשַׁבְּחִים אוֹתָנוּ עַל שֶׁנִּשְׁאַרְנוּ דְּבֵקִים בּוֹ. אֲנִי רוֹצֶה לִהְיוֹת כָּזֶה – יֶלֶד שֶׁדָּבֵק בַּה׳ תָּמִיד, כִּי ׳הוּא חַיֵּינוּ׳ 💛

✨ לְסִכּוּם

  • פָּרָשַׁת הַאֲזִינוּ הִיא הַפָּרָשָׁה הַחֲמִשִּׁים וְשָׁלֹשׁ מִתּוֹךְ חֲמִשִּׁים וְאַרְבַּע – הַפָּרָשָׁה שֶׁלִּפְנֵי הָאַחֲרוֹנָה, וְכֻלָּהּ שִׁירָה.
  • ה׳ צִוָּה: ׳כִּתְבוּ לָכֶם אֶת הַשִּׁירָה הַזֹּאת... שִׂימָהּ בְּפִיהֶם׳ – לֹא רַק לִכְתֹּב, אֶלָּא לָדַעַת בְּעַל פֶּה. רַשִׁ״י: ׳הַשִּׁירָה הַזֹּאת׳ הִיא מִ׳הַאֲזִינוּ הַשָּׁמַיִם׳ עַד ׳וְכִפֶּר אַדְמָתוֹ עַמּוֹ׳.
  • הַשִּׁירָה הִיא עֵדָה: כְּשֶׁיָּבוֹאוּ צָרוֹת וְ׳הֶסְתֵּר פָּנִים׳ – הַשְׁגָּחָה נִסְתֶּרֶת בְּלִי נִסִּים גְּלוּיִים – יִזְכְּרוּ שֶׁה׳ יָדַע מֵרֹאשׁ וְהִבְטִיחַ לְרַחֵם.
  • בְּשִׁירָה יֵשׁ תַּקְבִּילוֹת (׳הַאֲזִינוּ׳–׳וְתִשְׁמַע׳) וּדִמּוּיִים (הַתּוֹרָה כְּמוֹ מָטָר וְכַטַּל). שָׁמַיִם וָאָרֶץ הֵם הָעֵדִים, ׳עֵדִים שֶׁהֵן קַיָּמִים לְעוֹלָם׳.
  • ה׳ שָׁמַר עָלֵינוּ בַּמִּדְבָּר ׳כְּאִישׁוֹן עֵינוֹ׳, וְ׳כְּנֶשֶׁר יָעִיר קִנּוֹ׳: מֵעִיר בַּעֲדִינוּת, מְרַחֵף נוֹגֵעַ וְאֵינוֹ נוֹגֵעַ, וְנוֹשֵׂא עַל כְּנָפָיו – ׳מוּטָב שֶׁיִּכָּנֵס הַחֵץ בִּי וְלֹא בִּבְנִי׳.
  • ׳וַיִּשְׁמַן יְשֻׁרוּן וַיִּבְעָט׳ – הַשֶּׁפַע הוּא נִסָּיוֹן, וְצָרִיךְ לִזְכֹּר מִי נָתַן אוֹתוֹ. וּבְכָל זֹאת ה׳ ׳מִתְנֶחָם׳ – כְּשֶׁהוּא רוֹאֶה ׳כִּי אָזְלַת יָד׳, הוּא מִתְמַלֵּא רַחֲמִים וּמוֹשִׁיעַ.
  • שִׁירַת הַאֲזִינוּ כְּתוּבָה ׳אָרִיחַ עַל גַּבֵּי אָרִיחַ׳, וְשִׁירַת הַיָּם ׳אָרִיחַ עַל גַּבֵּי לְבֵנָה׳. בַּמִּקְדָּשׁ שָׁרוּ אוֹתָהּ בְּשִׁשָּׁה חֲלָקִים – ׳הַזִּי״ו לָ״ךְ׳ – וְזוֹ חֲלֻקַּת הָעֲלִיּוֹת עַד הַיּוֹם.
  • הַסִּיּוּם: ׳הַרְנִינוּ גוֹיִם עַמּוֹ׳ – הָאֻמּוֹת יְשַׁבְּחוּ אֶת יִשְׂרָאֵל שֶׁדָּבְקוּ בַּה׳ וְלֹא עֲזָבוּהוּ. וּמֹשֶׁה מְסַכֵּם: ׳כִּי לֹא דָבָר רֵק הוּא מִכֶּם, כִּי הוּא חַיֵּיכֶם׳.

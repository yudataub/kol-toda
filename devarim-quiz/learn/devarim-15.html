נִסָּיוֹן הַשֶּׁפַע

הַחֹמֶר הַמָּלֵא: בָּתִּים מְלֵאִים כָּל טוּב – וְלֵב שֶׁזּוֹכֵר מִי נָתַן אוֹתָם (פָּרָשַׁת וָאֶתְחַנַּן)
🎯 לַחִידוֹן🏠 כָּל הַחִידוֹנִים
בַּשִּׁעוּר הַזֶּה מֹשֶׁה רַבֵּנוּ מַזְהִיר אֶת בְּנֵי יִשְׂרָאֵל מִפְּנֵי סַכָּנָה רוּחָנִית מְיֻחֶדֶת: לֹא סַכָּנָה שֶׁל רָעָב וְלֹא שֶׁל מִלְחָמָה, אֶלָּא דַּוְקָא שֶׁל שֶׁפַע.
בְּאֶרֶץ יִשְׂרָאֵל יְחַכּוּ לָעָם עָרִים בְּנוּיוֹת, בָּתִּים מְלֵאִים כָּל טוּב, בּוֹרוֹת מַיִם חֲצוּבִים וּכְרָמִים נְטוּעִים – הַכֹּל מוּכָן, בְּלִי שׁוּם מַאֲמָץ. וְכַאֲשֶׁר יֵשׁ לָאָדָם הַכֹּל, קַל לוֹ לִשְׁכֹּחַ מִי נָתַן לוֹ.
נִלְמַד אֶת אַרְבַּע הַמַּתָּנוֹת שֶׁל ׳אֲשֶׁר לֹא׳, אֶת הָאַזְהָרוֹת שֶׁבָּאוֹת אַחֲרֵיהֶן, אֶת הָאִסּוּר לְנַסּוֹת אֶת ה׳, אֶת הַמִּצְוָה הַגְּדוֹלָה ׳וְעָשִׂיתָ הַיָּשָׁר וְהַטּוֹב׳, וְאֶת שְׁאֵלַת הַבֵּן הֶחָכָם.
וּבְעִקָּר נִלְמַד דָּבָר יָפֶה: כָּל הַשֶּׁפַע הַזֶּה הוּא מַתָּנָה שֶׁל אַהֲבָה. ה׳ נָתַן לָנוּ אוֹתוֹ בְּחֶסֶד, וְכָל מָה שֶׁהוּא מְבַקֵּשׁ מֵאִתָּנוּ הוּא שֶׁלֹּא נִשְׁכַּח אוֹתוֹ, וְנִזְכֹּר שֶׁהוּא אוֹהֵב אוֹתָנוּ 💛
בַּשִּׁעוּר: 🌧️ פְּתִיחָה: ז׳ בְּמַרְחֶשְׁוָן – לֵב שָׁבוּר וְלֵב שָׁלֵם 📋 צָרִיךְ אוֹ רוֹצֶה? – מָה אָדָם וּמָה עַם זְקוּקִים לוֹ 🏙️ אַרְבַּע מַתָּנוֹת – ׳אֲשֶׁר לֹא׳ ⚠️ הִשָּׁמֶר לְךָ פֶּן תִּשְׁכַּח – נִסָּיוֹן הַשֶּׁפַע 🔗 סִבָּה וְתוֹצָאָה – וְהָרַמְבַּ״ן עַל זִכְרוֹן יְמֵי הָעֹנִי 🙇 תִּירָא, תַּעֲבֹד, תִּשָּׁבֵעַ – וְלֹא אַחֲרֵי אֱלֹהֵי הַשְּׁכֵנִים 💧 לֹא תְנַסּוּ אֶת ה׳ – וְהָרַמְבַּ״ן 🛡️ שָׁמוֹר תִּשְׁמְרוּן – וְה׳ הוֹדֵף אֶת הָאוֹיְבִים 🤝 וְעָשִׂיתָ הַיָּשָׁר וְהַטּוֹב – לִפְנִים מִשּׁוּרַת הַדִּין 👦 כִּי יִשְׁאָלְךָ בִּנְךָ מָחָר – שְׁאֵלַת הַבֵּן הֶחָכָם 🌱 לְטוֹב לָנוּ כָּל הַיָּמִים – לָמָּה ה׳ נָתַן לָנוּ מִצְווֹת

🌧️פְּתִיחָה: ז׳ בְּמַרְחֶשְׁוָן – לֵב שָׁבוּר וְלֵב שָׁלֵם

הַשִּׁעוּר נִפְתָּח בְּתַאֲרִיךְ מְיֻחָד בְּלוּחַ הַשָּׁנָה: ז׳ בְּמַרְחֶשְׁוָן. מִן הַיּוֹם הַזֶּה מַתְחִילִים בְּאֶרֶץ יִשְׂרָאֵל לְבַקֵּשׁ גְּשָׁמִים בַּתְּפִלָּה – ׳וְתֵן טַל וּמָטָר לִבְרָכָה׳. אֲנַחְנוּ פּוֹנִים אֶל ה׳ וּמְבַקְּשִׁים מִמֶּנּוּ מַיִם, כִּי אֲנַחְנוּ יוֹדְעִים שֶׁרַק הוּא נוֹתֵן גֶּשֶׁם.

הַמִּשְׁנָה בְּמַסֶּכֶת תַּעֲנִית מְלַמֶּדֶת מָה עוֹשִׂים אִם עָבַר זְמַן רַב וְהַגֶּשֶׁם לֹא יוֹרֵד. בַּהַתְחָלָה יְחִידִים צַדִּיקִים מִתְעַנִּים, אַחַר כָּךְ כָּל הַצִּבּוּר, וְאִם עֲדַיִן לֹא יָרַד גֶּשֶׁם – יוֹצְאִים לִרְחוֹבָהּ שֶׁל עִיר וּמִתְפַּלְּלִים תְּפִלָּה גְּדוֹלָה בְּרַבִּים. גַּם הַיּוֹם, בְּשָׁנִים שֶׁל עֲצִירַת גְּשָׁמִים, עוֹרְכִים תְּפִלָּה מְיֻחֶדֶת עַל הַגְּשָׁמִים.

וְאֶת מִי בּוֹחֲרִים לִהְיוֹת הַשְּׁלִיחַ שֶׁעוֹבֵר לִפְנֵי הַתֵּבָה בַּתְּפִלָּה הַזֹּאת? לֹא אֶת בַּעַל הַקּוֹל הַיָּפֶה בְּיוֹתֵר, אֶלָּא אָדָם זָקֵן (בַּעַל נִסָּיוֹן חַיִּים), רָגִיל בַּתְּפִלָּה, שֶׁיֵּשׁ לוֹ בָּנִים וְשֶׁבֵּיתוֹ רֵיקָם – כְּדֵי שֶׁיְּהֵא לִבּוֹ שָׁלֵם בַּתְּפִלָּה.

🤔 שְׁנֵי לְבָבוֹת
אֶפְשָׁר לְצַיֵּר עַל הַלּוּחַ שְׁנֵי לְבָבוֹת: אֶחָד שָׁלֵם, שֶׁל אָדָם שָׂבֵעַ שֶׁיֵּשׁ לוֹ הַכֹּל, וְאֶחָד שָׁבוּר, שֶׁל אָדָם שֶׁחָסֵר לוֹ.
וְלִשְׁאֹל אֶת הַתַּלְמִידִים: מִי מֵהֶם יִתְפַּלֵּל מִכָּל הַלֵּב?
הָרַבִּי מִקּוֹצְק אָמַר: ׳אֵין דָּבָר שָׁלֵם יוֹתֵר מִלֵּב שָׁבוּר׳.

אָדָם שֶׁיֵּשׁ לוֹ הַכֹּל וְהוּא שָׂבֵעַ, אֵינוֹ מַרְגִּישׁ כָּל כָּךְ שֶׁהוּא תָּלוּי בַּה׳, וְהַזְּעָקָה שֶׁלּוֹ אֵינָהּ בָּאָה מִתּוֹךְ מַחְסוֹר אֲמִתִּי. מִי שֶׁחָסֵר לוֹ – מִתְפַּלֵּל מִכָּל הַלֵּב. וּמִי שֶׁיֵּשׁ לוֹ הַכֹּל, צָרִיךְ לְהִתְאַמֵּץ כְּדֵי לִזְכֹּר אֶת ה׳.
וְזוֹ בְּדִיּוּק הַשְּׁאֵלָה שֶׁל הַפְּסוּקִים שֶׁנִּלְמַד: מָה קוֹרֶה לָעָם כְּשֶׁפִּתְאוֹם יֵשׁ לוֹ הַכֹּל?

💛 אַהֲבַת הַשֵּׁם
גַּם כְּשֶׁהַבַּיִת שֶׁלִּי מָלֵא, אֲנִי רוֹצֶה שֶׁהַלֵּב שֶׁלִּי יִהְיֶה פָּתוּחַ אֶל ה׳ כְּמוֹ שֶׁל מִי שֶׁמְּבַקֵּשׁ גֶּשֶׁם. לֹא כִּי חָסֵר לִי – אֶלָּא כִּי אֲנִי אוֹהֵב אוֹתוֹ וְרוֹצֶה לְדַבֵּר אִתּוֹ 💛

📋צָרִיךְ אוֹ רוֹצֶה? – מָה אָדָם וּמָה עַם זְקוּקִים לוֹ

לִפְנֵי שֶׁנִּכָּנֵס לַפְּסוּקִים, נַחֲשֹׁב רֶגַע: יֵשׁ דְּבָרִים שֶׁאָדָם צָרִיךְ – אֹכֶל, מַיִם, בַּיִת, בְּגָדִים – בְּלִי שֶׁהֵם אִי אֶפְשָׁר לִחְיוֹת. וְיֵשׁ דְּבָרִים שֶׁאָדָם רַק רוֹצֶה – נָעִים לוֹ שֶׁיִּהְיוּ, אֲבָל אֶפְשָׁר גַּם בִּלְעֲדֵיהֶם.

🤔 לַמּוֹרֶה: חֲמִשָּׁה חֲלוֹנוֹת
אֶפְשָׁר לְבַקֵּשׁ מִכָּל תַּלְמִיד לִכְתֹּב דָּבָר אֶחָד בְּכָל כּוֹתֶרֶת:
דְּבָרִים שֶׁאֲנִי צָרִיךְ וְיֵשׁ לִי בְּלִי שֶׁבִּקַּשְׁתִּי.
דְּבָרִים שֶׁאֲנִי רוֹצֶה וְיֵשׁ לִי בְּלִי שֶׁבִּקַּשְׁתִּי.
דְּבָרִים שֶׁאֲנִי צָרִיךְ, בִּקַּשְׁתִּי וְקִבַּלְתִּי.
דְּבָרִים שֶׁאֲנִי רוֹצֶה, בִּקַּשְׁתִּי וְקִבַּלְתִּי.
דְּבָרִים שֶׁאֵין לִי וְהָיִיתִי רוֹצֶה לְקַבֵּל.
וְאַחַר כָּךְ לָדוּן: מָתַי הַהַרְגָּשָׁה נְעִימָה יוֹתֵר – כְּשֶׁמְּקַבְּלִים בְּלִי לְבַקֵּשׁ, אוֹ אַחֲרֵי בַּקָּשׁוֹת? וּמָה מַרְגִּישׁ מִי שֶׁרוֹצֶה וְלֹא מְקַבֵּל?

וְשֶׁאֱלָה גְּדוֹלָה: אֵיךְ מַרְגִּישׁ אָדָם שֶׁיֵּשׁ לוֹ הַכֹּל וְאֵינוֹ צָרִיךְ לְבַקֵּשׁ כְּלוּם? סָבִיר שֶׁהַתַּלְמִידִים יֹאמְרוּ שֶׁאֵין מְצִיאוּת כָּזֹאת: לְכָל אֶחָד חָסֵר מַשֶּׁהוּ. כֻּלָּנוּ נִבְרֵאנוּ חֲסֵרִים.
וְלָמָּה ה׳ בָּרָא אוֹתָנוּ כָּךְ? כְּדֵי שֶׁנִּפְנֶה אֵלָיו. הַחִסָּרוֹן הוּא מַתָּנָה: הוּא זֶה שֶׁמְּחַבֵּר אוֹתִי אֶל אַבָּא שֶׁבַּשָּׁמַיִם, וְגַם אֶל הָאֲנָשִׁים שֶׁסְּבִיבִי, כִּי כֻּלָּנוּ זְקוּקִים זֶה לָזֶה.

וְלֹא רַק אָדָם פְּרָטִי – גַּם עַם זָקוּק לִדְבָרִים. עַם צָרִיךְ מָקוֹם לָגוּר בּוֹ, מַיִם, אֹכֶל וּבִטָּחוֹן.
וּלְמָה בְּדִיּוּק הִזְדַּקֵּק עַם יִשְׂרָאֵל כְּשֶׁיָּצָא מִן הַמִּדְבָּר וְנִכְנַס לָאָרֶץ? לְעָרִים לָגוּר בָּהֶן, לְבָתִּים עִם אֹכֶל, לְבוֹרוֹת מַיִם, וּלְעֵצִים שֶׁנּוֹתְנִים פֵּרוֹת. בְּדִיּוּק אֶת אַרְבַּעַת הַדְּבָרִים הָאֵלֶּה ה׳ נוֹתֵן לָהֶם מוּכָנִים!

💛 אַהֲבַת הַשֵּׁם
ה׳ יוֹדֵעַ בְּדִיּוּק מָה אֲנִי צָרִיךְ, עוֹד לִפְנֵי שֶׁבִּקַּשְׁתִּי. הוּא הֵכִין לְעַמּוֹ עָרִים וּבָתִּים וּמַיִם – וְגַם לִי הוּא מֵכִין כָּל יוֹם אֶת מָה שֶׁאֲנִי צָרִיךְ, מִתּוֹךְ אַהֲבָה 💛

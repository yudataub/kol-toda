🌱 רַשִׁ״י: ׳וְלֹא כַּעֲשָׂרָה שְׁבָטִים׳

׳וַיֹּאמֶר לִיהוּדָה לִדְרוֹשׁ אֶת ה׳ אֱלֹהֵי אֲבוֹתֵיהֶם׳. רַשִׁ״י מְבָאֵר: ׳וְלֹא כַּעֲשָׂרָה שְׁבָטִים שֶׁדָּרְשׁוּ לֵאלֹהִים אֲחֵרִים חֲדָשִׁים מִקָּרוֹב בָּאוּ, וְאֶת אֱלֹהֵי אֲבוֹתָם לֹא דָרְשׁוּ׳. הַמִּלָּה ׳אֲבוֹתֵיהֶם׳ מְלַמֶּדֶת: הַדֶּרֶךְ שֶׁל אָסָא הִיא הַדֶּרֶךְ שֶׁל אַבְרָהָם יִצְחָק וְיַעֲקֹב.

לִדְרשׁ אֶת ה׳ שֶׁל הָאָבוֹת – לֹא דְּבָרִים חֲדָשִׁים וְזָרִים.

🤝 רַשִׁ״י: ׳בְּשֶׁדָּרַשְׁנוּ – הִנִּיחַ לָנוּ׳

׳כִּי דָרַשְׁנוּ אֶת ה׳ אֱלֹהֵינוּ דָּרַשְׁנוּ וַיָּנַח לָנוּ מִסָּבִיב׳. רַשִׁ״י מְבָאֵר: ׳לְפִי שֶׁדְּרָשְׁנוּהוּ הִנִּיחַ לָנוּ מִסָּבִיב׳. וְעַל ׳עוֹדֶנּוּ הָאָרֶץ לְפָנֵינוּ׳ רַשִׁ״י: ׳בְּעוֹד שֶׁהַמַּלְכוּת נְכוֹנָה לְפָנֵינוּ׳, וּמְצוּדַת דָּוִד: ׳נְחַזֵּק הֶעָרִים בְּעוֹדָם לְפָנֵינוּ עַד לֹא בָּא הָאוֹיֵב׳. כְּלוֹמַר: מְכִינִים כְּשֶׁשָּׁקֵט.

הַשֶּׁקֶט הוּא הַזְּמַן הֲכִי טוֹב לְהִתְכּוֹנֵן.

⚖️ רַשִׁ״י: ׳אֵין עִמְּךָ לַעְזוֹר׳

רַשִׁ״י מְבָאֵר אֶת הַתְּפִלָּה: ׳אַף עַל פִּי שֶׁאָנוּ מְעַטִּים וְהֵם רַבִּים כְּחוֹל הַיָּם, וְאֵין כֹּחַ לָנוּ בִּפְנֵיהֶם – עִמְּךָ לַעְזוֹר בֵּין רַב לְאֵין כֹּחַ. כְּלוֹמַר בַּיָּדוּעַ שֶׁיֵּשׁ לְךָ כֹּחַ לַעְזוֹר, לְפִיכָךְ עָזְרֵנוּ׳. אָסָא מַתְחִיל בְּמָה שֶׁהוּא יוֹדֵעַ: ה׳ יָכוֹל.

הַתְּפִלָּה הַטּוֹבָה מַתְחִילָה בְּמָה שֶׁאֲנִי יוֹדֵעַ עַל ה׳.

🕯️

📜 הַפָּסוּקוְיָמִים רַבִּים לְיִשְׂרָאֵל לְלֹא אֱלֹהֵי אֱמֶת וּלְלֹא כֹּהֵן מוֹרֶה וּלְלֹא תוֹרָה

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה עַל הַמּוֹרִים שֶׁמְּלַמְּדִים אוֹתִי תּוֹרָה 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק ט״ו, פָּסוּק ג׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בְּבִטָּחוֹן

🕯️ רַשִׁ״י: ׳לְלֹא כֹּהֵן מוֹרֶה׳

עֲזַרְיָהוּ מְתָאֵר עָבָר: ׳וְיָמִים רַבִּים לְיִשְׂרָאֵל לְלֹא אֱלֹהֵי אֱמֶת וּלְלֹא כֹּהֵן מוֹרֶה וּלְלֹא תוֹרָה׳. רַשִׁ״י: ׳שֶׁאֵין מִי שֶׁשּׁוֹאֵל לָהֶם שׁוּם דְּבַר תּוֹרָה׳. וּמְצוּדַת דָּוִד: ׳עַל הַכֹּהֵן לְהוֹרוֹת דַּרְכֵי ה׳... וּלְלֹא תוֹרָה – לֹא יַחְפְּצוּ בְּתוֹרַת ה׳׳. כְּשֶׁאֵין מוֹרֶה וְתוֹרָה, הָעָם אָבוּד.

בְּלִי מוֹרֶה וּבְלִי תּוֹרָה הָעָם אָבוּד – לָכֵן יֵשׁ לִשְׁמֹר עַל לִמּוּד הַתּוֹרָה.

🔙 רַשִׁ״י: ׳וַיָּשָׁב בַּצַּר לוֹ׳

׳וַיָּשָׁב בַּצַּר לוֹ עַל ה׳ אֱלֹהֵי יִשְׂרָאֵל וַיְבַקְשֻׁהוּ וַיִּמָּצֵא לָהֶם׳. רַשִׁ״י: ׳עַתָּה כְּשֶׁבָּאָה צָרָה לָהֶם בִּקְשׁוּ עַל ה׳... לְפִיכָךְ נִמְצָא לָהֶם׳. וּמְצוּדַת דָּוִד: ׳כַּאֲשֶׁר יִהְיֶה צַר לְיִשְׂרָאֵל יָשׁוּב אֶל ה׳, וְכַאֲשֶׁר יְבַקְשֶׁהוּ יִמָּצֵא לָהֶם׳. הַצָּרָה יְכוֹלָה לִהְיוֹת פֶּתַח לַחֲזָרָה.

גַּם צָרָה יְכוֹלָה לְהַחֲזִיר אוֹתָנוּ לַה׳.

📜 הַפָּסוּק…בְּהִשָּׁעֶנְךָ עַל מֶלֶךְ אֲרָם וְלֹא נִשְׁעַנְתָּ עַל ה׳ אֱלֹהֶיךָ

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי מִתְבַּלְבֵּל, אֲנִי זוֹכֵר אֵיךְ ה׳ עָזַר לִי 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק ט״ז, פָּסוּק ז׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּבִטָּחוֹן

⚠️ רַשִׁ״י: ׳בְּהִשָּׁעֶנְךָ עַל מֶלֶךְ אֲרָם׳

שָׁנִים רַבּוֹת אַחַר כָּךְ, כְּשֶׁבַּעְשָׁא מֶלֶךְ יִשְׂרָאֵל בָּנָה אֶת הָרָמָה, אָסָא שָׁלַח כֶּסֶף וְזָהָב לְבֶן הֲדַד מֶלֶךְ אֲרָם. הַנָּבִיא חֲנָנִי אוֹמֵר: ׳בְּהִשָּׁעֶנְךָ עַל מֶלֶךְ אֲרָם וְלֹא נִשְׁעַנְתָּ עַל ה׳ אֱלֹהֶיךָ׳. רַשִׁ״י מְבָאֵר: ׳אִם בָּטַחְתָּ בַּקָּדוֹשׁ בָּרוּךְ הוּא כַּאֲשֶׁר בָּטַחְתָּ בּוֹ כְּשֶׁבָּאוּ עָלֶיךָ הַכּוּשִׁים – לֹא נִמְלַט מִיָּדְךָ׳. הַנָּבִיא מַזְכִּיר לוֹ אֶת הַנִּצָּחוֹן הָרִאשׁוֹן.

הַנָּבִיא מְזַכֵּר לְאָסָא מָה עָזַר לוֹ בְּעָבָר.

👁️ מְצוּדַת דָּוִד: ׳עֵינָיו מְשֹׁטְטוֹת בְּכָל הָאָרֶץ׳

חֲנָנִי מוֹסִיף: ׳כִּי ה׳ עֵינָיו מְשֹׁטְטוֹת בְּכָל הָאָרֶץ לְהִתְחַזֵּק עִם לְבָבָם שָׁלֵם אֵלָיו׳. מְצוּדַת דָּוִד: ׳לָתֵת חוֹזֶק עִם אֲשֶׁר לְבָבָם שָׁלֵם אֵלָיו לִבְטֹחַ בּוֹ׳. ה׳ מְחַפֵּשׁ אֶת מִי שֶׁלִּבּוֹ שָׁלֵם – כְּדֵי לְחַזֵּק אוֹתוֹ. וּבְסוֹף אָסָא חָלָה בְּרַגְלָיו ׳וְגַם בְּחָלְיוֹ לֹא דָרַשׁ אֶת ה׳ כִּי בָרֹפְאִים׳ – מְצוּדַת דָּוִד: ׳שָׂם בִּטְחוֹנוֹ בָּרוֹפְאִים לְבַדָּם׳.

ה׳ מְחַפֵּשׁ לֵב שָׁלֵם כְּדֵי לְחַזֵּק אוֹתוֹ. וְהַשְּׁגִיאָה הִיא לִבְטֹחַ בָּרוֹפְאִים ׳לְבַדָּם׳, בְּלִי לְהִתְפַּלֵּל.

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְפָרְשִׁים מַסְבִּירִים: ה׳ עוֹזֵר כְּשֶׁאֲנַחְנוּ בּוֹטְחִים בּוֹ. בְּסוֹף יָמָיו אָסָא שָׁכַח אֶת הַלֶּקַח, וְהַנָּבִיא מְזַכֵּר לוֹ בְּאַהֲבָה: בְּשָׁעָה שֶׁבָּטַחְתָּ בַּה׳ – נִצַּחְתָּ.

  • אָסָא אָמַר לַה׳: ׳אֵין עִמְּךָ לַעְזוֹר בֵּין רַב לְאֵין כֹּחַ׳. מָתַי הִרְגַּשְׁתִּי חָלָשׁ וְקָרָאתִי לַה׳?
  • ׳דָּרַשְׁנוּ וַיָּנַח לָנוּ׳ – אֵיךְ אֲנִי יָכוֹל לְנַצֵּל זְמַן שָׁקֵט כְּדֵי לְהִתְקָרֵב לַה׳?
  • הַנָּבִיא חֲנָנִי הִזְכִּיר לְאָסָא אֶת הַנִּצָּחוֹן הָרִאשׁוֹן. מָה אֲנִי רוֹצֶה לִזְכֹּר מִמָּקוֹם שֶׁבּוֹ ה׳ עָזַר לִי?

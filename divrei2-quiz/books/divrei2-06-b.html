יְהוֹשָׁפָט וְהָעָם קוֹרְאִים אֶל ה׳ מוּל צָבָא עֲצוּם, וְהַלְוִיִּם יוֹצְאִים לִפְנֵי הַצָּבָא בְּשִׁיר – וה׳ מְנַצֵּחַ בִּמְקוֹמָם

📜 הַפָּסוּק…וַתִּתְּנָהּ לְזֶרַע אַבְרָהָם אֹהַבְךָ לְעוֹלָם

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אוֹתָנוּ, וּמַתָּנוֹתָיו אֵינָן נֶעֱלָמוֹת 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק כ׳, פָּסוּק ז׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּתְּפִלָּה וּבַשִּׁיר

🎁 רַשִׁ״י: מַתָּנָה לְאוֹהֵב

׳וַתִּתְּנָהּ לְזֶרַע אַבְרָהָם אֹהַבְךָ לְעוֹלָם׳ – רַשִׁ״י מְבָאֵר: ׳לְפִיכָךְ עָלֶיךָ לְהַחֲזִיקָהּ בְּיָדָם... כִּי אֲפִלּוּ מֶלֶךְ וַאֲפִלּוּ שׁוּם אָדָם כְּשֶׁנָּתַן מַתָּנָה לְאוֹהֲבוֹ וּבָא אַחֵר לְגָזְלָהּ מִמֶּנּוּ – דּוֹחֵק בְּעַצְמוֹ לְהַחֲזִיקָהּ בְּיָדוֹ׳. יְהוֹשָׁפָט מִתְפַּלֵּל כְּבֵן אֶל אָבִיו: אַתָּה נָתַתָּ לָנוּ אֶת הָאָרֶץ מֵאַהֲבָתְךָ.

ה׳ נָתַן לָנוּ אֶת הָאָרֶץ מֵאַהֲבָה, וה׳ אֵינוֹ מַפְקִיר אֶת מַתָּנָתוֹ.

1

📜 הַפָּסוּק…נַעַמְדָה לִפְנֵי הַבַּיִת הַזֶּה וּלְפָנֶיךָ כִּי שִׁמְךָ בַּבַּיִת הַזֶּה וְנִזְעַק אֵלֶיךָ מִצָּרָתֵנוּ וְתִשְׁמַע וְתוֹשִׁיעַ

💛 אַהֲבַת הַשֵּׁם: אֲנִי יוֹדֵעַ שֶׁתְּפִלּוֹת הָאָבוֹת עוֹזְרוֹת לִי גַּם הַיּוֹם 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק כ׳, פָּסוּק ט׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּתְּפִלָּה וּבַשִּׁיר

🏠 רַשִׁ״י: ׳נַעַמְדָה לִפְנֵי הַבַּיִת הַזֶּה׳

יְהוֹשָׁפָט מַזְכִּיר אֶת הַבְּטָחַת שְׁלֹמֹה: ׳אִם תָּבוֹא עָלֵינוּ רָעָה חֶרֶב שְׁפוֹט וְדֶבֶר וְרָעָב נַעַמְדָה לִפְנֵי הַבַּיִת הַזֶּה... וְנִזְעַק אֵלֶיךָ מִצָּרָתֵנוּ וְתִשְׁמַע וְתוֹשִׁיעַ׳. רַשִׁ״י: ׳בְּבֵית הַמִּקְדָּשׁ׳, וְ׳כַּאֲשֶׁר הִתְפַּלֵּל לְפָנֶיךָ שְׁלֹמֹה וְהִבְטַחְתּוֹ לַעֲשׂוֹת כָּךְ׳. הַתְּפִלָּה שֶׁל שְׁלֹמֹה בַּחֲנֻכַּת הַבַּיִת – הִיא עַכְשָׁיו מַמָּשׁ עוֹזֶרֶת.

הַתְּפִלָּה שֶׁל שְׁלֹמֹה בַּחֲנֻכַּת הַבַּיִת עוֹזֶרֶת גַּם לְיוֹם צָרָה.

2

📜 הַפָּסוּק…וַאֲנַחְנוּ לֹא נֵדַע מַה נַּעֲשֶׂה כִּי עָלֶיךָ עֵינֵינוּ

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁיֵּשׁ קֹשִׁי, אֲנִי לֹא בּוֹרֵחַ אֶלָּא נִשְׁעָן עַל ה׳ 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק כ׳, פְּסוּקִים י״ב–י״ג (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּתְּפִלָּה וּבַשִּׁיר

👀 רַשִׁ״י: ׳כִּי עָלֶיךָ עֵינֵינוּ׳

׳כִּי אֵין בָּנוּ כֹּחַ... וַאֲנַחְנוּ לֹא נֵדַע מַה נַּעֲשֶׂה כִּי עָלֶיךָ עֵינֵינוּ׳. רַשִׁ״י: ׳לְהוֹשִׁיעֵנוּ׳. וְעַל ׳גַּם טַפָּם נְשֵׁיהֶם וּבְנֵיהֶם׳ רַשִׁ״י: ׳כְּמִנְהָג שֶׁהַגְּדוּדִים בָּאִים עַל הָאָרֶץ – בּוֹרְחִים מִן הַכְּפָרִים לְעָרֵי מִבְצָרָם טַפָּם נְשֵׁיהֶם וּבְנֵיהֶם׳. אֲבָל כָּאן הֵם לֹא בּוֹרְחִים, אֶלָּא עוֹמְדִים כֻּלָּם לִפְנֵי ה׳.

לֹא בּוֹרְחִים מִן הַצָּרָה אֶלָּא עוֹמְדִים לִפְנֵי ה׳.

3

📜 הַפָּסוּק…אַתֶּם אַל תִּירְאוּ וְאַל תֵּחַתּוּ מִפְּנֵי הֶהָמוֹן הָרָב הַזֶּה כִּי לֹא לָכֶם הַמִּלְחָמָה כִּי לֵאלֹהִים

💛 אַהֲבַת הַשֵּׁם: הַבִּטָּחוֹן בַּה׳ הוּא הַמָּגֵן שֶׁלִּי 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק כ׳, פָּסוּק ט״ו (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּתְּפִלָּה וּבַשִּׁיר

🛑 רַשִׁ״י: ׳אַל תִּירְאוּ... כִּי לֹא לָכֶם הַמִּלְחָמָה׳

רַשִׁ״י מְבָאֵר אֶת תְּשׁוּבַת הַנָּבִיא: ׳אַף עַל פִּי שֶׁהֵם רַבִּים מִכֶּם אַל תִּירְאוּ מֵהֶם׳. וְעַל ׳כִּי לֹא לָכֶם הַמִּלְחָמָה׳: ׳וַדַּאי כַּאֲשֶׁר אֲמַרְתֶּם שֶׁעַל הַקָּדוֹשׁ בָּרוּךְ הוּא עֵינֵיכֶם וּבִטְחוֹנְכֶם – כֵּן יַעֲשֶׂה׳. כֹּל מִי שֶׁמְּדַבֵּר אֶל ה׳ בִּבְטָחוֹן, ה׳ עוֹנֶה.

ה׳ עוֹנֶה לְמִי שֶׁבּוֹטֵחַ בּוֹ.

4

📜 הַפָּסוּק…

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי מַאֲמִין, אֲנִי קָם בְּשִׂמְחָה לַעֲשׂוֹת מָה שֶׁצָּרִיךְ 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק כ׳, פָּסוּק י״ז (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּתְּפִלָּה וּבַשִּׁיר

🌅 רַשִׁ״י: ׳מָחָר צְאוּ... וְאֶלְחַם בַּעַדְכֶם׳

רַשִׁ״י מְבָאֵר אֶת ׳מָחָר צְאוּ לִפְנֵיהֶם וַה׳ עִמָּכֶם׳: ׳וְאֶלְחַם בַּעַדְכֶם׳, וְאֶת תְּגוּבַת הָעָם: ׳וְהֵם הֶאֱמִינוּ וְנִזְדָּרְזוּ וַיַּשְׁכִּימוּ בַבֹּקֶר׳. אָדָם שֶׁמַּאֲמִין לֹא מְחַכֶּה – הוּא קָם בַּבֹּקֶר וְיוֹצֵא לַדֶּרֶךְ.

אֱמוּנָה אֲמִתִּית גּוֹרֶמֶת לָקוּם בְּזְרִיזוּת.

5

📜 הַפָּסוּק…וְאֹמְרִים הוֹדוּ לַה׳ כִּי לְעוֹלָם חַסְדּוֹ

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת בַּחֲזִית שֶׁל הַהוֹדָיָה 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק כ׳, פָּסוּק כ״א (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּתְּפִלָּה וּבַשִּׁיר

🎼 מְצוּדַת דָּוִד: ׳לִפְנֵי הֶחָלוּץ׳

מְצוּדַת דָּוִד מְבָאֵר: ׳בְּעֵת שֶׁיָּצְאוּ לִפְנֵי הֶחָלוּץ יְשׁוֹרְרוּ וִיהַלְלוּ וִיהְיוּ אוֹמְרִים הוֹדוּ וְגוֹ׳׳, וְעַל ׳לְהַדְרַת קֹדֶשׁ׳: ׳עַל הַמָּקוֹם בָּרוּךְ הוּא, יֹאמַר שֶׁהוּא מְהֻדָּר וּמְפֹאָר בִּקְדֻשָּׁה׳. הַלְוִיִּם לֹא הָיוּ אַחֲרֵי הַצָּבָא אֶלָּא לְפָנָיו: הָרִאשׁוֹנִים לְהַלֵּל.

הַמְּהַלְלִים הוֹלְכִים בַּחֲזִית.

6

📜 הַפָּסוּק…כִּי שָׁם בֵּרֲכוּ אֶת ה׳

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁהַמָּקוֹם שֶׁאֲנִי נִמְצָא בּוֹ יִהְיֶה מָקוֹם שֶׁל תּוֹדָה 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק כ׳, פְּסוּקִים כ״ו; כ״ח (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּתְּפִלָּה וּבַשִּׁיר

🌸 רַשִׁ״י: ׳עֵמֶק בְּרָכָה׳

רַשִׁ״י מְבָאֵר: ׳נִקְהֲלוּ לְאוֹתוֹ הָעֵמֶק שֶׁלְּאַחַר כָּךְ נִקְרָא עֵמֶק בְּרָכָה עַל שֵׁם הַמַּעֲשֶׂה... כִּי שָׁם בֵּרֲכוּ אֶת ה׳ וְהוֹדוּ עַל כָּל הַטּוֹבָה אֲשֶׁר עָשָׂה לָהֶם׳. וְעַל ׳בִּנְבָלִים וּבְכִנֹּרוֹת וּבַחֲצֹצְרוֹת׳: ׳הֲרֵי שְׁלוֹשָׁה כְּלֵי שִׁיר כְּנֶגֶד שְׁלוֹשָׁה – כָּל אִישׁ יְהוּדָה וִירוּשָׁלַיִם וִיהוֹשָׁפָט׳. הַמָּקוֹם קִבֵּל שֵׁם שֶׁל תּוֹדָה.

מָקוֹם שֶׁבֵּרְכוּ בּוֹ אֶת ה׳ נִקְרָא לָעַד עַל שֵׁם הַבְּרָכָה.

💡

רָמָה ב׳ – עִיּוּן בַּתְּפִלָּה וּבַשִּׁיר

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְפָרְשִׁים מַסְבִּירִים אֶת הַהִגָּיוֹן שֶׁל יְהוֹשָׁפָט: ה׳ כְּבָר נָתַן לָנוּ אֶת הָאָרֶץ, וְהַמִּקְדָּשׁ נִבְנָה כְּדֵי שֶׁנָּבוֹא אֵלָיו בַּצָּרָה. לָכֵן הַתְּפִלָּה נִמְשֶׁכֶת בַּבִּטָּחוֹן, וְהַשִּׁיר בָּא לִפְנֵי הַנֵּס.

🤔 חוֹשְׁבִים עַל זֶה

  • יְהוֹשָׁפָט שָׁלַח אֶת הַמְשׁוֹרְרִים לִפְנֵי הַצָּבָא. אֵיזֶה שִׁיר אֲנִי יָכוֹל לָשִׁיר בְּיוֹם קָשֶׁה?
  • ׳כִּי עָלֶיךָ עֵינֵינוּ׳ – מָתַי הִרְגַּשְׁתִּי שֶׁאֲנִי לֹא יוֹדֵעַ מָה לַעֲשׂוֹת, וְהִתְפַּלַּלְתִּי?
  • ׳לֹא לָכֶם הַמִּלְחָמָה׳ – אֵיךְ אֲנִי יָכוֹל לְהַאֲמִין שֶׁה׳ עוֹזֵר, וְעֲדַיִן לַעֲשׂוֹת אֶת הַחֵלֶק שֶׁלִּי?

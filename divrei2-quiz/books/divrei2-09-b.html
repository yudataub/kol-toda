📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק כ״ט, פָּסוּק ג׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּטָהֳרָה וְשִׂמְחָה

🔧 רַשִׁ״י: ׳וַיְחַזְּקֵם׳

לָמָּה כָּתוּב ׳וַיְחַזְּקֵם׳, לֹא רַק ׳וַיִּפְתַּח׳? רַשִׁ״י מְבָאֵר: ׳לְפִי שֶׁאָחָז קָצַץ אֶת כָּל כְּלֵי בֵּית הָאֱלֹהִים׳. הַדְּלָתוֹת הָיוּ פְּגוּמוֹת. חִזְקִיָּהוּ לֹא רַק פָּתַח – הוּא תִּקֵּן וְחִזֵּק. לִפְעָמִים כְּדֵי לַחֲזֹר צָרִיךְ לְתַקֵּן אֶת מָה שֶׁנִּשְׁבַּר.

לַחֲזֹר זֶה לֹא רַק לִפְתֹּחַ – זֶה גַּם לְתַקֵּן.

📜 הַפָּסוּק…וְהוֹצִיאוּ אֶת הַנִּדָּה מִן הַקֹּדֶשׁ

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁבַּלֵּב שֶׁלִּי יִהְיֶה מָקוֹם רַק לְמָה שֶׁקָּדוֹשׁ 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק כ״ט, פָּסוּק ה׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּטָהֳרָה וְשִׂמְחָה

🧼 רַשִׁ״י: ׳וְהוֹצִיאוּ אֶת הַנִּדָּה׳

׳וְהוֹצִיאוּ אֶת הַנִּדָּה מִן הַקֹּדֶשׁ׳ – רַשִׁ״י מְבָאֵר: ׳עֲבוֹדָה זָרָה... מָאוּס יוֹתֵר מִן הַמֵּת הַמּוּטָל שָׁלֵם׳. הַמִּלָּה ׳נִדָּה׳ מְרַמֶּזֶת עַל דָּבָר מָאוּס מְאֹד; הַמֶּלֶךְ רוֹצֶה שֶׁהַיְּרֵאִים יְנַקּוּ אֶת הַמָּקוֹם מִכָּל מָה שֶׁאֵינוֹ שַׁיָּךְ.

מְנַקִּים מִן הַמָּקוֹם הַקָּדוֹשׁ כָּל מָה שֶׁאֵינוֹ שַׁיָּךְ אֵלָיו.

🕯️

📜 הַפָּסוּקגַּם סָגְרוּ דַּלְתוֹת הָאוּלָם וַיְכַבּוּ אֶת הַנֵּרוֹת וּקְטֹרֶת לֹא הִקְטִירוּ וְעֹלָה לֹא הֶעֱלוּ בַקֹּדֶשׁ לֵאלֹהֵי יִשְׂרָאֵל

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִפְתֹּחַ דְּלָתוֹת שֶׁל תּוֹרָה 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק כ״ט, פָּסוּק ז׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּטָהֳרָה וְשִׂמְחָה

🕯️ רַשִׁ״י: ׳וַיְכַבּוּ אֶת הַנֵּרוֹת׳

׳גַּם סָגְרוּ דַּלְתוֹת הָאוּלָם וַיְכַבּוּ אֶת הַנֵּרוֹת... וְעוֹלָה לֹא הֶעֱלוּ בַקֹּדֶשׁ לֵאלֹהֵי יִשְׂרָאֵל׳. רַשִׁ״י מְבָאֵר: ׳אֲבָל לֵאלֹהִים אֲחֵרִים עָשׂוּ בָּמוֹת לְכָל עִיר וָעִיר׳. כְּלוֹמַר, הַבַּיִת נִסְגַּר – אֲבָל הָעָם לֹא נִשְׁאַר בְּלִי עֲבוֹדָה; הוּא חִפֵּשׂ דְּרָכִים אֲחֵרוֹת, וְהַמֶּלֶךְ מַחֲזִיר אוֹתוֹ.

כְּשֶׁסּוֹגְרִים בַּיִת אֶחָד, אֲנָשִׁים מוֹצְאִים דְּרָכִים אֲחֵרוֹת – חָשׁוּב לִפְתֹּחַ אֶת הָאֱמֶת.

👨‍👦

📜 הַפָּסוּקבָּנַי עַתָּה אַל תִּשָּׁלוּ כִּי בָכֶם בָּחַר ה׳ לַעֲמֹד לְפָנָיו לְשָׁרְתוֹ

💛 אַהֲבַת הַשֵּׁם: אֲנִי חָשׁוּב לַה׳, וּפָעוּלָה שֶׁלִּי חֲשׁוּבָה 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק כ״ט, פָּסוּק י״א (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּטָהֳרָה וְשִׂמְחָה

👨‍👦 רַשִׁ״י: ׳אַל תִּשָּׁלוּ׳

׳בָּנַי עַתָּה אַל תִּשָּׁלוּ׳ – רַשִׁ״י מְבָאֵר: ׳לְשׁוֹן שְׁגָגָה׳. כְּלוֹמַר, ׳אַל תִּטְעוּ׳, ׳אַל תִּתְרַשְּׁלוּ׳. חִזְקִיָּהוּ מַזְהִיר אֶת הַלְוִיִּם בְּאַהֲבָה: ׳בָּכֶם בָּחַר ה׳ לַעֲמֹד לְפָנָיו׳. כָּל אֶחָד חָשׁוּב; אִם תַּחְשְׁבוּ ׳אֲנִי רַק אֶחָד׳ – תִּטְעוּ.

כָּל אֶחָד מֵאִתָּנוּ נִבְחַר – וְלָכֵן אֵין מָקוֹם לִטְעוֹת מִתּוֹךְ רְשָׁלָנוּת.

🧩

📜 הַפָּסוּקוַיָּבֹאוּ הַכֹּהֲנִים לִפְנִימָה בֵית ה׳ לְטַהֵר

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹשֶׂה אֶת הַחֵלֶק שֶׁלִּי בְּשִׂמְחָה 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק כ״ט, פָּסוּק ט״ז (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּטָהֳרָה וְשִׂמְחָה

🧩 רַשִׁ״י: מִי נִכְנַס פְּנִימָה

׳וַיָּבֹאוּ הַכֹּהֲנִים לִפְנִימָה בֵית ה׳ לְטַהֵר... לַחֲצַר בֵּית ה׳׳. רַשִׁ״י מְבָאֵר: ׳כִּי לֹא עַל הַלְוִיִּם לִכָּנֵס פְּנִימָה׳, וְהַלְוִיִּם קִבְּלוּ בַּחֲצֵר. כָּל קְבוּצָה עָשְׂתָה אֶת הַחֵלֶק שֶׁלָּהּ: כֹּהֲנִים בְּפְּנִים, לְוִיִּם בַּחוּץ. הַשְּׁמִירָה עַל הַסֵּדֶר הִיא חֵלֶק מִן הַטָּהֳרָה.

בַּעֲבוֹדַת ה׳ לְכָל אֶחָד יֵשׁ תַּפְקִיד, וּמְכַבְּדִים אֶת הַגְּבוּלוֹת.

🎼

📜 הַפָּסוּק…כִּי בְיַד ה׳ הַמִּצְוָה

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׁר כִּי ה׳ אִשֵּׁר אֶת הַשִּׁיר 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק כ״ט, פְּסוּקִים כ״ה–כ״ו (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּטָהֳרָה וְשִׂמְחָה

🎼 רַשִׁ״י: ׳בְּמִצְוַת דָּוִיד... כִּי בְיַד ה׳ הַמִּצְוָה׳

הַשִּׁיר בַּקָּרְבָּן אֵינוֹ כָּתוּב בַּתּוֹרָה. רַשִׁ״י מְבָאֵר: ׳לְשׁוֹרֵר הַלְוִיִּם בִּכְלֵי שִׁיר לֹא כָּתוּב בַּתּוֹרָה שֶׁיְּהֵא קָרְבָּן טָעוּן שִׁירָה, אֶלָּא מִצְוַת ה׳ שֶׁצִּוָּה בְּיַד נְבִיאָיו׳. כְּלוֹמַר, הַנְּבִיאִים דָּוִד, גָּד וְנָתָן הוֹסִיפוּ שִׁיר, וה׳ אִשֵּׁר. וְהַכֹּהֲנִים בַּחֲצֹצְרוֹת – ׳כְּכָתוּב וּבְנֵי אַהֲרֹן יִתְקְעוּ׳.

הַשִּׁיר הוּא מִצְוָה שֶׁקִּבַּלְנוּ מִן הַנְּבִיאִים.

📜 הַפָּסוּקוַיִּשְׂמַח יְחִזְקִיָּהוּ וְכָל הָעָם עַל הַהֵכִין הָאֱלֹהִים לָעָם כִּי בְּפִתְאֹם הָיָה הַדָּבָר

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ כִּי ה׳ מֵכִין אֶת הַלֵּב שֶׁלִּי 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק כ״ט, פָּסוּק ל״ו (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּטָהֳרָה וְשִׂמְחָה

😊 רַשִׁ״י: ׳עַל הַהֵכִין הָאֱלֹהִים׳

בְּסוֹף הַפֶּרֶק: ׳וַיִּשְׂמַח יְחִזְקִיָּהוּ וְכָל הָעָם עַל הַהֵכִין הָאֱלֹהִים לָעָם כִּי בְּפִתְאֹם הָיָה הַדָּבָר׳. רַשִׁ״י מְבָאֵר: ׳עַל הַהֵכִין הָאֱלֹהִים לָעָם – לַעֲבֹדוֹ וְלַעֲשׂוֹת רְצוֹנוֹ׳. כְּלוֹמַר, הַשִּׂמְחָה הִיא עַל כָּךְ שֶׁה׳ הֵכִין אֶת לֵב הָעָם. וְ׳בְּפִתְאֹם׳ – בִּמְהִירוּת, כִּי הַכֹּל קָרָה תּוֹךְ הַשָּׁנָה הָרִאשׁוֹנָה.

ה׳ מְסַיֵּעַ לִי לְהַכִּין אֶת הַלֵּב.

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְפָרְשִׁים מְדַיְּקִים מָה הָיָה קָשֶׁה בְּפִתּוּחַ הַבַּיִת: לְחַזֵּק אֶת מָה שֶׁנִּפְגַּע, לְטַהֵר, וְלִקְרֹא לְכֻלָּם לַחֲזֹר. וַאֲנַחְנוּ שׁוֹמְעִים שֶׁהַשִּׂמְחָה הָיְתָה גְּדוֹלָה כִּי ה׳ עַצְמוֹ הֵכִין אֶת הַלֵּב.

  • חִזְקִיָּהוּ פָּתַח אֶת הַדְּלָתוֹת בַּשָּׁנָה הָרִאשׁוֹנָה. מָה אֲנִי רוֹצֶה לְהַתְחִיל הַשָּׁבוּעַ בְּלִי לְחַכּוֹת?
  • הַמֶּלֶךְ קָרָא לַלְוִיִּם ׳בָּנַי׳. מָתַי מִישֶׁהוּ אָמַר לִי מִלָּה חַמָּה שֶׁהִקְרִיבָה אוֹתִי?
  • ׳טִהַרְנוּ אֶת כָּל בֵּית ה׳׳ – אֵיךְ אֲנִי יָכוֹל לְטַהֵר מַשֶּׁהוּ בַּבַּיִת, בְּעַצְמִי אוֹ בַּחֲבֵרוּת?

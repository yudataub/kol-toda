עָזְרֵנוּ ה׳ אֱלֹהֵינוּ

אָסָא קוֹרֵא אֶל ה׳ מוּל צָבָא עֲצוּם וּמְנַצֵּחַ – וְהַנָּבִיא מַזְכִּיר לוֹ שֶׁבְּבִטָּחוֹן בַּה׳ הַכֹּחַ

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק י״ד, פְּסוּקִים א׳–ב׳

רָמָה א׳ – ׳עָזְרֵנוּ ה׳ אֱלֹהֵינוּ׳

🪓 ׳וַיַּעַשׂ אָסָא הַטּוֹב וְהַיָּשָׁר׳

אָסָא מֶלֶךְ יְהוּדָה ׳וַיַּעַשׂ... הַטּוֹב וְהַיָּשָׁר בְּעֵינֵי ה׳ אֱלֹהָיו׳. הוּא מֵסִיר אֶת מִזְבְּחוֹת הַנֵּכָר וְהַבָּמוֹת, מְשַׁבֵּר אֶת הַמַּצֵּבוֹת וְגוֹדֵעַ אֶת הָאֲשֵׁרִים. הוּא לֹא מַמְתִּין שֶׁמִּישֶׁהוּ אַחֵר יַתְחִיל – הוּא מַתְחִיל בְּעַצְמוֹ.

אָסָא עָשָׂה הַטּוֹב וְהַיָּשָׁר בְּעֵינֵי ה׳, וְהֵסִיר אֶת מַה שֶׁמַּרְחִיק מִמֶּנּוּ.

1
📚

💛 אַהֲבַת הַשֵּׁם: לִלְמֹד תּוֹרָה זֶה הַדֶּרֶךְ שֶׁלִּי לָדַעַת מָה ה׳ רוֹצֶה 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק י״ד, פְּסוּקִים ג׳–ד׳

רָמָה א׳ – ׳עָזְרֵנוּ ה׳ אֱלֹהֵינוּ׳

📚 ׳לִדְרוֹשׁ אֶת ה׳... וְלַעֲשׂוֹת הַתּוֹרָה׳

אָסָא ׳אָמַר לִיהוּדָה לִדְרוֹשׁ אֶת ה׳ אֱלֹהֵי אֲבוֹתֵיהֶם וְלַעֲשׂוֹת הַתּוֹרָה וְהַמִּצְוָה׳. וּמָה קָרָה? ׳וַתִּשְׁקֹט הַמַּמְלָכָה לְפָנָיו׳. הָעָם הִתְחִיל לִלְמֹד, וְהָאָרֶץ הָיְתָה שְׁקֵטָה.

הַמֶּלֶךְ לִמֵּד אֶת הָעָם לִדְרשׁ אֶת ה׳ – וְכָךְ בָּא שֶׁקֶט.

2
🏰

💛 אַהֲבַת הַשֵּׁם: תּוֹדָה, ה׳, עַל הַיָּמִים הַשְּׁקֵטִים שֶׁנָּתַתָּ לִי 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק י״ד, פְּסוּקִים ה׳–ו׳

רָמָה א׳ – ׳עָזְרֵנוּ ה׳ אֱלֹהֵינוּ׳

🏰 ׳דָּרַשְׁנוּ וַיָּנַח לָנוּ׳

בַּשָּׁנִים הַשְּׁקֵטוֹת אָסָא בּוֹנֶה עָרֵי מְצוּרָה, חוֹמָה, מִגְדָּלִים וְדַלְתַיִם. הוּא אוֹמֵר לָעָם: ׳עוֹדֶנּוּ הָאָרֶץ לְפָנֵינוּ כִּי דָרַשְׁנוּ אֶת ה׳ אֱלֹהֵינוּ דָּרַשְׁנוּ וַיָּנַח לָנוּ מִסָּבִיב׳. כְּשֶׁיֵּשׁ שֶׁקֶט, מִשְׁתַּמְּשִׁים בּוֹ כְּדֵי לְהָכִין.

בְּיָמִים שְׁקֵטִים מְכִינִים אֶת עַצְמֵנוּ.

3
🌅

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁהַקֹּשִׁי גָּדוֹל, אֲנִי לֹא לְבַד 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק י״ד, פְּסוּקִים ז׳–ח׳

רָמָה א׳ – ׳עָזְרֵנוּ ה׳ אֱלֹהֵינוּ׳

🌅 צָבָא עֲצוּם

אֲבָל פִּתְאוֹם: ׳וַיֵּצֵא אֲלֵיהֶם זֶרַח הַכּוּשִׁי בְּחַיִל אֶלֶף אֲלָפִים וּמַרְכָּבוֹת שְׁלֹשׁ מֵאוֹת׳. מוּל צָבָא שֶׁל מִילְיוֹן, לְאָסָא יֵשׁ צָבָא קָטָן יוֹתֵר – שְׁלוֹשׁ מֵאוֹת אֶלֶף מִיהוּדָה וּמָאתַיִם וּשְׁמוֹנִים אֶלֶף מִבִּנְיָמִן. הַמִּסְפָּרִים לֹא בְּצַד שֶׁלּוֹ.

הָאוֹיֵב גָּדוֹל מִמֶּנּוּ בְּהַרְבֵּה.

4
🙏

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי נִשְׁעָן עַל ה׳, אֲנִי לֹא פּוֹחֵד 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק י״ד, פָּסוּק י׳

רָמָה א׳ – ׳עָזְרֵנוּ ה׳ אֱלֹהֵינוּ׳

🙏 ׳אֵין עִמְּךָ לַעְזוֹר בֵּין רַב לְאֵין כֹּחַ׳

אָסָא קוֹרֵא: ׳ה׳ אֵין עִמְּךָ לַעְזוֹר בֵּין רַב לְאֵין כֹּחַ עָזְרֵנוּ ה׳ אֱלֹהֵינוּ כִּי עָלֶיךָ נִשְׁעַנּוּ וּבְשִׁמְךָ בָאנוּ עַל הֶהָמוֹן הַזֶּה׳. אֵין הֶבְדֵּל לַה׳ בֵּין חָזָק לְחַלָּשׁ – הוּא עוֹזֵר לְמִי שֶׁנִּשְׁעָן עָלָיו.

אֵין הֶבְדֵּל לַה׳ בֵּין רַבִּים לִמְעַטִּים; הוּא עוֹזֵר לְמִי שֶׁנִּשְׁעָן עָלָיו.

5
🏃

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹמֵעַ אוֹתִי כְּשֶׁאֲנִי קוֹרֵא אֵלָיו 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק י״ד, פְּסוּקִים י״א–י״ג

רָמָה א׳ – ׳עָזְרֵנוּ ה׳ אֱלֹהֵינוּ׳

🏃 ׳וַיִּגֹּף ה׳ אֶת הַכּוּשִׁים׳

הַתְּשׁוּבָה מַגִּיעָה מִיָּד: ׳וַיִּגֹּף ה׳ אֶת הַכּוּשִׁים לִפְנֵי אָסָא וְלִפְנֵי יְהוּדָה וַיָּנֻסוּ הַכּוּשִׁים׳. הַצָּבָא הָעֲצוּם בּוֹרֵחַ, וְאָסָא וְהָעָם חוֹזְרִים עִם שָׁלָל. הַנִּצָּחוֹן הוּא שֶׁל ה׳.

ה׳ עָזַר לָהֶם, וְהָאוֹיֵב בָּרַח.

6
📣

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת עִם ה׳ – וְהוּא עִמִּי 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק ט״ו, פְּסוּקִים א׳–ב׳

רָמָה א׳ – ׳עָזְרֵנוּ ה׳ אֱלֹהֵינוּ׳

📣 עֲזַרְיָהוּ בֶּן עוֹדֵד: ׳ה׳ עִמָּכֶם בִּהְיוֹתְכֶם עִמּוֹ׳

כְּשֶׁאָסָא חוֹזֵר, יוֹצֵא לִקְרָאתוֹ הַנָּבִיא עֲזַרְיָהוּ בֶּן עוֹדֵד: ׳שְׁמָעוּנִי אָסָא... ה׳ עִמָּכֶם בִּהְיוֹתְכֶם עִמּוֹ וְאִם תִּדְרְשֻׁהוּ יִמָּצֵא לָכֶם וְאִם תַּעַזְבֻהוּ יַעֲזֹב אֶתְכֶם׳. ה׳ עִם מִי שֶׁעִמּוֹ.

ה׳ עִמָּנוּ כְּשֶׁאֲנַחְנוּ עִמּוֹ.

7
🎺

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לָשִׁיר וְלִשְׂמֹחַ בְּכָל לֵב 💛

📖 דִּבְרֵי הַיָּמִים ב, פֶּרֶק ט״ו, פְּסוּקִים ח׳; י״ב; י״ד–ט״ו

רָמָה א׳ – ׳עָזְרֵנוּ ה׳ אֱלֹהֵינוּ׳

🎺 ׳וַיָּבֹאוּ בַבְּרִית׳

אָסָא מִתְחַזֵּק, מַעֲבִיר אֶת הַשִּׁקּוּצִים וּמְחַדֵּשׁ אֶת מִזְבַּח ה׳. כָּל הָעָם נִקְבָּץ לִירוּשָׁלַיִם, זוֹבֵחַ, וְ׳בָּאוּ בַבְּרִית לִדְרשׁ אֶת ה׳ אֱלֹהֵי אֲבוֹתֵיהֶם בְּכָל לְבָבָם וּבְכָל נַפְשָׁם׳. הֵם נִשְׁבְּעוּ בְּקוֹל גָּדוֹל, בַּחֲצוֹצְרוֹת וּבְשׁוֹפָרוֹת, ׳וַיִּשְׂמְחוּ... כִּי בְכָל לְבָבָם נִשְׁבָּעוּ׳.

שִׂמְחָה שֶׁל הִתְחַיְּבוּת לֵב שָׁלֵם לַה׳.

מָה לוֹקְחִים מִרָמָה א׳?

אָסָא מֶלֶךְ יְהוּדָה מַעֲבִיר אֶת הָעֲבוֹדָה הַזָּרָה וּמְלַמֵּד אֶת הָעָם לִדְרשׁ אֶת ה׳. כְּשֶׁבָּא צָבָא עֲצוּם הוּא לֹא מִסְתַּכֵּל עַל הַמִּסְפָּרִים אֶלָּא קוֹרֵא אֶל ה׳: ׳עָזְרֵנוּ... כִּי עָלֶיךָ נִשְׁעַנּוּ׳. וה׳ עוֹזֵר.

  • הַנָּבִיא חֲנָנִי הִזְכִּיר לְאָסָא אֶת הַנִּצָּחוֹן הָרִאשׁוֹן. מָה אֲנִי רוֹצֶה לִזְכֹּר מִמָּקוֹם שֶׁבּוֹ ה׳ עָזַר לִי?

תְּפִלַּת הַמַּקְדִּימִים - מִזְמוֹר אֱמוּנָה טֶרֶם יְשׁוּעָה

תְּפִלַּת הַמַּקְדִּימִים - מִזְמוֹר אֱמוּנָה טֶרֶם יְשׁוּעָה

שַׁעַר רִאשׁוֹן: הַשִּׁירָה הַקּוֹדֶמֶת

רִבּוֹן כָּל הָעוֹלָמִים, אֵל הַהִשְׂתַּנּוּת וְהַנִּצְחִיּוּת

הִנֵּה נִשְׁמָתִי עוֹמֶדֶת לְפָנֶיךָ כְּאִשָּׁה לִפְנֵי יָם, מַבִּיטָה אֶל מַרְחֲבֵי הַתְּהוֹם, חֲרֵדָה וּבוֹטַחַת כְּאַחַת. הָיִיתִי כְּגֶבֶר – מַאֲמִין רַק כְּשֶׁרוֹאֶה, וְכַמָּה אֲנִי מִתְגַּעְגֵּעַ לִהְיוֹת כְּאִשָּׁה – מַאֲמִינָה טֶרֶם רְאִיָּה.

אָבִי, לַמְּדֵנִי סוֹד הַצִּפּוֹר הַמְזַמֶּרֶת עוֹד בַּחֲשֵׁכָה, הַמְּבַשֶּׂרֶת אוֹר יוֹם בְּעוֹד לַיְלָה אוֹפֵף; לַמְּדֵנִי סוֹד הַתֹּף הַנִּלְקָח מֵאֶרֶץ עֲבָדִים עַל מְנָת לְנַגֵּן עִמּוֹ בְּחוֹף הַיָּם הַנִּבְקָע.

הֵן בְּכָל תְּהוֹם בְּחַיַּי אַתָּה מַצְמִיחַ בְּקִיעָה, אַתָּה יוֹצֵר נְתִיב בְּתוֹךְ הַיָּם הַסּוֹעֵר. אַךְ אֲנִי, בְּקֹטֶן אֱמוּנָתִי, מְחַכֶּה לִרְאוֹת דֶּרֶךְ, מְחַכֶּה לְקַבֵּל אוֹת, מְחַכֶּה לִרְאוֹת מִצְרַיִם מֵתִים עַל שְׂפַת הַיָּם.

יְהִי רָצוֹן שֶׁתְּלַמְּדֵנִי, אָבִי, לַחְבֹּשׁ לֹא רַק בִּגְדֵי זָכָר, אֶלָּא גַּם אֶת אֱמוּנַת הַנָּשִׁים הַצִּדְקָנִיּוֹת. שֶׁאֶזְכֶּה לְזַמֵּר לְךָ עוֹד בְּתוֹךְ הֵיכְלֵי הַגָּלוּת, לְהָכִין לִי תֹּף וּמָחוֹל בְּלֵב שְׁבוּר וְאָפֵל.

שַׁעַר שֵׁנִי: הַקְּדִימָה הַמְּקֻדֶּשֶׁת

אֵל אָדוֹן, בּוֹחֵן לְבָבוֹת וּמַבִּיט בִּכְלָיוֹת

"יְדִידִים רוֹמְמוּךָ בְּשִׁירָה קִדְּמוּךָ" - מִלִּים צוֹרְבוֹת אֵשׁ בְּלִבִּי הַמְּהַסֵּס. קִדְּמוּךָ – אֵיךְ מְקַדְּמִים אֶת מִי שֶׁרוֹאֶה כָּל עָתִיד? הֵם הִקְדִּימוּ אֶת הַשִּׁירָה לַיְּשׁוּעָה, לֹא כָּמוֹנִי.

אָנָּא, תְּחַבֵּר אוֹתִי אֶל שֹׁרֶשׁ נִשְׁמַת הַנָּשִׁים, אֶל מִרְיָם, אֶל הַבּוֹטְחוֹת, אֶל יוֹדְעוֹת הַנֵּס טֶרֶם הִגָּלוֹתוֹ. בַּאֲשֶׁר אֲנִי עוֹמֵד, בְּקֹשִׁי, בְּיָגוֹן, בְּשִׁבְרוֹן לֵב, תֵּן בִּי כֹּחַ לְהַרְקִיד אֶת תּוֹפִי הַדּוֹמֵם.

לַמְּדֵנִי לִשְׁמֹעַ קוֹל בְּרָקִיעַ חָשׁוּךְ, קוֹל דּוֹדִי בֵּין הַבְּתָרִים, בֵּין רִגְבֵי לֵב שָׁבוּר. לַמְּדֵנִי לִרְקֹד טֶרֶם תִּתְגַּלֶּה בַּדֶּרֶךְ, לִשְׁמֹעַ שִׁירָה בַּחֲרִישִׁיּוּת הַלַּיְלָה.

אֵיךְ אָקוּם מֵאִישׁ לְאִשָּׁה, מֵרוֹאֶה לְיוֹדַעַת, מִמְּחַכֶּה לַנֵּס לִמְזַמֶּרֶת טֶרֶם הוֹפָעָתוֹ? אֵיךְ אֶרְפָּא מִתַּחֲלוּאֵי הַהַמְתָּנָה הַסּוֹעֶרֶת, וְאֶהְיֶה בְּתוֹךְ צַלְמִי הַגַּשְׁמִי לְבֶת מַלְכוּת שֶׁאֵינָהּ סָפֵק?

שַׁעַר שְׁלִישִׁי: תֹּף וּמְחוֹלוֹת בְּשַׁעֲרֵי הַזְּמַן

יוֹדֵעַ צְפוּנוֹת הַלֵּב, רוֹאֶה אֶת הַלַּהַט הַמְּכֻסֶּה בְּאֵפֶר

אֲנִי מְבַקֵּשׁ מִמְּךָ, בּוֹרֵא עוֹלָמִי, שׁוֹבֵר יַמִּים וּבוֹנֶה יַבָּשׁוֹת, תֵּן בִּי כֹּחַ לִבְנוֹת דֶּרֶךְ בְּיַם סוּף שֶׁל הַנֶּפֶשׁ, וְלָצֵאת מֵאִישׁ לְאִשָּׁה, מִנְּבִיעַ לִמְרַקֵּד.

עוֹד לִפְנֵי שֶׁאֶרְאֶה אֶת יְשׁוּעָתְךָ הַגְּדוֹלָה, עוֹד לִפְנֵי שֶׁאֶרְאֶה אֶת נֵס הַגְּאֻלָּה, עוֹד לִפְנֵי שֶׁתִּבָּקַע הַתְּהוֹם לִשְׁנַיִם, אֲהַלֵּל וַאֲזַמֵּר לְךָ בְּלֵב שָׂמֵחַ.

אָבִי, אֵין אָנוּ יוֹדְעִים כֵּיצַד לְקַבֵּל אֶת יְשׁוּעָתְךָ, יוֹדְעִים רַק אֶת גָּלוּתֵנוּ הַמָּרָה, אֶת מַשָּׂא הַיּוֹם. אַךְ אָנָּא, תֵּן בָּנוּ כֹּחַ לְסַגֵּל אֶת אֱמוּנַת מִרְיָם, לֶאֱחֹז בְּתֹף עוֹד בְּטֶרֶם הַיָּם נִבְקָע.

מִי יִתֵּן וְאֶזְכֶּה לְשָׁמְרִי שַׁבָּת כְּכַלָּה, בְּעוֹד הַחֹל מִסָּבִיב מְכַלֶּה מְלַאכְתּוֹ. כִּי סוֹד הָאֱמוּנָה הַנְּשִׁית הִיא הַיְכֹלֶת לִמְצֹא מְנוּחָה בְּעוֹד הַמְּנוּחָה לֹא בָאָה לָעוֹלָם.

שַׁעַר רְבִיעִי: הַשְּׂמֵחִים בְּטֶרֶם שִׂמְחָה

אֵל חַנּוּן, צוּר הַשִּׁירָה וְהַנְּגִינָה

"בְּשׁוּב ה' אֶת שִׁיבַת צִיּוֹן, הָיִינוּ כְּחֹלְמִים", וְהֵם יִשְׁאֲלוּ: "הִגְדִּיל ה' לַעֲשׂוֹת עִם אֵלֶּה?" וְהַתְּשׁוּבָה הַמַּפְלִיאָה תִּהְיֶה: "הִגְדִּיל ה' לַעֲשׂוֹת עִמָּנוּ, הָיִינוּ שְׂמֵחִים".

הָיִינוּ שְׂמֵחִים – לֹא בַּסּוֹף הַגָּלוּת אֶלָּא בְּתוֹכָהּ. הָיִינוּ שְׂמֵחִים – לֹא בַּגְּאֻלָּה אֶלָּא בְּצִפִּיָּתֵנוּ לָהּ. הָיִינוּ שְׂמֵחִים – לֹא בַּיֹּבֶשׁ אֶלָּא בְּעוֹדֵנוּ בַּמַּיִם, כִּי בְּתוֹךְ מַיִם הַמָּרָה כְּבָר הִרְגַּשְׁנוּ אֶת מְתִיקוּת הַיְּשׁוּעָה.

אוֹר אֵינְסוֹפִי, מְקוֹר כָּל הַבְּרָכוֹת וְהַנֶּחָמוֹת, תֵּן לִי לִשְׂמוֹחַ בְּתוֹךְ מְצוּקַת הַגָּלוּת, לָשִׁיר בְּעוֹדִי בְּאֶרֶץ נְכָר, לִרְקֹד בְּעוֹד שַׁלְשְׁלָאוֹת מְכַבְּדוֹת אֶת קַרְסֻלַּי.

אָבִי, אֲנִי רוֹצֶה לִהְיוֹת מֵהָרוֹקְדִים לִפְנֵי הָאוֹר, לִהְיוֹת כִּנּוֹר הַמְּזַמֵּר לִפְנֵי זְרִיחַת הַשֶּׁמֶשׁ, לִהְיוֹת קָנֶה סוּף הַרוֹעֵד בַּבְּרָכָה טֶרֶם יוֹרֵד גֶּשֶׁם, לִהְיוֹת מִדְרָשׁ לַגָּלוּת הַמַּמְצִיא אֶת הַגְּאֻלָּה.

שַׁעַר חֲמִישִׁי: הִיא שֶׁעָמְדָה לַאֲבוֹתֵינוּ וְלָנוּ

אוֹר עַתִּיק, רוֹאֶה הַנּוֹלָד וְחוֹמֵל עַל הַכְּמֵהִים

נִסְתָּרִים הֵם דַּרְכֵי הַגְּאֻלָּה, אֲדוֹנִי, אַךְ אִם נֹאמַר אֶת סוֹד הָאֱמוּנָה הַנְּשִׁית: לֹא הַיְּשׁוּעָה מְבִיאָה אֶת הַתּוֹדָה, אֶלָּא הַתּוֹדָה מְבִיאָה אֶת הַיְּשׁוּעָה.

בְּכָל דּוֹר וָדוֹר עוֹמְדִים עָלֵינוּ לְכַלּוֹתֵנוּ, וְהַקָּדוֹשׁ בָּרוּךְ הוּא מַצִּילֵנוּ מִיָּדָם. אַךְ "הִיא" – הִיא מִי? הִיא שָׁרָה. הִיא הַזֵּמֶרֶת. הִיא הָאֱמוּנָה. הִיא הַמַּקְדִּימָה אֶת הַהוֹדָיָה לַיְּשׁוּעָה.

בִּזְכוּת הָאֱמוּנָה הַמְּזַמֶּרֶת טֶרֶם רְאוֹת יְשׁוּעָה, בִּזְכוּת הַמַּקְדִּימִים אֶת הַתּוֹף לְשִׁירַת הַיָּם, בִּזְכוּת מִרְיָם הַנְּבִיאָה שֶׁשָּׁמְרָה אֶת הַתּוֹף בְּמִצְרַיִם – עָמְדָה לָנוּ זְכוּת הַגְּאֻלָּה.

הַיּוֹם, בְּתוֹךְ גָּלוּתֵנוּ הַפְּנִימִית וְהַחִיצוֹנִית, אֲנִי מְקַבֵּל עָלַי אֶת הַמַּלְכוּת הַזּוֹ: לְהוֹדוֹת עַל מַה שֶׁטֶּרֶם נִתָּן, לְזַמֵּר אֶת שֶׁטֶּרֶם הִתְגַּלָּה, לִשְׂמֹחַ בְּשֶׁעָתִיד, לַחְיוֹת בְּהִתְגַּלּוּת שֶׁטֶּרֶם נִפְרְשָׂה.

שַׁעַר שִׁשִּׁי: הָאֱמוּנָה הַנְּשִׁית

חוֹנֵן דַּעַת לָאָדָם, מְלַמֵּד לֶאֱנוֹשׁ בִּינָה

וּבְמַעֲמַקֵּי נְשָׁמָתִי אֲנִי שׁוֹאֵל אוֹתְךָ, אָבִי: מַה סּוֹדָן שֶׁל הַנָּשִׁים שֶׁהֵבִיאוּ תֻּפִּים מִמִּצְרַיִם? מַה יָּדְעוּ הֵן שֶׁאֲנִי, הָאִישׁ, עֲדַיִן לֹא יוֹדֵעַ? אֵיךְ אֶזְכֶּה לַמִּבָּט הַחוֹדֵר אֶל מֵעֵבֶר לַמַּיִם?

וְאַתָּה עוֹנֶה בְּשֶׁקֶט, בְּנִגּוּן קַדְמוֹנִי: הַנָּשִׁים יָדְעוּ אֶת הַמִּשְׁלֵי הָעָמֹק מִנִּי כָּל, שֶׁסּוֹד הַגְּאֻלָּה הוּא שִׂמְחַת הַהַמְתָּנָה, שֶׁנֵּס הַיָּם קָרַע אֶת דֶּרֶךְ הַמְתִיקוּת שֶׁבְּאֶמְצַע הַמָּרִירוּת.

אָנָּא, נָא, חַדֵּשׁ בְּקִרְבִּי תֹּף נְשִׁי, תֵּן לִי לְהַאֲמִין כְּחֵרֵשׁ שֶׁלֹּא שָׁמַע, כְּעִוֵּר שֶׁלֹּא רָאָה, כִּתְמִים בְּאֱמוּנָה, כְּנָשִׁים צִדְקָנִיּוֹת הַיּוֹדְעוֹת אֶת אֲשֶׁר טֶרֶם הָיָה.

אֶת שִׁירַת הַיָּם, לְשׁוֹן זָכָר, יָשִׁירוּ הַגְּבָרִים בְּנֵס גָּלוּי, עַל גְּבוּרוֹת אֵל. אַךְ אֶת שִׁירַת הַנָּשִׁים, סוֹד הָאֱמוּנָה, הַיּוֹם אֲנִי מְבַקֵּשׁ לְנַגֵּן בְּעוֹדִי טוֹבֵעַ בַּיָּם.

חֲתִימָה: בְּשִׁירָה קִדְּמוּךָ

מְקוֹר הַחַיִּים וְהָאַהֲבָה, דּוֹדִי צַח וְאָדֹם

וְכָעֵת, בְּלִבִּי מְתַפֵּעֵם, אֲנִי מַכְרִיז בְּפָנֶיךָ: לֹא אַחְכֶּה לְחוֹף מִבְטַחִים כְּדֵי לָשִׁיר, לֹא אַמְתִּין לְנֵס גָּלוּי כְּדֵי לְהוֹדוֹת, לֹא אֲיַחֵל לִבְקִיעַת הַמַּיִם כְּדֵי לְהַאֲמִין.

אֶלָּא עַכְשָׁו, בְּעוֹדִי שָׁקוּעַ בְּתוֹךְ הַגָּלוּת, אֲנִי מֵנִיף אֶת תֹּפִי מִמִּצְרַיִם, וְרוֹקֵד עַל חוֹף הַיָּם. בְּעוֹדִי מוּקָף גַּלִּים קָשִׁים, גָּלֵי הַתּוֹדָעָה וְהַקְּשָׁיִים, אֲנִי מְרִים אֶת נִגּוּן הָאֱמוּנָה כְּמַלְכוּת נְשִׁית.

יְהִי רָצוֹן מִלְּפָנֶיךָ, בּוֹרֵא כָּל הָעוֹלָמוֹת, שֶׁבִּזְכוּת שִׁירָתִי הַמַּקְדִּימָה אֶת הַנֵּס, תִּשְׁלַח לִי וּלְכָל עַמְּךָ אֶת הַיְּשׁוּעָה הַשְּׁלֵמָה, אֶת חֲלוֹם כָּל הַדּוֹרוֹת הַנִּכְסָפִים לְצִיּוֹן.

וְעַד בּוֹא הַיּוֹם הַהוּא, אֲנִי כְּאִשָּׁה מְצַפֶּה, כְּמִרְיָם הַנְּבִיאָה, בְּיָד אַחַת אוֹחֶזֶת בְּתֻפִּים וּבְיָד אַחֶרֶת רוֹקֶמֶת אֶת בִּגְדֵי הַגְּאֻלָּה, כִּי אֲנִי מַאֲמִין שֶׁ"בְּשִׁירָה קִדְּמוּךָ" הוּא סוֹד קִיּוּמֵנוּ.

אָמֵן וְאָמֵן.

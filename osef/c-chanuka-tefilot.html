תפילות בסגנון ליקוטי תפילות ופיוטים על אור החנוכה, ההידור וההודיה.

פיוט הידור המצווה והאדם

חזן: הִדּוּר הַמִּצְוָה מַה הוּא? הִדּוּר הָאָדָם מַהוּ? שְׁנֵי דְרָכִים לַעֲבוֹדָה שְׁנֵי שְׁבִילִים לְהוֹד קוֹנוֹ

קהל: יֵשׁ מְהַדֵּר בַּחֵפֶץ הַנָּאֶה וְיֵשׁ מְהַדֵּר בְּלֵב הַמִּתְאַוֶּה זֶה בְּיֹפִי הַכְּלִי מִשְׁתַּדֵּל וְזֶה בְּמַאֲמַץ נַפְשׁוֹ מִתְעַלֶּה

חזן: רַבֵּנוּ רַשִׁ"י מֵאִיר עֵינֵינוּ: "הַמְהַדְּרִין אַחַר הַמִּצְווֹת" לֹא אָמַר "מְהַדְּרִים אֶת הַמִּצְווֹת" כִּי הָעִקָּר - הַחִפּוּשׂ, הַהִשְׁתַּדְּלוּת

קהל: מְחַזֵּר אַחַר הַמִּצְוָה בְּעַצְמוֹ מְבַקֵּשׁ לַעֲשׂוֹת בִּמְסִירוּת נַפְשׁוֹ לֹא דַי בְּהִדּוּר הַחִיצוֹנִי כִּי אִם בְּמַאֲמַץ הַלֵּב הַפְּנִימִי

חזן: וְהָרַמְבַּ"ם בְּאוֹר תּוֹרָתוֹ מְלַמְּדֵנוּ: הַדֵּר הַחֵפֶץ בְּיָפְיוֹ כְּלוּלָב נָאֶה, אֶתְרוֹג מְהֻדָּר כֵּן הַמִּצְוָה תִּתְיַפֶּה בְּהוֹדָהּ

קהל: שְׁנֵי אוֹרוֹת מְאִירִים יַחְדָּיו זֶה בַּחוּץ וְזֶה בִּפְנִים זֶה בְּיֹפִי הַמַּעֲשֶׂה הַנִּרְאֶה וְזֶה בְּעֹמֶק הַלֵּב הַמִּתְכַּוֵּן

חזן וקהל יחד: אַשְׁרֵי הַזּוֹכֶה לִשְׁנֵיהֶם לְהַדֵּר הַמִּצְוָה בְּתֹאַר וָהוֹד וּלְהַדֵּר נַפְשׁוֹ בַּעֲבוֹדָה לְהִתְקַדֵּשׁ בִּקְדֻשַּׁת הַבּוֹרֵא

נַעֲלֶה בַּקֹּדֶשׁ בִּשְׁנֵי הַדְּרָכִים נְהַדֵּר הַמִּצְוָה בְּגוּף וָנֶפֶשׁ עַד יָאִיר אוֹרֵנוּ כְּאוֹר שִׁבְעָתַיִם בְּהִדּוּר הַמַּעֲשֶׂה וְהִדּוּר הַנֶּפֶשׁ

אור ההידור

חזן: מַה נָּאוּ הַנֵּרוֹת, מַה יָּפוּ הַלֶּהָבוֹת בְּהִדּוּר מִצְוָה מְאִירִים הָאוֹרוֹת

קהל: בְּכָל יוֹם נוֹסִיף, בְּכָל לַיְלָה נַעֲלֶה מִנֵּר לְנֵר נִתְעַלֶּה, מֵאוֹר לְאוֹר נַגְבִּיהַ

חזן: הִדּוּר הַמִּצְוָה - סוֹד הַהִתְעַלּוּת מוֹסִיף וְהוֹלֵךְ - דֶּרֶךְ הַהִתְקַדְּשׁוּת

קהל: אוֹר קָטָן מַדְלִיק אוֹרוֹת גְּדוֹלִים מִצְוָה קְטַנָּה פּוֹתַחַת שְׁעָרִים נִשְׂגָּבִים

ראיית הניסים

חזן: בְּכָל נֵס קָטָן רוֹאִים נִסִּים גְּדוֹלִים בְּפַךְ שֶׁמֶן אֶחָד מְגַלִּים אוֹרוֹת נִפְלָאִים

קהל: הַטֶּבַע כֻּלּוֹ - נֵס בְּתוֹךְ נֵס כָּל רֶגַע וָרֶגַע - פֶּלֶא בְּתוֹךְ פֶּלֶא

חזן: עֵינַיִם פְּקוּחוֹת רוֹאוֹת נִפְלָאוֹת לֵב מֵבִין מַכִּיר בַּנִּסִּים הַנִּסְתָּרוֹת

קדושת הבית

חזן: בַּיִת קָטָן כְּמִקְדָּשׁ מְעַט נֵרוֹת דּוֹלְקִים בּוֹ כִּמְנוֹרַת הַקֹּדֶשׁ

קהל: בְּכָל בַּיִת יְהוּדִי שׁוֹכֵן אוֹר עֶלְיוֹן בְּכָל חַלּוֹן מֵאִיר נֵר הָאֱמוּנָה

חזן: מִקְדָּשׁ בָּנוּי מֵאוֹר וּקְדֻשָּׁה בַּיִת מוּאָר בְּאַהֲבַת תּוֹרָה

אינסופיות האור

חזן: אֵין סוֹף לְמַעְלָה, אֵין סוֹף לְמַטָּה בְּכָל נֵר מֵאִיר אוֹר אֵין־סוֹפִי

קהל: כָּל לַיְלָה עוֹלֶה, כָּל יוֹם מִתְעַלֶּה מֵאוֹר לְאוֹר, מִמַּעֲלָה לְמַעֲלָה

חזן ְוקהל יחד: אוֹר חֲנֻכָּה מֵאִיר לְדוֹרוֹת פּוֹתֵחַ שַׁעֲרֵי אוֹרָה אֵין־סוֹפִיִּים

נַעֲלֶה בַּקֹּדֶשׁ, נוֹסִיף בְּאוֹר עַד שֶׁיָּאִיר אוֹר חָדָשׁ עַל צִיּוֹן

שיר האור וההודיה

חזן: נְשַׁבֵּחַ בְּשִׁיר וּבְזָמִיר לְמִי שֶׁאוֹר נֵרוֹ מֵאִיר אוֹר הַהוֹדָיָה בּוֹ נִגְלֶה בְּכָל נֵר קָדוֹשׁ שֶׁעוֹלֶה

קהל: אוֹר וְהוֹדָיָה אֶחָד הֵם הַלֶּהָבָה וְהַתּוֹדָה אַחַת הֵן לֹא סִימָן לְבַד הוּא הַנֵּר אֶלָּא הַהוֹדָיָה בּוֹ תִּתְבָּאֵר

חזן: קָבְעוּ חֲכָמִים בְּבִינָה מִצְוָה קְדוֹשָׁה וַעֲדִינָה לְהַדְלִיק נֵרוֹת וּלְהַלֵּל לְמִי שֶׁאוֹרוֹ לֹא יִכְלֶּל

קהל: הַנֵּרוֹת הַלָּלוּ מְסַפְּרִים עַל תְּשׁוּעוֹת וְעַל נִסִּים נִסְתָּרִים לֹא רַק זֵכֶר לְפַךְ הַשֶּׁמֶן אֶלָּא הוֹדָיָה עַל כָּל הַזְּמַן

חזן: כְּשֶׁאָנוּ מַדְלִיקִים בְּלֵב טָהוֹר מִתְעוֹרֶרֶת הַהוֹדָיָה כְּמַעְיָן טָהוֹר וּכְשֶׁאָנוּ רוֹאִים אֶת הָאוֹר נִפְתָּח הַלֵּב לְהוֹדוֹת עוֹד וָעוֹד

קהל: אוֹר הַהוֹדָיָה, אוֹר הַתּוֹדָה מְאִירִים יַחַד בְּנֶפֶשׁ הוֹמִיָּה הַמַּעֲשֶׂה וְהַכַּוָּנָה מִתְאַחֲדִים בְּשַׁלְהֶבֶת אַחַת לַמָּרוֹם עוֹלִים

חזן וקהל יחד: וְכָל הָרוֹאֶה נֵר דּוֹלֵק נִשְׁמָתוֹ לְהוֹדוֹת תִּתְעוֹרֵר וְכָל הַמַּבִּיט בְּאוֹר הַנֵּרוֹת יִמָּלֵא לִבּוֹ בְּשִׁירוֹת וְתוֹדוֹת

סיום: יְהִי רָצוֹן שֶׁתַּעֲלֶה הוֹדָיָתֵנוּ כְּאוֹר הַנֵּרוֹת לִפְנֵי אֲדוֹנֵנוּ וּכְשֶׁם שֶׁהָאוֹר עוֹלֶה וּמֵאִיר כֵּן תַּעֲלֶה תְּפִלָּתֵנוּ לִפְנֵי שׁוֹכֵן שָׁמִיר

תפילה בסגנון ליקוטי תפילות - אינסופיות האור

רִבּוֹנוֹ שֶׁל עוֹלָם, אֵין סוֹף בְּרוּךְ הוּא. אַתָּה בְּאֵין סוֹפְךָ מְמַלֵּא כָּל עָלְמִין, וְאֵין מָקוֹם פָּנוּי מִמֶּךָ. בְּעֹמֶק חֲסָדֶיךָ נָתַתָּ לָנוּ מִצְוַת נֵר חֲנֻכָּה, שֶׁהִיא כְּלִי לְגַלּוֹת אֶת הָאֵין סוֹף שֶׁבְּךָ בָּעוֹלָם.

אָנָּא ה', זַכֵּנִי לְהָבִין אֶת סוֹד הָאֵין סוֹף שֶׁבַּמִּצְוָה הַזֹּאת. כְּשֶׁם שֶׁאֵין שִׁעוּר תַּחְתּוֹן לַמִּצְוָה, וַאֲפִלּוּ עָנִי שֶׁאֵין לוֹ מַה יֹּאכַל חַיָּב בָּהּ, כָּךְ אֵין גְּבוּל עֶלְיוֹן לְהִדּוּרָהּ, וְהַכֹּל בִּכְלַל הַמִּצְוָה.

רִבּוֹנוֹ שֶׁל עוֹלָם, אַתָּה יוֹדֵעַ שֶׁלִּפְעָמִים אֲנִי מְצַמְצֵם אֶת עַצְמִי, מַגְבִּיל אֶת הוֹדָיָתִי, שָׂם גְּבוּלוֹת לַעֲבוֹדָתִי. אֲבָל אַתָּה לִמַּדְתָּ אוֹתָנוּ שֶׁבַּהוֹדָיָה אֵין גְּבוּלוֹת, בַּהַכָּרַת הַטּוֹב אֵין מְצָרִים.

עָזְרֵנִי, אָבִי, לְהַגִּיעַ לְהוֹדָיָה אֵין סוֹפִית, לְהַכָּרַת הַטּוֹב שֶׁאֵין לָהּ קֵץ וְתַכְלִית. שֶׁאֶזְכֶּה לְהוֹדוֹת לְךָ בְּכָל לְבָבִי, בְּכָל נַפְשִׁי וּבְכָל מְאֹדִי, בְּלִי גְּבוּל וּמִדָּה.

רִבּוֹנוֹ שֶׁל עוֹלָם, בּוֹנֵה בַּיִת וּמְכוֹנֵן הֵיכָל. אַתָּה בְּחַסְדְּךָ הַגָּדוֹל נָתַתָּ לָנוּ מִקְדָּשׁ מְעַט - אֶת הַבַּיִת הַיְּהוּדִי. נָתַתָּ לָנוּ מָקוֹם לְהַשְׁרָאַת שְׁכִינָתְךָ, מָקוֹם שֶׁמִּמֶּנּוּ יוֹצֵא אוֹר לָעוֹלָם כֻּלּוֹ.

אָנָּא ה', זַכֵּנִי לְהַדְלִיק נֵרוֹת חֲנֻכָּה בִּקְדֻשָּׁה וּבְטָהֳרָה, שֶׁיִּהְיוּ נֵרוֹתַי כְּמוֹ נֵרוֹת הַמְּנוֹרָה בְּבֵית הַמִּקְדָּשׁ. שֶׁיְּהֵא בֵּיתִי מָקוֹם שֶׁל תּוֹרָה וּתְפִלָּה, מָקוֹם שֶׁל קְדֻשָּׁה וְטָהֳרָה, מָקוֹם שֶׁמְּשַׁפִּיעַ אוֹר וּבְרָכָה עַל כָּל יוֹשְׁבָיו.

רִבּוֹנוֹ שֶׁל עוֹלָם, אַתָּה יוֹדֵעַ שֶׁלִּפְעָמִים אֲנִי שׁוֹכֵחַ אֶת קְדֻשַּׁת בֵּיתִי, שׁוֹכֵחַ שֶׁזֶּה מִקְדָּשׁ מְעַט. אֲבָל בְּחַג הַחֲנֻכָּה, כְּשֶׁהַנֵּרוֹת דּוֹלְקִים, אֲנִי נִזְכָּר שֶׁבֵּיתִי הוּא מָקוֹם קָדוֹשׁ, מְקוֹם שֶׁאַתָּה שׁוֹכֵן בּוֹ.

עָזְרֵנִי לְהַמְשִׁיךְ אֶת אוֹר הַחֲנֻכָּה לְכָל הַשָּׁנָה כֻּלָּהּ, שֶׁיִּהְיֶה בֵּיתִי תָּמִיד מָקוֹם שֶׁל אוֹר וּקְדֻשָּׁה. זַכֵּנִי לְגַדֵּל בָּנִים תַּלְמִידֵי חֲכָמִים, שֶׁיִּינְקוּ מִקְדֻשַּׁת הַבַּיִת וְיַמְשִׁיכוּ אֶת שַׁלְשֶׁלֶת הַדּוֹרוֹת.

רִבּוֹנוֹ שֶׁל עוֹלָם, אַתָּה הֲרֵי יוֹדֵעַ שֶׁלִּפְעָמִים אֲנִי נוֹפֵל לְמַחְשָׁבוֹת שֶׁל "יוֹצְאִים יְדֵי חוֹבָה", שֶׁדַּי בְּקִיּוּם הַמִּצְוָה בְּהִדּוּר חִיצוֹנִי. אֲבָל אַתָּה לִמַּדְתָּ אוֹתָנוּ דֶּרֶךְ רַשִׁ"י הַקָּדוֹשׁ שֶׁהָעִקָּר הוּא לִהְיוֹת "מְהַדְּרִין אַחַר הַמִּצְווֹת" - לִרְדֹּף, לְחַפֵּשׂ, לְהִשְׁתּוֹקֵק לַעֲשׂוֹת אֶת הַמִּצְוָה בְּעַצְמֵנוּ, מִתּוֹךְ חִבּוּר פְּנִימִי עָמֹק.

תפילה על קדושת הבית

ריבון העולמים, כשם שקידשת את בית מקדשך וטיהרת את היכלך, כן תקדש את ביתי להיות משכן לשכינתך.

תן בי כוח להדליק נר של קדושה בביתי, להאיר באור תורתך את כל פינות חיי.

יהי רצון מלפניך שיהיה ביתי כמקדש מעט, מקום של תורה ותפילה, של שלום ואהבה.

תפילה על עומק האינסופיות

אדון כל העולמים, אתה שאין סוף לגדולתך, תן בליבי את ההבנה שאין גבול לעלייה בעבודתך.

כמו הנרות המוסיפים והולכים, כן תגדל אהבתי אליך. כמו האור המתפשט והולך, כן יתרחב ליבי בעבודתך.

למדני להבין שכל הישג הוא רק התחלה, שכל מדרגה היא שער לעוד עלייה בקדושה.

חזן: הִנֵּה יָמִים בָּאִים, יְמֵי חוֹתָם וְאוֹר מִיּוֹם הַכִּפּוּרִים עַד נֵרוֹת הַמָּאוֹר הַטַּהֵר מְטַהֵר, הַמְּכַפֵּר מְכַפֵּר עַד יָבוֹא חֲנֻכָּה וְיַשְׁלִים הַגּוֹמֵר

קהל: יוֹם כִּפּוּר - רֵאשִׁית הַטָּהֳרָה חֲנֻכָּה - גְּמַר הַכַּפָּרָה מִסְּלִיחָה לִסְלִיחָה נָרוּץ עַד אוֹר הַנֵּרוֹת יָצוּץ

חזן: שְׁלֹשָׁה חֳדָשִׁים מֵרֹאשׁ הַשָּׁנָה עַד יְמֵי הַנֵּס וְהַתְּבוּנָה כְּהֶכֵּר הָעֻבָּר בַּיְּצִירָה כֵּן יֻכַּר טָהֳרַת הַנְּשָׁמָה הַבְּהִירָה

קהל: נֵר דּוֹלֵק עַל רֹאשֵׁנוּ תּוֹרַת אֱמֶת מְאִירָה בְּקִרְבֵּנוּ מֵאוֹר הַגָּנוּז בִּנְשָׁמוֹתֵינוּ מֵאִיר דֶּרֶךְ תְּשׁוּבָתֵנוּ

חזן: בְּהֵיכַל קָדְשְׁךָ נִמְצָא הָאוֹר בְּנֵר חֲנֻכָּה נִגְלֶה הַמָּאוֹר מִסּוֹד הַבִּינָה עַד הַהוֹד יוֹרֵד הַשֶּׁפַע בְּסוֹד הַיִּחוּד

קהל וחזן יחד: אָנָּא, טַהֵר לִבֵּנוּ בִּשְׁלֵמוּת הָאֵר נַפְשֵׁנוּ בִּקְדֻשַּׁת הַנֵּרוֹת מִגְּמַר דִּין לְגְמַר טָהֳרָה מִכִּפּוּר לַחֲנֻכָּה בְּאוֹרָה

תפילת האור וההודיה

חזן: אוֹר הַהוֹדָיָה מְרַצֵּד בַּנֵּרוֹת כָּל לֶהָבָה מַבִּיעָה תְּהִלּוֹת מַה נִּפְלָא הָאוֹר הַשּׁוֹפֵעַ מִמָּרוֹם מְסַפֵּר גְּבוּרוֹת אֵל עַד אֵין מָקוֹם

קהל: יִתְבָּרַךְ שִׁמְךָ בְּפִי כָּל הָאוֹרוֹת יִתְרוֹמַם זִכְרְךָ בְּשַׁלְהֶבֶת הַנֵּרוֹת כָּל נֵר מַכְרִיז: הוֹדוּ לַה' כִּי טוֹב כָּל אוֹר מְסַפֵּר: חַסְדּוֹ גָּבַר עָלֵינוּ מְאֹד

חזן: לֹא רַק סִימָן, לֹא רַק זֵכֶר לְנֵס אֶלָּא הוֹדָיָה חַיָּה, אוֹר קָדוֹשׁ נִכְנָס בְּכָל הַדְלָקָה - תְּפִלָּה זַכָּה עוֹלָה בְּכָל רְאִיָּה - נְשָׁמָה לְךָ מוֹדָה

קהל: הַנֵּרוֹת הַלָּלוּ קֹדֶשׁ הֵם אֵין לָנוּ רְשׁוּת לְהִשְׁתַּמֵּשׁ בָּהֶם רַק לִרְאוֹתָם, רַק לְהוֹדוֹת דַּרְכָּם עַל כָּל הַנִּסִּים שֶׁעָשִׂיתָ עִמָּנוּ מֵעוֹלָם

חזן וקהל יחד: בְּכָל נֵר שֶׁדּוֹלֵק - תּוֹדָה מִתְעוֹרֶרֶת בְּכָל אוֹר שֶׁמֵּאִיר - הַכָּרָה מִתְגַּלֶּה אוֹר הַהוֹדָיָה יִמְשִׁיךְ לְהָאִיר עַד יָבוֹא יוֹם בּוֹ אוֹר חָדָשׁ עַל צִיּוֹן תָּאִיר

סיום: יְהִי רָצוֹן שֶׁתַּעֲלֶה הוֹדָיָתֵנוּ כְּרֵיחַ נִיחוֹחַ לִפְנֵי כִּסֵּא כְבוֹדֶךָ וְתִתְקַבֵּל תְּפִלָּתֵנוּ בְּאוֹר פָּנֶיךָ כְּמוֹ שֶׁקִבַּלְתָּ בְּרָצוֹן אֶת נֵרוֹת אֲבוֹתֵינוּ

תפילת האור והתודה

רִבּוֹן הָעוֹלָמִים, אֲדוֹן הָאוֹרִים, אֲשֶׁר בְּרַחֲמֶיךָ הָעֲצוּמִים נָתַתָּ בְּלִבִּי נֵר קָדוֹשׁ שֶׁל הוֹדָיָה, לְהוֹדוֹת וּלְהַלֵּל לְשִׁמְךָ הַגָּדוֹל בְּכָל הֲוָיָה.

אָנָּא, חַזֵּק אֶת נַפְשִׁי הַכּוֹסֶפֶת לְהַדְלִיק אֶת נֵר הַהוֹדָיָה בְּלֵב רוֹעֵד. שֶׁיִּהְיֶה כָּל נֵר שֶׁאַדְלִיק בְּחַלּוֹנִי בִּטּוּי אֱמֶת לַהֲמוֹן תּוֹדוֹת הַמְּמַלְּאוֹת אֶת כָּל הֲוָיָתִי.

אָבִי שֶׁבַּשָּׁמַיִם, פְּתַח אֶת לִבִּי לְהַרְגִּישׁ אֶת סוֹד הָאוֹר הַקָּדוֹשׁ. שֶׁתִּהְיֶה הַדְלָקָתִי לֹא רַק מַעֲשֶׂה חִיצוֹנִי, אֶלָּא בִּטּוּי עָמֹק שֶׁל הַכָּרַת טוּבְךָ הָאֵין-סוֹפִי.

לַמְּדֵנִי לְהוֹדוֹת בְּכָל נֵר וָנֵר עַל כָּל הַנִּסִּים וְהַנִּפְלָאוֹת, עַל הַתְּשׁוּעוֹת וְעַל הַמִּלְחָמוֹת, עַל חַסְדְּךָ הַגָּדוֹל הַמֵּאִיר אוֹתִי בְּכָל הַמַּצָּבוֹת.

רִבּוֹנוֹ שֶׁל עוֹלָם, תֵּן בִּי אֶת הַכֹּחַ לִרְאוֹת בְּכָל נֵר אֶת אוֹר אַהֲבָתְךָ הַבּוֹעֶרֶת. שֶׁיִּהְיוּ עֵינַי וְלִבִּי פְּתוּחִים לְהוֹדוֹת וּלְהַלֵּל בְּכָל רֶגַע מֵחַיַּי.

אָנָּא, קַבֵּל בְּרַחֲמִים וּבְרָצוֹן אֶת תְּפִלַּת הוֹדָיָתִי הַבּוֹעֶרֶת, הַמִּתְעַלָּה אֵלֶיךָ עִם כָּל נֵר כְּרֵיחַ נִיחוֹחַ, כְּקָרְבַּן תּוֹדָה טָהוֹר.

וּכְשֶׁם שֶׁהַנֵּרוֹת מְאִירִים בַּחֹשֶׁךְ, כֵּן תָּאִיר נִשְׁמָתִי בְּאוֹר הוֹדָיָתָהּ, לְהַכִּיר וּלְהוֹדוֹת עַל כָּל חַסְדֶּיךָ, עַד יָבוֹא הַיּוֹם בּוֹ יָאִיר אוֹר חָדָשׁ עַל צִיּוֹן.

תפילת מהות החנוכה

רִבּוֹנוֹ שֶׁל עוֹלָם, פּוֹעֵל נִסִּים וְנִפְלָאוֹת, אַתָּה בְּחָכְמָתְךָ הָעֲמֻקָּה גִּלִּיתָ לָנוּ בַּחֲנֻכָּה סוֹד נִפְלָא: נֵס בְּתוֹךְ נֵס, אוֹר בְּתוֹךְ אוֹר. הִצַּלְתָּ אֶת עַמְּךָ מִיַּד הַיְּוָנִים הָרְשָׁעִים, וְכְדֵי שֶׁנֵּדַע שֶׁהַכֹּל מִמְּךָ, עָשִׂיתָ לָנוּ נֵס נוֹסָף בְּפַךְ הַשֶּׁמֶן.

אָנָּא ה', זַכֵּנִי לְהָבִין אֶת עֹמֶק חָכְמַת חֲכָמֵינוּ זִכְרוֹנָם לִבְרָכָה, שֶׁלֹּא תִּקְּנוּ רַק לְהַדְלִיק נֵרוֹת לְזֵכֶר הַנֵּס, אֶלָּא נָתְנוּ לָנוּ דֶּרֶךְ נִפְלָאָה לְהוֹדוֹת לְךָ - שֶׁהַהַדְלָקָה עַצְמָהּ הִיא הַהוֹדָיָה. בְּכָל פַּעַם שֶׁאֲנִי מַדְלִיק אֶת הַנֵּרוֹת, תֵּן בְּלִבִּי לְהַרְגִּישׁ שֶׁבְּמַעֲשֶׂה זֶה עַצְמוֹ אֲנִי מוֹדֶה לְךָ עַל כָּל הַנִּסִּים וְהַנִּפְלָאוֹת.

רִבּוֹנוֹ שֶׁל עוֹלָם, לַמְּדֵנִי לִרְאוֹת בְּנֵס הַשֶּׁמֶן הַגָּלוּי אֶת נֵס הַמִּלְחָמָה הַנִּסְתָּר. כְּשֵׁם שֶׁהַשֶּׁמֶן דָּלַק בְּדֶרֶךְ נֵס, כֵּן תֵּן לִי לְהַכִּיר שֶׁכָּל נִצְחוֹנֵנוּ הָיָה נֵס גָּמוּר מֵאִתְּךָ. וּכְשֵׁם שֶׁלֹּא הָיִיתָ צָרִיךְ לְנֵס פַּךְ הַשֶּׁמֶן, אֶלָּא עָשִׂיתָ אוֹתוֹ כְּדֵי לְגַלּוֹת לָנוּ אֶת יָדְךָ הַגְּדוֹלָה, כֵּן זַכֵּנִי לִרְאוֹת אֶת יָדְךָ בְּכָל דָּבָר.

שיר ליטורגי - נס בתוך נס

חזן: שְׁנֵי נִסִּים בְּאוֹר אֶחָד מְאִירִים נֵס גָּלוּי וְנֵס נִסְתָּר מְאֻחָדִים פַּךְ הַשֶּׁמֶן מַרְאֶה בְּאוֹר בָּהִיר שֶׁכָּל הַיְשׁוּעָה מֵאֵת הַכַּבִּיר

קהל: וַאֲנַחְנוּ מַדְלִיקִים וּמוֹדִים בְּמַעֲשֶׂה אֶחָד שְׁנֵיהֶם מְאֻחָדִים הַהַדְלָקָה הִיא הַהוֹדָיָה וְהַהוֹדָיָה הִיא הַהַדְלָקָה

חזן: חָכְמַת חֲכָמֵינוּ הֶעֱמִיקָה לִרְאוֹת שֶׁאֵין הַנֵּרוֹת רַק סִימָן לְהוֹדוֹת אֶלָּא בָּם עַצְמָם מְגַלִּים אָנוּ אֶת הַהוֹדָיָה הַטְּמוּנָה בְּלִבֵּנוּ

קהל וחזן יחד: וּבְכָל שָׁנָה כְּשֶׁנַּדְלִיק מֵחָדָשׁ נִרְאֶה בָּאוֹר אֶת הַנֵּס הַמְחֻדָּשׁ וְנוֹדֶה לְךָ בְּהַדְלָקַת הַנֵּרוֹת עַל כָּל הַנִּסִּים, הַנִּגְלִים וְהַנִּסְתָּרוֹת

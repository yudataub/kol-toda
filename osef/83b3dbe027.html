תְּפִלַּת הַנִּקָּיוֹן - מֵאָבָק לְהוֹדָיָה

תְּפִלַּת הַנִּקָּיוֹן - מֵאָבָק לְהוֹדָיָה

שַׁעַר רִאשׁוֹן: אוֹר לְאַרְבָּעָה עָשָׂר

רִבּוֹן הָעוֹלָמִים, בּוֹחֵר בַּיִת יְהוּדִי לְמִשְׁכָּנוֹ

אֵל עוֹלָם, בַּעַל הָרַחֲמִים וְהַחֶסֶד, הִנְנִי עוֹמֵד לְפָנֶיךָ בְּעֶרֶב קָדוֹשׁ זֶה, לְאַחַר שֶׁחָלְפוּ שְׁלֹשָׁה עָשָׂר יָמִים מֵרֹאשׁ חֹדֶשׁ. "אֶחָד" עַל גַּבֵּי "אֶחָד", וְעַכְשָׁו הִגִּיעַ זְמַן הַחִבּוּר.

לְאוֹר הַנֵּר הִנְנִי מְחַפֵּשׂ חָמֵץ, מְבַקֵּשׁ לְנַקּוֹת אֶת הַחָמֵץ מִבֵּיתִי, אַךְ יוֹדֵעַ אֲנִי, אָבִי וּמַלְכִּי, שֶׁעִקַּר הַחָמֵץ מְסֻתָּר בְּלִבִּי.

כְּמוֹ נַעַר שֶׁמַּגִּיעַ לְגִיל אַרְבַּע-עֶשְׂרֵה, וּמְקַבֵּל עָלָיו עֹל מַלְכוּת שָׁמַיִם, כָּךְ אֲנִי מְבַקֵּשׁ לְהִתְחַדֵּשׁ בְּלֵילָה זֶה, לִהְיוֹת בֵּן חוֹרִין מְשֻׁחְרָר מִכָּל אֵל זָר.

אָבִי שֶׁבַּשָּׁמַיִם, בַּעֲבוֹדָתִי הַגַּשְׁמִית לְנַקּוֹת אֶת בֵּיתִי, אֲנִי מְבַקֵּשׁ סִיַּעְתָּא דִשְׁמַיָּא לְנַקּוֹת גַּם אֶת לִבִּי. הַמְּחִצָּה הַמַּפְרִידָה בֵּינִי וּבֵינְךָ, אַבְקַת הַחָמֵץ הַמּוֹנַעַת מִמֶּנִּי לְהוֹדוֹת לְךָ בֶּאֱמֶת.

שַׁעַר שֵׁנִי: כָּאן גָּר יְהוּדִי

אָדוֹן הָאַהֲבָה, הָרוֹקֵד עַל פִּתְחֵי בָּתֵּינוּ

וְאַתָּה, בּוֹרְאִי וְגוֹאֲלִי, לֹא דִלַּגְתָּ מֵעָלַי, אֶלָּא רָקַדְתָּ עָלַי. בְּאוֹתוֹ לַיְלָה בְּמִצְרַיִם, בִּימֵי קֶדֶם, קָפַצְתָּ וְרָקַדְתָּ בְּאַהֲבָה וְקָרָאתָ: "כָּאן גָּר יְהוּדִי!"

כְּמוֹ הַצַּדִּיק רַבִּי זוּשָׁא מֵאָנִיפּוֹלִי, שֶׁרָקַד וְשָׁר בְּשִׂמְחָה עַל שֻׁלְחָנוֹ, אֲנִי מְבַקֵּשׁ לְהִתְעוֹרֵר בְּהַכָּרָה עֲמֻקָּה: זֶה בֵּית יְהוּדִי, זֶה בֵּית יְהוּדִי, זֶה בֵּית יְהוּדִי!

בְּכָל נִקּוּי וְנִקּוּי, בְּכָל פֵּרוּר וּפֵרוּר, אֲנִי מְטַהֵר אֶת מִשְׁכָּנְךָ הַפְּנִימִי, מְקוֹם הַשְּׁכִינָה. בְּכָל תְּנוּעַת מְטַאֲטֵא, בְּכָל הֶבְרֵק מַגֵּב, אֲנִי מַכְרִיז: "מִקְדָּשׁ לָךְ אֲנִי מֵכִין, אָבִי".

לֹא מִתּוֹךְ כַּעַס וְטִרְחָה אֲנִי עוֹבֵד, אֶלָּא מִתּוֹךְ שִׂמְחָה וְהוֹדָיָה עַל זְכוּתִי, שֶׁבָּחַרְתָּ בִּי לִהְיוֹת לְךָ עֶבֶד אוֹהֵב, לִהְיוֹת לְךָ דִּירָה בַּתַּחְתּוֹנִים.

שַׁעַר שְׁלִישִׁי: הָאֵל הַזָּר שֶׁבַּלֵּב

צוּר הַנְּשָׁמוֹת, בּוֹנֶה אֱמוּנָה בַּלֵּבָבוֹת

רִבּוֹנוֹ שֶׁל עוֹלָם, בְּלִבִּי שׁוֹכֵן יֵצֶר הָרָע, "אֵל זָר" הַמְּבַקֵּשׁ לַעֲשׂוֹת פֵּרוּד בֵּינֵנוּ. מְמַלֵּא הוּא אֶת לִבִּי בִּדְאָגוֹת וּמְרִירוּת, מוֹנֵעַ מִמֶּנִּי מִלַּעֲלֹץ בְּאַהֲבָתְךָ.

בִּימֵי הַנִּקָּיוֹן הָאֵלֶּה, עָזְרֵנִי לְהַכְנִיעַ אוֹתוֹ, לְסַלֵּק אֶת הַמַּחְמֶצֶת מֵעִסַּת חַיַּי, לִמְחֹק אֶת הַמְּחִיצָה הַמַּפְרִידָה בֵּינֵנוּ, כְּדֵי שֶׁאוּכַל לְהַכִּיר בְּטוּבְךָ לִי.

כַּאֲשֶׁר אֲנִי מְנַקֶּה אֶת חֲרָכֵי בֵּיתִי, אָנָּא, נַקֵּה גַּם אֶת נִקְבוּבִיּוֹת לִבִּי, לְמַעַן אֶזְכֶּה לְהוֹדוֹת לְךָ בְּלֵב שָׁלֵם, וְתִתְגַּלֶּה בִּי הָאֱמוּנָה הַטְּהוֹרָה.

כְּשֵׁם שֶׁהֶחָמֵץ תּוֹפֵחַ וּמִתְנַשֵּׂא, כָּךְ גַּם הָאֵל הַזָּר שֶׁבְּקִרְבִּי מִתְנַפֵּחַ בְּגַאֲוָה. סַיְּעֵנִי לְהִתְנַקּוֹת מִמֶּנּוּ, אֶל עֶרְכַּת מַצָּה דַקָּה, עֶרְכַּת בִּטּוּל וַעֲנָוָה בִּפְנֵי רְצוֹנְךָ.

שַׁעַר רְבִיעִי: הַנִּקָּיוֹן כְּרִקּוּד שֶׁל אַהֲבָה

מַצְמִיחַ יְשׁוּעוֹת, מְזַכֵּךְ הַלֵּב הַשָּׁבוּר

אָבִי, מְלַמְּדֵנִי דֶּרֶךְ חֲדָשָׁה לְנַקּוֹת, דֶּרֶךְ שֶׁל שִׂמְחָה, דֶּרֶךְ שֶׁל הוֹדָיָה, לֹא כְּעֹל כָּבֵד, אֶלָּא כְּרִקּוּד קָדוֹשׁ, לֹא כְּמַשָּׂא, אֶלָּא כְּמַתָּנָה.

בְּכָל פֵּרוּר שֶׁאֲנִי מוֹצֵא וּמְסַלֵּק, בְּכָל הַבְרָקַת שַׁיִשׁ וְנִקּוּי חַלּוֹן, אֲנִי מְסַלֵּק גַּם קְלִפָּה רוּחָנִית, וְנִפְתָּח לוֹ חַלּוֹן לְאַהֲבָה וְהוֹדָיָה.

מֵאֵרַע הַמַּטְלִית עֲשֵׂה לִי מִזְבֵּחַ, מֵהַמַּגֵּב עֲשֵׂה לִי שַׁרְבִיט קָדוֹשׁ, שֶׁבּוֹ אַכְנִיס אֶת בֵּיתִי לִקְדֻשַּׁת הַחַג, וְאֶזְכֶּה לְהָכִין מִשְׁכָּן רָאוּי לִשְׁכִינָתְךָ.

בְּכָל זָוִית שֶׁאֲנִי מְנַקֶּה בְּבֵיתִי, אֲנִי מְטַהֵר גַּם זָוִית בְּנַפְשִׁי. כִּי "כָּאן גָּר יְהוּדִי" אֵינוֹ רַק הַכְרָזָה, אֶלָּא חוֹבָה וּזְכוּת לְהָכִין מָקוֹם לַמֶּלֶךְ.

שַׁעַר חֲמִישִׁי: לְהַצְמִיחַ שִׁירָה חֲדָשָׁה

מְחוֹלֵל הַגְּאֻלָּה, מְחַבֵּר הַנְּשָׁמוֹת

בּוֹרְאִי הַנֶּאֱמָן, חוֹשֵׂף מַעְיָנוֹת תְּהוֹם, אֲנִי מְבַקֵּשׁ לְהַכְנִיס אֶת הָאֱמוּנָה אֶל לִבִּי, שֶׁתְּהֵא שָׁם כְּמַעְיָן חַי, מְפַכֶּה וְנוֹבֵעַ, וְלֹא כִּידִיעָה יְבֵשָׁה בְּמֹחִי לְבַד.

עָזְרֵנִי לְסַדֵּר אֶת לִבִּי לִקְרַאת לֵיל הַסֵּדֶר, כְּמוֹ שֶׁאֲנִי מְסַדֵּר אֶת בֵּיתִי וּמְמַרֵק אוֹתוֹ. שֶׁאֶזְכֶּה לָשֶׁבֶת בְּלֵיל קָדוֹשׁ זֶה וְלִפְרֹץ בְּשִׁירָה: "שִׁבְּחוּ גְאוּלִים עַל גְּאֻלָּתֵנוּ וְעַל פְּדוּת נַפְשֵׁנוּ".

אָנָּא, בְּכֹחַ הַמִּצְוָה הַזֹּאת שֶׁל בִּעוּר חָמֵץ, עָזְרֵנִי לְהִנָּצֵל מֵהִרְהוּרִים רָעִים וּדְאָגוֹת, לְהִנָּצֵל מִמַּחְשָׁבוֹת זָרוֹת וְטִרְדוֹת, וְלִפְתֹּחַ אֶת לִבִּי לְהַכָּרָה בְּאַהֲבָתְךָ אֵלַי.

כְּמוֹ שֶׁהָאָבָק נֶאֱסָף אֶל תּוֹךְ הַמַּטְלִית, כָּךְ אֱסֹף אֶת כָּל הַדְּאָגוֹת וְהַפְּחָדִים שֶׁלִּי, וְתַשְׁלִיכֵם אֶל מְקוֹם שֶׁלֹּא יִמָּצְאוּ וְלֹא יִזָּכְרוּ, וְהַצְמַח בְּלִבִּי נִצָּנֵי אֱמוּנָה וְהוֹדָיָה.

שַׁעַר שִׁשִּׁי: לְהִתְחַתֵּן עִם הַבּוֹרֵא

קְדוֹשׁ יִשְׂרָאֵל וְגוֹאֲלוֹ, חָתָן הַנֶּאֱמָן

אָבִי, חָתָן יִשְׂרָאֵל, קְרָבוֹ מוֹעֵד הַפֶּסַח, הַלַּיְלָה בּוֹ אַתָּה עָתִיד לְהִכָּנֵס אֶל בִּיתִי. בְּנִקָּיוֹן זֶה אֲנִי מְכַבֵּד וּמְיַפֶּה אֶת חַדְרֵי בֵּיתִי, מְקַשֵּׁט אֶת הִיכָלְךָ, מַצִּיעַ אֶת מִטָּתְךָ.

כִּי כַּאֲשֶׁר אַתָּה בּוֹחֵר בִּי, בּוֹרְאִי, אֵין זוֹ רַק גְּאֻלָּה, אֶלָּא גַּם חֲתֻנָּה. זוֹהִי לֹא רַק יְצִיאַת מִצְרַיִם, אֶלָּא גַּם כְּנִיסָה לְבֵרִית אֵירוּסִין וְנִשּׂוּאִין.

עָשְׂרִי נָא לִי בְּרָכָה וְהַצְלָחָה בַּעֲבוֹדָתִי, שֶׁתְּהֵא לְרָצוֹן לְפָנֶיךָ וּלְנַחַת רוּחַ, שֶׁתַּעֲזֹר לִי לְקַבֵּל אֶת יוֹם הַכִּפּוּרִים הַקָּטָן הַזֶּה, בְּאַהֲבָה גְּמוּרָה וּבְלֵב שָׁלֵם.

וְאֶזְכֶּה בְּלֵיל הַקַּדֵּשׁ לָשֶׁבֶת כְּבֶן חוֹרִין, לְהָסֵב עִמְּךָ בְּאַהֲבָה, לִשְׁתּוֹת מִכּוֹס הַיְּשׁוּעָה, וּלְהַרְגִּישׁ אֶת הַקִּרְבָה בֵּינֵינוּ, כְּקִרְבַת אִישׁ וְאִשָּׁה, כְּדָבֵקוּת חָתָן וְכַלָּה.

חֲתִימָה: מִמְּזוּזָה לִמְזוּזָה

רַחוּם וְחַנּוּן, רוֹקֵד עַל פִּתְחֵי בָתֵּינוּ

הִנֵּה עוֹמֵד אֲנִי בְּפֶתַח בֵּיתִי, אָבִי הַטּוֹב, מְטַהֵר אֶת הַמַּשְׁקוֹף וְאֶת הַמְּזוּזוֹת, כְּשֵׁם שֶׁבְּמִצְרַיִם שָׂמוּ אֲבוֹתֵינוּ דָּם עַל הַמְּזוּזוֹת, וְאַתָּה רָקַדְתָּ וְדִלַּגְתָּ בְּאַהֲבָה עֲצוּמָה.

בְּכָל בֵּית יְהוּדִי יֵשׁ קְדֻשָּׁה מְיֻחֶדֶת, כִּי בַּיִת יְהוּדִי הוּא הַמִּקְדָּשׁ הַפְּרָטִי, וְאַתָּה אוֹהֵב אוֹתָנוּ כְּאָב אֶת בָּנָיו, וְשָׂמֵחַ בָּנוּ בְּשִׂמְחַת חָתָן עַל כַּלָּה.

יְהִי רָצוֹן, שֶׁתָּחוֹן עָלַי וְעַל כָּל בְּנֵי בֵּיתִי, וּתְסַיְּעֵנִי לְנַקּוֹת אֶת הַחָמֵץ מִבֵּיתִי וּמִלִּבִּי, שֶׁאֶזְכֶּה לַחֹג אֶת חַג הַמַּצּוֹת בְּטָהֳרָה, וְלוֹמַר הַלֵּל בְּפֶה מָלֵא וּבְלֵב שָׁלֵם.

וִיקֻיַּם בָּנוּ מִקְרָא שֶׁכָּתוּב: "הַשָּׁמַיִם שָׁמַיִם לַה', וְהָאָרֶץ נָתַן לִבְנֵי אָדָם", שֶׁנִּזְכֶּה לַעֲשׂוֹת מֵהָאָרֶץ מִשְׁכָּן לָךְ, אָבִי, וְתִשְׁרֶה שְׁכִינָתְךָ בְּתוֹכֵנוּ תָּמִיד.

אָמֵן כֵּן יְהִי רָצוֹן.

יוֹדוּ לְךָ רַעְיוֹנַי

(אֶשְׂמַח בְּךָ)

מאת הצדיק
רבי ישראל אבוחצירא זצוק"ל
הבבא סאלי

א הודיה על יצירתי ועל מתן תורה
יוֹדוּ לְךָ רַעְיוֹנַי אֵל מִבֶּטֶן יוֹצְרִי עַל קִרְבָתְךָ בְּסִינַי לְהָאִיר אֶת נֵרִי עַל כֵּן בְּתַחֲנוּנַי אֲפָאֵר בְּשִׁירִי בְּכָל יָמַי וְשָׁנַי עַד לָעַד לְעוֹלָם
אֶשְׂמַח בְּךָ אֵל עוֹלָם אֶשְׂמַח בְּךָ גּוֹאֵל נַפְשִׁי אֶת גְאֻלַּת עוֹלָם
פירוש
רַעְיוֹנַי – מחשבותי. פתיחת הפיוט מעלה בזיכרון את פיוטו המפורסם של ר' ישראל נג'ארה "יודוך רעיוני". (ראו דניאל ז, כח)
אֵל מִבֶּטֶן יוֹצְרִי – הקב"ה, שיצרני בבטן אמי. (ירמיה א, ה: בְּטֶרֶם אֶצָּרְךָ בַבֶּטֶן יְדַעְתִּיךָ)
עַל קִרְבָתְךָ בְּסִינַי – על כך שהיית קרוב אלינו במעמד הר סיני וקרבתנו שם לעבודתך.
לְהָאִיר אֶת נֵרִי – להאיר את נשמתי באור התורה.
בְּכָל יָמַי וְשָׁנַי – כמו שכתוב: "בְּכָל־יוֹם אֲבָרְכֶךָּ וַאֲהַלְלָה שִׁמְךָ לְעוֹלָם וָעֶד". (תהלים קמה, ב)
ב בחירת ישראל מכל העמים
שְׂשׂוֹן לִבִּי הֵמָּה בְּזָכְרִי חֲסָדָיו כִּי בָחַר מִכָּל אֻמָּה יִשְׂרָאֵל עֲבָדָיו נֵצֶר נְטִיעָה תַמָּה הָאָבוֹת יְדִידָיו אֶבֶן צֶדֶק שְׁלֵמָה מִשֹׁרֶשׁ גִּדְּלָם
אֶשְׂמַח בְּךָ אֵל עוֹלָם...
פירוש
שְׂשׂוֹן לִבִּי הֵמָּה – כאשר אני זוכר את חסדיו של הקב"ה, אני שמח ומתרגש. (תהלים קיט, קיא)
כִּי בָחַר מִכָּל אֻמָּה – חסדו הגדול של הקב"ה עימנו הוא שבחר בנו מכל העמים להיות לו לעבדים. (ויקרא כה, נה; דברים ז, ו)
נֵצֶר נְטִיעָה תַמָּה – עם ישראל, שהם נצר (ענף) לאבות, המכונים "ידידים". (ספרי דברים, שנב)
אֶבֶן צֶדֶק שְׁלֵמָה – אנשים ישרים ותמימים – עם ישראל. (דברים כה, טו)
ג בקשה להנהגה כבראשונה
רְצוֹנְךָ אֶשְׁאֲלָה חַי בַּעַל הָרָצוֹן כַּאֲשֶׁר בַּתְּחִלָּה נָהַגְתָּ אֶת הַצֹּאן בְּמִרְעֶה טוֹב וְנִפְלָא וְהָיָה כָל עֶלְצוֹן רוֹעֵה יִשְׂרָאֵל סֶלָה הַאֲזֵן אֶת קוֹלָם
אֶשְׂמַח בְּךָ אֵל עוֹלָם...
פירוש
רְצוֹנְךָ אֶשְׁאֲלָה, חַי בַּעַל הָרָצוֹן – בקשתי שטוחה לפניך, האל, שאתה הוא מקור הרצון, היכול למלא כל רצון.
כַּאֲשֶׁר בַּתְּחִלָּה – כמו שהנהגת את עם ישראל כצאן ביד משה ואהרן. (תהלים עז, יב-כא)
בְּמִרְעֶה טוֹב וְנִפְלָא – וסיפקת את כל מחסורם בשליו ובמן – לחם פלאי מן השמים.
רוֹעֵה יִשְׂרָאֵל סֶלָה – אנו מבקשים ממך, הרועה הנצחי של עם ישראל, שמע את קולם וגאל אותם. (תהלים פ, ב)
ד בקשה לנקמה באויבים
אַתָּה הוּא הָאֱלֹהִים הָעֵד וְהַדַּיָּן קוּם נָא לִקְרַאת גְבָהִים צְאִי נָא מְזֻיָּן לִנְקֹם נִקְמָתְךָ מֵהֶם לְהַשְׁלִים כָּל עִנְיָן שִׁמְךָ וְכִסְאֲךָ שֶׁהֵם מֵעֵין כָּל חַי נֶעְלָם
אֶשְׂמַח בְּךָ אֵל עוֹלָם...
פירוש
אַתָּה הוּא הָאֱלֹהִים, הָעֵד וְהַדַּיָּן – על פי המשנה: "שהוא אל הוא היוצר... הוא הדיין הוא עד". (אבות ד, כב)
קוּם נָא לִקְרַאת גְבָהִים – צא נגד הרשעים (גבהי לב), אומות העולם.
צְאִי נָא מְזֻיָּן – צא מצויד בכלי קרב והשמד אותם. (ישעיהו מב, יג)
לְהַשְׁלִים כָּל עִנְיָן, שִׁמְךָ וְכִסְאֲךָ – שאין כסא ה' שלם בעולם עד שיכרית את זרע עמלק. (שמות יז, טז)
ה תקווה לחסד ה'
לְעֹשֵׂה נִפְלָאוֹת לְבַדּוֹ אֶת יְרִיבַי יָרִיב אֲקַוֶּה תָמִיד חַסְדּוֹ הַשְׁכֵּם וְהַעֲרֵב וְלֹא יָסֵךְ בַּעֲדוֹ אֶת טִירָתוֹ יַחֲרִיב וְאָז נָשִׁיר לִכְבוֹדוֹ הַשִּׁירִים בְּמָשְׁלָם
אֶשְׂמַח בְּךָ אֵל עוֹלָם...
פירוש
לְעֹשֵׂה נִפְלָאוֹת לְבַדּוֹ – כינוי לקב"ה. (תהלים קלו, ד)
אֶת יְרִיבַי יָרִיב – שלא אצטרך אני להלחם באויבים אלא ה' ילחם לי. (תהלים לה, א)
אֲקַוֶּה תָמִיד חַסְדּוֹ – לא אפסיק מלקוות לחסד ה'.
וְלֹא יָסֵךְ בַּעֲדוֹ – שלא יגן, יסוכך, על אויבי אלא יחריב את מבצריו.
וְאָז נָשִׁיר לִכְבוֹדוֹ – ואז, כשמלכות ה' תהיה שלמה, נוכל לשיר לו שיר חדש ושלם על גאולתנו. (הגדה של פסח)
ו חיזוק ציון והמקדש
חַזֵּק צִיּוֹן מוֹעֲדֵנוּ אֲשֶׁר מְכֻוֶּנֶת מוּל זְבוּל מִקְדָּשֵׁנוּ שֶׁהִיא הַנּוֹתֶנֶת כֹּחַ עֶלְיוֹן לְגַנּוֹ עֵדֶן הַמְכוֹנֶנֶת אֲשֶׁר בָּם כָּל חֶפְצֵנוּ הֵם מִקְדָּשׁ וְאוּלָם
אֶשְׂמַח בְּךָ אֵל עוֹלָם...
פירוש
חַזֵּק צִיּוֹן מוֹעֲדֵנוּ – חזק את ציון, ירושלים ומקום המקדש. (ישעיהו לג, כ)
אֲשֶׁר מְכֻוֶּנֶת מוּל זְבוּל מִקְדָּשֵׁנוּ – ירושלים של מטה מכוונת כנגד ירושלים ובית מקדש של מעלה. (זוהר חדש, ויצא מח ע"א)
שֶׁהִיא הַנּוֹתֶנֶת כֹּחַ עֶלְיוֹן לְגַנּוֹ – היא המשפיעה ונותנת כח לגן העדן השמימי.
הֵם מִקְדָּשׁ וְאוּלָם – בגן העדן השמימי נמצא המקדש והאולם. (חגיגה יב ע"ב)
ז ההבטחה לאברהם אבינו
זֹאת הָיְתָה לִּי בְעָנְיִי גְּדוֹלָה הַבְטָחָה לְאַב הָמוֹן נִסְיִי בֵּין בְּתָרִים זָרָחָה לֵאמֹר מָתַי וְאַיֵּה וּמְצֶאנָה מְנוּחָה בְּךָ כָּל מַאֲוַיַי אֵל תַּאֲוַת עוֹלָם
אֶשְׂמַח בְּךָ אֵל עוֹלָם...
פירוש
זֹאת הָיְתָה לִּי בְעָנְיִי – בימי צער הגלות אנו מתנחמים בהבטחתך הגדולה לאברהם אבינו. (תהלים קיט, נ)
לְאַב הָמוֹן נִסְיִי – אברהם אבינו, אביהם של עם ישראל, מכונה "אב המון" ו"נִסְיִי" על שום הנסיונות שנתנסה בהם.
בֵּין בְּתָרִים זָרָחָה – בברית בין הבתרים הובטח לאברהם שעתידים בניו להיגאל ולרשת את הארץ. (בראשית טו, יג-כא)
לֵאמֹר מָתַי וְאַיֵּה – בלשונו של ריה"ל בפיוט "ישן אל תרדם" – קץ הגאולה הובטח לאברהם ובזה אנו מתנחמים.
ח תפילה לגאולה
קֵץ יָמִים גַּלֵּה לְעַם אֵלֶּה עֲמוּסֵי תְלָאוֹת גַּדֵּל צַעֲרִי וַעֲמָלִי בְפִלְאֵי-פְלָאוֹת מְצַפִּים אָבוֹת וְעוֹלָלֵי עֲשֵׂה לְטוֹבָה אוֹת כְּלִילוֹ גְדִילֵי בָנֶיךָ עַל תִּלָּם
אֶשְׂמַח בְּךָ אֵל עוֹלָם אֶשְׂמַח בְּךָ גּוֹאֵל נַפְשִׁי אֶת גְאֻלַּת עוֹלָם
פירוש
קֵץ יָמִים גַּלֵּה – הבא כבר את הגאולה לעם ישראל ששבע כבר מצרות רבות ורעות ומתלאות הגלות.
גַּדֵּל צַעֲרִי וַעֲמָלִי בְפִלְאֵי-פְלָאוֹת – כשם שסבלנו והצטערנו רבות, כך הפלאות שתעשה עמנו תהיינה כפולות ומכופלות.
עֲשֵׂה לְטוֹבָה אוֹת – כל עם ישראל, הקטנים והגדולים מייחלים לישועתך ומבקשים שתראה להם אות לטובה. (תהלים פו, יז)
כְּלִילוֹ גְדִילֵי, בָנֶיךָ עַל תִּלָּם – החזר עטרה ליושנה והושב את עם ישראל על מכונו. "כְּלִילוֹ" ו"גְדִילֵי" במשמעות של עטרה, כתר. (עבודה זרה מא ע"א)

אָעוּפָה אֶשְׁכֹּנָה

פִּיּוּט עַל כִּסּוּפִים לְהַקָּדוֹשׁ בָּרוּךְ הוּא

רַבִּי יִצְחָק אֲבִיחֲצִירָא

אָעוּפָה אֶשְׁכֹּנָה וְאַרְחִיקָה נְדֹד
בַּמִּדְבָּר אָלִינָה וְאוּלַי אֶמְצָא דּוֹד
מָה זֶה אוֹמֵר?
הַמְשׁוֹרֵר מִתְאַר שֶׁהוּא רוֹצֶה לָעוּף כְּמוֹ צִפּוֹר וּלְנַדֵּד בַּמִּדְבָּר, רַק כְּדֵי לִמְצֹא אֶת הַקָּדוֹשׁ בָּרוּךְ הוּא (שֶׁהוּא קוֹרֵא לוֹ "דּוֹד" - כְּמוֹ חָבֵר אָהוּב). כָּל כָּךְ הוּא מִתְגַּעְגֵּעַ לְהַשֵּׁם!
נֶשֶׁק אַהֲבָתוֹ בְּלִבִּי בֹּעֵרָה
מִיּוֹם פְּרֵדָתוֹ נַפְשִׁי עָלַי מָרָה
מָה זֶה אוֹמֵר?
הָאַהֲבָה לְהַקָּדוֹשׁ בָּרוּךְ הוּא בּוֹעֶרֶת בְּלִבּוֹ כְּמוֹ אֵשׁ! מִיּוֹם שֶׁהוּא מַרְגִּישׁ רָחוֹק מֵהַשֵּׁם (בְּעַקְבוֹת הַגָּלוּת), הַנֶּפֶשׁ שֶׁלּוֹ עֲצוּבָה וּמָרָה.
יְדִיד מֶנִי בָּרַח הָלַךְ עֲזָבַנִי
אֵיזוֹ דֶּרֶךְ אָרַח וְאֵלְכָה גַּם אֲנִי
מָה זֶה אוֹמֵר?
הוּא מַרְגִּישׁ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא (שֶׁהוּא הַיְּדִיד הֲכִי טוֹב שֶׁלּוֹ) הָלַךְ מִמֶּנּוּ. אָז הוּא שׁוֹאֵל: "בְּאֵיזֶה דֶּרֶךְ אֲנִי צָרִיךְ לָלֶכֶת כְּדֵי לִמְצֹא אוֹתוֹ שׁוּב?"
יָצָאתִי לְבַקֵּשׁ דּוֹדִי בֵּין חֲבֵרִים
נִלְכַּדְתִּי בְּמוֹקֵשׁ הִכּוּנִי הַשֹּׁמְרִים
מָה זֶה אוֹמֵר?
הוּא יָצָא לְחַפֵּשׂ אֶת הַקָּדוֹשׁ בָּרוּךְ הוּא, אֲבָל בַּדֶּרֶךְ נִתְקַל בְּקָשָׁיִים וּמְכְשׁוֹלִים (כְּמוֹ שׁוֹמְרִים שֶׁמַּכִּים אוֹתוֹ). זֶה מְסַמֵּל אֶת הַקְּשָׁיִים שֶׁעַם יִשְׂרָאֵל עוֹבֵר בַּגָּלוּת.
צִפִּיתִי לְדוֹדִי מָתַי יָבוֹא אֵלַי
יַלְבִּישֵׁנִי עֶדְיִי וִירַחֵם עָלַי
מָה זֶה אוֹמֵר?
הוּא מְחַכֶּה וּמְצַפֶּה: מָתַי הַקָּדוֹשׁ בָּרוּךְ הוּא יָבוֹא אֵלָיו? מָתַי הוּא יְקַשֵּׁט אוֹתוֹ שׁוּב בְּתַכְשִׁיטִים (כְּלוֹמַר, יַחֲזִיר לְעַם יִשְׂרָאֵל אֶת הַכָּבוֹד שֶׁלּוֹ) וִירַחֵם עָלָיו?

💫 רֶגַע שֶׁל מַחֲשָׁבָה 💫

עַד עַכְשָׁיו הַמְּשׁוֹרֵר דִּבֵּר עַל הַכִּסּוּפִים שֶׁלּוֹ לְהַקָּדוֹשׁ בָּרוּךְ הוּא.
עַכְשָׁיו - הַשֵּׁם יִתְבָּרַךְ עוֹנֶה לוֹ!
בּוֹאוּ נִרְאֶה מָה הוּא אוֹמֵר...

חֲשֹף זְרוֹעֲךָ לְקַבֵּץ פְּזוּרִים
גַּלֵּה קֵץ יִשְׁעֲךָ וְדִגְלְךָ הָרֵם
מָה זֶה אוֹמֵר?
הַמְּשׁוֹרֵר מִתְפַּלֵּל: "הַקָּדוֹשׁ בָּרוּךְ הוּא, גַּלֵּה אֶת הַכֹּחַ הַגָּדוֹל שֶׁלְּךָ! קַבֵּץ אֶת כָּל עַם יִשְׂרָאֵל מִכָּל הָעוֹלָם! גַּלֵּה לָנוּ מָתַי תָּבוֹא הַגְּאֻלָּה!"
קוּמִי יְחִידָתִי וְשׁוּבִי בִּתְשׁוּבָה
אֲחוֹתִי רַעְיָתִי הִנֵּה גוֹאֲלֵךְ בָּא
מָה זֶה אוֹמֵר?
עַכְשָׁיו הַקָּדוֹשׁ בָּרוּךְ הוּא מְדַבֵּר אֶל עַם יִשְׂרָאֵל (שֶׁהוּא קוֹרֵא לוֹ "יְחִידָתִי" - הָאֲהוּבָה שֶׁלִּי): "קוּמִי! תַּחְזְרִי בִּתְשׁוּבָה! הַגּוֹאֵל שֶׁלָּךְ - הַמָּשִׁיחַ - כְּבָר בָּא!"
חֲמוּדָה יְקָרָה רַבַּת הַמַּעֲלוֹת
גַּם מִפָּז נִבְחָרָה צְאִי נָא בִמְחוֹלוֹת
מָה זֶה אוֹמֵר?
הַקָּדוֹשׁ בָּרוּךְ הוּא אוֹמֵר לְעַם יִשְׂרָאֵל: "אַתְּ חֲמוּדָה וִיקָרָה לִי יוֹתֵר מִזָּהָב! צְאִי וְשִׂמְחִי בְּמָחוֹל!" - הוּא מְבַשֵּׂר לָנוּ שִׂמְחָה גְדוֹלָה!
זָכַרְתִּי לָךְ חֶסֶד נְעוּרַיִךְ נְעוּרִים
הֵיכָלֵךְ אֲיַסֵּד בְּאַבְנֵי סַפִּירִים
מָה זֶה אוֹמֵר?
הַשֵּׁם יִתְבָּרַךְ אוֹמֵר: "אֲנִי זוֹכֵר אֶת הַטּוּב שֶׁעָשִׂית בִּימֵי הַנְּעוּרִים שֶׁלָּךְ (בְּמִדְבָּר סִינַי). אֲנִי אֶבְנֶה לָךְ אֶת בֵּית הַמִּקְדָּשׁ מֵחָדָשׁ, מֵאֲבָנִים יְקָרוֹת וּמְפֹאָרוֹת!"
קוֹל דּוֹדִי הִנֵּה בָּא מְדַלֵּג עַל הֶהָרִים
קוּמִי לָךְ אֲהוּבָה כִּי בָּא קֵץ דְּרוֹרִים
מָה זֶה אוֹמֵר?
הַפִּיּוּט מִסְתַּיֵּם בְּבִשּׂוֹרָה נִפְלָאָה: "שִׁמְעוּ! קוֹל הַגּוֹאֵל בָּא! הוּא קוֹפֵץ עַל הֶהָרִים מֵרֹב שִׂמְחָה! קוּמִי, אֲהוּבָה שֶׁלִּי - כִּי הִגִּיעַ זְמַן הַחֵרוּת וְהַגְּאֻלָּה!"

✨ הַמֶּסֶר שֶׁל הַפִּיּוּט ✨

הַפִּיּוּט הַזֶּה מְלַמֵּד אוֹתָנוּ כַּמָּה הַקָּדוֹשׁ בָּרוּךְ הוּא אוֹהֵב אוֹתָנוּ,
וְכַמָּה אֲנַחְנוּ צְרִיכִים לְהִתְגַּעְגֵּעַ אֵלָיו!

גַּם כְּשֶׁאֲנַחְנוּ בְּקָשָׁיִים, הַשֵּׁם לֹא שׁוֹכֵחַ אוֹתָנוּ.
הוּא מְחַכֶּה שֶׁנַּחֲזֹר אֵלָיו, וּמְבַשֵּׂר לָנוּ גְּאֻלָּה שְׁלֵמָה!

🕊️ שֶׁנִּזְכֶּה בִּמְהֵרָה לִרְאוֹת אֶת הַגְּאֻלָּה! 🕊️

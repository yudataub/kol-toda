תְּפִלַּת שִׁבְעַת הַיָּמִים – הַשְּׁכִינָה בְּמַאֲכַל הָאֱמוּנָה

תְּפִלַּת שִׁבְעַת הַיָּמִים – הַשְּׁכִינָה בְּמַאֲכַל הָאֱמוּנָה

שַׁעַר רִאשׁוֹן: מַצָּה – אֲכִילַת הַנְּשָׁמָה

רִבּוֹן הָעוֹלָמִים, בּוֹרֵא נְפָשׁוֹת רַבּוֹת וְחֶסְרוֹנָן

הִנֵּה בָאתִי לְפָנֶיךָ, אֲהוּבִי, מְבַקֵּשׁ לִטְעֹם מִמַּאֲכַל הָאֱמוּנָה, לֶחֶם דַּק, לֶחֶם עֹנִי, פְּרוּסַת נִשְׁמָתְךָ הַמְּרֻבַּעַת. הִנֵּה יָדַי פְּרוּשׂוֹת לְקַבֵּל אֶת הַמַּצָּה, כְּנַפְשִׁי הַיּוֹנָה פְּרוּשַׂת כְּנָפַיִם אֵלֶיךָ.

אִם בְּלֵיל קַדֵּשׁ יֵשׁ חוֹבָה לֶאֱכֹל מִמֶּנָּה, הֲלֹא בְּכָל שִׁבְעַת הַיָּמִים יֵשׁ מִצְוָה מְתוּקָה, כְּעֵדוּת הַגָּאוֹן: "כָּל שִׁבְעָה מִצְוָה מִדְּאוֹרַיְתָא" לְהִתְרַפֵּק עַל דּוֹדִי בְּכָל נְגִיסָה וּנְגִיסָה.

הֲרֵינִי רוֹצֶה, אֵל חַי, לְקַיֵּם מִצְוַת עֲשֵׂה שֶׁל "שִׁבְעַת יָמִים מַצּוֹת תֹּאכֵלוּ", כִּי בְּכָל אֲכִילָה נִכְנָסִים אֶל עֲמָקֵי הַמְּהֵימְנוּתָא, כְּדִבְרֵי הַזֹּהַר: "כַּד אָכְלוּ מַצָּה בְּנֵי יִשְׂרָאֵל, עָאֲלוּ לִמְהֵימְנוּתָא", וּבְכָל לְעִיסָה נִמְתָּחִים חוּטֵי הָאֱמוּנָה בֵּין נַפְשִׁי לְנַפְשֶׁךָ.

שַׁעַר שֵׁנִי: הַמַּצָּה כִּמְרִיבָה קְדוֹשָׁה

אָדוֹן הַנִּפְלָאוֹת, צַר וּמְרִיב כָּל אוֹיְבֵי נַפְשִׁי

"מַצָּה, מַה הִיא?" שׁוֹאֵל הַזֹּהַר, וְעוֹנֶה: "עֲנִי מְרִיבָה", לְשׁוֹן מַצָּה וּמְרִיבָה. מְרִיבָה הִיא זוֹ עִם כֹּחוֹת הַסִּטְרָא אַחֲרָא, נִלְחֶמֶת בְּכָל צַד רַע, בְּכָל סִטְרִין בִּישִׁין.

רִבּוֹנוֹ שֶׁל עוֹלָם, בְּתוֹךְ חֲלָלִי פְּנִימָה מְרִיבוֹת רַבּוֹת – מְרִיבַת הַנֶּפֶשׁ, מְרִיבַת הַיֵּצֶר. אֲנִי מְבַקֵּשׁ, בְּאָכְלִי מַצָּה, שֶׁתִּכָּנֵס בִּי מְרִיבָתְךָ הַקְּדוֹשָׁה, שֶׁהֲרֵי אַתָּה אִישׁ מִלְחָמָה, ה' שְׁמֶךָ.

לְחֶם עֹנִי זֶה נִרְאֶה כְּדַל וּפָשׁוּט, אַךְ רַב-עָצְמָה הוּא בְּעוֹלָמוֹת הָעֶלְיוֹנִים. הָאֵר עֵינַי לִרְאוֹת בְּכָל שִׁבְעַת הַיָּמִים, אֵיךְ מַצָּה זוֹ מְגָרֶשֶׁת פְּנִימָה אֶת כָּל עַבְדוּת מִצְרַיִם.

בְּכָל כַּזַּיִת מַלְאָכִים חֲדָשִׁים נוֹלָדִים בִּי, מְגִנֵּי רוּחַ, עוֹרְכֵי מְרִיבָה קְדוֹשָׁה בְּשֵׁם ה', וְהִנֵּה כַּדּוּרִיּוֹת דָּם אֱמוּנָה זוֹרְמוֹת בְּקִרְבִּי, לוֹחֲמוֹת בְּכָל חֲשֵׁכָה וָזֹר.

שַׁעַר שְׁלִישִׁי: לֶחֶם הָאֵרוּסִין

אוֹהֵב עַמּוֹ יִשְׂרָאֵל, דּוֹדִי הַדָּבֵק בִּי תָּמִיד

בְּמִצְרַיִם הִתְאָרַסְנוּ לְךָ, אֲהוּבִי, אַתָּה לָקַחְתָּ אוֹתָנוּ לְעַם בִּדְמֵי כְּלוּלוֹת. לֵיל הַסֵּדֶר הוּא לֵיל הַכְּלוּלוֹת שֶׁלָּנוּ, וְשִׁבְעַת הַיָּמִים הֵם שֶׁבַע בְּרָכוֹת שֶׁל אַהֲבָתֵנוּ.

לֶחֶם הַמַּצּוֹת, לֶחֶם חֶפְזוֹן שֶׁל אַהֲבָה, מַזְכִּיר לִי שֶׁלֹּא הִתְמַהְמַהְתָּ אֲפִלּוּ רֶגַע, בָּחַרְתָּ לִגְאֹל אוֹתִי בִּתְשׁוּקָה בּוֹעֶרֶת, לֹא יָכֹלְתָּ לְהִתְאַפֵּק לִרְאוֹת אוֹתִי בְּצַעַר.

אֵיךְ אוּכַל שֶׁלֹּא לֶאֱכֹל שִׁבְעַת יָמִים לֶחֶם זֶה? אֵיךְ אוּכַל שֶׁלֹּא לִטְעֹם טַעַם אַהֲבָתְךָ בְּכָל סְעוּדָה? יְקָרָה לִי, רִבּוֹנִי, הַמִּצְוָה הַזּוֹ מִפָּז, מִצְוַת שִׁבְעַת יְמֵי הָאַהֲבָה.

הֲרֵינִי כַּלָּה הַטּוֹעֶמֶת מַעֲדַנֵּי חָתָן, שְׁבִיעָה וּרְעֵבָה, מְלֵאָה וַחֲסֵרָה – שְׁבִיעָה מִדַּעַת אֱלֹהִים וּרְעֵבָה עוֹד, מְלֵאָה מֵאַהֲבָתְךָ וַחֲסֵרָה לְךָ תָּמִיד.

שַׁעַר רְבִיעִי: מַצָּת הַבְּחִירָה הָאֱלֹהִית

מַלְכִּי וֵאלֹהַי, בּוֹחֵר עַמּוֹ יִשְׂרָאֵל בְּאַהֲבָה

בְּחַג זֶה אֲנִי טוֹעֵם אֶת "אַתָּה בְחַרְתָּנוּ", וַאֲפִלּוּ בְּמַאֲכָלִי מְרַחֶפֶת נְשִׁימַת בְּחִירָתְךָ. אֲנִי מַרְגִּישׁ אוֹתָהּ עַל שְׂפָתַי, בֵּין שִׁנַּי, יוֹרֶדֶת אֶל מְעַי וְנִבְלַעַת בִּנְתִיבֵי גוּפִי.

"מַה טּוֹב חֶלְקֵנוּ וּמַה נָּעִים גּוֹרָלֵנוּ", הִנֵּה הַשֶּׁבַח הַמְּרַצֵּד בֵּין רְסִיסֵי הַמַּצָּה, וְעַל כָּל כַּזַּיִת מִתְעַלֶּה הַקְּרִיאָה: "אַשְׁרֵינוּ שֶׁאָנוּ מַשְׁכִּימִים וּמַעֲרִיבִים לָךְ".

יְקָרָה לִי, רִבּוֹנִי, הַמִּצְוָה שֶׁל שִׁבְעַת יָמִים מַצּוֹת תֹּאכֵלוּ – כִּי כָל פְּרוּסַת מַצָּה הִיא עֵדוּת עַל בְּחִירָתְךָ בִי, וְהַשִּׂמְחָה בְּ"אַתָּה בְחַרְתָּנוּ" הִיא עֲבוֹדָתִי הַגְּדוֹלָה בַּחַג, כְּמַאֲמַר רַב "יְסוֹד וְשֹׁרֶשׁ הָעֲבוֹדָה".

בְּכָל אֲכִילָה וַאֲכִילָה מְלַחֵשׁ לִבִּי: "מוֹדֶה אֲנִי לָךְ שֶׁבָּחַרְתָּ בִּי מִכָּל הָעַמִּים", וּמִן הַפֶּה יוֹצֵא הָאֶבֶל וְנִכְנָס הַטּוּב – אֶבֶל מִצְרַיִם נִמְחָק וְנִכְנָס אוֹר גְּאוּלָה.

שַׁעַר חֲמִישִׁי: מַצָּה כְּאוֹת בָּעוֹלָמוֹת

קָדוֹשׁ יִשְׂרָאֵל וְגוֹאֲלוֹ, יוֹצֵר אוֹתוֹת בַּבְּרִיאָה

מְלַמְּדִים אוֹתָנוּ חֲכָמֵינוּ: הַמַּצָּה הִיא "אוֹת" בְּמוֹעֵד, אוֹת יָקָר מֵאוֹת תְּפִלִּין, אוֹת שֶׁמַּקִּיף אֶת כָּל שִׁבְעַת הַיָּמִים. אָנָּא, הָאֵר עֵינַי לְהַכִּיר בְּכָל אֲכִילָה אֶת הָאוֹת הַמֻּפְלָא שֶׁנָּתַתָּ בְּתוֹכִי.

הַמַּצָּה אֵינָהּ רַק מַאֲכָל פָּשׁוּט, הִיא כְּתַב רָזִין דַּקִּיק מִמְּךָ אֵלַי, סֵפֶר תּוֹרָה זָעִיר שֶׁאֲנִי בּוֹלֵעַ, וְהוּא מְפַעֲנֵחַ בִּמְעַי אֶת סוֹדוֹת הָאֱמוּנָה.

כְּשֶׁהַמַּצָּה יוֹרֶדֶת לְמַטָּה, הַנְּשָׁמָה מִתְעַלָּה לְמַעְלָה, כְּשֶׁהַמַּצָּה נִבְלַעַת, הָאֱמוּנָה מִתְגַּלָּה, כְּשֶׁהַמַּצָּה מִתְפָּרֶקֶת, שְׁמְךָ הָאֱלֹהִי מִתְחַבֵּר, וְאוֹתִיּוֹת רְשׁוּמוֹת בִּי חֲקִיקוֹת חֲקִיקוֹת.

וּכְשֶׁאֲנִי מַמְשִׁיךְ לֶאֱכֹל גַּם בַּיָּמִים הַבָּאִים, מִתְחַדֶּשֶׁת הָאֱמוּנָה בְּכָל יוֹם מֵחָדָשׁ, וְהָאוֹת הוֹלֵךְ וּמִתְחַזֵּק, נִרְשָׁם וְנִטְבָּע, עַד שֶׁאֲנִי כֻּלִּי אוֹת אֶחָד לְךָ, אֱלֹהַי.

שַׁעַר שִׁשִּׁי: טַעַם הַמַּצָּה לָאֹרֶךְ הַשָּׁנָה

שׁוֹמֵר עַמּוֹ יִשְׂרָאֵל לָעַד, מַשְׁגִּיחַ עָלֵינוּ תָּמִיד

כָּתוּב בְּתוֹרָתְךָ: "לְמַעַן תִּזְכֹּר אֶת יוֹם צֵאתְךָ מֵאֶרֶץ מִצְרַיִם כֹּל יְמֵי חַיֶּיךָ", וְרַבּוֹתֵינוּ דָּרְשׁוּ: "כֹּל יְמֵי חַיֶּיךָ" – לְהָבִיא אֶת יְמוֹת הַמָּשִׁיחַ. וְאַף אֲנִי מִתְפַּלֵּל: יְהִי רָצוֹן שֶׁטַּעַם מַצּוֹת אֵלּוּ יִשָּׁאֵר עִמִּי כָּל הַשָּׁנָה, וַאֲפִלּוּ לִימוֹת הַמָּשִׁיחַ.

שִׁבְעַת יְמֵי הַמַּצָּה הֵם חֲנֻכַּת הַנֶּפֶשׁ לְשָׁנָה שְׁלֵמָה, שִׁבְעָה מַעְיָנוֹת שֶׁאֲנִי שׁוֹתֶה מֵהֶם אֱמוּנָה, וְהַמַּיִם הַקְּדוֹשִׁים הָאֵלֶּה יַמְשִׁיכוּ לִזְרֹם בְּתוֹךְ עוֹרְקַי כָּל הַשָּׁנָה כֻּלָּהּ.

זַכֵּנִי, מְקוֹר הַחַיִּים, לְהַשְׁאִיר בְּזִכְרוֹנִי אֶת הַמִּצְוָה שֶׁל "שִׁבְעַת יָמִים מַצּוֹת תֹּאכֵלוּ", שֶׁאַף לְאַחַר הַפֶּסַח אֶמְשִׁיךְ לְהַרְגִּישׁ שֶׁאֲנִי בָּחוּר, אָהוּב, קָרוֹב אֵלֶיךָ.

וּכְשֶׁאֲנִי אוֹכֵל לֶחֶם כָּל הַשָּׁנָה, עָמֹק בְּלִבִּי יִזְרַח זֵכֶר הַמַּצָּה, הַלֶּחֶם שֶׁל הָאֱמוּנָה, שֶׁל הָאַהֲבָה, לֶחֶם הַגְּאֻלָּה שֶׁשָּׁמַרְתָּ לִי מֵרֵאשִׁית.

חֲתִימָה: הָאֲכִילָה הַפְּנִימִית

מְחַיֶּה חַיִּים בְּטוּבוֹ, מֵזִין עוֹלָמוֹת בְּרַחֲמִים

הִנֵּה הָרַעְיוֹן הַמִּתְנַשֵּׂא מֵעַל כָּל הַדְּבָרִים: בְּכָל עֵת שֶׁאֲנִי אוֹכֵל מַצָּה בְּיָמִים אֵלּוּ, אֵינִי רַק אוֹכֵל מַאֲכָל – אֲנִי בּוֹלֵעַ אוֹתִיּוֹת, אוֹכֵל אַהֲבָה, אוֹכֵל בְּחִירָה, אוֹכֵל רָזִין עִלָּאִין.

הַמַּצָּה הִיא סֻלָּם רוּחָנִי – פְּרוּסוֹת פְּרוּסוֹת אֲנִי מִתְעַלֶּה בְּמַדְרֵגוֹת שֶׁל אֱמוּנָה. הַמַּצָּה הִיא מִכְחוֹל – בְּנְגִיסוֹת נְגִיסוֹת מְצַיֵּר בְּנַפְשִׁי צוּרַת הַמְלוּכָה הָאֱלֹהִית.

אָבִי שֶׁבַּשָּׁמַיִם, אוֹדְךָ בְּכָל לִבִּי עַל מִצְוַת אֲכִילַת מַצָּה כָּל שִׁבְעַת הַיָּמִים, כִּי בָהּ אַתָּה מְזַכֶּה אוֹתִי שֶׁאַכְנִיס לְתוֹכִי אֶת סוֹד הַגְּאֻלָּה בְּצוּרָה מַמָּשִׁית וּמְזִינָה.

וּבַיּוֹם הַשְּׁבִיעִי, כְּשֶׁאֲנִי אוֹכֵל אֶת הַסְּעוּדָה הָאַחֲרוֹנָה, אֲנִי מוֹדֶה לְךָ עַל שִׁבְעַת יְמֵי הַבְּרָכָה שֶׁזָּכִיתִי בָּהֶם. הִנֵּה עוֹמֵד אֲנִי בְּסוֹף שֶׁבַע הַבְּרָכוֹת, אוֹחֵז בִּכְלִי הִתְקַשְּׁרוּתֵנוּ, וְיוֹדֵעַ: אֲנִי שֶׁלְּךָ, אַתָּה שֶׁלִּי, אַתָּה בְחַרְתָּנוּ.

יְהִי רָצוֹן מִלְּפָנֶיךָ, אֵל רָם וְנִשָּׂא, שֶׁזְּכוּת אֲכִילַת הַמַּצָּה, תַּעֲמֹד לִי וּלְכָל יִשְׂרָאֵל חֲבֵרִים, וּכְשֵׁם שֶׁזָּכִינוּ לֶאֱכֹל מַצָּה בְּפֶסַח זֶה, כֵּן נִזְכֶּה לֶאֱכֹל מִן הַזְּבָחִים וּמִן הַפְּסָחִים בְּבֵית מִקְדָּשְׁךָ הָעֲתִידִי.

וִיקֻיַּם בָּנוּ מִקְרָא שֶׁכָּתוּב: "כִּי הוֹצִיאֲךָ ה' אֱלֹהֶיךָ מֵאֶרֶץ מִצְרַיִם מִבֵּית עֲבָדִים – מִיַּד חֲזָקָה וּבִזְרוֹעַ נְטוּיָה, עַל כֵּן צִוְּךָ ה' אֱלֹהֶיךָ לַעֲשׂוֹת אֶת יוֹם הַשַׁבָּת".

אָמֵן, כֵּן יְהִי רָצוֹן.

שִׁיר לְזֵכֶר הָרַב אֱלִיָּהוּ רוֹטֶנְבֶּרְג זַצַ"ל

קִינַת קִרְיַת אַתָּא

הֲלֹא נִבְכֶּה, קִרְיַת אַתָּא הָאֲהוּבָה?
הֲלֹא נִסְפְּדָה לְאַבִּיר לֵב וְרוּחַ נְדִיבָה?
כָּל רְחוֹבוֹתַיִךְ יַעַמְדוּ בְּשַׁעַר דּוֹמְמִים
כָּל שְׂדֵרוֹתַיִךְ בְּכאב, בְּצַעֲר דּוֹמְעִים
לְאִישׁ נֶאֱמָן שֶׁמִּמֵּנוּ הָלַךְ
לְאוֹהֵב אלֹקִים שֶׁמִּמֵּנוּ נִלְקַח
יֶחֱרַשׁ בְּאֶבְלוֹ רְחוֹב הַמְּיַסְּדִים
יְלַוּוּהוּ בְּכָבוֹד תּוֹשְׁבֵי עִיר אוֹהֲדִים
שָׁם עוֹדֶנָּה הַיְשִׁיבָה הַקְּטַנָּה, צוֹמַחַת וּמִתְפַּתַּחַת
מְלַוָּה אוֹתוֹ הַתּוֹרָה בְּאַהֲבָתָהּ אוֹתָהּ אָהָב, הָרוֹתַחַת
עוֹלֶה קוֹל הַתּוֹרָה, הַמּוּסָר, הַהֲלָכָה, מִמֶּנָּה
כְּמִנְחָה
נִשְׁמַת הַמְּיַסֵּד בָּהּ שׁוֹכֶנֶת וּמִתְבָּרֶכֶת
מִתּוֹכָהּ
בֵּית הַכְּנֶסֶת "דֶּרֶךְ אֱמוּנָה" יַעַן וְיֹאמַר:
"עֵינַי דֹּמְעוֹת עַל הַמִּתְפַּלֵּל הַנֶּעדָּר"
הֲלֹא פֹּה עָמַד לִפְנֵי עַמּוּד הַתְּפִלָּה
קוֹלוֹ הַנֶּאֱמָן עוֹד מְהַדְהֵד בֵּין כְּתָלַי, כְּעוֹלָה
הָעִיר כֻּלָּהּ הוֹמָה וְסוֹעֶרֶת כַּיָּם
נִבְנְתָה קְהִלָּה מִכֹּחוֹ, בְּיֶזַע, בְּדָם
מֵחוֹמוֹת בָּתֶּיהָ עַד גְּבוּלָהּ
מִי יַחֲזִיק בְּיָדָיו הַטּוֹבוֹת אֶת מַשָּׂא הַקְּהִלָּה?
כָּל מוֹסְדוֹת הַחִנּוּךְ יַעַמְדוּ בְדוּמִיָּה
לְמִי עוֹד הָיְתָה כָּזוֹ סִיעְתָּא דִשְׁמַיָּא?
מִי יַעֲמֹד בַּפֶּרֶץ לְהָקִים מוֹקְדֵי קְדֻשָּׁה? לְסַלֵּק חֶשְׁכָה?
מִי יִתְגַּלְגֵּל בָּעוֹלָם לְהָבִיא קֶמַח וּתְמִיכָה?
קְהִלַּת יִשְׂרָאֵל בָּעִיר תֵּלֵךְ אֲבֵלָה
מֵעַמּוּדֵי הַתָּווֶךְ שֶׁלָּנוּ הָיָה
מֵרִאשׁוֹנֵי הַמִּתְיַשְּׁבִים בְּזוֹ הָעִיר
אָבַד נְשִׂיא וּמַשְׁפִּיעַ בַּדּוֹר, וּמֵאִיר
אַף הַמּוֹעֵצָה הָעִירוֹנִית תַּעֲמֹד וְתִתְאַבֵּל
אִישׁ יוֹשֶׁר וּמֵישָׁרִים הָיָה
פָּעַל לְמַעַן כֻּלָּם בְּלִי הֶבְדֵּל
כִּבֵּד כָּל לָשׁוֹן וְעֵדָה, אוֹהֵב יִשְׂרָאֵל
מִי יְעוֹרֵר אוֹתָנוּ לִתְשׁוּבָה?
הַאִם הַכֹּל נִגְמַר?
מִי יְכַוֵּן אוֹתָנוּ לַתּוֹרָה?
יִצְעֲקוּ הָרְחוֹבוֹת בְּקוֹל מָר
הֵיכָן הַהוֹלֵךְ בָּנוּ בְּזְמַן צָרָה וּבְעִתָּה?
שָׁמַיִם, אַל תֶּחֱרְשׁוּ, אַל תִּדֹּמִי קִרְיַת אַתָּא!
אַךְ מִתּוֹךְ הָאֲבֵלוּת תֵּצֵא הַתְּשׁוּעָה
לֹא יַעֲזֹב ה' אֶת עַמּוֹ בְּכָל שָׁעָה
תּוֹרָתוֹ וּמוֹרַשְׁתּוֹ הַכֹּל שָׁרִיר וְקַיָּם
הֲלֹא יַעַמְדוּ בִּמְקוֹמוֹ פֹּה בָּעִיר, זַרְעוֹ וְזַרְעָם
הַמּוֹסָדוֹת פּוֹרְחִים אֶת אֱמוּנָתוֹ לוֹמְדִים
אֵין עוֹד מִלְבַדּוֹ כָּל אֶחָד יִזְכּוֹר
לַעֲשׂוֹת וּלְקַיֵּם כָּל אֶחָד מִדַּרְכּוֹ יִקַּח דְּבַר מָה לִשְׁמֹר
וּשְׁמוֹ הַטּוֹב יִזָּכֵר לְדוֹר וָדוֹר
יִתְנַחֵם לִבֵּנוּ בִּזְכוּתוֹ הָעֲצוּמָה
כִּי הִנְחִיל לָעוֹלָם מַתָּנָה יְקָרָה
דּוֹר תַּלְמִידֵי חֲכָמִים וּמַעֲשִׂים טוֹבִים
נוֹשְׂאֵי הַלַּפִּיד בְּכָל הַדְּרָכִים
וְעַד יוֹם בּוֹא מָשִׁיחַ צִדְקֵנוּ
נִזְכֹּר אֶת הָאִישׁ שֶׁבָּנָה אֶת עִירֵנוּ
דְּבוּקָה וּכְרוּכָה בִּשְׁמוֹ קִרְיַת אַתָּא
נוֹדֶה לַה' עַל יוֹם אֲשֶׁר אֵלֶיהָ בָּאתָ
תְּהֵא נִשְׁמָתוֹ צְרוּרָה בִּצְרוֹר הַחַיִּים
תָּנוּחַ בִּמְנוּחָה נְכוֹנָה עִם צַדִּיקִים
וְזִכְרוֹ בָּרוּךְ לְעוֹלָם וָעֶד

שִׁיר זֶה נִכְתַּב בְּאַהֲבָה וּבְכָבוֹד לְזֵכֶר נִשְׁמַת הָרַב אֱלִיָּהוּ יִצְחָק רוֹטֶנְבֶּרְג זַצַ"ל
מְיַסֵּד עוֹלָם הַתּוֹרָה בְּקִרְיַת אַתָּא

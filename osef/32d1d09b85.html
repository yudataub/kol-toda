תְּפִלַּת הַפֶּה הַמְדַבֵּר - סוֹד הַמָּן וְסְעֻדַּת אֶסְתֵּר

תְּפִלַּת הַפֶּה הַמְדַבֵּר - סוֹד הַמָּן וְסְעֻדַּת אֶסְתֵּר

שַׁעַר רִאשׁוֹן: הַפֶּה הַנִּגְאָל

רִבּוֹן כָּל הָעוֹלָמִים, הַשּׁוֹכֵן בְּנִגּוּן הַדּוֹמֵם וְהַשָּׂרָף כְּאֶחָד

הִנֵּה בָּאתִי לְפָנֶיךָ בְּפֶה נִרְעָשׁ וְנִפְעָם, פֶּה שֶׁהָיָה שָׁקוּעַ בְּמֵצַר מִצְרַיִם, פִּי-רָע, וְעַתָּה נִגְאַל לְמֶרְחַב הַהוֹדָיָה, פֶּה-סַח, לִהְיוֹת כְּלִי בְּיָדְךָ לְהַגָּדַת תְּהִלּוֹתֶיךָ.

אָנָּא הַרְחַב פִּי וּמַלְּאֵהוּ אוֹתִיּוֹת קְדוֹשׁוֹת, צֵרוּפֵי אֱמוּנָה שֶׁיַּעֲלוּ כְּרֵיחַ נִיחוֹחַ לְפָנֶיךָ. קַדֵּשׁ אֶת שְׂפָתַי מִכָּל דִּבּוּר פָּגוּם וְנָמוּךְ, כִּי אֵין תְּפִלָּה קוֹדֶמֶת לַהוֹדָיָה.

אַתָּה יָדַעְתָּ אֶת מַאֲבַק הַפֶּה שֶׁלִּי, הַנִּכְבָּל בְּיַד הַשְׁתִּיקָה הַקְּשָׁה, מְבַקֵּשׁ לְהִשְׁתַּחֲרֵר אֶל מֶרְחַב הָאֵין סוֹף, מְבַקֵּשׁ לְהִבָּקַע אֶל מִדְבְּרֵי הַשֶּׁבַח.

זְכֹר כִּי בְּפִי כָּל אָדָם הִנַּחְתָּ נִשְׁמַת עוֹלָם, הַלַּהַב הַדַּק שֶׁל צֶלֶם אֱלֹהִים דּוֹלֵק בְּמִלּוֹתָיו. בְּלֵיל שִׁמֻּרִים זֶה, הַשְׂבִּיעֵנִי מִסּוֹדוֹת הַפֶּסַח, שֶׁאוֹמַר כְּאָדָם וְלֹא כְּכֶלֶב, שֶׁאוֹדֶה בְּפֶה וְלֹא רַק בְּלֵב.

שַׁעַר שֵׁנִי: דִּבּוּר הַמְּשַׁחְרֵר

קְדוֹשׁ יִשְׂרָאֵל וְגוֹאֲלוֹ, אֲדוֹן הַמִּלִּים וְהַדְּמָמָה

אָנָּא, הָאֵר לִי בְּסוֹד הַגָּאוֹן הַגָּדוֹל: הַמִּתְפַּלֵּל רַק בַּחֲדְרֵי לִבּוֹ וְלֹא בְּשְׂפָתָיו, כְּמוֹהוּ כְּמַבְעִיר אֵשׁ בְּלִי אֲוִיר, כְּמוֹהוּ כְּשׁוֹתֵל זַרְעִים בְּלִי אֲדָמָה.

גַּלֵּה לִי אֶת קְדֻשַּׁת הַמִּלָּה הַיּוֹצֵאת מִן הַפֶּה, כִּי הֲלֹא בַּמִּלָּה בָּרָאתָ עוֹלָמוֹת וְכוֹכָבִים, בַּמִּלָּה קָרַעְתָּ יָם וְהֶאֱפַלְתָּ אוֹר שֶׁמֶשׁ, וְשִׂפְתוֹתַי לָמָּה תִּדֹּמְנָה?

אָנָּא, עֲשֵׂה אֶת שְׂפָתַי כְּהֵיכָל מִקְדָּשְׁךָ, אֶת מִלּוֹתַי כְּלֶחֶם הַפָּנִים עַל שֻׁלְחָן מְרֻבָּע, אֶת נְשִׁימוֹתַי כְּקְטֹרֶת סַמִּים עוֹלָה מִן הַמַּחְתָּה, אוֹתִיּוֹת נְשָׁמָתִי יְאִירוּ כְּנֵרוֹת הַמְּנוֹרָה.

יֵדַע לְשׁוֹנִי סוֹד הָאָדָם מוּל הַכֶּלֶב: הַמְּדַבֵּר מוּל הַנּוֹבֵחַ, הַמְּשַׁבֵּחַ מוּל הַלֵּב. יִגַּע פִּי בְּמַעֲלַת מְקוֹר חַיִּים, יֵדַע לָרִיב אֶת רִיב הַפֶּה קָדֳשֶׁךָ.

שַׁעַר שְׁלִישִׁי: הַפֶּה כְּנָהָר הַמִּתְחַדֵּשׁ

נוֹתֵן הַתּוֹרָה, פּוֹתֵחַ שְׁעָרִים עֶלְיוֹנִים

בְּיוֹם פְּסָח זֶה, עָשֵׂה בִּי שְׁתֵּי פְּעֻלּוֹת עֶלְיוֹנוֹת: גַּלֵּה בִּי כֹּחַ הַהוֹדָיָה עַל כָּל מִקְרֵי הֶעָבָר, וְהַצְמַח בִּי שֶׁאִיפַת הַנֶּפֶשׁ לַהֲכָנַת הֶעָתִיד. שְׁנֵיהֶם חֲשׁוּבִים בְּעֵינֶיךָ כִּשְׁנֵי כַּנְפֵי הַסְּפִירָה.

יָדַעְתִּי אֲדוֹנִי אֶת שְׁנֵי צִדְדֵי הַמַּטְבֵּעַ: צַד אֶחָד סוֹפֵר "טז ימים לָעֹמֶר" – חוֹתֵם שְׁמְךָ הַיָּהּ, צַד שֵׁנִי מְנַקֵּב "לד ימים עוֹד לְקַבָּלַת הַתּוֹרָה". שְׁנֵיהֶם אֱמֶת, שְׁנֵיהֶם חַיִּים.

עֲשֵׂה אוֹתִי כְּמַקְרִיב הָעֹמֶר בַּיּוֹם הַזֶּה: יָדִי מְנִיפָה אֶת רֵאשִׁית הַקָּצִיר לִפְנֵי מִזְבַּחֲךָ, פִּי מוֹדֶה עַל שְׂעוֹרִים – מַאֲכַל בְּהֵמָה, וְעַיִן נַפְשִׁי צוֹפָה אֶל חִטֵּי שָׁבוּעוֹת.

יְלַמְּדֵנִי הָעֹמֶר עֲבוֹדַת הַנֶּפֶשׁ הַמְּתֻקֶּנֶת: לְהוֹדוֹת עַל הַמְּעַט, לִשְׂמֹחַ בַּקַּיָּם, וְעִם זֹאת לְהַרְחִיב פֶּה בִּתְפִלָּה עַל הֶעָתִיד, שֶׁכֵּן "וּמְמַּכּוֹתַי אֶרְפָּאֵךְ" – מֵהַשְּׂעוֹרִים נַגִּיעַ לַחִטִּים.

שַׁעַר רְבִיעִי: סְעֻדַּת אֶסְתֵּר וְהַמָּן הַתָּלוּי

אָדוֹן הַנְּקָמוֹת, מַפִּיל גֵּאִים וּמַגְבִּיהַּ שְׁפָלִים

אֶל נְקֻדַּת פֶּלֶא אֲנִי צוֹפֶה הַיּוֹם: בְּיוֹם הָעֹמֶר נִתְלָה הַמָּן הָרָשָׁע. לֹא בְּמִקְרֶה נִקְשַׁר אוֹתוֹ צוֹרֵר הַיְּהוּדִים עִם רֵאשִׁית הַקָּצִיר וַהֲבָאַת הַבִּכּוּרִים.

אַתָּה גִּלִּיתָ לָנוּ סוֹד הַמָּן – הַמוֹן הוּא, הָרוֹצֶה עוֹד וְעוֹד, תָּמִיד לֹא שָׂבֵעַ, "וְכָל זֶה אֵינֶנּוּ שֹׁוֶה לִי" – זוֹעֵק בְּכַפְיוּת טוֹבָה, עֵינָיו רַק בְּמַה שֶּׁאֵין לוֹ, לֹא בַּקַּיָּם.

וַאֲנִי מוֹדֶה לְפָנֶיךָ עַל תְּרוּפַת הַנֶּפֶשׁ: בְּיוֹם שֶׁמַּקְרִיבִים הָעֹמֶר, מוֹדִים עַל הַטִּפָּה הַקְּטַנָּה, מְשַׁבְּחִים עַל מַה שֶׁהִבְשִׁיל בַּשָּׂדֶה, נִתְלֶה הַמָּן עַל עֵץ כַּפְיוּת הַטּוֹבָה.

עָשֵׂה בִּי, בּוֹרְאִי, נֵס פּוּרִים קָדוֹשׁ: תְּלֵה בְּתוֹכִי אֶת הַקּוֹל הַבּוֹכֶה "עוֹד וְעוֹד", אֶת הָעַיִן הַצָּרָה שֶׁלֹּא רוֹאָה אֶת הַטּוֹב הַקַּיָּם, אֶת הַלֵּב הֶהָמוֹן שֶׁאֵינוֹ מִסְתַּפֵּק לְעוֹלָם.

אָנָּא, פְּתַח בִּי פֶּתַח שֶׁל הוֹדָיָה, פִּתְחֵי תּוֹדָה פְּתוּחִים לִרְוָחָה. הַנֵּר הַשָּׁבוּר וְהַקָּטָן הַזּוֹהֵר בְּתוֹכִי יַרְאֶה לִי אֶת הָעוֹלָם כֻּלּוֹ בְּאוֹר הַהַכָּרָה.

שַׁעַר חֲמִישִׁי: תְּפִלַּת הַסְּפִירָה

אֵל עֶלְיוֹן, הַמְּאִיר בְּחַלּוֹנוֹת הַלֵּב

אֲנִי עוֹמֵד בְּיוֹם סְפִירַת הָעֹמֶר, יוֹם שָׁח בְּדַקּוּת וּבְמִדָּה, "הַיּוֹם טז יָמִים לָעֹמֶר", וְאַתָּה דּוֹרֵשׁ מִמֶּנִּי לְהַזְכִּיר בְּכָל יוֹם אֶת הָעֹמֶר, כְּדֵי שֶׁלֹּא אֶשְׁכַּח – אֲפִלּוּ בְּדֶרֶךְ לְמַתַּן תּוֹרָה.

זֶה סוֹד הַחַיִּים: בְּדַרְכִּי לְהַר הַחֲמִשִּׁים, אַל אֶשְׁכַּח אֶת תַּחְתִּיּוֹת הָהָר, אֶת צַעֲדִי הָרִאשׁוֹן. גַּם כַּאֲשֶׁר אֲטַפֵּס בְּמִשְׁעוֹל הַקֹּדֶשׁ, אֶת הַמָּקוֹם מִמֶּנּוּ בָּאתִי אֶזְכֹּר וְאוֹדֶה.

בְּכָל סְפִירָה וּסְפִירָה, אֲנִי מְשַׁכְלֵל אֶת נַפְשִׁי: וְהָיָה עֶרֶב – אֲנִי מוֹדֶה עַל יְצִיאַת מִצְרַיִם, וְהָיָה בֹקֶר – אֲנִי מְבַקֵּשׁ לְהִתְקַדֵּשׁ בְּסִינַי, אֶחָד בְּאֶחָד נִגָּשִׁים, יוֹם אֶחָד.

אַתָּה יוֹדֵעַ אֶת מָצוֹר הַנֶּפֶשׁ: לִפְעָמִים מַעְיָן הַהוֹדָיָה נִסְתָּם, אֵינִי רוֹאֶה אֶת הַשְּׂעוֹרָה הַצּוֹמַחַת, רַק אֶת הַדֶּרֶךְ הָאֲרֻכָּה וְהַחִטָּה הָרְחוֹקָה.

אָנָּא, חַדֵּשׁ יָמַי כְּקֶדֶם, בְּפֶה מָלֵא הוֹדָיָה, בְּלֵב רָחָב וְעֵינַיִם פְּקוּחוֹת, לִרְאוֹת הַטִּפָּה הַקְּטַנָּה שֶׁל חֶסֶד בְּתוֹךְ יָם הַצַּעַר, וְלוֹמַר בְּשִׂמְחָה: "מִזְמוֹר לְתוֹדָה".

שַׁעַר שִׁשִּׁי: לְשׁוֹן הַטְּבוּלָה בְּטַל תְּחִיָּה

מַנְחִיל שְׂפַת הַקֹּדֶשׁ לְיְצוּרֵי חֹמֶר וָדָם

הַיּוֹם אֲנִי אוֹכֵל זֵכֶר לִסְעֻדַּת אֶסְתֵּר, אוֹכֵל וְנִזְכָּר בִּפּוּר שֶׁנָּפַל בְּמֶרְחַק דּוֹרוֹת. מַה נּוֹרָא הוּא זְמַן, מַה נִפְלָא הוּא זִכָּרוֹן, שֶׁטַּעַם אֶסְתֵּר עוֹדֶנּוּ חַי עַל שְׂפָתַי.

אָנָּא, תֵּן לִפָתַי לִהְיוֹת כִּשְׂפָתֶיהָ שֶׁל אֶסְתֵּר: לִשְׁתֹּק כַּאֲשֶׁר צָרִיךְ, לְדַבֵּר כַּאֲשֶׁר הַשָּׁעָה דּוֹרֶשֶׁת, לְהַסְתִּיר אֶת הַפְּנִימִיּוּת בִּשְׁעַת הַסְתָּרָה, וּלְגַלּוֹת אֶת הַזֶּהוּת בְּשַׁעַר הַמֶּלֶךְ.

בְּהִלּוּךְ שְׂפָתַי, כְּמוֹ בִּמְגִלָּה מְדֻיֶּקֶת, יִכָּתֵב סִפּוּר הַהַצָּלָה שֶׁלִּי. בְּכָל דִּבּוּר, בְּכָל מִלָּה, בְּכָל אוֹת, אֲנִי עוֹלֶה מִבַּיִת רִאשׁוֹן אֶל בַּיִת שֵׁנִי.

מִי יִתֵּן וְשִׂפְתוֹתַי תִּהְיֶינָה דְּלָתוֹת הַנְּבוּאָה, אָזְנַי תִּהְיֶינָה חַלּוֹנוֹת הַשְּׁכִינָה. לָמָּה נִקְרָא שְׁמָהּ "דֶּלֶת"? כִּי הִיא דַּלָּה וְחֲלוּלָה, פָּנוּי הַמָּקוֹם לְהַכְנִיס דְּבַר מֶלֶךְ הָעוֹלָם.

אָנָּא, פְּתַח אֶת דַּלְתוֹת פִּי הַחֲתוּמוֹת, אֶת שְׁעָרֵי לְשׁוֹנִי הַנְּעוּלִים. זַכֵּנִי לַעֲלוֹת מִדַּרְגָּה לְדַרְגָּה בַּסֻּלָּם, מִן הַדָּל"ת אֶל הַה"א, מִן הַשְּׂעוֹרִים אֶל הַחִטִּים.

חֲתִימָה: שִׁבְעַת נֵרוֹת הַהוֹדָיָה

אֲדוֹן עוֹלָם אֲשֶׁר מָלַךְ, פּוֹתֵחַ שְׁעָרֵי אוֹר

אֲנִי עוֹמֵד לְפָנֶיךָ כְּמַקְדִּישׁ אֶת הַפֶּה, כְּמַקְדִּישׁ אֶת הַמִּלָּה, כְּמַקְדִּישׁ אֶת הַנְּשִׁימָה. הָמֵר נָא אֶת הַ"פַּרְעֹה" שֶׁבְּתוֹכִי בְּ"פֶּה-רוֹאֶה", אֶת הַ"פִּי-רָע" הָפֹךְ בְּרַחֲמֶיךָ לְ"פֶּה-סַח".

בְּכָל עֵת שֶׁאֲנִי סוֹפֵר אֶת הָעֹמֶר, אֶזְכֹּר לְהוֹדוֹת עַל כָּל קוֹמָה שֶׁהִתְקַדַּמְתִּי, וְגַם לִשְׁאֹף לְהִתְעַלּוֹת עוֹד וָעוֹד, כִּי לֹא יוּכַל הָאָדָם לַעֲמֹד בְּמָקוֹם אֶחָד.

וּבְשַׁחֲרִית הַחַג, כַּאֲשֶׁר אוֹמַר אוֹ לֹא אוֹמַר אֶת מִזְמוֹר הַתּוֹדָה, תִּהְיֶה כָּל נְשִׁימָתִי הוֹדָיָה, כָּל תְּנוּעָתִי תְּהִלָּה, כָּל פְּעִימַת לֵב שֶׁבַח, לְךָ, בּוֹרֵא הַפֶּה וְנוֹתֵן הַסְּפִירָה.

יְהִי רָצוֹן מִלְּפָנֶיךָ, פּוֹקֵחַ עֵינַיִם חֲסוּמוֹת, שֶׁאֶזְכֶּה לְתָלוֹת אֶת הַמָּן שֶׁבִּי עַל עֵץ הַהוֹדָיָה, שֶׁאַכְנִיעַ וַאֲמַגֵּר אֶת הַהָמוֹן הַשּׁוֹאֵג "עוֹד וְעוֹד", וּמִזְמוֹר חָדָשׁ יִכָּתֵב עַל לוּחַ לִבִּי.

וְכַאֲשֶׁר אֲנִי אוֹכֵל זֵכֶר לְסְעֻדַּת אֶסְתֵּר, אֶשְׁמַע בַּמַּאֲכָל קוֹלוֹת הַהוֹדָיָה שֶׁל דּוֹרוֹת; חֶסֶד אָחוּז בְּחֶסֶד, פֶּלֶא אָחוּז בְּפֶלֶא, וְהַפֶּה פּוֹתֵחַ שַׁעַר גְּאֻלָּה.

אָמֵן, כֵּן יְהִי רָצוֹן.

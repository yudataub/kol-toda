🏠 מָבוֹא: מִצְוַת חֲנֻכָּה – חוֹבַת הָאָדָם בְּתוֹךְ בֵּיתוֹ

מִצְווֹת רַבּוֹת קְשׁוּרוֹת לַמֶּרְחָב הַפִּיזִי שֶׁל הַבַּיִת, אַךְ אוֹפְיָן אֵינוֹ זֶהֶה. מִצְווֹת מְסוּיָּמוֹת, כְּמוֹ מְזוּזָה אוֹ מַעֲקֶה, הֵן "חוֹבַת הַבַּיִת" – הַחִיּוּב חָל עַל הַמִּבְנֶה עַצְמוֹ, וּמִי שֶׁאֵין לוֹ בַּיִת, פָּטוּר מֵהֶן. מִצְווֹת אֲחֵרוֹת, כְּמוֹ אֲכִילַת מַצָּה בְּפֶסַח אוֹ יְשִׁיבָה בְּסֻכָּה, הֵן "חוֹבַת הָאָדָם" – חִיּוּב אִישִׁי הַמּוּטָל עַל כָּל יְהוּדִי, לְלֹא קֶשֶׁר לִמְקוֹם מְגוּרָיו. אִם כֵּן, מַהוּ אוֹפְיָהּ שֶׁל מִצְוַת הַדְלָקַת נֵר חֲנֻכָּה?

הַשְּׁאֵלָה מִתְחַדֶּדֶת בְּמִקְרֶה קִיצוֹן שֶׁל אָדָם שֶׁאֵין לוֹ בַּיִת, הַמִּתְגּוֹרֵר בָּרְחוֹב. מִכֵּיוָן שֶׁמְּקוֹם הַהַדְלָקָה הָאִידֵאָלִי הוּא "פֶּתַח הַבַּיִת", הַאִם הוּא פָּטוּר מֵהַמִּצְוָה? אִם הַחִיּוּב הוּא עַל הַבַּיִת, הַתְּשׁוּבָה הִיא כֵּן. אַךְ אִם הַחִיּוּב הוּא עַל הָאָדָם, הֲרֵי שֶׁעָלָיו לִמְצֹא פִּתָּרוֹן.

פְּסִיקָתוֹ שֶׁל הָרַב אֱלִישִׁיב זַצַ"ל:

הַחִיּוּב לְהַדְלִיק נֵר חֲנֻכָּה הוּא חוֹבַת הָאָדָם ('חוֹבַת גַּבְרָא'). הַחוֹבָה הִיא כָּה גְדוֹלָה, עַד כְּדֵי כָּךְ שֶׁאָדָם חֲסַר בַּיִת חַיָּב לָצֵאת וּלְחַזֵּר עַל הַפְּתָחִים כְּדֵי לֶאֱסֹף כֶּסֶף וְלִשְׂכֹּר דִּירָה לִימֵי הַחֲנֻכָּה, וּבִלְבַד שֶׁיְּקַיֵּם אֶת הַמִּצְוָה כַּהֲלָכָתָהּ.

כָּאן מִתְגַּלָּה הַיְחִידִיּוּת הַמּוּפְלָאָה שֶׁל מִצְוַת נֵר חֲנֻכָּה: זוֹהִי חוֹבָה אִישִׁית, אַךְ מְקוֹם קִיּוּמָהּ הוּא דַוְקָא וּבְאֹפֶן מֻבְהָק בְּתוֹךְ הַבַּיִת. הַפָּרָדוֹקְס הַיְּחִידִי הַזֶּה – חוֹבָה אִישִׁית הַמִּתְקַיֶּמֶת דַוְקָא בַּבַּיִת – מַצְבִּיעַ עַל תַּכְלִית עֲמֻקָּה יוֹתֵר. כְּדֵי לְחַשֹּׂף אוֹתָהּ, עָלֵינוּ לְפַעֲנֵחַ אֶת סוֹדוֹ שֶׁל שֵׁם הֶחָג עַצְמוֹ: חֲנֻכָּה.

📖 סוֹדוֹ שֶׁל הַשֵּׁם "חֲנֻכָּה": שָׁלֹשׁ קוֹמוֹת שֶׁל מַשְׁמָעוּת

שְׁמוֹ שֶׁל חָג אֵינוֹ רַק תַּוִּית הִיסְטוֹרִית; הוּא מֵכִיל בְּתוֹכוֹ אֶת הַמַּהוּת הַפְּנִימִית וְהָרַעְיוֹנוֹת הָעֲמֻקִּים הַגְּלוּמִים בּוֹ. נִתּוּחַ הַשֵּׁם "חֲנֻכָּה" חוֹשֵׂף שָׁלֹשׁ רָמוֹת שֶׁל הֲבָנָה, שֶׁכָּל אַחַת מֵהֶן מַעֲמִיקָה אֶת קוֹדְמָתָהּ וּמְגַלָּה טֶפַח נוֹסָף בְּסוֹדוֹ שֶׁל הֶחָג.

1️⃣ הַפֵּרוּשׁ הַפָּשׁוּט: חָנוּ בְּכ"ה

הַפֵּרוּשׁ הַמֻּכָּר וְהַנָּפוֹץ בְּיוֹתֵר לְשֵׁם הֶחָג הוּא רָאשֵׁי תֵבוֹת וָרֶמֶז הִיסְטוֹרִי: הַמַּכַּבִּים חָנוּ מִמִּלְחַמְתָּם בְּיוֹם כ"ה בְּכִסְלֵו. לְאַחַר נִצְחוֹנָם עַל הַיְּוָנִים, הֵם הִגִּיעוּ לִמְנוּחָה וּלְנַחֲלָה בְּתַאֲרִיךְ זֶה.

שְׁאֵלוֹת שֶׁמַּעֲלֶה הַחִידָ"א (רַבִּי חַיִּים יוֹסֵף דָּוִד אַזוּלַאי):

• מַדּוּעַ הֶחָג נִקְרָא עַל שֵׁם הַחֲנִיָּה וְהַמְּנוּחָה, וְלֹא עַל שֵׁם הַנִּצָּחוֹן הָאַדִּיר וְהַנֵּס הַגָּדוֹל?
• מַדּוּעַ נִבְחֲרָה הַלָּשׁוֹן "חָנוּ", וְלֹא "נָחוּ", כְּפִי שֶׁנֶּאֱמַר עַל חַג הַפּוּרִים "כַּיָּמִים אֲשֶׁר נָחוּ בָהֶם הַיְּהוּדִים מֵאֹיְבֵיהֶם"?

שְׁאֵלוֹת אֵלוּ מוֹבִילוֹת אוֹתָנוּ אֶל הָרוֹבֶד הַבָּא שֶׁל הַמַּשְׁמָעוּת.

2️⃣ עֹמֶק נוֹסָף: חֲנֻכַּת הַמִּקְדָּשׁ

פֵּרוּשׁ עָמֹק יוֹתֵר, הַמּוּבָא עַל יְדֵי הַחִידָ"א, הַמַּהַר"ל וְהָרַמְבַּ"ן, גּוֹרֵס כִּי מְקוֹר הַשֵּׁם אֵינוֹ בִּפְעֻלַּת הַחֲנִיָּה, אֶלָּא בַּחֲנֻכַּת הַמִּקְדָּשׁ. כַּאֲשֶׁר נִכְנְסוּ הַחַשְׁמוֹנָאִים לְהֵיכָל, הֵם מָצְאוּ אוֹתוֹ טָמֵא וּמְחֻלָּל. הַפְּעֻלָּה הָרִאשׁוֹנָה שֶׁעָשׂוּ כְּדֵי לְהָשִׁיב אֶת הַקְּדֻשָּׁה לִמְקוֹמָהּ הָיְתָה הַדְלָקַת הַמְּנוֹרָה הַטְּהוֹרָה בְּשֶׁמֶן טָהוֹר.

הַדְלָקַת הַנֵּרוֹת הִיא פְּעֻלַּת הַחֲנֻכָּה עַצְמָהּ, הַמְקַדֶּשֶׁת מֵחָדָשׁ אֶת הַמֶּרְחָב וּמַשִּׁיבָה אֶת הַשְּׁכִינָה לְמִשְׁכָּנָהּ.

לְפִי פֵּרוּשׁ זֶה, שֵׁם הֶחָג מְבַטֵּא אֶת מַהוּת הַיּוֹם – לֹא רַק זִכָּרוֹן הַנִּצָּחוֹן, אֶלָּא חֲגִיגַת חֲנֻכַּת הַבַּיִת, הַשָּׁבַת הַקְּדֻשָּׁה לִמְקוֹמָהּ בְּאֶמְצָעוּת הָאוֹר.

3️⃣ סוֹד הַסּוֹדוֹת: חָנָה-כ"ה, הַשְּׁכִינָה חוֹנָה

הָרוֹבֶד הַפְּנִימִי וְהַמִּיסְטִי בְּיוֹתֵר קוֹשֵׁר אֶת שֵׁם הֶחָג לְהַשְׁרָאַת הַשְּׁכִינָה עַצְמָהּ. בְּסִפְרֵי הַקַּבָּלָה וְהַחֲסִידוּת, הַשְּׁכִינָה – הַגִּלּוּי הָאֱלֹהִי בָּעוֹלָם הַזֶּה – מְכֻנָּה בָּאוֹתִיּוֹת כ"ה. רֶמֶז לְכָךְ נִמְצָא בִּפְסוּקִים כְּמוֹ "כֹּה תְּבָרְכוּ אֶת בְּנֵי יִשְׂרָאֵל" אוֹ "כֹּה אָמַר ה'".

כִּי לְעוֹלָם חַסְדּוֹ
הוֹדוּ לַה' כִּי טוֹב כִּי לְעוֹלָם חַסְדּוֹ
כֹּה תְבָרְכוּ אֶת בְּנֵי יִשְׂרָאֵל אָמוֹר לָהֶם

מַדּוּעַ דַּוְקָא כ"ה? הָאוֹת כ' הִיא "כַּף הַדִּמְיוֹן", הַמְסַמֶּלֶת גִּלּוּי חֶלְקִי, אוֹר הַמֻּשָּׂג דֶּרֶךְ מָשָׁל וְדִמְיוֹן, בְּדוֹמֶה לְאוֹר הַלְּבָנָה. זֹאת בְּנִגּוּד לִנְבוּאָתוֹ הַיְּשִׁירָה שֶׁל מֹשֶׁה רַבֵּנוּ, שֶׁהָיְתָה בָּאַסְפַּקְלַרְיָה הַמְּאִירָה, בִּבְחִינַת "זֶה הַדָּבָר אֲשֶׁר דִּבֶּר ה'". רֹב הַנְּבִיאִים חָווּ אֶת הַגִּלּוּי הָאֱלֹהִי בִּבְחִינַת "כֹּה אָמַר", כְּלוֹמַר, דֶּרֶךְ הַשְּׁכִינָה.

הַחִבּוּר הַמּוּפְלָא:

כַּאֲשֶׁר הַיְּוָנִים טִמְּאוּ אֶת הַמִּקְדָּשׁ, הַשְּׁכִינָה הִסְתַּלְּקָה. בְּיוֹם כ"ה בְּכִסְלֵו, עִם טִהוּר הַמִּקְדָּשׁ וְהַדְלָקַת הַמְּנוֹרָה, הַשְּׁכִינָה (הַמְּכֻנָּה כ"ה) חָנְתָה וְשָׁבָה לִשְׁכֹּן בִּמְקוֹמָהּ.

מִכָּאן הַשֵּׁם: חָנוּ-כ"ה!

עִקָּרוֹן זֶה פּוֹעֵל גַּם כַּיּוֹם, עַל פִּי הַכְּלָל שֶׁל "בַּיָּמִים הָהֵם, בַּזְּמַן הַזֶּה". כַּאֲשֶׁר אָנוּ מַדְלִיקִים נֵרוֹת חֲנֻכָּה בְּבֵיתֵנוּ, אָנוּ יוֹצְרִים מֵחָדָשׁ אֶת אוֹתוֹ הַתַּהֲלִיךְ הָרוּחָנִי: אָנוּ מַזְמִינִים אֶת הַשְּׁכִינָה, הַנִּקְרֵאת כ"ה, לַחֲנוֹת בְּתוֹךְ בֵּיתֵנוּ.

רַעְיוֹן זֶה בָּא לִידֵי בִּטּוּי בַּמִּנְהָג הַנָּהוּג בִּמְקוֹמוֹת רַבִּים, לִפְתֹּחַ אֶת קְרִיאַת הַתּוֹרָה בְּיוֹם הָרִאשׁוֹן שֶׁל חֲנֻכָּה בְּפָרָשַׁת "כֹּה תְבָרְכוּ אֶת בְּנֵי יִשְׂרָאֵל", כְּרֶמֶז לְכָךְ שֶׁהַשְּׁכִינָה, הַנִּקְרֵאת "כֹּה", חוֹנָה בְּקִרְבֵּנוּ.

🔑 פֶּתַח הַבַּיִת: שַׁעַר לַשְּׁכִינָה

מִקּוּם הַדְלָקַת הַנֵּרוֹת בְּפֶתַח הַבַּיִת אֵינוֹ פְּרָט טֶכְנִי, אֶלָּא מַהוּתִי וְעָמֹק. הַפֶּתַח הוּא הַמַּעֲבָר בֵּין הַפְּרָטִי לַצִּבּוּרִי, בֵּין הַפְּנִים לַחוּץ, וְהוּא הַמָּקוֹם שֶׁדַּרְכּוֹ נִכְנֶסֶת הַבְּרָכָה – וְהַשְּׁכִינָה – אֶל הַבַּיִת.

אֵין זוֹ הַפַּעַם הָרִאשׁוֹנָה בְּהִיסְטוֹרְיָה הַיְּהוּדִית שֶׁפֶּתַח הַבַּיִת מְשַׁמֵּשׁ מוֹקֵד לִפְעֻלָּה רוּחָנִית מְכוֹנֶנֶת. בִּיצִיאַת מִצְרַיִם, צִוּוּ בְּנֵי יִשְׂרָאֵל לִמְרֹחַ אֶת דַּם קָרְבָּן הַפֶּסַח עַל הַמַּשְׁקוֹף וְעַל שְׁתֵּי הַמְּזוּזוֹת, וּפְעֻלָּה זוֹ הִבְדִּילָה בֵּין בָּתֵּי יִשְׂרָאֵל לְבָתֵּי הַמִּצְרִים וְהֵבִיאָה עֲלֵיהֶם אֶת הַגְּאֻלָּה.

פֵּרוּשׁ ה'כְּלִי יָקָר' (רַבִּי שְׁלֹמֹה אֶפְרַיִם מִלוּנְצִ'יץ):

הַקָּבָּ"ה נִקְרָא 'מַשְׁקִיף'... הַמַּשְׁקוֹף הוּא צוּרַת הָאוֹת ח'. הֵיכָן שׁוֹרֶה הַקָּבָּ"ה? הֵיכָן שֶׁיֵּשׁ זוּג שֶׁמְּאֻחָדִים – שְׁתֵּי אוֹתִיּוֹת ז' (הַמְּזוּזוֹת) וְהַקָּבָּ"ה, הָאֶחָד (הַמַּשְׁקוֹף), רוֹכֵב וְשׁוֹרֶה עֲלֵיהֶם.

ה'כְּלִי יָקָר' מַסְבִּיר אֶת הָרַעְיוֹן הַמּוּפְלָא הַבָּא:

  • הַמַּשְׁקוֹף (הַקּוֹרָה הָאוֹפְקִית) וּשְׁתֵּי הַמְּזוּזוֹת (הַקּוֹרוֹת הָאֲנָכִיּוֹת) יוֹצְרִים יַחַד אֶת צוּרָתָהּ שֶׁל הָאוֹת ח'.
  • הָאוֹת ח' בְּגִימַטְרִיָּה הִיא 8, כְּמִסְפַּר יְמֵי הַחֲנֻכָּה.
  • שְׁתֵּי הַמְּזוּזוֹת, הַדּוֹמוֹת לְאוֹת ז', מְסַמְּלוֹת אֶת הַזּוּג הַחַי בַּבַּיִת, שְׁנֵי אֲנָשִׁים הַמִּתְאַחֲדִים וְיוֹצְרִים אֶת הַבָּסִיס לִקְדֻשָּׁה.
  • הַמִּסְפָּר 8 מְסַמֵּל אֶת הַמְּצִיאוּת שֶׁמֵּעַל הַטֶּבַע (הַמְסֻמָּל בַּמִּסְפָּר 7). הוּא מְבַטֵּא אֶת רַעְיוֹן "אֶחָד רוֹכֵב עַל שֶׁבַע" – הַקָּבָּ"ה (הָאֶחָד), מַשְׁרֶה אֶת קְדֻשָּׁתוֹ עַל הָעוֹלָם הַטִּבְעִי וְעַל הָאַחְדוּת בַּבַּיִת.

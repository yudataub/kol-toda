תפילת המלמד הצדיק - מעמקי הנשמה

פְּתִיחָה בְּהִתְבּוֹדְדוּת

רִבּוֹנוֹ שֶׁל עוֹלָם, מְקוֹר כָּל הַנְּשָׁמוֹת,

אָנֹכִי עֶבְדְּךָ הַנִּכְנַע, בָּאתִי לִפְנֵי כִּסֵּא רַחֲמֶיךָ כְּחֶרֶס נִשְׁבָּר הַמְבַקֵּשׁ תִּקּוּן. הִנְנִי יוֹדֵעַ כִּי אֵין בִּי כְּלוּם, וְכָל מַה שֶּׁאֲנִי חוֹשֵׁב שֶׁיֵּשׁ בִּי - הַכֹּל הֶבֶל וְכָל רוּחַ רָעָה. אֲבָל בְּרַחֲמֶיךָ הָרַבִּים, בָּחַרְתָּ בִּי לִהְיוֹת שָׁלִיחַ בְּמַעֲשֵׂה בְרֵאשִׁית הַנִּמְשָׁךְ - לִתֵּן צוּרָה לַנְּשָׁמוֹת, לְעוֹרֵר יְצִירוֹת יָדֶיךָ מֵחֶלְמָתָן.

תְּפִלָּה לְהַרְחָקַת הַנָּפֶשׁ הַבַּהֲמִית

אֱלֹהַי, הַצִּילֵנִי מִיִּצְרִי הָרַע הַמִּתְחַבֵּא בְּתוֹךְ מַעֲשֵׂי הַקֹּדֶשׁ. הַרְחֵק מִמֶּנִּי גַּאֲוָתָן וְכַעֲסָן, אַף וְחֵמָה וְכָל מִינֵי רוּחַ רָעָה. כִּי יָדַעְתִּי כִּי אִם אֶהְיֶה כְּלִי מְטֻנָּף, אֵיךְ יוּכַל לַעֲבֹר דְּרָכִי אוֹר טָהוֹר? וְאִם לִבִּי יִהְיֶה סוּגַר, אֵיךְ יִפְתְּחוּ לְבָבוֹת בָּנַי וּבְנוֹתַי?

זַכֵּנִי לְהִתְבַּטֵּל בֶּאֱמֶת לְפָנֶיךָ, עַד שֶׁיִּהְיֶה כָּל מַעֲשַׂי כְּמוֹ צִנּוֹר חָלוּל הַמַּעֲבִיר מַיִם חַיִּים מִמְּקוֹרָם הָעֶלְיוֹן אֶל הַצְּמֵאִים לְמַטָּה.

בַּקָּשָׁה לְגִלּוּי אוֹר פְּנִימִי

רִבּוֹנוֹ שֶׁל עוֹלָם, גַּל אֶת עֵינַי וְאַבִּיטָה נִפְלָאוֹת מִתּוֹרָתֶךָ - לֹא רַק בִּכְתוּבֶיךָ הַקְּדוֹשִׁים, אֶלָּא גַּם בְּנִשְׁמוֹת קְדוֹשׁוֹת אֵלֶּה שֶׁהוֹשַׁבְתָּ לְפָנַי.

זַכֵּנִי לְהַכִּיר אֶת הָאוֹר הַגָּנוּז בְּכָל נֶפֶשׁ וְנֶפֶשׁ - גַּם בְּזוֹ שֶׁנִּרְאֵית כְּמוֹ חֲשֵׁכָה עַזָּה, גַּם בְּזוֹ שֶׁמִּתְעַטֶּפֶת בִּקְלִפּוֹת קָשׁוֹת. הוֹרֵנִי לִרְאוֹת שֶׁבְּכָל יֶלֶד יֵשׁ נִיצוֹץ אֱלֹקִי הַמְחַכֶּה לְהַתְלַהֲבוּת, וְכָל נְשָׁמָה הִיא עוֹלָם מָלֵא וּבְרִיאָה שְׁלֵמָה בִּפְנֵי עַצְמָהּ.

תְּפִלָּה לְכֹחַ עֶלְיוֹן בִּשְׁעַת נִסָּיוֹן

וּכְשֶׁיָּבֹאוּ עֵתוֹת הַמִּבְחָן - וְיָבֹאוּ, כִּי לֹא נִתְּנָה תּוֹרָה אֶלָּא לְמַלְאֲכֵי הַשָּׁרֵת, וַאֲנַחְנוּ בְּנֵי אָדָם חַסְרֵי מַעְשֶׂה - תֵּן לִי לְהַאֲמִין שֶׁגַּם מִתּוֹךְ הַכִּשָּׁלוֹן יְכוֹלִים לִפְרֹחַ נִצָּנִים חֲדָשִׁים.

כְּשֶׁאֶרְאֶה תַּלְמִיד נוֹפֵל וְלֹא אוּכַל לְהָקִימוֹ, הַזְכִּירֵנִי כִּי גַּם אַבְרָהָם אָבִינוּ לֹא הִצְלִיחַ עִם כֻּלָּם. כְּשֶׁאֶרְאֶה מַעֲשַׂי הוֹלְכִים לְאִבּוּד, חַזְּקֵנִי בֶּאֱמוּנָה כִּי אֵין דָּבָר הַהוֹלֵךְ לְאִבּוּד לְעוֹלָמִים - כִּי אִם נִטַּע בְּאֱמֶת וּבְאַהֲבָה, יִצְמַח בִּזְמַנּוֹ, אֲפִלּוּ אַחַר שָׁנִים רַבּוֹת.

בַּקָּשָׁה לְקִדּוּשׁ הַזְּמַן וְהַמָּקוֹם

יְהִי רָצוֹן שֶׁבֵּית הַמִּדְרָשׁ שֶׁלִּי יִהְיֶה מִקְדָּשׁ מְעַט - מָקוֹם שֶׁהַשְּׁכִינָה שׁוֹרָה בוֹ, שֶׁכָּל נֶפֶשׁ שֶׁתִּכָּנֵס לְשָׁם תַּרְגִּישׁ כִּי הִיא בִּמְקוֹם קָדוֹשׁ. זַכֵּנִי שֶׁכָּל מִלָּה שֶׁתֵּצֵא מִפִּי תִּהְיֶה עַטוּרָה בִּכְוָנוֹת קְדוֹשׁוֹת, וְכָל שְׁתִיקָה שֶׁלִּי תִּהְיֶה מְלֵאָה בְּאַהֲבָה.

תְּפִלָּה לְחִבּוּר הַדּוֹרוֹת

רִבּוֹנוֹ שֶׁל עוֹלָם, זַכֵּנִי לִהְיוֹת חוּלְיָה נֶאֱמֶנֶת בְּשַׁלְשֶׁלֶת הַמָּסֹרֶת הַקְּדוֹשָׁה. מַה שֶּׁקִּבַּלְתִּי מֵרַבּוֹתַי בְּאַהֲבָה, אֶמְסֹר לְתַלְמִידַי בְּאַהֲבָה כְּפוּלָה. וּמַה שֶּׁלֹּא זָכִיתִי לְהָבִין, יַבִּינוּ הֵם וִיַשְׁלִימוּ מַה שֶּׁחָסַר בִּי.

הַרְאֵנִי כִּי אֲנִי רַק שָׁלִיחַ בְּמַעֲשֶׂה אֲלָפִים שָׁנָה, וְעָלַי לְהַעֲבִיר הָאֵשׁ הַקְּדוֹשָׁה מִדּוֹר לְדוֹר כְּמוֹ שֶׁהֶעֱבִירוּ לִי רַבּוֹתַי הַקְּדוֹשִׁים.

חֲתִימָה בְּהַכְנָעָה וְתִקְוָה

אֱלֹהַי, אִם בָּחַרְתָּ בְּכֶלִי שָׁבוּר כָּמוֹנִי לַעֲבוֹדָתְךָ הַקְּדוֹשָׁה, הֲרֵי זֶה אוֹת שֶׁאַתָּה רוֹצֶה דַּוְקָא בְּכֵלִים שְׁבוּרִים - כִּי מֵהֶם יוֹצֵא הָאוֹר הַכִּי עָמֹק וְהַכִּי אֲמִתִּי.

הֱיֵה נָא עִמִּי בְּכָל יוֹם וְיוֹם, בְּכָל תַּלְמִיד וְתַלְמִיד. וּזְכוּת רַבּוֹתַי הַקְּדוֹשִׁים תָּגֵן עָלַי מִלְמַעְלָה, וּזְכוּת תַּלְמִידַי הַטְּהוֹרִים תִּזְכֶּה אוֹתִי מִלְּמַטָּה.

יְהִי נֹעַם יְיָ אֱלֹהֵינוּ עָלֵינוּ, וּמַעֲשֵׂה יָדֵינוּ כּוֹנְנָה עָלֵינוּ - וּמַעֲשֵׂה יָדֵינוּ כּוֹנְנֵהוּ לְעוֹלָם וָעֶד.

נֶצַח יִשְׂרָאֵל לֹא יְשַׁקֵּר.

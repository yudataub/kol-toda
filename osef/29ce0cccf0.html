🌍 מָבוֹא: הָעוֹלָם נִבְרָא בְּמַאֲמָר - כֹּחַ הַדִּבּוּר

כַּאֲשֶׁר הַקָּדוֹשׁ בָּרוּךְ הוּא בָּחַר לִבְרֹא אֶת הָעוֹלָם, הוּא לֹא הִשְׁתַּמֵּשׁ בְּכֹחַ פִיזִי. לֹא הִרְכִּיב אֲבָנִים, לֹא עִצֵּב חֹמֶר, וְלֹא יָצַר דָּבָר בְּיָדָיִם. הוּא פָּשׁוּט דִּבֵּר. "וַיֹּאמֶר אֱלֹהִים יְהִי אוֹר - וַיְהִי אוֹר". עֲשָׂרָה מַאֲמָרוֹת, עֲשָׂרָה דִבּוּרִים, וְהִנֵּה - עוֹלָם שָׁלֵם נוֹצַר מֵאַיִן לְיֵשׁ.

"בִּדְבַר ה' שָׁמַיִם נַעֲשׂוּ
וּבְרוּחַ פִּיו כָּל צְבָאָם"
(תְּהִלִּים לג, ו)

מַה הַמָּסָר שֶׁל בְּרִיאָה זוֹ? הַקָּבָּ"ה רוֹצֶה לְלַמְּדֵנוּ שֶׁהַדִּבּוּר הוּא כֹּחַ עֶלְיוֹן. הַמִּלִּים שֶׁלָּנוּ אֵינָן סְתָם רְעָשִׁים בָּאֲוִיר. כָּל מִלָּה שֶׁיּוֹצֵאת מֵהַפֶּה הִיא כֹּחַ בּוֹרֵא. מִלָּה יְכוֹלָה לִבְנוֹת אָדָם, לְחַזֵּק אוֹתוֹ, לְהַעֲלוֹת אוֹתוֹ. וּמִלָּה יְכוֹלָה - חַס וְשָׁלוֹם - לְהַרְסוֹ לַגַּמְרֵי.

הַחִלּוּק הַמְיֻחָד שֶׁל הָאָדָם:

חַז"ל מְלַמְּדִים אוֹתָנוּ: "מַה הַחִלּוּק בֵּין הָאָדָם לַבְּהֵמָה? הַבְּהֵמָה נִקְרֵאת 'נֶפֶשׁ חַיָּה', וְהָאָדָם נִקְרָא 'נֶפֶשׁ חַיָּה מְדַבֶּרֶת'". הַדִּבּוּר הוּא מַה שֶּׁעוֹשֶׂה אוֹתָנוּ לַאֲנָשִׁים, לֹא סְתָם חַיּוֹת מְדַבְּרוֹת!

אָבָל - וְזֶה הָעִקָּר - עִם כֹּחַ כָּל כָּךְ גָּדוֹל בָּאָה גַּם אַחֲרָיוּת כָּבֵדָה. אִם הָאָדָם מְשַׁתֵּף פְּעֻלָּה עִם הַקָּבָּ"ה בִּבְרִיאַת הָעוֹלָם דֶּרֶךְ הַדִּבּוּר שֶׁלּוֹ, הוּא חַיָּב לָדַעַת שֶׁכָּל מִלָּה שֶׁהוּא אוֹמֵר מַשְׁפִּיעָה עַל הַמְּצִיאוּת.

⚔️ לָשׁוֹן הָרָע - הַנֶּשֶׁק הַקָּטְלָנִי ⚔️

💀 לָשׁוֹן הָרָע - חֶרֶב פִּיפִיּוֹת הַהוֹרֶגֶת שְׁלֹשָׁה

לָשׁוֹן הָרָע הִיא מִלָּה שֶׁנֶּאֶמְרָה עַל אָדָם אַחֵר, גַּם אִם הִיא אֱמֶת לַאֲמִתָּהּ, אֲבָל הִיא פּוֹגַעַת בּוֹ אוֹ מַשְׁפִּילָה אוֹתוֹ. אֵין זֶה מַשְׁנֶה אִם סִפַּרְתָּ דָּבָר שֶׁקֶר אוֹ דָּבָר אֱמֶת - אִם זֶה גּוֹרֵם לְאָדָם נֶזֶק, זֶה לָשׁוֹן הָרָע.

אָמַר רַבִּי יַנַּאי: "לָשׁוֹן הָרָע הוֹרֶגֶת שְׁלֹשָׁה:
אֶת הָאוֹמְרוֹ, אֶת הַמְקַבְּלוֹ, וְאֶת מִי שֶׁמְּדֻבָּר עָלָיו"

מִי נֶהֱרָג מִלָּשׁוֹן הָרָע?

1️⃣ הַמְדַבֵּר - הוֹרֵג אֶת עַצְמוֹ

כַּאֲשֶׁר אַתָּה מְדַבֵּר לָשׁוֹן הָרָע, אַתָּה מַרְעִיל אֶת נַפְשְׁךָ. הַמִּלִּים הָרָעוֹת פּוֹגְעוֹת בִּנְשָׁמָה שֶׁלְּךָ, מְרַחֲקוֹת אוֹתְךָ מֵהַקָּבָּ"ה, וְהוֹרְסוֹת אֶת הַצֶּלֶם אֱלֹהִים שֶׁבְּךָ.

2️⃣ הַמְקַבֵּל - שֻׁתָּף לָרֶצַח

כַּאֲשֶׁר אַתָּה מַאֲזִין לְלָשׁוֹן הָרָע, אַתָּה לֹא סְתָם "שׁוֹמֵעַ" - אַתָּה שֻׁתָּף מָלֵא לַחֵטְא. חַז"ל אוֹמְרִים שֶׁהַמְקַבֵּל חָמוּר יוֹתֵר מֵהָאוֹמֵר, כִּי בְּלִי מַאֲזִין - אֵין לְמִי לְדַבֵּר!

3️⃣ מִי שֶׁמְּדֻבָּר עָלָיו - הַקָּרְבָּן הַיָּשִׁיר

זֶה הַנֶּזֶק הַגָּלוּי וְהַבָּרוּר בְּיוֹתֵר. לָשׁוֹן הָרָע הוֹרֶגֶת אֶת הָאָדָם בְּעֵינֵי הָאֲחֵרִים. פַּעַם אַחַת, כְּשֶׁהַמִּלָּה יוֹצֵאת - אִי אֶפְשָׁר לְהַחֲזִיר אוֹתָהּ. הַנֶּזֶק נִשְׁאָר לָעַד.

📖 סִפּוּר הַנּוֹצָה - הַחָפֵץ חַיִּים

פַּעַם אַחַת בָּא אָדָם אֶל הַחָפֵץ חַיִּים וְאָמַר: "רַבִּי, אֲנִי מִתְחָרֵט עַל לָשׁוֹן הָרָע שֶׁדִּבַּרְתִּי. מָה אוּכַל לַעֲשׂוֹת כְּדֵי לְתַקֵּן?"

הֶחָפֵץ חַיִּים אָמַר לוֹ: "קַח כָּרִית מְלֵאָה נוֹצוֹת, עֲלֵה לְגַג הַבַּיִת, וּפַזֵּר אֶת כָּל הַנּוֹצוֹת בָּרוּחַ".

הָאִישׁ עָשָׂה כֵּן, וְחָזַר. "וְעַכְשָׁיו", אָמַר הֶחָפֵץ חַיִּים, "לֵךְ וְאַסֹף אֶת כָּל הַנּוֹצוֹת".

"אֲבָל רַבִּי, זֶה בִּלְתִּי אֶפְשָׁרִי! הָרוּחַ פִּזְּרָה אוֹתָן לְכָל עֵבֶר!"

"כָּךְ גַּם לָשׁוֹן הָרָע", אָמַר הֶחָפֵץ חַיִּים בְּעֶצֶב. "מִלָּה שֶׁיּוֹצֵאת - לֹא חוֹזֶרֶת. הַנֶּזֶק כְּבָר נַעֲשָׂה, וְאִי אֶפְשָׁר לְתַקְּנוֹ לְגַמְרֵי."

⚖️ חֻמְרָה יְתֵרָה ⚖️

🔥 חֻמְרַת לָשׁוֹן הָרָע - יוֹתֵר מִשְּׁלֹשֶׁת הָעֲבֵרוֹת הַחֲמוּרוֹת

הַדָּבָר הַמְפַתֵּחַ בְּיוֹתֵר שֶׁצָּרִיךְ לָדַעַת: חַז"ל אוֹמְרִים שֶׁלָשׁוֹן הָרָע שְׁקוּלָה כְּנֶגֶד שָׁלֹשׁ עֲבֵרוֹת חֲמוּרוֹת: עֲבוֹדָה זָרָה, גִּלּוּי עֲרָיוֹת, וּשְׁפִיכוּת דָּמִים!

"כָּל הַמְסַפֵּר לָשׁוֹן הָרָע
כְּאִלּוּ כָּפַר בָּעִקָּר"
(עֲרָכִין טו, ב)

אֵיךְ זֶה יִתָּכֵן? אֵיךְ אֶפְשָׁר לְהַשְׁווֹת מִלָּה אַחַת לַעֲבֵרוֹת הַחֲמוּרוֹת בְּיוֹתֵר?

הַהֶסְבֵּר הָעָמֹק:

עֲבֵרָה חֲמוּרָה הַקֶּשֶׁר לְלָשׁוֹן הָרָע
עֲבוֹדָה זָרָה כַּאֲשֶׁר מְדַבְּרִים לָשׁוֹן הָרָע, מַפְרִידִים בֵּין אָדָם לְהַקָּבָּ"ה. זֶה כְּמוֹ כְּפִירָה בּוֹ.
גִּלּוּי עֲרָיוֹת לָשׁוֹן הָרָע "מְגַלָּה" דְּבָרִים שֶׁצְּרִיכִים לְהִשָּׁאֵר מְכֻסִּים. פּוֹגַעַת בְּצְנִיעוּת הַנֶּפֶשׁ.
שְׁפִיכוּת דָּמִים לָשׁוֹן הָרָע "שׁוֹפֶכֶת" אֶת שְׁמוֹ הַטּוֹב שֶׁל הָאָדָם, הוֹרֶגֶת אוֹתוֹ רוּחָנִית וְחֶבְרָתִית.

מַדּוּעַ לָשׁוֹן הָרָע כָּל כָּךְ חֲמוּרָה?

הָרַב דֶּסְלֶר זַצַ"ל מְסַבֵּר: עֲבֵרוֹת חֲמוּרוֹת אֲחֵרוֹת נַעֲשׂוֹת בִּדְרַךְ כְּלָל בְּיֵצֶר רָחָב, בִּדְחָק, וּבְמַצַּב חֵרוּם. אֲבָל לָשׁוֹן הָרָע? זֶה קַל, זֶה נוֹח, זֶה נַעֲשֶׂה כָּל יוֹם בְּלִי מַחְשָׁבָה. וּבְגָלָל זֶה - זֶה מְסֻכָּן פִּי כַּמָּה וְכַמָּה!

הַקְּלִיפָּה שֶׁל לָשׁוֹן הָרָע

יֵשׁ מַשֶּׁהוּ עוֹד יוֹתֵר עָמֹק. הַמְקֻבָּלִים אוֹמְרִים שֶׁכָּל עֲבֵרָה בָּעוֹלָם יוֹצֶרֶת "קְלִיפָּה" - כֹּחַ רוּחָנִי שֶׁל טֻמְאָה. אֲבָל לָשׁוֹן הָרָע יוֹצֶרֶת קְלִיפָּה מְיֻחֶדֶת וּמְסֻכֶּנֶת, שֶׁהִיא מַפְרִידָה בֵּין אָדָם לְאָדָם, בֵּין אָדָם לְהַקָּבָּ"ה, וּבֵין אָדָם לְעַצְמוֹ.

"מָוֶת וְחַיִּים בְּיַד לָשׁוֹן
וְאֹהֲבֶיהָ יֹאכַל פִּרְיָהּ"
(מִשְׁלֵי יח, כא)

הַפָּסוּק הַזֶּה מְלַמֵּד אוֹתָנוּ שֶׁהַלָּשׁוֹן הִיא כְּלִי נֶשֶׁק - אֶפְשָׁר לְהַשְׁתַּמֵּשׁ בָּהּ לְטוֹב אוֹ לְרַע. כְּמוֹ סַכִּין: אֶפְשָׁר לַחְתֹּךְ לֶחֶם וּלְהַאֲכִיל, וְאֶפְשָׁר - חַס וְשָׁלוֹם - לִפְגֹּעַ. הַבְּחִירָה בְּיָדֵינוּ.

🌟 לָשׁוֹן טוֹב - הַכֹּחַ לִבְנוֹת 🌟

💚 לָשׁוֹן טוֹב - כֹּחַ הַבִּנְיָן וְהַתִּקּוּן

אִם לָשׁוֹן הָרָע הִיא כֹּחַ כָּל כָּךְ הַרְסָנִי, הֲרֵי שֶׁהַהֶפֶךְ הַמּוּחְלָט שֶׁלָּהּ - לָשׁוֹן טוֹב - הוּא כֹּחַ בִּנְיָן אָדִיר! כְּשֶׁם שֶׁמִּלָּה רָעָה הוֹרֶגֶת, מִלָּה טוֹבָה מַחֲיָה!

מַהוּ לָשׁוֹן טוֹב?

  • לְדַבֵּר טוֹב עַל אֲחֵרִים - לְשַׁבֵּחַ, לְעוֹדֵד, לְחַזֵּק
  • לִרְאוֹת אֶת הַטּוֹב בָּאָדָם - גַּם כְּשֶׁיֵּשׁ חֲסָרוֹנוֹת
  • לִתֵּן לֶאֱנָשִׁים "שָׁם טוֹב" - לְהַעֲלוֹת אֶת הַמּוּנִיטִין שֶׁלָּהֶם
  • לְהוֹדוֹת וּלְהַכִּיר תּוֹדָה - לֹא לְקַחַת דְּבָרִים כִּמְּוּבָנִים מֵאֵלָיו

הַכֹּחַ הַמּוּפְלָא שֶׁל מִלָּה טוֹבָה

חַז"ל אוֹמְרִים: "דְּבָרִים הַיּוֹצְאִים מִן הַלֵּב - נִכְנָסִים אֶל הַלֵּב". כְּשֶׁאַתָּה אוֹמֵר לְאָדָם מִלָּה טוֹבָה מֵהַלֵּב, זֶה לֹא רַק "נִימּוּס" - זֶה כֹּחַ שֶׁבּוֹנֶה אֶת הָאָדָם!

מָה קוֹרֶה לְמִי שֶׁמְקַבֵּל מִלָּה טוֹבָה?

  • הוּא מַרְגִּישׁ מֻעֲרָךְ וְאָהוּב
  • הוּא מִתְמַלֵּא בִּטָּחוֹן עַצְמִי
  • הוּא רוֹצֶה לְהִמָּשֵׁךְ וּלְהִשְׁתַּפֵּר
  • הוּא מִתְחַזֵּק לְהַמְשִׁיךְ

מָה קוֹרֶה לְמִי שֶׁנּוֹתֵן מִלָּה טוֹבָה?

  • הוּא מְקָרֵב אֶת הַלֵּב שֶׁלּוֹ לְהַקָּבָּ"ה
  • הוּא יוֹצֵר אוֹר וְשִׂמְחָה בָּעוֹלָם
  • הוּא זוֹכֶה לִברָכָה מִמַּעַל
  • הוּא בּוֹנֶה עוֹלָם טוֹב יוֹתֵר

📖 סִפּוּר הַמִּלָּה שֶׁהִצִּילָה חַיִּים

בְּעִיר צְפַת הָיָה יֶלֶד צָעִיר שֶׁלָּמַד בְּיֶשִׁיבָה. הוּא הָיָה תַּלְמִיד חָלָשׁ, וְהָרַבָּנִים כְּבָר רָצוּ לְסַלֵּק אוֹתוֹ מֵהַיְשִׁיבָה. יוֹם אֶחָד, רַב גָּדוֹל בָּא לְבַקֵּר בַּיְשִׁיבָה. הוּא רָאָה אֶת הַיֶּלֶד, וְאָמַר לוֹ בְּחִבָּה: "אֲנִי רוֹאֶה בְּךָ כֹּחַ גָּדוֹל. תַּמְשִׁיךְ לִלְמֹד, וְתִהְיֶה לְתַלְמִיד חָכָם גָּדוֹל!"

הַמִּלִּים הָאֵלֶּה שִׁנּוּ אֶת כָּל חַיָּיו שֶׁל הַיֶּלֶד. הוּא הִרְגִּישׁ שֶׁמִּשֶּׁהוּ מַאֲמִין בּוֹ, וְהִתְחִיל לְהִשְׁתַּדֵּל בְּלִמּוּד. הַיֶּלֶד הַזֶּה הָיָה - הָרַב שַׁךְ זַצַ"ל, אֶחָד הַגְּדוֹלִים שֶׁבַּדּוֹר!

מִלָּה אַחַת טוֹבָה יְכוֹלָה לִבְנוֹת דּוֹרוֹת!

"תַּפּוּחַ זָהָב בְּמַשְׂכִּיּוֹת כָּסֶף
דָּבָר דָּבוּר עַל אָפְנָיו"
(מִשְׁלֵי כה, יא)

הַפָּסוּק הַזֶּה מְתָאֵר מִלָּה טוֹבָה כְּ"תַּפּוּחַ זָהָב בְּמַשְׂכִּיּוֹת כָּסֶף" - כְּתַכְשִׁיט יָקָר וּמְשֻׁבָּח. וּכְמוֹ תַּכְשִׁיט - גַּם מִלָּה טוֹבָה צְרִיכָה לִהְיוֹת "עַל אָפְנָיו" - בִּזְמַן הַנָּכוֹן, בַּמָּקוֹם הַנָּכוֹן, וּבְאֹפֶן הַנָּכוֹן.

📚 מַעֲשִׂיּוֹת שֶׁמְּחַזְּקוֹת 📚

✨ מַעֲשִׂיּוֹת מְחַזְּקוֹת - סִפּוּרִים מֵהַחַיִּים

סִפּוּר 1: הָרַב שַׁמְשׁוֹן רָפָאֵל הִירְשׁ וְהַמִּכְתָּב

הָרַב הִירְשׁ הָיָה רָב גָּדוֹל בְּגֶרְמַנְיָה. פַּעַם אַחַת כָּתַב לוֹ אָדָם מִכְתָּב מָלֵא בִּקּוֹרֶת וְתוֹכָחָה קָשָׁה. הָרַב יָכוֹל הָיָה לְהִתְעַלֵּם מִמֶּנּוּ, אֲבָל הוּא בָּחַר לַעֲשׂוֹת מַשֶּׁהוּ אַחֵר.

הוּא כָּתַב מִכְתָּב תְּשׁוּבָה אָרֹךְ, בּוֹ הוּא הוֹדָה לָאִישׁ עַל הַדְּאָגָה, הִסְבִּיר אֶת עַמְדָתוֹ בְּאֹהֶב, וּבֵרֵךְ אוֹתוֹ בִּברָכוֹת חַמּוֹת. בְּסוֹף הַמִּכְתָּב הוּא כָּתַב: "יְהִי רָצוֹן שֶׁכָּל דִּבּוּרֵינוּ יִהְיוּ לְכַוָּנַת שָׁמַיִם, וְשֶׁנִּזְכֶּה לְדַבֵּר תָּמִיד בְּאַהֲבָה".

הָאִישׁ שֶׁקִּבֵּל אֶת הַמִּכְתָּב הָיָה כָּל כָּךְ נִרְגָּשׁ, שֶׁהוּא הִתְחִיל לְהִתְקָרֵב לְתוֹרָה וְהָפַךְ לְתַלְמִיד חָשׁוּב בִּישִׁיבָה.

סִפּוּר 2: מֹשֶׁה רַבֵּנוּ וְהַחֲנִית

אִמְרוּ חַז"ל: כְּשֶׁמֹּשֶׁה רַבֵּנוּ רָאָה שֶׁבְּנֵי יִשְׂרָאֵל חָטְאוּ בְּעֵגֶל הַזָּהָב, הוּא יָכוֹל הָיָה לְבַקֵּשׁ מֵהַקָּבָּ"ה לְהַשְׁמִידָם. אֲבָל הוּא לֹא עָשָׂה זֹאת. הוּא בִּקֵּשׁ רַחֲמִים עֲלֵיהֶם!

לָמָּה? כִּי מֹשֶׁה יָדַע שֶׁמִלָּה אַחַת שֶׁל הַגָּנָה עַל עַם יִשְׂרָאֵל שְׁקוּלָה כְּנֶגֶד כָּל הַקִּטְרוּגִים. זֶה כֹּחַ הַלָּשׁוֹן הַטּוֹב - לְהַגֵּן, לְהָגֵן, וּלְהָגֵּן.

סִפּוּר 3: הַמּוֹרָה שֶׁשִּׁנְּתָה חַיִּים

הָיְתָה תַּלְמִידָה צְעִירָה שֶׁחָשְׁבָה שֶׁהִיא לֹא יְכוֹלָה לְהַצְלִיחַ בְּשׁוּם דָּבָר. הִיא הִרְגִּישָׁה חֲסָרַת תּוֹעֶלֶת, וּכְמַעַט וִיתְרָה עַל הַכֹּל. יוֹם אֶחָד, מוֹרָה שֶׁלָּהּ רָאֲתָה בָּהּ מַשֶּׁהוּ מְיֻחָד. הִיא קָרְאָה לָהּ וְאָמְרָה: "אֲנִי רוֹאָה בָּךְ כִּשְׁרוֹן מְיֻחָד. אַל תִּוַּתְרִי!"

הַמִּלִּים הָאֵלֶּה נָתְנוּ לַתַּלְמִידָה כֹּחַ לְהַמְשִׁיךְ. הַיּוֹם, אוֹתָהּ תַּלְמִידָה הִיא מוֹרָה מְצֻיֶּנֶת בְּעַצְמָהּ, וְהִיא אוֹמֶרֶת: "מִלָּה אַחַת טוֹבָה הִצִּילָה אֶת חַיַּי".

  7. תְּפִלָּה יוֹמִית - תִּתְפַּלֵּל כָּל יוֹם: "אֱלֹהַי, נְצֹר לְשׁוֹנִי מֵרָע וּשְׂפָתַי מִדַּבֵּר מִרְמָה"

ד. מוֹצִיא שֵׁם רָע הוּא רַק בְּפָרְהֶסְיָא, לָשׁוֹן הָרָע גַּם בְּצִנְעָה

תְּשׁוּבָה נְכוֹנָה: ב'

הֶסְבֵּר: לָשׁוֹן הָרָע הִיא דִּבּוּר שֶׁפּוֹגֵעַ בָּאָדָם, גַּם אִם הוּא אֱמֶת. מוֹצִיא שֵׁם רָע הוּא כְּשֶׁמְּדַבְּרִים שֶׁקֶר עַל אָדָם, וְזֶה עוֹד יוֹתֵר חָמוּר.

שְׁאֵלָה 3: מַדּוּעַ לָשׁוֹן הָרָע נֶחְשֶׁבֶת חֲמוּרָה כִּשְׁלֹשֶׁת הָעֲבֵרוֹת הַחֲמוּרוֹת?

א. כִּי הִיא פּוֹגַעַת בִּשְׁלֹשָׁה אֲנָשִׁים בוֹ-זְמַנִית

ב. כִּי הִיא קַלָּה לַעֲשִׂיָּה וְנַעֲשֵׂית כָּל הַזְּמַן בְּלִי מַחְשָׁבָה

ג. כִּי הִיא מַפְרִידָה בֵּין אָדָם לְהַקָּבָּ"ה

ד. כָּל הַתְּשׁוּבוֹת נְכוֹנוֹת

תְּשׁוּבָה נְכוֹנָה: ד'

הֶסְבֵּר: כָּל הַסִּבּוֹת הַנֶּזְכָּרוֹת נְכוֹנוֹת וּמְשַׁלְּבוֹת זוֹ אֶת זוֹ. לָשׁוֹן הָרָע חֲמוּרָה בִּגְלַל הַקְּלוּת שֶׁבַּעֲשִׂיָּתָהּ, הַנֶּזֶק שֶׁהִיא גּוֹרֶמֶת לִשְׁלֹשָׁה, וְהַפִּרוּד הָרוּחָנִי שֶׁהִיא יוֹצֶרֶת.

שְׁאֵלָה 4: מָה הַמָּסָר הַמֶּרְכָּזִי שֶׁל סִפּוּר הַנּוֹצוֹת שֶׁל הַחָפֵץ חַיִּים?

א. שֶׁלָּשׁוֹן הָרָע מִתְפַּשֶּׁטֶת מָהֵר

ב. שֶׁמִּלָּה שֶׁיּוֹצֵאת מֵהַפֶּה לֹא אֶפְשָׁר לְהַחֲזִיר אוֹתָהּ

ג. שֶׁצָּרִיךְ לִזְהֹר שֶׁלֹּא לְאַבֵּד נוֹצוֹת

ד. שֶׁאֶפְשָׁר לְתַקֵּן לָשׁוֹן הָרָע עַל-יְדֵי תְּפִלָּה

תְּשׁוּבָה נְכוֹנָה: ב'

הֶסְבֵּר: הַמָּסָל הַמֶּרְכָּזִי הוּא שֶׁכְּמוֹ שֶׁלֹּא אֶפְשָׁר לֶאֱסֹף אֶת הַנּוֹצוֹת לַאֲחַר שֶׁהָרוּחַ פִּזְּרָה אוֹתָן, כָּךְ גַּם מִלָּה רָעָה שֶׁיּוֹצֵאת אִי אֶפְשָׁר לְהַחֲזִיר לְגַמְרֵי.

שְׁאֵלָה 5: מָה זוֹ הַדֶּרֶךְ הַטּוֹבָה בְּיוֹתֵר לְהִתְמוֹדֵד עִם מִישֶׁהוּ שֶׁמְּנַסֶּה לְסַפֵּר לְךָ לָשׁוֹן הָרָע?

א. לִשְׁמֹעַ אֲבָל לֹא לְהַאֲמִין

ב. לְהַפֵּר אֶת הַשִּׂיחָה בְּעָדִינוּת וְלוֹמַר "אֲנִי מַעֲדִיף לֹא לִשְׁמֹעַ"

ג. לְהִתְרַגֵּשׁ וּלְהַגִּיד לוֹ שֶׁהוּא חוֹטֵא

ד. לְעָזֹב אֶת הַמָּקוֹם מִיָּד

תְּשׁוּבָה נְכוֹנָה: ב'

הֶסְבֵּר: הַדֶּרֶךְ הַטּוֹבָה בְּיוֹתֵר הִיא לְהַפֵּר אֶת הַשִּׂיחָה בְּעָדִינוּת וּבְכָבוֹד, מִבְּלִי לְהַשְׁפִּיל אֶת הַמְדַבֵּר. זֶה מוֹנֵעַ אֶת הַחֵטְא וְגַם שׁוֹמֵר עַל הָאֲוִירָה הַטּוֹבָה.

שְׁאֵלָה 6: מָה הַמַּשְׁמָעוּת שֶׁל "מָוֶת וְחַיִּים בְּיַד לָשׁוֹן"?

א. שֶׁהַלָּשׁוֹן יְכוֹלָה לְדַבֵּר עַל מָוֶת וְחַיִּים

ב. שֶׁהַלָּשׁוֹן הִיא כְּלִי שֶׁאֶפְשָׁר לְהַשְׁתַּמֵּשׁ בּוֹ לְטוֹב (חַיִּים) אוֹ לְרַע (מָוֶת)

ג. שֶׁלָשׁוֹן הָרָע גּוֹרֶמֶת לְמָוֶת פִיזִי

ד. שֶׁהַלָּשׁוֹן חָזָקָה יוֹתֵר מֵהַיָּד

תְּשׁוּבָה נְכוֹנָה: ב'

הֶסְבֵּר: הַפָּסוּק מְלַמֵּד אוֹתָנוּ שֶׁהַלָּשׁוֹן הִיא כְּלִי נֶשֶׁק, וְהַבְּחִירָה בְּיָדֵינוּ אֵיךְ לְהַשְׁתַּמֵּשׁ בָּהּ - לִבְנוֹת וּלְהַחֲיוֹת, אוֹ לַהֲרֹס וּלְהָמִית.

שְׁאֵלָה 7: מָה הַכֹּחַ הַמְיֻחָד שֶׁל מִלָּה טוֹבָה?

א. הִיא גּוֹרֶמֶת לָאָדָם לְהִרְגִּישׁ טוֹב רֶגַע אֶחָד

ב. הִיא יְכוֹלָה לִבְנוֹת אֶת הָאָדָם, לְחַזֵּק אוֹתוֹ, וְלִפְעָמִים אֲפִלּוּ לְשַׁנּוֹת אֶת כָּל חַיָּיו

ג. הִיא מְבַטֶּלֶת אֶת לָשׁוֹן הָרָע שֶׁדֻּבְּרָה קוֹדֶם

ד. הִיא גּוֹרֶמֶת לָאָדָם לְהִשְׁתַּפֵּר מִיָּד

תְּשׁוּבָה נְכוֹנָה: ב'

הֶסְבֵּר: כְּפִי שֶׁרָאִינוּ בַּסִּפּוּרִים, מִלָּה טוֹבָה אַחַת יְכוֹלָה לִבְנוֹת אָדָם שָׁלֵם, לָתֵת לוֹ בִּטָּחוֹן, וְלִפְעָמִים אֲפִלּוּ לְשַׁנּוֹת אֶת כָּל מַסְלוּל חַיָּיו לְטוֹבָה.

הַמִּלָּה הָאַחֲרוֹנָה

הַקָּבָּ"ה נָתַן לָנוּ אֶת הַכֹּחַ הַגָּדוֹל בְּיוֹתֵר בָּעוֹלָם - כֹּחַ הַדִּבּוּר.
עִם כֹּחַ זֶה אֶפְשָׁר לִבְנוֹת עוֹלָמוֹת, וְאֶפְשָׁר - חַס וְשָׁלוֹם - לַחֲרֹב אוֹתָם.

כָּל יוֹם, בְּכָל שִׂיחָה, אָנוּ עוֹמְדִים בִּפְנֵי בְּחִירָה:
לִבְנוֹת אוֹ לַהֲרֹס?

הַבְּחִירָה הִיא שֶׁלָּנוּ, וְהָאַחֲרָיוּת - גַּם כֵּן.

שֶׁנִּזְכֶּה לִשְׁמֹר אֶת לְשׁוֹנֵנוּ מֵרָע
וְלִבְנוֹת בְּכָל מִלָּה עוֹלָם שֶׁל טוֹב, אַהֲבָה וְאוֹר!

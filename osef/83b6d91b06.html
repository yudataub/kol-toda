💎 מָבוֹא: מִצְוָה חֲבִיבָה עַד לִמְאֹד

מִצְוַת נֵר חֲנֻכָּה נִצֶּבֶת בְּמָקוֹם יְחִידִי בְּעוֹלָמָהּ שֶׁל הַהֲלָכָה. הִיא אֵינָהּ עוֹד מִצְוָה מִן הַמִּנְיָן, אֶלָּא מִצְוָה שֶׁחֲכָמִים הִפְלִיגוּ בְּחֲשִׁיבוּתָהּ וּבְמַעֲלָתָהּ. כְּדֵי לְהַעֲרִיךְ אֶת יְמֵי הַחֲנֻכָּה כָּרָאוּי וּלְהִתְחַבֵּר אֶל הָאוֹר הַמְיֻחָד שֶׁלָּהֶם, חִיּוּנִי לְהָבִין אֶת עֹמֶק מַשְׁמָעוּתָהּ שֶׁל מִצְוָה זוֹ.

הָרַמְבַּ"ם, בְּכָתְבוֹ עַל הִלְכוֹת חֲנֻכָּה, מִשְׁתַּמֵּשׁ בְּבִטּוּי שֶׁאֵין לוֹ אָח וָרֵעַ בְּכָל חִבּוּרוֹ הַהֲלָכָתִי הַמּוֹנוּמֶנְטָלִי "מִשְׁנֵה תּוֹרָה":

"מִצְוַת נֵר חֲנֻכָּה
מִצְוָה חֲבִיבָה הִיא עַד לִמְאֹד"

(רַמְבַּ"ם, הִלְכוֹת חֲנֻכָּה, פֶּרֶק ד', הֲלָכָה י"ב)

הַגְדָרָה זוֹ אֵינָהּ נִשְׁאֶרֶת בִּתְחוּם הָרִגְשִׁי אוֹ הָרוּחָנִי בִּלְבָד, אֶלָּא מִתְתַּרְגֶּמֶת לַהֲלָכָה לְמַעֲשֶׂה בְּאֹפֶן חֲסַר תַּקְדִים, וְאַף מַבְעִיל בְּחֻמְרָתוֹ.

הַשְׁוָאָה מַפְתִּיעָה בְּחֻמְרָתָהּ:

בְּעוֹד שֶׁבְּרֹב מִצְווֹת עֲשֵׂה, אָדָם שֶׁאֶמְצָעָיו מֻגְבָּלִים פָּטוּר אִם עֲלוּת הַמִּצְוָה עוֹלָה עַל חֲמִישִׁית מִנְּכָסָיו, בְּנֵר חֲנֻכָּה הַדִּין שׁוֹנֶה לְחַלּוּטִין.

חֻמְרָתָהּ שֶׁל מִצְוַת נֵר חֲנֻכָּה גְדוֹלָה יוֹתֵר מִמִּצְווֹת יְסוֹד כְּמוֹ תְּפִלִּין, מְזוּזָה וַאֲפִלּוּ אֲכִילַת מַצָּה בְּלֵיל הַסֵּדֶר!

בְּהִלְכוֹת חֲנֻכָּה נִפְסָק כִּי גַּם אָדָם שֶׁאֵין לוֹ כֶּסֶף כְּלָל, "צָרִיךְ לְחַזֵּר עַל הַפְּתָחִים" – לְבַקֵּשׁ נְדָבוֹת – כְּדֵי שֶׁיּוּכַל לְהַשִּׂיג שֶׁמֶן אוֹ נֵרוֹת וּלְהַדְלִיק. זוֹהִי חֻמְרָה שֶׁאֵין כְּדוּגְמָתָהּ, הַמְעִידָה עַל מַעֲמָד יוֹצֵא דֹּפֶן.

הִתְבּוֹנְנוּת זוֹ מַעֲלָה מִיָּד אֶת הַשְּׁאֵלָה הַמִּתְבַּקֶּשֶׁת: מַהִי הַסִּבָּה לַחֲשִׁיבוּתָהּ הָעֲצוּמָה וְהַיְּחִידִית שֶׁל מִצְוָה זוֹ, הַמְצַיֶּבֶת אוֹתָהּ מֵעַל לְמִצְווֹת יְסוֹד אֲחֵרוֹת בַּתּוֹרָה? מַהוּ הַסּוֹד הַטָּמוּן בְּנֵרוֹת הַחֲנֻכָּה שֶׁהוֹפֵךְ אוֹתָם ל"חֲבִיבִים עַד לִמְאֹד"?

❓ אַרְבַּע קֻשְׁיוֹת יְסוֹד ❓

🤔 אַרְבַּע קֻשְׁיוֹת יְסוֹד בְּנֵר חֲנֻכָּה

כְּדֵי לְפַעֲנֵחַ אֶת סוֹדָהּ שֶׁל מִצְוַת נֵר חֲנֻכָּה, עָלֵינוּ רֵאשִׁית לְהִתְמוֹדֵד עִם אַרְבַּע שְׁאֵלוֹת מַהוּתִיּוֹת הַמַּבְדִּילוֹת אוֹתָהּ בְּאֹפֶן מֻבְהָק מִשְּׁאָר הַמִּצְווֹת. שְׁאֵלוֹת אֵלּוּ חוֹשְׂפוֹת אֶת הַמִּבְנֶה הַיְּחִידִי שֶׁל הַמִּצְוָה וּמוֹבִילוֹת אוֹתָנוּ אֶל שֹׁרֶשׁ מַשְׁמָעוּתָהּ.

אַרְבַּע הַשְּׁאֵלוֹת הַגְּדוֹלוֹת:

1️⃣ הַחֲבִיבוּת וְהַחֻמְרָה:
מַדּוּעַ מִצְוַת נֵר חֲנֻכָּה מוּגְדֶּרֶת כְּ"חֲבִיבָה עַד לִמְאֹד", וְחֻמְרָתָהּ הַהֲלָכָתִית גְּדוֹלָה יוֹתֵר מִשֶּׁל מִצְווֹת אֲחֵרוֹת, עַד כְּדֵי חִיּוּבוֹ שֶׁל עָנִי לְחַזֵּר עַל הַפְּתָחִים כְּדֵי לְקַיְּמָהּ?
2️⃣ מְקוֹם קִיּוּם הַמִּצְוָה:
מַדּוּעַ, בְּנִגּוּד לְכָל הַמִּצְווֹת שֶׁאוֹתָן נִתָּן לְקַיֵּם כִּמְעַט בְּכָל מָקוֹם – עַל הַגָּג, בַּבּוֹיְדֶם, בַּמִּקְלָט – אֶת נֵר חֲנֻכָּה יֵשׁ לְהַדְלִיק דַוְקָא "בְּפֶתַח בֵּיתוֹ" וְלֹא בְּשׁוּם מָקוֹם אַחֵר?
3️⃣ דִּין "אִשְׁתּוֹ כְּגוּפוֹ":
כֵּיצַד יִתָּכֵן שֶׁבְּנֵר חֲנֻכָּה אָדָם יוֹצֵא יְדֵי חוֹבָה בְּהַדְלָקָה שֶׁל אִשְׁתּוֹ אוֹ שֶׁל בֶּן בַּיִת אַחֵר, אֲפִלּוּ אִם הוּא אֵינוֹ נִמְצָא פִיזִית בַּבַּיִת? הֲרֵי בְּכָל הַמִּצְווֹת הָאֲחֵרוֹת הַחִיּוּב הוּא אִישִׁי וְאֵינוֹ נִתָּן לְהַעֲבָרָה!
4️⃣ קְדֻשַּׁת הַנֵּרוֹת:
מַהוּ פֶּשֶׁר הָאֲמִירָה שֶׁאָנוּ שָׁרִים "הַנֵּרוֹת הַלָּלוּ קֹדֶשׁ הֵם"? כֵּיצַד חֵפֶץ גַּשְׁמִי כְּמוֹ נֵרוֹת וְשֶׁמֶן מִתְקַדֵּשׁ, וּמַהוּ מְקוֹר קְדֻשָּׁתָם הָעֶלְיוֹנָה, עַד כְּדֵי כָּךְ שֶׁאֵין לָנוּ רְשׁוּת לְהִשְׁתַּמֵּשׁ בָּהֶם?

הַתְּשׁוּבָה לְכָל הַשְּׁאֵלוֹת הַלָּלוּ טְמוּנָה בִּיסוֹד אֶחָד,
עָמֹק וְנִפְלָא, הַמִּסְתַּתֵּר בְּדִבְרֵי הָרַמְבַּ"ן!

🕯️ יְסוֹד הַיְסוֹדוֹת 🕯️

⭐ יְסוֹד הַיְסוֹדוֹת: נֶחָמַת אַהֲרֹן וָרֶמֶז הָרַמְבַּ"ן

הַמַּפְתֵּחַ לְהֲבָנַת סוֹדוֹת הַחֲנֻכָּה נִמְצָא בְּפֵרוּשׁוֹ שֶׁל הָרַמְבַּ"ן עַל פָּרָשַׁת בְּהַעֲלֹתְךָ, שָׁם הוּא דָּן בְּנֶחָמָה הַמְיֻחֶדֶת שֶׁהֵעֱנִיק הַקָּבָּ"ה לְאַהֲרֹן הַכֹּהֵן.

לְאַחַר שֶׁכָּל נְשִׂיאֵי הַשְּׁבָטִים הִשְׁתַּתְּפוּ בַּחֲנֻכַּת הַמִּשְׁכָּן וְהִקְרִיבוּ אֶת קָרְבְּנוֹתֵיהֶם, הִצְטַעֵר אַהֲרֹן עַל כָּךְ שֶׁשִּׁבְטוֹ, שֵׁבֶט לֵוִי, לֹא זָכָה לָקַחַת חֵלֶק בַּשִּׂמְחָה זוֹ. בִּתְגוּבָה לְצַעֲרוֹ, הִבְטִיחַ לוֹ הַקָּבָּ"ה: "שֶׁלְּךָ גְדוֹלָה מִשֶּׁלָּהֶם".

הָרַמְבַּ"ן שׁוֹאֵל קֻשְׁיָה נוֹקֶבֶת: הַנֶּחָמָה שֶׁנִּתְּנָה לְאַהֲרֹן הָיְתָה זְכוּת הַדְלָקַת נֵרוֹת הַמְּנוֹרָה בְּבֵית הַמִּקְדָּשׁ, זְכוּת שֶׁהוּבְטְחָה לוֹ "לְעוֹלָם". אַךְ כֵּיצַד הַבְטָחָה זוֹ יְכוֹלָה לִהְיוֹת נִצְחִית, הֲרֵי בֵּית הַמִּקְדָּשׁ חָרֵב וְהַמְּנוֹרָה אֵינֶנָּה עוֹד?

תְּשׁוּבָתוֹ שֶׁל הָרַמְבַּ"ן הִיא יְסוֹד הַיְסוֹדוֹת לְהֲבָנַת הַחֲנֻכָּה:

"הָרֶמֶז הוּא עַל חֲנֻכָּה שֶׁל נֵרוֹת
שֶׁהָיְתָה בְּבַיִת שֵׁנִי...

יֵשׁ חֲנֻכָּה אַחֶרֶת
שֶׁאֲנִי עוֹשֶׂה בָּהּ נִסִּים לְעַם יִשְׂרָאֵל...

וְהִיא חֲנֻכַּת הַמְּנוֹרָה"

(רַמְבַּ"ן, פָּרָשַׁת בְּהַעֲלֹתְךָ)

הַסּוֹד הָאַדִּיר שֶׁמְּגַלֶּה הָרַמְבַּ"ן:

הַבְטָחָתוֹ שֶׁל הַקָּבָּ"ה לְאַהֲרֹן לֹא בָטְלָה עִם חֻרְבַּן הַבַּיִת. הִיא מַמְשִׁיכָה לְהִתְקַיֵּם דֶּרֶךְ נֵרוֹת הַחֲנֻכָּה!

הַחֲנֻכִּיָּה שֶׁאָנוּ מַדְלִיקִים בְּבָתֵּינוּ אֵינָהּ רַק זֵכֶר לְנֵס פַּךְ הַשֶּׁמֶן; הִיא הַמְשֵׁךְ יָשִׁיר וְחַי שֶׁל מְנוֹרַת בֵּית הַמִּקְדָּשׁ!

הַמַּהְפַּךְ הַתּוֹדָעָתִי:

עַל פִּי יְסוֹד זֶה, כַּאֲשֶׁר יְהוּדִי נִגָּשׁ לְהַדְלִיק נֵרוֹת חֲנֻכָּה, הוּא אֵינוֹ מְבַצֵּעַ פְּעֻלָּה סִמְבּוֹלִית בִּלְבָד. בְּאוֹתָהּ שָׁעָה:

  • 🙋‍♂️ הוּא הוֹפֵךְ לִהְיוֹת אַהֲרֹן הַכֹּהֵן
  • 🕎 הַחֲנֻכִּיָּה שֶׁלּוֹ הִיא מְנוֹרַת הַמִּקְדָּשׁ
  • ✨ הוּא מְקַיֵּם אֶת מִצְוַת "בְּהַעֲלֹתְךָ אֶת הַנֵּרֹת" בְּאֹפֶן הַמַּמָּשִׁי בְּיוֹתֵר

יְסוֹד זֶה מַסְבִּיר מַדּוּעַ מִנְהָגִים רַבִּים בְּהַדְלָקַת הַנֵּרוֹת נוֹעֲדוּ לְחַקּוֹת אֶת עֲבוֹדַת הַמִּקְדָּשׁ:

  • אָנוּ מְהַדְּרִים לְהַדְלִיק בְּשֶׁמֶן זַיִת, כְּפִי שֶׁהָיָה בַּמְּנוֹרָה
  • רַבִּים נוֹהֲגִים לְהִתְפַּלֵּל מִנְחָה לִפְנֵי הַהַדְלָקָה, כְּפִי שֶׁבַּמִּקְדָּשׁ הִקְרִיבוּ אֶת קָרְבָּן הַתָּמִיד שֶׁל בֵּין הָעַרְבַּיִם וְרַק לְאַחֲרָיו הִדְלִיקוּ אֶת הַמְּנוֹרָה

כַּעֵת, כְּשֶׁאָנוּ אוֹחֲזִים בִּיסוֹד רוּחָנִי זֶה,
אָנוּ יְכוֹלִים לַחֲזוֹר אֶל אַרְבַּע הַקֻּשְׁיוֹת
וְלִרְאוֹת כֵּיצַד הֵן נִפְתָּרוֹת בִּמְלוֹאָן! 💡

💡 הַתְּמוּנָה מִתְבָּרֶרֶת 💡

🔍 הַתְּמוּנָה מִתְבָּרֶרֶת: פִּתָּרוֹן אַרְבַּע הַקֻּשְׁיוֹת

לְאוֹר יְסוֹדוֹ הַמֵּאִיר שֶׁל הָרַמְבַּ"ן, כָּל הַתְּמִיהוֹת שֶׁהִצַּגְנוּ מִתְיַשְּׁבוֹת וּמִתְחַבְּרוֹת לִתְמוּנָה אַחַת שְׁלֵמָה, הַמְגַלָּה אֶת עָצְמָתָם שֶׁל יְמֵי הַחֲנֻכָּה.

4️⃣ מַדּוּעַ הַנֵּרוֹת קֹדֶשׁ הֵם?

הַתְּשׁוּבָה לַקֻּשְׁיָה הָרְבִיעִית הוֹפֶכֶת לִפְשׁוּטָה וּמוּבֶנֶת מֵאֵלֶיהָ.

אָנוּ אוֹמְרִים "הַנֵּרוֹת הַלָּלוּ קֹדֶשׁ הֵם" מִפְּנֵי שֶׁהֵם בֶּאֱמֶת קֹדֶשׁ.

הֵם אֵינָם חֵפֶץ רָגִיל, אֶלָּא הַמְשֵׁךְ יָשִׁיר שֶׁל מְנוֹרַת בֵּית הַמִּקְדָּשׁ, שֶׁהָיְתָה כְּלִי קֹדֶשׁ קָדָשִׁים. קְדֻשָּׁתָהּ שֶׁל הַמְּנוֹרָה הַמְקוֹרִית מַאֲצִילָה עַל הַחֲנֻכִּיּוֹת שֶׁלָּנוּ, וְהוֹפֶכֶת אוֹתָן לִכְלֵי מִקְדָּשׁ בְּתוֹךְ בֵּיתֵנוּ!

2️⃣ מַדּוּעַ מַדְלִיקִים דַוְקָא בַּבַּיִת? הַבַּיִת כְּמִקְדָּשׁ מְעַט

אִם הַחֲנֻכִּיָּה הִיא הַמְּנוֹרָה וְהָאָדָם הַמַּדְלִיק הוּא הַכֹּהֵן, הֵיכָן נִמְצָא בֵּית הַמִּקְדָּשׁ?

חֲזַ"ל מְלַמְּדִים אוֹתָנוּ אֶת הַתְּשׁוּבָה בִּדְרָשָׁתָם עַל הַפָּסוּק:

"וְעָשׂוּ לִי מִקְדָּשׁ וְשָׁכַנְתִּי בְּתוֹכָם"
(שְׁמוֹת כ"ה, ח')
חֲכָמִים מְדַיְּקִים: לֹא נֶאֱמַר "וְשָׁכַנְתִּי בְּתוֹכוֹ", אֶלָּא "וְשָׁכַנְתִּי בְּתוֹכָם" – בְּתוֹךְ כָּל אֶחָד וְאֶחָד מִיִּשְׂרָאֵל. הַשְׁרָאַת הַשְּׁכִינָה בִּמְלוֹאָהּ מִתְרַחֶשֶׁת בַּבַּיִת הַיְּהוּדִי, בַּמָּקוֹם הַמְאַחֵד אִישׁ וְאִשָּׁה, הוֹרִים וִילָדִים.

מִכָּאן נוֹלֶדֶת הַמַּסְקָנָה הַמַּדְהִימָה:

🏛️ בִּימֵי הַחֲנֻכָּה 🏛️

הַבַּיִת הַיְּהוּדִי הוֹפֵךְ לִהְיוֹת בֵּית הַמִּקְדָּשׁ!

לָכֵן, הַמִּצְוָה הִיא לְהַדְלִיק אֶת "הַמְּנוֹרָה" (הַחֲנֻכִּיָּה)
דַוְקָא בְּ"מִקְדָּשׁ" (הַבַּיִת).

3️⃣ כֵּיצַד אָדָם אֶחָד מוֹצִיא אֶת כֻּלָּם יְדֵי חוֹבָה?

כְּשֵׁם שֶׁבְּבֵית הַמִּקְדָּשׁ הַגָּדוֹל, הַדְלָקַת הַמְּנוֹרָה עַל יְדֵי הַכֹּהֵן הַמְשָׁרֵת פָּטְרָה אֶת כָּל עַם יִשְׂרָאֵל, כָּךְ גַּם בְּ"מִקְדָּשׁ מְעַט" הַבַּיִתִי.

כַּאֲשֶׁר אֶחָד מִבְּנֵי הַבַּיִת מַדְלִיק אֶת הַחֲנֻכִּיָּה, הוּא פּוֹעֵל כְּ"כֹּהֵן" שֶׁל אוֹתוֹ "מִקְדָּשׁ", וּפְעֻלָּתוֹ מוֹצִיאָה יְדֵי חוֹבָה אֶת כָּל שְׁאָר בְּנֵי הַבַּיִת הַמִּשְׁתַּיְּכִים לְאוֹתוֹ מִקְדָּשׁ.

הַזֹּהַר הַקָּדוֹשׁ מַעֲמִיק רַעְיוֹן זֶה וּמַסְבִּיר שֶׁהַבַּיִת הַיְּהוּדִי הַשָּׁלֵם הוּא מִבְנֶה אֱלֹהִי, הַמְרַמֵּז לִשְׁמוֹ שֶׁל הַקָּבָּ"ה (י-ה-ו-ה):

  • י' כְּנֶגֶד הָאִישׁ 👨
  • ה' הָרִאשׁוֹנָה כְּנֶגֶד הָאִשָּׁה 👩
  • ו' כְּנֶגֶד הַבָּנִים 👦👦
  • ה' הָאַחֲרוֹנָה כְּנֶגֶד הַבָּנוֹת 👧👧

כְּשֶׁהַמְּנוֹרָה דּוֹלֶקֶת בַּמִּקְדָּשׁ הַמִּשְׁפַּחְתִּי, אוֹרָהּ פּוֹטֵר אֶת כָּל חֶלְקֵי הַמִּבְנֶה הַקָּדוֹשׁ הַזֶּה, גַּם אֶת אֵלּוּ שֶׁאֵינָם נוֹכְחִים פִיזִית בְּאוֹתָהּ שָׁעָה.

1️⃣ מַדּוּעַ הַמִּצְוָה חֲבִיבָה עַד לִמְאֹד?

כַּעֵת, אָנוּ מַגִּיעִים לְשִׂיא הַהֲבָנָה. וְכִי יֵשׁ לְךָ דָבָר נִפְלָא מִזֶּה?

מִצְוַת נֵר חֲנֻכָּה חֲבִיבָה כָּל כָּךְ
כִּי הִיא מַתָּנָה שֶׁאֵין גְּדוֹלָה מִמֶּנָּה!

הַקָּבָּ"ה מְאַפְשֵׁר לְכָל יְהוּדִי,
לְמֶשֶׁךְ שְׁמוֹנָה יָמִים בַּשָּׁנָה,
לְהַפֹּךְ אֶת הַמְּצִיאוּת הַיּוֹמְיוֹמִית שֶׁלּוֹ
לִמְצִיאוּת שֶׁל קְדֻשָּׁה עֶלְיוֹנָה

הַמַּתָּנָה הַגְּדוֹלָה:

  • 🏠 לְהַפֹּךְ אֶת בֵּיתוֹ לְמִקְדָּשׁ
  • 👤 לְהַפֹּךְ אֶת עַצְמוֹ לְכֹהֵן גָּדוֹל
  • 🕎 לְהַפֹּךְ אֶת הַחֲנֻכִּיָּה שֶׁלּוֹ לִמְנוֹרַת זָהָב טָהוֹר

זוֹהִי הַזְדַּמְּנוּת נְדִירָה לְהִתְחַבֵּר לִקְדֻשַּׁת הַמִּקְדָּשׁ
בְּאֹפֶן אִישִׁי וּבִלְתִּי אֶמְצָעִי!

אֵין פֶּלֶא שֶׁמִּצְוָה כָּזוֹ, הַמְרוֹמֶמֶת אֶת כָּל הֲוָיָתֵנוּ,
הִיא "חֲבִיבָה עַד לִמְאֹד"! ✨

📖 הַיִּשּׁוּם הַמַּעֲשִׂי 📖

🎵 הַיִּשּׁוּם הַמַּעֲשִׂי: הֲלָכָה, שִׁירָה וְהַכָּרַת הַטּוֹב

הַהֲבָנָה הָעֲמֻקָּה שֶׁבֵּיתֵנוּ הוּא מִקְדָּשׁ בִּימֵי הַחֲנֻכָּה אֵינָהּ נִשְׁאֶרֶת רַעְיוֹן תֵּאוֹרֵטִי, אֶלָּא מִתְתַּרְגֶּמֶת לַהֲלָכוֹת מַעֲשִׂיּוֹת וּלְהִתְנַהֲגוּת רוּחָנִית נִדְרֶשֶׁת.

1️⃣ הַבַּיִת הוֹפֵךְ לְהֵיכַל שִׁירָה

כְּשֵׁם שֶׁבְּבֵית הַמִּקְדָּשׁ הָיוּ הַלְוִיִּם עוֹמְדִים עַל דּוּכָנָם וְשָׁרִים שִׁירֵי הַלֵּל וְהוֹדָיָה לַה', כָּךְ גַּם הַבַּיִת שֶׁלָּנוּ, לְאַחַר שֶׁהוּדְלְקָה בוֹ "הַמְּנוֹרָה", צָרִיךְ לְהִתְמַלֵּא בְּשִׁירָה, בְּהוֹדָיָה וּבְשִׂמְחָה.

רְגָעִים אֵלּוּ הֵם זְמַן לְהִתְכַּנְסוּת מִשְׁפַּחְתִית, לָשִׁיר יַחַד "מָעוֹז צוּר" וּפִיּוּטִים נוֹסָפִים, וּלְהַפֹּךְ אֶת הַבַּיִת לְמָקוֹם שֶׁל קוֹל תּוֹדָה.

2️⃣ חוֹבַת הַזְּהִירוּת וְהַכָּרַת הַטּוֹב

הַשֻּׁלְחָן עָרוּךְ פּוֹסֵק כִּי בְּמִצְוַת נֵר חֲנֻכָּה "צָרִיךְ לְהִזָּהֵר בָּהּ מְאֹד". מְעַנְיֵן לְגַלּוֹת שֶׁהַבִּטּוּי הַנָּדִיר "לְהִזָּהֵר מְאֹד" מוֹפִיעַ בַּשֻּׁלְחָן עָרוּךְ בִּשְׁלוֹשָׁה הֶקְשָׁרִים בִּלְבָד, וּלְכֻלָּם מְכַנֶּה מְשֻׁתָּף אֶחָד: הַכָּרַת הַטּוֹב, שֶׁהִיא יְסוֹד הַיַּהֲדוּת.

שְׁלוֹשׁ מִצְווֹת שֶׁצָּרִיךְ "לְהִזָּהֵר מְאֹד":

הַמִּצְוָה הַמַּהוּת הַמְשֻׁתֶּפֶת
בִּרְכַּת הַתּוֹרָה הַכָּרַת הַטּוֹב עַל מַתְּנַת הַתּוֹרָה 📖
כִּבּוּד הוֹרִים הַכָּרַת הַטּוֹב לַהוֹרִים שֶׁהֱבִיאוּנוּ לָעוֹלָם 👨‍👩‍👧‍👦
נֵר חֲנֻכָּה הַכָּרַת הַטּוֹב עַל הַנִּסִּים וְעַל מַתְּנַת הַקְּדֻשָּׁה 🕎

הַזְּהִירוּת הַנִּדְרֶשֶׁת אֵינָהּ רַק בִּפְרָטֵי הַהֲלָכָה, אֶלָּא בָּעִקָּר בַּתֹּכֶן הַפְּנִימִי: לְהִזָּהֵר לְהוֹדוֹת וּלְהַלֵּל מִכָּל הַלֵּב עַל הַמַּתָּנוֹת הַנִּפְלָאוֹת שֶׁקִּבַּלְנוּ – הַבַּיִת שֶׁהָפַךְ לְמִקְדָּשׁ, וְהַזְּכוּת לִהְיוֹת כֹּהֲנִים הַמַּדְלִיקִים אֶת אוֹר ה'.

3️⃣ כְּנֶגֶד הַיְּוָנִים: קְדֻשָּׁה בַּבַּיִת

רַעְיוֹן זֶה עוֹמֵד כְּנֶגֶד מַהוּת הַמַּאֲבָק בַּיְּוָנִים. הֵם בִּקְשׁוּ לַעֲקוֹר אֶת הַקְּדֻשָּׁה, לְטַמֵּא אֶת הַשֶּׁמֶן, לְהַשְׁכִּיחַ אֶת הַתּוֹרָה וּלְהַפֹּךְ אֶת חַיֵּינוּ לְחֻלִּין.

כְּנֶגֶד רְצוֹנָם לְטַמֵּא, נָתַן לָנוּ הַקָּבָּ"ה קְדֻשָּׁה עֶלְיוֹנָה הַמִּתְגַּלָּה דַוְקָא בַּמָּקוֹם הַמֶּרְכָּזִי וְהָאִינְטִימִי בְּיוֹתֵר – הַבַּיִת הַיְּהוּדִי!

💫 סִכּוּם וּבְרָכָה

יְמֵי הַחֲנֻכָּה מַזְמִינִים אוֹתָנוּ לְמַסָּע רוּחָנִי. הִתְחַלְנוּ מִקֻּשְׁיוֹת וּתְמִיהוֹת עַל יְחִידִיּוּתָהּ שֶׁל מִצְוַת נֵר חֲנֻכָּה, וְדֶרֶךְ יְסוֹדוֹ הַמֵּאִיר שֶׁל הָרַמְבַּ"ן, הִגַּעְנוּ לַהֲבָנָה הַמְלֵאָה:

לְמֶשֶׁךְ שְׁמוֹנָה יָמִים,
הַבַּיִת שֶׁלָּנוּ הוּא בֵּית הַמִּקְדָּשׁ 🏛️

זוֹהִי אֵינָהּ מְלִיצָה,
אֶלָּא אֱמֶת רוּחָנִית עֲמֻקָּה
בַּעֲלַת הַשְׁלָכוֹת מַעֲשִׂיּוֹת

מִתּוֹךְ הֲבָנָה זוֹ, אָנוּ נִקְרָאִים לְהַפֹּךְ אֶת בָּתֵּינוּ לְמָקוֹם שֶׁל:

  • 🎵 שִׁירָה – לְשׁוֹרֵר יַחַד "מָעוֹז צוּר" וּלְהַלֵּל לַה'
  • 🙏 הוֹדָיָה – לְהַכִּיר טוֹבָה עַל הַמַּתָּנָה הַגְּדוֹלָה
  • ✨ קְדֻשָּׁה – לְהַרְגִּישׁ שֶׁהַבַּיִת הוּא מִקְדָּשׁ מַמָּשׁ

בְּרָכָה לְחַג הַחֲנֻכָּה

שֶׁנִּזְכֶּה כֻּלָּנוּ לַחֲנֻכָּה שָׂמֵחַ,
חֲנֻכָּה שֶׁל קְדֻשָּׁה,
חֲנֻכָּה שֶׁמָּלֵא שִׁירָה וְהוֹדָיָה,

וְנִקְבַּע אֶת שְׁמוֹנַת יְמֵי חֲנֻכָּה אֵלּוּ
לְהוֹדוֹת וּלְהַלֵּל לְשִׁמְךָ הַגָּדוֹל!

חֲנֻכָּה שָׂמֵחַ! 🕎✨

📚 מְקוֹרוֹת וּמָרֵאי מְקוֹמוֹת 📚

מְקוֹרוֹת רָאשִׁיִּים

  • רַמְבַּ"ם – מִשְׁנֵה תּוֹרָה, הִלְכוֹת חֲנֻכָּה, פֶּרֶק ד', הֲלָכָה י"ב
  • רַמְבַּ"ן – פֵּרוּשׁ עַל הַתּוֹרָה, פָּרָשַׁת בְּהַעֲלֹתְךָ
  • שֻׁלְחָן עָרוּךְ, אֹרַח חַיִּים, הִלְכוֹת חֲנֻכָּה
  • מִשְׁנָה בְּרוּרָה – הִלְכוֹת חֲנֻכָּה
  • גְּמָרָא – מַסֶּכֶת שַׁבָּת, דַּף כ"ג, עַמּוּד א'
  • שְׁמוֹת כ"ה, ח' – "וְעָשׂוּ לִי מִקְדָּשׁ וְשָׁכַנְתִּי בְּתוֹכָם"
  • זֹהַר הַקָּדוֹשׁ – עַל מִבְנֵה הַבַּיִת הַיְּהוּדִי
  • פִּיּוּט "מָעוֹז צוּר" – שִׁיר מָסוֹרְתִּי לְחַג הַחֲנֻכָּה
נִכְתַּב וְנֶעֱרַךְ עַל יְדֵי:
הָרַב יְהוּדָה טָאוּב

קִרְיַת אַתָּא

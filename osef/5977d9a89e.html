מגילת הגאולה

פֶּרֶק: עֲלִיַּת תַּלְמִידֵי הַגָּאוֹן מִוִּילְנָא

וַיְהִי בְּאֶרֶץ לִיטָא, וַיִּזְרַח אוֹר גָּדוֹל בְּיִשְׂרָאֵל, הוּא הַגָּאוֹן רַבִּי אֵלִיָּהוּ בֶּן־שְׁלֹמֹה זַלְמָן מִוִּילְנָא׃ וַיְהִי אִישׁ קָדוֹשׁ וְטָהוֹר, אֲשֶׁר לִבּוֹ כְּלֵב הָאַרְיֵה, גָּדוֹל בַּתּוֹרָה וּבַחָכְמָה, וְנוֹדָע בְּכָל־תְּפוּצוֹת יִשְׂרָאֵל כְּאַחַד מִגְּדוֹלֵי הַדּוֹרוֹת׃

וַיְהִי לֵב הַגָּאוֹן מִוִּילְנָא בּוֹעֵר בְּאַהֲבַת אֶרֶץ יִשְׂרָאֵל אַהֲבָה עַזָּה כַּמָּוֶת, וַתְּהִי לוֹ תְּשׁוּקָה גְדוֹלָה לַעֲלוֹת וּלְהִשְׁתַּקֵּעַ בָּאָרֶץ אֲשֶׁר עֵינֵי ה׳ אֱלֹהֶיךָ בָּהּ׃ וַיַּחְשֹׁב עַל־מִצְוַת יִשּׁוּב אֶרֶץ־יִשְׂרָאֵל יוֹמָם וָלַיְלָה, וַיֹּאמֶר בְּלִבּוֹ: "אֵלְכָה נָּא וְאֶרְאֶה אֶת־הָאָרֶץ הַטּוֹבָה הַזֹּאת, וְאֶשְׁכֹּן בָּהּ כָּל־יְמֵי חַיַּי"׃

וַיְהִי בִּשְׁנַת חֲמֵשֶׁת אֲלָפִים חֲמֵשׁ מֵאוֹת וְשִׁשִּׁים וְחָמֵשׁ, וַיָּקָם הַגָּאוֹן וַיִּשָּׂא אֶת־רַגְלָיו לַעֲלוֹת לְאֶרֶץ יִשְׂרָאֵל׃ וַיִּסַּע בַּדֶּרֶךְ, וַיְהִי בְּהַגִּיעוֹ לְעִיר קֶנִיגְסְבֶּרְג בְּאַשְׁכְּנַז, וַיִּכְתֹּב אִגֶּרֶת לְבֵיתוֹ, וְכֹה כָּתַב: "וַאֲנִי תְּהִלָּה לָאֵל נוֹסֵעַ לְאֶרֶץ הַקְּדוֹשָׁה שֶׁהַכֹּל מְצַפִּים לִרְאוֹתָהּ"׃

אַךְ לֹא זָכָה הַגָּאוֹן מִוִּילְנָא לְהַשְׁלִים אֶת־מַסָּעוֹ, כִּי אִישׁ אֶחָד רָדַף אַחֲרָיו מִוִּילְנָא וּמָצָא אוֹתוֹ בַּדֶּרֶךְ, וַיִּתְחַנֵּן לְפָנָיו בְּכִי וּדְמָעוֹת שֶׁיָּשׁוּב לְעִירוֹ, כִּי אַנְשֵׁי הָעִיר לֹא יָכְלוּ לָשֵׂאת אֶת־פְּרֵדָתוֹ׃ וַיַּעַשׂ כֵּן הַגָּאוֹן, וַיָּשָׁב לְעִירוֹ, אַךְ נַפְשׁוֹ נוֹתְרָה קְשׁוּרָה בְּאֶרֶץ יִשְׂרָאֵל׃

וַיְצַו אֶת־תַּלְמִידָיו הַנֶּאֱמָנִים לַעֲלוֹת וּלְהִתְיַשֵּׁב בְּאֶרֶץ יִשְׂרָאֵל, וְלָשׂוּם יְסוֹדוֹת לַיִּשּׁוּב הַיְּהוּדִי בָּהּ׃ וַיְשַׁנֵּן בְּאָזְנֵיהֶם אֶת־דִּבְרֵי הַזֹּהַר וְהַמְּקֻבָּלִים עַל־אוֹדוֹת הַגְּאֻלָּה הָעֲתִידָה לָבוֹא, וְעַל־הַחוֹבָה לִפְעֹל בְּדֶרֶךְ הַטֶּבַע לְקֵרוּב הַגְּאֻלָּה׃

וַיְהִי אַחֲרֵי מוֹת הַגָּאוֹן בִּשְׁנַת תקנ"ח, וַיָּקָם דּוֹר חָדָשׁ מִתַּלְמִידָיו, אֲשֶׁר לֹא יָדְעוּ מְנוּחָה לְנַפְשָׁם עַד אֲשֶׁר יְקַיְּמוּ אֶת־צַוָּאַת רַבָּם הַגָּדוֹל וְהַקָּדוֹשׁ׃ וַיִּתְאַסְּפוּ הַתַּלְמִידִים וַיָּשִׂימוּ מַעֲשֶׂה, וַיָּקוּמוּ לַעֲלוֹת לְאֶרֶץ יִשְׂרָאֵל׃

וַיְהִי בִּשְׁנַת חֲמֵשֶׁת אֲלָפִים חֲמֵשׁ מֵאוֹת שִׁשִּׁים וְתֵשַׁע, הִיא שְׁנַת תקס"ט, הָיְתָה הָעֲלִיָּה הָרִאשׁוֹנָה שֶׁל תַּלְמִידֵי הַגָּאוֹן מִוִּילְנָא׃ וַיַּעַל רַבִּי מְנַחֵם מֶנְדֶל בֶּן־רַבִּי בַּרוּךְ מִשְּׁקְלוֹב, הוּא מַנְהִיג הַקְּבוּצָה הָרִאשׁוֹנָה, וְעִמּוֹ עוֹד כְּעֶשְׂרִים מִשְׁפָּחוֹת מִתַּלְמִידֵי הַגְר"א, וַיָּבוֹאוּ לְאֶרֶץ יִשְׂרָאֵל׃

וַיִּתְיַשְּׁבוּ הָעוֹלִים בָּעִיר צְפַת, אֲשֶׁר הָיְתָה אָז עִיר הַקֹּדֶשׁ וְהַחָכְמָה, מָקוֹם שֶׁחִבְּרוּ בּוֹ חַכְמֵי הָאֱמֶת, הַמְקֻבָּלִים, אֶת־לִמּוּדֵיהֶם הַגְּדוֹלִים׃ וַיָּחֵלּוּ לִבְנוֹת בָּהּ בָּתִּים וּבָתֵּי־מִדְרָשׁ, וַיַּעַבְדוּ אֶת־ה׳ בְּהַתְמָדָה וּבִקְדֻשָּׁה׃

וַיְהִי אַחֲרֵי הָעֲלִיָּה הָרִאשׁוֹנָה, וַיִּשְׁלַח רַבִּי מְנַחֵם מֶנְדֶל מִשְּׁקְלוֹב אִגְּרוֹת אֶל־אֶחָיו הַנִּשְׁאָרִים בַּגּוֹלָה, וַיִּקְרָא לָהֶם לַעֲלוֹת וּלְהִשְׁתַּקֵּעַ בָּאָרֶץ׃ וַיְשַׁלֵּחַ אֶת־תַּלְמִידוֹ רַבִּי הִלֵּל בֶּן־רַבִּי יַעֲקֹב רִבְלִין לְהָבִיא עוֹד עוֹלִים, וַיְהִי רַבִּי הִלֵּל לְשָׁלִיחַ הָעֲלִיָּה׃

וַיְהִי בִּשְׁנַת תקע"א, הִיא שְׁנַת חֲמֵשֶׁת אֲלָפִים חֲמֵשׁ מֵאוֹת שִׁבְעִים וְאַחַת, וַתְּהִי הָעֲלִיָּה הַשְּׁנִיָּה שֶׁל תַּלְמִידֵי הַגְּר"א׃ וַיַּעַל רַבִּי יִשְׂרָאֵל בֶּן־רַבִּי שְׁמוּאֵל מִשְּׁקְלוֹב וְעִמּוֹ קְהַל רַב שֶׁל תַּלְמִידִים וַחֲסִידִים, וַיָּבוֹאוּ לְצְפַת׃

וַיְהִי כַּאֲשֶׁר גָּדְלָה הָעִיר צְפַת וַתֵּצַר לָהֶם, וַיַּעֲלוּ חֵלֶק מִן הָעוֹלִים לִירוּשָׁלַיִם, הִיא עִיר הַקֹּדֶשׁ וְתִפְאֶרֶת יִשְׂרָאֵל׃ וַיִּתְיַשְּׁבוּ בָּהּ וַיִּבְנוּ בָּהּ בָּתִּים וּבָתֵּי־מִדְרָשׁ, וַיָּקִימוּ בֵית הַכְּנֶסֶת הַגָּדוֹל "בֵּית יַעֲקֹב", הוּא "חוּרְבַּת רַבִּי יְהוּדָה הֶחָסִיד"׃

וַיְהִי בִּשְׁנַת תקפ"ז, וַיָּבוֹא רַבִּי יְהוֹשֻׁעַ בֶּן־רַבִּי שְׁלֹמֹה זַלְמָן צַיְטְלִין וְעִמּוֹ קְבוּצָה גְּדוֹלָה מִתַּלְמִידֵי הַגְּר"א, וַיַּעֲלוּ יְרוּשָׁלַיְמָה׃ וַיִּבְנוּ בָּהּ בָּתִּים וַיְיַסְּדוּ מוֹסָדוֹת לַתּוֹרָה וּלְתַלְמוּדֵיהָ, וַיָּקִימוּ קְהִלָּה גְּדוֹלָה וּמְפֹאָרָה׃

וַיְהִי בְּעֵת הַהִיא, וְכָל־אֶרֶץ יִשְׂרָאֵל שׁוֹמֵמָה הָיְתָה, וְהַיִּשּׁוּב הַיְּהוּדִי בָּהּ מְעַט וְדַל, וְהַדְּרָכִים מְסֻכָּנוֹת, וּבָאָרֶץ מַחֲלוֹת רָעוֹת, וַתְּהִי נַפְשָׁם בְּכַפָּם׃ אַךְ הֵם בִּטְּלוּ רְצוֹנָם מִפְּנֵי רְצוֹן קוֹנָם, וַיֵּאָחֲזוּ בָּאָרֶץ וַיִּפְרוּ וַיִּרְבּוּ וַיִּמָּלְאוּ׃

וַיְהִי לְאַחַר יָמִים, וַיִּרְאוּ הָעוֹלִים כִּי הָעִיר יְרוּשָׁלַיִם צָרָה לָהֶם, כִּי הָיוּ תּוֹשָׁבֶיהָ רַבִּים וּדְחוּקִים בֵּין הַחוֹמוֹת׃ וַיַּחְשְׁבוּ מַחֲשָׁבוֹת לֵאמֹר: "נְבַקֵּשׁ לָנוּ מִחְיָה מְרֻוַּחַת מִחוּץ לַחוֹמוֹת, וְנִבְנֶה לָנוּ שְׁכוּנוֹת חֲדָשׁוֹת וּמְרֻוָּחוֹת"׃

וַיְהִי בִּשְׁנַת תרכ"ח, וַיָּקוּמוּ תַּלְמִידֵי הַגְּר"א וַיָּחֵלּוּ לִבְנוֹת אֶת־הַשְּׁכוּנָה הָרִאשׁוֹנָה מִחוּץ לְחוֹמוֹת הָעִיר, הִיא שְׁכוּנַת "מֵאָה שְׁעָרִים"׃ וַיְסַיְּעוּ בְּיָדָם מֹשֶׁה מוֹנְטִיפִיוֹרִי, הוּא הַשַּׂר הַיְּהוּדִי מִמֶּלֶךְ בְּרִיטַנְיָה, וְיְהוּדִים נְדִיבִים מִכָּל־הָעוֹלָם׃

וְאַחֲרֵי־כֵן בָּנוּ עוֹד שְׁכוּנוֹת רַבּוֹת, כֵּן יְמִין מֹשֶׁה, וְכֵן מִשְׁכְּנוֹת יִשְׂרָאֵל, וּבָתֵּי מַחֲסֶה, וּנְחָלַת שִׁבְעָה, וְעוֹד שְׁכוּנוֹת רַבּוֹת׃ וַיָּשִׂימוּ לִירוּשָׁלַיִם בְּגָדִים חֲדָשִׁים, וַיַּרְחִיבוּ אֶת־גְּבוּלֶיהָ, וַיַּעֲשׂוּ לָהּ פָּרְזוֹת וּמְגָדִּים׃

וְאֵלֶּה הָיוּ מַעֲשֵׂי רַבּוֹתֵינוּ תַּלְמִידֵי הַגְּר"א, אֲשֶׁר הָיוּ לְאָבוֹת הַתְּחִיָּה הָעִבְרִית בְּאֶרֶץ יִשְׂרָאֵל׃ הֵם הִנִּיחוּ יְסוֹדוֹת אֵיתָנִים לְיִשּׁוּב הָאָרֶץ, וְלָהֶם זְכוּת רִאשׁוֹנִים בַּפְּעֻלָּה לְקִבּוּץ גָּלֻיּוֹת וּלְקֵרוּב גְּאֻלָּתֵנוּ, כַּאֲשֶׁר חָזָה רַבָּם הַגָּדוֹל וְהַקָּדוֹשׁ, הַגָּאוֹן מִוִּילְנָא׃

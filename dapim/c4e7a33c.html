מְגִלַּת הַנִּסִּים

📜✨🇮🇱

מַעֲשֵׂה מִלְחֶמֶת יִשְׂרָאֵל עִם אִירָן

בִּשְׁנַת תשפ"ה לִבְרִיאַת הָעוֹלָם

י"ז - כ"ו בְּסִיוָן

2

📖 הַקְדָּמָה לִמְגִלַּת הַנִּסִּים

מַדּוּעַ חָשׁוּב לִקְרֹא אֶת מְגִלַּת הַנִּסִּים

בְּיָמִים שֶׁבָּהֶם הִתְרַגַּלְנוּ לְסַכָּנוֹת וּלְהַצָּלָה כְּאִלּוּ הָיוּ שִׁגְרָה, חוֹבָה עָלֵינוּ לַעֲצֹר, לְהָרִים עֵינַיִם וְלוֹמַר: "הַשֵּׁם עָשָׂה זֹאת!"

"וְזָכַרְתָּ אֶת ה' אֱלֹהֶיךָ, כִּי הוּא הַנֹּתֵן לְךָ כֹּחַ לַעֲשׂוֹת חָיִל"
(דְּבָרִים ח', י"ח)

הָאוֹר הַחַיִּים הַקָּדוֹשׁ: "פֵּרוּשׁ שֶׁצָּרִיךְ לָתֵת לִבּוֹ עַל טוֹבָתוֹ, כִּי מֵה' הָיְתָה לוֹ"

3

📜 יְסוֹד מִקְרָאִי לִקְבִיעַת יוֹם שִׂמְחָה

"וּבְיוֹם שִׂמְחַתְכֶם וּבְמוֹעֲדֵיכֶם וּבְרָאשֵׁי חׇדְשֵׁיכֶם – וּתְקַעְתֶּם בַּחֲצֹצְרוֹת"
(בַּמִּדְבָּר י', י')

דִּבְרֵי הָאֶבֶן עֶזְרָא:

"וּבְיוֹם שִׂמְחַתְכֶם – כְּשֶׁשַּׁבְתֶּם מֵאֶרֶץ אוֹיֵב, אוֹ נִצַּחְתֶּם הָאוֹיֵב הַבָּא עֲלֵיכֶם, וְקִבַּעְתֶּם יוֹם שִׂמְחָה כִּימֵי פוּרִים וְשִׁבְעַת יְמֵי חִזְקִיָּה."

זוֹ אֵינֶנָּה רַק דְּרָשָׁה – אֶלָּא פְּשַׁט הַפָּסוּק!

4
פֶּרֶק א' - וַיְהִי בִּימֵי הָרָשָׁע מֵאִירָן
👑⚔️

וַיְהִי בִּימֵי הָמָן הָרָשָׁע מֵאִירָן, אִישׁ צוֹרֵר וְאוֹיֵב לְעַם יִשְׂרָאֵל, אֲשֶׁר בִּקֵּשׁ לְהַשְׁמִיד וְלַהֲרֹג וּלְאַבֵּד אֶת כׇּל הַיְּהוּדִים בְּאֶרֶץ יִשְׂרָאֵל.

וְהָמָן הַזֶּה חֲמֵינִי שְׁמוֹ, וְנָשָׂא עַצְמוֹ לְמֶלֶךְ עַל מְדִינַת פָּרַס הַגְּדוֹלָה, וְהֵקִים תַּחְתָּיו שָׂרֵי צָבָא רְשָׁעִים: חוּסֵין סְלָאמִי מְפַקֵּד הַמִּשְׁמָרוֹת, וּמֻחַמַּד בַּגְרִי רֹאשׁ הַמַּטֶּה הַכְּלָלִי.

וַיִּיעַץ הָמָן עִם יוֹעֲצָיו לַעֲשׂוֹת נֶשֶׁק אָיֹם לְכַלּוֹת אֶת עַם יִשְׂרָאֵל, וַיָּכֶן אֶלֶף טִילִים וְאַלְפֵי כטב"מים רָעִים.

"וּבְכׇל זֹאת לֹא יָדַע הָמָן הָרָשָׁע כִּי יַד ה' תִּהְיֶה נְטוּיָה לְהַצִּיל אֶת עַמּוֹ יִשְׂרָאֵל"
5
פֶּרֶק ב' - וַתְּהִי אֶסְתֵּר לַמַּלְכוּת בְּיִשְׂרָאֵל
👑🇮🇱

וּבַיָּמִים הָהֵם מָלַךְ עַל יִשְׂרָאֵל נְתַנְיָהוּ הַמֶּלֶךְ, וּבִנְיָמִין זֶה שְׁמוֹ, וַיְהִי לוֹ יוֹעֵץ חָכָם שְׁמוֹ יִשְׂרָאֵל כַּץ וְשַׂר צָבָא נֶאֱמָן שְׁמוֹ אֵיָּל זָמִיר.

וְהַמֶּלֶךְ נְתַנְיָהוּ הָיָה אִישׁ חָכָם וּבַעַל מַדָּע, יוֹדֵעַ לְדַבֵּר לִפְנֵי מַלְכֵי הָאָרֶץ.

וַיְהִי כַּאֲשֶׁר הִגִּיעוּ שְׁמוּעוֹת רָעוֹת מֵאִירָן, וַיִּקְרָא הַמֶּלֶךְ לְכׇל שָׂרָיו וַעֲבָדָיו: "הֵן הָמָן מִפָּרַס מְבַקֵּשׁ אֶת רָעָתֵנוּ!"

וַיֶּחֱרַד הַמֶּלֶךְ חֲרָדָה גְדוֹלָה, וַיֵּצֵא לַכּוֹתֶל הַמַּעֲרָבִי בִּירוּשָׁלַיִם לְהִתְפַּלֵּל.

"אַל תִּירָא, כִּי אָנֹכִי עִמָּךְ, לָקוּם וּלְהַכּוֹת אֶת אוֹיְבֶיךָ. זֶה הַזְּמַן וְהָעֵת לִגְבוּרָה."
6
פֶּרֶק ג' - וַיֵּצֵא הַפּוּר עַל הַיּוֹם הָרָע
🌙⚡

וַיְהִי בְּלֵיל י"ז בְּחֹדֶשׁ סִיוָן, בְּשָׁעָה שָׁלֹשׁ לִפְנוֹת בֹּקֶר, וַיֵּצֵא פוּר רַע עַל אִירָן.

וּבַלַּיְלָה הַהוּא לֹא יָשַׁן הַמֶּלֶךְ נְתַנְיָהוּ: "הַיּוֹם נָתַן ה' אֶת אוֹיְבֵינוּ בְּיָדֵנוּ!"

וַיֵּלְכוּ שָׂרֵי הַצָּבָא בְּמָטוֹסִים רַבִּים - מָאתַיִם מְטוֹסֵי מִלְחָמָה, וַיָּטוּסוּ כִּנְשָׁרִים בַּשָּׁמַיִם אֶל אֶרֶץ פָּרַס הָרְחוֹקָה.

"וְלֹא רָאָה אוֹתָם אִישׁ וְלֹא שָׁמַע אָדָם, כִּי מַלְאֲכֵי ה' הִסְתִּירוּ אוֹתָם בֶּעֱנָנִים"

וּבְעֶצֶם הַלַּיְלָה הַהוּא יָשְׁבוּ הָמָן וְכׇל שָׂרָיו הָרָעִים בִּטְהֶרָן, חוֹשְׁבִים כִּי בְטוּחִים הֵם. וְלֹא יָדְעוּ כִּי מָוֶת נוֹקֵף עַל דַּלְתוֹתֵיהֶם!

7
פֶּרֶק ד' - וַתִּפֹּל פֻּרְעָנוּת גְּדוֹלָה עַל אִירָן
⚡💥

וַיָּמׇת בְּאוֹתוֹ הַלַּיְלָה חוּסֵין סְלָאמִי מְפַקֵּד הַמִּשְׁמָרוֹת, וְגַם מֻחַמַּד בַּגְרִי רֹאשׁ הַמַּטֶּה הַכְּלָלִי מֵת. וְעִמָּהֶם מֵתוּ עֶשְׂרִים שָׂרֵי צָבָא נוֹסָפִים, וְתִשְׁעָה חַכְמֵי הָרַע עֹשֵׂי נֶשֶׁק הַכְּלָיָה הַגַּרְעִינִי.

וַיְהִי בֹקֶר וַתִּמְצָא אִירָן כִּי נִכְרְתָה רָאשֶׁיהָ וּגְדוֹלֶיהָ, וַיִּתְמְהוּ כׇל הָעָם: "אֵיךְ יָכוֹל לִהְיוֹת הַדָּבָר הַזֶּה?"

וַיְבַקֵּשׁ הָמָן הָרָשָׁע לִשְׁגֹּר אֶלֶף טִילִים רָעִים, אַךְ וְהִנֵּה נֵס גָּדוֹל - לֹא יָצְאוּ כִּי אִם מָאתַיִם בִּלְבַד!

"וַיִּירְטוּ מַלְאֲכֵי ה' תִּשְׁעִים וַחֲמִשָּׁה מִתּוֹךְ כׇּל מֵאָה טִילִים"

וַיִּזְעַק הָמָן בְּקוֹל מַר: "אֵיךְ נֶהֶפְכוּ עָלֵינוּ הַמַּזָּלוֹת!"

8
פֶּרֶק ה' - וַתָּבוֹא יְשׁוּעָה לְעַם יִשְׂרָאֵל
👶🏥✨

וּבָעֵת הַהִיא נִסִּים רַבִּים נַעֲשׂוּ לְעַם יִשְׂרָאֵל:

  • תִּינוֹק בֶּן שִׁשָּׁה יָמִים נִמְצָא חַי וְשָׁלֵם מִתַּחַת לֶהֱרִיסוֹת
  • בָתֵּי חוֹלִים הִתְרוֹקְנוּ דַּקּוֹת לִפְנֵי שֶׁהַטִּילִים נָפְלוּ
  • כׇל עוֹבְרֵי אֹרַח נִצְּלוּ בְּאֹפֶן פִּלְאִי

וַיְהִי בְּאוֹתָם הַיָּמִים, וְגַם מֶלֶךְ אַמֶּרִיקָה הַגְּדוֹלָה טְרַמְפּ נָתַן אֶת יָדוֹ לַעֲזֹר: "יִשְׂרָאֵל הִיא חַבֶּרְתֵּנוּ הָאֲמִינָה!"

וַיִּקַּח שִׁשָּׁה מְטוֹסֵי B-2 הַגְּדוֹלִים, וַיַּכֶּה אֶת מְעָרוֹת הָרֶשַׁע - אֶת פּוֹרְדוֹ וְאֶת נַתַּנְז וְאֶת אִיסְפַּהָן.

"וַתִּהְיֶינָה שְׁמָמָה וְחֳרָבוֹת עוֹלָמִים"

וַיֹּאמֶר טְרַמְפּ בְּקוֹל גָּדוֹל: "הוּשְׁמְדוּ מַתְקְנֵי הַגַּרְעִין שֶׁל אִירָן לְחַלּוּטִין!"

9
פֶּרֶק ו' - וַיַּכִּירוּ הָעַמִּים בְּיַד ה'
🌍🙏

וַיִּרְאוּ כׇל עַמֵּי הָאָרֶץ אֶת הַנִּסִּים הַגְּדוֹלִים, וַיִּהְיוּ נִדְהָמִים לֵאמֹר: "אָכֵן אֱלֹהֵי יִשְׂרָאֵל הוּא הָאֱלֹהִים!"

וְאַף שֶׁעָלָה הָמָן עַל דַּעְתּוֹ לִסְגֹּר אֶת מִיצְרֵי הַהוֹרְמוּז לְהַפְסִיק נֶפֶט הָעוֹלָם, לֹא הִצְלִיח בִּזְדוֹנוֹ.

וַתּוּקַמְנָה מַעַרְכוֹת הַהֲגָנָה שֶׁל יִשְׂרָאֵל בְּמֻפְלָא, וְכׇל כטב"ם רַע וּשְׁלִיחַ מָוֶת נִתְפַּס בִּרְשָׁתָן.

מַלְכֵי אֵירוֹפָּה קָרְאוּ לְשָׁלוֹם, וְגַם בְּרוּסִיָא וּבְסִין אָמְרוּ: "דַּי לַמִּלְחָמָה!"

"בָּרוּךְ אַתָּה ה' אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, שֶׁעָשָׂה לָנוּ נִסִּים בַּמָּקוֹם הַזֶּה"
10
פֶּרֶק ז' - וַתָּבוֹא קְלָלָה עַל אִירָן
💔⚰️

וַיְהִי מִקֵּץ שְׁנֵים עָשָׂר יוֹם, וְלֹא הָיָה עוֹד מַעֲמָד בְּרוּחַ אִירָן הָרְשָׁעָה.

  • נֶהֶרְסוּ מַתְקְנֵי הָרֶשַׁע
  • מֵתוּ כׇל שָׂרֵי הַצָּבָא הַגְּדוֹלִים
  • וְאֵין מִי שֶׁיּוּכַל לְנַהֵל אֶת הַמִּלְחָמָה

וַיִּבְכּוּ נְשֵׁי אִירָן עַל בְּנֵיהֶן הַנּוֹפְלִים: "לָמָה יָצָאנוּ לְמִלְחָמָה נֶגֶד עַם ה' הַנִּצְחִי?"

וַיִּשָּׁבַע הָמָן הָרָשָׁע חֲמֵינִי שְׁבוּעָה שָׁוְא: "עוֹד אֶלָּחֵם עִם יִשְׂרָאֵל!" אֲבָל לֹא הָיְתָה לוֹ עוֹד יְכֹלֶת.

"זֹאת הַדֶּרֶךְ לְכׇל מִי שֶׁמֵּרִים יָד בֵּאלֹהֵי יִשְׂרָאֵל וּבְעַמּוֹ"
11
פֶּרֶק ח' - וַתִּגְדַּל שִׂמְחַת יִשְׂרָאֵל
🎉🇮🇱

וּבְכׇל מְדִינַת יִשְׂרָאֵל הָיְתָה שִׂמְחָה וְשָׂשׂוֹן - שִׂמְחַת נִסִּים וְשִׂמְחַת נְקָמָה מֵהָאוֹיְבִים!

וַיֵּלְכוּ בְּנֵי יִשְׂרָאֵל לְבָתֵּי כְנֵסִיּוֹת לְהוֹדוֹת וּלְהַלֵּל.

"הֲיָדְעוּ נָא כׇל עַמֵּי הָאָרֶץ כִּי ה' אֱלֹהֵי יִשְׂרָאֵל הוּא הָאֱלֹהִים בַּשָּׁמַיִם מִמַּעַל וְעַל הָאָרֶץ מִתָּחַת"

וַיִּתֵּן הוֹדָיָה לַה' עַל שֶׁלֹּא נָפַל אַף חַיָּל יִשְׂרְאֵלִי בְּאֶרֶץ הָאוֹיֵב, וְכׇל הַטַּיָּסִים חָזְרוּ לְבָתֵּיהֶם בְּשָׁלוֹם.

וַיֹּאמֶר הַמֶּלֶךְ נְתַנְיָהוּ: "רְאוּ נָא כֵּיצַד הַסִּיַּעְתָּא דִּשְׁמַיָּא הִיא הָיְתָה הַסִּיעָה הַגְּדוֹלָה בְּיוֹתֵר בַּקּוֹאָלִיצְיָה שֶׁלָּנוּ!"

12
פֶּרֶק ט' - וַיִּהְיוּ הַיָּמִים הָאֵלֶּה לְזִכָּרוֹן
📅📜

וּמֵאוֹתוֹ הַיּוֹם וָאֵילָךְ יִקְרְאוּ הַיָּמִים הָאֵלֶּה "יְמֵי הַנִּסִּים" - לְזֵכֶר הַנִּפְלָאוֹת שֶׁעָשָׂה ה' לְעַמּוֹ יִשְׂרָאֵל.

מִיּוֹם י"ז בְּסִיוָן וְעַד כ"ו בְּסִיוָן
עֲשָׂרָה יְמֵי נִסִּים וּגְבוּרָה

וַיִּכְתֹּב הַמֶּלֶךְ נְתַנְיָהוּ בְּסֵפֶר דִּבְרֵי הַיָּמִים אֶת כׇּל אֲשֶׁר קָרָה, לְעֵדוּת לַדּוֹרוֹת הָעֲתִידִים כִּי ה' לוֹחֵם לְיִשְׂרָאֵל.

וַיִּתְקְנוּ חַכְמֵי יִשְׂרָאֵל לִקְרֹא בְּכׇל שָׁנָה וְשָׁנָה מְגִלָּה זוֹ, לְהַזְכִּיר אֶת גְּדֻלַּת הָאֵל וְאֶת נִפְלְאוֹתָיו.

"בְּכׇל דּוֹר וָדוֹר עוֹמְדִים עָלֵינוּ לְכַלּוֹתֵנוּ, וְהַקָּדוֹשׁ בָּרוּךְ הוּא מַצִּילֵנוּ מִיָּדָם"
13
פֶּרֶק י' - בִּרְכַּת הַגּוֹאֵל
🙏✨

וְעַתָּה, בְּנֵי יִשְׂרָאֵל, זִכְרוּ נָא אֶת הַיָּמִים הָאֵלֶּה וְאַל תִּשְׁכְּחוּ!

זִכְרוּ כִּי בְּעֵת שֶׁכׇל אֻמּוֹת הָעוֹלָם אָמְרוּ שֶׁאֵין תִּקְוָה לְיִשְׂרָאֵל מוּל אִירָן הַגְּדוֹלָה, ה' אֱלֹהֵינוּ הֶרְאָה שֶׁהוּא יָחִיד וּמְיֻחָד בְּעוֹלָמוֹ.

  • זִכְרוּ כִּי אֵין לְהַבְטִיחַ בְּחַיִל וּבְכֹחַ
  • אֶלָּא בְּרוּחַ ה' צְבָאוֹת
  • עַם יִשְׂרָאֵל חַי, וֵאלֹהֵי יִשְׂרָאֵל חַי לְעוֹלָם וָעֶד
"שְׁמַע יִשְׂרָאֵל, ה' אֱלֹהֵינוּ, ה' אֶחָד!"

וַיֹּאמֶר כׇל הָעָם: "אָמֵן, כֵּן יְהִי רָצוֹן!"

14

בִּרְכַּת הַסִּיּוּם

🌟🙏📜
בָּרוּךְ הַגּוֹאֵל אֶת עַמּוֹ יִשְׂרָאֵל מִכׇּל צָרוֹתֵיהֶם

זֹאת מְגִלַּת הַנִּסִּים שֶׁקָּרוּ לְעַם יִשְׂרָאֵל

בִּשְׁנַת תשפ"ה לִבְרִיאַת הָעוֹלָם

בִּימֵי הַמֶּלֶךְ נְתַנְיָהוּ בֶּן בֶּנְצִי

"וְעַל כֵּן יְמֵי הַנִּסִּים הָאֵלֶּה לֹא יַעַבְרוּ מִתּוֹךְ הַיְּהוּדִים, וְזִכְרָם לֹא יָסוּף מִזַּרְעָם"

נִשְׁלְמָה מְגִלַּת הַנִּסִּים בְּשָׁלוֹם

15

נֶעֱרַךְ עַל יְדֵי

יְהוּדָה טָאוּב

עִם אַמְצָעֵי הַבִּינָה הַמְּלָאכוּתִית בְּעֶזְרַת הָאָדָם

לְזֵכֶר וּלְתִהִלָּה שֶׁל הַנִּסִּים הַגְּדוֹלִים

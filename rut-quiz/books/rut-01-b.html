רָעָב בְּבֵית לֶחֶם, אֱלִימֶלֶךְ יוֹצֵא מֵהָאָרֶץ, וְנָעֳמִי חוֹזֶרֶת כְּשֶׁה׳ ׳פּוֹקֵד׳ אֶת עַמּוֹ – מִבְחָן שֶׁל נְדִיבוּת וְשֶׁל בִּטָּחוֹן

📜 הַפָּסוּקוַיְהִי בִּימֵי שְׁפֹט הַשֹּׁפְטִים וַיְהִי רָעָב בָּאָרֶץ

💛 אַהֲבַת הַשֵּׁם: ה׳ מֵנִיעַ הִיסְטוֹרְיָה גְדוֹלָה דֶּרֶךְ אֲנָשִׁים פְּשׁוּטִים 💛

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

⚖️ ׳בִּימֵי שְׁפֹט הַשֹּׁפְטִים׳

רַשִׁ״י: ׳לִפְנֵי מְלֹךְ מֶלֶךְ שָׁאוּל, שֶׁהָיוּ הַדּוֹרוֹת מִתְפַּרְנְסִים עַל יְדֵי שׁוֹפְטִים׳. ׳וּבִימֵי אִבְצָן הָיָה... אִבְצָן זֶה בֹּעַז׳ – בֹּעַז הוּא הַשּׁוֹפֵט בְּאוֹתוֹ זְמַן.

רַשִׁ״י: מְגִלַּת רוּת קָרְתָה בִּימֵי הַשּׁוֹפְטִים, וּבֹעַז הוּא הַשּׁוֹפֵט הַנִּקְרָא אִבְצָן.

1

📜 הַפָּסוּק…וַיֵּלֶךְ אִישׁ מִבֵּית לֶחֶם יְהוּדָה לָגוּר בִּשְׂדֵי מוֹאָב

💛 אַהֲבַת הַשֵּׁם: ׳עַיִן טוֹבָה׳ הִיא לִרְצוֹת לָתֵת וְלִשְׂמֹחַ בָּזֶה 💛

📖 רוּת, פֶּרֶק א׳, פָּסוּק א׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🧱 ׳צָרוּת הָעַיִן׳

רַשִׁ״י מְבָאֵר מָה הָיָה חֶטְאוֹ שֶׁל אֱלִימֶלֶךְ: ׳עָשִׁיר גָּדוֹל הָיָה וּפַרְנַס הַדּוֹר. וְיָצָא מֵאֶרֶץ יִשְׂרָאֵל לְחוּץ לָאָרֶץ מִפְּנֵי צָרוּת הָעָיִן, שֶׁהָיְתָה עֵינוֹ צָרָה בָעֲנִיִּים הַבָּאִים לְדָחְקוֹ׳. ׳צָרוּת הָעַיִן׳ הִיא לֹא לִרְצוֹת לָתֵת. אֱלִימֶלֶךְ יָכֹל הָיָה לַעֲזֹר, אֲבָל בָּחַר לְהִסְתַּלֵּק.

רַשִׁ״י: אֱלִימֶלֶךְ עָזַב אֶת הָאָרֶץ מִפְּנֵי שֶׁלֹּא רָצָה לָתֵת לָעֲנִיִּים.

2

📜 הַפָּסוּק…אֶפְרָתִים מִבֵּית לֶחֶם יְהוּדָה

💛 אַהֲבַת הַשֵּׁם: מַה שֶּׁחָשׁוּב בֶּאֱמֶת הוּא הַבְּחִירָה בְּה׳, לֹא הַמַּעֲמָד 💛

📖 רוּת, פֶּרֶק א׳, פָּסוּק ב׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

👑 ׳אֶפְרָתִים׳ – חֲשׁוּבִים

רַשִׁ״י מְבָאֵר ׳אֶפְרָתִים׳ – ׳חֲשׁוּבִים׳, וּמֵבִיא רְאָיָה: ׳הִשִּׂיא עֶגְלוֹן מֶלֶךְ מוֹאָב אֶת בִּתּוֹ לְמַחְלוֹן, דְּאָמַר מַר, רוּת בִּתּוֹ שֶׁל עֶגְלוֹן הָיְתָה׳. רוּת הָיְתָה בַּת מֶלֶךְ מוֹאָב! וְהִיא בְּחָרָה לְהִצְטָרֵף לְעַם יִשְׂרָאֵל בְּעֹנִי, מִתּוֹךְ אֱמוּנָה.

רַשִׁ״י: רוּת הָיְתָה בַּת מֶלֶךְ מוֹאָב, אֲבָל בָּחֲרָה בַּדֶּרֶךְ שֶׁל עַם יִשְׂרָאֵל.

3

📜 הַפָּסוּקוַיָּמוּתוּ גַם־שְׁנֵיהֶם מַחְלוֹן וְכִלְיוֹן וַתִּשָּׁאֵר הָאִשָּׁה מִשְּׁנֵי יְלָדֶיהָ וּמֵאִישָׁהּ

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי מַרְגִּישׁ אַזְהָרָה קַלָּה, אֲנִי מְתַקֵּן מִיָּד 💛

📖 רוּת, פֶּרֶק א׳, פָּסוּק ה׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

📉 ׳גַּם שְׁנֵיהֶם׳

רַשִׁ״י שָׂם לֵב לַמִּלָּה ׳גַּם׳: ׳בַּתְּחִלָּה לָקוּ בְמָמוֹנָם וּמֵתוּ גְמַלֵּיהֶם וּמִקְנֵיהֶם, וְאַחַר כָּךְ מֵתוּ גַם הֵם׳. הַמְּסָרִים מֵה׳ מַתְחִילִים קַלִּים וְגָדֵלִים אִם לֹא שׁוֹמְעִים. הַפָּסוּק מְסַכֵּם: נָעֳמִי נִשְׁאֲרָה ׳מִשְּׁנֵי יְלָדֶיהָ וּמֵאִישָׁהּ׳.

רַשִׁ״י: הַתְּחִלָּה הָיְתָה פְּגִיעָה בַּמָּמוֹן, וְרַק אַחַר כָּךְ הַדְּבָרִים הַקָּשִׁים יוֹתֵר.

4

📜 הַפָּסוּק…כִּי שָׁמְעָה בִּשְׂדֵה מוֹאָב כִּי־פָקַד ה׳ אֶת־עַמּוֹ לָתֵת לָהֶם לָחֶם

💛 אַהֲבַת הַשֵּׁם: ה׳ זוֹכֵר אוֹתִי וְנוֹתֵן לִי אֶת הַלֶּחֶם שֶׁאֲנִי צָרִיךְ 💛

📖 רוּת, פֶּרֶק א׳, פָּסוּק ו׳

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🍞 ׳פָּקַד ה׳ אֶת עַמּוֹ׳

הַמִּלָּה ׳פָּקַד׳ פֵּרוּשָׁהּ שֶׁה׳ שָׂם לֵב וְזָכַר אֶת עַמּוֹ. ה׳ לֹא שָׁכַח – הוּא נָתַן ׳לָהֶם לָחֶם׳. וְהַפָּסוּק מְסַפֵּר שֶׁנָּעֳמִי ׳שָׁמְעָה בִּשְׂדֵה מוֹאָב׳, כְּלוֹמַר הַחֲדָשׁוֹת הַטּוֹבוֹת מֵאֶרֶץ יִשְׂרָאֵל הִגִּיעוּ אֲפִלּוּ לְמוֹאָב. כְּשֶׁיֵּשׁ בְּרָכָה בַּבַּיִת, הַשֵּׁם שֶׁלָּהּ יוֹצֵא לָרָחוֹק.

הַמִּלָּה ׳פָּקַד׳ מַרְאָה שֶׁה׳ זוֹכֵר אֶת עַמּוֹ וְנוֹתֵן לָהֶם לֶחֶם.

5

📜 הַפָּסוּקוַתֵּצֵא מִן־הַמָּקוֹם אֲשֶׁר הָיְתָה־שָׁמָּה וּשְׁתֵּי

💛 אַהֲבַת הַשֵּׁם: בְּעִיר שֶׁאֲנִי בָּהּ, אֲנִי רוֹצֶה לִהְיוֹת מִי שֶׁיַּרְגִּישׁוּ אֶת הַחֶסְרוֹן שֶׁלּוֹ 💛

📖 רוּת, פֶּרֶק א׳, פָּסוּק ז׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

✨ ׳יְצִיאַת צַדִּיק׳

רַשִׁ״י: ׳וַתֵּצֵא מִן הַמָּקוֹם – לָמָּה נֶאֱמַר? הֲרֵי כְבָר נֶאֱמַר ״וַתָּשָׁב מִשְּׂדֵי מוֹאָב״... אֶלָּא מַגִּיד שֶׁיְּצִיאַת צַדִּיק מִן הַמָּקוֹם נִכֶּרֶת וְעוֹשָׂה רֹשֶׁם׳. הוּא מְדַמֶּה לְיַעֲקֹב: ׳וְכֵן ״וַיֵּצֵא יַעֲקֹב מִבְּאֵר שָׁבַע״׳. הָעִיר הַמּוֹאֲבִית הִרְגִּישָׁה אֶת הָעֲזִיבָה שֶׁל נָעֳמִי וְרוּת.

רַשִׁ״י: יְצִיאַת צַדִּיק מֵעִיר ׳נִכֶּרֶת׳ – הָעִיר מַרְגִּישָׁה שֶׁחָסֵר בָּהּ הַזִּיו וְהַהוֹד.

6

📜 הַפָּסוּק…וְהֵמָּה בָּאוּ בֵּית לֶחֶם בִּתְחִלַּת קְצִיר שְׂעֹרִים

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי חוֹזֵר אֶל ה׳, הַזְּמַן עֲדַיִן בְּתִחְלַת הַקָּצִיר 💛

📖 רוּת, פֶּרֶק א׳, פָּסוּק כ״ב (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🌅 ׳בִּתְחִלַּת קְצִיר שְׂעֹרִים׳

רַשִׁ״י מְבָאֵר אֶת זְמַן הַחֲזָרָה: ׳בִּקְצִירַת הָעֹמֶר הַכָּתוּב מְדַבֵּר׳. בְּחַג הַפֶּסַח מַקְרִיבִים אֶת קָרְבַּן הָעֹמֶר מִן הַשְּׂעוֹרִים הֶחֳדָשׁוֹת, וְאָז מַתְחִיל סֵפֶר הָעֹמֶר שֶׁמּוֹבִיל לְשָׁבוּעוֹת. נָעֳמִי וְרוּת הִגִּיעוּ בְּדִיּוּק בִּתְחִלַּת הַדֶּרֶךְ מִפֶּסַח לְשָׁבוּעוֹת, מִן הַגְּאֻלָּה אֶל מַתַּן תּוֹרָה.

רַשִׁ״י: נָעֳמִי וְרוּת הִגִּיעוּ בִּזְמַן קְצִירַת הָעֹמֶר, תְּחִלַּת הַדֶּרֶךְ מִפֶּסַח לְשָׁבוּעוֹת.

7
💡

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

מָה לוֹקְחִים מִרָמָה ב׳?

רַשִׁ״י מְבָאֵר שֶׁאֱלִימֶלֶךְ הָיָה פַּרְנַס הַדּוֹר וְיָצָא מִצָּרוּת עַיִן, וְשֶׁרוּת הָיְתָה בַּת מֶלֶךְ מוֹאָב; וְשֶׁהֵם בָּאוּ בִּקְצִירַת הָעֹמֶר.

…כִּי שָׁמְעָה בִּשְׂדֵה מוֹאָב כִּי־פָקַד ה׳ אֶת־עַמּוֹ לָתֵת לָהֶם לָחֶם

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • אֱלִימֶלֶךְ הָיָה עָשִׁיר וְיָצָא מֵהָאָרֶץ כְּדֵי לֹא לַעֲזֹר לַעֲנִיִּים. מָתַי בְּפַעַם הָאַחֲרוֹנָה עָזַרְתִּי לְמִישֶׁהוּ בְּלִי שֶׁיְּבַקֵּשׁ?
  • ׳יְצִיאַת צַדִּיק מִן הַמָּקוֹם נִכֶּרֶת׳. מָה יַרְגִּישׁוּ בַּכִּתָּה אוֹ בַּבַּיִת כְּשֶׁאֲנִי לֹא נִמְצָא?
  • ה׳ ׳פָּקַד׳ אֶת עַמּוֹ וְנָתַן לֶחֶם. עַל אֵיזֶה לֶחֶם שֶׁה׳ נוֹתֵן לִי אֲנִי רוֹצֶה לְהוֹדוֹת הַיּוֹם?

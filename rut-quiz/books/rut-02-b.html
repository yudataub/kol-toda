עַמֵּךְ עַמִּי וֵאלֹהַיִךְ אֱלֹהָי

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

רוּת בּוֹחֶרֶת בְּעַם יִשְׂרָאֵל וּבְה׳, נָעֳמִי מוֹדִיעָה לָהּ מַה כָּרוּךְ בַּדְּבָר, וְהַגֵּרוּת מִתְקַבֶּלֶת בְּאַהֲבָה

📖 פֶּרֶק א׳

🎯 לַחִידוֹן

📜 הַפָּסוּקיִתֵּן ה׳ לָכֶם וּמְצֶאןָ מְנוּחָה אִשָּׁה

💛 אַהֲבַת הַשֵּׁם: חֶסֶד אֲמִתִּי הוּא כְּשֶׁאֲנִי עוֹשֶׂה אוֹתוֹ בְּלִי לְצַפּוֹת לִתְמוּרָה 💛

📖 רוּת, פֶּרֶק א׳, פָּסוּק ט׳

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🏡 ׳וּמְצֶאןָ מְנוּחָה׳

נָעֳמִי מְבָרֶכֶת: ׳יִתֵּן ה׳ לָכֶם וּמְצֶאןָ מְנוּחָה אִשָּׁה בֵּית אִישָׁהּ׳. ׳מְנוּחָה׳ אֵינָהּ רַק מָקוֹם לָגוּר – הִיא תְּחוּשַׁת בֵּיתִיּוּת. וְהַבְּרָכָה הָרִאשׁוֹנָה: ׳יַעַשׂ ה׳ עִמָּכֶם חֶסֶד כַּאֲשֶׁר עֲשִׂיתֶם עִם הַמֵּתִים וְעִמָּדִי׳ – חֶסֶד שֶׁל אֱמֶת, כִּי הוּא נַעֲשֶׂה לְמִי שֶׁלֹּא יָכוֹל לְהַחֲזִיר.

נָעֳמִי מְבָרֶכֶת בְּחֶסֶד וּבִמְנוּחָה, וְקוֹרֵאת לַחֶסֶד שֶׁעָשׂוּ ׳עִם הַמֵּתִים׳ – חֶסֶד שֶׁאֵין מְצַפִּים לְתַגְמוּל עָלָיו.

1

📜 הַפָּסוּקהֲלָהֵן תְּשַׂבֵּרְנָה עַד אֲשֶׁר יִגְדָּלוּ הֲלָהֵן

💛 אַהֲבַת הַשֵּׁם: הַכֵּנוּת שֶׁל נָעֳמִי הִיא אַהֲבָה שֶׁלֹּא רוֹצָה לְהַעֲמִיס 💛

📖 רוּת, פֶּרֶק א׳, פָּסוּק י״ג (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

⏳ ׳הֲלָהֵן תְּשַׂבֵּרְנָה׳

נָעֳמִי שׁוֹאֶלֶת: ׳הֲלָהֵן תְּשַׂבֵּרְנָה עַד אֲשֶׁר יִגְדָּלוּ׳? רַשִׁ״י: ׳בִּתְמִיהַּ, שֶׁמָּא לָהֶם תְּצַפֶּינָה עַד אֲשֶׁר יִגְדָּלוּ? לְשׁוֹן ״שִׂבְרוֹ עַל ה׳ אֱלֹהָיו״׳. ׳תֵּעָגֵנָה׳ – ׳לְשׁוֹן אָסוּר וְכָלוּא׳. נָעֳמִי אוֹמֶרֶת: אַתֶּן צְעִירוֹת; אַל תִּשְׁבּוּ סְגוּרוֹת בִּשְׁבִיל מַשֶּׁהוּ שֶׁאֵין לִי לְהַבְטִיחַ.

רַשִׁ״י: נָעֳמִי שׁוֹאֶלֶת בִּתְמִיהָה אִם הַכַּלּוֹת יְחַכּוּ שָׁנִים, וְאוֹמֶרֶת שֶׁאֵין לָהּ מָה לְהַבְטִיחַ.

2

📜 הַפָּסוּק…וְרוּת דָּבְקָה בָּהּ

💛 אַהֲבַת הַשֵּׁם: אֲנִי בוֹחֵר ׳לִדְבֹּק׳ בִּטּוֹב, גַּם כְּשֶׁאֲחֵרִים הוֹלְכִים לְמָקוֹם אַחֵר 💛

📖 רוּת, פֶּרֶק א׳, פָּסוּק י״ד

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🫂 ׳וְרוּת דָּבְקָה בָּהּ׳

הַמְּגִלָּה מַצִּיגָה זוֹ מוּל זוֹ שְׁתֵּי כַלּוֹת: ׳וַתִּשַּׁק עָרְפָּה לַחֲמוֹתָהּ וְרוּת דָּבְקָה בָּהּ׳. עָרְפָּה עָשְׂתָה אֶת הַנָּכוֹן לְפִי מָה שֶׁהָיְתָה צְפוּיָה לַעֲשׂוֹת; אֲבָל רוּת בָּחֲרָה בְּדֶרֶךְ אַחֶרֶת. ׳דָּבְקָה׳ הוּא כְּמוֹ ׳דָּבוּק׳ – לֹא שְׁנֵי אֲנָשִׁים לְצַד זֶה, אֶלָּא קֶשֶׁר שֶׁל לֵב.

הַמִּלָּה ׳דָּבְקָה׳ מְתָאֶרֶת קֶשֶׁר שֶׁל לֵב, שֶׁרוּת בָּחֲרָה בּוֹ.

3

📜 הַפָּסוּק…הִנֵּה שָׁבָה יְבִמְתֵּךְ אֶל־עַמָּהּ וְאֶל־אֱלֹהֶיהָ

💛 אַהֲבַת הַשֵּׁם: בְּחִירַת הָעַם הִיא גַּם בְּחִירַת הָאֱלֹהִים – אֲנִי בּוֹחֵר בְּה׳ 💛

📖 רוּת, פֶּרֶק א׳, פָּסוּק ט״ו (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

⛪ ׳אֶל עַמָּהּ וְאֶל אֱלֹהֶיהָ׳

נָעֳמִי אוֹמֶרֶת לְרוּת: ׳הִנֵּה שָׁבָה יְבִמְתֵּךְ אֶל עַמָּהּ וְאֶל אֱלֹהֶיהָ׳. רַשִׁ״י מַעֲרִיךְ אֶת הַטַּעַם: הַ׳שִּׁי״ן שֶׁל ׳שָׁבָה׳ עִם טַעַם לְמַעְלָה כִּי הוּא ׳לְשׁוֹן עָבַר׳, אֲבָל ׳וּבַבֹּקֶר הִיא שָׁבָה׳ עִם טַעַם לְמַטָּה כִּי הוּא ׳לְשׁוֹן הֹוֶה׳. שִׂימוּ לֵב לְדִקְדּוּק: עָרְפָּה כְּבָר שָׁבָה. וְנָעֳמִי מַצִּיבָה אֶת הַשְּׁאֵלָה: ׳עַמָּהּ וֵאלֹהֶיהָ׳ – הָעַם וְהָאֱמוּנָה שֶׁל עָרְפָּה.

רַשִׁ״י: ׳שָׁבָה׳ בַּפָּסוּק הוּא עָבָר – עָרְפָּה כְּבָר חָזְרָה, וְנָעֳמִי מַצִּיעָה לְרוּת לַעֲשׂוֹת אֶת אוֹתוֹ דָּבָר.

4

📜 הַפָּסוּק…כִּי אֶל־אֲשֶׁר תֵּלְכִי אֵלֵךְ וּבַאֲשֶׁר תָּלִינִי

💛 אַהֲבַת הַשֵּׁם: כָּל מִצְוָה שֶׁאֲנִי מְקַבֵּל הִיא בְּחִירָה בַּה׳ 💛

📖 רוּת, פֶּרֶק א׳, פָּסוּק ט״ז (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🗺️ ׳בַּאֲשֶׁר תֵּלְכִי אֵלֵךְ׳

רַשִׁ״י מְבָאֵר אֶת פֶּתַח הַגֵּרוּת: ׳מִתּוֹךְ דְּבָרֶיהָ שֶׁל רוּת אַתָּה לָמֵד מַה שֶּׁאָמְרָה לָהּ נָעֳמִי׳. נָעֳמִי: ׳אָסוּר לָנוּ לָצֵאת חוּץ לַתְּחוּם בַּשַּׁבָּת׳ – רוּת: ׳בַּאֲשֶׁר תֵּלְכִי אֵלֵךְ׳. נָעֳמִי: ׳אָסוּר לָנוּ לְהִתְיַחֵד נְקֵבָה עִם זָכָר שֶׁאֵינוֹ אִישָׁהּ׳ – רוּת: ׳בַּאֲשֶׁר תָּלִינִי אָלִין׳. כָּל פָּסוּק הוּא תְּשׁוּבָה לְמִצְוָה אֶחָת.

רַשִׁ״י: כָּל חֵלֶק מִדִּבְרֵי רוּת הוּא קַבָּלָה שֶׁל הֲלָכָה שֶׁנָּעֳמִי אָמְרָה: תְּחוּם שַׁבָּת וְיִחוּד.

5

📜 הַפָּסוּק…אָלִין עַמֵּךְ עַמִּי וֵאלֹהַיִךְ אֱלֹהָי

💛 אַהֲבַת הַשֵּׁם: מִצְווֹת הַתּוֹרָה הֵן מַתָּנָה שֶׁרוּת בָּחֲרָה בָּהּ בְּעֵינַיִם פְּקוּחוֹת 💛

📖 רוּת, פֶּרֶק א׳, פָּסוּק ט״ז (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

📚 תַּרְיַ״ג מִצְוֹת וְשֶׁתַּיִם קְבָרִים

רַשִׁ״י מַמְשִׁיךְ: נָעֳמִי: ׳עַמֵּנוּ מֻבְדָּלִים מִשְּׁאָר עַמִּים בְּתַרְיַ״ג מִצְוֹת׳ – רוּת: ׳עַמֵּךְ עַמִּי׳. ׳אָסוּר לָנוּ עֲבוֹדַת כּוֹכָבִים׳ – ׳אֱלֹהַיִךְ אֱלֹהָי׳. וְנָעֳמִי הִסְבִּירָה שֶׁיֵּשׁ ׳אַרְבַּע מִיתוֹת בֵּית דִּין׳ וּ׳שְׁנֵי קְבָרִים׳ – וְרוּת: ׳בַּאֲשֶׁר תָּמוּתִי אָמוּת׳ ... ׳וְשָׁם אֶקָּבֵר׳. רוּת הִסְכִּימָה לְהִתְחַיֵּב בְּכָל הַמִּצְוֹת, כֻּלָּן.

רַשִׁ״י: רוּת קִבְּלָה עָלֶיהָ אֶת תַּרְיַ״ג הַמִּצְוֹת, אֶת הָאִסּוּר עַל עֲבוֹדָה זָרָה, וְאֶת כָּל דִּין הַתּוֹרָה.

6

📜 הַפָּסוּקוַתֵּלַכְנָה שְׁתֵּיהֶם

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב כָּל מִי שֶׁבּוֹחֵר בּוֹ, וַאֲנִי אוֹהֵב גַּם כָּךְ 💛

📖 רוּת, פֶּרֶק א׳, פְּסוּקִים י״ח–י״ט (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🎗️ ׳כַּמָּה חֲבִיבִים הַגֵּרִים׳

רַשִׁ״י מַקְשִׁיב: ׳וַתֶּחְדַּל לְדַבֵּר אֵלֶיהָ׳ – ׳אֵין מַרְבִּין עָלָיו וְאֵין מְדַקְדְּקִין עָלָיו׳. וְעַל ׳וַתֵּלַכְנָה שְׁתֵּיהֶם׳ הוּא מֵבִיא: ׳אָמַר רַבִּי אַבָּהוּ: ״בֹּא וּרְאֵה כַּמָּה חֲבִיבִים הַגֵּרִים לִפְנֵי הַקָּדוֹשׁ בָּרוּךְ הוּא. כֵּיוָן שֶׁנָּתְנָה דַעְתָּהּ לְהִתְגַּיֵּר, הִשְׁוָה אוֹתָהּ הַכָּתוּב לְנָעֳמִי״׳. הַפָּסוּק אוֹמֵר ׳שְׁתֵּיהֶם׳ – שְׁוֵות.

רַשִׁ״י: אַחֲרֵי שֶׁרוּת קִבְּלָה אֶת הַגֵּרוּת, הַכָּתוּב מַשְׁוֶה אוֹתָהּ לְנָעֳמִי – הַגֵּרִים חֲבִיבִים לִפְנֵי ה׳.

💡

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

מָה לוֹקְחִים מִרָמָה ב׳?

רַשִׁ״י מְבָאֵר שֶׁנָּעֳמִי הוֹדִיעָה לְרוּת מִקְצַת עֳנָשִׁים וּמִצְוֹת, וְרוּת קִבְּלָה אֶת הַכֹּל, וְהַכָּתוּב הִשְׁוָה אוֹתָהּ לְנָעֳמִי.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • נָעֳמִי אָמְרָה לְרוּת מָה הַגֵּרוּת מְחַיֶּבֶת, וְרוּת בָּחֲרָה. מָה אֲנִי בּוֹחֵר בְּכָל יוֹם בְּאַהֲבָה, וְלֹא מִתּוֹךְ הֶרְגֵּל?
  • אֵין מַרְבִּים עַל גֵּר שֶׁמַּחְלִיט. אֵיךְ אֲנִי מְקַבֵּל חָבֵר חָדָשׁ שֶׁרוֹצֶה לְהִצְטָרֵף לַקְּבוּצָה?
  • הַגְּרָ״א מְלַמֵּד שֶׁגַּם הַצָּרוֹת הֵן רַחֲמֵי ה׳. מָתַי קֶשִׁי הֵבִיא אוֹתִי קָרוֹב יוֹתֵר לְה׳?

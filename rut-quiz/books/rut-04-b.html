רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🕊️ ׳ה׳ עִמָּכֶם׳

בְּמִדְרַשׁ רוּת רַבָּה נֶאֱמַר שֶׁ׳עָמַד בֹּעַז וּבֵית דִּינוֹ וְהִסְכִּימוּ לִשְׁאֹל לְשָׁלוֹם׳ בְּשֵׁם ה׳, כְּדֵי שֶׁיָּרִימוּ אֶת שֵׁם ה׳ בְּפִיהֶם בְּכָל יוֹם. הַבְּרָכָה ׳ה׳ עִמָּכֶם׳ הִיא בְּרָכָה שֶׁמְּצִיינֶת שֶׁה׳ נִמְצָא בַּשָּׂדֶה – גַּם בֵּין הָעוֹבְדִים וּבַעַל הַשָּׂדֶה.

הַמִּדְרָשׁ: בֹּעַז וּבֵית דִּינוֹ תִּקְּנוּ לִשְׁאֹל לְשָׁלוֹם בְּשֵׁם ה׳.

1

📜 הַפָּסוּק…אַל־תֵּלְכִי לִלְקֹט בְּשָׂדֶה אַחֵר וְגַם לֹא תַעֲבוּרִי מִזֶּה וְכֹה תִדְבָּקִין עִם־נַעֲרֹתָי

💛 אַהֲבַת הַשֵּׁם: חֶסֶד הוּא לְדַאֲגָה שֶׁהַשֵּׁנִי לֹא יִהְיֶה לְבַד 💛

📖 רוּת, פֶּרֶק ב׳, פָּסוּק ח׳

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🍃 ׳וְכֹה תִדְבָּקִין׳

בֹּעַז אוֹמֵר לְרוּת: ׳אַל תֵּלְכִי לִלְקֹט בְּשָׂדֶה אַחֵר וְגַם לֹא תַעֲבוּרִי מִזֶּה וְכֹה תִדְבָּקִין עִם נַעֲרֹתָי׳. ׳תִדְבָּקִין׳ מְלַמֵּד שֶׁבֹּעַז לֹא רוֹצֶה שֶׁתִּהְיֶה לְבַדָּהּ – הוּא רוֹצֶה שֶׁתִּהְיֶה בְּחֶבְרָה, מוּגֶנֶת וְשׁוֹמֶרֶת עַל כְּבוֹדָהּ.

בֹּעַז דָּאַג שֶׁרוּת תִּהְיֶה בְּחֶבְרָה בְּטוּחָה וְלֹא לְבַדָּהּ.

2

📜 הַפָּסוּק…וְצָמִת וְהָלַכְתְּ אֶל־הַכֵּלִים וְשָׁתִית מֵאֲשֶׁר יִשְׁאֲבוּן הַנְּעָרִים

💛 אַהֲבַת הַשֵּׁם: מִי שֶׁאֵינוֹ מַכְלִים אֶת הַשֵּׁנִי נוֹתֵן לוֹ מַתָּנָה גְּדוֹלָה 💛

📖 רוּת, פֶּרֶק ב׳, פָּסוּק ט׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🥤 ׳אַל תִּכָּלְמִי׳

רַשִׁ״י מְפָרֵשׁ אֶת ׳וְצָמִת וְהָלַכְתְּ אֶל הַכֵּלִים׳: ׳וְאִם תִּצְמְאִי אַל תִּכָּלְמִי מִלֶּכֶת וְלִשְׁתּוֹת מִכְּלֵי הַמַּיִם אֲשֶׁר יִשְׁאֲבוּן הַנְּעָרִים׳. בֹּעַז יוֹדֵעַ שֶׁאָדָם בְּמַצָּב קָשֶׁה מִתְבַּיֵּשׁ לְבַקֵּשׁ אֲפִלּוּ מַיִם, וְלָכֵן הוּא נוֹתֵן לָהּ רְשׁוּת מֵרֹאשׁ.

רַשִׁ״י: בֹּעַז אוֹמֵר לְרוּת שֶׁלֹּא תִתְבַּיֵּשׁ לִשְׁתּוֹת מִן הַכֵּלִים שֶׁהַנְּעָרִים שָׁאֲבוּ.

3

📜 הַפָּסוּק…וַתַּעַזְבִי אָּבִיךְ וְאִמֵּךְ וְאֶרֶץ מוֹלַדְתֵּךְ וַתֵּלְכִי אֶל־עַם אֲשֶׁר לֹא־יָדַעַתְּ תְּמוֹל שִׁלְשׁוֹם

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹקִיר אֶת מִי שֶׁעוֹזֵב אֶת הַנּוֹחַ וּבוֹחֵר בְּה׳ 💛

📖 רוּת, פֶּרֶק ב׳, פָּסוּק י״א

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🛣️ ׳תְּמוֹל שִׁלְשׁוֹם׳

בֹּעַז מַדְגִּישׁ אֶת הַמַּעֲשֶׂה הַגָּדוֹל שֶׁל רוּת: הִיא עָזְבָה ׳אָבִיךְ וְאִמֵּךְ וְאֶרֶץ מוֹלַדְתֵּךְ׳ וְהָלְכָה ׳אֶל עַם אֲשֶׁר לֹא יָדַעַתְּ תְּמוֹל שִׁלְשׁוֹם׳. הַמִּלִּים ׳תְּמוֹל שִׁלְשׁוֹם׳ מְלַמְּדוֹת שֶׁהִיא לֹא הִכִּירָה אֶת הָעָם הַזֶּה כְּלָל – וְהִיא בָּחֲרָה בּוֹ מֵאַהֲבָה. הַמַּעֲשֶׂה שֶׁל אַבְרָהָם אָבִינוּ (׳לֶךְ לְךָ מֵאַרְצְךָ׳) חוֹזֵר בָּהּ.

רוּת עָזְבָה מוֹלֶדֶת וּמִשְׁפָּחָה וּבָאָה לְעַם שֶׁלֹּא הִכִּירָה, כְּמוֹ אַבְרָהָם אָבִינוּ.

4

📜 הַפָּסוּקיְשַׁלֵּם ה׳ פָּעֳלֵךְ וּתְהִי מַשְׂכֻּרְתֵּךְ שְׁלֵמָה מֵעִם ה׳ אֱלֹהֵי יִשְׂרָאֵל אֲשֶׁר־בָּאת לַחֲסוֹת תַּחַת־כְּנָפָיו

💛 אַהֲבַת הַשֵּׁם: מִי שֶׁעוֹשֶׂה טוֹב בְּלִי לְצַפּוֹת, ה׳ מְשַׁלֵּם לוֹ בְּשָׁלֵם 💛

📖 רוּת, פֶּרֶק ב׳, פָּסוּק י״ב

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

💰 ׳מַשְׂכֻּרְתֵּךְ שְׁלֵמָה׳

בֹּעַז מְבָרֵךְ: ׳יְשַׁלֵּם ה׳ פָּעֳלֵךְ וּתְהִי מַשְׂכֻּרְתֵּךְ שְׁלֵמָה׳. ׳מַשְׂכֹּרֶת שְׁלֵמָה׳ פֵּרוּשָׁהּ שֶׁלֹּא יַחְסַר לָהּ דָּבָר. וְהוּא מוֹסִיף מַדּוּעַ זָכְתָה: ׳אֲשֶׁר בָּאת לַחֲסוֹת תַּחַת כְּנָפָיו׳ – כִּי הִיא רָצְתָה בֶּאֱמוּנָה, וְלֹא בִּשְׁבִיל שָׂכָר. הַשָּׂכָר בָּא מֵה׳.

בֹּעַז מְבָרֵךְ שֶׁה׳ יְשַׁלֵּם לְרוּת שָׂכָר שָׁלֵם, כִּי בָּאָה לַחֲסוֹת תַּחַת כַּנְפֵי ה׳.

5

📜 הַפָּסוּק…כִּי נִחַמְתָּנִי וְכִי דִבַּרְתָּ עַל־לֵב שִׁפְחָתֶךָ

💛 אַהֲבַת הַשֵּׁם: הַמִּלִּים הַטּוֹבוֹת שֶׁלִּי יְכוֹלוֹת לְנַחֵם 💛

📖 רוּת, פֶּרֶק ב׳, פָּסוּק י״ג (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🤍 ׳נִחַמְתָּנִי׳

רַשִׁ״י מְבָאֵר אֶת תְּשׁוּבַת רוּת: ׳לֹא אֶהְיֶה כְּאַחַת שִׁפְחֹתֶיךָ – אֵינֶנִּי חֲשׁוּבָה כְּאַחַת מִן הַשְּׁפָחוֹת שֶׁלְּךָ׳. כְּלוֹמַר: ׳אֲפִלּוּ הַשְּׁפָחוֹת שֶׁלְּךָ חֲשׁוּבוֹת מִמֶּנִּי׳. דִּבְרֵי בֹּעַז ׳נִחֲמוּ׳ אוֹתָהּ וְ׳דִבְּרוּ עַל לֵב׳. הַדִּבּוּר הַטּוֹב הוּא מַתָּנָה, אֲפִלּוּ לְמִי שֶׁלֹּא מַמָּשׁ זָקוּק לְלֶחֶם.

רַשִׁ״י: רוּת אוֹמֶרֶת שֶׁהִיא פְּחוּתָה מִן הַשְּׁפָחוֹת, וְדִבְרֵי בֹּעַז נִחֲמוּ אוֹתָהּ.

6

📜 הַפָּסוּק…אֲשֶׁר לֹא־עָזַב חַסְדּוֹ אֶת־הַחַיִּים וְאֶת־הַמֵּתִים

💛 אַהֲבַת הַשֵּׁם: ה׳ לֹא עוֹזֵב אוֹתִי וְלֹא אֶת מִי שֶׁאֲהַבְתִּי 💛

📖 רוּת, פֶּרֶק ב׳, פָּסוּק כ׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

♾️ ׳אֶת הַחַיִּים וְאֶת הַמֵּתִים׳

נָעֳמִי מְבָרֶכֶת: ׳בָּרוּךְ הוּא לַה׳ אֲשֶׁר לֹא עָזַב חַסְדּוֹ אֶת הַחַיִּים וְאֶת הַמֵּתִים׳. רַשִׁ״י מְבָאֵר: ה׳ הוּא ׳שֶׁזָּן וּמְפַרְנֵס אֶת הַחַיִּים וְנִטְפָּל בְּצָרְכֵי הַמֵּתִים׳. בֹּעַז, שֶׁדָּאַג לַחַיּוֹת – רוּת וְנָעֳמִי – יִדְאַג גַּם לִשְׁמוֹ שֶׁל אֱלִימֶלֶךְ שֶׁנִּפְטַר. ׳קָרוֹב לָנוּ הָאִישׁ מִגֹּאֲלֵנוּ הוּא׳ – הוּא הַגּוֹאֵל.

רַשִׁ״י: ה׳ זָן אֶת הַחַיִּים וְדָאַג גַּם לַמֵּתִים; בֹּעַז הוּא הַגּוֹאֵל שֶׁיַּמְשִׁיךְ אֶת הַחֶסֶד.

מָה לוֹקְחִים מִרָמָה ב׳?

רַשִׁ״י מְבָאֵר ׳אַל תִּכָּלְמִי׳ וְ׳לֹא אֶהְיֶה כְּאַחַת שִׁפְחֹתֶיךָ׳, וְאֶת חֶסֶד ה׳ לַחַיִּים וְלַמֵּתִים.

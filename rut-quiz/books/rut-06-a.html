בֹּעַז לוֹקֵחַ אֶת רוּת, וְהִיא נִהְיֵית אִשְׁתּוֹ. הַכָּתוּב אוֹמֵר: ׳וַיִּתֵּן ה׳ לָהּ הֵרָיוֹן וַתֵּלֶד בֵּן׳. הַפָּסוּק מַדְגִּישׁ אֶת הַמָּקוֹר: הַיֶּלֶד הוּא מַתָּנָה מֵה׳. לְאַחַר שָׁנִים שֶׁל צָרוֹת, ה׳ נוֹתֵן שִׂמְחָה.

בֹּעַז וְרוּת נִשְׂאוּ, וה׳ נָתַן לָהֶם בֵּן.

הַשְּׁכֵנוֹת פּוֹנוֹת אֶל נָעֳמִי: ׳בָּרוּךְ ה׳ אֲשֶׁר לֹא הִשְׁבִּית לָךְ גֹּאֵל הַיּוֹם וְיִקָּרֵא שְׁמוֹ בְּיִשְׂרָאֵל׳. הֵן מוֹדוֹת לַה׳. נָעֳמִי, שֶׁאָמְרָה ׳קְרֶאןָ לִי מָרָא׳, עַכְשָׁיו שׁוֹמַעַת בְּרָכָה: ה׳ לֹא עָזַב אוֹתָהּ.

הַשְּׁכֵנוֹת מוֹדוֹת לַה׳ עַל הַגֹּאֵל שֶׁנּוֹלַד לְנָעֳמִי.

הַשְּׁכֵנוֹת אוֹמְרוֹת לְנָעֳמִי: ׳וְהָיָה לָךְ לְמֵשִׁיב נֶפֶשׁ וּלְכַלְכֵּל אֶת שֵׂיבָתֵךְ, כִּי כַלָּתֵךְ אֲשֶׁר אֲהֵבַתֶךְ יְלָדַתּוּ אֲשֶׁר הִיא טוֹבָה לָךְ מִשִּׁבְעָה בָּנִים׳. שִׁבְעָה בָּנִים הוּא הַמִּסְפָּר הַמָּלֵא; וְרוּת, שֶׁהִיא כַּלָּה, טוֹבָה מִכֻּלָּם. הַחֶסֶד שֶׁלָּהּ הוּא הַמַּעֲלָה.

רוּת, שֶׁאָהֲבָה אֶת נָעֳמִי, טוֹבָה לָהּ מִשִּׁבְעָה בָּנִים.

🤱 ׳וַתְּהִי לוֹ לְאֹמֶנֶת׳

נָעֳמִי לוֹקַחַת אֶת הַיֶּלֶד: ׳וַתִּקַּח נָעֳמִי אֶת הַיֶּלֶד וַתְּשִׁתֵהוּ בְחֵיקָהּ וַתְּהִי לוֹ לְאֹמֶנֶת׳. הִיא מְגַדֶּלֶת אוֹתוֹ. מִי שֶׁאָמְרָה ׳הֵמַר שַׁדַּי לִי׳ זוֹכָה שֶׁיִּהְיֶה בְּחֵיקָהּ נֶכֶד. ה׳ הֶחֱזִיר לָהּ אֶת הַחַיִּים.

נָעֳמִי מְגַדֶּלֶת אֶת הַנֶּכֶד, וְה׳ הֶחֱזִיר לָהּ אֶת הַשִּׂמְחָה.

רָמָה א׳ – מֵרוּת עַד דָּוִד

📛 ׳וַתִּקְרֶאנָה שְׁמוֹ עוֹבֵד׳

הַשְּׁכֵנוֹת קוֹרְאוֹת לַיֶּלֶד שֵׁם: ׳יֻלַּד בֵּן לְנָעֳמִי׳, וְשֵׁמוֹ ׳עוֹבֵד׳. ׳הוּא אֲבִי יִשַׁי אֲבִי דָוִד׳. הַמִּלָּה ׳עוֹבֵד׳ קְרוֹבָה לְ׳עֲבוֹדָה׳, וְהִיא מַרְמֶזֶת עַל הַדֶּרֶךְ הַמֻּכָּרֶת: הַנֶּכֶד שֶׁל נָעֳמִי יִהְיֶה סַבָּא שֶׁל הַמֶּלֶךְ.

הַיֶּלֶד נִקְרָא עוֹבֵד, וְהוּא סַבּוֹ שֶׁל דָּוִד הַמֶּלֶךְ.

רָמָה א׳ – מֵרוּת עַד דָּוִד

🌳 ׳וְאֵלֶּה תּוֹלְדוֹת פָּרֶץ׳

הַמְּגִלָּה מִסְתַּיֶּמֶת בִּרְשִׁימַת יוּחֲסִין: ׳פֶּרֶץ הוֹלִיד אֶת חֶצְרוֹן, וְחֶצְרוֹן הוֹלִיד אֶת רָם, וְרָם הוֹלִיד אֶת עַמִּינָדָב׳. רַשִׁ״י מְבָאֵר: ׳לְפִי שֶׁיִּחֵס אֶת דָּוִד עַל שְׁמָהּ שֶׁל רוּת הַמּוֹאֲבִיָּה חָזַר וְיִחֲסוֹ עַל שֵׁם יְהוּדָה׳. דָּוִד הוּא מִן הַשֵּׁבֶט הַמַּלְכוּתִי, שֶׁל יְהוּדָה.

הַמְּגִלָּה מְסַיֶּמֶת בְּיִחוּס דָּוִד לְיְהוּדָה, וְלֹא רַק לְרוּת.

רָמָה א׳ – מֵרוּת עַד דָּוִד

📜 ׳נַחְשׁוֹן הוֹלִיד אֶת שַׂלְמָה׳

שִׁבְעָה דּוֹרוֹת מִפֶּרֶץ עַד דָּוִד: ׳עַמִּינָדָב הוֹלִיד אֶת נַחְשׁוֹן וְנַחְשׁוֹן הוֹלִיד אֶת שַׂלְמָה, וְשַׂלְמוֹן הוֹלִיד אֶת בֹּעַז וּבֹעַז הוֹלִיד אֶת עוֹבֵד, וְעֹבֵד הוֹלִיד אֶת יִשָׁי וְיִשַׁי הוֹלִיד אֶת דָּוִד׳. נַחְשׁוֹן הוּא נְשִׂיא שֵׁבֶט יְהוּדָה בַּמִּדְבָּר, וְדָוִד הוּא ׳נְעִים זְמִירוֹת יִשְׂרָאֵל׳.

מִנַּחְשׁוֹן, נְשִׂיא יְהוּדָה, דֶּרֶךְ בֹּעַז וְעוֹבֵד וְעַד דָּוִד – שִׁלְשֶׁלֶת הַמַּלְכוּת.

📖 רוּת, פֶּרֶק ב׳, פָּסוּק כ״ג; פֶּרֶק ד׳, פָּסוּק י״ז

רָמָה א׳ – מֵרוּת עַד דָּוִד

🌾 שָׁבוּעוֹת – חַג הַקָּצִיר

הַמְּגִלָּה נִקְרֵאת בְּשָׁבוּעוֹת, חַג הַקָּצִיר וּמַתַּן תּוֹרָה. הִיא מַתְחִילָה בִּקְצִיר הַשְּׂעוֹרִים וּמִסְתַּיֶּמֶת בִּקְצִיר הַחִטִּים; וְרוּת, הַגִּיּוֹרֶת, מְקַבֶּלֶת אֶת הַתּוֹרָה כְּמוֹ שֶׁעַם יִשְׂרָאֵל קִבֵּל בְּסִינַי. וְדָוִד הַמֶּלֶךְ נוֹלַד וְנִפְטַר בְּשָׁבוּעוֹת.

מְגִלַּת רוּת נִקְרֵאת בְּשָׁבוּעוֹת – חַג הַקָּצִיר וְהַתּוֹרָה, וְדָוִד נוֹלַד בּוֹ.

מָה לוֹקְחִים מִרָמָה א׳?

בֹּעַז וְרוּת מִתְחַתְּנִים, נוֹלָד עוֹבֵד, נָעֳמִי מְגַדֶּלֶת אוֹתוֹ, וְהַמְּגִלָּה מִסְתַּיֶּמֶת בִּרְשִׁימַת הַיּוּחֲסִין עַד דָּוִד.

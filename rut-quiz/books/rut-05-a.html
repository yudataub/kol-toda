בֹּעַז לוֹקֵחַ אַחְרָיוּת וּפוֹעֵל בְּגָלוּי בַּשַּׁעַר לִפְנֵי עֲשָׂרָה זְקֵנִים, בְּרָכָה ׳כְּרָחֵל וּכְלֵאָה׳ – וְהַכֹּל מִתּוֹךְ חֶסֶד וְהוֹדָיָה

רָמָה א׳ – בַּשַּׁעַר

🏠 ׳הֲלֹא אֲבַקֶּשׁ לָךְ מָנוֹחַ׳

אַחֲרֵי שָׁלוֹשׁ חֳדָשִׁים בְּבֵית לֶחֶם, נָעֳמִי אוֹמֶרֶת לְרוּת: ׳בִּתִּי הֲלֹא אֲבַקֶּשׁ לָךְ מָנוֹחַ אֲשֶׁר יִיטַב לָךְ׳. הִיא רוֹצָה לִמְצֹא לְרוּת בַּיִת נָאֶה וְיַצִּיב. וְהִיא מַזְכִּירָה אֶת בֹּעַז, ׳מֹדַעְתָּנוּ׳ – קְרוֹבֵנוּ – שֶׁזּוֹרֶה אֶת גֹּרֶן הַשְּׂעֹרִים.

נָעֳמִי דּוֹאֶגֶת לְרוּת וּמְחַפֶּשֶׂת לָהּ מָנוֹחַ וּבַיִת.

🙇

💛 אַהֲבַת הַשֵּׁם: כִּבּוּד הַהוֹרִים וְהַמְּבֻגָּרִים הוּא חֵלֶק מִן הָאַהֲבָה 💛

📖 רוּת, פֶּרֶק ג׳, פְּסוּקִים ה׳–ו׳

רָמָה א׳ – בַּשַּׁעַר

🙇 ׳כֹּל אֲשֶׁר תֹּאמְרִי אֵלַי אֶעֱשֶׂה׳

רוּת עוֹנָה לְנָעֳמִי בִּמְשׁוּבָה קְצָרָה: ׳כֹּל אֲשֶׁר תֹּאמְרִי אֵלַי אֶעֱשֶׂה׳. וְהַמִּקְרָא מוֹסִיף: ׳וַתֵּרֶד הַגֹּרֶן וַתַּעַשׂ כְּכֹל אֲשֶׁר צִוַּתָּה חֲמוֹתָהּ׳. רוּת מְכַבֶּדֶת אֶת נָעֳמִי וְעוֹשָׂה בְּתָמִימוּת אֶת מַה שֶּׁנָּעֳמִי אוֹמֶרֶת.

רוּת מְכַבֶּדֶת אֶת חֲמוֹתָהּ וְעוֹשָׂה כְּכֹל אֲשֶׁר צִוְּתָה.

🙏

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְבַקֵּשׁ עֶזְרָה מִמִּי שֶׁיָּכוֹל, בְּכָבוֹד 💛

📖 רוּת, פֶּרֶק ג׳, פְּסוּקִים ט׳–י׳

רָמָה א׳ – בַּשַּׁעַר

🙏 ׳כִּי גֹאֵל אָתָּה׳

רוּת פּוֹנָה אֶל בֹּעַז: ׳אָנֹכִי רוּת אֲמָתֶךָ, וּפָרַשְׂתָּ כְנָפֶךָ עַל אֲמָתְךָ כִּי גֹאֵל אָתָּה׳. רַשִׁ״י מְבָאֵר: ׳לִגְאֹל נַחֲלַת אִישִׁי... וַחֲמוֹתִי וַאֲנִי צְרִיכוֹת לִמְכֹּר נַחֲלָתֵנוּ, וְעַתָּה עָלֶיךָ לִקְנוֹת׳. הִיא מְבַקֶּשֶׁת מִבֹּעַז, כְּקָרוֹב, לָקַחַת אַחְרָיוּת. וּבֹעַז עוֹנֶה: ׳בְּרוּכָה אַתְּ לַה׳ בִּתִּי הֵיטַבְתְּ חַסְדֵּךְ הָאַחֲרוֹן מִן הָרִאשׁוֹן׳.

רוּת מְבַקֶּשֶׁת מִבֹּעַז לִגְאֹל אֶת נַחֲלַת אֱלִימֶלֶךְ, וּבֹעַז מְבָרֵךְ אוֹתָהּ.

3
🛡️

💛 אַהֲבַת הַשֵּׁם: שֵׁם טוֹב נִבְנֶה מִמַּעֲשִׂים קְטַנִּים שֶׁחוֹזְרִים עַל עַצְמָם 💛

📖 רוּת, פֶּרֶק ג׳, פָּסוּק י״א

רָמָה א׳ – בַּשַּׁעַר

🛡️ ׳אֵשֶׁת חַיִל אָתְּ׳

בֹּעַז מַרְגִּיעַ אוֹתָהּ: ׳וְעַתָּה בִּתִּי אַל תִּירְאִי כֹּל אֲשֶׁר תֹּאמְרִי אֶעֱשֶׂה לָּךְ, כִּי יוֹדֵעַ כָּל שַׁעַר עַמִּי כִּי אֵשֶׁת חַיִל אָתְּ׳. כָּל הַשַּׁעַר – כְּלוֹמַר הַזְּקֵנִים וְהַשׁוֹפְטִים שֶׁיּוֹשְׁבִים בַּשַּׁעַר – יוֹדְעִים עַל רוּת שֶׁהִיא ׳אֵשֶׁת חַיִל׳, אִשָּׁה שֶׁל כֹּחַ מוּסָרִי. הַשֵּׁם הַטּוֹב שֶׁלָּהּ הִקְדִּים אוֹתָהּ.

כָּל שַׁעַר עִירוֹ שֶׁל בֹּעַז יוֹדֵעַ שֶׁרוּת אֵשֶׁת חַיִל – שֵׁם טוֹב שֶׁבָּנְתָה בְּמַעֲשֶׂיהָ.

4
🌾

💛 אַהֲבַת הַשֵּׁם: אָדָם שֶׁמַּבְטִיחַ וּמִתְכַּוֵּן – לֹא נָח עַד שֶׁעוֹמֵד בַּדְּבָרִים 💛

📖 רוּת, פֶּרֶק ג׳, פְּסוּקִים י״ג–י״ח

רָמָה א׳ – בַּשַּׁעַר

🌾 ׳שֵׁשׁ שְׂעֹרִים׳

בֹּעַז מַבְטִיחַ: ׳אִם יִגְאָלֵךְ טוֹב יִגְאָל וְאִם לֹא יַחְפֹּץ לְגָאֳלֵךְ וּגְאַלְתִּיךְ אָנֹכִי חַי ה׳׳. הוּא נוֹתֵן לָהּ ׳שֵׁשׁ שְׂעֹרִים׳ וְאוֹמֵר ׳אַל תָּבוֹאִי רֵיקָם אֶל חֲמוֹתֵךְ׳. נָעֳמִי מְחַכָּה: ׳שְׁבִי בִתִּי עַד אֲשֶׁר תֵּדְעִין אֵיךְ יִפֹּל דָּבָר כִּי לֹא יִשְׁקֹט הָאִישׁ כִּי אִם כִּלָּה הַדָּבָר הַיּוֹם׳.

בֹּעַז מַבְטִיחַ לִגְאֹל, וְנָעֳמִי יוֹדַעַת שֶׁהוּא לֹא יָנוּחַ עַד שֶׁיִּסַּדֵּר אֶת הָעִנְיָן.

🏛️

💛 אַהֲבַת הַשֵּׁם: צֶדֶק וְחֶסֶד נַעֲשִׂים בְּגָלוּי וּבְיוֹשֶׁר 💛

רָמָה א׳ – בַּשַּׁעַר

🏛️ ׳וּבֹעַז עָלָה הַשַּׁעַר׳

בַּבֹּקֶר בֹּעַז ׳עָלָה הַשַּׁעַר וַיֵּשֶׁב שָׁם׳, בַּמָּקוֹם שֶׁבּוֹ יוֹשְׁבִים הַזְּקֵנִים וְדָנִים. כְּשֶׁהַגּוֹאֵל הַקָּרוֹב עוֹבֵר, בֹּעַז קוֹרֵא לוֹ: ׳סוּרָה שְׁבָה פֹּה פְּלֹנִי אַלְמֹנִי׳. ׳וַיִּקַּח עֲשָׂרָה אֲנָשִׁים מִזִּקְנֵי הָעִיר׳ – מִנְיָן, כְּדֵי שֶׁהַכֹּל יֵעָשֶׂה לְעֵינֵי עֵדִים וּבְפַרְהֶסְיָה.

בֹּעַז מְסַדֵּר אֶת הָעִנְיָן בְּשַׁעַר הָעִיר לְפָנֵי עֲשָׂרָה זְקֵנִים.

👞

💛 אַהֲבַת הַשֵּׁם: בֹּעַז לָקַח אַחְרָיוּת כְּשֶׁאַחֵר וִתֵּר 💛

📖 רוּת, פֶּרֶק ד׳, פְּסוּקִים ג׳–ח׳

רָמָה א׳ – בַּשַּׁעַר

👞 ׳שָׁלַף אִישׁ נַעֲלוֹ׳

הַגּוֹאֵל הַקָּרוֹב שׁוֹמֵעַ עַל שְׂדֵה אֱלִימֶלֶךְ וְאוֹמֵר ׳אָנֹכִי אֶגְאָל׳, אֲבָל כְּשֶׁבֹּעַז מוֹסִיף שֶׁצָּרִיךְ גַּם לָשֵׂאת אֶת רוּת, הוּא אוֹמֵר: ׳לֹא אוּכַל לִגְאָל לִי פֶּן אַשְׁחִית אֶת נַחֲלָתִי׳. ׳וַיֹּאמֶר הַגֹּאֵל לְבֹעַז קְנֵה לָךְ וַיִּשְׁלֹף נַעֲלוֹ׳. הַנַּעַל הִיא סִימָן לְקִנְיָן: ׳וְזֹאת הַתְּעוּדָה בְּיִשְׂרָאֵל׳.

הַגּוֹאֵל הַקָּרוֹב וִתֵּר, וּבְשְׁלִיפַת הַנַּעַל בֹּעַז קָנָה אֶת הַגְּאֻלָּה.

🎉 ׳כְּרָחֵל וּכְלֵאָה׳

בֹּעַז אוֹמֵר לַזְּקֵנִים: ׳עֵדִים אַתֶּם הַיּוֹם כִּי קָנִיתִי... וְגַם אֶת רוּת הַמֹּאֲבִיָּה אֵשֶׁת מַחְלוֹן קָנִיתִי לִי לְאִשָּׁה לְהָקִים שֵׁם הַמֵּת עַל נַחֲלָתוֹ׳. וְכָל הָעָם בְּרֵכוּ: ׳יִתֵּן ה׳ אֶת הָאִשָּׁה הַבָּאָה אֶל בֵּיתֶךָ כְּרָחֵל וּכְלֵאָה אֲשֶׁר בָּנוּ שְׁתֵּיהֶם אֶת בֵּית יִשְׂרָאֵל... וִיהִי בֵיתְךָ כְּבֵית פֶּרֶץ׳.

הָעָם בֵּרֵךְ אֶת בֹּעַז וְרוּת שֶׁיִּהְיֶה בֵּיתָם כְּבֵית יִשְׂרָאֵל וּכְבֵית פֶּרֶץ.

מָה לוֹקְחִים מִרָמָה א׳?

רוּת פּוֹנָה אֶל בֹּעַז כְּגוֹאֵל, בֹּעַז מְבָרֵךְ אוֹתָהּ ו׳אֵשֶׁת חַיִל׳, וְהַכֹּל מִתְבַּצֵּעַ בַּשַּׁעַר לִפְנֵי עֲשָׂרָה זְקֵנִים.

  • בֹּעַז לָקַח אַחְרָיוּת כְּשֶׁאַחֵר וִתֵּר. מָה לָקַחְתִּי עָלַי לְאַחְרָיוּת, וּמָה בָּרַחְתִּי מִמֶּנּוּ?
  • רוּת יָרְדָה לַגֹּרֶן לִפְנֵי שֶׁהִתְקַשְּׁטָה כְּדֵי לִשְׁמֹר עַל צְנִיעוּתָהּ. אֵיךְ אֲנִי שׁוֹמֵר עַל שְׁמִי?
  • הַמֶּלֶךְ דָּוִד נוֹלַד מִתּוֹךְ הוֹדָיָה. עַל מָה אֲנִי רוֹצֶה לְהוֹדוֹת לַה׳ הַיּוֹם?

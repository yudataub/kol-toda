רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🏛️ ׳כְּבֵית פֶּרֶץ׳

הָעָם בֵּרֵךְ אֶת בֹּעַז: ׳וִיהִי בֵיתְךָ כְּבֵית פֶּרֶץ אֲשֶׁר יָלְדָה תָמָר לִיהוּדָה׳. רַשִׁ״י מְבָאֵר: ׳כְּבֵית פֶּרֶץ – שֶׁיָּצְאוּ מִמֶּנּוּ׳. הִנֵּה, בֵּית פֶּרֶץ הוּא הַבַּיִת שֶׁמִּמֶּנּוּ יָצָא בֹּעַז וְיֵצֵא דָּוִד. הָעָם בֵּרֵךְ שֶׁגַּם בִּיתוֹ שֶׁל בֹּעַז יִהְיֶה שֹׁרֶשׁ לְמַלְכוּת.

רַשִׁ״י: ׳כְּבֵית פֶּרֶץ׳ הוּא הַבַּיִת שֶׁיָּצְאוּ מִמֶּנּוּ – וְהַבְּרָכָה הָיְתָה שֶׁגַּם בֹּעַז וְרוּת יַצְמִיחוּ מַלְכוּת.

🙏 ׳וַיִּתֵּן ה׳ לָהּ הֵרָיוֹן׳

הַפָּסוּק מְסַפֵּר בְּפָשְׁטוּת: ׳וַיִּקַּח בֹּעַז אֶת רוּת וַתְּהִי לוֹ לְאִשָּׁה... וַיִּתֵּן ה׳ לָהּ הֵרָיוֹן וַתֵּלֶד בֵּן׳. הַכָּתוּב לֹא אוֹמֵר ׳וַתַּהַר׳ אֶלָּא מְיַחֵס אֶת הַהֵרָיוֹן לָה׳ – ׳וַיִּתֵּן ה׳׳. כָּךְ הוּא מַדְגִּישׁ שֶׁרוּת זָכְתָה בַּיֶּלֶד בִּזְכוּת מַעֲשֶׂיהָ, אֲבָל הַנְּתִינָה הִיא מֵה׳.

הַכָּתוּב מַדְגִּישׁ שֶׁה׳ הוּא הַנּוֹתֵן שֶׁל הַהֵרָיוֹן וְהַבֵּן.

📜 הַפָּסוּק…וַעֲשֵׂה־חַיִל בְּאֶפְרָתָה וּקְרָא־שֵׁם בְּבֵית לָחֶם

💛 אַהֲבַת הַשֵּׁם: שֵׁם טוֹב וּגָדוֹל נִבְנֶה מִצְּדָקוֹת 💛

📖 רוּת, פֶּרֶק ד׳, פָּסוּק י״א (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

✨ ׳וּקְרָא שֵׁם בְּבֵית לָחֶם׳

רַשִׁ״י מְבָאֵר אֶת בִּרְכַּת הָעָם בַּשַּׁעַר: ׳וּקְרָא שֵׁם – כְּלוֹמַר, ״יִגְדַּל שְׁמֶךָ״׳. הַבְּרָכָה: ׳וַעֲשֵׂה חַיִל בְּאֶפְרָתָה וּקְרָא שֵׁם בְּבֵית לָחֶם׳. הָעָם מִתְפַּלֵּל שֶׁשְּׁמוֹ שֶׁל בֹּעַז יִתְפַּרְסֵם. וְאָכֵן שְׁמוֹ מְפֻרְסָם עַד הַיּוֹם, כִּי הוּא סַבּוֹ שֶׁל יִשַׁי וְדָוִד.

רַשִׁ״י: ׳וּקְרָא שֵׁם׳ פֵּרוּשׁוֹ ׳יִגְדַּל שְׁמֶךָ׳ – וְכָךְ הָיָה.

🥇

📜 הַפָּסוּק…כִּי כַלָּתֵךְ אֲשֶׁר־אֲהֵבַתֶךְ יְלָדַתּוּ אֲשֶׁר־הִיא טוֹבָה לָךְ מִשִּׁבְעָה בָּנִים

💛 אַהֲבַת הַשֵּׁם: הָאַהֲבָה שֶׁאֲנִי נוֹתֵן שָׁוָה יוֹתֵר מִמַּה שֶּׁאֲנִי מְקַבֵּל 💛

📖 רוּת, פֶּרֶק ד׳, פָּסוּק ט״ו

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🥇 ׳טוֹבָה לָךְ מִשִּׁבְעָה בָּנִים׳

הַשְּׁכֵנוֹת אוֹמְרוֹת לְנָעֳמִי: ׳כִּי כַלָּתֵךְ אֲשֶׁר אֲהֵבַתֶךְ יְלָדַתּוּ אֲשֶׁר הִיא טוֹבָה לָךְ מִשִּׁבְעָה בָּנִים׳. בְּסִפּוּר חַנָּה נֶאֱמַר ׳טוֹב לָךְ... מֵעֲשָׂרָה בָּנִים׳; כָּאן ׳שִׁבְעָה׳ – הַמִּסְפָּר הַמָּלֵא בַּתַּנַ״ךְ. רוּת, שֶׁאֲהֲבָה בְּלִי לְצַפּוֹת, שָׁוָה יוֹתֵר מִמִּשְׁפָּחָה שְׁלֵמָה.

הַשְּׁכֵנוֹת מְשַׁבְּחוֹת אֶת רוּת: אַהֲבָתָהּ לְנָעֳמִי שָׁוָה יוֹתֵר מִשִּׁבְעָה בָּנִים.

📜 הַפָּסוּקוַתִּקַּח נָעֳמִי אֶת־הַיֶּלֶד וַתְּשִׁתֵהוּ בְחֵיקָהּ וַתְּהִי־לוֹ לְאֹמֶנֶת

💛 אַהֲבַת הַשֵּׁם: מִי שֶׁמְּחַנֵּךְ בֶּאֱמוּנָה בּוֹנֶה עֲתִיד 💛

📖 רוּת, פֶּרֶק ד׳, פָּסוּק ט״ז

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🪑 ׳אֹמֶנֶת׳

הַפָּסוּק אוֹמֵר שֶׁנָּעֳמִי ׳הָיְתָה לוֹ לְאֹמֶנֶת׳. ׳אֹמֶנֶת׳ הִיא מִי שֶׁמְּגַדֶּלֶת בְּאֶמוּנָה. נָעֳמִי הִנִּיחָה אֶת הַיֶּלֶד ׳בְחֵיקָהּ׳ – הוּא קָרוֹב אֶל לִבָּהּ. הָאֱמוּנָה הִיא הַתְּכוּנָה הַמְחַבֶּרֶת בֵּין שְׁנֵי הַדּוֹרוֹת. הַנֶּכֶד לֹא רַק יֶלֶד – הוּא תִּקְוָה.

נָעֳמִי מְגַדֶּלֶת אֶת הַנֶּכֶד בֶּאֱמוּנָה, וְהוּא הַתִּקְוָה שֶׁל הַמִּשְׁפָּחָה.

📜 הַפָּסוּק…יֻלַּד־בֵּן לְנָעֳמִי וַתִּקְרֶאנָה שְׁמוֹ עוֹבֵד

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת ׳עוֹבֵד׳ – עוֹבֵד אֶת ה׳ 💛

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🌿 ׳עוֹבֵד׳ – שֵׁם שֶׁנִּתְנוּ הַשְּׁכֵנוֹת

הַשְּׁכֵנוֹת קוֹרְאוֹת לַיֶּלֶד ׳עוֹבֵד׳ וְאוֹמְרוֹת ׳יֻלַּד בֵּן לְנָעֳמִי׳. שִׂימוּ לֵב: הֵן לֹא אוֹמְרוֹת ״לְרוּת״ אֶלָּא ״לְנָעֳמִי״, כִּי בִּשְׁבִילָהּ כָּל הַסִּפּוּר: הַנֶּכֶד הוּא הַנִּחוּם שֶׁלָּהּ. וְהַמִּלָּה ׳עוֹבֵד׳ מְלַמֶּדֶת שֶׁהַתִּקְוָה הִיא לַעֲבֹד אֶת ה׳.

הַשֵּׁם ׳עוֹבֵד׳ מְרַמֵּז שֶׁהַדּוֹר הַבָּא יַעֲבֹד אֶת ה׳.

רָמָה ב׳ – עִיּוּן בְּרַשִׁ״י

🧭 ׳וְאֵלֶּה תּוֹלְדוֹת פָּרֶץ׳

רַשִׁ״י: ׳לְפִי שֶׁיִּחֵס אֶת דָּוִד עַל שְׁמָהּ שֶׁל רוּת הַמּוֹאֲבִיָּה חָזַר וְיִחֲסוֹ עַל שֵׁם יְהוּדָה׳. בַּפְּסוּקִים הַקּוֹדְמִים דָּוִד הוּא ׳אֲבִי יִשַׁי׳ וְהַנֶּכֶד שֶׁל רוּת; עַכְשָׁיו הוּא נֶכֶד שֶׁל פֶּרֶץ בֶּן יְהוּדָה. הַמְּגִלָּה לֹא מַסְתִּירָה אֶת רוּת, וְגַם לֹא מַעֲלִימָה אֶת שֶׁרֶשׁ הַמַּלְכוּת שֶׁל יְהוּדָה.

רַשִׁ״י: הַמְּגִלָּה חוֹזֶרֶת לְיַחֵס אֶת דָּוִד לְיְהוּדָה, כְּדֵי שֶׁהַמַּלְכוּת תְּיֻחַס גַּם לְשֶׁבֶט יְהוּדָה.

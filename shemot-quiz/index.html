🔥 חִידוֹנֵי חֻמַּשׁ שְׁמוֹת

49 חִידוֹנִים לְפִי תָּכְנִית הַלִּמּוּדִים שֶׁל מִשְׂרַד הַחִנּוּךְ (תַּנַ״ךְ חֶמֶ״ד), לְכִתָּה ג׳. בְּכָל חִידוֹן כַּרְטִיסֵי לִמּוּד עִם אַהֲבַת ה׳ 💛 וְשָׁלוֹשׁ רָמוֹת: תְּמוּנוֹת, שְׁאֵלוֹת לַחֲכָמִים, וּבָרָמָה הַשְּׁלִישִׁית מִשְׂחָקִים – בְּכָל פַּעַם מִשְׂחָק אַחֵר

🧺 שִׁעְבּוּד מִצְרַיִם וּשְׁלִיחוּת מֹשֶׁה

1. וְאֵלֶּה שְׁמוֹת – שִׁעְבּוּד וּמְיַלְּדוֹתפֶּרֶק א׳
2. הַתֵּבָה עַל שְׂפַת הַיְאֹרפֶּרֶק ב׳, פְּסוּקִים א׳–י׳
3. מֹשֶׁה יוֹצֵא אֶל אֶחָיופֶּרֶק ב׳, פְּסוּקִים י״א–כ״ב
4. הַסְּנֶה בֹּעֵר בָּאֵשׁפֶּרֶק ב׳ כ״ג – פֶּרֶק ג׳ י״ב
5. ׳אֶהְיֶה אֲשֶׁר אֶהְיֶה׳ וְהָאוֹתוֹתפֶּרֶק ג׳ י״ג – פֶּרֶק ד׳ ט׳
6. ׳כְּבַד פֶּה׳ – הַחֲשָׁשׁוֹת שֶׁל מֹשֶׁהפֶּרֶק ד׳ י׳–ל״א
7. אֶל פַּרְעֹה: ׳שַׁלַּח אֶת עַמִּי׳פֶּרֶק ה׳, פְּסוּקִים א׳–י״ד
8. ׳לָמָּה הֲרֵעֹתָה׳ – תְּלוּנַת הַשּׁוֹטְרִיםפֶּרֶק ה׳ פָּסוּק ט״ו – פֶּרֶק ו׳ פָּסוּק א׳
9. אַרְבַּע לְשׁוֹנוֹת שֶׁל גְּאֻלָּהפֶּרֶק ו׳ פָּסוּק ב׳ – פֶּרֶק ז׳ פָּסוּק י״ג

🐸 הַמַּכּוֹת

10. מַכַּת דָּם – ׳בְּזֹאת תֵּדַע כִּי אֲנִי ה׳׳פֶּרֶק ז׳, פְּסוּקִים י״ד–כ״ה
11. צְפַרְדֵּעַ, כִּנִּים וְעָרוֹב – ׳וְהִפְלֵיתִי אֶת אֶרֶץ גֹּשֶׁן׳פֶּרֶק ז׳ כ״ו – פֶּרֶק ח׳ כ״ח
12. דֶּבֶר, שְׁחִין וּבָרָד – ׳בַּעֲבוּר תֵּדַע כִּי אֵין כָּמֹנִי׳פֶּרֶק ט׳
13. אַרְבֶּה וְחֹשֶׁךְ – אוֹר בְּמוֹשְׁבֹתָםפֶּרֶק י׳, פְּסוּקִים א׳–כ״ט
14. סִכּוּם: תֵּשַׁע הַמַּכּוֹת – ׳בַּעֲבוּר תֵּדַע׳פֶּרֶק ז׳ פָּסוּק י״ד – פֶּרֶק י׳ פָּסוּק כ״ט
15. הַהַתְרָאָה הָאַחֲרוֹנָה – ׳לֹא יֶחֱרַץ כֶּלֶב לְשֹׁנוֹ׳פֶּרֶק י״א, פְּסוּקִים א׳–י׳

🍞 הַפֶּסַח וִיצִיאַת מִצְרַיִם

16. ׳הַחֹדֶשׁ הַזֶּה לָכֶם׳ – קָרְבַּן הַפֶּסַחפֶּרֶק י״ב, פְּסוּקִים א׳–כ״ח
17. בַּחֲצִי הַלַּיְלָה – יְצִיאַת מִצְרַיִםפֶּרֶק י״ב, פְּסוּקִים כ״ט–מ״ב
18. חֻקַּת הַפֶּסַח וְזִכָּרוֹן הַיְּצִיאָהפֶּרֶק י״ב, פְּסוּקִים מ״ג–נ״א; פֶּרֶק י״ג, פְּסוּקִים א׳–ט״ז

🌊 יַם סוּף וְהַמַּסָּע בַּמִּדְבָּר

19. עַמּוּד הֶעָנָן וּקְרִיעַת יַם סוּףפֶּרֶק י״ג, פְּסוּקִים י״ז–כ״ב – פֶּרֶק י״ד
20. שִׁירַת הַיָּםפֶּרֶק ט״ו, פְּסוּקִים א׳–כ״א
21. מָרָה וְאֵילִםפֶּרֶק ט״ו, פְּסוּקִים כ״ב–כ״ז
22. הַמָּן – לֶחֶם מִן הַשָּׁמַיִםפֶּרֶק ט״ז
23. מַסָּה וּמְרִיבָה – ׳הֲיֵשׁ ה׳ בְּקִרְבֵּנוּ׳?פֶּרֶק י״ז, פְּסוּקִים א׳–ז׳
24. מִלְחֶמֶת עֲמָלֵק – ׳וִידֵי מֹשֶׁה כְּבֵדִים׳פֶּרֶק י״ז, פְּסוּקִים ח׳–ט״ז

⛰️ יִתְרוֹ וּמַתַּן תּוֹרָה

25. יִתְרוֹ וַעֲצָתוֹפֶּרֶק י״ח
26. מַעֲמַד הַר סִינַיפֶּרֶק י״ט
27. ׳אָנֹכִי ה׳ אֱלֹהֶיךָ׳פֶּרֶק כ׳, פְּסוּקִים א׳–ז׳
28. זָכוֹר וְכַבֵּד – שַׁבָּת, הוֹרִים וַחֲבֵרִיםפֶּרֶק כ׳, פְּסוּקִים ח׳–כ״ב

⚖️ מִשְׁפָּטִים

29. עֶבֶד עִבְרִי – שֵׁשׁ שָׁנִים וּבַשְּׁבִיעִית חָפְשִׁיפֶּרֶק כ״א, פְּסוּקִים א׳–ו׳
30. נִזְקֵי הַגּוּף – כָּל אָדָם יָקָרפֶּרֶק כ״א, פְּסוּקִים י״ח–י״ט, כ״ד–כ״ה
31. בּוֹר וְשׁוֹר – מִי אַחְרַאי?פֶּרֶק כ״א, פְּסוּקִים ל״ג–ל״ו
32. גְּנֵבָה וְנִזְקֵי מָמוֹן – ׳חֲמִשָּׁה בָקָר׳פֶּרֶק כ״א, פָּסוּק ל״ז – פֶּרֶק כ״ב, פָּסוּק ה׳
33. אַרְבָּעָה שׁוֹמְרִים – חִנָּם, שָׂכָר, שׁוֹאֵל וְשׂוֹכֵרפֶּרֶק כ״ב, פְּסוּקִים ו׳–י״ד
34. גֵּר, עָנִי וַאֲבֵדָהפֶּרֶק כ״ב, פְּסוּקִים כ׳–כ״ו; פֶּרֶק כ״ג, פְּסוּקִים ד׳–ה׳, ט׳
35. מִשְׁפַּט צֶדֶקפֶּרֶק כ״ג, פְּסוּקִים א׳–ג׳, ו׳–ח׳ · פֶּרֶק י״ח, פָּסוּק כ״א
36. שְׁמִטָּה, שַׁבָּת וְשָׁלֹשׁ רְגָלִיםפֶּרֶק כ״ג, פְּסוּקִים י׳–כ״ו
37. ״נַעֲשֶׂה וְנִשְׁמָע״ – בְּרִית בְּהַר סִינַיפֶּרֶק כ״ד, פְּסוּקִים א׳–י״ח

🕎 צִוּוּי הַמִּשְׁכָּן

38. ״וְעָשׂוּ לִי מִקְדָּשׁ״ – הָאָרוֹן, הַשֻּׁלְחָן וְהַמְּנוֹרָהפֶּרֶק כ״ה, פְּסוּקִים א׳–מ׳
39. הַמִּשְׁכָּן וְהֶחָצֵרפֶּרֶק כ״ו – פֶּרֶק כ״ז, פָּסוּק כ״א
40. בִּגְדֵי כְּהֻנָּה – לְכָבוֹד וּלְתִפְאָרֶתפֶּרֶק כ״ח, פְּסוּקִים א׳–מ״ג
41. הַכֹּהֲנִים, מַחֲצִית הַשֶּׁקֶל וְהָאֻמָּנִיםפֶּרֶק כ״ט, פְּסוּקִים מ״ה–מ״ו; פֶּרֶק ל׳, פְּסוּקִים י״א–ט״ז; פֶּרֶק ל״א, פְּסוּקִים א׳–י״א
42. סִכּוּם: כְּלֵי הַמִּשְׁכָּן וּבֵית הַכְּנֶסֶתפְּרָקִים כ״ה–ל׳ (כְּלֵי הַמִּשְׁכָּן)

🪨 חֵטְא הָעֵגֶל וְהַלּוּחוֹת הַשְּׁנִיִּים

43. חֵטְא הָעֵגֶל, תְּפִלַּת מֹשֶׁה וּשְׁבִירַת הַלּוּחוֹתפֶּרֶק ל״א פָּסוּק י״ח – פֶּרֶק ל״ב פָּסוּק כ׳
44. ׳מִי לַה׳ אֵלָי׳ – מֹשֶׁה עוֹלֶה לְכַפֵּר עַל עַמּוֹפֶּרֶק ל״ב, פְּסוּקִים כ״א–ל״ה
45. אֹהֶל מוֹעֵד – ׳הַרְאֵנִי נָא אֶת כְּבֹדֶךָ׳פֶּרֶק ל״ג, פְּסוּקִים א׳–כ״ג
46. הַלּוּחוֹת הַשְּׁנִיִּיםפֶּרֶק ל״ד, פְּסוּקִים א׳–י״א, כ״ז–ל״ה

⛺ עֲשִׂיַּת הַמִּשְׁכָּן וַהֲקָמָתוֹ

47. נִדְבַת הָעָם וַעֲשִׂיַּת הַמִּשְׁכָּןפֶּרֶק ל״ה – פֶּרֶק ל״ח (בְּחִירָה)
48. הֲקָמַת הַמִּשְׁכָּן – ׳כַּאֲשֶׁר צִוָּה ה׳ אֶת מֹשֶׁה׳פֶּרֶק מ׳, פְּסוּקִים א׳–ל״ג
49. ׳וּכְבוֹד ה׳ מָלֵא אֶת הַמִּשְׁכָּן׳ – סִכּוּם הַחֻמָּשׁפֶּרֶק מ׳, פְּסוּקִים ל״ד–ל״ח · סִכּוּם הַחֻמָּשׁ

חוּט הַשָּׁנִי

רָמָה ב׳ – עִיּוּן בַּסִּפּוּר

שְׁנֵי מְרַגְּלִים בִּירִיחוֹ מִסְתַּתְּרִים בַּגָּג, וְרָחָב מְצִילָה אוֹתָם – חֶסֶד שֶׁחוֹזֵר בַּחֲזָרָה

📖 פֶּרֶק ב׳

🎯 לַחִידוֹן

📜 הַפָּסוּקוַיִּשְׁלַח יְהוֹשֻׁעַ בִּן נוּן מִן הַשִּׁטִּים שְׁנַיִם אֲנָשִׁים מְרַגְּלִים חֶרֶשׁ

💛 אַהֲבַת הַשֵּׁם: ה׳ עוֹזֵר לִי לָדַעַת מָה לַעֲשׂוֹת 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק ב׳, פָּסוּק א׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּסִּפּוּר

✨ רַשִׁ״י: ׳חֶרֶשׁ׳

׳וַיִּשְׁלַח יְהוֹשֻׁעַ... שְׁנַיִם אֲנָשִׁים מְרַגְּלִים חֶרֶשׁ׳. רַשִׁ״י מְבָאֵר: ׳חֶרֶשׁ – בְּרָז׳, כְּלוֹמַר בְּסוֹד. יְהוֹשֻׁעַ אָמַר לָהֶם: ׳עֲשׂוּ עַצְמְכֶם כְּחֵרְשִׁים׳, כְּדֵי שֶׁלֹּא יַסְתִּירוּ דְּבָרִים מִפְּנֵיכֶם. וְעוֹד פֵּרוּשׁ: ׳חֶרֶס׳ – הֵם נָשְׂאוּ קְדֵרוֹת, כְּדֵי לְהֵרָאוֹת כְּקַדָּרִים.

רַשִׁ״י: מְרַגְּלִים ׳חֶרֶשׁ׳ – בְּסוֹד, וְאוּלַי גַּם בִּלְבוּשׁ שֶׁל קַדָּרִים.

1

📜 הַפָּסוּקלְכוּ רְאוּ אֶת הָאָרֶץ וְאֶת יְרִיחוֹ

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לָנוּ כֹּחַ גַּם לַדָּבָר הֲכִי קָשֶׁה 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק ב׳, פָּסוּק א׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּסִּפּוּר

🔍 מִנַּיִן לָנוּ? לָמָּה דַּוְקָא יְרִיחוֹ?

׳רְאוּ אֶת הָאָרֶץ וְאֶת יְרִיחוֹ׳. יְרִיחוֹ הָיְתָה חֵלֶק מִן הָאָרֶץ, אָז לָמָּה כָּתוּב אוֹתָהּ בְּנִפְרָד? רַשִׁ״י מְבָאֵר: ׳שֶׁהָיְתָה קָשָׁה כְּנֶגֶד כֻּלָּם, לְפִי שֶׁהָיְתָה יוֹשֶׁבֶת עַל הַסְּפָר׳ – הִיא הָעִיר הָרִאשׁוֹנָה בַּגְּבוּל. וּמְצוּדַת דָּוִד: הֵם בָּדְקוּ ׳אִם נָמֵס לִבָּם׳.

רַשִׁ״י: יְרִיחוֹ הָיְתָה הֲכִי קָשָׁה, כִּי יָשְׁבָה עַל הַגְּבוּל.

2

📜 הַפָּסוּקוְהִיא הֶעֱלָתַם הַגָּגָה וַתִּטְמְנֵם בְּפִשְׁתֵּי הָעֵץ הָעֲרֻכוֹת לָהּ עַל הַגָּג

💛 אַהֲבַת הַשֵּׁם: ה׳ מִסְתִּיר אוֹתִי כְּשֶׁאֲנִי צָרִיךְ 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק ב׳, פָּסוּק ו׳ (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּסִּפּוּר

🌾 מְצוּדַת דָּוִד: ׳בְּפִשְׁתֵּי הָעֵץ׳

מָה הֵם ׳פִּשְׁתֵּי הָעֵץ׳? מְצוּדַת דָּוִד מְבָאֵר: ׳בְּפִשְׁתִּים שֶׁהֵם עֲדַיִן בְּתוֹךְ הָעֵץ, וְהֵם הַקָּנִים שֶׁל פִּשְׁתָּן׳. הַפִּשְׁתִּים ׳הָעֲרוּכוֹת׳ הָיוּ מְסֻדָּרוֹת עַל הַגָּג לְיַבֵּשׁ, וּמִתַּחְתֵּיהֶן אִי אֶפְשָׁר לִרְאוֹת אִישׁ.

מְצוּדַת דָּוִד: ׳פִּשְׁתֵּי הָעֵץ׳ הֵם קְנֵי פִּשְׁתָּן.

3

📜 הַפָּסוּקוְעַתָּה הִשָּׁבְעוּ נָא לִי בַּה׳ כִּי עָשִׂיתִי עִמָּכֶם חֶסֶד

💛 אַהֲבַת הַשֵּׁם: ה׳ עוֹשֶׂה אִתִּי חֶסֶד כָּל יוֹם, אַף שֶׁלֹּא עָשִׂיתִי לוֹ כְּלוּם 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק ב׳, פָּסוּק י״ב (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּסִּפּוּר

🤝 מְצוּדַת דָּוִד: ׳חֶסֶד׳ לִפְנֵי שֶׁקִּבְּלָה

רָחָב אָמְרָה: ׳כִּי עָשִׂיתִי עִמָּכֶם חֶסֶד׳. מְצוּדַת דָּוִד מְבָאֵר: ׳חֶסֶד – כִּי עֲדַיִן לֹא עֲשִׂיתֶם לִי טוֹבָה מְאוּמָה׳. כְּלוֹמַר, הִיא עָזְרָה לָהֶם מִבְּלִי שֶׁקִּבְּלָה דָּבָר, וְעַכְשָׁיו הִיא מְבַקֶּשֶׁת שֶׁגַּם הֵם יַעֲשׂוּ חֶסֶד.

מְצוּדַת דָּוִד: חֶסֶד הוּא לַעֲזֹר בְּלִי שֶׁקִּבַּלְנוּ קֹדֶם.

4

📜 הַפָּסוּקכִּי ה׳ אֱלֹהֵיכֶם הוּא אֱלֹהִים בַּשָּׁמַיִם מִמַּעַל וְעַל הָאָרֶץ מִתָּחַת

💛 אַהֲבַת הַשֵּׁם: כֻּלָּם מַכִּירִים בְּכֹחַ שֶׁל ה׳, וַאֲנִי שָׂמֵחַ שֶׁהוּא אוֹהֵב אוֹתִי 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק ב׳, פָּסוּק י״א (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּסִּפּוּר

💛 אַהֲבַת ה׳: ׳הוּא אֱלֹהִים׳

רָחָב אָמְרָה: ׳כִּי ה׳ אֱלֹהֵיכֶם הוּא אֱלֹהִים בַּשָּׁמַיִם מִמַּעַל וְעַל הָאָרֶץ מִתָּחַת׳. מְצוּדַת דָּוִד מְבָאֵר: ׳הוּא אֱלֹהִים – אִם כֵּן הַיְכֹלֶת בְּיָדוֹ עַל כֻּלָּם׳. אֲפִלּוּ מִי שֶׁגָּר בְּיְרִיחוֹ הֵבִין שֶׁה׳ שׁוֹלֵט בְּכָל הָעוֹלָם.

מְצוּדַת דָּוִד: ה׳ שׁוֹלֵט בַּשָּׁמַיִם וּבָאָרֶץ – וְהַכֹּל בְּיָדוֹ.

5

📜 הַפָּסוּקהִנֵּה אֲנַחְנוּ בָאִים בָּאָרֶץ אֶת תִּקְוַת חוּט הַשָּׁנִי הַזֶּה תִּקְשְׁרִי בַּחַלּוֹן אֲשֶׁר הוֹרַדְתֵּנוּ בוֹ

💛 אַהֲבַת הַשֵּׁם: יֵשׁ לִי תִּקְוָה כִּי ה׳ שׁוֹמֵר עָלַי 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק ב׳, פָּסוּק י״ח (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּסִּפּוּר

🧵 רַשִׁ״י: ׳תִּקְוַת׳

׳אֶת תִּקְוַת חוּט הַשָּׁנִי הַזֶּה תִּקְשְׁרִי בַּחַלּוֹן׳. מָה פֵּרוּשׁ ׳תִּקְוַת׳? רַשִׁ״י מְבָאֵר: ׳לְשׁוֹן קַו וְחֶבֶל׳. אָז ׳תִּקְוַת חוּט הַשָּׁנִי׳ הִיא הַחוּט, וְגַם הַתִּקְוָה: מִי שֶׁקָּשַׁר אוֹתוֹ, תִּקְוָתוֹ לְהִנָּצֵל.

רַשִׁ״י: ׳תִּקְוָה׳ פֵּרוּשָׁהּ גַּם חֶבֶל – הַחֶבֶל שֶׁל הַתִּקְוָה.

6

📜 הַפָּסוּקוַתֹּאמֶר לָהֶם הָהָרָה לֵכוּ פֶּן יִפְגְּעוּ בָכֶם הָרֹדְפִים וְנַחְבֵּתֶם שָׁמָּה שְׁלֹשֶׁת יָמִים

💛 אַהֲבַת הַשֵּׁם: ה׳ מְכַוֵּן אוֹתִי לַדֶּרֶךְ הַטּוֹבָה 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק ב׳, פָּסוּק ט״ז (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּסִּפּוּר

⏳ רַשִׁ״י: ׳עַד שׁוֹב הָרֹדְפִים׳

רָחָב אָמְרָה: ׳וְנַחְבֵּתֶם שָׁמָּה שְׁלֹשֶׁת יָמִים עַד שׁוֹב הָרֹדְפִים׳. מִנַּיִן יָדְעָה שֶׁהֵם יָשׁוּבוּ אַחֲרֵי שְׁלֹשָׁה יָמִים? רַשִׁ״י מְבָאֵר: ׳רָאֲתָה בְּרוּחַ הַקֹּדֶשׁ שֶׁיָּשׁוּבוּ לְסוֹף שְׁלֹשֶׁת יָמִים׳.

רַשִׁ״י: רָחָב יָדְעָה בְּרוּחַ הַקֹּדֶשׁ כַּמָּה זְמַן לְהִתְחַבֵּא.

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְפָרְשִׁים מְגַלִּים שֶׁחֶסֶד אֲמִתִּי הוּא חֶסֶד בְּלִי לְחַכּוֹת לִגְמוּל, וּמִי שֶׁעוֹשֶׂה אוֹתוֹ מוֹצֵא תִּקְוָה – ׳תִּקְוַת חוּט הַשָּׁנִי׳. וְה׳ הוּא אֱלֹהִים בַּשָּׁמַיִם וּבָאָרֶץ.

🤔 חוֹשְׁבִים עַל זֶה

  • רָחָב עָשְׂתָה חֶסֶד עִם הַמְרַגְּלִים, וְהֵם עָשׂוּ חֶסֶד בַּחֲזָרָה. מָתַי עָשִׂיתִי חֶסֶד וְקִבַּלְתִּי חֶסֶד?
  • ׳שָׁמַעְנוּ אֵת אֲשֶׁר הוֹבִישׁ ה׳׳ – גַּם בְּיְרִיחוֹ שָׁמְעוּ עַל נִסִּים. אֵיזֶה נֵס אֲנִי רוֹצֶה לְסַפֵּר הַשָּׁבוּעַ?
  • מָה אֲנִי יָכוֹל לַעֲשׂוֹת הַיּוֹם בִּשְׁבִיל מִישֶׁהוּ בְּלִי לְחַכּוֹת לְתוֹדָה?

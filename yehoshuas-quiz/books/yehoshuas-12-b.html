וְהָאָרֶץ שָׁקְטָה

רָמָה ב׳ – מַה אוֹמְרִים הַמְפָרְשִׁים

מְלָכִים רַבִּים מִצְּפוֹן הָאָרֶץ נֶאֱסָפִים בְּמֵי מֵרוֹם, וְה׳ אוֹמֵר לִיהוֹשֻׁעַ ׳אַל תִּירָא׳. בַּסּוֹף ׳וְהָאָרֶץ שָׁקְטָה מִמִּלְחָמָה׳

📖 פְּרָקִים י״א–י״ב

🎯 לַחִידוֹן
🗺️

📜 הַפָּסוּקוְאֶל הַמְּלָכִים אֲשֶׁר מִצְּפוֹן בָּהָר וּבָעֲרָבָה נֶגֶב כִּנֲרוֹת וּבַשְּׁפֵלָה וּבְנָפוֹת דּוֹר מִיָּם

💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לָנוּ אֶרֶץ יָפָה עִם כִּנֶּרֶת, הָרִים וְעֵמֶק, וַאֲנִי מוֹדֶה לוֹ 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק י״א, פָּסוּק ב׳ (רַשִׁ״י)

רָמָה ב׳ – מַה אוֹמְרִים הַמְפָרְשִׁים

🗺️ רַשִׁ״י: ׳כִּנְרוֹת׳ וְ׳נָפוֹת דּוֹר׳

יָבִין שׁוֹלֵחַ אֶל מְלָכִים ׳נֶגֶב כִּנְרוֹת׳ וְ׳בְּנָפוֹת דּוֹר׳. רַשִׁ״י מְבָאֵר: ׳נֶגֶב כִּנְרוֹת – דְּרוֹם גִּינוֹסָר׳, אֵזוֹר הַכִּנֶּרֶת. ׳בְּנָפוֹת דּוֹר – פִּלְכֵי דּוֹר׳, כְּלוֹמַר חֶלְקֵי אָרֶץ. ׳מִיָּם – מַעֲרָבָה׳, לְכִוּוּן הַיָּם.

רַשִׁ״י מְבָאֵר מִלִּים שֶׁל מָקוֹם: כִּנְרוֹת, נָפוֹת דּוֹר, מִיָּם.

1
🌊

📜 הַפָּסוּקוַיִּרְדְּפוּם עַד צִידוֹן רַבָּה וְעַד מִשְׂרְפוֹת מַיִם וְעַד בִּקְעַת מִצְפֶּה מִזְרָחָה

💛 אַהֲבַת הַשֵּׁם: ה׳ בָּרָא שֶׁמֶשׁ וְיָם וּמֶלַח, וַאֲנִי מִשְׁתָּאֶה מִיצִירַת הַבּוֹרֵא 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק י״א, פָּסוּק ח׳ (רַשִׁ״י)

רָמָה ב׳ – מַה אוֹמְרִים הַמְפָרְשִׁים

🌊 רַשִׁ״י: ׳מִשְׂרְפוֹת מַיִם׳

הַמְּלָכִים בּוֹרְחִים עַד ׳מִשְׂרְפוֹת מַיִם׳. רַשִׁ״י מְבָאֵר בְּשֵׁם תַּרְגּוּם יוֹנָתָן: ׳חֲרִיצֵי יַמָּא׳ – ׳שֶׁעוֹשִׂין חֲרִיצִין וּמֵי הַיָּם יוֹצְאִין לְתוֹכָן, וְנִשְׂרָפִין מֵחוֹם הַשֶּׁמֶשׁ וְנַעֲשִׂים מֶלַח׳. מָקוֹם שֶׁהַשֶּׁמֶשׁ הוֹפֶכֶת בּוֹ מַיִם לְמֶלַח.

׳מִשְׂרְפוֹת מַיִם׳ – בְּרֵכוֹת שֶׁהַשֶּׁמֶשׁ הוֹפֶכֶת אֶת מֵימֵיהֶן לְמֶלַח.

2
🏰

📜 הַפָּסוּקכִּי חָצוֹר לְפָנִים הִיא רֹאשׁ כָּל הַמַּמְלָכוֹת הָאֵלֶּה

💛 אַהֲבַת הַשֵּׁם: ה׳ מְסַדֵּר גַּם אֶת הַדְּבָרִים הַגְּדוֹלִים וְגַם אֶת הַקְּטַנִּים 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק י״א, פְּסוּקִים י׳, י״ג (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – מַה אוֹמְרִים הַמְפָרְשִׁים

🏰 מְצוּדַת דָּוִד: ׳רֹאשׁ כָּל הַמַּמְלָכוֹת׳

מְצוּדַת דָּוִד מְבָאֵר: חָצוֹר הָיְתָה רֹאשׁ הַמַּמְלָכוֹת, ׳וּלְזֶה פָּתַח בָּהּ תְּחִלָּה׳, כִּי כְּשֶׁלּוֹקְחִים אֶת הָעִיר הַחֲשׁוּבָה, כָּל הַשְּׁאָר מִתְּפָּרֵק. וְרַשִׁ״י עַל ׳זוּלָתִי אֶת חָצוֹר לְבַדָּהּ׳: ׳בַּמָּסוֹרֶת שְׂרָפָהּ, מֹשֶׁה צִוָּה לוֹ וּמָסַר לוֹ כֵּן׳.

רַשִׁ״י: מֹשֶׁה צִוָּה לִשְׂרֹף אֶת חָצוֹר, וִיהוֹשֻׁעַ עָשָׂה כֵּן.

3
🧱

📜 הַפָּסוּקרַק כָּל הֶעָרִים הָעֹמְדוֹת עַל תִּלָּם לֹא שְׂרָפָם יִשְׂרָאֵל זוּלָתִי אֶת חָצוֹר לְבַדָּהּ שָׂרַף יְהוֹשֻׁעַ

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹמֵר עַל הַבָּתִּים וְעַל הָעָרִים שֶׁל יִשְׂרָאֵל 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק י״א, פָּסוּק י״ג (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – מַה אוֹמְרִים הַמְפָרְשִׁים

🧱 רַשִׁ״י: ׳עַל תִּלָּם׳

הַפָּסוּק אוֹמֵר: ׳הֶעָרִים הָעֹמְדוֹת עַל תִּלָּם׳. רַשִׁ״י מְבָאֵר: ׳שֶׁלֹּא נֶהֱרְסָה חוֹמָתָם, כְּמוֹ יְרִיחוֹ שֶׁנָּפְלָה חוֹמָתָהּ׳. מְצוּדַת דָּוִד: ׳בְּגָבְהָם, שֶׁלֹּא נָפְלוּ חוֹמוֹתֵיהֶם׳. הֶעָרִים נִשְׁאֲרוּ בְּמָקוֹמָן עַל הַתֵּל שֶׁלָּהֶן.

רַשִׁ״י: ׳עַל תִּלָּם׳ – שֶׁלֹּא נֶהֱרְסָה חוֹמָתָם.

4
⛰️

📜 הַפָּסוּקמִן הָהָר הֶחָלָק הָעוֹלֶה שֵׂעִיר וְעַד בַּעַל גָּד בְּבִקְעַת הַלְּבָנוֹן תַּחַת הַר חֶרְמוֹן

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לָנוּ אֶרֶץ, וַאֲנַחְנוּ יְכוֹלִים לְהוֹסִיף וְלִבְנוֹת בָּהּ 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק י״א, פָּסוּק י״ז (רַשִׁ״י)

רָמָה ב׳ – מַה אוֹמְרִים הַמְפָרְשִׁים

⛰️ רַשִׁ״י: ׳הָהָר הֶחָלָק׳

גְּבוּל הַכִּבּוּשׁ הוּא ׳מִן הָהָר הֶחָלָק הָעוֹלֶה שֵׂעִיר וְעַד בַּעַל גָּד׳. רַשִׁ״י מְבָאֵר: ׳טוּרָא שְׁעִיעַ׳, הַר חָלָק. וְעוֹד מְלַמֵּד: ׳וְלֹא הִסְפִּיק לִכְבּוֹשׁ בְּחַיָּיו אֶת כָּל הַמֶּצֶר עַד הַצָּפוֹן׳. יֵשׁ עוֹד עֲבוֹדָה לַדּוֹרוֹת הַבָּאִים.

רַשִׁ״י: יְהוֹשֻׁעַ לֹא הִסְפִּיק לִכְבּוֹשׁ אֶת כָּל הַצָּפוֹן, יֵשׁ עוֹד לְהַמְשִׁיךְ.

5
🔗

📜 הַפָּסוּקכַּאֲשֶׁר צִוָּה ה׳ אֶת מֹשֶׁה עַבְדּוֹ כֵּן צִוָּה מֹשֶׁה אֶת יְהוֹשֻׁעַ וְכֵן עָשָׂה יְהוֹשֻׁעַ לֹא הֵסִיר דָּבָר מִכֹּל אֲשֶׁר צִוָּה ה׳ אֶת מֹשֶׁה

💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן אֶת הַתּוֹרָה מִדּוֹר לְדוֹר, וְהִיא הִגִּיעָה גַּם אֵלַי 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק י״א, פָּסוּק ט״ו

רָמָה ב׳ – מַה אוֹמְרִים הַמְפָרְשִׁים

🔗 מִנַּיִן לָנוּ שֶׁיְּהוֹשֻׁעַ הִמְשִׁיךְ אֶת מֹשֶׁה?

מִן הַפָּסוּק: ׳כַּאֲשֶׁר צִוָּה ה׳ אֶת מֹשֶׁה עַבְדּוֹ, כֵּן צִוָּה מֹשֶׁה אֶת יְהוֹשֻׁעַ, וְכֵן עָשָׂה יְהוֹשֻׁעַ׳. הַשַּׁרְשֶׁרֶת: ה׳ אֶל מֹשֶׁה, מֹשֶׁה אֶל יְהוֹשֻׁעַ, וִיהוֹשֻׁעַ עוֹשֶׂה. ׳לֹא הֵסִיר דָּבָר׳ – שׁוּם דָּבָר לֹא חָסֵר.

ה׳ ← מֹשֶׁה ← יְהוֹשֻׁעַ: הַתּוֹרָה עוֹבֶרֶת מִדּוֹר לְדוֹר.

6
💛

📜 הַפָּסוּקוַיֹּאמֶר ה׳ אֶל יְהוֹשֻׁעַ אַל תִּירָא מִפְּנֵיהֶם כִּי מָחָר כָּעֵת הַזֹּאת אָנֹכִי נֹתֵן

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁה׳ אוֹמֵר ׳אַל תִּירָא׳, הַלֵּב שֶׁלִּי נִרְגָּע 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק י״א, פָּסוּק ו׳

רָמָה ב׳ – מַה אוֹמְרִים הַמְפָרְשִׁים

💛 ׳כִּי מָחָר כָּעֵת הַזֹּאת׳

ה׳ אוֹמֵר לִיהוֹשֻׁעַ ׳אַל תִּירָא מִפְּנֵיהֶם׳, וְלֹא רַק זֹאת. הוּא קוֹבֵעַ זְמַן: ׳כִּי מָחָר כָּעֵת הַזֹּאת׳. ה׳ אֵינוֹ מַבְטִיחַ סְתָם, הוּא אוֹמֵר בְּדִיּוּק מָתַי. כָּךְ הַלֵּב שֶׁל יְהוֹשֻׁעַ שָׁקֵט.

ה׳ מַבְטִיחַ בְּדִיּוּק, וְהַלֵּב נִרְגָּע.

מָה לוֹקְחִים מִרָמָה ב׳?

רַשִׁ״י וּמְצוּדַת דָּוִד מַסְבִּירִים מִלִּים קָשׁוֹת: כִּנְרוֹת, תִּלָּם, מִשְׂרְפוֹת מַיִם. הֵם מַרְאִים שֶׁיְּהוֹשֻׁעַ עָשָׂה בְּדִיּוּק מַה שֶּׁמֹּשֶׁה צִוָּה, וְה׳ הִבְטִיחַ וְקִיֵּם.

  • יְהוֹשֻׁעַ ׳לֹא הֵסִיר דָּבָר׳ מִמָּה שֶׁמֹּשֶׁה צִוָּה. אֵיזֶה דָּבָר קָטָן אֲנִי רוֹצֶה לִשְׁמֹר הַשָּׁנָה בְּדִיּוּק?

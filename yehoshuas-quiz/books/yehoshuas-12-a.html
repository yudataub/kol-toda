📜 סֵפֶר יְהוֹשֻׁעַ · וְהָאָרֶץ שָׁקְטָה · רָמָה א׳

וְהָאָרֶץ שָׁקְטָה

רָמָה א׳ – מֵי מֵרוֹם

מְלָכִים רַבִּים מִצְּפוֹן הָאָרֶץ נֶאֱסָפִים בְּמֵי מֵרוֹם, וְה׳ אוֹמֵר לִיהוֹשֻׁעַ ׳אַל תִּירָא׳. בַּסּוֹף ׳וְהָאָרֶץ שָׁקְטָה מִמִּלְחָמָה׳

📖 פְּרָקִים י״א–י״ב

🎯 לַחִידוֹן

💛 אַהֲבַת הַשֵּׁם: ה׳ יוֹדֵעַ מָה קוֹרֶה בְּכָל מָקוֹם, גַּם בַּמֶּרְחָקִים 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק י״א, פְּסוּקִים א׳–ג׳

רָמָה א׳ – מֵי מֵרוֹם

📨 יָבִין שׁוֹלֵחַ הוֹדָעָה

בְּצָפוֹן הָאָרֶץ יוֹשֵׁב יָבִין מֶלֶךְ חָצוֹר. כְּשֶׁהוּא שׁוֹמֵעַ מָה קָרָה בַּדָּרוֹם, הוּא שׁוֹלֵחַ הוֹדָעָה לְמַלְכֵי הַצָּפוֹן, וְלִמְלָכִים שֶׁבָּהָר וּבָעֲרָבָה וּבַשְּׁפֵלָה. הוּא רוֹצֶה שֶׁכֻּלָּם יִתְאַחֲדוּ.

יָבִין מֶלֶךְ חָצוֹר מַזְמִין מְלָכִים רַבִּים לְהִתְאַחֵד.

1

💛 אַהֲבַת הַשֵּׁם: אֲפִלּוּ כְּשֶׁהַצָּבָא נִרְאֶה גָּדוֹל מְאֹד, ה׳ גָּדוֹל מִכֻּלָּם 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק י״א, פְּסוּקִים ד׳–ה׳

רָמָה א׳ – מֵי מֵרוֹם

🐎 עַם רָב כַּחוֹל

הַמְּלָכִים וְהַצְּבָאוֹת יוֹצְאִים: ׳עַם רָב כַּחוֹל אֲשֶׁר עַל שְׂפַת הַיָּם לָרֹב׳. יֵשׁ לָהֶם גַּם ׳סוּס וָרֶכֶב רַב מְאֹד׳. כֻּלָּם נִפְגָּשִׁים וּמַחֲנִים יַחַד אֶל ׳מֵי מֵרוֹם׳.

צָבָא עֲצוּם כְּמוֹ חוֹל הַיָּם מִתְאַסֵּף בְּמֵי מֵרוֹם.

2

💛 אַהֲבַת הַשֵּׁם: ה׳ מַבְטִיחַ וּמְקַיֵּם. אֲנִי בּוֹטֵחַ בּוֹ 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק י״א, פָּסוּק ו׳

רָמָה א׳ – מֵי מֵרוֹם

💪 ׳אַל תִּירָא מִפְּנֵיהֶם׳

ה׳ אוֹמֵר לִיהוֹשֻׁעַ: ׳אַל תִּירָא מִפְּנֵיהֶם׳. ה׳ גַּם אוֹמֵר מָתַי: ׳כִּי מָחָר כָּעֵת הַזֹּאת׳. עוֹד הוּא אוֹמֵר לַעֲקֹר אֶת הַסּוּסִים וְלִשְׂרֹף אֶת הַמֶּרְכָּבוֹת, כְּדֵי שֶׁלֹּא יִהְיֶה לָהֶם יוֹתֵר כְּלֵי מִלְחָמָה.

ה׳ מַבְטִיחַ לִיהוֹשֻׁעַ: ׳אַל תִּירָא׳, מָחָר זֶה יִגָּמֵר.

3

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי הַצְלָחָה בְּדִיּוּק בַּזְּמַן שֶׁהוּא הִבְטִיחַ 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק י״א, פְּסוּקִים ז׳–ח׳

רָמָה א׳ – מֵי מֵרוֹם

⚡ פִּתְאֹם

יְהוֹשֻׁעַ וְהָעָם בָּאִים ׳עֲלֵיהֶם עַל מֵי מֵרוֹם פִּתְאֹם׳. ׳וַיִּתְּנֵם ה׳ בְּיַד יִשְׂרָאֵל׳. הַמְּלָכִים בּוֹרְחִים עַד צִידוֹן רַבָּה, וְעַד ׳מִשְׂרְפוֹת מַיִם׳ וְעַד בִּקְעַת מִצְפֶּה. הַנִּצָּחוֹן בָּא מֵה׳.

יְהוֹשֻׁעַ בָּא פִּתְאֹם, וְה׳ נוֹתֵן אֶת הַנִּצָּחוֹן.

4

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹהֵב לַעֲשׂוֹת מַה שֶּׁה׳ מְבַקֵּשׁ, כִּי אֲנִי יוֹדֵעַ שֶׁזֶּה לְטוֹבָתִי 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק י״א, פָּסוּק ט׳

רָמָה א׳ – מֵי מֵרוֹם

🔥 עוֹשִׂים כְּמוֹ שֶׁה׳ אָמַר

הַפָּסוּק אוֹמֵר: ׳וַיַּעַשׂ לָהֶם יְהוֹשֻׁעַ כַּאֲשֶׁר אָמַר לוֹ ה׳׳. הוּא עוֹקֵר אֶת הַסּוּסִים וְשׂוֹרֵף אֶת הַמֶּרְכָּבוֹת, בְּדִיּוּק כְּמוֹ שֶׁה׳ אָמַר. אֵין עוֹד כְּלֵי מִלְחָמָה לַצָּבָא הַזֶּה.

יְהוֹשֻׁעַ עוֹשֶׂה בְּדִיּוּק מַה שֶּׁה׳ אָמַר.

5

💛 אַהֲבַת הַשֵּׁם: ה׳ מְסַדֵּר אֶת הָעוֹלָם בְּסֵדֶר, וַאֲנִי סוֹמֵךְ עָלָיו 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק י״א, פְּסוּקִים י׳–י״ג

רָמָה א׳ – מֵי מֵרוֹם

🏰 חָצוֹר – רֹאשׁ הַמַּמְלָכוֹת

הַפָּסוּק מוֹסִיף: ׳כִּי חָצוֹר לְפָנִים הִיא רֹאשׁ כָּל הַמַּמְלָכוֹת הָאֵלֶּה׳. הִיא הָעִיר הַחֲשׁוּבָה בַּצָּפוֹן. יְהוֹשֻׁעַ לוֹקֵחַ אוֹתָהּ ׳וְאֶת חָצוֹר שָׂרַף בָּאֵשׁ׳. שְׁאָר הֶעָרִים נִשְׁאָרוֹת עוֹמְדוֹת עַל תִּלָּן.

חָצוֹר הָיְתָה רֹאשׁ הַמַּמְלָכוֹת, וְהִיא הָעִיר שֶׁנִּשְׂרְפָה.

6

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי שׁוֹמֵר אֶת הַמִּצְווֹת, אֲנִי כְּמוֹ יְהוֹשֻׁעַ, וְה׳ שָׂמֵחַ בִּי 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק י״א, פְּסוּקִים י״ד–ט״ו

רָמָה א׳ – מֵי מֵרוֹם

📜 ׳לֹא הֵסִיר דָּבָר׳

יִשְׂרָאֵל לוֹקְחִים אֶת הַשָּׁלָל וְהַבְּהֵמָה כְּמוֹ שֶׁהֻתַּר. הַפָּסוּק מְסַכֵּם: ׳כַּאֲשֶׁר צִוָּה ה׳ אֶת מֹשֶׁה עַבְדּוֹ כֵּן צִוָּה מֹשֶׁה אֶת יְהוֹשֻׁעַ וְכֵן עָשָׂה יְהוֹשֻׁעַ׳. הוּא ׳לֹא הֵסִיר דָּבָר׳ מִכָּל מַה שֶּׁה׳ צִוָּה.

יְהוֹשֻׁעַ עָשָׂה בְּדִיּוּק כְּמוֹ שֶׁמֹּשֶׁה לִמֵּד אוֹתוֹ.

7

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן שָׁלוֹם אַחֲרֵי הַמַּאֲמָץ, וַאֲנִי מוֹדֶה לוֹ 💛

📖 יְהוֹשֻׁעַ, פֶּרֶק י״א, פְּסוּקִים ט״ז–י״ח, כ״ג; פֶּרֶק י״ב, פָּסוּק כ״ד

רָמָה א׳ – מֵי מֵרוֹם

🌿 ׳וְהָאָרֶץ שָׁקְטָה׳

יְהוֹשֻׁעַ לוֹקֵחַ אֶת הָאָרֶץ ׳כְּכֹל אֲשֶׁר דִּבֶּר ה׳ אֶל מֹשֶׁה׳. הַמִּלְחָמוֹת נִמְשְׁכוּ ׳יָמִים רַבִּים׳. בַּסּוֹף ׳וְהָאָרֶץ שָׁקְטָה מִמִּלְחָמָה׳. פֶּרֶק י״ב מְסַכֵּם: ׳כָּל מְלָכִים שְׁלֹשִׁים וְאֶחָד׳.

בַּסּוֹף ׳וְהָאָרֶץ שָׁקְטָה מִמִּלְחָמָה׳.

8

מָה לוֹקְחִים מִרָמָה א׳?

צָבָא גָּדוֹל כַּחוֹל אֲשֶׁר עַל שְׂפַת הַיָּם, וִיהוֹשֻׁעַ שׁוֹמֵעַ מֵה׳: ׳אַל תִּירָא׳. ה׳ אוֹהֵב אוֹתָנוּ, מַבְטִיחַ וּמְקַיֵּם, וִיהוֹשֻׁעַ עוֹשֶׂה בְּדִיּוּק כְּמוֹ שֶׁצֻּוָּה.

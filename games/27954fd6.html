בַּיִת 1 - אֶנֶרְגֶטִי
עַמְרָם וְיוֹכֶבֶד, מִשְׁפַּחַת לֵוִי
תִּינוֹק נוֹלַד, יָפֶה וְטוֹב מְאוֹד כִּי
בַּיִת מָלֵא אוֹרָה, נֵס גָּדוֹל וְנִפְלָא
שְׁלוֹשָׁה חֳדָשִׁים הֶחְבִּיאוּ, צָפְנוּ וְהִצִּילוּ תִּקְוָה
טְרוּם פִּזְמוֹן - בּוֹנֶה הִתְרַגְּשׁוּת
תֵּיבַת גֹּמֶא בַּיְאוֹר, מִרְיָם שׁוֹמֶרֶת מֵרָחוֹק
בַּת פַּרְעֹה מוֹצֵאת, וְלִבָּהּ רַךְ וְחָזָק
"מִן הַמַּיִם מְשִׁיתִיהוּ" - מֹשֶׁה שְׁמוֹ קָרְאָה
הַשֵּׁם הַגָּדוֹל שָׁמַר עָלָיו, גָּאֲלוֹ מִן הַצָּרָה
פִּזְמוֹן - מְאוֹד תּוֹפֵס ⭐
מֹשֶׁה מֹשֶׁה רַבֵּנוּ הַגָּדוֹל
נוֹלַד בְּנֵס, גָּדַל בְּאוֹמֶץ וּבְכֹחַ מֵהַשֵּׁם הַגָּדוֹל
יָצָא אֶל אֶחָיו, רָאָה בְּסִבְלוֹתָם
אָהַב אוֹתָם בְּכָל לִבּוֹ, עָזַר וְהֵקֵל מֵעֲמָלָם
בַּיִת 2
בָּאַרְמוֹן גָּדַל, אֲבָל לֹא שָׁכַח
אֶת עַמּוֹ בְּצָרָה, אֶת אֶחָיו הִזְדָּרֵז לִפְקֹד כָּרָה
נָתַן כְּתֵפוֹ לַעֲזוֹר, סִבֵּל עִמָּהֶם יַחַד
עַל הַכָּבוֹד וְהַתַּעֲנוּגִים, וִיתֵּר בְּלֵב אֶחָד
טְרוּם פִּזְמוֹן
רָשָׁע הִכָּה עִבְרִי, מֹשֶׁה הִצִּילוֹ מִמָּוֶת
בָּרַח לְמִדְיָן הָרְחוֹקָה, הַשֵּׁם עוֹזֵר בָּעֵת
בְּנוֹת יִתְרוֹ הוֹשִׁיעַ, הִשְׁקָה אֶת צֹאנָן
אוֹמֶץ לֵב וְטוּב לֵב, זוֹ דַּרְכּוֹ תָּמִיד כָּאן
פִּזְמוֹן - אֶנֶרְגְיָה גְבוֹהָה ⭐⭐
מֹשֶׁה מֹשֶׁה רַבֵּנוּ הַגָּדוֹל

גֶּשֶׁר - יוֹתֵר אִיטִי, רֶגַע לִמּוּד
מִרְיָם הָאָחוֹת, עָמְדָה בְּגִבּוּרָה
צִפּוֹרָה הָאִשָּׁה, זָרְזָה לִקְרֹא בְּעֶזְרָה
יוֹכֶבֶד הָאִמָּא, נָתְנָה אַהֲבָה וֶאֱמוּנָה
בַּת פַּרְעֹה גִּדְּלָה, חָמְלָה בְּמֶסֶר נַפְשָׁהּ תְּמִימָה

בַּיִת 3 - מְנַצֵּחַ
"גֵּר הָיִיתִי בְּאֶרֶץ נָכְרִיָּה"
גֵּרְשֹׁם בְּנוֹ קָרָא, לֹא שָׁכַח אֶת אֶחָיו בַּמַּצוֹר וּבַצָּרָה
רֹעֶה צֹאן בַּמִּדְבָּר, לוֹמֵד וּמִתְכּוֹנֵן
הַשֵּׁם עִמּוֹ תָּמִיד, לִגְאֻלָּה הוּא יִבָּחֵר וְיִקָּרֵא פֶּן
פִּזְמוֹן סִיּוּם - עִם הַרְמוֹנִיּוֹת וּמְחִיאוֹת כַּפַּיִם ⭐⭐⭐

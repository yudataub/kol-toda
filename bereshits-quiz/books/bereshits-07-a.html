עֵשָׂו כּוֹעֵס עַל יַעֲקֹב, וְיִצְחָק שׁוֹלֵחַ אוֹתוֹ לְחָרָן, אֶל לָבָן אֲחִי אִמּוֹ. ׳וַיֵּצֵא יַעֲקֹב מִבְּאֵר שָׁבַע וַיֵּלֶךְ חָרָנָה׳. הוּא הוֹלֵךְ לְבַד, עִם מַקֵּל בַּיָּד, בְּדֶרֶךְ אֲרֻכָּה.

💛 אַהֲבַת הַשֵּׁם: ה׳ מְבָרֵךְ אֶת מַעֲשֵׂה הַיָּדַיִם שֶׁלִּי 💛

📖 בְּרֵאשִׁית, פֶּרֶק ל׳, פְּסוּקִים כ״ה–מ״ג

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🐑 ׳וַיִּפְרֹץ הָאִישׁ מְאֹד מְאֹד׳

יַעֲקֹב רוֹצֶה לַחֲזֹר הַבַּיְתָה. רַשִׁ״י מְסַבֵּר: הוּא לֹא רָצָה לָצֵאת בְּלִי רְשׁוּת. לָבָן מְבַקֵּשׁ: הִשָּׁאֵר, ׳וַיְבָרְכֵנִי ה׳ בִּגְלָלֶךָ׳. הַכְּבָשִׂים וְהָעִזִּים הַנְּקֻדִּים יִהְיוּ שֶׁל יַעֲקֹב – וְה׳ מְבָרֵךְ אוֹתוֹ: ׳וַיִּפְרֹץ הָאִישׁ מְאֹד מְאֹד׳.

ה׳ מְבָרֵךְ אֶת מִי שֶׁעוֹבֵד בְּיֹשֶׁר.

💛 אַהֲבַת הַשֵּׁם: ה׳ קוֹרֵא לָנוּ הַבַּיְתָה – לְאֶרֶץ יִשְׂרָאֵל 💛

📖 בְּרֵאשִׁית, פֶּרֶק ל״א, פְּסוּקִים א׳–ט״ז

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🏠 ׳שׁוּב אֶל אֶרֶץ אֲבוֹתֶיךָ׳

יַעֲקֹב רוֹאֶה שֶׁלָּבָן כְּבָר לֹא מִסְתַּכֵּל עָלָיו יָפֶה ׳כִּתְמוֹל שִׁלְשׁוֹם׳. ה׳ אוֹמֵר לוֹ: ׳שׁוּב אֶל אֶרֶץ אֲבוֹתֶיךָ... וְאֶהְיֶה עִמָּךְ׳. יַעֲקֹב קוֹרֵא לְרָחֵל וּלְלֵאָה לַשָּׂדֶה, וְהֵן עוֹנוֹת: ׳כֹּל אֲשֶׁר אָמַר אֱלֹהִים אֵלֶיךָ עֲשֵׂה׳.

כָּל הַמִּשְׁפָּחָה מַסְכִּימָה לַעֲשׂוֹת מָה שֶׁה׳ אוֹמֵר.

💛 אַהֲבַת הַשֵּׁם: ה׳ מֵגֵן עָלַי מִפְּנֵי מִי שֶׁרוֹצֶה לְהָרַע לִי 💛

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🐫 לָבָן רוֹדֵף

יַעֲקֹב מַרְכִּיב אֶת הַמִּשְׁפָּחָה עַל הַגְּמַלִּים וְיוֹצֵא לַדֶּרֶךְ. אַחֲרֵי שְׁלֹשָׁה יָמִים לָבָן שׁוֹמֵעַ, וְרוֹדֵף אַחֲרָיו עַד הַר הַגִּלְעָד. אֲבָל בַּלַּיְלָה ה׳ בָּא אֵלָיו בַּחֲלוֹם: ׳הִשָּׁמֶר לְךָ פֶּן תְּדַבֵּר עִם יַעֲקֹב מִטּוֹב עַד רָע׳.

ה׳ שׁוֹמֵר עַל יַעֲקֹב מִפְּנֵי לָבָן.

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🔍 לָבָן מְחַפֵּשׂ

לָבָן כּוֹעֵס: ׳לָמָּה גָנַבְתָּ אֶת אֱלֹהָי?׳ – רָחֵל לָקְחָה אֶת הַתְּרָפִים שֶׁל אָבִיהָ. רַשִׁ״י מְסַבֵּר: הִיא רָצְתָה ׳לְהַפְרִישׁ אֶת אָבִיהָ מֵעֲבוֹדָה זָרָה׳. לָבָן מְחַפֵּשׂ בְּכָל הָאֹהָלִים – וְלֹא מוֹצֵא.

רָחֵל רָצְתָה שֶׁאָבִיהָ יַפְסִיק לַעֲבֹד עֲבוֹדָה זָרָה.

💛 אַהֲבַת הַשֵּׁם: ה׳ רָאָה כַּמָּה יַעֲקֹב עָבַד – וְרוֹאֶה גַּם אוֹתִי 💛

📖 בְּרֵאשִׁית, פֶּרֶק ל״א, פְּסוּקִים ל״ו–מ״ב

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

☀️ ׳בַיּוֹם אֲכָלַנִי חֹרֶב׳

עַכְשָׁו יַעֲקֹב מְדַבֵּר: ׳זֶה עֶשְׂרִים שָׁנָה אָנֹכִי עִמָּךְ׳. בַּיּוֹם הָיָה לִי חַם, בַּלַּיְלָה קַר, וְלֹא יָשַׁנְתִּי – כְּדֵי לִשְׁמֹר עַל הַצֹּאן שֶׁלְּךָ. רַשִׁ״י מְסַבֵּר: כָּל כֶּבֶשׂ שֶׁנִּגְנַב, בַּיּוֹם אוֹ בַּלַּיְלָה – ׳הַכֹּל שִׁלַּמְתִּי׳.

יַעֲקֹב שָׁמַר עַל הַצֹּאן שֶׁל לָבָן כְּמוֹ עַל שֶׁלּוֹ.

💛 אַהֲבַת הַשֵּׁם: ה׳ עוֹשֶׂה שָׁלוֹם גַּם עִם מִי שֶׁרָדַף אַחֲרַי 💛

📖 בְּרֵאשִׁית, פֶּרֶק ל״א, פְּסוּקִים מ״ג–נ״ד

🗿 גַּל עֵד

לָבָן אוֹמֵר: בּוֹא נַעֲשֶׂה בְּרִית. יַעֲקֹב מַעֲמִיד אֶבֶן, וְכֻלָּם עוֹשִׂים גַּל שֶׁל אֲבָנִים. לָבָן קוֹרֵא לוֹ ׳יְגַר שָׂהֲדוּתָא׳ בַּאֲרָמִית, וְיַעֲקֹב קוֹרֵא לוֹ ׳גַּלְעֵד׳ – רַשִׁ״י: ׳גַּל עֵד׳. בַּבֹּקֶר לָבָן מְנַשֵּׁק אֶת הַנְּכָדִים, מְבָרֵךְ וְחוֹזֵר לִמְקוֹמוֹ.

הַגַּל עֵד שֶׁשּׁוּם צַד לֹא יַזִּיק לַשֵּׁנִי.

⛺ מַחֲנַיִם

יַעֲקֹב מַמְשִׁיךְ בַּדֶּרֶךְ הַבַּיְתָה, ׳וַיִּפְגְּעוּ בוֹ מַלְאֲכֵי אֱלֹהִים׳. רַשִׁ״י מְסַבֵּר: הַמַּלְאָכִים שֶׁל אֶרֶץ יִשְׂרָאֵל בָּאוּ לִקְרָאתוֹ, לְלַוּוֹת אוֹתוֹ לָאָרֶץ. יַעֲקֹב קוֹרֵא לַמָּקוֹם ׳מַחֲנָיִם׳ – שְׁנֵי מַחֲנוֹת.

יַעֲקֹב חוֹזֵר הַבַּיְתָה – וְה׳ שׁוֹלֵחַ לוֹ שׁוֹמְרִים.

מָה לוֹקְחִים מִרָמָה א׳?

יַעֲקֹב יָצָא לְבַד, עִם מַקֵּל, וְחָזַר עִם מִשְׁפָּחָה גְּדוֹלָה. בְּכָל הַדֶּרֶךְ – בַּלַּיְלָה עַל הָאֶבֶן, אֵצֶל לָבָן הָרַמַּאי וּבַבְּרִיחָה – ה׳ קִיֵּם אֶת הַהַבְטָחָה: ׳וְהִנֵּה אָנֹכִי עִמָּךְ׳.

בִּלְעָדָי אֱלֹהִים יַעֲנֶה אֶת שְׁלוֹם פַּרְעֹה

בִּלְעָדָי – אֱלֹהִים יַעֲנֶה

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

כָּל סִפּוּר פָּרָשַׁת מִקֵּץ: יוֹסֵף יוֹצֵא מִבֵּית הַסֹּהַר, פּוֹתֵר אֶת חֲלוֹמוֹת פַּרְעֹה, נַעֲשֶׂה מִשְׁנֶה לַמֶּלֶךְ – וְהָאַחִים יוֹרְדִים לְמִצְרַיִם

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹטֵחַ בַּה׳, וְהוּא שׁוֹמֵר עָלַי 💛

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

⛓️ ׳וַיְהִי מִקֵּץ שְׁנָתַיִם׳

יוֹסֵף יוֹשֵׁב בְּבֵית הַסֹּהַר. שַׂר הַמַּשְׁקִים שָׁכַח אוֹתוֹ, וְעָבְרוּ עוֹד שְׁנָתַיִם. רַשִׁ״י מְסַבֵּר: יוֹסֵף בִּקֵּשׁ מִשַּׂר הַמַּשְׁקִים שֶׁיִּזְכֹּר אוֹתוֹ – וַאֲנַחְנוּ בּוֹטְחִים רַק בַּה׳. ׳אַשְׁרֵי הַגֶּבֶר אֲשֶׁר שָׂם ה׳ מִבְטַחוֹ׳.

מִי שֶׁבָּאֱמֶת יָכוֹל לַעֲזֹר – זֶה ה׳.

🐄

💛 אַהֲבַת הַשֵּׁם: ה׳ מְגַלֶּה מָה שֶׁיִּהְיֶה, כְּדֵי לִשְׁמֹר עַל הָעוֹלָם 💛

📖 בְּרֵאשִׁית, פֶּרֶק מ״א, פְּסוּקִים א׳–ד׳

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🐄 שֶׁבַע פָּרוֹת

פַּרְעֹה חוֹלֵם: הוּא עוֹמֵד עַל הַיְאֹר – נְהַר הַנִּילוּס. עוֹלוֹת שֶׁבַע פָּרוֹת יָפוֹת וּשְׁמֵנוֹת, וְאַחֲרֵיהֶן שֶׁבַע פָּרוֹת רָזוֹת. הָרָזוֹת אוֹכְלוֹת אֶת הַשְּׁמֵנוֹת! רַשִׁ״י מְסַבֵּר: זֶה סִימָן שֶׁכָּל הַשִּׂמְחָה שֶׁל הַשָּׂבָע תִּשָּׁכַח בִּשְׁנוֹת הָרָעָב.

חֲלוֹם מוּזָר – וְיֵשׁ לוֹ פִּתְרוֹן.

🌾

💛 אַהֲבַת הַשֵּׁם: ה׳ מְכִין אֶת הַדֶּרֶךְ לְיוֹסֵף, צַעַד אַחַר צַעַד 💛

📖 בְּרֵאשִׁית, פֶּרֶק מ״א, פְּסוּקִים ה׳–ח׳

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🌾 שֶׁבַע שִׁבֳּלִים

פַּרְעֹה נִרְדַּם שׁוּב וְחוֹלֵם: שֶׁבַע שִׁבֳּלִים מְלֵאוֹת, וְשֶׁבַע שִׁבֳּלִים דַּקּוֹת שֶׁבּוֹלְעוֹת אוֹתָן. בַּבֹּקֶר ׳וַתִּפָּעֶם רוּחוֹ׳ – הוּא מְבֹהָל. הַחֲכָמִים מְנַסִּים לִפְתֹּר. רַשִׁ״י מְסַבֵּר: הֵם פָּתְרוּ, אֲבָל הַפִּתְרוֹן לֹא נִכְנַס לָאָזְנַיִם שֶׁל פַּרְעֹה.

רַק פִּתְרוֹן אֲמִתִּי יַרְגִּיעַ אֶת פַּרְעֹה.

3
🍷

💛 אַהֲבַת הַשֵּׁם: ה׳ זוֹכֵר אוֹתִי, גַּם כְּשֶׁנִּדְמֶה שֶׁכֻּלָּם שָׁכְחוּ 💛

📖 בְּרֵאשִׁית, פֶּרֶק מ״א, פְּסוּקִים ט׳–י״ג

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🍷 שַׂר הַמַּשְׁקִים נִזְכָּר

פִּתְאוֹם שַׂר הַמַּשְׁקִים נִזְכָּר: ׳אֶת חֲטָאַי אֲנִי מַזְכִּיר הַיּוֹם׳. הוּא מְסַפֵּר: בְּבֵית הַסֹּהַר הָיָה ׳נַעַר עִבְרִי עֶבֶד׳, וְהוּא פָּתַר לָנוּ חֲלוֹמוֹת. רַשִׁ״י מְסַבֵּר: גַּם עַכְשָׁו הוּא מְדַבֵּר עַל יוֹסֵף בְּזִלְזוּל – ׳נַעַר׳, ׳עִבְרִי׳, ׳עֶבֶד׳.

גַּם כְּשֶׁאֲנָשִׁים שׁוֹכְחִים – ה׳ לֹא שׁוֹכֵחַ.

🏃

💛 אַהֲבַת הַשֵּׁם: ה׳ יָכוֹל לְהוֹצִיא אוֹתִי מִכָּל מָקוֹם קָשֶׁה 💛

📖 בְּרֵאשִׁית, פֶּרֶק מ״א, פָּסוּק י״ד

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🏃 מֵהַבּוֹר – אֶל הַמֶּלֶךְ

פַּרְעֹה שׁוֹלֵחַ, ׳וַיְרִיצֻהוּ מִן הַבּוֹר׳ – מְמַהֲרִים לְהוֹצִיא אֶת יוֹסֵף. ׳וַיְגַלַּח וַיְחַלֵּף שִׂמְלֹתָיו׳. רַשִׁ״י מְסַבֵּר: יוֹסֵף הִסְתַּפֵּר וְהֶחֱלִיף בְּגָדִים מִפְּנֵי כְּבוֹד הַמַּלְכוּת. אַחֲרֵי שָׁנִים קָשׁוֹת – בְּיוֹם אֶחָד הַכֹּל מִשְׁתַּנֶּה.

הַיְשׁוּעָה שֶׁל ה׳ יְכוֹלָה לָבוֹא בְּרֶגַע.

5
🙏

💛 אַהֲבַת הַשֵּׁם: כָּל מָה שֶׁאֲנִי יוֹדֵעַ – מַתָּנָה מֵה׳ 💛

📖 בְּרֵאשִׁית, פֶּרֶק מ״א, פְּסוּקִים ט״ו–ט״ז

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🙏 ׳בִּלְעָדָי׳

פַּרְעֹה אוֹמֵר: שָׁמַעְתִּי שֶׁאַתָּה יוֹדֵעַ לִפְתֹּר חֲלוֹמוֹת. יוֹסֵף עוֹנֶה: ׳בִּלְעָדָי, אֱלֹהִים יַעֲנֶה אֶת שְׁלוֹם פַּרְעֹה׳. רַשִׁ״י מְסַבֵּר: ׳אֵין הַחָכְמָה מִשֶּׁלִּי׳ – ה׳ יִתֵּן תְּשׁוּבָה בַּפֶּה שֶׁלִּי. גַּם מוּל הַמֶּלֶךְ הֶחָזָק, יוֹסֵף מְדַבֵּר עַל ה׳.

יוֹסֵף לֹא לוֹקֵחַ אֶת הַכָּבוֹד לְעַצְמוֹ.

6
🔮

💛 אַהֲבַת הַשֵּׁם: ה׳ מַנְהִיג אֶת הָעוֹלָם, וְיוֹדֵעַ מָה יִהְיֶה 💛

📖 בְּרֵאשִׁית, פֶּרֶק מ״א, פְּסוּקִים כ״ה–ל״ב

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🔮 הַפִּתְרוֹן

יוֹסֵף פּוֹתֵר: ׳אֵת אֲשֶׁר הָאֱלֹהִים עֹשֶׂה הִגִּיד לְפַרְעֹה׳. שֶׁבַע הַפָּרוֹת הַטּוֹבוֹת – שֶׁבַע שָׁנִים שֶׁל שֹׂבַע, הַרְבֵּה אֹכֶל. אַחֲרֵיהֶן – שֶׁבַע שְׁנוֹת רָעָב. וְלָמָּה פַּעֲמַיִם? ׳וּמְמַהֵר הָאֱלֹהִים לַעֲשׂוֹתוֹ׳ – זֶה יָבוֹא בְּקָרוֹב.

שְׁנֵי הַחֲלוֹמוֹת הֵם בְּעֶצֶם חֲלוֹם אֶחָד.

🏛️

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי שֵׂכֶל לְהִתְכּוֹנֵן מֵרֹאשׁ 💛

📖 בְּרֵאשִׁית, פֶּרֶק מ״א, פְּסוּקִים ל״ג–ל״ו

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🏛️ ׳אִישׁ נָבוֹן וְחָכָם׳

יוֹסֵף נוֹתֵן עֵצָה: ׳יֵרֶא פַרְעֹה אִישׁ נָבוֹן וְחָכָם׳ – שֶׁיִּמְצָא אִישׁ חָכָם וִימַנֶּה אוֹתוֹ עַל מִצְרַיִם. בִּשְׁנוֹת הַשָּׂבָע יַאַסְפוּ אֹכֶל וְיִשְׁמְרוּ בֶּעָרִים. וְכָךְ בִּשְׁנוֹת הָרָעָב יִהְיֶה לֶחֶם לְכֻלָּם.

חָכָם חוֹשֵׁב מֵרֹאשׁ עַל מָה שֶׁיִּהְיֶה.

📖 בְּרֵאשִׁית, פֶּרֶק מ״א, פְּסוּקִים ל״ז–מ״ו

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

👑 מִשְׁנֶה לַמֶּלֶךְ

פַּרְעֹה אוֹמֵר: ׳הֲנִמְצָא כָזֶה אִישׁ אֲשֶׁר רוּחַ אֱלֹהִים בּוֹ?׳ וְאוֹמֵר לְיוֹסֵף: ׳אֵין נָבוֹן וְחָכָם כָּמוֹךָ׳. הוּא נוֹתֵן לְיוֹסֵף טַבַּעַת, בִּגְדֵי שֵׁשׁ וְשַׁרְשֶׁרֶת זָהָב, וּמַרְכִּיב אוֹתוֹ בַּמֶּרְכָּבָה הַשְּׁנִיָּה. ׳רַק הַכִּסֵּא אֶגְדַּל מִמֶּךָּ׳ – יוֹסֵף הוּא הַשֵּׁנִי לַמֶּלֶךְ.

יוֹסֵף בֶּן שְׁלוֹשִׁים, וְהוּא מוֹשֵׁל עַל כָּל מִצְרַיִם.

🧺 ׳כְּחוֹל הַיָּם׳

בְּשֶׁבַע שְׁנוֹת הַשָּׂבָע הָאָרֶץ נוֹתֶנֶת הַרְבֵּה תְּבוּאָה. יוֹסֵף אוֹסֵף אֹכֶל בְּכָל עִיר. רַשִׁ״י מְסַבֵּר: הוּא שָׂם בַּתְּבוּאָה מֵעֲפַר הַמָּקוֹם, כְּדֵי שֶׁלֹּא תֵּרָקֵב. יֵשׁ כָּל כָּךְ הַרְבֵּה – ׳כְּחוֹל הַיָּם׳, עַד שֶׁהִפְסִיקוּ לִסְפֹּר!

יוֹסֵף שׁוֹמֵר אֶת הַטּוֹב לַיָּמִים הַקָּשִׁים.

👶 מְנַשֶּׁה וְאֶפְרַיִם

יוֹסֵף הִתְחַתֵּן, וְנוֹלְדוּ לוֹ שְׁנֵי בָּנִים. לַגָּדוֹל קָרָא ׳מְנַשֶּׁה׳ – ׳כִּי נַשַּׁנִי אֱלֹהִים אֶת כָּל עֲמָלִי׳: ה׳ עָזַר לִי לִשְׁכֹּחַ אֶת הַצָּרוֹת. לַקָּטָן קָרָא ׳אֶפְרַיִם׳ – ׳כִּי הִפְרַנִי אֱלֹהִים בְּאֶרֶץ עָנְיִי׳: ה׳ הִצְמִיחַ אוֹתִי גַּם בְּמָקוֹם קָשֶׁה.

שְׁנֵי הַשֵּׁמוֹת הֵם תּוֹדָה לַה׳.

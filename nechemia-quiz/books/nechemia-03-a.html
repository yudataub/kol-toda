הַגְּדוֹלִים נוֹתְנִים דֻּגְמָה – הֵם הַמַּתְחִילִים.

כָּל אֶחָד בּוֹנֶה לְיַד הַבַּיִת שֶׁלּוֹ – וְכָךְ הָעֲבוֹדָה נִגְמֶרֶת מַהֵר.

אֲנָשִׁים פְּשׁוּטִים מִתְּקוֹעַ בָּנוּ פַּעֲמַיִם – גַּם כְּשֶׁהַגְּדוֹלִים לֹא רָצוּ.

הָאוֹיְבִים לוֹעֲגִים: ׳אַפִּילּוּ שׁוּעָל יַפִּיל אֶת הַחוֹמָה׳.

הַתְּשׁוּבָה לַלַּעַג הִיא תְּפִלָּה – וְאַחֲרֶיהָ מַמְשִׁיכִים לִבְנוֹת.

בִּגְלַל הַלֵּב שֶׁל הָעָם – הַחוֹמָה עוֹלָה עַד חֶצְיָהּ.

הָאוֹיְבִים מִתְאַגְּדִים: ׳וַיִּקְשְׁרוּ כֻלָּם יַחְדָּו לָבוֹא לְהִלָּחֵם בִּירוּשָׁלִָם וְלַעֲשׂוֹת לוֹ תּוֹעָה׳. נְחֶמְיָה אוֹמֵר: ׳וַנִּתְפַּלֵּל אֶל אֱלֹהֵינוּ וַנַּעֲמִיד מִשְׁמָר עֲלֵיהֶם יוֹמָם וָלַיְלָה׳ וְגַם ׳אֶל הַחֹרִים וְאֶל הַסְּגָנִים וְאֶל יֶתֶר הָעָם אַל תִּירְאוּ מִפְּנֵיהֶם אֶת אֲדֹנָי הַגָּדוֹל וְהַנּוֹרָא זְכֹרוּ׳.

נְחֶמְיָה מִתְפַּלֵּל וּמַעֲמִיד שׁוֹמְרִים – שְׁנֵיהֶם יַחַד.

⚔️ חֶרֶב בְּיָד אַחַת

מִן הַיּוֹם הַהוּא: ׳חֲצִי נְעָרַי עֹשִׂים בַּמְּלָאכָה וְחֶצְיָם מַחֲזִיקִים׳. ׳הַבּוֹנִים בַּחוֹמָה וְהַנֹּשְׂאִים בַּסֶּבֶל עֹמְשִׂים בְּאַחַת יָדוֹ עֹשֶׂה בַמְּלָאכָה וְאַחַת מַחֲזֶקֶת הַשָּׁלַח׳. וְתוֹקֵעַ הַשּׁוֹפָר עוֹמֵד לְיַד נְחֶמְיָה: ׳בִּמְקוֹם אֲשֶׁר תִּשְׁמְעוּ אֶת קוֹל הַשּׁוֹפָר שָׁמָּה תִּקָּבְצוּ אֵלֵינוּ אֱלֹהֵינוּ יִלָּחֶם לָנוּ׳. עוֹבְדִים ׳מֵעֲלוֹת הַשַּׁחַר עַד צֵאת הַכּוֹכָבִים׳.

בִּיָד אַחַת עוֹשִׂים מְלָאכָה וּבְיָד אַחַת אוֹחֲזִים אֶת הַחֶרֶב.

מָה לוֹקְחִים מִרָמָה א׳?

הַחוֹמָה נִבְנֵית בִּידֵי כֻּלָּם: כֹּהֲנִים, צוֹרְפִים, רוֹקְחִים וַאֲנָשִׁים פְּשׁוּטִים, כָּל אֶחָד לְיַד הַבַּיִת שֶׁלּוֹ. הָאוֹיְבִים לוֹעֲגִים וּמְאַיְּמִים, אֲבָל ׳הַלֵּב שֶׁל הָעָם לַעֲשׂוֹת׳ חָזָק, וּבְיָד אַחַת בּוֹנִים וּבְיָד אַחַת אוֹחֲזִים חֶרֶב. ה׳ אוֹהֵב אוֹתָנוּ וְנִלְחָם בִּשְׁבִילֵנוּ.

  • כָּל אֶחָד בָּנָה ׳לְנֶגֶד בֵּיתוֹ׳ – אֶת הַחֵלֶק שֶׁלּוֹ. מָהוּ הַחֵלֶק שֶׁלִּי בְּבִנְיַן הַבַּיִת, הַכִּתָּה אוֹ הַקְּהִלָּה?

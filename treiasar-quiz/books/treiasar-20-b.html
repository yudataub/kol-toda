שִׂימוּ לְבַבְכֶם עַל דַּרְכֵיכֶם

חַגַּי קוֹרֵא לְשִׁבֵי צִיּוֹן לִבְנוֹת אֶת בֵּית ה׳, וּמַבְטִיחַ: ׳אֲנִי אִתְּכֶם׳

🎯 לַחִידוֹן
🕰️

📜 הַפָּסוּקהָעָם הַזֶּה אָמְרוּ לֹא עֶת־בֹּא עֶת־בֵּית ה׳ לְהִבָּנוֹת

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁה׳ אוֹמֵר ׳עַכְשָׁו׳, אֲנִי לֹא מְחַכֶּה לְ׳אַחַר כָּךְ׳ 💛

📖 חַגַּי, פֶּרֶק א׳, פָּסוּק ב׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🕰️ רַשִׁ״י: ׳לֹא עֶת בֹּא׳ – הַטַּעֲנָה שֶׁל הָעָם

רַשִׁ״י מְבָאֵר אֶת טַעֲנַת הָעָם ׳לֹא עֶת בֹּא עֶת בֵּית ה׳ לְהִבָּנוֹת׳: מִפְּנֵי שֶׁאוֹיְבֵיהֶם בִּטְּלוּ אוֹתָם בַּפַּעַם הָרִאשׁוֹנָה, הֵם ׳סְבוּרִים שֶׁלֹּא אֲקַיֵּם אֶת דְּבָרַי׳. וּמְצוּדַת דָּוִד: ׳עֲדַיִן לֹא בָּא הָעֵת׳ – אֲבָל ׳כְּבָר נִשְׁלְמוּ שִׁבְעִים שָׁנָה׳ שֶׁאָמַר יִרְמְיָהוּ. כְּלוֹמַר: הָעָם הִתְיָאֵשׁ מֵהַזְּמַן, אֲבָל ה׳ אָמַר שֶׁהַזְּמַן כְּבָר הִגִּיעַ.

הָעָם אָמַר ׳לֹא הַזְּמַן׳ – וְחַגַּי עוֹנֶה שֶׁהַזְּמַן כְּבָר הִגִּיעַ.

1
🪵

📜 הַפָּסוּקהַעֵת לָכֶם אַתֶּם לָשֶׁבֶת בְּבָתֵּיכֶם סְפוּנִים וְהַבַּיִת הַזֶּה חָרֵב

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לָשִׂים אֶת מַה שֶּׁקָּשׁוּר לַה׳ בַּמָּקוֹם הָרִאשׁוֹן 💛

📖 חַגַּי, פֶּרֶק א׳, פָּסוּק ד׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🪵 רַשִׁ״י: ׳סְפוּנִים׳ – מְכֻסִּים בְּנִסְרֵי אֲרָזִים

רַשִׁ״י מְבָאֵר אֶת ׳בְּבָתֵּיכֶם סְפוּנִים׳: ׳מְכֻסִּים בְּנִסְרֵי אֲרָזִים׳. וּמְצוּדַת דָּוִד: ׳וְכִי רַק לְבַד לָכֶם בָּא הָעֵת שֶׁאַתֶּם תֵּשְׁבוּ בְּבָתֵּיכֶם הַסְּפוּנִים וּמְכֻסִּין בְּתִקְרוֹת וְהַבַּיִת הַזֶּה יִשָּׁאֵר חָרֵב׳. הַפַּרְטִים מַרְאִים: בָּתִּים מֻשְׁקָעִים וְנָאִים, וְהַמִּקְדָּשׁ נִשְׁאָר בְּלִי קִירוֹת. לֹא הַבַּיִת הַיָּפֶה הוּא הַבְּעָיָה אֶלָּא הַסֵּדֶר שֶׁל הַדְּבָרִים.

הַבְּעָיָה אֵינָהּ בַּבָּתִּים הַיָּפִים – אֶלָּא בְּכָךְ שֶׁבֵּית ה׳ נִשְׁאַר אַחֲרוֹן.

2
👛

📜 הַפָּסוּקזְרַעְתֶּם הַרְבֵּה וְהָבֵא מְעָט אָכוֹל וְאֵין־לְשָׂבְעָה שָׁתוֹ וְאֵין־לְשָׁכְרָה לָבוֹשׁ וְאֵין־לְחֹם לוֹ וְהַמִּשְׂתַּכֵּר מִשְׂתַּכֵּר אֶל־צְרוֹר נָקוּב

💛 אַהֲבַת הַשֵּׁם: הַבְּרָכָה בָּאָה מֵה׳, וְלָכֵן אֲנִי מְבַקֵּשׁ אוֹתָהּ מִמֶּנּוּ 💛

📖 חַגַּי, פֶּרֶק א׳, פָּסוּק ו׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

👛 רַשִׁ״י: ׳צְרוֹר נָקוּב׳

רַשִׁ״י מְפָרֵט אֶת ׳זְרַעְתֶּם הַרְבֵּה וְהָבֵא מְעָט׳: ׳בַּעֲוֹן בִּכּוּרִים שֶׁבִּטְּלוּ… בַּעֲוֹן בִּטּוּל מְנָחוֹת… שֶׁנִּטַּל טַעַם הַיַּיִן לְפִי שֶׁבִּטְּלוּ הַנְּסָכִים… בַּעֲוֹן בִּגְדֵי כְּהֻנָּה שֶׁבִּטְּלוּ׳. וְעַל ׳מִשְׂתַּכֵּר אֶל צְרוֹר נָקוּב׳: ׳כָּל רֶוַח שֶׁאַתֶּם עוֹשִׂים הוֹלֵךְ וְכָלֶה כְּנוֹתֵן מָעוֹתָיו בְּקֶשֶׁר בֶּגֶד נָקוּב׳. כְּשֶׁהַמִּקְדָּשׁ חָרֵב, חֲסֵרָה בְּרָכָה בְּכָל מָקוֹם.

הַכֶּסֶף הוֹלֵךְ וְנוֹפֵל מִן הַכִּיס הַנָּקוּב – כְּשֶׁחֲסֵרָה הַבְּרָכָה.

3
🔤

📜 הַפָּסוּקעֲלוּ הָהָר וַהֲבֵאתֶם עֵץ וּבְנוּ הַבָּיִת וְאֶרְצֶה־בּוֹ וְאֶכָּבְדָה אָמַר ה׳

💛 אַהֲבַת הַשֵּׁם: ה׳ מִתְרַצֶּה בִּי גַּם כְּשֶׁחָסֵר בִּי מַשֶּׁהוּ, וְהוּא אוֹהֵב אוֹתִי כְּמוֹ שֶׁאֲנִי 💛

📖 חַגַּי, פֶּרֶק א׳, פָּסוּק ח׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🔤 רַשִׁ״י: ׳וְאֶכָּבְדָה׳ חָסֵר הֵא

בַּפָּסוּק ׳וְאֶרְצֶה בּוֹ וְאֶכָּבְדָה׳ רַשִׁ״י מְבָאֵר: ׳וְאֶכָּבְדָה – חָסֵר הֵ״א, כְּנֶגֶד חֲמִשָּׁה דְבָרִים שֶׁהָיוּ בַּמִּקְדָּשׁ הָרִאשׁוֹן וְלֹא הָיוּ בַּמִּקְדָּשׁ שֵׁנִי: אָרוֹן, אוּרִים וְתוּמִּים, אֵשׁ, שְׁכִינָה וְרוּחַ הַקֹּדֶשׁ׳, ׳כְּדְאִיתָא בְּמַסֶּכֶת יוֹמָא׳. הַכְּתִיב ׳וְאֶכָּבֵד׳ בְּלִי הֵא מְרַמֵּז לְחֲמִשָּׁה דְּבָרִים חֲסֵרִים. אַף עַל פִּי כֵן ה׳ אוֹמֵר: ׳וְאֶרְצֶה בּוֹ׳.

גַּם בַּמִּקְדָּשׁ הַשֵּׁנִי שֶׁחָסְרוּ בּוֹ דְּבָרִים – ה׳ אוֹמֵר: ׳וְאֶרְצֶה בּוֹ׳.

🛡️

📜 הַפָּסוּקוַיֹּאמֶר חַגַּי מַלְאַךְ ה׳ בְּמַלְאֲכוּת ה׳ לָעָם לֵאמֹר אֲנִי אִתְּכֶם נְאֻם־ה׳

💛 אַהֲבַת הַשֵּׁם: ה׳ אִתִּי, וְלָכֵן אֲנִי יָכוֹל לְהַתְחִיל גַּם כְּשֶׁאֲנִי מְפַחֵד 💛

📖 חַגַּי, פֶּרֶק א׳, פְּסוּקִים י״ג–י״ד (מְצוּדַת דָּוִד וְרַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🛡️ מְצוּדַת דָּוִד: לָמָּה ׳אֲנִי אִתְּכֶם׳?

מְצוּדַת דָּוִד מְבָאֵר אֶת ׳אֲנִי אִתְּכֶם׳: ׳עֲדַיִן פָּחֲדוּ לִבְנוֹת הַבַּיִת בְּלֹא רְשׁוּת הַמֶּלֶךְ, וּלְכָךְ אָמַר לָהֶם: אֲנִי עִמָּכֶם וְאַל תִּפְחָדוּ׳. רַשִׁ״י עַל הַפָּסוּק הַבָּא: ׳הִתְחִילוּ מְסַתְּתִים בָּאֲבָנִים וּמְנַסְּרִים עֵצִים בְּשִׁשִּׁי בְּעֶשְׂרִים וְאַרְבָּעָה לַחֹדֶשׁ׳. הַפַּחַד הָיָה אֲמִתִּי, וְהַהַבְטָחָה הִיא מַה שֶּׁנָּתְנָה לָהֶם אֹמֶץ לְהַתְחִיל.

אֲפִלּוּ כְּשֶׁיֵּשׁ פַּחַד אֲמִתִּי – ׳אֲנִי אִתְּכֶם׳ נוֹתֵן אֹמֶץ לְהַתְחִיל.

⚖️

📜 הַפָּסוּקהֲלוֹא כָמֹהוּ כְּאַיִן בְּעֵינֵיכֶם וְעַתָּה חֲזַק זְרֻבָּבֶל נְאֻם־ה׳

💛 אַהֲבַת הַשֵּׁם: גַּם דָּבָר קָטָן שֶׁעוֹשִׂים לַה׳ הוּא גָּדוֹל בְּעֵינָיו 💛

📖 חַגַּי, פֶּרֶק ב׳, פְּסוּקִים ג׳–ד׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

⚖️ שְׁנֵי הַבָּתִּים: רִאשׁוֹן וְאַחֲרוֹן

רַשִׁ״י מְבָאֵר אֶת ׳מִי בָכֶם הַנִּשְׁאָר׳: ׳יוֹדֵעַ אֲנִי כִּי מִי שֶׁרָאָה הָרִאשׁוֹן אֵין זֶה בְּעֵינָיו אֶלָּא כְּאַיִן׳. מִי שֶׁזָּכַר אֶת הַבַּיִת שֶׁבָּנָה שְׁלֹמֹה בְּעֹשֶׁר רַב, מַשְׁוֶה וְרוֹאֶה פַּעַר. וּמְצוּדַת דָּוִד עַל ׳חֲזַק׳: ׳עִם שֶׁאֵין לוֹ עֵרֶךְ עִם הַבַּיִת הָרִאשׁוֹן, מִכָּל מָקוֹם תָּבוֹא לָכֶם הַבְּרָכָה עַל יָדוֹ׳. אֲנַחְנוּ לֹא מוֹדְדִים אֶת הַבַּיִת רַק לְפִי הַגֹּדֶל – אֶלָּא לְפִי מַה שֶׁה׳ נוֹתֵן בּוֹ.

ה׳ לֹא מוֹדֵד לְפִי הַגֹּדֶל אֶלָּא לְפִי הַלֵּב וְהַבְּרָכָה שֶׁהוּא נוֹתֵן.

📜 הַפָּסוּקהַעוֹד הַזֶּרַע בַּמְּגוּרָה… מִן־הַיּוֹם הַזֶּה אֲבָרֵךְ

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹמֵר: ׳מִן הַיּוֹם הַזֶּה אֲבָרֵךְ׳ – כְּבָר הַיּוֹם יֵשׁ לִי הִזְדַּמְּנוּת לְהַתְחִיל 💛

📖 חַגַּי, פֶּרֶק ב׳, פְּסוּקִים י״ח–י״ט (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🌱 ׳מִן הַיּוֹם הַזֶּה אֲבָרֵךְ׳

הַנָּבִיא חוֹזֵר שׁוּב וְשׁוֹאֵל: ׳שִׂימוּ נָא לְבַבְכֶם מִן הַיּוֹם הַזֶּה וָמָעְלָה׳. רַשִׁ״י מְבָאֵר: ׳עֲדַיִן לֹא זְרַעְתֶּם בַּשָּׁנָה הַזֹּאת… וּמֵעַתָּה תִּזְרְעוּ בִּזְמַן בְּרָכָה, כִּי בְּבִנְיַן הַבַּיִת יָבוֹא בְּרָכָה בְּמַעֲשֵׂה יְדֵיכֶם׳. וּמְצוּדַת דָּוִד: ׳מִיּוֹם הַתְחָלַת הַבִּנְיָן אֶשְׁלַח בְּרָכָה׳. הִנֵּה הַבְּרָכָה מַתְחִילָה בְּדִיּוּק מִן הַיּוֹם שֶׁמַּתְחִילִים לִבְנוֹת.

הַבְּרָכָה מַתְחִילָה מִן הַיּוֹם שֶׁבּוֹ מַתְחִילִים לִבְנוֹת לַה׳.

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מְגַלִּים: הַבַּיִת הַשֵּׁנִי חָסְרוּ בּוֹ חֲמִשָּׁה דְבָרִים, וְה׳ אָמַר ׳וְאֶרְצֶה בּוֹ׳. ה׳ מוֹדֵד אֶת הַלֵּב וְלֹא אֶת הַגֹּדֶל, וְהַבְּרָכָה מַתְחִילָה ׳מִן הַיּוֹם הַזֶּה׳.

שִׂימוּ לְבַבְכֶם עַל דַּרְכֵיכֶם

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • חַגַּי אוֹמֵר ׳שִׂימוּ לְבַבְכֶם עַל דַּרְכֵיכֶם׳. מָה הַדָּבָר הַחָשׁוּב שֶׁאֲנִי דּוֹחֶה, וּמָה אֲנִי יָכוֹל לְהַתְחִיל הַשָּׁבוּעַ?
  • ׳אֲנִי אִתְּכֶם נְאֻם ה׳׳. מָתַי הִרְגַּשְׁתִּי שֶׁה׳ אִתִּי כְּשֶׁהִתְחַלְתִּי דָּבָר קָשֶׁה?
  • הַבַּיִת הַשֵּׁנִי נִרְאָה קָטָן, וְה׳ אָמַר ׳וְאֶרְצֶה בּוֹ׳. עַל אֵיזֶה דָּבָר קָטָן שֶׁעָשִׂיתִי אֲנִי רוֹצֶה שֶׁה׳ יִתְרַצֶּה בּוֹ?

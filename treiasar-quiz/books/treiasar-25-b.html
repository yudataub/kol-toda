📜 הַפָּסוּקמַשָּׂא דְבַר־ה׳ אֶל־יִשְׂרָאֵל בְּיַד מַלְאָכִי

💛 אַהֲבַת הַשֵּׁם: הַנְּבוּאוֹת הֵן הָרוּחַ שֶׁל סִינַי – וַאֲנִי חֵלֶק מִן הַשַּׁלְשֶׁלֶת 💛

📖 מַלְאָכִי, פֶּרֶק א׳, פָּסוּק א׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

📜 רַשִׁ״י: כָּל הַנְּבִיאִים עָמְדוּ עַל הַר סִינַי

עַל הַמִּלִּים ׳בְּיַד מַלְאָכִי׳ רַשִׁ״י מְבָאֵר: ׳כְּבָר הָיָה מָסוּר בְּיָדוֹ זֶה יָמִים רַבִּים. מִכָּאן דָּרְשׁוּ רַבּוֹתֵינוּ בְּבָרַיְתָא דִּמְכִילְתָּא שֶׁכָּל הַנְּבִיאִים עָמְדוּ עַל הַר סִינַי וְשָׁם נִמְסְרוּ לָהֶם הַנְּבוּאוֹת׳. וְרַשִׁ״י מַזְכִּיר פָּסוּק מִיְּשַׁעְיָהוּ: ׳מֵעֵת הֱיוֹתָהּ שָׁם אֲנִי׳. לְפִי זֶה, הַנְּבוּאוֹת הָאֲחֵרוֹנוֹת הֵן חֵלֶק מִמַּתַּן תּוֹרָה, וְהַנָּבִיא הָאַחֲרוֹן מְחַבֵּר אוֹתָנוּ לְסִינַי.

כָּל הַנְּבִיאִים, גַּם הָאַחֲרוֹן, קִבְּלוּ אֶת דִּבְרֵיהֶם בְּסִינַי.

1
🏞️

📜 הַפָּסוּקאָהַבְתִּי אֶתְכֶם אָמַר ה׳ וַאֲמַרְתֶּם בַּמָּה אֲהַבְתָּנוּ הֲלוֹא־אָח עֵשָׂו לְיַעֲקֹב נְאֻם־ה׳ וָאֹהַב אֶת־יַעֲקֹב

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אוֹתִי בִּגְלַל מִי שֶׁאֲנִי, וְלֹא רַק בִּגְלַל הַהוֹרִים 💛

📖 מַלְאָכִי, פֶּרֶק א׳, פְּסוּקִים ב׳–ג׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🏞️ רַשִׁ״י: ׳וָאֹהַב אֶת יַעֲקֹב׳ – אֶרֶץ חֶמְדָּה

רַשִׁ״י מְבָאֵר אֶת ׳וָאֹהַב אֶת יַעֲקֹב׳: ׳לָתֵת לוֹ אֶת אֶרֶץ חֶמְדָּה׳. וְעַל ׳וְאֶת עֵשָׂו שָׂנֵאתִי׳ הוּא מְבָאֵר: ׳לְדָחְפוֹ אֶל אֶרֶץ שֵׂעִיר מִפְּנֵי יַעֲקֹב אָחִיו, וּמִנְהַג שֶׁבָּעוֹלָם מִי שֶׁיֵּשׁ לוֹ שְׁנֵי בָּנִים לַבְּכוֹר הוּא בּוֹרֵר לוֹ מָנָה יָפָה׳. מְצוּדַת דָּוִד: ׳אָהַבְתִּי אֶתְכֶם גַּם מִפְּאַת עַצְמֵיכֶם בְּצֵרוּף אַהֲבָתִי לָכֶם מִפְּאַת הָאָבוֹת׳. הָאַהֲבָה בָּאָה גַּם בִּגְלַל הָאָבוֹת, וְגַם בִּגְלַל כָּל אֶחָד מֵאִתָּנוּ.

ה׳ אוֹהֵב אוֹתָנוּ גַּם בִּגְלַל הָאָבוֹת וְגַם בִּגְלַל עַצְמֵנוּ.

2
🐑

📜 הַפָּסוּקוְכִי־תַגִּשׁוּן עִוֵּר לִזְבֹּחַ אֵין רָע וְכִי תַגִּישׁוּ פִּסֵּחַ וְחֹלֶה אֵין רָע הַקְרִיבֵהוּ נָא לְפֶחָתֶךָ הֲיִרְצְךָ אוֹ הֲיִשָּׂא פָנֶיךָ

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לָתֵת לַה׳ אֶת הַמֻּבְחָר, כְּמוֹ שֶׁנּוֹתְנִים לְאוֹרֵחַ חָשׁוּב 💛

📖 מַלְאָכִי, פֶּרֶק א׳, פָּסוּק ח׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🐑 רַשִׁ״י: ׳אֵין רָע׳ – וְכִי אֵין זֶה רָע?

עַל ׳אֵין רָע׳ רַשִׁ״י מְבָאֵר: ׳וְכִי אֵין דָּבָר זֶה רָע׳. הַנָּבִיא שׁוֹאֵל בִּתְמִיהָה. וּמְצוּדַת דָּוִד: ׳הַקְרִיבֵהוּ נָא לְפֶחָתֶךָ – בְּחוֹן אַתָּה הַדָּבָר וְהַקְרֵב אַחַת מֵאֵלֶּה לְדוֹרוֹן לְפֶחָתְךָ: הַאִם יִהְיֶה מְרֻצֶּה לְךָ בָּזֶה הַדּוֹרוֹן?׳. כְּלוֹמַר: אִם לְמוֹשֵׁל אָדָם לֹא מְבִיאִים דּוֹרוֹן פָּגוּם, כָּל שֶׁכֵּן לַמֶּלֶךְ שֶׁל כָּל הָעוֹלָם.

מַה שֶּׁלֹּא נָבִיא לְמוֹשֵׁל – לֹא נָבִיא לַמֶּלֶךְ שֶׁל כָּל הָעוֹלָם.

⚖️

📜 הַפָּסוּקכִּי מִמִּזְרַח־שֶׁמֶשׁ וְעַד־מְבוֹאוֹ גָּדוֹל שְׁמִי בַּגּוֹיִם וּבְכָל־מָקוֹם מֻקְטָר מֻגָּשׁ לִשְׁמִי וּמִנְחָה טְהוֹרָה

💛 אַהֲבַת הַשֵּׁם: הַתְּפִלָּה שֶׁלִּי, בְּכָל מָקוֹם, יְקָרָה לַה׳ כְּמוֹ מִנְחָה טְהוֹרָה 💛

📖 מַלְאָכִי, פֶּרֶק א׳, פָּסוּק י״א (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

⚖️ שְׁנֵי פֵּרוּשִׁים: ׳מִנְחָה טְהוֹרָה׳

הַפָּסוּק ׳כִּי מִמִּזְרַח שֶׁמֶשׁ וְעַד מְבוֹאוֹ גָּדוֹל שְׁמִי בַּגּוֹיִם וּבְכָל מָקוֹם מֻקְטָר מֻגָּשׁ לִשְׁמִי וּמִנְחָה טְהוֹרָה׳ מִתְפָּרֵשׁ בִּשְׁנֵי אֳפָנִים. רַשִׁ״י מֵבִיא פֵּרוּשׁ רִאשׁוֹן: אֲפִלּוּ מִי שֶׁיֵּשׁ לוֹ עֲבוֹדָה זָרָה יוֹדֵעַ שֶׁה׳ הוּא עַל כֻּלָּם. וּפֵרוּשׁ שֵׁנִי: ׳רַבּוֹתֵינוּ פֵּרְשׁוּ אֵלּוּ תַּלְמִידֵי חֲכָמִים הָעוֹסְקִים בְּהִלְכוֹת עֲבוֹדָה בְּכָל מָקוֹם, וְכֵן כָּל תְּפִלּוֹת יִשְׂרָאֵל שֶׁמִּתְפַּלְּלִין בְּכָל מָקוֹם הֲרֵי הֵן לִי כְּמִנְחָה טְהוֹרָה׳. שְׁנֵי הַפֵּרוּשִׁים מַגִּיעִים לְאוֹתָהּ נְקֻדָּה: שְׁמוֹ שֶׁל ה׳ גָּדוֹל בְּכָל הָעוֹלָם, וּתְפִלַּת יִשְׂרָאֵל בְּכָל מָקוֹם יְקָרָה לוֹ.

רַשִׁ״י: תְּפִלּוֹת יִשְׂרָאֵל בְּכָל מָקוֹם הֵן כְּמִנְחָה טְהוֹרָה, וּשְׁמוֹ שֶׁל ה׳ גָּדוֹל בְּכָל הָעוֹלָם.

📜 הַפָּסוּקתּוֹרַת אֱמֶת הָיְתָה בְּפִיהוּ וְעַוְלָה לֹא־נִמְצָא בִשְׂפָתָיו בְּשָׁלוֹם וּבְמִישׁוֹר הָלַךְ אִתִּי וְרַבִּים הֵשִׁיב מֵעָוֺן

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִרְדֹּף שָׁלוֹם וּלְקָרֵב אֲחֵרִים, כְּמוֹ אַהֲרֹן 💛

📖 מַלְאָכִי, פֶּרֶק ב׳, פָּסוּק ו׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

☮️ רַשִׁ״י: ׳בְּשָׁלוֹם וּבְמִישׁוֹר הָלַךְ אִתִּי׳

רַשִׁ״י מְבָאֵר אֶת ׳בְּשָׁלוֹם וּבְמִישׁוֹר הָלַךְ אִתִּי׳: ׳אַהֲרֹן וְאֶלְעָזָר וּפִנְחָס׳. וְעַל ׳וְרַבִּים הֵשִׁיב מֵעָוֺן׳: ׳וְכֵן בְּמַעֲשֵׂה הָעֵגֶל הֵשִׁיבוּ כָּל שִׁבְטֵיהֶם מֵעָוֺן, שֶׁנֶּאֱמַר: וַיֵּאָסְפוּ אֵלָיו כָּל בְּנֵי לֵוִי׳. מְצוּדַת דָּוִד: ׳בַּעֲבוּר מִצְוָתִי הָיָה רוֹדֵף לַעֲשׂוֹת שָׁלוֹם בֵּין הַבְּרִיּוֹת וּלְהַדְרִיכָם בְּדֶרֶךְ מִישׁוֹר׳. מִי שֶׁהוֹלֵךְ בְּשָׁלוֹם וּבְיֹשֶׁר – מֵשִׁיב רַבִּים לַטּוֹב.

אַהֲרֹן, אֶלְעָזָר וּפִנְחָס הָלְכוּ בְּשָׁלוֹם וְהֵשִׁיבוּ רַבִּים מֵעָוֺן.

📚 רַשִׁ״י: ׳שִׂפְתֵי כֹהֵן יִשְׁמְרוּ דַעַת׳

רַשִׁ״י מְבָאֵר אֶת ׳כִּי שִׂפְתֵי כֹהֵן יִשְׁמְרוּ דַעַת׳: ׳עֲלֵיהֶם מוּטָל לִשְׁמֹר דַּעַת, לָמָּה? שֶׁהֲרֵי תּוֹרָה יְבַקְשׁוּ מִפִּיהוּ, שֶׁכְּבָר דָּבָר זֶה מָסוּר לָהֶם: יוֹרוּ מִשְׁפָּטֶיךָ לְיַעֲקֹב׳. וְעַל ׳כִּי מַלְאַךְ ה׳ צְבָאוֹת הוּא׳: ׳שְׁלוּחוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא׳. מְצוּדַת דָּוִד: ׳הוּא שְׁלוּחוֹ שֶׁל הַמָּקוֹם לְהוֹרוֹתָם הַדֶּרֶךְ הַיָּשָׁר׳. מִי שֶׁלּוֹמֵד תּוֹרָה הוּא שָׁלִיחַ שֶׁל ה׳ לְלַמֵּד אֶת הַדֶּרֶךְ.

מִי שֶׁלּוֹמֵד וּמְלַמֵּד תּוֹרָה הוּא שָׁלִיחַ שֶׁל ה׳.

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

👨‍👩‍👦 רַשִׁ״י: ׳בְּרִית אֲבוֹתֵינוּ׳

עַל הַפָּסוּק ׳מַדּוּעַ נִבְגַּד אִישׁ בְּאָחִיו לְחַלֵּל בְּרִית אֲבֹתֵינוּ׳ רַשִׁ״י מְבָאֵר: ׳בְּרִית שֶׁכָּרַת הַקָּדוֹשׁ בָּרוּךְ הוּא עִם אֲבוֹתֵינוּ בְּסִינַי׳. מְצוּדַת דָּוִד: ׳הֲלוֹא אָב אֶחָד לְכֻלָּנוּ – כִּי כֻּלָּנוּ בְּנֵי אָדָם הָרִאשׁוֹן, וְיֵשׁ לָנוּ קִרְבָה מִפְּאַת הַגּוּף, וַהֲלֹא אֵל אֶחָד בְּרָאָנוּ – נַפְשׁוֹת הַיְשָׁרִים אֲצוּלוֹת מֵאִתּוֹ׳. אֲנַחְנוּ אַחִים מִשְּׁנֵי צְדָדִים: אָדָם הָרִאשׁוֹן, וְנֶפֶשׁ שֶׁה׳ נָתַן.

כֻּלָּנוּ אַחִים – בִּגּוּף וּבְנֶפֶשׁ – וְלָכֵן אֵין בּוֹגְדִים בָּאָח.

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מְגַלִּים שֶׁה׳ אוֹהֵב אוֹתָנוּ גַּם בִּגְלַל הָאָבוֹת וְגַם בִּגְלַל עַצְמֵנוּ, שֶׁתְּפִלּוֹת יִשְׂרָאֵל בְּכָל מָקוֹם הֵן מִנְחָה טְהוֹרָה, וְשֶׁכֻּלָּנוּ אַחִים.

אָהַבְתִּי אֶתְכֶם אָמַר ה׳

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • הַנָּבִיא הָאַחֲרוֹן פּוֹתֵחַ ב׳אָהַבְתִּי אֶתְכֶם׳. בְּמָה אֲנִי רוֹאֶה אֶת אַהֲבַת ה׳ בְּחַיַּי?
  • ׳הַקְרִיבֵהוּ נָא לְפֶחָתֶךָ׳. אֵיזֶה דָּבָר אֲנִי עוֹשֶׂה לַה׳ בִּשְׁאֵרִית כֹּחַ, וְאֵיךְ אֶפְשָׁר לִתֵּן לוֹ אֶת הַמֻּבְחָר?
  • ׳הֲלוֹא אָב אֶחָד לְכֻלָּנוּ׳. אֵיךְ אֲנִי יָכוֹל לְהִתְנַהֵג הַשָּׁבוּעַ אֶל חָבֵר כְּמוֹ אָח?

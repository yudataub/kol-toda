💛 אַהֲבַת הַשֵּׁם: ה׳ מוֹלִיךְ אֶת הָעוֹלָם בְּלִי שֶׁאֲנִי אַרְגִּישׁ, וְאֲנִי מוֹדֶה עַל כָּךְ 💛

רָמָה ב׳ – עִיּוּן בַּמַּרְאוֹת

🛢️ רַשִׁ״י: הַשֶּׁמֶן בָּא מֵעַצְמוֹ

רַשִׁ״י מְפָרֵט אֶת הַמַּרְאֶה: ׳וְגֻלָּה עַל רֹאשָׁהּ – כְּמִין סֵפֶל גָּדוֹל עָגֹל׳, וְ׳שִׁבְעָה וְשִׁבְעָה מוּצָקוֹת – לְכָל נֵר וְנֵר בָּאִים שִׁבְעָה צִנּוֹרוֹת קְטַנִּים שֶׁהַשֶּׁמֶן זָב מִן הַגֻּלָּה דֶּרֶךְ אוֹתָן מוּצָקוֹת׳. וְעַל הַשְּׁאֵלָה ׳מָה אֵלֶּה׳ הוּא מְבָאֵר: ׳מָה זֶה שֶׁהַזֵּיתִים נִמְסָקִים מֵאֲלֵיהֶם וְהַשֶּׁמֶן בָּא אֶל הַנֵּרוֹת מֵאֵלָיו׳. הַפֶּלֶא הוּא שֶׁהַכֹּל פּוֹעֵל בְּלִי מַאֲמָץ שֶׁל אָדָם.

הַשֶּׁמֶן בָּא אֶל הַנֵּרוֹת מֵאֵלָיו – כְּמוֹ שֶׁה׳ מְסַיֵּעַ לְבַד.

רָמָה ב׳ – עִיּוּן בַּמַּרְאוֹת

⚖️ שְׁנֵי פֵּרוּשִׁים: ׳בְּרוּחִי׳

רַשִׁ״י מְבָאֵר אֶת ׳לֹא בְחַיִל וְלֹא בְכֹחַ׳: ׳כְּשֵׁם שֶׁהַזֵּיתִים וְהַשֶּׁמֶן הַזֶּה נִגְמָר מֵאֵלָיו לְכָל דְּבָרָיו, כָּךְ לֹא בְחַיִל וְלֹא בְכֹחַ שֶׁבָּכֶם תַּעֲשׂוּ אֶת בִּנְיַן בֵּיתִי׳, אֶלָּא ׳אֲנִי אֶתֵּן רוּחִי עַל דָּרְיָוֶשׁ וְיַנִּיחַ לָכֶם לִבְנוֹת׳. מְצוּדַת דָּוִד מְפָרֵשׁ עַל מֶלֶךְ הַמָּשִׁיחַ: מֶמְשַׁלְתּוֹ ׳לֹא תִהְיֶה בַּעֲבוּר חֵילוֹ… אֲבָל יִמְשֹׁל בָּהֶם מִבְּלִי טֹרַח מִלְחָמָה׳. בִּשְׁנֵי הַפֵּרוּשִׁים: הָרוּחַ שֶׁל ה׳ מַכְרִיעָה, לֹא כֹּחַ הָאָדָם.

בִּשְׁנֵי הַפֵּרוּשִׁים, הַנִּצָּחוֹן בָּא בְּרוּחַ ה׳ וְלֹא בְּכֹחַ הַצָּבָא.

💛 אַהֲבַת הַשֵּׁם: ה׳ מַעֲבִיר אֶת הַמְּכשׁוֹלִים שֶׁנִּרְאִים לִי בִּלְתִּי עֲבִירִים 💛

📖 זְכַרְיָה, פֶּרֶק ד׳, פָּסוּק ז׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּמַּרְאוֹת

🏔️ רַשִׁ״י: ׳הַר הַגָּדוֹל׳ – שָׂרֵי עֵבֶר הַנָּהָר

רַשִׁ״י מְבָאֵר אֶת ׳מִי אַתָּה הַר הַגָּדוֹל׳: אֵלֶּה ׳שָׂרֵי עֵבֶר הַנָּהָר׳, ׳שֶׁבִּטַּלְתֶּם הַמְּלָאכָה עַד כֹּה, מֵעַתָּה תִּהְיֶה לִפְנֵי זְרֻבָּבֶל מִישׁוֹר – אֵין לָכֶם שְׁרָרָה וּגְדֻלָּה עוֹד עָלָיו׳. הַהַר הוּא אֲנָשִׁים שֶׁעִכְּבוּ אֶת הַבִּנְיָן. מְצוּדַת דָּוִד מַסְבִּיר עַל מֶלֶךְ גּוֹג: ׳מַה כֹּחַ יֵשׁ בְּךָ… לְמוּלוֹ תִּהְיֶה כְּאֶרֶץ מִישׁוֹר׳. כָּל מְכשׁוֹל גָּדוֹל שׁוֹקֵעַ מוּל רְצוֹן ה׳.

הַמְּכשׁוֹלִים הַגְּדוֹלִים – אֲנָשִׁים שֶׁעִכְּבוּ – הוֹפְכִים מִישׁוֹר לִפְנֵי רְצוֹן ה׳.

רַשִׁ״י מְבָאֵר: ׳יְדֵי זְרֻבָּבֶל יִסְּדוּ הַבַּיִת הַזֶּה – מִתְּחִלָּה וְעַד סוֹף בִּימֵי כוֹרֶשׁ הָרִאשׁוֹן׳, ׳וְיָדָיו תְּבַצַּעְנָה – אוֹתוֹ עַתָּה׳, ׳תְּבַצַּעְנָה לְשׁוֹן גְּמַר׳. מְצוּדַת דָּוִד: ׳זְרֻבָּבֶל עַצְמוֹ הִנִּיחַ הַיְּסוֹד… כֵּן יָדָיו יַשְׁלִימוּ׳. יֵשׁ פֹּה סֵדֶר: יְסוֹד, אֶמְצַע, וְגָמַר – וְהַיָּדַיִם הֵן אוֹתָן יָדַיִם. מִי שֶׁהִתְחִיל, ה׳ נוֹתֵן לוֹ לְסַיֵּם.

אוֹתָן יָדַיִם שֶׁהִתְחִילוּ – יַגְמְרוּ, בְּעֶזְרַת ה׳.

😢 רַשִׁ״י: מֵאַכְזָבָה לְשִׂמְחָה

רַשִׁ״י מְבָאֵר: ׳כִּי מִי בַז לְיוֹם – שֶׁיִּוָּסֵד הַבַּיִת אֲשֶׁר הֻקְטַן בְּעֵינֵיהֶם׳, וּמַזְכִּיר מַה שֶּׁכָּתוּב בְּעֶזְרָא: ׳וְרַבִּים מֵהַכֹּהֲנִים אֲשֶׁר רָאוּ הַבַּיִת הָרִאשׁוֹן בְּיָסְדוֹ זֶה הַבַּיִת בְּעֵינֵיהֶם בּוֹכִים בְּקוֹל גָּדוֹל׳. וּבִנְבוּאַת חַגַּי ׳הֲלוֹא כָמֹהוּ כְּאַיִן׳. אֲבָל עַתָּה ׳יִשְׂמְחוּ כְּשֶׁיִּרְאוּ אֶת הָאֶבֶן הַבְּדִיל׳. מְצוּדַת דָּוִד: ׳הַשִּׂמְחוֹת הָאֵלֶּה… כָּפוּל שִׁבְעָה בְּעֵרֶךְ הַתּוּגָה׳. הַבֶּכִי שֶׁהָיָה יִהְפֹּךְ לְשִׂמְחָה גְּדוֹלָה.

הַבֶּכִי עַל הַקּוֹטֶן יִהְפֹּךְ לְשִׂמְחָה כְּפוּלָה, כְּשֶׁיִּרְאוּ אֶת הַבִּנְיָן גָּדֵל.

🔀 רַשִׁ״י: שְׁנֵי פֵּרוּשִׁים לְ׳בְנֵי הַיִּצְהָר׳

עַל ׳שְׁנֵי בְנֵי הַיִּצְהָר׳ רַשִׁ״י מַבִּיא שְׁנֵי פֵּרוּשִׁים. הָרִאשׁוֹן: ׳קֶרֶן כְּהֻנָּה וּמַלְכוּת הַנִּמְשָׁחִים בְּשֶׁמֶן הַמִּשְׁחָה׳. הַשֵּׁנִי: ׳יֵצֶר טוֹב וְיֵצֶר הָרָע מִתְהַפֵּךְ לְטוֹב בִּזְכוּת הַתּוֹרָה׳. מְצוּדַת דָּוִד מַסְכִּים לַפֵּרוּשׁ הָרִאשׁוֹן – הַמָּשִׁיחַ וְהַכֹּהֵן הַגָּדוֹל הַנִּמְשָׁחִים בְּשֶׁמֶן. שְׁנֵי הַפֵּרוּשִׁים מַלְמְּדִים: הַשֶּׁמֶן – הַתּוֹרָה וְהַקְּדֻשָּׁה – הוּא מָה שֶׁמַּאִיר אֶת הַנֵּרוֹת.

אוֹ מֶלֶךְ וְכֹהֵן, אוֹ שְׁנֵי הַיְּצָרִים שֶׁמִּתְהַפְּכִים לְטוֹב – הַשֶּׁמֶן מַאִיר.

רָמָה ב׳ – עִיּוּן בַּמַּרְאוֹת

⚖️ רַשִׁ״י: ׳זֹאת הָאָלָה׳

מָה בַּמְּגִלָּה? רַשִׁ״י: ׳זֹאת הָאָלָה – זֹאת שֶׁכָּתוּב בָּהּ פּוּרְעָנוּת נְקִמַת שְׁבוּעוֹת שֶׁקֶר׳. מְצוּדַת דָּוִד: ׳אָלָה… בְּגָזְרָה עַל פְּנֵי כָּל אֶרֶץ יִשְׂרָאֵל׳. וְעַל ׳הוֹצֵאתִיהָ׳: ׳אַף שֶׁעַד הֵנָּה… הֶאֱרַכְתִּי אַפִּי, אֲבָל כַּאֲשֶׁר נִתְמַלְּאָה הַסְּאָה הוֹצֵאתִי אֶת הַמְּגִלָּה׳. ה׳ מַמְתִּין הַרְבֵּה לְפָנֵי שֶׁהוּא פּוֹעֵל, כִּי הוּא ׳אֶרֶךְ אַפַּיִם׳. אֲבָל הוּא רוֹצֶה שֶׁאֲנַחְנוּ נִהְיֶה יְשָׁרִים לְפָנָיו וְלַחֲבֵרֵינוּ.

ה׳ מַאֲרִיךְ אַפּוֹ כְּדֵי שֶׁנְּתַקֵּן, וְרוֹצֶה שֶׁנִּהְיֶה יְשָׁרִים.

רָמָה ב׳ – עִיּוּן בַּמַּרְאוֹת

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מְגַלִּים שֶׁהָהָר הַגָּדוֹל הוּא מְכשׁוֹלִים, שֶׁהַבֶּכִי עַל הַקּוֹטֶן יִהְפֹּךְ לְשִׂמְחָה כְּפוּלָה, וְשֶׁהַשֶּׁמֶן הוּא הַתּוֹרָה וְהַקְּדֻשָּׁה. ה׳ אוֹהֵב וְעוֹזֵר.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳לֹא בְחַיִל וְלֹא בְכֹחַ כִּי אִם בְּרוּחִי׳. מָתַי הִרְגַּשְׁתִּי שֶׁה׳ עוֹזֵר לִי יוֹתֵר מִכֹּחַ שֶׁלִּי?
  • ׳כִּי מִי בַז לְיוֹם קְטַנּוֹת׳. מָה הַתְחָלָה קְטַנָּה שֶׁעָשִׂיתִי לָאחֲרוֹנָה, וְאֵיךְ אֲנִי יָכוֹל לַחֲשֹׁב עָלֶיהָ בְּכָבוֹד?
  • הַנֵּר הַקָּטָן מֵאִיר הַרְבֵּה. עַל אֵיזֶה דָּבָר קָטָן אֲנִי שָׂמֵחַ וּמוֹדֶה הַיּוֹם לַה׳?

📜 הַפָּסוּקעוֹד אַרְבָּעִים יוֹם וְנִינְוֵה נֶהְפָּכֶת

🔄 רַשִׁ״י: ׳נֶהְפָּכֶת׳ – שְׁתֵּי לְשׁוֹנוֹת

רַשִׁ״י מְבָאֵר ׳נֶהְפָּכֶת׳: ׳נֶחֱרֶבֶת׳. וְהוּא מוֹסִיף: ׳לֹא אָמַר נֶחֱרֶבֶת כִּי נֶהְפָּכֶת מְשַׁמֵּשׁ שְׁתֵּי לְשׁוֹנוֹת, רַע וְטוֹב: אִם לֹא יַעֲשׂוּ תְשׁוּבָה – נֶחֱרֶבֶת, וְאִם יַעֲשׂוּ תְשׁוּבָה – אָז נֶהְפָּכֶת׳. כְּלוֹמַר: אַנְשֵׁי נִינְוֵה יַהַפְכוּ מֵרָעָה לְטוֹבָה. יוֹנָה אָמַר מִלָּה אַחַת בְּשֶׁתֵּי מַשְׁמָעֻיּוֹת, וְהֵם בָּחֲרוּ בַּטּוֹבָה.

׳נֶהְפָּכֶת׳ יָכוֹל לִהְיוֹת הֲרִיסָה, וְיָכוֹל לִהְיוֹת הֲפִיכָה לְטוֹבָה – וְהַבְּחִירָה בִּידֵי נִינְוֵה.

רָמָה ב׳ – עִיּוּן בַּתְּשׁוּבָה

⏳ מְצוּדַת דָּוִד: יֵשׁ זְמַן

מְצוּדַת דָּוִד מְבָאֵר: ׳הִנֵּה יֵשׁ עוֹד זְמַן אַרְבָּעִים יוֹם, וְאַחַר כָּךְ תִּהְיֶה נִינְוֵה נֶהְפָּכֶת׳. וְעוֹד: ׳כִּמְהֻפֶּכֶת סְדֹם וַעֲמֹרָה׳. וְעַל פָּסוּק ב׳: ׳הַקְּרִיאָה אֲשֶׁר אָנֹכִי דֹּבֵר אֵלֶיךָ׳ – ׳וְלֹא הֻצְרַךְ לְפָרֵשׁ כִּי מֵאֵלָיו יוּבַן׳. הַיְּדִיעָה הַפְּשׁוּטָה, עִם זְמַן לַחֲשֹׁב, מַסְפִּיקָה כְּדֵי לְשַׁנּוֹת עִיר שְׁלֵמָה.

מְצוּדַת דָּוִד: יֵשׁ אַרְבָּעִים יוֹם – זְמַן מַסְפִּיק כְּדֵי לָשׁוּב.

📜 הַפָּסוּקוַיִּגַּע הַדָּבָר אֶל מֶלֶךְ נִינְוֵה וַיָּקָם מִכִּסְאוֹ וַיַּעֲבֵר אַדַּרְתּוֹ מֵעָלָיו

💛 אַהֲבַת הַשֵּׁם: תְּשׁוּבָה מַתְחִילָה בַּעֲנָוָה – אֲנִי מוּכָן לֵרֵד מִן הַכִּסֵּא שֶׁלִּי 💛

🧥 רַשִׁ״י: ׳אַדַּרְתּוֹ׳ – לְבוּשׁ יְקָרוֹ

רַשִׁ״י מְבָאֵר ׳אַדַּרְתּוֹ׳: ׳לְבוּשׁ יְקָרֵיהּ׳. וּמְצוּדַת דָּוִד: ׳קָם מִכִּסְאוֹ לָשֶׁבֶת עַל הָאָרֶץ וְהֶעֱבִיר מֵעָלָיו לְבוּשׁ מַלְכוּת׳. הַמֶּלֶךְ מוּכָן לְוַתֵּר עַל הַכָּבוֹד שֶׁלּוֹ כְּדֵי לָשׁוּב. הַפַּעַם הַגָּדוֹל בְּיוֹתֵר הוּא הָרִאשׁוֹן שֶׁיּוֹרֵד מִגֹּבַהּ.

רַשִׁ״י וּמְצוּדַת דָּוִד: הַמֶּלֶךְ מוֹתֵר עַל בִּגְדֵי הַמַּלְכוּת – מִי שֶׁגָּדוֹל, הוּא הָרִאשׁוֹן שֶׁעוֹנֶה.

📜 הַפָּסוּקוַיַּזְעֵק וַיֹּאמֶר בְּנִינְוֵה מִטַּעַם הַמֶּלֶךְ וּגְדֹלָיו לֵאמֹר

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁכֻּלָּם עוֹשִׂים תְּשׁוּבָה יַחַד, קַל יוֹתֵר לְכָל אֶחָד לַחֲזֹר 💛

📖 יוֹנָה, פֶּרֶק ג׳, פָּסוּק ז׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּתְּשׁוּבָה

📣 רַשִׁ״י: ׳מִטַּעַם הַמֶּלֶךְ וּגְדֹלָיו׳

רַשִׁ״י מְבָאֵר ׳מִטַּעַם הַמֶּלֶךְ וּגְדוֹלָיו׳: ׳מֵעֲצַת הַמֶּלֶךְ וּגְדוֹלָיו צִוּוּ לְהַכְרִיז׳. כְּלוֹמַר, לֹא הַמֶּלֶךְ לְבַדּוֹ, אֶלָּא כָּל גְּדוֹלֵי הָעִיר יַחַד. וּמְצוּדַת דָּוִד: ׳בַּעֲבוּר עֶגְמַת נֶפֶשׁ צִוּוּ לִמְנֹעַ מַאֲכָל מִבְּהֵמוֹת, לְמַעַן יִכָּנַע לֵב הָאֲנָשִׁים׳. הַכֹּל קוֹרֶה יַחַד, בַּהֲסכָּמָה.

הַמֶּלֶךְ וְהַגְּדוֹלִים מַכְרִיזִים יַחַד, כְּדֵי שֶׁהַלֵּב שֶׁל כֻּלָּם יִכָּנַע.

📜 הַפָּסוּקוְיִתְכַּסּוּ שַׂקִּים הָאָדָם וְהַבְּהֵמָה וְיִקְרְאוּ אֶל אֱלֹהִים בְּחָזְקָה

📖 יוֹנָה, פֶּרֶק ג׳, פָּסוּק ח׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּתְּשׁוּבָה

🙏 רַשִׁ״י: ׳וְיִקְרְאוּ אֶל אֱלֹהִים בְּחָזְקָה׳

רַשִׁ״י מְבָאֵר ׳בְּחָזְקָה׳: ׳אָסְרוּ אִמָּהוֹת לְבַד וְהַוְּלָדוֹת לְבַד. אָמְרוּ לְפָנָיו: רִבּוֹנוֹ שֶׁל עוֹלָם, אִם אֵין אַתָּה מְרַחֵם עָלֵינוּ אֵין אָנוּ מְרַחֲמִים עַל אֵלּוּ׳. כְּלוֹמַר, הֵם גַּם לוֹמְדִים לְרַחֵם עַל הַבְּהֵמוֹת, וּמְבַקְשִׁים רַחֲמִים מֵה׳. וּמְצוּדַת דָּוִד: ׳וְכָל אֶחָד יָשׁוּב מִדַּרְכּוֹ הָרָעָה וּמִן הֶחָמָס אֲשֶׁר עָשׂוּ בְּכַפֵּיהֶם׳.

אַנְשֵׁי נִינְוֵה מְרַחֲמִים אֵלּוּ עַל אֵלּוּ, וּמְבַקְשִׁים שֶׁה׳ יְרַחֵם עֲלֵיהֶם.

📜 הַפָּסוּקמִי יוֹדֵעַ יָשׁוּב וְנִחַם הָאֱלֹהִים וְשָׁב מֵחֲרוֹן אַפּוֹ וְלֹא נֹאבֵד

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹמֵעַ גַּם תִּקְוָה קְטַנָּה וַהֲלִיכָה כֵּנָה אֵלָיו 💛

רָמָה ב׳ – עִיּוּן בַּתְּשׁוּבָה

❓ רַשִׁ״י: ׳מִי יוֹדֵעַ׳

רַשִׁ״י מְבָאֵר ׳מִי יוֹדֵעַ׳: ׳עֲבֵרוֹת שֶׁבְּיָדוֹ יָשׁוּב׳. וּמְצוּדַת דָּוִד: ׳מִי הַיּוֹדֵעַ דַּרְכֵי תְשׁוּבָה יָשׁוּב לַה׳ וְאָז יִנָּחֵם הָאֱלֹהִים וְיָשׁוּב מֵחֲרוֹן אַפּוֹ׳. כָּל אֶחָד צָרִיךְ לִבְדֹּק מָה בְּיָדוֹ וְלָשׁוּב. וְהַתִּקְוָה: אָדָם לֹא בָּטוּחַ, אֲבָל ה׳ יוֹדֵעַ לְקַבֵּל.

תְּשׁוּבָה אֲמִתִּית מַתְחִילָה בִּבְדִיקַת מָה שֶׁבְּיָדִי, וּבְתִקְוָה שֶׁה׳ יִנָּחֵם.

📜 הַפָּסוּקוַיַּרְא הָאֱלֹהִים אֶת מַעֲשֵׂיהֶם כִּי שָׁבוּ מִדַּרְכָּם הָרָעָה

רַשִׁ״י מְבָאֵר ׳וַיִּנָּחֶם הָאֱלֹהִים׳: ׳נִתְעַשֵּׁת עַל הָרָעָה לָשׁוּב מִמֶּנָּה׳. וּמְצוּדַת דָּוִד עַל ׳כִּי שָׁבוּ׳: ׳כִּי כֵן עָשׂוּ כִּדְבַר הַמֶּלֶךְ׳. כְּלוֹמַר, נִינְוֵה עָשְׂתָה בְּפוֹעַל מָה שֶׁהַמֶּלֶךְ אָמַר, וְה׳ רָאָה אֶת הַמַּעֲשִׂים. הַפָּסוּק אוֹמֵר ׳אֶת מַעֲשֵׂיהֶם׳, וְלֹא אֶת הַצּוֹם.

ה׳ רוֹאֶה אֶת הַמַּעֲשִׂים שֶׁלָּנוּ, וּכְשֶׁהֵם מִשְׁתַּנִּים – הוּא מִתְעַשֵּׁת.

  • הַמֶּלֶךְ יָרַד מִן הַכִּסֵּא וְהוֹרִיד אֶת בִּגְדֵי הַמַּלְכוּת. מַהוּ ׳הַכִּסֵּא׳ שֶׁלִּי שֶׁקָּשֶׁה לִי לָרֶדֶת מִמֶּנּוּ כְּשֶׁאֲנִי צָרִיךְ לוֹמַר ׳סְלִיחָה׳?
  • הַפָּסוּק אוֹמֵר שֶׁה׳ רָאָה ׳אֶת מַעֲשֵׂיהֶם׳. אֵיזֶה מַעֲשֶׂה אֶחָד אֲנִי יָכוֹל לְשַׁנּוֹת הַשָּׁבוּעַ כְּדֵי לַעֲשׂוֹת תְּשׁוּבָה אֲמִתִּית?
  • הַמֶּלֶךְ אָמַר ׳מִי יוֹדֵעַ׳ וְהוֹסִיף תִּקְוָה. אֵיךְ אֲנִי יָכוֹל לַחֲזֹר אֶל ה׳ גַּם כְּשֶׁאֲנִי לֹא בָּטוּחַ שֶׁזֶּה יַצְלִיחַ?

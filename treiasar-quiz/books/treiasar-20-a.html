חַגַּי הוּא אֶחָד מִשְּׁלוֹשָׁה הַנְּבִיאִים הָאַחֲרוֹנִים, בִּימֵי שִׁבֵי צִיּוֹן. הַסֵּפֶר פּוֹתֵחַ בְּתַאֲרִיךְ מְדֻיָּק: ׳בִּשְׁנַת שְׁתַּיִם לְדָרְיָוֶשׁ הַמֶּלֶךְ בַּחֹדֶשׁ הַשִּׁשִּׁי בְּיוֹם אֶחָד לַחֹדֶשׁ׳. דְּבַר ה׳ בָּא אֶל זְרֻבָּבֶל פַּחַת יְהוּדָה וְאֶל יְהוֹשֻׁעַ הַכֹּהֵן הַגָּדוֹל. רַשִׁ״י מְבָאֵר שֶׁזֶּה דָּרְיָוֶשׁ מֶלֶךְ פָּרַס. הָעָם חָזַר מִבָּבֶל, וְהַנָּבִיא בָּא לְעוֹרֵר אוֹתוֹ.

חַגַּי מְדַבֵּר אֶל שִׁבֵי צִיּוֹן שֶׁחָזְרוּ מִבָּבֶל – אֶל הַמְּנַהֲיגִים וְאֶל הָעָם.

הַנָּבִיא שׁוֹאֵל: אֵיךְ אֶפְשָׁר לָשֶׁבֶת בְּבָתִּים יָפִים כְּשֶׁבֵּית ה׳ חָרֵב?

חַגַּי אוֹמֵר: ׳שִׂימוּ לְבַבְכֶם עַל דַּרְכֵיכֶם׳ – תִּתְבּוֹנְנוּ בְּמַה שֶּׁקּוֹרֶה לָכֶם. ׳זְרַעְתֶּם הַרְבֵּה וְהָבֵא מְעָט אָכוֹל וְאֵין לְשָׂבְעָה שָׁתוֹ וְאֵין לְשָׁכְרָה׳ – זְרַעְתֶּם הַרְבֵּה וְהִצְלַחְתֶּם מְעַט, אֲכַלְתֶּם וְלֹא שְׂבַעְתֶּם. אַתֶּם עוֹבְדִים קָשֶׁה וְהַבְּרָכָה חֲסֵרָה. אִם תִּתְבּוֹנְנוּ תָּבִינוּ מַדּוּעַ.

׳שִׂימוּ לְבַבְכֶם׳ – עוֹצְרִים, חוֹשְׁבִים, מְחַבְּרִים בֵּין הַמַּעֲשֶׂה לַתּוֹצָאָה.

וְעַכְשָׁו בָּאָה הַהוֹרָאָה בְּרוּרָה: ׳עֲלוּ הָהָר וַהֲבֵאתֶם עֵץ וּבְנוּ הַבַּיִת וְאֶרְצֶה בּוֹ וְאֶכָּבְדָה אָמַר ה׳׳. כָּל אֶחָד עוֹלֶה לָהָר וּמֵבִיא עֵץ, וְיַחַד בּוֹנִים. וְה׳ אוֹמֵר בְּאַהֲבָה: ׳וְאֶרְצֶה בּוֹ׳ – אֲנִי אֶתְרַצֶּה בַּבַּיִת, גַּם אִם יִהְיֶה קָטָן. כְּשֶׁלֹּא בּוֹנִים, ׳כָּלְאוּ שָׁמַיִם מִטָּל׳, אֲבָל בְּבַקָּשָׁה לִבְנוֹת יֵשׁ רָצוֹן.

עוֹלִים לָהָר, מְבִיאִים עֵץ, וְבוֹנִים – וה׳ אוֹמֵר: ׳וְאֶרְצֶה בּוֹ׳.

רָמָה א׳ – עֲלוּ הָהָר וּבְנוּ

💪 ׳וַיָּעַר ה׳ אֶת רוּחַ זְרֻבָּבֶל׳

הָעָם שׁוֹמֵעַ: ׳וַיִּשְׁמַע זְרֻבָּבֶל… וִיהוֹשֻׁעַ… וְכֹל שְׁאֵרִית הָעָם בְּקוֹל ה׳׳, ׳וַיִּירְאוּ הָעָם מִפְּנֵי ה׳׳. ׳וַיָּעַר ה׳ אֶת רוּחַ זְרֻבָּבֶל… וְאֶת רוּחַ יְהוֹשֻׁעַ וְאֶת רוּחַ כֹּל שְׁאֵרִית הָעָם וַיָּבֹאוּ וַיַּעֲשׂוּ מְלָאכָה בְּבֵית ה׳ צְבָאוֹת אֱלֹהֵיהֶם׳. כֻּלָּם יַחַד – הַמְּנַהֵל, הַכֹּהֵן וְהָעָם – שָׁמְעוּ וְקָמוּ לַעֲבוֹדָה, כִּי ה׳ עוֹרֵר אֶת לִבָּם.

כְּשֶׁשּׁוֹמְעִים אֶת קוֹל ה׳ וְקָמִים יַחַד – הָעֲבוֹדָה מִתְחִילָה.

בְּאֶמְצַע הַדְּבָרִים בָּאָה הַהַבְטָחָה הַקְּצָרָה וְהַחֲזָקָה: ׳וַיֹּאמֶר חַגַּי מַלְאַךְ ה׳ בְּמַלְאֲכוּת ה׳ לָעָם לֵאמֹר אֲנִי אִתְּכֶם נְאֻם ה׳׳. חַגַּי הוּא שָׁלִיחַ ה׳, וּמְבַשֵּׂר: ה׳ נִמְצָא אִתְּכֶם. אֵין לִפְחֹד. אָדָם שֶׁיּוֹדֵעַ שֶׁה׳ אִתּוֹ מַתְחִיל לִבְנוֹת בְּאֹמֶץ.

הַמִּשְׁפָּט הַקָּצָר ׳אֲנִי אִתְּכֶם׳ נוֹתֵן כֹּחַ לְהַתְחִיל.

👴 ׳מִי בָכֶם הַנִּשְׁאָר אֲשֶׁר רָאָה׳

חֹדֶשׁ אַחֵר, וַחֲשָׁשׁ חָדָשׁ: זְקֵנִים שֶׁזָּכְרוּ אֶת הַבַּיִת הָרִאשׁוֹן רָאוּ שֶׁהַבַּיִת הַחָדָשׁ נִרְאֶה קָטָן. חַגַּי שׁוֹאֵל: ׳מִי בָכֶם הַנִּשְׁאָר אֲשֶׁר רָאָה אֶת הַבַּיִת הַזֶּה בִּכְבוֹדוֹ הָרִאשׁוֹן… הֲלוֹא כָמֹהוּ כְּאַיִן בְּעֵינֵיכֶם׳. וְהוּא עוֹנֶה: ׳חֲזַק זְרֻבָּבֶל… וַחֲזַק כָּל עַם הָאָרֶץ וַעֲשׂוּ כִּי אֲנִי אִתְּכֶם׳. וְעוֹד הַבְטָחָה: ׳וְרוּחִי עֹמֶדֶת בְּתוֹכְכֶם אַל תִּירָאוּ׳.

גַּם כְּשֶׁהַבַּיִת נִרְאֶה קָטָן, ה׳ אוֹמֵר: ׳חֲזַק… כִּי אֲנִי אִתְּכֶם׳.

הַבַּיִת הָאַחֲרוֹן קָטָן עַכְשָׁו – וְה׳ מַבְטִיחַ לוֹ כָּבוֹד גָּדוֹל וְשָׁלוֹם.

מָה לוֹקְחִים מִרָמָה א׳?

חַגַּי אָמַר לַשָּׁבִים מִבָּבֶל: אַתֶּם בּוֹנִים בָּתִּים יָפִים, וְה׳ מְחַכֶּה שֶׁתִּבְנוּ לוֹ בַּיִת. וּבְאַהֲבָה הוּא מַבְטִיחַ: ׳וְאֶרְצֶה בּוֹ׳ וְ׳אֲנִי אִתְּכֶם׳. ה׳ אֵינוֹ מְבַקֵּשׁ בַּיִת מְפֹאָר – הוּא מְבַקֵּשׁ שֶׁנָּבוֹא אֵלָיו.

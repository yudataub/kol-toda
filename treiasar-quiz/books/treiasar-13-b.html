רַשִׁ״י מְבָאֵר ׳וַיֵּרַע אֶל יוֹנָה׳: ׳אָמַר: עַכְשָׁיו יֹאמְרוּ הָעַכּוּ״ם שֶׁאֲנִי נְבִיא הַשֶּׁקֶר׳. וּמְצוּדַת דָּוִד: ׳בַּנְּבוּאָה נֶאֱמַר לוֹ שֶׁנִּיחַם ה׳ וְהָיָה רַע בְּעֵינָיו, וְחָרָה עַל עַצְמוֹ כְּדֶרֶךְ הַמֵּצֵר׳. יוֹנָה כּוֹעֵס, אֲבָל לֹא עַל נִינְוֵה: הוּא דּוֹאֵג לַשֵּׁם הַטּוֹב שֶׁלּוֹ, וְשֶׁלֹּא יֹאמְרוּ שֶׁהוּא שֶׁקֶר.

יוֹנָה כּוֹעֵס כִּי הוּא פּוֹחֵד שֶׁיֹּאמְרוּ שֶׁנְּבוּאָתוֹ שֶׁל שֶׁקֶר.

יוֹנָה אוֹמֵר: ׳הֲלוֹא זֶה דְבָרִי עַד הֱיוֹתִי עַל אַדְמָתִי׳. רַשִׁ״י מְבָאֵר: ׳יוֹדֵעַ אֲנִי שֶׁאִם יַחְזְרוּ בִּתְשׁוּבָה לֹא תַּחֲרִיבֵם וְאֶהְיֶה שַׁקְרָן בְּעֵינֵיהֶם׳. וּמְצוּדַת דָּוִד מוֹסִיף: ׳הָיָה טוֹב בְּעֵינַי שֶׁלֹּא יָשׁוּבוּ וְיֹאבְדוּ... וְלֹא יִהְיוּ רְצוּעָה מַרְדּוּת עַל יִשְׂרָאֵל׳, ׳וּלְפִי דִּבְרֵי חֲזַ״ל... בִּרְאוֹתָם כֹּחַ הַתְּשׁוּבָה׳.

יוֹנָה יָדַע שֶׁה׳ חַנּוּן וְרַחוּם, וְלָכֵן בָּרַח; הוּא דָּאַג לְעַמּוֹ וְלִכְבוֹדוֹ.

🎯 מְצוּדַת דָּוִד: ׳הֲרָאֶה לוֹ מוֹפֵת׳

עַל ׳הַהֵיטֵב חָרָה לָךְ׳ מְצוּדַת דָּוִד מְבָאֵר: ׳בְּהֵ״א הַתֵּמַהּ, וְכָאוֹמֵר: הִנֵּה אַרְאֵךְ בְּמוֹפֵת שֶׁאֵין מִן הַדִּין שֶׁיֶּחֱרֶה לְךָ׳. ה׳ אֵינוֹ מוֹכִיחַ אֶת יוֹנָה בְּדִבּוּר, אֶלָּא מַרְאֶה לוֹ מָשָׁל חַי: קִיקָיוֹן, תּוֹלַעַת וְרוּחַ. כָּךְ יוֹנָה יָבִין בְּעַצְמוֹ.

ה׳ מְלַמֵּד אֶת יוֹנָה דֶּרֶךְ מָשָׁל וְחֲוָיָה, וְלֹא דֶּרֶךְ כַּעַס.

🌱 רַשִׁ״י וּמְצוּדַת דָּוִד: הַקִּיקָיוֹן

רַשִׁ״י מְבָאֵר: ׳קִיקָיוֹן – עֵשֶׂב הַגָּדֵל לְמַעְלָה בַּעֲנָפִים רַבִּים וּמַצִּיל׳. וְ׳לְהַצִּיל לוֹ מֵרָעָתוֹ׳ – ׳מֵחוֹם הַשֶּׁמֶשׁ׳. וּמְצוּדַת דָּוִד מַסְבִּיר לָמָּה זֶה עָדִיף מִצֵּל הַסֻּכָּה: ׳צֵל הַסֻּכָּה הוּא דָּבָר שֶׁאֵינוֹ מִתְמִיד כִּי הוֹלֵךְ וּמִתְיַבֵּשׁ... וְלֹא כֵן הַקִּיקָיוֹן הַיּוֹנֵק רְטִיבוּת הָאָרֶץ׳. כָּךְ ה׳ מְפַנֵּק אֶת יוֹנָה בְּצֵל טוֹב יוֹתֵר מִזֶּה שֶׁעָשָׂה לְעַצְמוֹ.

ה׳ נוֹתֵן לְיוֹנָה צֵל טוֹב יוֹתֵר מִן הַסֻּכָּה שֶׁעָשָׂה, כִּי הַקִּיקָיוֹן ׳יוֹנֵק רְטִיבוּת הָאָרֶץ׳.

רַשִׁ״י מְבָאֵר ׳חֲרִישִׁית׳: ׳אָמְרוּ רַבּוֹתֵינוּ: בְּשָׁעָה שֶׁמְּנַשֶּׁבֶת מַשְׁתֶּקֶת כָּל הָרוּחוֹת מִפָּנֶיהָ וְהִיא חַמָּה מְאֹד׳. וּמְצוּדַת דָּוִד: ׳רוּחַ קָדִים – הִיא הָעַזָּה שֶׁבָּרוּחוֹת וְחַמָּה בְּיוֹתֵר׳. וְיוֹנָה יוֹשֵׁב בְּמִזְרַח הָעִיר, וְלָכֵן הָרוּחַ פּוֹגַעַת בּוֹ יָשָׁר. ׳וַיִּתְעַלָּף׳ – הוּא מִתְעַלֵּף בְּגַלְלָהּ.

הָרוּחַ שֶׁה׳ שׁוֹלֵחַ עַזָּה וְחַמָּה, וְיוֹנָה לוֹמֵד כַּמָּה צֵל זֶה חָשׁוּב.

📜 הַפָּסוּקאַתָּה חַסְתָּ עַל הַקִּיקָיוֹן אֲשֶׁר לֹא עָמַלְתָּ בּוֹ וְלֹא גִדַּלְתּוֹ

💛 אַהֲבַת הַשֵּׁם: ה׳ מְרַחֵם עַל כָּל מַה שֶּׁבָּרָא, וְאֲנִי רוֹצֶה לְרַחֵם כְּמוֹ שֶׁהוּא 💛

⚖️ רַשִׁ״י: ׳לֹא עָמַלְתָּ בּוֹ׳

רַשִׁ״י מְבָאֵר: ׳לֹא עָמַלְתָּ בּוֹ – בַּחֲרִישָׁה וּזְרִיעָה וְהַשְׁקָאַת מַיִם׳. וְ׳שֶׁבִּן לַיְלָה׳ – ׳לֹא גָדַל אֶלָּא לַיְלָה אֶחָד׳. וּמְצוּדַת דָּוִד: ׳הַדֶּרֶךְ שֶׁאָדָם חָס וּמִצְטַעֵר עַל אֲבֵדַת דָּבָר שֶׁעָמַל בּוֹ, וְלֹא כֵן עַל דָּבָר שֶׁלֹּא עָמַל בּוֹ׳. ה׳ מַשְׁוֶה: אַתָּה חַסְתָּ עַל קִיקָיוֹן שֶׁלֹּא עָמַלְתָּ בּוֹ, וַאֲנִי בָּרָאתִי אֶת נִינְוֵה.

אִם יוֹנָה חָס עַל קִיקָיוֹן שֶׁלֹּא עָמַל בּוֹ, ה׳ חָס בְּוַדַּאי עַל מַעֲשֵׂה יָדָיו.

רַשִׁ״י מְבָאֵר ׳אֲשֶׁר לֹא יָדַע וְגוֹ׳׳: ׳קְטַנִּים׳. וְעַל ׳וּבְהֵמָה רַבָּה׳: ׳בְּנֵי אָדָם גְּדוֹלִים וְדַעְתָּן כִּבְהֵמָה, שֶׁאֵינָם מַכִּירִים מִי בְּרָאָם׳. וּמְצוּדַת דָּוִד: ׳הֵם הַקְּטַנִּים שֶׁאֵין לַחְשֹׁב לָהֶם עָוֹן... וְעַל כָּל אֵלֶּה רָאוּי לָחוּס וְלַחְמֹל׳. ה׳ מְרַחֵם עַל הַקְּטַנִּים וְעַל הַבְּהֵמוֹת, וְעַל מִי שֶׁעוֹד לֹא הִכִּיר אוֹתוֹ.

ה׳ מְרַחֵם עַל קְטַנִּים, עַל בְּהֵמוֹת, וְעַל מִי שֶׁעוֹד לֹא הִכִּיר אוֹתוֹ.

🤔 חוֹשְׁבִים עַל זֶה

  • יוֹנָה כָּעַס כִּי חָשַׁשׁ מִשֵּׁם רַע, אֲבָל ה׳ רָצָה לְלַמֵּד אוֹתוֹ רַחֲמִים. מָתַי כָּעַסְתִּי וְהִבַּנְתִּי אַחַר כָּךְ שֶׁזֶּה בִּגְלַל שֶׁדָּאַגְתִּי לְעַצְמִי?
  • ה׳ שָׁאַל ׳הַהֵיטֵב חָרָה לָךְ׳ בְּשֶׁקֶט. אֵיךְ אֲנִי יָכוֹל לְהָגִיב בְּרַכּוּת כְּשֶׁחָבֵר כּוֹעֵס עָלַי?
  • יוֹנָה הִכִּיר טוֹבָה לַקִּיקָיוֹן. עַל אֵיזֶה ׳צֵל׳ קָטָן שֶׁה׳ נוֹתֵן לִי כָּל יוֹם אֲנִי רוֹצֶה לוֹמַר תּוֹדָה?

רָמָה א׳ – שׁוּבוּ אֵלַי

🔥 ׳וְיָשַׁב מְצָרֵף וּמְטַהֵר כֶּסֶף׳

הַנָּבִיא אוֹמֵר שֶׁה׳ שׁוֹלֵחַ שָׁלִיחַ לְפַנּוֹת דֶּרֶךְ, וְשֶׁבַּסּוֹף ׳וְיָשַׁב מְצָרֵף וּמְטַהֵר כֶּסֶף׳. הַצּוֹרֵף יוֹשֵׁב וּמְנַקֶּה אֶת הַכֶּסֶף, כְּדֵי שֶׁיִּהְיֶה זַךְ וְנָקִי. כָּךְ ה׳ מְנַקֶּה אֶת הַלֵּב שֶׁלָּנוּ, ׳וְעָרְבָה לַה׳ מִנְחַת יְהוּדָה וִירוּשָׁלָיִם כִּימֵי עוֹלָם וּכְשָׁנִים קַדְמֹנִיּוֹת׳ – הַמִּנְחָה תִּהְיֶה נְעִימָה וּמְרֻצָּה כְּמוֹ בַּעבָר.

ה׳ כְּצוֹרֵף שֶׁמְּנַקֶּה כֶּסֶף – כְּדֵי שֶׁנֵּהְיֶה זַכִּים וּנְעִימִים לְפָנָיו.

💛 אַהֲבַת הַשֵּׁם: אֲנִי יָרֵא אֶת ה׳ כְּשֶׁאֲנִי הוֹגֵן וְיָשָׁר עִם כָּל אָדָם 💛

📖 מַלְאָכִי, פֶּרֶק ג׳, פָּסוּק ה׳

רָמָה א׳ – שׁוּבוּ אֵלַי

⚖️ ׳וְלֹא יְרֵאוּנִי׳

ה׳ מְבַקֵּשׁ יֹשֶׁר: ׳וְקָרַבְתִּי אֲלֵיכֶם לַמִּשְׁפָּט וְהָיִיתִי עֵד מְמַהֵר… וּבְעֹשְׁקֵי שְׂכַר שָׂכִיר אַלְמָנָה וְיָתוֹם וּמַטֵּי גֵר וְלֹא יְרֵאוּנִי אָמַר ה׳ צְבָאוֹת׳. מִי שֶׁלֹּא מְשַׁלֵּם שָׂכָר הוֹגֵן לַפּוֹעֵל, מִי שֶׁעוֹשֵׁק אֶת הַחַלָּשׁ וְאֶת הַגֵּר – הוּא לֹא יָרֵא אֶת ה׳. הַיִּרְאָה הִיא לֹא פַּחַד אֶלָּא יֹשֶׁר בְּעֵסֶק עִם אֲנָשִׁים.

יִרְאַת ה׳ מִתְבַּטֵּאת בְּיֹשֶׁר: שָׂכָר הוֹגֵן וְלֹא לַעֲשֹׁק אֶת הַחַלָּשׁ.

🏔️

💛 אַהֲבַת הַשֵּׁם: ה׳ נֶאֱמָן אֵלַי וּמְחַכֶּה לִי, גַּם כְּשֶׁסַּרְתִּי 💛

📖 מַלְאָכִי, פֶּרֶק ג׳, פְּסוּקִים ו׳–ז׳

רָמָה א׳ – שׁוּבוּ אֵלַי

🏔️ ׳כִּי אֲנִי ה׳ לֹא שָׁנִיתִי׳

ה׳ אוֹמֵר לָעָם: ׳כִּי אֲנִי ה׳ לֹא שָׁנִיתִי וְאַתֶּם בְּנֵי יַעֲקֹב לֹא כְלִיתֶם׳. ה׳ אֵינוֹ מִשְׁתַּנֶּה, וְלָכֵן הַבְּנֵי יַעֲקֹב עוֹדָם קַיָּמִים. וְאָז הַהַזְמָנָה: ׳לְמִימֵי אֲבֹתֵיכֶם סַרְתֶּם מֵחֻקַּי וְלֹא שְׁמַרְתֶּם שׁוּבוּ אֵלַי וְאָשׁוּבָה אֲלֵיכֶם׳. אַף עַל פִּי שֶׁהָעָם סָר – ה׳ מַמְתִּין לָהֶם.

ה׳ לֹא מִשְׁתַּנֶּה, וְלָכֵן תָּמִיד אֶפְשָׁר לָשׁוּב אֵלָיו.

3
🌾

💛 אַהֲבַת הַשֵּׁם: אֲנִי נוֹתֵן חֵלֶק מִמַּה שֶּׁה׳ נָתַן לִי, וְשָׂמֵחַ בַּנְּתִינָה 💛

📖 מַלְאָכִי, פֶּרֶק ג׳, פְּסוּקִים ח׳–ט׳

רָמָה א׳ – שׁוּבוּ אֵלַי

🌾 ׳הַמַּעֲשֵׂר וְהַתְּרוּמָה׳

ה׳ שׁוֹאֵל: ׳הֲיִקְבַּע אָדָם אֱלֹהִים כִּי אַתֶּם קֹבְעִים אֹתִי׳. הָעָם שׁוֹאֵל ׳בַּמֶּה קְבַעֲנוּךָ׳, וה׳ עוֹנֶה: ׳הַמַּעֲשֵׂר וְהַתְּרוּמָה׳. מִי שֶׁלֹּא נוֹתֵן אֶת הַמַּעֲשֵׂר, כְּאִלּוּ גּוֹזֵל. הָאֲדָמָה וְהַיְּבוּל הֵם מַתָּנָה מֵה׳, וְחֵלֶק קָטָן מִמֶּנָּה צָרִיךְ לַחֲזֹר אֶל מִי שֶׁזָּקוּק לוֹ: הַכֹּהֲנִים, הַלְוִיִּם וְהָעֲנִיִּים.

הַמַּעֲשֵׂר הוּא חֵלֶק מִן הַמַּתָּנָה שֶׁחוֹזֵר אֶל מִי שֶׁצָּרִיךְ.

🌧️

💛 אַהֲבַת הַשֵּׁם: ה׳ מַזְמִין אוֹתִי לָתֵת, וּמַבְטִיחַ לִתֵּן הַרְבֵּה יוֹתֵר 💛

📖 מַלְאָכִי, פֶּרֶק ג׳, פְּסוּקִים י׳–י״ב

רָמָה א׳ – שׁוּבוּ אֵלַי

🌧️ ׳וּבְחָנוּנִי נָא בָּזֹאת׳

וְאָז הַהַבְטָחָה הַמְּיֻחֶדֶת: ׳הָבִיאוּ אֶת כָּל הַמַּעֲשֵׂר אֶל בֵּית הָאוֹצָר וִיהִי טֶרֶף בְּבֵיתִי וּבְחָנוּנִי נָא בָּזֹאת אָמַר ה׳ צְבָאוֹת אִם לֹא אֶפְתַּח לָכֶם אֵת אֲרֻבּוֹת הַשָּׁמַיִם וַהֲרִיקֹתִי לָכֶם בְּרָכָה עַד בְּלִי דָי׳. ה׳ מַזְמִין אוֹתָנוּ לְנַסּוֹת – אִם נִתֵּן, הוּא יִתֵּן בְּרָכָה גְּדוֹלָה. ׳וְאִשְּׁרוּ אֶתְכֶם כָּל הַגּוֹיִם כִּי תִהְיוּ אַתֶּם אֶרֶץ חֵפֶץ׳.

ה׳ מַבְטִיחַ: מִי שֶׁנּוֹתֵן יִקַּח בְּרָכָה ׳עַד בְּלִי דָי׳.

📖

💛 אַהֲבַת הַשֵּׁם: ה׳ מַקְשִׁיב לִשְׂיחוֹת הַטּוֹבוֹת שֶׁלִּי עִם חֲבֵרַי וְחוֹמֵל עָלַי כְּמוֹ אָב 💛

📖 מַלְאָכִי, פֶּרֶק ג׳, פְּסוּקִים ט״ז–י״ז

רָמָה א׳ – שׁוּבוּ אֵלַי

📖 ׳אָז נִדְבְּרוּ יִרְאֵי ה׳׳

וְעַל זֶה הַתְּשׁוּבָה הַיָּפָה בַּסֵּפֶר: ׳אָז נִדְבְּרוּ יִרְאֵי ה׳ אִישׁ אֶת רֵעֵהוּ וַיַּקְשֵׁב ה׳ וַיִּשְׁמָע וַיִּכָּתֵב סֵפֶר זִכָּרוֹן לְפָנָיו לְיִרְאֵי ה׳ וּלְחֹשְׁבֵי שְׁמוֹ׳. אֲנָשִׁים הוֹלְכִים וּמְדַבְּרִים זֶה עִם זֶה בְּיִרְאַת ה׳, וה׳ ׳מַקְשִׁיב וְשׁוֹמֵעַ׳ וְכוֹתֵב הַכֹּל בְּסֵפֶר זִכָּרוֹן. ׳וְחָמַלְתִּי עֲלֵיהֶם כַּאֲשֶׁר יַחְמֹל אִישׁ עַל בְּנוֹ הָעֹבֵד אֹתוֹ׳.

שִׂיחָה טוֹבָה בֵּין חֲבֵרִים נִכְתֶּבֶת בְּסֵפֶר זִכָּרוֹן, וה׳ חוֹמֵל כְּמוֹ אָב.

🌅

💛 אַהֲבַת הַשֵּׁם: ה׳ מַאִיר אֶת חַיַּי בְּאוֹר שֶׁל רְפוּאָה וְשִׂמְחָה 💛

רָמָה א׳ – שׁוּבוּ אֵלַי

🌅 ׳שֶׁמֶשׁ צְדָקָה וּמַרְפֵּא בִּכְנָפֶיהָ׳

וְהַסּוֹף הַמֵּאִיר: ׳וְזָרְחָה לָכֶם יִרְאֵי שְׁמִי שֶׁמֶשׁ צְדָקָה וּמַרְפֵּא בִּכְנָפֶיהָ וִיצָאתֶם וּפִשְׁתֶּם כְּעֶגְלֵי מַרְבֵּק׳. מִי שֶׁיָּרֵא אֶת ה׳ – שֶׁמֶשׁ צְדָקָה זוֹרַחַת לוֹ, וְיֵשׁ רְפוּאָה בְּאוֹרָהּ. וְיֵצְאוּ וְיִקְפְּצוּ כְּמוֹ עֲגָלִים בְּשָׂדֶה. זוֹ תְּמוּנָה שֶׁל שִׂמְחָה וְחַיִּים.

לְמִי שֶׁיָּרֵא אֶת ה׳ זוֹרַחַת שֶׁמֶשׁ צְדָקָה וּמַרְפֵּא.

הַפְּסוּקִים הָאַחֲרוֹנִים שֶׁל הַנְּבִיאִים: ׳זִכְרוּ תּוֹרַת מֹשֶׁה עַבְדִּי… הִנֵּה אָנֹכִי שֹׁלֵחַ לָכֶם אֵת אֵלִיָּה הַנָּבִיא לִפְנֵי בּוֹא יוֹם ה׳ הַגָּדוֹל וְהַנּוֹרָא וְהֵשִׁיב לֵב אָבוֹת עַל בָּנִים וְלֵב בָּנִים עַל אֲבוֹתָם׳. אֵלִיָּהוּ יָבוֹא לְחַבֵּר בֵּין הַדּוֹרוֹת: אָבוֹת וּבָנִים יַחֲזְרוּ לִהְיוֹת קְרוֹבִים, וְיָשִׁיבוּ אֶת הַלֵּב לַה׳.

אֵלִיָּהוּ יִשִּׂים שָׁלוֹם בֵּין אָבוֹת לְבָנִים וְיָשִׁיב אֶת הַלֵּב.

מָה לוֹקְחִים מִרָמָה א׳?

מַלְאָכִי מַזְכִּיר שֶׁה׳ לֹא מִשְׁתַּנֶּה, מַזְמִין אוֹתָנוּ לָשׁוּב וְלָתֵת, שׁוֹמֵעַ אֶת הַשִּׂיחוֹת הַטּוֹבוֹת וְחוֹמֵל כְּמוֹ אָב. הוּא מַבְטִיחַ אוֹר וּרְפוּאָה, וּמְסַיֵּם בְּתִקְוָה שֶׁל אֵלִיָּהוּ.

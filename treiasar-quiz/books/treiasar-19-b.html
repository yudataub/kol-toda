📜 תְּרֵי עָשָׂר · יָשִׂישׂ עָלַיִךְ בְּשִׂמְחָה · רָמָה ב׳

יָשִׂישׂ עָלַיִךְ בְּשִׂמְחָה

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

צְפַנְיָה קוֹרֵא לָשׁוּב וּלְבַקֵּשׁ עֲנָוָה, וּמְסַיֵּם בְּשִׁיר: ה׳ בְּקִרְבֵּנוּ, שָׂמֵחַ עָלֵינוּ וְאוֹהֵב

📖 פְּרָקִים א׳–ג׳

🎯 לַחִידוֹן

📜 הַפָּסוּקהִתְקוֹשְׁשׁוּ וָקוֹשּׁוּ הַגּוֹי לֹא נִכְסָף

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַתְחִיל בְּעַצְמִי – וּבָזֶה אֲנִי מוֹדֶה לַה׳ שֶׁנּוֹתֵן לִי הִזְדַּמְּנוּת לְתַקֵּן 💛

📖 צְפַנְיָה, פֶּרֶק ב׳, פָּסוּק א׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🌾 רַשִׁ״י: ׳הִתְקוֹשְׁשׁוּ וָקוֹשּׁוּ׳

רַשִׁ״י מְבָאֵר: ׳הִתְקוֹשְׁשׁוּ – הִתְלַקְּטוּ וְאִסְפוּ יַחַד, כְּמוֹ לְקוֹשֵׁשׁ קַשׁ׳. ׳וָקוֹשּׁוּ – מַעֲשֵׂיכֶם הַשְׁווּ מַעֲשֵׂיכֶם לְדַעַת קוֹנְכֶם׳. וְרַשִׁ״י מוֹסִיף: חֲכָמֵינוּ דָּרְשׁוּ ׳קַשֵּׁט עַצְמְךָ וְאַחַר כָּךְ קַשֵּׁט אֲחֵרִים׳. קֹדֶם מְתַקְּנִים אֶת עַצְמֵנוּ, וְאַחַר כָּךְ אֶת הַסְּבִיבָה.

קֹדֶם מְיַשְּׁרִים אֶת עַצְמֵנוּ לִרְצוֹן ה׳, וְאַחַר כָּךְ מְסַיְּעִים לַאֲחֵרִים.

1

📜 הַפָּסוּקבְּטֶרֶם לֶדֶת חֹק כְּמֹץ עָבַר יוֹם בְּטֶרֶם לֹא־יָבוֹא עֲלֵיכֶם חֲרוֹן אַף־ה׳

💛 אַהֲבַת הַשֵּׁם: ה׳ מַמְתִּין לִי לִפְנֵי הַגְּזֵרָה, כִּי הוּא רוֹצֶה שֶׁאֶשּׁוּב 💛

📖 צְפַנְיָה, פֶּרֶק ב׳, פָּסוּק ב׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

⏱️ מְצוּדַת דָּוִד: ׳בְּטֶרֶם לֶדֶת חֹק׳

הַנָּבִיא אוֹמֵר: ׳בְּטֶרֶם לֶדֶת חֹק כְּמֹץ עָבַר יוֹם׳. מְצוּדַת דָּוִד מְבָאֵר: ׳עַד לֹא יוּלַּד חֹק הַגְּזֵרָה׳, וְ׳כְּמֹץ עָבַר יוֹם׳ – כְּמוֹ מֹץ הָעוֹבֵר בָּרוּחַ. רַשִׁ״י: ׳לִפְנֵי בֹּא הַגְּזֵרָה׳. כְּלוֹמַר: הַגְּזֵרָה עֲדַיִן לֹא נוֹלְדָה, וְיֵשׁ עוֹד זְמַן לְתַקֵּן. כְּשֶׁשָּׁבִים בַּזְּמַן, הַפּוּרְעָנוּת עוֹבֶרֶת כְּמֹץ.

בְּטֶרֶם לֶדֶת חֹק – עוֹד לֹא מְאֻחָּר, וְאֶפְשָׁר לָשׁוּב.

2

📜 הַפָּסוּקבַּקְּשׁוּ אֶת־ה׳ כָּל־עַנְוֵי הָאָרֶץ אֲשֶׁר מִשְׁפָּטוֹ פָּעָלוּ בַּקְּשׁוּ־צֶדֶק בַּקְּשׁוּ עֲנָוָה

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא צָרִיךְ לְהִיוֹת מֻשְׁלָם כְּדֵי לְבַקֵּשׁ אֶת ה׳ 💛

📖 צְפַנְיָה, פֶּרֶק ב׳, פָּסוּק ג׳ (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🙇 מְצוּדַת דָּוִד: ׳אוּלַי תִּסָּתְרוּ׳

מְצוּדַת דָּוִד מְבָאֵר אֶת ׳בַּקְּשׁוּ צֶדֶק בַּקְּשׁוּ עֲנָוָה׳: ׳וְאַתֶּם יֶתֶר הָעָם בַּקְּשׁוּ מֵעַתָּה צֶדֶק וַעֲנָוָה, אוּלַי עַל יְדֵי זֶה תִּהְיוּ נִסְתָּרִים בְּיוֹם אַף ה׳׳. וְעַל הַמִּלִּים ׳אֲשֶׁר מִשְׁפָּטוֹ פָּעָלוּ׳ הוּא מְבָאֵר: ׳אֲשֶׁר עָשׂוּ מִשְׁפַּט ה׳׳. יֵשׁ שְׁנֵי סוּגִים: מִי שֶׁכְּבָר עוֹשֶׂה מִשְׁפַּט ה׳, וּמִי שֶׁעוֹד צָרִיךְ לְבַקֵּשׁ צֶדֶק וַעֲנָוָה.

גַּם מִי שֶׁעוֹד לֹא שָׁלֵם יָכוֹל לְבַקֵּשׁ צֶדֶק וַעֲנָוָה – ׳אוּלַי׳ יִסָּתֵר.

3

📜 הַפָּסוּקנְבִיאֶיהָ פֹּחֲזִים אַנְשֵׁי בֹּגְדוֹת כֹּהֲנֶיהָ חִלְּלוּ־קֹדֶשׁ חָמְסוּ תוֹרָה

💛 אַהֲבַת הַשֵּׁם: מָה שֶׁאֲנִי לוֹמֵד אֲנִי רוֹצֶה לַעֲבֹר הָלְאָה, כִּי זֶה חֶסֶד גָּדוֹל 💛

📖 צְפַנְיָה, פֶּרֶק ג׳, פָּסוּק ד׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

📚 רַשִׁ״י: ׳חָמְסוּ תּוֹרָה׳

הַנָּבִיא מוֹכִיחַ אֶת הַכֹּהֲנִים: ׳כֹּהֲנֶיהָ חִלְּלוּ קֹדֶשׁ חָמְסוּ תּוֹרָה׳. רַשִׁ״י מְבָאֵר: ׳חָמְסוּ תּוֹרָה – לֹא הוֹרוּ אוֹתָהּ לְשׁוֹאֲלֵיהֶם׳. וּמְצוּדַת דָּוִד: ׳עַל הַכֹּהֵן לְלַמֵּד, כְּמוֹ שֶׁנֶּאֱמַר שִׂפְתֵי כֹהֵן יִשְׁמְרוּ דַעַת וְתוֹרָה יְבַקְשׁוּ מִפִּיהוּ׳. מִי שֶׁיּוֹדֵעַ תּוֹרָה וְלֹא מְלַמֵּד אוֹתָהּ – כְּאִלּוּ גָּזַל אוֹתָהּ מִן הַשּׁוֹאֲלִים.

תּוֹרָה צְרִיכָה לְהִתְלַמֵּד וּלְהִתְלַמֵּד לָאֲחֵרִים – אֲחֶרֶת הִיא ׳נֶחְמְסָה׳.

4

📜 הַפָּסוּקה׳ צַדִּיק בְּקִרְבָּהּ לֹא יַעֲשֶׂה עַוְלָה בַּבֹּקֶר בַּבֹּקֶר מִשְׁפָּטוֹ יִתֵּן לָאוֹר

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי זוֹכֵר שֶׁה׳ בְּתוֹכִי, קַל לִי לִהְיוֹת יָשָׁר 💛

📖 צְפַנְיָה, פֶּרֶק ג׳, פָּסוּק ה׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🔆 רַשִׁ״י: ה׳ צַדִּיק בְּקִרְבָּהּ

רַשִׁ״י מְבָאֵר: ׳ה׳ צַדִּיק – הִשְׁרָה שְׁכִינָתוֹ בְּקִרְבָּם׳. ׳מִשְׁפָּטוֹ יִתֵּן לָאוֹר – דָּן דִּין אֱמֶת לַאֲמִתּוֹ׳. ׳וְלֹא יוֹדֵעַ עַוָּל בֹּשֶׁת – וְכָל זֶה לֹא נָתְנוּ שׁוֹפְטֵי עָוֶל שֶׁבְּתוֹכָהּ אֶל לִבָּם לְהִתְבַּיֵּשׁ מִן הַצַּדִּיק הַשּׁוֹכֵן בֵּינֵיהֶם׳. מְצוּדַת דָּוִד: ׳בַּבֹּקֶר בַּבֹּקֶר – בְּכָל יוֹם וָיוֹם׳. ה׳ קָרוֹב כָּל כָּךְ, וְשׁוֹפְטֵי הָעָוֶל אֵינָם מִתְבַּיְּשִׁים.

ה׳ נִמְצָא בְּתוֹךְ הָעִיר, וּמִי שֶׁזּוֹכֵר זֹאת מִתְבַּיֵּשׁ לַעֲשׂוֹת עָוֶל.

5

📜 הַפָּסוּקה׳ אֱלֹהַיִךְ בְּקִרְבֵּךְ גִּבּוֹר יוֹשִׁיעַ יָשִׂישׂ עָלַיִךְ בְּשִׂמְחָה יַחֲרִישׁ בְּאַהֲבָתוֹ יָגִיל עָלַיִךְ בְּרִנָּה

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי חוֹזֵר בִּתְשׁוּבָה, ה׳ לֹא מַכְלִים אוֹתִי – הוּא מְקַבֵּל אוֹתִי בְּאַהֲבָה 💛

📖 צְפַנְיָה, פֶּרֶק ג׳, פָּסוּק י״ז (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🤍 רַשִׁ״י: ׳יַחֲרִישׁ בְּאַהֲבָתוֹ׳

מָה פֵּרוּשׁ ׳יַחֲרִישׁ בְּאַהֲבָתוֹ׳? רַשִׁ״י מְבָאֵר: ׳יְכַסֶּה עַל פְּשָׁעַיִךְ בְּאַהֲבָתוֹ׳. וּמְצוּדַת דָּוִד: ׳בַּעֲבוּר אַהֲבָתוֹ לָךְ יִשְׁתֹּק עַל הָעֲוֹנוֹת… וְלֹא יַזְכִּירָם לְךָ לְהַכְלִימֶךָ׳. ה׳ גִּבּוֹר שֶׁמּוֹשִׁיעַ, וּמִתּוֹךְ אַהֲבָה הוּא אֵינוֹ מַזְכִּיר לָנוּ אֶת הַטָּעוּיוֹת שֶׁעָבְרוּ. זוֹ אַהֲבָה שֶׁמְּכַסָּה וְאֵינָהּ מַכְלִימָה.

מֵאַהֲבָה, ה׳ מְכַסֶּה עַל הַפְּשָׁעִים וְלֹא מַכְלִים אוֹתָנוּ.

📜 הַפָּסוּקנוּגֵי מִמּוֹעֵד אָסַפְתִּי מִמֵּךְ הָיוּ מַשְׂאֵת עָלֶיהָ חֶרְפָּה

💛 אַהֲבַת הַשֵּׁם: ה׳ מַחְזִיר אֵלָיו גַּם אֶת מִי שֶׁהִתְרַחֵק וְגַם אֶת מִי שֶׁכּוֹאֵב 💛

📖 צְפַנְיָה, פֶּרֶק ג׳, פָּסוּק י״ח (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🔀 שְׁנֵי בֵּאוּרִים: ׳נוּגֵי מִמּוֹעֵד׳

הַפָּסוּק ׳נוּגֵי מִמּוֹעֵד אָסַפְתִּי מִמֵּךְ׳ מִתְפָּרֵשׁ בִּשְׁנֵי אֳפָנִים. רַשִׁ״י: ׳אָסַפְתִּי אֶת הַמּוֹצָאִים מִמּוֹעֲדֵי שֶׁלֹּא הָיוּ מְשַׁמְּרִים שַׁבָּתוֹת וִימִים טוֹבִים׳ – ׳נוּגֵי׳ מִלְּשׁוֹן הוֹצָאָה. מְצוּדַת דָּוִד: ׳בָּנַיִךְ שֶׁהָיוּ מְלֵאֵי יָגוֹן בַּעֲבוּר אִחוּר זְמַן הַגְּאֻלָּה הִכְנַסְתִּים אֵלַיִךְ׳. אֶחָד מַבִּיט עַל מִי שֶׁהִתְרַחֵק, וְהַשֵּׁנִי עַל מִי שֶׁכּוֹאֵב – וּבִשְׁנֵיהֶם ה׳ מְקַבֵּץ וּמַחְזִיר.

בֵּין אִם הֵם רְחוֹקִים אוֹ כּוֹאֲבִים – ה׳ מְקַבֵּץ אוֹתָם וּמַחְזִיר.

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מְגַלִּים שֶׁ׳יַחֲרִישׁ בְּאַהֲבָתוֹ׳ פֵּרוּשׁוֹ אַהֲבָה שֶׁמְּכַסָּה אֶת הַפְּשָׁעִים, וְשֶׁה׳ מְקַבֵּץ גַּם אֶת מִי שֶׁהִתְרַחֵק וְגַם אֶת מִי שֶׁכּוֹאֵב. הַתּוֹכַחָה וְהָאַהֲבָה חַיּוֹת יַחַד.

יָשִׂישׂ עָלַיִךְ בְּשִׂמְחָה יַחֲרִישׁ בְּאַהֲבָתוֹ יָגִיל עָלַיִךְ בְּרִנָּה

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • צְפַנְיָה אוֹמֵר ׳בַּקְּשׁוּ צֶדֶק בַּקְּשׁוּ עֲנָוָה׳. בְּאֵיזוֹ מִדָּה אֲנִי רוֹצֶה לְהִתְחַזֵּק הַשָּׁבוּעַ – וּמָה אֶעֱשֶׂה כְּדֵי לְבַקֵּשׁ אוֹתָהּ?
  • ׳קַשֵּׁט עַצְמְךָ וְאַחַר כָּךְ קַשֵּׁט אֲחֵרִים׳. מָה דָּבָר קָטָן אֶחָד שֶׁאֲנִי יָכוֹל לְתַקֵּן בְּעַצְמִי הַיּוֹם?
  • ה׳ ׳יָשִׂישׂ עָלַיִךְ בְּשִׂמְחָה… יָגִיל עָלַיִךְ בְּרִנָּה׳. אֵיךְ אֲנִי מַרְגִּישׁ כְּשֶׁאֲנִי חוֹשֵׁב שֶׁה׳ שָׂמֵחַ עָלַי?

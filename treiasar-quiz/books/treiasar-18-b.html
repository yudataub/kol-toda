וְצַדִּיק בֶּאֱמוּנָתוֹ יִחְיֶה

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

חֲבַקּוּק שׁוֹאֵל ׳עַד אָנָה׳, עוֹמֵד עַל מִשְׁמַרְתּוֹ וּמְקַבֵּל תְּשׁוּבָה – וְאוֹהֵב אֶת ה׳ גַּם כְּשֶׁאֵין לוֹ כְּלוּם

📖 פְּרָקִים א׳–ג׳

📜 הַסֵּפֶר הַמָּלֵא
🔭

📜 הַפָּסוּקעַד־אָנָה ה׳ שִׁוַּעְתִּי וְלֹא תִשְׁמָע אֶזְעַק אֵלֶיךָ חָמָס וְלֹא תוֹשִׁיעַ

💛 אַהֲבַת הַשֵּׁם: הַנָּבִיא אוֹהֵב אֶת עַמּוֹ וּמִתְפַּלֵּל עֲלֵיהֶם מֵרֹאשׁ, וְכָךְ גַּם אֲנִי יָכוֹל 💛

📖 חֲבַקּוּק, פֶּרֶק א׳, פָּסוּק ב׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🔭 רַשִׁ״י: ׳עַד אָנָה׳ – הַנָּבִיא רוֹאֶה מֵרָחוֹק

רַשִׁ״י מְבָאֵר עַל ׳עַד אָנָה ה׳ שִׁוַּעְתִּי׳: ׳צוֹפֶה הָיָה בְּרוּחַ הַקֹּדֶשׁ שֶׁעָתִיד נְבוּכַדְנֶצַּר לִמְלֹךְ… וּלְהִיוֹת מֵצֵר לְיִשְׂרָאֵל׳ – ׳וְעַל זֹאת הָיָה מִתְאוֹנֵן וּמִתְפַּלֵּל׳. וּמְצוּדַת דָּוִד מוֹסִיף: ׳הַנָּבִיא הָיָה מִתְלוֹנֵן עַל שַׁלְוַת בָּבֶל וְגֹדֶל מֶמְשַׁלְתָּהּ׳. כְּלוֹמַר: עוֹד לִפְנֵי שֶׁקָּרָה, הוּא כְּבָר מִתְפַּלֵּל עַל הַצָּרָה.

חֲבַקּוּק מִתְפַּלֵּל עוֹד לִפְנֵי שֶׁהַצָּרָה בָּאָה – כְּמוֹ אָבִי שֶׁדּוֹאֵג לִבְנוֹ.

1
⚔️

📜 הַפָּסוּקאָז חָלַף רוּחַ וַיַּעֲבֹר וְאָשֵׁם זוּ כֹחוֹ לֵאלֹהוֹ

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹמֵר תּוֹדָה לַה׳ עַל כָּל הַצְלָחָה, כִּי הוּא הַנּוֹתֵן 💛

📖 חֲבַקּוּק, פֶּרֶק א׳, פָּסוּק י״א (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

⚔️ מְצוּדַת דָּוִד: ׳זוֹ כֹחוֹ לֵאלֹהוֹ׳

הַפָּסוּק מְתָאֵר אֶת מֶלֶךְ הַכַּשְׂדִּים: ׳אָז חָלַף רוּחַ וַיַּעֲבֹר וְאָשֵׁם זוּ כֹחוֹ לֵאלֹהוֹ׳. רַשִׁ״י מְבָאֵר: ׳הַכֹּחַ הַזֶּה הוּא נוֹתֵן לֵאלוֹהוֹ׳, וּמְצוּדַת דָּוִד: הוּא תּוֹלֶה אֶת הַהַצְלָחָה בְּאֱלִילָיו ׳וְלֹא ה׳ פָּעַל כָּל זֹאת׳. הַמַּנְצֵחַ שֶׁאֵינוֹ מוֹדֶה לַה׳ עַל הַכֹּחַ שֶׁקִּבֵּל – מְאֻשָּׁם. לָמַדְנוּ: גַּם נִצָּחוֹן הוּא מַתָּנָה, וַאֲנַחְנוּ צְרִיכִים לְהוֹדוֹת עָלָיו.

מִי שֶׁמַּצְלִיחַ וְאוֹמֵר ׳זֶה הַכֹּחַ שֶׁלִּי׳ שׁוֹכֵחַ שֶׁהַכֹּחַ הוּא מַתָּנָה.

2
⭕

📜 הַפָּסוּקעַל־מִשְׁמַרְתִּי אֶעֱמֹדָה וְאֶתְיַצְּבָה עַל־מָצוֹר וַאֲצַפֶּה לִרְאוֹת מַה־יְדַבֶּר־בִּי

💛 אַהֲבַת הַשֵּׁם: מִי שֶׁבֶּאֱמֶת רוֹצֶה לִשְׁמֹעַ אֶת ה׳, מוּכָן לַעֲמֹד וּלְחַכּוֹת 💛

📖 חֲבַקּוּק, פֶּרֶק ב׳, פָּסוּק א׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

⭕ רַשִׁ״י: חֲבַקּוּק עָג עוּגָה

עַל הַפָּסוּק ׳עַל מִשְׁמַרְתִּי אֶעֱמֹדָה׳ רַשִׁ״י מְסַפֵּר: ׳עָג עוּגָה חֲבַקּוּק וְעָמַד בְּתוֹכָהּ וְאָמַר: לֹא אָזוּז מִכָּאן עַד שֶׁאֶשְׁמַע מַה יְדַבֵּר בִּי עַל שְׁאֵלָתִי זֹאת׳. כְּלוֹמַר: הוּא צִיֵּר עִגּוּל בֶּעָפָר וְהִתְיַצֵּב בְּתוֹכוֹ כְּדֵי שֶׁלֹּא לָזוּז. לִשְׁאֹל אֶת ה׳ זֶה חָשׁוּב, וְלַחֲכוֹת לַתְּשׁוּבָה בְּסַבְלָנוּת זֶה חָשׁוּב עוֹד יוֹתֵר.

חֲבַקּוּק מִתְיַצֵּב בְּעִגּוּל וְאוֹמֵר: ׳לֹא אָזוּז עַד שֶׁאֶשְׁמַע׳.

3
🏃

📜 הַפָּסוּקכְּתוֹב חָזוֹן וּבָאֵר עַל־הַלֻּחוֹת לְמַעַן יָרוּץ קוֹרֵא בוֹ

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁנָּבִין אוֹתוֹ, וְלָכֵן הוּא דִּבֵּר בְּפָשְׁטוּת וּבְבֵרוּר 💛

📖 חֲבַקּוּק, פֶּרֶק ב׳, פָּסוּק ב׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

🏃 מְצוּדַת דָּוִד: ׳יָרוּץ קוֹרֵא בוֹ׳

ה׳ אָמַר לַנָּבִיא לִכְתֹּב ׳לְמַעַן יָרוּץ קוֹרֵא בוֹ׳. מְצוּדַת דָּוִד מְבָאֵר: ׳בָּאֵר אוֹתָהּ עַל הַלּוּחוֹת וּמְפֹרָשׁ וְלֹא בַּחִידוֹת׳, כְּדֵי שֶׁהַקּוֹרֵא ׳יוּכַל לִקְרוֹת בּוֹ בִּמְהִירוּת בְּמִעַט הִתְבּוֹנְנוּת׳. רַשִׁ״י: ׳בָּאֲרֵהוּ הֵיטֵב אֲשֶׁר יָרוּץ הַקּוֹרֵא בּוֹ לִקְרוֹתוֹ מַהֵר שֶׁלֹּא יִכָּשֵׁל׳. דִּבְרֵי ה׳ נִכְתְּבוּ בְּבֵרוּר, כְּדֵי שֶׁכֻּלָּם יָבִינוּ וְיִזְכְּרוּ.

דִּבְרֵי הַנְּבוּאָה נִכְתְּבוּ בְּבֵרוּר – כְּדֵי שֶׁכָּל אֶחָד יָבִין וְיִלְמַד.

4
⏳

📜 הַפָּסוּקאִם־יִתְמַהְמָהּ חַכֵּה־לוֹ כִּי־בֹא יָבֹא לֹא יְאַחֵר

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁהַדָּבָר מִתְמַהְמֵהַּ, אֲנִי לֹא מִתְיָאֵשׁ, כִּי ה׳ אוֹהֵב אוֹתָנוּ וְיָבִיא אֶת הַטּוֹב 💛

📖 חֲבַקּוּק, פֶּרֶק ב׳, פָּסוּק ג׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

⏳ רַשִׁ״י: ׳אִם יִתְמַהְמָהּ חַכֵּה לוֹ׳

רַשִׁ״י מְבָאֵר: ׳אִם יִתְמַהְמָהּ לָבוֹא שֶׁיַּאֲרִיךְ לָכֶם אוֹתוֹ מוֹעֵד – חַכֵּה לוֹ כִּי לֹא יְאַחֵר׳. וּמְצוּדַת דָּוִד: ׳אַף אִם יִתְמַהְמָהּ זְמַן רַב, עִם כָּל זֶה אַתָּה יִשְׂרָאֵל חַכֵּה לוֹ וְלֹא תִתְיָאֵשׁ מִמֶּנּוּ׳. הַמִּלָּה ׳חַכֵּה׳ אֵינָהּ רַק לְחַכּוֹת בְּשֶׁקֶט – הִיא לְקַוּוֹת בְּלֵב מָלֵא. הַגְּאֻלָּה בָּאָה בִּזְמַנָּהּ, וְלָכֵן אֵין לְהִתְיָאֵשׁ.

׳חַכֵּה לוֹ׳ – לֹא מִתְיָאֲשִׁים, כִּי מָה שֶׁה׳ אָמַר בֹּא יָבֹא.

5
⚖️

📜 הַפָּסוּקהִנֵּה עֻפְּלָה לֹא־יָשְׁרָה נַפְשׁוֹ בּוֹ וְצַדִּיק בֶּאֱמוּנָתוֹ יִחְיֶה

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי מַסְפִּיק, וַאֲנִי לֹא צָרִיךְ לָרוּץ אַחֲרֵי עוֹד וָעוֹד 💛

📖 חֲבַקּוּק, פֶּרֶק ב׳, פָּסוּק ד׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

⚖️ שְׁנֵי דְּרָכִים: עֻפְּלָה וֶאֱמוּנָה

הַפָּסוּק מַנִּיחַ זֶה מוּל זֶה שְׁנֵי אֲנָשִׁים. רַשִׁ״י מְבָאֵר ׳עֻפְּלָה׳ – ׳לְשׁוֹן עַזּוּת׳, וְאֶת ׳לֹא יָשְׁרָה נַפְשׁוֹ בּוֹ׳ – ׳אֵין רוּחוֹ מִתְקָרֶרֶת עָלָיו לוֹמַר: דַּי בְּמַה שֶּׁקָּנִיתִי כְּבָר׳. מִי שֶׁגֵּאֶה לְעוֹלָם אֵינוֹ שָׂבֵעַ וְאֵין לוֹ שֶׁקֶט. מוּלוֹ ׳צַדִּיק בֶּאֱמוּנָתוֹ יִחְיֶה׳ – מִי שֶׁסּוֹמֵךְ עַל ה׳ חַי בְּשַׁלְוָה.

הַגֵּאֶה אוֹמֵר ׳עוֹד וְעוֹד׳ וְאֵינוֹ שָׂבֵעַ, וְהַצַּדִּיק חַי בֶּאֱמוּנָה וְשָׂבֵעַ.

💞

📜 הַפָּסוּקה׳ שָׁמַעְתִּי שִׁמְעֲךָ יָרֵאתִי ה׳ פָּעָלְךָ בְּקֶרֶב שָׁנִים חַיֵּיהוּ בְּקֶרֶב שָׁנִים תּוֹדִיעַ בְּרֹגֶז רַחֵם תִּזְכּוֹר

💛 אַהֲבַת הַשֵּׁם: ה׳ שֶׁלִּי מְרַחֵם, גַּם כְּשֶׁקָּשֶׁה לִי לְהָבִין אֶת הַדֶּרֶךְ 💛

📖 חֲבַקּוּק, פֶּרֶק ג׳, פָּסוּק ב׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

💞 מְצוּדַת דָּוִד: ׳בְּרֹגֶז רַחֵם תִּזְכּוֹר׳

בִּתְפִלָּתוֹ חֲבַקּוּק אוֹמֵר: ׳ה׳ שָׁמַעְתִּי שִׁמְעֲךָ יָרֵאתִי… בְּרֹגֶז רַחֵם תִּזְכּוֹר׳. מְצוּדַת דָּוִד מְבָאֵר: ׳שֶׁאַף בִּזְמַן הָרֹגֶז וְהַגָּלוּת תִּזְכּוֹר לְרַחֵם עֲלֵיהֶם׳. הַנָּבִיא מְבַקֵּשׁ: גַּם כְּשֶׁה׳ כּוֹעֵס, שֶׁיִּזְכֹּר אֶת הָרַחֲמִים. כָּךְ הוּא מַכִּיר אֶת ה׳ – כְּמִי שֶׁהָרַחֲמִים שֶׁלּוֹ עוֹמְדִים גַּם בְּתוֹךְ הַדִּין.

גַּם בְּתוֹךְ הַדִּין, חֲבַקּוּק מְבַקֵּשׁ רַחֲמִים – וּמַאֲמִין שֶׁה׳ יִזְכֹּר אוֹתָם.

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מְגַלִּים נָבִיא שֶׁדּוֹאֵג לְעַמּוֹ עוֹד לִפְנֵי הַצָּרָה, עוֹמֵד בְּעִגּוּל וְלֹא זָז עַד שֶׁהוּא שׁוֹמֵעַ, וּמְבַקֵּשׁ ׳בְּרֹגֶז רַחֵם תִּזְכּוֹר׳. כָּךְ נִרְאֵית אַהֲבָה לַה׳ שֶׁאֵינָהּ מוֹתֶרֶת עַל שְׁאֵלוֹת.

וְצַדִּיק בֶּאֱמוּנָתוֹ יִחְיֶה

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • חֲבַקּוּק שׁוֹאֵל ׳עַד אָנָה׳ וְה׳ עוֹנֶה ׳חַכֵּה לוֹ׳. מָתַי הִרְגַּשְׁתִּי שֶׁאֲנִי צָרִיךְ לְחַכּוֹת לְדָבָר טוֹב, וְאֵיךְ עָזַר לִי לְהַאֲמִין?
  • ׳וַאֲנִי בַּה׳ אֶעְלוֹזָה׳ – גַּם כְּשֶׁחָסֵר. עַל מָה אֲנִי יָכוֹל לִשְׂמֹחַ הַיּוֹם, גַּם אִם לֹא הַכֹּל מְסֻדָּר?
  • מַה זֶּה אוֹמֵר לִחְיוֹת ׳בֶּאֱמוּנָתוֹ׳? אֵיזֶה מַעֲשֶׂה קָטָן אֲנִי יָכוֹל לַעֲשׂוֹת הַשָּׁבוּעַ שֶׁמַּרְאֶה שֶׁאֲנִי סוֹמֵךְ עַל ה׳?

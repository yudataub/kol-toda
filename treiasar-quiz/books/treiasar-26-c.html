רָמָה ג׳ – מַעֲמִיקִים

🧪 ׳עַשֵּׂר בִּשְׁבִיל שֶׁתִּתְעַשֵּׁר׳

בַּגְּמָרָא (תַּעֲנִית ט׳ ע״א) נֶאֱמַר: ׳״עַשֵּׂר תְּעַשֵּׂר״ – עַשֵּׂר בִּשְׁבִיל שֶׁתִּתְעַשֵּׁר׳. רַבִּי יוֹחָנָן מָצָא אֶת הַיֶּלֶד שֶׁל רֵישׁ לָקִישׁ וְשָׁאַל אוֹתוֹ: ״מְנָא לָךְ?״ – מִנַּיִן לְךָ? וְהַיֶּלֶד אָמַר: ״זִיל נַסִּי״ – לֵךְ וְנַסֶּה. שָׁאַל רַבִּי יוֹחָנָן: ׳וּמִי שְׁרֵי לְנַסּוֹיֵיהּ לְהַקָּדוֹשׁ בָּרוּךְ הוּא? וְהָכְתִיב: לֹא תְנַסּוּ אֶת ה׳׳, וְהַתְּשׁוּבָה: חוּץ מִזּוֹ, שֶׁנֶּאֱמַר ׳וּבְחָנוּנִי נָא בָּזֹאת׳. וּמַאי ״עַד בְּלִי דָי״? ׳עַד שֶׁיִּבְלוּ שִׂפְתוֹתֵיכֶם מִלּוֹמַר ״דַּי״׳.

בְּמַעֲשֵׂר מוּתָּר לִבְחוֹן אֶת ה׳: ׳עַשֵּׂר בִּשְׁבִיל שֶׁתִּתְעַשֵּׁר׳ – וְהַשֶּׁפַע יִגְמַר רַק כְּשֶׁנֹּאמַר ׳דַּי׳.

📜 הַפָּסוּקהָבִיאוּ אֶת־כָּל־הַמַּעֲשֵׂר

רָמָה ג׳ – מַעֲמִיקִים

🤲 מַעֲשֵׂר עָנִי בִּימֵינוּ

בְּשִׁעוּר נֶאֱמַר: ׳מִכָּל הַתְּרוּמוֹת וְהַמַּעַשְׂרוֹת, אֶת רֻבָּן אָנוּ מַפְרִישִׁים וּמַנִּיחִים בַּצַּד, כִּי אֵין לָנוּ כֹּהֵן טָהוֹר וְלֹא לֵוִי מְיֻחָס׳. ׳אֲבָל מַעֲשֵׂר עָנִי הוּא הַמִּצְוָה שֶׁעוֹד אֶפְשָׁר לְקַיֵּם בִּמְלֹאָהּ: לָתֵת לֶעָנִי אֶת מַה שֶּׁמַּגִּיעַ לוֹ׳. אֶת דְּבַר מַלְאָכִי ׳וְעֹשְׁקֵי שְׂכַר שָׂכִיר… וּמַטֵּי גֵר׳ אֲנַחְנוּ שׁוֹמְעִים גַּם בִּימֵינוּ: מִי שֶׁנּוֹתֵן לֶעָנִי מַה שֶּׁמַּגִּיעַ לוֹ, מְקַיֵּם אֶת הַמִּצְוָה בִּמְלֹאָהּ.

מַעֲשֵׂר עָנִי הוּא הַמִּצְוָה שֶׁאֶפְשָׁר לְקַיֵּם הַיּוֹם בִּמְלֹאָהּ.

2

📜 הַפָּסוּקוְעָרְבָה לַה׳ מִנְחַת יְהוּדָה וִירוּשָׁלִָם כִּימֵי עוֹלָם וּכְשָׁנִים קַדְמֹנִיּוֹת

💛 אַהֲבַת הַשֵּׁם: הַתְּפִלָּה שֶׁלִּי מְבַקֶּשֶׁת לִהְיוֹת נְעִימָה וּמְרֻצָּה לִפְנֵי ה׳ 💛

📖 מַלְאָכִי, פֶּרֶק ג׳, פָּסוּק ד׳

רָמָה ג׳ – מַעֲמִיקִים

🛐 ׳וְעָרְבָה לַה׳׳ – בְּסוֹף הָעֲמִידָה

אֶת הַפָּסוּק ׳וְעָרְבָה לַה׳ מִנְחַת יְהוּדָה וִירוּשָׁלָיִם כִּימֵי עוֹלָם וּכְשָׁנִים קַדְמֹנִיּוֹת׳ אָנוּ אוֹמְרִים בְּסוֹף תְּפִלַּת הָעֲמִידָה, אַחֲרֵי ׳אֱלֹהַי נְצוֹר׳. אָנוּ מְבַקְּשִׁים שֶׁהַתְּפִלָּה שֶׁלָּנוּ תִּהְיֶה נְעִימָה לַה׳ כְּמוֹ הַמִּנְחָה בְּבֵית הַמִּקְדָּשׁ. כָּךְ מַלְאָכִי נִשְׁמָע בְּכָל תְּפִלָּה: אֲנִי אוֹמֵר אֶת דְּבָרָיו בְּסוֹף כָּל עֲמִידָה.

בְּסוֹף כָּל עֲמִידָה אֲנִי מְבַקֵּשׁ: ׳וְעָרְבָה לַה׳׳ – שֶׁהַתְּפִלָּה תִּהְיֶה נְעִימָה.

3

📜 הַפָּסוּקהִנֵּה אָנֹכִי שֹׁלֵחַ לָכֶם אֵת אֵלִיָּה הַנָּבִיא לִפְנֵי בּוֹא יוֹם ה׳ הַגָּדוֹל וְהַנּוֹרָא

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַזְכִּיר אֶת אֵלִיָּהוּ הַנָּבִיא בְּאַהֲבָה, כִּי הוּא מְבַשֵּׂר גְּאֻלָּה 💛

📖 מַלְאָכִי, פֶּרֶק ג׳, פְּסוּקִים כ״ג–כ״ד (מְצוּדַת דָּוִד)

רָמָה ג׳ – מַעֲמִיקִים

🍷 אֵלִיָּהוּ הַנָּבִיא – שַׁבָּת הַגָּדוֹל וְהַבְדָּלָה

אֶת הַפְטָרַת שַׁבָּת הַגָּדוֹל קוֹרְאִים מִמַּלְאָכִי, וְהִיא נִגְמֶרֶת: ׳הִנֵּה אָנֹכִי שֹׁלֵחַ לָכֶם אֵת אֵלִיָּה הַנָּבִיא לִפְנֵי בּוֹא יוֹם ה׳ הַגָּדוֹל וְהַנּוֹרָא וְהֵשִׁיב לֵב אָבוֹת עַל בָּנִים׳. בְּסוֹף הַפְטָרָה חוֹזְרִים עַל פָּסוּק כ״ג, כְּדֵי לְסַיֵּם בְּטוֹב. אֵלִיָּהוּ הַנָּבִיא נִזְכָּר גַּם בְּשִׁיר שֶׁל הַבְדָּלָה בְּמוֹצָאֵי שַׁבָּת, וּבְלֵיל הַסֵּדֶר פּוֹתְחִים לוֹ אֶת הַדֶּלֶת. מְצוּדַת דָּוִד: ׳בִּזְכוּת הַתּוֹרָה אֶשְׁלַח לָכֶם אֶת אֵלִיָּהוּ הַנָּבִיא׳.

מַלְאָכִי מַבְטִיחַ שֶׁאֵלִיָּהוּ יָבוֹא – וְאָנוּ מַזְכִּירִים אוֹתוֹ בְּשַׁבָּת הַגָּדוֹל, בַּהַבְדָּלָה וּבַסֵּדֶר.

4

📜 הַפָּסוּקזִכְרוּ תּוֹרַת מֹשֶׁה עַבְדִּי אֲשֶׁר צִוִּיתִי אוֹתוֹ בְחֹרֵב עַל־כָּל־יִשְׂרָאֵל חֻקִּים וּמִשְׁפָּטִים

💛 אַהֲבַת הַשֵּׁם: אֲנִי זוֹכֵר אֶת הַתּוֹרָה, כִּי בָּהּ הַכֹּל מַתְחִיל וְנִגְמָר 💛

📖 מַלְאָכִי, פֶּרֶק ג׳, פָּסוּק כ״ב (מְצוּדַת דָּוִד)

רָמָה ג׳ – מַעֲמִיקִים

📕 ׳זִכְרוּ תּוֹרַת מֹשֶׁה׳ – סוֹף הַנְּבִיאִים

לִפְנֵי שֶׁהַנָּבִיא הָאַחֲרוֹן מְסַיֵּם, הוּא מַחְזִיר אוֹתָנוּ לַהַתְחָלָה: ׳זִכְרוּ תּוֹרַת מֹשֶׁה עַבְדִּי אֲשֶׁר צִוִּיתִי אוֹתוֹ בְחֹרֵב עַל כָּל יִשְׂרָאֵל חֻקִּים וּמִשְׁפָּטִים׳. מְצוּדַת דָּוִד: ׳בְּכָל דּוֹר וָדוֹר זִכְרוּ הַתּוֹרָה שֶׁמָּסַר לָכֶם מֹשֶׁה עַבְדִּי, כִּי בִּשְׁמִירָתָהּ תְּקַבְּלוּ הַגְּמוּל׳. וְ׳חֻקִּים וּמִשְׁפָּטִים׳ – ׳חֻקִּים׳ הֵם דְּבָרִים שֶׁלֹּא נוֹדַע טַעֲמָם, וּ׳מִשְׁפָּטִים׳ הֵם הַנּוֹדָעִים בְּטַעֲמָם. כָּל הַנְּבִיאִים עוֹמְדִים עַל הַתּוֹרָה.

הַנָּבִיא הָאַחֲרוֹן חוֹזֵר לַהַתְחָלָה: זִכְרוּ אֶת הַתּוֹרָה.

רָמָה ג׳ – מַעֲמִיקִים

📖 מָה מַלְאָכִי מְלַמֵּד אוֹתִי הַיּוֹם

מַלְאָכִי מְלַמֵּד: לִהְיוֹת יָשָׁר עִם הַחַלָּשׁ, לָתֵת מַעֲשֵׂר וּצְדָקָה, לָשׁוּב אֶל ה׳ שֶׁלֹּא מִשְׁתַּנֶּה, וּלְדַבֵּר עִם חֲבֵרִים דְּבָרִים שֶׁמְּחַזְּקִים. הוּא מְלַמֵּד שֶׁה׳ זוֹכֵר כָּל שִׂיחָה טוֹבָה, חוֹמֵל כְּמוֹ אָב, וְשֶׁבַּסּוֹף שֶׁמֶשׁ צְדָקָה זוֹרַחַת. וְהַנָּבִיא הָאַחֲרוֹן מְסַיֵּם בְּחִבּוּר בֵּין אָבוֹת וּבָנִים וּבְתוֹרַת מֹשֶׁה – כְּדֵי שֶׁהַנְּבוּאָה תַּמְשִׁיךְ בְּתוֹכֵנוּ.

לָשׁוּב, לָתֵת, לְדַבֵּר טוֹב – וה׳ זוֹכֵר, חוֹמֵל וּמֵאִיר.

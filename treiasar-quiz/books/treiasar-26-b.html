ה׳ מַאֲרִיךְ אַפּוֹ, אֲבָל אֵינוֹ מִשְׁתַּנֶּה – הוּא אוֹהֵב טוֹב וְשׂוֹנֵא רַע.

רַשִׁ״י מְבָאֵר אֶת ׳הֲיִקְבַּע אָדָם אֱלֹהִים׳: ׳רַבּוֹתֵינוּ דָּרְשׁוּ לְשׁוֹן גְּזֵלָה וְלָשׁוֹן אֲרָמִי הוּא׳. ׳הַמַּעֲשֵׂר וְהַתְּרוּמָה – שֶׁאַתֶּם גּוֹזְלִים מִן הַכֹּהֲנִים וּמִן הַלְוִיִּם הִיא גְּזֵלָתִי׳. מְצוּדַת דָּוִד: ׳עַל זֶה אָשִׁיב שֶׁאַתֶּם עוֹשְׁקִים מִמֶּנִּי הַמַּעֲשֵׂר וְהַתְּרוּמָה כִּי אֵינְכֶם נוֹתְנִים אוֹתָם אֶל הַכֹּהֲנִים וְהַלְוִיִּם מְשָׁרְתַי׳. הַכֹּהֲנִים וְהַלְוִיִּם הֵם מְשָׁרְתֵי ה׳, וְהַמַּעֲשֵׂר הוּא פַּרְנָסָתָם.

מִי שֶׁלֹּא נוֹתֵן מַעֲשֵׂר גּוֹזֵל אֶת מְשָׁרְתֵי ה׳ – וְכָךְ גַּם אֶת ה׳.

רַשִׁ״י מְבָאֵר: ׳אָז נִדְבְּרוּ – אֲנִי מֵשִׁיב עַל דִּבְרֵיכֶם: בַּעֲשׂוֹת הָרְשָׁעִים רִשְׁעָם וְהַטּוֹבִים הָלְכוּ קְדוֹרַנִּית, נִדְבְּרוּ יִרְאֵי ה׳ שֶׁלֹּא לִדָּבֵק בְּמַעֲשֵׂיהֶם הָרָעִים, וַאֲנִי לֹא נִשְׁכְּחוּ דִבְרֵיהֶם מִלְּפָנַי… וְצִוִּיתִי לִכְתֹּב לָהֶם סֵפֶר זִכָּרוֹן׳. מְצוּדַת דָּוִד: ׳נִדְבְּרוּ גַּם הֵם אִישׁ אֶל רֵעֵהוּ… עַד שֶׁמָּצְאוּ בְּדַעְתָּם כִּי כָּל דְּרָכָיו מִשְׁפָּט אֵל אֱמוּנָה׳. אֲנָשִׁים שֶׁשָּׁאֲלוּ, דִּבְּרוּ וְהִתְחַזְּקוּ – וה׳ כָּתַב.

הַשִּׂיחָה בֵּין חֲבֵרִים שֶׁמִּתְחַזְּקִים בַּאֱמוּנָה נִכְתֶּבֶת לִפְנֵי ה׳.

רַשִׁ״י מְבָאֵר אֶת ׳לַיּוֹם אֲשֶׁר אֲנִי עֹשֶׂה סְגֻלָּה׳: ׳אֲשֶׁר אָצַרְתִּי וְהִנַּחְתִּי לְשַׁלֵּם בּוֹ גְּמוּלִי, שָׁם אֶרְאֶה אֶתְכֶם מַה בֵּין צַדִּיק לְרָשָׁע׳. ׳סְגֻלָּה – אוֹצָר׳. וּמְצוּדַת דָּוִד עַל ׳וְחָמַלְתִּי עֲלֵיהֶם כַּאֲשֶׁר יַחְמֹל אִישׁ עַל בְּנוֹ הָעֹבֵד אֹתוֹ׳: ׳כַּאֲשֶׁר יַחְמֹל אִישׁ עַל בְּנוֹ הַנִּזְהָר בַּעֲבוֹדָתוֹ׳. אָב חוֹמֵל עַל בֵּן שֶׁעוֹבֵד אוֹתוֹ – וה׳ חוֹמֵל עַל מִי שֶׁעוֹבֵד אוֹתוֹ. הַצַּדִּיקִים הֵם אוֹצָר יָקָר בְּעֵינָיו.

🔀 שְׁנֵי פֵּרוּשִׁים: ׳וְהֵשִׁיב לֵב אָבוֹת׳

רַשִׁ״י מְבָאֵר אֶת ׳וְהֵשִׁיב לֵב אָבוֹת עַל בָּנִים׳: ׳לְהַקָּדוֹשׁ בָּרוּךְ הוּא, עַל יְדֵי בָּנִים – יֹאמַר לַבָּנִים דֶּרֶךְ אַהֲבָה וְרָצוֹן: לְכוּ וְדַבְּרוּ אֶל אֲבוֹתֵיכֶם לֶאֱחֹז בְּדַרְכֵי הַמָּקוֹם, וְכֵן וְלֵב בָּנִים עַל אֲבוֹתָם. כָּךְ שָׁמַעְתִּי… לָשׂוּם שָׁלוֹם בָּעוֹלָם׳. מְצוּדַת דָּוִד: ׳הוּא יָשִׁיב לֵב הָאָבוֹת לַמָּקוֹם עַל יְדֵי הַבָּנִים׳, ׳כִּי תְּחִיַּת הַמֵּתִים תִּהְיֶה עַל יָדוֹ׳ – וְהַבָּנִים וְהָאָבוֹת יְסַפְּרוּ זֶה לָזֶה. שְׁנֵי פֵּרוּשִׁים: אֶחָד – שָׁלוֹם בְּעוֹלָם בֵּין הַדּוֹרוֹת; הַשֵּׁנִי – עֵדוּת שֶׁל הָאָבוֹת וְהַבָּנִים זֶה לָזֶה.

רַשִׁ״י: שָׁלוֹם בֵּין הַדּוֹרוֹת; מְצוּדַת דָּוִד: הָאָבוֹת וְהַבָּנִים מְסַפְּרִים זֶה לָזֶה וְשָׁבִים לַה׳.

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מְגַלִּים שֶׁה׳ מַאֲרִיךְ אַפּוֹ אֲבָל אֵינוֹ מִשְׁתַּנֶּה, שֶׁהַנְּתִינָה הִיא ׳בְּחִינָה׳ שֶׁלּוֹ, וְשֶׁהַצְּדָקָה הִיא ׳שֶׁמֶשׁ׳ שֶׁמְּאִירָה לַנּוֹתֵן. כָּל זֶה מֵאַהֲבָה.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳שׁוּבוּ אֵלַי וְאָשׁוּבָה אֲלֵיכֶם׳. מָה הַדָּבָר שֶׁאֲנִי רוֹצֶה לַחֲזֹר אֵלָיו, וּמָה אֲנִי יָכוֹל לַעֲשׂוֹת הַשָּׁבוּעַ?
  • ׳וְחָמַלְתִּי עֲלֵיהֶם כַּאֲשֶׁר יַחְמֹל אִישׁ עַל בְּנוֹ הָעֹבֵד אֹתוֹ׳. אֵיךְ אֲנִי מַרְגִּישׁ כְּשֶׁאֲנִי חוֹשֵׁב שֶׁה׳ חוֹמֵל עָלַי כְּמוֹ אָב?
  • ׳וְהֵשִׁיב לֵב אָבוֹת עַל בָּנִים׳. אֵיךְ אֲנִי יָכוֹל לְחַבֵּר אֶת הַלֵּב שֶׁלִּי עִם הַהוֹרִים וְהַסָּבִים שֶׁלִּי?

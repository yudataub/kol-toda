🫏 רַשִׁ״י: מֶלֶךְ הַמָּשִׁיחַ הָעָנָו

רַשִׁ״י מְבָאֵר אֶת ׳הִנֵּה מַלְכֵּךְ יָבוֹא לָךְ׳: ׳אִי אֶפְשָׁר לְפָתְרוֹ אֶלָּא עַל מֶלֶךְ הַמָּשִׁיחַ, שֶׁנֶּאֱמַר בּוֹ ׳וּמָשְׁלוֹ מִיָּם עַד יָם׳ וְלֹא מָצִינוּ מוֹשֵׁל לְיִשְׂרָאֵל כָּזֶה בִּימֵי בַּיִת שֵׁנִי׳. ׳עָנִי – עַנְוְתָן׳, ׳וְרוֹכֵב עַל חֲמוֹר – מִדַּת עֲנָוָה הִיא׳. מְצוּדַת דָּוִד: ׳צַדִּיק בְּמַעֲשָׂיו וּבְמִלְחֲמוֹתָיו יִהְיֶה נוֹשָׁע וּמַצְלִיחַ׳. מִי שֶׁגָּדוֹל בֶּאֱמֶת – עָנָו.

הַמֶּלֶךְ הַגָּדוֹל בְּיוֹתֵר נִקְרָא ׳עָנִי׳ – כִּי הַגְּדֻלָּה הָאֲמִתִּית הִיא עֲנָוָה.

רַשִׁ״י מְבָאֵר ׳וְהִכְרַתִּי רֶכֶב׳: ׳שֶׁלֹּא יִהְיוּ צְרִיכִין לָהּ׳. וּמְצוּדַת דָּוִד עַל ׳וְדִבֶּר שָׁלוֹם לַגּוֹיִם׳: ׳הַמֶּלֶךְ הַמָּשִׁיחַ לֹא יַכְנִיעַ אֶת כֻּלָּם עַל יְדֵי מִלְחָמָה כִּי אִם בְּדִבְרֵי שָׁלוֹם לִהְיוֹת סָרִים לְמִשְׁמַעְתּוֹ, וְזֶה יִהְיֶה הָעִקָּר וְהַמִּלְחָמָה טְפֵלָה לוֹ׳. כְּלוֹמַר: הֶעָרֶךְ הָרִאשׁוֹן הוּא שָׁלוֹם וְדִבּוּר, וְלֹא נֶשֶׁק.

הָעִקָּר בְּיוֹם הַמָּשִׁיחַ הוּא שָׁלוֹם וְדִבּוּר – וְהַמִּלְחָמָה טְפֵלָה.

הַתִּקְוָה קוֹשֶׁרֶת אֶת הַלֵּב לַה׳ כְּמוֹ חֶבֶל שֶׁלֹּא נִפְרָד.

ה׳ שׁוֹלֵחַ גֶּשֶׁם אֲפִלּוּ בִּשְׁבִיל אָדָם אֶחָד וְעֵשֶׂב אֶחָד.

רַשִׁ״י מְבָאֵר אֶת ׳אֶשְׁרְקָה לָהֶם׳: ׳כְּדֶרֶךְ הַשּׁוֹרְקִים לְסִימָן, אוֹת לָבוֹא אֵלָיו הַטּוֹעֶה בַּדֶּרֶךְ׳. ׳וַאֲקַבְּצֵם – לְעֵת הַקֵּץ לֶעָתִיד׳. ׳כִּי פְדִיתִים – כְּשֶׁאֶפְדֵּם׳. ׳וְרָבוּ – בַּגּוֹלָה כְּמוֹ שֶׁרָבוּ בְּמִצְרַיִם׳. מְצוּדַת דָּוִד: ׳אָעִיר לִבָּם לָשׁוּב לְאַרְצָם׳. הַשְּׁרִיקָה הִיא סִימָן לַטּוֹעֶה שֶׁיֵּשׁ מִי שֶׁקּוֹרֵא לוֹ.

ה׳ שׁוֹרֵק לַטּוֹעֶה בַּדֶּרֶךְ, וּמְעוֹרֵר אֶת הַלֵּב לָשׁוּב.

ה׳ בּוֹחֵר בָּנוּ וְאוֹמֵר ׳עַמִּי הוּא׳, וְאָנוּ בּוֹחֲרִים בּוֹ וְאוֹמְרִים ׳ה׳ אֱלֹהָי׳.

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מְגַלִּים שֶׁגְּדֻלָּה אֲמִתִּית הִיא עֲנָוָה, שֶׁהַתִּקְוָה קוֹשֶׁרֶת אוֹתָנוּ לַה׳, וְשֶׁה׳ שׁוֹרֵק לַטּוֹעֶה בַּדֶּרֶךְ וְשׁוֹלֵחַ גֶּשֶׁם גַּם לְאָדָם אֶחָד.

וְהָיָה ה׳ לְמֶלֶךְ עַל־כָּל־הָאָרֶץ בַּיּוֹם הַהוּא יִהְיֶה ה׳ אֶחָד וּשְׁמוֹ אֶחָד

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • הַמֶּלֶךְ הָעָתִיד הוּא ׳עָנִי וְרֹכֵב עַל חֲמוֹר׳. מָה אֲנִי לוֹמֵד מִכָּךְ עַל גְּדֻלָּה אֲמִתִּית?
  • ׳אֲסִירֵי הַתִּקְוָה׳. עַל אֵיזוֹ תִּקְוָה אֲנִי לֹא מוֹתֵר, וְאֵיךְ אֲנִי שׁוֹמֵר עָלֶיהָ?
  • ׳וְהָיָה לְעֵת עֶרֶב יִהְיֶה אוֹר׳. מָתַי הִרְגַּשְׁתִּי שֶׁדַּוְקָא בְּסוֹף הַיּוֹם בָּא אוֹר?

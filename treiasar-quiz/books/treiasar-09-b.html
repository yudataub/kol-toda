׳הִנֵּה קָטֹן נְתַתִּיךָ בַּגּוֹיִם בָּזוּי אַתָּה מְאֹד׳ – רַשִׁ״י: ׳כְּלַפֵּי שֶׁקְּרָאוֹ אָבִיו בְּנוֹ הַגָּדוֹל וְאִמּוֹ בְּנָהּ הַגָּדוֹל, אָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא: לְפָנַי קָטֹן הוּא׳. וְ׳רַבּוֹתֵינוּ דָּרְשׁוּ: קָטֹן – שֶׁלֹּא הָיָה לָהֶם לֹא כְּתָב וְלֹא לָשׁוֹן׳, ו׳בָּזוּי – שֶׁלֹּא הָיוּ מַעֲמִידִין מֶלֶךְ בֶּן מֶלֶךְ׳. מְצוּדַת דָּוִד: ׳בִּתְחִלַּת מֶמְשַׁלְתְּךָ בְּאֶרֶץ שֵׂעִיר הָיִיתָ קָטָן בָּעַכּוּ״ם וּמְבֻזֶּה׳. מִי שֶׁהָיָה קָטָן וְהִתְגַּדֵּל לֹא אָמוּר לִשְׁכֹּחַ מֵאַיִן בָּא.

׳וְעָלוּ מוֹשִׁעִים בְּהַר צִיּוֹן לִשְׁפֹּט אֶת הַר עֵשָׂו׳ – רַשִׁ״י: ׳שָׂרֵי יִשְׂרָאֵל מוֹשִׁיעִים בְּהַר צִיּוֹן… לִפָּרַע מֵהַר עֵשָׂו אֲשֶׁר עָשׂוּ לְיִשְׂרָאֵל׳. מְצוּדַת דָּוִד: ׳הַמּוֹשִׁיעִים הֵם מֶלֶךְ הַמָּשִׁיחַ וְשָׂרָיו׳. ׳וְהָיְתָה לַה׳ הַמְּלוּכָה׳ – רַשִׁ״י: ׳לִמֶּדְךָ שֶׁאֵין מַלְכוּתוֹ שְׁלֵמָה עַד שֶׁיִּפָּרַע מֵעֲמָלֵק׳. הַמַּלְכוּת הָאֲמִתִּית שֶׁל ה׳ תִּתְגַּלֶּה כְּשֶׁהָרַע יִגָּמֵר וְכָל הָעוֹלָם יַכִּיר בּוֹ.

הַמַּלְכוּת הַשְּׁלֵמָה שֶׁל ה׳ תִּתְגַּלֶּה כְּשֶׁהָרַע יִגָּמֵר – אָז ׳תִּהְיֶה לַה׳ הַמְּלוּכָה׳.

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מַסְבִּירִים אֶת הַחֵטְא בִּשְׁלֹשָׁה צְעָדִים: לֹא לַעֲזֹר, לֹא לְהִסְתַּכֵּל בִּבְשֶׁת הָאָח, לֹא לִשְׂמֹחַ לְאֵידוֹ. מִי שֶׁלֹּא עוֹזֵר – ה׳ מְחַשֵּׁב אוֹתוֹ כְּמִי שֶׁפָּגַע.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳בְּיוֹם עֲמָדְךָ מִנֶּגֶד׳ – עֹבַדְיָה מוֹכִיחַ מִי שֶׁעָמַד מִן הַצַּד כְּשֶׁאָחִיו בְּצָרָה. מָתַי יָכֹלְתִּי לַעֲזֹר לְחָבֵר וְלֹא עָזַרְתִּי, וּמָה אֲנִי רוֹצֶה לַעֲשׂוֹת הַפַּעַם?
  • ׳וְאַל תִּשְׂמַח לִבְנֵי יְהוּדָה בְּיוֹם אָבְדָם׳ – אֵיךְ אֲנִי מַרְגִּישׁ כְּשֶׁלְּמִישֶׁהוּ שֶׁלֹּא אוֹהֵב אוֹתִי קוֹרֶה מַשֶּׁהוּ רַע? מָה ה׳ מְבַקֵּשׁ מִמֶּנִּי?
  • ׳וְהָיְתָה לַה׳ הַמְּלוּכָה׳ – ה׳ הוּא הַמֶּלֶךְ שֶׁל כֻּלָּנוּ. מָה אֲנִי עוֹשֶׂה בַּבֹּקֶר כְּדֵי לְקַבֵּל עָלַי אֶת מַלְכוּתוֹ?

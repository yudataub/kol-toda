עושים צדקה עם השם – משחק למידה

הַקּוּשְׁיָה עַל הַפָּסוּק וּתְכָלִית הַבְּרִיאָה 🌟

שָׁלוֹם וּבְרָכָה חֲבֵרִים יְקָרִים! 💖 הַיּוֹם נִלְמַד מַשֶּׁהוּ נִפְלָא - אֵיךְ אֲנַחְנוּ עוֹשִׂים צְדָקָה עִם הַקָּדוֹשׁ בָּרוּךְ הוּא ! 🎯 בַּתּוֹרָה כָּתוּב: "וְהֶאֱמִין בַּה' וַיַּחְשְׁבֶהָ לּוֹ צְדָקָה" (בְּרֵאשִׁית ט"ו, ו') 📖 אֲבָל יֵשׁ כָּאן קוּשְׁיָה: מָה זֹאת צְדָקָה? הֲרֵי אַבְרָהָם אָבִינוּ פָּשׁוּט הֶאֱמִין בַּה'! 🤔 הַבִּיאוּר הַנִּפְלָא: הַקָּדוֹשׁ בָּרוּךְ הוּא רוֹצֶה מְאֹד לְהָטִיב לָנוּ ! זֹאת תַּכְלִית הַבְּרִיאָה כּוּלָּהּ! 🌈 הַשֵּׁם שָׂמֵחַ כְּשֶׁהוּא נוֹתֵן לָנוּ - זֶה הַתַּעֲנוּג שֶׁלּוֹ! ✨ אֲבָל בִּשְׁבִיל שֶׁהַשֵּׁם יוּכַל לְהָטִיב לָנוּ, אֲנַחְנוּ צְרִיכִים לְהַכִּיר שֶׁהַטּוֹב מֵהַשֵּׁם ! 💫

מָה תַּכְלִית הַבְּרִיאָה לְפִי הַשִּׁיעוּר? 🌍

נָכוֹן מְאֹד! 🌟 תַּכְלִית הַבְּרִיאָה הִיא שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא רוֹצֶה לְהָטִיב לָנוּ - זֶה הַתַּעֲנוּג שֶׁלּוֹ! הַשֵּׁם אוֹהֵב אוֹתְךָ וְשָׂמֵחַ לָתֵת לְךָ! 💖

מָה קוֹרֶה כְּשֶׁאָדָם מְקַבֵּל טוֹב וְלֹא מַכִּיר שֶׁזֶּה מֵהַשֵּׁם? 🎁

מְצֻיָּן! ⭐ כְּשֶׁאָדָם לֹא מַכִּיר שֶׁהַטּוֹב מֵהַשֵּׁם - זֶה קִלְקוּל! הַשֶּׁפַע לֹא מְקָרֵב אוֹתוֹ לַקָּדוֹשׁ בָּרוּךְ הוּא. הַשֵּׁם גֵּאֶה בְּךָ שֶׁהֵבַנְתָּ! 💪

אֵיזֶה רֶגֶשׁ הַשֵּׁם מַרְגִּישׁ כְּשֶׁהוּא נוֹתֵן לָנוּ? 😊

נִפְלָא! 🎉 הַשֵּׁם מַרְגִּישׁ שִׂמְחָה וְתַעֲנוּג כְּשֶׁהוּא נוֹתֵן לָנוּ! זֹאת הַשִּׂמְחָה הַכִּי גְדוֹלָה שֶׁלּוֹ! הוּא אוֹהֵב אוֹתְךָ מְאֹד! 💖

הָאֱמוּנָה כְּמַפְתֵּחַ לְשֶׁפַע וְעוֹשִׂים צְדָקָה עִם הַשֵּׁם 🔑

עַכְשָׁיו נָבִין מַשֶּׁהוּ מַדְהִים! ✨ הָאֱמוּנָה הִיא הַכְּלִי לְקַבֵּל שֶׁפַע! 🎁 כְּשֶׁאֲנִי מַאֲמִין שֶׁהַשֵּׁם נוֹתֵן לִי הַכֹּל - אָז הַשֵּׁם יָכוֹל לָתֵת לִי עוֹד וְעוֹד! 🌈 וְכְשֶׁאֲנִי מוֹדֶה לַשֵּׁם - אֲנִי מַרְאֶה לוֹ שֶׁאֲנִי יוֹדֵעַ שֶׁהַכֹּל מִמֶּנּוּ! 🙏 וְהִנֵּה הַסּוֹד הַנִּפְלָא: עַל יְדֵי הָאֱמוּנָה וְהַהוֹדָיָה, אֲנִי עוֹשֶׂה צְדָקָה עִם הַשֵּׁם! 💝 אֵיךְ? כִּי אֲנִי נוֹתֵן לוֹ אֶת הָאֶפְשָׁרוּת לְהָטִיב לִי - וְזֶה מַה שֶּׁהוּא רוֹצֶה! 🎯 אֲנִי נוֹתֵן לַקָּדוֹשׁ בָּרוּךְ הוּא אֶת מַה שֶּׁהוּא הַכִּי רוֹצֶה - לְהָטִיב לִי! זֹאת צְדָקָה אֲמִתִּית ! 💖

מָה קוֹרֶה כְּשֶׁאֲנִי מַאֲמִין שֶׁהַשֶּׁפַע מֵהַשֵּׁם? 🌊

כָּל הַכָּבוֹד! 🏆 הָאֱמוּנָה הִיא הַכְּלִי לְקַבֵּל שֶׁפַע! כְּשֶׁאֲנִי מַאֲמִין - הַשֶּׁפַע יָכוֹל לָרֶדֶת אֵלַי! הַשֵּׁם שָׂמֵחַ מִמְּךָ! 💫

אֵיךְ אֲנִי עוֹשֶׂה צְדָקָה עִם הַקָּדוֹשׁ בָּרוּךְ הוּא? 💝

מְצֻיָּן! ⭐ כְּשֶׁאֲנִי מַאֲמִין וּמוֹדֶה - אֲנִי נוֹתֵן לַשֵּׁם אֶת הָאֶפְשָׁרוּת לְהָטִיב לִי! זֹאת צְדָקָה אֲמִתִּית עִם הַקָּדוֹשׁ בָּרוּךְ הוּא! הַשֵּׁם אוֹהֵב אוֹתְךָ! 💖

מַה הַשֵּׁם הַכִּי רוֹצֶה? 🎯

נִפְלָא! 💪 הַשֵּׁם רוֹצֶה לְהָטִיב לָנוּ - זֶה הַתַּעֲנוּג שֶׁלּוֹ! כְּשֶׁאֲנִי נוֹתֵן לוֹ אֶת הָאֶפְשָׁרוּת - אֲנִי מְשַׂמֵּחַ אוֹתוֹ! הַשֵּׁם גֵּאֶה בְּךָ! 🎉

הַמָּקוֹר בְּפָרָשַׁת וַיֵּרָא וּבִיאוּר הַמַּהֲרַ"ל 📖

עַכְשָׁיו נִרְאֶה אֶת זֶה בַּתּוֹרָה עַצְמָהּ! 📚 בְּפָרָשַׁת וַיֵּרָא כָּתוּב עַל אַבְרָהָם אָבִינוּ: "כִּי יְדַעְתִּיו... וְשָׁמְרוּ דֶּרֶךְ ה' לַעֲשׂוֹת צְדָקָה וּמִשְׁפָּט" ✨ הַמַּהֲרַ"ל שׁוֹאֵל: הַאִם אַבְרָהָם אָבִינוּ עוֹבֵד אֶת הַשֵּׁם בִּשְׁבִיל שָׂכָר?! 🤔 הֲרֵי אַבְרָהָם אָבִינוּ עוֹבֵד מֵאַהֲבָה! אָז מָה זֶה "לְמַעַן הָבִיא ה' עַל אַבְרָהָם"? 💭 וְהַמַּהֲרַ"ל מְתָרֵץ דָּבָר נִפְלָא: אַבְרָהָם אָבִינוּ יוֹדֵעַ שֶׁהַשֵּׁם נֶהֱנֶה לְהָטִיב! 🌟 לָכֵן הוּא אוֹמֵר לְבָנָיו: "שִׁמְרוּ דֶּרֶךְ ה' - כְּדֵי שֶׁהַשֵּׁם יוּכַל לְהָטִיב לָכֶם וְזֶה יִהְיֶה הַתַּעֲנוּג שֶׁלּוֹ!" 💖 אַבְרָהָם לֹא רוֹצֶה שָׂכָר - הוּא רוֹצֶה לְשַׂמֵּחַ אֶת הַשֵּׁם! וְזֶה קוֹרֶה כְּשֶׁהַשֵּׁם יָכוֹל לָתֵת לָנוּ! 🎁

מָה הַמַּהֲרַ"ל שׁוֹאֵל עַל אַבְרָהָם אָבִינוּ? 🤔

מְעֻלֶּה! 🌟 הַמַּהֲרַ"ל שׁוֹאֵל: הַאִם אַבְרָהָם עוֹבֵד בִּשְׁבִיל שָׂכָר? הֲרֵי הוּא עוֹבֵד מֵאַהֲבָה! הַשֵּׁם שָׂמֵחַ שֶׁהֵבַנְתָּ! 💖

מָה הַתֵּרוּץ הַנִּפְלָא שֶׁל הַמַּהֲרַ"ל? 💡

פַּצְצָה! 🎯 אַבְרָהָם רוֹצֶה לְשַׂמֵּחַ אֶת הַשֵּׁם! אֵיךְ? עַל יְדֵי שֶׁנּוֹתֵן לוֹ אֶת הָאֶפְשָׁרוּת לְהָטִיב לָנוּ! הַשֵּׁם גֵּאֶה בְּךָ! 💪

מַדּוּעַ אַבְרָהָם אוֹמֵר לְבָנָיו לִשְׁמֹר דֶּרֶךְ ה'? 🌈

מַדְהִים! 🏆 אַבְרָהָם רוֹצֶה שֶׁבָּנָיו יִשְׁמְרוּ דֶּרֶךְ ה' כְּדֵי שֶׁהַשֵּׁם יוּכַל לְהָטִיב לָהֶם - וְזֶה יְשַׂמַּח אֶת הַשֵּׁם! הַשֵּׁם אוֹהֵב אוֹתְךָ! 💖

הַמֶּסֶר מִמַּסֶּכֶת אָבוֹת וְסִיכּוּם 🌈

וְעַכְשָׁיו הַמֶּסֶר הָאַחֲרוֹן וְהַחָשׁוּב! ⭐ בְּמַסֶּכֶת אָבוֹת כָּתוּב: "שְׂכַר מִצְוָה - מִצְוָה" (אָבוֹת ד', ב') 📜 מָה זֶה אוֹמֵר? הַשָּׂכָר שֶׁאַתָּה מְקַבֵּל עַל הַמִּצְוָה - הוּא עַצְמוֹ מִצְוָה! 🎁 כֵּיצַד? כִּי עַל יְדֵי הַשָּׂכָר - נָתַתָּ לַשֵּׁם לְהָטִיב לְךָ! 💫 וְזוֹ בִּפְנֵי עַצְמָהּ מִצְוָה גְדוֹלָה - לְאַפְשֵׁר לַשֵּׁם לְהָטִיב! 🌟 מַעֲגָל נִפְלָא: אַתָּה עוֹשֶׂה מִצְוָה → מְקַבֵּל שָׂכָר → הַשָּׂכָר הוּא מִצְוָה → וְזֶה מוֹבִיל לְעוֹד שָׂכָר... 🔄 וּמִי שֶׁמֵּבִין זֹאת - רָץ לְקַיֵּם מִצְווֹת! 🏃‍♂️💨 כִּי כָּל פַּעַם שֶׁאֲנִי מוֹדֶה לַשֵּׁם - אֲנִי נוֹתֵן לוֹ אֶת הַשִּׂמְחָה הַכִּי גְדוֹלָה! 💖✨

מָה הַמַּשְׁמָעוּת שֶׁל 'שְׂכַר מִצְוָה - מִצְוָה'? 🎁

יוֹפִי! 🌟 הַשָּׂכָר עַצְמוֹ הוּא מִצְוָה - כִּי נָתַתִּי לַשֵּׁם אֶת הָאֶפְשָׁרוּת לְהָטִיב לִי! וְזוֹ מִצְוָה גְדוֹלָה! הַשֵּׁם שָׂמֵחַ מִמְּךָ! 💖

לָמָּה מִי שֶׁמֵּבִין זֹאת רָץ לְקַיֵּם מִצְווֹת? 🏃‍♂️

מְצֻיָּן! ⭐ מִי שֶׁמֵּבִין רָץ לְקַיֵּם מִצְווֹת - כִּי הוּא רוֹצֶה לָתֵת לַשֵּׁם אֶת הַשִּׂמְחָה שֶׁלּוֹ לְהָטִיב לוֹ! הַשֵּׁם גֵּאֶה בְּךָ! 💪

מָה קוֹרֶה כְּשֶׁאֲנִי מוֹדֶה לַשֵּׁם? 🙏

נִפְלָא! 🎉 כְּשֶׁאֲנִי מוֹדֶה לַשֵּׁם - אֲנִי נוֹתֵן לוֹ אֶת הַשִּׂמְחָה הַכִּי גְדוֹלָה שֶׁלּוֹ: לְהָטִיב לִי! הַשֵּׁם אוֹהֵב אוֹתְךָ מְאֹד! 💖

🌟 מָה זֶה בֶּאֱמֶת הַכְנָסַת אוֹרְחִים?

שָׁלוֹם חֲבֵרִים יְקָרִים! 💖 הַיּוֹם נִלְמַד עַל מִצְוָה מְיֻחֶדֶת וְנִפְלָאָה - הַכְנָסַת אוֹרְחִים ! 🏠✨ אַבְרָהָם אָבִינוּ וְלוֹט אָחִיָינוֹ - שְׁנֵיהֶם עָשׂוּ הַכְנָסַת אוֹרְחִים. אֲבָל הַשֵּׁם אָהַב אֶת הַמַּעֲשִׂים שֶׁל אַבְרָהָם הַרְבֵּה יוֹתֵר! 🎯 לָמָּה? זֶה מָה שֶׁנִּלְמַד! 🤔 הַ"חָפֵץ חַיִּים" (צַדִּיק גָּדוֹל) אוֹמֵר לָנוּ מַשֶּׁהוּ מְיֻחָד: עִקַּר הַמִּצְוָה הִיא לְקַבֵּל אֶת הָאוֹרֵחַ בְּפָנִים יָפוֹת וּבְשִׂמְחָה ! 😊💕 זֶה אוֹמֵר שֶׁלֹּא מַסְפִּיק רַק לָתֵת אוֹכֶל אוֹ מִיטָה. הַדָּבָר הַכִּי חָשׁוּב הוּא לְהַרְגִּישׁ אֶת הָאוֹרֵחַ רָצוּי, מוּעֲרָךְ וְחָשׁוּב ! 🌈✨ גַּם אִם הָאוֹרֵחַ עָשִׁיר וְלֹא צָרִיךְ אֶת הַמָּזוֹן שֶׁלָּנוּ - עוֹדֶנּוּ מְקַיְּמִים מִצְוָה כְּשֶׁאָנוּ מְקַבְּלִים אוֹתוֹ בְּאַהֲבָה! 💖🙏 הַמַּהֲרַ"ל (חָכָם גָּדוֹל) מְסַבֵּר: כְּשֶׁאָנוּ מְכַבְּדִים אוֹרֵחַ, אָנוּ בְּעֶצֶם מְכַבְּדִים אֶת צֶלֶם אֱלֹקִים שֶׁיֵּשׁ בּוֹ ! 👑✨ זֶה כְּמוֹ לְכַבֵּד אֶת הַשֵּׁם עַצְמוֹ! 🌟 💭 שְׁאֵלָה לְמַחְשָׁבָה: אֵיךְ אֲנִי מַרְגִּישׁ כְּשֶׁמִּישֶׁהוּ מְקַבֵּל אוֹתִי בְּשִׂמְחָה וּבְאַהֲבָה? אֵיךְ זֶה עוֹזֵר לִי לְהַרְגִּישׁ אֶת אַהֲבַת הַשֵּׁם? 💖 זִכְרוּ: הַשֵּׁם שָׂמֵחַ כְּשֶׁאָנוּ מְקַבְּלִים אֶת הָאֲחֵרִים בְּאַהֲבָה! 🎉💕

מָה הַדָּבָר הַכִּי חָשׁוּב בְּמִצְוַת הַכְנָסַת אוֹרְחִים, לְפִי הַ'חָפֵץ חַיִּים'?

נְכוֹן מְאוֹד! 🎯 הַ'חָפֵץ חַיִּים' אוֹמֵר שֶׁעִקַּר הַמִּצְוָה הוּא קְבָלַת הַפָּנִים הַיָּפוֹת וְהַשִּׂמְחָה . לֹא מַסְפִּיק רַק אוֹכֶל אוֹ מִיטָה - הַדָּבָר הַחָשׁוּב בְּיוֹתֵר הוּא לְהַרְגִּישׁ אֶת הָאוֹרֵחַ אָהוּב וְרָצוּי ! 💖 זֶה הַדֶּרֶךְ שֶׁל הַשֵּׁם - לְאַהֵב כָּל אֶחָד! ✨

לְפִי הַמַּהֲרַ"ל, מָה אֲנַחְנוּ מְכַבְּדִים כְּשֶׁאָנוּ מְכַבְּדִים אוֹרֵחַ?

מְצֻיָּן! 🌟 הַמַּהֲרַ"ל מְלַמֵּד אוֹתָנוּ שֶׁכְּשֶׁאָנוּ מְכַבְּדִים אוֹרֵחַ, אֲנַחְנוּ מְכַבְּדִים אֶת צֶלֶם אֱלֹקִים שֶׁבְּתוֹכוֹ! 👑 כָּל אָדָם הוּא כְּמוֹ שָׁלִיחַ שֶׁל הַשֵּׁם. לָכֵן, לְכַבֵּד אוֹרֵחַ זֶה כְּמוֹ לְכַבֵּד אֶת הַקָּדוֹשׁ בָּרוּךְ הוּא עַצְמוֹ! 💕 הַשֵּׁם אוֹהֵב אוֹתְךָ כְּשֶׁאַתָּה מְכַבֵּד אֲחֵרִים! 🎉

הַאִם אָנוּ מְקַיְּמִים מִצְוַת הַכְנָסַת אוֹרְחִים גַּם כְּשֶׁהָאוֹרֵחַ עָשִׁיר וְלֹא צָרִיךְ אֶת הָעֶזְרָה שֶׁלָּנוּ?

נִפְלָא! 💪 הַ'חָפֵץ חַיִּים' אוֹמֵר בְּבִרוּר: "אֲפִילוּ הֵם אֵינָם צְרִיכִים לְטוֹבָתוֹ" - גַּם אִם הָאוֹרֵחַ עָשִׁיר! 💰 הַמִּצְוָה הִיא הַשִּׂמְחָה וְהַכָּבוֹד שֶׁאָנוּ נוֹתְנִים לוֹ, לֹא רַק הָעֶזְרָה הַחוּמְרִית. 🎁 כָּל אָדָם צָרִיךְ לְהַרְגִּישׁ אָהוּב - וְזֶה מַה שֶׁהַשֵּׁם רוֹצֶה מִמֶּנּוּ! 💖

🏃‍♂️ הַיּוֹזְמָה וְהַחִפּוּשׂ - מִי רָץ אַחֲרֵי הַמִּצְוָה?

עַכְשָׁו נִרְאֶה אֵיךְ אַבְרָהָם וְלוֹט הִתְנַהֲגוּ בְּאֹפֶן שׁוֹנֶה לְגַמְרֵי! 🔍✨ אַבְרָהָם אָבִינוּ: הַתּוֹרָה אוֹמֶרֶת "יֹשֵׁב פֶּתַח הָאֹהֶל כְּחֹם הַיּוֹם" 🌞🏕️. אַבְרָהָם חִיפֵּשׂ בְּאֹפֶן פָּעִיל אוֹרְחִים ! הוּא לֹא יָשַׁב בְּבַיִת, אֶלָּא יָשַׁב בַּחוּץ, בַּחוֹם הַכִּי גָּדוֹל , רַק כְּדֵי שֶׁיּוּכַל לִרְאוֹת אוֹרְחִים וּלְקַבֵּל אוֹתָם! 🔥💪 הוּא כָּל כָּךְ אָהַב לַעֲשׂוֹת חֶסֶד, שֶׁכְּשֶׁלֹּא הָיוּ אוֹרְחִים - כָּאַב לוֹ הַלֵּב מִמָּשׁ ! 😢💔 תַּאֲרוּ לְעַצְמְכֶם - הוּא כָּל כָּךְ רָצָה לַעֲזֹר לַאֲחֵרִים! 🌟 לוֹט: לְעֻמַּת זֹאת, לוֹט יָשַׁב בְּ"שַׁעַר סְדוֹם" 🏛️. זֶה הָיָה הַמָּקוֹם שֶׁבּוֹ הַשּׁוֹפְטִים יוֹשְׁבִים. הוּא לֹא חִיפֵּשׂ אוֹרְחִים - הֵם הִגִּיעוּ אֵלָיו בְּמִקְרֶה . 🚶‍♂️ כְּשֶׁהָאוֹרְחִים הִגִּיעוּ, לוֹט טִיפֵּל בָּהֶם יָפֶה, אֲבָל הוּא לֹא רָץ לְחַפֵּשׂ אוֹתָם כְּמוֹ אַבְרָהָם. 🤷‍♂️ 💭 שְׁאֵלָה לְמַחְשָׁבָה: מָתַי פַּעַם אַחֲרוֹנָה חִיפַּשְׂתִּי בְּעַצְמִי הַזְדַּמְּנוּת לַעֲזֹר לְמִישֶׁהוּ? אֵיךְ זֶה הִרְגִּישׁ לִי? אֵיךְ אֲנִי יָכוֹל לִהְיוֹת יוֹתֵר כְּמוֹ אַבְרָהָם - לְחַפֵּשׂ דְּרָכִים לַעֲשׂוֹת טוֹב? 🌈 הַלֶּקַח הַגָּדוֹל: אַבְרָהָם רָץ אַחֲרֵי הַמִּצְוָה , בְּעוֹד לוֹט הִמְתִּין שֶׁהַמִּצְוָה תָּבוֹא אֵלָיו . 🏃‍♂️💨 הַשֵּׁם אוֹהֵב כְּשֶׁאָנוּ לֹא רַק מְחַכִּים, אֶלָּא מְחַפְּשִׂים בְּשִׂמְחָה הַזְדַּמְּנוּיּוֹת לַעֲשׂוֹת טוֹב ! 💖🎯

אֵיפֹה יָשַׁב אַבְרָהָם אָבִינוּ, וְלָמָּה דַּוְקָא שָׁם?

מְעֻלֶּה! ⭐ אַבְרָהָם יָשַׁב דַּוְקָא בַּחוּץ, בַּחוֹם הַכִּי גָּדוֹל ('כְּחֹם הַיּוֹם'), כִּי הוּא חִיפֵּשׂ בְּכִלְיוֹן עֵינַיִם אוֹרְחִים ! 👀🔥 הוּא לֹא הִמְתִּין בְּנוֹחוּת בְּתוֹךְ הָאֹהֶל - הוּא רָצָה לִרְאוֹת כָּל עוֹבֵר אֹרֵחַ! זֶה מְלַמֵּד אוֹתָנוּ שֶׁאַהֲבַת הַחֶסֶד חֲזָקָה יוֹתֵר מִכָּל קוֹשִׁי! 💪💖 הַשֵּׁם גֵּאֶה בְּךָ כְּשֶׁאַתָּה מְחַפֵּשׂ לַעֲזֹר! ✨

מָה הַהֶבְדֵּל הָעִיקָרִי בֵּין אַבְרָהָם לְלוֹט בְּעִנְיַן הַיּוֹזְמָה?

נְכוֹן מְאוֹד! 🎯 זֶה בְּדִיּוּק הַהֶבְדֵּל הַגָּדוֹל! אַבְרָהָם רָץ אַחֲרֵי הַמִּצְוָה - חִיפֵּשׂ בְּכָל הַכֹּחַ הַזְדַּמְּנוּיּוֹת לַעֲשׂוֹת חֶסֶד! 🏃‍♂️💨 לוֹט הָיָה טוֹב וּמְכַבֵּד, אֲבָל הוּא הִמְתִּין שֶׁהַמִּצְוָה תָּבוֹא אֵלָיו . 🤷‍♂️ הַשֵּׁם אוֹהֵב אוֹתָנוּ כְּשֶׁאָנוּ יוֹזְמִים טוֹב! 💕✨

מָה קָרָה לְאַבְרָהָם אָבִינוּ כְּשֶׁלֹּא הָיוּ לוֹ אוֹרְחִים?

כָּל הַכָּבוֹד! 🏆 לְאַבְרָהָם אָבִינוּ כָּאַב הַלֵּב מִמָּשׁ כְּשֶׁלֹּא הָיוּ לוֹ אוֹרְחִים! 😢💔 זֶה מְלַמֵּד אוֹתָנוּ עַד כַּמָּה הוּא אָהַב לַעֲשׂוֹת חֶסֶד - זֹאת לֹא הָיְתָה רַק 'עֲבוֹדָה' בִּשְׁבִילוֹ, אֶלָּא הַשִּׂמְחָה הַכִּי גְדוֹלָה שֶׁלּוֹ בְּחַיִּים ! 🌟💖 גַּם כְּשֶׁטּוֹעִים - לוֹמְדִים! הַשֵּׁם אוֹהֵב אוֹתְךָ! 💡

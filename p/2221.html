🌟 שִׁיעוּר 1: מַדּוּעַ חֲנֻכָּה הִיא שְׁמוֹנָה יָמִים?

מַהוּ הַקֶּשֶׁר הַמְּיֻחָד בֵּין הַמִּלִּים 'שְׁמוֹנֶה' וְ'נְשָׁמָה'?

לִשְׁתֵּיהֶן יֵשׁ אוֹתָן אוֹתִיּוֹת בְּדִיּוּק, וְזֶה מְרַמֵּז עַל קֶשֶׁר עָמֹק.

לִשְׁתֵּיהֶן יֵשׁ אֶת אוֹתָהּ הַגִּימַטְרִיָּה, וְזֶה מְסַמֵּל חִבּוּר רוּחָנִי.

שְׁתֵּיהֶן מוֹפִיעוֹת בְּאוֹתוֹ פָּסוּק בַּתּוֹרָה בְּהֶקְשֵׁר שֶׁל חֲנֻכָּה.

שְׁתֵּיהֶן מַתְחִילוֹת בָּאוֹת שׁ', הַמְּסַמֶּלֶת שְׁלֵמוּת וְקִרְבַת אֱלֹהִים.

הַמִּלִּים 'שְׁמוֹנֶה' וְ'נְשָׁמָה' מֻרְכָּבוֹת מֵאוֹתָן הָאוֹתִיּוֹת בְּדִיּוּק, מַה שֶּׁמְּלַמֵּד שֶׁהַנְּשָׁמָה שַׁיֶּכֶת לְ'עוֹלַם הַשְּׁמוֹנֶה' – הָעוֹלָם שֶׁמֵּעַל הַטֶּבַע.

מַדּוּעַ נִקְבְּעוּ יְמֵי חֲנֻכָּה כִּימֵי 'הַלֵּל וְהוֹדָיָה'?

כִּי הַיְּוָנִים נִסּוּ לִפְגֹּעַ בַּנְּשָׁמָה, וְהוֹדָיָה הִיא הַתַּעֲנוּג שֶׁל הַנְּשָׁמָה.

כִּי זֶה מְסַמֵּל אֶת נֵס פַּךְ הַשֶּׁמֶן שֶׁדָּלַק שְׁמוֹנָה יָמִים רְצוּפִים.

כִּי הַחַשְׁמוֹנָאִים הִתְפַּלְּלוּ וְשָׁרוּ הַלֵּל לִפְנֵי כָּל קְרָב שֶׁיָּצְאוּ אֵלָיו.

כִּי זֶה מִנְהָג קָדוּם שֶׁהִתְחִיל כְּבָר בִּזְמַן בֵּית הַמִּקְדָּשׁ הָרִאשׁוֹן.

הַיְּוָנִים רָצוּ לִפְגֹּעַ בַּנְּשָׁמָה שֶׁל עַם יִשְׂרָאֵל. הַתַּעֲנוּג הָאֲמִתִּי שֶׁל הַנְּשָׁמָה הוּא לְהוֹדוֹת וּלְהַלֵּל אֶת ה', וְלָכֵן זוֹהִי מַהוּת הֶחָג.

עוֹלַם הַשְּׁמוֹנֶה, שֶׁאֵלָיו שַׁיֶּכֶת הַנְּשָׁמָה, מְסַמֵּל:

מְצִיאוּת רוּחָנִית שֶׁנִּמְצֵאת מֵעַל חֻקֵּי הַטֶּבַע הָרְגִילִים.

אֶת שְׁמוֹנַת הַשְּׁעָרִים שֶׁהָיוּ בְּבֵית הַמִּקְדָּשׁ הַשֵּׁנִי.

אֶת שְׁמוֹנֶת סִפְרֵי הַיְסוֹד שֶׁל חָכְמַת יְוָן הָעַתִּיקָה.

אֶת שְׁמוֹנֶה הַשָּׁנִים שֶׁלָּקַח לִבְנוֹת מֵחָדָשׁ אֶת בֵּית הַמִּקְדָּשׁ.

הַמִּסְפָּר שְׁמוֹנֶה מְיַצֵּג אֶת מַה שֶּׁנִּמְצָא מִחוּץ לַטֶּבַע וּמֵעָלָיו, בְּדִיּוּק כְּמוֹ הַנְּשָׁמָה שֶׁהִיא 'חֵלֶק אֱלוֹקָה מִמַּעַל'.

🌍 שִׁיעוּר 2: הַסּוֹד שֶׁל הַמִּסְפָּרִים 6, 7 ו-8

לְפִי הַנִּלְמָד, מַה מְסַמֵּל הַמִּסְפָּר 6, וְכֵיצַד הוּא מִתְקַשֵּׁר לַיְּוָנִים?

הוּא מְסַמֵּל אֶת הַטֶּבַע, וְהַמִּלָּה 'יָוָן' בְּגִימַטְרִיָּה הִיא 66.

הוּא מְסַמֵּל אֶת שִׁשָּׁה סִדְרֵי מִשְׁנָה, שֶׁהַיְּוָנִים רָצוּ לְבַטֵּל.

הוּא מְסַמֵּל אֶת שֵׁשֶׁת הַיָּמִים שֶׁבָּהֶם נִצְּחוּ הַחַשְׁמוֹנָאִים.

הוּא מְסַמֵּל אֶת שֵׁשׁ הַמִּצְווֹת הָעִקָּרִיּוֹת שֶׁעֲלֵיהֶן נִלְחֲמוּ.

הַמִּסְפָּר 6 מְיַצֵּג אֶת הָעוֹלָם הַגַּשְׁמִי (6 צְדָדִים), וְהַיְּוָנִים הֶאֱמִינוּ רַק בַּטֶּבַע. לָכֵן, בְּגִימַטְרִיָּה, הַמִּלָּה 'יָוָן' שָׁוָה 66.

מָה הַהֶבְדֵּל הָעִקָּרִי בֵּין הַמַּשְׁמָעוּת שֶׁל 7 לְמַשְׁמָעוּת שֶׁל 8?

7 הוּא קְדֻשָּׁה שֶׁבְּתוֹךְ הַטֶּבַע, ו-8 הוּא קְדֻשָּׁה שֶׁמֵּעַל הַטֶּבַע.

7 מְסַמֵּל אֶת הַתּוֹרָה שֶׁבִּכְתָב, ו-8 אֶת הַתּוֹרָה שֶׁבְּעַל-פֶּה.

7 מְסַמֵּל אֶת בְּרִית הַמִּילָה, ו-8 אֶת בְּרִית הַשַּׁבָּת.

7 מְסַמֵּל אֶת עַם יִשְׂרָאֵל, ו-8 מְסַמֵּל אֶת אֻמּוֹת הָעוֹלָם.

הַמִּסְפָּר 7, כְּמוֹ הַשַּׁבָּת, מְיַצֵּג אֶת הָרוּחָנִיּוּת שֶׁנִּמְצֵאת בְּתוֹךְ מִסְגֶּרֶת הַטֶּבַע. הַמִּסְפָּר 8, לְעֻמַּת זֹאת, פּוֹרֵץ אֶת גְּבוּלוֹת הַטֶּבַע.

בִּימוֹת הַמָּשִׁיחַ, מַה יִּהְיֶה מְיֻחָד בַּכִּנּוֹר, וּמַה זֶה מְלַמֵּד אוֹתָנוּ?

יִהְיוּ לוֹ שְׁמוֹנָה נִימִים, מַה שֶּׁמְּלַמֵּד שֶׁנִּחְיֶה בִּמְצִיאוּת שֶׁמֵּעַל הַטֶּבַע.

הוּא יְנַגֵּן מֵעַצְמוֹ, מַה שֶּׁמְּלַמֵּד עַל הַשְׁגָּחָה פְּרָטִית גְּלוּיָה.

הוּא יִהְיֶה עָשׂוּי מִזָּהָב טָהוֹר, מַה שֶּׁמְּלַמֵּד עַל הַשֶּׁפַע הַגָּדוֹל שֶׁיִּהְיֶה.

יִהְיוּ לוֹ שִׁבְעָה נִימִים, מַה שֶּׁמְּלַמֵּד עַל שְׁלֵמוּת הַבְּרִיאָה.

כַּיּוֹם, בְּחֻקֵּי הַטֶּבַע יֵשׁ רַק 7 תַּוִּים, וְלָכֵן הַכִּנּוֹר הוּא בַּעַל 7 נִימִים. הַכִּנּוֹר שֶׁל יְמוֹת הַמָּשִׁיחַ יִהְיֶה בַּעַל 8 נִימִים, לְהוֹרוֹת עַל הִתְעַלּוּת מֵעַל הַטֶּבַע.

🔑 שִׁיעוּר 3: רְמָזִים נוֹסָפִים לַמִּסְפָּר 8

כֵּיצַד שְׁמָם שֶׁל הַ'חַשְׁמוֹנָאִים' רוֹמֵז לַמִּסְפָּר שְׁמוֹנֶה?

הַשֵּׁם מֵכִיל אֶת הָאוֹת ח' (שֶׁשָּׁוָה 8) וְאֶת הַמִּלָּה 'שְׁמוֹנֶה'.

הָיוּ בַּמִּשְׁפָּחָה שְׁמוֹנָה בָּנִים שֶׁנִּלְחֲמוּ בַּקְּרָב הָרִאשׁוֹן.

הַגִּימַטְרִיָּה שֶׁל הַמִּלָּה 'חַשְׁמוֹנַאי' שָׁוָה לִשְׁמוֹנֶה כָּפוּל שְׁמוֹנֶה.

הֵם הִצְלִיחוּ לְשַׁחְרֵר אֶת בֵּית הַמִּקְדָּשׁ בְּתוֹךְ שְׁמוֹנָה יָמִים.

הַשֵּׁם 'חַשְׁמוֹנַאי' מַרְאֶה לָנוּ שׁוּב אֶת הַהֶקְשֵׁר לְעוֹלַם הַשְּׁמוֹנֶה, עוֹלַם הַנְּשָׁמָה, שֶׁעָלָיו הֵם נִלְחֲמוּ.

מַדּוּעַ נוֹהֲגִים לוֹמַר אֶת פֶּרֶק קי"ט בִּתְהִלִּים כְּשֶׁמִּתְפַּלְּלִים עַל הַנְּשָׁמָה?

כִּי בְּפֶרֶק זֶה, לְכָל אוֹת יֵשׁ שְׁמוֹנָה פְּסוּקִים, וְזֶה מְחַבֵּר לְעוֹלַם הַשְּׁמוֹנֶה שֶׁל הַנְּשָׁמָה.

כִּי הַפֶּרֶק נִכְתַּב עַל יְדֵי דָּוִד הַמֶּלֶךְ בְּגִיל שְׁמוֹנִים וּשְׁמוֹנֶה.

כִּי הַפֶּרֶק מַזְכִיר אֶת הַמִּלָּה 'נְשָׁמָה' בְּדִיּוּק שְׁמוֹנֶה פְּעָמִים.

כִּי הַפֶּרֶק מְדַבֵּר עַל שְׁמוֹנֶה מַעֲלוֹת שֶׁעוֹבֶרֶת הַנְּשָׁמָה לְאַחַר הַמָּוֶת.

הַמִּבְנֶה הַמְּיֻחָד שֶׁל הַפֶּרֶק, שֶׁבּוֹ לְכָל אוֹת יֵשׁ שְׁמוֹנָה פְּסוּקִים, קוֹשֵׁר אוֹתוֹ בְּאֹפֶן עָמֹק לְמַהוּת הַנְּשָׁמָה וּלְעוֹלַם הַשְּׁמוֹנֶה.

אֵיזֶה רֶמֶז נוֹסָף לַמִּסְפָּר שְׁמוֹנֶה מָצָאנוּ בְּנֵס חֲנֻכָּה?

הַמִּלָּה 'שֶׁמֶן' דּוֹמָה בְּאוֹתִיּוֹתֶיהָ לַמִּלָּה 'שְׁמוֹנֶה'.

מִנְיַן הַחַיָּלִים הַיְּוָנִים בַּצָּבָא הָיָה חֵלֶק מִכְּפוּלוֹת שֶׁל שְׁמוֹנֶה.

פַּךְ הַשֶּׁמֶן נִמְצָא בַּחֶדֶר הַשְּׁמִינִי מִיָּמִין בְּבֵית הַמִּקְדָּשׁ.

הַחוֹתָם שֶׁל הַכֹּהֵן הַגָּדוֹל הָיָה בְּצוּרַת מְתֻמָּן (שְׁמוֹנֶה צְלָעוֹת).

הַקֶּשֶׁר בֵּין הַמִּלִּים 'שֶׁמֶן' וְ'שְׁמוֹנֶה' מַדְגִּישׁ שׁוּב כֵּיצַד הַנֵּס הַגַּשְׁמִי (הַשֶּׁמֶן) רוֹמֵז לְמַשְׁמָעוּת רוּחָנִית עֲמֻקָּה יוֹתֵר (עוֹלַם הַשְּׁמוֹנֶה).

🎶 שִׁיעוּר 4: שִׁבְחֵי הַבְּרִיאָה בִּקְבוּצוֹת שֶׁל 8!

כֵּיצַד מְחַלֵּק דָּוִד הַמֶּלֶךְ אֶת הַמְהַלְּלִים בְּפֶרֶק קמ"ח בִּתְהִלִּים?

לְאַרְבַּע קְבוּצוֹת שׁוֹנוֹת, כַּאֲשֶׁר בְּכָל קְבוּצָה יֵשׁ שְׁמוֹנָה מְהַלְּלִים.

לִשְׁמוֹנֶה קְבוּצוֹת שׁוֹנוֹת, כַּאֲשֶׁר בְּכָל קְבוּצָה יֵשׁ שִׁבְעָה מְהַלְּלִים.

לִשְׁתֵּי קְבוּצוֹת, אַחַת בַּשָּׁמַיִם וְאַחַת בָּאָרֶץ, לְלֹא מִסְפָּר קָבוּעַ.

לְפִי סֵדֶר הָאָלֶף-בֵּית, כַּאֲשֶׁר לְכָל אוֹת יֵשׁ מְהַלֵּל אֶחָד.

הַמִּבְנֶה הַמַּדְהִים שֶׁל הַפֶּרֶק מַרְאֶה סֵדֶר שֶׁל 4 קְבוּצוֹת, וּבְכָל אַחַת 8 מְהַלְּלִים. זֶה מַדְגִּישׁ אֶת הַמִּסְפָּר שְׁמוֹנֶה כְּמִסְפָּר שֶׁל הַלֵּל וְהוֹדָיָה.

מִי נִכְלָל בִּקְבוּצַת 'הָאֱנוֹשׁוּת' שֶׁמְּהַלֶּלֶת אֶת ה'?

מַלְכֵי-אֶרֶץ, כָּל-לְאֻמִּים, שָׂרִים, שׁוֹפְטִים, בַּחוּרִים, בְּתוּלוֹת, זְקֵנִים וּנְעָרִים.

כֹּהֲנִים, לְוִיִּים, יִשְׂרְאֵלִים, גֵּרִים, יְתוֹמִים וְאַלְמָנוֹת.

הֶהָרִים, הַגְּבָעוֹת, הָעֵצִים, הַחַיּוֹת וְהָעוֹפוֹת.

שֶׁמֶשׁ, יָרֵחַ, כּוֹכָבִים, מַלְאָכִים וּצְבָאוֹת הַשָּׁמַיִם.

הַקְּבוּצָה הָרְבִיעִית בַּפֶּרֶק מוֹנָה שְׁמוֹנָה סוּגִים שׁוֹנִים שֶׁל בְּנֵי אָדָם, מִכָּל הַמַּעֲמָדוֹת וְהַגִּילָאִים, שֶׁכֻּלָּם מְהַלְּלִים אֶת ה'.

אֵילוּ מֵהַבָּאִים שַׁיָּכִים לִקְבוּצַת 'הַשָּׁמַיִם וְצִבְאוֹתָם'?

שֶׁמֶשׁ וְיָרֵחַ, כּוֹכְבֵי אוֹר וּמַלְאָכִים.

אֵשׁ וּבָרָד, שֶׁלֶג וְקִיטוֹר.

עֵץ פְּרִי וְכָל אֲרָזִים, חַיָּה וּבְהֵמָה.

מַלְכֵי אֶרֶץ וְכָל לְאֻמִּים, שָׂרִים וְשׁוֹפְטִים.

הַקְּבוּצָה הָרִאשׁוֹנָה שֶׁדָּוִד הַמֶּלֶךְ מוֹנֶה הִיא הַקְּבוּצָה שֶׁל הַבְּרוּאִים הַשְּׁמֵימִיִּים, הַגַּשְׁמִיִּים וְהָרוּחָנִיִּים.

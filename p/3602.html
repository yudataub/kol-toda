חֲשָׁשׁוֹת וְאוֹתוֹת 🌟

מָשֶׁה רַבֵּנוּ קִבֵּל שְׁלִיחוּת גְּדוֹלָה 💫 - לְגָאֵל אֶת עַם יִשְׂרָאֵל מִמִּצְרַיִם! אֲבָל הוּא חָשַׁשׁ מְאוֹד 😟. הוּא אָמַר: "הֵן לֹא יַאֲמִינוּ לִי" - הֵם לֹא יַאֲמִינוּ שֶׁהַשֵּׁם שָׁלַח אוֹתִי! אֲבָל הַקָּדוֹשׁ בָּרוּךְ הוּא כָּעַס עָלָיו קְצָת 😔. לָמָּה? כִּי מָשֶׁה דִּבֶּר לָשׁוֹן הָרָע עַל עַם יִשְׂרָאֵל! הוּא חָשַׁב שֶׁהֵם לֹא יַאֲמִינוּ, אֲבָל הָאֱמֶת הִיא שֶׁהֵם כֵּן הָיוּ מוּכָנִים לְהַאֲמִין ! 💖 הַשֵּׁם נָתַן לְמָשֶׁה שְׁלוֹשָׁה אוֹתוֹת מֻפְלָאִים ✨: 1. הַמַּטֶּה שֶׁהָפַךְ לְנָחָשׁ 🐍 - הַנָּחָשׁ מְסַמֵּל לָשׁוֹן הָרָע. הַשֵּׁם רָצָה לְלַמֵּד אֶת מָשֶׁה: תִּזָּהֵר מִדִּבּוּר רָע! אֲבָל יֵשׁ תִּקְוָה 🌈 - הַנָּחָשׁ חָזַר לִהְיוֹת מַטֶּה, כָּךְ אֶפְשָׁר לְתַקֵּן לָשׁוֹן הָרָע! 2. הַיָּד הַמְּצֹרַעַת כַּשֶּׁלֶג ❄️ - צָרַעַת הִיא עוֹנֶשׁ עַל לָשׁוֹן הָרָע. אֲבָל הַשֵּׁם יָכוֹל גַּם לְרַפֵּא! הַיָּד חָזְרָה לִהְיוֹת בְּרִיאָה 💪. 3. מֵי הַיְאוֹר נֶהֶפְכוּ לְדָם 🔴 - זֶה הָיָה רֶמֶז לַמַּכּוֹת שֶׁיָּבוֹאוּ עַל מִצְרַיִם. הַשֵּׁם מְרָאֶה שֶׁהוּא שׁוֹלֵט עַל כָּל הָעוֹלָם!

מָה הָיָה הַחֶשֶׁשׁ הָרִאשׁוֹן שֶׁל מָשֶׁה?

מָשֶׁה חָשַׁשׁ שֶׁהָעָם לֹא יַאֲמִין לוֹ. הוּא אָמַר: 'הֵן לֹא יַאֲמִינוּ לִי'.

מָה מְסַמֵּל הַנָּחָשׁ שֶׁהַמַּטֶּה הָפַךְ אֵלָיו?

הַנָּחָשׁ מְסַמֵּל לָשׁוֹן הָרָע 🐍. הַשֵּׁם לִמֵּד אֶת מָשֶׁה לְהִזָּהֵר מִדִּבּוּר רָע!

כַּמָּה אוֹתוֹת נָתַן הַשֵּׁם לְמָשֶׁה?

הַשֵּׁם נָתַן שְׁלוֹשָׁה אוֹתוֹת ✨: מַטֶּה לְנָחָשׁ, יָד מְצֹרַעַת, וּמַיִם לְדָם!

קוֹשִׁי הַדִּבּוּר וְאַהֲרוֹן הַכֹּהֵן 💬

אַף עַל פִּי שֶׁהַשֵּׁם נָתַן לְמָשֶׁה אוֹתוֹת, הוּא עוֹד חָשַׁשׁ 😰. מָשֶׁה אָמַר: "כְּבַד פֶּה וּכְבַד לָשׁוֹן אָנֹכִי" - קָשֶׁה לִי לְדַבֵּר! אֵיךְ אֲנִי אוּכַל לְהוֹבִיל אֶת הָעָם? הַקָּדוֹשׁ בָּרוּךְ הוּא הִרְגִּיעַ אוֹתוֹ בְּאַהֲבָה ❤️: "מִי שָׂם פֶּה לָאָדָם?" אֲנִי, הַשֵּׁם, בָּרָאתִי אֶת הַפֶּה שֶׁלְּךָ! אֲנִי אֶעֱזוֹר לְךָ לְדַבֵּר! 💪 זֶה לִמֵּד אוֹתָנוּ מַשֶּׁהוּ חָשׁוּב מְאוֹד: הַשֵּׁם לֹא מְחַפֵּשׂ אֲנָשִׁים מֻשְׁלָמִים 🌟. הוּא מְחַפֵּשׂ אֲנָשִׁים שֶׁרוֹצִים לְהִתְגַּבֵּר עַל הַקְּשָׁיִים שֶׁלָּהֶם בְּעֶזְרָתוֹ! מָשֶׁה גַּם חָשַׁשׁ מִדָּבָר נוֹסָף 😔: אוּלַי אַהֲרוֹן אָחִיו יְקַנֵּא בּוֹ? אַהֲרוֹן הָיָה נָבִיא שֶׁל הָעָם במָשֶׁךְ שְׁמוֹנִים שָׁנָה. עַכְשָׁיו מָשֶׁה חוֹזֵר וְלוֹקֵחַ אֶת מְקוֹמוֹ? אֲבָל הַשֵּׁם הִרְגִּיעַ אֶת מָשֶׁה: "הִנֵּה הוּא יֹצֵא לִקְרָאתֶךָ וְרָאֲךָ וְשָׂמַח בְּלִבּוֹ" 😊. אַהֲרוֹן לֹא רַק שֶׁלֹּא יְקַנֵּא - הוּא יִשְׂמַח בְּלֵב שָׁלֵם! 💖 וּבְזְכוּת הַשִּׂמְחָה הַטְּהוֹרָה שֶׁל אַהֲרוֹן 🎉, הוּא זָכָה לִלְבּוֹשׁ אֶת הָאוּרִים וְהַתּוּמִּים - כֵּלִים מְקוּדָּשִׁים שֶׁהָיוּ עַל הַחוֹשֶׁן! זֶה לִמֵּד אוֹתָנוּ: כְּשֶׁאֲנַחְנוּ שְׂמֵחִים בֶּאֱמֶת בְּהַצְלָחַת הַזּוּלָת, אֲנַחְנוּ זוֹכִים לְאוֹר פְּנִימִי שֶׁמֵּאִיר אֶת חַיֵּינוּ! ✨

מָה הָיָה קוֹשִׁי הַדִּבּוּר שֶׁל מָשֶׁה?

מָשֶׁה הָיָה 'כְּבַד פֶּה וּכְבַד לָשׁוֹן' - הוּא מִגְמֵג וְקָשֶׁה לוֹ לְדַבֵּר 💬.

אֵיךְ הִגִּיב אַהֲרוֹן כְּשֶׁפָּגַשׁ אֶת מָשֶׁה?

אַהֲרוֹן שָׂמַח בְּלִבּוֹ וְנִשֵּׁק לְמָשֶׁה 😊💖! הוּא לֹא קִנֵּא כְּלָל!

מָה זָכָה אַהֲרוֹן בִּזְכוּת שִׂמְחָתוֹ?

בִּזְכוּת שִׂמְחָתוֹ הַטְּהוֹרָה ✨, אַהֲרוֹן זָכָה לָאוּרִים וְהַתּוּמִּים!

הַדֶּרֶךְ לְמִצְרַיִם וְסַכָּנָה ⚠️

מָשֶׁה הֵבִין שֶׁעָלָיו לָצֵאת לְמִצְרַיִם 🚶. אֲבָל לִפְנֵי כֵן, הוּא נִפְרַד מִיִּתְרוֹ חוֹתְנוֹ שֶׁנָּתַן לוֹ בַּיִת במָשֶׁךְ אַרְבָּעִים שָׁנָה 🏡. מָשֶׁה אָמַר לוֹ: "אֵלְכָה נָּא וְאָשׁוּבָה אֶל אַחַי" - אֲנִי רוֹצֶה לִרְאוֹת אֶת מִשְׁפַּחְתִּי. יִתְרוֹ בֵּרַךְ אוֹתוֹ: "לֵךְ לְשָׁלוֹם" 🙏 - תֵּלֵךְ וְתַחֲזוֹר בְּשָׁלוֹם! זוֹ בְּרָכָה טוֹבָה וִיפָה. הַשֵּׁם הוֹדִיעַ לְמָשֶׁה: "מֵתוּ כָּל הָאֲנָשִׁים הַמְבַקְשִׁים אֶת נַפְשֶׁךָ" ✅. עַכְשָׁיו הַדֶּרֶךְ בְּטוּחָה! מָשֶׁה לָקַח אֶת אִשְׁתּוֹ צִפּוֹרָה וְאֶת שְׁנֵי בָּנָיו - גֵּרְשׁוֹם וֶאֱלִיעֶזֶר - וְיָצְאוּ לַדֶּרֶךְ. אֲבָל אָז קָרָה מַשֶּׁהוּ נוֹרָא 😱! מַלְאַךְ ה' הוֹפִיעַ וּבִקֵּשׁ לְהָמִית אֶת מָשֶׁה! לָמָּה?! הַסִּבָּה: מָשֶׁה לֹא מָל אֶת בְּנוֹ גֵּרְשׁוֹם בַּיּוֹם הַשְּׁמִינִי לְלֵדָתוֹ 👶. הוּא דָּחָה אֶת בְּרִית הַמִּילָה - מִצְוָה חֲשׁוּבָה מְאוֹד! אוּלַי מָשֶׁה חָשַׁב: אֲנִי טָרוּד בְּגָאֻלַּת עַם יִשְׂרָאֵל 🎯, אֶעֱשֶׂה אֶת זֶה מְאֻחָר יוֹתֵר. אֲבָל הַשֵּׁם לִמֵּד אוֹתוֹ: אֵין שׁוּם דָּבָר שֶׁדּוֹחֶה קִיּוּם מִצְוָה בִּזְמַנָּהּ! ⏰ צִפּוֹרָה, אִשְׁתּוֹ, הֵבִינָה מַהֵר מָה קוֹרֶה 💪. הִיא לָקְחָה צוּר (אֶבֶן חַדָּה), מָלָה אֶת בְּנָהּ, וְהִצִּילָה אֶת מָשֶׁה מֵהַמָּוֶת! 🙌 זֶה הַשִּׁיעוּר הַחָשׁוּב: מִצְווֹת צְרִיכוֹת לְהִתְבַּצַּע בִּזְמַן, בְּלִי עִכּוּבִים וּבְלִי דְּחִיּוֹת ⚡. אֵין תִּרוּץ טוֹב לִדְחוֹת מִצְוָה!

מָה אָמַר יִתְרוֹ לְמָשֶׁה לִפְנֵי שֶׁיָּצָא?

יִתְרוֹ בֵּרַךְ: 'לֵךְ לְשָׁלוֹם' 🙏 - בְּרָכָה טוֹבָה לָלֶכֶת וְלַחֲזוֹר בְּשָׁלוֹם!

לָמָּה מַלְאַךְ ה' בִּקֵּשׁ לְהָמִית אֶת מָשֶׁה?

מָשֶׁה לֹא מָל אֶת בְּנוֹ בִּזְמַן 👶. זֶה לִמֵּד: אֵין לִדְחוֹת מִצְווֹת!

מִי הִצִּיל אֶת מָשֶׁה מֵהַמָּוֶת?

צִפּוֹרָה הַגִּבּוֹרָה 💪 מָלָה אֶת הַבֵּן וְהִצִּילָה אֶת מָשֶׁה!

הַמִּפְגָּשׁ הַמַּרְגִּישׁ וְהָאֱמוּנָה 💖

הַשֵּׁם צִוָּה עַל אַהֲרוֹן: "לֵךְ לִקְרַאת מֹשֶׁה הַמִּדְבָּרָה" 🏜️. אַהֲרוֹן יָצָא לַמִּדְבָּר לַמִּפְגָּשׁ הִיסְטוֹרִי עִם אָחִיו שֶׁלֹּא רָאָה אוֹתוֹ במָשֶׁךְ עֶשְׂרוֹת שָׁנִים! הֵם נִפְגָּשׁוּ בְּהַר הָאֱלֹהִים - הַר סִינַי ⛰️ - הַמָּקוֹם הַקָּדוֹשׁ בְּיוֹתֵר! "וַיִּרְאֵהוּ וַיִּשַּׁק לוֹ" 💋 - אַהֲרוֹן רָאָה אֶת מָשֶׁה וְנִשֵּׁק אוֹתוֹ מִתּוֹךְ שִׂמְחָה אֲמִתִּית, בְּלִי קִנְאָה! חֲשָׁשׁוֹתָיו שֶׁל מָשֶׁה הִתְבַּדּוּ 🎉! אַהֲרוֹן אֵינוֹ כּוֹעֵס, אֵינוֹ מְקַנֵּא - הוּא שָׂמֵחַ בְּלֵב שָׁלֵם! מָשֶׁה סִפֵּר לְאַהֲרוֹן אֶת כָּל דִּבְרֵי ה' וְאֶת כָּל הָאוֹתוֹת 💫. יַחַד, מָשֶׁה וְאַהֲרוֹן אָסְפוּ אֶת כָּל זִקְנֵי בְּנֵי יִשְׂרָאֵל 👨‍👨‍👦‍👦. זֶה הָיָה רֶגַע מְכוֹנֵן! לָרִאשׁוֹנָה מֵאָז שָׁנִים שֶׁל שִׁעְבּוּד, הָעָם שׁוֹמֵעַ בְּשׂוֹרַת גְּאֻלָּה 🎺! אַהֲרוֹן סִפֵּר לָהֶם אֶת כָּל דִּבְרֵי ה', וּמָשֶׁה עָשָׂה אֶת הָאוֹתוֹת לְעֵינֵיהֶם ✨. וְאָז הַרֶּגַע שֶׁמָּשֶׁה חָשַׁשׁ מִמֶּנּוּ הִתְגַּשֵּׁם - אֲבָל בְּדִיּוּק הֵפֶךְ!

אַרְבָּעָה נְאוּמִים — מְשַׂחֲקִים בְּסִוּוּג 📜

מִשְׂחָק לִמּוּדִי לְכִתָּה ג׳ • אַרְבָּעָה שִׁעוּרִים • שְׁלוֹשִׁים וּשְׁתַּיִם שְׁאֵלוֹת מִיּוּן

שִׁעוּר א׳ — נְאוּם הַתּוֹכֵחָה: מוֹכִיחַ כִּי אוֹהֵב

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: חֵטְא הַמְּרַגְּלִים?

חֵטְא הַמְּרַגְּלִים הוּא סִפּוּר עָבָר שֶׁמֹּשֶׁה מַזְכִּיר כְּתוֹכֵחָה, וְלָכֵן הוּא שַׁיָּךְ לַנְּאוּם הָרִאשׁוֹן.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: קְרִיאַת שְׁמַע?

קְרִיאַת שְׁמַע הִיא מִצְוָה, וְהִיא מוֹפִיעָה מִיָּד אַחֲרֵי פֶּרֶק ה׳ — בְּתוֹךְ נְאוּם הַמִּצְווֹת.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: מַעֲמַד הַר גְּרִזִים וְהַר עֵיבָל?

עֲמִידַת הָעָם עַל שְׁנֵי הֶהָרִים הִיא טֶקֶס שֶׁל כְּרִיתַת בְּרִית, וְלָכֵן שַׁיֶּכֶת לִנְאוּם הַבְּרִית.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: שִׁירַת הַאֲזִינוּ?

שִׁירַת הַאֲזִינוּ נֶאֶמְרָה בִּימֵי חַיָּיו הָאַחֲרוֹנִים שֶׁל מֹשֶׁה, וְלָכֵן שַׁיֶּכֶת לִנְאוּם הַפְּרֵדָה.

שִׁעוּר ב׳ — נְאוּם הַמִּצְווֹת: לִחְיוֹת לְפִי דְּבַר הַשֵּׁם

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: אַרְבָּעִים שְׁנוֹת הַנְּדוּדִים בַּמִּדְבָּר כְּעֹנֶשׁ?

הָעֹנֶשׁ עַל חֵטְא הַמְּרַגְּלִים הוּא חֵלֶק מִסִּפּוּר הַמַּסָּע שֶׁמֹּשֶׁה מוֹכִיחַ עָלָיו בַּנְּאוּם הָרִאשׁוֹן.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: פָּרָשַׁת וְהָיָה אִם שָׁמֹעַ?

פָּרָשַׁת וְהָיָה אִם שָׁמֹעַ מְלַמֶּדֶת מִצְווֹת וּשְׂכַר קִיּוּמָן, וְלָכֵן שַׁיֶּכֶת לִנְאוּם הַמִּצְווֹת.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: עֶשְׂרִים וְאַרְבַּע הַקְּלָלוֹת שֶׁבְּפֶרֶק כ״ח?

הַקְּלָלוֹת נֶאֶמְרוּ כְּחֵלֶק מִטֶּקֶס הַבְּרִית עַל הֶהָרִים, וְלָכֵן שַׁיָּכוֹת לִנְאוּם הַבְּרִית.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: בִּרְכַּת מֹשֶׁה לִשְׁבָטִים, וְזֹאת הַבְּרָכָה?

מֹשֶׁה בֵּרַךְ כָּל שֵׁבֶט בְּבִרְכָה מְיֻחֶדֶת מַמָּשׁ לִפְנֵי מוֹתוֹ, וְלָכֵן זֶה שַׁיָּךְ לִנְאוּם הַפְּרֵדָה.

שִׁעוּר ג׳ — נְאוּם הַבְּרִית: הַר גְּרִזִים וְהַר עֵיבָל

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: הָאַזְהָרָה מִפְּנֵי עֲבוֹדָה זָרָה שֶׁבְּפֶרֶק ד׳?

הָאַזְהָרָה נֶאֶמְרָה בְּפֶרֶק ד׳, בְּתוֹךְ נְאוּם הַתּוֹכֵחָה, לִפְנֵי שֶׁמַּתְחִיל הַנְּאוּם הַשֵּׁנִי.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: מִצְוַת שִׁלּוּחַ הַקֵּן?

שִׁלּוּחַ הַקֵּן הִיא מִצְוָה מְעַשִּׂית, כְּמוֹ שְׁאָר הַמִּצְווֹת שֶׁבִּנְאוּם הַמִּצְווֹת.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: הַבְּרִית שֶׁנִּכְרְתָה בְּעַרְבוֹת מוֹאָב?

זוֹ בְּרִית נוֹסֶפֶת שֶׁמֹּשֶׁה כּוֹרֵת עִם הָעָם, וְלָכֵן שַׁיֶּכֶת לִנְאוּם הַבְּרִית עַצְמוֹ.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: מִצְוַת הַקְהֵל?

מִצְוַת הַקְהֵל נֶאֶמְרָה לְקַרְאַת פְּרֵדַת מֹשֶׁה, כְּדֵי שֶׁכָּל דּוֹר יִשְׁמַע אֶת הַתּוֹרָה יַחַד.

שִׁעוּר ד׳ — נְאוּם הַפְּרֵדָה: מְבָרֵךְ עַד הַנְּשִׁימָה הָאַחֲרוֹנָה

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: הַזְכָּרַת חֵטְא הָעֵגֶל בְּהַר חוֹרֵב?

גַּם חֵטְא הָעֵגֶל מֻזְכָּר כְּתוֹכֵחָה מִתּוֹךְ סִפּוּר הֶעָבָר, וְלָכֵן שַׁיָּךְ לִנְאוּם הַתּוֹכֵחָה.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: דִּינֵי מֶלֶךְ יִשְׂרָאֵל?

דִּינֵי הַמֶּלֶךְ הֵם חֹק מְעַשִּׂי מִתּוֹךְ אוֹסֵף הַמִּצְווֹת שֶׁבַּנְּאוּם הַשֵּׁנִי.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: הַנִּסְתָּרוֹת לַה׳ אֱלֹהֵינוּ וְהַנִּגְלוֹת לָנוּ?

פָּסוּק זֶה נֶאֱמַר בְּסוֹף נְאוּם הַבְּרִית, וּמְלַמֵּד שֶׁכָּל הָעָם עָרֵב לְקִיּוּם הַבְּרִית יַחַד.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: הַפְּסוּקִים שֶׁמְּעִידִים שֶׁלֹּא קָם עוֹד נָבִיא כְּמֹשֶׁה?

פְּסוּקִים אֵלֶּה חוֹתְמִים אֶת הַתּוֹרָה כֻּלָּהּ, בְּתוֹךְ נְאוּם הַפְּרֵדָה, אַחֲרֵי מִיתַת מֹשֶׁה.

שִׁעוּר ה׳ — סִבּוּב מִיּוּן: כָּל הַנּוֹשְׂאִים מְעֹרְבָבִים

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: הַזְכָּרַת מַעֲמַד הַר סִינַי כְּהַקְדָּמָה לַתּוֹכֵחָה?

מֹשֶׁה מַזְכִּיר אֶת מַעֲמַד הַר סִינַי כְּחֵלֶק מִסִּפּוּר הֶעָבָר בְּנְאוּם הַתּוֹכֵחָה, לִפְנֵי שֶׁהוּא חוֹזֵר עָלָיו בַּמִּצְווֹת.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: אִסּוּר רִבִּית?

אִסּוּר רִבִּית הוּא דִּין מְעַשִּׂי בֵּין אָדָם לַחֲבֵרוֹ, בְּדִיּוּק כְּמוֹ שְׁאָר חֻקֵּי נְאוּם הַמִּצְווֹת.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: עֲנִיַּת אָמֵן שֶׁל הָעָם אַחֲרֵי כָּל בְּרָכָה וּקְלָלָה?

עֲנִיַּת הָאָמֵן הָיְתָה חֵלֶק מִטֶּקֶס הַר גְּרִזִים וְהַר עֵיבָל, בְּתוֹךְ נְאוּם הַבְּרִית.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: עֲלִיַּת מֹשֶׁה לְהַר נְבוֹ וּרְאִיַּת הָאָרֶץ?

עֲלִיַּת הַר נְבוֹ הִיא הָרֶגַע הָאַחֲרוֹן בְּחַיֵּי מֹשֶׁה, וְלָכֵן שַׁיֶּכֶת לִנְאוּם הַפְּרֵדָה.

שִׁעוּר ו׳ — סִבּוּב אַלִּיפוּת: הַכֹּל בְּיַחַד

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: מִלְחֶמֶת סִיחוֹן וְעוֹג וְכִבּוּשׁ עֵבֶר הַיַּרְדֵּן?

כִּבּוּשׁ עֵבֶר הַיַּרְדֵּן הוּא הַהֶמְשֵׁךְ הַמֻּצְלָח שֶׁל סִפּוּר הַמַּסָּעוֹת בְּנְאוּם הַתּוֹכֵחָה.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: חֲזָרָה עַל עֲשֶׂרֶת הַדִּבְּרוֹת?

עֲשֶׂרֶת הַדִּבְּרוֹת נִשְׁנוֹת בְּפֶתַח נְאוּם הַמִּצְווֹת, כְּבָסִיס לְכָל שְׁאָר הַחֻקִּים.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: פָּרָשַׁת הַתְּשׁוּבָה, וְשַׁבְתָּ עַד ה׳ אֱלֹהֶיךָ?

פָּרָשַׁת הַתְּשׁוּבָה מוֹפִיעָה בְּסוֹף נְאוּם הַבְּרִית, כְּפֵתְרוֹן לְמִי שֶׁכָּשַׁל בַּקְּלָלוֹת שֶׁלִּפְנֵי כֵן.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: מִנּוּי יְהוֹשֻׁעַ כְּמַנְהִיג הַבָּא?

מִנּוּי יְהוֹשֻׁעַ קוֹרֶה מַמָּשׁ לִפְנֵי מוֹתוֹ שֶׁל מֹשֶׁה, כְּחֵלֶק מֵהֲכָנַת הָעָם לַפְּרֵדָה.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: עָרֵי הַמִּקְלָט שֶׁהֻבְדְּלוּ בְּעֵבֶר הַיַּרְדֵּן?

הַבְדָּלַת עָרֵי הַמִּקְלָט נִזְכֶּרֶת בְּסוֹף פֶּרֶק ד׳, בְּתוֹךְ נְאוּם הַתּוֹכֵחָה עַצְמוֹ.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: מִצְוַת הַמַּעֲשֵׂר?

מִצְוַת הַמַּעֲשֵׂר הִיא דִּין מְעַשִּׂי בְּתוֹךְ אוֹסֵף הַחֻקִּים שֶׁל נְאוּם הַמִּצְווֹת.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: הַפָּסוּק כִּי קָרוֹב אֵלֶיךָ הַדָּבָר מְאֹד?

פָּסוּק זֶה נֶאֱמַר לִקְרַאת סוֹף נְאוּם הַבְּרִית, כְּעִדּוּד לִפְנֵי הַתְּשׁוּבָה וְהַקִּיּוּם.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: כְּתִיבַת סֵפֶר הַתּוֹרָה וּמְסִירָתוֹ לַלְוִיִּם?

מֹשֶׁה כּוֹתֵב וּמוֹסֵר אֶת סֵפֶר הַתּוֹרָה מַמָּשׁ לִפְנֵי מוֹתוֹ, כְּחֵלֶק מִנְּאוּם הַפְּרֵדָה.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: תֵּאוּר הַמַּסָּע מֵחוֹרֵב וְעַד קָדֵשׁ בַּרְנֵעַ?

תֵּאוּר תְּחִלַּת הַמַּסָּע הוּא חֵלֶק מִסִּפּוּר הַמַּסָּעוֹת שֶׁפּוֹתֵחַ אֶת נְאוּם הַתּוֹכֵחָה.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: מִצְוַת הַבִּכּוּרִים?

הֲבָאַת הַבִּכּוּרִים הִיא מִצְוָה מְעַשִּׂית הַנִּמְנֵית עִם שְׁאָר חֻקֵּי נְאוּם הַמִּצְווֹת.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: הָאַזְהָרָה הַחֲמוּרָה אִם לֹא יִשְׁמְעוּ לְקוֹל ה׳?

הָאַזְהָרוֹת הַחֲמוּרוֹת נֶאֶמְרוּ עַל הַר עֵיבָל, כְּחֵלֶק מִטֶּקֶס נְאוּם הַבְּרִית.

לְאֵיזֶה נְאוּם שַׁיָּךְ הַנּוֹשֵׂא: מִיתַת מֹשֶׁה בְּהַר נְבוֹ?

מִיתַת מֹשֶׁה הִיא הָרֶגַע הַחוֹתֵם אֶת נְאוּם הַפְּרֵדָה וְאֶת הַתּוֹרָה כֻּלָּהּ.

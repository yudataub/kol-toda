כֹּחָהּ שֶׁל תְּהִלָּה

⛶ פתח במסך מלא
📄 תוכן העמוד בטקסט

🌟 לְמַעַן תְּהִלָּתוֹ: מַטְּרַת הַבְּרִיאָה

כְּדֵי שֶׁיִּלְמְדוּ אֶת כָּל הַתּוֹרָה.

כְּדֵי שֶׁיִּשְׁמְרוּ אֶת כָּל הַמִּצְווֹת.

כְּדֵי שֶׁיְּסַפְּרוּ וִיפָרְטוּ אֶת תְּהִלָּתוֹ.

כְּדֵי שֶׁיִּהְיוּ בְּרִיאִים וַחֲזָקִים.

הַפָּסוּק "עַם זוּ יָצַרְתִּי לִי, תְּהִלָּתִי יְסַפֵּרוּ" מְצַיֵּן בִּמְפֹרָשׁ שֶׁהַמַּטָּרָה הִיא סִפּוּר הַתְּהִלָּה.

מְצֻיָּן! זֶה בְּדִיּוּק יִעוּדֵנוּ הָעֶלְיוֹן.

שֶׁנּוּכַל לִלְמֹד תּוֹרָה בְּשִׂמְחָה.

שֶׁנּוּכַל לְהַמְשִׁיךְ לִהְיוֹת "תְּהִלָּתוֹ".

כְּדֵי שֶׁנּוּכַל לַעֲבֹד וּלְפַרְנֵס.

כְּדֵי שֶׁלֹּא נְצַעֵר אֶת מִשְׁפַּחְתֵּנוּ.

אָנוּ מְבַקְּשִׁים רְפוּאָה לֹא כְּמַטָּרָה בִּפְנֵי עַצְמָהּ, אֶלָּא כְּדֵי שֶׁנּוּכַל לְמַלֵּא אֶת יִעוּדֵנוּ - לִהְיוֹת "תְּהִלָּתוֹ", כְּלוֹמַר, לְשַׁבֵּחַ אוֹתוֹ.

עֲבוֹדָה נֶהֱדֶרֶת! הֵבַנְתָּ אֶת הָעֹמֶק.

כְּבוֹד וְשֵׁם ה' מִתְגַּדֵּל וּמִתְהַלֵּל.

הֵם מְקַבְּלִים עַל כָּךְ שָׂכָר חוֹמְרִי.

הֵם עוֹשִׂים טוֹבָה גְּדוֹלָה לַבּוֹרֵא.

הֵם מוֹכִיחִים שֶׁהֵם עַם נִבְחָר.

הַשִּׁעוּר מַדְגִּישׁ שֶׁבִּפְעֻלַּת הַשֶּׁבַח "מִתְהַלֵּל וּמִתְגַּדֵּל כְּבוֹד הַשֵּׁם וּשְׁמוֹ".

נָכוֹן מְאוֹד! כָּךְ מַגְדִּילִים כְּבוֹד שָׁמַיִם.

👑 יוֹשֵׁב תְּהִלּוֹת יִשְׂרָאֵל

עַל כִּסֵּא הַכָּבוֹד בַּשָּׁמַיִם.

עַל מַעֲשֵׂי הַצְּדָקָה שֶׁלָּנוּ.

עַל הַתְּהִלּוֹת שֶׁל עַם יִשְׂרָאֵל.

עַל כָּל הַזָּהָב שֶׁבָּעוֹלָם.

הַפָּסוּק הַמְּפֹרָשׁ הוּא "יוֹשֵׁב תְּהִלּוֹת יִשְׂרָאֵל". הַתְּהִלּוֹת הֵן "הַכֵּס" שֶׁעָלָיו הוּא מוֹצֵא אֶת עִקַּר הַנַּחַת שֶׁלּוֹ.

יָפֶה מְאוֹד! תְּשׁוּבָה מְדֻיֶּקֶת.

לְשִׂמְחַת חָתָן בְּיוֹם חֻפָּתוֹ.

לְשִׂמְחַת אָב הַשּׁוֹמֵעַ "תּוֹדָה" מִיַּלְדּוֹ.

לְשִׂמְחַת מֶלֶךְ שֶׁנִּצַּח בַּמִּלְחָמָה.

לְשִׂמְחַת אָדָם שֶׁזָּכָה בִּפְרָס.

הַמָּשָׁל הַמֶּרְכָּזִי בַּשִּׁעוּר הוּא עַל אָב שֶׁשּׁוֹמֵעַ אֶת יַלְדּוֹ הַקָּטָן, שֶׁבְּקֹשִׁי מְדַבֵּר, אוֹמֵר "תּוֹדָה", וּמִתְמַלֵּא שִׂמְחָה.

נָכוֹן לַחֲלוּטִין! זֶה מַמְחִישׁ אֶת הָרַעְיוֹן.

מַשְׁמָעוּת שֶׁל צִפִּיָּה וְהִתְעַכְּבוּת.

מַשְׁמָעוּת שֶׁל מְנוּחָה וְשַׁלְוָה.

מַשְׁמָעוּת שֶׁל מְלוּכָה וּשְׁלִיטָה.

מַשְׁמָעוּת שֶׁל דִּין וּמִשְׁפָּט.

עַל פִּי רַשִׁ"י, "יְשִׁיבָה" הִיא גַּם "לְשׁוֹן עִכּוּב", כְּלוֹמַר הַקָּבָּ"ה מִתְעַכֵּב וּמְצַפֶּה לִשְׁמֹעַ אֶת תְּהִלּוֹתֵינוּ.

אַתָּה לוֹמֵד מַהֵר! מְצֻיָּן!

🙏 יְהוּדִי = הוֹדָאָה

מֵהַמִּלָּה "יֶדַע", לְלַמֵּד עַל חֲשִׁיבוּת הַלִּמּוּד.

מֵהַמִּלָּה "יָחִיד", לְלַמֵּד עַל יִחוּדִיּוּתֵנוּ.

מֵהַמִּלָּה "הוֹדָאָה", לְלַמֵּד שֶׁמַּהוּתֵנוּ לְהוֹדוֹת.

מֵהַמִּלָּה "אַהֲבָה", לְלַמֵּד עַל אַהֲבַת ה'.

הַשִּׁעוּר קוֹשֵׁר בְּאֹפֶן יָשִׁיר בֵּין "יְהוּדִי" לְ"הוֹדָאָה", וּמַסְבִּיר שֶׁזּוֹ מַהוּתוֹ שֶׁל הַיְּהוּדִי.

אַתָּה בַּדֶּרֶךְ הַנְּכוֹנָה! עֲבוֹדָה מְעֻלָּה.

בַּאֲמִירַת "שְׁמַע יִשְׂרָאֵל".

בַּאֲמִירַת בִּרְכַּת "עַל הַנִּסִּים".

בַּאֲמִירַת מִזְמוֹר תְּהִלִּים.

בַּאֲמִירַת "מוֹדֶה אֲנִי לְפָנֶיךָ".

הַשִּׁעוּר מְצַיֵּן שֶׁכָּל יְהוּדִי פּוֹתֵחַ אֶת יוֹמוֹ בְּהוֹדָאָה לַה': "מוֹדֶה אֲנִי לְפָנֶיךָ".

בִּדְיוּק כָּכָה! הַהַתְחָלָה הֲכִי חֲשׁוּבָה.

אֲמִירַת "עַל הַנִּסִּים" וְ"הַלֵּל".

הַדְלָקַת נֵרוֹת וּסְעוּדוֹת מִצְוָה.

תְּפִלַּת מוּסָף וּקְרִיאָה בַּתּוֹרָה.

מַתַּן צְדָקָה וּמִשְׁלוֹחַ מָנוֹת.

הַשִּׁעוּר מַדְגִּישׁ אֶת שְׁתֵּי הַתּוֹסָפוֹת הַתְּפִלָּתִיּוֹת הַמֶּרְכָּזִיּוֹת בַּחֲנֻכָּה: "עַל הַנִּסִּים" וְ"הַלֵּל", שֶׁכֻּלָּן עִנְיָן שֶׁל שֶׁבַח.

כָּל הַכָּבוֹד! אַתָּה מַכִּיר אֶת עִנְיְנֵי הַיּוֹם.

✨ הַכֹּחַ לְהוֹרִיד שֶׁפַע

לוֹמַר אוֹתָהּ בְּקוֹל רָם מְאֹד.

לְהִתְבּוֹנֵן וְלַחְשֹׁב עַל הַטּוֹבוֹת.

לְהִתְפַּלֵּל רַק בְּבֵית כְּנֶסֶת.

לַחֲזֹר עַל הַמִּלִּים מֵאָה פְּעָמִים.

הַשִּׁעוּר מַדְגִּישׁ אֶת הַחֲשִׁיבוּת שֶׁל "לְהִתְבּוֹנֵן בַּדָּבָר לַחְשֹׁב עַל כָּל הַדְּבָרִים הַנִּפְלָאִים".

נָכוֹן מְאוֹד! הִתְבּוֹנְנוּת הִיא הַמַּפְתֵּחַ.

שַׁעֲרֵי גַּן עֵדֶן נִפְתָּחִים.

יוֹרְדִים חֲסָדִים גְּדוֹלִים וִישׁוּעוֹת.

הַמַּלְאָכִים מַפְסִיקִים לָשִׁיר.

הַקָּבָּ"ה כּוֹתֵב אֶת שְׁמָם בַּסֵּפֶר.

הַשִּׁעוּר מְצַטֵּט יְשִׁירוֹת מֵהַסְּפָרִים שֶׁפְּעֻלָּה זוֹ "נוֹרָא חֲסָדִים מִלְמַעְלָה וּמֵבִיא יְשׁוּעוֹת גְּדוֹלוֹת".

בְּדִיּוּק! תְּשׁוּבָה מַבְרִיקָה.

רַק לָאָדָם הַמְּשַׁבֵּחַ עַצְמוֹ.

רַק לִכְלַל עַם יִשְׂרָאֵל.

גַּם לָאָדָם הַפְּרָטִי וְגַם לַכְּלָל.

רַק לְצַדִּיקֵי הַדּוֹר.

הַטֶּקְסְט מְצַיֵּן בִּמְפֹרָשׁ שֶׁהַיְשׁוּעוֹת הֵן "לַפְּרָט וְלַכְּלָל", כְּלוֹמַר, הֵן אִישִׁיּוֹת וְקְהִלָּתִיּוֹת.

אַתָּה קוֹלֵט אֶת הַכֹּחַ הַמְּשֻׁתָּף! מְעוּלֶה.

נַעֲשֶׂה וְנִשְׁמָע - סוֹד הָאֱמוּן בְּאַבָּא

⛶ פתח במסך מלא
📄 תוכן העמוד בטקסט

🎁 הַקּוּפְסָה הַמְּסַתּוֹרִית - לִסְמוֹךְ עַל אַבָּא

🌟 חֲבֵרִים יְקָרִים! בּוֹאוּ נְשַׂחֵק מַשְׂחָק! 🎯 אִם אָדָם זָר יִתֵּן לָכֶם קוּפְסָה סְגוּרָה וְיֹאמַר: ״קַח אֶת זֶה!״ - תִּפְחֲדוּ לִפְתּוֹחַ, נָכוֹן? 😰 אֲבָל אִם אַבָּא אוֹ אִמָּא יִתְּנוּ לָכֶם קוּפְסָה וְיֹאמְרוּ: ״זֹאת מַתָּנָה בִּשְׁבִילְכֶם!״ - תִּפְתְּחוּ בְּשִׂמְחָה! 💖 לָמָּה? כִּי אַתֶּם יוֹדְעִים שֶׁאַבָּא וְאִמָּא אוֹהֲבִים אֶתְכֶם וְרוֹצִים רַק בְּטוֹבָתְכֶם! 🌈 כָּךְ בְּדִיּוּק הִרְגִּישׁוּ עַם יִשְׂרָאֵל כְּשֶׁקִּבְּלוּ אֶת הַתּוֹרָה! הֵם סָמְכוּ עַל הַקָּדוֹשׁ בָּרוּךְ הוּא בְּאַהֲבָה שְׁלֵמָה! ✨

מַדּוּעַ לֹא נִפְחַד לִפְתּוֹחַ קוּפְסָה מֵאַבָּא וְאִמָּא?

נָכוֹן מְאֹד! כְּשֶׁאַבָּא וְאִמָּא נוֹתְנִים לָנוּ מַשֶּׁהוּ, אָנוּ סוֹמְכִים עֲלֵיהֶם מִתּוֹךְ אַהֲבָה! כָּךְ גַּם עַם יִשְׂרָאֵל סָמַךְ עַל הַקָּדוֹשׁ בָּרוּךְ הוּא! 💖

מָה הַהֶבְדֵּל בֵּין קוּפְסָה מֵאָדָם זָר לְקוּפְסָה מֵאַבָּא?

מְצֻיָּן! הָאֱמוּן וְהָאַהֲבָה הֵם הַמַּפְתֵּחַ! 🗝️ זֶהוּ הַסּוֹד שֶׁל ״נַעֲשֶׂה וְנִשְׁמָע״ - לִסְמוֹךְ עַל הַשֵּׁם בְּאַהֲבָה! ✨

אֵיךְ הִרְגִּישׁוּ עַם יִשְׂרָאֵל כְּשֶׁקִּבְּלוּ אֶת הַתּוֹרָה?

נִפְלָא! עַם יִשְׂרָאֵל סָמַךְ עַל הַקָּדוֹשׁ בָּרוּךְ הוּא כְּמוֹ יֶלֶד עַל אָבִיו! הַשֵּׁם גֵּאֶה בְךָ! 🏆

📜 ״עַמָּא פַּזִּיזָא״ - מָה אָמַר הַצָּדוֹקִי?

🕍 רַבָּא הַגָּדוֹל יָשַׁב וְלָמַד תּוֹרָה בְּכָל כֹּחוֹ! הוּא הָיָה כָּל כָּךְ מִתְרַגֵּשׁ מֵהַלִּמּוּד! 📚✨ הִגִּיעַ צָדוֹקִי (אָדָם שֶׁלֹּא הֶאֱמִין בְּדֶרֶךְ חֲכָמִים) וְרָאָה אוֹתוֹ. הַצָּדוֹקִי לָעַג: ״אַתֶּם עַם פַּזִּיז!״ (עַם נִמְהָר וְלֹא זָהִיר) 😠 ״לָמָּה?״ שָׁאַל רַבָּא. הַצָּדוֹקִי אָמַר: ״הִקְדַּמְתֶּם פֶּה לְאֹזֶן!״ אֲמַרְתֶּם ״נַעֲשֶׂה״ לִפְנֵי שֶׁשְּׁמַעְתֶּם מָה צָרִיךְ לַעֲשׂוֹת! 🤔 ״הַיִּיתֶם צְרִיכִים קוֹדֶם לִבְדּוֹק אִם אַתֶּם מְסוּגָּלִים, וְרַק אָז לְהַסְכִּים!״ אֲבָל רַבָּא יָדַע סוֹד גָּדוֹל שֶׁהַצָּדוֹקִי לֹא הֵבִין... 💎

מַהִי הַטַּעֲנָה שֶׁל הַצָּדוֹקִי נֶגֶד עַם יִשְׂרָאֵל?

מְצֻיָּן! הַצָּדוֹקִי חָשַׁב שֶׁזּוֹ ״פַּזִּיזוּת״, אֲבָל בֶּאֱמֶת זוֹ הָיְתָה אַהֲבָה וּבִטָּחוֹן! 💖

מַדּוּעַ הַצָּדוֹקִי קָרָא לְעַם יִשְׂרָאֵל ״עַם פַּזִּיז״?

נָכוֹן! הַצָּדוֹקִי לֹא הֵבִין שֶׁכְּשֶׁיֵּשׁ אֱמוּן וְאַהֲבָה - אֵין צֹרֶךְ לִבְדּוֹק! הַשֵּׁם שָׂמֵחַ בְּךָ! ✨

מָה רַבָּא הָיָה עוֹשֶׂה כְּשֶׁהִגִּיעַ הַצָּדוֹקִי?

כָּל הַכָּבוֹד! רַבָּא הָיָה מְלֻמָּד גָּדוֹל שֶׁאָהַב אֶת הַתּוֹרָה מְאֹד! הַשֵּׁם אוֹהֵב אוֹתְךָ! 💪

💰 מָשָׁל הָאָב הֶעָשִׁיר וְהַבֵּן הַחַשְׁדָּן

🏭 הָיָה פַּעַם אָב עָשִׁיר מְאֹד שֶׁהָיָה לוֹ מִפְעָל עֲנָק! 💎 הַבֵּן עָבַד אֶצֶל אַבָּא, קִבֵּל מַשְׂכֹּרֶת וְגָדַל אִתּוֹ. הָאָב הָיָה תָּמִיד טוֹב אֵלָיו וְאָהַב אוֹתוֹ מְאֹד! 💖 כְּשֶׁהָאָב הִזְדַּקֵּן, הוּא רָצָה לָתֵת לַבֵּן אֶת כָּל הַמִּפְעָל בְּמַתָּנָה (צַוָּאָה). 🎁 הָאָב הֵגִישׁ לַבֵּן אֶת הַטְּפָסִים לַחֲתִימָה. אֲבָל הַבֵּן אָמַר: ״רֶגַע אַבָּא, אֲנִי לֹא חוֹתֵם! אֲנִי רוֹצֶה לְהָבִיא עוֹרֵךְ דִּין שֶׁיִּבְדֹּק...״ 😢 הָאָב נִפְגַּע עָמֹק: ״כָּל חַיַּי עָבַדְתִּי בִּשְׁבִילְךָ, נָתַתִּי לְךָ הַכֹּל, אֲנִי אוֹהֵב אוֹתְךָ - וְאַתָּה חוֹשֵׁד שֶׁאֲנִי רוֹצֶה לְהָרַע לְךָ ?!״ 💔

מָה רָצָה הָאָב לָתֵת לַבֵּן?

נָכוֹן מְאֹד! הָאָב רָצָה לָתֵת הַכֹּל מֵאַהֲבָה! כָּךְ גַּם הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לָנוּ אֶת הַתּוֹרָה! 💖

מַדּוּעַ הָאָב נִפְגַּע מֵהַבֵּן?

מְצֻיָּן! כְּשֶׁמִּישֶׁהוּ חוֹשֵׁד בָּנוּ אַחֲרֵי שֶׁאָהַבְנוּ אוֹתוֹ - זֶה כּוֹאֵב מְאֹד! הַשֵּׁם גֵּאֶה בְּהַבָנָה שֶׁלְּךָ! ✨

מָה הַבֵּן הָיָה צָרִיךְ לַעֲשׂוֹת?

נִפְלָא! כְּשֶׁיֵּשׁ אֱמוּן וְאַהֲבָה - חוֹתְמִים מִיָּד! כָּךְ עַם יִשְׂרָאֵל אָמְרוּ ״נַעֲשֶׂה וְנִשְׁמָע״! 🏆

💎 הַסּוֹד הַגָּדוֹל - תְּשׁוּבַת רַבָּא

🌟 עַכְשָׁו רַבָּא עָנָה לַצָּדוֹקִי: ״אֲנַן דִּסְגִינַן בִּשְׁלִימוּתָא!״ זֶה אוֹמֵר: ״אֲנַחְנוּ הוֹלְכִים עִם הַשֵּׁם בִּתְמִימוּת וּבְשַׁלְמוּת!״ 💖 רַשִׁ״י מְסַבֵּר: הִתְהַלַּכְנוּ עִמּוֹ כְּמוֹ אֲנָשִׁים שֶׁעוֹשִׂים מִתּוֹךְ אַהֲבָה וְסָמַכְנוּ עָלָיו שֶׁלֹּא יְבַקֵּשׁ מֵאִתָּנוּ מַשֶּׁהוּ רַע! 🌈 אֲנַחְנוּ מַכִּירִים אֶת אַבָּא (הַקָּדוֹשׁ בָּרוּךְ הוּא)! הוּא ״חִכּוֹ מַמְתַקִּים וְכֻלּוֹ מַחֲמַדִּים״ - הוּא רַק רוֹצֶה לְהֵיטִיב לָנוּ! ✨ לָכֵן חָתַמְנוּ מִיָּד (״נַעֲשֶׂה וְנִשְׁמָע״) בְּלִי לִבְדּוֹק, כִּי יָדַעְנוּ שֶׁזּוֹ הַמַּתָּנָה הַכִּי טוֹבָה שֶׁיֵּשׁ ! 🎁 זוֹ לֹא ״פַּזִּיזוּת״ - זוֹ חָכְמָה שֶׁל אַהֲבָה וּבִטָּחוֹן! 💪

מָה פֵּרוּשׁ ״אֲנַן דִּסְגִינַן בִּשְׁלִימוּתָא״?

כָּל הַכָּבוֹד! לָלֶכֶת בִּתְמִימוּת זֶה לִסְמוֹךְ עַל הַשֵּׁם מֵאַהֲבָה! הַשֵּׁם שָׂמֵחַ בְּךָ! 💖

עַל מָה סָמְכוּ עַם יִשְׂרָאֵל כְּשֶׁאָמְרוּ ״נַעֲשֶׂה וְנִשְׁמָע״?

מְצֻיָּן! הֵם בָּטְחוּ שֶׁהַשֵּׁם אוֹהֵב אוֹתָם כְּמוֹ אָב! זוֹ הָאֱמוּנָה הַכִּי חֲזָקָה! ✨

מַדּוּעַ ״נַעֲשֶׂה וְנִשְׁמָע״ זוֹ לֹא פַּזִּיזוּת אֶלָּא חָכְמָה?

נִפְלָא! זוֹ הַחָכְמָה הָעֲמֻקָּה בִּיוֹתֵר - לִסְמוֹךְ מֵאַהֲבָה! הַשֵּׁם גֵּאֶה בְךָ מְאֹד! 🏆

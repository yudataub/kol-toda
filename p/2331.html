🔹 שָׁלָב 1: בֵּרוּר מַקְדִּים וְנִתּוּחַ (לַמּוֹרֶה)

הַנּוֹשֵׂא: בית המקדש השני – לא רק בניין של אבנים, אלא מרכז פועם של אהבת ה', ניסים גלויים ואחדות ישראל.

הַמְּקוֹרוֹת בַּבָּסִיס: משנה אבות פ"ה מ"ז (עשרת הניסים), ותלמוד ירושלמי (פאה) / שיר השירים רבה (ז) על הניסים של "שומר ישראל" בזמן העלייה לרגל.

הַקּוֹנְפְלִיקְט (הַמֶּתַח הַסִּפּוּרִי):

  • הַפַּחַד הַטִּבְעִי לַעֲזֹב בַּיִת רֵיק מֵאֲחוֹר כְּשֶׁיֵּשׁ אוֹיְבִים מִסָּבִיב.
  • הַצְּפִיפוּת הַגְּדוֹלָה בָּעֲזָרָה שֶׁנִּרְאֵית כְּבִלְתִּי אֶפְשָׁרִית לַעֲמִידָה, וְכָל שֶׁכֵּן לְהִשְׁתַּחֲוָיָה.

נְקֻדַּת הַשִּׂיא: גִּלּוּי הַנֵּס בְּבֵית הַמִּקְדָּשׁ כְּשֶׁלְּכֻלָּם נִהְיֶה מָקוֹם, וְהַחֲזָרָה הַבַּיְתָה אֶל הַשּׁוֹמֵר הַפִּלְאִי בַּדֶּלֶת.

🔹 שָׁלָב 2: הַסִּפּוּר הַמָּלֵא – "הַבַּיִת שֶׁל כֻּלָּם וְהַשּׁוֹמֵר הַפִּלְאִי"

פְּתִיחָה: הַדֶּלֶת הַפְּתוּחָה

יְלָדִים, נַסּוּ לְדַמְיֵן – הַאִם הֱיִיתֶם טָסִים לְטִיּוּל שֶׁל שְׁבוּעַיִם, וּמַשְׁאִירִים אֶת הַדֶּלֶת שֶׁל הַבַּיִת פְּתוּחָה לְרְוָחָה? בְּלִי מַנְעוּל? בְּלִי אַזְעָקָה? בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ הָיָה קַיָּם, הָיְתָה מִצְוָה מְיֻחֶדֶת מְאֹד בַּחַגִּים: "עֲלִיָּה לָרֶגֶל". מֵאוֹת אַלְפֵי יְהוּדִים עָזְבוּ אֶת הַבָּתִּים שֶׁלָּהם בַּגָּלִיל וּבַדָּרוֹם, וְעָלוּ לִירוּשָׁלַיִם.

בִּכְפָר קָטָן אֶחָד, גָּר יְהוּדִי טוֹב וּתְמִים-לֵב. הוּא הֶעֱמִיס אֶת הַחֲמוֹר שֶׁלּוֹ בְּפֵרוֹת לְבִכּוּרִים, אָרַז צֵידָה לַדֶּרֶךְ וְקָרָא לִבְנֵי מִשְׁפַּחְתּוֹ עִם חִיּוּךְ גָּדוֹל: "קָדִימָה! יוֹצְאִים לְבֵית הַמִּקְדָּשׁ!". לְפֶתַע, קָפַץ אֶחָד הַשְּׁכֵנִים (שֶׁלֹּא הָיָה יְהוּדִי), צָחַק מִמֶּנּוּ וְאָמַר: "אַתָּה מְשֻׁגָּע! אַתָּה יוֹצֵא לִירוּשָׁלַיִם וּמַשְׁאִיר פֹּה אֶת הַתַּרְנְגֹלוֹת שֶׁלְּךָ? אֶת הַזָּהָב שֶׁלְּךָ? אֶת תְּבוּאַת הַשָּׁדֶה? אֲנִי בְּעַצְמִי אֶגְנֹב לְךָ הַכֹּל בָּרֶגַע שֶׁתֵּלֵךְ!". אֲבָל הַיְּהוּדִי הִבִּיט לַשָּׁמַיִם, לֹא פָּחַד כְּלָל, וְאָמַר: "אֲנִי מְקַקֵּם אֶת הַמִּצְוָה שֶׁל ה' יִתְבָּרַךְ. ה' בְּעַצְמוֹ הִבְטִיחַ בַּתּוֹרָה: 'וְלֹא יַחְמֹד אִישׁ אֶת אַרְצְךָ בַּעֲלֹתְךָ לֵרָאוֹת אֶת פְּנֵי ה''. יֵשׁ לִי אֶת הַשּׁוֹמֵר הֲכִי טוֹב בָּעוֹלָם!". וְכָך, הוּא הָלַךְ לִירוּשָׁלַיִם בְּלִי בֵּל שָׂמֵחַ.

הָעֲזָרָה הַצְּפוּפָה וְעֶשֶׂר הַנִּסִּים

כְּשֶׁהִגִּיעוּ הַיְהוּדִים לִירוּשָׁלַיִם, הֵם נִכְנְסוּ לְבֵית הַמִּקְדָּשׁ. יְלָדִים, אֵיזֶה הָדָר! אֵיזוֹ קְדֻשָּׁה! הֵם יָדְעוּ שֶׁבְּבֵית הַמִּקְדָּשׁ יֵשׁ עֲשָׂרָה נִסִּים גְּלוּיִים שֶׁקּוֹרִים בְּכָל יוֹם, וְהֵם מַרְאִים כַּמָּה ה' אוֹהֵב אֶת יִשְׂרָאֵל: הֵם הִבִּיטוּ לַמִּזְבֵּחַ, הָיְתָה שָׁם רוּחַ סְעָרָה חֲזָקָה שֶׁנָּשְׁבָה בֶּהָרִים, אֲבָל – נֵס! – הֶעָשָׁן שֶׁל עֲצֵי הַמִּזְבֵּחַ עָלָה יָשָׁר לַשָּׁמַיִם כְּמוֹ עַמּוּד שֶׁל זְכוּכִית, וְהָרוּחַ בִּכְלָל לֹא הֵזִיזָה אוֹתוֹ! הֵם רָאוּ שֶׁשּׁוֹחֲטִים שָׁם אֲלָפִים שֶׁל כְּבָשִׂים לְקָרְבָּנוֹת. בְּדֶרֶךְ כְּלָל מָקוֹם כָּזֶה מָלֵא בִּזְבוּבִים, נָכוֹן? אֲבָל – נֵס! – לֹא נִרְאָה אֲפִלּוּ זְבוּב אֶחָד כָּל יְמֵי בֵּית הַמִּקְדָּשׁ! מָקוֹם נָקִי וְטָהוֹר לְגַמְרֵי. וְלַמְרוֹת שֶׁיָּרַד גֶּשֶׁם עַז פַּעַם אַחַת, גְּשָׁמִים לְעוֹלָם לֹא כִּבּוּ אֶת מְדוּרַת עֲצֵי הַמִּזְבֵּחַ הַבּוֹעֶרֶת!

אֲבָל הַנֵּס הֲכִי מְרַגֵּשׁ הָיָה הַנֵּס שֶׁל הָאֲנָשִׁים הֶחָמִים שֶׁבָּאוּ מִכָּל קְצוֹת הָאָרֶץ. הָיוּ שָׁם מאוֹת אַלְפֵי אֲנָשִׁים בְּתוֹךְ "עֶזְרַת יִשְׂרָאֵל". מָקוֹם לֹא גָּדוֹל. הַצְּפִיפוּת הָיְתָה כָּל כָּךְ גְּדוֹלָה, עַד שֶׁהַיְהוּדִים נִדְחֲפוּ זֶה בָּזֶה, מַמָּשׁ "כָּתֵף צְמוּדָה לְכָתֵף". כְּשֶׁהֵם עָמְדוּ, הֵם הָיוּ צְפוּפִים כְּמוֹ קֻפְסַת גַּפְרוּרִים. אִי אֶפְשָׁר הָיָה לְהָזִיז יָד! וְאָז, הִגִּיעַ הָרֶגַע שֶׁבּוֹ הַכֹּהֵן הַגָּדוֹל אָמַר אֶת הַשֵּׁם הַמְפֹרָשׁ. מָה עָשׂוּ כֻּלָּם? רָצוּ לִפֹּל עַל פְּנֵיהֶם עַל הָרִצְפָּה, לְהִשְׁתַּחֲווֹת. אֲבָל אֵיךְ אֶפְשָׁר?! מֵאוֹת אַלְפֵי אֲנָשִׁים יִפְּלוּ אֶחָד עַל הַשֵּׁנִי וְיִמְחֲצוּ זֶה אֶת זֶה!

בָּרֶגַע שֶׁכֻּלָּם הִתְכּוֹפְפוּ כְּדֵי לְהִשְׁתַּחֲווֹת – נֵס עָצוּם! – פִּתְאוֹם בֵּית הַמִּקְדָּשׁ כְּאִלּוּ נִמְתַּח וְהִתְרַחֵב! כָּל אֶחָד שֶׁהִשְׁתַּחֲוָה גִּלָּה לְפֶתַע שֶׁיֵּשׁ מִסְּבִיבוֹ מֶרְחָב גָּדוֹל (כְּאַרְבַּע אַמּוֹת), אַף אֶחָד לֹא נָגַע בּוֹ, אַף אֶחָד לֹא הִפְרִיעַ לוֹ וְאַף אֶחָד לֹא שָׁמַע אֶת הַתְּפִלָּה הַסּוֹדִית וְהַשְּׁקֵטָה שֶׁלּוֹ! כְּשֶׁיְּהוּדִים מִתְבַּטְּלִים כְּלַפֵּי ה', לְלֹא גַּאֲוָה, פִּתְאוֹם יֵשׁ מָקוֹם לְכֻלָּם! וְאַף מִתְפַּלֵּל מֵעוֹלָם לֹא אָמַר שֶׁאֵין מָקוֹם לִישׁוֹן בִּירוּשָׁלַיִם.

הַשּׁוֹמֵר בַּדֶּלֶת

הַחַג נִגְמַר. הַיְהוּדִי הַתָּמִים חָזַר לַכְּפָר שֶׁלּוֹ, רָכוּב עַל הַחֲמוֹר, מְלֵא הִתְרַגְּשׁוּת וּקְדֻשָּׁה מִבֵּית הַמִּקְדָּשׁ. אֲבָל... לְפֶתַע לִבּוֹ דָּפַק. מָה הוּא יִמְצָא בַּבַּיִת הַפָּתוּחַ? הַאִם ה\שָּׁכֵן הָרָשָׁע רוֹקֵן לוֹ אֶת הַבַּיִת וְגָנב הַכֹּל? הוּא מִתְקָרֵב לַשַּׁעַר וְנֶעֱצָר מֻפְתָּע מִמָּה שֶׁרָאוּ עֵינָיו.

עַל מַנְעוּל דַּלְתּוֹ, הָיָה כָּרוּךְ לְסִיבּוּבִים – נָחָשׁ אֵימְתָנִי וְעָצוּם! מֵרִים אֶת רֹאשׁוֹ וּמוֹצִיא אֶת לְשׁוֹנוֹ קָדִימָה בִּלְחִישָׁה: חְסְסְס... מַמָּשׁ שׁוֹמֵר עַל הַדֶּלֶת! עַל עֲרֵמַת הַחִטִּים שֶׁלּוֹ יָשַׁב אֲרִי עָצוּם שֶׁשָּׁאַג! כָּל מִי שֶׁנִּסָּה לְהִתְקָרֵב, בָּרַח כְּשֶׁנַּפְשׁוֹ תּוֹבַעת עַל יָדָיו. הַקָּדוֹשׁ בָּרוּךְ הוּא גִּיֵּס אֶת שׁוֹמְרֵי הַטֶּבַע שֶׁלּוֹ כְּדֵי שֶׁיִּשְׁמְרוּ עַל הַיְהוּדִי שֶׁשָּׁמַר אֶת הַמִּצְווֹת. מִיָּד כְּשֶׁבַּעַל הַבַּיִת הִגִּיעַ, הַנָּחָשׁ גָּלַשׁ אֶל הַשָּׂדֶה וְהָאַרְיֵה חָזַר לַיַּעַר. הַיְהוּדִי נִכְנַס לְבֵיתוֹ, מָצָא אֶת כָּל רְכוּשׁוֹ בִּשְׁלֵמוּת, נָשָׂא עֵינָיו לַשָּׁמַיִם וְאָמַר לְבוֹרֵא עוֹלָם: "תּוֹדָה רַבָּה לְךָ, הַשּׁוֹמֵר הַנֶּאֱמָן שֶׁל עַם יִשְׂרָאֵל!".

🔹 שָׁלָב 3: חִבּוּר עֶרְכִּי – ME-WE-YOU

ME (הַיְהוּדִי בַּסִּפּוּר / עוֹלֵי הָרֶגֶל): הֵם הָיוּ מוּכָנִים לַעֲזֹב הַכֹּל, לִמְסֹר נֶפֶשׁ, וְגַם לִמְעֹךְ קְצָת אֶת הָאֱגוֹ וְהַגַּאֲוָה שֶׁלָּהֶם כְּדֵי לַעֲמֹד צְפוּפִים עִם עוֹד מֵאוֹת אֲלָפִים בְּבֵית הַמִּקְדָּשׁ. וּבִזְכוּת זֶה הֵם קִבְּלוּ שְׁמִירָה נִסִּית וּמֶרְחָב נִפְלָא.

WE (אֲנַחְנוּ): גַּם אֲנַחְנוּ לִפְעָמִים מְפַחֲדִים שֶׁאִם נַעֲשֶׂה מַשֶּׁהוּ טוֹב – נַפְסִיד מַשֶּׁהוּ אַחֵר (נַפְסִיד זְמַן, כֶּסֶף, אֶת תּוֹרֵנוּ בַּמִּשְׂחָק). לִפְעָמִים גַּם מַרְגִּישׁ לָנוּ "צָפוּף" כְּשֶׁמִּישֶׁהוּ אַחֵר מִשְׁתַּמֵּשׁ בַּחֲפָצִים שֶׁלָּנוּ אוֹ נִמְצָא בַּסְּבִיבָה שֶׁלָּנוּ.

YOU (אַתָּה): אַתָּה צָרִיךְ לִזְכֹּר: כְּשֶׁאַתָּה בְּאֶמְצַע לַעֲשׂוֹת מִצְוָה, ה' "שׁוֹמֵר עָלֶיךָ עִם אַרְיֵה וְנָחָשׁ!" – הוּא לֹא יִתֵּן שֶׁתַּתְפְסִיד בִּגְלל מִצְוָה. וּכְשֶׁאַתָּה מִשְׁתַּחֲוֶה בַּלֵּב (מוְותֵּר עַל גַּאֲוָה, אוֹהֵב אֶת יִשְׂרָאֵל), פִּתְאוֹם תִּרְאֶה שֶׁהַלֵּב וְהַחֶדֶר מִתְרַחֲבִים, וְיֵשׁ מָקוֹם לְכָל הַחֲבֵרִים בְּשָׁלוֹם וּבְאַהֲבָה!

🔹 שָׁלָב 4: מַדְרִיךְ לַמּוֹרֶה / לַמְסַפֵּר וְלַמֵּכִין (מַצֶּגֶת)

א. שְׁאֵלַת פְּתִיחָה (לְעוֹרֵר עִנְיָן): "יְלָדִים, אִם כָּל מִשְׁפַּחְתְּכֶם יוֹצֵאת לְחֻפְשָׁה שֶׁל שְׁבוּעַיִם, וְאַבָּא פִּתְאוֹם אוֹמֵר: 'עִזְבוּ אֶת הַדֶּלֶת פְּתוּחָה, אַל תִּנְעֲלוּ!' – מָה הֱיִיתֶם אוֹמְרִים לוֹ? בַּשִּׁעוּר הַיּוֹם נַחֲזֹר לִתְקוּפָה שֶׁכֻּלָּם עָשׂוּ אֶת זֶה בְּשִׂמְחָה!"

ב. הֶעָרוֹת בִּצּוּעַ (לַמְסַפֵּר/לַמַּצֶּגֶת):

  • הַנָּחָשׁ וְהָאַרְיֵה: לְהַמְחִישׁ קוֹלִית ("חְסְסְסְס... שְׁאָגָה!"). (לְמַצֶּגֶת: צִיּוּר שֶׁל נָחָשׁ שׁוֹמֵר בַּדֶּלֶת).
  • הַצְּפִיפוּת ("עוֹמְדִים צְפוּפִים"): לְהִדָּחֵק עִם הַגּוּף פִיזִית כְּדֵי שֶׁיָּבִינוּ שֶׁאִי אֶפְשָשר לִנְשֹׁם מֵרֹב צְפִיפוּת, וְאָז בַּהִשְׁתַּחֲוָיָה לִפְרֹשׂ אֶת הַיָּדัיִם רָחָב וְלוֹמַר בְּפֶלֶא: "פִּתְאוֹם... יֵשׁ מָקוֹם לְכֻלָּם!".
  • עֲשַׁן הַמִּזְבֵּחַ וְהַזְּבוּבִים: בַּמַּצֶּגֶת: מִזְבֵּחַ שֶׁל אוֹר שֶׁהֶעָשָׁן שֶׁלּוֹ עָלָה וְיָשָׁר לַשָּׁמַיִם לַמְרוֹת עֲנָנִים וְרוּחַ טֶבַע בָּרְקַע.

פְּעִילוּת מְסַכֶּמֶת ("נֵס מָקוֹם" מַעֲשִׂי)

מִשְׂחַק עִתּוֹן אוֹ חִישׁוּק: מַנִּיחִים חִישׁוּק אֶחָד (אוֹ דַּף עִתּוֹן עֲנָק) עַל רִצְפַּת הַכִּתָּה. מְבַקְשִׁים מִ-4-5 יְלָדִים לַעֲמֹד בְּתוֹכוֹ בְּלִי לִגַּעַט בָּרִצְפָּה בַּחוּץ. הֵם נוֹגְעִים זֶה בָּזֶה. מַסְבִּירִים שֶׁכְּדֵי שֶׁכֻּלָּם יִשָּׁאֲרוּ בַּחִישׁוּק הֵם חַיָּבִים לַחֲבֹק כְּתֵפַיִם וּלְוַתֵּר אֶחָד לַשֵּׁנִי וּלְתָמְכוֹ זֶה בָּזֶה.

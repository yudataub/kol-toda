שִׁיעוּר 1: מֹשֶׁה רַבֵּנוּ וְהַסְּנֶה הַבּוֹעֵר 🔥

שָׁלוֹם חֲבֵרִים יְקָרִים! 💖 הַיּוֹם נִלְמַד סִפּוּר מַרְתֵּק עַל מֹשֶׁה רַבֵּנוּ. 🌟 יוֹם אֶחָד, מֹשֶׁה רַבֵּנוּ הָלַךְ לְטַיֵּל בַּמִּדְבָּר. הוּא רָעָה אֶת הַצֹּאן וְהָלַךְ רָחוֹק רָחוֹק... 🐑 וּפִתְאֹם, הוּא רָאָה מַשֶּׁהוּ מְיֻחָד מְאֹד! 🤩 הוּא רָאָה סְנֶה בּוֹעֵר בָּאֵשׁ - אֲבָל הַסְּנֶה לֹא נִשְׂרַף! ✨ וַאֲוֹוֹוֹ! זֶה הָיָה כָּל כָּךְ מְיֻחָד! אֵיךְ זֶה יָכוֹל לִהְיוֹת? 😲 מֹשֶׁה נִגַּשׁ יוֹתֵר קָרוֹב לִרְאוֹת... וְשָׁם, מִתּוֹךְ הַסְּנֶה, בּוֹרֵא עוֹלָם קוֹרֵא לוֹ: "מֹשֶׁה! מֹשֶׁה!" 🙏 הַשֵּׁם אָמַר לוֹ: "שַׁל נְעָלֶיךָ מֵעַל רַגְלֶיךָ, כִּי הַמָּקוֹם אֲשֶׁר אַתָּה עוֹמֵד עָלָיו אַדְמַת קֹדֶשׁ הוּא!" 💫 מֹשֶׁה הֵבִין שֶׁזֶּה רֶגַע מְיֻחָד! הַשֵּׁם מְדַבֵּר אִתּוֹ! 💖

מָה מֹשֶׁה רַבֵּנוּ רָאָה בַּמִּדְבָּר שֶׁהָיָה כָּל כָּךְ מְיֻחָד?

נָכוֹן מְאֹד! 🔥 מֹשֶׁה רָאָה סְנֶה בּוֹעֵר בָּאֵשׁ, אֲבָל הַסְּנֶה לֹא נִשְׂרַף. זֶה הָיָה נֵס גָּדוֹל!

מִי קָרָא לְמֹשֶׁה מִתּוֹךְ הַסְּנֶה הַבּוֹעֵר?

כָּל הַכָּבוֹד! 💫 בּוֹרֵא עוֹלָם עַצְמוֹ קָרָא לְמֹשֶׁה. זֶה הָיָה רֶגַע קָדוֹשׁ וּמְיֻחָד מְאֹד!

מָה הַשֵּׁם אָמַר לְמֹשֶׁה לַעֲשׂוֹת כְּשֶׁהוּא מִתְקָרֵב לַסְּנֶה?

יָפֶה מְאֹד! 👞 הַשֵּׁם אָמַר לוֹ לְהוֹרִיד אֶת הַנְּעָלַיִם, כִּי הַמָּקוֹם הַזֶּה קָדוֹשׁ. הַשֵּׁם שָׂמֵחַ כְּשֶׁאָנוּ מְכַבְּדִים אֶת הַקְּדֻשָּׁה!

שִׁיעוּר 2: כִּי יָדַעְתִּי אֶת מַכְאוֹבָיו - אֶמְפַּתְיָה אֱלֹהִית 💙

עַכְשָׁיו נִלְמַד מַשֶּׁהוּ מְיֻחָד מְאֹד! 🌟 אַתֶּם יוֹדְעִים מָה זֶה אֶמְפַּתְיָה? 🤔 אֶמְפַּתְיָה זֶה כְּשֶׁאַתָּה מַרְגִּישׁ אֶת מָה שֶׁהַשֵּׁנִי מַרְגִּישׁ! 💖 לְמָשָׁל: כְּשֶׁמִּישֶׁהוּ מְחַיֵּךְ - גַּם אַתָּה פִּתְאֹם מְחַיֵּךְ! 😊 כְּשֶׁמִּישֶׁהוּ עָצוּב - גַּם אַתָּה מַרְגִּישׁ עָצוּב. זֶה עוֹבֵר מֵאָדָם לְאָדָם! ✨ וְעַכְשָׁיו, בּוֹאוּ נִשְׁמַע מַה הַשֵּׁם אָמַר לְמֹשֶׁה: 🙏 "רָאֹה רָאִיתִי אֶת עֳנִי עַמִּי אֲשֶׁר בְּמִצְרַיִם, וְאֶת צַעֲקָתָם שָׁמַעְתִּי מִפְּנֵי נֹגְשָׂיו, כִּי יָדַעְתִּי אֶת מַכְאֹבָיו" 💙 שִׂימוּ לֵב לַמִּלִּים: רָאִיתִי... שָׁמַעְתִּי... וְגַם יָדַעְתִּי! 🌟 מָה זֶה "יָדַעְתִּי אֶת מַכְאֹבָיו"? 🤔 זֶה שֶׁהַשֵּׁם מַרְגִּישׁ אֶת הַכְּאֵב שֶׁלָּנוּ! הוּא לֹא רַק רוֹאֶה וְשׁוֹמֵעַ - הוּא גַּם מַרְגִּישׁ! 💖 כְּמוֹ אִמָּא שֶׁמַּרְגִּישָׁה מָה כּוֹאֵב לַיֶּלֶד, גַּם בְּלִי שֶׁהוּא אוֹמֵר - כָּךְ הַשֵּׁם מַרְגִּישׁ אֶת הַכְּאֵב שֶׁלָּנוּ! 🙏 זֶה אוֹמֵר שֶׁאָנוּ אַף פַּעַם לֹא לְבַד! הַשֵּׁם תָּמִיד מַרְגִּישׁ אוֹתָנוּ! ✨

מָה זֶה אֶמְפַּתְיָה?

מְצֻיָּן! ⭐ אֶמְפַּתְיָה זֶה לְהַרְגִּישׁ מָה שֶׁהַשֵּׁנִי מַרְגִּישׁ. זֹאת תְּכוּנָה נִפְלָאָה שֶׁהַשֵּׁם נָתַן לָנוּ!

מָה הַשֵּׁם אָמַר לְמֹשֶׁה עַל עַם יִשְׂרָאֵל בְּמִצְרַיִם?

נִפְלָא! 💙 הַשֵּׁם אָמַר 'יָדַעְתִּי אֶת מַכְאוֹבָיו' - הוּא מַרְגִּישׁ אֶת כְּאֵבָם. הַשֵּׁם אוֹהֵב אוֹתָנוּ וּמַרְגִּישׁ אוֹתָנוּ תָּמִיד!

לְמָה זֶה דּוֹמֶה שֶׁהַשֵּׁם מַרְגִּישׁ אֶת הַכְּאֵב שֶׁלָּנוּ?

יָפֶה מְאֹד! 💖 כְּמוֹ אִמָּא שֶׁמַּרְגִּישָׁה מָה כּוֹאֵב לַיֶּלֶד - כָּךְ הַשֵּׁם מַרְגִּישׁ אוֹתָנוּ. אָנוּ אַף פַּעַם לֹא לְבַד! הַשֵּׁם תָּמִיד אִתָּנוּ!

שִׁיעוּר 3: מֹשֶׁה מְפַחֵד - מִי אָנֹכִי? 😰

עַכְשָׁיו הַשֵּׁם אוֹמֵר לְמֹשֶׁה מַשֶּׁהוּ מְאֹד חָשׁוּב: 🎯 "לֵךְ וְהוֹצֵא אֶת עַם יִשְׂרָאֵל מִמִּצְרַיִם!" 🙏 וּמֹשֶׁה רַבֵּנוּ? הוּא מְפַחֵד! 😰 הוּא אוֹמֵר: "מִי אָנֹכִי כִּי אֵלֵךְ אֶל פַּרְעֹה?" 🤔 מֹשֶׁה חוֹשֵׁב: אֲנִי? לָלֶכֶת אֶל פַּרְעֹה הַמֶּלֶךְ הַחָזָק? הוּא כָּל כָּךְ מְפַחֵד! 👑 פַּרְעֹה הָיָה הַמֶּלֶךְ הַכִּי חָזָק בָּעוֹלָם! מֹשֶׁה חוֹשֵׁב: "אוּלַי הוּא יַהֲרֹג אוֹתִי..." 😨 וְעוֹד בְּעָיָה! מֹשֶׁה שׁוֹאֵל: "וְכִי אוֹצִיא אֶת בְּנֵי יִשְׂרָאֵל מִמִּצְרַיִם?" 🤷 מֹשֶׁה חוֹשֵׁב: "אֵיךְ עַם יִשְׂרָאֵל יַאֲמִינוּ לִי? אֲנִי רַק אָדָם אֶחָד!" 😟 לִפְעָמִים גַּם אָנוּ מַרְגִּישִׁים קְטַנִּים, נָכוֹן? אֲבָל תִּרְאוּ מַה הַשֵּׁם עוֹשֶׂה! 💪 הַשֵּׁם לֹא זוֹרֵק אֶת מֹשֶׁה - הוּא מְחַזֵּק אוֹתוֹ! זֶה כָּל כָּךְ יָפֶה! 💖

מָה הַשֵּׁם בִּקֵּשׁ מִמֶּשֶׁה לַעֲשׂוֹת?

נָכוֹן! 🎯 הַשֵּׁם בִּקֵּשׁ מִמֶּשֶׁה לְהוֹצִיא אֶת עַם יִשְׂרָאֵל מִמִּצְרַיִם. זֹאת הָיְתָה מִשִּׁימָה גְּדוֹלָה!

לָמָּה מֹשֶׁה פָּחַד מִפַּרְעֹה?

מְצֻיָּן! 💪 מֹשֶׁה פָּחַד כִּי פַּרְעֹה הָיָה הַמֶּלֶךְ הַחָזָק בְּיוֹתֵר. אֲבָל הַשֵּׁם חָזָק יוֹתֵר!

מָה עוֹד דָּאַג מֹשֶׁה?

נִפְלָא! 🌟 מֹשֶׁה דָּאַג שֶׁעַם יִשְׂרָאֵל לֹא יַאֲמִינוּ לוֹ. זֹאת הָיְתָה בְּעָיָה גְּדוֹלָה. אֲבָל הַשֵּׁם יֵשׁ לוֹ פִּתְרוֹן!

שִׁיעוּר 4: הַהַבְטָחָה - כִּי אֶהְיֶה עִמָּךְ! 💫

וְעַכְשָׁיו בָּא הַחֵלֶק הַכִּי יָפֶה וּמְחַזֵּק! 🌟 הַשֵּׁם עוֹנֶה לְמֹשֶׁה וְאוֹמֵר לוֹ: 🙏 "כִּי אֶהְיֶה עִמָּךְ!" 💙 מָה זֶה אוֹמֵר? אֵין לְךָ מָה לִפְחֹד מִפַּרְעֹה! אֲנִי אֶהְיֶה אִתְּךָ! 💪 הַשֵּׁם יִשְׁמֹר עָלֶיךָ, יַעֲזֹר לְךָ, יִהְיֶה לְיָדְךָ! ✨ וּמָה עִם הַבְּעָיָה הַשְּׁנִיָּה? אֵיךְ עַם יִשְׂרָאֵל יַאֲמִינוּ? 🤔 הַשֵּׁם אוֹמֵר: "וְזֶה לְּךָ הָאוֹת..." 🎯 מָה יִהְיֶה הָאוֹת? שֶׁאַחֲרֵי שֶׁתּוֹצִיא אֶת עַם יִשְׂרָאֵל מִמִּצְרַיִם... 🌈 "תַּעַבְדוּן אֶת הָאֱלֹהִים עַל הָהָר הַזֶּה!" ⛰️ מָה זֶה? זֶה הַר סִינַי! שָׁם תְּקַבְּלוּ אֶת הַתּוֹרָה! 📖 כָּל עַם יִשְׂרָאֵל יִרְאוּ אֶת הַשֵּׁם מִתְגַּלֶּה, יִשְׁמְעוּ אֶת קוֹלוֹ - וְאָז כֻּלָּם יֵדְעוּ שֶׁמֶּשֶׁה הוּא הַנָּבִיא הָאֲמִתִּי! 🎺 רוֹאִים? הַשֵּׁם תָּמִיד דּוֹאֵג לָנוּ! 💖 גַּם כְּשֶׁאָנוּ מְפַחֲדִים, הַשֵּׁם אוֹמֵר: "אֲנִי אֶהְיֶה עִמָּךְ!" 🙏

מָה הַשֵּׁם הִבְטִיחַ לְמֹשֶׁה כְּשֶׁפָּחַד מִפַּרְעֹה?

כָּל הַכָּבוֹד! 💙 הַשֵּׁם אָמַר 'כִּי אֶהְיֶה עִמָּךְ' - אֲנִי אֶהְיֶה אִתְּךָ לְעוֹלָם! זֹאת הַבְטָחָה נִפְלָאָה!

מָה הָיָה הָאוֹת שֶׁהַשֵּׁם נָתַן שֶׁעַם יִשְׂרָאֵל יַאֲמִינוּ לְמֹשֶׁה?

מְצֻיָּן! ⛰️ הָאוֹת הָיָה שֶׁכֻּלָּם יְקַבְּלוּ אֶת הַתּוֹרָה בְּהַר סִינַי וְיִרְאוּ אֶת הַשֵּׁם מִתְגַּלֶּה!

מָה הַשִּׁיעוּר הַכִּי חָשׁוּב מֵהַסִּפּוּר הַזֶּה?

נִפְלָא! 🌟 הַשִּׁיעוּר הַכִּי חָשׁוּב: הַשֵּׁם תָּמִיד אִתָּנוּ! גַּם כְּשֶׁאָנוּ מְפַחֲדִים, הַשֵּׁם אוֹמֵר: 'אֲנִי אֶהְיֶה עִמָּךְ!' 💖

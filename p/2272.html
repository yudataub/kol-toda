אֶרֶץ יִשְׂרָאֵל - אֶרֶץ הַפֵּירוֹת 🍇🌳

🌟 שָׁלוֹם יְקָרִים! הַיּוֹם נִלְמַד עַל מַשֶּׁהוּ מְיֻחָד מְאוֹד! הַקָּדוֹשׁ בָּרוּךְ הוּא בָּרָא אֶת הָעוֹלָם עִם אֲרָצוֹת רַבּוֹת , אֲבָל אֶרֶץ יִשְׂרָאֵל הִיא מְיֻחֶדֶת בְּמִשֶּׁהוּ נִפְלָא! 💖 אֶרֶץ מִצְרַיִם מְיֻחֶדֶת בַּיְרָקוֹת - קִשּׁוּאִים, בְּצָלִים, שׁוּמִים וְאֲבַטִּיחִים 🥒 אֲבָל אֶרֶץ יִשְׂרָאֵל מְיֻחֶדֶת בַּפֵּירוֹת! 🍇🍎🍊 בַּתּוֹרָה כָּתוּב: ״אֶרֶץ חִטָּה וּשְׂעֹרָה וְגֶפֶן וּתְאֵנָה וְרִמּוֹן, אֶרֶץ זֵית שֶׁמֶן וּדְבָשׁ״ - אֵלּוּ שִׁבְעַת הַמִּינִים ! 🌾 גַּם בְּגַן עֵדֶן הָיוּ רַק עֵצִים וּפֵירוֹת ! הַקָּדוֹשׁ בָּרוּךְ הוּא אָמַר לאָדָם הָרִאשׁוֹן: ״מִכֹּל עֵץ הַגָּן אָכֹל תֹּאכֵל״ - זוֹ מִצְוָה לֶאֱכֹל פֵּירוֹת! 🌳

בַּמָּה מִשְׁתַּבַּחַת אֶרֶץ יִשְׂרָאֵל לְעוּמַּת מִצְרַיִם? 🤔

נָכוֹן! אֶרֶץ יִשְׂרָאֵל מִשְׁתַּבַּחַת בַּפֵּירוֹת, וּמִצְרַיִם בְּיַרְקוֹת. הַפֵּירוֹת הֵם יוֹתֵר רוּחָנִיִּים! 🍇

כַּמָּה מִינִים מְיֻחָדִים יֵשׁ שֶׁבָּהֶם מִשְׁתַּבַּחַת אֶרֶץ יִשְׂרָאֵל? 🌾

מְצֻיָּן! שִׁבְעַת הַמִּינִים הֵם: חִטָּה, שְׂעוֹרָה, גֶּפֶן, תְּאֵנָה, רִמּוֹן, זַיִת וּתְמָר! 🌟

מָה אָכְלוּ בְּגַן עֵדֶן לְפִי הַתּוֹרָה? 🌳

נָכוֹן מְאוֹד! הַקָּדוֹשׁ בָּרוּךְ הוּא צִוָּה: ״מִכֹּל עֵץ הַגָּן אָכֹל תֹּאכֵל״ - פֵּירוֹת! 🍎

מָה זָכְרוּ בְּנֵי יִשְׂרָאֵל מִמִּצְרַיִם בַּמִּדְבָּר? 🥒

יָפֶה! בְּנֵי יִשְׂרָאֵל זָכְרוּ אֶת הַיְרָקוֹת שֶׁל מִצְרַיִם - דָּגִים, קִשּׁוּאִים, אֲבַטִּיחִים, בְּצָלִים וְשׁוּמִים! 🧅

אֲכִילָה שֶׁמְּקָרֶבֶת לַשֵּׁם 🍎💫

🌟 הַבַּעַל שֵׁם טוֹב וְתַלְמִידָיו גִּלּוּ לָנוּ סוֹד נִפְלָא! לָמָּה הַקָּדוֹשׁ בָּרוּךְ הוּא בָּרָא אֶת הָאָדָם שֶׁצָּרִיךְ לֶאֱכֹל? 🤔 הַבַּת עַיִן הָיָה רַבִּי צַדִּיק שֶׁעָלָה לְאֶרֶץ יִשְׂרָאֵל לִפְנֵי כְּ-150 שָׁנָה! הוּא מְסַפֵּר שֶׁכְּשֶׁאָדָם אוֹכֵל פֵּירוֹת אֶרֶץ יִשְׂרָאֵל - הוּא יָכוֹל לְהִתְקָרֵב לַקָּדוֹשׁ בָּרוּךְ הוּא! 💖 הוּא מְסַפֵּר: שִׁבְעַת הַמִּינִים הֵם כְּנֶגֶד שֶׁבַע מִדּוֹת טוֹבוֹת - חֶסֶד, גְּבוּרָה, תִּפְאֶרֶת, נֶצַח, הוֹד, יְסוֹד וּמַלְכוּת! 👑 כְּשֶׁאָדָם אוֹכֵל פְּרִי יָפֶה וְרוֹאֶה כַּמָּה הוּא מְיֻחָד - הוּא מַרְגִּישׁ כַּמָּה הַשֵּׁם אוֹהֵב אוֹתוֹ ! 💕 לָכֵן בָּרָא הַקָּדוֹשׁ בָּרוּךְ הוּא כָּל כָּךְ הַרְבֵּה סוּגֵי פֵּירוֹת - שֶׁכָּל פַּעַם נִתְפָּעֵל מֵחָדָשׁ! 🎉

לָמָּה הַקָּדוֹשׁ בָּרוּךְ הוּא בָּרָא שֶׁנִּצְטָרֵךְ לֶאֱכֹל? 💭

נָכוֹן! דֶּרֶךְ הָאֲכִילָה אָנוּ מַרְגִּישִׁים כַּמָּה הַשֵּׁם אוֹהֵב אוֹתָנוּ וּבָרָא לָנוּ דְּבָרִים יָפִים! 💖

כְּנֶגֶד מָה הֵם שִׁבְעַת הַמִּינִים לְפִי הַבַּת עַיִן? 🌾

מְצֻיָּן! שִׁבְעַת הַמִּינִים הֵם כְּנֶגֶד שֶׁבַע מִדּוֹת: חֶסֶד, גְּבוּרָה, תִּפְאֶרֶת, נֶצַח, הוֹד, יְסוֹד וּמַלְכוּת! 👑

מִי הָיָה הַבַּת עַיִן? 📖

יָפֶה! הַבַּת עַיִן הָיָה רַב צַדִּיק שֶׁעָלָה לְצְפַת וְאָהַב מְאוֹד אֶת אֶרֶץ יִשְׂרָאֵל וּפֵירוֹתֶיהָ! 🌟

לָמָּה הַקָּדוֹשׁ בָּרוּךְ הוּא בָּרָא הַרְבֵּה סוּגֵי פֵּירוֹת? 🍇🍊🍎

נָכוֹן מְאוֹד! הַקָּדוֹשׁ בָּרוּךְ הוּא רוֹצֶה שֶׁכָּל פַּעַם נִתְרַגֵּשׁ מֵחָדָשׁ וְנַרְגִּישׁ אֶת אַהֲבָתוֹ! 💕

בִּרְכַּת הַמָּזוֹן - הוֹדָיָה מֵהַלֵּב 🙏💖

🌟 הַיּוֹם נִלְמַד סוֹד נִפְלָא עַל בִּרְכַּת הַמָּזוֹן! כָּתוּב בַּתּוֹרָה: ״וְאָכַלְתָּ וְשָׂבָעְתָּ וּבֵרַכְתָּ אֶת ה׳ אֱלֹקֶיךָ עַל הָאָרֶץ הַטּוֹבָה״ הַסִּידוּרוֹ שֶׁל שַׁבָּת שׁוֹאֵל: לָמָּה כָּתוּב ״עַל הָאָרֶץ הַטּוֹבָה״ ? הֲרֵי גַּם בְּחוּץ לָאָרֶץ מְבָרְכִים! 🤔 וְהוּא מְסַבֵּר דָּבָר מַדְהִים: בְּאֶרֶץ יִשְׂרָאֵל, הַפֵּירוֹת כָּל כָּךְ קְדוֹשִׁים - שֶׁכְּשֶׁאָדָם אוֹכֵל וְשָׂבֵעַ, מִיָּד מִתְחַשֵּׁק לוֹ מְאוֹד לְהוֹדוֹת לַהַשֵּׁם! 💖 לֹא בְּתוֹר חוֹבָה - אֶלָּא מִתּוֹךְ אַהֲבָה וְהִתְלַהֲבוּת ! הַתּוֹרָה אוֹמֶרֶת: בָּאָרֶץ הַטּוֹבָה הַזֹּאת, כְּשֶׁתֹּאכַל וְתִשְׂבַּע - יִתְחַשֵּׁק לְךָ מֵעַצְמְךָ לְבָרֵךְ אֶת הַשֵּׁם! 🎉 הָרַב מָרְדְּכַי אֵלִיָּהוּ זַצַ״ל לֹא הִסְכִּים לֶאֱכֹל כְּלוּם מֵחוּץ לָאָרֶץ - אֲפִילּוּ מַיִם לָקַח אִתּוֹ מֵעין גֶּדִי! 💧

מָה מְיֻחָד בְּאֶרֶץ יִשְׂרָאֵל לְגַבֵּי בִּרְכַּת הַמָּזוֹן? 🙏

מְצֻיָּן! בְּאֶרֶץ יִשְׂרָאֵל הַפֵּירוֹת כָּל כָּךְ קְדוֹשִׁים שֶׁמִּתְחַשֵּׁק לְךָ לְהוֹדוֹת לַהַשֵּׁם מֵאַהֲבָה! 💖

מָה עָשָׂה הָרַב מָרְדְּכַי אֵלִיָּהוּ כְּשֶׁנָּסַע לְחוּץ לָאָרֶץ? ✈️

נָכוֹן! הָרַב מָרְדְּכַי אֵלִיָּהוּ לָקַח אִתּוֹ אֲפִילּוּ מַיִם מֵעין גֶּדִי - כָּל כָּךְ אָהַב אֶת אֶרֶץ יִשְׂרָאֵל! 💧

מָה כָּתוּב בַּתּוֹרָה עַל בְּרָכָה אַחֲרֵי אֲכִילָה בָּאָרֶץ? 📖

יָפֶה מְאוֹד! זֶה הַפָּסוּק שֶׁמְּלַמֵּד אוֹתָנוּ עַל בִּרְכַּת הַמָּזוֹן בְּאֶרֶץ יִשְׂרָאֵל! 🌟

מָה הַהֶבְדֵּל בֵּין לְבָרֵךְ ״כְּחוֹבָה״ לְבֵין לְבָרֵךְ ״מֵאַהֲבָה״? 💕

נָכוֹן מְאוֹד! כְּשֶׁמְּבָרְכִים מֵאַהֲבָה - זֶה בָּא מֵהַלֵּב, בְּהִתְלַהֲבוּת, כִּי רוֹצִים לְהוֹדוֹת! 💖

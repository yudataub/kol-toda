הוֹדָיָה: שַׁעַר הַכְּנִיסָה אֶל ה'

תֹּכֶן הָעִנְיָנִים

  • הַקֶּשֶׁר הָרְבִיעִי: חֲנֻכָּה כְּחַג הַהוֹדָיָה
  • חִידַת הָרְבִיעִי: רְמָזִים מִפָּרָשַׁת וַיֵּשֶׁב
  • דֶּלֶת אֶל ה': מַשְׁמָעוּת הַהוֹדָיָה וְהַדְלָקַת הַנֵּרוֹת
  • מִבְחַן חִדּוּד
  • מְקוֹרוֹת

1. הַקֶּשֶׁר הָרְבִיעִי: חֲנֻכָּה כְּחַג הַהוֹדָיָה

לְחַג הַחֲנֻכָּה מַעֲמָד יִחוּדִי בְּלוּחַ הַשָּׁנָה הַיְּהוּדִי. לְצַד שְׁלֹשֶׁת הָרְגָלִים – פֶּסַח, שָׁבוּעוֹת וְסֻכּוֹת – הוּא נֶחְשָׁב לַ"חַג הָרְבִיעִי". אַךְ בְּנִגּוּד אֲלֵיהֶם, שֶׁמְּצֻוִּים מִן הַתּוֹרָה, חֲנֻכָּה הוּא מִדִּבְרֵי חֲכָמִים. מַהוּת זוֹ מַדְגִּישָׁה אֶת הָעֹמֶק הָרַעְיוֹנִי שֶׁלּוֹ. חֲזַ"ל קָבְעוּ אֶת מַהוּתוֹ הַמֶּרְכָּזִית שֶׁל הַחַג כְּ"יָמִים שֶׁל הוֹדָיָה". לֹא רֶגַע אֶחָד שֶׁל תּוֹדָה, אֶלָּא מַצָּב מִתְמַשֵּׁךְ, שְׁמוֹנָה יָמִים שֶׁכָּכל מַהוּתָם סוֹבֶבֶת סְבִיב פְּעֻלַּת הַהוֹדָיָה לַה'. הֲבָנַת מוּשָׂג זֶה כִּפְעֻלָּה אסטרטגית מִתְמַשֶּׁכֶת, וְלֹא כְּאֵרוּעַ חַד-פַּעֲמִי, הִיא הַמַּפְתֵּחַ לְהָפִיכַת הַהוֹדָיָה מֵאֵרוּעַ חַד-פַּעֲמִי לְמַצָּב תּוֹדָעָתִי קָבוּعַ, הַפּוֹתֵחַ שְׁעָרִים רוּחָנִיִּים.

הַמִּסְפָּר אַרְבַּע חוֹזֵר וּמֵהַדְהֵד בַּחַג בְּאֹפֶן מֻבְהָק: אָנוּ עוֹמְדִים בְּיוֹם רְבִיעִי לַשָּׁבוּעַ, מַדְלִיקִים נֵר רְבִיעִי, וְחוֹגְגִים אֶת הַחַג הָרְבִיעִי עַצְמוֹ. רִבּוּי זֶה שֶׁל הַמִּסְפָּר אַרְבַּע אֵינוֹ מִקְרִי, אֶלָּא מְכַוֵּן אוֹתָנוּ לְהִתְמַקֵּד בַּמַּהוּת הַמְּזֻקֶּקֶת שֶׁל הַחַג – מַהוּת הַהוֹדָיָה.

הִתְמַקְּדוּת זוֹ מוֹבִילָה אוֹתָנוּ לִשְׁאֵלָה מֶרְכָּזִית, הַנִּצֶּבֶת בִּבְסִיס הַמַּאֲמָר: מַדּוּעַ אָנוּ מַדְלִיקִים אֶת נֵרוֹת הַחֲנֻכָּה דַּוְקָא בַּדֶּלֶת? הַתְּשׁוּבָה, כְּפִי שֶׁנִּרְאֶה, טְמוּנָה בַּקֶּשֶׁר הָעָמֹק שֶׁבֵּין הַמִּצְוָה, הָאוֹת הָרְבִיעִית בָּאָלֶף-בֵּית – הָאוֹת ד' – וְרַעְיוֹן הַהוֹדָיָה עַצְמוֹ.

כְּדֵי לְפַעְנֵחַ קֶשֶׁר זֶה, נִפְנֶה אֶל פָּרָשַׁת הַשָּׁבוּעַ, פָּרָשַׁת "וַיֵּשֶׁב", שָׁם טְמוּנִים רְמָזִים עֲמֻקִּים הַמְּגַלִּים אֶת סוֹד הַהוֹדָיָה וְהַקֶּשֶׁר שֶׁלָּהּ לַמִּסְפָּר אַרְבַּע וּלְחַג הַחֲנֻכָּה.

2. חִידַת הָרְבִיעִי: רְמָזִים מִפָּרָשַׁת וַיֵּשֶׁב

כְּדֵי לַחְשֹׂף אֶת הַשְּׁכָבוֹת הַנִּסְתָּרוֹת הַקּוֹשְׁרוֹת אֶת הַמִּסְפָּר אַרְבַּע, יְהוּדָה, חֲנֻכָּה וְהוֹדָיָה, הֻצְּגָה חִידָה מֻרְכֶּבֶת וְרַב-מְמַדִּית. נִתּוּחַ חִידָה זוֹ אֵינוֹ רַק שַׁעֲשׁוּעַ שִׂכְלִי, אֶלָּא כְּלִי לְהַעֲמָקָה, הַחוֹשֵׂף כֵּיצַד אֵרוּעִים וּפְסוּקִים בְּפָרָשַׁת הַשָּׁבוּעַ מְהַדְהֲדִים אֶת מַהוּת הַהוֹדָיָה שֶׁאָנוּ מְצַיְּנִים בַּחֲנֻכָּה.

הַחִידָה: לַבֵּן הָרְבִיעִי נוֹלְדוּ בָּרְבִיעִי יְלָדִים, כְּנֶגֶד הַבְּרִיאָה שֶׁבָּרְבִיעִי. שֶׁעֲלֵיהֶם בָּרְבִיעִי אָנוּ עוֹשִׂים "רְבִיעִי", וְנִקְרָא עֲלֵיהֶם בַּתּוֹרָה בָּרְבִיעִי שֶׁתָּחוּל תָּמִיד סָמוּךְ לַחַג הָרְבִיעִי. וּבְסָמוּךְ לָרְבִיעִי עָשָׂה פְּעֻלָּה שֶׁל "רְבִיעִי", וּבָזֶה הִצִּיל אֶת הָרְבִיעִי. וּבְפָסוּק זֶה נִרְמַז הַחַג הָרְבִיעִי, עִם הַדִּין שֶׁל הָרְבִיעִי שֶׁחָל סָמוּךְ לָרְבִיעִי. וּבִזְכוּת מַעֲשֶׂה זֶה נוֹלַד לוֹ וְלָהּ בֵּן, שֶׁהָאוֹת הָרְבִיעִית כְּפוּלָה בּוֹ.

פִּתְרוֹן הַחִידָה:

  • הַבֵּן הָרְבִיעִי וְהַלֵּדָה הָרְבִיעִית: "הַבֵּן הָרְבִיעִי" הוּא יְהוּדָה, בְּנָהּ הָרְבִיעִי שֶׁל לֵאָה. "בָּרְבִיעִי נוֹלְדוּ יְלָדִים" מִתְיַחֵס לַלֵּדָה הָרְבִיעִית שֶׁל יְהוּדָה (לְאַחַר עֵר, אוֹנָן וְשֵׁלָה) – הַתְּאוֹמִים פֶּרֶץ וָזֶרַח שֶׁנּוֹלְדוּ לוֹ מִתָּמָר. יְלָדִים אֵלּוּ, כְּפִי שֶׁמְּבָאֲרִים הַמְּפָרְשִׁים (רַבֵּנוּ בַּחְיֵי, רַמְבַּ"ן), הֵם כְּנֶגֶד הַבְּרִיאָה שֶׁל יוֹם רְבִיעִי – הַשֶּׁמֶשׁ וְהַיָּרֵחַ. פֶּרֶץ כְּנֶגֶד הַלְּבָנָה שֶׁ"נִפְרֶצֶת" בְּחִדּוּשָׁהּ, וְזֶרַח כְּנֶגֶד הַשֶּׁמֶשׁ הַזּוֹרַחַת.
  • הוֹדָיָה בָּרְבִיעִי: "שֶׁעֲלֵיהֶם בָּרְבִיעִי אָנוּ עוֹשִׂים 'רְבִיעִי'" – כְּלוֹמַר, הוֹדָיָה. עַל הַיָּרֵחַ אֶפְשָׁר לְבָרֵךְ וּלְהוֹדוֹת רַק מֵהַיּוֹם הָרְבִיעִי לַמּוֹלָד. עַל הַשֶּׁמֶשׁ מוֹדִים בְּ"בִרְכַּת הַחַמָּה", פַּעַם בְּ-28 שָׁנָה, הַחָלָה תָּמִיד בְּיוֹם רְבִיעִי.
  • הָרֶמֶז בַּפָּסוּק "וַיְהִי כְּמִשְׁלֹשׁ חֳדָשִׁים": סִפּוּרָם שֶׁל יְהוּדָה וְתָמָר מַגִּיעַ לְשִׂיאוֹ כַּאֲשֶׁר מִתְגַּלֶּה הֵרָיוֹנָהּ שֶׁל תָּמָר. בַּעֲלֵי הַדְּרָשׁ רוֹאִים בַּפָּסוּק רֶמֶז עָמֹק לַחֲנֻכָּה וּלְדִין הַמִּשְׁנָה, וְקוֹרְאִים אֶת הַכָּתוּב כְּצֹפֶן שֶׁכָּל מִלָּה בּוֹ מְרַמֶּזֶת עַל פְּרָט אַחֵר: "וַיְהִי כְּמִשְׁלֹשׁ חֳדָשִׁים" – כְּלוֹמַר, סָמוּךְ לַחֹדֶשׁ הָרְבִיעִי (תִּשְׁרֵי, חֶשְׁוָן, כִּסְלֵו), בִּזְמַן חַג הַחֲנֻכָּה. "וַיֹּאמֶר יְהוּדָה" – זֶהוּ הַתַּנָּא רַבִּי יְהוּדָה בַּמִּשְׁנָה (ב"ק ו, ו). "הוֹצִיאוּהָ" – הוֹצִיאוּ אֶת נֵר הַחֲנֻכָּה לִרְשׁוּת הָרַבִּים כַּמִּצְוָה, "וְתִשָּׂרֵף" – וְאַף אִם יִשָּׂרֵף פִּשְׁתָּן שֶׁעַל גַּב הַגָּמָל, הַמַּדְלִיק פָּטוּר, כְּדֵי לְעוֹדֵד אֶת פִּרְסוּם הַנֵּס.
  • הוֹדָיַת יְהוּדָה וְהַצָּלַת הָרְבִיעִי: בְּדִיּוּק בְּאוֹתוֹ זְמַן, "סָמוּךְ לָרְבִיעִי" (סוֹף הַחֹדֶשׁ הַשְּׁלִישִׁי), עָשָׂה יְהוּדָה אֶת פְּעֻלַּת הַהוֹדָיָה הַגְּדוֹלָה שֶׁלּוֹ כְּשֶׁהִכִּיר בַּחֲפָצִים וְאָמַר: "צָדְקָה מִמֶּנִּי". פְּעֻלָּה זוֹ, שֶׁהִיא מַהוּת הָ"רְבִיעִי", הִצִּילָה אֶת הָרְבִיעִי – אֶת צֶאֱצָאָיו, פֶּרֶץ וָזֶרַח, שֶׁהָיוּ הַלֵּדָה הָרְבִיעִית שֶׁלּוֹ.
  • שְׂכַר הַהוֹדָיָה – דָּוִד הַמֶּלֶךְ: הַשָּׂכָר הַגָּדוֹל בְּיוֹתֵר עַל מַעֲשֵׂה הַהוֹדָיָה הָיָה לְדוֹרוֹת. מִפֶּרֶץ יָצָא דָּוִד הַמֶּלֶךְ, שֶׁבִּשְׁמוֹ "הָאוֹת הָרְבִיעִית (ד') כְּפוּלָה". דָּוִד הַמֶּלֶךְ עַצְמוֹ הָפַךְ לְסֵמֶל הַהוֹדָיָה לְעַם יִשְׂרָאֵל, בְּחַבְּרוֹ אֶת סֵפֶר תְּהִלִּים – סֵפֶר הַהוֹדָיָה הַנִּצְחִי לַה'.

כָּל הָרְמָזִים הַלָּלוּ, הַשְּׁזוּרִים זֶה בָּזֶה בְּאֳמָּנוּת, מְבַסְּסִים אֶת הַקֶּשֶׁר הַמַּהוּתִי בֵּין הַמִּסְפָּר אַרְבַּע לְבֵין פְּעֻלַּת הַהוֹדָיָה, שֶׁשִּׂיא בִּטּוּיָהּ בְּדָוִד הַמֶּלֶךְ. כָּעֵת, עִם הֲבָנָה זוֹ, נוּכַל לָשׁוּב לַשְּׁאֵלָה הַמְּקוֹרִית וּלְפַעְנֵחַ מַדּוּעַ כֹּחַ רוּחָנִי זֶה קָשׁוּר דַּוְקָא לַמָּקוֹם הַפִּיזִי שֶׁל הַדֶּלֶת.

3. דֶּלֶת אֶל ה': מַשְׁמָעוּת הַהוֹדָיָה וְהַדְלָקַת הַנֵּרוֹת

לְאַחַר שֶׁפֵּרַקְנוּ אֶת הָרְמָזִים הַטְּמוּנִים בְּפָרָשַׁת וַיֵּשֶׁב, אָנוּ מַגִּיעִים לְשִׂיא הַהֲבָנָה: פִּעְנוּחַ מַשְׁמָעוּתָהּ הַסִּמְלִית שֶׁל הָאוֹת ד' עַצְמָהּ. הָאוֹת הָרְבִיעִית אֵינָהּ רַק צִיּוּן מִסְפָּרִי, אֶלָּא הִיא נוֹשֵׂאת בְּתוֹכָהּ אֶת מַהוּת הַשַּׁעַר וְהַפֶּתַח. הֲבָנָה זוֹ הִיא הַמַּפְתֵּחַ הַסּוֹפִי לִשְׁאֵלַת הַדְלָקַת נֵרוֹת הַחֲנֻכָּה בַּדֶּלֶת.

דָּוִד הַמֶּלֶךְ, שֶׁכָּל מַהוּתוֹ הוֹדָיָה, כּוֹתֵב בְּסֵפֶר תְּהִלִּים (קי"ח, יט-כ):

הַדְּרָשׁ עַל פָּסוּק זֶה מְגַלֶּה סוֹד גָּדוֹל: מַהוּ "זֶה הַשַּׁעַר לַה'"? זוֹהִי פְּעֻלַּת הַהוֹדָיָה עַצְמָהּ – "אוֹדֶה יָ-הּ"! כְּלוֹמַר, הַהוֹדָיָה הִיא הַשַּׁעַר, הַפֶּתַח וְהַדֶּרֶךְ לְהִתְקָרֵב אֶל בּוֹרֵא עוֹלָם. אִם אָדָם רוֹצֶה לִפְתֹּחַ אֶת הַשַּׁעַר וְלָבוֹא בִּקְהַל ה', הַדֶּרֶךְ הִיא דֶּרֶךְ הַתּוֹדָה.

אֶת הַקֶּשֶׁר הַנִּרְמָז בִּתְהִלִּים בֵּין הַהוֹדָיָה לְבֵין הַשַּׁעַר, חוֹשֵׂף הַזֹּהַר הַקָּדוֹשׁ בְּאֹפֶן יָשִׁיר, כְּשֶׁהוּא קוֹשֵׁר זֹאת לְמַהוּתָהּ שֶׁל הָאוֹת הָרְבִיעִית עַצְמָהּ:

בְּהַשְׁמָטוֹת הַזֹּהַר (ח"א ר"נ, א) כָּתוּב: "וְד' הוּא מִלְּשׁוֹן שַׁעַר וְדֶלֶת". הַזֹּהַר קוֹבֵעַ כִּי הָאוֹת ד' הִיא שַׁעַר וּפֶתַח. הַהוֹדָיָה, שֶׁסִּמְלָהּ הוּא הַמִּסְפָּר אַרְבַּע וְהָאוֹת ד', קְשׁוּרָה בְּאֹפֶן מַהוּתִי לְרַעְיוֹן הַדֶּלֶת כִּכְנִיסָה וּכְהִתְקָרְבוּת.

כָּעֵת, כָּל הַחֲלָקִים מִתְחַבְּרִים לִתְמוּנָה אַחַת שְׁלֵמָה וּבְהִירָה: הַהוֹדָיָה (הַקְּשׁוּרָה לָאוֹת ד') הִיא הַדֶּלֶת (ד') לְהִתְקָרְבוּת לַה'. מִשּׁוּם כָּךְ, אָנוּ מַדְלִיקִים אֶת נֵרוֹת הַחֲנֻכָּה, הַמְּסַמְּלִים אֶת הַהוֹדָיָה עַל הַנִּסִּים, דַּוְקָא בְּפֶתַח הַבַּיִת – בַּדֶּלֶת. בְּמַעֲשֶׂה זֶה, אָנוּ מַמְחִישִׁים בָּעוֹלָם הַפִיזִי רַעְיוֹן רוּחָנִי עָמֹק: דֶּרֶךְ הַהוֹדָיָה אָנוּ פּוֹתְחִים שַׁעַר לִקְדֻשָּׁה וּלְקִרְבַת אֱלֹהִים.

הַמֶּסֶר הַנּוֹבֵעַ מִכָּךְ הוּא קְרִיאָה לְהִתְחַזְּקוּת. עָלֵינוּ לְהַרְגִּיל אֶת עַצְמֵנוּ לְהוֹדוֹת לְמֶלֶךְ מַלְכֵי הַמְּלָכִים עַל כָּל דָּבָר וְדָבָר – עַל הַהַצְלָחָה, עַל הַזְּכוּת לִלְמֹד תּוֹרָה וּלְקַיֵּם מִצְוֹת, וְעַל הַזְּכוּת לִהְיוֹת יְהוּדִים. הַהוֹדָיָה הַמִּתְמֶדֶת הִיא הַשַּׁעַר, וְהִיא הַדֶּלֶת שֶׁדַּרְכָּהּ "צַדִּיקִים יָבֹאוּ בוֹ".

4. 🎯 מִבְחַן חִדּוּד

1. מַהִי הַמַּשְׁמָעוּת הָעִקָּרִית שֶׁל חֲנֻכָּה כַּ"חַג הָרְבִיעִי", לְפִי הַשִּׁעוּר?

  • א) שֶׁהוּא הַחַג הָרְבִיעִי בְּחָשִׁיבוּתוֹ לְאַחַר שְׁלֹשֶׁת הָרְגָלִים.
  • ב) שֶׁהַמִּסְפָּר אַרְבַּע מְרַמֵּז עַל מַאֲבַק הַמַּכַּבִּים בְּאַרְבַּע מַלְכֻיּוֹת.
  • ג) שֶׁהַקֶּשֶׁר לַמִּסְפָּר אַרְבַּע מְכַוֵּן אֶת הַהִתְמַקְּדוּת בְּמַהוּת הַהוֹדָיָה.
  • ד) שֶׁהוּא מִתְקַיֵּם בְּסָמוּךְ לַחֹדֶשׁ הָרְבִיעִי בַּשָּׁנָה, הוּא חֹדֶשׁ טֵבֵת.

תְּשׁוּבָה נְכוֹנָה: ג'

נִמּוּק: הַשִּׁעוּר מַדְגִּישׁ שֶׁהַחֲזָרָה עַל הַמִּסְפָּר אַרְבַּע (יוֹם רְבִיעִי, נֵר רְבִיעִי, חַג רְבִיעִי) אֵינָהּ מִקְרִית אֶלָּא מְכַוֶּנֶת לְהִתְמַקֵּד בְּרַעְיוֹן הַהוֹדָיָה, הַקָּשׁוּר בְּעֹמֶק לָאוֹת ד' וְלַמִּסְפָּר אַרְבַּע.

2. כֵּיצַד מְקַשֵּׁר הַשִּׁעוּר בֵּין פֶּרֶץ וָזֶרַח, בְּנֵי יְהוּדָה, לְבֵין יוֹם הַבְּרִיאָה הָרְבִיעִי?

  • א) שְׁנֵיהֶם נוֹלְדוּ בְּיוֹם רְבִיעִי בַּשָּׁבוּעַ.
  • ב) הֵם רוֹמְזִים לַשֶּׁמֶשׁ וְלַיָּרֵחַ, שֶׁנִּבְרְאוּ בַּיּוֹם הָרְבִיעִי.
  • ג) הֵם הָיוּ הַדּוֹר הָרְבִיעִי לְאַבְרָהָם אָבִינוּ.
  • ד) סִפּוּרָם מֻזְכָּר בַּסֵּפֶר הָרְבִיעִי שֶׁל הַתּוֹרָה, סֵפֶר בַּמִּדְבָּר.

תְּשׁוּבָה נְכוֹנָה: ב'

נִמּוּק: הַשִּׁעוּר מְצַטֵּט מְפָרְשִׁים הַמְּסַבִּירִים שֶׁפֶּרֶץ (כְּנֶגֶד הַלְּבָנָה הַנִּפְרֶצֶת) וְזֶרַח (כְּנֶגֶד הַשֶּׁמֶשׁ הַזּוֹרַחַת) הֵם כְּנֶגֶד הַמְּאוֹרוֹת שֶׁנִּבְרְאוּ בַּיּוֹם הָרְבִיעִי.

3. מַהוּ הָרֶמֶז לְדִין נֵר חֲנֻכָּה בַּפָּסוּק "וַיְהִי כְּמִשְׁלֹשׁ חֳדָשִׁים וַיֹּאמֶר יְהוּדָה הוֹצִיאוּהָ וְתִשָּׂרֵף"?

  • א) הָרֶמֶז הוּא שֶׁיְּהוּדָה קָבַע אֶת זְמַן הַדְלָקַת הַנֵּרוֹת.
  • ב) הָרֶמֶז הוּא לְרַבִּי יְהוּדָה בַּמִּשְׁנָה, שֶׁפָּטַר אֶת הַמַּדְלִיק מִנְּזָקִים שֶׁנִּגְרְמוּ מִנֵּר חֲנֻכָּה.
  • ג) הָרֶמֶז הוּא שֶׁיֵּשׁ לִשְׂרֹף אֶת הַפְּתִילוֹת וְהַשֶּׁמֶן שֶׁנּוֹתְרוּ מִנֵּר חֲנֻכָּה.
  • ד) הָרֶמֶז הוּא שֶׁחַג הַחֲנֻכָּה נִקְבַּע לְאַחַר שְׁלֹשָׁה חֳדָשִׁים שֶׁל מִלְחָמָה.

תְּשׁוּבָה נְכוֹנָה: ב'

נִמּוּק: הַדְּרָשׁ מְקַשֵּׁר אֶת "וַיֹּאמֶר יְהוּדָה" לְרַבִּי יְהוּדָה בַּמִּשְׁנָה, וְאֶת "הוֹצִיאוּהָ וְתִשָּׂרֵף" לְדִינוֹ שֶׁיֵּשׁ לְהוֹצִיא אֶת הַנֵּר, וְאַף אִם יִגְרֹם לִשְׂרֵפָה – הַמַּדְלִיק פָּטוּר.

4. מַדּוּעַ, עַל פִּי הַזֹּהַר הַקָּדוֹשׁ וּפְסוּקֵי תְּהִלִּים, קְשׁוּרָה הַהוֹדָיָה דַּוְקָא לָאוֹת ד'?

  • א) כִּי הָאוֹת ד' מְסַמֶּלֶת דַּלּוּת וַעֲנָוָה, שֶׁהֵן תְּנַאי לְהוֹדָיָה אֲמִתִּית.
  • ב) כִּי דָּוִד הַמֶּלֶךְ, אֲבִי הַהוֹדָיָה, נוֹלַד בַּיּוֹם הָרְבִיעִי לַחֹדֶשׁ.
  • ג) כִּי הָאוֹת ד' רוֹמֶזֶת לְ"דֶּלֶת" וּלְ"שַׁעַר", וְהַהוֹדָיָה הִיא הַשַּׁעַר לְהִתְקָרֵב לַה'.
  • ד) כִּי יֵשׁ אַרְבַּע רָמוֹת שֶׁל הוֹדָיָה שֶׁצָּרִיךְ לַעֲבֹר כְּדֵי לְהַגִּיעַ לִשְׁלֵמוּת.

תְּשׁוּבָה נְכוֹנָה: ג'

נִמּוּק: הַשִּׁעוּר מֵבִיא אֶת דִּבְרֵי הַזֹּהַר שֶׁהָאוֹת ד' הִיא לְשׁוֹן דֶּלֶת, וּמְקַשֵּׁר זֹאת לַפָּסוּק בִּתְהִלִּים "זֶה-הַשַּׁעַר לַה'", וּמַסְבִּיר שֶׁהַהוֹדָיָה עַצְמָהּ הִיא הַשַּׁעַר וְהַדֶּלֶת לְקִרְבַת ה'.

5. מַהִי הַמַּטָּרָה הָרוּחָנִית שֶׁל הַדְלָקַת נֵרוֹת חֲנֻכָּה דַּוְקָא בַּדֶּלֶת, לְפִי הַסִּכּוּם שֶׁל הַשִּׁעוּר?

  • א) כְּדֵי לְפַרְסֵם אֶת הַנֵּס בְּאֹפֶן הַבָּרוּר וְהַנִּרְאֶה בְּיוֹתֵר לְכָל הָעוֹבְרִים וְהַשָּׁבִים.
  • ב) כְּדֵי לִשְׁמֹר עַל הַבַּיִת מִפְּנֵי מַזִּיקִים, כְּפִי שֶׁהַדֶּלֶת שׁוֹמֶרֶת עַל הַבַּיִת.
  • ג) כְּדֵי לְהַזְכִּיר אֶת דָּוִד הַמֶּלֶךְ, שֶׁבִּשְׁמוֹ שְׁתֵּי אוֹתִיּוֹת ד' כְּמוֹ דֶּלֶת.
  • ד) כְּדֵי לְהַמְחִישׁ בְּאֹפֶן פִיזִי אֶת הָרַעְיוֹן שֶׁהַהוֹדָיָה (הַנֵּר) הִיא הַדֶּלֶת לְקִרְבַת ה'.

תְּשׁוּבָה נְכוֹנָה: ד'

נִמּוּק: הַמַּסְקָנָה הַסּוֹפִית הִיא שֶׁהַדְלָקַת הַנֵּר (הַמְּסַמֵּל הוֹדָיָה) בַּדֶּלֶת הִיא מַעֲשֶׂה סִמְלִי עָמֹק, הַמְּבַטֵּא בָּעוֹלָם הַגַּשְׁמִי אֶת הָאֱמֶת הָרוּחָנִית שֶׁהַהוֹדָיָה הִיא הַכְּנִיסָה וְהַפֶּתַח לְקִרְבַת אֱלֹהִים.

5. 📚 מְקוֹרוֹת

  • פָּרָשַׁת וַיֵּשֶׁב

        const correctAnswers = {

            q1: "ב",

            q2: "ב",

            q3: "ג",

            q4: "ג",

            q5: "ב",

            q6: "ב",

            q7: "ג",

            q8: "ג",

            q9: "ב",

            q10: "ב"

        };


        function checkAnswers() {

            const form = document.getElementById('quiz-form');

            const results = form.querySelectorAll('.result');

            results.forEach(result => result.style.display = 'none'); // Reset results


            let score = 0;

            let total = 0;


            for (const [question, correctAnswer] of Object.entries(correctAnswers)) {

                const userAnswer = form.querySelector(`input[name="${question}"]:checked`);

                const result = document.getElementById(`result-${question}`);

                if (userAnswer) {

                    total++;

                    if (userAnswer.value === correctAnswer) {

                        result.textContent = "נכון!";

                        result.className = "result correct";

                        score++;

                    } else {

                        result.textContent = "לא נכון, התשובה הנכונה היא: " + correctAnswer;

                        result.className = "result incorrect";

                    }

                    result.style.display = 'block';

                }

            }


            alert(`סיימת! הציון שלך: ${score} מתוך ${total}`);

        }

שִׁיעוּר 1: חֲנֻכָּה - חַג הַהוֹדָיָה הָרְבִיעִי

לְחַג הַחֲנֻכָּה מַעֲמָד יִחוּדִי בְּלוּחַ הַשָּׁנָה הַיְּהוּדִי. לְצַד שְׁלֹשֶׁת הָרְגָלִים – פֶּסַח, שָׁבוּעוֹת וְסֻכּוֹת – הוּא נֶחְשָׁב לַ"חַג הָרְבִיעִי". אַךְ בְּנִגּוּד אֲלֵיהֶם, שֶׁמְּצֻוִּים מִן הַתּוֹרָה, חֲנֻכָּה הוּא מִדִּבְרֵי חֲכָמִים. חֲזַ"ל קָבְעוּ אֶת מַהוּתוֹ הַמֶּרְכָּזִית שֶׁל הַחַג כְּ"יָמִים שֶׁל הוֹדָיָה". לֹא רֶגַע אֶחָד שֶׁל תּוֹדָה, אֶלָּא מַצָּב מִתְמַשֵּׁךְ, שְׁמוֹנָה יָמִים שֶׁכָּל מַהוּתָם סוֹבֶבֶת סְבִיב פְּעֻלַּת הַהוֹדָיָה לַה'. הַמִּסְפָּר אַרְבַּע חוֹזֵר וּמֵהַדְהֵד בַּחַג בְּאֹפֶן מֻבְהָק: אָנוּ עוֹמְדִים בְּיוֹם רְבִיעִי לַשָּׁבוּעַ, מַדְלִיקִים נֵר רְבִיעִי, וְחוֹגְגִים אֶת הַחַג הָרְבִיעִי עַצְמוֹ. רִבּוּי זֶה שֶׁל הַמִּסְפָּר אַרְבַּע אֵינוֹ מִקְרִי, אֶלָּא מְכַוֵּן אוֹתָנוּ לְהִתְמַקֵּד בַּמַּהוּת הַמְּזֻקֶּקֶת שֶׁל הַחַג – מַהוּת הַהוֹדָיָה.

מַהוּ הַהֶבְדֵּל הָעִקָּרִי בֵּין חַג הַחֲנֻכָּה לְבֵין שְׁלֹשֶׁת הָרְגָלִים?

חֲנֻכָּה הוּא מִדִּבְרֵי חֲכָמִים, בְּעוֹד שֶׁהָרְגָלִים מְצֻוִּים מִן הַתּוֹרָה

חֲנֻכָּה מִתְקַיֵּם בַּחוֹרֶף, בְּעוֹד שֶׁהָרְגָלִים בְּעוֹנוֹת אַחֵרוֹת

חֲנֻכָּה אָרוֹךְ יוֹתֵר מֵהָרְגָלִים וְנִמְשָׁךְ שְׁמוֹנָה יָמִים

חֲנֻכָּה קָשׁוּר לְנִצָּחוֹן צְבָאִי, בְּעוֹד שֶׁהָרְגָלִים חַקְלָאִיִּים

כֵּיצַד הִגְדִּירוּ חֲזַ"ל אֶת מַהוּת חַג הַחֲנֻכָּה?

יָמִים שֶׁל הוֹדָיָה - מַצָּב מִתְמַשֵּׁךְ שֶׁל תּוֹדָה לַה'

יָמִים שֶׁל שִׂמְחָה - חֲגִיגָה עַל נִצָּחוֹן הַמַּכַּבִּים

יָמִים שֶׁל תְּשׁוּבָה - הִזְדַּמְּנוּת לָשׁוּב אֶל ה'

יָמִים שֶׁל טָהֳרָה - זֵכֶר לְטִהוּר הַמִּקְדָּשׁ

מַדּוּעַ הַמִּסְפָּר אַרְבַּע חוֹזֵר כָּל כָּךְ הַרְבֵּה בְּחַג הַחֲנֻכָּה?

כְּדֵי לְכַוֵּן אוֹתָנוּ לְהִתְמַקֵּד בְּמַהוּת הַהוֹדָיָה שֶׁל הַחַג

כִּי הַמַּכַּבִּים נִלְחֲמוּ נֶגֶד אַרְבַּע אִימְפֶּרְיוֹת גְּדוֹלוֹת

כִּי הַנֵּס קָרָה בַּשָּׁבוּעַ הָרְבִיעִי שֶׁל הַחֹדֶשׁ

כִּי הָיוּ אַרְבָּעָה אַחִים חַשְׁמוֹנָאִים רָאשִׁיִּים

שִׁיעוּר 2: יְהוּדָה, תָּמָר וְסוֹד הָרְבִיעִי

כְּדֵי לַחְשֹׂף אֶת הַשְּׁכָבוֹת הַנִּסְתָּרוֹת הַקּוֹשְׁרוֹת אֶת הַמִּסְפָּר אַרְבַּע, יְהוּדָה, חֲנֻכָּה וְהוֹדָיָה, אָנוּ לוֹמְדִים מִפָּרָשַׁת וַיֵּשֶׁב חִידָה מֻרְכֶּבֶת. יְהוּדָה הוּא הַבֵּן הָרְבִיעִי שֶׁל לֵאָה. הַתְּאוֹמִים פֶּרֶץ וָזֶרַח, שֶׁנּוֹלְדוּ לוֹ מִתָּמָר, הֵם הַלֵּדָה הָרְבִיעִית שֶׁלּוֹ (לְאַחַר עֵר, אוֹנָן וְשֵׁלָה). הַמְּפָרְשִׁים מְבָאֲרִים שֶׁיְּלָדִים אֵלּוּ הֵם כְּנֶגֶד הַבְּרִיאָה שֶׁל יוֹם רְבִיעִי – הַשֶּׁמֶשׁ וְהַיָּרֵחַ. פֶּרֶץ כְּנֶגֶד הַלְּבָנָה שֶׁ"נִפְרֶצֶת" בְּחִדּוּשָׁהּ, וְזֶרַח כְּנֶגֶד הַשֶּׁמֶשׁ הַזּוֹרַחַת. עַל הַיָּרֵחַ אֶפְשָׁר לְבָרֵךְ וּלְהוֹדוֹת רַק מֵהַיּוֹם הָרְבִיעִי לַמּוֹלָד. עַל הַשֶּׁמֶשׁ מוֹדִים בְּ"בִרְכַּת הַחַמָּה", פַּעַם בְּ-28 שָׁנָה, הַחָלָה תָּמִיד בְּיוֹם רְבִיעִי!

מִי הָיָה יְהוּדָה בְּמִבְנֵה הַמִּשְׁפָּחָה שֶׁל יַעֲקֹב?

הַבֵּן הָרְבִיעִי שֶׁל לֵאָה, שֶׁמֵּהֶם יָצָא מֶלֶךְ יִשְׂרָאֵל

הַבֵּן הַשְּׁלִישִׁי שֶׁל רָחֵל, שֶׁהָיָה הַנֶּאֱמָן בְּיוֹתֵר

הַבֵּן הָרִאשׁוֹן שֶׁל לֵאָה, שֶׁקִּבֵּל אֶת הַבְּכוֹרָה

הַבֵּן הַחֲמִישִׁי שֶׁל יַעֲקֹב, שֶׁנִּבְחַר לְהַנְהָגָה

כֵּיצַד קְשׁוּרִים פֶּרֶץ וָזֶרַח, בְּנֵי יְהוּדָה וְתָמָר, לְיוֹם הַבְּרִיאָה הָרְבִיעִי?

הֵם מְסַמְּלִים אֶת הַשֶּׁמֶשׁ וְהַיָּרֵחַ שֶׁנִּבְרְאוּ בַּיּוֹם הָרְבִיעִי

הֵם נוֹלְדוּ בְּיוֹם רְבִיעִי בַּשָּׁבוּעַ בְּדִיּוּק

הֵם הָיוּ הַדּוֹר הָרְבִיעִי מֵאַבְרָהָם אָבִינוּ

שְׁנֵיהֶם חָיוּ אַרְבַּע מֵאוֹת שָׁנָה כְּמוֹ הַיָּרֵחַ

מָתַי אֶפְשָׁר לְבָרֵךְ בִּרְכַּת הַלְּבָנָה, וּמַה הַקֶּשֶׁר לַמִּסְפָּר אַרְבַּע?

רַק מֵהַיּוֹם הָרְבִיעִי לַמּוֹלָד, כְּשֶׁהַיָּרֵחַ נִרְאֶה בְּבִרוּר

רַק בְּיוֹם רְבִיעִי בַּשָּׁבוּעַ, כְּיוֹם בְּרִיאַת הַמְּאוֹרוֹת

רַק אַרְבַּע פְּעָמִים בַּשָּׁנָה, בְּרָאשֵׁי הָעוֹנוֹת

רַק בְּחֹדֶשׁ הָרְבִיעִי, כִּסְלֵו, בִּזְמַן הַחֲנֻכָּה

שִׁיעוּר 3: פְּעֻלַּת הַהוֹדָיָה שֶׁל יְהוּדָה

סִפּוּרָם שֶׁל יְהוּדָה וְתָמָר מַגִּיעַ לְשִׂיאוֹ כַּאֲשֶׁר מִתְגַּלֶּה הֵרָיוֹנָהּ שֶׁל תָּמָר. בַּעֲלֵי הַדְּרָשׁ רוֹאִים בַּפָּסוּק "וַיְהִי כְּמִשְׁלֹשׁ חֳדָשִׁים" רֶמֶז עָמֹק לַחֲנֻכָּה. "וַיְהִי כְּמִשְׁלֹשׁ חֳדָשִׁים" – סָמוּךְ לַחֹדֶשׁ הָרְבִיעִי (כִּסְלֵו), בִּזְמַן חַג הַחֲנֻכָּה! "וַיֹּאמֶר יְהוּדָה" – רַבִּי יְהוּדָה בַּמִּשְׁנָה. "הוֹצִיאוּהָ" – הוֹצִיאוּ אֶת נֵר הַחֲנֻכָּה לִרְשׁוּת הָרַבִּים! "וְתִשָּׂרֵף" – וְאִם יִשָּׂרֵף פִּשְׁתָּן, הַמַּדְלִיק פָּטוּר! בְּדִיּוּק בְּאוֹתוֹ זְמַן, "סָמוּךְ לָרְבִיעִי", עָשָׂה יְהוּדָה אֶת פְּעֻלַּת הַהוֹדָיָה הַגְּדוֹלָה שֶׁלּוֹ כְּשֶׁהִכִּיר בַּחֲפָצִים וְאָמַר: "צָדְקָה מִמֶּנִּי". פְּעֻלָּה זוֹ הִצִּילָה אֶת צֶאֱצָאָיו, פֶּרֶץ וָזֶרַח – הַלֵּדָה הָרְבִיעִית שֶׁלּוֹ. מִפֶּרֶץ יָצָא דָּוִד הַמֶּלֶךְ, שֶׁבִּשְׁמוֹ הָאוֹת ד' כְּפוּלָה!

מַה הָרֶמֶז שֶׁבַּפָּסוּק 'וַיְהִי כְּמִשְׁלֹשׁ חֳדָשִׁים' לְחַג הַחֲנֻכָּה?

זֶה רוֹמֵז לִזְמַן הַחֲנֻכָּה - סָמוּךְ לַחֹדֶשׁ הָרְבִיעִי, כִּסְלֵו

זֶה רוֹמֵז שֶׁהַנֵּס אָרַךְ שְׁלֹשָׁה חֳדָשִׁים

זֶה רוֹמֵז שֶׁהַמִּלְחָמָה נִמְשְׁכָה שְׁלֹשָׁה חֳדָשִׁים

זֶה רוֹמֵז לִשְׁלֹשָׁה מוֹעֲדִים שֶׁעוֹבְרִים לִפְנֵי הַחֲנֻכָּה

מַה הַקֶּשֶׁר בֵּין 'הוֹצִיאוּהָ וְתִשָּׂרֵף' לְדִין נֵר חֲנֻכָּה?

רַבִּי יְהוּדָה פּוֹטֵר אֶת הַמַּדְלִיק מִנְּזָקִים שֶׁנִּגְרְמוּ מֵהַנֵּר

צָרִיךְ לִשְׂרֹף אֶת כָּל הַפְּתִילוֹת שֶׁנּוֹתְרוּ מֵהַחֲנֻכָּה

יֵשׁ לְהוֹצִיא אֶת הַנֵּר לִרְחוֹב וּלְהַדְלִיק שָׁם

אָסוּר לְהַדְלִיק נֵר חֲנֻכָּה בִּמְקוֹם שֶׁיֵּשׁ פִּשְׁתָּן

מָה הָיָה הַשָּׂכָר הַגָּדוֹל עַל פְּעֻלַּת הַהוֹדָיָה שֶׁל יְהוּדָה?

מִפֶּרֶץ יָצָא דָּוִד הַמֶּלֶךְ, שֶׁהָאוֹת ד' כְּפוּלָה בִּשְׁמוֹ

הוּא זָכָה לִהְיוֹת הַשֵּׁבֶט הַמַּנְהִיג בְּעַם יִשְׂרָאֵל

צֶאֱצָאָיו הָיוּ הַכֹּהֲנִים בַּמִּקְדָּשׁ

הוּא קִבֵּל אֶת הַבְּכוֹרָה בְּמָקוֹם רְאוּבֵן

שִׁיעוּר 4: הָאוֹת ד' - דֶּלֶת אֶל ה'

לְאַחַר שֶׁפֵּרַקְנוּ אֶת הָרְמָזִים מִפָּרָשַׁת וַיֵּשֶׁב, אָנוּ מַגִּיעִים לְשִׂיא הַהֲבָנָה: פִּעְנוּחַ מַשְׁמָעוּתָהּ הַסִּמְלִית שֶׁל הָאוֹת ד' עַצְמָהּ. דָּוִד הַמֶּלֶךְ כּוֹתֵב בִּתְהִלִּים: "פִּתְחוּ-לִי שַׁעֲרֵי-צֶדֶק אָבֹא-בָם אוֹדֶה יָ-הּ. זֶה-הַשַּׁעַר לַה' צַדִּיקִים יָבֹאוּ בוֹ." הַדְּרָשׁ מְגַלֶּה: מַהוּ "זֶה הַשַּׁעַר לַה'"? זוֹהִי פְּעֻלַּת הַהוֹדָיָה עַצְמָהּ – "אוֹדֶה יָ-הּ"! הַזֹּהַר הַקָּדוֹשׁ חוֹשֵׂף זֹאת בְּאֹפֶן יָשִׁיר: "וְד' הוּא מִלְּשׁוֹן שַׁעַר וְדֶלֶת". הָאוֹת ד' הִיא שַׁעַר וּפֶתַח! מִשּׁוּם כָּךְ, אָנוּ מַדְלִיקִים אֶת נֵרוֹת הַחֲנֻכָּה דַּוְקָא בְּפֶתַח הַבַּיִת – בַּדֶּלֶת. בְּמַעֲשֶׂה זֶה, אָנוּ מַמְחִישִׁים בָּעוֹלָם הַפִּיזִי רַעְיוֹן רוּחָנִי עָמֹק: דֶּרֶךְ הַהוֹדָיָה אָנוּ פּוֹתְחִים שַׁעַר לִקְדֻשָּׁה וּלְקִרְבַת אֱלֹהִים!

מַהוּ 'הַשַּׁעַר לַה'' לְפִי דָּוִד הַמֶּלֶךְ בִּתְהִלִּים?

פְּעֻלַּת הַהוֹדָיָה עַצְמָהּ - 'אוֹדֶה יָ-הּ' - הִיא הַשַּׁעַר

שַׁעַר הַמִּקְדָּשׁ הַמִּזְרָחִי שֶׁבּוֹ נִכְנָסִים לְהִתְפַּלֵּל

לִמּוּד הַתּוֹרָה שֶׁפּוֹתֵחַ אֶת הַדֶּרֶךְ לַה'

עֲנָוָה וְדַלּוּת רוּחַ שֶׁמְּאַפְשָׁרוֹת קִרְבָה

מָה מְגַלֶּה הַזֹּהַר הַקָּדוֹשׁ עַל הָאוֹת ד'?

שֶׁהָאוֹת ד' הִיא מִלְּשׁוֹן שַׁעַר וְדֶלֶת - פֶּתַח לִקְדֻשָּׁה

שֶׁהָאוֹת ד' מְסַמֶּלֶת דַּלּוּת וַעֲנָוָה בִּלְבַד

שֶׁהָאוֹת ד' קְשׁוּרָה רַק לְדָוִד הַמֶּלֶךְ

שֶׁהָאוֹת ד' הִיא הָאוֹת הַמְּיוּחֶדֶת לִתְפִלָּה

מַדּוּעַ אָנוּ מַדְלִיקִים אֶת נֵרוֹת הַחֲנֻכָּה דַּוְקָא בַּדֶּלֶת?

כְּדֵי לְהַמְחִישׁ שֶׁהַהוֹדָיָה (הַנֵּר) הִיא הַדֶּלֶת לְקִרְבַת ה'

כְּדֵי שֶׁהָעוֹבְרִים יִרְאוּ אֶת הַנֵּר בְּקַלּוּת

כְּדֵי לְהַזְכִּיר שֶׁדָּוִד נוֹלַד עִם אוֹת ד' כְּפוּלָה

כְּדֵי לִשְׁמֹר עַל הַבַּיִת מִסַּכָּנוֹת חִיצוֹנִיּוֹת

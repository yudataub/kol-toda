תְּפִלַּת לוֹט וְאַבְרָהָם - סוֹד הַיְשׁוּעָה 🌟

שָׁלוֹם וּבְרָכָה, יְלָדִים יְקָרִים! 💖 הַיּוֹם נְגַלֶּה סוֹד מַדְהִים מֵעוֹלַם הַתְּפִלָּה! ✨ הָיָה פַּעַם סִפּוּר מְיֻחָד: אַבְרָהָם אָבִינוּ וְלוֹט שְׁנֵיהֶם הִתְפַּלְּלוּ עַל סְדוֹם 🙏 אֲבָל... אַבְרָהָם אָבִינוּ, הַצַּדִּיק הַגָּדוֹל 🌟, לֹא הִצְלִיחַ לְהַצִּיל אֶת סְדוֹם 😢 וְלוֹט? לוֹט כֵּן הִצְלִיחַ! 🎉 אֵיךְ זֶה אֶפְשָׁרִי? הַסּוֹד הַגָּדוֹל: אַבְרָהָם אָבִינוּ הִתְפַּלֵּל "מִצַּד מַעֲשִׂים" 📜 - הוּא אָמַר: "הִנֵּה נָא הוֹאַלְתִּי לְדַבֵּר". הוּא חָשַׁב שֶׁבִּזְכוּת מַעֲשָׂיו הַטּוֹבִים הַקָּדוֹשׁ בָּרוּךְ הוּא יַעֲנֶה לוֹ 💪 אֲבָל לוֹט הִתְפַּלֵּל "מִצַּד רַחֲמִים" 💕 - הוּא הִרְגִּישׁ: לֹא מַגִּיעַ לִי כְּלוּם! רַק בּוֹרֵא עוֹלָם רוֹצֶה לָתֵת לִי מִתּוֹךְ אַהֲבָה! 🌈 💭 חִשְׁבוּ רֶגַע: אֵיךְ אֲנִי מִתְפַּלֵּל? הַאִם אֲנִי חוֹשֵׁב "מַגִּיעַ לִי" אוֹ "הַשֵּׁם אוֹהֵב אוֹתִי וְרוֹצֶה לָתֵת לִי"? 💖 כְּשֶׁאָדָם מְבַקֵּשׁ מִצַּד רַחֲמִים - אֵין גְּבוּל לְמַה שֶׁהוּא יָכוֹל לְקַבֵּל! 🎁✨ כִּי לְמִדַּת חַסְדּוֹ שֶׁל בּוֹרֵא עוֹלָם אֵין גְּבוּל! 🌟💫 זֶה כְּמוֹ לְקַבֵּל מַתָּנָה מֵאַבָּא שֶׁאוֹהֵב אוֹתְךָ - לֹא בִּגְלַל שֶׁאַתָּה מוּכְרָח לִהְיוֹת מוּשְׁלָם, אֶלָּא בִּגְלַל שֶׁהוּא פָּשׁוּט אוֹהֵב אוֹתְךָ! 💖

מִי הִצְלִיחַ לְהַצִּיל אֶת סְדוֹם בִּתְפִלָּתוֹ? 🙏

מַה הַהֶבְדֵּל בֵּין תְּפִלָּה 'מִצַּד מַעֲשִׂים' לִתְפִלָּה 'מִצַּד רַחֲמִים'? 💭

מַה קוֹרֶה כְּשֶׁאָדָם מִתְפַּלֵּל מִצַּד רַחֲמִים? 🌟

סוֹד י״ג מִדּוֹת הָרַחֲמִים ✨

עַכְשָׁו נְגַלֶּה עוֹד סוֹד גָּדוֹל! 🎁 הַקָּדוֹשׁ בָּרוּךְ הוּא גִּלָּה לְמשֶׁה רַבֵּנוּ: 📜 "כָּל פַּעַם שֶׁתִּהְיוּ צְרִיכִים יְשׁוּעָה 🆘 - תֹּאמְרוּ שָׁלוֹשׁ עֶשְׂרֵה מִדּוֹת שֶׁל רַחֲמִים! " ⭐ "הַשֵּׁם, הַשֵּׁם, אֵל רַחוּם וְחַנּוּן..." 🙏💫 אֲבָל... 🤔 הַרְבֵּה אֲנָשִׁים שׁוֹאֲלִים: אֵיךְ זֶה עוֹבֵד? מָה מְיֻחָד בְּמִלִּים הָאֵלֶּה? הַסּוֹד הַנִּפְלָא: לֹא מַסְפִּיק סְתָם לוֹמַר אֶת הַמִּלִּים! 📣 צָרִיךְ לָחִיּוֹת אוֹתָן בְּכָל הַלֵּב! 💗 כְּשֶׁאַתָּה אוֹמֵר "הַשֵּׁם הַשֵּׁם אֵל רַחוּם וְחַנּוּן" - תַּרְגִּישׁ בְּכָל נְשִׁימָה: 😌✨ 💖 "הַשֵּׁם רַחוּם עָלַי!" 🌟 "הוּא חוֹנֵן אוֹתִי בְּאַהֲבָה!" 🎁 "הוּא רַב חֶסֶד אִתִּי!" 💭 תַּחְשְׁבוּ: כַּמָּה פְּעָמִים הַיּוֹם הַשֵּׁם עָשָׂה אִתְּכֶם חֶסֶד? 🌈 הִתְעוֹרַרְתֶּם בַּבֹּקֶר? יָכוֹל לִרְאוֹת? לִשְׁמוֹעַ? לֶאֱכוֹל? 🍞👀👂 כְּשֶׁאַתָּה מִתְבּוֹנֵן בְּחַסְדֵי הַשֵּׁם וּמַרְגִּישׁ: "קָטֹנְתִּי מִכָּל הַחֲסָדִים!" 🙏 - אָז כְּשֶׁאַתָּה אוֹמֵר י״ג מִדּוֹת, הֵן פּוֹעֲלוֹת! 🔥 זֶה כְּמוֹ לִפְתּוֹחַ דֶּלֶת לְאוֹצָר 💎 - הַמַּפְתֵּחַ הוּא לְהַרְגִּישׁ אֶת אַהֲבַת הַשֵּׁם בְּאֱמֶת! 🔑✨

מָה הַקָּדוֹשׁ בָּרוּךְ הוּא אָמַר לְמשֶׁה רַבֵּנוּ לַעֲשׂוֹת כְּשֶׁיִּהְיוּ צְרִיכִים יְשׁוּעָה? 🆘

אֵיךְ י״ג מִדּוֹת הָרַחֲמִים פּוֹעֲלוֹת בְּאֱמֶת? 🔥

אֵיזוֹ הַרְגָּשָׁה צְרִיכָה לִהְיוֹת לָנוּ כְּשֶׁמִּתְבּוֹנְנִים בְּחַסְדֵי הַשֵּׁם? 🌈

הָאוֹת ד׳ - שׁוֹרֶשׁ הַהוֹדָאָה 🔤

עַכְשָׁו חִידָה מַדְהִימָה! 🧩✨ בּוֹאוּ נִרְאֶה מַשֶּׁהוּ מְיֻחָד בְּמִלָּה "הוֹדָאָה": אִם נַהֲפוֹךְ אֶת הַמִּלָּה "הוֹדָאָה" לְכָל מִינֵי צְדָדִים 🔄 - כָּל הָאוֹתִיּוֹת יִפְּלוּ חוּץ מֵאוֹת אַחַת! 💚 מוֹדִים - נוֹפֵל הַה׳, נִשְׁאַר רַק ד׳ 💙 יוֹדוּ לַשֵּׁם חַסְדּוֹ - נוֹפֵל הַי׳ וְהַו׳, נִשְׁאַר ד׳ 💛 נוֹדֶה לְךָ - נוֹפְלִים אוֹתִיּוֹת, נִשְׁאַר ד׳ 🧡 יְהוּדָה - הַבֵּן הָרְבִיעִי, גַּם כָּאן יֵשׁ ד׳! תָּמִיד, תָּמִיד, תָּמִיד - הָאוֹת ד׳ תִּשָּׁאֵר! 🎯 לָמָּה? 🤔 כִּי בַּגְּמָרָא לָמַדְנוּ: ד׳ = דַּל (דָּל, עָנִי, צָרִיךְ) 💫 💭 חִשְׁבוּ עַל זֶה: הוֹדָאָה אֲמִתִּית בָּאָה רַק כְּשֶׁאֲנִי מַרְגִּישׁ שֶׁלֹּא מַגִּיעַ לִי! שֶׁאֲנִי "דַּל" וְהַשֵּׁם נוֹתֵן לִי מַתָּנָה! 🎁✨ כְּשֶׁאַתָּה חוֹשֵׁב "הַכֹּל מַגִּיעַ לִי" 😤 - אַתָּה לֹא יָכוֹל לְהוֹדוֹת! כִּי אֵין מַתָּנָה, יֵשׁ רַק "מַה שֶּׁמַּגִּיעַ לִי". אֲבָל כְּשֶׁאַתָּה מַרְגִּישׁ "קָטֹנְתִּי מִכָּל הַחֲסָדִים" 🙏💕 - אָז כָּל דָּבָר קָטָן הוּא מַתָּנָה! אָז אַתָּה מוֹדֶה עַל הַכֹּל! דָּוִד הַמֶּלֶךְ 👑 - יֵשׁ לוֹ שְׁנֵי דַּלִּים (ד-ו-ד)! הוּא הָיָה כְּמוֹ הַלְּבָנָה 🌙 - אֵין לוֹ אוֹר עַצְמִי, הַכֹּל הוּא מְקַבֵּל מֵהַשֵּׁם! 💫 לָכֵן דָּוִד הַמֶּלֶךְ חִבֵּר אֶת סֵפֶר תְּהִלִּים 📖 - סֵפֶר הַהוֹדָאָה הַגָּדוֹל שֶׁל עַם יִשְׂרָאֵל! 🎵✨

מָה מְיֻחָד בָּאוֹת ד׳ בְּכָל מִלּוֹת הַהוֹדָאָה? 🔤

מַה הַקֶּשֶׁר בֵּין הָאוֹת ד׳ לְבֵין הוֹדָאָה אֲמִתִּית? 💭

לָמָּה דָּוִד הַמֶּלֶךְ נִקְרָא 'אֲבִי הַהוֹדָאָה' וְחִבֵּר אֶת סֵפֶר תְּהִלִּים? 👑

שְׁנֵי סוּגֵי חַיִּים - אֵיזֶה תִּבְחַר? 🌈

עַכְשָׁו בּוֹאוּ נִרְאֶה מַשֶּׁהוּ מְיֻחָד מְאוֹד! 🎯 יֵשׁ שְׁנֵי סוּגֵי אֲנָשִׁים בָּעוֹלָם: 😤 טִיפּוּס א׳: "הַכֹּל מַגִּיעַ לִי!" 🔴 חַי בְּתְחוּשָׁה: "חַיָּבִים לִי!" 😞 כָּל הַיּוֹם סוֹבֵל: "לָמָּה לֹא קִבַּלְתִּי?!" 💔 תָּמִיד מְאֻכְזָב - "זֶה לֹא קִבַּלְתִּי, זֶה לֹא קִבַּלְתִּי" 😣 הַפָּסוּק אוֹמֵר: "רַבִּים מַכְאוֹבִים לָרָשָׁע" - חוֹטֵף מַכּוֹת מִכָּל עֵבֶר! 😊 טִיפּוּס ב׳: "קָטֹנְתִּי מִכָּל הַחֲסָדִים!" 💚 חַי בְּהַכָּרַת טוֹב מִתְמִידָה! 😃 כָּל הַיּוֹם שָׂמֵחַ: "תִּרְאוּ כַּמָּה קִבַּלְתִּי!" 💖 מוֹדֶה עַל כָּל דָּבָר קָטָן - גַּם עַל אֲוִיר לִנְשׁוֹם! 🌟 הַפָּסוּק אוֹמֵר: "וְהַבּוֹטֵחַ בַּה׳ חֶסֶד יְסוֹבְבֶנּוּ" - מֻקָּף חֲסָדִים מִכָּל עֵבֶר! 💭 שְׁאֵלָה חֲשׁוּבָה: אֵיזֶה טִיפּוּס אֲנִי רוֹצֶה לִהְיוֹת? 🤔 הַאִם אֲנִי חַי בְּ"הַכֹּל מַגִּיעַ לִי" אוֹ בְּ"תּוֹדָה לְהַשֵּׁם עַל כָּל דָּבָר"? 🙏✨ הַמַּהֲלָךְ הַמַּעֲשִׂי: 📝 💚 כָּל בֹּקֶר - כִּתְבוּ 5 חֲסָדִים שֶׁהַשֵּׁם עָשָׂה אִתְּכֶם אֶתְמוֹל 🙏 לִפְנֵי תְּפִלָּה - קִרְאוּ אֶת הָרְשִׁימָה וְהַרְגִּישׁוּ: "קָטֹנְתִּי!" ✨ אִמְרוּ י״ג מִדּוֹת בְּהַרְגָּשָׁה - הַרְגִּישׁוּ: "ה׳ רַחוּם עָלַי!" 💖 תּוֹדָה עַל כָּל דָּבָר קָטָן - גַּם עַל כּוֹס מַיִם! גַּם עַל חִיּוּךְ! כְּשֶׁתַּתְחִילוּ לָחִיּוֹת כָּךְ - אֵין גְּבוּל לְמַה שֶׁתּוּכְלוּ לְקַבֵּל! 🎁🌊 כִּי אַתֶּם פּוֹנִים לִמְקוֹר הָרַחֲמִים הַבִּלְתִּי מֻגְבָּל שֶׁל הַשֵּׁם! 💫✨

מַה קוֹרֶה לְאָדָם שֶׁחַי בְּתְחוּשָׁה 'הַכֹּל מַגִּיעַ לִי'? 😤

מָה הַפָּסוּק מַבְטִיחַ לְמִי שֶׁחַי בְּהַכָּרַת טוֹב? 🌟

מַה הַמַּעֲשֶׂה הָרִאשׁוֹן שֶׁאֲנִי יָכוֹל לַעֲשׂוֹת כְּדֵי לָחִיּוֹת בְּהַכָּרַת טוֹב? 📝

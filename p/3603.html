מֹשֶׁה רַבֵּנוּ וְהַסְּנֶה הַבּוֹעֵר 🔥

מֹשֶׁה רַבֵּנוּ הָיָה רוֹעֶה צֹאן בַּמִּדְבָּר. 🐑 פִּתְאֹם הוּא רָאָה מַשֶּׁהוּ מוּזָר מְאוֹד - סְנֶה שֶׁבּוֹעֵר בָּאֵשׁ, אֲבָל לֹא נִשְׂרָף! 🌿🔥 מֹשֶׁה הִתְקָרֵב לִרְאוֹת, וְאָז קָרָא לוֹ הַקָּדוֹשׁ בָּרוּךְ הוּא מִתּוֹךְ הַסְּנֶה: "מֹשֶׁה, מֹשֶׁה!" 💖 הַקָּדוֹשׁ בָּרוּךְ הוּא אָמַר לְמֹשֶׁה: "רָאִיתִי אֶת צַעַר עַם יִשְׂרָאֵל בְּמִצְרַיִם. שָׁמַעְתִּי אֶת בִּכְיָם. 😢" "אֲנִי רוֹצֶה שֶׁתֵּלֵךְ לְפַרְעֹה וְתוֹצִיא אֶת עַמִּי מִמִּצְרַיִם!" ✨ מֹשֶׁה פָּחַד. הוּא אָמַר: "מִי אֲנִי? אֵיךְ אֲנִי אֵלֵךְ לְפַרְעֹה?" 😰 אֲבָל הַקָּדוֹשׁ בָּרוּךְ הוּא אָמַר לוֹ: "אַל תִּפְחַד! אֲנִי אֶהְיֶה עִמְּךָ תָּמִיד!" 💪 זֶה הָיָה הָרֶגַע שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא בָּחַר בְּמֹשֶׁה לִהְיוֹת הַמַּנְהִיג שֶׁל עַם יִשְׂרָאֵל! 👑

מָה מֹשֶׁה רַבֵּנוּ רָאָה בַּמִּדְבָּר?

נָכוֹן מְאוֹד! מֹשֶׁה רָאָה סְנֶה שֶׁבּוֹעֵר אֲבָל לֹא נִשְׂרָף - זֶה הָיָה נֵס! 🔥✨

מָה הַקָּדוֹשׁ בָּרוּךְ הוּא בִּקֵּשׁ מִמֹּשֶׁה לַעֲשׂוֹת?

מְצֻיָּן! הַקָּדוֹשׁ בָּרוּךְ הוּא בָּחַר בְּמֹשֶׁה לְהוֹצִיא אֶת עַם יִשְׂרָאֵל מִמִּצְרַיִם! 💪🌟

מַה הַקָּדוֹשׁ בָּרוּךְ הוּא הִבְטִיחַ לְמֹשֶׁה?

כָּל הַכָּבוֹד! הַקָּדוֹשׁ בָּרוּךְ הוּא הִבְטִיחַ לִהְיוֹת עִם מֹשֶׁה תָּמִיד. זֶה הַהַבְטָחָה הַכִּי חֲשׁוּבָה! 💖🙏

אֶהְיֶה אֲשֶׁר אֶהְיֶה - הַשֵּׁם תָּמִיד אִתָּנוּ 💫

מֹשֶׁה רַבֵּנוּ שָׁאַל שְׁאֵלָה חֲשׁוּבָה: "כְּשֶׁעַם יִשְׂרָאֵל יִשְׁאֲלוּ אוֹתִי - אֵיךְ קוֹרְאִים לַקָּדוֹשׁ בָּרוּךְ הוּא?" 🤔 הַקָּדוֹשׁ בָּרוּךְ הוּא עָנָה לוֹ בְּשֵׁם מְיֻחָד: "אֶהְיֶה אֲשֶׁר אֶהְיֶה" ✨ מָה זֶה אוֹמֵר? אֲנִי נִמְצָא אִתְּכֶם עַכְשָׁיו, וַאֲנִי אֶהְיֶה אִתְּכֶם גַּם בַּצָּרָה הַבָּאָה! 💖 זֶה אוֹמֵר שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא נִמְצָא אִתָּנוּ תָּמִיד - בִּשְׂמָחוֹת וְגַם בִּקְשָׁיִים! 🌈 יֵשׁ שִׁיר יָפֶה שֶׁאוֹמֵר: "הַשֵּׁם נִמְצָא בְּכָל הָעוֹלָם" 🎵 זֶה נָכוֹן! הַקָּדוֹשׁ בָּרוּךְ הוּא נִמְצָא בְּכָל מָקוֹם - בַּבַּיִת, בַּגַּן, בַּכִּתָּה, וּבַמִּשְׂחָקִים שֶׁלָּנוּ! 🏠🌳📚 כְּשֶׁאָנוּ מִתְפַּלְּלִים, אָנוּ יְכוֹלִים לְהַרְגִּישׁ שֶׁהַשֵּׁם קָרוֹב אֵלֵינוּ. 🙏💫 זֶה הַשֵּׁם הַכִּי יָפֶה - שֵׁם שֶׁמְּלַמֵּד אוֹתָנוּ שֶׁאֲנַחְנוּ לֹא לְבַד לְעוֹלָם! 👨‍👩‍👧‍👦✨

אֵיזֶה שֵׁם הַקָּדוֹשׁ בָּרוּךְ הוּא אָמַר לְמֹשֶׁה?

נִפְלָא! 'אֶהְיֶה אֲשֶׁר אֶהְיֶה' זֶה הַשֵּׁם שֶׁמְּלַמֵּד שֶׁהַשֵּׁם תָּמִיד אִתָּנוּ! 💫✨

מָה הַשֵּׁם 'אֶהְיֶה אֲשֶׁר אֶהְיֶה' מְלַמֵּד אוֹתָנוּ?

מְצֻיָּן! הַשֵּׁם תָּמִיד אִתָּנוּ - בַּבַּיִת, בַּכִּתָּה, וּבְכָל מָקוֹם! 🏠💖

מָתַי הַקָּדוֹשׁ בָּרוּךְ הוּא אוֹמֵר שֶׁהוּא יִהְיֶה אִתָּנוּ?

פַּצְצָה! הַשֵּׁם אִתָּנוּ תָּמִיד - בַּשְּׂמָחוֹת וּבַקְּשָׁיִים. הוּא אוֹהֵב אוֹתָנוּ תָּמִיד! 💪💖

פָּקוֹד פָּקַדְתִּי - גְּאֻלָּה כְּפוּלָה 🎯

הַקָּדוֹשׁ בָּרוּךְ הוּא אָמַר לְמֹשֶׁה מִלִּים מְיֻחָדוֹת: "פָּקוֹד פָּקַדְתִּי אֶתְכֶם" 💫 שִׂמְתֶּם לֵב? הַמִּלָּה "פָּקוֹד" חוֹזֶרֶת פַּעֲמַיִם! 🤔 זֶה בִּגְלַל שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא עָשָׂה שְׁנֵי דְּבָרִים גְּדוֹלִים לְעַם יִשְׂרָאֵל: ✨ דָּבָר רִאשׁוֹן: הוֹצִיא אוֹתָם מִמִּצְרַיִם - שִׁחְרֵר אוֹתָם מֵהָעֲבוֹדָה הַקָּשָׁה! 🏃‍♂️💨 דָּבָר שֵׁנִי: הִכָּה אֶת הַמִּצְרִים בְּעֶשֶׂר מַכּוֹת - הֵגֵן עַל עַם יִשְׂרָאֵל! 💪⚡ זֶה כְּמוֹ כְּשֶׁמִּישֶׁהוּ עוֹזֵר לָנוּ בִּשְׁנֵי דְּרָכִים: גַּם מוֹצִיא אוֹתָנוּ מִהַבֵּעָיָה, וְגַם דּוֹאֵג שֶׁלֹּא יִפְגְּעוּ בָּנוּ שׁוּב! 🛡️💖 הַקָּדוֹשׁ בָּרוּךְ הוּא לֹא רַק מוֹשִׁיעַ אוֹתָנוּ - הוּא גַּם שׁוֹמֵר עָלֵינוּ תָּמִיד! 🙏✨ זֶה מְלַמֵּד אוֹתָנוּ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא עוֹשֶׂה הַכֹּל בִּשְׁלֵמוּת וּבְאַהֲבָה! 💫💖

לָמָּה הַמִּלָּה 'פָּקוֹד' חוֹזֶרֶת פַּעֲמַיִם?

כָּל הַכָּבוֹד! הַכֶּפֶל מְלַמֵּד שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא עָשָׂה שְׁנֵי דְּבָרִים - הוֹצִיא וְהִגֵּן! 🎯💪

מָה הָיוּ שְׁנֵי הַדְּבָרִים שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא עָשָׂה?

נִפְלָא! הוֹצִיא מִמִּצְרַיִם וְהִכָּה אֶת הַמִּצְרִים - גְּאֻלָּה כְּפוּלָה! ⭐💫

מָה זֶה מְלַמֵּד אוֹתָנוּ עַל הַקָּדוֹשׁ בָּרוּךְ הוּא?

מְצֻיָּן! הַקָּדוֹשׁ בָּרוּךְ הוּא עוֹשֶׂה הַכֹּל בִּשְׁלֵמוּת וְשׁוֹמֵר עָלֵינוּ תָּמִיד! 🛡️💖

הַנֵּס שֶׁל חֵן הָעָם - אַהֲבָה בִּלְתִּי צְפוּיָה 💝

עַכְשָׁיו יְלָדִים, תִּשְׁמְעוּ נֵס מֻפְלָא! ✨ אַחֲרֵי כָּל הַמַּכּוֹת שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא הִכָּה אֶת מִצְרַיִם, מָה אַתֶּם חוֹשְׁבִים שֶׁהַמִּצְרִים הִרְגִּישׁוּ? 🤔 בְּטַח שׁוֹנְאִים אֶת עַם יִשְׂרָאֵל, נָכוֹן? ❌ אֲבָל לֹא! הַקָּדוֹשׁ בָּרוּךְ הוּא עָשָׂה נֵס מְיֻחָד: "וְנָתַתִּי אֶת חֵן הָעָם בְּעֵינֵי מִצְרַיִם" 💫 זֶה אוֹמֵר שֶׁפִּתְאוֹם - הַמִּצְרִים אָהֲבוּ אֶת הַיְּהוּדִים! 💖 הֵם נָתְנוּ לָהֶם בְּשִׂמְחָה כֶּסֶף, זָהָב, וּבְגָדִים יְקָרִים! 👑💎 אֵיךְ זֶה יָכוֹל לִהְיוֹת? כִּי הַקָּדוֹשׁ בָּרוּךְ הוּא יָכוֹל לְשַׁנּוֹת לְבָבוֹת בְּרֶגַע! 🌟 זֶה מְלַמֵּד אוֹתָנוּ שֶׁאֵין דָּבָר שֶׁקָּשֶׁה עַל הַקָּדוֹשׁ בָּרוּךְ הוּא. הוּא יָכוֹל לַעֲשׂוֹת נִסִּים בְּכָל רֶגַע! 🙏✨ גַּם הַיּוֹם, הַקָּדוֹשׁ בָּרוּךְ הוּא יָכוֹל לַעֲזוֹר לָנוּ בְּדְרָכִים שֶׁלֹּא חָשַׁבְנוּ עֲלֵיהֶן! 💪💖

מָה קָרָה לַמִּצְרִים אַחֲרֵי הַמַּכּוֹת?

וָאוֹ! זֶה הַנֵּס - הַמִּצְרִים אָהֲבוּ אֶת הַיְּהוּדִים וְנָתְנוּ מַתָּנוֹת! 🎁💖

מָה הַמִּצְרִים נָתְנוּ לַיְּהוּדִים?

נָכוֹן! הֵם נָתְנוּ כֶּסֶף, זָהָב וּבְגָדִים - הַרְבֵּה מָאוֹד עֹשֶׁר! 💎👑✨

מָה הַנֵּס הַזֶּה מְלַמֵּד אוֹתָנוּ?

מְצֻיָּן! הַקָּדוֹשׁ בָּרוּךְ הוּא יָכוֹל לְשַׁנּוֹת לְבָבוֹת וְלַעֲשׂוֹת נִסִּים! 🌟💫

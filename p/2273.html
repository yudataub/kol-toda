שִׁבְעַת הַמִּינִים טוּ בִּשְׁבָט

⛶ פתח במסך מלא
📄 תוכן העמוד בטקסט

🌾 מָה זֶה שִׁבְעַת הַמִּינִים?

🎉 בּוֹא נִלְמַד עַל הַמַּתָּנָה הַכִי מְיֻחֶדֶת שֶׁהַשֵּׁם נָתַן לְאֶרֶץ יִשְׂרָאֵל! 💚 הַתּוֹרָה אוֹמֶרֶת: ⭐ "אֶרֶץ חִטָּה וּשְׂעֹרָה וְגֶפֶן וּתְאֵנָה וְרִמּוֹן, אֶרֶץ זֵית שֶׁמֶן וּדְבָשׁ" בּוֹא נִסְפֹּר - כַּמָּה מִינִים יֵשׁ פֹּה? 🤔 🌾 1. חִטָּה 🌾 2. שְׂעֹרָה 🍇 3. גֶּפֶן (עֲנָבִים) 🫐 4. תְּאֵנָה 🍎 5. רִמּוֹן 🫒 6. זַיִת 🌴 7. תָּמָר (דְּבַשׁ = תְּמָרִים מְתוּקִים!) בְּדִיּוּק 7 מִינִים! 🌟 לָמָּה דַּוְקָא 7? 💭 כִּי 7 זֶה מִסְפָּר קָדוֹשׁ! ✨ 7 יְמֵי הַשָּׁבוּעַ, שַׁבָּת בַּיּוֹם הַ-7, שְׁמִיטָה כָּל 7 שָׁנִים! 📅 וְאֶרֶץ יִשְׂרָאֵל נִשְׁתַּבְּחָה בְּ 7 מִינִים מְיֻחָדִים! 🏆 אֵלּוּ הַפֵּרוֹת שֶׁבָּהֶם הַשֵּׁם בֵּרַךְ אֶת אֶרֶץ יִשְׂרָאֵל! 💖

כַּמָּה מִינִים יֵשׁ בְּשִׁבְעַת הַמִּינִים?

מְצֻיָּן! 🌟 בְּדִיּוּק 7 מִינִים! חִטָּה, שְׂעֹרָה, גֶּפֶן, תְּאֵנָה, רִמּוֹן, זַיִת וְתָמָר. הַשֵּׁם שָׂמֵחַ כְּשֶׁאַתָּה זוֹכֵר אוֹתָם! 💖

אֵיזֶה פָּסוּק מַזְכִּיר אֶת שִׁבְעַת הַמִּינִים?

נִפְלָא! ⭐ הַפָּסוּק 'אֶרֶץ חִטָּה וּשְׂעֹרָה...' מוֹנֶה אֶת כָּל שִׁבְעַת הַמִּינִים! זֶה הַמַּתָּנָה שֶׁל הַשֵּׁם לָנוּ! הוּא אוֹהֵב אוֹתְךָ! 💚

לָמָּה דַּוְקָא 7 מִינִים?

אַלּוּף! 🏆 7 זֶה מִסְפָּר קָדוֹשׁ - 7 יָמִים, שַׁבָּת, שְׁמִיטָה! לָכֵן גַּם אֶרֶץ יִשְׂרָאֵל נִשְׁתַּבְּחָה בְּ-7 מִינִים קְדוֹשִׁים! הַשֵּׁם גֵּאֶה בְּךָ! 💪

🎁 לָמָּה מְבִיאִים בִּכּוּרִים רַק מִשִּׁבְעַת הַמִּינִים?

עַכְשָׁיו בּוֹא נִלְמַד מַשֶּׁהוּ מְרַגֵּשׁ! 💖 מָה זֶה בִּכּוּרִים? 🤔 בִּכּוּרִים = הַפֵּרוֹת הָרִאשׁוֹנִים שֶׁגָּדְלוּ בַּשָּׂדֶה! 🌱✨ בִּזְמַן שֶׁבֵּית הַמִּקְדָּשׁ הָיָה קַיָּם, הָיוּ מְבִיאִים אֶת הַפֵּרוֹת הָרִאשׁוֹנִים לִירוּשָׁלַיִם! 🏛️ לָמָּה? כְּדֵי לְהוֹדוֹת לַשֵּׁם! 🙏 הָאִכָּר שָׂמֵחַ וְאָמַר: "תּוֹדָה לְךָ הַשֵּׁם שֶׁנָּתַתָּ לִי פֵּרוֹת! 💚" אֲבָל שִׂימוּ לֵב - לֹא מְבִיאִים בִּכּוּרִים מִכָּל פְּרִי! ❌ רַק מִשִּׁבְעַת הַמִּינִים! ⭐ לָמָּה דַּוְקָא מֵהֶם? 🤔 כִּי בָּהֶם נִשְׁתַּבְּחָה אֶרֶץ יִשְׂרָאֵל! 🌟 כְּשֶׁאֲנִי רוֹצֶה לְהוֹדוֹת לַשֵּׁם, אֲנִי מֵבִיא לוֹ אֶת הַמַּתָּנָה הַכִי טוֹבָה! 🎁 וּמָה הַמַּתָּנָה הַכִי טוֹבָה? שִׁבְעַת הַמִּינִים שֶׁבָּהֶם הִשְׁתַּבְּחָה הָאָרֶץ! 💖 לֹא תָפוּחַ, לֹא בָּנָנָה, לֹא תַּפּוּז - רַק שִׁבְעַת הַמִּינִים! 🌾🍇🫒 זֶה אוֹמֵר לַשֵּׁם: "אֲנִי מֵבִיא לְךָ אֶת הַכִי מְיֻחָד שֶׁיֵּשׁ לִי!" 💕

מָה זֶה בִּכּוּרִים?

מְעֻלֶּה! 🌟 בִּכּוּרִים זֶה הַפֵּרוֹת הָרִאשׁוֹנִים שֶׁמְּבִיאִים לַשֵּׁם בִּירוּשָׁלַיִם כְּדֵי לְהוֹדוֹת! זוֹ דֶּרֶךְ לוֹמַר 'תּוֹדָה'! הַשֵּׁם שָׂמֵחַ מִמְּךָ! 💖

מִמָּה מְבִיאִים בִּכּוּרִים?

נִפְלָא! ⭐ רַק מִשִּׁבְעַת הַמִּינִים! לֹא מִתַּפּוּחַ, לֹא מִבָּנָנָה - רַק מֵהַמִּינִים שֶׁבָּהֶם נִשְׁתַּבְּחָה הָאָרֶץ! הַשֵּׁם אוֹהֵב אוֹתְךָ! 💚

לָמָּה מְבִיאִים בִּכּוּרִים דַּוְקָא מִשִּׁבְעַת הַמִּינִים?

פְּצָצָה! 🎯 כְּשֶׁמּוֹדִים לַשֵּׁם, מְבִיאִים לוֹ אֶת הַכִי מְיֻחָד! שִׁבְעַת הַמִּינִים הֵם הַמַּתָּנָה שֶׁל הָאָרֶץ! הַשֵּׁם גֵּאֶה בְּךָ! 💪

🍇 בְּרָכָה רִאשׁוֹנָה - אֵיזֶה פְּרִי לְבָרֵךְ קֹדֶם?

עַכְשָׁיו בּוֹא נִלְמַד חֹק מְעַנְיֵן! 🤓 תְּדַמְיֵן: יֵשׁ לְךָ עַל הַשֻּׁלְחָן תַּפּוּחַ 🍎 וְתָמָר 🌴 עַל אֵיזֶה פְּרִי תְּבָרֵךְ רִאשׁוֹן? 🤔 עַל הַתָּמָר! ⭐ לָמָּה? כִּי הוּא מִשִּׁבְעַת הַמִּינִים! 🌟 חֲכָמִים קָבְעוּ: שִׁבְעַת הַמִּינִים קוֹדְמִים לִבְרָכָה! 💚 גַּם אִם אַתָּה יוֹתֵר אוֹהֵב תַּפּוּחַ - עַל הַתָּמָר מְבָרְכִים קֹדֶם! 😊 וּמָה אִם יֵשׁ לְךָ שְׁנֵי פֵּרוֹת מִשִּׁבְעַת הַמִּינִים? 🍇🫒 אָז יֵשׁ סֵדֶר מְיֻחָד! 📋 הַסֵּדֶר לְפִי הַפָּסוּק: 🫒 זַיִת - קָרוֹב לְמִלָּה "אֶרֶץ" בַּפָּסוּק! 🌴 תָּמָר - גַּם הוּא קָרוֹב! 🍇 גֶּפֶן (עֲנָבִים) 🫐 תְּאֵנָה 🍎 רִמּוֹן 🌾 חִטָּה וּשְׂעֹרָה אָז לְדֻגְמָה: יֵשׁ לְךָ רִמּוֹן וְזַיִת - עַל הַזַּיִת מְבָרְכִים קֹדֶם! 🫒 לָמָּה? כִּי הוּא יוֹתֵר קָרוֹב לְמִלָּה "אֶרֶץ" בַּפָּסוּק! 📖 זוֹ דֶּרֶךְ לְהַרְאוֹת כָּבוֹד לַפֵּרוֹת שֶׁל אֶרֶץ יִשְׂרָאֵל! 💖

יֵשׁ לִי תַּפּוּחַ וְתָמָר - עַל מָה אֲבָרֵךְ רִאשׁוֹן?

מְצֻיָּן! 🌟 עַל הַתָּמָר! כִּי שִׁבְעַת הַמִּינִים קוֹדְמִים לִבְרָכָה - גַּם אִם אַתָּה יוֹתֵר אוֹהֵב תַּפּוּחַ! הַשֵּׁם שָׂמֵחַ מֵהַכָּבוֹד שֶׁלְּךָ! 💖

אֵיזֶה פֵּרוֹת קוֹדְמִים לִבְרָכָה?

נִפְלָא! ⭐ שִׁבְעַת הַמִּינִים תָּמִיד קוֹדְמִים - כִּי בָּהֶם נִשְׁתַּבְּחָה אֶרֶץ יִשְׂרָאֵל! זוֹ דֶּרֶךְ לְהַרְאוֹת כָּבוֹד! הַשֵּׁם אוֹהֵב אוֹתְךָ! 💚

יֵשׁ לִי רִמּוֹן וְזַיִת - עַל מָה אֲבָרֵךְ קֹדֶם?

אַלּוּף! 🏆 עַל הַזַּיִת! כִּי יֵשׁ סֵדֶר - כֹּל מַה שֶּׁיּוֹתֵר קָרוֹב לְ'אֶרֶץ' בַּפָּסוּק, יוֹתֵר חָשׁוּב! זֶה מַרְאֶה אַהֲבָה! הַשֵּׁם גֵּאֶה בְּךָ! 💪

🙏 בְּרָכָה אַחֲרוֹנָה - עַל הָאָרֶץ וְעַל פֵּרוֹתֶיהָ

וְעַכְשָׁיו הַחֵלֶק הָאַחֲרוֹן וְהַמְּרַגֵּשׁ! 💖 אַחֲרֵי שֶׁאָכַלְנוּ פְּרִי - מָה עוֹשִׂים? מְבָרְכִים בְּרָכָה אַחֲרוֹנָה! 🙏 עַל רֹב הַפֵּרוֹת אוֹמְרִים: "בּוֹרֵא נְפָשׁוֹת רַבּוֹת..." 📖 אֲבָל עַל שִׁבְעַת הַמִּינִים? בְּרָכָה מְיֻחֶדֶת! ⭐ אוֹמְרִים: "עַל הָאָרֶץ וְעַל פֵּרוֹתֶיהָ" 💚 (אוֹ "עַל הַמִּחְיָה" עַל חִטָּה וּשְׂעֹרָה, "עַל הַגֶּפֶן" עַל יַיִן וַעֲנָבִים) לָמָּה בְּרָכָה מְיֻחֶדֶת? 🤔 כִּי שִׁבְעַת הַמִּינִים הֵם הַמַּתָּנָה הַמְּיֻחֶדֶת שֶׁל אֶרֶץ יִשְׂרָאֵל! 🎁 כְּשֶׁמְּבָרְכִים "עַל הָאָרֶץ וְעַל פֵּרוֹתֶיהָ" - אָנוּ אוֹמְרִים: 💕 "תּוֹדָה לְךָ הַשֵּׁם עַל אֶרֶץ יִשְׂרָאֵל וְעַל הַפֵּרוֹת הַמְּיֻחָדִים שֶׁלָּהּ!" 🌈 זוֹ בְּרָכָה מְלֵאַת אַהֲבָה - אַהֲבָה לָאָרֶץ וְאַהֲבָה לַשֵּׁם! 💖 לְדֻגְמָה: 📝 אָכַלְתָּ תַּפּוּז? → "בּוֹרֵא נְפָשׁוֹת..." אָכַלְתָּ תָּמָר? → "עַל הָאָרֶץ וְעַל פֵּרוֹתֶיהָ!" ⭐ רוֹאִים? שִׁבְעַת הַמִּינִים מְיֻחָדִים מֵהַתְּחָלָה וְעַד הַסּוֹף! 🌟 מִבִּכּוּרִים, דֶּרֶךְ בְּרָכָה רִאשׁוֹנָה, וְעַד בְּרָכָה אַחֲרוֹנָה! 🙌

אֵיזוֹ בְּרָכָה אַחֲרוֹנָה עַל שִׁבְעַת הַמִּינִים?

מְעֻלֶּה! 🌟 'עַל הָאָרֶץ וְעַל פֵּרוֹתֶיהָ' - בְּרָכָה מְיֻחֶדֶת! כִּי שִׁבְעַת הַמִּינִים מְיֻחָדִים! הַשֵּׁם שָׂמֵחַ מִבִּרְכוֹתֶיךָ! 💖

מָה אוֹמְרִים בִּבְרָכָה הַמְּיֻחֶדֶת?

נִפְלָא! ⭐ 'עַל הָאָרֶץ וְעַל פֵּרוֹתֶיהָ!' זוֹ דֶּרֶךְ לְהוֹדוֹת לַשֵּׁם עַל אֶרֶץ יִשְׂרָאֵל וְעַל הַפֵּרוֹת שֶׁלָּהּ! הַשֵּׁם אוֹהֵב אוֹתְךָ! 💚

לָמָּה בְּרָכָה מְיֻחֶדֶת דַּוְקָא לְשִׁבְעַת הַמִּינִים?

פְּצָצָה! 🎯 כִּי הֵם הַמַּתָּנָה הַמְּיֻחֶדֶת שֶׁל הָאָרֶץ! לָכֵן מְבָרְכִים עֲלֵיהֶם בְּרָכָה מְיֻחֶדֶת מְלֵאַת אַהֲבָה! הַשֵּׁם גֵּאֶה בְּךָ! 💪

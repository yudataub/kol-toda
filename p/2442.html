חֵלֶק א׳

לָמָּה הַפָּסוּק חוֹזֵר פַּעֲמַיִם? 🌊

בְּפָרָשַׁת בְּשַׁלַּח, כְּשֶׁבְּנֵי יִשְׂרָאֵל יוֹצְאִים מִמִּצְרַיִם, קוֹרֶה מַשֶּׁהוּ מִדְהִים — הַיָּם נִבְקָע! 🌊✨ אֲבָל כְּשֶׁאֲנַחְנוּ קוֹרְאִים אֶת הַתּוֹרָה, אֲנַחְנוּ רוֹאִים שֶׁיֵּשׁ שְׁנֵי פְּסוּקִים שֶׁמְּסַפְּרִים עַל הַמַּעֲבָר בַּיָּם! פָּסוּק כ״ב: "וַיָּבֹאוּ בְנֵי יִשְׂרָאֵל בְּתוֹךְ הַיָּם בַּיַּבָּשָׁה וְהַמַּיִם לָהֶם חוֹמָה מִימִינָם וּמִשְּׂמֹאלָם" פָּסוּק כ״ט: "וּבְנֵי יִשְׂרָאֵל הָלְכוּ בַּיַּבָּשָׁה בְּתוֹךְ הַיָּם וְהַמַּיִם לָהֶם חֹמָה מִימִינָם וּמִשְּׂמֹאלָם" שְׁאֵלָה טוֹבָה: לָמָּה הַתּוֹרָה אוֹמֶרֶת אֶת אוֹתוֹ הַדָּבָר פַּעֲמַיִם? 🤔 💡 בַּתּוֹרָה אֵין מִלָּה מְיֻתֶּרֶת! כָּל מִלָּה, כָּל פְּסוּק — מְלַמֵּד אוֹתָנוּ מַשֶּׁהוּ חָשׁוּב וּמְיֻחָד. כָּאן מִתְחַבֵּא סוֹד עָמֹק עַל שְׁנֵי סוּגִים שֶׁל יְהוּדִים! 🌟 הַשֵּׁם אוֹהֵב כָּל תַּלְמִיד שֶׁשּׁוֹאֵל שְׁאֵלוֹת טוֹבוֹת! אַתָּה כְּבָר בְּדֶרֶךְ לְגַלּוֹת אֶת הַסּוֹד!

כַּמָּה פְּעָמִים מְסַפֶּרֶת הַתּוֹרָה עַל מַעֲבָר בְּנֵי יִשְׂרָאֵל בַּיָּם?

פַּעַם אַחַת בְּלֶבֶד בְּפָרָשָׁה

שְׁנֵי פְּסוּקִים מְסַפְּרִים עַל הַמַּעֲבָר

שָׁלֹשׁ פְּעָמִים בְּשָׁלֹשׁ פָּרָשִׁיּוֹת

אַרְבַּע פְּעָמִים בְּסֵפֶר שְׁמוֹת

הַתּוֹרָה כּוֹתֶבֶת שְׁנֵי פְּסוּקִים (כ״ב וְכ״ט) שֶׁמְּסַפְּרִים עַל הַמַּעֲבָר. כָּל פָּסוּק מְגַלֶּה סוֹד אַחֵר!

מַה הָעִקָּרוֹן הַחָשׁוּב בִּקְרִיאַת הַתּוֹרָה שֶׁלָּמַדְנוּ?

כָּל מִלָּה בַּתּוֹרָה מְלַמֶּדֶת מַשֶּׁהוּ חָשׁוּב

פְּסוּקִים חוֹזְרִים כְּדֵי שֶׁנִּזְכֹּר אוֹתָם

הַתּוֹרָה חוֹזֶרֶת כִּי הַסּוֹפֵר טָעָה

פְּסוּקִים חוֹזְרִים כְּדֵי לְמַלֵּא מָקוֹם בַּסֵּפֶר

בַּתּוֹרָה אֵין מִלָּה מְיֻתֶּרֶת — כָּל חֲזָרָה מְלַמֶּדֶת דָּבָר חָדָשׁ וְחָשׁוּב!

מִי הִיה הָרִאשׁוֹן שֶׁקָּפַץ לַיָּם לִפְנֵי כּוּלָּם?

מֹשֶׁה רַבֵּנוּ שֶׁהֵרִים אֶת מַטֵּהוּ

אַהֲרֹן שֶׁהָיָה כֹּהֵן גָּדוֹל בְּיִשְׂרָאֵל

נַחְשׁוֹן בֶּן עַמִּינָדָב מִשֵּׁבֶט יְהוּדָה

כָּלֵב בֶּן יְפֻנֶּה הַגִּבּוֹר וְהֶאָמִיץ

נַחְשׁוֹן בֶּן עַמִּינָדָב מִשֵּׁבֶט יְהוּדָה קָפַץ לַיָּם בְּאֵימוּן מָלֵא! אַחֲרָיו קָפַץ כָּל שֵׁבֶט יְהוּדָה!

הַסּוֹד שֶׁל הַסֵּדֶר — יַבָּשָׁה אוֹ יָם? 🌊🏝️

עַכְשָׁו נִגַּשׁ לַסּוֹד הָאֲמִתִּי! 🔍✨ אִם תִּקְרָא שׁוֹב אֶת שְׁנֵי הַפְּסוּקִים, תִּמְצָא הֶבְדֵּל קָטָן אֲבָל עָצוּם: פָּסוּק כ״ב (רִאשׁוֹן): " בְּתוֹךְ הַיָּם בַּיַּבָּשָׁה " — קוֹדֶם יָם, אַחַר כָּךְ יַבָּשָׁה פָּסוּק כ״ט (שֵׁנִי): " בַּיַּבָּשָׁה בְּתוֹךְ הַיָּם " — קוֹדֶם יַבָּשָׁה, אַחַר כָּךְ יָם 🌊 הַ"קוֹל אֵלִיָּהוּ" מְסַבִּיר: הַסֵּדֶר בַּפָּסוּק מְגַלֶּה אֵיךְ כָּל אֶחָד חָוָה אֶת הַנֵּס! מִי שֶׁנֶּכְנַס לַיָּם קוֹדֶם — חָוָה קוֹדֶם אֶת הַיָּם, וְאַחַר כָּךְ אֶת הַיַּבָּשָׁה. מִי שֶׁחִכָּה לִרְאוֹת יַבָּשָׁה — רָאָה קוֹדֶם יַבָּשָׁה. זֶה לֹא רַק הֶבְדֵּל בַּמִּלִּים — זֶה הֶבְדֵּל בִּלְבָבוֹת! 💫 💭 אֵיךְ אַתָּה חוֹשֵׁב — קָל יוֹתֵר לָלֶכֶת לְמַיִם עֲמֻקִּים לְפָנֵי שֶׁרוֹאִים יַבָּשָׁה, אוֹ אַחֲרֵי? מָה הַשֵּׁם רוֹצֶה מִמֶּנּוּ?

בְּפָסוּק כ״ב — מָה כָּתוּב קוֹדֶם: יָם אוֹ יַבָּשָׁה?

קוֹדֶם יָם, אַחַר כָּךְ יַבָּשָׁה

קוֹדֶם יַבָּשָׁה, אַחַר כָּךְ יָם

שְׁנֵיהֶם יַחַד בְּאוֹתוֹ זְמַן

הַפָּסוּק לֹא מְזַכִּיר יַבָּשָׁה כְּלָל

פָּסוּק כ״ב אוֹמֵר: 'בְּתוֹךְ הַיָּם בַּיַּבָּשָׁה' — קוֹדֶם יָם, אַחַר כָּךְ יַבָּשָׁה!

מָה מְגַלֶּה לָנוּ הַסֵּדֶר הַשׁוֹנֶה בִּשְׁנֵי הַפְּסוּקִים?

אֵיךְ כָּל אֶחָד חָוָה אֶת הַנֵּס אֲחֶרֶת

שֶׁהַיָּם נִבְקַע פַּעֲמַיִם שְׁתֵּי פְּעָמִים

שֶׁהָיוּ שְׁנֵי כְּבִישִׁים שׁוֹנִים בַּיָּם

שֶׁחֲצִי הָעָם עָבַר בַּבֹּקֶר וַחֲצִי בַּלַּיְלָה

הַסֵּדֶר מְגַלֶּה אֶת חֲוָיַת כָּל אֶחָד — מִי נִכְנַס קוֹדֶם לַיָּם וּמִי הִמְתִּין לִרְאוֹת יַבָּשָׁה!

בְּפָסוּק כ״ט — מָה כָּתוּב קוֹדֶם: יַבָּשָׁה אוֹ יָם?

קוֹדֶם יַבָּשָׁה, אַחַר כָּךְ יָם

קוֹדֶם יָם, אַחַר כָּךְ יַבָּשָׁה

רַק יַבָּשָׁה בְּלִי אֲזְכַּרַת יָם

רַק יָם בְּלִי אֲזְכַּרַת יַבָּשָׁה

פָּסוּק כ״ט אוֹמֵר: 'בַּיַּבָּשָׁה בְּתוֹךְ הַיָּם' — קוֹדֶם יַבָּשָׁה, אַחַר כָּךְ יָם!

חֵלֶק ג׳

שְׁנֵי סוּגֵי יְהוּדִים — אֱמוּנָה וּבִטָּחוֹן 🙏

הַ"קוֹל אֵלִיָּהוּ" מְגַלֶּה לָנוּ סוֹד מַדְהִים עַל בְּנֵי יִשְׂרָאֵל! 🌟 הָיוּ שְׁנֵי סוּגִים שֶׁל אֲנָשִׁים בְּיוֹם קְרִיעַת יַם סוּף: 🌊 סוּג רִאשׁוֹן — אַנְשֵׁי בִּטָּחוֹן: צַדִּיקִים שֶׁהֶאֱמִינוּ בַּשֵּׁם בְּלֵב שָׁלֵם! הֵם הָלְכוּ לַיָּם עַד שֶׁהַמַּיִם הִגִּיעוּ עַד פֶּה. הֵם לֹא חִכּוּ לִרְאוֹת יַבָּשָׁה — הֵם פָּשׁוּט סָמְכוּ עַל הַשֵּׁם! לָהֶם — קוֹדֶם יָם, אַחַר כָּךְ יַבָּשָׁה. 🏝️ סוּג שֵׁנִי — אַנְשֵׁי רְאִיָּה: הֵם לֹא הִסְפִּיקוּ לִבְטֹּחַ כָּל כָּךְ — הֵם הִמְתִּינוּ עַד שֶׁרָאוּ בְּעֵינֵיהֶם אֶת הַיַּבָּשָׁה! רַק אֶחֱרֵי שֶׁרָאוּ — נִכְנְסוּ. לָהֶם — קוֹדֶם יַבָּשָׁה, אַחַר כָּךְ יָם. הַמִּדְרָשׁ מְסַפֵּר שֶׁהַמַּלְאָכִים שָׁאֲלוּ: "מַה נִּשְׁתַּנּוּ אֵלּוּ מֵאֵלּוּ?" 🤔 וְהַשֵּׁם עָנָה: "זְכוּת הָאֱמוּנָה וְהַבִּטָּחוֹן כְּדַאי לָהֶם!" — הָאֱמוּנָה עָשְׂתָה אֶת הֶהֶבְדֵּל! 💭 אֵיךְ אֲנִי מַרְגִּישׁ אֶת אַהֲבַת הַשֵּׁם כְּשֶׁאֲנִי סוֹמֵךְ עָלָיו גַּם לִפְנֵי שֶׁרוֹאֶה?

אֵיזֶה סוּג אֲנָשִׁים נִכְנַס לַיָּם לִפְנֵי שֶׁרָאָה יַבָּשָׁה?

אַנְשֵׁי בִּטָּחוֹן וְאֱמוּנָה חֲזָקָה

אַנְשֵׁי כֹּחַ וּגְבוּרָה גּוּפָנִית

אַנְשֵׁי עֹשֶׁר שֶׁהִשְׁקִיעוּ כֶּסֶף רַב

אַנְשֵׁי חָכְמָה שֶׁחָשְׁבוּ הַרְבֵּה

אַנְשֵׁי הַבִּטָּחוֹן — הַצַּדִּיקִים שֶׁהֶאֱמִינוּ — נִכְנְסוּ לַיָּם לִפְנֵי שֶׁרָאוּ יַבָּשָׁה!

מַה עָנָה הַשֵּׁם לַמַּלְאָכִים שֶׁשָּׁאֲלוּ: 'מַה נִּשְׁתַּנּוּ אֵלּוּ מֵאֵלּוּ'?

זְכוּת הָאֱמוּנָה וְהַבִּטָּחוֹן כְּדַאי לָהֶם

הֵם עָשׂוּ מִצְווֹת יוֹתֵר מִכָּל הָאֲחֵרִים

הֵם מִשְׁפַּחַת נַחְשׁוֹן הַמְיֻחֶדֶת

הֵם לָמְדוּ הַרְבֵּה תוֹרָה בְּמִצְרַיִם

הַשֵּׁם אָמַר: 'זְכוּת הָאֱמוּנָה וְהַבִּטָּחוֹן כְּדַאי לָהֶם' — זוֹ זְכוּת גְּדוֹלָה!

מָה עָשׂוּ אַנְשֵׁי הַסּוּג הַשֵּׁנִי לִפְנֵי שֶׁנִּכְנְסוּ לַיָּם?

הִמְתִּינוּ עַד שֶׁרָאוּ יַבָּשָׁה בְּעֵינֵיהֶם

שָׁרוּ שִׁירִים לְהַשֵּׁם בְּקוֹל גָּדוֹל

הִסְתַּכְּלוּ עַל מֹשֶׁה וְחִכּוּ לְאוֹת

הִתְפַּלְּלוּ וְצָמוּ לִפְנֵי הַכְּנִיסָה

הֵם הִמְתִּינוּ לִרְאוֹת יַבָּשָׁה בְּעֵינֵיהֶם — וְרַק אָז נִכְנְסוּ. זֶה הַהֶבְדֵּל הַגָּדוֹל!

חֵלֶק ד׳

חֹמָה אוֹ חמה — שׁוֹנֶה בְּנִיקּוּד! 📜

יֵשׁ עוֹד הֶבְדֵּל קָטָן וּמַדְהִים בֵּין שְׁנֵי הַפְּסוּקִים! 😮 פָּסוּק כ״ב (רִאשׁוֹן): "וְהַמַּיִם לָהֶם חוֹמָה " — עִם ו' וְנִיקּוּד מָלֵא ✅ פָּסוּק כ״ט (שֵׁנִי): "וְהַמַּיִם לָהֶם חֹמָה " — חָסֵר ו', נִיקּוּד שׁוֹנֶה 🤔 💡 הַ"קוֹל אֵלִיָּהוּ" מְסַבִּיר: לַמַּאֲמִינִים שֶׁנִּכְנְסוּ קוֹדֶם — הַמַּיִם הָיוּ לָהֶם חוֹמָה מְלֵאָה, בְּטוּחָה, שְׁלֵמָה. לִשְׁאָר — הָיָה קִטְרוּג: מַה נִּשְׁתַּנּוּ אֵלּוּ מֵאֵלּוּ? לָכֵן הַמִּלָּה חָסֵר. כְּלוֹמַר — הַנִּיקּוּד מְגַלֶּה אֶת הָרוּחָנִיּוּת! 🌟 אַנְשֵׁי הָאֱמוּנָה קִבְּלוּ נֵס שָׁלֵם — חוֹמָה מְלֵאָה. אַנְשֵׁי הָרְאִיָּה קִבְּלוּ גַּם הֵם נֵס, אֲבָל הָיָה קִטְרוּג — לָכֵן חֹמָה חֲסֵרָה. 💖 הַשּׁוֹרָה הַתַּחְתּוֹנָה: הַשֵּׁם הִצִּיל אֶת כָּל בְּנֵי יִשְׂרָאֵל — גַּם הַמַּאֲמִינִים וְגַּם אֵלֶּה שֶׁחִכּוּ לִרְאוֹת! הוּא אוֹהֵב אֶת כָּל עַמּוֹ! 🌊 אֵיךְ אֲנִי יָכוֹל לְחַזֵּק אֶת הָאֱמוּנָה שֶׁלִּי כָּל יוֹם, כָּל רֶגַע?

מָה הַהֶבְדֵּל בֵּין 'חוֹמָה' בְּפָסוּק כ״ב לְ'חֹמָה' בְּפָסוּק כ״ט?

חוֹמָה מָלֵא בְּו', חֹמָה חָסֵר ו'

שְׁנֵי הַפְּסוּקִים בְּאוֹתוֹ הַנִּיקּוּד

חוֹמָה פֵּרוּשָׁהּ קִיר, חֹמָה פֵּרוּשָׁהּ שְׁמִירָה

חֹמָה גְּדוֹלָה יוֹתֵר מֵחוֹמָה

בְּפָסוּק כ״ב: 'חוֹמָה' עִם ו', מָלֵא. בְּפָסוּק כ״ט: 'חֹמָה' בְּלִי ו', חָסֵר. הַהֶבְדֵּל הַקָּטָן מְגַלֶּה הַרְבֵּה!

לְמִי הָיְתָה חוֹמָה מְלֵאָה (עִם ו') — לְאַנְשֵׁי הַבִּטָּחוֹן אוֹ לַמַּמְתִּינִים?

לְאַנְשֵׁי הַבִּטָּחוֹן שֶׁנִּכְנְסוּ קוֹדֶם

לַמַּמְתִּינִים שֶׁחִכּוּ לִרְאוֹת יַבָּשָׁה

לִשְׁנֵיהֶם בִּשְׁוֶה בְּאוֹתָה מִדָּה

לְאֵלֶּה שֶׁנִּכְנְסוּ אַחֲרוֹנִים לַיָּם

לְאַנְשֵׁי הַבִּטָּחוֹן — חוֹמָה מְלֵאָה! הָאֱמוּנָה שֶׁלָּהֶם הֵבִיאָה נֵס שָׁלֵם!

מָה לָמַדְנוּ מֵהַנִּיקּוּד הַשּׁוֹנֶה עַל אַהֲבַת הַשֵּׁם לְעַמּוֹ?

הַשֵּׁם הִצִּיל אֶת כָּל עַמּוֹ — גַּם הַמַּאֲמִינִים

הַשֵּׁם הִצִּיל רַק אֶת הַצַּדִּיקִים הַגְּמוּרִים

רַק אַנְשֵׁי הָאֱמוּנָה עָבְרוּ בְּשָׁלוֹם

הַשֵּׁם עָנַשׁ אֶת הַמַּמְתִּינִים עַל חִכּוּיָם

הַשֵּׁם הִצִּיל אֶת כָּל בְּנֵי יִשְׂרָאֵל — גַּם מִי שֶׁחִכָּה לִרְאוֹת! הוּא אוֹהֵב אֶת כָּל עַמּוֹ! 💖

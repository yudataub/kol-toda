הַכְנָסַת אוֹרְחִים

⛶ פתח במסך מלא
📄 תוכן העמוד בטקסט

🌟 אַבְרָהָם אָבִינוּ - מֶלֶךְ הַחֶסֶד

אַבְרָהָם אָבִינוּ הָיָה הָאִישׁ הַכִּי טוֹב בָּעוֹלָם! 💖 הוּא אָהַב לְקַבֵּל אוֹרְחִים יוֹתֵר מִכֹּל דָּבָר אַחֵר. הַתּוֹרָה מְסַפֶּרֶת שֶׁהוּא יָשַׁב בְּפֶתַח הָאֹהֶל בְּחֹם הַיּוֹם וְחִיפֵּשׂ אוֹרְחִים שֶׁיּוּכַל לְאָרֵחַ! 🏕️✨ תַּאֲרוּ לָכֶם: אַבְרָהָם הָיָה חוֹלֶה, בַּיּוֹם הַשְּׁלִישִׁי אַחֲרֵי הַמִּילָה, וְהַחֹם הָיָה נוֹרָא! 🌡️ אֲבָל הוּא לֹא הִסְתַּכֵּל עַל עַצְמוֹ - הוּא רָצָה לַעֲשׂוֹת חֶסֶד! 💪 כְּשֶׁרָאָה שְׁלֹשָׁה אוֹרְחִים מֵרָחוֹק, הוּא רָץ לִקְרָאתָם וְהִשְׁתַּחֲוָה לָהֶם! 🏃‍♂️ אַבְרָהָם לֹא רַק הִזְמִין אוֹתָם - הוּא עָשָׂה לָהֶם סְעוּדָה מַלְכוּתִית! 🍖🍞 הוּא רָץ לִבְחוֹר עֵגֶל רַךְ וָטוֹב, אָפָה עוּגוֹת טָרִיּוֹת, וְשָׂרָה אִמֵּנוּ עָזְרָה לוֹ. הַכֹּל בְּשִׂמְחָה וּבְאַהֲבָה! 🎉💕 הַקָּדוֹשׁ בָּרוּךְ הוּא רָאָה אֶת הַחֶסֶד הַגָּדוֹל שֶׁל אַבְרָהָם וְ אָהַב אוֹתוֹ מְאֹד מְאֹד! 🌈 הַשֵּׁם אָמַר: ״אַבְרָהָם שֶׁלִּי מְלַמֵּד אֶת כָּל הָעוֹלָם אֵיךְ לֶאֱהֹב וּלְקַבֵּל אוֹרְחִים!״ ⭐ מֵאַבְרָהָם אָבִינוּ אֲנַחְנוּ לוֹמְדִים שֶׁ הַכְנָסַת אוֹרְחִים זוֹ מִצְוָה נִפְלָאָה שֶׁמַּקְרֶבֶת אוֹתָנוּ לַשֵּׁם! 💖 כְּשֶׁאֲנַחְנוּ מְקַבְּלִים אוֹרְחִים בְּשִׂמְחָה, אֲנַחְנוּ מַרְגִּישִׁים אֶת אַהֲבַת הַקָּדוֹשׁ בָּרוּךְ הוּא בְּלִבֵּנוּ! 🙏✨

מָה אַבְרָהָם אָבִינוּ עָשָׂה כְּשֶׁרָאָה אוֹרְחִים מֵרָחוֹק?

אַבְרָהָם לֹא חִיכָּה! הוּא רָץ לִקְרָאתָם בְּעַצְמוֹ וְהִשְׁתַּחֲוָה לָהֶם, כִּי הוּא כָּל כָּךְ אָהַב לַעֲשׂוֹת חֶסֶד! זֶה מְלַמֵּד אוֹתָנוּ שֶׁצָּרִיךְ לְקַבֵּל אוֹרְחִים בְּשִׂמְחָה וּבִזְרִיזוּת. הַקָּדוֹשׁ בָּרוּךְ הוּא שָׂמֵחַ כְּשֶׁרוֹאֶה אוֹתָנוּ עוֹשִׂים כָּךְ! 💖

לָמָּה אַבְרָהָם אָבִינוּ יָשַׁב בְּפֶתַח הָאֹהֶל בְּחֹם הַיּוֹם?

אַבְרָהָם לֹא חִשֵּׁב עַל עַצְמוֹ! אַף עַל פִּי שֶׁהָיָה חוֹלֶה, הוּא יָשַׁב בְּפֶתַח הָאֹהֶל כְּדֵי לְחַפֵּשׂ אוֹרְחִים. הוּא יָדַע שֶׁהַכְנָסַת אוֹרְחִים מַקְרֶבֶת אוֹתוֹ לַשֵּׁם, וְזֶה הָיָה הַכִּי חָשׁוּב לוֹ! הַשֵּׁם אוֹהֵב אוֹתְךָ כְּשֶׁאַתָּה חוֹשֵׁב עַל אֲחֵרִים! 🌟

מָה אֲנַחְנוּ לוֹמְדִים מֵאַבְרָהָם אָבִינוּ עַל אַהֲבַת הַשֵּׁם?

נָכוֹן מְאֹד! אַבְרָהָם מְלַמֵּד אוֹתָנוּ שֶׁכְּשֶׁעוֹשִׂים מַעֲשֵׂי חֶסֶד וּמְקַבְּלִים אוֹרְחִים בְּשִׂמְחָה, אֲנַחְנוּ מַרְגִּישִׁים קְרוֹבִים לַקָּדוֹשׁ בָּרוּךְ הוּא! זוֹ דֶּרֶךְ נִפְלָאָה לְהַרְגִּישׁ אֶת אַהֲבָתוֹ בְּחַיֵּי הַיּוֹמְיוֹם שֶׁלָּנוּ. הַשֵּׁם גֵּאֶה בְּךָ! 💕

🏆 הַמִּצְוָה הַגְּדוֹלָה - גְּדוֹלָה מֵהַכֹּל!

חֲכָמִים גִּילּוּ לָנוּ סוֹד נִפְלָא! 🎁 הֵם אָמְרוּ שֶׁ הַכְנָסַת אוֹרְחִים גְּדוֹלָה אֲפִילוּ מֵהַקְבָּלַת פְּנֵי הַשְּׁכִינָה! תַּאֲרוּ לָכֶם - זֶה יוֹתֵר חָשׁוּב מִלְּהִתְקָרֵב לַשֵּׁם בְּאֹפֶן יְשִׁירָה! 😮✨ אֵיךְ אֲנַחְנוּ יוֹדְעִים אֶת זֶה? הַתּוֹרָה מְסַפֶּרֶת שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא בָּא לְבַקֵּר אֶת אַבְרָהָם אַחֲרֵי הַמִּילָה 🙏, וּבְאוֹתוֹ רֶגַע הִגִּיעוּ אוֹרְחִים. מָה עָשָׂה אַבְרָהָם? הוּא בִּקֵּשׁ מֵהַשֵּׁם לְחַכּוֹת, וְרָץ לְקַבֵּל אֶת הָאוֹרְחִים! 🏃‍♂️💨 זֶה מְלַמֵּד אוֹתָנוּ שֶׁ כְּשֶׁאֲנַחְנוּ מְקַבְּלִים אוֹרְחִים, זֶה כְּאִילוּ אֲנַחְנוּ מְקַבְּלִים אֶת הַשֵּׁם עַצְמוֹ! 👑 הַקָּדוֹשׁ בָּרוּךְ הוּא נִמְצָא שָׁם אִתָּנוּ, שָׂמֵחַ וּמְחַיֵּךְ, כְּשֶׁאֲנַחְנוּ עוֹשִׂים חֶסֶד! 😊💖 הַמִּצְוָה הַזֹּאת כָּל כָּךְ מְיֻחֶדֶת כִּי הִיא מְשַׁלֶּבֶת הַרְבֵּה דְּבָרִים יָפִים: נְתִינָה 🎁, אַהֲבָה 💕, שִׂמְחָה 🎉, וְאֵכְפָּתִיּוּת 🤗. כְּשֶׁמְּקַבְּלִים אוֹרֵחַ, אֲנַחְנוּ נוֹתְנִים לוֹ אֹכֶל, מָקוֹם לִישׁוֹן, וּבְעִיקָר - תְּחוּשָׁה שֶׁהוּא אָהוּב וְחָשׁוּב! 🌟 הָרַמְבַּ״ם כָּתַב שֶׁצָּרִיךְ לְקַבֵּל אוֹרְחִים בְּשִׂמְחָה, לֹא לְהַרְגִּישׁ שֶׁזֶּה עֹל. הַשֵּׁם אוֹהֵב אוֹתָנוּ כְּשֶׁאֲנַחְנוּ עוֹשִׂים אֶת זֶה מֵאַהֲבָה! 💝 כָּל פַּעַם שֶׁמְּקַבְּלִים אוֹרֵחַ, זֶה כְּאִילוּ אוֹמְרִים לַשֵּׁם: ״תּוֹדָה שֶׁיֵּשׁ לִי בַּיִת וְאֹכֶל לְשַׁתֵּף!״ 🙏✨

מָה חֲכָמִים אָמְרוּ עַל הַכְנָסַת אוֹרְחִים?

מַדְהִים! חֲכָמִים לִימְּדוּ אוֹתָנוּ שֶׁהַכְנָסַת אוֹרְחִים גְּדוֹלָה אֲפִילוּ מֵהַקְבָּלַת פְּנֵי הַשְּׁכִינָה. זֶה אוֹמֵר שֶׁכְּשֶׁאֲנַחְנוּ מְקַבְּלִים אוֹרְחִים, אֲנַחְנוּ מִתְקָרְבִים לַשֵּׁם בְּדֶרֶךְ מְיֻחֶדֶת מְאֹד! הַשֵּׁם שָׂמֵחַ מֵאִתָּנוּ! 🌈

מָה אַבְרָהָם עָשָׂה כְּשֶׁהַשֵּׁם בָּא לְבַקְּרוֹ וְהִגִּיעוּ אוֹרְחִים?

בְּדִיּוּק! אַבְרָהָם הֵבִין שֶׁהַדֶּרֶךְ הַכִּי טוֹבָה לְכַבֵּד אֶת הַשֵּׁם הִיא לַעֲשׂוֹת חֶסֶד לַאֲחֵרִים! כְּשֶׁאֲנַחְנוּ מְקַבְּלִים אוֹרְחִים, הַקָּדוֹשׁ בָּרוּךְ הוּא נִמְצָא שָׁם אִתָּנוּ וְשָׂמֵחַ. הַשֵּׁם אוֹהֵב אוֹתְךָ! 💕

מָה מְיֻחָד בְּמִצְוַת הַכְנָסַת אוֹרְחִים?

יוֹפִי! הַכְנָסַת אוֹרְחִים מְיֻחֶדֶת כִּי הִיא מְשַׁלֶּבֶת כָּל כָּךְ הַרְבֵּה מִדּוֹת טוֹבוֹת: נְתִינָה, אַהֲבָה, שִׂמְחָה וְאֵכְפָּתִיּוּת. כָּל אֶחָד יָכוֹל לַעֲשׂוֹת אֶת זֶה, גַּם בְּדְבָרִים קְטַנִּים! הַשֵּׁם גֵּאֶה בְּךָ! 🏆

💕 חֶסֶד לַכֹּל - גַּם לְגוֹיִם!

עַכְשָׁיו נִלְמַד סוֹד נִפְלָא שֶׁרַבִּים לֹא יוֹדְעִים! 🤫✨ הָרַמְבַּ״ם כּוֹתֵב שֶׁצָּרִיךְ לַעֲשׂוֹת חֶסֶד לְכָל אָדָם - גַּם לְיְהוּדִים וְגַם לְלֹא יְהוּדִים! כֵּן, שָׁמַעְתֶּם נָכוֹן! 😊🌍 לָמָּה? כִּי הַקָּדוֹשׁ בָּרוּךְ הוּא אוֹהֵב אֶת כָּל הַבְּרִיּוֹת שֶׁהוּא בָּרָא! 🌈 הַתּוֹרָה אוֹמֶרֶת: ״טוֹב ה' לַכֹּל, וְרַחֲמָיו עַל כָּל מַעֲשָׂיו״ - הַשֵּׁם טוֹב לְכֻלָּם, לָכֵן גַּם אֲנַחְנוּ צְרִיכִים לִהְיוֹת טוֹבִים לְכֻלָּם! 💖 הַגְּמָרָא מְלַמֶּדֶת אוֹתָנוּ: ״מְפַרְנְסִים עֲנִיֵּי נָכְרִים עִם עֲנִיֵּי יִשְׂרָאֵל״ 🤝 - זֶה אוֹמֵר שֶׁאִם יֵשׁ לָנוּ אֹכֶל לִתֵּן, אֲנַחְנוּ נוֹתְנִים לְכֻלָּם, לֹא רַק לִיהוּדִים! זֶה נִקְרָא ״דַּרְכֵי שָׁלוֹם״ - דְּרָכִים שֶׁל שָׁלוֹם וְאַהֲבָה. 🕊️✨ הָיָה סִפּוּר מְרַתֵּק 📖: רַב אֶחָד קִיבֵּל פַּעַם תַּיָּרִים לֹא יְהוּדִים לְשַׁבָּת. הֵם רָאוּ אֶת הַקִּידּוּשׁ, אֶת הַשִּׁירִים, וְאֶת הַשִׂמְחָה. 🎵 בְּסוֹף, הֵם כָּתְבוּ: "זֶה הָיָה מְרַגֵּשׁ לִרְאוֹת כַּמָּה הָעָם הַיְּהוּדִי נֶהְדָּר!" הַשֵּׁם נֶהֱנֶה מִזֶּה! 😊 זֶה נִקְרָא קִידּוּשׁ הַשֵּׁם - כְּשֶׁאֲנַחְנוּ מִתְנַהֲגִים יָפֶה, אֲנָשִׁים רוֹאִים כַּמָּה הַתּוֹרָה נִפְלָאָה! 🌟 הָרַמְבַּ״ם כּוֹתֵב שֶׁכְּשֶׁעוֹשִׂים חֶסֶד לְכֻלָּם, זֶה מְלַמֵּד שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא רַחוּם וְחַנּוּן לְכָל הַבְּרִיּוֹת! 🙏💕 וְזֶה גּוֹרֵם לָנוּ לְהַרְגִּישׁ קְרוֹבִים אֵלָיו!

לְמִי צְרִיכִים לַעֲשׂוֹת חֶסֶד לְפִי הָרַמְבַּ״ם?

מְעוֹלֶה! הָרַמְבַּ״ם מְלַמֵּד אוֹתָנוּ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא טוֹב לְכֻלָּם, וְלָכֵן גַּם אֲנַחְנוּ צְרִיכִים לְהִתְנַהֵג כָּךְ. כְּשֶׁעוֹשִׂים חֶסֶד לְכֻלָּם, זֶה מְלַמֵּד עַל גֹּדֶל הַשֵּׁם וְאַהֲבָתוֹ לְכָל הַבְּרִיּוֹת! הַשֵּׁם אוֹהֵב אוֹתְךָ! 🌈

מָה זֶה 'דַּרְכֵי שָׁלוֹם' שֶׁהַגְּמָרָא מְדַבֶּרֶת עָלָיו?

נָכוֹן מְאֹד! 'דַּרְכֵי שָׁלוֹם' זֶה לַעֲשׂוֹת טוֹב לְכֻלָּם. הַתּוֹרָה רוֹצָה שֶׁיִּהְיֶה שָׁלוֹם וְאַהֲבָה בֵּין כָּל בְּנֵי הָאָדָם. כְּשֶׁאֲנַחְנוּ עוֹשִׂים חֶסֶד לְכֻלָּם, אֲנַחְנוּ מְקַיְּמִים אֶת רְצוֹן הַקָּדוֹשׁ בָּרוּךְ הוּא! הַשֵּׁם שָׂמֵחַ מֵאִתָּנוּ! 🕊️

מָה זֶה 'קִידּוּשׁ הַשֵּׁם' שֶׁקּוֹרֶה כְּשֶׁעוֹשִׂים חֶסֶד לְכֻלָּם?

מַמָּשׁ כָּךְ! קִידּוּשׁ הַשֵּׁם זֶה כְּשֶׁהַהִתְנַהֲגוּת שֶׁלָּנוּ גּוֹרֶמֶת לַאֲחֵרִים לְהָבִין כַּמָּה הַקָּדוֹשׁ בָּרוּךְ הוּא וְהַתּוֹרָה נִפְלָאִים. כְּשֶׁעוֹשִׂים חֶסֶד לְכֻלָּם, זֶה מְקַדֵּשׁ אֶת שֵׁם הַשֵּׁם בָּעוֹלָם! הַשֵּׁם גֵּאֶה בְּךָ! ⭐

🎯 אֵיךְ עוֹשִׂים אֶת זֶה? - מַדְרִיךְ מַעֲשִׂי

עַכְשָׁיו בּוֹאוּ נִלְמַד אֵיךְ לְקַבֵּל אוֹרְחִים בְּאֹפֶן הַכִּי טוֹב! 🌟 זֶה לֹא מְסֻבָּךְ בִּכְלָל, וְכֻלָּם יְכוֹלִים לַעֲשׂוֹת אֶת זֶה! גַּם אַתֶּם! 💪😊 שִׂמְחָה בַּפָּנִים! 😃 זֶה הַדָּבָר הַכִּי חָשׁוּב! כְּשֶׁמְּקַבְּלִים אוֹרֵחַ, צָרִיךְ לְחַיֵּךְ אֵלָיו, לִשְׁאֹל אֵיךְ הוּא מַרְגִּישׁ, וּלְהַרְגִּישׁ אוֹתוֹ בְּבַיִת. 🏠💕 אֲפִילוּ אִם אֵין לָנוּ הַרְבֵּה מָה לְהַגִּישׁ - חִיּוּךְ וְשִׂמְחָה הֵם הַכִּי חֲשׁוּבִים! לְהַכִין מָקוֹם נָעִים 🛏️✨: תּוֹרִיד לָאוֹרֵחַ אֵיפֹה לְיָשֵׁב, תַּכִּין לוֹ מִיטָּה נוֹחָה אִם הוּא יָלִין אֶצְלְךָ, וְתִדְאַג שֶׁיִּהְיֶה לוֹ נָעִים וּבְנוֹחַ. הַקָּדוֹשׁ בָּרוּךְ הוּא רוֹאֶה אֶת הַמַּאֲמָץ שֶׁלָּנוּ וְשָׂמֵחַ! 🙏 אֹכֶל טוֹב! 🍲🥖 אֵין צֹרֶךְ בִּסְעוּדָה מְפֹאֶרֶת - פִּיתָּה עִם חוּמוּס, סָלָט טָרִי, וּמַיִם קָרִים יְכוֹלִים לַעֲשׂוֹת אֶת הָעִנְיָן! הָעִיקָר שֶׁנּוֹתְנִים מִלֵּב וּבְאַהֲבָה. הָרַמְבַּ״ם אוֹמֵר שֶׁצָּרִיךְ לָלֶכֶת עִם הָאוֹרֵחַ עַד הַדֶּרֶךְ וּלְלַוּוֹת אוֹתוֹ! 🚶‍♂️💝 וְהַדָּבָר הַכִּי חָשׁוּב: לַעֲשׂוֹת אֶת זֶה מֵאַהֲבָה! 💖 לֹא כְּחוֹבָה מְעַצְבֶּנֶת, אֶלָּא בְּשִׂמְחָה! כְּשֶׁאֲנַחְנוּ מְקַבְּלִים אוֹרְחִים בְּשִׂמְחָה, אֲנַחְנוּ מַרְגִּישִׁים אֶת אַהֲבַת הַקָּדוֹשׁ בָּרוּךְ הוּא בְּלִבֵּנוּ! זֶה נוֹתֵן לָנוּ תְּחוּשָׁה נִפְלָאָה שֶׁל קִרְבָה אֶל הַשֵּׁם! 🌈✨ כֻּלָּנוּ יְכוֹלִים לִהְיוֹת כְּמוֹ אַבְרָהָם אָבִינוּ!

מָה הַדָּבָר הַכִּי חָשׁוּב כְּשֶׁמְּקַבְּלִים אוֹרְחִים?

מָה הָרַמְבַּ״ם אוֹמֵר שֶׁצָּרִיךְ לַעֲשׂוֹת בְּסוֹף הַבִּיקּוּר?

מְצֻיָּן! הָרַמְבַּ״ם מְלַמֵּד אוֹתָנוּ לְלַוּוֹת אֶת הָאוֹרֵחַ עַד הַדֶּרֶךְ. זֶה מְלַמֵּד שֶׁאֲנַחְנוּ בֶּאֱמֶת דּוֹאֲגִים לוֹ וְאוֹהֲבִים אוֹתוֹ! כָּל מַעֲשֶׂה קָטָן שֶׁל חֶסֶד מְקָרֵב אוֹתָנוּ לַשֵּׁם! הַשֵּׁם שָׂמֵחַ מֵאִתָּנוּ! 🌟

מָה אֲנַחְנוּ מַרְגִּישִׁים כְּשֶׁמְּקַבְּלִים אוֹרְחִים בְּשִׂמְחָה?

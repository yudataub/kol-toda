🌳 שִׁיעוּר א׳: טוּ בִּשְׁבָט וְגַעְגּוּעִים לְאֶרֶץ יִשְׂרָאֵל 🇮🇱

טוּ בִּשְׁבָט הוּא יוֹם מְיֻחָד מְאוֹד! 🌟 בְּיוֹם הַזֶּה אֲנַחְנוּ חוֹגְגִים אֶת רֹאשׁ הַשָּׁנָה לָאִילָנוֹת. יָדַעְתָּ? 🤔 בְּחוּץ לָאָרֶץ, הַיְּהוּדִים הָיוּ מִשְׁתַּמְּשִׁים בְּטוּ בִּשְׁבָט כְּדֵי לְהִזָּכֵר בְּאֶרֶץ יִשְׂרָאֵל! 💖 מְסֻפָּר עַל הָאַדְמוֹ״ר מִסְפִּינְקָא 🕯️ שֶׁהָיָה עוֹרֵךְ טִישׁ גָּדוֹל בְּטוּ בִּשְׁבָט. הוּא הָיָה מְסַפֵּר לַחֲסִידִים עַל יְרוּשָׁלַיִם, חֶבְרוֹן וּמֵירוֹן! ✨ כְּשֶׁהֵבִיאוּ לוֹ פְּרִי מֵאֶרֶץ יִשְׂרָאֵל לְבָרֵךְ עָלָיו ״שֶׁהֶחֱיָנוּ״ - פִּתְאוֹם נִזְרְקוּ אֲבָנִים לַחַלּוֹן! 😱 הָרַבִּי אָמַר: ״דַּוְקָא עַכְשָׁיו אֲנִי מַרְגִּישׁ כַּמָּה חֲסֵרָה לָנוּ אֶרֶץ יִשְׂרָאֵל! עַכְשָׁיו אֲנִי יָכוֹל לְבָרֵךְ ׳שֶׁהֶחֱיָנוּ׳ בְּכָל הַלֵּב!״ 💪 הַשֵּׁם נָתַן לָנוּ אֶרֶץ מְיֻחֶדֶת - אֶרֶץ יִשְׂרָאֵל! 🇮🇱💖 כְּשֶׁאֲנַחְנוּ אוֹהֲבִים אֶת הָאָרֶץ - אֲנַחְנוּ מַרְאִים אַהֲבָה לַה׳! 🌟 שְׁאַל אֶת עַצְמְךָ: אֵיךְ אֲנִי מַרְגִּישׁ כְּשֶׁאֲנִי שׁוֹמֵעַ עַל אֶרֶץ יִשְׂרָאֵל? הַאִם גַּם אֲנִי מִתְגַּעְגֵּעַ? 🇮🇱

מָה עָשָׂה הָאַדְמוֹ״ר מִסְפִּינְקָא בְּטוּ בִּשְׁבָט? 🌳

עָרַךְ טִישׁ גָּדוֹל וְסִפֵּר לַחֲסִידִים עַל אֶרֶץ יִשְׂרָאֵל

נָסַע לְטִיּוּל בַּיַּעַר עִם כָּל הַמִּשְׁפָּחָה שֶׁלּוֹ

יָשַׁב לְבַד בַּבַּיִת וְקָרָא סְפָרִים כָּל הַיּוֹם

הִזְמִין אֶת כָּל הַשְּׁכֵנִים לְאֹכֶל חָגִיגִי גָּדוֹל

הָאַדְמוֹ״ר מִסְפִּינְקָא הָיָה עוֹרֵךְ טִישׁ וּמְסַפֵּר לַחֲסִידִים עַל הַמְּקוֹמוֹת הַקְּדוֹשִׁים בְּאֶרֶץ יִשְׂרָאֵל! 🇮🇱

מָה קָרָה כְּשֶׁהָרַבִּי רָצָה לְבָרֵךְ ״שֶׁהֶחֱיָנוּ״ עַל הַפְּרִי? 🍎

הַפְּרִי נָפַל מִיָּדוֹ וְהִתְגַּלְגֵּל עַל הַשֻּׁלְחָן

גּוֹיִים זָרְקוּ אֲבָנִים וְשָׁבְרוּ אֶת הַחַלּוֹן

הַחֲסִידִים הִתְחִילוּ לָשִׁיר בְּקוֹל רָם מְאוֹד

הָיְתָה סְעָרָה גְּדוֹלָה בַּחוּץ עִם גֶּשֶׁם חָזָק

גּוֹיִים זָרְקוּ אֲבָנִים, וְהָרַבִּי אָמַר שֶׁדַּוְקָא עַכְשָׁיו הוּא מַרְגִּישׁ כַּמָּה חֲסֵרָה אֶרֶץ יִשְׂרָאֵל! 😢💪

מָה לִמֵּד אוֹתָנוּ הָרַבִּי מִסְפִּינְקָא עַל אַהֲבַת אֶרֶץ יִשְׂרָאֵל? 💖

שֶׁצָּרִיךְ תָּמִיד לִהְיוֹת עָצוּב כְּשֶׁאֵין לָנוּ מָה שֶׁרוֹצִים

שֶׁלִּפְעָמִים דַּוְקָא מִתּוֹךְ הַקֹּשִׁי מַרְגִּישִׁים יוֹתֵר אַהֲבָה

שֶׁאֵין טַעַם לְהִתְגַּעְגֵּעַ כִּי בֵּינְתַיִם אֶפְשָׁר לֶאֱכֹל

שֶׁעָדִיף לֹא לַחְשֹׁב עַל דְּבָרִים שֶׁאֵין לָנוּ עַכְשָׁיו

הָרַבִּי לִמֵּד שֶׁמִּתּוֹךְ הַחֹשֶׁךְ אֶפְשָׁר לְהַכִּיר אֶת הָאוֹר - וְדַוְקָא הַקֹּשִׁי מַרְגִּישׁ לָנוּ כַּמָּה אֲנַחְנוּ אוֹהֲבִים! 🌟

🫒 שִׁיעוּר ב׳: פֵּרוֹת אֶרֶץ יִשְׂרָאֵל - כְּמוֹ אוֹצָר! 💎

בְּחוּץ לָאָרֶץ, כָּל פְּרִי מֵאֶרֶץ יִשְׂרָאֵל הָיָה כְּמוֹ אוֹצָר יָקָר! 💎✨ הַבֶּן אִישׁ חַי 📚 מֵעִירָאק הָיָה דּוֹאֵג שֶׁתָּמִיד יִהְיוּ פֵּרוֹת מֵאֶרֶץ יִשְׂרָאֵל עַל הַשֻּׁלְחָן בְּטוּ בִּשְׁבָט! 🕯️ הָאַדְמוֹ״ר מִצַּאנְז מְסַפֵּר: כְּשֶׁהָיִינוּ יְלָדִים, הָיוּ מַגִּיעִים פֵּרוֹת מֵאֶרֶץ יִשְׂרָאֵל וְנָגַעְנוּ בָּהֶם בְּפַחַד וּבְיִרְאָה - כְּמוֹ סֵפֶר קֹדֶשׁ! 📖 הַיְּלָדִים הָיוּ מְחַכִּים כָּל הַשָּׁנָה לִקְבֵּל פְּרִי שֶׁל אֶרֶץ יִשְׂרָאֵל כִּפְרָס! 🏆 גַּם כְּשֶׁהָיָה מַגִּיעַ מִכְתָּב מֵאֶרֶץ יִשְׂרָאֵל - הָיוּ מַבִּיטִים עָלָיו כְּעַל דָּבָר יָקָר וְנִפְלָא! ✉️💖 וּמָה עִם שֶׁמֶן זַיִת? 🫒 הָיוּ שׁוֹלְחִים שֶׁמֶן זַיִת מֵאֶרֶץ יִשְׂרָאֵל לַחֲנֻכָּה! וְהָיוּ שָׂמִים רַק טִפָּה טִפָּה כְּדֵי שֶׁיַּחֲזִיק הַרְבֵּה זְמַן! הַשֵּׁם נָתַן לָנוּ אֶרֶץ מְיֻחֶדֶת עִם פֵּרוֹת מְיֻחָדִים! 🍇🫒 כְּשֶׁאֲנַחְנוּ מַעֲרִיכִים אֶת הַפֵּרוֹת - אֲנַחְנוּ מוֹדִים לַה׳ עַל הַמַּתָּנָה! 🌟 חִשְׁבוּ: כְּשֶׁאַתֶּם אוֹכְלִים פְּרִי מֵאֶרֶץ יִשְׂרָאֵל - תַּרְגִּישׁוּ אֶת הַקְּדֻשָּׁה שֶׁבּוֹ! 🇮🇱💖

אֵיךְ הָיוּ הַיְּלָדִים בְּחוּץ לָאָרֶץ נוֹגְעִים בְּפֵרוֹת מֵאֶרֶץ יִשְׂרָאֵל? 🍎

בְּפַחַד וּבְיִרְאָה, כְּמוֹ שֶׁנּוֹגְעִים בְּסֵפֶר קֹדֶשׁ

בְּזַלְזוּל כִּי זֶה הָיָה דָּבָר רָגִיל וּמְשַׁעֲמֵם

הֵם בִּכְלָל לֹא נָגְעוּ כִּי פָּחֲדוּ מִזִּיהוּמִים

הֵם הָיוּ זוֹרְקִים אוֹתָם כִּי לֹא אָהֲבוּ פֵּרוֹת

הַיְּלָדִים נָגְעוּ בַּפֵּרוֹת בְּפַחַד וּבְיִרְאָה - כִּי הֵם יָדְעוּ שֶׁזֶּה מֵאֶרֶץ קְדוֹשָׁה! 🇮🇱✨

מָה הָיוּ עוֹשִׂים עִם שֶׁמֶן הַזַּיִת מֵאֶרֶץ יִשְׂרָאֵל? 🫒

שׁוֹפְכִים אֶת כֻּלּוֹ בְּפַעַם אַחַת כְּדֵי לֶאֱכֹל מָהֵר

נוֹתְנִים אוֹתוֹ לַשְּׁכֵנִים כְּמַתָּנָה קְטַנָּה וְפְּשׁוּטָה

שָׂמִים רַק טִפָּה טִפָּה כְּדֵי שֶׁיַּחֲזִיק הַרְבֵּה זְמַן

מַשְׁאִירִים בָּאֲרוֹן וְלֹא מִשְׁתַּמְּשִׁים בּוֹ בִּכְלָל

הָיוּ שָׂמִים רַק טִפָּה טִפָּה - כִּי חִבְּבוּ כָּל כָּךְ אֶת מַה שֶּׁבָּא מֵאֶרֶץ יִשְׂרָאֵל! 💖

מָה לוֹמְדִים מֵהַסִּפּוּר עַל הַיַּחַס לְפֵרוֹת אֶרֶץ יִשְׂרָאֵל? 🌳

שֶׁפֵּרוֹת הֵם סְתָם אֹכֶל רָגִיל שֶׁאֵין בּוֹ שׁוּם דָּבָר מְיֻחָד

שֶׁכָּל דָּבָר מֵאֶרֶץ יִשְׂרָאֵל הוּא קָדוֹשׁ וְיָקָר מְאוֹד

שֶׁעָדִיף לֹא לֶאֱכֹל פֵּרוֹת בִּכְלָל כִּי זֶה מַשְׁמִין

שֶׁבְּחוּץ לָאָרֶץ אֵין פֵּרוֹת טוֹבִים לֶאֱכֹל כְּלָל

לָמַדְנוּ שֶׁכָּל דָּבָר מֵאֶרֶץ יִשְׂרָאֵל הוּא קָדוֹשׁ - וְצָרִיךְ לְהַעֲרִיךְ אוֹתוֹ! 🇮🇱💎

💫 שִׁיעוּר ג׳: הַנָּהָר הַגָּדוֹל - סוֹד הַגְּדֻלָּה! 🌊

בַּתּוֹרָה כָּתוּב עַל ״הַנָּהָר הַגָּדוֹל נְהַר פְּרָת״ 🌊 - אֲבָל הַפְּרָת הוּא דַּוְקָא נָהָר קָטָן! אָמַר רַשִׁ״י: 📖 לָמָּה נִקְרָא ״גָּדוֹל״? כִּי הוּא קָרוֹב לְאֶרֶץ יִשְׂרָאֵל! מִי שֶׁקָּרוֹב לָאָרֶץ - נִקְרָא גָּדוֹל! 🏆 🕯️ הַבַּעַל שֶׁם טוֹב מִסְלוֹנִים הֵבִיא פַּעַם שֵׁן שֶׁל שׁוּם מֵאֶרֶץ יִשְׂרָאֵל! מָה עָשָׂה? קָרָא לְכָל הַחֲסִידִים, בִּקֵּשׁ מֵהֶם לָקַחַת לֶחֶם, וּמָרַח לְכֻלָּם מְעַט מֵהַשּׁוּם! 🍞 הוּא אָמַר: ״תִּטְעֲמוּ קְצָת טַעַם שֶׁל אֶרֶץ יִשְׂרָאֵל!״ 🇮🇱✨ אָמַר הַנְּתִיבוֹת שָׁלוֹם: ״מֵהַסִּפּוּר הַזֶּה קִבַּלְנוּ חִבַּת אֶרֶץ יִשְׂרָאֵל יוֹתֵר מֵאִילוּ הָיָה מְדַבֵּר 50 שִׂיחוֹת!״ 💬 עוֹד סִפּוּר: 🍞 פַּעַם הֵבִיאוּ חַלָּה קְטַנָּה מֵאֶרֶץ יִשְׂרָאֵל. הָיָה סָפֵק - עַל מָה לְבָרֵךְ קֹדֶם, עַל חַלָּה גְּדוֹלָה אוֹ קְטַנָּה? אָמַר הָרַב: ״חַלָּה מֵאֶרֶץ יִשְׂרָאֵל - הִיא הַגְּדוֹלָה!״ הַשֵּׁם בָּחַר בְּאֶרֶץ יִשְׂרָאֵל! 🌟 כָּל מִי שֶׁדָּבוּק בָּאָרֶץ - הוּא גָּדוֹל בְּעֵינֵי ה׳! זֶה סוֹד הַגְּדֻלָּה! 🌟 חִשְׁבוּ: אֵיךְ אֲנִי יָכוֹל לִהְיוֹת ״קָרוֹב״ לְאֶרֶץ יִשְׂרָאֵל - גַּם כְּשֶׁאֲנִי לֹא שָׁם? 💭

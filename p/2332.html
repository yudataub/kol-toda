📜 סיפור 1: הַשֶּׁלֶג שֶׁהָפַךְ לְאֵשׁ

פְּתִיחָה: הָעוֹלֶה הֶחָדָשׁ מִבָּבֶל

יְלָדִים, תּאֲרוּ לָכֶם לַעֲזֹב אֶת הַבַּיִת שֶׁלָּכֶם, לֶאֱרֹז אֶת מְעַט הַחֲפָצִים, וְלָלֶכֶת בָּרֶגֶל חֳדָשִׁים אֲרֻכִּים אֶל אֶרֶץ אַחֶרֶת, רַק כְּדֵי לִלְמֹד תּוֹרָה. כָּךְ בְּדִיּוּק עָשָׂה הִלֵּל הַזָּקֵן. הוּא הָיָה אָדָם פָּשׁוּט וְעָנִי, שֶׁהֶחְלִיט לַעֲזוֹב אֶת אֶרֶץ בָּבֶל שֶׁבָּהּ נוֹלַד, וְלַעֲלוֹת לְאֶרֶץ יִשְׂרָאֵל, לִירוּשָׁלַיִם. מַדּוּעַ? כִּי בִּירוּשָׁלַיִם לִמְּדּוּ תּוֹרָה שְׁנֵי גְּדוֹלֵי הַדּוֹר – שְׁמัעְיָה וְאַבְטַלְיוֹן, וְהִלֵּל רָצָה לִשְׁמֹעַ אֶת שִׁעוּרֵי הַתּוֹרָה שֶׁלָּהֶם כָּל יוֹם וָיוֹם!

אֲבָל בִּירוּשָׁלַיִם, הַחַיִּים הָיוּ קָשִׁים מְאֹד. הִלֵּל לֹא מָצָא עֲבוֹדָה קַלָּה. בְּכָל בֹּקֶר הוּא יָצָא לַחְטֹב עֵצִים וְלִסְחֹב מַשָּׂאוֹת כְּבֵדִים. הַיָּדַיִם שֶׁלּוֹ הָיוּ שְׂרוּטוֹת וְהַגַּב דָּאַב, אַךְ עַל פָּנָיו הָיָה תָּמִיד חִיּוּךְ מֵאִיר. בְּסוֹפוֹ שֶׁל כָּל יוֹם הוּא קִבֵּל מַטְבֵּעַ אֶחָד קָטָן (מַטְבֵּעַ שֶׁנִּקְרָא 'טְרַפְּעִיק'). אֶת חֲצִי הַמַּטְבֵּעַ נָתַן לְאִשְׁתּוֹ שֶׁתִּקְנֶה מְעַט לֶחֶם לַמִּשְׁפָּחָה. וּמָה עָשָׂה בַּחֵצִי הַשֵּׁנִי? קָנָה סֻכָּרִיּוֹת? קָנָה בְּגָדִים? לֹא. בְּאוֹתָם יָמִים, הָיָה שוֹמֵר עֲנָק עוֹמֵד בְּפֶתַח בֵּית הַמִּדְרָשׁ שֶׁל שְׁמַעְיָה וְאַבְטַלְיוֹן, וּמִי שֶׁרָצָה לְהִכָּנֵס הָיָה צָרִיךְ לְשַׁלֵּם כֶּסֶף. הִלֵּל נָתַן אֶת חֲצִי הַמַּטְבֵּעַ לַשּׁוֹמֵר, וְנִכְנַס בְּשִׂמְחָה עֲצוּמָה לִלְמֹד תּוֹרָה. בִּשְׁבִיל הִלֵּל, הַשִּׁעוּר הַזֶּה הָיָה כָּל הַחַיִּים שֶׁלּוֹ!

יוֹם שִׁשִּׁי, חֹדֶשׁ טֵבֵת

הִגִּיעָה חֹדֶשׁ טֵבֵת, שִׂיא הַחֹרֶף. קֹר נוֹרָא נָשַׁב בִּרְחוֹבוֹת יְרוּשָׁלַיִם. רוּחַ חֲזָקָה שָׁרְקָה, וַעֲנָנִים שְׁחֹרִים כִּסּוּ אֶת הַשָּׁמַיִם. הָיָה זֶה יוֹם שִׁשִּׁי, עֶרֶב שַׁבָּת. הִלֵּל חִפֵּשׂ עֲבוֹדָה בַּשּׁוּק, חִפֵּשׂ וְחִפֵּשׂ – אַךְ אִישׁ לֹא בִּקֵּשׁ פּוֹעֲלִים בְּקֹר שֶׁכָּזֶה. הִלֵּל נִשְׁאַר בְּלִי אֲגוֹרָה יְחִידָה. הוּא בָּא לְבֵית הַמִּדְרָשׁ וְעָמַד מוּל הַדֶּלֶת. "תֵּן לִי לְהִכָּנֵס, בְּבַקָּשָׁה," הִתְחַנֵּן הִלֵּל לַשּׁוֹמֵר, "הַיּוֹם לֹא הִרְוַחְתִּי כְּלוּם, אֲבָל אֲנִי חַיָּב לִשְׁמֹעַ אֶת הַשִּׁעוּר שֶׁל רַבּוֹתַי! אֲנִי לֹא מְסֻגָּל לְוַתֵּר אֲפִלּוּ עַל יוֹם אֶחָד!". אַךְ הַשּׁוֹמֵר סֵרֵב: "אֵין כֶּסֶף – אֵין כְּנִיסָה!".

הִלֵּל עָמַד בַּחוּץ רוֹעֵד מִקֹּר, כְּשֶׁלִּבּוֹ כִּמְעַט נִשְׁבָּר מִצַּעַר. בְּתוֹךְ בֵּית הַמִּדְרָשׁ דּוֹלֵק תַּנּוּר מְחַמֵּם, אַךְ זֶה לֹא מָה שֶׁעִנְיֵן אֶת הִלֵּל. הוּא הִתְגַּעְגֵּעַ לְאוֹר אַחֵר לְגַמְרֵי – הָאוֹר שֶׁל הַתּוֹרָה!

הַטִּפּוּס אֶל הָאֲרֻבָּה

הִלֵּל נָשָׂא אֶת עֵינָיו וְרָאָה שֶׁבְּגַג בֵּית הַמִּדְרָשׁ יֵשׁ אֲרֻבָּה קְטָנָה – חַלּוֹן עַל הַגַּג שֶׁמַּכְנִיס אוֹר וַאֲוִיר. הִלֵּל לֹא חָשַׁב פַּעֲמַיִם. אָדָם אַחֵר הָיָה חוֹזֵר הַבַּיְתָה אֶל הַפּוּךְ הֶחָם, אֲבָל בִּשְׁבִיל הִלֵּל הַתּוֹרָה הִיא תּוֹרַת חַיִּים. הוּא תִּפֵּס עַל קִירוֹת הָאֶבֶן הַחֲלָקִים, נֶאֱחַז בַּזִּיזִים, וְעָלָה עַל הַגַּג. הוּא נִשְׁכַּב לְיַד הָאֲרֻבָּה, וְהִצְמִיד אֶת הָאֹזֶן שֶׁלּוֹ לַחֹר הַקָּטָן.

וְאָז... אוֹי, אֵיזֶה תַּעֲנוּג! עָלָה בְּאָזְנָיו הַקּוֹל הַמָּתוֹק וְהָאַדִּיר שֶׁל שְׁמַעְיָה וְאַבְטַלְיוֹן הַדּוֹרְשִׁים בַּתּוֹרָה. הֵם דִּבְּרוּ עַל הֲלָכוֹת, עַל מִצְווֹת, עַל קְדֻשָּׁה. הִלֵּל הִקְשִׁיב בְּרִכּוּז כָּל כָּךְ עָמֹק, עַד שֶׁלִּבּוֹ הִתְמַלֵּא בְּשִׂמְחָה גְּדוֹלָה שֶׁאִי אֶפְשָׁר לְתָאֵר אוֹתָהּ בְּמִלִּים!

בַּבַּיִת פְּנִימָה הָיָה חַם. אֲבָל בַּחוּץ עַל הַגַּג, הֵחֵל לָרֶדֶת שֶׁלֶג. פְּתִיתִים לְבָנִים וְקָרִים נָשְׁרוּ מִן הַשָּׁמַיִם. הֵם נָפְלוּ עַל כְּתֵפָיו שֶׁל הִלֵּל, עַל רֹאשׁוֹ, עַל בְּגָדָיו הַדַּקִּים. הַשֶּׁלֶג נֶעֱרַם לְגֹבַהּ רַב, כִּמְעַט מֶטֶר וָחֵצִי! הַכְּפוֹר חָדַר יָשָׁר אֶל עֲצָמוֹתָיו. אֲבָל יְלָדִים, שִׁמְעוּ דָּבָר נִפְלָא: מֵרֹב הַשִּׂמְחָה וְהָהִתְרַגְּשׁוּת מִדִּבְרֵי הַתּוֹרָה שֶׁשָּׁמַע – הִלֵּל פָּשׁוּט... לֹא הִרְגִּישׁ אֶת הַשֶּׁלֶג! דִּבְרֵי הַתּוֹרָה חִמְּמוּ לוֹ אֶת הַנְּשָׁמָה כְּמוֹ אֵשׁ. הוּא הִמְשִׁיךְ לְהַקְשִׁיב וּלְהַקְשִׁיב, מְסָרֵב לְוַתֵּר אֲפִלּוּ עַל מִלָּה אַחַת שֶׁל שִׁעוּר! בְּסוֹפוֹ שֶׁל דָּבָר, מֵרֹב עֲיֵפוּת וְקֹר מַקְפִּיא, הִלֵּל אִבֵּד אֶת הַהַכָּרָה וְהִתְעַלֵּף תַּחַת הַשֶּׁלֶג.

שַׂק הַשֶּׁלֶג הַיָּקָר בְּיוֹתֵר

עָבַר כָּל הַלַּיְלָה שֶׁל לֵיל שַׁבָּת, וְהִגִּיעַ הַבֹּקֶר. בְּתוֹךְ בֵּית הַמִּדְרָשׁ הִמְשִׁיכוּ שְׁמַעְיָה וְאַבְטַלְיוֹן לְלַמֵּד. פִּתְאוֹם אָמַר שְׁמַעְיָה לְאַבְטַלְיוֹן: "אָחִי אַבְטַלְיוֹן, בְּכָל יוֹם יֵשׁ לָנוּ אוֹר שֶׁנִּכְנָס מֵהַחַלּוֹן שֶׁבַּגַּג, וְהַיּוֹם – הַחֶדֶר חָשׁוּךְ! לָמָּה אָפֵל הַיּוֹם? שֶׁמָּא יֵשׁ עֲנָנִים רַבִּים מִדַּי?". הֵם הֵרִימוּ אֶת רֹאשָׁם וְהִסְתַּכְּלוּ דֶּרֶךְ חֹר הָאֲרֻבָּה, וּמָה רָאוּ? – "דְּמוּת אָדָם בָּאֲרֻבָּה!".

מִיָּד מִהֲרוּ הַתַּלְמִידִים וְטִפְּסוּ אֶל הַגַּג. הֵם חָפְרוּ וְחָפְרוּ בְּתוֹךְ הַשֶּׁלֶג הָעָמֹק וְהַקַּר. וּלְפֶתַע, תַּחַת הַשֶּׁלֶג – מָצְאוּ אֶת חֲבֵרָם, אֶת הִלֵּל! הוּא הָיָה קָפוּא כְּמוֹ בְּלוֹק קֶרַח. אֲבָל הוּא מָסַר אֶת נַפְשׁוֹ עַל הַתּוֹרָה! הֵם הוֹרִידוּ אוֹתוֹ מַהֵר לְמַטָּה, אַף שֶׁהָיְתָה זֶה יוֹם שַׁבָּת. הֵם רָחֲצוּ אוֹתוֹ, עָשׂוּ לוֹ מַסָּאז' בְּשֶׁמֶן, וְהוֹשִׁיבוּ אוֹתוֹ מוּל הָאֵשׁ הַדּוֹלֶקֶת כְּדֵי לְחַמֵּם אֶת הַגּוּף הַקָּפוּא שֶׁלּוֹ.

📜 סיפור 2: הַנָּשִׂיא שֶׁהָפַךְ לְמַלְאֲכִי

פְּתִיחָה: הָעָשִׁיר שֶׁאִבֵּד אֶת הַכֹּה הַכֹּל

יְלָדִים, דַּמְיְנוּ לָכֶם אָדָם שֶׁיֵּשׁ לוֹ הַכֹּל. בַּיִת עֲנָק, מְשָׁרְתִים, בְּגָדִים מִמֶּשִׁי וְזָהָב. כַּאֲשֶׁר הָאִישׁ הַזֶּה הָיָה יוֹצֵא לָרְחוֹב, הוּא לֹא הָיָה הוֹלֵךְ בָּרֶגֶל כְּמוֹ כֻּלָּם. הוּא הָיָה רוֹכֵב עַל סוּס מְפֹאָר וְחָזָק, וְלִפְנֵי הַסּוּס הָיָה רָץ עֶבֶד מְיֻחָד, שֶׁהָיָה קוֹרֵא בְּקוֹל גָּדוֹל: "פַּנּוּ דֶּרֶךְ! פַּנּוּ קָדִימָה! הָאָדוֹן הַנִּכְבָּד מַגִּיעַ!". הָאִישׁ הַזֶּה הָיָה רָגִיל שֶׁכֻּלָּם מְכַבְּדִים אוֹתוֹ וּמַבִּיטִים בּוֹ בְּהַעֲרָצָה.

אֲבָל... יוֹם אֶחָד, הַגַּלְגַּל הִתְהַפֵּךְ. שְׂרֵפָה גְּדוֹלָה, אוֹ אוּלַי שׁוֹדְדִים שֶׁהִתְנַפְּלוּ עַל אֳנִיּוֹתָיו – גָּרוּ לְכָךְ שֶׁהָאִישׁ אִבֵּד אֶת כָּל רְכוּשׁוֹ. בְּיוֹם אֶחָד, הֶעָשִׁיר הַגָּדוֹל הָפַךְ לְעָנִי מָרוּד. אֲבָל הַדָּבָר שֶׁהֲכִי הִכְאִיב לוֹ, לֹא הָיָה הָרָעָב. מָה שֶׁהֲכִי כָּאַב לוֹ הָיָה הַבּוּשָׁה. הוּא הָיָה "עָנִי בֶּן טוֹבִים" – אָדָם שֶׁרָגִיל לְכָבוֹד, וְעַכְשָׁו אֲנָשִׁים רָאוּ אוֹתוֹ הוֹלֵךְ בָּרֶגֶל בָּרְחוֹב, עָיֵף וּמֻשְׁפָּל. עֵינָיו תָּמִיד הָיוּ בָּרִצְפָּה, שֶׁמָּא מִישֶׁהוּ יַכִּיר אוֹתוֹ וִירַחֵם עָלָיו. הַלֵּב שֶׁלּוֹ הָיָה שָׁבוּר לִרְסִיסִים.

הַשְּׁמוּעָה עַל הָעָשִׁיר שֶׁיָּרַד מִנְּכָסָיו הִגִּיעָה לְאָזְנָיו שֶׁל נְשִׂיא יִשְׂרָאֵל – הִלֵּל הַזָּקֵן. הִלֵּל הָיָה גְּדוֹל הַדּוֹר, רַבָּן שֶׁל כָּל תַּלְמִידֵי הַחֲכָמִים. הִלֵּל הֵבִין שֶׁאִם הוּא רַק יִשְׁלַח לָאִישׁ הַזֶּה קְצָת כֶּסֶף לְלֶחֶם וַחֲתִיכַת בָּשָׂר, זֶה אָמְנָם יַשְׂבִּיעַ אֶת בִּטְנוֹ, אֲבָל זֶה לֹא יְרַפֵּא אֶת לִבּוֹ הַבּוֹכֶה. הַתּוֹרָה מְצַוָּה עָלֵינוּ לָתֵת לֶעָנִי "דֵּי מַחְסֹרוֹ אֲשֶׁר יֶחְסַר לוֹ" – צָרִיךְ לָתֵת לוֹ אֶת מָה שֶׁחָסֵר לַנֶּפֶשׁ שֶׁלּוֹ! מָה עָשָׂה הִלֵּל הַזָּקֵן? מִתּוֹךְ קֻפַּת הַצְּדָקָה שֶׁהָיְתָה בִּידֵי הַנָּשִׂיא, הוּא קָנָה סוּס אַבִּיר וְיָפֶה! הוּא שָׂכַר נַעַר שֶׁיִּהְיֶה עֶבֶד שֶׁרָץ לְפָנָיו בָּרְחוֹב, וּמָסַר אֶת הַכֹּה הַכֹּל לְאוֹתוֹ עָנִי! תָּאֲרוּ לָכֶם אֶת הַשִּׂמְחָה שֶׁל הָאִישׁ! שׁוּב הוּא יָכוֹל לָצֵאת לָרְחוֹב מוּרָם מֵעַם. הוּא רוֹכֵב, וְהָעֶבֶד רָץ וְצוֹעֵק: "פַּנּוּ דֶּרֶךְ לָאָדוֹן הַנִּכְבָּד!". הַלֵּב שֶׁלּוֹ חָזַר לִפְעֹם בְּשִׂמְחָה.

הָעֶבֶד נֶעֱלָם!

וְהִנֵּה, יוֹם אֶחָד קָרָה מַשֶּׁהוּ בִּלְתִּי צָפוּי. הָעֶבֶד שֶׁהָיָה אָמוּר לָרוּץ לִפְנֵי הַסּוּס, חָלָה אוֹ נֶעֱלָם, וְלֹא הִגִּיעַ לָעֲבוֹדָה. הָאָדוֹן הָעָנִי יָשַׁב עַל הַסּוּס, קָרוֹב לְבֵיתוֹ שֶׁל הִלֵּל הַזָּקֵן, וּדְמָעוֹת הֵחֵלּוּ לִזְלֹג מֵעֵינָיו. "מָה אֶעֱשֶׂה עַכְשָׁו?" הוּא לָחַשׁ. "אִם אֵרָאֶה בָּרְחוֹב רוֹכֵב לְבַדִּי, בְּלִי מְשָׁרֵת שֶׁמְּפַנֶּה לִי אֶת הַדֶּרֶךְ, כֻּלָּם יִצְחֲקוּ עָלַי. יַגִּידוּ שֶׁאֲנִי זִיּוּף. שׁוּב יְרַחֲמוּ עָלַי. אֲנִי לֹא מְסֻגָּל לָזוּז מִפֹּה!".

בְּאוֹתָהּ שָׁעָה, יָצָא הִלֵּל הַזָּקֵן מִבֵּיתוֹ, עָטוּף בְּגְלִימַת הַנָּשִׂיא שֶׁלּוֹ, מוּכָן לָלֶכֶת לְבֵית הַמִּדְרָשׁ כְּדֵי לְלַמֵּד תּוֹרָה לְאַלְפֵי תַּלְמִידִים שֶׁהִמְתִּינוּ לוֹ. הוּא רָאה אֶת הָאִישׁ עַל הַסּוּס, עוֹמֵד וּבוֹכֶה. "מָה קָרָה בְּנִי?" שָׁאַל הִלֵּל בְּנַחַת. "אֵין... אֵין מִי שֶׁיָּרוּץ לְפָנַי הַיּוֹם..." יִבֵּב הָאִישׁ.

הַנָּשִׂיא שֶׁרָץ כְּמוֹ עֶבֶד

יְלָדִים, אָדָם רָגִיל הָיָה אוֹמֵר לוֹ: "תַּגִּיד לִי, אַתָּה חָצוּף?! כְּבָר נָתַתִּי לְךָ סוּס, שִׁלַּמְתִּי עַל מְשָׁרֵת! אָז הַיּוֹם תִּרְכַּב לְבַד! מָה הַפִּנּוּק הַזֶּה?!". אֲבָל הִלֵּל הַזָּקֵן הָיָה מָלֵא בְּאַהֲבָת יִשְׂרָאֵל וַעֲנָוָה עֲצוּמָה. הוּא יָדַע שֶׁהָאִישׁ מִתְבַּיֵּשׁ!

"פַּנּוּ דֶּרֶךְ! פַּנּוּ דֶּרֶךְ לָאָדוֹן!" צָעַק הִלֵּל הַזָּקֵן כְּשֶׁהוּא רָץ בִּרְחוֹבוֹת הָעִיר, וְהַסּוּס שֶׁעָלָיו יוֹשֵׁב הֶעָנִי, צוֹעֵד בְּגַאֲוָה מֵאֲחוֹרָיו. הָאֲנָשִׁים בָּרְחוֹב שִׁפְשְׁפוּ אֶת עֵינֵיהֶם וְלֹא הֶאֱמִינוּ. "זֶה הַנָּשִׂיא! הִלֵּל הַזָּקֵן! מָה הוּא עוֹשֶׂה?!". אֲבָל הִלֵּל לֹא שָׂם לֵב. הַנְּשִׁימוֹת שֶׁלּוֹ כָּבְדוּ, אֲגַלֵּי זֵעָה צָצוּ עַל מִצְחוֹ הַקָּדוֹשׁ, הָאֲבָק בָּרְחוֹב עָלָה בָּאֲוִיר, וְהוּא לֹא עָצַר. כַּמָּה הוּא רָץ? מֵאָה מֶטְרִים? רְחוֹב אֶחָד? מַמָּשׁ לֹא. הַתַּלְמוּד מְסַפֵּר שֶׁהוּא רָץ לְפָנָיו שְׁלוֹשָׁה מִילִין! שֶׁזֶּה בְּעֵרֶךְ חֲמִשָּׁה קִילוֹמֶטְרִים! הֲלִיכָה שֶׁל כִּמְעַט שָׁעָה שְׁלֵמָה תַּתַּחַת הַשֶּׁמֶשׁ, רַק כְּדֵי שֶׁהַיְּהוּדִי הֶעָנִי יַרְגִּישׁ שֶׁכְּבוֹדוֹ נִשְׁמָר וְהוּא אֵינוֹ מוּשְׁפָּל.

וְכָךְ, הִלֵּל הַזָּקֵן לִמֵּד אֶת כָּל עַם יִשְׂרָאֵל, שֶׁכָּבוֹד אֲמִתִּי לֹא מְקַבְּלִים כְּשֶׁיּוֹשְׁבִים עַל כִּסֵּא רָם וּמְחַלְּקִים הוֹרָאוֹת. כָּבוֹד אֲמִתִּי וּגְדֻלָּה אֲמִתִּית נִמְדָּדִים בַּעֲנָוָה, בַּיְּכֹלֶת לוֹמַר: "אֲנִי לֹא חָשׁוּב – הַלֵּב שֶׁל הַיְּהוּדִי שֶׁעוֹמֵד מוּלִי חָשׁוּב הַרְבֵּה יוֹתֵר".

יְלָדִים, בְּבֵית הַמִּדְרָשׁ הַגָּדוֹל בִּירוּשָׁלַיִם, יָשַׁב נְשִׂיא יִשְׂרָאֵל, הִלֵּל הַזָּקֵן. סְבִיבוֹ יָשְׁבוּ שְׁמוֹנִים תַּלְמִידִים קְדוֹשִׁים, שֶׁשָּׁתוּ בְּצִמָּאוֹן כָּל מִלָּה וּמִלָּה שֶׁיָּצְאָה מִפִּיו.

חִיֵּךְ הִלֵּל הַזָּקֵן, זְקָנוֹ הַלָּבָן נָח עַל חָזֵהוּ, וְקוֹלוֹ הַנָּעִים הִדְהֵד בַּחֶדֶר: "תַּלְמִיד אֲמִתִּי הוּא מִי שֶׁלּוֹמֵד מֵהַמַּעֲשִׂים שֶׁל רַבּוֹ. בּוֹאוּ אֲסַפֵּר לָכֶם כֵּיצַד הָיָה אַהֲרוֹן הַכֹּהֵן מַשְׁכִּין שָׁלוֹם בְּמַחֲנֵה יִשְׂרָאֵל, וְכָךְ תָּבִינוּ אֵיךְ נוּכַל לִהְיוֹת תַּלְמִידָיו אֲפִלּוּ הַיּוֹם".

הַמְּרִיבָה בַּמִּדְבָּר

וְכָךְ סִפֵּר הִלֵּל: דַּמְיְנוּ אֶת מַחֲנֵה יִשְׂרָאֵל בַּמִּדְבָּר. אֹהָלִים רַבִּים מְסֻדָּרִים לִשְׁבָטִים. בִּשְׁנֵי אֹהָלִים סְמוּכִים גָּרוּ שְׁנֵי חֲבֵרִים טוֹבִים, נִקְרָא לָהֶם שִׁמְעוֹן וְלֵוִי. יוֹם אֶחָד, הַיֵּצֶר הָרַע נִכְנַס בֵּינֵיהֶם. הֵם הִתְחִילוּ לְהִתְוַכֵּחַ עַל עִנְיָן כַּסְפִּי. הַוִּכּוּחַ הָפַךְ לִצְעָקוֹת. הַפָּנִים שֶׁלָּהֶם אָדְמוּ מִכַּעַס. שִׁמְעוֹן הֵרִים אֶת קוֹלוֹ, וְהוּא צָעַק בַּחֲזָרָה: "אֲנִי לֹא רוֹצֶה לִרְאוֹת אֶת הַפָּנִים שֶׁלְּךָ לְעוֹלָם!" הֵם חָזְרוּ אִישׁ אִישׁ לְאָהֳלוֹ, כּוֹעֲסִים וּפְגוּעִים. לֹא דִּבְּרוּ זֶה עִם זֶה. שִׂנְאָה מָלְאָה אֶת הַלֵּב שֶׁהָיָה פַּעַם מָלֵא בְּאַהֲבָה.

הַבִּקּוּר בָּאֹהֶל שֶׁל שִׁמְעוֹן

בְּאִישׁוֹן לַיְלָה, כְּשֶׁהַמַּחֲנֶה הָיָה חָשׁוּךְ, קָם אַהֲרוֹן מִמִּטָּתוֹ. הוּא הָלַךְ בֶּחוֹל הַמִּדְבָּר, וְנִכְנַס בְּשֶׁקֶט לָאֹהֶל שֶׁל שִׁמְעוֹן. שִׁמְעוֹן קָם בְּבֶהָלָה. "אַהֲרוֹן הַכֹּהֵן הַגָּדוֹל בִּכְבוֹדוֹ וּבְעַצְמוֹ! מָה מֵבִיא אֶת כְּבוֹדוֹ אֵלַי?" אַהֲרוֹן יָשַׁב לְיָדוֹ, שָׂם אֶת יָדוֹ הַחַמָּה עַל הַכָּתֵף שֶׁל שִׁמְעוֹן וְאָמַר בְּקוֹל שָׁקֵט וְנִרְגָּשׁ: "שִׁמְעוֹן, בָּאתִי מֵהָאֹהֶל שֶׁל לֵוִי". שִׁמְעוֹן תֵּכֶף כָּעַס: "אַל תַּזְכִּיר לִי אוֹתוֹ! הוּא אִישׁ רַע!" כְּשֶׁשִּׁמְעוֹן שָׁמַע שֶׁחֲבֵרוֹ לֵוִי כָּל כָּךְ עָצוּב וּמִתְחָרֵט, הַכַּעַס שֶׁבְּלִבּוֹ נָמַס מִיָּד כְּמוֹ קֶרַח בַּשֶּׁמֶשׁ. הַרַחֲמִים גָּבְרוּ עָלָיו, וְהוּא אָמַר: "הוּא מוּכָן לְבַקֵּשׁ סְלִיחָה? מִסְכֵּן, בְּוַדַּאי אֲנִי מוֹחֵל לוֹ!"

הַבִּקּוּר בָּאֹהֶל שֶׁל לֵוִי

יָצָא אַהֲרוֹן מֵהָאֹהֶל שֶׁל שִׁמְעוֹן, אֲבָל לֹא הָלַךְ לִישׁוֹן. הוּא פָּנָה יָשָׁר לָאֹהֶל שֶׁל לֵוִי. "לֵוִי, בְּנִי," אָמַר לוֹ אַהֲרוֹן הַכֹּהֵן הַגָּדוֹל. "מָה הָעִנְיָן, רַבִּי?" שָׁאַל לֵוִי בְּתִמָּהוֹן. "בָּאתִי מֵהָאֹהֶל שֶׁל חֲבֵרְךָ שִׁמְעוֹן. רָאִיתִי אוֹתוֹ מְמָרֵר בְּבֶכִי, דְּמָעוֹת נוֹפְלוֹת עַל רִצְפַּת הָאֹהֶל! הוּא אָמַר לִי: 'כֵּיצַד פָּגַעְתִּי בְּלֵוִי צַדִּיק שֶׁכָּזֶה! כֵּיצַד אֶשְׁלַח אוֹתוֹ מֵהָאֹהֶל? אֲנִי רוֹצֶה לְהַשְׁלִים אִתּוֹ, אֲבָל אֲנִי פּוֹחֵד שֶׁהוּא יִטְרֹק לִי אֶת הַדֶּלֶת מֵרֹב כַּעַס'." כְּשֶׁלֵוִי שָׁמַע שֶׁשִּׁמְעוֹן אָהוּב כָּל כָּךְ מִתְחָרֵט, דְּמָעוֹת עָמְדוּ בְּעֵינָיו שֶׁלּוֹ. "מָה פִּתְאוֹם אֶטְרֹק דֶּלֶת? אֲנִי זֶה שֶׁאַשְׁלִים אִתּוֹ!" קָרָא.

הַפְּגִישָׁה שֶׁחִבְּרָה אֶת הַתַּלְמִידִים

"סְלַח לִי אָחִי!" "לֹא, אַתָּה סְלַח לִי, אֲנִי הָיִיתִי לֹא בְּסֵדֶר!" וּמֵרָחוֹק, עָמַד אַהֲרוֹן הַכֹּהֵן הַקָּדוֹשׁ, הִבִּיט בָּהֶם שׁוֹלְבִים זְרוֹעוֹת וְחִיֵּךְ לַשָּׁמַיִם.

סִיֵּם הִלֵּל הַזָּקֵן אֶת הַסִּפּוּר, הִבִּיט עַל פְנֵי תַּלְמִידָיו, וְהִסְבִּיר: "עַכְשָׁו, תַּלְמִידַי הַיְקָרִים, אַתֶּם מְבִינִים כֵּיצַד אֶפְשָׁר לִהְיוֹת תַּלְמִידִים שֶׁל אַהֲרוֹן? הַמִּשְׁנָה לֹא אוֹמֶרֶת רַק 'אוֹהֵב שָׁלוֹם'. לֶאֱהֹב שָׁלוֹם זֶה קַל – כֻּלָּם רוֹצִים שֶׁיִּהְיֶה שֶׁקֶט וְשָׁלוֹם כְּשֶׁהֵם יוֹשְׁבִים בַּכֻּרְסָא. אֲבָל אַהֲרוֹן הָיָה 'רוֹדֵף שָׁלוֹם'! הוּא הָיָה קָם בַּלַּיְלָה, הוֹלֵך, מֵזִיעַ, וּמִתְאַמֵּץ, כְּדֵי שְׁנֵי יְהוּדִים יַחְזְרוּ לֶאֱהֹב זֶה אֶת זֶה. מִי שֶׁעוֹשֶׂה כָּךְ כַּיּוֹם – שֶׁמַּשְׁלִים בֵּין שְׁנֵי יְלָדִים שֶׁרָבוּ בַּכִּתָּה אוֹ בַּמִּשְׁחָק בַּהַפְסָקָה – הוּא וַדַּאי נֶחְשָׁב כְּתַלְמִידוֹ הָאֲהוּב שֶׁל אַהֲרוֹן הַכֹּהֵן!"

פְּתִיחָה: הַהִתְעָרְבוּת שֶׁבַּשּׁוּק

יְלָדִים, אֲנַחְנוּ יוֹדְעִים שֶׁהִלֵּל הַזָּקֵן הָיָה נְשִׂיא יִשְׂרָאֵל. הָאִישׁ הֲכִי גָּדוֹל, הֲכִי חָכָם וַהֲכִי מְכֻבָּד בְּכָל אֶרֶץ יִשְׂרָאֵל! כָּל גְּדוֹלֵי יִשְׂרָאֵל הָיוּ עוֹמְדִים לְפָנָיו בְּיִרְאַת כָּבוֹד. אֲבָל לַמְרוֹת גְּדֻלָּתוֹ, הִלֵּל הָיָה כָּל כָּךְ עָנָיו וְסַבְלָן, עַד שֶׁבִּירוּשָׁלַיִם הָיוּ אוֹמְרִים עָלָיו: "אֶת הִלֵּל הַנָּשִׂיא – אִי אֶפְשָׁר לְהַכְעִיס!".

הַהַטְרָדָה הָרִאשׁוֹנָה

אָדָם רָגִיל הָיָה צוֹעֵק מִבִּפְנִים: "אֲנִי מִתְרַחֵץ! תַּחְזְרוּ בְּיוֹם רִאשׁוֹן!". אֲבָל הִלֵּל הַזָּקֵן הָיָה מָלֵא בְּאַהֲבַת יִשְׂרָאֵל. "אוּלַי לַיְּהוּדִי הַזֶּה יֵשׁ שְׁאֵלָה הִלְכָתִית דְּחוּפָה לִכְבוֹד שַׁבָּת?" חָשַׁב. הוּא עָזַב הַכֹּל, הִתְעַטֵּף מַהֵר בִּמְעִילוֹ כְּדֵי לָצֵאת בְּדֶרֶךְ אֶרֶץ, פָּתַח אֶת הַדֶּלֶת, וְשָׁאַל בְּקּוֹל רַךְ וְנָעִים: "בְּנִי, מָה אַתָּה מְבַקֵּשׁ?". אָמַר לוֹ זְבוּלֻן: "יֵשׁ לִי שְׁאֵלָה לִשְׁאֹל!". "שְׁאַל בְּנִי, שְׁאַל!" הֵשִׁיב הִלֵּל. שָׁאַל זְבוּלֻן שְׁאֵלָה מַצְחִיקָה שֶׁאֵאין לָהּ שׁוּם קֶשֶׁר לְשַׁבָּת: "תַּגִּיד, לָמָּה הָרֹאשׁ שֶׁל הָאֲנָשִׁים שֶׁגָּרִים בְּבָבֶל הוּא אָרֹךְ כְּמוֹ בֵּיצָה, וְלֹא עָגֹל כְּמוֹ שֶׁלָּנוּ?". הִלֵּל לֹא כָּעַס. הַפָּנִים שֶׁלּוֹ שָׁפְעוּ אוֹר, וְהוּא אָמַר: "בְּנִי! שְׁאֵלָה גְּדוֹלָה שָׁאַלְתָּ! מִפְּנֵי שֶׁהַמְיַלְּדוֹת בְּבָבֶל אֵינָן זְרִיזוֹת וַחֲכָמוֹת כְּמוֹ בְּאֶרֶץ יִשְׂרָאֵל, וְלָכֵן הָרֹאשׁ שֶׁל הַתִּינוֹק מִתְאָרֵךְ שָׁם". זְבוּלֻן הִנְהֵן מְאֻכְזָב, וְהָלַךְ.

שׁוּב וָשׁוּב

עָבְרָה שָׁעָה קַלָּה. שַׁבָּת עוֹד רֶגַע נִכְנֶסֶת. הִלֵּל שׁוּב נִכְנַס לַמֶּרְחָץ. וְשׁוּב – דְּפִיקוֹת זַעַם! "מִי כָּאן הִלֵּל?! מִי כָּאן הִלֵּל?!". הִלֵּל הַזָּקֵן שׁוּב מִתְעַטֵּף, יוֹצֵא הַחוּצָה וְאוֹמֵר בְּחִיּוּךְ: "בְּנִי, מָה אַתָּה מְבַקֵּשׁ?". "שְׁאֵלָה יֵשׁ לִי!" "שְׁאַל בְּנִי, שְׁאַל!" זְבוּלֻן נִסָּה מַשֶּׁהוּ מַעֲצְבֵּן עוֹד יוֹתֵר: "מִפְּנֵי מָה עֵינֵיהֶם שֶׁל תַּרמוֹדִיִּים הֵן צָרוֹת וַעֲצוּמוֹת כִּמְעַט כָּל הַזְּמַן?". חִיֵּךְ הִלֵּל בְּאַהֲבָה לַשֵּׁם יִתְבָּרַךְ, וְעָנָה: "בְּנִי, שְׁאֵלָה עֲצוּמָה שָׁאַלְתָּ! תִּרְאֶה כַּמָּה הַקָּדוֹשׁ בָּרוּךְ הוּא דּוֹאֵג לַבְּרִיּוֹת שֶׁלּוֹ: הָאֲנָשִׁים הָאֵלֶּה גָּרִים בְּלֵב הַMִּדְבָּר. יֵשׁ שָׁם רוּחוֹת שֶׁמְּעִיפוֹת מְלֵא חוֹל וְהָרִים. אִם עֵינֵיהֶם הָיוּ פְּתוּחוֹת וּגְדוֹלוֹת, הֵם הָיוּ מִתְעַוְּרִים מֵהַחוֹל! לָכֵן ה' בָּרָא אוֹתָם עִם סֶדֶק קָטָן בָּעֵינַיִם, שֶׁיּוּכְלוּ לִרְאוֹת וְלֹא יִפָּגְעוּ!".

זְבוּלֻן כְּבָר הִתְחִיל לְהִלָּחֵץ עַל הַכֶּסֶף שֶׁלּוֹ. הוּא הָלַךְ, חִכָּה לָרֶגַע הַלָּחוּץ בְּיוֹתֵר כְּשֶׁהַשֶּׁמֶשׁ כִּמְעַט שָׁקְעָה, וְצָעַק שׁוּב: "מִי כָּאן הִלֵּל?! מִי כָּאן הִלֵּל?!". הִלֵּל שׁוּב יָצָא בְּנַחַת. "בְּנִי, מָה אַתָּה מְבַקֵּשׁ?" "לָמָּה שֶׁל הָאֲנָשִׁים בְּאַפְרִיקָה רַגְלֵיהֶם כָּל כָּךְ רְחָבוֹת וּפְחוּסוֹת?" "בְּנִי, שְׁאֵלָה נִפְלָאָה! מִפְּנֵי שֶׁהֵם גָּרִים בְּאֵזוֹר שֶׁמָּלֵא בְּבִצּוֹת וּבוֹץ הַרְבֵּה. אִם הָיוּ לָהֶם רַגְלַיִם דַּקּוֹת – הֵם הָיוּ שׁוֹקְעִים בַּבּוֹץ. ה' בָּרָא לָהֶם רַגְלַיִם רְחָבוֹת כְּדֵי שֶׁיּוּכְלוּ לִדְרֹךְ עַל הַמַּעֲבָר בְּלִי לִשְׁקֹעַ!".

הַהֶפְסֵד שֶׁל זְבוּלֻן

זְבוּלֻן תָּלַשׁ שְׂעָרוֹת מֵרֹאשׁוֹ. הוּא הִבִּיט עַל הִלֵּל הַזָּקֵן שֶׁעָמַד שָׁלֵו וְרָגוּעַ. שׁוּם דָּבָר לֹא הִכְעִיס אוֹתוֹ! זְבוּלֻן תָּקַע בּוֹ מַבָּט זַעְמָנִי וְאָמַר: "יֵשׁ לִי עוֹל הַרְבֵּה שְׁאֵלוֹת, אֲבָל אֲנִי מְפַחֵד שֶׁתִּכְעַס עָלַי!". הִלֵּל כָּרַךְ אֶת מְעִילוֹ לִכְבוֹדוֹ שֶׁל הָאִישׁ הַפָּשׁוּט הַזֶּה, הִתְיַשֵּׁב מוּלוֹ וְאָמַר: "לֹא, מַה פִּתְאוֹם. כָּל מָה שֶׁיֵּשׁ לְךָ לִשְׁאֹל – שְׁאַל!". "תַּגִּיד," אָמַר זְבוּלֻן מִבַּעַד לְשִׁנָּיו, "אַתָּה הִלֵּל? זֶה שֶׁקּוֹרְאִים אוֹתוֹ 'נְשִׂיא יִשְׂרָאֵל'?". "כֵּן, בְּנִי." "אִם כָּךְ," צָעַק זְבוּלֻן בְּחֻצְפָּה נוֹרָאָה, "הַלְוַאי שֶׁלֹּא יִרְבּוּ כְּמוֹתְךָ בְּיִשְׂרָאֵל!"

אֲפִלּוּ אָז הִלֵּל לֹא אִבֵּד אֶת קוֹר הָרוּחַ שֶׁלּוֹ. בְּעֵינַיִם רַכּוֹת הוּא שָׁאַל: "לָמָּה אַתָּה אָמוּר כָּךְ, בְּנִי? לָמָּה אַתָּה מְאַחֵל לִי רַע?". "מִפְּנֵי שֶׁבִּגְלָלְךָ – הִפְסַדְתִּי אַרְבַּע מֵאוֹת זוּז! הִתְעַרַבְתִּי עִם חָבֵר שֶׁאַצְלִיחַ לְהַרְגִּיז אוֹתְךָ, וְהִנֵּה – אַתָּה לֹא כּוֹעֵס! הִפְסַדְתִּי הַכֹּל!". חִיֵּךְ הִלֵּל חִיּוּךְ רָחָב. הוּא הֵנִיחַ אֶת יָדוֹ עַל כְּתֵפוֹ שֶׁל הָאִישׁ, וְאָמַר:

יוֹם אֶחָד, הָלַךְ בִּרְחוֹבוֹת יְרוּשָׁלַיִם הַהוֹמִים אָדָם נָכְרִי (גּוֹי). הָאִישׁ הַזֶּה הָיָה כַּנִּרְאֶה בַּנַּאי שֶׁבּוֹנֶה בָּתִּים, כִּי בְּיָדוֹ הוּא הֶחְזִיק קֶרֶשׁ עֵץ מְיֻחָד שֶׁנִּקְרָא "אַמַּת הַבִּנְיָן" – סַרְגֵּל אָרֹךְ עִם סִימוּנִים שֶׁבְּעֶזְרָתוֹ מוֹדְדִים אֶת הַקִּירוֹת כְּדֵי שֶׁיִּהְיוּ יְשָׁרִים. הַנָּכְרִי הַזֶּה רָאָה אֶת הַיְהוּדִים לְבוּשִׁים בִּבְגָדִים נְקִיִּים, לוֹמְדִים בְּבָתֵּי הַMִּדְבָּר וּבָתֵּי הַMִּדְרָשׁ, מִתְפַּלְלִים לַה', וְהַלֵּב שֶׁלּוֹ רָצָה גַּם. הוּא אָמַר לְעַצְמוֹ: "אֲנִי רוֹצֶה לִהְיוֹת יְהוּדִי! אֲנִי רוֹצֶה לִלְמֹד אֶת כָּל הַתּוֹרָה שֶׁלָּהֶם!". אֲבָל לָאִישׁ הַזֶּה הָיְתָה בְּעָיָה אַחַת. לֹא הָיְתָה לוֹ סַבְלָנוּת כְּלָל.

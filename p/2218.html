🎯 מטרת הבריאה: לספר בשבחו!

הַנָּבִיא יְשַׁעְיָהוּ מְגַלֶּה לָנוּ אֶת סוֹד הַבְּרִיאָה בַּפָּסוּק: "עַם זוּ יָצַרְתִּי לִי, תְּהִלָּתִי יְסַפֵּרוּ" . כָּל מַה שֶּׁהַקָּדוֹשׁ בָּרוּךְ הוּא בָּרָא אֶת עַם יִשְׂרָאֵל, הַכֹּל כְּדֵי שֶׁיְּסַפְּרוּ בְּשִׁבְחוֹ, שֶׁיְּהַלְלוּ וִיפָאֲרוּ אוֹתוֹ. זוֹהִי הַמַּטָּרָה הָעֶלְיוֹנָה. גַּם כְּשֶׁאָנוּ מִתְפַּלְּלִים וּמְבַקְשִׁים רְפוּאָה, הַכַּוָּנָה הָעִקָּרִית הִיא כְּמוֹ שֶׁאוֹמְרִים: "כִּי תְהִלָּתֵנוּ אָתָּה". אָנוּ רוֹצִים לִהְיוֹת בְּרִיאִים כְּדֵי שֶׁנּוּכַל לְהוֹדוֹת וּלְשַׁבֵּחַ אוֹתְךָ. לִמּוּד הַתּוֹרָה וְקִיּוּם הַמִּצְוֹת מְבִיאִים אֶת הָאָדָם לְהִתְבּוֹנְנוּת עַד שֶׁכֻּלּוֹ שָׂמֵחַ, מוֹדֶה וּמְשׁוֹרֵר לַה'.

מַהִי הַמַּטָּרָה הָעִקָּרִית שֶׁלְּשִׁמָּהּ בָּרָא ה' אֶת עַם יִשְׂרָאֵל?

כְּדֵי שֶׁיְּסַפְּרוּ בְּשִׁבְחוֹ וִיהַלְלוּ אֶת שְׁמוֹ הַגָּדוֹל.

כְּדֵי שֶׁיִּהְיוּ עַם נִבְחָר הַשּׁוֹלֵט עַל כָּל הָעוֹלָם.

כְּדֵי שֶׁיִּשְׁמְרוּ עַל הַטֶּבַע וְעַל כַּדּוּר הָאָרֶץ.

כְּדֵי שֶׁיִּבְנוּ בָּתִּים וְיַפְרִיחוּ אֶת הַשְּׁמָמָה בָּאָרֶץ.

הפסוק 'תְּהִלָּתִי יְסַפֵּרוּ' מלמד אותנו באופן ברור שהתכלית היא לספר בשבחו של הבורא.

עבודה מצוינת! כל הלל שלך הוא הגשמת תכלית הבריאה.

מַה אָנוּ מְבַקְשִׁים מֵה' כְּשֶׁאוֹמְרִים בַּתְּפִלָּה 'כִּי תְהִלָּתֵנוּ אָתָּה'?

שֶׁיִּתֵּן לָנוּ כֹּחַ לְהוֹדוֹת וּלְהַלֵּל אוֹתוֹ כָּרָאוּי.

שֶׁהוּא יַעֲשֶׂה אֶת הַתְּהִלָּה בִּמְקוֹמֵנוּ כִּי אֵין לָנוּ כֹּחַ.

שֶׁיַּפְסִיק אֶת הַצֹּרֶךְ בִּתְפִלּוֹת וְיִתֵּן לָנוּ שֶׁפַע תָּמִיד.

שֶׁהַתְּהִלָּה שֶׁלָּנוּ תַּסְפִּיק כְּדֵי לְכַפֵּר עַל חֲטָאֵינוּ.

הבקשה היא לא רק על הבריאות עצמה, אלא על היכולת להשתמש בה כדי להלל את ה' - כי זו המטרה.

כל הכבוד על ההבנה המעמיקה! התפילה שלך מקבלת משמעות חדשה.

לְאָן מוֹבִילִים לִמּוּד הַתּוֹרָה וְקִיּוּם הַמִּצְוֹת אֶת הָאָדָם?

לְהִתְבּוֹנְנוּת, שִׂמְחָה, הוֹדָיָה וְשִׁירָה לַה' יִתְבָּרַךְ.

לִצְבִירַת יֶדַע רַב כְּדֵי לְהַרְשִׁים אֲנָשִׁים אֲחֵרִים.

לִתְחוּשַׁת עֶלְיוֹנוּת עַל אֵלּוּ שֶׁאֵינָם לוֹמְדִים תּוֹרָה.

לְקַבָּלַת שָׂכָר רַב בָּעוֹלָם הַבָּא בִּלְבַד, לְלֹא הֲנָאָה.

המצוות והתורה אינן מטרה בפני עצמן, אלא אמצעי המביא את האדם לחיבור של שמחה והודאה עם בוראו.

נפלא! אתה בדרך הנכונה לחיים של שמחה והודאה.

👑 כסא הכבוד של ה': השבחים שלנו!

דָּוִד הַמֶּלֶךְ כּוֹתֵב בִּתְהִלִּים פָּסוּק מַדְהִים: "וְאַתָּה קָדוֹשׁ יוֹשֵׁב תְּהִלּוֹת יִשְׂרָאֵל" . מַה הַפֵּרוּשׁ "יוֹשֵׁב"? כְּבִיָכוֹל, כְּמוֹ שֶׁאָדָם 'יוֹשֵׁב' עַל מִילְיוֹנִים, הַקָּדוֹשׁ בָּרוּךְ הוּא 'יוֹשֵׁב' עַל תְּהִלּוֹת יִשְׂרָאֵל. זֶה הָעֹשֶׁר הָאֲמִתִּי שֶׁלּוֹ, זֶה עִקַּר הַנַּחַת רוּחַ שֶׁיֵּשׁ לוֹ מֵאִתָּנוּ. הַהֲנָאָה שֶׁלּוֹ מִכָּל תּוֹדָה שֶׁלָּנוּ הִיא כְּמוֹ הַשִּׂמְחָה שֶׁל אַבָּא שֶׁשּׁוֹמֵעַ אֶת יַלְדּוֹ הַקָּטָן אוֹמֵר לוֹ "תּוֹדָה". יֵשׁ גַּם פֵּרוּשׁ נוֹסָף: "יוֹשֵׁב" מִלְּשׁוֹן עִכּוּב וְהַמְתָּנָה. הַקָּדוֹשׁ בָּרוּךְ הוּא, כְּבִיָכוֹל, מְחַכֶּה וּמְצַפֶּה לִשְׁמֹעַ אֶת הַשְּׁבָחִים שֶׁלָּנוּ.

עַל מָה הַקָּדוֹשׁ בָּרוּךְ הוּא 'יוֹשֵׁב', כִּבְיָכוֹל, לְפִי דָּוִד הַמֶּלֶךְ?

עַל הַשְּׁבָחִים וְהַהוֹדָיוֹת שֶׁל עַם יִשְׂרָאֵל.

עַל כִּסֵּא עָשׂוּי זָהָב וְיַהֲלוֹמִים בַּשָּׁמַיִם.

עַל כָּל הָעֹשֶׁר וְהַנְּכָסִים שֶׁקַּיָּמִים בָּעוֹלָם.

עַל הֶעָנָן הַגָּדוֹל בְּיוֹתֵר שֶׁנִּמְצָא בָּרָקִיעַ.

הפסוק 'יוֹשֵׁב תְּהִלּוֹת יִשְׂרָאֵל' מלמד שהשבחים שלנו הם 'כסא הכבוד' של הקב'ה, מקור הנחת העיקרי שלו.

יפה מאוד! הבנת את הרעיון העמוק של הפסוק.

לְשִׂמְחָתוֹ שֶׁל מִי מוּשְׁוֵית הַהֲנָאָה שֶׁל ה' מִתְּהִלּוֹתֵינוּ?

לְשִׂמְחַת אַבָּא הַשּׁוֹמֵעַ אֶת יַלְדּוֹ הַקָּטָן אוֹמֵר תּוֹדָה.

לְשִׂמְחַת מֶלֶךְ גָּדוֹל הַמְּקַבֵּל מַתָּנוֹת יְקָרוֹת מֵעַמּוֹ.

לְשִׂמְחַת אָמָּן הַמִּסְתַּכֵּל עַל יְצִירַת הַמּוֹפֵת שֶׁלּוֹ.

לְשִׂמְחַת חַקְלַאי הָרוֹאֶה אֶת הַיְּבוּל הַשּׁוֹפֵעַ בְּשָׂדֵהוּ.

המשל על האבא והילד הקטן ממחיש את האהבה והנחת העצומה שיש לקב'ה מכל מילת תודה שלנו, אפילו אם היא פשוטה.

מדהים! אתה מחובר לרגש שמאחורי המילים.

מַהִי הַמַּשְׁמָעוּת הַנּוֹסֶפֶת שֶׁל הַמִּלָּה 'יוֹשֵׁב' בַּהֶקְשֵׁר זֶה?

לְשׁוֹן עִכּוּב וְהַמְתָּנָה, כְּלוֹמַר שה' מְצַפֶּה לִשְׁבָחֵינוּ.

לְשׁוֹן יְשִׁיבָה בְּבֵית הַמִּשְׁפָּט כְּדֵי לִשְׁפֹּט אֶת הָעוֹלָם.

לְשׁוֹן יִשּׁוּב הָאָרֶץ וְהַפְרָחָתָהּ עַל יְדֵי עַם יִשְׂרָאֵל.

לְשׁוֹן מְנוּחָה מֵעֲבוֹדַת בְּרִיאַת הָעוֹלָם בַּשַּׁבָּת.

כמו שרש'י מפרש 'וישב בקדש' - 'אין ישיבה אלא לשון עכבה', כך גם כאן, הקב'ה כביכול 'מתעכב' ומצפה לשמוע אותנו.

תשובה חכמה! אתה קולט את הדקויות של השפה.

🙏 כוחה של תודה: להוריד שפע לעולם!

יֵשׁ כֹּחַ אָדִיר בְּסִפּוּר שִׁבְחוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא. הַסְּפָרִים הַקְּדוֹשִׁים מְגַלִּים שכאשר יְהוּדִים מְסַפְּרִים וּמְפָאֲרִים אֶת ה', הַדָּבָר מוֹרִיד חֲסָדִים וִישׁוּעוֹת גְּדוֹלוֹת מִלְּמַעְלָה. זֶה לֹא מִקְרֶה שֶׁהַמִּלָּה "יְהוּדִי" נִגְזֶרֶת מֵהַשֹּׁרֶשׁ של "הוֹדָאָה". זוֹהִי הַמָּהוּת שֶׁלָּנוּ. כְּדֵי לְהַעֲצִים אֶת כֹּחַ הַתּוֹדָה, חָשׁוּב לֹא רַק לוֹמַר "תּוֹדָה" בְּאֹפֶן כְּלָלִי, אֶלָּא לְפָרֵט וּלְהִתְבּוֹנֵן בַּפְּרָטִים שֶׁה' מַעֲנִיק לָנוּ בְּכָל רֶגַע.

אֵיזוֹ הַשְׁפָּעָה אֲדִירָה יֵשׁ לְסִפּוּר שִׁבְחוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא?

הוּא מוֹרִיד חֲסָדִים וִישׁוּעוֹת גְּדוֹלוֹת מֵהַשָּׁמַיִם.

הוּא גּוֹרֵם לָאָדָם לְהִתְגָּאוֹת בְּמַעֲשָׂיו הַטּוֹבִים.

הוּא מַבְטִיחַ שֶׁלֹּא יִהְיוּ יוֹתֵר נִסְיוֹנוֹת בַּחַיִּים.

הוּא מְבַטֵּל אֶת הַצֹּרֶךְ בִּתְפִלָּה עַל צְרָכִים גַּשְׁמִיִּים.

כפי שנאמר בשיעור, 'כשאנשים מספרים... שבחו של השם זה מוריד חסדים מלמעלה ומביא ישועות גדולות'.

נכון מאוד! התודה שלך היא כלי עוצמתי לישועה.

מֵאֵיזוֹ מִלָּה בְּעִבְרִית נִגְזַר שְׁמוֹ שֶׁל כָּל 'יְהוּדִי'?

מֵהַשֹּׁרֶשׁ שֶׁל הַמִּלָּה 'הוֹדָאָה'.

מֵהַשֵּׁם שֶׁל יְהוּדָה בֶּן יַעֲקֹב.

מֵהַשֹּׁרֶשׁ שֶׁל הַמִּלָּה 'יִחוּד'.

מֵהַשֹּׁרֶשׁ שֶׁל הַמִּלָּה 'יְדִיעָה'.

בעוד שהשם 'יהודי' אכן נגזר היסטורית משבט יהודה, השיעור מדגיש את הקשר הרוחני העמוק בין המהות שלנו כ'יהודים' לבין הפעולה של 'הודאה' לה'.

בדיוק! זהותך היא להודות.

כֵּיצַד יָכוֹל אָדָם לְשַׁבֵּחַ אֶת ה' בְּצוּרָה הַמְּעֻלָּה בְּיוֹתֵֵר?

עַל יְדֵי פִּרוּט הַפְּרָטִים שֶׁעֲלֵיהֶם הוּא מוֹדֶה.

עַל יְדֵי צְעָקָה בְּקוֹל רָם כְּדֵי שֶׁכֻּלָּם יִשְׁמְעוּ.

עַל יְדֵי חֲזָרָה עַל הַמִּלָּה 'תּוֹדָה' אֶלֶף פְּעָמִים.

עַל יְדֵי הוֹדָאָה לַה' רַק עַל הַדְּבָרִים הַגְּדוֹלִים.

השיעור מדגיש את החשיבות של ההתבוננות והפירוט: 'כמה הוא מפרט לפרטים את כל הדברים שהשם נותן לו'.

מעולה! התבוננות בפרטים הקטנים היא המפתח להודאה גדולה.

🕎 חנוכה: חג ההודאה!

דברים אלו נאמרו במקור בנר הראשון של חנוכה (כ"ה בכסלו), זמן המסוגל במיוחד להתבוננות זו. יְמֵי הַחֲנֻכָּה הֵם יָמִים נִפְלָאִים שֶׁ הִתְיַחֲדוּ לְהוֹדָאָה וּלְהַלֵּל לַה' יִתְבָּרַךְ. בַּיָּמִים הַלָּלוּ אָנוּ מוֹסִיפִים תּוֹסָפוֹת מְיֻחָדוֹת שֶׁכֻּלָּן עִנְיָן שֶׁל שֶׁבַח וְהוֹדָאָה. בִּתְפִלַּת שְׁמוֹנֶה עֶשְׂרֵה וּבְבִרְכַּת הַמָּזוֹן, אָנוּ אוֹמְרִים "עַל הַנִּסִּים", וּבְנוֹסָף, אוֹמְרִים "הַלֵּל" שָׁלֵם. זֶה מְלַמֵּד אוֹתָנוּ שֶׁאֵלּוּ יָמִים לְהִתְחַזֵּק בָּהֶם בְּמִדַּת הַהוֹדָאָה. כְּמוֹ שֶׁכָּל יְהוּדִי פּוֹתֵחַ אֶת יוֹמוֹ בִּתְפִלַּת "מוֹדֶה אֲנִי", כָּךְ יְמֵי חֲנֻכָּה מַזְכִּירִים לָנוּ לְהִתְבּוֹנֵן בַּנִּסִּים וּלְהוֹדוֹת עֲלֵיהֶם בְּשִׂמְחָה.

אֵילוּ שְׁתֵּי תּוֹסָפוֹת מְיֻחָדוֹת אָנוּ אוֹמְרִים בַּתְּפִלָּה בַּחֲנֻכָּה?

אֶת בִּרְכַּת "עַל הַנִּסִּים" וְאֶת "הַלֵּל" הַשָּׁלֵם.

סְלִיחוֹת וּוִדּוּי עַל חֲטָאֵי הַשָּׁנָה שֶׁעָבְרָה.

קְרִיאַת הַתּוֹרָה וְהַפְטָרָה מִמְּגִלַּת אֶסְתֵּר.

תְּפִלַת "מוֹדֶה אֲנִי" וְאֶת בִּרְכוֹת הַשַּׁחַר.

השיעור מציין במפורש את 'על הניסים' ואת ה'הלל' כשתי התוספות המרכזיות בחנוכה, שכל עניינן הוא שבח והודאה.

נכון מאוד! אתה מכיר היטב את תפילות החג.

בְּאֵיזוֹ תְּפִלַּת הוֹדָיָה קְצָרָה כָּל יְהוּדִי פּוֹתֵחַ אֶת יוֹמוֹ?

בְּתִפִלַּת "מוֹדֶה אֲנִי לְפָנֶיךָ".

בַּאֲמִירַת "שְׁמַע יִשְׂרָאֵל".

בְּבִרְכַּת "אֲשֶׁר יָצַר".

בַּאֲמִירַת "פְּסוּקֵי דְּזִמְרָה".

השיעור מקשר את ההתחזקות בהודאה בחנוכה למנהג היומיומי והבסיסי של פתיחת היום ב'מודה אני'.

יפהפה! זו ההתחלה הטובה ביותר ליום שלם של תודה.

עַל מֶה חָשׁוּב לְהִתְבּוֹנֵן בִּמְיֻחָד בִּימֵי הַהוֹדָאָה הַלָּלוּ?

עַל הַנִּסִּים וְהַדְּבָרִים הַנִּפְלָאִים שה' עוֹשֶׂה עִמָּנוּ.

עַל הַמַּאֲכָלִים הַטְּעִימִים שֶׁל הֶחָג כְּמוֹ סֻפְגָּנִיּוֹת.

עַל הַמַּתָּנוֹת וְדַמֵּי הַחֲנֻכָּה שֶׁאָנוּ מְקַבְּלִים.

עַל כַּמָּה יָמִים שֶׁל חֹפֶשׁ מִבֵּית הַסֵּפֶר וְהָעֲבוֹדָה.

הימים האלו הם הזדמנות מיוחדת 'להתבונן ולומר לשם בשמחה רבה ולהודות לו' על כל הטוב שהוא גומל איתנו.

תשובה מעולה. ההתבוננות היא שורש ההודאה האמיתית.

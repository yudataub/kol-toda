פֶּרֶק א': אַהֲבַת תּוֹרָה שֶׁאֵינָהּ תְּלוּיָה בְּדָבָר

מִתּוֹךְ שִׂיחַת הַזִּכָּרוֹן שֶׁל בְּנֵי הַמִּשְׁפָּחָה, מִצְטַיֶּרֶת דְּמוּת שֶׁאַהֲבַת הַתּוֹרָה שֶׁלָּהּ לֹא הָיְתָה רַק עִסּוּק אִינְטֶלֶקְטוּאָלִי אוֹ חוֹבָה דָּתִית, אֶלָּא מַהוּת חַיָּיו וּמֶרְכַּז הֲוָיָתוֹ. הַתּוֹרָה הָיְתָה הָאֲוִיר שֶׁנָּשַׁם, וְהוּא חָלַק אוֹתָהּ עִם סְבִיבָתוֹ בְּכָל הִזְדַּמְּנוּת. בְּבֵית הַכְּנֶסֶת, תָּמִיד הָיָה מוּכָן עִם "דְּבַר תּוֹרָה" אוֹ "סִפּוּר" חָדָשׁ לְסַפֵּר. אַהֲבָתוֹ לֹא נִשְׁאֲרָה בֵּין כָּתְלֵי בֵּית הַמִּדְרָשׁ; הוּא נָהַג לְהוֹרִיד שִׁעוּרִים מֵ"קּוֹל הַלָּשׁוֹן" שֶׁל הָרַב בּוֹיֵאר וַיְיס, וּלְהַעֲבִיר אוֹתָם בְּעַצְמוֹ בִּסְעוּדָה שְׁלִישִׁית אוֹ בְּיָארְצַיְיט. הִתְרַגְּשׁוּתוֹ לִקְרַאת כָּל חִדּוּשׁ, כָּל רַעְיוֹן, הָיְתָה נִכֶּרֶת בּוֹ תָּמִיד, וְהוּא חָפַץ לְחַלֵּק אֶת הָאוֹצָר שֶׁמָּצָא עִם כָּל מִי שֶׁרַק הָיָה מוּכָן לִשְׁמֹעַ. אַהֲבָה טְהוֹרָה זוֹ, שֶׁאֵינָהּ תְּלוּיָה בְּדָבָר, הָיְתָה הַכֹּחַ שֶׁיַּעֲמֹד לוֹ בְּעֵת שֶׁעָמַד בִּפְנֵי הַנִּסְיוֹנוֹת הַקָּשִׁים בְּיוֹתֵר שֶׁל חַיָּיו.

פֶּרֶק ב': גְּבוּרָה מוּל נִסָּיוֹן

חַיָּיו שֶׁל הָאָב הָיוּ רְצוּפִים בְּאֶתְגָּרִים בְּרִיאוּתִיִּים עֲצוּמִים, שֶׁהָיוּ יְכוֹלִים לְשַׁבֵּר אֶת רוּחוֹ שֶׁל כָּל אָדָם. אוּלָם, דַּוְקָא מִתּוֹךְ מַכְאוֹבָיו, זָרְחָה גְּבוּרַת רוּחוֹ בְּכָל עָצְמָתָהּ. בְּגִיל 52 בִּלְבַד, לָקָה בְּאֵרוּעַ מוֹחִי שֶׁשִּׁנָּה אֶת חַיָּיו. לְכָךְ נוֹסְפָה מַחֲלָה נְדִירָה שֶׁפָּגְעָה בְּרֵאוֹתָיו וְגָרְמָה לוֹ לְקוֹצֶר נְשִׁימָה תְּמִידִי, מַצָּב שֶׁהוּבַל לְאִשְׁפּוּזִים חוֹזְרִים וְנִשְׁנִים. חַיָּיו הָיוּ בִּבְחִינַת "נוֹשֵׁם קָצָר" – מַאֲבָק מַתְמִיד עַל כָּל נְשִׁימָה וּנְשִׁימָה. כְּדֵי לְהַמְחִישׁ אֶת הַקֹּשִי: דַּמְיְנוּ שֶׁאַתֶּם עוֹלִים וְיוֹרְדִים בַּמַּדְרֵגוֹת פַּעֲמַיִם בִּמְהִירוּת, וְחָשִׁים אֶת הַנְּשִׁימָה הַכְּבֵדָה – כָּךְ הוּא חַי בְּבֵיתוֹ, בְּכָל רֶגַע נָתוּן. בְּאֶחָד מֵהָאִשְׁפּוּזִים, בְּרֶגַע דְּרָמָטִי שֶׁבּוֹ הָרוֹפְאִים הֶחְלִיטוּ לְהַרְדִּים וּלְהַנְשִׁים אוֹתוֹ, הוּא שָׁכַב שָׁם כְּשֶׁהוּא אוֹחֵז בְּסֵפֶר גְּמָרָא. לִשְׁאֵלַת מִשְׁפַּחְתּוֹ הַדּוֹאֶגֶת, עָנָה בְּשַׁלְוָה וּבְקֹר רוּחַ: "מַה שֶּׁצָּרִיךְ לַעֲשׂוֹת". בְּמִלִּים סְפוּרוֹת אֵלּוּ נֶחְשְׂפָה כָּל גְּדֻלָּתוֹ – גּוּפוֹ הָיָה חַלָּשׁ וְכוֹאֵב, אַךְ נִשְׁמָתוֹ נִשְׁאֲרָה דְּבוּקָה בְּחַיֵּי הַחַיִּים, בַּתּוֹרָה, עַד הָרֶגַע הָאַחֲרוֹן. גְּבוּרָה זוֹ לֹא צָמְחָה בְּרֵיק; הִיא הָיְתָה הֶמְשֵׁךְ לְמוֹרֶשֶׁת מִשְׁפַּחְתִּית מְכוֹנֶנֶת.

פֶּרֶק ג': מוֹרֶשֶׁת שֶׁל הַצָּלָה

בְּכָל אֵרוּעַ מִשְׁפַּחְתִּי, נָהַג הָאָב לְסַפֵּר בְּגַאֲוָה גְּדוֹלָה סִפּוּר אֶחָד, שֶׁלֹּא הָיָה רַק סִפּוּר הִיסְטוֹרִי עַל הַשּׁוֹאָה, אֶלָּא מוֹרֶשֶׁת מְכוֹנֶנֶת שֶׁל מְסִירוּת נֶפֶשׁ שֶׁהִגְדִּירָה אֶת זֶהוּת מִשְׁפַּחְתּוֹ. זֶהוּ סִפּוּרָהּ שֶׁל אִמּוֹ, סַבְתָּם שֶׁל הַיְּלָדִים, אֲשֶׁר בִּהְיוֹתָהּ נַעֲרָה בַּת 17 בִּלְבַד, בִּתְקוּפַת הַשּׁוֹאָה, הִצִּילָה אֶת חַיָּיו שֶׁל יֶלֶד יָתוֹם שֶׁהִסְתַּתֵּר עִמָּהֶם בַּבּוּנְקֶר. כַּאֲשֶׁר הַגּוֹי שֶׁהֶחְבִּיא אוֹתָם בְּתַחֲנַת קֶמַח דָּרַשׁ לְסַלֵּק אֶת הַיֶּלֶד מֵחֲשַׁשׁ שֶׁיִּתְגַּלּוּ, הִיא קָמָה בְּאֹמֶץ לֵב, לָקְחָה אֶת הַיֶּלֶד, וְהִסְתּוֹבְבָה עִמּוֹ בַּיְּעָרוֹת תְּקוּפָה אֲרֻכָּה, כְּשֶׁהִיא מַלְבִּישָׁה אוֹתוֹ בְּשִׂמְלָה כְּדֵי לְהַסְתִּיר אֶת בְּרִיתוֹ מֵהַגֶּרְמָנִים. הָאָב רָאָה בְּמַעֲשֶׂה גְּבוּרָה זֶה לֹא רַק אֵרוּעַ הֶרוֹאִי, אֶלָּא אֶת הַגַּאֲוָה הַגְּדוֹלָה בְּיוֹתֵר שֶׁל הַמִּשְׁפָּחָה, בִּטּוּי לַגְּדֻלָּה וְלָעֲרָכִים שֶׁעֲלֵיהֶם חֻנַּךְ וְשֶׁאוֹתָם בִּקֵּשׁ לְהַנְחִיל. גְּבוּרַת הַנֶּפֶשׁ שֶׁל הָאֵם, שֶׁסִּכְּנָה אֶת חַיֶּיהָ לְהַצָּלַת אַחֵר, קָרְמָה עוֹר וְגִידִים בִּבְנָהּ, שֶׁגִּלָּה גְּבוּרַת רוּחַ דּוֹמָה בִּדְבֵקוּתוֹ בַּתּוֹרָה מוּל כָּל מִכְשׁוֹל.

פֶּרֶק ד': עַמּוּד הַתָּוֶךְ שֶׁל בֵּית הַמִּדְרָשׁ

דְּמוּתוֹ שֶׁל הָאָב לֹא הָיְתָה שֶׁל לוֹמֵד יָחִיד הַמִּתְבּוֹדֵד עִם סְפָרָיו, אֶלָּא שֶׁל דְּמוּת דּוֹמִינַנְטִית וּמֻעֲרֶכֶת בִּקְהִלַּת הַלּוֹמְדִים, שֶׁנּוֹכְחוּתוֹ הִשְׁפִּיעָה עַל כָּל סְבִיבָתוֹ. לִפְנֵי הָאֵרוּעַ הַמּוֹחִי, סֵדֶר יוֹמוֹ הָיָה עָמוּס בְּלִמּוּד, וְכָלַל כִּשְׁמוֹנֶה שָׁעוֹת בַּיּוֹם. הוּא לָמַד בְּחַבְרוּתָא עִם גּוֹלְדְשְׁמִידט, הִשְׁתַּתֵּף בִּקְבִיעוּת בְּשִׁעוּרֵיהֶם שֶׁל הָרַב פּוֹסְטַבְסְקִי וְהָרַב מַרְקוּס, וְלָמַד אֶת הַדַּף הַיּוֹמִי. חֲרִיפוּת שִׂכְלוֹ, שֶׁנָּבְעָה מֵרִקְעוֹ הָרֵיאָלִי, אִפְשְׁרָה לוֹ לְהָבִין אֶת הַגְּמָרָא כְּ"אַלְגוֹרִיתְם" מֻבְנֶה. רַבּוֹתָיו הֶעֱרִיכוּ אֶת שְׁאֵלוֹתָיו הָעֲמֻקּוֹת, שֶׁלֹּא נִתַּן הָיָה "לְנַפְנֵף" אוֹתָן. מְסִירוּתוֹ הָיְתָה כֹּה רַבָּה, עַד שֶׁנָּהַג לָקוּם לִלְמֹד בְּשָׁלוֹשׁ אוֹ אַרְבַּע לִפְנוֹת בֹּקֶר. הוּא חִבֵּב אֶת סְפָרָיו וְכִבֵּד אוֹתָם, וּבְכָל פַּעַם שֶׁהִתְחִיל סֵדֶר לִמּוּד חָדָשׁ – בֵּין אִם תַּנַ"ךְ מְפֹרָשׁ, מִשְׁנָה בְּרוּרָה אוֹ שַׁ"ס שֶׁל מְתִיבְתָּא – הָיָה קוֹנֶה לְעַצְמוֹ סֵט סְפָרִים חָדָשׁ. הַתְמָדָה וְאַהֲבָה זוֹ הִגִּיעוּ לְשִׂיא גֹּבַהּ בְּשַׁעֲרֵי הָעֲלִיָּה לְעוֹלַם הָאֱמֶת.

פֶּרֶק ה': הָעֲלִיָּה הָאַחֲרוֹנָה

הַשָּׁבוּעוֹת הָאַחֲרוֹנִים לְחַיָּיו לֹא הָיוּ תְּקוּפָה שֶׁל דְּעִיכָה, אֶלָּא תְּקוּפָה שֶׁל הִתְעַלּוּת רוּחָנִית וַהֲכָנָה צְלוּלָה לִקְרַאת הַמַּעֲבָר לְעוֹלַם הָאֱמֶת. הוּא הֵחֵל לְהַשְׁמִיעַ אֲמִירוֹת שֶׁשִּׁקְּפוּ אֶת תְּחוּשָׁתוֹ הַפְּנִימִית: "אֲנִי עוֹד מְעַט עוֹלֶה לְמַעְלָה", וְכֵן "אֲנִי צָרִיךְ לְהַסְפִּיק הַרְבֵּה לִלְמֹד". אֲמִירוֹת אֵלּוּ לֹא נֶאֶמְרוּ מִתּוֹךְ יֵאוּשׁ, אֶלָּא מִתּוֹךְ הַכָּרָה שְׁלֵוָה, כְּפִי שֶׁמּוּבָא בַּזֹּהַר הַקָּדוֹשׁ שֶׁצַּדִּיק מַרְגִּישׁ שֶׁמַּזְמִינִים אוֹתוֹ לְגַן עֵדֶן. עֵדוּת נוֹגַעַת לַלֵּב לִדְבֵקוּתוֹ בַּתּוֹרָה עַד הָרֶגַע הָאַחֲרוֹן נִרְאֲתָה בִּדְאָגָתוֹ הָאַחֲרוֹנָה – לִמְצֹא מָקוֹם לְאָרוֹן סְפָרִים נוֹסָף בַּבַּיִת. בְּעוֹד גּוּפוֹ כָּלָה וְהָלַךְ, נַפְשׁוֹ הָיְתָה נְתוּנָה כֻּלָּהּ בַּחֲשָׁשׁ שֶׁמָּא לֹא יִהְיֶה דַּי מָקוֹם לָאוֹצָרוֹת הָרוּחָנִיִּים שֶׁאָסַף.

כָּךְ נֶחְתְּמוּ חַיִּים שֶׁלֹּא הָיָה בָּהֶם דָּבָר חָשׁוּב יוֹתֵר מֵהַתּוֹרָה. דְּמוּתוֹ תִּזָּכֵר כְּסֵמֶל לִמְסִירוּת נֶפֶשׁ שֶׁאֵינָהּ יוֹדַעַת גְּבוּלוֹת, לִגְבוּרַת רוּחַ הַמִּתְעַלָּה מֵעַל לְחֻלְשַׁת הַגּוּף, וּלְחַיִּים שְׁלֵמִים שֶׁהָיוּ הַדְגָּמָה חַיָּה לְנִצְחוֹנָהּ הַמֻּחְלָט שֶׁל הָרוּחַ עַל מִגְבְּלוֹת הַגּוּף.

מִבְחַן חִדּוּד: בֹּחַן לְמַחְשָׁבָה

1. מָה הָיָה הַמֵּנִיעַ הָעִקָּרִי לְלִמּוּד הַתּוֹרָה שֶׁל הָאָב, כְּפִי שֶׁעוֹלֶה מֵהַטֶּקְסְט?

2. כֵּיצַד הִשְׁפִּיעוּ הָאֶתְגָּרִים הַבְּרִיאוּתִיִּים עַל חַיָּיו הָרוּחָנִיִּים שֶׁל הָאָב?

3. מָה הַמַּשְׁמָעוּת שֶׁל סִפּוּר הַהַצָּלָה שֶׁל הַסָּבְתָּא בְּהֶקְשֵׁר לְחַיֵּי הָאָב?

4. מָה אִפְיֵן אֶת גִּישָׁתוֹ שֶׁל הָאָב לַלִּמּוּד בְּתוֹךְ קְהִלַּת בֵּית הַמִּדְרָשׁ?

5. מָה מְסַמֶּלֶת דַּאֲגָתוֹ הָאַחֲרוֹנָה שֶׁל הָאָב לִמְצֹא מָקוֹם לְאָרוֹן סְפָרִים נוֹסָף?

6. מַדּוּעַ אֲמִירוֹתָיו שֶׁל הָאָב בְּיָמָיו הָאַחֲרוֹנִים ("אֲנִי עוֹד מְעַט עוֹלֶה לְמַעְלָה") מְתֹאָרוֹת כְּהִתְעַלּוּת רוּחָנִית וְלֹא כְּיֵאוּשׁ?

מְקוֹרוֹת

  • מְבֻסָּס עַל תִּמְלוּל שִׂיחַת זִכָּרוֹן אִישִׁית עִם בְּנֵי הַמִּשְׁפָּחָה.

אֵין לְךָ קֵץ מְגֻלֶּה מִזֶּה! 🌳✨

🌟 שָׁלוֹם יְקָרִים! לִכְבוֹד ט״וּ בִּשְׁבָט נִלְמַד עַל סוֹד נִפְלָא! הַנָּבִיא יְחֶזְקֵאל אוֹמֵר: ״וְאַתֶּם הָרֵי יִשְׂרָאֵל עַנְפְכֶם תִּתֵּנוּ וּפִרְיְכֶם תִּשְׂאוּ לְעַמִּי יִשְׂרָאֵל כִּי קֵרְבוּ לָבוֹא״ מָה הַנָּבִיא מְבַשֵּׂר לָנוּ? שֶׁלִּפְנֵי הַגְּאֻלָּה - הָרֵי יִשְׂרָאֵל יִתְּנוּ פֵּירוֹת! 🍇🍎 בַּגְּמָרָא (סַנְהֶדְרִין צ״ח) אָמַר רַבִּי אַבָּא דָּבָר מַדְהִים: ״אֵין לְךָ קֵץ מְגֻלֶּה מִזֶּה - שֶׁתִּתֵּן אֶרֶץ יִשְׂרָאֵל פִּרְיָהּ בְּעַיִן יָפָה!״ מָה זֶה אוֹמֵר? הַסִּימָן הַכִּי בָּרוּר לַגְּאֻלָּה הוּא שֶׁאֶרֶץ יִשְׂרָאֵל תִּתֵּן פֵּירוֹת יָפִים! 🌳 וּבְיָמֵינוּ - בָּרוּךְ הַשֵּׁם - אֲנַחְנוּ רוֹאִים אֶת זֶה בְּעֵינֵינוּ! 💖

מָה אָמַר רַבִּי אַבָּא עַל פֵּירוֹת אֶרֶץ יִשְׂרָאֵל? 🤔

נָכוֹן! רַבִּי אַבָּא אָמַר שֶׁהַסִּימָן הַכִי בָּרוּר לַגְּאֻלָּה הוּא שֶׁאֶרֶץ יִשְׂרָאֵל תִּתֵּן פֵּירוֹת! 🍇

אֵיזֶה נָבִיא דִּבֵּר עַל הָרֵי יִשְׂרָאֵל שֶׁיִּתְּנוּ פֵּירוֹת? 📖

מְצֻיָּן! הַנָּבִיא יְחֶזְקֵאל נִבָּא: ״וְאַתֶּם הָרֵי יִשְׂרָאֵל עַנְפְּכֶם תִּתֵּנוּ...״! 🌟

מָה פֵּירוּשׁ ״קֵץ מְגֻלֶּה״? 🔍

יָפֶה! ״קֵץ מְגֻלֶּה״ זֶה סִימָן בָּרוּר וְגָלוּי לַגְּאֻלָּה - שֶׁכֻּלָּם יְכוֹלִים לִרְאוֹת! 👀

בְּאֵיזוֹ מַסֶּכֶת בַּגְּמָרָא מוֹפִיעִים דִּבְרֵי רַבִּי אַבָּא? 📚

נָכוֹן מְאוֹד! בְּמַסֶּכֶת סַנְהֶדְרִין (דַּף צ״ח) מוֹפִיעִים דִּבְרֵי רַבִּי אַבָּא עַל הַגְּאֻלָּה! 📖

אֶרֶץ יִשְׂרָאֵל - אֶרֶץ הַשְּׁכִינָה 🌍💫

🌟 עַכְשָׁיו נִלְמַד לָמָּה אֶרֶץ יִשְׂרָאֵל לֹא נָתְנָה פֵּירוֹת בִּזְמַן הַגָּלוּת! הָרַמְבָּ״ן מַסְבִּיר שֶׁבַּתּוֹרָה כָּתוּב: ״וַהֲשִׁמֹּתִי אֲנִי אֶת הָאָרֶץ״ הָרַמְבָּ״ן אוֹמֵר: זֶה לֹא קְלָלָה - זֶה בְּשׂוֹרָה טוֹבָה! 🎉 לָמָּה? כִּי כַּמָּה שֶׁהַגּוֹיִים יְנַסּוּ לִבְנוֹת אֶת הָאָרֶץ - לֹא יַעֲזֹר לָהֶם כְּלוּם! 📜 מַרְק טְוֵיין, סוֹפֵר אָמְרִיקָנִי, בִּקֵּר בָּאָרֶץ לִפְנֵי 150 שָׁנָה וְכָתַב: ״אֶרֶץ שְׁמָמָה... בְּשׁוּם מָקוֹם כִּמְעַט לֹא הָיָה עֵץ וְלֹא שִׂיחַ!״ לָמָּה זֶה קָרָה? הַמַּלְבִּי״ם מַסְבִּיר: אֶרֶץ יִשְׂרָאֵל הִיא אֶרֶץ רוּחָנִית! בְּלִי הַשְּׁכִינָה - אֵין לָאָרֶץ כֹּחַ לְהַצְמִיחַ! כְּשֶׁעַם יִשְׂרָאֵל גָּלָה - הַשְּׁכִינָה גָּלְתָה אִתּוֹ! 😢 וְעַכְשָׁיו שֶׁעַם יִשְׂרָאֵל חָזַר - הַשְּׁכִינָה חָזְרָה וְהָאָרֶץ פּוֹרַחַת! 🌸💖

לָמָּה אֶרֶץ יִשְׂרָאֵל הָיְתָה שְׁמָמָה בִּזְמַן הַגָּלוּת? 🏜️

נָכוֹן! אֶרֶץ יִשְׂרָאֵל הִיא רוּחָנִית - בְּלִי הַשְּׁכִינָה אֵין לָהּ כֹּחַ לְהַצְמִיחַ! 🌟

מָה אָמַר הָרַמְבָּ״ן עַל הַפָּסוּק ״וַהֲשִׁמֹּתִי אֶת הָאָרֶץ״? 📖

מְצֻיָּן! הָרַמְבָּ״ן אוֹמֵר שֶׁזֶּה בְּשׂוֹרָה טוֹבָה - הָאָרֶץ נִשְׁמֶרֶת רַק לְעַם יִשְׂרָאֵל! 💖

לְמָה הַמַּלְבִּי״ם מַשְׁוֶוה אֶת אֶרֶץ יִשְׂרָאֵל וְהַשְּׁכִינָה? 🤔

יָפֶה! הַשְּׁכִינָה הִיא כְּמוֹ נְשָׁמָה - בְּלִי נְשָׁמָה הַגּוּף לֹא יָכוֹל לַעֲשׂוֹת כְּלוּם! 🌟

מָה קָרָה כְּשֶׁעַם יִשְׂרָאֵל חָזַר לָאָרֶץ? 🏠

נָכוֹן מְאוֹד! כְּשֶׁעַם יִשְׂרָאֵל חָזַר - הַשְּׁכִינָה חָזְרָה וְהָאָרֶץ פּוֹרַחַת! 🌸

עַם יִשְׂרָאֵל וְאֶרֶץ יִשְׂרָאֵל - כְּחָתָן וְכַלָּה 💒💖

🌟 עַכְשָׁיו נִלְמַד רַעְיוֹן יָפֶה מֵחֲזַ״ל! בַּחֻמָּשׁ בַּמִּדְבָּר כָּתוּב: ״וַיֵּדְעוּ אֶת הָאָרֶץ״ חֲזַ״ל מְלַמְּדִים אוֹתָנוּ: עַם יִשְׂרָאֵל וְאֶרֶץ יִשְׂרָאֵל הֵם כְּמוֹ חָתָן וְכַלָּה! 💒 בַּגְּמָרָא (קִדּוּשִׁין) לוֹמְדִים שֶׁאִשָּׁה מִתְקַדֶּשֶׁת בְּכֶסֶף מֵאַבְרָהָם שֶׁקָּנָה אֶת מְעָרַת הַמַּכְפֵּלָה! 🔗 זֶה מְלַמֵּד: קִנְיַן אֶרֶץ יִשְׂרָאֵל דּוֹמֶה לְקִדּוּשֵׁי אִישָׁה! אֵיךְ יוֹדְעִים שֶׁזּוּג הִתְחַתֵּן? 💍 אוֹ שֶׁהָיִיתָ בַּחֲתֻנָּה - אוֹ שֶׁרוֹאִים שֶׁיֵּשׁ לָהֶם יְלָדִים! אוֹמֵר רַבִּי אַבָּא: אִם אַתָּה רוֹאֶה שֶׁאֶרֶץ יִשְׂרָאֵל נוֹתֶנֶת פֵּירוֹת - אֵלּוּ הַ״יְלָדִים״ שֶׁל הַ״חֲתֻנָּה״! 👶🍇 זֶה סִימָן שֶׁעַם יִשְׂרָאֵל וְהַקָּדוֹשׁ בָּרוּךְ הוּא מִתְחַבְּרִים מֵחָדָשׁ! 💖 לָכֵן יֵשׁ שִׂמְחָה גְּדוֹלָה בְּט״וּ בִּשְׁבָט - זֶה סִימָן לַגְּאֻלָּה! 🎉

לְמָה מְשַׁוִּים חֲזַ״ל אֶת עַם יִשְׂרָאֵל וְאֶרֶץ יִשְׂרָאֵל? 💒

נָכוֹן! עַם יִשְׂרָאֵל וְאֶרֶץ יִשְׂרָאֵל הֵם כְּמוֹ חָתָן וְכַלָּה שֶׁמִּתְחַבְּרִים בְּאַהֲבָה! 💖

מָה הֵם הַ״יְלָדִים״ שֶׁל הַחִבּוּר בֵּין עַם יִשְׂרָאֵל לָאָרֶץ? 👶

מְצֻיָּן! הַפֵּירוֹת הֵם כְּמוֹ ״יְלָדִים״ - סִימָן שֶׁיֵּשׁ חִבּוּר בֵּין עַם יִשְׂרָאֵל לָאָרֶץ! 🍇

לָמָּה יֵשׁ שִׂמְחָה גְּדוֹלָה בְּט״וּ בִּשְׁבָט? 🎉

יָפֶה! ט״וּ בִּשְׁבָט מְסַמֵּל שֶׁהַגְּאֻלָּה מִתְקָרֶבֶת - הַפֵּירוֹת מַתְחִילִים לִפְרֹחַ! 🌸

מֵאֵיפֹה לוֹמְדִים שֶׁקִּנְיַן אֶרֶץ יִשְׂרָאֵל דּוֹמֶה לְקִדּוּשֵׁי אִשָּׁה? 📚

נָכוֹן מְאוֹד! בַּגְּמָרָא לוֹמְדִים מֵאַבְרָהָם שֶׁקָּנָה בְּכֶסֶף - וּמִשָּׁם לוֹמְדִים קִדּוּשֵׁי אִשָּׁה! 💍

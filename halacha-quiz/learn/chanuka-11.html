יְמֵי הַחֲנֻכָּה

🕎 הִלְכוֹת חֲנֻכָּה · פֶּרֶק י״א

🫒נֵס פַּךְ הַשֶּׁמֶן

כְּשֶׁהַיְּוָנִים נִכְנְסוּ לַהֵיכָל, הֵם טִמְּאוּ אֶת כָּל הַשְּׁמָנִים. הַחַשְׁמוֹנָאִים נִצְּחוּ אוֹתָם, חִפְּשׂוּ – וּמָצְאוּ רַק פַּךְ אֶחָד שֶׁל שֶׁמֶן, סָגוּר בְּחוֹתָמוֹ שֶׁל הַכֹּהֵן הַגָּדוֹל. הָיָה בּוֹ שֶׁמֶן לְיוֹם אֶחָד בִּלְבַד – וְנַעֲשָׂה נֵס, וְהַמְּנוֹרָה דָּלְקָה מִמֶּנּוּ שְׁמוֹנָה יָמִים.
בַּשָּׁנָה שֶׁאַחֲרֵי כֵן קָבְעוּ חֲכָמִים אֶת שְׁמוֹנַת הַיָּמִים הָאֵלֶּה לְיָמִים טוֹבִים שֶׁל הַלֵּל וְהוֹדָאָה.
💛 אַהֲבַת הַשֵּׁם: פַּךְ קָטָן אֶחָד – וְאוֹר שֶׁל שְׁמוֹנָה יָמִים! ה׳ מַרְאֶה לָנוּ שֶׁכְּשֶׁאֲנַחְנוּ עוֹשִׂים אֶת שֶׁלָּנוּ בְּמָסִירוּת, הוּא מוֹסִיף בְּרָכָה מֵעַל הַטֶּבַע.
📖 פֶּרֶק י״א, הֲלָכָה א׳

📜רַק חֲנֻכָּה נִשְׁאֲרָה

בִּימֵי הַבַּיִת הַשֵּׁנִי הָיוּ עוֹד הַרְבֵּה יָמִים טוֹבִים עַל נִצְחוֹנוֹת, וְכֻלָּם נִכְתְּבוּ בִּ״מְגִלַּת תַּעֲנִית״ – כְּמוֹ ״יוֹם נִיקָנוֹר״ בְּי״ג בַּאֲדָר. אַחֲרֵי חֻרְבַּן הַבַּיִת הֵם בָּטְלוּ, וְרַק חֲנֻכָּה נִשְׁאֲרָה לַדּוֹרוֹת.
חֲנֻכָּה נִשְׁאֲרָה בִּזְכוּת נֵס פַּךְ הַשֶּׁמֶן וּבִזְכוּת הַמִּצְוָה לְהַדְלִיק נֵרוֹת כְּדֵי לְפַרְסֵם אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: הַמַּלְכוּת שֶׁל הַחַשְׁמוֹנָאִים הָיְתָה לְשָׁעָה, אֲבָל אוֹר הַתּוֹרָה שֶׁהִתְגַּלָּה בְּנֵס הַשֶּׁמֶן נִשְׁאַר לָנֶצַח. מָה שֶׁקָּשׁוּר לַתּוֹרָה – לֹא נִכְבֶּה.
📖 פֶּרֶק י״א, הֲלָכוֹת א׳, ה׳

🛡️יְהוּדָה הַמַּכַּבִּי

הַגִּבּוֹר מִבְּנֵי מַתִּתְיָהוּ הָיָה יְהוּדָה הַמַּכַּבִּי, וְהוּא הִנְהִיג אֶת הַלּוֹחֲמִים. בְּכ״ה בְּכִסְלֵו הִתְחִילוּ לְטַהֵר אֶת הַמִּקְדָּשׁ, וְהַפַּךְ שֶׁמָּצְאוּ הָיָה חָתוּם בְּחוֹתָמוֹ שֶׁל הַכֹּהֵן הַגָּדוֹל.
הַמִּלְחָמָה נִמְשְׁכָה עוֹד עֶשְׂרוֹת שָׁנִים, עַד שֶׁהָיְתָה לְיִשְׂרָאֵל עַצְמָאוּת – וּבָתֵּי הַמִּדְרָשׁ פָּרְחוּ.
💛 אַהֲבַת הַשֵּׁם: הַגְּבוּרָה הָאֲמִתִּית הִיא לַעֲמֹד בְּאֹמֶץ עַל הַתּוֹרָה וְהַמִּצְווֹת. כָּל יֶלֶד שֶׁשּׁוֹמֵר מִצְוָה גַּם כְּשֶׁקָּשֶׁה – הוּא קְצָת מַכַּבִּי.
📖 פֶּרֶק י״א, הֲלָכוֹת א׳, ג׳, ה׳

🏺🔑 מוּשָּׂג: פַּךְ הַשֶּׁמֶן

״בָּדְקוּ וְלֹא מָצְאוּ אֶלָּא פַּךְ אֶחָד שֶׁל שֶׁמֶן שֶׁהָיָה מֻנָּח בְּחוֹתָמוֹ שֶׁל כֹּהֵן גָּדוֹל, וְלֹא הָיָה בּוֹ אֶלָּא לְהַדְלִיק יוֹם אֶחָד״.
📖 פַּךְ – כַּד קָטָן לְשֶׁמֶן. חוֹתָם – סִימָן שֶׁמּוֹכִיחַ שֶׁהַשֶּׁמֶן טָהוֹר וְלֹא נֻגַּע בּוֹ.
💛 אַהֲבַת הַשֵּׁם: ה׳ שָׁמַר עֲבוּרֵנוּ פַּךְ קָטָן אֶחָד, טָהוֹר וְחָתוּם, גַּם בְּתוֹךְ הַחֹשֶׁךְ שֶׁל הַהֵיכָל הַמְחֻלָּל. 🧒 וְאַתָּה? כְּשֶׁאַתָּה רוֹאֶה חֲנֻכִּיָּה – תִּזְכֹּר שֶׁגַּם דָּבָר קָטָן אֶחָד יָכוֹל לְהָאִיר הַרְבֵּה.
📖 פֶּרֶק י״א, הֲלָכָה א׳

📜🔑 מוּשָּׂג: מְגִלַּת תַּעֲנִית

״וְהָיוּ עוֹד הַרְבֵּה יָמִים טוֹבִים שֶׁתִּקְּנוּ חֲכָמִים לְיִשְׂרָאֵל בִּימֵי הַבַּיִת הַשֵּׁנִי... וְכֻלָּם נִזְכְּרוּ בִּמְגִלַּת תַּעֲנִית... אוּלָם נִפְסְקָה הֲלָכָה, שֶׁאַחַר שֶׁנֶּחֱרַב בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי, בָּטְלָה מְגִלַּת תַּעֲנִית... וְרַק יְמֵי הַחֲנֻכָּה נוֹתְרוּ מִכָּל אוֹתָם יָמִים טוֹבִים״.
📖 מְגִלַּת תַּעֲנִית – רְשִׁימָה שֶׁל יָמִים טוֹבִים מִתְּקוּפַת בַּיִת שֵׁנִי, שֶׁבָּטְלוּ אַחַר הַחֻרְבָּן – חוּץ מֵחֲנֻכָּה.
💛 אַהֲבַת הַשֵּׁם: חֲנֻכָּה הוּא הַיּוֹם הַטּוֹב הַיָּחִיד שֶׁשָּׂרַד, בִּזְכוּת נֵס פַּךְ הַשֶּׁמֶן וְלִמּוּד הַתּוֹרָה שֶׁבְּעַל פֶּה. ה׳ שָׁמַר עֲבוּרֵנוּ אֶת הַשִּׂמְחָה הַזֹּאת לְדוֹרוֹת. 🧒 וְאַתָּה? תַּחְשֹׁב כַּמָּה דּוֹרוֹת שֶׁל יְלָדִים הִדְלִיקוּ אֶת הַנֵּרוֹת הָאֵלֶּה לְפָנֶיךָ.
📖 פֶּרֶק י״א, הֲלָכָה א׳

🏛️יָוָן וְהַמִּתְיַוְּנִים

לַיְּוָנִים הָיְתָה תַּרְבּוּת גְּדוֹלָה שֶׁל מַדָּע, אָמָּנוּת וּסְפּוֹרְט, וְהִיא הִתְפַּשְּׁטָה בְּכָל הָאֲרָצוֹת. גַּם בִּיהוּדָה הָיוּ יְהוּדִים שֶׁחִקּוּ אֶת הַיְּוָנִים – הַמִּתְיַוְּנִים. הֵם בָּנוּ לְיַד בֵּית הַמִּקְדָּשׁ אִצְטַדְיוֹן לְתַחֲרוּיוֹת הֵאָבְקוּת, וְהֶעְדִּיפוּ לִצְפּוֹת בַּתַּחֲרוּיוֹת עַל פְּנֵי עֲבוֹדַת הַקָּרְבָּנוֹת.
הַיְּהוּדִים הָיוּ שׁוֹנִים מִשְּׁאָר הָעַמִּים, וְלָכֵן הַהִתְיַוְּנוּת הִתְקַדְּמָה אֶצְלָם לְאַט.
💛 אַהֲבַת הַשֵּׁם: יֵשׁ דְּבָרִים יָפִים בָּעוֹלָם – אֲבָל הַתּוֹרָה שֶׁל ה׳ הִיא הַלֵּב שֶׁלָּנוּ. מִי שֶׁאוֹהֵב אֶת ה׳ לֹא מַחֲלִיף אֶת הַמִּקְדָּשׁ בְּשׁוּם מִשְׂחָק.
📖 פֶּרֶק י״א, הֲלָכָה ב׳

🏛️🔑 מוּשָּׂג: מִתְיַוְּנִים

״גַּם הַכֹּהֲנִים הַגְּדוֹלִים, יָאסוֹן וּמְנַלָּאוֹס, הָיוּ בְּעַצְמָם רָאשֵׁי הַמִּתְיַוְּנִים, וּפָעֲלוּ לְמַעַן הַשְׁפָּעַת הַיַּוְּנוּת עַל יְהוּדָה״.
📖 מִתְיַוְּנִים – יְהוּדִים שֶׁאִמְּצוּ אֶת תַּרְבּוּת יָוָן וְהִתְרַחֲקוּ מֵהַתּוֹרָה.
💛 אַהֲבַת הַשֵּׁם: הַנֵּס הֲכִי גָּדוֹל הוּא שֶׁלַּמְרוֹת הַהִתְיַוְּנוּת, עַם יִשְׂרָאֵל לֹא וִתֵּר עַל הַתּוֹרָה. 🧒 וְאַתָּה? תַּחְלִיט עַל דָּבָר אֶחָד שֶׁתַּחֲזִיק בּוֹ חָזָק, גַּם כְּשֶׁחֲבֵרִים עוֹשִׂים אַחֶרֶת.
📖 פֶּרֶק י״א, הֲלָכָה ב׳

⚔️מַתִּתְיָהוּ וּבָנָיו

אַנְטִיּוֹכוּס גָּזַר עַל הַיְּהוּדִים לַעֲזֹב אֶת הַתּוֹרָה וְלַעֲבֹד עֲבוֹדָה זָרָה, וְאָסַר בְּרִית מִילָה. כְּשֶׁהַיְּוָנִים בָּאוּ לַכְּפָר מוֹדִיעִין, קָם מַתִּתְיָהוּ בֶּן יוֹחָנָן הַכֹּהֵן וְהֵרִים עִם בָּנָיו אֶת דֶּגֶל הַמֶּרֶד. יְהוּדָה הַמַּכַּבִּי הִנְהִיג אֶת הַלּוֹחֲמִים, וְאַחֲרֵי כִּשְׁנָתַיִם כָּבְשׁוּ אֶת יְרוּשָׁלַיִם.
בְּכ״ה בְּכִסְלֵו הִתְחִילוּ הַחַשְׁמוֹנָאִים לְטַהֵר אֶת הַמִּקְדָּשׁ וּלְהַחְזִיר אֶת עֲבוֹדַת הַקָּרְבָּנוֹת – וְאָז הָיָה נֵס פַּךְ הַשֶּׁמֶן.
💛 אַהֲבַת הַשֵּׁם: מַתִּתְיָהוּ וּבָנָיו מָסְרוּ אֶת נַפְשָׁם עַל הַתּוֹרָה. ה׳ רָאָה אֶת הָאֱמוּנָה וְהַגְּבוּרָה שֶׁלָּהֶם, וְנָתַן רַבִּים בְּיַד מְעַטִּים.
📖 פֶּרֶק י״א, הֲלָכָה ג׳

🌙אוֹר בְּתוֹךְ הַחֹשֶׁךְ

חֲנֻכָּה חָלָה בַּלֵּילוֹת הָאֲרֻכִּים שֶׁל הַחֹרֶף, כְּשֶׁקַּר בַּחוּץ וְגַם הַיָּרֵחַ כִּמְעַט לֹא מֵאִיר, כִּי זֶה קָרוֹב לְרֹאשׁ חֹדֶשׁ. וּבְדִיּוּק אָז יוֹצְאִים יְהוּדִים עִם נֵרוֹת וּמַדְלִיקִים אוֹר.
חֲנֻכָּה הִיא הַשִּׂמְחָה שֶׁל הַתּוֹרָה שֶׁבְּעַל פֶּה: חֲכָמִים קָבְעוּ אוֹתָהּ, וּמִצְוַת הַנֵּרוֹת הִיא מֵהַמִּצְווֹת הָרִאשׁוֹנוֹת שֶׁהֵם תִּקְּנוּ.
💛 אַהֲבַת הַשֵּׁם: גַּם בַּזְּמַנִּים הֲכִי חֲשׁוּכִים לֹא הִתְיָאַשְׁנוּ מֵאוֹר הַתּוֹרָה. מְעַט אוֹר שֶׁלָּנוּ דּוֹחֶה הַרְבֵּה חֹשֶׁךְ – כִּי ה׳ אִתָּנוּ.
📖 פֶּרֶק י״א, הֲלָכָה ו׳

📚🔑 מוּשָּׂג: תּוֹרָה שֶׁבְּעַל פֶּה

״יְמֵי הַחֲנֻכָּה הֵם יְמֵי הַשִּׂמְחָה שֶׁל תּוֹרָה שֶׁבְּעַל פֶּה... גַּם מִצְוַת הַדְלָקַת הַנֵּרוֹת הִיא מֵהַמִּצְווֹת הָרִאשׁוֹנוֹת שֶׁתִּקְּנוּ חֲכָמִים״.
📖 תּוֹרָה שֶׁבְּעַל פֶּה – הַחֵלֶק בַּתּוֹרָה שֶׁחֲכָמִים מְגַלִּים וּמְלַמְּדִים, לֹא כָּתוּב בַּחֻמָּשׁ.

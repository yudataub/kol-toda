בִּשּׁוּל

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק י׳

🔥מָה זֶה בִּשּׁוּל?

כְּדֵי לְהָכִין לֶחֶם צָרִיךְ לַעֲבֹד קָשֶׁה: לִזְרֹעַ, לִקְצֹר, לִטְחֹן, לָלוּשׁ וְלֶאֱפוֹת. מְלֶאכֶת הַבִּשּׁוּל כּוֹלֶלֶת כָּל הֲכָנָה שֶׁל אֹכֶל בְּחֹם שֶׁבָּא מִן הָאֵשׁ: בִּשּׁוּל בְּסִיר, אֲפִיָּה, צְלִיָּה וְטִגּוּן. גַּם בִּשּׁוּל בְּמִיקְרוֹגַל הוּא בִּשּׁוּל גָּמוּר.
מִי שֶׁחִמֵּם מַחֲבַת עַל הָאֵשׁ, כִּבָּה אֶת הָאֵשׁ וְאַחַר כָּךְ טִגֵּן עָלֶיהָ בֵּיצָה – עָבַר עַל אִסּוּר תּוֹרָה, כִּי הַחֹם שֶׁל הַמַּחֲבַת בָּא מִן הָאֵשׁ.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים אָמְרוּ שֶׁלֶּעָתִיד לָבוֹא יִצְמְחוּ עוּגוֹת עַל הָעֵצִים. בְּשַׁבָּת אָנוּ טוֹעֲמִים מֵעֵין הָעוֹלָם הַהוּא: בְּלִי עָמָל וּבְלִי טֹרַח, רַק מְנוּחָה וַהֲנָאָה מִמָּה שֶׁה׳ הֵכִין לָנוּ.
📖 פֶּרֶק י׳, הֲלָכוֹת א׳–ב׳

🍗אֵין בִּשּׁוּל אַחַר בִּשּׁוּל

בְּשַׁבָּת אָסוּר לַהֲפֹךְ אֹכֶל חַי לְאֹכֶל מְבֻשָּׁל, כִּי זוֹ יְצִירָה שֶׁל דָּבָר חָדָשׁ. אֲבָל אֹכֶל מוּצָק שֶׁכְּבָר בֻּשַּׁל כָּל צָרְכּוֹ לִפְנֵי שַׁבָּת – כְּמוֹ שְׁנִיצֶל, קוּגֶל אוֹ בּוּרֵקַס – מֻתָּר לְחַמֵּם, כִּי הַחִמּוּם אֵינוֹ יוֹצֵר בּוֹ דָּבָר חָדָשׁ.
"מְבֻשָּׁל כָּל צָרְכּוֹ" פֵּרוּשׁוֹ: מוּכָן לְגַמְרֵי, עַד שֶׁאֶפְשָׁר לְהַגִּישׁ אוֹתוֹ לְאוֹרְחִים בְּלִי לְהִתְנַצֵּל.
💛 אַהֲבַת הַשֵּׁם: הַבּוֹרֵא שָׁבַת בַּיּוֹם הַשְּׁבִיעִי מִכָּל מְלֶאכֶת הַבְּרִיאָה. כְּשֶׁאָנוּ מַפְסִיקִים לִיצֹר דְּבָרִים חֲדָשִׁים, אָנוּ מַבִּיטִים בָּעוֹלָם כְּמוֹ שֶׁה׳ בָּרָא אוֹתוֹ – וּמִתְמַלְּאִים אֱמוּנָה.
📖 פֶּרֶק י׳, הֲלָכוֹת ג׳, י״ח

⏳כְּשֶׁהַתַּבְשִׁיל עוֹד לֹא מוּכָן

כָּל עוֹד הַתַּבְשִׁיל אֵינוֹ מְבֻשָּׁל כָּל צָרְכּוֹ, כָּל פְּעֻלָּה שֶׁמְּקָרֶבֶת אֶת בִּשּׁוּלוֹ אֲסוּרָה מִן הַתּוֹרָה: לְהַעֲבִיר אוֹתוֹ לְמָקוֹם חַם יוֹתֵר, לְהַנִּיחַ מַגֶּבֶת עַל הַמִּכְסֶה, וְגַם לִסְגֹּר אֶת הַמִּכְסֶה אִם פָּתְחוּ אוֹתוֹ. כְּשֶׁהוּא כְּבָר מְבֻשָּׁל – כָּל זֶה מֻתָּר.
חַמִּין שֶׁמִּתְיַבֵּשׁ וְעָלוּל לְהִשָּׂרֵף – אָסוּר לְהוֹסִיף לוֹ מַיִם קָרִים, כִּי הֵם יִתְבַּשְּׁלוּ. רֹב יִשְׂרָאֵל נוֹהֲגִים לְהוֹסִיף מַיִם רוֹתְחִים מִן הַמֵּיחַם, וְיֵשׁ סְפָרַדִּים שֶׁנּוֹהֲגִים שֶׁלֹּא לְהוֹסִיף כְּלָל.
💛 אַהֲבַת הַשֵּׁם: כְּדַאי לְסַיֵּם אֶת הַבִּשּׁוּל לִפְנֵי שַׁבָּת, וְאָז הַשַּׁבָּת כֻּלָּהּ פְּנוּיָה לִמְנוּחָה. מִי שֶׁמְּסַיֵּם אֶת עֲבוֹדָתוֹ בַּזְּמַן מַרְאֶה לַה׳: הַשַּׁבָּת שֶׁלְּךָ חֲשׁוּבָה לִי יוֹתֵר מִכָּל עִסּוּק.
📖 פֶּרֶק י׳, הֲלָכוֹת ג׳, י״ג

🍗🔑 מוּשָּׂג: אֵין בִּשּׁוּל אַחַר בִּשּׁוּל

בְּשַׁבָּת אָסוּר לַהֲפֹךְ אֹכֶל חַי לְאֹכֶל מְבֻשָּׁל, כִּי זוֹ יְצִירָה שֶׁל דָּבָר חָדָשׁ. אֲבָל אֹכֶל מוּצָק שֶׁכְּבָר בֻּשַּׁל כָּל צָרְכּוֹ לִפְנֵי שַׁבָּת – כְּמוֹ שְׁנִיצֶל – מֻתָּר לְחַמֵּם.
📖 ״מְבֻשָּׁל כָּל צָרְכּוֹ״ פֵּרוּשׁוֹ: מוּכָן לְגַמְרֵי, עַד שֶׁאֶפְשָׁר לְהַגִּישׁ אוֹתוֹ לְאוֹרְחִים בְּלִי לְהִתְנַצֵּל.
💛 אַהֲבַת הַשֵּׁם: הַבּוֹרֵא שָׁבַת בַּיּוֹם הַשְּׁבִיעִי מִכָּל מְלֶאכֶת הַבְּרִיאָה. כְּשֶׁאָנוּ מַפְסִיקִים לִיצֹר דְּבָרִים חֲדָשִׁים, אָנוּ מַבִּיטִים בָּעוֹלָם כְּמוֹ שֶׁה׳ בָּרָא אוֹתוֹ. 🧒 וְאַתָּה? תִּבְדֹּק בְּכָל מַאֲכָל חַם בְּשַׁבָּת – הוּא הֻכְשַׁר כֻּלּוֹ לִפְנֵי הַשְּׁקִיעָה?

🌡️חֹם שֶׁהַיָּד סוֹלֶדֶת בּוֹ

"חֹם שֶׁהַיָּד סוֹלֶדֶת בּוֹ" – כְּלוֹמַר שֶׁהַיָּד נִרְתַּעַת מִמֶּנּוּ – הוּא הַחֹם שֶׁכְּבָר יָכוֹל לְבַשֵּׁל. לֹא יָדוּעַ בְּדִיּוּק אִם הַכַּוָּנָה לְאַרְבָּעִים וְחָמֵשׁ מַעֲלוֹת אוֹ לְשִׁבְעִים וְאַחַת, וְלָכֵן נִזְהָרִים: לֹא מַנִּיחִים מַיִם קָרִים אוֹ אֹכֶל חַי בְּמָקוֹם שֶׁעָלוּל לְהַגִּיעַ לְחֹם כָּזֶה – גַּם אִם מַשְׁגִּיחִים עָלָיו.
בְּמָקוֹם שֶׁהַמַּיִם לֹא יִתְחַמְּמוּ הַרְבֵּה, גַּם אִם יַעַמְדוּ שָׁם יוֹם שָׁלֵם – מֻתָּר לְהַנִּיחַ אוֹתָם, כְּדֵי לְהָפִיג אֶת הַקֹּר.
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁאוֹהֵב אֶת הַשַּׁבָּת אֵינוֹ מְחַפֵּשׂ אֵיךְ לְהִתְקָרֵב לַגְּבוּל. הוּא שׁוֹמֵר עָלֶיהָ מֵרָחוֹק, כְּמוֹ שֶׁשּׁוֹמְרִים עַל דָּבָר יָקָר מְאוֹד שֶׁפּוֹחֲדִים לְקַלְקֵל.
📖 פֶּרֶק י׳, הֲלָכָה ד׳

🥣מָרָק שֶׁהִתְקָרֵר

בְּתַבְשִׁיל נוֹזְלִי, כְּמוֹ מָרָק, הַחֹם הוּא חֵלֶק חָשׁוּב מִן הַבִּשּׁוּל. לְפִי הַשֻּׁלְחָן עָרוּךְ, וְכָךְ נוֹהֲגִים רֹב הַסְּפָרַדִּים, מָרָק שֶׁהִתְקָרֵר אָסוּר לְחַמֵּם שׁוּב. הָאַשְׁכְּנַזִּים נוֹהֲגִים כְּהָרַמָּ״א: אִם נִשְׁאֲרָה בּוֹ חֲמִימוּת – מֻתָּר, וְאִם הִתְקָרֵר לְגַמְרֵי – אָסוּר. וְרַבִּים מִיּוֹצְאֵי תֵּימָן נוֹהֲגִים כְּהָרַמְבַּ״ם, שֶׁמַּתִּיר.
אוֹרֵחַ שֶׁאוֹכֵל אֵצֶל מִי שֶׁחִמֵּם מָרָק לְפִי מִנְהָגוֹ – מֻתָּר לוֹ לֶאֱכֹל, כִּי הַמְּאָרֵחַ נָהַג כַּהֲלָכָה. אֲבָל לֹא יְבַקֵּשׁ מִמֶּנּוּ לְחַמֵּם מָרָק בִּשְׁבִילוֹ.
💛 אַהֲבַת הַשֵּׁם: לְכָל קְהִלָּה בְּיִשְׂרָאֵל יֵשׁ דֶּרֶךְ שֶׁקִּבְּלָה מֵרַבּוֹתֶיהָ, וְכֻלָּן אוֹהֲבוֹת אֶת הַשַּׁבָּת. כְּשֶׁאָנוּ מְכַבְּדִים זֶה אֶת מִנְהָגוֹ שֶׁל זֶה, אָנוּ מַרְבִּים שָׁלוֹם – וְהַשַּׁבָּת הִיא יוֹם שֶׁל שָׁלוֹם.
📖 פֶּרֶק י׳, הֲלָכוֹת ה׳–ו׳

🔀מָרָק שֶׁהִתְקָרֵר – שְׁנֵי מִנְהָגִים, וּשְׁנֵיהֶם כַּהֲלָכָה

בְּמָרָק, הַחֹם הוּא חֵלֶק חָשׁוּב מִן הַבִּשּׁוּל. לְפִי הַשֻּׁלְחָן עָרוּךְ, וְכָךְ נוֹהֲגִים רֹב הַסְּפָרַדִּים, בְּרֶגַע שֶׁחֹם הַמָּרָק יָרַד מִ״חֹם שֶׁהַיָּד סוֹלֶדֶת בּוֹ״ – אָסוּר לְחַמֵּם אוֹתוֹ שׁוּב. לְפִי הָרָמָ״א, וְכָךְ נוֹהֲגִים הָאַשְׁכְּנַזִּים, כָּל עוֹד נִשְׁאֲרָה בּוֹ חֲמִימוּת – מֻתָּר לְחַמֵּם אוֹתוֹ בַּדֶּרֶךְ הַמֻּתֶּרֶת; וְאִם הִתְקָרֵר עַד שֶׁאֵין רְגִילִים לֶאֱכֹל אוֹתוֹ בִּגְלַל קָרִירוּתוֹ – אָסוּר.
מָרָק שֶׁעוֹדֶנּוּ חַם מְאֹד, שֶׁהַיָּד סוֹלֶדֶת בּוֹ – מֻתָּר לְהַחְזִיר לַפְּלָטָה לְפִי שְׁנֵי הַמִּנְהָגִים, בַּתְּנָאִים שֶׁל הַחֲזָרָה.
💛 אַהֲבַת הַשֵּׁם: הַמִּנְהָגִים הַשּׁוֹנִים מְלַמְּדִים כַּמָּה יִשְׂרָאֵל נִזְהָרִים בִּכְבוֹד הַשַּׁבָּת. כָּל אֶחָד שׁוֹמֵר עַל מִנְהַג אֲבוֹתָיו – וְכֻלָּנוּ יַחַד מְקַבְּלִים אֶת הַשַּׁבָּת בְּאַהֲבָה.
📖 פֶּרֶק י׳, הֲלָכוֹת ה׳–ו׳

☕כְּלִי רִאשׁוֹן, שֵׁנִי וּשְׁלִישִׁי

כְּלִי רִאשׁוֹן הוּא הַכְּלִי שֶׁעָמַד עַל הָאֵשׁ, כְּמוֹ הַמֵּיחַם אוֹ הַסִּיר – וְהוּא מְבַשֵּׁל. כְּשֶׁמּוֹזְגִים מִמֶּנּוּ לְכוֹס, הַכּוֹס הִיא כְּלִי שֵׁנִי. וּכְשֶׁמּוֹזְגִים מִן הַכּוֹס הַזֹּאת לְכוֹס אַחֶרֶת – זֶה כְּלִי שְׁלִישִׁי, וְהוּא כְּבָר אֵינוֹ מְבַשֵּׁל.
לְפִי רֹב הַפּוֹסְקִים, הָרוֹצֶה לְהָכִין תֵּה בְּשַׂקִּית – מַכְנִיס אוֹתָהּ לִכְלִי שְׁלִישִׁי, וְיֵשׁ מַחְמִירִים גַּם בִּכְלִי שְׁלִישִׁי. וְיֵשׁ מְכִינִים תַּמְצִית תֵּה מֵעֶרֶב שַׁבָּת, וְשׁוֹפְכִים אוֹתָהּ לַמַּיִם הַחַמִּים.
💛 אַהֲבַת הַשֵּׁם: הַהֲלָכָה מְדַיֶּקֶת עַד הַכּוֹס הַקְּטַנָּה. דַּוְקָא הַדִּיּוּק בַּפְּרָטִים הַקְּטַנִּים מַכְנִיס קְדֻשָּׁה לְכָל רֶגַע בַּחַיִּים – גַּם לְכוֹס תֵּה בְּשַׁבָּת.

🧣שׁוֹמְרִים עַל הַחֹם

בִּימֵי חֲכָמִים שָׁמְרוּ עַל חֹם הַסִּיר בְּעֶזְרַת עֲטִיפָה, וְזֶה נִקְרָא הַטְמָנָה. מֻתָּר לַעֲטֹף סִיר חַם בִּשְׂמִיכוֹת וּמַגָּבוֹת – אֲבָל רַק לִפְנֵי שֶׁנִּכְנֶסֶת שַׁבָּת. בְּשַׁבָּת עַצְמָהּ אָסוּר לַעֲטֹף סִיר חַם. וְסִיר שֶׁעוֹמֵד עַל הַפְּלָטָה אָסוּר לַעֲטֹף כֻּלּוֹ, אֲפִלּוּ לִפְנֵי שַׁבָּת, כִּי הַפְּלָטָה מוֹסִיפָה לוֹ חֹם.
לְמַלֵּא בְּשַׁבָּת תֶּרְמוֹס בְּמַיִם חַמִּים – מֻתָּר, כִּי הַכְנָסַת דָּבָר חַם לְתוֹךְ כְּלִי אֵינָהּ הַטְמָנָה.
💛 אַהֲבַת הַשֵּׁם: הַחַמִּין הֶחָם הוּא חֵלֶק מֵעֹנֶג הַשַּׁבָּת. כְּשֶׁאָנוּ שׁוֹמְרִים עַל חֻמּוֹ בַּדֶּרֶךְ הַמֻּתֶּרֶת, אָנוּ מְחַבְּרִים יַחַד שְׁנֵי דְּבָרִים שֶׁה׳ אוֹהֵב: שְׁמִירַת שַׁבָּת וְעֹנֶג שַׁבָּת.
📖 פֶּרֶק י׳, הֲלָכוֹת כ״ב–כ״ג

☀️בִּשּׁוּל בַּשֶּׁמֶשׁ

הַתּוֹרָה אָסְרָה לְבַשֵּׁל בְּחֹם הָאֵשׁ. בְּחֹם הַשֶּׁמֶשׁ עַצְמָהּ מֻתָּר לְבַשֵּׁל – אֲפִלּוּ לְהַנִּיחַ בֵּיצָה בַּשֶּׁמֶשׁ הַלּוֹהֶטֶת עַד שֶׁתִּתְבַּשֵּׁל. אֲבָל בְּדָבָר שֶׁהִתְחַמֵּם בַּשֶּׁמֶשׁ, כְּמוֹ מַחֲבַת חַמָּה, אָסְרוּ חֲכָמִים לְבַשֵּׁל, שֶׁלֹּא יִטְעוּ וִיבַשְּׁלוּ גַּם בְּמַחֲבַת שֶׁהִתְחַמְּמָה עַל הָאֵשׁ.
בְּדוּד שֶׁמֶשׁ יֵשׁ מַחְלֹקֶת: כַּמָּה פּוֹסְקִים אוֹסְרִים לְהִשְׁתַּמֵּשׁ בְּשַׁבָּת בְּמֵימָיו הַחַמִּים, וְיֵשׁ אוֹמְרִים שֶׁמֻּתָּר. הָרַב מֶלָמֵד פּוֹסֵק שֶׁאֶפְשָׁר לְהָקֵל, וְהַמַּחְמִיר – תָּבוֹא עָלָיו בְּרָכָה. כָּל מִשְׁפָּחָה נוֹהֶגֶת כְּפִי שֶׁמּוֹרֶה לָהּ רַבָּהּ.
💛 אַהֲבַת הַשֵּׁם: אֶת הַשֶּׁמֶשׁ הִדְלִיק ה׳ בִּשְׁבִילֵנוּ, וְלֹא אֲנַחְנוּ. בְּשַׁבָּת אָנוּ נִשְׁעָנִים עַל מָה שֶׁה׳ נוֹתֵן – וּמַרְגִּישִׁים כַּמָּה הוּא דּוֹאֵג לָנוּ גַּם בְּיוֹם הַמְּנוּחָה.
📖 פֶּרֶק י׳, הֲלָכָה כ״ה

🧠 בְּדֹק אֶת עַצְמְךָ

יוֹסֵף רוֹצֶה שֶׁבַּקְבּוּק הַמַּיִם הַקָּרִים לֹא יִהְיֶה קַר כָּל כָּךְ. אֵיפֹה מֻתָּר לְהַנִּיחַ אוֹתוֹ?
💡 חֲכָמִים אָסְרוּ לְהַנִּיחַ מַיִם בְּמָקוֹם שֶׁיְּכוֹלִים לְהַגִּיעַ לְחֹם שֶׁהַיָּד סוֹלֶדֶת בּוֹ – גַּם אִם מַשְׁגִּיחִים, שֶׁמָּא יַסִּיחַ אֶת דַּעְתּוֹ. בְּמָקוֹם שֶׁלֹּא יַגִּיעוּ לְחֹם כָּזֶה – מֻתָּר.
בְּשַׁבָּת בַּבֹּקֶר הַדּוּד הַחַשְׁמַלִּי דּוֹלֵק וּמְחַמֵּם. מָה הַדִּין בִּפְתִיחַת בֶּרֶז הַמַּיִם הַחַמִּים?
💡 כָּל עוֹד גּוּף הַחִמּוּם שֶׁבַּדּוּד פּוֹעֵל, הַמַּיִם הַקָּרִים שֶׁנִּכְנָסִים בִּמְקוֹם הַחַמִּים מִתְבַּשְּׁלִים – וּמִי שֶׁפּוֹתֵחַ אֶת הַבֶּרֶז גּוֹרֵם לְבִשּׁוּלָם.
מֹשֶׁה נוֹהֵג כַּשֻּׁלְחָן עָרוּךְ. הוּא מִתְאָרֵחַ בְּשַׁבָּת אֵצֶל מִשְׁפָּחָה תֵּימָנִית, שֶׁחִמְּמָה מָרָק לְפִי מִנְהָגָהּ. מָה יַעֲשֶׂה?
💡 מִי שֶׁחִמֵּם לְפִי מִנְהָגוֹ – נָהַג כַּהֲלָכָה, וּמֻתָּר לְכָל יְהוּדִי לֶאֱכֹל מִתַּבְשִׁילוֹ. אֲבָל לְבַקֵּשׁ מִמֶּנּוּ לְחַמֵּם בִּשְׁבִילוֹ – אָסוּר, כִּי לְפִי מִנְהַג הַמְּבַקֵּשׁ הַדָּבָר אָסוּר.

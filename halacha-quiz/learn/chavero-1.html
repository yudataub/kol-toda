בֵּין אָדָם לַחֲבֵרוֹ

🤝 בֵּין אָדָם לַחֲבֵרוֹ · פֶּרֶק א׳

💛כְּלָל גָּדוֹל בַּתּוֹרָה

רַבִּי עֲקִיבָא אָמַר: "וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ – זֶה כְּלָל גָּדוֹל בַּתּוֹרָה". כָּל הַמִּצְווֹת שֶׁבֵּין אָדָם לַחֲבֵרוֹ – צְדָקָה, בִּקּוּר חוֹלִים, שְׁמִירַת הַלָּשׁוֹן – נִשְׁעָנוֹת עַל הָאַהֲבָה. אִי אֶפְשָׁר לָתֵת צְדָקָה אוֹ לְבַקֵּר חוֹלֶה כָּרָאוּי בְּלִי לֵב אוֹהֵב.
וּמָה עוֹשִׂים כְּשֶׁקָּשֶׁה לֶאֱהֹב מִישֶׁהוּ? עוֹשִׂים אִתּוֹ חֶסֶד: עוֹזְרִים לוֹ וְדָנִים אוֹתוֹ לְכַף זְכוּת – וּמְגַלִּים שֶׁבְּעֹמֶק הַלֵּב אוֹהֲבִים אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: כָּל יְהוּדִי אוֹהֵב בְּעֶצֶם אֶת אָחִיו, כִּי בְּשֹׁרֶשׁ הַנְּשָׁמָה כֻּלָּם מְאֻחָדִים. מִי שֶׁלּוֹמֵד לֶאֱהֹב אֶת חֲבֵרָיו יוֹצֵא מִן הַדְּאָגָה רַק לְעַצְמוֹ – וְלִבּוֹ נִפְתָּח גַּם לֶאֱמוּנָה וּלְאַהֲבַת ה׳.
📖 פֶּרֶק א׳, הֲלָכוֹת א׳–ג׳

🤐שְׁמִירַת הַלָּשׁוֹן

יֵשׁ שְׁלוֹשָׁה סוּגִים שֶׁל דִּבּוּר אָסוּר. רְכִילוּת – לְסַפֵּר לְחָבֵר "פְּלוֹנִי אָמַר עָלֶיךָ כָּךְ", אֲפִלּוּ שֶׁזֶּה נָכוֹן. לָשׁוֹן הָרָע – לְסַפֵּר דְּבַר אֱמֶת בִּגְנוּת חָבֵר. הוֹצָאַת שֵׁם רַע – לְסַפֵּר עָלָיו שְׁקָרִים.
חֲזַ״ל אָמְרוּ שֶׁלָּשׁוֹן הָרָע שָׁקוּל כְּנֶגֶד שָׁלֹשׁ הָעֲבֵרוֹת הַחֲמוּרוֹת. וְלֹא יוֹשְׁבִים עִם מִי שֶׁרָגִיל לְסַפֵּר לָשׁוֹן הָרָע, וְלֹא שׁוֹמְעִים לִדְבָרָיו.
💛 אַהֲבַת הַשֵּׁם: בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי חָרַב בַּעֲוֹן שִׂנְאַת חִנָּם וְלָשׁוֹן הָרָע, וְהָרַב קוּק אָמַר שֶׁבְּאַהֲבַת חִנָּם הוּא יִבָּנֶה. כָּל מִלָּה טוֹבָה עַל חָבֵר מְקָרֶבֶת אֶת הַגְּאֻלָּה.
📖 פֶּרֶק א׳, הֲלָכוֹת ב׳, י״א

🗣️לֹא שׂוֹנְאִים – מְדַבְּרִים

הַתּוֹרָה אוֹמֶרֶת: "לֹא תִשְׂנָא אֶת אָחִיךָ בִּלְבָבֶךָ". אָסוּר לִשְׂנֹא יְהוּדִי, אֲפִלּוּ מִי שֶׁהֶעֱלִיב אוֹתְךָ. מָה עוֹשִׂים? לֹא שׁוֹתְקִים וְשׂוֹנְאִים בַּלֵּב, אֶלָּא הוֹלְכִים אֶל הֶחָבֵר וְאוֹמְרִים לוֹ בְּנַחַת: "לָמָּה עָשִׂיתָ לִי כָּךְ?"
וְאִם הֶחָבֵר מְבַקֵּשׁ סְלִיחָה – צָרִיךְ לִמְחֹל לוֹ. וְאִם אִי אֶפְשָׁר לְדַבֵּר אִתּוֹ, מוֹחִים אֶת הַכַּעַס מִן הַלֵּב וְסוֹלְחִים לוֹ בַּלֵּב – וְזוֹ מִדַּת חֲסִידוּת.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה מַאֲמִינָה שֶׁאֶפְשָׁר לִפְתֹּר כָּל בְּעָיָה בֵּין חֲבֵרִים בְּשִׂיחָה כֵּנָה. מִי שֶׁמְּדַבֵּר בְּיֹשֶׁר בִּמְקוֹם לִשְׂנֹא, שׁוֹמֵר עַל לֵב נָקִי – לֵב שֶׁה׳ אוֹהֵב.
📖 פֶּרֶק א׳, הֲלָכוֹת ד׳, ח׳

💬שִׂיחָה בִּמְקוֹם נְקָמָה

טוֹב לְדַבֵּר עִם הֶחָבֵר שֶׁפָּגַע, בִּזְמַן מַתְאִים וּבְנַחַת, וּלְבָרֵר לָמָּה עָשָׂה כָּךְ. אוּלַי בִּכְלָל הָיָה צָרִיךְ אֶת הַפַּטִּישׁ בְּאוֹתוֹ רֶגַע? אָסוּר לְהַסְתִּיר אֶת הַשִּׂנְאָה וּלְחַכּוֹת לִשְׁעַת כֹּשֶׁר לִנְקֹם – הָרַמְבַּ״ם קוֹרֵא לָזֶה מִנְהַג הָרְשָׁעִים.
כְּשֶׁאֵין סִכּוּי לְהִתְפַּיֵּס בְּשִׂיחָה, כִּי הַפּוֹגֵעַ לֹא מוּכָן לְהָבִין – מֻתָּר לִמְחֹל לוֹ בַּלֵּב וְלִשְׁכֹּחַ מֵהָעִנְיָן. הַתּוֹרָה הִקְפִּידָה שֶׁלֹּא תִּשָּׁאֵר שִׂנְאָה.
💛 אַהֲבַת הַשֵּׁם: אַבְשָׁלוֹם שָׁתַק וְשָׂנֵא אֶת אַמְנוֹן בְּלִבּוֹ, וְהַסּוֹף הָיָה רַע מְאוֹד. מִי שֶׁמְּדַבֵּר בְּיֹשֶׁר בּוֹנֶה אֵמוּן וְיָכוֹל לֶאֱהֹב בֶּאֱמֶת – וְאֶת הַמִּצְוָה הַזֹּאת ה׳ חוֹתֵם בִּשְׁמוֹ: "וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ, אֲנִי ה׳".
📖 פֶּרֶק א׳, הֲלָכוֹת ד׳, ו׳, ח׳

🤍מִי נִקְרָא רָשָׁע?

״לֹא תִשְׂנָא אֶת אָחִיךָ בִּלְבָבֶךָ״. מִי שֶׁנִּכְשָׁל מִפַּעַם לְפַעַם וּפוֹגֵעַ בַּחֲבֵרָיו – אֵינוֹ רָשָׁע, וְאָסוּר לִשְׂנֹא אוֹתוֹ. רַק מִי שֶׁפּוֹגֵעַ בַּאֲנָשִׁים בְּאֹפֶן קָבוּעַ, וְגַם אַחֲרֵי שֶׁהֵעִירוּ לוֹ מַמְשִׁיךְ בְּדַרְכּוֹ הָרָעָה – נִקְרָא רָשָׁע.
וַאֲפִלּוּ אֶת הָרָשָׁע צָרִיךְ לֶאֱהֹב בִּפְנִימִיּוּתוֹ, כִּי הוּא יְהוּדִי וְנִשְׁמָתוֹ קְדוֹשָׁה – וּמִתְפַּלְּלִים שֶׁיַּחְזֹר בִּתְשׁוּבָה.
💛 אַהֲבַת הַשֵּׁם: גַּם יְהוּדִי שֶׁחָטָא נִקְרָא ״בֵּן לַה׳״. כְּשֶׁאֲנַחְנוּ רוֹאִים אֶת הַטּוֹב שֶׁבְּתוֹךְ כָּל אֶחָד – אֲנַחְנוּ מִסְתַּכְּלִים כְּמוֹ שֶׁה׳ מִסְתַּכֵּל.
📖 פֶּרֶק א׳, הֲלָכָה ה׳

🔨לֹא תִקֹּם וְלֹא תִטֹּר

רְאוּבֵן בִּקֵּשׁ מִשִּׁמְעוֹן פַּטִּישׁ, וְשִׁמְעוֹן סֵרֵב. לְמָחֳרָת שִׁמְעוֹן מְבַקֵּשׁ מֵרְאוּבֵן מַבְרֵג. אִם רְאוּבֵן יֹאמַר "לֹא אַשְׁאִיל לְךָ, כְּמוֹ שֶׁאַתָּה לֹא הִשְׁאַלְתָּ לִי" – עָבַר עַל "לֹא תִקֹּם". וְאִם יִתֵּן וְיוֹסִיף עֲקִיצָה, כְּמוֹ "תִּרְאֶה שֶׁאֲנִי לֹא כָּמוֹךָ" – עָבַר עַל "לֹא תִטֹּר".
הַדֶּרֶךְ הַנְּכוֹנָה: לִמְחוֹת אֶת הַכַּעַס מִן הַלֵּב וּלְהַשְׁאִיל בְּסֵבֶר פָּנִים יָפוֹת. גַּם מִי שֶׁנּוֹתֵן בְּלִי מִלָּה, אֲבָל שׁוֹמֵר אֶת הַכַּעַס בְּלִבּוֹ – עוֹבֵר עַל "לֹא תִטֹּר".
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁמַּצְלִיחַ לֹא לִנְקֹם וְלֹא לִנְטֹר הוּא אָדָם מְאֻשָּׁר. לִבּוֹ טוֹב, וְאַהֲבָתוֹ לַבְּרִיּוֹת חֲזָקָה וְאֵינָהּ תְּלוּיָה בְּשׁוּם דָּבָר – כְּמוֹ אַהֲבַת ה׳ אֵלֵינוּ.
📖 פֶּרֶק א׳, הֲלָכָה ו׳

✋עוֹצְרִים אֶת הָרִיב

אָדָם נִתְקַל בְּאֶבֶן, נָפַל וְנִפְצַע בַּיָּד. הַאִם הַיָּד תִּקַּח פַּטִּישׁ וְתַכֶּה אֶת הָרֶגֶל? כָּךְ כָּל יִשְׂרָאֵל הֵם כְּמוֹ גּוּף אֶחָד. אִם חָבֵר פָּגַע בְּךָ וְתַחְזִיר לוֹ – רַק יִתּוֹסֵף עוֹד כְּאֵב, וּשְׁנֵיכֶם תִּפָּגְעוּ.
רִיב נִמְשָׁךְ רַק כְּשֶׁיֵּשׁ שְׁנֵי צְדָדִים. כְּשֶׁאֶחָד לֹא מַחֲזִיר – הָרִיב נֶעֱצָר כְּשֶׁהוּא עוֹד קָטָן, כְּמוֹ כַּדּוּר שֶׁלֶג שֶׁעוֹצְרִים אוֹתוֹ בְּרֹאשׁ הַמִּדְרוֹן.
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁלֹּא נוֹקֵם יִהְיוּ לוֹ פָּחוֹת אוֹיְבִים, וְהוּא יִהְיֶה שָׂמֵחַ וְעָנָו. כֹּחוֹתָיו יִהְיוּ פְּנוּיִים לַדְּבָרִים הַטּוֹבִים שֶׁה׳ שָׁלַח אוֹתוֹ לַעֲשׂוֹת בָּעוֹלָם.
📖 פֶּרֶק א׳, הֲלָכָה ז׳

⚖️דָּנִים לְכַף זְכוּת

"בְּצֶדֶק תִּשְׁפֹּט עֲמִיתֶךָ" – וַחֲכָמִים אָמְרוּ: "הֱוֵי דָּן אֶת כָּל הָאָדָם לְכַף זְכוּת". כְּשֶׁרוֹאִים חָבֵר עוֹשֶׂה דָּבָר שֶׁאֶפְשָׁר לְהָבִין לְטוֹבָה וְאֶפְשָׁר לְהָבִין לְרָעָה – מִצְוָה לְהָבִין אוֹתוֹ לְטוֹבָה. וְכֵן כְּשֶׁשּׁוֹמְעִים עָלָיו סִפּוּר.
וְכָל שֶׁכֵּן אָדָם צַדִּיק וִירֵא שָׁמַיִם: גַּם אִם עָשָׂה דָּבָר שֶׁנִּרְאֶה רַע – מְחַפְּשִׂים לוֹ הֶסְבֵּר טוֹב, אֲפִלּוּ הֶסְבֵּר דָּחוּק.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים אָמְרוּ: מִי שֶׁדָּן אֶת חֲבֵרוֹ לְכַף זְכוּת – גַּם בַּשָּׁמַיִם דָּנִים אוֹתוֹ לְכַף זְכוּת. כְּשֶׁאֲנַחְנוּ רוֹאִים אֶת הַנְּשָׁמָה הַטּוֹבָה שֶׁבַּחֲבֵרִים, ה׳ רוֹאֶה אֶת הַטּוֹב שֶׁבָּנוּ.
📖 פֶּרֶק א׳, הֲלָכוֹת ט׳–י׳

🤔🔑 מוּשָּׂג: דָּן לְכַף זְכוּת

״הֱוֵי דָן אֶת כָּל הָאָדָם לְכַף זְכוּת״ (אָבוֹת א׳, ו׳). ״בְּצֶדֶק תִּשְׁפֹּט עֲמִיתֶךָ״ (וַיִּקְרָא י״ט, ט״ו).
📖 כַּף זְכוּת – הַצַּד הַטּוֹב שֶׁבְּמַעֲשֵׂה הֶחָבֵר, לְעֻמַּת כַּף חוֹבָה – הַצַּד הָרָע.
💛 אַהֲבַת הַשֵּׁם: כָּל אָדָם בְּנִשְׁמָתוֹ הוּא טוֹב, וּמִי שֶׁדָּן לְכַף זְכוּת קוֹבֵעַ שֶׁהַנְּשָׁמָה הַטְּהוֹרָה הִיא עִקַּר הָאָדָם. 🧒 וְאַתָּה? חָבֵר לֹא עָנָה לְךָ שָׁלוֹם? חַפֵּשׂ הֶסְבֵּר טוֹב לִפְנֵי שֶׁאַתָּה נִפְגָּע.
📖 פֶּרֶק א׳, הֲלָכָה ט׳

🔎מָתַי חוֹבָה לָדוּן לְכַף זְכוּת

כְּשֶׁאֶפְשָׁר לְהָבִין מַעֲשֶׂה לְטוֹבָה וּלְרָעָה בְּאֹפֶן הֶגְיוֹנִי – מִצְוָה לְהָבִין לְטוֹבָה. אֲבָל כְּשֶׁהַמַּעֲשֶׂה נִרְאֶה רַע, וְקָשֶׁה מְאוֹד לְפָרֵשׁ אוֹתוֹ לְטוֹבָה – אֵין חוֹבָה לָדוּן לְכַף זְכוּת אָדָם רָגִיל, שֶׁאֵינוֹ צַדִּיק מְיֻחָד וְגַם לֹא רָשָׁע.
אֲבָל אָדָם יְרֵא שָׁמַיִם – דָּנִים לְכַף זְכוּת גַּם בְּהֶסְבֵּר דָּחוּק. וְכָל שֶׁהָאָדָם צַדִּיק יוֹתֵר, צָרִיךְ לְהִתְאַמֵּץ יוֹתֵר. מִרְיָם הַנְּבִיאָה נֶעֶנְשָׁה עַל שֶׁלֹּא דָּנָה אֶת מֹשֶׁה רַבֵּנוּ לְכַף זְכוּת.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים הִזְהִירוּ: "הַחוֹשֵׁד בִּכְשֵׁרִים לוֹקֶה בְּגוּפוֹ". מִי שֶׁמְּכַבֵּד אֶת יִרְאֵי ה׳ וּמְחַפֵּשׂ אֶת הַטּוֹב שֶׁבָּהֶם, מְכַבֵּד אֶת הַתּוֹרָה שֶׁהֵם חַיִּים לְפִיהָ.
📖 פֶּרֶק א׳, הֲלָכָה י׳

🤔🔑 מוּשָּׂג: הַחוֹשֵׁד בִּכְשֵׁרִים

״הֲרֵי אֲפִלּוּ אֶת מַעֲשָׂיו שֶׁל מֹשֶׁה רַבֵּנוּ פֵּרַשׁ קֹרַח בְּאֹפֶן שְׁלִילִי... וְאִם דָּן אֶת הַצַּדִּיק לְכַף חוֹבָה – ״הַחוֹשֵׁד בִּכְשֵׁרִים לוֹקֶה בְּגוּפוֹ״ (שַׁבָּת צ״ז, א׳)״.
📖 הַחוֹשֵׁד בִּכְשֵׁרִים – מִי שֶׁחוֹשֵׁב רַע עַל אָדָם יָשָׁר בְּלִי סִבָּה, נֶעְנָש בְּעַצְמוֹ.
💛 אַהֲבַת הַשֵּׁם: אֲפִלּוּ עַל מֹשֶׁה רַבֵּנוּ אֶפְשָׁר לְפָרֵש לְרָעָה, וּבְכָל זֹאת חוֹבָה לָדוּן לְכַף זְכוּת. 🧒 וְאַתָּה? לִפְנֵי שֶׁאַתָּה חוֹשֵׁד בְּמִי שֶׁהוּא יָשָׁר – חֲשֹׁב פַּעֲמַיִם.
📖 פֶּרֶק א׳, הֲלָכָה י׳

🤫📜 לְשׁוֹן חֲכָמִים: רְכִילוּת

״רָכִיל הוּא הַהוֹלֵךְ כְּרוֹכֵל מֵאָדָם לְאָדָם, וְאוֹמֵר: כָּךְ אָמַר עָלֶיךָ פְּלוֹנִי... אַף שֶׁדְּבָרָיו אֱמֶת״.
📖 רְכִילוּת – לְהַעֲבִיר מֵאֶחָד לַשֵּׁנִי ״כָּךְ אָמַר עָלֶיךָ״, גַּם אִם זֶה נָכוֹן.
💛 אַהֲבַת הַשֵּׁם: דִּבּוּרִים כָּאֵלֶּה, גַּם אִם הֵם אֱמֶת, עֲלוּלִים לִגְרֹם לְסִכְסוּכִים וּלְקַלְקֵל יְדִידוּת יָפָה. 🧒 וְאַתָּה? שׁוֹמֵעַ ״כָּךְ אָמַר עָלֶיךָ״? אַל תַּעְבִּיר הָלְאָה.
📖 פֶּרֶק א׳, הֲלָכָה י״א

🤫📜 לְשׁוֹן חֲכָמִים: לָשׁוֹן הָרָע

״אִסּוּר לָשׁוֹן הָרָע שָׁקוּל כְּנֶגֶד שָׁלוֹש הָעֲבֵרוֹת הַחֲמוּרוֹת בְּיוֹתֵר: עֲבוֹדָה זָרָה, גִּלּוּי עֲרָיוֹת וּשְׁפִיכוּת דָּמִים״.
📖 לָשׁוֹן הָרָע – לְסַפֵּר דִּבְרֵי אֱמֶת בִּגְנוּתוֹ שֶׁל חָבֵר, שֶׁגּוֹרְמִים לוֹ קָלוֹן.
💛 אַהֲבַת הַשֵּׁם: חֻרְבַּן בֵּית שֵׁנִי בָּא בַּעֲוֹן שִׂנְאַת חִנָּם וְלָשׁוֹן הָרָע. 🧒 וְאַתָּה? יוֹדֵעַ מַשֶּׁהוּ רַע עַל חָבֵר? שְׁאַל אֶת עַצְמְךָ אִם צָרִיךְ בֶּאֱמֶת לְסַפֵּר.
📖 פֶּרֶק א׳, הֲלָכָה י״א

📖הֶחָפֵץ חַיִּים

רַבִּי אַלֶכְּסַנְדְּרִי הִכְרִיז שֶׁיֵּשׁ לוֹ תְּרוּפָה לַאֲרִיכוּת יָמִים, וְאָמַר: "מִי הָאִישׁ הֶחָפֵץ חַיִּים... נְצֹר לְשׁוֹנְךָ מֵרָע". רַבִּי יִשְׂרָאֵל מֵאִיר מֵרָאדִין כָּתַב סֵפֶר עַל דִּינֵי לָשׁוֹן הָרָע וּרְכִילוּת, וּקְרָאוֹ "חָפֵץ חַיִּים" – וְעַד הַיּוֹם קוֹרְאִים לוֹ בְּשֵׁם סִפְרוֹ.
בְּגִיל תִּשְׁעִים חָגַג הֶחָפֵץ חַיִּים יוֹם הֻלֶּדֶת, כְּדֵי לְפַרְסֵם שֶׁהַהַבְטָחָה מִתְקַיֶּמֶת: מִי שֶׁשּׁוֹמֵר אֶת לְשׁוֹנוֹ – מַאֲרִיךְ יָמִים. הוּא חִבֵּר גַּם אֶת הַסֵּפֶר "מִשְׁנָה בְּרוּרָה".
💛 אַהֲבַת הַשֵּׁם: כָּל פַּעַם שֶׁאֲנַחְנוּ עוֹצְרִים אֶת הַלָּשׁוֹן מִדִּבּוּר רַע – אֲנַחְנוּ עוֹשִׂים רְצוֹן ה׳ וּמַרְבִּים אַהֲבָה בָּעוֹלָם. הֶחָפֵץ חַיִּים הָיָה דֻּגְמָה חַיָּה לְאַהֲבַת יִשְׂרָאֵל.
📖 פֶּרֶק א׳, הֲלָכָה י״ב

🪟מְסַפְּרִים כְּדֵי לְהַצִּיל

כְּדֵי לְהַצִּיל חָבֵר מִנֵּזֶק – צָרִיךְ לְהַזְהִיר אוֹתוֹ, גַּם אִם לֹא שָׁאַל, בִּתְנַאי שֶׁהַכַּוָּנָה טְהוֹרָה וְלֹא מַגְזִימִים. וּמִי שֶׁשָּׁבַר חַלּוֹן שֶׁל הַצִּבּוּר – חַיָּב לְשַׁלֵּם. קֹדֶם מְנַסִּים לְשַׁכְנֵעַ אוֹתוֹ לְשַׁלֵּם, וְאִם הוּא מְסָרֵב – מְסַפְּרִים לָאַחְרָאִים.
"לֹא תַעֲמֹד עַל דַּם רֵעֶךָ" – חַיָּבִים לְהַצִּיל גַּם אֶת הַמָּמוֹן שֶׁל הַחָבֵר וְשֶׁל הַצִּבּוּר, וְלָכֵן אֵין כָּאן לָשׁוֹן הָרָע. הָאִסּוּר לֹא נוֹעַד לְהָגֵן עַל מִי שֶׁמַּזִּיק.
💛 אַהֲבַת הַשֵּׁם: לִפְעָמִים הָאַהֲבָה הָאֲמִתִּית הִיא לַעֲזֹר לְחָבֵר לְתַקֵּן אֶת מָה שֶׁקִּלְקֵל. ה׳ רוֹצֶה חֶבְרָה שֶׁל צֶדֶק וְיֹשֶׁר, וְגַם הַשּׁוֹבֵר מַרְוִיחַ כְּשֶׁהוּא מְתַקֵּן.
📖 פֶּרֶק א׳, הֲלָכוֹת י״ג, כ״א

😳לֹא מַעֲלִיבִים בְּדִבּוּרִים

"וְלֹא תוֹנוּ אִישׁ אֶת עֲמִיתוֹ" – אָסוּר לְצַעֵר חָבֵר בְּדִבּוּר. זֶה נִקְרָא "אוֹנָאַת דְּבָרִים", וְהוּא חָמוּר יוֹתֵר מִלִּגְרֹם לֶחָבֵר הֶפְסֵד שֶׁל כֶּסֶף. לְמָשָׁל, לְחָבֵר שֶׁקָּרָה לוֹ דָּבָר קָשֶׁה – לֹא אוֹמְרִים "זֶה עֹנֶשׁ עַל הַחֲטָאִים שֶׁלְּךָ".
הַמַּלְבִּין פְּנֵי חֲבֵרוֹ בָּרַבִּים – כְּאִלּוּ שׁוֹפֵךְ דָּמִים: מֵרֹב בּוּשָׁה הוּא מַאֲדִים, וְאַחַר כָּךְ פָּנָיו מַחֲוִירוֹת. תָּמָר הֶעֱדִיפָה לְהִשָּׂרֵף בָּאֵשׁ וְלֹא לְבַיֵּשׁ אֶת יְהוּדָה בָּרַבִּים.
💛 אַהֲבַת הַשֵּׁם: עַל הָאִסּוּר הַזֶּה הוֹסִיפָה הַתּוֹרָה: "וְיָרֵאתָ מֵאֱלֹהֶיךָ". ה׳ רוֹאֶה אֶת הַלֵּב הַנִּפְגָּע שֶׁל כָּל יֶלֶד, וּמִי שֶׁשּׁוֹמֵר עַל כְּבוֹד חֲבֵרוֹ – שׁוֹמֵר עַל כְּבוֹדָהּ שֶׁל בְּרִיָּה שֶׁה׳ בָּרָא.
📖 פֶּרֶק א׳, הֲלָכָה י״ד

😳🔑 מוּשָּׂג: הַלְבָּנַת פָּנִים

״נוֹח לוֹ לְאָדָם שֶׁיַּפִּיל עַצְמוֹ לְכִבְשַׁן הָאֵש וְאַל יַלְבִּין פְּנֵי חֲבֵרוֹ בָּרַבִּים״ (בָּבָא מְצִיעָא נ״ט, א׳).
📖 הַלְבָּנַת פָּנִים – לְבַיֵּש אָדָם בָּרַבִּים, עַד שֶׁפָּנָיו מַלְבִּינִים מִבּוּשָׁה.
💛 אַהֲבַת הַשֵּׁם: תָּמָר הֶעְדִּיפָה לֵישָׂרֵף בְּאֵש מִלְּהַעְלִיב אֶת יְהוּדָה בָּרַבִּים. 🧒 וְאַתָּה? אַל תְּצַחֵק עַל טָעוּת שֶׁל חָבֵר מוּל כֻּלָּם.
📖 פֶּרֶק א׳, הֲלָכָה י״ד

🌍אַהֲבַת הַגֵּר

לַגֵּר שֶׁהִצְטָרֵף לְעַם יִשְׂרָאֵל יֵשׁ מִצְוָה מְיֻחֶדֶת: "וַאֲהַבְתֶּם אֶת הַגֵּר", וְאִסּוּר מְיֻחָד לְצַעֵר אוֹתוֹ: "וְגֵר לֹא תוֹנֶה". הָאוֹהֵב אֶת הַגֵּר מְקַיֵּם שְׁתֵּי מִצְווֹת עֲשֵׂה, וְהַמְצַעֵר אוֹתוֹ עוֹבֵר עַל שְׁנֵי לָאוִין.
לָמָּה? כִּי הַגֵּר עָשָׂה מַעֲשֶׂה גָּדוֹל – עָזַב אֶת עַמּוֹ וְאַרְצוֹ וּבָחַר לְהִצְטָרֵף אֵלֵינוּ. וְגַם כִּי הוּא עָלוּל לִהְיוֹת רָגִישׁ יוֹתֵר, וְלָכֵן שׁוֹקְלִים הֵיטֵב כָּל מִלָּה כְּלַפָּיו.
💛 אַהֲבַת הַשֵּׁם: הַגֵּר בָּחַר לְהִכָּנֵס תַּחַת כַּנְפֵי הַשְּׁכִינָה. כְּשֶׁאֲנַחְנוּ אוֹהֲבִים אוֹתוֹ, אֲנַחְנוּ אוֹהֲבִים אָדָם שֶׁאָהַב אֶת ה׳ וְאֶת תּוֹרָתוֹ בְּכָל לִבּוֹ.
📖 פֶּרֶק א׳, הֲלָכָה ט״ו

📢מוֹחִים מִתּוֹךְ אַהֲבָה

מִי שֶׁרוֹאֶה חָבֵר עוֹשֶׂה עֲבֵרָה – מִצְוָה עָלָיו לִמְחוֹת בּוֹ. הַתּוֹכָחָה בָּאָה מִתּוֹךְ אַהֲבָה וְאַחְרָיוּת: כְּמוֹ שֶׁדּוֹאֲגִים שֶׁהֶחָבֵר לֹא יִהְיֶה רָעֵב, כָּךְ דּוֹאֲגִים שֶׁלֹּא יַזִּיק לְנִשְׁמָתוֹ. לָכֵן "הוֹכֵחַ תּוֹכִיחַ" נֶאֱמַר לְיַד "וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ".
מִי שֶׁיָּכוֹל הָיָה לְמַנֹּעַ עֲבֵרָה וְשָׁתַק – כְּאִלּוּ חָטָא בְּעַצְמוֹ. בְּמַחֲלֹקֶת קֹרַח רֹב הָעָם עָמַד מִן הַצַּד וְשָׁתַק, וְנֶעֱנַשׁ עַל כָּךְ.
💛 אַהֲבַת הַשֵּׁם: "כִּי אֶת אֲשֶׁר יֶאֱהַב ה׳ יוֹכִיחַ" – תּוֹכָחָה אֲמִתִּית הִיא סִימָן שֶׁל אַהֲבָה. מִי שֶׁאִכְפַּת לוֹ מֵחֲבֵרוֹ לֹא נוֹתֵן לוֹ לִפֹּל, כְּמוֹ אָב אוֹהֵב.
📖 פֶּרֶק א׳, הֲלָכָה ט״ז

📢📜 לְשׁוֹן חֲכָמִים: מֶחָאָה

״מִצְוָה עַל אָדָם מִיִּשְׂרָאֵל שֶׁרוֹאֶה אֶת חֲבֵרוֹ חוֹטֵא – שֶׁיִּמְחֶה בְּיָדוֹ״.
📖 מֶחָאָה – לְהַבְהִיר בְּקוֹל שֶׁמַּשֶּׁהוּ לֹא בְּסֵדֶר, כְּדֵי לַעְצֹר אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: הַמֶּחָאָה לֹא נוֹבֶעֶת מִשִּׂנְאָה, אֶלָּא מֵאַהֲבָה וְאַחְרָיוּת לְמַצָּבוֹ שֶׁל הֶחָבֵר. 🧒 וְאַתָּה? רוֹאֶה מִישֶׁהוּ עוֹשֶׂה דָּבָר לֹא בְּסֵדֶר – קְרָא לְמִי שֶׁיָּכוֹל לַעְצֹר אוֹתוֹ.
📖 פֶּרֶק א׳, הֲלָכָה ט״ז

📢📜 לְשׁוֹן חֲכָמִים: עַד שֶׁיִּמְחֶה בְּכֹחוֹ

״כֹּל שֶׁאֶפְשָׁר לוֹ לִמְחוֹת בְּאַנְשֵׁי בֵּיתוֹ וְלֹא מָחָה – נִתְפָּש עַל אַנְשֵׁי בֵּיתוֹ. בְּאַנְשֵׁי עִירוֹ – נִתְפָּש עַל אַנְשֵׁי עִירוֹ״ (שַׁבָּת נ״ד, ב׳).
📖 מִי שֶׁהָיָה יָכוֹל לִמְחוֹת וְלֹא מָחָה – נֶחְשָׁב כְּאִלּוּ חָטָא בְּעַצְמוֹ.
💛 אַהֲבַת הַשֵּׁם: הַמֶּחָאָה מְבַטְּאָה אֶת הָאַחְרָיוּת שֶׁל כֻּלָּנוּ זֶה לָזֶה. 🧒 וְאַתָּה? אַל תַּגִּיד ״זֶה לֹא עִנְיָנִי״ – כָּשֶׁאַתָּה יָכוֹל לַעְצֹר דָּבָר רַע.
📖 פֶּרֶק א׳, הֲלָכָה ט״ז

🙇🔑 מוּשָּׂג: חֲנוּפָה

״אָסוּר לְהִתְחַנֵּף לָרְשָׁעִים... שֶׁהֲרֵי כָּל הַמַּחְנִיף לְרָשָׁע מְחַזְּקוֹ וּמְעוֹדְדוֹ לְהַמְשִׁיךְ בִּדְרָכָיו״.
📖 חֲנוּפָה – לְהַחְמִיא לְמִי שֶׁעָשָׂה דָּבָר רַע, כְּדֵי לֹא לְהַרְגִּיז אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: הַמַּחְנִיף לָרָשָׁע נֶעְשֶׂה שֻׁתָּף בְּחִזּוּק הָרַע בָּעוֹלָם. 🧒 וְאַתָּה? חָבֵר עָשָׂה מַעֲשֶׂה לֹא יָפֶה – אַל תְּשַׁבֵּח אוֹתוֹ עָלָיו.
📖 פֶּרֶק א׳, הֲלָכָה י״ז

⚔️מַחֲלֹקֶת – וּמַחֲלֹקֶת לְשֵׁם שָׁמַיִם

חֲכָמִים אָמְרוּ: כָּל הַמַּחֲזִיק בְּמַחֲלֹקֶת עוֹבֵר בְּלָאו, שֶׁנֶּאֱמַר: "וְלֹא יִהְיֶה כְקֹרַח וְכַעֲדָתוֹ". מַחֲלֹקֶת חֲמוּרָה מֵרִיב רָגִיל, כִּי אֲנָשִׁים מִתְמַסְּרִים לִשְׂנֹא זְמַן רַב. וּמַחֲלֹקֶת נֶגֶד תַּלְמִיד חָכָם אוֹ נֶגֶד הָרַב – חֲמוּרָה עוֹד יוֹתֵר.
אֲבָל וִכּוּחַ עִנְיָנִי בְּלִי שִׂנְאָה אִישִׁית, כְּמוֹ שֶׁל הִלֵּל וְשַׁמַּאי, הוּא מַחֲלֹקֶת לְשֵׁם שָׁמַיִם – מִתּוֹכָהּ הַדְּבָרִים מִתְבָּרְרִים, וְסוֹפָהּ לְהִתְקַיֵּם.
💛 אַהֲבַת הַשֵּׁם: הַשָּׁלוֹם וְהָאַחְדוּת מְגַלִּים אֶת הָאֱמוּנָה בַּה׳ הָאֶחָד: כְּמוֹ שֶׁה׳ אֶחָד, כָּךְ עַם יִשְׂרָאֵל צָרִיךְ לִהְיוֹת אֶחָד.
📖 פֶּרֶק א׳, הֲלָכוֹת י״ח–י״ט

🏜️מִדַּת סְדוֹם

אַנְשֵׁי סְדוֹם הָיוּ עֲשִׁירִים, אֲבָל לֹא רָצוּ שֶׁאֲחֵרִים יֵהָנוּ מֵאַרְצָם – גַּם כְּשֶׁזֶּה לֹא עָלָה לָהֶם כְּלוּם. חֲכָמִים אָמְרוּ: כּוֹפִין עַל מִדַּת סְדוֹם. כְּשֶׁזֶּה נֶהֱנֶה וְזֶה לֹא חָסֵר – מְחַיְּבִים אֶת הָאָדָם לָתֵת לַאֲחֵרִים לֵהָנוֹת מִשֶּׁלּוֹ.
בַּמִּשְׁנָה בְּאָבוֹת: "הָאוֹמֵר שֶׁלִּי שֶׁלִּי וְשֶׁלְּךָ שֶׁלְּךָ" – זוֹ מִדָּה בֵּינוֹנִית, וְיֵשׁ אוֹמְרִים שֶׁזּוֹ מִדַּת סְדוֹם. מִי שֶׁחוֹשֵׁב כָּךְ קָרוֹב לְמִדַּת סְדוֹם.
💛 אַהֲבַת הַשֵּׁם: ה׳ מַשְׁפִּיעַ טוֹב עַל כָּל הַבְּרִיּוֹת בְּעַיִן טוֹבָה. כְּשֶׁאֲנַחְנוּ שְׂמֵחִים לָתֵת לַאֲחֵרִים לֵהָנוֹת מִמָּה שֶׁיֵּשׁ לָנוּ, אֲנַחְנוּ הוֹלְכִים בִּדְרָכָיו.
📖 פֶּרֶק א׳, הֲלָכָה כ׳

🏘️🔑 מוּשָּׂג: מִדַּת סְדוֹם

״כּוֹפִים עַל מִדַּת סְדוֹם״ – מְחַיְּבִים אָדָם לָתֵת לַאֲחֵרִים לֵיהָנוֹת מִדְּבָרִים שֶׁשַּׁיָּכִים לוֹ, כְּשֶׁ״זֶה נֶהֱנֶה וְזֶה לֹא חָסֵר״.
📖 מִדַּת סְדוֹם – רְצוֹת שֶׁלֹּא יֵהָנוּ אֲחֵרִים, גַּם כְּשֶׁזֶּה לֹא עוֹלֶה לָנוּ בִּכְלוּם.
💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁנִּהְיֶה הֶפֶך מֵאַנְשֵׁי סְדוֹם, שֶׁלֹּא רָצוּ לָתֵת לָאֲחֵרִים לֵיהָנוֹת מֵהַטּוֹב שֶׁלָּהֶם. 🧒 וְאַתָּה? יֵש לְךָ מַשֶּׁהוּ שֶׁלֹּא חָסֵר לְךָ אִם חָבֵר יִשְׁתַּמֵּש בּוֹ? תֵּן לוֹ.
📖 פֶּרֶק א׳, הֲלָכָה כ׳

🏫לֹא נִגְרָרִים אַחֲרֵי הַכִּתָּה

כִּתָּה שֶׁרוֹצָה לְהֵעָלֵם לִפְנֵי הַשִּׁעוּר כְּדֵי לִצְחֹק עַל הַמּוֹרֶה – אָסוּר לְהִצְטָרֵף. זֶה מַעֲלִיב מְאוֹד אֶת הַמּוֹרֶה, וְזוֹ מַמָּשׁ הַלְבָּנַת פָּנִים בָּרַבִּים. אֵין לָנוּ מֻשָּׂג עִם אֵילוּ קְשָׁיִים הַמּוֹרֶה מִתְמוֹדֵד, וּמָה עֶלְבּוֹן כָּזֶה עָלוּל לִגְרֹם לוֹ.
אַחְדוּת הַכִּתָּה טוֹבָה רַק כְּשֶׁהִיא מְחַזֶּקֶת אֶת הַטּוֹב. לָכֵן הֱיֵה גִּבּוֹר: הוֹדַע שֶׁאַתָּה נִשְׁאָר, וְנַסֵּה לְשַׁכְנֵעַ עוֹד חֲבֵרִים לְהִשָּׁאֵר אִתְּךָ.
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁעוֹמֵד בִּגְבוּרָה לְצַד הַטּוֹב, גַּם כְּשֶׁרַבִּים הוֹלְכִים לְצַד אַחֵר, נֶאֱמָן לַטּוֹב שֶׁה׳ שָׂם בְּלִבּוֹ. וְלִפְעָמִים צַעַד אֶחָד אַמִּיץ מְשַׁנֶּה אֶת כָּל הַכִּתָּה לְטוֹבָה.
📖 פֶּרֶק א׳, הֲלָכָה כ״ב

🧠 בְּדֹק אֶת עַצְמְךָ

שִׁמְעוֹן לֹא הִשְׁאִיל לִרְאוּבֵן אֶת הָאוֹפַנַּיִם. לְמָחֳרָת שִׁמְעוֹן מְבַקֵּשׁ כַּדּוּר, וּרְאוּבֵן נוֹתֵן וְאוֹמֵר: "תִּרְאֶה שֶׁאֲנִי לֹא כָּמוֹךָ". מָה הַדִּין?
💡 הוּא לֹא נָקַם, כִּי הִשְׁאִיל. אֲבָל כָּל לַעַג שֶׁקָּשׁוּר לְמָה שֶׁהָיָה אֶתְמוֹל הוּא נְטִירָה. צָרִיךְ לִמְחוֹת אֶת הַכַּעַס מִן הַלֵּב.
אַרְיֵה פָּגַע בְּדָוִד, וְדָוִד כּוֹעֵס עָלָיו. מָה הַדֶּרֶךְ שֶׁהַתּוֹרָה מַעְדִּיפָה?
💡 הַתּוֹרָה רוֹצָה שִׂיחָה כֵּנָה: "לָמָּה עָשִׂיתָ לִי כָּךְ?". שְׁתִיקָה עִם שִׂנְאָה בַּלֵּב הִיא מִנְהַג הָרְשָׁעִים, וְסִפּוּר לְכָל הַכִּתָּה הוּא לָשׁוֹן הָרָע.
רָאִיתָ אֶת רַב בֵּית הַכְּנֶסֶת, אִישׁ יְרֵא שָׁמַיִם, יוֹצֵא בְּאֶמְצַע הַתְּפִלָּה. מָה נָכוֹן לַחְשֹׁב?
💡 אֵצֶל אָדָם צַדִּיק, כָּל זְמַן שֶׁאֶפְשָׁר לְפָרֵשׁ לְטוֹבָה – אֲפִלּוּ בְּהֶסְבֵּר דָּחוּק – מִצְוָה לָדוּן אוֹתוֹ לְכַף זְכוּת. "הַחוֹשֵׁד בִּכְשֵׁרִים לוֹקֶה בְּגוּפוֹ".
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

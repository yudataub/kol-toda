הִלְכוֹת גְּנֵבַת דַּעַת

🤝 בֵּין אָדָם לַחֲבֵרוֹ · פֶּרֶק ג׳

🙈גְּנֵבַת דַּעַת

אָסוּר לִגְרֹם לְחָבֵר לַחְשֹׁב שֶׁעָשִׂינוּ לְמַעֲנוֹ טוֹבָה, כְּשֶׁבֶּאֱמֶת לֹא עָשִׂינוּ. זוֹ "גְּנֵבַת דַּעַת". הִיא יְכוֹלָה לִהְיוֹת בְּדִבּוּר, וְגַם בְּמַעֲשֶׂה אוֹ בְּהַצָּגָה. רַבֵּנוּ יוֹנָה כָּתַב שֶׁאֵצֶל חֲכָמִים גְּנֵבַת דַּעַת חֲמוּרָה אֲפִלּוּ מִגָּזֵל.
הַמָּקוֹר: "לֹא תִּגְנֹבוּ" – גַּם גְּנֵבַת לֵב. אַבְשָׁלוֹם גָּנַב אֶת לֵב אַנְשֵׁי יִשְׂרָאֵל: הוּא עָשָׂה אֶת עַצְמוֹ כְּאִלּוּ הוּא דּוֹאֵג לָהֶם, וּבֶאֱמֶת דָּאַג רַק לְעַצְמוֹ.
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁגּוֹנְבִים אֶת דַּעְתּוֹ שׁוּב וָשׁוּב מִתְאַכְזֵב, וְעָלוּל לְהַפְסִיק לְהַאֲמִין בַּחֲבֵרִים. יְדִידוּת שֶׁל אֱמֶת שׁוֹמֶרֶת עַל הַלֵּב הַפָּתוּחַ שֶׁה׳ נָתַן לָנוּ, הַלֵּב שֶׁיּוֹדֵעַ לֶאֱהֹב.
📖 פֶּרֶק ג׳, הֲלָכוֹת א׳–ב׳

🍰עֶזְרָה שֶׁל אֱמֶת

יֵשׁ מִי שֶׁמְּחַכֶּה שֶׁהַכֹּל יִהְיֶה מוּכָן אֵצֶל הַשְּׁכֵנִים, וְרַק אָז מַצִּיעַ עֶזְרָה וּמִצְטַעֵר שֶׁלֹּא זָכָה לַעֲזֹר. זוֹ גְּנֵבַת דַּעַת. וְכֵן אָסוּר לוֹמַר לְאוֹרֵחַ "פָּתַחְנוּ אֶת הַבַּקְבּוּק לִכְבוֹדְךָ" – כְּשֶׁמִּמֵּילָא רָצִינוּ לִפְתֹּחַ אוֹתוֹ.
וְאָסוּר לְהַגִּיד לָאוֹרֵחַ שֶׁהַיַּיִן מְשֻׁבָּח מְאוֹד, כְּשֶׁהוּא פָּשׁוּט – רַק כְּדֵי שֶׁיַּכִּיר לָנוּ טוֹבָה.
💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹאֶה אֶת הַלֵּב, לֹא רַק אֶת הַהַצָּגָה. עֶזְרָה קְטַנָּה שֶׁל אֱמֶת שָׁוָה יוֹתֵר מֵהַרְבֵּה מִלִּים יָפוֹת.
📖 פֶּרֶק ג׳, הֲלָכוֹת ד׳, ח׳

🎁הַזְמָנָה מִן הַלֵּב

אוֹרֵחַ הִגִּיעַ בִּזְמַן הָאֲרוּחָה. אִם בַּעַל הַבַּיִת לֹא רוֹצֶה שֶׁיֹּאכַל אִתּוֹ – עָדִיף שֶׁלֹּא יַזְמִין, כִּי הַזְמָנָה כָּזֹאת הִיא "אֶחָד בַּפֶּה וְאֶחָד בַּלֵּב". אוֹ שֶׁיְּשַׁנֶּה אֶת הַהַרְגָּשָׁה, וְיַזְמִין בִּרְצוֹן אֲמִתִּי.
אִם יוֹדֵעַ שֶׁהָאוֹרֵחַ יְסָרֵב, אֲבָל בְּלִבּוֹ מַסְכִּים לְאָרֵחַ – מֻתָּר לְהַזְמִין פַּעַם אוֹ פַּעֲמַיִם, מִשּׁוּם דֶּרֶךְ אֶרֶץ. אֲבָל לְהַפְצִיר רַק כִּי יוֹדְעִים שֶׁיְּסָרֵב – זוֹ גְּנֵבַת דַּעַת. וְכֵן מִי שֶׁשּׁוֹלֵחַ מַתָּנָה רַק כִּי יוֹדֵעַ שֶׁלֹּא יְקַבְּלוּ אוֹתָהּ.
💛 אַהֲבַת הַשֵּׁם: וּמִי שֶׁבֶּאֱמֶת אוֹהֵב אֶת הָאוֹרֵחַ – מֻתָּר לוֹ לְהַפְצִיר בּוֹ, אֲפִלּוּ שֶׁיּוֹדֵעַ שֶׁלֹּא יָבוֹא. רָאוּי שֶׁאָדָם יְגַלֶּה אֶת אַהֲבָתוֹ לַחֲבֵרוֹ, וְזוֹ הָאַהֲבָה שֶׁה׳ רוֹצֶה בָּהּ.
📖 פֶּרֶק ג׳, הֲלָכוֹת ד׳–ה׳

🎁🔑 מוּשָּׂג: אֶחָד בַּפֶּה וְאֶחָד בַּלֵּב

״אִם בְּעַל הַבַּיִת אֵינוֹ רוֹצֶה לְהַזְמִין, וּבְכָל זֹאת מַזְמִין, יִמָּצֵא אוֹמֵר אֶחָד בַּפֶּה וְאֶחָד בַּלֵּב, וְזֶה כַּמּוּבָן שֶׁקֶר״.
📖 אֶחָד בַּפֶּה וְאֶחָד בַּלֵּב – לְדַבֵּר הֶפֶךְ מִמָּה שֶׁחוֹשְׁבִים בֶּאֱמֶת.
💛 אַהֲבַת הַשֵּׁם: עָדִיף שֶׁלֹּא לְהַזְמִין בִּכְלָל, מִלְּהַזְמִין בְּפֶה שֶׁאֵינוֹ שָׁוֶה לַלֵּב. 🧒 וְאַתָּה? כְּשֶׁאַתָּה מַצִּיעַ מַשֶּׁהוּ – תִּרְצֶה בֶּאֱמֶת שֶׁהַשֵּׁנִי יַסְכִּים.
📖 פֶּרֶק ג׳, הֲלָכָה ד׳

🎁🔑 מוּשָּׂג: גְּנֵבַת דַּעַת

״אִם בְּתוֹךְ לִבּוֹ יֵשׁ רָצוֹן אֲמִתִּי לְהַזְמִין, מֻתָּר. אֲבָל אִם אֵינוֹ מְעוּנְיָן לְהַזְמִין, וּבְכָל זֹאת מַפְצִיר – זוֹ גְּנֵבַת דַּעַת״.
📖 גְּנֵבַת דַּעַת – לְיַצֵּר בְּלֵב הֶחָבֵר רֹשֶׁם לֹא נָכוֹן, כְּאִלּוּ אוֹהֲבִים אוֹתוֹ מְאֹד, כְּשֶׁזֶּה לֹא אֱמֶת.
💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁהַלֵּב וְהַפֶּה יִהְיוּ שָׁוִים, וְלֹא אֶחָד בַּפֶּה וְאֶחָד בַּלֵּב. 🧒 וְאַתָּה? מַצִּיעַ עֶזְרָה – רַק כְּשֶׁאַתָּה בֶּאֱמֶת רוֹצֶה לְעֲזֹר.
📖 פֶּרֶק ג׳, הֲלָכָה ה׳

🐔גַּם דַּעְתּוֹ שֶׁל גּוֹי

שְׁמוּאֵל אָמַר לַשַּׁמָּשׁ שֶׁלּוֹ לָתֵת מַתָּנָה לַגּוֹי שֶׁהֶעֱבִיר אוֹתָם אֶת הַנָּהָר. הַשַּׁמָּשׁ נָתַן תַּרְנְגֹלֶת טְרֵפָה וְאָמַר שֶׁהִיא כְּשֵׁרָה, כְּדֵי שֶׁהַגּוֹי יַחֲזִיק לוֹ טוֹבָה. שְׁמוּאֵל הִקְפִּיד עָלָיו וְנָזַף בּוֹ.
לַגּוֹי אֵין הֶבְדֵּל בַּמְּחִיר – וּבְכָל זֹאת זוֹ גְּנֵבַת דַּעַת, כִּי הוּא חוֹשֵׁב שֶׁקִּבֵּל מַתָּנָה חֲשׁוּבָה יוֹתֵר. זֶה אִסּוּר גָּמוּר, וְלֹא רַק מִדַּת חֲסִידוּת.
💛 אַהֲבַת הַשֵּׁם: הָאֱמֶת אֵינָהּ רַק לְמַעַן הַחֲבֵרִים הַקְּרוֹבִים. מִי שֶׁהוֹלֵךְ בְּדַרְכֵי ה׳ יָשָׁר עִם כָּל אָדָם, וְכָךְ שֵׁם שָׁמַיִם מִתְכַּבֵּד עַל יָדוֹ.
📖 פֶּרֶק ג׳, הֲלָכָה ו׳

🚶הוּא טָעָה מֵעַצְמוֹ

רָבָא וְרַב סָפְרָא יָצְאוּ מִן הָעִיר, וּפָגְשׁוּ אֶת מָר זוּטְרָא שֶׁהָיָה בַּדֶּרֶךְ אֲלֵיהֶם. הוּא חָשַׁב שֶׁיָּצְאוּ לִכְבוֹדוֹ. רַב סָפְרָא אָמַר לוֹ: "לֹא יָדַעְנוּ שֶׁאַתָּה בָּא". רָבָא שָׁאַל אוֹתוֹ: לָמָּה צִעַרְתָּ אוֹתוֹ? הֲרֵי הוּא טָעָה מֵעַצְמוֹ.
הַהֲלָכָה כְּרָבָא: גְּנֵבַת דַּעַת אֲסוּרָה רַק כְּשֶׁגּוֹרְמִים לֶחָבֵר לַחְשֹׁב טוֹבָה שֶׁאֵינָהּ. אִם הוּא חָשַׁב כָּךְ מֵעַצְמוֹ – אֵין חוֹבָה לְתַקֵּן אוֹתוֹ. רַב סָפְרָא הִקְפִּיד בִּמְיֻחָד עַל הָאֱמֶת, וְלָכֵן הֶעֱדִיף לוֹמַר.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה דּוֹרֶשֶׁת יֹשֶׁר, וְגַם רְגִישׁוּת: אֵין צֹרֶךְ לְצַעֵר אָדָם בְּלִי סִבָּה. כָּךְ מִתְחַבְּרוֹת הָאֱמֶת וְהָאַהֲבָה, שֶׁתַּרְוַיְהוּ מִדּוֹתָיו שֶׁל ה׳.
📖 פֶּרֶק ג׳, הֲלָכָה ז׳

👑לִכְבוֹד הַשְּׁכֵנִים

אָדָם חָשׁוּב שֶׁבָּא אֶל שְׁכֵנָיו בְּשִׂמְחָה אוֹ בְּצַעַר, וְיֵשׁ שָׁם אוֹרְחִים – מֻתָּר לוֹ לְהַצִּיעַ עֶזְרָה אוֹ לְהָבִיא סִיר אֹכֶל, גַּם כְּשֶׁהוּא יוֹדֵעַ שֶׁאֵין צֹרֶךְ, וְשֶׁבַּסּוֹף יַחֲזִיר אֶת הַסִּיר הַבַּיְתָה.
כָּאן אֵין גְּנֵבַת דַּעַת, כִּי הָעִקָּר הוּא הַכָּבוֹד: עַצְם הָרָצוֹן שֶׁלּוֹ לַעֲזֹר מְכַבֵּד אֶת הַשְּׁכֵנִים בְּעֵינֵי הָאוֹרְחִים, וְזֶה נוֹתֵן לָהֶם נַחַת רוּחַ.
💛 אַהֲבַת הַשֵּׁם: לִפְעָמִים הַיַּחַס חָשׁוּב יוֹתֵר מֵהַמַּתָּנָה. ה׳ רוֹצֶה שֶׁנְּכַבֵּד זֶה אֶת זֶה וְנַרְבֶּה שִׂמְחָה בְּלֵב חֲבֵרֵינוּ.
📖 פֶּרֶק ג׳, הֲלָכָה ח׳

🧠 בְּדֹק אֶת עַצְמְךָ

אוֹרֵחַ הִגִּיעַ בִּזְמַן הָאֲרוּחָה. אַבָּא יוֹדֵעַ שֶׁהוּא מְמַהֵר וִיסָרֵב, אֲבָל בְּלִבּוֹ שָׂמֵחַ לְאָרֵחַ אוֹתוֹ. מָה הַדִּין?
💡 כְּשֶׁיֵּשׁ בַּלֵּב רָצוֹן לְאָרֵחַ, הַהַזְמָנָה אֵינָהּ שֶׁקֶר – וּבְלִי הַזְמָנָה הָאוֹרֵחַ עָלוּל לְהֵעָלֵב. אֲבָל לְהַפְצִיר רַק כְּדֵי לְהֵרָאוֹת אוֹהֵב – זוֹ גְּנֵבַת דַּעַת.
שִׁמְעוֹן לֹא רוֹצֶה שֶׁחֲבֵרוֹ יֹאכַל אֶצְלוֹ, אֲבָל מַפְצִיר בּוֹ – כִּי הוּא יוֹדֵעַ שֶׁהֶחָבֵר יְסָרֵב. מָה הַדִּין?
💡 הַהַפְצָרָה נוֹעֲדָה רַק לְהֵרָאוֹת כְּאוֹהֵב גָּדוֹל, וּבֶאֱמֶת הוּא לֹא רוֹצֶה. זוֹ גְּנֵבַת דַּעַת – וְכֵן מִי שֶׁשּׁוֹלֵחַ מַתָּנָה רַק כִּי יוֹדֵעַ שֶׁלֹּא יְקַבְּלוּ אוֹתָהּ.
רוֹנִי יָשַׁב בַּתַּחֲנָה. דָּוִד יָרַד מֵהָאוֹטוֹבּוּס וְחָשַׁב שֶׁרוֹנִי בָּא לְקַבֵּל אוֹתוֹ, אֲבָל רוֹנִי סְתָם חִכָּה שָׁם. מָה הַדִּין?
💡 גְּנֵבַת דַּעַת הִיא רַק כְּשֶׁגּוֹרְמִים לֶחָבֵר לַחְשֹׁב טוֹבָה שֶׁאֵינָהּ. כְּשֶׁהוּא טָעָה מֵעַצְמוֹ – אֵין חוֹבָה לְתַקֵּן, כְּמוֹ שֶׁאָמַר רָבָא. רַב סָפְרָא הֶעֱדִיף לוֹמַר, מִתּוֹךְ הִדּוּר בָּאֱמֶת.

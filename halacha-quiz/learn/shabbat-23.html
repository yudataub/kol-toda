מֻקְצֶה

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק כ״ג

📵מָה זֶה מֻקְצֶה?

חֲכָמִים אָסְרוּ לְטַלְטֵל בְּשַׁבָּת דְּבָרִים שֶׁאֵינָם רְאוּיִים לְשַׁבָּת, שֶׁאָדָם "מַקְצֶה" – מַפְרִישׁ – אוֹתָם מִדַּעְתּוֹ. לְמָשָׁל: אֲבָנִים, חוֹל וּבַעֲלֵי חַיִּים; תַּפּוּחֵי אֲדָמָה חַיִּים; כֶּסֶף, מַצְלֵמָה וּכְלֵי נְגִינָה; טֶלֶפוֹן נַיָּד, עֵט וּפַטִּישׁ.
אָסוּר לְהָזִיז מֻקְצֶה בַּיָּדַיִם, אֲבָל מֻתָּר לָגַעַת בּוֹ בְּלִי לְהָזִיז, וּמֻתָּר לְכַסּוֹת אוֹתוֹ בְּמַפָּה. מַאֲכָלִים וּסְפָרִים מֻתָּר לְטַלְטֵל תָּמִיד.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת גַּם הַיָּדַיִם נָחוֹת. כְּשֶׁהַיָּדַיִם לֹא עֲסוּקוֹת בְּטֶלֶפוֹן וּבְכֶסֶף, הֵן פְּנוּיוֹת לְחַבֵּק, לְהַגִּישׁ אֹכֶל וּלְדַפְדֵּף בְּסֵפֶר.
📖 פֶּרֶק כ״ג, הֲלָכוֹת א׳–ד׳, ט׳

✋גַּם הַיָּדַיִם שׁוֹבְתוֹת

לָמָּה תִּקְּנוּ חֲכָמִים אֶת אִסּוּר הַמֻּקְצֶה? כְּדֵי שֶׁלֹּא נְבַלֶּה אֶת הַשַּׁבָּת בְּסִדּוּר חֲפָצִים וְאוֹצָרוֹת, כְּדֵי שֶׁגַּם הַיָּדַיִם יָנוּחוּ, וּכְדֵי לְהַרְחִיק אוֹתָנוּ מִמְּלָאכָה. כָּךְ הַשַּׁבָּת נִכֶּרֶת גַּם אֵצֶל מִי שֶׁאֵינוֹ עוֹבֵד בִּמְלָאכָה כָּל הַשָּׁבוּעַ.
הַמֻּקְצֶה מַתְאִים בֵּין הַמַּחֲשָׁבָה לַיָּדַיִם: מָה שֶׁאֵינוֹ רָאוּי לְשַׁבָּת – גַּם הַיָּדַיִם לֹא נוֹגְעוֹת בּוֹ כְּדֵי לְהָזִיז.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁהַיָּדַיִם נָחוֹת, גַּם הַלֵּב נָח. הַשַּׁבָּת מְלַמֶּדֶת אוֹתָנוּ לִהְיוֹת שְׁלֵמִים – בְּמַחֲשָׁבָה, בְּדִבּוּר וּבְמַעֲשֶׂה – עִם ה׳.
📖 פֶּרֶק כ״ג, הֲלָכָה א׳

📵🔑 מוּשָּׂג: מֻקְצֶה

חֲכָמִים אָסְרוּ לְטַלְטֵל בְּשַׁבָּת דְּבָרִים שֶׁאֵינָם רְאוּיִים לְשַׁבָּת, שֶׁאָדָם ״מַקְצֶה״ – מַפְרִישׁ – אוֹתָם מִדַּעְתּוֹ. לְמָשָׁל: אֲבָנִים וְכֶסֶף, מַצְלֵמָה וּטֶלֶפוֹן נַיָּד.
📖 אָסוּר לְהָזִיז מֻקְצֶה בַּיָּדַיִם, אֲבָל מַאֲכָלִים וּסְפָרִים מֻתָּר לְטַלְטֵל תָּמִיד.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת גַּם הַיָּדַיִם נָחוֹת. כְּשֶׁהַיָּדַיִם לֹא עֲסוּקוֹת בְּטֶלֶפוֹן וּבְכֶסֶף, הֵן פְּנוּיוֹת לְחַבֵּק, לְהַגִּישׁ אֹכֶל וּלְדַפְדֵּף בְּסֵפֶר. 🧒 וְאַתָּה? תַּשְׁאִיר אֶת הַטֶּלֶפוֹן בַּצַּד כָּל הַשַּׁבָּת, וְתָשִׂים לָיַד הַמִּטָּה סֵפֶר.
📖 פֶּרֶק כ״ג, הֲלָכוֹת א׳–ד׳, ט׳

🔨פַּטִּישׁ, קְלִפּוֹת וְרֶגֶל

כְּלִי שֶׁעוֹשִׂים בּוֹ בְּדֶרֶךְ כְּלָל מְלָאכָה אֲסוּרָה, כְּמוֹ פַּטִּישׁ אוֹ מִסְפָּרַיִם, מֻתָּר לְהָזִיז רַק לְצֹרֶךְ: פַּטִּישׁ כְּדֵי לְפַצֵּחַ אֱגוֹזִים, מִסְפָּרַיִם כְּדֵי לִפְתֹּחַ שַׂקִּית חָלָב, אוֹ כְּשֶׁהַכְּלִי מֻנָּח בַּמָּקוֹם שֶׁצְּרִיכִים. אֲבָל אֶבֶן – אָסוּר לְהָזִיז גַּם לְפִצּוּחַ אֱגוֹזִים.
קְלִפּוֹת וְגַרְעִינִים עַל הַשֻּׁלְחָן – מֻתָּר לְפַנּוֹת בַּיָּד לַפַּח, כִּי הֵם מַפְרִיעִים. וּמֻקְצֶה שֶׁעַל הָרִצְפָּה מֻתָּר לְהָזִיז בָּרֶגֶל.
💛 אַהֲבַת הַשֵּׁם: הַהֲלָכָה דּוֹאֶגֶת שֶׁהַבַּיִת יִהְיֶה נָקִי וְנָעִים בְּשַׁבָּת. כָּךְ הַשַּׁבָּת גַּם קְדוֹשָׁה וְגַם נְעִימָה – כְּמוֹ מַלְכָּה שֶׁבָּאָה לְבַקֵּר.
📖 פֶּרֶק כ״ג, הֲלָכוֹת ג׳, ז׳, י״ב, י״ד

🔨📜 לְשׁוֹן חֲכָמִים: מֻקְצֶה בָּרֶגֶל

כְּלִי שֶׁעוֹשִׂים בּוֹ בְּדֶרֶךְ כְּלָל מְלָאכָה אֲסוּרָה, כְּמוֹ פַּטִּישׁ אוֹ מִסְפָּרַיִם, מֻתָּר לְהָזִיז רַק לְצֹרֶךְ. אֲבָל מֻקְצֶה שֶׁעַל הָרִצְפָּה, כְּמוֹ מַטְבֵּעַ, מֻתָּר לְהָזִיז בָּרֶגֶל.
📖 קְלִפּוֹת וְגַרְעִינִים עַל הַשֻּׁלְחָן – מֻתָּר לְפַנּוֹת בַּיָּד לַפַּח, כִּי הֵם מַפְרִיעִים.
💛 אַהֲבַת הַשֵּׁם: הַהֲלָכָה דּוֹאֶגֶת שֶׁהַבַּיִת יִהְיֶה נָקִי וְנָעִים בְּשַׁבָּת. 🧒 וְאַתָּה? אִם רָאִית מַטְבֵּעַ בָּרִצְפָּה בְּשַׁבָּת – הַזֵּז אוֹתוֹ בָּרֶגֶל, וְלֹא בַּיָּד.
📖 פֶּרֶק כ״ג, הֲלָכוֹת ג׳, ז׳, י״ב, י״ד

🧺בָּסִיס לְדָבָר הָאָסוּר

מִי שֶׁהִנִּיחַ כֶּסֶף עַל שֻׁלְחָן לִפְנֵי שַׁבָּת, כְּדֵי שֶׁיִּשָּׁאֵר שָׁם בְּשַׁבָּת – גַּם הַשֻּׁלְחָן נַעֲשָׂה מֻקְצֶה, כִּי הוּא "בָּסִיס לְדָבָר הָאָסוּר". וְגַם אִם הַכֶּסֶף נָפַל בְּאֶמְצַע הַשַּׁבָּת – הַשֻּׁלְחָן נִשְׁאָר מֻקְצֶה כָּל הַשַּׁבָּת, כִּי בִּתְחִלַּת הַשַּׁבָּת, בֵּין הַשְּׁמָשׁוֹת, הוּא הָיָה בָּסִיס.
אִם שָׁכַח אֶת הַכֶּסֶף עַל הַשֻּׁלְחָן – הַשֻּׁלְחָן אֵינוֹ מֻקְצֶה. מַטִּים אוֹתוֹ וּמַפִּילִים אֶת הַכֶּסֶף. וּמִי שֶׁמָּצָא כֶּסֶף בְּכִיס בִּגְדוֹ – יְנַעֵר אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: מָה שֶׁאָנוּ מַחְלִיטִים לִפְנֵי שַׁבָּת קוֹבֵעַ אֵיךְ תֵּרָאֶה הַשַּׁבָּת. הֲכָנָה טוֹבָה בְּעֶרֶב שַׁבָּת הִיא חֵלֶק מִכְּבוֹד הַשַּׁבָּת.
📖 פֶּרֶק כ״ג, הֲלָכוֹת ה׳, י׳

🧺📜 לְשׁוֹן חֲכָמִים: בָּסִיס לְדָבָר הָאָסוּר

מִי שֶׁהִנִּיחַ כֶּסֶף עַל שֻׁלְחָן לִפְנֵי שַׁבָּת, כְּדֵי שֶׁיִּשָּׁאֵר שָׁם בְּשַׁבָּת – גַּם הַשֻּׁלְחָן נַעֲשָׂה מֻקְצֶה, כִּי הוּא ״בָּסִיס לְדָבָר הָאָסוּר״. גַּם אִם הַכֶּסֶף נָפַל בְּאֶמְצַע הַשַּׁבָּת – הַשֻּׁלְחָן נִשְׁאָר מֻקְצֶה כָּל הַשַּׁבָּת.
📖 אִם שָׁכַח אֶת הַכֶּסֶף עַל הַשֻּׁלְחָן בְּלִי כַּוָּנָה – הַשֻּׁלְחָן אֵינוֹ מֻקְצֶה.
💛 אַהֲבַת הַשֵּׁם: מָה שֶׁאָנוּ מַחְלִיטִים לִפְנֵי שַׁבָּת קוֹבֵעַ אֵיךְ תֵּרָאֶה הַשַּׁבָּת. 🧒 וְאַתָּה? בֶּעֶרֶב שַׁבָּת, פַּנֵּה מֵהַשֻּׁלְחָנוֹת כֶּסֶף וְכֵלִים שֶׁלֹּא תִּרְצֶה שֶׁיַּפְרִיעוּ לְךָ בַּשַּׁבָּת.
📖 פֶּרֶק כ״ג, הֲלָכוֹת ה׳, י׳

📱טֶלֶפוֹן, סִיר וְתַנּוּר

טֶלֶפוֹן נַיָּד מְשַׁמֵּשׁ בְּעִקָּר לְדִבּוּר, וְלָכֵן הוּא כְּלִי שֶׁמְּלַאכְתּוֹ לְאִסּוּר. מֻתָּר לְהָזִיז אוֹתוֹ לְצֹרֶךְ גּוּפוֹ – כְּדֵי לִרְאוֹת אֶת הַשָּׁעָה, אֲבָל לֹא לָשֵׂאת אוֹתוֹ בַּכִּיס לְשֵׁם כָּךְ. וְאִם הַשָּׁעוֹן הַמְּעוֹרֵר שֶׁבּוֹ מְצַלְצֵל וּמַפְרִיעַ – מֻתָּר לְהַרְחִיק אוֹתוֹ.
סִיר שֶׁיֵּשׁ בּוֹ תַּבְשִׁיל – מֻתָּר לְטַלְטֵל, כִּי הוּא טָפֵל לָאֹכֶל. וּמֻתָּר לִפְתֹּחַ אֶת דֶּלֶת הַתַּנּוּר כְּדֵי לְהוֹצִיא מִמֶּנּוּ מַאֲכָלִים.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת הַטֶּלֶפוֹן שׁוֹתֵק, וְאָז אֲנַחְנוּ שׁוֹמְעִים זֶה אֶת זֶה. זוֹ מַתָּנָה שֶׁל שֶׁקֶט – זְמַן לִהְיוֹת יַחַד בְּאֱמֶת.
📖 פֶּרֶק כ״ג, הֲלָכָה ח׳

📱🔑 מוּשָּׂג: כְּלִי שֶׁמְּלַאכְתּוֹ לְאִסּוּר

טֶלֶפוֹן נַיָּד מְשַׁמֵּשׁ בְּעִקָּר לְדִבּוּר, וְלָכֵן הוּא כְּלִי שֶׁמְּלַאכְתּוֹ לְאִסּוּר. מֻתָּר לְהָזִיז אוֹתוֹ לְצֹרֶךְ גּוּפוֹ אוֹ מְקוֹמוֹ, אֲבָל לֹא לָשֵׂאת אוֹתוֹ סְתָם.
📖 וְאִם הַשָּׁעוֹן הַמְּעוֹרֵר שֶׁבּוֹ מְצַלְצֵל וּמַפְרִיעַ – מֻתָּר לְהַרְחִיק אוֹתוֹ, כְּדֵי שֶׁהַצִּלְצוּל לֹא יַפְרִיעַ.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת הַטֶּלֶפוֹן שׁוֹתֵק, וְאָז אֲנַחְנוּ שׁוֹמְעִים זֶה אֶת זֶה. 🧒 וְאַתָּה? אִם טֶלֶפוֹן מְצַלְצֵל בְּשַׁבָּת – הַרְחֵק אוֹתוֹ בְּלִי לִלְחֹץ עָלָיו.
📖 פֶּרֶק כ״ג, הֲלָכָה ח׳

🧽נִשְׁפַּךְ מִיץ – בְּמָה סוֹפְגִים?

אָסוּר לְהִשְׁתַּמֵּשׁ בִּכְלִי אוֹ בְּבֶגֶד כָּךְ שֶׁלֹּא יִהְיֶה עוֹד רָאוּי לַתַּפְקִיד שֶׁלּוֹ – זֶה כְּאִלּוּ סוֹתְרִים אוֹתוֹ. לָכֵן מַשְׁקֶה שֶׁנִּשְׁפַּךְ עַל הָרִצְפָּה – לֹא סוֹפְגִים בִּבְגָדִים, אֲבָל מֻתָּר לִסְפֹּג אוֹתוֹ בִּסְמַרְטוּט.
מֻתָּר לַאֲנָשִׁים רַבִּים לְנַגֵּב יָדַיִם בְּמַגֶּבֶת, עַד שֶׁכְּבָר לֹא תִּהְיֶה רְאוּיָה, וּמֻתָּר לְהַנִּיחַ שַׂקִּית בְּפַח הָאַשְׁפָּה. אֲבָל אָסוּר לְהַרְטִיב כָּרִים וּכְסָתוֹת אוֹ לְטַנֵּף בְּגָדִים.
💛 אַהֲבַת הַשֵּׁם: גַּם הַחֲפָצִים שֶׁבַּבַּיִת נָחִים בַּשַּׁבָּת וּמְשַׁמְּשִׁים כָּל אֶחָד בְּתַפְקִידוֹ – כָּךְ נִשְׁמֶרֶת שַׁלְוַת הַבַּיִת כֻּלּוֹ.
📖 פֶּרֶק כ״ג, הֲלָכָה י״א

🪚מָה שֶׁהִגִּיעַ לַיָּד בְּהֶתֵּר

מִי שֶׁהֵרִים פַּטִּישׁ בְּהֶתֵּר, כְּדֵי לְפַצֵּחַ אֱגוֹזִים, רַשַּׁאי לְהַחֲזִירוֹ לִמְקוֹמוֹ. וּמִי שֶׁאָכַל פְּרִי וְנִשְׁאֲרוּ בְּיָדוֹ קְלִפּוֹת – מֻתָּר לוֹ לָשֵׂאת אוֹתָן עַד הַפַּח. אֲבָל מִי שֶׁהֵרִים מֻקְצֶה בְּטָעוּת – מַנִּיחַ אוֹתוֹ מִיָּד.
כְּלִי שֶׁנִּשְׁבַּר וְאֵין בְּשִׁבְרָיו שׁוּם שִׁמּוּשׁ – מֻקְצֶה. אֲבָל אִם הַשְּׁבָרִים מְסֻכָּנִים, מֻתָּר לְפַנּוֹת אוֹתָם בַּיָּדַיִם. כַּפְתּוֹר שֶׁנָּפַל מִבֶּגֶד – אֵינוֹ מֻקְצֶה.
💛 אַהֲבַת הַשֵּׁם: הַהֲלָכָה שׁוֹמֶרֶת עַל הַבְּרִיאוּת וְעַל הַכָּבוֹד שֶׁלָּנוּ. ה׳ לֹא רוֹצֶה שֶׁשּׁוּם יֶלֶד יִפָּצַע בְּשַׁבָּת – הַשַּׁבָּת נִתְּנָה לְשִׂמְחָה וּלְחַיִּים.
📖 פֶּרֶק כ״ג, הֲלָכוֹת י״ג, ט״ו

🧠 בְּדֹק אֶת עַצְמְךָ

דָּן הִנִּיחַ מַטְבְּעוֹת עַל שֻׁלְחָן קָטָן לִפְנֵי שַׁבָּת, כְּדֵי שֶׁיִּהְיוּ שָׁם. בְּשַׁבָּת הֵן נָפְלוּ. הַאִם מֻתָּר עַכְשָׁו לְהָזִיז אֶת הַשֻּׁלְחָן?
💡 מָה שֶׁהָיָה מֻקְצֶה בְּבֵין הַשְּׁמָשׁוֹת נִשְׁאָר מֻקְצֶה כָּל הַשַּׁבָּת. דָּן הִנִּיחַ אֶת הַמַּטְבְּעוֹת בְּכַוָּנָה, וְהַשֻּׁלְחָן נַעֲשָׂה בָּסִיס לְדָבָר הָאָסוּר.
הַשָּׁעוֹן הַמְּעוֹרֵר בַּטֶּלֶפוֹן הִתְחִיל לְצַלְצֵל בְּשַׁבָּת וּמַפְרִיעַ לַמִּשְׁפָּחָה לָנוּחַ. מָה מֻתָּר?
💡 טֶלֶפוֹן הוּא כְּלִי שֶׁמְּלַאכְתּוֹ לְאִסּוּר: אָסוּר לִלְחֹץ עָלָיו, אֲבָל מֻתָּר לְהָזִיז אוֹתוֹ לְצֹרֶךְ מְקוֹמוֹ – כְּדֵי שֶׁהַצִּלְצוּל לֹא יַפְרִיעַ.
מִיכָאֵל שָׁפַךְ מִיץ עַל הָרִצְפָּה בְּשַׁבָּת, וְאֵין לְיָדוֹ מַטְלִית. מָה נָכוֹן?
💡 מַשְׁקֶה שֶׁנִּשְׁפַּךְ עַל הָרִצְפָּה – אָסוּר לִסְפֹּג בִּבְגָדִים, וְאָסוּר לְהַרְטִיב כָּרִים, כִּי מְקַלְקְלִים אוֹתָם מִלְּהִיּוֹת רְאוּיִים לְתַפְקִידָם. אֲבָל מֻתָּר לִסְפֹּג בִּסְמַרְטוּט, שֶׁנּוֹעַד לְכָךְ, וּבִלְבַד שֶׁלֹּא סוֹחֲטִים אוֹתוֹ.

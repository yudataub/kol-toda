ל״ג בָּעֹמֶר

📅 רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת · פֶּרֶק ה׳

🔥ל״ג בָּעֹמֶר

בְּל״ג בָּעֹמֶר שְׂמֵחִים: לֹא אוֹמְרִים תַּחֲנוּן, לֹא מִתְעַנִּים, וּלְכָל הַמִּנְהָגִים מֻתָּר לָשִׁיר, לְנַגֵּן וְלִרְקֹד. בּוֹ פָּסְקוּ תַּלְמִידֵי רַבִּי עֲקִיבָא לָמוּת, וְהוּא יוֹם הַהִלּוּלָא שֶׁל רַבִּי שִׁמְעוֹן בַּר יוֹחָאי. רַבִּים עוֹלִים לְמֵירוֹן, לְקִבְרוֹ, וּמַדְלִיקִים מְדוּרוֹת.
רַבִּי שִׁמְעוֹן הִתְחַבֵּא מִן הָרוֹמָאִים עִם בְּנוֹ בִּמְעָרָה שְׁתֵּים עֶשְׂרֵה שָׁנָה, וְנַעֲשָׂה לָהֶם נֵס: צָמַח שָׁם עֵץ חָרוּב וְנָבַע מַעְיָן. הַמְּדוּרָה וְהָעֲלִיָּה לְמֵירוֹן הֵן מִנְהָג שֶׁל רְשׁוּת, וְאֵינָן חוֹבָה.
💛 אַהֲבַת הַשֵּׁם: "כִּי נֵר מִצְוָה וְתוֹרָה אוֹר". כְּמוֹ שֶׁהָאֵשׁ יוֹצֵאת מֵעֵצִים קָרִים וּמְאִירָה, כָּךְ הַתּוֹרָה מְאִירָה וּמְחַמֶּמֶת אֶת הָעוֹלָם.
📖 פֶּרֶק ה׳, הֲלָכוֹת א׳–ג׳, ה׳

🔥🔑 מוּשָּׂג: ל״ג בָּעֹמֶר

״נָהֲגוּ חֲסִידִים לְהַדְלִיק מְדוּרוֹת בְּל״ג בָּעֹמֶר, לִרְמֹז לְאוֹר הַגָּדוֹל שֶׁל סוֹדוֹת הַתּוֹרָה שֶׁנִּתְגַּלּוּ עַל יְדֵי רַבִּי שִׁמְעוֹן בַּר יוֹחָאי״.
📖 ל״ג בָּעֹמֶר – הַיּוֹם הַשְּׁלֹשִׁים וּשְׁלֹשָׁה לַסְּפִירָה, שֶׁבּוֹ שָׂמֵחִים וּמַדְלִיקִים מְדוּרוֹת.
💛 אַהֲבַת הַשֵּׁם: ״כִּי נֵר מִצְוָה וְתוֹרָה אוֹר״ – הָאֵשׁ מַזְכִּירָה לָנוּ שֶׁהַתּוֹרָה מְאִירָה אֶת הָעוֹלָם הֶחָשׁוּךְ. ה׳ נוֹתֵן לָנוּ בְּכָל דּוֹר מוֹרִים שֶׁמַּמְשִׁיכִים לְהַעֲבִיר אוֹר זֶה. 🧒 וְאַתָּה? בְּל״ג בָּעֹמֶר שֵׁב לְיַד מְדוּרָה וְתַחְשֹׁב עַל מוֹרֶה אֶחָד שֶׁלִּמֵּד אוֹתְךָ דָּבָר טוֹב.
📖 פֶּרֶק ה׳, הֲלָכָה א׳, ה׳

🕯️רַבִּי שִׁמְעוֹן בַּר יוֹחָאי

רַבִּי שִׁמְעוֹן דִּבֵּר נֶגֶד מַלְכוּת רוֹמָא הָרְשָׁעָה, וְנִגְזַר עָלָיו לֵהָרֵג. הוּא בָּרַח עִם בְּנוֹ, רַבִּי אֶלְעָזָר, לִמְעָרָה, וְנַעֲשָׂה לָהֶם נֵס: צָמַח שָׁם עֵץ חָרוּב וְנָבַע מַעְיָן. שְׁתֵּים עֶשְׂרֵה שָׁנָה לָמְדוּ שָׁם תּוֹרָה, וְעוֹד שָׁנָה אַחַת – עַד שֶׁלָּמְדוּ לְהַעֲרִיךְ גַּם אֶת הָעוֹלָם הַזֶּה.
בְּיוֹם פְּטִירָתוֹ גִּלָּה רַבִּי שִׁמְעוֹן סוֹדוֹת גְּדוֹלִים בַּתּוֹרָה, וְרָצָה שֶׁיִּשְׂמְחוּ בְּיוֹם זֶה. לָכֵן קוֹרְאִים לוֹ "הִלּוּלָא" – כְּמוֹ שִׂמְחַת חֲתֻנָּה.
💛 אַהֲבַת הַשֵּׁם: רַבִּי שִׁמְעוֹן מָסַר אֶת נַפְשׁוֹ עַל כְּבוֹד יִשְׂרָאֵל וְעַל הַתּוֹרָה. מִמֶּנּוּ אָנוּ לוֹמְדִים מָה זֶה לִדְבֹּק בַּתּוֹרָה בְּכָל הַלֵּב.
📖 פֶּרֶק ה׳, הֲלָכוֹת ב׳–ג׳

🕯️🔑 מוּשָּׂג: הִלּוּלַת רַשְׁבִּ״י

״כְּבָר מֵאוֹת שָׁנִים נוֹהֲגִים בְּמֵירוֹן, לְיַד קִבְרוֹ שֶׁל רַבִּי שִׁמְעוֹן בַּר יוֹחָאי, לְהַדְלִיק מְדוּרָה לִכְבוֹד הִלּוּלַת רַבִּי שִׁמְעוֹן בַּר יוֹחָאי״.
📖 הִלּוּלָא – יוֹם הִסְתַּלְּקוּתוֹ מִן הָעוֹלָם שֶׁל צַדִּיק גָּדוֹל, שֶׁבּוֹ שָׂמֵחִים עַל תּוֹרָתוֹ שֶׁנִּשְׁאֲרָה.
💛 אַהֲבַת הַשֵּׁם: בְּיוֹם הִלּוּלָתוֹ גִּלָּה רַשְׁבִּ״י לְתַלְמִידָיו סוֹדוֹת גְּדוֹלִים בַּתּוֹרָה – שִׂמְחָה עַל אוֹר שֶׁלֹּא נֶעְלָם גַּם בְּיוֹם פְּטִירָה. 🧒 וְאַתָּה? אִם תֵּדַע עַל צַדִּיק שֶׁמִּסְתַּלֵּק בְּיוֹם מְסֻיָּם – לְמַד דָּבָר אֶחָד מִתּוֹרָתוֹ בְּאוֹתוֹ יוֹם.
📖 פֶּרֶק ה׳, הֲלָכָה ב׳

🙏רַבִּי עֲקִיבָא וְקִבְרֵי צַדִּיקִים

רַבִּי עֲקִיבָא הִתְחִיל לִלְמֹד בִּתְנָאִים קָשִׁים, וְנַעֲשָׂה גְּדוֹל הַתּוֹרָה שֶׁבְּעַל פֶּה. כְּשֶׁהָרוֹמָאִים הוֹצִיאוּ אוֹתוֹ לְהֵהָרֵג עַל שֶׁלִּמֵּד תּוֹרָה, קָרָא קְרִיאַת שְׁמַע וְקִבֵּל עָלָיו עֹל מַלְכוּת שָׁמַיִם, וְהֶאֱרִיךְ בַּמִּלָּה "אֶחָד" עַד שֶׁיָּצְאָה נִשְׁמָתוֹ.
לְיַד קִבְרֵי צַדִּיקִים מִתְפַּלְּלִים רַק אֶל ה׳, וּמְבַקְּשִׁים שֶׁיְּקַבֵּל אֶת הַתְּפִלָּה בִּזְכוּת הַצַּדִּיק. וּמִי שֶׁרָגִיל לְהַשְׁלִיךְ בְּגָדִים לַמְּדוּרָה בְּמֵירוֹן – עָדִיף שֶׁיִּתְרֹם אוֹתָם לִצְדָקָה.
💛 אַהֲבַת הַשֵּׁם: רַבִּי עֲקִיבָא אָמַר: "כָּל יָמַי הָיִיתִי מִצְטַעֵר... מָתַי יָבֹא לְיָדִי וַאֲקַיְּמֶנּוּ". זוֹ אַהֲבַת ה׳ בְּכָל הַנֶּפֶשׁ – עַד הָרֶגַע הָאַחֲרוֹן.
📖 פֶּרֶק ה׳, הֲלָכוֹת ד׳, ז׳

🧒תִּסְפֹּרֶת רִאשׁוֹנָה בְּגִיל שָׁלֹשׁ

יֵשׁ נוֹהֲגִים שֶׁלֹּא לְסַפֵּר בֵּן קָטָן עַד גִּיל שָׁלֹשׁ, וְאָז מְסַפְּרִים אוֹתוֹ וּמַשְׁאִירִים לוֹ פֵּאוֹת – וְזוֹ אַחַת הַמִּצְווֹת הָרִאשׁוֹנוֹת שֶׁמְּחַנְּכִים אוֹתוֹ בָּהּ: "לֹא תַקִּפוּ פְּאַת רֹאשְׁכֶם". רַבִּים מְסַפְּרִים אֶת הַקְּטַנִּים בְּל״ג בָּעֹמֶר לְיַד קֶבֶר רַבִּי שִׁמְעוֹן בְּמֵירוֹן, וְעוֹשִׂים שִׂמְחָה.
אֵין חוֹבָה לִנְהֹג כָּךְ. הַמִּנְהָג אֵינוֹ מֻזְכָּר בָּרַמְבַּ״ם וּבַשֻּׁלְחָן עָרוּךְ, וְכָל הָרוֹצֶה רַשַּׁאי לְסַפֵּר אֶת בְּנוֹ גַּם לִפְנֵי גִּיל שָׁלֹשׁ.
💛 אַהֲבַת הַשֵּׁם: הַפֵּאוֹת מְסַמְּנוֹת שֶׁהַיֶּלֶד יְהוּדִי. כְּבָר מִן הַתִּסְפֹּרֶת הָרִאשׁוֹנָה הוּא לוֹמֵד לִשְׂמֹחַ בַּמִּצְווֹת.
📖 פֶּרֶק ה׳, הֲלָכָה ו׳

🧠 בְּדֹק אֶת עַצְמְךָ

לָמָּה נוֹהֲגִים לְסַפֵּר יֶלֶד בִּפְעַם הָרִאשׁוֹנָה בְּגִיל שָׁלֹשׁ וּלְהַשְׁאִיר לוֹ פֵּאוֹת?
💡 בְּגִיל שָׁלֹשׁ מַתְחִילִים לְחַנֵּךְ אֶת הַיֶּלֶד לְמִצְווֹת, וְהַפֵּאוֹת הֵן מִן הַמִּצְווֹת הָרִאשׁוֹנוֹת. זֶה מִנְהָג וְלֹא חוֹבָה, וְהוּא אֵינוֹ מֻזְכָּר בָּרַמְבַּ״ם וּבַשֻּׁלְחָן עָרוּךְ.
אֵיךְ נָכוֹן לְהִתְפַּלֵּל לְיַד קֶבֶר שֶׁל צַדִּיק?
💡 מִתְפַּלְּלִים רַק אֶל ה׳. הַמִּתְפַּלֵּל אֶל הַצַּדִּיק עוֹבֵר בְּאִסּוּר. בְּתוֹךְ הַתְּפִלָּה מֻתָּר לְבַקֵּשׁ מִן ה׳ שֶׁיְּקַבֵּל אוֹתָהּ בִּזְכוּת הַצַּדִּיק.
כְּשֶׁהוֹצִיאוּ אֶת רַבִּי עֲקִיבָא לַהֲרִיגָה, הוּא קָרָא קְרִיאַת שְׁמַע. מָה עָנָה לְתַלְמִידָיו שֶׁשָּׁאֲלוּ ״עַד כָּאן?״
💡 רַבִּי עֲקִיבָא אָמַר: ״כָּל יָמַי הָיִיתִי מִצְטַעֵר עַל פָּסוּק זֶה – בְּכָל נַפְשְׁךָ, אֲפִלּוּ נוֹטֵל אֶת נִשְׁמָתְךָ. אָמַרְתִּי: מָתַי יָבֹא לְיָדִי וַאֲקַיְּמֶנּוּ?״ וְהֶאֱרִיךְ בְּ״אֶחָד״ עַד שֶׁיָּצְאָה נִשְׁמָתוֹ.

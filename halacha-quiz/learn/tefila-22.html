דִּינֵי קְרִיאַת הַתּוֹרָה

🙏 הִלְכוֹת תְּפִלָּה · פֶּרֶק כ״ב

💧שֵׁנִי, חֲמִישִׁי וְשַׁבָּת

בַּמִּדְבָּר הָלְכוּ יִשְׂרָאֵל שְׁלֹשָׁה יָמִים וְלֹא מָצְאוּ מַיִם. חֲכָמִים הֵבִינוּ: הַתּוֹרָה הִיא כְּמוֹ מַיִם – הִיא מְחַיָּה אֶת הַנְּשָׁמָה, וּבְלִי תּוֹרָה הַנְּשָׁמָה צְמֵאָה.
לָכֵן תִּקֵּן מֹשֶׁה רַבֵּנוּ לִקְרֹא בַּתּוֹרָה בְּשַׁבָּת, בְּשֵׁנִי וּבַחֲמִישִׁי – שֶׁלֹּא יַעַבְרוּ שְׁלֹשָׁה יָמִים בְּלִי קְרִיאַת תּוֹרָה.
💛 אַהֲבַת הַשֵּׁם: כְּמוֹ שֶׁה׳ נוֹתֵן מַיִם לְכָל הָעוֹלָם, כָּךְ הוּא נוֹתֵן לָנוּ אֶת הַתּוֹרָה – מַיִם חַיִּים לַנְּשָׁמָה. כָּל קְרִיאָה הִיא פְּגִישָׁה חֲדָשָׁה אִתּוֹ.
📖 פֶּרֶק כ״ב, הֲלָכָה א׳

💋סֵפֶר הַתּוֹרָה עוֹבֵר לְיָדִי

כְּשֶׁמּוֹצִיאִים אֶת סֵפֶר הַתּוֹרָה מִן הָאָרוֹן, הַחַזָּן מוֹלִיךְ אוֹתוֹ אֶל הַבִּימָה בֵּין הַמִּתְפַּלְּלִים.
כָּל מִי שֶׁהַסֵּפֶר עוֹבֵר לְיָדוֹ – מְנַשֵּׁק אוֹתוֹ וּמְלַוֶּה אוֹתוֹ מְעַט.
💛 אַהֲבַת הַשֵּׁם: אֲנַחְנוּ מְנַשְּׁקִים אֶת הַתּוֹרָה כְּמוֹ שֶׁמְּנַשְּׁקִים מַתָּנָה יְקָרָה מֵאַבָּא אוֹהֵב. הַתּוֹרָה הִיא הַמַּתָּנָה שֶׁה׳ נָתַן לָנוּ בְּאַהֲבָה.
📖 פֶּרֶק כ״ב, הֲלָכָה ג׳

🙌הַגְבָּהָה וּנְשִׁיקַת הַסֵּפֶר

מַגְבִּיהִים אֶת סֵפֶר הַתּוֹרָה כְּדֵי לְהַרְאוֹת אֶת הָאוֹתִיּוֹת לְכָל הַקָּהָל. הַסְּפָרַדִּים מַגְבִּיהִים לִפְנֵי הַקְּרִיאָה, וְרֹב הָאַשְׁכְּנַזִּים – אַחֲרֶיהָ. הַמַּגְבִּיהַּ מַקְפִּיד שֶׁכֻּלָּם יִרְאוּ, וְהַטּוֹב בְּיוֹתֵר – לְהִסְתּוֹבֵב לְאַט סִבּוּב שָׁלֵם.
חוֹלֶה אוֹ מְצֻנָּן לֹא יְנַשֵּׁק אֶת הַסֵּפֶר בְּפִיו, כְּדֵי שֶׁלֹּא יַדְבִּיק אֲחֵרִים. אֶפְשָׁר לִגַּע בַּסֵּפֶר בַּיָּד וּלְנַשֵּׁק אֶת הַיָּד.
💛 אַהֲבַת הַשֵּׁם: כְּבוֹד הַתּוֹרָה וּשְׁמִירָה עַל הַחֲבֵרִים הוֹלְכִים יַחַד. מִי שֶׁאוֹהֵב אֶת ה׳ – אוֹהֵב גַּם אֶת הַבְּרִיּוֹת שֶׁלּוֹ.
📖 פֶּרֶק כ״ב, הֲלָכוֹת ג׳–ד׳

👀״וְזֹאת הַתּוֹרָה״

כְּשֶׁמַּגְבִּיהִים אֶת סֵפֶר הַתּוֹרָה, מִצְוָה עַל כָּל הָאֲנָשִׁים וְהַנָּשִׁים שֶׁרוֹאִים אֶת הַכְּתָב לִכְרֹעַ וְלוֹמַר: ״וְזֹאת הַתּוֹרָה אֲשֶׁר שָׂם מֹשֶׁה לִפְנֵי בְּנֵי יִשְׂרָאֵל״.
רַבִּים מֵהָאַשְׁכְּנַזִּים לֹא נָהֲגוּ לִכְרֹעַ, וְיֵשׁ שֶׁלִּמְּדוּ זְכוּת עַל מִנְהָגָם. אֲבָל לְכַתְּחִלָּה רָאוּי לִכְרֹעַ כְּשֶׁרוֹאִים אֶת הַכְּתָב.
💛 אַהֲבַת הַשֵּׁם: מֵהַסַּפְסָל הָאַחֲרוֹן וּמֵעֶזְרַת הַנָּשִׁים – כָּל אֶחָד וְכָל אַחַת מַבִּיטִים בְּאוֹתִיּוֹת הַתּוֹרָה וְאוֹמְרִים: זוֹ הַתּוֹרָה שֶׁה׳ נָתַן לָנוּ.
📖 פֶּרֶק כ״ב, הֲלָכָה ד׳

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

הַגְבָּהַת סֵפֶר הַתּוֹרָה: הַמִּנְהָג הַמְּקוֹרִי הָיָה לְהַגְבִּיהַּ לִפְנֵי הַקְּרִיאָה, וְכָךְ נוֹהֲגִים הַסְּפָרַדִּים וּמִקְצָת הָאַשְׁכְּנַזִּים. רֹב הָאַשְׁכְּנַזִּים מַגְבִּיהִים אַחֲרֵי הַקְּרִיאָה – כְּדֵי שֶׁכֻּלָּם יֵדְעוּ שֶׁהָעִקָּר הוּא לִשְׁמֹעַ אֶת הַקְּרִיאָה, וּמִתּוֹךְ כָּךְ מַרְאִים אֶת הַתּוֹרָה לָעָם.
עֲמִידָה בְּקַדִּישׁ: רֹב הָאַשְׁכְּנַזִּים עוֹמְדִים כְּשֶׁעוֹנִים לְקַדִּישׁ וּלְ״בָרְכוּ״. לְמִנְהַג רֹב הַסְּפָרַדִּים אֵין חוֹבָה לַעֲמֹד: מִי שֶׁיָּשַׁב – יָכוֹל לְהַמְשִׁיךְ לָשֶׁבֶת, וּמִי שֶׁעָמַד – נִשְׁאָר לַעֲמֹד.
💛 אַהֲבַת הַשֵּׁם: שְׁנֵי הַמִּנְהָגִים כַּהֲלָכָה. כָּל אֶחָד נוֹהֵג כְּמִנְהַג אֲבוֹתָיו – וְכֻלָּנוּ יַחַד עוֹנִים בְּכָל הַכֹּחַ: ״אָמֵן, יְהֵא שְׁמֵהּ רַבָּא מְבָרַךְ״.
📖 פֶּרֶק כ״ב, הֲלָכָה ד׳; פֶּרֶק ט״ז, הֲלָכָה ג׳

📜בַּעַל הַקּוֹרֵא וְהָעוֹלֶה

פַּעַם כָּל עוֹלֶה קָרָא בְּעַצְמוֹ. הַיּוֹם בַּעַל הַקּוֹרֵא קוֹרֵא בִּשְׁבִיל כֻּלָּם – כְּדֵי שֶׁלֹּא יִתְבַּיֵּשׁ מִי שֶׁאֵינוֹ יוֹדֵעַ לִקְרֹא. כָּל עוֹלֶה מְבָרֵךְ לִפְנֵי הַקְּרִיאָה וְאַחֲרֶיהָ.
הָעוֹלֶה קוֹרֵא בְּלַחַשׁ עִם בַּעַל הַקּוֹרֵא, כִּי הוּא זֶה שֶׁבֵּרֵךְ – וְאִם לֹא יִקְרָא, בִּרְכוֹתָיו עֲלוּלוֹת לִהְיוֹת לְבַטָּלָה.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים דָּאֲגוּ שֶׁאַף יְהוּדִי לֹא יִתְבַּיֵּשׁ לַעֲלוֹת לַתּוֹרָה. הַתּוֹרָה שַׁיֶּכֶת לְכֻלָּנוּ, וְה׳ אוֹהֵב כָּל אֶחָד וְאֶחָד מִמֶּנּוּ.
📖 פֶּרֶק כ״ב, הֲלָכוֹת ה׳–ו׳

🤫שֶׁקֶט – קוֹרְאִים בַּתּוֹרָה!

בַּעַל הַקּוֹרֵא קוֹרֵא בְּקוֹל מִן הַסֵּפֶר, וְהָעוֹלֶה לַתּוֹרָה קוֹרֵא אִתּוֹ בְּלַחַשׁ, מִלָּה בְּמִלָּה.
מֵהָרֶגַע שֶׁפּוֹתְחִים אֶת הַסֵּפֶר – אָסוּר לְדַבֵּר, אֲפִלּוּ בְּדִבְרֵי הֲלָכָה.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁה׳ מְדַבֵּר אֵלֵינוּ בַּתּוֹרָה, אֲנַחְנוּ מַקְשִׁיבִים לוֹ בְּשֶׁקֶט – כְּמוֹ בֵּן שֶׁמַּקְשִׁיב לְאָבִיו הָאוֹהֵב.
📖 פֶּרֶק כ״ב, הֲלָכוֹת ו׳, ח׳

⏳מְמַהֵר? מָה הֲכִי חָשׁוּב

אָסוּר לָצֵאת מִבֵּית הַכְּנֶסֶת בְּאֶמְצַע קְרִיאַת הַתּוֹרָה. מִי שֶׁמֻּכְרָח – יֵצֵא בֵּין עוֹלֶה לְעוֹלֶה, כְּשֶׁהַסֵּפֶר סָגוּר. וּבְסוֹף הַתְּפִלָּה: אַשְׁרֵי וּקְדֻשָּׁה דְּסִדְרָא (בְּ"וּבָא לְצִיּוֹן") חֲשׁוּבִים יוֹתֵר מִתַּחֲנוּן, שֶׁהוּא רְשׁוּת.
מִי שֶׁמֻּכְרָח לָצֵאת לִפְנֵי סוֹף הַתְּפִלָּה – יִשָּׁאֵר לְפָחוֹת עַד אַחֲרֵי קְדֻשָּׁה דְּסִדְרָא, וְלֹא יָסִיר תְּפִלִּין לִפְנֵי כֵן.
💛 אַהֲבַת הַשֵּׁם: עַל קְדֻשָּׁה דְּסִדְרָא אָמְרוּ חֲכָמִים שֶׁבִּזְכוּתָהּ הָעוֹלָם עוֹמֵד. גַּם כְּשֶׁאֲנַחְנוּ מְמַהֲרִים, אֲנַחְנוּ שׁוֹמְרִים לַה׳ אֶת הַחֵלֶק הֶחָשׁוּב בְּיוֹתֵר.
📖 פֶּרֶק כ״ב, הֲלָכָה ח׳; פֶּרֶק כ״ג, הֲלָכָה ג׳

🧠 בְּדֹק אֶת עַצְמְךָ

נָתָן מְצֻנָּן וּמִשְׁתַּעֵל, וְסֵפֶר הַתּוֹרָה עוֹבֵר לְיָדוֹ. מָה כְּדַאי לוֹ לַעֲשׂוֹת?
💡 חוֹלֶה לֹא יְנַשֵּׁק בְּפִיו, כְּדֵי שֶׁלֹּא יַדְבִּיק אֶת הַמִּתְפַּלְּלִים. נְגִיעָה בַּיָּד וּנְשִׁיקַת הַיָּד הִיא גַּם דֶּרֶךְ שֶׁל כָּבוֹד – וְיֵשׁ שֶׁנּוֹהֲגִים כָּךְ תָּמִיד.
בַּעַל הַקּוֹרֵא קוֹרֵא בְּקוֹל בִּשְׁבִיל הָעוֹלֶה. לָמָּה הָעוֹלֶה צָרִיךְ בְּכָל זֹאת לִקְרֹא אִתּוֹ בְּלַחַשׁ?
💡 הָעוֹלֶה הוּא שֶׁמְּבָרֵךְ עַל הַקְּרִיאָה, וְלָכֵן עָלָיו לִקְרֹא גַּם בְּעַצְמוֹ, בְּלַחַשׁ, מִלָּה בְּמִלָּה. בַּעַל הַקּוֹרֵא קוֹרֵא בְּקוֹל כְּדֵי שֶׁלֹּא יִתְבַּיֵּשׁ מִי שֶׁאֵינוֹ יוֹדֵעַ לִקְרֹא.
הַמַּגְבִּיהַּ הֵרִים אֶת סֵפֶר הַתּוֹרָה הַפָּתוּחַ. מָה כְּדַאי לוֹ לַעֲשׂוֹת עַכְשָׁו?
💡 מַגְבִּיהִים כְּדֵי לְהַרְאוֹת אֶת אוֹתִיּוֹת הַתּוֹרָה לְכָל הַקָּהָל. מִי שֶׁפּוֹנֶה רַק מְעַט לַצְּדָדִים – רַבִּים לֹא רוֹאִים. סִבּוּב שָׁלֵם וְאִטִּי מַרְאֶה לְכֻלָּם.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

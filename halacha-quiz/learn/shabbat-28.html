חוֹלֶה שֶׁאֵינוֹ מְסֻכָּן

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק כ״ח

🤒שָׁלֹשׁ דַּרְגּוֹת שֶׁל חֹלִי

חוֹלֶה מְסֻכָּן – עוֹשִׂים בִּשְׁבִילוֹ הַכֹּל, כְּמוֹ בְּיוֹם חֹל. חוֹלֶה רָגִיל, שֶׁשּׁוֹכֵב בַּמִּטָּה וְאֵין סַכָּנָה לְחַיָּיו – מֻתָּר לְבַקֵּשׁ מִגּוֹי לַעֲשׂוֹת בִּשְׁבִילוֹ מְלָאכָה, וּמֻתָּר לוֹ לִקַּח תְּרוּפוֹת. וּמִי שֶׁמִּתְהַלֵּךְ כְּבָרִיא וְיֵשׁ לוֹ רַק מֵחוּשׁ קַל – נִזְהָר בְּכָל אִסּוּרֵי הַשַּׁבָּת.
יֶלֶד קָטָן שֶׁצָּרִיךְ מְאוֹד דָּבָר מְסֻיָּם – דִּינוֹ כְּחוֹלֶה, גַּם אִם אֵינוֹ שׁוֹכֵב בַּמִּטָּה.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת אֲנַחְנוּ מְקַבְּלִים אֶת הַמְּצִיאוּת בְּשַׁלְוָה – גַּם מֵחוּשׁ קָטָן. אֲבָל כְּשֶׁיֵּשׁ מַחֲלָה אֲמִתִּית, חֲכָמִים הֵקֵלּוּ, כִּי מִצְוָה לִשְׁמֹר עַל בְּרִיאוּת הַגּוּף.
📖 פֶּרֶק כ״ח, הֲלָכוֹת א׳–ב׳

🔀תְּרוּפוֹת בְּשַׁבָּת – שְׁתֵּי שִׁיטוֹת

חֲכָמִים גָּזְרוּ שֶׁמִּי שֶׁיֵּשׁ לוֹ רַק מֵחוּשׁ לֹא יִקַּח תְּרוּפוֹת, שֶׁמָּא יָבוֹא לִטְחֹן צְמָחִים לִרְפוּאָה. יֵשׁ אוֹמְרִים שֶׁהַיּוֹם מֻתָּר לִקַּח תְּרוּפוֹת מִבֵּית חֲרֹשֶׁת, כִּי אִישׁ לֹא יָכִין אוֹתָן בַּבַּיִת. וְדַעַת רֹב הַפּוֹסְקִים שֶׁגַּם הַיּוֹם אָסוּר, כִּי גְּזֵרַת חֲכָמִים לֹא בְּטֵלָה מֵעַצְמָהּ, וְיֵשׁ עוֹד מִי שֶׁמֵּכִין תְּרוּפוֹת בֵּיתִיּוֹת.
הָרַב מֶלָמֵד: כְּשֶׁהַמֵּחוּשׁ רַק מַטְרִיד – נָכוֹן לְהַחְמִיר. כְּשֶׁהוּא גּוֹרֵם צַעַר מַמָּשׁ – אֶפְשָׁר לִקַּח תְּרוּפָה מִבֵּית חֲרֹשֶׁת. וּלְדַעַת רַבִּים גַּם בִּמְקוֹם צַעַר אָסוּר. חוֹלֶה שֶׁשּׁוֹכֵב בַּמִּטָּה, וְיֶלֶד קָטָן שֶׁצָּרִיךְ – מֻתָּר לָהֶם לִקַּח תְּרוּפוֹת.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה דּוֹאֶגֶת גַּם לַגּוּף וְגַם לְעֹנֶג הַשַּׁבָּת. כְּשֶׁהַפּוֹסְקִים שׁוֹקְלִים מָתַי לְהָקֵל, הֵם מְבַקְּשִׁים גַּם לִשְׁמֹר עַל גְּזֵרוֹת חֲכָמִים וְגַם לְרַחֵם עַל הַמִּצְטַעֵר.
📖 פֶּרֶק כ״ח, הֲלָכוֹת א׳–ב׳, ה׳

🍬מֵחוּשׁ קַל

חֲכָמִים גָּזְרוּ שֶׁמִּי שֶׁיֵּשׁ לוֹ רַק מֵחוּשׁ קַל לֹא יִתְעַסֵּק בִּרְפוּאוֹת, שֶׁמָּא יָבוֹא לִטְחֹן צְמָחִים לִתְרוּפָה. אֲבָל מַאֲכָלִים וּמַשְׁקִים שֶׁגַּם בְּרִיאִים אוֹכְלִים – מֻתָּרִים. לָכֵן סֻכָּרִיַּת רְפוּאָה לַגָּרוֹן – אֲסוּרָה, וְסֻכָּרִיָּה רְגִילָה – מֻתֶּרֶת. וּתְרוּפָה מִבֵּית חֲרֹשֶׁת: כְּשֶׁהַמֵּחוּשׁ רַק מַטְרִיד – נָכוֹן שֶׁלֹּא לִקַּח. כְּשֶׁהוּא גּוֹרֵם צַעַר מַמָּשׁ – הָרַב מֶלָמֵד כּוֹתֵב שֶׁאֶפְשָׁר לִקַּח, וּלְדַעַת רַבִּים גַּם אָז אָסוּר.
פְּעֻלּוֹת שֶׁאֵינָן תְּרוּפָה – מֻתָּרוֹת: בַּקְבּוּק חַם עַל בֶּטֶן כּוֹאֶבֶת, וְצֶמֶר גֶּפֶן בָּאֹזֶן. וְיֶלֶד שֶׁעוֹד לֹא בַּר מִצְוָה – מֻתָּר לָתֵת לוֹ תְּרוּפוֹת גַּם בְּמֵחוּשׁ.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת אֲנַחְנוּ לוֹמְדִים לִסְבֹּל מְעַט בְּשַׁלְוָה, וְלִבְטֹחַ בַּה׳ שֶׁהוּא רוֹפֵא כָּל בָּשָׂר.
📖 פֶּרֶק כ״ח, הֲלָכוֹת ג׳–ה׳, י״ב; פֶּרֶק כ״ד, הֲלָכָה ו׳

🩹פְּלַסְטֵר, קֶרַח וּמַדְחֹם

מֻתָּר לְהַדְבִּיק פְּלַסְטֵר כְּדֵי לְהָגֵן עַל פֶּצַע, וּמֻתָּר לְנַקּוֹת דָּם בִּנְיָר. אֲבָל אָסוּר לַחְתֹּךְ פְּלַסְטֵר אוֹ תַּחְבֹּשֶׁת – לָכֵן טוֹב לְהָכִין לִפְנֵי שַׁבָּת פְּלַסְטֵרִים בְּגֹדֶל מַתְאִים. עַל מַכָּה מֻתָּר לָשִׂים קֶרַח כְּדֵי שֶׁלֹּא תִּתְנַפַּח, וּכְשֶׁיֵּשׁ צֹרֶךְ – מֻתָּר לִמְדֹּד חֹם בְּמַדְחֹם רָגִיל שֶׁאֵינוֹ חַשְׁמַלִּי.
מִשְׁחָה לֹא מוֹרְחִים וְלֹא מַחֲלִיקִים. כְּשֶׁמֻּתָּר לְהִשְׁתַּמֵּשׁ בָּהּ – שָׂמִים אוֹתָהּ יָשָׁר מִן הַשְּׁפוֹפֶרֶת עַל הַפֶּצַע, בְּלִי לִמְרֹחַ.
💛 אַהֲבַת הַשֵּׁם: הַהֲלָכָה דּוֹאֶגֶת לָנוּ גַּם בִּזְמַן שֶׁכּוֹאֵב. כְּשֶׁמַּכִּירִים אֶת הַדְּרָכִים הַמֻּתָּרוֹת, אֶפְשָׁר לְטַפֵּל בְּאָחִיךָ הַקָּטָן בְּאַהֲבָה – וְגַם לִשְׁמֹר שַׁבָּת.
📖 פֶּרֶק כ״ח, הֲלָכוֹת ח׳–ט׳, י״א–י״ב

🧠 בְּדֹק אֶת עַצְמְךָ

לְאַבָּא כּוֹאֵב קְצָת הַגָּרוֹן, וְהוּא מִתְהַלֵּךְ כְּבָרִיא. מָה מֻתָּר לוֹ?
💡 מִי שֶׁיֵּשׁ לוֹ מֵחוּשׁ קַל לֹא מִשְׁתַּמֵּשׁ בִּרְפוּאוֹת. אֲבָל מָה שֶׁגַּם בְּרִיאִים אוֹכְלִים – מֻתָּר, גַּם אִם הוּא מֵקֵל עַל הַכְּאֵב.
לְיוֹסִי הַקָּטָן יֵשׁ פֶּצַע כּוֹאֵב, וְצָרִיךְ לָשִׂים עָלָיו מִשְׁחָה. אֵיךְ עוֹשִׂים זֹאת בְּשַׁבָּת?
💡 מְרִיחָה שֶׁמַּחֲלִיקָה אֶת הַמִּשְׁחָה הִיא מְלֶאכֶת מְמָרֵחַ, וְחִתּוּךְ תַּחְבֹּשֶׁת – מְחַתֵּךְ. לְקָטָן וּבִמְקוֹם צַעַר מֻתָּר לָשִׂים מִשְׁחָה בְּלִי לִמְרֹחַ.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

זִמּוּן

🍎 הִלְכוֹת בְּרָכוֹת · פֶּרֶק ה׳

👨‍👦‍👦זִמּוּן – מְבָרְכִים יַחַד

שְׁלוֹשָׁה שֶׁאָכְלוּ לֶחֶם יַחַד – חַיָּבִים לְזַמֵּן. הַמְּזַמֵּן אוֹמֵר: "נְבָרֵךְ שֶׁאָכַלְנוּ מִשֶּׁלּוֹ", וְכֻלָּם עוֹנִים: "בָּרוּךְ שֶׁאָכַלְנוּ מִשֶּׁלּוֹ וּבְטוּבוֹ חָיִינוּ". בַּעֲשָׂרָה מוֹסִיפִים "אֱלֹהֵינוּ". בְּמִסְעָדָה, כְּשֶׁכָּל אֶחָד יוֹשֵׁב בְּשֻׁלְחָן אַחֵר – לֹא מְזַמְּנִים.
נוֹהֲגִים לְכַבֵּד אֶת הָאוֹרֵחַ לְזַמֵּן, כְּדֵי שֶׁיְּבָרֵךְ בְּקוֹל אֶת בַּעַל הַבַּיִת. וְאִם יֵשׁ תַּלְמִיד חָכָם אוֹ כֹּהֵן – מְכַבְּדִים אוֹתָם.
💛 אַהֲבַת הַשֵּׁם: "גַּדְּלוּ לַה׳ אִתִּי וּנְרוֹמְמָה שְׁמוֹ יַחְדָּו" – כְּשֶׁמְּבָרְכִים יַחַד, כְּבוֹד ה׳ גָּדֵל. חֲבֵרִים שֶׁאוֹכְלִים יַחַד הוֹפְכִים לַחֲבוּרָה שֶׁמְּהַלֶּלֶת אֶת ה׳.
📖 פֶּרֶק ה׳, הֲלָכוֹת א׳–ב׳, ד׳–ה׳

👥🔑 מוּשָּׂג: זִמּוּן

״שְׁלֹשָׁה שֶׁאָכְלוּ לֶחֶם יַחַד, חַיָּבִים לְזַמֵּן״.
📖 זִמּוּן – בְּרָכַת הַקְדָּמָה שֶׁמּוֹסִיפִים לִפְנֵי בִּרְכַּת הַמָּזוֹן, כְּשֶׁשְּׁלוֹשָׁה (אוֹ יוֹתֵר) אָכְלוּ לֶחֶם יַחַד.
💛 אַהֲבַת הַשֵּׁם: עַל יְדֵי שֶׁמְּזַמְּנִים יַחַד, נַעֲשֶׂה קִדּוּשׁ הַשֵּׁם גָּדוֹל וְכָבוֹד שָׁמַיִם מִתְרַבֶּה, יוֹתֵר מִכְּשֶׁכֹּל אֶחָד מְבָרֵךְ לְחוֹד. 🧒 וְאַתָּה? כְּשֶׁאַתֶּם שְׁלוֹשָׁה בִּסְעוּדָה – תַּזְכִּיר לַחֲבֵרֶיךָ לְזַמֵּן יַחַד.
📖 פֶּרֶק ה׳, הֲלָכָה א׳

🤝לֹא מְפָרְקִים אֶת הַחֲבוּרָה

שְׁלוֹשָׁה, אַרְבָּעָה אוֹ חֲמִשָּׁה שֶׁאָכְלוּ יַחַד – אָסוּר לָהֶם לְהִתְחַלֵּק, כְּדֵי לֹא לְהַפְסִיד אֶת הַזִּמּוּן. הוֹלְכִים אַחַר הָרֹב: אִם רֹב הַחֲבוּרָה גָּמְרוּ וְרוֹצִים לְזַמֵּן – מִי שֶׁעוֹד אוֹכֵל עוֹצֵר, עוֹנֶה לַזִּמּוּן, וְאַחַר כָּךְ מַמְשִׁיךְ לֶאֱכֹל.
מִי שֶׁצָּרִיךְ לָצֵאת לְצֹרֶךְ גָּדוֹל, כְּמוֹ בַּחֲתֻנָּה אֲרֻכָּה, מֻתָּר לוֹ לְבָרֵךְ בְּלִי זִמּוּן.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנַחְנוּ מְוַתְּרִים קְצָת עַל הַנּוֹחוּת שֶׁלָּנוּ כְּדֵי לְבָרֵךְ יַחַד, כְּבוֹד ה׳ מִתְרַבֶּה. אַהֲבַת ה׳ וְאַהֲבַת הַחֲבֵרִים נִפְגָּשׁוֹת סְבִיב הַשֻּׁלְחָן.
📖 פֶּרֶק ה׳, הֲלָכוֹת ג׳, י״א

🍎מִי מִצְטָרֵף – וּמִי עוֹנֶה?

שִׁבְעָה שֶׁאָכְלוּ לֶחֶם וּשְׁלוֹשָׁה שֶׁאָכְלוּ פֵּרוֹת אוֹ שָׁתוּ מַשְׁקֶה (לֹא מַיִם) – מִצְטָרְפִים לַעֲשָׂרָה וּמַזְכִּירִים אֶת הַשֵּׁם. הַמְּזַמֵּן תָּמִיד מֵאֵלֶּה שֶׁאָכְלוּ לֶחֶם. וּכְשֶׁיֵּשׁ זִמּוּן שֶׁל שְׁלוֹשָׁה גְּבָרִים – גַּם הַנָּשִׁים וְהַבָּנוֹת שֶׁבַּסְּעוּדָה עוֹנוֹת.
מִי שֶׁלֹּא אָכַל וְשׁוֹמֵעַ זִמּוּן עוֹנֶה: "בָּרוּךְ וּמְבֹרָךְ שְׁמוֹ תָּמִיד לְעוֹלָם וָעֶד".
💛 אַהֲבַת הַשֵּׁם: אַף אֶחָד לֹא נִשְׁאָר בַּצַּד כְּשֶׁמְּהַלְּלִים אֶת ה׳. גַּם מִי שֶׁרַק נִכְנַס לַחֶדֶר מִצְטָרֵף לַשֶּׁבַח – כִּי כֻּלָּנוּ בָּנִים שֶׁל אוֹתוֹ אָב.
📖 פֶּרֶק ה׳, הֲלָכוֹת ה׳, ז׳, י׳

🍷כּוֹס שֶׁל בְּרָכָה

כְּשֶׁמְּזַמְּנִים – מִצְוָה (אֲבָל לֹא חוֹבָה) לְבָרֵךְ עַל כּוֹס יַיִן, "כּוֹס שֶׁל בְּרָכָה". הַכּוֹס שְׁלֵמָה, נְקִיָּה וּמְלֵאָה. בַּסּוֹף הַמְּזַמֵּן מְבָרֵךְ "בּוֹרֵא פְּרִי הַגֶּפֶן" וְשׁוֹתֶה.
אוֹחֲזִים אֶת הַכּוֹס בְּיַד יָמִין – הַיָּד הַחֲשׁוּבָה, מַגְבִּיהִים אוֹתָהּ טֶפַח מֵעַל הַשֻּׁלְחָן וּמִסְתַּכְּלִים בָּהּ, כְּדֵי לֹא לְהַסִּיחַ אֶת הַדַּעַת.
💛 אַהֲבַת הַשֵּׁם: יֵשׁ מִי שֶׁחוֹשֵׁב שֶׁבִּרְכַּת הַמָּזוֹן הִיא כְּמוֹ "מַס" שֶׁמְּשַׁלְּמִים אַחֲרֵי הָאֹכֶל. אֲבָל הִיא הַשִּׂיא הַשָּׂמֵחַ שֶׁל הַסְּעוּדָה – וְהַכּוֹס מַרְאָה שֶׁאֲנַחְנוּ מוֹדִים לַה׳ בְּשִׂמְחָה.
📖 פֶּרֶק ה׳, הֲלָכוֹת י״ג–ט״ו

🍇כּוֹס מִשֶּׁלּוֹ לְכָל אֶחָד

בְּסוֹף הַזִּמּוּן הַמְזַמֵּן מְבָרֵךְ ״בּוֹרֵא פְּרִי הַגֶּפֶן״ וְשׁוֹתֶה, וּמִצְוָה מִן הַמֻּבְחָר שֶׁגַּם כֻּלָּם יִטְעֲמוּ. אֲבָל אֵין לָתֵת לְחָבֵר לִשְׁתּוֹת מִכּוֹס שֶׁשָּׁתִינוּ מִמֶּנָּה: אוּלַי הוּא נִגְעָל וּמִתְבַּיֵּשׁ לוֹמַר, וְאוּלַי מַחֲלָה תַּעֲבֹר אֵלָיו.
הַדֶּרֶךְ הַמְהֻדֶּרֶת: עוֹד לִפְנֵי בִּרְכַּת הַמָּזוֹן מוֹזְגִים לְכָל מִי שֶׁרוֹצֶה לִטְעֹם כּוֹס מִשֶּׁלּוֹ, וְאַחֲרֵי הַבְּרָכָה כֻּלָּם שׁוֹתִים. וְכָךְ גַּם בַּקִּדּוּשׁ.
💛 אַהֲבַת הַשֵּׁם: גַּם בְּרֶגַע שֶׁל מִצְוָה חוֹשְׁבִים עַל הֶחָבֵר – שֶׁלֹּא יִתְבַּיֵּשׁ וְשֶׁלֹּא יִגָּעֵל. כָּךְ הַשִּׂמְחָה שֶׁל הַבְּרָכָה מַגִּיעָה לְכֻלָּם.
📖 פֶּרֶק ה׳, הֲלָכָה ט״ו

🧠 בְּדֹק אֶת עַצְמְךָ

שְׁלוֹשָׁה חֲבֵרִים אָכְלוּ לֶחֶם יַחַד. שְׁנַיִם סִיְּמוּ וְרוֹצִים לְזַמֵּן, וְהַשְּׁלִישִׁי עוֹד אוֹכֵל. מָה הַדִּין?
💡 הוֹלְכִים אַחַר הָרֹב. הַשְּׁלִישִׁי מַפְסִיק, עוֹנֶה לַזִּמּוּן, אַחַר כָּךְ מַמְשִׁיךְ לֶאֱכֹל, וּבַסּוֹף מְבָרֵךְ לְעַצְמוֹ. שְׁנַיִם לְבַד אֵינָם יְכוֹלִים לְזַמֵּן.
בִּסְעוּדָה גְּדוֹלָה שִׁבְעָה אָכְלוּ לֶחֶם, וּשְׁלוֹשָׁה אָכְלוּ רַק פֵּרוֹת וְשָׁתוּ מִיץ. אֵיךְ מְזַמְּנִים?
💡 כְּשֶׁרֹב נִכָּר – שִׁבְעָה – אָכְלוּ לֶחֶם, גַּם מִי שֶׁאָכַל כְּזַיִת פֵּרוֹת אוֹ שָׁתָה מַשְׁקֶה מִצְטָרֵף לָעֲשָׂרָה, וּמַזְכִּירִים אֶת הַשֵּׁם. אֲבָל הַמְּזַמֵּן צָרִיךְ לִהְיוֹת מִמִּי שֶׁאָכַל לֶחֶם.
בִּסְעוּדַת שַׁבָּת יוֹשְׁבִים אַבָּא וּשְׁנֵי בָּנִים גְּדוֹלִים אַחֲרֵי בַּר מִצְוָה, וְגַם אִמָּא וְהָאֲחָיוֹת. אַבָּא מְזַמֵּן. מִי עוֹנֶה?
💡 כְּשֶׁשְּׁלוֹשָׁה גְּבָרִים יוֹצְרִים חוֹבַת זִמּוּן, גַּם הַנָּשִׁים וְהַבָּנוֹת שֶׁאָכְלוּ חַיָּבוֹת לַעֲנוֹת. צָרִיךְ לְהִתְחַזֵּק בָּזֶה, כִּי הַרְבֵּה פְּעָמִים שׁוֹכְחִים.

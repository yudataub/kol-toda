הִלְכוֹת גְּנֵבָה

🤝 בֵּין אָדָם לַחֲבֵרוֹ · פֶּרֶק ד׳

🕵️גַּנָּב וְגַזְלָן

גַּנָּב הוּא מִי שֶׁלּוֹקֵחַ בַּסֵּתֶר, בְּלִי שֶׁהַבְּעָלִים יוֹדְעִים. גַּזְלָן הוּא מִי שֶׁחוֹטֵף בְּכֹחַ מִיַּד הַבְּעָלִים. שְׁנֵיהֶם חַיָּבִים לְהַחְזִיר אֶת מָה שֶׁלָּקְחוּ, אוֹ לְשַׁלֵּם אֶת שָׁוְיוֹ.
בִּזְמַן שֶׁהָיָה בֵּית דִּין שֶׁל דַּיָּנִים סְמוּכִים, גַּנָּב שֶׁהֵעִידוּ עָלָיו עֵדִים שִׁלֵּם כִּפְלַיִם. הַיּוֹם גַּם הַגַּנָּב מְשַׁלֵּם רַק אֶת שְׁוִי הַגְּנֵבָה.
💛 אַהֲבַת הַשֵּׁם: כָּל מָה שֶׁיֵּשׁ לְאָדָם – ה׳ נָתַן לוֹ. כְּשֶׁאֲנַחְנוּ שׁוֹמְרִים עַל מָה שֶׁשַּׁיָּךְ לַאֲחֵרִים, אֲנַחְנוּ מְכַבְּדִים אֶת מָה שֶׁה׳ נָתַן לְכָל אֶחָד.
📖 פֶּרֶק ד׳, הֲלָכָה א׳

🥄גַּם דָּבָר קָטָן

אָסוּר לִגְנֹב אֲפִלּוּ דָּבָר שֶׁשָּׁוֶה פָּחוֹת מִפְּרוּטָה. אֵין לְהִכָּנֵס לְבֵית הַשָּׁכֵן וְלָקַחַת מִשָּׁם אֲפִלּוּ כַּפִּית קְטַנָּה בְּלִי רְשׁוּת – גַּם לֹא עַל מְנַת לְהַחֲזִיר.
בִּדְבָרִים זוֹלִים מְאוֹד שֶׁאֲנָשִׁים לֹא מַקְפִּידִים עֲלֵיהֶם – כְּמוֹ גַּפְרוּר אֶחָד אוֹ עֵט פָּשׁוּט – מֻתָּר לְהִשְׁתַּמֵּשׁ בְּלִי רְשׁוּת, אֲבָל רַק כְּשֶׁהַבְּעָלִים לֹא נִמְצָא, וְהוּא אֵינוֹ קַפְּדָן. וּמִדַּת חֲסִידוּת לְהַחְמִיר גַּם בָּזֶה.
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁנִּזְהָר אֲפִלּוּ בְּכַפִּית שֶׁל חֲבֵרוֹ, לִבּוֹ נָקִי וְיָשָׁר לִפְנֵי ה׳. הַזְּהִירוּת בַּדְּבָרִים הַקְּטַנִּים בּוֹנָה אָדָם גָּדוֹל.
📖 פֶּרֶק ד׳, הֲלָכוֹת א׳, ג׳

🔁שׁוֹאֵל שֶׁלֹּא מִדַּעַת

גַּם מִי שֶׁלּוֹקֵחַ חֵפֶץ שֶׁל חָבֵר כְּדֵי לְהִשְׁתַּמֵּשׁ בּוֹ וּלְהַחֲזִיר, בְּלִי לְבַקֵּשׁ רְשׁוּת – נֶחְשָׁב גַּנָּב, אֲפִלּוּ לִזְמַן קָצָר. זֶה נִקְרָא "שׁוֹאֵל שֶׁלֹּא מִדַּעַת". כְּשֶׁהַבְּעָלִים נִמְצָא – פָּשׁוּט מְבַקְּשִׁים מִמֶּנּוּ.
גַּם אֵצֶל חָבֵר טוֹב: אִם יֵשׁ אֲפִלּוּ סָפֵק קָטָן שֶׁהוּא יַקְפִּיד – אָסוּר לָקַחַת.
💛 אַהֲבַת הַשֵּׁם: לְבַקֵּשׁ רְשׁוּת זֶה לֹא רַק נִימוּס – זֶה כָּבוֹד לֶחָבֵר וּלְמִצְוַת ה׳. כָּךְ כָּל אֶחָד מַרְגִּישׁ בָּטוּחַ עִם הַחֲפָצִים שֶׁלּוֹ, וְהָאַהֲבָה בֵּינֵינוּ גְּדֵלָה.
📖 פֶּרֶק ד׳, הֲלָכוֹת א׳, ד׳

😜לֹא גּוֹנְבִים בִּצְחוֹק

יֵשׁ מִי שֶׁלּוֹקֵחַ בִּצְחוֹק חֵפֶץ חָשׁוּב שֶׁל חָבֵר, וְנֶהֱנֶה לִרְאוֹת אוֹתוֹ דּוֹאֵג וּמְחַפֵּשׂ. זֶה אָסוּר מִן הַתּוֹרָה, כִּי הוּא גּוֹרֵם לֶחָבֵר צַעַר.
וּמִצַּד אִסּוּר הַגְּנֵבָה: יֵשׁ אוֹמְרִים שֶׁגַּם גְּנֵבָה בְּמִשְׂחָק אֲסוּרָה מִן הַתּוֹרָה, וְיֵשׁ אוֹמְרִים שֶׁחֲכָמִים אָסְרוּ אוֹתָהּ – כְּדֵי שֶׁלֹּא יִתְרַגְּלוּ לִגְנֹב בֶּאֱמֶת. לְכָל הַדֵּעוֹת אָסוּר.
💛 אַהֲבַת הַשֵּׁם: צְחוֹק אֲמִתִּי הוּא כְּשֶׁכֻּלָּם שְׂמֵחִים יַחַד. ה׳ רוֹצֶה שֶׁהַשִּׂמְחָה שֶׁלָּנוּ לֹא תָּבוֹא עַל חֶשְׁבּוֹן הַצַּעַר שֶׁל אַחֵר.
📖 פֶּרֶק ד׳, הֲלָכָה א׳

💰עֹדֶף בְּטָעוּת

סִכַּמְתֶּם עַל מְחִיר, וְהַמּוֹכֵר טָעָה בִּסְפִירַת הַכֶּסֶף וְנָתַן לְךָ עֹדֶף גָּדוֹל מִדַּי? חוֹבָה לְהַחְזִיר, אֲפִלּוּ שֶׁקֶל אֶחָד.
גַּם מִגּוֹי אָסוּר לִגְנֹב אֲפִלּוּ דָּבָר קָטָן, וְאָסוּר לְבַלְבֵּל אוֹתוֹ בַּחֶשְׁבּוֹן. וּמִי שֶׁקָּנָה מִגּוֹי בְּהַקָּפָה, וְהַגּוֹי שָׁכַח – חַיָּב לְשַׁלֵּם לוֹ.
💛 אַהֲבַת הַשֵּׁם: מְסֹרֶת בְּיִשְׂרָאֵל: מִי שֶׁהֶחֱזִיר לְגוֹי כֶּסֶף שֶׁהִגִּיעַ אֵלָיו בְּטָעוּת וְקִדֵּשׁ שֵׁם שָׁמַיִם – הִצְלִיחַ וְהִתְעַשֵּׁר. וּמִי שֶׁהִרְוִיחַ מִטָּעֻיּוֹת – לֹא רָאָה בְּרָכָה בְּכַסְפּוֹ.
📖 פֶּרֶק ד׳, הֲלָכוֹת ב׳, י״ז

🌍גַּם מִגּוֹי

גַּם מִגּוֹי אָסוּר לִגְנֹב אוֹ לִגְזֹל, אֲפִלּוּ דָּבָר קָטָן, וְאָסוּר לְבַלְבֵּל אוֹתוֹ בַּחֶשְׁבּוֹן. וְאִם הַגּוֹי טָעָה מֵעַצְמוֹ בַּחֶשְׁבּוֹן – הַיְּהוּדִי אֵינוֹ חַיָּב לְתַקֵּן, אֲבָל אִם יְתַקֵּן כְּדֵי לְקַדֵּשׁ שֵׁם שָׁמַיִם – הֲרֵי זֶה מְשֻׁבָּח.
וְאִם יֵשׁ חֲשָׁשׁ שֶׁהַגּוֹי יְגַלֶּה אֶת הַטָּעוּת וְיִתְרַעֵם, וְיִהְיֶה חִלּוּל ה׳ – חוֹבָה לְתַקֵּן אוֹתוֹ וּלְקַבֵּל רַק אֶת מָה שֶׁמַּגִּיעַ.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁיְּהוּדִי מִתְנַהֵג בְּיֹשֶׁר, כֻּלָּם רוֹאִים אֶת יֹפִי הַתּוֹרָה, וְשֵׁם ה׳ מִתְקַדֵּשׁ בָּעוֹלָם.
📖 פֶּרֶק ד׳, הֲלָכָה ב׳

🙋חָבֵר טוֹב שֶׁאֵינוֹ כָּאן

רוֹצִים לְהִשְׁתַּמֵּשׁ בְּחֵפֶץ שֶׁל חָבֵר טוֹב, וְהוּא לֹא נִמְצָא. אִם יֵשׁ אֲפִלּוּ אָחוּז אֶחָד שֶׁהוּא יִתְרַעֵם – לְכָל הַדֵּעוֹת אָסוּר, כִּי בְּסָפֵק שֶׁל אִסּוּר תּוֹרָה מַחְמִירִים. וְאִם בָּרוּר בְּמֵאָה אָחוּז שֶׁהוּא מַסְכִּים – יֵשׁ אוֹסְרִים בְּלִי רְשׁוּת מְפֹרֶשֶׁת וְיֵשׁ מַתִּירִים. נוֹהֲגִים לְהָקֵל, וּמִדַּת חֲסִידוּת לְהַחְמִיר.
בִּדְבָרִים זוֹלִים שֶׁלֹּא מַקְפִּידִים עֲלֵיהֶם – גַּפְרוּר, קֵיסָם, עֵט פָּשׁוּט – מֻתָּר רַק כְּשֶׁשְּׁנֵי תְּנָאִים מִתְקַיְּמִים: הַבְּעָלִים לֹא בַּסְּבִיבָה וְאִי אֶפְשָׁר לִשְׁאֹל, וְהוּא אֵינוֹ יָדוּעַ כְּקַפְּדָן.
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁמַּקְפִּיד לִשְׁאֹל גַּם כְּשֶׁאֶפְשָׁר הָיָה לְהָקֵל, מְגַלֶּה כַּמָּה יָקָר לוֹ רְצוֹן ה׳ – אֲפִלּוּ בִּדְבָרִים קְטַנִּים.
📖 פֶּרֶק ד׳, הֲלָכוֹת ג׳–ד׳

🔀בְּלִי רְשׁוּת – שְׁתֵּי שִׁיטוֹת

חָבֵר טוֹב אֵינוֹ כָּאן, וּבָרוּר בְּמֵאָה אָחוּז שֶׁהוּא מַסְכִּים שֶׁנִּשְׁתַּמֵּשׁ בַּחֵפֶץ שֶׁלּוֹ. לְדַעַת רַבִּי זַלְמָן מִלַּאדִי – כָּל זְמַן שֶׁלֹּא נָתַן רְשׁוּת מְפֹרֶשֶׁת, אָסוּר. וּלְדַעַת הַשַּׁ״ךְ – מֻתָּר, וְכָךְ נוֹהֲגִים. וּמִנְהַג חֲסִידוּת לְהַחְמִיר כְּשֶׁאֶפְשָׁר.
וְסִדּוּר שֶׁל אָדָם אַחֵר בְּבֵית הַכְּנֶסֶת, לִתְפִלָּה אַחַת אוֹ לְעִיּוּן קַל: לְפִי הַמִּשְׁנָה בְּרוּרָה – אָסוּר בְּלִי רְשׁוּת. לְפִי עֲרוּךְ הַשֻּׁלְחָן – מֻתָּר, וְכָךְ הַמִּנְהָג הַיּוֹם. וְיֵשׁ מַחְמִירִים.
💛 אַהֲבַת הַשֵּׁם: גַּם הַמְּקִלִּים וְגַם הַמַּחְמִירִים שׁוֹמְרִים עַל כְּבוֹד הַחָבֵר וְעַל מָמוֹנוֹ. מִי שֶׁזָּהִיר בְּמָמוֹן חֲבֵרוֹ – אָהוּב לְמַעְלָה וְנֶחְמָד לְמַטָּה.
📖 פֶּרֶק ד׳, הֲלָכוֹת ד׳, ח׳

🎒חֲפָצִים שֶׁהִתְחַלְּפוּ

מִי שֶׁהַחֲלִיפָה אוֹ הַתִּיק שֶׁלּוֹ הִתְחַלְּפוּ בְּאֵירוּעַ – לֹא יִשְׁתַּמֵּשׁ בַּחֵפֶץ שֶׁנִּשְׁאַר. אוּלַי מִישֶׁהוּ שְׁלִישִׁי לָקַח אֶת שֶׁלּוֹ, וְהַחֵפֶץ הַזֶּה שַׁיָּךְ לְאָדָם אַחֵר. מַכְרִיזִים שֶׁנִּמְצָא חֵפֶץ וְשֶׁאָבַד חֵפֶץ, כְּמוֹ בַּאֲבֵדָה.
אֲבָל מִטְרִיָּה אוֹ מְעִיל גֶּשֶׁם שֶׁהִתְחַלְּפוּ – נָהֲגוּ שֶׁכָּל אֶחָד מִשְׁתַּמֵּשׁ בֵּינְתַיִם בְּשֶׁל חֲבֵרוֹ, כִּי לֹא מַקְפִּידִים עֲלֵיהֶם, וּבְלִי זֶה שְׁנֵיהֶם יִרָּטְבוּ בַּגֶּשֶׁם. וְיֵשׁ מַחְמִירִים גַּם בָּזֶה.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנַחְנוּ שׁוֹמְרִים עַל חֵפֶץ שֶׁל אַחֵר כְּאִלּוּ הוּא שֶׁלָּנוּ, אֲנַחְנוּ מְקַיְּמִים "וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ" – וּמְשַׂמְּחִים אֶת ה׳ שֶׁצִּוָּה עָלֶיהָ.
📖 פֶּרֶק ד׳, הֲלָכָה ה׳

🍽️אֹכֶל מִן הַשִּׂמְחָה

בַּחֲתֻנָּה, אֶת הָאֹכֶל שֶׁנִּשְׁאַר בַּצַּלַּחַת הָאִישִׁית שֶׁלְּךָ מֻתָּר לָקַחַת הַבַּיְתָה, כִּי הוּא מִמֵּילָא יִזָּרֵק – בִּתְנַאי שֶׁעוֹשִׂים זֹאת בְּנִימוּס, בְּלִי לִפְגֹּעַ בִּכְבוֹד בַּעֲלֵי הַשִּׂמְחָה.
אֲבָל הָאֹכֶל שֶׁבַּקְּעָרוֹת הַכְּלָלִיּוֹת שַׁיָּךְ לְבַעֲלֵי הַשִּׂמְחָה אוֹ לְבַעַל הָאוּלָם, וְצָרִיךְ לְבַקֵּשׁ רְשׁוּת. וּמִי שֶׁהֻזְמַן עִם הַמִּשְׁפָּחָה – אֵינוֹ לוֹקֵחַ אֶת הַמָּנוֹת שֶׁל מִי שֶׁלֹּא בָּא.
💛 אַהֲבַת הַשֵּׁם: בָּאִים לַחֲתֻנָּה כְּדֵי לְשַׂמֵּחַ אֶת הֶחָתָן וְהַכַּלָּה. מִי שֶׁשּׁוֹמֵר עַל כְּבוֹדָם וְעַל מָמוֹנָם מוֹסִיף לַשִּׂמְחָה – שִׂמְחָה שֶׁה׳ אוֹהֵב.
📖 פֶּרֶק ד׳, הֲלָכָה ו׳

🍋סְפָרִים וְחֶפְצֵי מִצְוָה

נוֹחַ לוֹ לְאָדָם שֶׁעוֹשִׂים מִצְוָה בְּשֶׁלּוֹ. לָכֵן מֻתָּר לִטֹּל בְּבֵית הַכְּנֶסֶת אַרְבָּעָה מִינִים שֶׁל אַחֵר בְּלִי רְשׁוּת – כְּשֶׁאִי אֶפְשָׁר לִשְׁאֹל אוֹתוֹ, לְשִׁמּוּשׁ קָצָר, בְּאוֹתוֹ מָקוֹם, וּמַחֲזִירִים בְּדִיּוּק כְּמוֹ שֶׁהָיוּ. וְאִם יָדוּעַ שֶׁהַבְּעָלִים קַפְּדָן – אָסוּר.
סֵפֶר מִסִּפְרִיַּת בֵּית הַכְּנֶסֶת שַׁיָּךְ לַצִּבּוּר. אָסוּר לְהוֹצִיא אוֹתוֹ הַבַּיְתָה בְּלִי רְשׁוּת הַגַּבַּאי – גַּם כְּדֵי לִלְמֹד, וְגַם כְּשֶׁאַף אֶחָד לֹא לוֹמֵד בּוֹ.
💛 אַהֲבַת הַשֵּׁם: סִפְרֵי הַקֹּדֶשׁ וְאַרְבָּעַת הַמִּינִים הֵם כְּלֵי הַקְּדֻשָּׁה שֶׁלָּנוּ. מִצְוָה שֶׁנַּעֲשֵׂית בְּיֹשֶׁר, בְּלִי שֶׁמֶץ שֶׁל גְּנֵבָה, הִיא מִצְוָה שְׁלֵמָה לִפְנֵי ה׳.
📖 פֶּרֶק ד׳, הֲלָכוֹת ז׳, ט׳

📖סִדּוּר, טַלִּית וּסְפָרִים

סֵפֶר חֹל שֶׁל חָבֵר – כְּמוֹ כָּל חֵפֶץ, וְאָסוּר בְּלִי רְשׁוּת. בְּסֵפֶר קֹדֶשׁ, לְעִיּוּן קָצָר אוֹ לִתְפִלָּה אַחַת בְּסִדּוּר, כְּשֶׁאִי אֶפְשָׁר לִשְׁאֹל – יֵשׁ אוֹסְרִים, וְהַמִּנְהָג לְהָקֵל. אֲבָל לִלְמֹד זְמַן רַב בְּסֵפֶר שֶׁל אַחֵר – אָסוּר בְּלִי רְשׁוּת, כִּי הַסֵּפֶר מִתְבַּלֶּה. וְכַאֲשֶׁר הַבְּעָלִים כָּאן – תָּמִיד שׁוֹאֲלִים.
בְּטַלִּית וּתְפִלִּין שֶׁל אָדָם זָר – כַּמָּה פּוֹסְקִים כּוֹתְבִים שֶׁהַיּוֹם צָרִיךְ לְבַקֵּשׁ רְשׁוּת, כִּי רַבִּים מַקְפִּידִים, וְיֵשׁ מְקִלִּים. וְכָל שֶׁכֵּן כְּשֶׁהַטַּלִּית מֻסְתֶּרֶת. כְּשֶׁלֹּא יָדוּעַ שֶׁהַבְּעָלִים מַסְכִּים – עָדִיף לְהִתְפַּלֵּל בְּלִי טַלִּית מֵאֲשֶׁר לְהִכָּנֵס לַחֲשַׁשׁ גָּזֵל.
💛 אַהֲבַת הַשֵּׁם: גַּם הַמִּצְוָה הֲכִי יָפָה צְרִיכָה לְהֵעָשׂוֹת בְּיֹשֶׁר. ה׳ אוֹהֵב מִצְוָה שֶׁאֵין בָּהּ שׁוּם צַעַר לְאַחֵר.
📖 פֶּרֶק ד׳, הֲלָכוֹת ז׳–ח׳

🍯כַּד הַדְּבַשׁ

אָדָם הוֹלֵךְ עִם כַּד דְּבַשׁ, וּפִתְאוֹם הַכַּד נִשְׁבַּר. מוּלוֹ כַּד רֵיק שֶׁל אָדָם אַחֵר. מֻתָּר לוֹ לָקַחַת אֶת הַכַּד בְּלִי רְשׁוּת וּלְהַצִּיל בּוֹ אֶת הַדְּבַשׁ. כָּךְ תִּקֵּן יְהוֹשֻׁעַ בִּן נוּן כְּשֶׁנִּכְנְסוּ יִשְׂרָאֵל לָאָרֶץ.
הַתַּקָּנָה הִיא לְנֵזֶק פִּתְאוֹמִי, כְּשֶׁאֵין זְמַן לְחַפֵּשׂ אֶת הַבְּעָלִים. אִם הַבְּעָלִים נִמְצָא – עָלָיו לַעֲזֹר מִיָּד. וְאִם הַכְּלִי נִזּוֹק – מְשַׁלְּמִים עַל הַנֵּזֶק.
💛 אַהֲבַת הַשֵּׁם: יְהוֹשֻׁעַ בָּנָה חֶבְרָה שֶׁבָּהּ כָּל אֶחָד עוֹזֵר לְהַצִּיל אֶת מָמוֹן חֲבֵרוֹ. בְּאֶרֶץ הַקֹּדֶשׁ ה׳ רוֹצֶה שֶׁנִּהְיֶה כְּמוֹ מִשְׁפָּחָה אַחַת.
📖 פֶּרֶק ד׳, הֲלָכָה י״א

🚌רוֹאִים גְּנֵבָה

"לֹא תַעֲמֹד עַל דַּם רֵעֶךָ" – חַיָּבִים לְהַצִּיל גַּם אֶת הָרְכוּשׁ שֶׁל הֶחָבֵר. מִי שֶׁרוֹאֶה אָדָם גּוֹנֵב בַּחֲנוּת – לֹא עוֹמֵד מִן הַצַּד. קֹדֶם לוֹחֲשִׁים לוֹ שֶׁלֹּא יִגְנֹב, כְּדֵי לֹא לְבַיֵּשׁ אוֹתוֹ, וְאִם הוּא לֹא מַפְסִיק – אוֹמְרִים בְּקוֹל רָם, כְּדֵי שֶׁיָּבוֹאוּ לַעֲזֹר.
וְכֵן מִי שֶׁעוֹלֶה לָאוֹטוֹבּוּס בְּלִי לְשַׁלֵּם – אוֹמְרִים לוֹ בְּנַחַת שֶׁצָּרִיךְ לְשַׁלֵּם, וְאִם לֹא עוֹזֵר – אוֹמְרִים לַנֶּהָג. מִי שֶׁפּוֹחֵד שֶׁיִּפְגְּעוּ בּוֹ – אֵינוֹ חַיָּב לְהִתְעָרֵב מִיָּד, אֲבָל יִהְיֶה מוּכָן לְהִתְלוֹנֵן וּלְהָעִיד אַחַר כָּךְ.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁכֻּלָּם שׁוֹמְרִים זֶה עַל זֶה, הַגַּנָּבִים מִתְמַעֲטִים – וְזוֹ טוֹבָה גְּדוֹלָה לְעַם יִשְׂרָאֵל. כָּךְ נִרְאֵית אַהֲבָה שֶׁל אַחִים, בָּנִים שֶׁל אָב אֶחָד.
📖 פֶּרֶק ד׳, הֲלָכָה י״ב

🌾רֹב הָעוֹלָם נִכְשָׁלִים בְּגָזֵל

רַב יְהוּדָה אָמַר: רֹב הָעוֹלָם נִכְשָׁלִים בְּגָזֵל. רַשִׁ״י מַסְבִּיר: לֹא שֶׁרֹב הָאֲנָשִׁים מִתְכַּוְּנִים לִגְנֹב, אֶלָּא שֶׁהַמָּמוֹן שֶׁלָּהֶם חָבִיב עֲלֵיהֶם, וּמָמוֹן הֶחָבֵר פָּחוֹת חָשׁוּב בְּעֵינֵיהֶם – וְכָךְ הֵם מַתִּירִים לְעַצְמָם וְנִכְשָׁלִים בְּלִי לָשִׂים לֵב.
לָכֵן גַּם עוֹבֵד בְּמִשְׂרָד צָרִיךְ לְהִזָּהֵר: לֹא לְהִשְׁתַּמֵּשׁ בַּטֶּלֶפוֹן, בַּדַּפִּים אוֹ בַּמַּעֲטָפוֹת שֶׁל הָעֲבוֹדָה לְצָרְכָיו הַפְּרָטִיִּים, בְּלִי רְשׁוּת מְפֹרֶשֶׁת.
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁשּׁוֹמֵר עַל מָמוֹן חֲבֵרוֹ כְּמוֹ עַל שֶׁלּוֹ, מְקַיֵּם "וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ" בְּמַעֲשִׂים – וְזוֹ אַהֲבָה שֶׁה׳ רוֹאֶה וְשָׂמֵחַ בָּהּ.
📖 פֶּרֶק ד׳, הֲלָכָה י״ד

👓לִפְנֵי עִוֵּר – עֵצָה רָעָה

"לִפְנֵי עִוֵּר לֹא תִתֵּן מִכְשֹׁל", וְ"אָרוּר מַשְׁגֶּה עִוֵּר בַּדָּרֶךְ". חֲזַ״ל פֵּרְשׁוּ: מִי שֶׁאֵינוֹ מֵבִין בִּתְחוּם מְסֻיָּם – כְּמוֹ עִוֵּר בּוֹ. וּמִי שֶׁנּוֹתֵן לוֹ עֵצָה רָעָה, שֶׁגּוֹרֶמֶת לוֹ הֶפְסֵד אוֹ טִרְחָה – עוֹבֵר עַל אִסּוּר תּוֹרָה.
לְמָשָׁל, מוֹכֵר שֶׁיּוֹדֵעַ שֶׁמַּכְשִׁיר מְסֻיָּם מִתְקַלְקֵל מַהֵר, וּמְשַׁבֵּחַ אוֹתוֹ לְקוֹנֶה שֶׁבִּקֵּשׁ עֵצָה. וְאִם הוּא גַּם מַרְוִיחַ מִזֶּה – עָבַר גַּם עַל אִסּוּר אוֹנָאָה.
💛 אַהֲבַת הַשֵּׁם: אִי אֶפְשָׁר לְהוֹכִיחַ מָה הָיָה בַּלֵּב שֶׁל נוֹתֵן הָעֵצָה, אֲבָל ה׳ יוֹדֵעַ – וְלָכֵן נֶאֱמַר בָּזֶה "וְיָרֵאתָ מֵאֱלֹהֶיךָ".
📖 פֶּרֶק ד׳, הֲלָכָה ט״ו

🐑"בְּכָל כֹּחִי עָבַדְתִּי"

גַּם זְמַן הוּא מָמוֹן. פּוֹעֵל שֶׁמִּתְבַּטֵּל מְעַט כָּאן וּמְעַט כָּאן – גּוֹזֵל אֶת בַּעַל הַבַּיִת. יַעֲקֹב אָבִינוּ אָמַר: "כִּי בְּכָל כֹּחִי עָבַדְתִּי אֶת אֲבִיכֶן" – בַּיּוֹם אֲכָלוֹ חֹרֶב, וּבַלַּיְלָה קֶרַח.
וְצָרִיךְ הָעוֹבֵד לִישֹׁן וְלֶאֱכֹל כָּרָאוּי, כִּי אִם יִהְיֶה עָיֵף אוֹ רָעֵב – נִמְצָא גּוֹזֵל אֶת מִי שֶׁמַּעֲסִיק אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: גְּדוֹלָה בְּעֵינֵי הַקָּדוֹשׁ בָּרוּךְ הוּא הַמְּלָאכָה שֶׁל מִי שֶׁעוֹשֶׂה אוֹתָהּ בֶּאֱמוּנָה. ה׳ רָאָה אֶת יְגִיעַ כַּפָּיו שֶׁל יַעֲקֹב, וְהִצִּיל אוֹתוֹ מִלָּבָן.
📖 פֶּרֶק ד׳, הֲלָכָה ט״ז

🧠 בְּדֹק אֶת עַצְמְךָ

מָה הַהֶבְדֵּל בֵּין תִּיק שֶׁהִתְחַלֵּף בְּאֵירוּעַ לְבֵין מִטְרִיָּה שֶׁהִתְחַלְּפָה בְּיוֹם גֶּשֶׁם?
💡 בַּתִּיק אָסוּר לְהִשְׁתַּמֵּשׁ – אוּלַי הוּא שֶׁל אָדָם שְׁלִישִׁי. עַל מִטְרִיָּה וּמְעִיל גֶּשֶׁם לֹא מַקְפִּידִים, וּבְלִי שִׁמּוּשׁ שְׁנֵיהֶם יִרָּטְבוּ – לָכֵן נָהֲגוּ לְהָקֵל. וְיֵשׁ מַחְמִירִים גַּם בָּזֶה.
יוֹסִי צָרִיךְ לִכְתֹּב מִלָּה אַחַת. עַל הַשֻּׁלְחָן מֻנָּח עֵט פָּשׁוּט שֶׁל חָבֵר שֶׁיָּצָא. מָתַי מֻתָּר לוֹ לְהִשְׁתַּמֵּשׁ בּוֹ?
💡 בִּדְבָרִים זוֹלִים שֶׁאֲנָשִׁים לֹא מַקְפִּידִים עֲלֵיהֶם מֻתָּר – אִם הַבְּעָלִים לֹא כָּאן וְהוּא אֵינוֹ קַפְּדָן. אִם אֶפְשָׁר לִשְׁאֹל – חוֹבָה לִשְׁאֹל. וּמִדַּת חֲסִידוּת לְהַחְמִיר.
הַכַּדּוּר שֶׁל אָרִי מֻנָּח בֶּחָצֵר, וְאָרִי לֹא בַּבַּיִת. יֵשׁ סִכּוּי קָטָן שֶׁהוּא לֹא יַסְכִּים. מָה הַדִּין?
💡 שׁוֹאֵל שֶׁלֹּא מִדַּעַת נֶחְשָׁב גַּנָּב, גַּם לִזְמַן קָצָר. כְּשֶׁיֵּשׁ אֲפִלּוּ אָחוּז אֶחָד שֶׁהֶחָבֵר יִתְרַעֵם – לְכָל הַדֵּעוֹת אָסוּר.

פּוּרִים וּמִקְרָא מְגִלָּה

🎭 הִלְכוֹת פּוּרִים · פֶּרֶק ט״ו

🎭נֵס פּוּרִים – וְנַהֲפוֹךְ הוּא

הָמָן הָרָשָׁע, מִזֶּרַע עֲמָלֵק, גָּזַר לְהַשְׁמִיד אֶת כָּל הַיְּהוּדִים בְּי״ג בַּאֲדָר. מָרְדֳּכַי לֹא הִסְכִּים לִכְרֹעַ וּלְהִשְׁתַּחֲווֹת לְהָמָן. אֲבָל ה׳ הִקְדִּים רְפוּאָה לַמַּכָּה – אֶסְתֵּר כְּבָר הָיְתָה בְּבֵית הַמֶּלֶךְ, וְהִיא וּמָרְדֳּכַי הֵפֵרוּ אֶת עֲצַת הָמָן.
וְנַהֲפוֹךְ הוּא: הַיְּהוּדִים גָּבְרוּ עַל שׂוֹנְאֵיהֶם, וְאֶת הָמָן תָּלוּ עַל הָעֵץ שֶׁהֵכִין לְמָרְדֳּכַי.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁיִּשְׂרָאֵל חוֹטְאִים, הֵם נִקְרָאִים בָּנִים לַמָּקוֹם. ה׳ מְסַבֵּב אֶת הַכֹּל בַּסֵּתֶר – כְּדֵי לְהַצִּיל אֶת בָּנָיו וּלְגָאֳלָם.
📖 פֶּרֶק ט״ו, הֲלָכָה א׳

📖קִיְּמוּ וְקִבְּלוּ

בְּהַר סִינַי קִבְּלוּ יִשְׂרָאֵל אֶת הַתּוֹרָה אַחֲרֵי נִסִּים גְּדוֹלִים, וְכִמְעַט לֹא הָיְתָה לָהֶם בְּרֵרָה – "כָּפָה עֲלֵיהֶם הַר כְּגִיגִית". בִּימֵי אֲחַשְׁוֵרוֹשׁ יָכְלוּ הַיְּהוּדִים לְהִתְבּוֹלֵל וּלְהִנָּצֵל, וּבְכָל זֹאת בָּחֲרוּ מֵרָצוֹן לְהִשָּׁאֵר דְּבוּקִים בַּה׳, חָזְרוּ בִּתְשׁוּבָה וְהִתְחַזְּקוּ בַּמִּצְווֹת.
אָמְרוּ חֲכָמִים: "הֲדוּר קִבְּלוּהָ בִּימֵי אֲחַשְׁוֵרוֹשׁ", שֶׁנֶּאֱמַר "קִיְּמוּ וְקִבְּלוּ הַיְּהוּדִים" – קִיְּמוּ מָה שֶׁקִּבְּלוּ כְּבָר.
💛 אַהֲבַת הַשֵּׁם: בְּפוּרִים הִתְבָּרֵר שֶׁהַקֶּשֶׁר שֶׁל עַם יִשְׂרָאֵל לַה׳ וּלְתוֹרָתוֹ הוּא מֻחְלָט – מִתּוֹךְ אַהֲבָה, בְּלִי שׁוּם כְּפִיָּה. גַּם אֲנַחְנוּ יְכוֹלִים לִבְחֹר בַּתּוֹרָה מֵחָדָשׁ בְּכָל שָׁנָה.
📖 פֶּרֶק ט״ו, הֲלָכָה ב׳

✍️כִּתְבוּנִי לְדוֹרוֹת

אֶסְתֵּר בִּקְּשָׁה מִן הַחֲכָמִים: "כִּתְבוּנִי לְדוֹרוֹת" וְ"קִבְעוּנִי לְדוֹרוֹת". בַּסּוֹף כָּתְבוּ אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה אֶת מְגִלַּת אֶסְתֵּר בְּרוּחַ הַקֹּדֶשׁ, וְקָבְעוּ אֶת פּוּרִים לְדוֹרוֹת. אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה הָיוּ מֵאָה וְעֶשְׂרִים זְקֵנִים, וּבֵינֵיהֶם נְבִיאִים וַחֲכָמִים – וְגַם מָרְדֳּכַי הַיְּהוּדִי וְעֶזְרָא הַסּוֹפֵר.
שֶׁבַע מִצְווֹת בְּפוּרִים. אַרְבַּע מְיֻחָדוֹת לוֹ: מִקְרָא מְגִלָּה, מִשְׁלוֹחַ מָנוֹת, מַתָּנוֹת לָאֶבְיוֹנִים, וּמִשְׁתֶּה וְשִׂמְחָה. וְעוֹד שָׁלֹשׁ: קְרִיאַת הַתּוֹרָה, "עַל הַנִּסִּים", וְאִסּוּר הֶסְפֵּד וְתַעֲנִית.
💛 אַהֲבַת הַשֵּׁם: "אֶסְתֵּר סוֹף כָּל הַנִּסִּים" – נֵס פּוּרִים הוּא הָאַחֲרוֹן שֶׁנִּכְתַּב בַּתַּנַ״ךְ. מֵאָז ה׳ מַסְתִּיר אֶת פָּנָיו, וְעָלֵינוּ לִמְצֹא אוֹתוֹ גַּם בְּתוֹךְ הַהֶסְתֵּר.
📖 פֶּרֶק ט״ו, הֲלָכָה ג׳

✍️אֶסְתֵּר סוֹף כָּל הַנִּסִּים

נֵס פּוּרִים הוּא הַנֵּס הָאַחֲרוֹן שֶׁנִּכְתַּב בַּתַּנַ״ךְ, וּבִכְתִיבַת מְגִלַּת אֶסְתֵּר נֶחְתַּם הַתַּנַ״ךְ. מִצְווֹת הַפּוּרִים הֵן "מִדִּבְרֵי קַבָּלָה" – בֵּין מִצְוָה מִן הַתּוֹרָה לְבֵין מִצְוָה מִדִּבְרֵי חֲכָמִים.
פּוּרִים הוּא הַחֻלְיָה שֶׁמְּחַבֶּרֶת בֵּין הַתּוֹרָה שֶׁבִּכְתָב לַתּוֹרָה שֶׁבְּעַל פֶּה. מִמֶּנּוּ נִפְתַּח הַפֶּתַח לְלִמּוּד הַתּוֹרָה שֶׁבְּעַל פֶּה בִּימֵי הַבַּיִת הַשֵּׁנִי.
💛 אַהֲבַת הַשֵּׁם: בְּנֵס פּוּרִים ה׳ סִבֵּב אֶת הַכֹּל בַּסֵּתֶר, לְטוֹבַת יִשְׂרָאֵל. כָּךְ גַּם בַּחַיִּים שֶׁלָּנוּ: ה׳ מַנְהִיג אֶת הָעוֹלָם וְשׁוֹמֵר עָלֵינוּ – גַּם כְּשֶׁאֵינֶנּוּ רוֹאִים אוֹתוֹ.
📖 פֶּרֶק ט״ו, הֲלָכָה ג׳

🏰פְּרָזִים וּמֻקָּפִים

בְּרֹב הַמְּקוֹמוֹת חוֹגְגִים פּוּרִים בְּי״ד בַּאֲדָר – כִּי בְּי״ד נָחוּ הַיְּהוּדִים מִן הַמִּלְחָמָה. בְּשׁוּשַׁן הַבִּירָה נִלְחֲמוּ גַּם בְּי״ד, וְנָחוּ רַק בְּט״ו. לָכֵן תִּקְּנוּ: עִיר שֶׁהָיְתָה מֻקֶּפֶת חוֹמָה מִימוֹת יְהוֹשֻׁעַ בִּן נוּן – חוֹגֶגֶת בְּט״ו, וּשְׁאָר הַמְּקוֹמוֹת – "פְּרָזִים" – בְּי״ד.
הַיּוֹם רַק בִּירוּשָׁלַיִם חוֹגְגִים פּוּרִים בְּט״ו, כִּי רַק עָלֶיהָ יֵשׁ מָסֹרֶת בְּרוּרָה שֶׁהָיְתָה מֻקֶּפֶת חוֹמָה מִימוֹת יְהוֹשֻׁעַ.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים קָבְעוּ "מִימוֹת יְהוֹשֻׁעַ" מִפְּנֵי כְּבוֹד אֶרֶץ יִשְׂרָאֵל. גַּם בְּתוֹךְ שִׂמְחַת פּוּרִים – הַלֵּב שֶׁלָּנוּ קָשׁוּר לָאָרֶץ וְלִירוּשָׁלַיִם.
📖 פֶּרֶק ט״ו, הֲלָכָה ד׳; פֶּרֶק י״ז, הֲלָכָה א׳

📢פִּרְסוּם הַנֵּס

הַכֹּל חַיָּבִים בְּמִקְרָא מְגִלָּה – גְּבָרִים, נָשִׁים וְגֵרִים. הַשּׁוֹמֵעַ אֶת הַקְּרִיאָה יוֹצֵא יְדֵי חוֹבָה – בִּתְנַאי שֶׁהוּא שׁוֹמֵעַ מִמִּי שֶׁחַיָּב בַּמִּצְוָה. הַשּׁוֹמֵעַ מִקָּטָן שֶׁאֵינוֹ חַיָּב בְּמִצְווֹת – לֹא יָצָא. עִקַּר הַמִּצְוָה הוּא לְפַרְסֵם אֶת הַנֵּס: לְגַלּוֹת שֶׁה׳ מַנְהִיג אֶת הָעוֹלָם וּמְכַוֵּן אֶת הַכֹּל לְטוֹבָה.
כָּל כָּךְ חָשׁוּב לִשְׁמֹעַ בְּרֹב עָם, עַד שֶׁאֲפִלּוּ הַכֹּהֲנִים בַּמִּקְדָּשׁ דָּחוּ אֶת קָרְבַּן הַתָּמִיד כְּדֵי לִשְׁמֹעַ מְגִלָּה עִם הַצִּבּוּר, וְתַלְמִידֵי חֲכָמִים מַפְסִיקִים מִלִּמּוּדָם וְהוֹלְכִים לְבֵית הַכְּנֶסֶת.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁהַרְבֵּה יְהוּדִים שׁוֹמְעִים יַחַד, הָאֱמוּנָה מִתְחַזֶּקֶת בַּלְּבָבוֹת. כָּל אֶחָד שֶׁבָּא לְבֵית הַכְּנֶסֶת מוֹסִיף עוֹד קוֹל לַשִּׁיר הַגָּדוֹל: ה׳ הוּא הַמּוֹשִׁיעַ!
📖 פֶּרֶק ט״ו, הֲלָכָה ה׳

📜נָשִׁים וּבָנוֹת בְּמִקְרָא מְגִלָּה

גַּם נָשִׁים חַיָּבוֹת בְּמִקְרָא מְגִלָּה. לְדַעַת רַשִׁ״י וְהָרַמְבַּ״ם הֵן חַיָּבוֹת בִּקְרִיאָה כְּמוֹ הַגְּבָרִים, וּלְדַעַת בַּעַל הֲלָכוֹת גְּדוֹלוֹת – רַק בִּשְׁמִיעָה. בִּגְלַל הַמַּחֲלֹקֶת הוֹרוּ רֹב הָאַחֲרוֹנִים שֶׁאִשָּׁה לֹא תִּקְרָא כְּדֵי לְהוֹצִיא אִישׁ, אֶלָּא בִּשְׁעַת הַדְּחָק. אִשָּׁה יְכוֹלָה לִקְרֹא לְנָשִׁים אֲחֵרוֹת. לְדַעַת רֹב הַפּוֹסְקִים הִיא מְבָרֶכֶת כְּנֻסַּח הַגְּבָרִים, וְאִם הֵן עֶשֶׂר – גַּם "הָרָב אֶת רִיבֵנוּ". וְיֵשׁ אוֹמְרִים (בֶּן אִישׁ חַי, כַּף הַחַיִּים) שֶׁכְּשֶׁהַקְּרִיאָה עֲבוּר נָשִׁים אֵין מְבָרְכִים.
כְּשֶׁאִישׁ קוֹרֵא לַנָּשִׁים, הַמִּנְהָג הָרוֹוֵחַ שֶׁהַקּוֹרֵא מְבָרֵךְ עֲבוּרָן, וְיֵשׁ נוֹהֲגִים שֶׁאִשָּׁה אַחַת מְבָרֶכֶת – וּשְׁנֵי הַמִּנְהָגִים כְּשֵׁרִים. וְגַם לְאִשָּׁה אָסוּר לֶאֱכֹל לִפְנֵי שֶׁשָּׁמְעָה מְגִלָּה, אֲבָל אִם הִיא רְעֵבָה – מֻתָּר לָהּ לֶאֱכֹל מְעַט.
💛 אַהֲבַת הַשֵּׁם: הָרוֹצוֹת לְהַדֵּר יִשְׁמְעוּ מְגִלָּה מֵאִישׁ, וְאִם אֶפְשָׁר – בְּבֵית הַכְּנֶסֶת, כִּי בְּרֹב עָם הַנֵּס מִתְפַּרְסֵם יוֹתֵר. גַּם הַנָּשִׁים הָיוּ בַּנֵּס, וְכֻלָּנוּ מוֹדִים יַחַד לַה׳.
📖 פֶּרֶק ט״ו, הֲלָכוֹת ה׳–ז׳

📣אָרוּר הָמָן, בָּרוּךְ מָרְדֳּכַי

אֶת עֲשֶׂרֶת בְּנֵי הָמָן כּוֹתְבִים בְּשׁוּרוֹת יְשָׁרוֹת: הַשֵּׁמוֹת בְּצַד יָמִין, וְהַמִּלָּה "אֵת" בְּצַד שְׂמֹאל. אֶת הָאוֹת ו׳ שֶׁל "וַיְזָתָא" כּוֹתְבִים זְקוּפָה – לְלַמֵּד שֶׁתָּלוּ אֶת כֻּלָּם יַחַד. וְאַחֲרֵי הַקְּרִיאָה אוֹמְרִים: "אָרוּר הָמָן, בָּרוּךְ מָרְדֳּכַי… וְגַם חַרְבוֹנָא זָכוּר לַטּוֹב".
הַבְּרָכָה הָרִאשׁוֹנָה לִפְנֵי הַמְּגִלָּה: "אֲשֶׁר קִדְּשָׁנוּ בְּמִצְווֹתָיו וְצִוָּנוּ עַל מִקְרָא מְגִלָּה". וְעִקַּר עִנְיָנָהּ שֶׁל הַמְּגִלָּה – פִּרְסוּם הַנֵּס.
💛 אַהֲבַת הַשֵּׁם: כָּל הַקָּם נֶגֶד יִשְׂרָאֵל, עַמּוֹ שֶׁל ה׳, קָם נֶגֶד ה׳ עַצְמוֹ. וְה׳ שׁוֹמֵר עַל עַמּוֹ – כְּמוֹ אַבָּא שֶׁשּׁוֹמֵר עַל בָּנָיו.
📖 פֶּרֶק ט״ו, הֲלָכוֹת ה׳, י״ב–י״ג

🌙בַּלַּיְלָה וּבַיּוֹם

חַיָּב אָדָם לִקְרֹא אֶת הַמְּגִלָּה בַּלַּיְלָה וְלַחֲזֹר וְלִקְרֹא אוֹתָהּ בַּיּוֹם – כְּמוֹ שֶׁזָּעֲקוּ יִשְׂרָאֵל אֶל ה׳ בְּצָרָתָם יוֹם וָלַיְלָה. הַזְּרִיזִים קוֹרְאִים אֶת קְרִיאַת הַלַּיְלָה מִיָּד אַחֲרֵי עַרְבִית, וְאֶת קְרִיאַת הַיּוֹם מִיָּד אַחֲרֵי שַׁחֲרִית.
אָסוּר לֶאֱכֹל אוֹ לִישֹׁן לִפְנֵי קְרִיאַת הַמְּגִלָּה שֶׁל לַיְלָה – אֲבָל לִלְמֹד תּוֹרָה מֻתָּר.
💛 אַהֲבַת הַשֵּׁם: הַזְּרִיזִים מַקְדִּימִים לַמִּצְווֹת – כִּי מִי שֶׁאוֹהֵב, מְמַהֵר. כָּךְ אָנוּ מַרְאִים לַה׳ שֶׁאָנוּ שְׂמֵחִים לְקַיֵּם אֶת מִצְווֹתָיו.
📖 פֶּרֶק ט״ו, הֲלָכָה ו׳

🕰️זְמַן הַקְּרִיאָה

קְרִיאַת הַלַּיְלָה – כָּל הַלַּיְלָה, מִצֵּאת הַכּוֹכָבִים וְעַד עֲלוֹת הַשַּׁחַר. קְרִיאַת הַיּוֹם – כָּל הַיּוֹם, מֵהָנֵץ הַחַמָּה וְעַד הַשְּׁקִיעָה. לְדַעַת כַּמָּה מִגְּדוֹלֵי הָרִאשׁוֹנִים עִקַּר הַמִּצְוָה בַּיּוֹם, וְלָכֵן מְדַקְדְּקִים יוֹתֵר בִּקְרִיאַת הַיּוֹם וּמִשְׁתַּדְּלִים שֶׁתִּהְיֶה בְּרֹב עָם, וּלְפָחוֹת בְּמִנְיָן.
מִי שֶׁקָּשֶׁה לוֹ לְהַמְשִׁיךְ לָצוּם עַד אַחֲרֵי הַמְּגִלָּה – מֻתָּר לוֹ לִשְׁתּוֹת שְׁתִיָּה שֶׁאֵינָהּ מְשַׁכֶּרֶת, וּמִי שֶׁרָעֵב – לֶאֱכֹל מְעַט, כְּמוֹ פֵּרוֹת. אֲבָל לֹא סְעוּדָה.
💛 אַהֲבַת הַשֵּׁם: בַּלַּיְלָה וּבַיּוֹם – כָּל הַזְּמַן אָנוּ זוֹכְרִים אֶת חַסְדֵי ה׳. כְּמוֹ שֶׁזָּעֲקוּ אֵלָיו יוֹם וָלַיְלָה, כָּךְ מוֹדִים לוֹ יוֹם וָלַיְלָה.
📖 פֶּרֶק ט״ו, הֲלָכָה ו׳

🔀עוֹד מִנְהָגִים – וְכֻלָּם כַּהֲלָכָה

בִּקְרִיאַת הַתּוֹרָה: לְפִי הַשֻּׁלְחָן עָרוּךְ יֵשׁ נוֹהֲגִים לִקְרֹא אֶת הַפָּסוּק הָאַחֲרוֹן שֶׁל "וַיָּבֹא עֲמָלֵק" פַּעֲמַיִם, וּלְפִי הָרַמָּ״א – פַּעַם אַחַת. כְּשֶׁאִישׁ קוֹרֵא מְגִלָּה לְנָשִׁים: הַמִּנְהָג הָרוֹוֵחַ שֶׁהַקּוֹרֵא מְבָרֵךְ, וְיֵשׁ נוֹהֲגִים שֶׁאִשָּׁה אַחַת מְבָרֶכֶת.
כְּשֶׁאִשָּׁה קוֹרֵאת לְנָשִׁים: לְדַעַת רֹב הַפּוֹסְקִים הִיא מְבָרֶכֶת, וּלְדַעַת הַבֶּן אִישׁ חַי אֵין מְבָרְכִים. וּמְלָאכָה בְּלֵיל פּוּרִים: הָרַב מֶלָמֵד מַתִּיר, וְיֵשׁ מֵהָאַחֲרוֹנִים שֶׁאוֹסְרִים.
💛 אַהֲבַת הַשֵּׁם: כָּל אֶחָד שׁוֹמֵר עַל מִנְהַג מִשְׁפַּחְתּוֹ וְעַל פְּסַק רַבּוֹתָיו – בְּאַהֲבָה, בְּלִי לְזַלְזֵל בַּאֲחֵרִים. כָּל הַמִּנְהָגִים נוֹסְדוּ עַל יְדֵי גְּדוֹלֵי תּוֹרָה.
📖 פֶּרֶק ט״ו, הֲלָכוֹת ז׳, י״ד–ט״ו

👦יְלָדִים בַּמְּגִלָּה

מְחַנְּכִים בָּנִים וּבָנוֹת לִשְׁמֹעַ אֶת כָּל הַמְּגִלָּה כְּשֶׁהֵם מְבִינִים וִיכוֹלִים לִשְׁמֹעַ אֶת כֻּלָּהּ – רַבִּים מַגִּיעִים לְכָךְ אַחֲרֵי גִּיל תֵּשַׁע. מִנְהָג טוֹב לְהָבִיא גַּם בְּנֵי חָמֵשׁ אוֹ שֵׁשׁ, אֲבָל קְטַנִּים שֶׁעֲלוּלִים לְהַפְרִיעַ – לֹא מְבִיאִים.
כְּדֵי לְעוֹרֵר אֶת הַיְּלָדִים, הַקָּהָל אוֹמֵר בְּקוֹל רָם אַרְבָּעָה פְּסוּקִים: "אִישׁ יְהוּדִי", "וּמָרְדֳּכַי יָצָא", "לַיְּהוּדִים", "כִּי מָרְדֳּכַי" – וְהַחַזָּן חוֹזֵר עֲלֵיהֶם. וּמַכִּים אֶת הָמָן בְּרַעֲשָׁנִים – אֲבָל לֹא כְּשֶׁהַקּוֹרֵא מַמְשִׁיךְ לִקְרֹא.
💛 אַהֲבַת הַשֵּׁם: הָרַעֲשָׁן הוּא כְּלִי לְשִׂמְחָה, אֲבָל הָעִקָּר הוּא הַמְּגִלָּה. מִי שֶׁשּׁוֹמֵר עַל שֶׁקֶט כְּשֶׁצָּרִיךְ – עוֹזֵר לְכָל הַקָּהָל לְקַיֵּם אֶת הַמִּצְוָה.
📖 פֶּרֶק ט״ו, הֲלָכָה ח׳

📜מְגִלָּה כְּשֵׁרָה

מְגִלַּת אֶסְתֵּר הִיא אֶחָד מִסִּפְרֵי הַקֹּדֶשׁ, וְכוֹתְבִים אוֹתָהּ כְּמוֹ סֵפֶר תּוֹרָה: בְּיָד, בִּדְיוֹ שָׁחֹר עַל קְלָף. מְגִלָּה שֶׁנִּכְתְּבָה עַל נְיָר – פְּסוּלָה. מִי שֶׁאָמַר אֶת הַמְּגִלָּה בְּעַל פֶּה אוֹ קְרָאָהּ מִסֵּפֶר מֻדְפָּס – לֹא יָצָא.
הַמְּגִלָּה נִקְרֵאת גַּם "אִגֶּרֶת". לָכֵן לִפְנֵי הַקְּרִיאָה בַּצִּבּוּר פּוֹשְׁטִים אוֹתָהּ וְכוֹפְלִים דַּף עַל דַּף, כְּמוֹ מִכְתָּב – כְּדֵי לְפַרְסֵם אֶת הַנֵּס. אַחֲרֵי הַקְּרִיאָה גּוֹלְלִים אוֹתָהּ בַּחֲזָרָה.
💛 אַהֲבַת הַשֵּׁם: הַסּוֹפֵר כּוֹתֵב כָּל אוֹת בִּקְדֻשָּׁה וּבְדִיּוּק. כְּשֶׁאָנוּ שׁוֹמְעִים מִתּוֹךְ מְגִלָּה כְּשֵׁרָה, אָנוּ מִתְחַבְּרִים לְאוֹתוֹ סִפּוּר שֶׁכָּתְבוּ הַחֲכָמִים בְּרוּחַ הַקֹּדֶשׁ.
📖 פֶּרֶק ט״ו, הֲלָכוֹת ט׳–י״א

✉️מְגִלָּה שֶׁחֲסֵרוֹת בָּהּ אוֹתִיּוֹת

מְדַקְדְּקִים בִּכְתִיבַת הַמְּגִלָּה כְּמוֹ בְּסֵפֶר תּוֹרָה. אֲבָל יֵשׁ הֶבְדֵּל: בְּסֵפֶר תּוֹרָה שֶׁחֲסֵרָה בּוֹ אוֹת אַחַת – אֵין מְבָרְכִים. וּבַמְּגִלָּה, כָּל עוֹד עִקָּרָהּ כָּתוּב – אֶפְשָׁר לִקְרֹא בָּהּ בִּבְרָכָה, וּמַשְׁלִימִים אֶת הַחָסֵר מִסֵּפֶר אוֹ בְּעַל פֶּה. הַטַּעַם: הַמְּגִלָּה נִקְרֵאת "אִגֶּרֶת".
גַּם מִי שֶׁקָּרָא אֶת רֹב הַמְּגִלָּה מִתּוֹךְ מְגִלָּה כְּשֵׁרָה, וְחֵלֶק קָטָן אָמַר מִסֵּפֶר מֻדְפָּס אוֹ בְּעַל פֶּה – יָצָא. בִּתְנַאי שֶׁלֹּא הֶחְסִיר אַף מִלָּה.
💛 אַהֲבַת הַשֵּׁם: הָעִקָּר הוּא לְסַפֵּר אֶת הַסִּפּוּר שֶׁל הַנֵּס – אֵיךְ ה׳ הִצִּיל אֶת עַמּוֹ. כָּל מִלָּה מִן הַסִּפּוּר הַזֶּה מְחַזֶּקֶת אֶת הָאֱמוּנָה.
📖 פֶּרֶק ט״ו, הֲלָכוֹת ט׳–י׳

👂כָּל מִלָּה חֲשׁוּבָה

מִי שֶׁלֹּא שָׁמַע אֲפִלּוּ מִלָּה אַחַת – לֹא יָצָא. זוֹ הַהֲלָכָה הַחֲשׁוּבָה בְּיוֹתֵר בְּהִלְכוֹת הַמְּגִלָּה, כִּי בִּגְלַל הָרַעַשׁ יֵשׁ שֶׁמְּפַסְפְּסִים מִלִּים. הַפִּתְרוֹן: מִי שֶׁהֶחְסִיר מִלָּה – יַשְׁלִים אוֹתָהּ מִיָּד מִן הַסֵּפֶר שֶׁלְּפָנָיו, וְיַמְשִׁיךְ לִקְרֹא עַד שֶׁיַּגִּיעַ לַמָּקוֹם שֶׁל הַקּוֹרֵא.
מִי שֶׁשּׁוֹמֵעַ מִן הַקּוֹרֵא לֹא יִקְרָא יַחַד אִתּוֹ, כְּדֵי שֶׁלֹּא יַפְרִיעַ לְעַצְמוֹ וְלַשְּׁכֵנִים לִשְׁמֹעַ.
💛 אַהֲבַת הַשֵּׁם: כְּמוֹ שֶׁכָּל אוֹת בַּמְּגִלָּה חֲשׁוּבָה, כָּךְ כָּל יְהוּדִי חָשׁוּב לַה׳. מִי שֶׁמַּקְשִׁיב לְכָל מִלָּה – מַרְאֶה כַּמָּה הוּא מוֹקִיר אֶת דְּבַר ה׳.
📖 פֶּרֶק ט״ו, הֲלָכָה י׳

🙏בִּרְכוֹת הַמְּגִלָּה

לִפְנֵי קְרִיאַת הַמְּגִלָּה בַּלַּיְלָה מְבָרְכִים שָׁלֹשׁ בְּרָכוֹת: "עַל מִקְרָא מְגִלָּה", "שֶׁעָשָׂה נִסִּים לַאֲבוֹתֵינוּ" וְ"שֶׁהֶחֱיָנוּ". הַבְּרָכוֹת עוֹזְרוֹת לְכַוֵּן אֶת הַלֵּב לַמִּצְוָה. אַחֲרֵי הַקְּרִיאָה וְהַגְּלִילָה מְבָרְכִים "הָרָב אֶת רִיבֵנוּ" – אֲבָל רַק כְּשֶׁיֵּשׁ מִנְיָן.
בִּשְׁעַת הַבְּרָכוֹת נוֹהֲגִים רֹב יִשְׂרָאֵל לַעֲמֹד. אֶת הַמְּגִלָּה עַצְמָהּ אֶפְשָׁר לִשְׁמֹעַ בִּישִׁיבָה, וְרַק הַחַזָּן עוֹמֵד – מִפְּנֵי כְּבוֹד הַצִּבּוּר.
💛 אַהֲבַת הַשֵּׁם: "שֶׁעָשָׂה נִסִּים לַאֲבוֹתֵינוּ בַּיָּמִים הָהֵם בַּזְּמַן הַזֶּה" – ה׳ שֶׁהִצִּיל אָז, מַצִּיל וְשׁוֹמֵר עָלֵינוּ גַּם הַיּוֹם.
📖 פֶּרֶק ט״ו, הֲלָכוֹת י״א–י״ב

📖כַּסֵּדֶר וּבְרִיכּוּז

קוֹרְאִים אֶת הַמְּגִלָּה כַּסֵּדֶר. מִי שֶׁקָּרָא שֶׁלֹּא כַּסֵּדֶר – "לְמַפְרֵעַ" – לֹא יָצָא. לָכֵן מִי שֶׁהֶחְסִיר מִלָּה לֹא יַשְׁלִים אוֹתָהּ בַּסּוֹף, אֶלָּא מִיָּד. וּמִי שֶׁשָּׁמַע כְּשֶׁהוּא מִתְנַמְנֵם – לֹא יָצָא, כִּי בְּוַדַּאי הָיוּ מִלִּים שֶׁלֹּא שָׁמַע.
בִּשְׁמִיעָה דֶּרֶךְ מַכְשִׁירִים חַשְׁמַלִּיִּים – שִׁדּוּר חַי, טֶלֶפוֹן אוֹ רַמְקוֹל – נוֹהֲגִים לְהַחְמִיר, וְרַק כְּשֶׁאֵין בְּרֵרָה אַחֶרֶת סוֹמְכִים עַל הַמְּקִלִּים. גַּם מִי שֶׁאֵינוֹ מֵבִין עִבְרִית יוֹצֵא בִּשְׁמִיעָה, אִם הוּא מְכַוֵּן לַמִּצְוָה.
💛 אַהֲבַת הַשֵּׁם: הַמְּגִלָּה הִיא סִפּוּר שֶׁל הַשְׁגָּחָה – כָּל דָּבָר בָּא בִּזְמַנּוֹ וּבְסִדְרוֹ. כְּשֶׁאָנוּ שׁוֹמְעִים כַּסֵּדֶר, אָנוּ רוֹאִים אֵיךְ ה׳ סִדֵּר אֶת הַכֹּל לְטוֹבָה.

🔀"שֶׁהֶחֱיָנוּ" בַּיּוֹם – שְׁנֵי מִנְהָגִים

לְמִנְהַג הַסְּפָרַדִּים אֵין מְבָרְכִים "שֶׁהֶחֱיָנוּ" לִפְנֵי קְרִיאַת הַיּוֹם, כִּי הַ"שֶּׁהֶחֱיָנוּ" שֶׁל הַלַּיְלָה מוֹעִיל לְכָל הַיּוֹם. לְמִנְהַג הָאַשְׁכְּנַזִּים חוֹזְרִים וּמְבָרְכִים "שֶׁהֶחֱיָנוּ" גַּם בַּיּוֹם, כִּי קְרִיאַת הַיּוֹם הִיא מִצְוָה בִּפְנֵי עַצְמָהּ – וְהִיא הַקְּרִיאָה הַחֲשׁוּבָה.
בְּ"שֶׁהֶחֱיָנוּ" נָכוֹן לְכַוֵּן גַּם עַל שְׁאָר מִצְווֹת הַיּוֹם: מִשְׁלוֹחַ מָנוֹת, מַתָּנוֹת לָאֶבְיוֹנִים וּסְעוּדָה. הַסְּפָרַדִּים מְכַוְּנִים לְכָךְ בַּלַּיְלָה, וְהָאַשְׁכְּנַזִּים בַּיּוֹם.
💛 אַהֲבַת הַשֵּׁם: שְׁנֵי הַמִּנְהָגִים כַּהֲלָכָה, וְכָל אֶחָד שׁוֹמֵר עַל מִנְהַג מִשְׁפַּחְתּוֹ בְּאַהֲבָה. כֻּלָּנוּ מוֹדִים לְאוֹתוֹ ה׳ שֶׁהֶחֱיָנוּ וְקִיְּמָנוּ.

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

בַּלַּיְלָה כֻּלָּם מְבָרְכִים "שֶׁהֶחֱיָנוּ" לִפְנֵי הַמְּגִלָּה. בַּבֹּקֶר – לְמִנְהַג הַסְּפָרַדִּים לֹא מְבָרְכִים שׁוּב, כִּי בִּרְכַּת הַלַּיְלָה מוֹעִילָה לְכָל הַיּוֹם. לְמִנְהַג הָאַשְׁכְּנַזִּים מְבָרְכִים שׁוּב, כִּי קְרִיאַת הַיּוֹם הִיא מִצְוָה בִּפְנֵי עַצְמָהּ. נִמְצָא שֶׁהַסְּפָרַדִּים מְבָרְכִים "שֶׁהֶחֱיָנוּ" פַּעַם אַחַת, וְהָאַשְׁכְּנַזִּים – פַּעֲמַיִם.
בְּ"שֶׁהֶחֱיָנוּ" מְכַוְּנִים גַּם עַל מִשְׁלוֹחַ מָנוֹת, מַתָּנוֹת לָאֶבְיוֹנִים וּסְעוּדָה: הַסְּפָרַדִּים – בַּלַּיְלָה, וְהָאַשְׁכְּנַזִּים – בַּיּוֹם.
💛 אַהֲבַת הַשֵּׁם: לֹא מְשַׁכְנְעִים חָבֵר לְשַׁנּוֹת אֶת מִנְהַג מִשְׁפַּחְתּוֹ. כָּל אֶחָד שׁוֹמֵר עַל מִנְהָגוֹ בְּאַהֲבָה, וְכֻלָּנוּ שְׂמֵחִים יַחַד בְּאוֹתוֹ נֵס.
📖 פֶּרֶק ט״ו, הֲלָכָה י״ב

🔟עֲשֶׂרֶת בְּנֵי הָמָן

אֶת שְׁמוֹת עֲשֶׂרֶת בְּנֵי הָמָן כּוֹתְבִים בַּמְּגִלָּה בְּצוּרַת שִׁירָה מְיֻחֶדֶת, וְקוֹרְאִים אוֹתָם בִּנְשִׁימָה אַחַת – כִּי כֻּלָּם מֵתוּ יַחַד. וְאַחֲרֵי הַקְּרִיאָה אוֹמְרִים: "אָרוּר הָמָן, בָּרוּךְ מָרְדֳּכַי, אֲרוּרָה זֶרֶשׁ, בְּרוּכָה אֶסְתֵּר, אֲרוּרִים כָּל הָרְשָׁעִים, בְּרוּכִים כָּל הַצַּדִּיקִים, וְגַם חַרְבוֹנָא זָכוּר לַטּוֹב".
הַמִּנְהָג לְהַכּוֹת אֶת הָמָן מְבַטֵּא שִׂנְאָה לָרְשָׁעִים וְשִׂמְחָה עַל מַפַּלְתָּם. מֻתָּר לְהַמְשִׁיךְ בּוֹ – בִּתְנַאי שֶׁכָּל הַצִּבּוּר יוּכַל לִשְׁמֹעַ הֵיטֵב אֶת כָּל הַמְּגִלָּה.
💛 אַהֲבַת הַשֵּׁם: עַם יִשְׂרָאֵל מַאֲמִין בַּה׳ אֶחָד. כְּשֶׁהָרְשָׁעִים שֶׁקָּמוּ עָלֵינוּ נֶעֱנָשִׁים, מִתְגַּלֶּה לְכָל הָעוֹלָם שֶׁה׳ הוּא אֶחָד וְהוּא שׁוֹמֵר עַל עַמּוֹ.
📖 פֶּרֶק ט״ו, הֲלָכָה י״ג

🎉"עַל הַנִּסִּים" וּבִגְדֵי שַׁבָּת

בְּפוּרִים מוֹסִיפִים "עַל הַנִּסִּים" בִּבְרָכַת הַהוֹדָאָה שֶׁבַּתְּפִלָּה וּבְבִרְכַּת "נוֹדֶה לְּךָ" שֶׁבְּבִרְכַּת הַמָּזוֹן. בִּשְׁנֵי יְמֵי הַפּוּרִים אֵין אוֹמְרִים תַּחֲנוּן, אֵין מַסְפִּידִים וְאֵין מִתְעַנִּים.
נוֹהֲגִים לִלְבֹּשׁ בְּלֵיל פּוּרִים וּבַיּוֹם בִּגְדֵי שַׁבָּת וְיוֹם טוֹב.
💛 אַהֲבַת הַשֵּׁם: "עַל הַנִּסִּים" הִיא תּוֹדָה לַה׳ עַל הַיְשׁוּעָה. כְּשֶׁאָנוּ אוֹמְרִים תּוֹדָה, אָנוּ זוֹכְרִים שֶׁכָּל הַטּוֹב שֶׁיֵּשׁ לָנוּ – מַתָּנָה מִמֶּנּוּ.
📖 פֶּרֶק ט״ו, הֲלָכָה י״ד

📜"עַל הַנִּסִּים", קְרִיאַת הַתּוֹרָה וְהַלֵּל

מִי שֶׁשָּׁכַח "עַל הַנִּסִּים" – יָצָא. אִם נִזְכַּר לִפְנֵי שֶׁאָמַר אֶת שֵׁם ה׳ בְּסוֹף הַבְּרָכָה – יַחֲזֹר. וְאִם כְּבָר אָמַר – לֹא יַחֲזֹר, וְטוֹב שֶׁיֹּאמַר אוֹתוֹ בְּסוֹף הַתְּפִלָּה, אוֹ בְּבִרְכַּת הַמָּזוֹן יַחַד עִם "הָרַחֲמָן". בַּתּוֹרָה קוֹרְאִים שְׁלוֹשָׁה עוֹלִים בְּפָרָשַׁת "וַיָּבֹא עֲמָלֵק" – תִּשְׁעָה פְּסוּקִים, כִּי בָּהֶם נִגְמָר הָעִנְיָן. לְפִי הַשֻּׁלְחָן עָרוּךְ יֵשׁ נוֹהֲגִים לִקְרֹא אֶת הַפָּסוּק הָאַחֲרוֹן פַּעֲמַיִם, כְּדֵי לְהַשְׁלִים לַעֲשָׂרָה. וּלְפִי הָרַמָּ״א אֵין קוֹרְאִים אוֹתוֹ פַּעֲמַיִם.
לָמָּה אֵין אוֹמְרִים הַלֵּל בְּפוּרִים? נֶאֶמְרוּ שְׁלוֹשָׁה טְעָמִים: הַנֵּס הָיָה בְּחוּץ לָאָרֶץ; עֲדַיִן נִשְׁאַרְנוּ מְשֻׁעְבָּדִים לַאֲחַשְׁוֵרוֹשׁ; וּקְרִיאַת הַמְּגִלָּה הִיא בִּמְקוֹם הַהַלֵּל.
💛 אַהֲבַת הַשֵּׁם: רֶמֶז יֵשׁ בַּתִּשְׁעָה פְּסוּקִים: כָּל עוֹד לֹא נִמְחָה שְׁמוֹ שֶׁל עֲמָלֵק, שְׁמוֹ שֶׁל ה׳ עֲדַיִן אֵינוֹ מִתְגַּלֶּה בִּשְׁלֵמוּת. אָנוּ מִתְפַּלְּלִים לַיּוֹם שֶׁבּוֹ יִתְגַּלֶּה לְכֻלָּם.
📖 פֶּרֶק ט״ו, הֲלָכָה י״ד

🛠️מְלָאכָה בְּפוּרִים

פּוּרִים לֹא נִקְבַּע כְּיוֹם טוֹב שֶׁאָסוּר בִּמְלָאכָה, אֲבָל לִכְבוֹד הַיּוֹם נָהֲגוּ יִשְׂרָאֵל שֶׁלֹּא לַעֲשׂוֹת בּוֹ מְלָאכָה – וְהַמִּנְהָג מְחַיֵּב. אֵין יוֹצְאִים לָעֲבוֹדָה הַקְּבוּעָה, חוּץ מִמָּקוֹם שֶׁל הֶפְסֵד גָּדוֹל, אוֹ עָנִי שֶׁאֵין לוֹ מָה לֶאֱכֹל.
מֻתָּר: מְלָאכָה מְשַׂמַּחַת (כְּמוֹ לִשְׁתֹּל עֵץ נוֹי בַּגִּנָּה), מְלָאכָה שֶׁל מִצְוָה (כְּמוֹ לִכְתֹּב חִדּוּשֵׁי תּוֹרָה), וּמְלָאכָה קַלָּה (כְּמוֹ מִכְתָּב). כִּבּוּס בַּיָּד, תְּפִירָה וְתִסְפֹּרֶת – רַק לְצֹרֶךְ פּוּרִים. וּבְלֵיל פּוּרִים? יֵשׁ מֵהָאַחֲרוֹנִים שֶׁהֶחְמִירוּ לֶאֱסֹר מְלָאכָה גַּם בַּלַּיְלָה. הָרַב מֶלָמֵד כּוֹתֵב שֶׁכֵּיוָן שֶׁנֶּחְלְקוּ, אֵין מִנְהָג מֻסְכָּם, וְלָכֵן בַּלַּיְלָה אֵין אִסּוּר.
💛 אַהֲבַת הַשֵּׁם: אָמְרוּ: "כָּל הָעוֹשֶׂה מְלָאכָה בְּפוּרִים – אֵינוֹ רוֹאֶה מִמֶּנָּה סִימַן בְּרָכָה". כְּשֶׁמְּפַנִּים אֶת הַיּוֹם לַשִּׂמְחָה וְלַמִּצְווֹת, הַבְּרָכָה בָּאָה מֵה׳.
📖 פֶּרֶק ט״ו, הֲלָכָה ט״ו

🧠 בְּדֹק אֶת עַצְמְךָ

לָמָּה קַבָּלַת הַתּוֹרָה בִּימֵי אֲחַשְׁוֵרוֹשׁ גְּדוֹלָה בְּמִדָּה מְסֻיֶּמֶת מִזּוֹ שֶׁבְּסִינַי?
💡 בְּסִינַי, אַחֲרֵי כָּל הַנִּסִּים, כִּמְעַט לֹא הָיְתָה בְּרֵרָה. בְּפוּרִים, בְּלִי נִסִּים גְּלוּיִים וּבְלִי כְּפִיָּה, בָּחֲרוּ יִשְׂרָאֵל לְהִשָּׁאֵר דְּבוּקִים בַּה׳ – גַּם כְּשֶׁהַמְּחִיר הָיָה כָּבֵד.
עַד מָתַי אֶפְשָׁר לִקְרֹא אֶת הַמְּגִלָּה שֶׁל לַיְלָה?
💡 קְרִיאַת הַלַּיְלָה זְמַנָּהּ מִצֵּאת הַכּוֹכָבִים עַד עֲלוֹת הַשַּׁחַר. הַזְּרִיזִים קוֹרְאִים מִיָּד אַחֲרֵי עַרְבִית – אֲבָל זֶה רַק זְרִיזוּת, וְלֹא סוֹף הַזְּמַן.
יוֹסִי קָרָא אֶת רֹב הַמְּגִלָּה מִמְּגִלָּה כְּשֵׁרָה, וּפָסוּק אֶחָד אָמַר בְּעַל פֶּה. הַאִם יָצָא?
💡 מִי שֶׁקָּרָא אֶת עִקַּר הַמְּגִלָּה מִתּוֹךְ מְגִלָּה כְּשֵׁרָה, וְחֵלֶק אָמַר בְּעַל פֶּה – יָצָא, בִּתְנַאי שֶׁלֹּא הֶחְסִיר מִלָּה. אֲבָל כֻּלָּהּ מִסֵּפֶר מֻדְפָּס – לֹא יָצָא.

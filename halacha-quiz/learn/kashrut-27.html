הַגְּזֵרוֹת עַל מַאַכְלֵי גּוֹיִם

🍽️ הִלְכוֹת כַּשְׁרוּת · פֶּרֶק כ״ז

🧱הַבְדָּלָה – לֹא הַפְרָדָה

מִצְוָה לֶאֱהֹב כָּל אָדָם, כְּמוֹ אַבְרָהָם אָבִינוּ שֶׁפָּתַח אֶת אָהֳלוֹ לְכָל עוֹבְרֵי הַדְּרָכִים. אֲבָל כְּדֵי שֶׁעַם יִשְׂרָאֵל יָבִיא בְּרָכָה לָעוֹלָם, הוּא צָרִיךְ לִשְׁמֹר עַל יִחוּדוֹ. וּמִפְּנֵי שֶׁהָאֹכֶל מְקָרֵב לְבָבוֹת, גָּזְרוּ חֲכָמִים שֶׁלֹּא נֹאכַל יַיִן, פַּת וְתַבְשִׁילִים שֶׁל גּוֹיִם – גַּם כְּשֶׁהֵם עֲשׂוּיִים מִמּוּצָרִים כְּשֵׁרִים.
הַגְּזֵרוֹת נוֹעֲדוּ לִשְׁמֹר שֶׁלֹּא נִתְבּוֹלֵל בָּעַמִּים וְנִשְׁכַּח אֶת יִעוּדֵנוּ. לֹא מִשּׁוּם שֶׁאוֹכֵל אֶחָד יַזִּיק מִיָּד, אֶלָּא כְּדֵי שֶׁהַגְּבוּל בֵּינֵינוּ לְבֵין הָעַמִּים לֹא יִטַּשְׁטֵשׁ לְאַט לְאַט.
💛 אַהֲבַת הַשֵּׁם: הָרַב צְבִי יְהוּדָה הָיָה חוֹזֵר וְאוֹמֵר: "הַבְדָּלָה אֵינָהּ הַפְרָדָה". אֶפְשָׁר לִשְׁמֹר עַל הַיִּחוּד שֶׁלָּנוּ – וּבְכָל זֹאת לְכַבֵּד וְלֶאֱהֹב כָּל אָדָם.
📖 פֶּרֶק כ״ז, הֲלָכוֹת א׳, ז׳

💛"וִהְיִיתֶם לִי קְדֹשִׁים"

כָּל אִסּוּרֵי הַמַּאֲכָלִים נוֹעֲדוּ לְהַבְדִּיל אֶת יִשְׂרָאֵל וּלְקַדֵּשׁ אוֹתָם: "וִהְיִיתֶם לִי קְדֹשִׁים כִּי קָדוֹשׁ אֲנִי ה׳, וָאַבְדִּל אֶתְכֶם מִן הָעַמִּים לִהְיוֹת לִי". הַגְּזֵרוֹת עַל מַאַכְלֵי גּוֹיִם חָלוֹת עַל כָּל הַגּוֹיִם, כִּי מַטְּרָתָן לִמְנֹעַ קֵרוּב יָתֵר שֶׁיּוֹבִיל לְאַט לְאַט לְהִתְבּוֹלְלוּת.
עַל יְהוּדִי שֶׁאֵינוֹ שׁוֹמֵר תּוֹרָה וּמִצְווֹת – אֵין גְּזֵרַת פַּת וּבִשּׁוּלֵי גּוֹיִם, כִּי אֵין בּוֹ חֲשַׁשׁ הִתְבּוֹלְלוּת. אֲבָל צָרִיךְ לִבְדֹּק הֵיטֵב שֶׁמָּה שֶׁהֵכִין כָּשֵׁר.
💛 אַהֲבַת הַשֵּׁם: כָּל אֲרוּחָה כְּשֵׁרָה מַזְכִּירָה לָנוּ: ה׳ בָּחַר בָּנוּ לִהְיוֹת עַם קָדוֹשׁ. זוֹ לֹא מַעֲמָסָה – זוֹ זְכוּת.
📖 פֶּרֶק כ״ז, הֲלָכוֹת א׳, ז׳

👑דָּנִיֵּאל וַחֲבֵרָיו

נְבוּכַדְנֶצַּר מֶלֶךְ בָּבֶל לָקַח בַּשְּׁבִי נְעָרִים מִירוּשָׁלַיִם – דָּנִיֵּאל, חֲנַנְיָה, מִישָׁאֵל וַעֲזַרְיָה – כְּדֵי לְגַדְּלָם בְּאַרְמוֹנוֹ כְּמוֹ בַּבְלִים. אֲבָל דָּנִיֵּאל "שָׂם עַל לִבּוֹ אֲשֶׁר לֹא יִתְגָּאַל בְּפַתְבַּג הַמֶּלֶךְ וּבְיֵין מִשְׁתָּיו". הוּא וַחֲבֵרָיו לֹא אָכְלוּ מִפִּתָּם, מִיֵּינָם וּמִשַּׁמְנָם, וְהִסְתַּפְּקוּ בְּזֵרְעוֹנִים וּבְקִטְנִיּוֹת.
וּבְחַסְדֵי ה׳, דָּנִיֵּאל וַחֲבֵרָיו נִרְאוּ טוֹבִים וּבְרִיאִים מִכָּל הַנְּעָרִים שֶׁאָכְלוּ מִמַּאַכְלֵי הַמֶּלֶךְ.
💛 אַהֲבַת הַשֵּׁם: בִּזְכוּת שֶׁשָּׁמְרוּ עַל מַאַכְלֵיהֶם, עָמְדוּ דָּנִיֵּאל וַחֲבֵרָיו בַּנִּסָּיוֹן וְשָׁמְרוּ תּוֹרָה וּמִצְווֹת בְּאַרְמוֹן הַמֶּלֶךְ. הֵם דֻּגְמָה לְכָל יֶלֶד שֶׁשּׁוֹמֵר עַל זֶהוּתוֹ גַּם כְּשֶׁקָּשֶׁה.
📖 פֶּרֶק כ״ז, הֲלָכָה ג׳

🫒פַּת, יַיִן, שֶׁמֶן וְתַבְשִׁיל

בִּימֵי בֵּית שַׁמַּאי וּבֵית הִלֵּל גָּזְרוּ חֲכָמִים עַל פַּת, יַיִן וְשֶׁמֶן שֶׁל גּוֹיִם, וּבִגְזֵרָה נוֹסֶפֶת – גַּם עַל בִּשּׁוּלֵי גּוֹיִם. אֶת גְּזֵרַת הַשֶּׁמֶן רֹב יִשְׂרָאֵל לֹא יָכְלוּ לְקַיֵּם, וְלָכֵן רַבִּי יְהוּדָה נְשִׂיאָה וּבֵית דִּינוֹ הִתִּירוּ אוֹתוֹ – כִּי "אֵין גּוֹזְרִים גְּזֵרָה עַל הַצִּבּוּר אֶלָּא אִם כֵּן רֹב הַצִּבּוּר יְכוֹלִים לַעֲמֹד בָּהּ".
בַּפַּת הֵקֵלּוּ בִּשְׁעַת הַדְּחָק בְּ"פַת פַּלְטֵר" – פַּת שֶׁאוֹפֶה גּוֹי מוֹכֵר לָרַבִּים. וּבַיַּיִן, שֶׁמְּקָרֵב לְבָבוֹת יוֹתֵר מִכֻּלָּם – דַּוְקָא הֶחְמִירוּ עוֹד.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים הָיוּ קַפְּדָנִים וְגַם רַחֲמָנִים: הֵם שָׁמְרוּ עַל עַם יִשְׂרָאֵל, וְגַם דָּאֲגוּ שֶׁהַגְּזֵרוֹת לֹא יִהְיוּ כָּבֵד מִדַּי עַל הַצִּבּוּר.
📖 פֶּרֶק כ״ז, הֲלָכוֹת ד׳–ו׳

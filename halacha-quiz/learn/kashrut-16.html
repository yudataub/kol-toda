שִׁלּוּחַ הַקֵּן

🍽️ הִלְכוֹת כַּשְׁרוּת · פֶּרֶק ט״ז

🪺שִׁלּוּחַ הַקֵּן

מִי שֶׁמּוֹצֵא קֵן שֶׁל צִפּוֹר כְּשֵׁרָה, וְהָאֵם יוֹשֶׁבֶת עַל הַבֵּיצִים אוֹ עַל הָאֶפְרוֹחִים, וְהוּא רוֹצֶה אוֹתָם – לֹא לוֹקֵחַ אֶת הָאֵם עִם הַבָּנִים. קֹדֶם מְשַׁלְּחִים אֶת הָאֵם, וְרַק אַחַר כָּךְ לוֹקְחִים אֶת הַבֵּיצִים אוֹ הָאֶפְרוֹחִים.
מְשַׁלְּחִים אֶת הָאֵם בְּקוֹל אוֹ בְּמַכָּה קַלָּה עַל הַקֵּן. הַצִּפּוֹרִים הַכְּשֵׁרוֹת שֶׁמְּצוּיוֹת אֶצְלֵנוּ: יוֹנִים, תּוֹרִים וּדְרוֹר הַבַּיִת.
💛 אַהֲבַת הַשֵּׁם: הַצִּפּוֹר מוּכָנָה לְסַכֵּן אֶת עַצְמָהּ כְּדֵי לִשְׁמֹר עַל גּוֹזָלֶיהָ. הַתּוֹרָה מְלַמֶּדֶת אוֹתָנוּ לְכַבֵּד אֶת הָאַהֲבָה הַזֹּאת, וּמַבְטִיחָה: "לְמַעַן יִיטַב לָךְ וְהַאֲרַכְתָּ יָמִים".
📖 פֶּרֶק ט״ז, הֲלָכוֹת א׳, ד׳

🕊️מָתַי מְקַיְּמִים שִׁלּוּחַ הַקֵּן?

הַמִּצְוָה הִיא רַק בְּצִפּוֹר כְּשֵׁרָה שֶׁל בָּר, שֶׁאֵינָהּ שַׁיֶּכֶת לְאָדָם. מִי שֶׁמְּגַדֵּל תַּרְנְגוֹלוֹת אוֹ יוֹנִים – מֻתָּר לוֹ לָקַחַת אֶת הַבֵּיצִים מִתַּחַת לָאֵם, כִּי הַכֹּל שֶׁלּוֹ. וְאֵין מִצְוָה לְחַפֵּשׂ קֵן.
לְפִי עִקַּר הַהֲלָכָה, רַק מִי שֶׁצָּרִיךְ אֶת הַבֵּיצִים אוֹ הָאֶפְרוֹחִים מְשַׁלֵּחַ אֶת הָאֵם וְלוֹקֵחַ. מִי שֶׁאֵינוֹ צָרִיךְ אוֹתָם וְרוֹצֶה לְקַיֵּם גַּם אֶת דַּעַת הַפּוֹסְקִים הָאֲחֵרִים – יַבְרִיחַ אֶת הָאֵם, וְלֹא יִקַּח אֶת הַבֵּיצִים.
💛 אַהֲבַת הַשֵּׁם: הַמִּצְוָה מְלַמֶּדֶת רַחֲמִים, וְלָכֵן לֹא מְצַעֲרִים צִפּוֹר בְּלִי צֹרֶךְ. כְּשֶׁאָנוּ מְרַחֲמִים עַל הַבְּרִיּוֹת – ה׳ מְרַחֵם עָלֵינוּ.
📖 פֶּרֶק ט״ז, הֲלָכוֹת א׳, ג׳

🐦🔑 מוּשָּׂג: שִׁלּוּחַ הַקֵּן

״הַמּוֹצֵא קַן צִפּוֹר... וְהָאֵם רֹבֶצֶת עַל הָאֶפְרֹחִים אוֹ עַל הַבֵּיצִים – לֹא תִקַּח הָאֵם עַל הַבָּנִים, שַׁלֵּחַ תְּשַׁלַּח אֶת הָאֵם, וְאֶת הַבָּנִים תִּקַּח לָךְ״.
📖 שִׁלּוּחַ – שְׁלִיחָה, הַרְחָקָה. קֵן – הַבַּיִת שֶׁל הַצִּפּוֹר. קֹדֶם מַבְרִיחִים אֶת הָאֵם, וְרַק אָז מֻתָּר לָקַחַת.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה חָסָה עַל צַעַר הָאֵם, שֶׁלֹּא תִּרְאֶה אֵיךְ לוֹקְחִים מִמֶּנָּה אֶת יְלָדֶיהָ. 🧒 וְאַתָּה? אִם תִּמְצָא קֵן בַּגִּנָּה – אַל תִּגַּע בּוֹ בִּכְלָל, וְתַנִּיחַ לַצִּפּוֹרִים בְּשָׁלוֹם.
📖 פֶּרֶק ט״ז, הֲלָכָה א׳

🧠 בְּדֹק אֶת עַצְמְךָ

שִׁמְעוֹן מָצָא בַּגִּנָּה הַצִּבּוּרִית קֵן שֶׁל יוֹנַת בָּר. הוּא אֵינוֹ צָרִיךְ אֶת הַבֵּיצִים, אֲבָל רוֹצֶה לָצֵאת יְדֵי כָּל הַדֵּעוֹת. מָה יַעֲשֶׂה?
💡 יֵשׁ אוֹמְרִים שֶׁכְּבָר בְּהַבְרָחַת הָאֵם מְקַיְּמִים אֶת הַמִּצְוָה. וְכֵיוָן שֶׁאֵינוֹ צָרִיךְ אֶת הַבֵּיצִים – לֹא יִקַּח אוֹתָן, כְּדֵי שֶׁלֹּא לְצַעֵר וּלְהַשְׁחִית בְּחִנָּם.
לְסָבָא יֵשׁ לוּל תַּרְנְגוֹלוֹת בֶּחָצֵר, וְתַרְנְגֹלֶת אַחַת דּוֹגֶרֶת עַל בֵּיצִים. מָה הַדִּין?
💡 שִׁלּוּחַ הַקֵּן נוֹהֵג רַק בְּצִפּוֹר בָּר שֶׁאֵינָהּ שַׁיֶּכֶת לְאָדָם. הַתַּרְנְגוֹלוֹת שֶׁל סָבָא וּבֵיצֵיהֶן שֶׁלּוֹ, וּמֻתָּר לוֹ לָקַחַת כְּשֶׁיִּרְצֶה.

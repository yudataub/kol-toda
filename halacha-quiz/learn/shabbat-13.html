מְלָאכוֹת הַבֶּגֶד

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק י״ג

👔הַבֶּגֶד – כְּבוֹד הָאָדָם

אָדָם הָרִאשׁוֹן לֹא הָיָה צָרִיךְ בְּגָדִים, וְאַחֲרֵי הַחֵטְא נִצְרַךְ לָהֶם. הַבֶּגֶד מְכַסֶּה אֶת הַגּוּף, וְכָךְ הַנְּשָׁמָה בָּאָה לִידֵי בִּטּוּי – לָכֵן הַבֶּגֶד הוּא כְּבוֹדוֹ שֶׁל הָאָדָם. שְׁלֹשׁ עֶשְׂרֵה מְלָאכוֹת שֶׁל שַׁבָּת קְשׁוּרוֹת לַהֲכָנַת בְּגָדִים, וְשָׁרְשָׁן בִּירִיעוֹת הַמִּשְׁכָּן.
בֵּין הַמְּלָאכוֹת: מְלַבֵּן (כִּבּוּס), צוֹבֵעַ, טוֹוֶה, אוֹרֵג, קוֹשֵׁר, מַתִּיר, תּוֹפֵר וְקוֹרֵעַ.
💛 אַהֲבַת הַשֵּׁם: בִּגְדֵי שַׁבָּת הֵם דֶּרֶךְ לוֹמַר לַה׳: הַיּוֹם הַזֶּה מְיֻחָד לִי. כְּשֶׁאָנוּ לוֹבְשִׁים אוֹתָם, גַּם הַנְּשָׁמָה מִתְלַבֶּשֶׁת בְּשִׂמְחָה וּבִקְדֻשָּׁה.
📖 פֶּרֶק י״ג, הֲלָכוֹת א׳–ב׳

🧺כִּבּוּס

כִּבּוּס נַעֲשֶׂה בִּשְׁלֹשָׁה שְׁלַבִּים, וְכָל אֶחָד מֵהֶם אָסוּר מִן הַתּוֹרָה: הַשְׁרָיַת הַבֶּגֶד בְּמַיִם, שִׁפְשׁוּף, וּסְחִיטָה. לָכֵן אָסוּר לְהַשְׁרוֹת בֶּגֶד מְלֻכְלָךְ בְּמַיִם, וְאָסוּר לְהָסִיר כֶּתֶם בְּמַיִם אוֹ בְּרֹק. רַבִּים אוֹסְרִים לְהַרְטִיב אֲפִלּוּ בֶּגֶד נָקִי.
אֲבָל לְנַגֵּב יָדַיִם בְּמַגֶּבֶת – מֻתָּר, כִּי הַמַּיִם אֵינָם מְנַקִּים אֶת הַמַּגֶּבֶת, אֶלָּא לְהֶפֶךְ – מְלַכְלְכִים אוֹתָהּ מְעַט.
💛 אַהֲבַת הַשֵּׁם: בִּגְדֵי הַשַּׁבָּת הַנְּקִיִּים מוּכָנִים מִיּוֹם שִׁשִּׁי. כְּשֶׁאָנוּ נִזְהָרִים בָּהֶם בְּשַׁבָּת, אָנוּ מַרְאִים שֶׁכְּבוֹד הַשַּׁבָּת חָשׁוּב לָנוּ יוֹתֵר מִכָּל כֶּתֶם קָטָן.
📖 פֶּרֶק י״ג, הֲלָכוֹת ג׳–ד׳

🧥בֶּגֶד שֶׁנִּרְטַב וּכְתָמִים

בֶּגֶד שֶׁנִּרְטַב בַּגֶּשֶׁם – אָסוּר לִסְחֹט. מֻתָּר לְהַמְשִׁיךְ לִלְבֹּשׁ אוֹתוֹ, אֲבָל לֹא תּוֹלִים אוֹתוֹ עַל חֶבֶל כְּבִיסָה, כְּדֵי שֶׁלֹּא יַחְשְׁבוּ שֶׁכִּבְּסוּ בְּשַׁבָּת – אֶלָּא מַנִּיחִים עַל כִּסֵּא אוֹ קוֹלָב. וְלֹא מַנִּיחִים אוֹתוֹ צָמוּד לְתַנּוּר חִמּוּם, בְּמָקוֹם חַם מְאוֹד.
כֶּתֶם קַל – מֻתָּר לְהָסִיר בְּגֵרוּד אֶחָד בַּצִּפֹּרֶן, וְאַחֲרֵי הַפְסָקָה – עוֹד גֵּרוּד. נוֹצָה אוֹ חוּט שֶׁמֻּנָּחִים עַל הַבֶּגֶד – מֻתָּר לְהָסִיר. אֲבָל לְנַעֵר אֶת הַבֶּגֶד מֵאָבָק – אָסוּר.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים חָשְׁבוּ גַּם עַל מָה שֶּׁאֲחֵרִים יִרְאוּ וְיַחְשְׁבוּ. מִי שֶׁשּׁוֹמֵר שַׁבָּת כָּךְ שֶׁגַּם הָרוֹאִים יִלְמְדוּ מִמֶּנּוּ – מְקַדֵּשׁ שֵׁם שָׁמַיִם.
📖 פֶּרֶק י״ג, הֲלָכוֹת ג׳, ח׳

🧽נִשְׁפַּךְ מַשֶּׁהוּ

מְעַט מַיִם אוֹ מִיץ שֶׁנִּשְׁפְּכוּ עַל הַשֻּׁלְחָן – מֻתָּר לְנַגֵּב בְּמַגֶּבֶת. כְּשֶׁנִּשְׁפְּכוּ מַיִם רַבִּים, אָסוּר לִסְפֹּג אוֹתָם בְּמַגֶּבֶת אַחַת, שֶׁמָּא יִסְחֲטוּ אוֹתָהּ. אֶפְשָׁר לְהִשְׁתַּמֵּשׁ בְּכַמָּה מַגָּבוֹת אוֹ בִּנְיָר סוֹפֵג, וּמִן הָרִצְפָּה אֶפְשָׁר לִגְרֹף אֶת הַמַּיִם בְּמַגָּב.
סְפוֹג (סְקוֹטְשׁ) לִשְׁטִיפַת כֵּלִים – אָסוּר, כִּי סוֹחֲטִים אוֹתוֹ. וְלִשְׁטֹף אֶת הָרִצְפָּה בְּמַיִם – אָסוּר.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁמַּשֶּׁהוּ נִשְׁפָּךְ, אֶפְשָׁר לִשְׁמֹר עַל הַשַּׁבָּת וְעַל מַצַּב הָרוּחַ. מִי שֶׁמְּסַדֵּר בְּנַחַת וּבְדֶרֶךְ הַמֻּתֶּרֶת מְכַבֵּד אֶת הַשַּׁבָּת – וְאֶת כָּל בְּנֵי הַבַּיִת.
📖 פֶּרֶק י״ג, הֲלָכָה ה׳; פֶּרֶק ט״ו, הֲלָכָה ט׳

👟בּוֹץ עַל הַנַּעַל

בּוֹץ שֶׁנִּדְבַּק לַנַּעַל אוֹ לַבֶּגֶד – לֹא מְגָרְדִים וּמְפוֹרְרִים אוֹתוֹ בְּשַׁבָּת, כִּי זֶה כְּמוֹ טוֹחֵן. בִּשְׁעַת הַצֹּרֶךְ מֻתָּר לְהָסִיר אוֹתוֹ בְּשִׁנּוּי: לְהַכּוֹת נַעַל בְּנַעַל, אוֹ לִטְפֹּחַ עַל הַבֶּגֶד בְּגַב הַיָּד.
אָבָק עַל נַעֲלֵי עוֹר מֻתָּר לְהָסִיר בַּיָּד אוֹ בְּמַטְלִית – אֲבָל לֹא מַבְרִיקִים אוֹתָן.
💛 אַהֲבַת הַשֵּׁם: גַּם אַחֲרֵי הֲלִיכָה בַּשְּׁלוּלִיּוֹת – הַשַּׁבָּת מְלַמֶּדֶת אוֹתָנוּ לַעֲצֹר וְלַחְשֹׁב, וְכָךְ כָּל צַעַד שֶׁלָּנוּ נַעֲשֶׂה קָדוֹשׁ.
📖 פֶּרֶק י״ג, הֲלָכָה ו׳

🙏קִפּוּל טַלִּית

חֲכָמִים אָסְרוּ לְקַפֵּל בְּגָדִים בְּשַׁבָּת לְפִי הַקְּפָלִים הַקְּבוּעִים שֶׁלָּהֶם, כִּי זֶה נִרְאֶה כְּתִקּוּן הַבֶּגֶד. לָכֵן, לְדַעַת רֹב הַפּוֹסְקִים, מְקַפְּלִים אֶת הַטַּלִּית אַחֲרֵי הַתְּפִלָּה – אֲבָל לֹא בְּדִיּוּק לְפִי הַקְּפָלִים הַמְגֻהָצִים שֶׁלָּהּ.
כּוֹבַע שֶׁנִּמְעַךְ – מֻתָּר לְיַשֵּׁר. מָטוֹס אוֹ סְפִינָה מִנְּיָר – יְלָדִים רַשָּׁאִים לְהָקֵל, אֲבָל קִפּוּל אָמָּנוּתִי (אוֹרִיגָמִי) – אָסוּר.
💛 אַהֲבַת הַשֵּׁם: הַטַּלִּית עוֹטֶפֶת אוֹתָנוּ בַּתְּפִלָּה כְּמוֹ חִבּוּק. כְּשֶׁמְּקַפְּלִים אוֹתָהּ בְּכָבוֹד גַּם בְּשַׁבָּת, שׁוֹמְרִים גַּם עַל כְּבוֹד הַמִּצְוָה וְגַם עַל כְּבוֹד הַשַּׁבָּת.
📖 פֶּרֶק י״ג, הֲלָכָה ט׳; פֶּרֶק ט״ו, הֲלָכָה ז׳

🧵תּוֹפֵר וְקוֹרֵעַ

מֻתָּר לִסְגֹּר וְלִפְתֹּחַ כַּפְתּוֹרִים, רוֹכְסָן, לַחְצָנִיּוֹת וְסֶקוֹטְשׁ, כִּי הֵם נוֹעֲדוּ לִפְתִיחָה וּסְגִירָה. אֲבָל אָסוּר לְהַדְבִּיק נְיָרוֹת, אָסוּר לְחַבֵּר אוֹתָם בְּשַׁדְּכָן (מְהַדֵּק סִכּוֹת), וְאָסוּר לִקְרֹעַ נְיַר טוּאָלֶט אוֹ שַׂקִּיּוֹת מִגָּלִיל. אֶטֶב מִשְׂרָדִי – מֻתָּר.
לָכֵן מְכִינִים נְיַר טוּאָלֶט קָרוּעַ לִפְנֵי שַׁבָּת. אֲבָל שַׂקִּית שֶׁיֵּשׁ בָּהּ אֹכֶל – מֻתָּר לִקְרֹעַ, כְּמוֹ שֶׁמְּקַלְּפִים תַּפּוּז, וּמִשְׁתַּדְּלִים שֶׁלֹּא לִקְרֹעַ בִּמְקוֹם הָאוֹתִיּוֹת.
💛 אַהֲבַת הַשֵּׁם: הַהֲכָנָה בְּיוֹם שִׁשִּׁי – לִקְרֹעַ נְיָר, לְסַדֵּר וּלְהָכִין – הִיא כְּבָר חֵלֶק מִן הַשַּׁבָּת. מִי שֶׁמֵּכִין מֵרֹאשׁ מַרְאֶה שֶׁהוּא מְחַכֶּה לַשַּׁבָּת כְּמוֹ לְאוֹרֵחַ אָהוּב.
📖 פֶּרֶק י״ג, הֲלָכוֹת י׳–י״א

🩹פְּלַסְטֶר בְּשַׁבָּת

מֻתָּר לְהַדְבִּיק פְּלַסְטֶר בְּשַׁבָּת כְּשֶׁצָּרִיךְ, כִּי הַדְבָּקָה עַל הַגּוּף אֵינָהּ נֶחְשֶׁבֶת תְּפִירָה. וּבִשְׁעַת הַצֹּרֶךְ מֻתָּר גַּם לְהַדְבִּיק אוֹתוֹ עַל תַּחְבֹּשֶׁת.
מֻתָּר גַּם לְהִשְׁתַּמֵּשׁ בְּדַבְקִיַּת סִימָנִיָּה, שֶׁנִּדְבֶּקֶת וְנִפְתַּחַת שׁוּב וָשׁוּב – כְּמוֹ כַּפְתּוֹר.
💛 אַהֲבַת הַשֵּׁם: הַשַּׁבָּת הִיא יוֹם שֶׁל עֹנֶג, וְה׳ לֹא רוֹצֶה שֶׁנִּסְבֹּל. הַהֲלָכָה דּוֹאֶגֶת גַּם לַפֶּצַע הַקָּטָן שֶׁלָּנוּ.
📖 פֶּרֶק י״ג, הֲלָכָה י״ב

🎀קְשָׁרִים

קֶשֶׁר חָזָק וְקָבוּעַ, כְּמוֹ קֶשֶׁר הַתְּפִלִּין וְהַצִּיצִית – אָסוּר מִן הַתּוֹרָה, וְנִזְהָרִים שֶׁלֹּא לְהַדֵּק בְּשַׁבָּת אֶת קֶשֶׁר הַצִּיצִית. גַּם קֶשֶׁר כָּפוּל מְהֻדָּק – אָסוּר. אֲבָל עֲנִיבָה ("פַּרְפַּר") וְקֶשֶׁר אֶחָד – אֵינָם נֶחְשָׁבִים קֶשֶׁר, וּמֻתָּר לִקְשֹׁר אוֹתָם.
שַׂקִּית אֹכֶל קוֹשְׁרִים בַּעֲנִיבָה, אוֹ בְּקֶשֶׁר רָגִיל שֶׁל שְׁנֵי צִדֵּי הַשַּׂקִּית – וְלֹא בְּקֶשֶׁר אֶחָד מְהֻדָּק בַּקָּצֶה.
💛 אַהֲבַת הַשֵּׁם: הַקֶּשֶׁר הֶחָזָק בָּעוֹלָם הוּא הַקֶּשֶׁר שֶׁל עַם יִשְׂרָאֵל עִם ה׳. גַּם אֶת קִשְׁרֵי הַצִּיצִית אָנוּ שׁוֹמְרִים, וְהֵם מַזְכִּירִים לָנוּ אֶת כָּל הַמִּצְווֹת.
📖 פֶּרֶק י״ג, הֲלָכוֹת י״ג–ט״ו

👟שְׂרוֹכִים, חֲגוֹרָה וְצִפִּית

מֻתָּר לְהַכְנִיס חֲגוֹרָה לְמִכְנָסַיִם חֲדָשִׁים, כִּי הִיא לֹא נִשְׁאֶרֶת שָׁם תָּמִיד. וּמֻתָּר לְהַכְנִיס כַּר חָדָשׁ לְצִפִּית. אֲבָל שְׂרוֹךְ אוֹ רְצוּעָה שֶׁנִּשְׁאָרִים בַּבֶּגֶד תָּמִיד – אָסוּר לְהַכְנִיס, כִּי כָּךְ מַתְקִינִים אֶת הַבֶּגֶד לְשִׁמּוּשׁ.
שְׂרוֹךְ בְּנַעַל חֲדָשָׁה – כְּדֵי לָצֵאת יְדֵי כָּל הַדֵּעוֹת, מַשְׁחִילִים אוֹתוֹ בְּשַׁבָּת בְּצוּרָה מְשֻׁנָּה: בְּדִלּוּג עַל חֵלֶק מִן הַחוֹרִים, אוֹ רַק בַּחוֹרִים הָעֶלְיוֹנִים. כָּךְ בְּמוֹצָאֵי שַׁבָּת עוֹד צָרִיךְ לְהַשְׁחִיל אוֹתוֹ כָּרָאוּי.
💛 אַהֲבַת הַשֵּׁם: בַּשַּׁבָּת לֹא מְתַקְּנִים וְלֹא מַשְׁלִימִים כֵּלִים – אֶלָּא נָחִים וּמִשְׁתַּמְּשִׁים בְּמָה שֶׁהֵכַנּוּ. כָּךְ אָנוּ זוֹכְרִים שֶׁה׳ הוּא שֶׁהִשְׁלִים אֶת הָעוֹלָם.
📖 פֶּרֶק י״ג, הֲלָכָה ט״ז

🧠 בְּדֹק אֶת עַצְמְךָ

הַמְּעִיל שֶׁל אֵיתָן נִרְטַב בַּגֶּשֶׁם בַּדֶּרֶךְ לְבֵית הַכְּנֶסֶת. מָה יַעֲשֶׂה כְּשֶׁיַּגִּיעַ הַבַּיְתָה?
💡 סְחִיטָה הִיא חֵלֶק מִן הַכִּבּוּס. עַל חֶבֶל כְּבִיסָה לֹא תּוֹלִים, כְּדֵי שֶׁלֹּא יַחְשְׁבוּ שֶׁכִּבֵּס בְּשַׁבָּת. וְיִבּוּשׁ צָמוּד לְתַנּוּר, בְּחֹם גָּדוֹל, הוּא לִבּוּן הַבֶּגֶד – וְאָסוּר.
אַחֲרֵי תְּפִלַּת שַׁחֲרִית בְּשַׁבָּת רוֹצֶה אַבָּא לְקַפֵּל אֶת טַלִּיתוֹ. מָה הַדֶּרֶךְ שֶׁרֹב הַפּוֹסְקִים מַתִּירִים?
💡 הָאִסּוּר הוּא לְקַפֵּל לְפִי הַקְּפָלִים הַקְּבוּעִים, כִּי זֶה נִרְאֶה כְּתִקּוּן הַבֶּגֶד. קִפּוּל שֶׁאֵינוֹ מְדֻיָּק – מֻתָּר, וְכָךְ שׁוֹמְרִים גַּם עַל כְּבוֹד הַטַּלִּית.
בְּשַׁבָּת נִדְבַּק לְנֹעַם בּוֹץ עַל הַנַּעַל בַּדֶּרֶךְ לְבֵית הַכְּנֶסֶת. מָה מֻתָּר לוֹ לַעֲשׂוֹת?
💡 הַמֵּסִיר בּוֹץ וּמְפוֹרֵר אוֹתוֹ עוֹבֵר עַל אִסּוּר טוֹחֵן מִדִּבְרֵי חֲכָמִים. אֲבָל בִּשְׁעַת הַצֹּרֶךְ מֻתָּר לְהָסִיר אוֹתוֹ בְּשִׁנּוּי: לְהַכּוֹת נַעַל בְּנַעַל.

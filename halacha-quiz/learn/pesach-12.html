הִלְכוֹת מַצָּה

🍷 הִלְכוֹת פֶּסַח · פֶּרֶק י״ב

🫓מִצְוַת אֲכִילַת מַצָּה

מִצְוָה מִן הַתּוֹרָה לֶאֱכֹל מַצָּה בְּלֵיל ט״ו בְּנִיסָן, שֶׁנֶּאֱמַר: ״בָּעֶרֶב תֹּאכְלוּ מַצֹּת״. בִּשְׁאָר יְמֵי הַפֶּסַח אֵין חוֹבָה, אֲבָל הַמַּצָּה הִיא הַמַּאֲכָל הָעִקָּרִי שֶׁאוֹכְלִים בְּמָקוֹם הַלֶּחֶם.
אֵין יוֹצְאִים יְדֵי חוֹבָה בְּמַצָּה גְּנוּבָה אוֹ גְּזוּלָה. לָכֵן כְּשֶׁהַמּוֹכֵר מְבַקֵּשׁ לְקַבֵּל אֶת הַכֶּסֶף מִיָּד – מְשַׁלְּמִים לוֹ מִיָּד, וְלֹא דּוֹחִים אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: הַרְבֵּה מִגְּדוֹלֵי יִשְׂרָאֵל כָּתְבוּ שֶׁגַּם מִי שֶׁאוֹכֵל מַצָּה בְּכָל שִׁבְעַת הַיָּמִים מְקַיֵּם מִצְוָה. כָּךְ כָּל אֲרוּחָה בְּפֶסַח יְכוֹלָה לִהְיוֹת פְּגִישָׁה עִם הַחֵרוּת שֶׁה׳ נָתַן לָנוּ.
📖 פֶּרֶק י״ב, הֲלָכוֹת א׳, ח׳

🍘🔑 מוּשָּׂג: מַצָּה

״מִצְוָה מִן הַתּוֹרָה לֶאֱכֹל מַצָּה בְּלֵיל ט״ו בְּנִיסָן... וְכֵן בַּזֹּהַר קָרְאוּ לַמַּצָּה: ׳מֵיכְלָא דִּמְהֵימְנוּתָא׳ – לֶחֶם הָאֱמוּנָה״.
📖 מַצָּה – קֶמַח וָמַיִם שֶׁנֶּאֱפוּ בִּמְהִירוּת בְּלִי שֶׁהֶחְמִיצוּ; הַזֹּהַר קוֹרֵא לָהּ ״לֶחֶם הָאֱמוּנָה״.
💛 אַהֲבַת הַשֵּׁם: בַּאֲכִילַת הַמַּצָּה אָנוּ קוֹלְטִים אֶת הָאֱמוּנָה שֶׁה׳ מַשְׁגִּיחַ עָלֵינוּ וּבָחַר בָּנוּ. 🧒 וְאַתָּה? בְּכָל בִּיס מַצָּה בְּלֵיל הַסֵּדֶר, תִּזְכֹּר: ה׳ הוֹצִיא אוֹתָנוּ מִמִּצְרַיִם.
📖 פֶּרֶק י״ב, הֲלָכָה א׳; פֶּרֶק א׳, הֲלָכָה ו׳

🌾מַצָּה שְׁמוּרָה

הַתּוֹרָה אוֹמֶרֶת: ״וּשְׁמַרְתֶּם אֶת הַמַּצּוֹת״ – אֶת מַצּוֹת הַמִּצְוָה שֶׁל לֵיל הַסֵּדֶר שׁוֹמְרִים שְׁמִירָה מְיֻחֶדֶת שֶׁלֹּא יַחְמִיצוּ. לְכַתְּחִלָּה שׁוֹמְרִים אוֹתָן מִשְּׁעַת הַקְּצִירָה: קוֹצְרִים אֶת הַחִטִּים כְּשֶׁהֵן עוֹד מְעַט לַחוֹת, וּמְאַחְסְנִים אוֹתָן בְּמָקוֹם שֶׁאֵין בּוֹ חֲשַׁשׁ שֶׁל מַיִם.
גַּם מַצָּה שֶׁנִּשְׁמְרָה מִשְּׁעַת הַטְּחִינָה כְּשֵׁרָה לְמַצַּת מִצְוָה. אֲבָל כַּיּוֹם נוֹהֲגִים לְהַדֵּר, וּבְלֵיל הַסֵּדֶר אוֹכְלִים מַצּוֹת שְׁמוּרוֹת מִשְּׁעַת קְצִירָה.
💛 אַהֲבַת הַשֵּׁם: שׁוֹמְרִים עַל הַמַּצָּה כְּמוֹ שֶׁשּׁוֹמְרִים עַל דָּבָר יָקָר. כָּךְ אֲנַחְנוּ מַרְאִים לַה׳ שֶׁהַמִּצְוָה שֶׁלּוֹ חֲשׁוּבָה לָנוּ – כְּבָר מֵהַשָּׂדֶה.
📖 פֶּרֶק י״ב, הֲלָכָה ב׳

🏷️שְׁמוּרָה מִשְּׁעַת קְצִירָה אוֹ מִשְּׁעַת טְחִינָה

הַמַּצּוֹת הָרְגִילוֹת שְׁמוּרוֹת מִשְּׁעַת טְחִינָה, וְהֵן כְּשֵׁרוֹת לְכַתְּחִלָּה לְכָל הַפֶּסַח. הַמְּהַדְּרִים אוֹכְלִים כָּל הַפֶּסַח מַצּוֹת שְׁמוּרוֹת מִשְּׁעַת קְצִירָה, כִּי מְדַקְדְּקִים בָּהֶן יוֹתֵר: עוֹצְרִים אֶת הַמְּכוֹנוֹת כָּל שְׁמוֹנֶה עֶשְׂרֵה דַּקּוֹת וּמְנַקִּים אוֹתָן הֵיטֵב.
קֶמַח רָגִיל מֵהַחֲנוּת, בְּלִי הַשְׁגָּחָה לְפֶסַח – לֹא קוֹנִים, כִּי פְּעָמִים רַבּוֹת מַשְׁרִים אֶת הַחִטִּים בְּמַיִם לִפְנֵי הַטְּחִינָה, וְיֵשׁ חֲשַׁשׁ שֶׁהֶחְמִיצוּ.
💛 אַהֲבַת הַשֵּׁם: אֶפְשָׁר לְקַיֵּם אֶת הַהֲלָכָה, וְאֶפְשָׁר לְהוֹסִיף עָלֶיהָ הִדּוּר. מִי שֶׁאוֹהֵב מְחַפֵּשׂ אֵיךְ לְהַדֵּר בְּמִצְווֹת אֲהוּבוֹ.
📖 פֶּרֶק י״ב, הֲלָכוֹת ב׳, ה׳

🙌״לְשֵׁם מַצַּת מִצְוָה״

אֶת מַצּוֹת הַמִּצְוָה לָשִׁים וְאוֹפִים יְהוּדִים בְּנֵי מִצְוָה, שֶׁמְּכַוְּנִים שֶׁהַכֹּל נַעֲשֶׂה לְשֵׁם מַצַּת מִצְוָה – וְלֹא גּוֹיִים אוֹ יְלָדִים קְטַנִּים. בִּתְחִלַּת הָעֲבוֹדָה אוֹמְרִים בַּפֶּה: ״לְשֵׁם מַצַּת מִצְוָה״.
אֶת הַקְּצִירָה וְהַטְּחִינָה מֻתָּר לַעֲשׂוֹת גַּם עַל יְדֵי גּוֹיִים, בִּתְנַאי שֶׁיְּהוּדִי עוֹמֵד לְיָדָם וּמַשְׁגִּיחַ שֶׁלֹּא יִהְיֶה חִמּוּץ.
💛 אַהֲבַת הַשֵּׁם: הַמַּצָּה שֶׁל הַסֵּדֶר אֵינָהּ סְתָם לֶחֶם – הִיא נַעֲשֵׂית מֵהַהַתְחָלָה בִּשְׁבִיל ה׳. כְּשֶׁהַלֵּב מְכַוֵּן, גַּם קֶמַח וּמַיִם נַעֲשִׂים מִצְוָה.
📖 פֶּרֶק י״ב, הֲלָכָה ג׳

⚙️מַצּוֹת יָד וּמַצּוֹת מְכוֹנָה

מַצּוֹת מְכוֹנָה כְּשֵׁרוֹת לְפֶסַח כְּשֶׁיֵּשׁ עֲלֵיהֶן הַשְׁגָּחָה טוֹבָה. יֵשׁ אוֹמְרִים שֶׁאֵין יוֹצְאִים בָּהֶן בְּלֵיל הַסֵּדֶר, כִּי מְכוֹנָה לֹא יְכוֹלָה לְכַוֵּן. אֲבָל רֹב הַפּוֹסְקִים אוֹמְרִים שֶׁיּוֹצְאִים בָּהֶן: הָאָדָם מַפְעִיל אֶת הַמְּכוֹנָה לְשֵׁם מַצַּת מִצְוָה.
רַבִּים מְהַדְּרִים לְקַיֵּם אֶת הַמִּצְוָה בְּמַצּוֹת יָד. אֵין צֹרֶךְ שֶׁכָּל הַסְּעוּדָה תִּהְיֶה מִמַּצּוֹת יָד – רַק הַכְּזֵיתִים שֶׁל הַמִּצְוָה.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאָדָם לוֹחֵץ עַל הַכַּפְתּוֹר וְאוֹמֵר ״לְשֵׁם מַצַּת מִצְוָה״, גַּם הַמְּכוֹנָה עוֹבֶדֶת בִּשְׁבִיל ה׳. הַכַּוָּנָה שֶׁל הַלֵּב הוֹפֶכֶת כָּל כְּלִי לִכְלִי שֶׁל מִצְוָה.
📖 פֶּרֶק י״ב, הֲלָכָה ד׳

💧מַיִם שֶׁלָּנוּ וְלִישָׁה זְרִיזָה

לָשִׁים אֶת הַמַּצּוֹת בְּ״מַיִם שֶׁלָּנוּ״ – מַיִם שֶׁשּׁוֹאֲבִים מִבְּעוֹד יוֹם וּמַשְׁאִירִים לָלוּן כָּל הַלַּיְלָה בְּמָקוֹם קָרִיר. לֹא לָשִׁים בְּמַיִם פּוֹשְׁרִים, כִּי הַחֹם מְזָרֵז אֶת הַחִמּוּץ. וְלֹא נוֹתְנִים בָּעִסָּה מֶלַח אוֹ פִּלְפֵּל.
לֹא לָשִׁים בַּשֶּׁמֶשׁ אוֹ לְיַד הַתַּנּוּר, וְלֹא מַשְׁאִירִים אֶת הָעִסָּה בְּלִי עֵסֶק אֲפִלּוּ רֶגַע. כְּשֶׁהַיָּדַיִם שֶׁל הַלָּשׁ מִתְחַמְּמוֹת – הוּא מְצַנֵּן אוֹתָן בְּמַיִם קָרִים.
💛 אַהֲבַת הַשֵּׁם: הַזְּרִיזוּת בַּמַּצּוֹת מַזְכִּירָה לָנוּ אֶת הַיְּצִיאָה מִמִּצְרַיִם, כְּשֶׁאֲבוֹתֵינוּ יָצְאוּ בְּחִפָּזוֹן. מִי שֶׁאוֹהֵב מִצְוָה – רָץ אֵלֶיהָ וְלֹא מִתְעַצֵּל.
📖 פֶּרֶק י״ב, הֲלָכוֹת ו׳–ז׳

🔥אֲפִיַּת הַמַּצּוֹת

מְחַמְּמִים אֶת הַתַּנּוּר הֵיטֵב, כְּדֵי שֶׁהַבָּצֵק יַתְחִיל לְהֵאָפוֹת מִיָּד. מַצָּה שֶׁנֶּאֶפְתָה בְּחֹם הַשֶּׁמֶשׁ אֵינָהּ ״לֶחֶם״, וְאֵין יוֹצְאִים בָּהּ יְדֵי חוֹבָה. לֹא מְקַשְּׁטִים אֶת הַמַּצָּה בְּצִיּוּרִים, וְלֹא עוֹשִׂים מַצָּה עָבָה טֶפַח – שֶׁמָּא תַּחְמִיץ.
מִנְהַג הָאַשְׁכְּנַזִּים לַעֲשׂוֹת מַצּוֹת דַּקּוֹת וְקָשׁוֹת, שֶׁהָאֵשׁ שׁוֹלֶטֶת בָּהֶן הֵיטֵב. וּבַסְּפָרַדִּים יֵשׁ שֶׁנָּהֲגוּ לַעֲשׂוֹת מַצָּה בְּעֹבִי אֶצְבַּע, וְיֵשׁ שֶׁנָּהֲגוּ לַעֲשׂוֹתָהּ דַּקָּה כְּמוֹ הָאַשְׁכְּנַזִּים.
💛 אַהֲבַת הַשֵּׁם: הַמַּצָּה נִקְרֵאת ״לֶחֶם עֹנִי״ – פְּשׁוּטָה, בְּלִי קִשּׁוּטִים. דַּוְקָא הַפַּשְׁטוּת שֶׁלָּהּ מְלַמֶּדֶת אוֹתָנוּ עֲנָוָה לִפְנֵי ה׳.
📖 פֶּרֶק י״ב, הֲלָכָה ח׳

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

מַצָּה עֲשִׁירָה, שֶׁנִּלּוֹשָׁה בְּיַיִן אוֹ שֶׁמֶן: לְמִנְהַג הַסְּפָרַדִּים מֻתֶּרֶת בְּפֶסַח וּבְעֶרֶב פֶּסַח, וּלְמִנְהַג הָאַשְׁכְּנַזִּים אֲסוּרָה. וְעֹבִי הַמַּצָּה: הָאַשְׁכְּנַזִּים עוֹשִׂים מַצּוֹת דַּקּוֹת וְקָשׁוֹת, וּבַסְּפָרַדִּים יֵשׁ שֶׁעוֹשִׂים מַצָּה בְּעֹבִי אֶצְבַּע. וְיֵשׁ גַּם מַחֲלֹקוֹת בֵּין הַפּוֹסְקִים: הַאִם יוֹצְאִים בְּמַצַּת מְכוֹנָה, וְהַאִם מֻתָּר לֶאֱכֹל בְּעֶרֶב פֶּסַח עוּגָה מִקֶּמַח מַצָּה.
כַּיּוֹם רַבִּים מֵהַפּוֹסְקִים אוֹמְרִים שֶׁגַּם הַסְּפָרַדִּים יִמָּנְעוּ מִמַּצָּה עֲשִׁירָה, בִּגְלַל סְפֵקוֹת בְּדֶרֶךְ הַיִּצּוּר. וּמַצָּה עֲשִׁירָה שֶׁיֻּצְּרָה כַּדִּין – מִי שֶׁנּוֹהֵג לְאָכְלָהּ, אוֹכֵל רַק עַד סוֹף הַשָּׁעָה הַתְּשִׁיעִית.
💛 אַהֲבַת הַשֵּׁם: שְׁנֵי הַמִּנְהָגִים בְּנוּיִים עַל דִּבְרֵי גְּדוֹלֵי הַפּוֹסְקִים. כָּל אֶחָד שׁוֹמֵר עַל מִנְהַג אֲבוֹתָיו – וְכֻלָּנוּ אוֹכְלִים אֶת אוֹתָהּ מַצַּת מִצְוָה בְּלֵיל הַסֵּדֶר.
📖 פֶּרֶק י״ב, הֲלָכָה ח׳; פֶּרֶק י״ג, הֲלָכָה ז׳

🧠 בְּדֹק אֶת עַצְמְךָ

בְּלֵיל הַסֵּדֶר יֵשׁ לְאַבָּא רַק מַצּוֹת מְכוֹנָה שְׁמוּרוֹת, עִם הַשְׁגָּחָה טוֹבָה. מָה הַדִּין?
💡 רֹב הַפּוֹסְקִים אוֹמְרִים שֶׁיּוֹצְאִים בְּמַצּוֹת מְכוֹנָה, כִּי הָאָדָם שֶׁמַּפְעִיל אֶת הַמְּכוֹנָה מְכַוֵּן לְשֵׁם מַצַּת מִצְוָה. וְרַבִּים מְהַדְּרִים בְּמַצַּת יָד לַכְּזֵיתִים שֶׁל הַמִּצְוָה.
אֵילּוּ מַצּוֹת כְּשֵׁרוֹת לְכַתְּחִלָּה לְכָל יְמֵי הַפֶּסַח?
💡 הַמַּצּוֹת הָרְגִילוֹת שְׁמוּרוֹת מִשְּׁעַת טְחִינָה וּכְשֵׁרוֹת לְכַתְּחִלָּה לְכָל הַפֶּסַח. הַמְּהַדְּרִים אוֹכְלִים שְׁמוּרוֹת מִשְּׁעַת קְצִירָה. וְקֶמַח מֵהַחֲנוּת בְּלִי הַשְׁגָּחָה – יֵשׁ חֲשַׁשׁ שֶׁהֶחְמִיץ.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

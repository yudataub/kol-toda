כְּלָלֵי הַמְּלָאכוֹת

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק ט׳

🪑מָה נֶחְשָׁב מְלָאכָה?

הַתּוֹרָה אָסְרָה ״מְלֶאכֶת מַחֲשֶׁבֶת״ – יְצִירָה מְתֻכְנֶנֶת, כְּמוֹ בַּמִּשְׁכָּן, וְלֹא כָּל מַאֲמָץ. לְהַעֲבִיר כִּסְאוֹת וְשֻׁלְחָנוֹת בְּתוֹךְ הַבַּיִת – אֵינוֹ מְלָאכָה, אֲבָל לְהוֹצִיא מַחַט קְטַנָּה לִרְשׁוּת הָרַבִּים – מְלָאכָה. לִפְתֹּחַ וְלִסְגֹּר חַלּוֹן – מֻתָּר, אֲבָל לְהַרְכִּיב חַלּוֹן עַל צִירָיו – מְלָאכָה.
אַרְבַּע מְלָאכוֹת נִזְכְּרוּ בַּתּוֹרָה בִּמְפֹרָשׁ: חָרִישׁ, קָצִיר, הַבְעָרָה וְהוֹצָאָה – לְלַמֵּד שֶׁגַּם מְלָאכוֹת שֶׁהַחַיִּים תְּלוּיִים בָּהֶן, וְגַם מְלָאכָה שֶׁנִּרְאֵית קַלָּה, אֲסוּרוֹת.
💛 אַהֲבַת הַשֵּׁם: בְּשֵׁשֶׁת הַיָּמִים אָנוּ יוֹצְרִים וּבוֹנִים, וּבַשַּׁבָּת אָנוּ עוֹצְרִים וְזוֹכְרִים מִי בָּרָא אֶת הַכֹּל. הַשְּׁבִיתָה הִיא הַכְרָזָה שֶׁל אַהֲבָה: הָעוֹלָם שֶׁלְּךָ, ה׳.
📖 פֶּרֶק ט׳, הֲלָכָה א׳

🛠️ל״ט מְלָאכוֹת מִן הַמִּשְׁכָּן

לְיַד הַצִּוּוּי עַל בִּנְיַן הַמִּשְׁכָּן נֶאֱמַר: ״אַךְ אֶת שַׁבְּתֹתַי תִּשְׁמֹרוּ״. מִכָּאן לָמְדוּ: כָּל הַמְּלָאכוֹת שֶׁעָשׂוּ בְּבִנְיַן הַמִּשְׁכָּן – אֲסוּרוֹת בְּשַׁבָּת. הֵן שְׁלֹשִׁים וָתֵשַׁע ״אָבוֹת מְלָאכוֹת״, כְּמוֹ זוֹרֵעַ, קוֹצֵר, אוֹפֶה, כּוֹתֵב, בּוֹנֶה, מַבְעִיר וּמוֹצִיא מֵרְשׁוּת לִרְשׁוּת.
מְלָאכָה שֶׁדּוֹמָה לָ״אָב״ נִקְרֵאת ״תּוֹלָדָה״ – וְגַם הִיא אֲסוּרָה מִן הַתּוֹרָה, בְּדִיּוּק כְּמוֹ הָאָב.
💛 אַהֲבַת הַשֵּׁם: כָּל הָעוֹלָם יָכוֹל לִהְיוֹת מִשְׁכָּן לַשְּׁכִינָה. כְּשֶׁאָנוּ שׁוֹבְתִים מִן הַמְּלָאכוֹת הָאֵלֶּה בְּשַׁבָּת, אָנוּ מְגַלִּים אֶת הַטַּעַם הַפְּנִימִי שֶׁל כָּל הָעֲבוֹדָה שֶׁלָּנוּ בְּשֵׁשֶׁת הַיָּמִים.
📖 פֶּרֶק ט׳, הֲלָכָה ב׳

🔀בְּשִׁנּוּי וּבִשְׁנַיִם

מִן הַתּוֹרָה אָסוּר לַעֲשׂוֹת מְלָאכָה כְּדַרְכָּהּ. מִי שֶׁעוֹשֶׂה אוֹתָהּ ״בְּשִׁנּוּי״ – כְּמוֹ שֶׁכּוֹתֵב בַּיָּד שֶׁאֵינוֹ רָגִיל בָּהּ – לֹא עָבַר עַל אִסּוּר תּוֹרָה, אֲבָל חֲכָמִים אָסְרוּ גַּם זֹאת. וְכֵן שְׁנַיִם שֶׁעָשׂוּ יַחַד מְלָאכָה שֶׁכָּל אֶחָד יָכוֹל לַעֲשׂוֹת לְבַד – עָבְרוּ עַל אִסּוּר חֲכָמִים.
בְּסָפֵק שֶׁל אִסּוּר תּוֹרָה – מַחְמִירִים, וּבְסָפֵק שֶׁל אִסּוּר חֲכָמִים – מְקִלִּים.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים רָצוּ שֶׁהַשַּׁבָּת תִּהְיֶה שְׁבִיתָה אֲמִתִּית – לֹא ״לְהִתְחַכֵּם״ וְלַעֲשׂוֹת הַכֹּל בְּדֶרֶךְ אַחֶרֶת. מִי שֶׁאוֹהֵב אֶת הַשַּׁבָּת אֵינוֹ מְחַפֵּשׂ פִּרְצוֹת, אֶלָּא נָח בֶּאֱמֶת.
📖 פֶּרֶק ט׳, הֲלָכוֹת ג׳–ד׳

🧱גָּדֵר סְבִיב הַתּוֹרָה

הַתּוֹרָה צִוְּתָה אֶת הַחֲכָמִים: ״עֲשׂוּ מִשְׁמֶרֶת לְמִשְׁמַרְתִּי״ – לְהוֹסִיף גְּדֵרִים. לָכֵן אָסְרוּ: לְטַפֵּס עַל עֵץ אוֹ לִרְכֹּב עַל בְּהֵמָה – שֶׁמָּא יִקְטְפוּ עָנָף; לְנַגֵּן – שֶׁמָּא יְתַקְּנוּ כְּלִי נְגִינָה; לְבַקֵּשׁ מִגּוֹי לַעֲשׂוֹת מְלָאכָה; וּלְטַלְטֵל מֻקְצֶה.
עוֹד תִּקְּנוּ חֲכָמִים, כְּדֵי שֶׁהַשַּׁבָּת תִּהְיֶה יוֹם שֶׁל מְנוּחָה וּקְדֻשָּׁה: לֹא לְדַבֵּר עַל עִסְקֵי חוֹל, לֹא לָלֶכֶת בִּמְהִירוּת כְּמוֹ בְּחוֹל, וְלֹא לְהָכִין בְּשַׁבָּת דְּבָרִים לִימֵי הַחוֹל.
💛 אַהֲבַת הַשֵּׁם: גָּדֵר סְבִיב גַּן יָקָר אֵינָהּ עֹנֶשׁ – הִיא שְׁמִירָה. גְּזֵרוֹת חֲכָמִים שׁוֹמְרוֹת עָלֵינוּ שֶׁלֹּא נִכָּשֵׁל, וְשׁוֹמְרוֹת עַל הָאֲוִירָה הַשְּׁקֵטָה וְהַקְּדוֹשָׁה שֶׁל הַשַּׁבָּת.
📖 פֶּרֶק ט׳, הֲלָכָה ד׳

🦌בְּלִי כַּוָּנָה – וּמַה שֶּׁלֹּא מִתְקַיֵּם

מֻתָּר לִגְרֹר סַפְסָל אוֹ לָלֶכֶת עַל דֶּשֶׁא, אַף שֶׁאוּלַי יֵעָשֶׂה חָרִיץ בָּאֲדָמָה אוֹ יֵעָקְרוּ עֲשָׂבִים – כִּי אֵין מִתְכַּוְּנִים לְכָךְ, וְאֵין וַדָּאוּת. אֲבָל אִם בָּרוּר שֶׁהַמְּלָאכָה תֵּעָשֶׂה – אָסוּר. לְמָשָׁל, אָסוּר לִסְגֹּר דֶּלֶת כְּשֶׁיֵּשׁ צְבִי בַּבַּיִת, כִּי בְּוַדַּאי יִצּוֹד אוֹתוֹ.
הַתּוֹרָה אָסְרָה מְלָאכָה שֶׁמִּתְקַיֶּמֶת. קֶשֶׁר חָזָק שֶׁעוֹמֵד זְמַן רַב – אָסוּר, אֲבָל עֲנִיבָה רְגִילָה שֶׁפּוֹתְחִים בְּקַלּוּת – מֻתֶּרֶת. גַּם כְּתִיבָה עַל חוֹל אוֹ עַל אֵדִים בַּחַלּוֹן – אֲסוּרָה מִדִּבְרֵי חֲכָמִים.
💛 אַהֲבַת הַשֵּׁם: הַהֲלָכָה מְדַיֶּקֶת: הִיא לֹא רוֹצָה שֶׁנְּפַחֵד מִכָּל צַעַד, וְגַם לֹא שֶׁנַּעֲשֶׂה מְלָאכָה בְּלִי לָשִׂים לֵב. הַשַּׁבָּת מְלַמֶּדֶת אוֹתָנוּ לִחְיוֹת בְּשִׂמְחָה וּבִזְהִירוּת יַחַד.
📖 פֶּרֶק ט׳, הֲלָכוֹת ה׳, ח׳

🐜גַּם הֶרֶס – אָסוּר

הַתּוֹרָה אָסְרָה מְלָאכָה שֶׁל תִּקּוּן וּבְנִיָּה. מִי שֶׁעוֹשֶׂה מְלָאכָה בְּדֶרֶךְ שֶׁל הַשְׁחָתָה – ״מְקַלְקֵל״ – פָּטוּר מֵעֹנֶשׁ הַתּוֹרָה, אֲבָל חֲכָמִים אָסְרוּ. קוֹרֵעַ כְּדֵי לִתְפֹּר – אִסּוּר תּוֹרָה; קוֹרֵעַ סְתָם – אִסּוּר חֲכָמִים. וְכֵן הַדּוֹרֵס נְמָלִים אוֹ הוֹרֵג יַתּוּשִׁים – אָסוּר מִדִּבְרֵי חֲכָמִים.
בְּמַסֶּכֶת שַׁבָּת, ״פָּטוּר״ פֵּרוּשׁוֹ: פָּטוּר מֵעֹנֶשׁ – אֲבָל הַמַּעֲשֶׂה עַצְמוֹ אָסוּר.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת לֹא בּוֹנִים – וְגַם לֹא הוֹרְסִים. הַשַּׁבָּת הִיא יוֹם שֶׁל שַׁלְוָה, שֶׁבּוֹ נוֹתְנִים לָעוֹלָם שֶׁל ה׳ לִהְיוֹת כְּמוֹ שֶׁהוּא – וְנֶהֱנִים מִמֶּנּוּ.
📖 פֶּרֶק ט׳, הֲלָכָה ז׳

💧גְּרָמָא

הַתּוֹרָה אָסְרָה ״לַעֲשׂוֹת״ מְלָאכָה. כְּשֶׁהַמְּלָאכָה נַעֲשֵׂית מֵאֵלֶיהָ, וְהָאָדָם רַק גָּרַם לָהּ – זוֹ ״גְּרָמָא״. לְמָשָׁל, כְּשֶׁשְּׂרֵפָה מִתְפַּשֶּׁטֶת, מֻתָּר לְהַנִּיחַ סְבִיבָהּ כֵּלִים מְלֵאִים מַיִם – וּכְשֶׁהָאֵשׁ תַּגִּיעַ, הֵם יִתְבַּקְּעוּ וְהַמַּיִם יְכַבּוּ אוֹתָהּ.
גְּרָמָא מֻתֶּרֶת רַק לְצֹרֶךְ גָּדוֹל – לְמִצְוָה אוֹ כְּדֵי לִמְנֹעַ הֶפְסֵד. בְּלִי צֹרֶךְ גָּדוֹל – אָסוּר לִגְרֹם לִמְלָאכָה בְּשַׁבָּת.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים לִמְּדוּ אוֹתָנוּ לְהַבְחִין בְּדִיּוּק בֵּין מַעֲשֶׂה לִגְרִימָה. הַדִּיּוּק הַזֶּה בָּא מֵאַהֲבָה: לִשְׁמֹר עַל הַשַּׁבָּת – וְגַם לִדְאֹג לָאָדָם בִּשְׁעַת צָרָה.
📖 פֶּרֶק ט׳, הֲלָכָה ט׳

🫏גַּם הַבְּהֵמוֹת נָחוֹת

הַתּוֹרָה צִוְּתָה: ״לֹא תַעֲשֶׂה כָל מְלָאכָה – אַתָּה וּבִנְךָ וּבִתֶּךָ... וְשׁוֹרְךָ וַחֲמֹרְךָ״. לָכֵן אָסוּר לָתֵת לַיְּלָדִים לַעֲשׂוֹת בִּשְׁבִילֵנוּ מְלָאכָה בְּשַׁבָּת, וְאָסוּר לְהַעֲבִיד אֶת הַבְּהֵמוֹת – גַּם הֵן נָחוֹת. וַחֲכָמִים אָסְרוּ לְבַקֵּשׁ מִגּוֹי לַעֲשׂוֹת עֲבוּרֵנוּ מְלָאכָה בְּשַׁבָּת.
כֵּלִים אֵינָם צְרִיכִים לִשְׁבֹּת. לָכֵן מֻתָּר לְהַשְׁאִיל לְגוֹי כְּלִי שֶׁיַּעֲשֶׂה בּוֹ מְלָאכָה בְּשַׁבָּת לְעַצְמוֹ – כָּל עוֹד לֹא נִרְאֶה שֶׁהוּא עוֹבֵד בִּשְׁבִילֵנוּ.
💛 אַהֲבַת הַשֵּׁם: ״לְמַעַן יָנוּחַ עַבְדְּךָ וַאֲמָתְךָ כָּמוֹךָ״ – הַשַּׁבָּת הִיא מַתְּנַת מְנוּחָה לְכָל מִי שֶׁסְּבִיבֵנוּ. מִי שֶׁאוֹהֵב אֶת ה׳ דּוֹאֵג שֶׁגַּם הַחַלָּשִׁים, וְגַם בַּעֲלֵי הַחַיִּים, יָנוּחוּ.
📖 פֶּרֶק ט׳, הֲלָכָה י׳

🧠 בְּדֹק אֶת עַצְמְךָ

לָמָּה אָסְרוּ חֲכָמִים לְטַפֵּס עַל עֵץ בְּשַׁבָּת?
💡 זוֹ אַחַת מִגְּזֵרוֹת חֲכָמִים – ״גָּדֵר״ סְבִיב אִסּוּר תּוֹרָה: מִי שֶׁמְּטַפֵּס עַל עֵץ עָלוּל לִשְׁכֹּחַ וְלִקְטֹף עָנָף.
רוֹנִי כּוֹתֵב תָּמִיד בְּיַד יָמִין. הוּא אוֹמֵר: ״בְּשַׁבָּת אֶכְתֹּב בְּיַד שְׂמֹאל, וְזֶה לֹא נֶחְשָׁב״. מָה נָכוֹן?
💡 בְּשִׁנּוּי אֵין אִסּוּר תּוֹרָה, אֲבָל חֲכָמִים אָסְרוּ – כְּדֵי שֶׁלֹּא יָבוֹאוּ לַעֲשׂוֹת כְּדַרְכָּם, וּכְדֵי שֶׁהַשַּׁבָּת תִּהְיֶה שְׁבִיתָה אֲמִתִּית.
בֶּחָצֵר יֵשׁ דֶּשֶׁא. הַאִם מֻתָּר לָלֶכֶת עָלָיו בְּשַׁבָּת, אַף שֶׁאוּלַי יֵעָקְרוּ עֲשָׂבִים?
💡 זֶה ״דָּבָר שֶׁאֵינוֹ מִתְכַּוֵּן״ – וּמֻתָּר. אֲבָל אִם בָּרוּר שֶׁהַהֲלִיכָה תַּעֲקֹר עֲשָׂבִים – אָסוּר, כְּמוֹ מִי שֶׁסּוֹגֵר דֶּלֶת כְּשֶׁיֵּשׁ צְבִי בַּבַּיִת.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

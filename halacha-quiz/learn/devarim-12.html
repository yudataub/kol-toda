הַתְּשׁוּבָה: חֲזַ״ל אוֹמְרִים ׳שְׁלֹשָׁה שֻׁתָּפִין יֵשׁ בָּאָדָם׳ – הַקָּדוֹשׁ בָּרוּךְ הוּא, אָבִיו וְאִמּוֹ. הַהוֹרִים הֵם הַשֻּׁתָּפִים שֶׁל ה׳ בַּבְּרִיאָה שֶׁלִּי, וְלָכֵן מִי שֶׁמְּכַבֵּד אֶת הוֹרָיו – ה׳ מַחְשִׁיב לוֹ כְּאִלּוּ כִּבֵּד אוֹתוֹ.

מִיּוּן נוֹסָף: מִצְוַת עֲשֵׂה הִיא דָּבָר שֶׁצָּרִיךְ לַעֲשׂוֹת. מִצְוַת לֹא תַעֲשֶׂה הִיא אִסּוּר – דָּבָר שֶׁה׳ מְצַוֶּה שֶׁלֹּא לַעֲשׂוֹת.

וּבַעֲשֶׂרֶת הַדִּבְּרוֹת?
שָׁלֹשׁ מִצְווֹת עֲשֵׂה: לְהַאֲמִין בַּה׳ (אָנֹכִי), לִשְׁמֹר אֶת הַשַּׁבָּת, וּלְכַבֵּד אֶת הָאָב וְאֶת הָאֵם.
שֶׁבַע מִצְווֹת לֹא תַעֲשֶׂה: עֲבוֹדָה זָרָה, שְׁבוּעַת שָׁוְא, וְחָמֵשׁ הַמִּצְווֹת שֶׁבַּלּוּחַ הַשֵּׁנִי.
כָּךְ הַדִּבְּרוֹת כּוֹלְלוֹת גַּם דְּבָרִים שֶׁחַיָּבִים לַעֲשׂוֹת וְגַם דְּבָרִים שֶׁאָסוּר לַעֲשׂוֹת. אֶת הַשַּׁבָּת וְאֶת כִּבּוּד הוֹרִים נִלְמַד בְּהַרְחָבָה בַּשִּׁעוּר הַבָּא.

ה׳ נָתַן לָנוּ גַּם מִצְווֹת לַעֲשׂוֹת וְגַם דְּבָרִים לְהִזָּהֵר מֵהֶם. כְּשֶׁאֲנִי מְקַיֵּם מִצְוָה, וּכְשֶׁאֲנִי מִתְאַפֵּק וְלֹא לוֹקֵחַ מָה שֶׁאֵינוֹ שֶׁלִּי – בִּשְׁנֵי הַמִּקְרִים אֲנִי עוֹשֶׂה אֶת רְצוֹן ה׳, שֶׁאוֹהֵב אוֹתִי 💛

🤝בֵּין אָדָם לַחֲבֵרוֹ – מַעֲשֶׂה, דִּבּוּר וּמַחֲשָׁבָה

📖 פֶּרֶק ה׳, פְּסוּקִים י״ז–י״ח ←

חָמֵשׁ הַדִּבְּרוֹת שֶׁבַּלּוּחַ הַשֵּׁנִי הֵן הַבָּסִיס לְכָל הַחוֹבוֹת שֶׁלָּנוּ כְּלַפֵּי אָדָם אַחֵר: לֹא לִפְגֹּעַ בּוֹ בְּשׁוּם צוּרָה – לֹא בְּחַיָּיו, לֹא בְּמָמוֹנוֹ, וְלֹא בַּבְּרִית שֶׁבֵּין אִישׁ לְאִשְׁתּוֹ.

אֶפְשָׁר לְחַלֵּק אוֹתָן לִשְׁלוֹשָׁה חֲלָקִים:
בַּמַּעֲשֶׂה: לֹא תִּרְצָח, לֹא תִּנְאָף, לֹא תִּגְנֹב.
בַּדִּבּוּר: ׳לֹא תַעֲנֶה בְרֵעֲךָ עֵד שָׁוְא׳ – אָסוּר לְהָעִיד עֵדוּת שֶׁקֶר עַל אָדָם בְּבֵית הַדִּין.
בַּמַּחֲשָׁבָה: לֹא תַחְמֹד וְלֹא תִתְאַוֶּה.

׳לֹא תַחְמֹד׳ – אָסוּר לְתַכְנֵן תַּחְבּוּלוֹת אֵיךְ לְהָבִיא אֵלַי דָּבָר שֶׁשַּׁיָּךְ לְאָדָם אַחֵר. הַתּוֹרָה לֹא מִסְתַּפֶּקֶת בְּאִסּוּר לִגְנֹב בַּיָּדַיִם; הִיא שׁוֹמֶרֶת עָלֵינוּ כְּבָר מִן הַמַּחֲשָׁבָה הָרִאשׁוֹנָה.

💡 לֹא תַחְמֹד – וְלֹא תִתְאַוֶּה
בְּסֵפֶר שְׁמוֹת: ׳לֹא תַחְמֹד בֵּית רֵעֶךָ, לֹא תַחְמֹד אֵשֶׁת רֵעֶךָ וְעַבְדּוֹ וַאֲמָתוֹ...׳.
בְּחֻמַּשׁ דְּבָרִים: ׳וְלֹא תַחְמֹד אֵשֶׁת רֵעֶךָ וְלֹא תִתְאַוֶּה בֵּית רֵעֶךָ שָׂדֵהוּ...׳.
לְפִי רַשִׁ״י וְרֹב הַמְּפָרְשִׁים, ׳תִתְאַוֶּה׳ הוּא כְּמוֹ ׳תַחְמֹד׳ – אוֹתוֹ אִסּוּר בְּמִלָּה אַחֶרֶת.
הָרַמְבַּ״ם מְחַלֵּק בֵּינֵיהֶם: ׳לֹא תַחְמֹד׳ – כְּשֶׁאָדָם מַפְצִיר בַּחֲבֵרוֹ וְלוֹחֵץ עָלָיו עַד שֶׁהוּא מַשִּׂיג אֶת הַחֵפֶץ. ׳לֹא תִתְאַוֶּה׳ – כְּבָר כְּשֶׁחָשַׁב בְּלִבּוֹ אֵיךְ יַשִּׂיג אֶת מָה שֶׁל חֲבֵרוֹ.

׳וְלֹא תִתְאַוֶּה׳ – ׳לָא תָרוֹג׳, אַף הוּא לְשׁוֹן חֶמְדָּה, כְּמוֹ ׳נֶחְמָד לְמַרְאֶה׳, דְּמִתַּרְגְּמִינָן ׳דִּמְרַגַּג לְמֶחֱזֵי׳.

כְּלוֹמַר: אוּנְקְלוֹס מְתַרְגֵּם אֶת ׳וְלֹא תִתְאַוֶּה׳ בַּמִּלָּה הָאֲרָמִית ׳לָא תָרוֹג׳, וְאֶת ׳נֶחְמָד לְמַרְאֶה׳ (עַל הָעֵצִים בְּגַן עֵדֶן) בְּאוֹתָהּ מִלָּה. מִכָּאן שֶׁגַּם ׳תִתְאַוֶּה׳ הִיא מִלָּה שֶׁל חֶמְדָּה.

📚 צֵא וּלְמַד מֵאַחְאָב וְנָבוֹת
הָרַמְבַּ״ם מְלַמֵּד: הַתַּאֲוָה מְבִיאָה לִידֵי חִמּוּד, וְהַחִמּוּד מֵבִיא לִידֵי גָּזֵל. וְהוּא מְבִיא דֻּגְמָה מִסֵּפֶר מְלָכִים.
אַחְאָב, מֶלֶךְ יִשְׂרָאֵל, רָאָה אֶת הַכֶּרֶם שֶׁל נָבוֹת הַיִּזְרְעֵאלִי, שֶׁהָיָה סָמוּךְ לְאַרְמוֹנוֹ, וְרָצָה אוֹתוֹ. נָבוֹת סֵרֵב לִמְכֹּר אֶת נַחֲלַת אֲבוֹתָיו.
אַחְאָב לֹא הִשְׁלִים עִם זֶה, וְהַתַּאֲוָה שֶׁלֹּא עָצַר הֵבִיאָה בַּסּוֹף לְעֵדֵי שֶׁקֶר, לַהֲרִיגַת נָבוֹת וְלִגְזֵלַת הַכֶּרֶם (מְלָכִים א׳, פֶּרֶק כ״א).
מַחֲשָׁבָה אַחַת שֶׁלֹּא עָצְרוּ אוֹתָהּ – וְכִמְעַט כָּל הַלּוּחַ הַשֵּׁנִי נִשְׁבַּר.

הַדִּבְּרוֹת נוֹגְעוֹת לְכָל תְּחוּמֵי הַחַיִּים: לָאֱמוּנָה, לַגּוּף (לֹא תִּרְצָח), לַזְּמַן (הַשַּׁבָּת), לַמָּמוֹן (לֹא תִּגְנֹב), לַמִּשְׁפָּחָה (כִּבּוּד הוֹרִים) וְלַמִּשְׁפָּט (לֹא תַעֲנֶה).
יֵשׁ בָּהֶן מִצְווֹת בַּמַּעֲשֶׂה, בַּדִּבּוּר וּבַמַּחֲשָׁבָה. לֹא פֶּלֶא שֶׁהֵן הַמִּצְווֹת הַמְּפֻרְסָמוֹת בְּיוֹתֵר בְּכָל הָעוֹלָם.

ה׳ אוֹהֵב כָּל אָדָם, וְלָכֵן הוּא שׁוֹמֵר עַל חַיָּיו, עַל כְּבוֹדוֹ וְעַל מָה שֶׁיֵּשׁ לוֹ. וּכְשֶׁאֲנִי שָׂמֵחַ בְּמָה שֶׁיֵּשׁ לִי וְלֹא חוֹמֵד אֶת שֶׁל חֲבֵרִי, אֲנִי מַרְגִּישׁ שֶׁה׳ נָתַן לִי בְּדִיּוּק אֶת מָה שֶׁאֲנִי צָרִיךְ 💛

בֵּית סֵפֶר רוֹצֶה לְהוֹדִיעַ עַל הַכְּלָלִים שֶׁלּוֹ. לְתַלְמִידֵי כִּתָּה א׳ יַסְבִּירוּ אוֹתָם בְּדֶרֶךְ אַחַת, לְתַלְמִיד חָדָשׁ שֶׁמִּצְטָרֵף לְכִתָּה גְּבוֹהָה – בְּדֶרֶךְ אַחֶרֶת, וּלְמִי שֶׁכְּבָר מַכִּיר אוֹתָם – רַק יַזְכִּירוּ.
הַכְּלָלִים נִשְׁאָרִים אוֹתָם כְּלָלִים. מָה שֶׁמִּשְׁתַּנֶּה הוּא הַזְּמַן, הַמָּקוֹם, וְהָאֲנָשִׁים שֶׁאֲלֵיהֶם מְדַבְּרִים.

כָּךְ גַּם הַדִּבְּרוֹת. כְּשֶׁמַּשְׁוִים אֶת סֵפֶר שְׁמוֹת (פֶּרֶק כ׳) לְחֻמַּשׁ דְּבָרִים (פֶּרֶק ה׳) מְגַלִּים:
שֵׁשׁ דִּבְּרוֹת כְּתוּבוֹת כִּמְעַט מִלָּה בְּמִלָּה אוֹתוֹ דָּבָר: אָנֹכִי, לֹא יִהְיֶה, לֹא תִשָּׂא, לֹא תִּרְצָח, לֹא תִּנְאָף, לֹא תִּגְנֹב.
בְּאַרְבַּע יֵשׁ שִׁנּוּיִים: שַׁבָּת, כִּבּוּד הוֹרִים, לֹא תַעֲנֶה וְלֹא תַחְמֹד.

💡 הָאַרְבַּע שֶׁהִשְׁתַּנּוּ
שַׁבָּת: בִּשְׁמוֹת ׳זָכוֹר אֶת יוֹם הַשַּׁבָּת׳, וּבִדְבָרִים ׳שָׁמוֹר אֶת יוֹם הַשַּׁבָּת׳. גַּם הַטַּעַם הִשְׁתַּנָּה: בִּשְׁמוֹת – בְּרִיאַת הָעוֹלָם, וּבִדְבָרִים – יְצִיאַת מִצְרַיִם.
כִּבּוּד הוֹרִים: בִּדְבָרִים נוֹסַף ׳וּלְמַעַן יִיטַב לָךְ׳.
לֹא תַעֲנֶה: בִּשְׁמוֹת ׳עֵד שָׁקֶר׳, וּבִדְבָרִים ׳עֵד שָׁוְא׳ – שְׁתֵּי מִלִּים שֶׁפֵּרוּשָׁן אֶחָד.
לֹא תַחְמֹד: הַסֵּדֶר הִתְהַפֵּךְ (בִּשְׁמוֹת ׳בֵּית רֵעֶךָ׳ קֹדֶם, וּבִדְבָרִים ׳אֵשֶׁת רֵעֶךָ׳ קֹדֶם), נוֹסַף ׳וְלֹא תִתְאַוֶּה׳, וְנוֹסַף ׳שָׂדֵהוּ׳.
לַמּוֹרֶה: שִׁנּוּיִים שֶׁל אוֹת אַחַת, כְּמוֹ ׳וְלֹא תִּנְאָף׳ בִּמְקוֹם ׳לֹא תִּנְאָף׳, אֵינָם נֶחְשָׁבִים כָּאן.
📜 רַשִׁ״י: זָכוֹר וְשָׁמוֹר
׳שָׁמוֹר׳ – וּבָרִאשׁוֹנוֹת הוּא אוֹמֵר ׳זָכוֹר׳: שְׁנֵיהֶם בְּדִבּוּר אֶחָד וּבְתֵבָה אַחַת נֶאֶמְרוּ, וּבִשְׁמִיעָה אַחַת נִשְׁמְעוּ.

׳הָרִאשׁוֹנוֹת׳ הֵן הַדִּבְּרוֹת שֶׁבְּסֵפֶר שְׁמוֹת. ה׳ אָמַר ׳זָכוֹר׳ וְ׳שָׁמוֹר׳ בְּבַת אַחַת – דָּבָר שֶׁאָדָם אֵינוֹ יָכוֹל לַעֲשׂוֹת. אֶת זֶה אֲנַחְנוּ שָׁרִים בְּכָל לֵיל שַׁבָּת בְּ׳לְכָה דוֹדִי׳: ׳שָׁמוֹר וְזָכוֹר בְּדִבּוּר אֶחָד, הִשְׁמִיעָנוּ אֵל הַמְּיֻחָד׳.
זָכוֹר – מִצְוַת עֲשֵׂה: לְקַדֵּשׁ אֶת הַשַּׁבָּת בְּכָבוֹד וּבְעֹנֶג. שָׁמוֹר – לֹא תַעֲשֶׂה: לֹא לַעֲשׂוֹת מְלָאכָה.

אִם ה׳ אָמַר הַכֹּל בְּדִבּוּר אֶחָד, וְהַצִּוּוּי לֹא הִשְׁתַּנָּה – לָמָּה הַנֻּסָּח שׁוֹנֶה? שְׁתֵּי סִבּוֹת:
הַשּׁוֹמְעִים שׁוֹנִים: בְּסִינַי כָּל הָעוֹלָם הָיָה שֻׁתָּף לִירִידַת הַתּוֹרָה, וּבְחֻמַּשׁ דְּבָרִים מֹשֶׁה מְדַבֵּר רַק אֶל עַם יִשְׂרָאֵל. לָכֵן כָּאן הַשָּׂכָר מְפֹרָט יוֹתֵר (׳וּלְמַעַן יִיטַב לָךְ׳), וְהַטַּעַם לַשַּׁבָּת הוּא יְצִיאַת מִצְרַיִם, שֶׁשַּׁיֶּכֶת לְעַם יִשְׂרָאֵל.
הַזְּמַן שׁוֹנֶה: הָעָם עוֹמֵד לְהִכָּנֵס לְאֶרֶץ יִשְׂרָאֵל, וּלְכָל אֶחָד תִּהְיֶה שָׁם נַחֲלָה – בַּיִת וְשָׂדֶה. לָכֵן נוֹסַף ׳שָׂדֵהוּ׳, וְהַבַּיִת נִכְתַּב יַחַד עִם הַשָּׂדֶה.

💡 מִלּוֹן קָטָן לַדִּבְּרוֹת
׳פָּנִים בְּפָנִים׳ – הַמּוֹכֵר עַצְמוֹ מְדַבֵּר עִמָּכֶם; מֹשֶׁה אֵינוֹ סַרְסוּר.
׳עַל פָּנָי׳ – בְּכָל מָקוֹם אֲשֶׁר אֲנִי שָׁם.
׳לַשָּׁוְא׳ – לְחִנָּם: שְׁבוּעַת חִנָּם.
׳וְלֹא תִתְאַוֶּה׳ – אַף הוּא לְשׁוֹן חֶמְדָּה.
׳מִכּוּר הַבַּרְזֶל׳ – כּוּר הוּא כְּלִי שֶׁמְּזַקְּקִים בּוֹ אֶת הַזָּהָב.
׳סָמֶל׳ – צוּרָה.
💛 אַהֲבַת הַשֵּׁם
ה׳ מְדַבֵּר אֶל כָּל דּוֹר בַּדֶּרֶךְ שֶׁהוּא יָכוֹל לִשְׁמֹעַ. כְּמוֹ אַבָּא שֶׁמַּסְבִּיר לְכָל אֶחָד מִיְּלָדָיו בַּדֶּרֶךְ שֶׁלּוֹ – כָּךְ ה׳ מַסְבִּיר לָנוּ אֶת הַתּוֹרָה, כִּי הוּא רוֹצֶה שֶׁנָּבִין אוֹתָהּ וְנֹאהַב אוֹתָהּ 💛

🧠הִשָּׁמֶר לְךָ פֶּן תִּשְׁכַּח – לָמָּה הָיָה מַעֲמַד הַר סִינַי?

ט רַק הִשָּׁמֶר לְךָ וּשְׁמֹר נַפְשְׁךָ מְאֹד פֶּן תִּשְׁכַּח אֶת הַדְּבָרִים אֲשֶׁר רָאוּ עֵינֶיךָ וּפֶן יָסוּרוּ מִלְּבָבְךָ כֹּל יְמֵי חַיֶּיךָ וְהוֹדַעְתָּם לְבָנֶיךָ וְלִבְנֵי בָנֶיךָ י יוֹם אֲשֶׁר עָמַדְתָּ לִפְנֵי ה׳ אֱלֹהֶיךָ בְּחֹרֵב בֶּאֱמֹר ה׳ אֵלַי הַקְהֶל לִי אֶת הָעָם וְאַשְׁמִעֵם אֶת דְּבָרָי אֲשֶׁר יִלְמְדוּן לְיִרְאָה אֹתִי כָּל הַיָּמִים אֲשֶׁר הֵם חַיִּים עַל הָאֲדָמָה וְאֶת בְּנֵיהֶם יְלַמֵּדוּן

עוֹד לִפְנֵי הַדִּבְּרוֹת, בְּפֶרֶק ד׳, מֹשֶׁה מַזְהִיר: ׳רַק הִשָּׁמֶר לְךָ וּשְׁמֹר נַפְשְׁךָ מְאֹד פֶּן תִּשְׁכַּח אֶת הַדְּבָרִים אֲשֶׁר רָאוּ עֵינֶיךָ׳.
מָה אָסוּר לִשְׁכֹּחַ? אֶת מַעֲמַד הַר סִינַי – אֶת הַיּוֹם שֶׁבּוֹ עָמַדְנוּ לִפְנֵי ה׳ בְּחֹרֵב.

📜 רַשִׁ״י
׳רַק הִשָּׁמֶר לְךָ... פֶּן תִּשְׁכַּח׳ – אָז, כְּשֶׁלֹּא תִּשְׁכְּחוּ אוֹתָם וְתַעֲשׂוּם עַל אֲמִתָּתָם, תֵּחָשְׁבוּ חֲכָמִים וּנְבוֹנִים; וְאִם תְּעַוְּתוּ אוֹתָם מִתּוֹךְ שִׁכְחָה, תֵּחָשְׁבוּ שׁוֹטִים.

וְלֹא מַסְפִּיק לִזְכֹּר לְבַד: ׳וְהוֹדַעְתָּם לְבָנֶיךָ וְלִבְנֵי בָנֶיךָ׳. כָּל אַבָּא מְסַפֵּר לִילָדָיו, וְהֵם לִילְדֵיהֶם – וְכָךְ הַזִּכָּרוֹן עוֹבֵר מִדּוֹר לְדוֹר וְאֵינוֹ נִפְסָק. בִּזְכוּת הַשַּׁרְשֶׁרֶת הַזּוֹ אֲנַחְנוּ יוֹדְעִים הַיּוֹם מָה קָרָה בְּסִינַי.

📜 רַשִׁ״י
׳יִלְמְדוּן׳ – יֵלְפוּן, לְעַצְמָם.
׳יְלַמֵּדוּן׳ – יְאַלְּפוּן, לַאֲחֵרִים.

כִּמְעַט אוֹתָהּ מִלָּה, פַּעַם ׳לִלְמֹד׳ וּפַעַם ׳לְלַמֵּד׳: מִי שֶׁלּוֹמֵד אֶת הַתּוֹרָה, צָרִיךְ גַּם לְלַמֵּד אוֹתָהּ אֶת בָּנָיו.

בְּפֶרֶק ד׳ מֹשֶׁה מְגַלֶּה גַּם אֶת הַמַּטָּרוֹת שֶׁל הַמַּעֲמָד:
יִרְאַת ה׳: ׳וְאַשְׁמִעֵם אֶת דְּבָרָי אֲשֶׁר יִלְמְדוּן לְיִרְאָה אֹתִי כָּל הַיָּמִים׳ (פָּסוּק י׳).
הֲכָנָה לַחַיִּים בָּאָרֶץ: ׳לְלַמֵּד אֶתְכֶם חֻקִּים וּמִשְׁפָּטִים לַעֲשֹׂתְכֶם אֹתָם בָּאָרֶץ׳ (פָּסוּק י״ד).
לִהְיוֹת עַם ה׳: ה׳ הוֹצִיא אוֹתָנוּ ׳מִכּוּר הַבַּרְזֶל מִמִּצְרָיִם, לִהְיוֹת לוֹ לְעַם נַחֲלָה׳ (פָּסוּק כ׳).

׳וְאֹתִי צִוָּה ה׳ בָּעֵת הַהִוא לְלַמֵּד אֶתְכֶם׳ – תּוֹרָה שֶׁבְּעַל פֶּה.
׳מִכּוּר׳ – הוּא כְּלִי שֶׁמְּזַקְּקִים בּוֹ אֶת הַזָּהָב.

בְּסִינַי מֹשֶׁה קִבֵּל גַּם אֶת הַהֶסְבֵּרִים שֶׁל הַמִּצְווֹת, וְאוֹתָם לִמֵּד אֶת הָעָם.
וְהַכּוּר: מְחַמְּמִים בּוֹ אֶת הַזָּהָב בְּאֵשׁ חֲזָקָה, עַד שֶׁכָּל הַלִּכְלוּךְ נִפְרָד מִמֶּנּוּ וְנִשְׁאָר זָהָב טָהוֹר. אֶפְשָׁר לְהָבִין שֶׁגַּם הַשִּׁעְבּוּד בְּמִצְרַיִם הָיָה כְּמוֹ אֵשׁ, וּמִמֶּנּוּ יָצָא עַם יִשְׂרָאֵל כְּמוֹ זָהָב – עַם שֶׁל ה׳.

וְעוֹד דָּבָר: בָּאָרֶץ יִפְגְּשׁוּ בְּנֵי יִשְׂרָאֵל עַמִּים שֶׁעוֹבְדִים עֲבוֹדָה זָרָה. הַקֶּשֶׁר הֶחָזָק לַה׳ שֶׁנִּבְנָה בְּסִינַי, וְהַמִּצְווֹת שֶׁקִּבְּלוּ שָׁם, יַעַזְרוּ לָהֶם שֶׁלֹּא לִלְמֹד מִמַּעֲשֵׂיהֶם.

מֵאָז שֶׁנִּבְרָא הָאָדָם, אַף עַם לֹא שָׁמַע אֶת קוֹל ה׳ כְּמוֹ שֶׁשָּׁמְעוּ בְּנֵי יִשְׂרָאֵל. וְאַף אֱלוֹהַּ לֹא בָּא לָקַחַת לוֹ עַם מִתּוֹךְ עַם אַחֵר, כְּמוֹ שֶׁה׳ עָשָׂה לָנוּ בְּמִצְרַיִם.

׳אַתָּה הָרְאֵתָ לָדַעַת׳ – כְּשֶׁנָּתַן הַקָּדוֹשׁ בָּרוּךְ הוּא אֶת הַתּוֹרָה, פָּתַח לָהֶם שִׁבְעָה רְקִיעִים, וּכְשֵׁם שֶׁקָּרַע אֶת הָעֶלְיוֹנִים כָּךְ קָרַע אֶת הַתַּחְתּוֹנִים, וְרָאוּ שֶׁהוּא יְחִידִי.

וּמִכָּאן הַפָּסוּק שֶׁלּוֹמְדִים בְּעַל פֶּה: ׳וְיָדַעְתָּ הַיּוֹם וַהֲשֵׁבֹתָ אֶל לְבָבֶךָ כִּי ה׳ הוּא הָאֱלֹהִים בַּשָּׁמַיִם מִמַּעַל וְעַל הָאָרֶץ מִתָּחַת אֵין עוֹד׳.
לֹא מַסְפִּיק לָדַעַת בָּרֹאשׁ – צָרִיךְ לְהָשִׁיב אֶת זֶה אֶל הַלֵּב, שֶׁהָאֱמוּנָה תִּהְיֶה מֻרְגֶּשֶׁת בַּלֵּב וְתַשְׁפִּיעַ עַל הַמַּעֲשִׂים.

✨ לְסִכּוּם

  • עֲשֶׂרֶת הַדִּבְּרוֹת נִקְרָאוֹת ׳דִּבְּרוֹת׳ כִּי ה׳ עַצְמוֹ דִּבֵּר אוֹתָן אֶל כָּל הָעָם, פָּנִים בְּפָנִים. מֹשֶׁה הוּא רַק הַשָּׁלִיחַ, וְלָכֵן מִתְפַּלְּלִים רַק אֶל ה׳.
  • בְּסִינַי כָּרַת ה׳ בְּרִית עִם עַם יִשְׂרָאֵל – גַּם עִם הַדּוֹרוֹת הַבָּאִים, גַּם אִתָּנוּ. שָׁמְעוּ שָׁם קוֹל בְּלִי תְּמוּנָה, וְלָכֵן אָסוּר לַעֲשׂוֹת פֶּסֶל.
  • אָנֹכִי – הָאֱמוּנָה בַּה׳ – הִיא ׳יְסוֹד הַיְּסוֹדוֹת׳, וְעָלֶיהָ עוֹמֶדֶת כָּל הַתּוֹרָה.
  • שְׁנֵי לוּחוֹת: הָרִאשׁוֹן בֵּין אָדָם לַמָּקוֹם, וְהַשֵּׁנִי בֵּין אָדָם לַחֲבֵרוֹ – בַּמַּעֲשֶׂה, בַּדִּבּוּר וּבַמַּחֲשָׁבָה. בַּדִּבְּרוֹת שָׁלֹשׁ מִצְווֹת עֲשֵׂה וְשֶׁבַע מִצְווֹת לֹא תַעֲשֶׂה.
  • בְּחֻמַּשׁ דְּבָרִים שֵׁשׁ דִּבְּרוֹת זֵהוֹת וְאַרְבַּע שׁוֹנוֹת: שָׁמוֹר, וּלְמַעַן יִיטַב לָךְ, עֵד שָׁוְא, וְלֹא תִתְאַוֶּה. הַצִּוּוּי אֶחָד – רַק הַשּׁוֹמְעִים וְהַזְּמַן הִשְׁתַּנּוּ.
  • אָסוּר לִשְׁכֹּחַ אֶת מַעֲמַד הַר סִינַי, וְצָרִיךְ לְסַפֵּר עָלָיו מִדּוֹר לְדוֹר. וְכָל זֶה ׳תַּחַת אֲשֶׁר אָהַב׳: ה׳ אוֹהֵב אוֹתָנוּ, וְלֹא יַעֲזֹב אוֹתָנוּ לְעוֹלָם.

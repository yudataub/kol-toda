עָרְלָה וְנֶטַע רְבָעִי

🍽️ הִלְכוֹת כַּשְׁרוּת · פֶּרֶק ב׳

🌳עָרְלָה – שָׁלֹשׁ שָׁנִים מְחַכִּים

פֵּרוֹת שֶׁגָּדְלוּ בְּשָׁלֹשׁ הַשָּׁנִים הָרִאשׁוֹנוֹת שֶׁל עֵץ נִקְרָאִים "עָרְלָה". אָסוּר לֶאֱכֹל אוֹתָם, וְגַם אָסוּר לֵהָנוֹת מֵהֶם: לֹא לְהָרִיחַ, לֹא לִמְכֹּר וְלֹא לָתֵת לְאַחֵר. הָאִסּוּר חָל גַּם עַל עֵץ שֶׁנָּטַע גּוֹי.
הָעָלִים, הַפְּרָחִים וְהָעֵץ עַצְמוֹ – מֻתָּרִים. וְאֶתְרוֹג שֶׁל עָרְלָה פָּסוּל לְסֻכּוֹת.
💛 אַהֲבַת הַשֵּׁם: אֲנַחְנוּ רוֹאִים אֶת הַפֵּרוֹת גְּדֵלִים – וּמִתְאַפְּקִים, כִּי ה׳ בִּקֵּשׁ. חֲכָמִים אָמְרוּ שֶׁאָדָם הָרִאשׁוֹן לֹא הִתְאַפֵּק אֲפִלּוּ שָׁעָה אַחַת, וּבָנָיו מְחַכִּים לָעָרְלָה שָׁלֹשׁ שָׁנִים.
📖 פֶּרֶק ב׳, הֲלָכוֹת א׳–ג׳

🍇טַעַם הָעָרְלָה וְנֶטַע רְבָעִי

בְּשָׁלֹשׁ הַשָּׁנִים הָרִאשׁוֹנוֹת הַפֵּרוֹת מְעַטִּים וְלֹא מְשֻׁבָּחִים. ה׳ רָצָה שֶׁנַּתְחִיל לֶאֱכֹל בִּקְדֻשָּׁה וּבְהַלֵּל לַה׳ – בַּשָּׁנָה הָרְבִיעִית, כְּשֶׁהַפֵּרוֹת רַבִּים וְטוֹבִים. וְעוֹד לוֹמְדִים מִכָּךְ לְהִתְאַפֵּק – תְּכוּנָה חֲשׁוּבָה מְאוֹד לְהַצְלָחָה בַּחַיִּים.
פֵּרוֹת הַשָּׁנָה הָרְבִיעִית – "נֶטַע רְבָעִי" – הָיוּ אוֹכְלִים בִּירוּשָׁלַיִם, וְאֵין מַפְרִישִׁים מֵהֶם תְּרוּמוֹת וּמַעַשְׂרוֹת. הַיּוֹם פּוֹדִים אוֹתָם עַל מַטְבֵּעַ שֶׁשָּׁוֶה פְּרוּטָה, בִּבְרָכָה "עַל פִּדְיוֹן רְבָעִי", וְאָז מֻתָּר לֶאֱכֹל אוֹתָם. אֶת הַמַּטְבֵּעַ מְאַבְּדִים.
💛 אַהֲבַת הַשֵּׁם: הַהַמְתָּנָה מְלַמֶּדֶת אוֹתָנוּ שֶׁהַטּוֹב הָאֲמִתִּי שָׁוֶה לְחַכּוֹת לוֹ. מִי שֶׁמִּתְאַפֵּק לְמַעַן ה׳ – טוֹעֵם אַחַר כָּךְ פֵּרוֹת מְלֵאֵי בְּרָכָה.
📖 פֶּרֶק ב׳, הֲלָכוֹת ב׳, י׳; פֶּרֶק ח׳, הֲלָכָה ד׳

🍌רַק עֵץ פְּרִי

דִּין עָרְלָה חָל רַק עַל עֵץ, שֶׁהַגֶּזַע שֶׁלּוֹ נִשְׁאָר וּמִתְחַזֵּק מִשָּׁנָה לְשָׁנָה. עַל יְרָקוֹת אֵין דִּין עָרְלָה. הַבָּנָנָה גְּבוֹהָה כְּמוֹ עֵץ, אֲבָל בְּכָל שָׁנָה הִיא נוֹבֶלֶת וְחוֹזֶרֶת לִצְמֹחַ – וְלָכֵן הִיא נֶחְשֶׁבֶת יָרָק, וּמְבָרְכִים עָלֶיהָ "בּוֹרֵא פְּרִי הָאֲדָמָה".
גַּם חֲצִילִים וּפִלְפְּלִים אֵינָם עֵצִים לְעִנְיַן עָרְלָה, כִּי הֵם נוֹתְנִים פְּרִי כְּבָר בַּשָּׁנָה הָרִאשׁוֹנָה.
💛 אַהֲבַת הַשֵּׁם: ה׳ הִבְטִיחַ: "לְהוֹסִיף לָכֶם תְּבוּאָתוֹ" – מִי שֶׁשּׁוֹמֵר עָרְלָה זוֹכֶה לִבְרָכָה בַּפֵּרוֹת. כָּל הַמְתָּנָה לְמַעַן ה׳ מְבִיאָה טוֹב גָּדוֹל יוֹתֵר.
📖 פֶּרֶק ב׳, הֲלָכָה ח׳

🏪קוֹנִים עִם הֶכְשֵׁר

בְּאֶרֶץ יִשְׂרָאֵל צָרִיךְ לִדְאֹג שֶׁהַפֵּרוֹת וְהַיְּרָקוֹת שֶׁקּוֹנִים אֵינָם עָרְלָה, וְשֶׁהִפְרִישׁוּ מֵהֶם תְּרוּמוֹת וּמַעַשְׂרוֹת. לָכֵן קוֹנִים בַּחֲנוּת שֶׁיֵּשׁ לָהּ תְּעוּדַת כַּשְׁרוּת – כְּשֶׁאֶפְשָׁר, חוֹבָה לָצֵאת מִן הַסָּפֵק.
גַּם בְּמוּצְרֵי דָּגָן – קֶמַח, לֶחֶם וּפְתִיתִים – צָרִיךְ הֶכְשֵׁר שֶׁמַּקְפִּיד עַל אִסּוּר "חָדָשׁ", שֶׁעָלָיו נִלְמַד בָּרָמָה הַבָּאָה.
💛 אַהֲבַת הַשֵּׁם: מִצְוָה לְחַזֵּק אֶת הַמְּדַקְדְּקִים בְּכַשְׁרוּת. כְּשֶׁאֲנַחְנוּ קוֹנִים בְּהֶכְשֵׁר, אֲנַחְנוּ שׁוֹמְרִים עַל מִצְווֹת הָאָרֶץ – גַּם בְּלִי לִהְיוֹת חַקְלָאִים.
📖 פֶּרֶק ב׳, הֲלָכָה י״א; פֶּרֶק א׳, הֲלָכָה ו׳

🧠 בְּדֹק אֶת עַצְמְךָ

לָמָּה, לְפִי חֲכָמִים, צִוְּתָה הַתּוֹרָה לְחַכּוֹת שָׁלֹשׁ שָׁנִים לִפְנֵי שֶׁאוֹכְלִים מִפְּרִי הָעֵץ?
💡 פֵּרוֹת הַשָּׁנִים הָרִאשׁוֹנוֹת מְעַטִּים, וְאֵין רָאוּי לְהַלֵּל בָּהֶם אֶת ה׳. מְחַכִּים לַשָּׁנָה הָרְבִיעִית, וּבַהַמְתָּנָה לוֹמְדִים לְהִתְאַפֵּק – מַה שֶּׁאָדָם הָרִאשׁוֹן לֹא עָשָׂה.
הִגִּיעָה הַשָּׁנָה הָרְבִיעִית שֶׁל עֵץ הָרִמּוֹנִים. מָה עוֹשִׂים כְּדֵי לֶאֱכֹל מִפֵּרוֹתָיו בִּזְמַנֵּנוּ?
💡 פֵּרוֹת "נֶטַע רְבָעִי" קְדוֹשִׁים, וְאֵין מַפְרִישִׁים מֵהֶם תְּרוּמוֹת וּמַעַשְׂרוֹת. הַיּוֹם, כְּשֶׁאִי אֶפְשָׁר לֶאֱכֹל אוֹתָם בִּירוּשָׁלַיִם, פּוֹדִים אוֹתָם עַל שְׁוֵה פְּרוּטָה בִּבְרָכָה "עַל פִּדְיוֹן רְבָעִי", וּמְאַבְּדִים אֶת הַפְּרוּטָה.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

📜הַמִּצְוָה הָרִאשׁוֹנָה

בְּרִית מִילָה הִיא הַמִּצְוָה הָרִאשׁוֹנָה שֶׁנִּצְטַוָּה אַבְרָהָם אָבִינוּ, הַיְּהוּדִי הָרִאשׁוֹן. וְהִיא גַּם הַמִּצְוָה הָרִאשׁוֹנָה שֶׁכָּל תִּינוֹק יְהוּדִי מְקַיֵּם – כְּשֶׁהוּא בֶּן שְׁמוֹנָה יָמִים.
הַבְּרִית הִיא אוֹת שֶׁל קֶשֶׁר נִצְחִי בֵּין עַם יִשְׂרָאֵל לַה׳ – קֶשֶׁר שֶׁנֶּחְתַּם בְּגוּפֵנוּ. כְּשֶׁצִּוָּה ה׳ אֶת אַבְרָהָם עַל הַמִּילָה, הוּא הִבְטִיחַ לוֹ גַּם אֶת אֶרֶץ יִשְׂרָאֵל וְגַם זֶרַע רַב.
💛 אַהֲבַת הַשֵּׁם: בַּבְּרִית אָנוּ אוֹמְרִים לַה׳: אֲנַחְנוּ שֶׁלְּךָ, וְנִדְבַּק בְּךָ תָּמִיד – גַּם כְּשֶׁקָּשֶׁה. זוֹ מִצְוָה שֶׁל אַהֲבָה וּמְסִירוּת.
📖 פֶּרֶק ח׳, הֲלָכָה א׳

🌍בְּנֵי אַבְרָהָם וְאֶרֶץ יִשְׂרָאֵל

מִצְוַת הַמִּילָה מְיֻחֶדֶת לְיִשְׂרָאֵל – לְזַרְעוֹ שֶׁל יַעֲקֹב. כָּתוּב: ״כִּי בְיִצְחָק יִקָּרֵא לְךָ זָרַע״ – בְּיִצְחָק וְלֹא בְּיִשְׁמָעֵאל, וְגַם לֹא בְּכָל זַרְעוֹ שֶׁל יִצְחָק, אֶלָּא רַק בְּנֵי יַעֲקֹב.
יֵשׁ קֶשֶׁר מְיֻחָד בֵּין הַבְּרִית לְאֶרֶץ יִשְׂרָאֵל. לָכֵן בִּימֵי יְהוֹשֻׁעַ, לִפְנֵי שֶׁהִתְחִילוּ לִכְבֹּשׁ אֶת הָאָרֶץ, מָלוּ כָּל מִי שֶׁעוֹד לֹא נִמּוֹל. בַּבְּרִית מִתְגַּלֶּה שֶׁגַּם לַגּוּף יֵשׁ עֵרֶךְ קָדוֹשׁ, וּבְיִשּׁוּב הָאָרֶץ מִתְגַּלֶּה שֶׁיֵּשׁ עֵרֶךְ לְכָל עֲבוֹדָה בְּבִנְיַן הָאָרֶץ.
💛 אַהֲבַת הַשֵּׁם: ה׳ בָּחַר בָּנוּ, נָתַן לָנוּ אֶת בְּרִיתוֹ וְנָתַן לָנוּ אֶת אַרְצוֹ – וְאָנוּ שׁוֹמְרִים עַל שְׁתֵּיהֶן בְּאַהֲבָה.
📖 פֶּרֶק ח׳, הֲלָכוֹת א׳, י״ב

✡️🔑 מוּשָּׂג: בְּרִית

״מִצְוָה זוֹ מְסַמֶּלֶת יוֹתֵר מִכָּל שְׁאָר הַמִּצְווֹת, אֶת הַקֶּשֶׁר הַנִּצְחִי שֶׁבֵּין עַם יִשְׂרָאֵל לֵאלוֹקָיו, קֶשֶׁר שֶׁנֶּחְתַּם בִּבְשָׂרֵנוּ״.
📖 בְּרִית – הַבְטָחָה וְקֶשֶׁר שֶׁל נֶאֱמָנוּת בֵּין שְׁנַיִם, לָעַד. ״וַהֲקִמֹתִי אֶת בְּרִיתִי בֵּינִי וּבֵינֶךָ... לִבְרִית עוֹלָם״ – ה׳ כָּרַת בְּרִית עִם אַבְרָהָם וְעִם זַרְעוֹ.
💛 אַהֲבַת הַשֵּׁם: זוֹ ״הַמִּצְוָה הָרִאשׁוֹנָה שֶׁכָּל יְהוּדִי וִיהוּדִי מְקַיֵּם, בִּהְיוֹתוֹ בֶּן שְׁמוֹנָה יָמִים״ – עוֹד לִפְנֵי שֶׁיָּדַעְנוּ לְדַבֵּר, ה׳ כְּבָר קָשַׁר אוֹתָנוּ אֵלָיו. 🧒 וְאַתָּה? כְּשֶׁיֵּשׁ בְּרִית בַּמִּשְׁפָּחָה – בּוֹא בְּבִגְדֵי שַׁבָּת, ״לִכְבוֹד הַמִּצְוָה״.
📖 פֶּרֶק ח׳, הֲלָכָה א׳

⭐גְּדוֹלָה מִכָּל הַמִּצְווֹת

הַהֲלָכָה הִיא: ״מִצְוַת עֲשֵׂה לָאָב לָמוּל אֶת בְּנוֹ, וּגְדוֹלָה מִצְוָה זוֹ מִשְּׁאָר מִצְווֹת עֲשֵׂה״. לָכֵן כִּמְעַט כָּל הַיְּהוּדִים מְקַיְּמִים אוֹתָהּ בְּהִדּוּר.
יְהוּדִי הוּא מִי שֶׁאִמּוֹ יְהוּדִיָּה, אוֹ מִי שֶׁהִתְגַּיֵּר כַּהֲלָכָה. גַּם מִי שֶׁלֹּא זָכָה לִבְרִית – הוּא יְהוּדִי גָּמוּר, כִּי הַיַּהֲדוּת מַתְחִילָה בַּנְּשָׁמָה.
💛 אַהֲבַת הַשֵּׁם: הַבְּרִית מְגַלָּה אֶת הַנְּשָׁמָה הַקְּדוֹשָׁה שֶׁיֵּשׁ בְּכָל יְהוּדִי. כָּל יְהוּדִי הוּא בֵּן אָהוּב שֶׁל ה׳.
📖 פֶּרֶק ח׳, הֲלָכָה ב׳

👼📜 לְשׁוֹן חֲכָמִים: מַלְאַךְ הַבְּרִית

״הַכִּסֵּא שֶׁהוּכַן לִכְבוֹד אֵלִיָּהוּ הַנָּבִיא מַלְאַךְ הַבְּרִית״.
📖 מַלְאָךְ – שָׁלִיחַ שֶׁל ה׳. אֵלִיָּהוּ הַנָּבִיא נִקְרָא ״מַלְאַךְ הַבְּרִית״, וְלִכְבוֹדוֹ מְכִינִים כִּסֵּא בְּכָל בְּרִית מִילָה.
💛 אַהֲבַת הַשֵּׁם: בְּרִית הַמִּילָה ״מְבַטֵּאת יוֹתֵר מִכֹּל אֶת הִשְׁתַּיְּכוּתֵנוּ לְעַם יִשְׂרָאֵל״ – וְכָל בְּרִית הִיא שִׂמְחָה גְּדוֹלָה לְכָל הָעָם. 🧒 וְאַתָּה? בַּבְּרִית הַבָּאָה – בָּרֵךְ אֶת הַתִּינוֹק שֶׁיִּגְדַּל לְתוֹרָה וּלְמַעֲשִׂים טוֹבִים.
📖 פֶּרֶק ח׳, הֲלָכוֹת ב׳, י״א

8️⃣בַּיּוֹם הַשְּׁמִינִי – גַּם בְּשַׁבָּת

״וּבַיּוֹם הַשְּׁמִינִי יִמּוֹל״ – עוֹשִׂים אֶת הַבְּרִית בַּיּוֹם הַשְּׁמִינִי, לֹא לִפְנֵי וְלֹא אַחֲרֵי. גַּם כְּשֶׁהַיּוֹם הַשְּׁמִינִי חָל בְּשַׁבָּת – עוֹשִׂים אֶת הַבְּרִית בַּשַּׁבָּת עַצְמָהּ. אֶת כָּל מָה שֶׁצָּרִיךְ לַבְּרִית מְכִינִים לִפְנֵי שַׁבָּת.
אִם הַבְּרִית דּוֹחָה אֲפִלּוּ שַׁבָּת – בְּוַדַּאי שֶׁלֹּא דּוֹחִים אוֹתָהּ כְּדֵי שֶׁקְּרוֹבִים יַסְפִּיקוּ לָבוֹא. וַאֲפִלּוּ אִם הָאָב בְּחוּץ לָאָרֶץ – עוֹשִׂים אוֹתָהּ בַּיּוֹם הַשְּׁמִינִי בִּלְעָדָיו.
💛 אַהֲבַת הַשֵּׁם: הַזְּרִיזוּת בַּמִּצְוָה מַרְאָה כַּמָּה הִיא יְקָרָה לָנוּ. אָנוּ לֹא מְחַכִּים – אָנוּ רָצִים לְקַיֵּם אֶת מִצְוַת ה׳.
📖 פֶּרֶק ח׳, הֲלָכָה ד׳

⏳בְּרִיאוּת הַתִּינוֹק קוֹדֶמֶת

הַסִּבָּה הָעִקָּרִית לִדְחוֹת בְּרִית הִיא מַחֲלָה שֶׁל הַתִּינוֹק. אִם יֵשׁ אֲפִלּוּ צֵל שֶׁל חֲשָׁשׁ לְסַכָּנָה – דּוֹחִים אֶת הַבְּרִית עַד שֶׁהַתִּינוֹק יַבְרִיא לְגַמְרֵי. אֶת הַבְּרִית אֶפְשָׁר לַעֲשׂוֹת אַחַר כָּךְ, אֲבָל נֶפֶשׁ אַחַת מִיִּשְׂרָאֵל אִי אֶפְשָׁר לְהַחְזִיר.
רַק בְּרִית בִּזְמַנָּהּ – בַּיּוֹם הַשְּׁמִינִי – דּוֹחָה שַׁבָּת וְיוֹם טוֹב. בְּרִית שֶׁנִּדְחֲתָה אֵינָהּ דּוֹחָה שַׁבָּת: אִם הַיּוֹם הָרִאשׁוֹן שֶׁאֶפְשָׁר לָמוּל הוּא שַׁבָּת – מָלִים בְּיוֹם רִאשׁוֹן.
💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לָנוּ אֶת הַמִּצְווֹת כְּדֵי שֶׁנִּחְיֶה בָּהֶן. הַשְּׁמִירָה עַל הַתִּינוֹק הִיא בְּעַצְמָהּ מִצְוָה גְּדוֹלָה.
📖 פֶּרֶק ח׳, הֲלָכָה ד׳

🔀שְׁנֵי מִנְהָגִים בַּבְּרִית – וּשְׁנֵיהֶם כַּהֲלָכָה

בְּרִית שֶׁנִּדְחֲתָה, וְהַתִּינוֹק הִבְרִיא בְּיוֹם חֲמִישִׁי אוֹ שִׁשִּׁי: רַבִּים מִיּוֹצְאֵי סְפָרַד נוֹהֲגִים שֶׁלֹּא לָמוּל אָז, אֶלָּא לְחַכּוֹת לְיוֹם רִאשׁוֹן – כְּדֵי שֶׁלֹּא יִצְטָרְכוּ לְחַלֵּל עָלָיו שַׁבָּת. וְיוֹצְאֵי אַשְׁכְּנַז וְתֵימָן וְחֵלֶק מִיּוֹצְאֵי סְפָרַד מָלִים מִיָּד, כִּי מִצְוָה לְהִזְדָּרֵז. וּמָתַי בַּיּוֹם? לְדַעַת הַרְבֵּה פּוֹסְקִים עָדִיף בַּבֹּקֶר, כִּי זְרִיזוּת קוֹדֶמֶת לְהִדּוּר. וְיֵשׁ אוֹמְרִים, וְכָךְ דַּעַת הָרַבִּי מִלּוּבָּאוִיטְשׁ, שֶׁמֻּתָּר לִקְבֹּעַ אַחַר הַצָּהֳרַיִם, כְּדֵי שֶׁיָּבוֹאוּ יוֹתֵר אֲנָשִׁים וְתִרְבֶּה הַשִּׂמְחָה.
מִי מְבָרֵךְ רִאשׁוֹן? לְמִנְהַג רֹב יִשְׂרָאֵל – כָּל יוֹצְאֵי אַשְׁכְּנַז וּצְפוֹן אַפְרִיקָה וְרֹב יוֹצְאֵי סְפָרַד – הַמּוֹהֵל מְבָרֵךְ ״עַל הַמִּילָה״ וּמָל, וְאָז הָאָב מְבָרֵךְ. וּבִקְהִלּוֹת מְעַטּוֹת מִיּוֹצְאֵי סְפָרַד הָאָב מְבָרֵךְ רִאשׁוֹן ״לְהַכְנִיסוֹ״, וְאַחֲרָיו הַמּוֹהֵל. כָּל אֶחָד מַמְשִׁיךְ בְּמִנְהַג אֲבוֹתָיו.
💛 אַהֲבַת הַשֵּׁם: כָּל הַמִּנְהָגִים בָּאִים מֵאַהֲבָה לַמִּצְוָה: זֶה מְמַהֵר כִּי הוּא אוֹהֵב אוֹתָהּ, וְזֶה מְחַכֶּה כְּדֵי לִשְׁמֹר עָלֶיהָ וּלְהַרְבּוֹת בָּהּ שִׂמְחָה.
📖 פֶּרֶק ח׳, הֲלָכוֹת ד׳–ה׳, ט׳

🌅📜 לְשׁוֹן חֲכָמִים: זָרִיז הַמַּקְדִּים לַמִּצְוָה

״הַאִם עָדִיף לְהִזְדָּרֵז לְקַיֵּם אֶת הַבְּרִית בְּבֹקֶר הַיּוֹם הַשְּׁמִינִי, וְלִהְיוֹת זָרִיז הַמַּקְדִּים לַמִּצְוָה״ – ״שְׁאֵלָה עֶקְרוֹנִית זוֹ שַׁיֶּכֶת לְמִצְווֹת רַבּוֹת״.

לִפְנֵי הַבְּרִית מַנִּיחִים אֶת הַתִּינוֹק עַל כִּסֵּא מְיֻחָד, שֶׁהוּכַן לִכְבוֹד אֵלִיָּהוּ הַנָּבִיא – מַלְאַךְ הַבְּרִית. זֶה כִּבּוּד שֶׁנּוֹתְנִים לְאֶחָד הַמֻּזְמָנִים.
הַכִּבּוּד הֶחָשׁוּב בְּיוֹתֵר הוּא סַנְדָּק: הַתִּינוֹק מֻנָּח עַל בִּרְכָּיו בִּשְׁעַת הַבְּרִית, וְהוּא עוֹזֵר מְעַט לַמּוֹהֵל. בְּדֶרֶךְ כְּלָל מְכַבְּדִים בְּכָךְ אֶת הַסַּבָּא אוֹ תַּלְמִיד חָכָם.
💛 אַהֲבַת הַשֵּׁם: אֵלִיָּהוּ הַנָּבִיא נִקְרָא ״מַלְאַךְ הַבְּרִית״. כְּשֶׁאָנוּ מְכִינִים לוֹ כִּסֵּא, אָנוּ מַרְאִים שֶׁהַבְּרִית הִיא מַעֲמָד קָדוֹשׁ וּמְכֻבָּד.

📋כָּל הַכִּבּוּדִים

כְּדַאי לְתַכְנֵן אֶת הַכִּבּוּדִים מֵרֹאשׁ. אִם לֹא – הָאָב הַנִּרְגָּשׁ עָלוּל לְכַבֵּד אֶת מִי שֶׁעוֹמֵד לְיָדוֹ, וְלִשְׁכֹּחַ קְרוֹבִים חֲשׁוּבִים, וְהֵם יֵעָלְבוּ. אַחֲרֵי הַסַּנְדָּק יֵשׁ שְׁלֹשָׁה כִּבּוּדִים: כִּסֵּא שֶׁל אֵלִיָּהוּ, בְּרָכוֹת, וַעֲמִידָה לִבְרָכוֹת – מִי שֶׁמַּחְזִיק אֶת הַתִּינוֹק בִּזְמַן הַבְּרָכוֹת.
לִפְנֵי הַבְּרִית יֵשׁ כִּבּוּד מְיֻחָד לְזוּג נָשׂוּי – ״קְוָאטֶר״: הָאִשָּׁה לוֹקַחַת אֶת הַתִּינוֹק מֵאִמּוֹ וּמוֹסֶרֶת אוֹתוֹ לְבַעְלָהּ, וְהוּא מֵבִיא אוֹתוֹ אֶל אָבִיו. וְאֶפְשָׁר לְכַבֵּד עוֹד אֲנָשִׁים, שֶׁמַּעֲבִירִים אֶת הַתִּינוֹק מִכִּסֵּא שֶׁל אֵלִיָּהוּ עַד הַסַּנְדָּק.
💛 אַהֲבַת הַשֵּׁם: כִּבּוּד הוּא דֶּרֶךְ לְשַׁתֵּף אֶת הָאֲהוּבִים בַּמִּצְוָה. כְּשֶׁאָנוּ חוֹשְׁבִים מֵרֹאשׁ עַל כָּל אֶחָד – אָנוּ שׁוֹמְרִים עַל הַלֵּב שֶׁל כֻּלָּם.

🔀מִנְהֲגֵי הַכִּבּוּדִים וְהַפִּדְיוֹן

סַנְדָּק פַּעֲמַיִם: רַבִּים נוֹהֲגִים שֶׁאָב לֹא מְכַבֵּד אֶת אוֹתוֹ אָדָם לִהְיוֹת סַנְדָּק אֵצֶל שְׁנֵי בָּנָיו, כְּדֵי לְזַכּוֹת עוֹד אֲנָשִׁים בַּכִּבּוּד, שֶׁיֵּשׁ בּוֹ סְגֻלָּה לְעֹשֶׁר וּבְרָכָה. וְיֵשׁ נוֹהֲגִים שֶׁלֹּא לְהִמָּנַע מִכָּךְ. עֲמִידָה לִבְרָכוֹת: לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז וְחֵלֶק מִיּוֹצְאֵי סְפָרַד, ״סַנְדָּק שֵׁנִי״ מַחְזִיק אֶת הַתִּינוֹק בִּזְמַן הַבְּרָכוֹת. וְיֵשׁ מֵהַסְּפָרַדִּים שֶׁהַתִּינוֹק נִשְׁאָר עַל בִּרְכֵּי הַסַּנְדָּק.
בְּפִדְיוֹן הַבֵּן, אַחֲרֵי שֶׁהַכֹּהֵן מְבָרֵךְ עַל הַיַּיִן: לְמִנְהַג הַסְּפָרַדִּים הוּא לוֹקֵחַ הֲדַס, מְבָרֵךְ ״בּוֹרֵא עֲצֵי בְשָׂמִים״ וּמֵרִיחַ. לְמִנְהַג הָאַשְׁכְּנַזִּים לֹא לוֹקְחִים הֲדַס.
💛 אַהֲבַת הַשֵּׁם: מִשְׁפָּחוֹת שׁוֹנוֹת, מִנְהָגִים שׁוֹנִים – וְכֻלָּם מְכַבְּדִים זֶה אֶת זֶה, כִּי כֻּלָּם רוֹצִים לְהַדֵּר בְּמִצְווֹת ה׳.

💺🔑 מוּשָּׂג: כִּסֵּא שֶׁל אֵלִיָּהוּ

״לִפְנֵי הַבְּרִית מְכַבְּדִים אֶת אֶחָד הַמִּשְׁתַּתְּפִים בְּ׳כִסֵּא שֶׁל אֵלִיָּהוּ׳, כְּלוֹמַר בְּהַנָּחַת הַיֶּלֶד עַל הַכִּסֵּא שֶׁהוּכַן לִכְבוֹד אֵלִיָּהוּ הַנָּבִיא מַלְאַךְ הַבְּרִית״.
📖 כִּסֵּא שֶׁל אֵלִיָּהוּ – כִּסֵּא מְיֻחָד שֶׁמְּכִינִים בַּבְּרִית לִכְבוֹד אֵלִיָּהוּ הַנָּבִיא. לֹא ״כּוֹס שֶׁל אֵלִיָּהוּ״ שֶׁבְּלֵיל הַסֵּדֶר.
💛 אַהֲבַת הַשֵּׁם: בַּבְּרִית מְחַלְּקִים כַּמָּה כִּבּוּדִים, ״וּלְכָל אֶחָד מֵהֶם מַעֲלָה מְיֻחֶדֶת״ – כָּל אֶחָד זוֹכֶה לְחֵלֶק בַּשִּׂמְחָה שֶׁל מִצְוַת ה׳. 🧒 וְאַתָּה? בַּבְּרִית הַבָּאָה – חַפֵּשׂ אֶת הַכִּסֵּא הַמְּיֻחָד, וּזְכֹר לִכְבוֹד מִי הוּא עוֹמֵד שָׁם.

🧠 בְּדֹק אֶת עַצְמְךָ

הַתִּינוֹק חָלָה לִפְנֵי הַבְּרִית, וְהָרוֹפֵא חוֹשֵׁשׁ. מָתַי יַעֲשׂוּ אֶת הַבְּרִית?
💡 אִם יֵשׁ אֲפִלּוּ צֵל שֶׁל חֲשָׁשׁ לְסַכָּנָה, מַמְתִּינִים עַד שֶׁהַתִּינוֹק יַבְרִיא לְגַמְרֵי. מִי שֶׁמַּחְמִיר וּמָל תִּינוֹק חוֹלֶה – אֵינוֹ מְהַדֵּר, אֶלָּא מְסַכֵּן אוֹתוֹ.
הַבְּרִית נִדְחֲתָה כִּי הַתִּינוֹק חָלָה, וְהַיּוֹם הָרִאשׁוֹן שֶׁאֶפְשָׁר לָמוּל אוֹתוֹ הוּא שַׁבָּת. מָה עוֹשִׂים?
💡 רַק בְּרִית בִּזְמַנָּהּ, בַּיּוֹם הַשְּׁמִינִי, דּוֹחָה שַׁבָּת וְיוֹם טוֹב. בְּרִית שֶׁנִּדְחֲתָה – עוֹשִׂים בְּיוֹם רִאשׁוֹן, אוֹ אַחֲרֵי הַחַג.
הַסָּבִים גָּרִים רָחוֹק וְיַגִּיעוּ רַק בְּעוֹד שָׁבוּעַ. מָתַי עוֹשִׂים אֶת הַבְּרִית?
💡 אִם הַבְּרִית דּוֹחָה אֲפִלּוּ שַׁבָּת – קַל וָחֹמֶר שֶׁלֹּא דּוֹחִים אוֹתָהּ בִּשְׁבִיל קְרוֹבִים. אֲפִלּוּ כְּשֶׁהָאָב בְּחוּץ לָאָרֶץ, עוֹשִׂים אוֹתָהּ בַּיּוֹם הַשְּׁמִינִי בִּלְעָדָיו.

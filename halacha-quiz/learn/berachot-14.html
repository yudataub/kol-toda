בִּרְכַּת הָרֵיחַ

🍎 הִלְכוֹת בְּרָכוֹת · פֶּרֶק י״ד

👃רֵיחַ טוֹב – הֲנָאָה לַנְּשָׁמָה

חֲכָמִים לָמְדוּ מִן הַפָּסוּק ״כֹּל הַנְּשָׁמָה תְּהַלֵּל יָהּ״: אֵיזֶה דָּבָר הַנְּשָׁמָה נֶהֱנֵית מִמֶּנּוּ וְהַגּוּף לֹא? זֶה הָרֵיחַ! לָכֵן מְבָרְכִים לִפְנֵי שֶׁמְּרִיחִים רֵיחַ טוֹב.
עַל רֵיחַ אֵין בְּרָכָה אַחֲרוֹנָה – כְּשֶׁמַּפְסִיקִים לְהָרִיחַ, לֹא נִשְׁאָר מִן הַהֲנָאָה כְּלוּם.
💛 אַהֲבַת הַשֵּׁם: הָרֵיחַ הוּא מַתָּנָה עֲדִינָה מִה׳. כְּשֶׁאָנוּ שָׂמִים לֵב גַּם לַדְּבָרִים הַקְּטַנִּים וְהַדַּקִּים, הַנְּשָׁמָה מִתְעוֹרֶרֶת לֶאֱהֹב אוֹתוֹ יוֹתֵר.
📖 פֶּרֶק י״ד, הֲלָכָה א׳

🌿אֵיזוֹ בְּרָכָה עַל אֵיזֶה רֵיחַ?

עַל עֵץ אוֹ שִׂיחַ רֵיחָנִי, כְּמוֹ הֲדַס וְרוֹזְמָרִין – ״בּוֹרֵא עֲצֵי בְשָׂמִים״. עַל עֵשֶׂב רַךְ, כְּמוֹ נַרְקִיס – ״בּוֹרֵא עִשְׂבֵי בְשָׂמִים״. עַל פְּרִי שֶׁלּוֹקְחִים כְּדֵי לְהָרִיחַ – ״הַנּוֹתֵן רֵיחַ טוֹב בַּפֵּרוֹת״. עַל שְׁאָר רֵיחוֹת, כְּמוֹ תַּבְלִינִים וְקָפֶה – ״בּוֹרֵא מִינֵי בְשָׂמִים״.
״מִינֵי בְשָׂמִים״ מוֹעִילָה בְּדִיעֲבַד לְכָל רֵיחַ, וְלָכֵן בְּסָפֵק מְבָרְכִים אוֹתָהּ. בַּהַבְדָּלָה נוֹהֲגִים הָאַשְׁכְּנַזִּים לְבָרֵךְ תָּמִיד ״מִינֵי בְשָׂמִים״, וְהַסְּפָרַדִּים – עַל כָּל מִין אֶת בִּרְכָתוֹ.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים נָתְנוּ בְּרָכָה מְיֻחֶדֶת לְכָל סוּג רֵיחַ, כְּדֵי שֶׁנַּכִּיר כָּל מַתָּנָה בִּשְׁמָהּ. כָּךְ הַתּוֹדָה שֶׁלָּנוּ נַעֲשֵׂית מְדֻיֶּקֶת וּשְׁלֵמָה.
📖 פֶּרֶק י״ד, הֲלָכוֹת א׳, ה׳

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

בַּהַבְדָּלָה מְבָרְכִים עַל הַבְּשָׂמִים, וְיֵשׁ בָּזֶה שְׁנֵי מִנְהָגִים. לְמִנְהַג הָאַשְׁכְּנַזִּים מְבָרְכִים תָּמִיד ״בּוֹרֵא מִינֵי בְשָׂמִים״, כִּי רַבִּים אֵינָם יוֹדְעִים מָה הוּא עֵץ וּמָה הוּא עֵשֶׂב, וּבְרָכָה זוֹ מוֹעִילָה לְכָל רֵיחַ. לְמִנְהַג הַסְּפָרַדִּים, כְּשֶׁיּוֹדְעִים מָה הַבֹּשֶׂם, מְבָרְכִים עָלָיו אֶת הַבְּרָכָה הַמְּיֻחֶדֶת לוֹ.
לְמָשָׁל, עַל הֲדַס: לְמִנְהַג הָאַשְׁכְּנַזִּים – ״מִינֵי בְשָׂמִים״; לְמִנְהַג הַסְּפָרַדִּים – ״עֲצֵי בְשָׂמִים״, כִּי הַהֲדַס הוּא עֵץ.
💛 אַהֲבַת הַשֵּׁם: הָאַשְׁכְּנַזִּים נִזְהָרִים שֶׁלֹּא לִטְעוֹת בַּבְּרָכָה, וְהַסְּפָרַדִּים מְדַיְּקִים לְהוֹדוֹת עַל כָּל מִין בִּשְׁמוֹ. שְׁנֵי הַמִּנְהָגִים הֵם דֶּרֶךְ שֶׁל אַהֲבָה וּזְהִירוּת בִּכְבוֹד ה׳.
📖 פֶּרֶק י״ד, הֲלָכוֹת א׳, ה׳

🌺עוֹד דִּינֵי רֵיחַ

עַל בֹּשֶׂם שֶׁקִּבֵּל אֶת רֵיחוֹ מֵעֵצִים – ״עֲצֵי בְשָׂמִים״; עַל בֹּשֶׂם מֵחֳמָרִים מְלָאכוּתִיִּים – ״מִינֵי בְשָׂמִים״. מִי שֶׁמְּצֻנָּן וְלֹא מַרְגִּישׁ רֵיחַ, וּמִי שֶׁבּוֹדֵק בְּשָׂמִים רַק כְּדֵי לִבְחֹר מָה לִקְנוֹת – לֹא מְבָרֵךְ.
דִּבֵּר בֵּין הַבְּרָכָה לַהֲרָחָה – מְבָרֵךְ שׁוּב. וְאֶת הַהֲדַסִּים שֶׁל אַרְבָּעַת הַמִּינִים אָסוּר לְהָרִיחַ בְּסֻכּוֹת, כִּי הֵם הֻקְצוּ לַמִּצְוָה.
💛 אַהֲבַת הַשֵּׁם: הַבְּרָכָה צְמוּדָה לַהֲנָאָה – כְּמוֹ ״תּוֹדָה״ שֶׁאוֹמְרִים בְּרֶגַע שֶׁמְּקַבְּלִים מַתָּנָה. כָּךְ אָנוּ חַיִּים עִם ה׳ בְּכָל רֶגַע.
📖 פֶּרֶק י״ד, הֲלָכוֹת ב׳, ו׳

🧼רֵיחוֹת שֶׁלֹּא מְבָרְכִים עֲלֵיהֶם

מְבָרְכִים רַק עַל רֵיחַ שֶׁנּוֹעַד לֵהָנוֹת מִמֶּנּוּ. לֹא עַל מְטַהֵר אֲוִיר לַשֵּׁרוּתִים, סַבּוֹן אוֹ דֵּאוֹדוֹרַנְט, וְלֹא עַל רֵיחַ שֶׁנִּשְׁאָר בַּבְּגָדִים. גַּם עַל רֵיחַ לֶחֶם חַם לֹא מְבָרְכִים – הוּא רַק מְעוֹרֵר תֵּאָבוֹן.
גִּנָּה שֶׁשָּׁתְלוּ בָּהּ צְמָחִים כְּדֵי שֶׁיָּרִיחוּ אוֹתָם – מְבָרְכִים. פַּרְדֵּס שֶׁנּוֹעַד לְפֵרוֹת – מְבָרְכִים רַק אִם נִגָּשִׁים לְהָרִיחַ פֶּרַח.
💛 אַהֲבַת הַשֵּׁם: הַבְּרָכָה מְלַמֶּדֶת אוֹתָנוּ לַעֲצֹר וְלָשִׂים לֵב. מִי שֶׁעוֹצֵר לְהָרִיחַ פֶּרַח וּמְבָרֵךְ, פּוֹגֵשׁ אֶת ה׳ בְּתוֹךְ הַטֶּבַע.
📖 פֶּרֶק י״ד, הֲלָכוֹת ג׳–ה׳

🧴בֹּשֶׂם מִן הַבַּקְבּוּק

עַל בֹּשֶׂם מְבָרְכִים, כִּי הוּא נוֹעַד לֵהָנוֹת מֵרֵיחוֹ. מִי שֶׁמְּבַשֶּׂמֶת אֶת עַצְמָהּ – מְבָרֶכֶת עַל הָרֵיחַ שֶׁיּוֹצֵא מִבַּקְבּוּק הַבֹּשֶׂם. אֲבָל עַל רֵיחַ בֹּשֶׂם שֶׁנּוֹדֵף מֵאָדָם שֶׁהִתְבַּשֵּׂם – לֹא מְבָרְכִים, כִּי מְקוֹר הָרֵיחַ אֵינוֹ שָׁם.
וְכֵן לֹא מְבָרְכִים עַל רֵיחַ טוֹב שֶׁנּוֹדֵף מִבְּגָדִים שֶׁכִּבְּסוּ בְּאַבְקָה רֵיחָנִית, וְלֹא עַל קְרֶם יָדַיִם – כִּי הֵם נוֹעֲדוּ לָתֵת רֵיחַ בְּדָבָר אַחֵר.
💛 אַהֲבַת הַשֵּׁם: גַּם רֵיחַ עָדִין הוּא מַתָּנָה מֵה׳. כְּשֶׁמְּבָרְכִים עָלָיו, הַהֲנָאָה הַקְּטַנָּה נַעֲשֵׂית רֶגַע שֶׁל תּוֹדָה.
📖 פֶּרֶק י״ד, הֲלָכָה ג׳

🌱נַעֲנָע – כְּשֶׁיֵּשׁ סָפֵק

עֵץ בְּשָׂמִים – גִּבְעוֹלוֹ קָשֶׁה, וְהוּא מִתְקַיֵּם מִשָּׁנָה לְשָׁנָה. עֵשֶׂב – גִּבְעוֹלוֹ רַךְ. עַל נַעֲנָע הִסְתַּפְּקוּ אִם הִיא עֵץ אוֹ עֵשֶׂב, וְלָכֵן עָדִיף לְבָרֵךְ עָלֶיהָ "בּוֹרֵא מִינֵי בְשָׂמִים". וְכֵן עַל צֶמַח רֵיחָנִי שֶׁאֵין דֶּרֶךְ קַלָּה לָדַעַת מָה הוּא.
הַטּוֹעֶה וּמְבָרֵךְ עַל עֵץ "עִשְׂבֵי בְשָׂמִים", אוֹ עַל עֵשֶׂב "עֲצֵי בְשָׂמִים" – לֹא יָצָא. וְעַל תַּבְלִינִים שֶׁלּוֹקְחִים כְּדֵי לְהָרִיחַ – "מִינֵי בְשָׂמִים".
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאָנוּ מְדַקְדְּקִים בַּבְּרָכָה הַנְּכוֹנָה, אָנוּ לוֹמְדִים לְהַכִּיר אֶת הַבְּרִיאָה – וּלְהוֹדוֹת לַה׳ עַל כָּל פְּרָט קָטָן שֶׁבָּרָא.
📖 פֶּרֶק י״ד, הֲלָכָה ה׳

🧠 בְּדֹק אֶת עַצְמְךָ

שִׁמְעוֹן בֵּרֵךְ ״בּוֹרֵא עֲצֵי בְשָׂמִים״ עַל הֲדַס. לִפְנֵי שֶׁהֵרִיחַ, עָנָה לְחָבֵר עַל שְׁאֵלָה. מָה יַעֲשֶׂה?
💡 דִּבּוּר בֵּין הַבְּרָכָה לַהֲנָאָה מַפְסִיק, וְהַבְּרָכָה הָרִאשׁוֹנָה כְּבָר לֹא מוֹעִילָה. לָכֵן מְבָרֵךְ שׁוּב, וּמֵרִיחַ מִיָּד.
אָבִיתָר קָטַף עָלֶה שֶׁל נַעֲנָע כְּדֵי לְהָרִיחַ אוֹתוֹ. אֵיזוֹ בְּרָכָה עָדִיף שֶׁיְּבָרֵךְ?
💡 עַל נַעֲנָע הִסְתַּפְּקוּ אִם הִיא עֵץ אוֹ עֵשֶׂב, וְלָכֵן עָדִיף לְבָרֵךְ "מִינֵי בְשָׂמִים" – בְּרָכָה שֶׁמּוֹעִילָה לְכָל רֵיחַ. מִי שֶׁטּוֹעֶה בֵּין עֵץ לְעֵשֶׂב – לֹא יָצָא.
שִׁירָה מְבַשֶּׂמֶת אֶת עַצְמָהּ לִפְנֵי שִׂמְחָה מִשְׁפַּחְתִּית. עַל אֵיזֶה רֵיחַ הִיא מְבָרֶכֶת?
💡 בֹּשֶׂם נוֹעַד לֵהָנוֹת מֵרֵיחוֹ, וְלָכֵן מְבָרְכִים עָלָיו. הַמִּתְבַּשֶּׂמֶת מְבָרֶכֶת עַל הָרֵיחַ שֶׁיּוֹצֵא מִן הַבַּקְבּוּק. עַל רֵיחַ שֶׁנּוֹדֵף מֵאָדָם אוֹ מִבְּגָדִים – לֹא מְבָרְכִים, כִּי אֵין לוֹ עִקָּר.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

הִלְכוֹת הוֹצָאָה

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק כ״א

👕מָה יוֹצְאִים אִתּוֹ לָרְחוֹב?

בִּמְקוֹם שֶׁאֵין עֵרוּב אָסוּר לְהוֹצִיא חֵפֶץ מִן הַבַּיִת לָרְחוֹב, וְאָסוּר לְטַלְטֵל אוֹתוֹ בָּרְחוֹב. אֲבָל בְּגָדִים, נַעֲלַיִם וְכִפָּה – אֵינָם מַשָּׂא, כִּי הֵם חֵלֶק מִן הַגּוּף. מִי שֶׁרוֹצֶה לְהָבִיא חֻלְצָה לְחָבֵר יָכוֹל לִלְבֹּשׁ אוֹתָהּ עַל הַחֻלְצָה שֶׁלּוֹ, אֲבָל לֹא לָשֵׂאת אוֹתָהּ בַּיָּד.
מִשְׁקְפֵי רְאִיָּה, פְּלַסְטֵר וּפְלַטָּה לַשִּׁנַּיִם – מְשַׁמְּשִׁים אֶת הַגּוּף, וּמֻתָּר לָצֵאת בָּהֶם. מִשְׁקְפֵי שֶׁמֶשׁ – לֹא, כִּי בַּצֵּל עָלוּלִים לְהוֹרִיד אוֹתָם וְלָשֵׂאת בַּיָּד.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁהַיָּדַיִם רֵיקוֹת, הַלֵּב פָּנוּי. בְּשַׁבָּת אֲנַחְנוּ מְטַיְּלִים בְּלִי מַשָּׂא, וּמַרְגִּישִׁים אֶת הַחֵרוּת שֶׁה׳ נָתַן לָנוּ.
📖 פֶּרֶק כ״א, הֲלָכוֹת א׳, י׳, י״ב

⛺מְלֶאכֶת הַמִּשְׁכָּן

בַּמִּדְבָּר הֵבִיאוּ בְּנֵי יִשְׂרָאֵל אֶת הַתְּרוּמוֹת מִן הָאֹהָלִים שֶׁלָּהֶם אֶל הַמִּשְׁכָּן, עַד שֶׁמֹּשֶׁה צִוָּה לְהַפְסִיק. מִכָּאן לָמְדוּ שֶׁהוֹצָאָה מֵרְשׁוּת לִרְשׁוּת הִיא מְלָאכָה. לֹא רַק שִׁנּוּי בַּחֵפֶץ עַצְמוֹ הוּא מְלָאכָה – גַּם שִׁנּוּי בִּמְקוֹמוֹ.
לְכָל דָּבָר בָּעוֹלָם יֵשׁ מָקוֹם, וּבִמְקוֹמוֹ הוּא מוֹעִיל. לָכֵן ה׳ נִקְרָא "מָקוֹם" – הוּא נוֹתֵן מָקוֹם לְכָל הָעוֹלָם.
💛 אַהֲבַת הַשֵּׁם: שִׁשָּׁה יָמִים אֲנַחְנוּ מְזִיזִים וּמְסַדְּרִים אֶת הָעוֹלָם. בַּשַּׁבָּת אֲנַחְנוּ נִשְׁאָרִים בִּמְקוֹמֵנוּ, וּמַרְגִּישִׁים שֶׁה׳ הוּא הַמָּקוֹם הַבָּטוּחַ שֶׁל כֻּלָּנוּ.
📖 פֶּרֶק כ״א, הֲלָכָה א׳

🏠אַרְבַּע רְשׁוּיוֹת

רְשׁוּת הַיָּחִיד – מָקוֹם מֻקָּף מְחִצּוֹת, כְּמוֹ בַּיִת אוֹ חָצֵר סְגוּרָה, וּבוֹ מֻתָּר לְטַלְטֵל. רְשׁוּת הָרַבִּים – רְחוֹב רָחָב שֶׁרַבִּים עוֹבְרִים בּוֹ. כַּרְמְלִית – מָקוֹם פָּתוּחַ כְּמוֹ שָׂדֶה, שֶׁמִּן הַתּוֹרָה מֻתָּר בּוֹ, אֲבָל חֲכָמִים קָבְעוּ שֶׁדִּינוֹ כִּרְשׁוּת הָרַבִּים. וּמְקוֹם פְּטוּר – מָקוֹם קָטָן וְגָבוֹהַּ, כְּמוֹ סֶלַע צַר.
בָּרְחוֹב, אָדָם רַשַּׁאי לְטַלְטֵל רַק בְּתוֹךְ אַרְבַּע אַמּוֹת שֶׁסְּבִיבוֹ – כְּגֹדֶל אָדָם שׁוֹכֵב וְיָדָיו וְרַגְלָיו פְּשׁוּטוֹת.
💛 אַהֲבַת הַשֵּׁם: בַּיִת מֻקָּף הוּא מָקוֹם אֶחָד, שֶׁכֻּלּוֹ מְאֻחָד. כָּךְ גַּם עַם יִשְׂרָאֵל: כְּשֶׁאֲנַחְנוּ מְאֻחָדִים בְּאַהֲבָה, אֲנַחְנוּ כְּבַיִת אֶחָד לִפְנֵי ה׳.
📖 פֶּרֶק כ״א, הֲלָכוֹת ב׳–ד׳

🏠🔑 מוּשָּׂג: כַּרְמְלִית

רְשׁוּת הַיָּחִיד – מָקוֹם מֻקָּף מְחִצּוֹת, וּבוֹ מֻתָּר לְטַלְטֵל. רְשׁוּת הָרַבִּים – רְחוֹב רָחָב שֶׁרַבִּים עוֹבְרִים בּוֹ. כַּרְמְלִית – מָקוֹם פָּתוּחַ כְּמוֹ שָׂדֶה, שֶׁמִּן הַתּוֹרָה מֻתָּר בּוֹ, אֲבָל חֲכָמִים קָבְעוּ שֶׁדִּינוֹ כִּרְשׁוּת הָרַבִּים.
📖 בָּרְחוֹב, אָדָם רַשַּׁאי לְטַלְטֵל רַק בְּתוֹךְ אַרְבַּע אַמּוֹת שֶׁסְּבִיבוֹ.
💛 אַהֲבַת הַשֵּׁם: בַּיִת מֻקָּף הוּא מָקוֹם אֶחָד, שֶׁכֻּלּוֹ מְאֻחָד. 🧒 וְאַתָּה? כְּשֶׁאֵין עֵרוּב, זְכֹר שֶׁגַּם שָׂדֶה פָּתוּחַ אֵינוֹ בֶּאֱמֶת כְּמוֹ בַּיִת שֶׁלְּךָ – נִזְהָרִים בּוֹ בְּטִלְטוּל.
📖 פֶּרֶק כ״א, הֲלָכוֹת ב׳–ד׳

🤲שְׁנַיִם יַחַד וּבְשִׁנּוּי

מְלֶאכֶת הוֹצָאָה בְּנוּיָה מִשְּׁלֹשָׁה שְׁלַבִּים: עֲקִירַת הַחֵפֶץ מִמְּקוֹמוֹ, הַעֲבָרָתוֹ לִרְשׁוּת אַחֶרֶת, וְהַנָּחָתוֹ שָׁם. מִן הַתּוֹרָה אָסוּר רַק כְּשֶׁאָדָם אֶחָד עוֹשֶׂה אֶת כֻּלָּם, כְּדַרְכּוֹ – בַּיָּד, בַּכִּיס אוֹ בַּתִּיק.
חֲכָמִים אָסְרוּ גַּם כְּשֶׁשְּׁנַיִם מִתְחַלְּקִים בַּשְּׁלַבִּים, וְגַם כְּשֶׁמּוֹצִיאִים בְּשִׁנּוּי, כְּמוֹ מִטְפַּחַת בְּתוֹךְ הַנַּעַל. אֶת הַמְּלָאכָה עַצְמָהּ לֹא עוֹקְפִים.
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁאוֹהֵב לֹא מְחַפֵּשׂ "פִּרְצָה" כְּדֵי לְהַשִּׂיג אֶת מָה שֶׁהוּא רוֹצֶה. הוּא שׁוֹאֵל: מָה רוֹצֶה ה׳ שֶׁאֶעֱשֶׂה בַּיּוֹם הַקָּדוֹשׁ?
📖 פֶּרֶק כ״א, הֲלָכָה ה׳

🤲📜 לְשׁוֹן חֲכָמִים: שְׁנַיִם שֶׁעֲשׂוּהָ

מְלֶאכֶת הוֹצָאָה בְּנוּיָה מִשְּׁלֹשָׁה שְׁלַבִּים: עֲקִירַת הַחֵפֶץ מִמְּקוֹמוֹ, הַעֲבָרָתוֹ לִרְשׁוּת אַחֶרֶת, וְהַנָּחָתוֹ שָׁם. מִן הַתּוֹרָה אָסוּר רַק כְּשֶׁאָדָם אֶחָד עוֹשֶׂה אֶת כֻּלָּם, אֲבָל כְּשֶׁשְּׁנַיִם מִתְחַלְּקִים בַּשְּׁלַבִּים – אִסּוּר חֲכָמִים.
📖 חֲכָמִים אָסְרוּ גַּם כְּשֶׁשְּׁנַיִם מִתְחַלְּקִים בַּשְּׁלַבִּים, וְגַם כְּשֶׁמּוֹצִיאִים בְּשִׁנּוּי. אֶת הַמְּלָאכָה עַצְמָהּ לֹא עוֹקְפִים.
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁאוֹהֵב לֹא מְחַפֵּשׂ ״פִּרְצָה״ כְּדֵי לְהַשִּׂיג אֶת מָה שֶׁהוּא רוֹצֶה. 🧒 וְאַתָּה? אַל תּוֹשִׁיט חֵפֶץ בְּשַׁבָּת דֶּרֶךְ חַלּוֹן לְחָצֵר שֶׁאֵין בָּהּ עֵרוּב, גַּם אִם חָבֵר יַחְזִיק אוֹתוֹ בָּצַד שֶׁלּוֹ.
📖 פֶּרֶק כ״א, הֲלָכָה ה׳

🔑עֵרוּב, כַּפְתּוֹרִים וּמַפְתֵּחַ

רֹב יִשְׂרָאֵל נוֹהֲגִים לְטַלְטֵל בָּעִיר עַל סְמַךְ עֵרוּב שֶׁל צוּרַת הַפֶּתַח, וְלִדְעַת רַבִּים רָאוּי לְכַתְּחִלָּה לְהַחְמִיר שֶׁלֹּא לִסְמֹךְ עָלָיו בְּעִיר שֶׁיֵּשׁ בָּהּ רְחוֹבוֹת רְחָבִים, יוֹתֵר מִשֵּׁשׁ עֶשְׂרֵה אַמָּה. בְּכָל מָקוֹם: כַּפְתּוֹרִים, כִּיסִים וְכוֹבַע שֶׁמְּחֻבָּר לַמְּעִיל הֵם חֵלֶק מִן הַבֶּגֶד. וּמִי שֶׁאֵינוֹ יָכוֹל לָלֶכֶת בְּלִי מַקֵּל – הַמַּקֵּל כְּמוֹ נַעֲלָיו.
בִּמְקוֹם שֶׁאֵין עֵרוּב, מַשְׁחִילִים אֶת הַמַּפְתֵּחַ עַל שְׂרוֹךְ, וְהוּא מְשַׁמֵּשׁ אַבְזָם שֶׁל חֲגוֹרָה שֶׁחוֹגְרִים עַל הַמָּתְנַיִם.
💛 אַהֲבַת הַשֵּׁם: הָעֵרוּב מְחַבֵּר אֶת כָּל בָּתֵּי הָעִיר לְמָקוֹם אֶחָד, כְּדֵי שֶׁמִּשְׁפָּחוֹת יוּכְלוּ לְבַקֵּר זוֹ אֶת זוֹ בְּשַׁבָּת. זוֹ שַׁבָּת שֶׁל חֲבֵרוּת וְאַהֲבָה.
📖 פֶּרֶק כ״א, הֲלָכוֹת ט׳, י״א, י״ג, ט״ו

🔀דֵּעוֹת וּמִנְהָגִים בְּהוֹצָאָה

תַּכְשִׁיטִים בְּמָקוֹם שֶׁאֵין עֵרוּב: יֵשׁ אוֹמְרִים שֶׁמִּן הַדִּין אָסוּר, אֶלָּא שֶׁלֹּא מִחוּ בַּנָּשִׁים. וְיֵשׁ פּוֹסְקִים שֶׁלִּמְּדוּ זְכוּת – כִּי הַיּוֹם אֵין רְשׁוּת הָרַבִּים מִן הַתּוֹרָה, אוֹ כִּי תַּכְשִׁיטִים מְצוּיִים וְלֹא מוֹרִידִים אוֹתָם כְּדֵי לְהַרְאוֹתָם. אֶפְשָׁר לִסְמֹךְ עַל הַמְּקִלִּים, וְכָךְ נוֹהֲגוֹת הַנָּשִׁים. שְׁעוֹן יָד: יֵשׁ אוֹמְרִים שֶׁמֻּתָּר לָצֵאת בּוֹ רַק כְּשֶׁהוּא תַּכְשִׁיט, וְהַרְבֵּה פּוֹסְקִים מַתִּירִים, כִּי הוּא כְּמוֹ מַלְבּוּשׁ – וְהָעִקָּר כַּמְּקִלִּים.
כְּפָפוֹת: יֵשׁ אוֹמְרִים שֶׁלֹּא לָצֵאת בָּהֶן לְמָקוֹם שֶׁאֵין בּוֹ עֵרוּב, שֶׁמָּא יוֹרִידוּ אוֹתָן וְיִשְּׂאוּ בַּיָּד. לְכַתְּחִלָּה רָצוּי לְהַחְמִיר, וְהַמִּנְהָג לְהָקֵל. עֵרוּב: רֹב יִשְׂרָאֵל נוֹהֲגִים לְטַלְטֵל עַל סְמַךְ צוּרוֹת הַפֶּתַח, וְלִדְעַת רַבִּים רָאוּי לְכַתְּחִלָּה לְהַחְמִיר בְּעִיר שֶׁיֵּשׁ בָּהּ רְחוֹבוֹת רְחָבִים.
💛 אַהֲבַת הַשֵּׁם: הַמַּחֲלֹקֶת בֵּין חֲכָמִים הִיא לְשֵׁם שָׁמַיִם. כְּשֶׁאָנוּ מַכִּירִים אֶת שְׁתֵּי הַדֵּעוֹת, אָנוּ לוֹמְדִים לְכַבֵּד כָּל יְהוּדִי שֶׁהוֹלֵךְ בְּדֶרֶךְ רַבּוֹתָיו.
📖 פֶּרֶק כ״א, הֲלָכוֹת ט׳–י׳, י״ד–ט״ו

💍תַּכְשִׁיטִים וּשְׂמִיכָה

תַּכְשִׁיטִים שֶׁעַל הַגּוּף טְפֵלִים לַגּוּף, כְּמוֹ בְּגָדִים. אֲבָל חֲכָמִים חָשְׁשׁוּ שֶׁמָּא אִשָּׁה תּוֹרִיד תַּכְשִׁיט כְּדֵי לְהַרְאוֹת אוֹתוֹ לַחֲבֶרְתָּהּ, תִּשְׁכַּח וְתֵלֵךְ אִתּוֹ בַּיָּד אַרְבַּע אַמּוֹת בִּרְשׁוּת הָרַבִּים – וְלָכֵן אָסְרוּ לָצֵאת בְּעָגִילִים, צְמִידִים, טַבָּעוֹת וְשַׁרְשְׁרָאוֹת. וּכְבָר בִּימֵי הָרִאשׁוֹנִים נָהֲגוּ הַנָּשִׁים לָצֵאת בְּתַכְשִׁיטֵיהֶן, וְעַל כָּךְ נִלְמַד בָּרָמָה הַבָּאָה.
מִי שֶׁרוֹצָה לְהָבִיא שְׂמִיכָה אוֹ מַפָּה דֶּרֶךְ רְחוֹב שֶׁאֵין בּוֹ עֵרוּב – יְכוֹלָה לְהִתְעַטֵּף בָּהּ כְּדֶרֶךְ לְבוּשׁ, וְלָלֶכֶת בָּהּ. כְּלָל הַדָּבָר: כָּל מָה שֶׁעָלֶיהָ כְּדֶרֶךְ לְבוּשׁ – מֻתָּר.
💛 אַהֲבַת הַשֵּׁם: גַּם הַתַּכְשִׁיטִים וְהַבְּגָדִים הַיָּפִים הֵם חֵלֶק מִכְּבוֹד הַשַּׁבָּת. כְּשֶׁאָנוּ מִתְקַשְּׁטִים לִכְבוֹדָהּ וְשׁוֹמְרִים עַל הִלְכוֹתֶיהָ, אָנוּ מַרְאִים שֶׁהַשַּׁבָּת הִיא כְּמוֹ מַלְכָּה שֶׁבָּאָה אֵלֵינוּ.
📖 פֶּרֶק כ״א, הֲלָכוֹת י׳, י״ד

💍🔑 מוּשָּׂג: טָפֵל לַגּוּף

מַה שֶּׁמְּשַׁמֵּשׁ אֶת הַגּוּף כָּל הַזְּמַן, כְּמוֹ מִשְׁקְפֵי רְאִיָּה, טָפֵל לַגּוּף וּמֻתָּר לָצֵאת בּוֹ לִרְחוֹב בְּלִי עֵרוּב. אֲבָל תַּכְשִׁיטִים שֶׁמּוֹרִידִים כְּדֵי לְהַרְאוֹת לַחֲבֵרוֹת – חָשְׁשׁוּ חֲכָמִים שֶׁמָּא יִשְׁכְּחוּ וְיִשָּׂאוּ אוֹתָם בַּיָּד.
📖 מִשְׁקְפֵי שֶׁמֶשׁ מוֹרִידִים בַּצֵּל וְנוֹשְׂאִים בַּיָּד, וְכַדּוּר הוּא סְתָם מַשָּׂא – אֲסוּרִים בִּלְלִי עֵרוּב.
💛 אַהֲבַת הַשֵּׁם: גַּם הַתַּכְשִׁיטִים וְהַבְּגָדִים הַיָּפִים הֵם חֵלֶק מִכְּבוֹד הַשַּׁבָּת, אֲבָל חֲכָמִים דָּאֲגוּ שֶׁלֹּא נִכָּשֵׁל בִּטִּלְטוּל. 🧒 וְאַתָּה? בְּלִי עֵרוּב – צֵא לִרְחוֹב בְּמִשְׁקְפֵי רְאִיָּה, וְלֹא בְּמִשְׁקְפֵי שֶׁמֶשׁ.
📖 פֶּרֶק כ״א, הֲלָכוֹת י׳, י״ב, י״ד

🧠 בְּדֹק אֶת עַצְמְךָ

לָמָּה אָסוּר לָצֵאת בְּמִשְׁקְפֵי שֶׁמֶשׁ לִרְחוֹב שֶׁאֵין בּוֹ עֵרוּב, וּבְמִשְׁקְפֵי רְאִיָּה מֻתָּר?
💡 מַה שֶּׁמְּשַׁמֵּשׁ אֶת הַגּוּף טָפֵל לוֹ. אֶת מִשְׁקְפֵי הַשֶּׁמֶשׁ מוֹרִידִים כְּשֶׁמַּגִּיעִים לַצֵּל, וְיֵשׁ חֲשָׁשׁ שֶׁיִּשָּׂא אוֹתָם בַּיָּד אוֹ בַּכִּיס.
אֵין עֵרוּב. יוֹסֵף עוֹמֵד בַּבַּיִת, וּמוֹשִׁיט סֵפֶר דֶּרֶךְ הַחַלּוֹן לְחָבֵר שֶׁעוֹמֵד בָּרְחוֹב. מָה הַדִּין?
💡 יוֹסֵף עָקַר וְהֶעֱבִיר, וְהֶחָבֵר הִנִּיחַ – וְלָכֵן אֵין כָּאן אִסּוּר תּוֹרָה. אֲבָל חֲכָמִים אָסְרוּ, שֶׁלֹּא יָבוֹאוּ לְזַלְזֵל בִּמְלֶאכֶת הוֹצָאָה.
מֹשֶׁה מִתְאָרֵחַ בְּשַׁבָּת בְּיִשּׁוּב שֶׁאֵין בּוֹ עֵרוּב, וְרוֹצֶה לִנְעֹל אֶת הַדֶּלֶת כְּשֶׁהוּא יוֹצֵא לְבֵית הַכְּנֶסֶת. מָה יַעֲשֶׂה?
💡 הַכִּיס טָפֵל לַבֶּגֶד, אֲבָל הַמַּפְתֵּחַ שֶׁבְּתוֹכוֹ הוּא מַשָּׂא. כְּשֶׁהַמַּפְתֵּחַ מְשַׁמֵּשׁ אַבְזָם בַּחֲגוֹרָה – הוּא חֵלֶק מִן הַלְּבוּשׁ.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

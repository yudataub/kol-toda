💛שְׁמִינִי עֲצֶרֶת

שְׁמִינִי עֲצֶרֶת הוּא חַג בִּפְנֵי עַצְמוֹ: אֵין בּוֹ סֻכָּה וְאֵין בּוֹ לוּלָב, וּמְבָרְכִים עָלָיו ״שֶׁהֶחֱיָנוּ״ בַּקִּדּוּשׁ. בְּאֶרֶץ יִשְׂרָאֵל אָסוּר לֶאֱכֹל בּוֹ בַּסֻּכָּה לְשֵׁם מִצְוָה – כִּי זוֹ הוֹסָפָה עַל הַמִּצְוָה, ״בַּל תּוֹסִיף״.
בְּשִׁבְעַת יְמֵי הַחַג הִקְרִיבוּ שִׁבְעִים פָּרִים כְּנֶגֶד שִׁבְעִים אֻמּוֹת, וּבִשְׁמִינִי עֲצֶרֶת – פַּר אֶחָד, כְּנֶגֶד יִשְׂרָאֵל. כְּמוֹ מֶלֶךְ שֶׁאָמַר לְאוֹהֲבוֹ: ״עֲשֵׂה לִי סְעוּדָה קְטַנָּה, כְּדֵי שֶׁאֵהָנֶה מִמְּךָ״.
💛 אַהֲבַת הַשֵּׁם: בַּיּוֹם הַזֶּה אֵין מִצְוָה מְיֻחֶדֶת – רַק לִשְׂמֹחַ בַּה׳, בְּתוֹרָתוֹ וּבִישׁוּעָתוֹ. ״זֶה הַיּוֹם עָשָׂה ה׳ נָגִילָה וְנִשְׂמְחָה בוֹ״.
📖 פֶּרֶק ז׳, הֲלָכוֹת א׳–ב׳; פֶּרֶק ו׳, הֲלָכָה ה׳

💛🔑 מוּשָּׂג: שְׁמִינִי עֲצֶרֶת – רֶגֶל בִּפְנֵי עַצְמוֹ

״שְׁמִינִי עֲצֶרֶת הוּא חַג בִּפְנֵי עַצְמוֹ: אֵין בּוֹ סֻכָּה וְאֵין בּוֹ לוּלָב, וּמְבָרְכִים עָלָיו ׳שֶׁהֶחֱיָנוּ׳ בַּקִּדּוּשׁ״.
📖 עֲצֶרֶת – מִלְּשׁוֹן ״לְהֵעָצֵר״ וְלְהִתְאַסֵּף: יוֹם שֶׁבּוֹ ה׳ מְבַקֵּשׁ מֵאִתָּנוּ לְהִשָּׁאֵר עוֹד יוֹם אֶחָד אִתּוֹ.
💛 אַהֲבַת הַשֵּׁם: בַּיּוֹם הַזֶּה אֵין מִצְוָה מְיֻחֶדֶת – רַק לִשְׂמֹחַ בַּה׳, בְּתוֹרָתוֹ וּבִישׁוּעָתוֹ. 🧒 וְאַתָּה? בִּשְׁמִינִי עֲצֶרֶת – תִּשְׂמַח בְּפָשְׁטוּת, בְּלִי שׁוּם סִבָּה מְיֻחֶדֶת.
📖 פֶּרֶק ז׳, הֲלָכוֹת א׳–ב׳

💛קָשָׁה עָלַי פְּרֵדַתְכֶם

״עֲצֶרֶת״ – מִלְּשׁוֹן עֲצִירָה וְהִתְאַסְּפוּת. מָשָׁל לְמֶלֶךְ שֶׁהִזְמִין אֶת בָּנָיו לִסְעוּדָה כַּמָּה יָמִים. כְּשֶׁהִגִּיעַ זְמַנָּם לָלֶכֶת, אָמַר: ״בָּנַי, בְּבַקָּשָׁה מִכֶּם, עִכְּבוּ עִמִּי עוֹד יוֹם – קָשָׁה עָלַי פְּרֵדַתְכֶם״.
בִּשְׁמִינִי עֲצֶרֶת הַשִּׂמְחָה אֵינָהּ תְּלוּיָה בְּסֻכָּה אוֹ בְּלוּלָב, אֶלָּא בְּכָךְ שֶׁאָנוּ בָּנָיו וְעַמּוֹ שֶׁל ה׳. הַגְּרָ״א אָמַר שֶׁזֶּה יוֹם הַשִּׂמְחָה הַגָּדוֹל מִכָּל יְמֵי הֶחָג.
💛 אַהֲבַת הַשֵּׁם: בַּיּוֹם הַזֶּה אָנוּ אוֹצְרִים בַּלֵּב אֶת כָּל הָאוֹר שֶׁל הַשָּׁנָה – כְּדֵי לִשְׁמֹר אוֹתוֹ לְתָמִיד.
📖 פֶּרֶק ז׳, הֲלָכָה ג׳

🌧️מַשִּׁיב הָרוּחַ וּמוֹרִיד הַגֶּשֶׁם

בְּסֻכּוֹת לֹא מַזְכִּירִים גֶּשֶׁם, כִּי גֶּשֶׁם בַּחַג מַפְרִיעַ לַיְּשִׁיבָה בַּסֻּכָּה. מַתְחִילִים לְהַזְכִּיר ״מַשִּׁיב הָרוּחַ וּמוֹרִיד הַגֶּשֶׁם״ בִּתְפִלַּת מוּסָף שֶׁל שְׁמִינִי עֲצֶרֶת, כְּשֶׁכָּל הַצִּבּוּר בְּבֵית הַכְּנֶסֶת. וְאוֹמְרִים אָז ״תְּפִלַּת הַגֶּשֶׁם״ בְּכַוָּנָה וּבְתַחֲנוּנִים.
אֶת הַבַּקָּשָׁה ״וְתֵן טַל וּמָטָר לִבְרָכָה״ מַתְחִילִים בְּאֶרֶץ יִשְׂרָאֵל רַק בְּלֵיל ז׳ בְּמַרְחֶשְׁוָן – כְּדֵי שֶׁעוֹלֵי הָרֶגֶל הָאַחֲרוֹנִים, שֶׁבָּאוּ מִנְּהַר פְּרָת, יַגִּיעוּ לְבֵיתָם בְּלִי גֶּשֶׁם.
💛 אַהֲבַת הַשֵּׁם: הַמַּיִם נוֹתְנִים חַיִּים לַגּוּף, וְהַתּוֹרָה – לַנְּשָׁמָה. בִּתְפִלַּת הַגֶּשֶׁם מְבַקְּשִׁים גַּם שָׁנָה בְּרוּכָה בְּתוֹרָה.
📖 פֶּרֶק ז׳, הֲלָכָה ד׳

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

תְּפִלַּת הַגֶּשֶׁם: הַסְּפָרַדִּים וְחֵלֶק מִן הָאַשְׁכְּנַזִּים אוֹמְרִים אוֹתָהּ לִפְנֵי תְּפִלַּת מוּסָף. רֹב הָאַשְׁכְּנַזִּים אוֹמְרִים אוֹתָהּ בְּתוֹךְ חֲזָרַת הַחַזָּן שֶׁל מוּסָף, וְהַגַּבַּאי מַכְרִיז לִפְנֵי תְּפִלַּת הַלַּחַשׁ ״מַשִּׁיב הָרוּחַ וּמוֹרִיד הַגֶּשֶׁם״. הַחַזָּן הָאַשְׁכְּנַזִּי לוֹבֵשׁ ״קִיטְל״ וּמְנַגֵּן בְּנִגּוּן הַיָּמִים הַנּוֹרָאִים.
הַקָּפוֹת שִׂמְחַת תּוֹרָה: לְמִנְהַג הָאַשְׁכְּנַזִּים נוֹשְׂאִים אֶת כָּל סִפְרֵי הַתּוֹרָה וְרוֹקְדִים אִתָּם בְּכָל הַהַקָּפוֹת. לְמִנְהַג הַסְּפָרַדִּים, בְּרֹב הַזְּמַן הַסְּפָרִים מֻנָּחִים עַל הַבִּימָה, אֶחָד אוֹחֵז סֵפֶר תּוֹרָה, וְהַצִּבּוּר מַקִּיף וְרוֹקֵד סְבִיבָם.
💛 אַהֲבַת הַשֵּׁם: כָּל הַמִּנְהָגִים טוֹבִים, וְכֻלָּם רוֹקְדִים לִכְבוֹד אוֹתָהּ תּוֹרָה. כְּשֶׁאָנוּ שְׂמֵחִים יַחַד – כָּל עֵדָה בְּדַרְכָּהּ – הַתּוֹרָה מְאַחֶדֶת אֶת כֻּלָּנוּ.
📖 פֶּרֶק ז׳, הֲלָכוֹת ד׳, ז׳

🌧️🔑 מוּשָּׂג: תְּפִלַּת הַגֶּשֶׁם

״בְּסֻכּוֹת לֹא מַזְכִּירִים גֶּשֶׁם... מַתְחִילִים לְהַזְכִּיר ׳מַשִּׁיב הָרוּחַ וּמוֹרִיד הַגֶּשֶׁם׳ בִּתְפִלַּת מוּסָף שֶׁל שְׁמִינִי עֲצֶרֶת״.
📖 תְּפִלַּת הַגֶּשֶׁם – תְּפִלָּה מְיֻחֶדֶת בִּשְׁמִינִי עֲצֶרֶת, שֶׁבָּהּ מְבַקְּשִׁים מֵה׳ עַל גִּשְׁמֵי הַשָּׁנָה הַבָּאָה.
💛 אַהֲבַת הַשֵּׁם: הַמַּיִם נוֹתְנִים חַיִּים לַגּוּף, וְהַתּוֹרָה – לַנְּשָׁמָה. בִּתְפִלַּת הַגֶּשֶׁם מְבַקְּשִׁים גַּם שָׁנָה בְּרוּכָה בְּתוֹרָה. 🧒 וְאַתָּה? כְּשֶׁיֵּרֵד הַגֶּשֶׁם הָרִאשׁוֹן – תַּגִּיד תּוֹדָה לַה׳.
📖 פֶּרֶק ז׳, הֲלָכָה ד׳

📜שִׂמְחַת תּוֹרָה

בִּשְׁמִינִי עֲצֶרֶת מְסַיְּמִים אֶת קְרִיאַת הַתּוֹרָה, וְלָכֵן הוּא נִקְרָא ״שִׂמְחַת תּוֹרָה״. לִכְבוֹד הַסִּיּוּם מַעֲלִים לַתּוֹרָה אֶת כָּל הַמִּתְפַּלְּלִים. הַיְּלָדִים הַקְּטַנִּים עוֹלִים יַחַד בַּעֲלִיַּת ״כָּל הַנְּעָרִים״, עִם אֶחָד מִנִּכְבַּדֵי הַקָּהָל, וְחוֹזְרִים אַחֲרָיו עַל הַבְּרָכוֹת מִלָּה בְּמִלָּה.
הָעוֹלֶה שֶׁמְּסַיֵּם אֶת הַתּוֹרָה נִקְרָא ״חֲתַן תּוֹרָה״. מִיָּד אַחֲרָיו מַתְחִילִים לִקְרֹא מִ״בְּרֵאשִׁית״ – וְהָעוֹלֶה נִקְרָא ״חֲתַן בְּרֵאשִׁית״, כִּי אֲפִלּוּ לְרֶגַע אֵינֶנּוּ נִפְרָדִים מִן הַתּוֹרָה.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁכֻּלָּם עוֹלִים לַתּוֹרָה – זָקֵן וְצָעִיר, חָכָם וְתָם – אָנוּ מַרְאִים שֶׁלְּכָל יְהוּדִי יֵשׁ חֵלֶק בַּתּוֹרָה.
📖 פֶּרֶק ז׳, הֲלָכוֹת ה׳–ו׳

📜📜 לְשׁוֹן חֲכָמִים: כָּל הַנְּעָרִים

״אֶת הַקְּטַנִּים מַעֲלִים יַחַד עִם אֶחָד מִנִּכְבְּדֵי הַקָּהָל לַעֲלִיַּת ׳כָּל הַנְּעָרִים׳״.
📖 כָּל הַנְּעָרִים – עֲלִיָּה מְיֻחֶדֶת בְּשִׂמְחַת תּוֹרָה, שֶׁבָּהּ כָּל הַיְּלָדִים עוֹלִים יַחַד לַתּוֹרָה.
💛 אַהֲבַת הַשֵּׁם: בְּכָךְ שֶׁכֻּלָּם עוֹלִים לַתּוֹרָה, מַרְאִים שֶׁלְּכָל יְהוּדִי, בֵּין גָּדוֹל בֵּין קָטָן, יֵשׁ חֵלֶק בַּתּוֹרָה. 🧒 וְאַתָּה? בְּשִׂמְחַת תּוֹרָה – עֲלֵה לְ״כָל הַנְּעָרִים״ בְּקוֹל רָם וְשָׂמֵחַ.
📖 פֶּרֶק ז׳, הֲלָכָה ה׳

📜חֲתָנֵי הַתּוֹרָה

אַרְבַּע עֲלִיּוֹת חֲשׁוּבוֹת בְּשִׂמְחַת תּוֹרָה: ״חֲתַן מְעֹנָה״ – שֶׁעוֹלֶה עִם כָּל הַנְּעָרִים, ״חֲתַן תּוֹרָה״ – שֶׁמְּסַיֵּם אֶת הַתּוֹרָה, ״חֲתַן בְּרֵאשִׁית״ – שֶׁמַּתְחִיל אוֹתָהּ מֵחָדָשׁ בְּסֵפֶר שֵׁנִי, וְהַ״מַּפְטִיר״ – שֶׁקּוֹרֵא בְּסֵפֶר שְׁלִישִׁי עַל קָרְבְּנוֹת הַחַג.
הַהַפְטָרָה – מִתְּחִלַּת סֵפֶר יְהוֹשֻׁעַ, כִּי תַּכְלִית הַתּוֹרָה לְקַיְּמָהּ בָּאָרֶץ, וְשָׁם כָּתוּב: ״לֹא יָמוּשׁ סֵפֶר הַתּוֹרָה הַזֶּה מִפִּיךָ״. וּמִי שֶׁעָלָה לַ״חֲתַן תּוֹרָה״ לֹא יַעֲלֶה מִיָּד גַּם לַ״חֲתַן בְּרֵאשִׁית״.
💛 אַהֲבַת הַשֵּׁם: אֲפִלּוּ לְרֶגַע אֵינֶנּוּ נִפְרָדִים מִן הַתּוֹרָה: מְסַיְּמִים – וּמִיָּד מַתְחִילִים מֵחָדָשׁ, כִּי הִיא חַיֵּינוּ.
📖 פֶּרֶק ז׳, הֲלָכָה ו׳

📜📜 לְשׁוֹן חֲכָמִים: חֲתַן תּוֹרָה

״הָעֲלִיָּה שֶׁבָּהּ מְסַיְּמִים אֶת הַתּוֹרָה, וְהָעוֹלֶה אֵלֶיהָ נִקְרָא ׳חֲתַן תּוֹרָה׳״.
📖 חֲתַן תּוֹרָה – כְּמוֹ חָתָן בְּיוֹם חֲתֻנָּתוֹ: מִי שֶׁזּוֹכֶה לַעֲלִיָּה שֶׁל סִיּוּם הַתּוֹרָה בְּשִׂמְחַת תּוֹרָה.
💛 אַהֲבַת הַשֵּׁם: אֲפִלּוּ בְּרֶגַע שֶׁל סִיּוּם – שִׂמְחָה גְּדוֹלָה כְּמוֹ בַּחֲתֻנָּה, כִּי סִיַּמְנוּ לִלְמֹד אֶת כָּל הַתּוֹרָה. 🧒 וְאַתָּה? בְּסִיּוּם סֵפֶר אוֹ פָּרָשָׁה שֶׁלָּמַדְתָּ – עֲשֵׂה חֲגִיגָה קְטַנָּה.
📖 פֶּרֶק ז׳, הֲלָכָה ו׳

📜📜 לְשׁוֹן חֲכָמִים: חֲתַן בְּרֵאשִׁית

״מִיָּד לְאַחַר סִיּוּם הַתּוֹרָה מַעֲלִים עוֹלֶה נוֹסָף לִקְרֹא בָּהּ מִבְּרֵאשִׁית, וְקוֹרְאִים לוֹ ׳חֲתַן בְּרֵאשִׁית׳״.
📖 חֲתַן בְּרֵאשִׁית – מִי שֶׁזּוֹכֶה לַעֲלִיָּה שֶׁל הַתְחָלַת הַתּוֹרָה, מִיָּד אַחֲרֵי שֶׁמְּסַיְּמִים אוֹתָהּ.
💛 אַהֲבַת הַשֵּׁם: אֲפִלּוּ לְרֶגַע אֵינֶנּוּ נִפְרָדִים מִן הַתּוֹרָה: מְסַיְּמִים – וּמִיָּד מַתְחִילִים מֵחָדָשׁ, כִּי הִיא חַיֵּינוּ. 🧒 וְאַתָּה? כְּשֶׁאַתָּה מְסַיֵּם סֵפֶר – תִּבְחַר כְּבָר אֵיזֶה סֵפֶר תִּלְמַד אַחֲרָיו.
📖 פֶּרֶק ז׳, הֲלָכָה ו׳

💃הַקָּפוֹת וְרִקּוּדִים

בְּלֵיל שִׂמְחַת תּוֹרָה וּבַיּוֹם עוֹרְכִים שֶׁבַע הַקָּפוֹת עִם סִפְרֵי הַתּוֹרָה, וּמַרְבִּים בְּרִקּוּדִים לִכְבוֹד הַתּוֹרָה. אַף שֶׁבְּשַׁבָּת וּבְיוֹם טוֹב אֵין רוֹקְדִים וְאֵין מוֹחֲאִים כַּפַּיִם, לִכְבוֹד שִׂמְחַת הַתּוֹרָה הִתִּירוּ.
כְּשֶׁמַּקִּיפִים וְרוֹקְדִים עִם סִפְרֵי הַתּוֹרָה – כָּל הַנִּמְצָאִים עוֹמְדִים. מִי שֶׁקָּשֶׁה לוֹ – יוֹשֵׁב, וְעוֹמֵד בִּתְחִלַּת כָּל הַקָּפָה. גַּם הַנָּשִׁים וְהַטַּף בָּאִים לְבֵית הַכְּנֶסֶת לִכְבוֹד שִׂמְחַת הַתּוֹרָה.
💛 אַהֲבַת הַשֵּׁם: הַגְּרָ״א הָיָה רוֹקֵד לִפְנֵי סֵפֶר הַתּוֹרָה בְּכָל כֹּחוֹ, וּפָנָיו מְאִירוֹת כְּלַפִּיד. אֵין שִׂמְחָה גְּדוֹלָה יוֹתֵר מִשִּׂמְחַת הַתּוֹרָה.
📖 פֶּרֶק ז׳, הֲלָכוֹת ז׳–ח׳

🧠 בְּדֹק אֶת עַצְמְךָ

לָמָּה מַתְחִילִים לְבַקֵּשׁ ״וְתֵן טַל וּמָטָר״ רַק בְּז׳ בְּמַרְחֶשְׁוָן?
💡 חֲכָמִים דָּאֲגוּ לְעוֹלֵי הָרֶגֶל שֶׁבָּאוּ מֵרָחוֹק, וְדָחוּ אֶת הַבַּקָּשָׁה בַּחֲמִשָּׁה עָשָׂר יוֹם – עַד שֶׁהָאַחֲרוֹן, שֶׁבָּא מִנְּהַר פְּרָת, יַגִּיעַ לְבֵיתוֹ. וְגַם הַיּוֹם מַמְשִׁיכִים בַּמִּנְהָג הַזֶּה, שֶׁמַּזְכִּיר אֶת יְמֵי הַמִּקְדָּשׁ.
לָמָּה הַהַפְטָרָה שֶׁל שִׂמְחַת תּוֹרָה הִיא מִתְּחִלַּת סֵפֶר יְהוֹשֻׁעַ?
💡 כָּל תַּכְלִית הַתּוֹרָה לְקַיְּמָהּ בָּאָרֶץ. וְעוֹד, יֵשׁ שָׁם פְּסוּקִים עַל עֵרֶךְ הַתּוֹרָה: ״לֹא יָמוּשׁ סֵפֶר הַתּוֹרָה הַזֶּה מִפִּיךָ וְהָגִיתָ בּוֹ יוֹמָם וָלַיְלָה״. וְרָאוּי לְהַתְחִיל מִיָּד לִלְמֹד נְבִיאִים.
מִי שֶׁעָלָה לַתּוֹרָה כְּ״חֲתַן תּוֹרָה״ – הַאִם יַעֲלֶה מִיָּד גַּם לַ״חֲתַן בְּרֵאשִׁית״?
💡 מִי שֶׁעָלָה כְּבָר עִם כָּל הַצִּבּוּר – מֻתָּר לוֹ לַעֲלוֹת שׁוּב לְאַחַת הָעֲלִיּוֹת הַחֲשׁוּבוֹת. אֲבָל ״חֲתַן תּוֹרָה״ לֹא יַעֲלֶה מִיָּד לַ״חֲתַן בְּרֵאשִׁית״, כִּי זֶה פּוֹגֵעַ בִּכְבוֹד הַסֵּפֶר הָרִאשׁוֹן – כְּאִלּוּ הָיָה פָּסוּל. וּמֻתָּר לְהַעֲלוֹת כֹּהֵן אוֹ לֵוִי לָעֲלִיּוֹת הָאֵלֶּה.

חָדָשׁ

🍽️ הִלְכוֹת כַּשְׁרוּת · פֶּרֶק א׳

🌾חָדָשׁ וְקָרְבַּן הָעֹמֶר

בְּט״ז בְּנִיסָן הָיוּ מַקְרִיבִים בַּמִּקְדָּשׁ אֶת "הָעֹמֶר" – רֵאשִׁית הַשְּׂעוֹרִים שֶׁל הַשָּׁנָה. עַד אָז אָסוּר לֶאֱכֹל מִן הַתְּבוּאָה הַחֲדָשָׁה שֶׁל חֲמֵשֶׁת מִינֵי הַדָּגָן – לֹא לֶחֶם, לֹא גַּרְעִינִים קְלוּיִים וְלֹא גַּרְעִינִים חַיִּים. הָאִסּוּר נִקְרָא "חָדָשׁ".
גַּם הַיּוֹם, כְּשֶׁאֵין מִקְדָּשׁ, הָאִסּוּר קַיָּם: בְּאֶרֶץ יִשְׂרָאֵל הַתְּבוּאָה הַחֲדָשָׁה מֻתֶּרֶת אַחֲרֵי שֶׁנִּגְמָר יוֹם ט״ז בְּנִיסָן. תְּבוּאָה שֶׁהִשְׁרִישָׁה בָּאֲדָמָה לִפְנֵי יוֹם הָעֹמֶר – מֻתֶּרֶת מִמֶּנּוּ.
💛 אַהֲבַת הַשֵּׁם: מִן הַתְּבוּאָה מְכִינִים אֶת עִקַּר הַמָּזוֹן. כְּשֶׁאֲנַחְנוּ מַקְדִּישִׁים לַה׳ אֶת הָרֵאשִׁית, כָּל הָאֲכִילָה שֶׁל הַשָּׁנָה נַעֲשֵׂית מְחֻבֶּרֶת לַקְּדֻשָּׁה.
📖 פֶּרֶק א׳, הֲלָכוֹת א׳–ג׳

🌱🔑 מוּשָּׂג: חָדָשׁ

״אָסוּר לֶאֱכֹל מֵהַתְּבוּאָה הַחֲדָשָׁה לִפְנֵי הַקְרָבַת הָעֹמֶר... לֶחֶם וְקָלִי וְכַרְמֶל לֹא תֹאכְלוּ עַד עֶצֶם הַיּוֹם הַזֶּה״.
📖 חָדָשׁ – תְּבוּאָה שֶׁגָּדְלָה הַשָּׁנָה, שֶׁאָסוּר לֶאֱכֹל מִמֶּנָּה עַד יוֹם הָעֹמֶר, ט״ז בְּנִיסָן.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה מְלַמֶּדֶת לְהַקְדִּים אֶת ה׳ לְעַצְמֵנוּ: קֹדֶם מַקְרִיבִים לוֹ אֶת רֵאשִׁית הַתְּבוּאָה, וְרַק אָז אוֹכְלִים מִמֶּנָּה. 🧒 וְאַתָּה? לִפְנֵי שֶׁאַתָּה טוֹעֵם מַשֶּׁהוּ חָדָשׁ וְטוֹב – תַּגִּיד תּוֹדָה לַה׳.
📖 פֶּרֶק א׳, הֲלָכוֹת א׳, ג׳

🧠 בְּדֹק אֶת עַצְמְךָ

מָתַי מֻתָּר בְּאֶרֶץ יִשְׂרָאֵל לֶאֱכֹל מִן הַחִטָּה הַחֲדָשָׁה שֶׁל הַשָּׁנָה?
💡 גַּם בְּלִי מִקְדָּשׁ אִסּוּר "חָדָשׁ" קַיָּם, וְהַתְּבוּאָה הַחֲדָשָׁה מֻתֶּרֶת אַחֲרֵי שֶׁנִּגְמָר יוֹם ט״ז בְּנִיסָן – "עַד עֶצֶם הַיּוֹם הַזֶּה". קָרְבַּן שְׁתֵּי הַלֶּחֶם בְּשָׁבוּעוֹת הִתִּיר רַק לְהַקְרִיב בַּמִּקְדָּשׁ מִן הַחִטָּה הַחֲדָשָׁה.
בַּחֲנוּת בְּאֶרֶץ יִשְׂרָאֵל רָאָה נָתָן עוּגִיּוֹת מִחוּץ לָאָרֶץ, עִם הֶכְשֵׁר שֶׁל חוּץ לָאָרֶץ בִּלְבַד. מָה הַדִּין?
💡 בְּאֶרֶץ יִשְׂרָאֵל מַחְמִירִים בִּסְפֵק ״חָדָשׁ״. כָּל מוּצָר שֶׁיֵּשׁ בּוֹ דָּגָן בְּכַשְׁרוּת שֶׁל חוּץ לָאָרֶץ, בְּלִי אִשּׁוּר שֶׁל הָרַבָּנוּת, וְלֹא צֻיַּן בּוֹ שֶׁהוּא מִתְּבוּאָה יְשָׁנָה – אָסוּר מֵחֲמַת הַסָּפֵק, כִּי רֹב הַכַּשְׁרֻיּוֹת בְּחוּץ לָאָרֶץ מְקִלּוֹת בָּזֶה.
🔑 מוּשָּׂג: עַד אֵיזֶה יוֹם אָסוּר לֶאֱכֹל מֵהַתְּבוּאָה הַ'חָדָשׁ' שֶׁל הַשָּׁנָה?
💡 לְאַחַר שֶׁחָרַב הַמִּקְדָּשׁ, מֻתָּר לֶאֱכֹל מֵהַתְּבוּאָה הַחֲדָשָׁה רַק מִסּוֹף יוֹם ט״ז בְּנִיסָן, יוֹם הָעֹמֶר.

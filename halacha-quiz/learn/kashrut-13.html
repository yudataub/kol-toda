עֲצֵי פְּרִי וּבַל תַּשְׁחִית

🍽️ הִלְכוֹת כַּשְׁרוּת · פֶּרֶק י״ג

🌳עֲצֵי פְּרִי

מִי שֶׁנּוֹטֵעַ עֵץ פְּרִי מוֹסִיף מָזוֹן לָעוֹלָם, וְנַעֲשֶׂה שֻׁתָּף לַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁנָּטַע בְּעַצְמוֹ גַּן בְּעֵדֶן. הַתּוֹרָה אָסְרָה לַחַיָּלִים, אֲפִלּוּ בִּשְׁעַת מִלְחָמָה, לִכְרֹת עֲצֵי פְּרִי: "לֹא תַשְׁחִית אֶת עֵצָהּ".
גַּם עֵצִים, שִׂיחִים וּפְרָחִים שֶׁמְּגַדְּלִים בַּגִּנּוֹת יֵשׁ לָהֶם עֵרֶךְ, וְהַמַּשְׁחִית אוֹתָם עוֹבֵר עַל אִסּוּר. וּמִצְוָה לִשְׁמֹר עַל נִקְיוֹן הַטֶּבַע וְלֹא לִזְרֹק פְּסֹלֶת אֶלָּא לַפַּח.
💛 אַהֲבַת הַשֵּׁם: "כִּי הָאָדָם עֵץ הַשָּׂדֶה" – כְּמוֹ עֵץ פְּרִי, גַּם אָדָם טוֹב נוֹתֵן מִטּוּבוֹ לַאֲחֵרִים. כְּשֶׁאֲנַחְנוּ שׁוֹמְרִים עַל הָעֵצִים, אֲנַחְנוּ שׁוֹמְרִים עַל הָעוֹלָם הַיָּפֶה שֶׁה׳ בָּרָא בִּשְׁבִילֵנוּ.
📖 פֶּרֶק י״ג, הֲלָכוֹת א׳–ב׳, י״ג

🪓מָתַי מֻתָּר לַעֲקֹר עֵץ פְּרִי

אִם אֲפִלּוּ בְּמִלְחָמָה אָסוּר לְהַשְׁחִית עֲצֵי פְּרִי – קַל וָחֹמֶר בִּימֵי שָׁלוֹם, וּמִכָּאן נִלְמָד כָּל אִסּוּר "בַּל תַּשְׁחִית". אֲבָל כְּשֶׁעוֹקְרִים עֵץ לְצֹרֶךְ חָשׁוּב בְּהַרְבֵּה, כְּמוֹ לִבְנוֹת בַּיִת לְמִשְׁפָּחָה, אוֹ כְּשֶׁהָעֵץ מַזִּיק – שׁוֹרָשָׁיו מְבַקְּעִים צִנּוֹרוֹת אוֹ שֶׁהוּא מֵבִיא חֲרָקִים רַבִּים לַבַּיִת – אֵין זוֹ הַשְׁחָתָה.
עֵץ שֶׁמַּסְתִּיר אֶת הָאוֹר בַּחַלּוֹן – אִם אֶפְשָׁר, מִסְתַּפְּקִים בְּקִצּוּץ הָעֲנָפִים, גַּם אִם צָרִיךְ לַחֲזֹר עַל זֶה כָּל כַּמָּה חֳדָשִׁים. וּבְסָפֵק – שׁוֹאֲלִים רַב.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה מְלַמֶּדֶת אוֹתָנוּ לְכַבֵּד כָּל דָּבָר שֶׁמֵּבִיא בְּרָכָה לָעוֹלָם. גַּם כְּשֶׁמֻּתָּר לַעֲקֹר – עוֹשִׂים זֹאת בְּמַחְשָׁבָה וּבִזְהִירוּת, וְלֹא בְּזִלְזוּל.
📖 פֶּרֶק י״ג, הֲלָכוֹת ב׳–ה׳, ז׳

🔨בַּל תַּשְׁחִית

מִצְוָה מִן הַתּוֹרָה לֹא לְהַשְׁחִית שׁוּם דָּבָר מוֹעִיל: אֹכֶל, כֵּלִים, בְּגָדִים וְגַם אֶת הַטֶּבַע. מִי שֶׁשּׁוֹבֵר כְּלִי אוֹ קוֹרֵעַ בֶּגֶד בְּלִי צֹרֶךְ – עוֹבֵר עַל "בַּל תַּשְׁחִית".
כְּשֶׁיֵּשׁ סִבָּה חֲשׁוּבָה – זוֹ לֹא הַשְׁחָתָה. לָכֵן הֶחָתָן שׁוֹבֵר כּוֹס מִתַּחַת לַחֻפָּה, כְּדֵי לִזְכֹּר אֶת חֻרְבַּן בֵּית הַמִּקְדָּשׁ. אֲבָל לִשְׁבֹּר כֵּלִים מִתּוֹךְ כַּעַס – אָסוּר.
💛 אַהֲבַת הַשֵּׁם: ה׳ הֶרְאָה לְאָדָם הָרִאשׁוֹן אֶת כָּל עֲצֵי גַּן עֵדֶן וְאָמַר לוֹ: "תֵּן דַּעְתְּךָ שֶׁלֹּא תְּקַלְקֵל וְתַחֲרִיב אֶת עוֹלָמִי". מִי שֶׁשּׁוֹמֵר עַל הַחֲפָצִים מַרְאֶה שֶׁהוּא מַעֲרִיךְ אֶת הַמַּתָּנוֹת שֶׁל ה׳.
📖 פֶּרֶק י״ג, הֲלָכוֹת ז׳–ח׳

🌍שׁוֹמְרִים עַל הַכֹּל

שְׁלוֹשָׁה דְּבָרִים לָמַדְנוּ לִשְׁמֹר: אֶת הָעוֹלָם – "לֹא תַשְׁחִית"; אֶת הַגּוּף – "וְנִשְׁמַרְתֶּם מְאֹד לְנַפְשֹׁתֵיכֶם"; וְאֶת קְדֻשַּׁת הָאֲכִילָה – בְּהֶכְשֵׁר וּבִטְבִילַת כֵּלִים.
אִם אָסוּר לְהַשְׁחִית אֹכֶל – כָּל שֶׁכֵּן שֶׁאָסוּר לְהַשְׁחִית אֶת הַגּוּף בִּפְגִיעָה בַּבְּרִיאוּת.
💛 אַהֲבַת הַשֵּׁם: "וְכָל מַה שֶּׁבָּרָאתִי – בִּשְׁבִילְךָ בָּרָאתִי", אָמַר ה׳ לְאָדָם הָרִאשׁוֹן. מִי שֶׁשּׁוֹמֵר עַל הָעוֹלָם וְעַל גּוּפוֹ – מְכַבֵּד אֶת הַמַּתָּנָה שֶׁל ה׳.
📖 פֶּרֶק י״ג, הֲלָכָה ז׳; פֶּרֶק ל״ו, הֲלָכָה א׳

🍞כְּבוֹד הַלֶּחֶם

חֲכָמִים הִזְהִירוּ לִשְׁמֹר עַל הָאֹכֶל, וּבְמִיּוּחָד עַל הַלֶּחֶם. לֹא מַעֲבִירִים כּוֹס מְלֵאָה מֵעַל לֶחֶם, שֶׁמָּא תִּשָּׁפֵךְ עָלָיו, וְלֹא מַנִּיחִים עָלָיו כְּלִי מְלֻכְלָךְ. וְלֹא יוֹשְׁבִים עַל תִּיק שֶׁיֵּשׁ בּוֹ אֹכֶל שֶׁעָלוּל לְהִמָּעֵךְ.
לֶחֶם אָסוּר לִזְרֹק, אֲפִלּוּ עַל שֻׁלְחָן נָקִי. סֻכָּרִיּוֹת עֲטוּפוֹת מֻתָּר לִזְרֹק, כִּי הֵן לֹא מִתְלַכְלְכוֹת. וּמִי שֶׁמִּשְׁתַּמֵּשׁ בַּלֶּחֶם כְּמוֹ כַּף – יֹאכַל אוֹתוֹ בַּסּוֹף.
💛 אַהֲבַת הַשֵּׁם: הַלֶּחֶם הוּא עִקַּר הַמָּזוֹן שֶׁה׳ נוֹתֵן לָנוּ. כְּשֶׁאֲנַחְנוּ מְכַבְּדִים אוֹתוֹ, אֲנַחְנוּ אוֹמְרִים תּוֹדָה לַזָּן וּמְפַרְנֵס לַכֹּל.
📖 פֶּרֶק י״ג, הֲלָכָה ט׳

🍽️לֹא לְהַגְזִים, וְלֹא לִזְרֹק

מִי שֶׁשָּׂבֵעַ אֵינוֹ צָרִיךְ לְסַיֵּם אֶת מָה שֶׁבַּצַּלַּחַת, כִּי אֲכִילָה יְתֵרָה מַזִּיקָה לַגּוּף – וְהַגּוּף חָשׁוּב יוֹתֵר. לָכֵן הָעִקָּר הוּא לֹא לְמַלֵּא אֶת הַצַּלַּחַת יוֹתֵר מִדַּי מֵרֹאשׁ.
אַחֲרֵי סְעוּדָה גְּדוֹלָה לֹא זוֹרְקִים אֹכֶל טוֹב לַפַּח: שׁוֹמְרִים בַּמְּקָרֵר, וּמְחַלְּקִים לִקְרוֹבִים, לִשְׁכֵנִים אוֹ לַעֲנִיִּים. וּבְגָדִים וַחֲפָצִים שֶׁעוֹד טוֹבִים – מַעֲבִירִים לְמִי שֶׁיֵּהָנֶה מֵהֶם.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנַחְנוּ נוֹתְנִים לְאַחֵר אֶת מָה שֶׁנִּשְׁאַר לָנוּ, אֲנַחְנוּ מַצִּילִים אֶת הַדָּבָר וְגַם מְקַיְּמִים מִצְוַת צְדָקָה. כָּךְ מַה שֶּׁה׳ נָתַן לָנוּ מַמְשִׁיךְ לְהָבִיא בְּרָכָה.
📖 פֶּרֶק י״ג, הֲלָכוֹת י׳–י״א

🧹פֵּרוּרִים וּשְׁאֵרִיּוֹת

חֲתִיכוֹת לֶחֶם גְּדוֹלוֹת שֶׁנִּשְׁאֲרוּ לֹא מַרְטִיבִים בִּשְׁטִיפַת הַשֻּׁלְחָן, אֶלָּא עוֹטְפִים וּמַנִּיחִים בַּפַּח בְּדֶרֶךְ כָּבוֹד. גַּם פֵּרוּרִים קְטַנִּים לֹא מַשְׁלִיכִים עַל הָרִצְפָּה, אֶלָּא אוֹסְפִים לַפַּח. מְעַט פֵּרוּרִים מֻתָּר לְפַזֵּר בֶּחָצֵר לַצִּפּוֹרִים.
לֶחֶם שֶׁמֻּנָּח בַּדֶּרֶךְ וַאֲנָשִׁים דּוֹרְכִים עָלָיו – מְרִימִים אוֹתוֹ לַצַּד. וְלֹא מַנִּיחִים אֹכֶל וּשְׁתִיָּה מִתַּחַת לַמִּטָּה.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים אָמְרוּ שֶׁהַמַּלְאָךְ הַמְמֻנֶּה עַל הַפַּרְנָסָה שׁוֹכֵן בְּמָקוֹם נָקִי. מִי שֶׁמְּכַבֵּד אֶת הַטּוֹבָה שֶׁה׳ נוֹתֵן – זוֹכֶה לִבְרָכָה.
📖 פֶּרֶק י״ג, הֲלָכָה י״ב; פֶּרֶק ל״ו, הֲלָכָה ח׳

🧠 בְּדֹק אֶת עַצְמְךָ

עֵץ לִימוֹן בֶּחָצֵר מַסְתִּיר אֶת הָאוֹר בְּחַלּוֹן הַחֶדֶר. מָה נָכוֹן לַעֲשׂוֹת?
💡 הַסְתָּרַת הַחַלּוֹן הִיא נֵזֶק, אֲבָל כְּשֶׁאֶפְשָׁר לְתַקֵּן בְּקִצּוּץ הָעֲנָפִים – לֹא עוֹקְרִים, גַּם אִם זֶה טֹרַח. וּמֻתָּר לִגְזֹם עֲנָפִים, כִּי הָעֵץ נִשְׁאָר בָּרִיא.
הַתּוֹרָה אָסְרָה עַל חַיָּלִים לִכְרֹת עֲצֵי פְּרִי בְּמִלְחָמָה. מָה לוֹמְדִים מִזֶּה לִימֵי שָׁלוֹם?
💡 אִם אֲפִלּוּ בְּסַעֲרַת הַמִּלְחָמָה אָסוּר לְהַשְׁחִית – קַל וָחֹמֶר בִּימֵי שָׁלוֹם. מִכָּאן נִלְמָד הָאִסּוּר לְהַשְׁחִית כֵּלִים, בְּגָדִים, אֹכֶל וְטֶבַע.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

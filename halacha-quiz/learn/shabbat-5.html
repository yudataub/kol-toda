תּוֹרָה וּתְפִלָּה בְּשַׁבָּת

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק ה׳

📖שַׁבָּת – יוֹם שֶׁל תּוֹרָה

״לֹא נִתְּנוּ שַׁבָּתוֹת אֶלָּא לַעֲסֹק בָּהֶם בְּדִבְרֵי תוֹרָה״. חֲכָמִים אָמְרוּ: חֲצִי הַשַּׁבָּת לַה׳ – בְּתוֹרָה וּתְפִלָּה, וְחֶצְיָהּ לְעֹנֶג – בַּאֲכִילָה, שְׁתִיָּה וּמְנוּחָה. לוֹמְדִים בְּשִׂמְחָה, וּבְעִקָּר דְּבָרִים שֶׁמַּדְרִיכִים אֵיךְ לִחְיוֹת – הֲלָכָה, אֱמוּנָה וּמוּסָר.
אַבָּא שֶׁלּוֹמֵד תּוֹרָה עִם בָּנָיו – כְּאִלּוּ לִמֵּד גַּם אֶת נְכָדָיו עַד סוֹף כָּל הַדּוֹרוֹת. וְהַשַּׁבָּת, שֶׁבָּהּ נִתְּנָה הַתּוֹרָה, הִיא הַיּוֹם הַמַּתְאִים בְּיוֹתֵר.
💛 אַהֲבַת הַשֵּׁם: בְּמוֹצָאֵי שַׁבָּת שׁוֹאֵל ה׳ אֶת כָּל יְהוּדִי: אֵיזֶה חִדּוּשׁ חִדַּשְׁתָּ הַשַּׁבָּת? הַתּוֹרָה בְּשַׁבָּת אֵינָהּ מַשָּׂא – הִיא עֹנֶג שֶׁל הַנְּשָׁמָה.
📖 פֶּרֶק ה׳, הֲלָכוֹת א׳–ב׳

😴שֵׁינָה וַאֲכִילָה – בְּמִדָּה

״שֵׁינָה בְּשַׁבָּת תַּעֲנוּג״ – מִי שֶׁיָּשֵׁן בְּכָל יוֹם שֶׁבַע שָׁעוֹת, יָכוֹל לִישֹׁן בְּשַׁבָּת קְצָת יוֹתֵר וְלִהְיוֹת רַעֲנָן. אֲבָל לֹא יִישַׁן בְּשַׁבָּת כְּדֵי לַעֲבֹד בְּמוֹצָאֵי שַׁבָּת – כִּי אָז הוּא עוֹשֶׂה אֶת הַשַּׁבָּת מְשָׁרֶתֶת שֶׁל יְמֵי הַחוֹל.
לֹא אוֹכְלִים יוֹתֵר מִדַּי בַּסְּעוּדוֹת: אֲכִילָה מֻפְרֶזֶת מְעַיֶּפֶת, וְאָז אֵין כֹּחַ לִלְמֹד וּלְשׂוֹחֵחַ יָפֶה עִם הַמִּשְׁפָּחָה.
💛 אַהֲבַת הַשֵּׁם: הַשַּׁבָּת אֵינָהּ יוֹם לְהַשְׁלָמַת שָׁעוֹת – הִיא הַיּוֹם הֶחָשׁוּב בְּיוֹתֵר בַּשָּׁבוּעַ. מְנוּחָה נְכוֹנָה נוֹתֶנֶת כֹּחַ לְהִתְעַנֵּג עַל ה׳ וְעַל תּוֹרָתוֹ.
📖 פֶּרֶק ה׳, הֲלָכָה ג׳

🏃רָצִים לִשְׁמֹעַ תּוֹרָה

מֹשֶׁה רַבֵּנוּ תִּקֵּן לִדְרֹשׁ בְּשַׁבָּת לִפְנֵי הַצִּבּוּר בְּעִנְיְנֵי הַיּוֹם. בְּשַׁבָּת לֹא הוֹלְכִים בִּמְהִירוּת כְּמוֹ בְּחוֹל, אֲבָל לִדְרָשָׁה שֶׁל תּוֹרָה – רָצִים: ״לְעוֹלָם יָרוּץ אָדָם לִדְבַר הֲלָכָה, וַאֲפִלּוּ בְּשַׁבָּת״. בִּזְמַן הַדְּרָשָׁה לֹא עוֹרְכִים סְעוּדָה וְלֹא מְטַיְּלִים.
נָהֲגוּ לָלֶכֶת בְּסוֹף הַתְּפִלָּה אֶל רַב בֵּית הַכְּנֶסֶת וְלוֹמַר לוֹ ״שַׁבָּת שָׁלוֹם״ – כָּךְ מְקַיְּמִים אֶת מִצְוַת הַקְבָּלַת פְּנֵי הָרַב.
💛 אַהֲבַת הַשֵּׁם: ״אַגְרָא דְּפִרְקָא רִהֲטָא״ – עִקַּר הַשָּׂכָר שֶׁל הַדְּרָשָׁה הוּא עַל הָרִיצָה אֵלֶיהָ. כְּשֶׁרָצִים לַתּוֹרָה, מַרְאִים לַה׳ כַּמָּה הִיא יְקָרָה לָנוּ – וְהַשְּׁכִינָה שׁוֹרָה בֵּינֵינוּ.
📖 פֶּרֶק ה׳, הֲלָכוֹת ד׳, ט״ו

📜קְרִיאַת הַתּוֹרָה

כְּבָר מִימֵי מֹשֶׁה רַבֵּנוּ קוֹרְאִים בְּסֵפֶר תּוֹרָה בְּשַׁבָּת, וּבְשַׁחֲרִית עוֹלִים שִׁבְעָה – כְּנֶגֶד שִׁבְעַת יְמֵי הַשָּׁבוּעַ. כָּל שַׁבָּת קוֹרְאִים אֶת פָּרָשַׁת הַשָּׁבוּעַ, וּבְכָל שָׁנָה מְסַיְּמִים אֶת כָּל הַתּוֹרָה בְּשִׂמְחַת תּוֹרָה. אִם בְּשַׁבָּת אַחַת לֹא קָרְאוּ – בַּשַּׁבָּת הַבָּאָה קוֹרְאִים שְׁתֵּי פָּרָשִׁיּוֹת.
אַחֲרֵי הַפָּרָשָׁה קוֹרֵא הַמַּפְטִיר ״הַפְטָרָה״ – קֶטַע מִן הַנְּבִיאִים שֶׁקָּשׁוּר לַפָּרָשָׁה אוֹ לַיּוֹם.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה נִתְּנָה בְּשַׁבָּת. כְּשֶׁכָּל עַם יִשְׂרָאֵל קוֹרֵא בְּאוֹתָהּ שַׁבָּת אֶת אוֹתָהּ פָּרָשָׁה – כֻּלָּנוּ כְּמוֹ מִשְׁפָּחָה אַחַת שֶׁעוֹמֶדֶת שׁוּב לְיַד הַר סִינַי.
📖 פֶּרֶק ה׳, הֲלָכוֹת ו׳–ז׳

📜הַהַפְטָרָה וּקְרִיאַת מִנְחָה

פַּעַם גָּזְרוּ הַשִּׁלְטוֹנוֹת שֶׁלֹּא לִקְרֹא בַּתּוֹרָה, וַחֲכָמִים תִּקְּנוּ לִקְרֹא בִּמְקוֹמָהּ בַּנְּבִיאִים. כְּשֶׁבָּטְלָה הַגְּזֵרָה הִמְשִׁיכוּ לִקְרֹא גַּם הַפְטָרָה – לְפָחוֹת עֶשְׂרִים וְאֶחָד פְּסוּקִים. הַמַּפְטִיר קוֹרֵא קֹדֶם כַּמָּה פְּסוּקִים בַּתּוֹרָה, כְּדֵי שֶׁלֹּא יֵרָאֶה שֶׁהַנָּבִיא שָׁוֶה לַתּוֹרָה.
עֶזְרָא הַסּוֹפֵר תִּקֵּן לִקְרֹא בַּתּוֹרָה גַּם בְּמִנְחָה שֶׁל שַׁבָּת: שְׁלֹשָׁה עוֹלִים קוֹרְאִים אֶת תְּחִלַּת הַפָּרָשָׁה הַבָּאָה – בִּשְׁבִיל מִי שֶׁעָסוּק כָּל הַשָּׁבוּעַ וְלֹא שָׁמַע אוֹתָהּ בְּשֵׁנִי וּבַחֲמִישִׁי.
💛 אַהֲבַת הַשֵּׁם: דָּוִד הַמֶּלֶךְ אָמַר: אַחֲרֵי סְעוּדָה רַבִּים הוֹלְכִים לְבַטָּלָה, וַאֲנַחְנוּ – אַחֲרֵי שֶׁאָכַלְנוּ וְשָׁתִינוּ – בָּאִים לְהִתְפַּלֵּל: ״וַאֲנִי תְפִלָּתִי לְךָ ה׳ עֵת רָצוֹן״.
📖 פֶּרֶק ה׳, הֲלָכוֹת ז׳–ח׳

📘שְׁנַיִם מִקְרָא וְאֶחָד תַּרְגּוּם

חֲכָמִים תִּקְּנוּ שֶׁכָּל אֶחָד יִקְרָא בְּכָל שָׁבוּעַ אֶת הַפָּרָשָׁה פַּעֲמַיִם, וּפַעַם אַחַת אֶת תַּרְגּוּם אֻנְקְלוֹס – אוֹ אֶת פֵּרוּשׁ רַשִׁ״י. הָעוֹשֶׂה כֵּן – מַאֲרִיכִים לוֹ יָמָיו וּשְׁנוֹתָיו. אֶפְשָׁר לִקְרֹא הַכֹּל בְּיוֹם שִׁשִּׁי, אוֹ עֲלִיָּה אַחַת בְּכָל יוֹם.
הַזְּמַן הַטּוֹב – עַד סְעוּדַת שַׁבָּת בַּבֹּקֶר. מִי שֶׁלֹּא הִסְפִּיק – מַשְׁלִים עַד מִנְחָה, וְאִם לֹא – עַד סוֹף יוֹם שְׁלִישִׁי.
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁקּוֹרֵא אֶת הַפָּרָשָׁה בְּעַצְמוֹ מַגִּיעַ לַשַּׁבָּת כְּשֶׁהוּא כְּבָר מַכִּיר אוֹתָהּ – וְהַקְּרִיאָה בְּבֵית הַכְּנֶסֶת נַעֲשֵׂית פְּגִישָׁה עִם חָבֵר טוֹב.
📖 פֶּרֶק ה׳, הֲלָכוֹת ט׳–י׳

🙏תְּפִלּוֹת הַשַּׁבָּת וְ״וַיְכֻלּוּ״

לִפְנֵי יוֹתֵר מֵאַרְבַּע מֵאוֹת שָׁנָה הִתְחִילוּ הַמְּקֻבָּלִים בִּצְפַת לְקַבֵּל אֶת הַשַּׁבָּת בְּמִזְמוֹרִים, וְרַבִּי שְׁלֹמֹה אַלְקָבֵּץ חִבֵּר אֶת ״לְכָה דוֹדִי״. בַּעֲמִידָה שֶׁל שַׁבָּת שֶׁבַע בְּרָכוֹת: שָׁלֹשׁ רִאשׁוֹנוֹת, שָׁלֹשׁ אַחֲרוֹנוֹת, וּבָאֶמְצַע בְּרָכָה אַחַת עַל קְדֻשַּׁת הַשַּׁבָּת. וּמוֹסִיפִים תְּפִלַּת מוּסָף.
אֶת ״וַיְכֻלּוּ״ אוֹמְרִים שָׁלֹשׁ פְּעָמִים בְּלֵיל שַׁבָּת: בַּעֲמִידָה; אַחֲרֶיהָ – כָּל הַצִּבּוּר יַחַד, בְּקוֹל וּבַעֲמִידָה; וּבַקִּדּוּשׁ.
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁאוֹמֵר ״וַיְכֻלּוּ״ מֵעִיד שֶׁה׳ בָּרָא אֶת הָעוֹלָם – וְנַעֲשֶׂה כְּאִלּוּ שֻׁתָּף לַה׳ בְּמַעֲשֵׂה בְרֵאשִׁית. שְׁנֵי מַלְאָכִים מַנִּיחִים יְדֵיהֶם עַל רֹאשׁוֹ וְאוֹמְרִים: ״וְסָר עֲוֹנֶךָ״.
📖 פֶּרֶק ה׳, הֲלָכוֹת י״א–י״ב, י״ד

🌙טָעוּת בַּתְּפִלָּה

בַּעֲמִידָה שֶׁל שַׁבָּת לֹא אוֹמְרִים אֶת שְׁלֹשׁ עֶשְׂרֵה הַבְּרָכוֹת הָאֶמְצָעִיּוֹת שֶׁל חוֹל – כְּדֵי לֹא לְהַטְרִיחַ, וְכִי לֹא מַתְאִים לְבַקֵּשׁ בְּשַׁבָּת עַל דְּבָרִים שֶׁעֲלוּלִים לְצַעֵר. מִי שֶׁהִתְחִיל בְּטָעוּת בְּרָכָה שֶׁל חוֹל – מְסַיֵּם אוֹתָהּ, וְאָז עוֹבֵר לְבִרְכַּת הַשַּׁבָּת.
מִי שֶׁשָּׁכַח לְגַמְרֵי אֶת בִּרְכַּת הַשַּׁבָּת וְנִזְכַּר לִפְנֵי סוֹף הַתְּפִלָּה – חוֹזֵר לְבִרְכַּת הַשַּׁבָּת. וְאִם כְּבָר סִיֵּם אֶת כָּל הַתְּפִלָּה – חוֹזֵר וּמִתְפַּלֵּל מֵהַהַתְחָלָה.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת אֲנַחְנוּ לֹא בָּאִים לַה׳ עִם רְשִׁימַת בַּקָּשׁוֹת, אֶלָּא כְּמוֹ בֵּן שֶׁבָּא פָּשׁוּט לִהְיוֹת עִם אָבִיו – וּמְבַקֵּשׁ רַק: ״רְצֵה בִמְנוּחָתֵנוּ״.
📖 פֶּרֶק ה׳, הֲלָכָה י״א

🌃מֵעֵין שֶׁבַע

פַּעַם הָיוּ בָּתֵּי הַכְּנֶסֶת בַּשָּׂדֶה, וְהָיָה מְסֻכָּן לַחֲזֹר מֵהֶם לְבַד בַּלַּיְלָה. לָכֵן תִּקְּנוּ שֶׁהַחַזָּן יֹאמַר אַחֲרֵי עַרְבִית שֶׁל שַׁבָּת בְּרָכָה מְיֻחֶדֶת, שֶׁיֵּשׁ בָּהּ תַּמְצִית שֶׁל שֶׁבַע בִּרְכוֹת הַשַּׁבָּת – וּבֵינְתַיִם הַמְאַחֲרִים מְסַיְּמִים וְחוֹזְרִים עִם כֻּלָּם.
אַף שֶׁהַיּוֹם בָּתֵּי הַכְּנֶסֶת בְּתוֹךְ הָעִיר, הַתַּקָּנָה נִשְׁאֲרָה בִּמְקוֹמָהּ. בְּמִנְיָן בְּבַיִת פְּרָטִי – יֵשׁ נוֹהֲגִים לְאָמְרָהּ וְיֵשׁ שֶׁלֹּא.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים דָּאֲגוּ שֶׁאַף אֶחָד לֹא יִשָּׁאֵר מֵאָחוֹר. כָּךְ גַּם אֲנַחְנוּ – מְחַכִּים לֶחָבֵר שֶׁמִּתְעַכֵּב, וְהוֹלְכִים יַחַד.

🧠 בְּדֹק אֶת עַצְמְךָ

לָמָּה הַמַּפְטִיר קוֹרֵא קֹדֶם כַּמָּה פְּסוּקִים בַּתּוֹרָה, וְרַק אַחַר כָּךְ אֶת הַהַפְטָרָה?
💡 חֲכָמִים רָצוּ שֶׁיִּהְיֶה בָּרוּר שֶׁהַקְּרִיאָה בַּנָּבִיא אֵינָהּ שְׁקוּלָה לְבַדָּהּ כְּנֶגֶד הַתּוֹרָה. הַגְּזֵרָה הָיְתָה דַּוְקָא עַל הַתּוֹרָה, וְשִׁבְעַת הָעוֹלִים כְּבָר סִיְּמוּ אֶת הַפָּרָשָׁה.
לָמָּה תִּקֵּן עֶזְרָא לִקְרֹא בַּתּוֹרָה גַּם בְּמִנְחָה שֶׁל שַׁבָּת?
💡 בְּמִנְחָה קוֹרְאִים שְׁלֹשָׁה עוֹלִים אֶת תְּחִלַּת הַפָּרָשָׁה הַבָּאָה. מִי שֶׁעָסוּק בַּחֲנוּתוֹ וְלֹא בָּא בְּשֵׁנִי וּבַחֲמִישִׁי – שׁוֹמֵעַ אוֹתָהּ לְפָחוֹת פַּעַם אַחַת, בְּשַׁבָּת, כְּשֶׁכֻּלָּם פְּנוּיִים.
דָּנִי לֹא סִיֵּם ״שְׁנַיִם מִקְרָא וְאֶחָד תַּרְגּוּם״ עַד סְעוּדַת שַׁבָּת בַּבֹּקֶר. מָה יַעֲשֶׂה?
💡 מִי שֶׁלֹּא הִסְפִּיק עַד הַסְּעוּדָה – מַשְׁלִים עַד מִנְחָה, וְאַחַר כָּךְ עַד סוֹף יוֹם שְׁלִישִׁי, כִּי שְׁלֹשֶׁת הַיָּמִים הָרִאשׁוֹנִים קְשׁוּרִים לַשַּׁבָּת שֶׁעָבְרָה. וּבְכָל מִקְרֶה מַשְׁלִימִים עַד שִׂמְחַת תּוֹרָה.

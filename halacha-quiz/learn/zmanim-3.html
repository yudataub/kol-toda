מִנְהֲגֵי אֲבֵלוּת בִּסְפִירַת הָעֹמֶר

📅 רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת · פֶּרֶק ג׳

📚תַּלְמִידֵי רַבִּי עֲקִיבָא

בֵּין פֶּסַח לְשָׁבוּעוֹת מֵתוּ עֶשְׂרִים וְאַרְבָּעָה אֶלֶף תַּלְמִידֵי רַבִּי עֲקִיבָא, מִפְּנֵי שֶׁלֹּא נָהֲגוּ כָּבוֹד זֶה לָזֶה. לָכֵן נוֹהֲגִים בַּיָּמִים הָאֵלֶּה מִקְצָת מִנְהֲגֵי אֲבֵלוּת: לֹא מִתְחַתְּנִים, לֹא מִסְתַּפְּרִים, וְלֹא עוֹרְכִים רִקּוּדִים.
רַבִּי עֲקִיבָא לֹא הִתְיָאֵשׁ. הוּא לִמֵּד תַּלְמִידִים חֲדָשִׁים – רַבִּי מֵאִיר, רַבִּי יְהוּדָה, רַבִּי יוֹסֵי, רַבִּי שִׁמְעוֹן וְרַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ – וְהֵם הֶעֱמִידוּ אֶת הַתּוֹרָה.
💛 אַהֲבַת הַשֵּׁם: בַּיָּמִים הָאֵלֶּה אָנוּ מִשְׁתַּדְּלִים לְתַקֵּן אֶת הַיְּחָסִים שֶׁבֵּין אָדָם לַחֲבֵרוֹ. מִי שֶׁמְּכַבֵּד אֶת חֲבֵרוֹ – מְכַבֵּד אֶת ה׳ שֶׁבָּרָא אוֹתוֹ.
📖 פֶּרֶק ג׳, הֲלָכָה א׳

🤝כָּבוֹד זֶה לָזֶה

רַבִּי עֲקִיבָא אָמַר לְתַלְמִידָיו הַחֲדָשִׁים: "בָּנַי, הָרִאשׁוֹנִים לֹא מֵתוּ אֶלָּא שֶׁהָיְתָה עֵינָם צָרָה אֵלּוּ לָאֵלּוּ – תְּנוּ דַּעְתְּכֶם שֶׁלֹּא תַּעֲשׂוּ כְּמַעֲשֵׂיהֶם". וְהַתַּלְמִידִים הַחֲדָשִׁים הֶעֱמִידוּ אֶת הַתּוֹרָה לְכָל הַדּוֹרוֹת.
אֶחָד מֵהֶם הָיָה רַבִּי שִׁמְעוֹן בַּר יוֹחָאי, שֶׁהִתְחַבֵּא עִם בְּנוֹ בִּמְעָרָה שְׁתֵּים עֶשְׂרֵה שָׁנָה וְלָמַד שָׁם תּוֹרָה.
💛 אַהֲבַת הַשֵּׁם: תּוֹרָה בְּלִי כָּבוֹד לַחֲבֵרִים אֵינָהּ שְׁלֵמָה. כְּשֶׁאָנוּ שְׂמֵחִים בְּהַצְלָחַת הֶחָבֵר – אָנוּ מַמְשִׁיכִים אֶת דַּרְכָּם שֶׁל רַבִּי מֵאִיר וְרַבִּי שִׁמְעוֹן.
📖 פֶּרֶק ג׳, הֲלָכָה א׳; פֶּרֶק ה׳, הֲלָכָה ג׳

📆עַד מָתַי הָאֲבֵלוּת?

לְמִנְהַג הָאַשְׁכְּנַזִּים – לֹא מִסְתַּפְּרִים וְלֹא מְנַגְּנִים עַד ל״ג בָּעֹמֶר, וּבְיוֹם ל״ג עַצְמוֹ כְּבָר מֻתָּר לְהִסְתַּפֵּר. אַחַר כָּךְ נִמְנָעִים רַק מִשְּׂמָחוֹת גְּדוֹלוֹת. לְמִנְהַג רַבִּים מִן הַסְּפָרַדִּים – הָאֲבֵלוּת נִמְשֶׁכֶת עַד הַבֹּקֶר שֶׁל ל״ד בָּעֹמֶר: בְּל״ג בָּעֹמֶר מֻתָּר לָשִׁיר וְלִרְקֹד לִכְבוֹד רַבִּי שִׁמְעוֹן, אֲבָל לֹא לְהִסְתַּפֵּר.
בִּקְהִלּוֹת רַבּוֹת שֶׁל סְפָרַדִּים, כְּמוֹ בְּטוּרְקִיָּה וּבְמִצְרַיִם, מְסַיְּמִים אֶת הָאֲבֵלוּת בְּל״ג בָּעֹמֶר. וּכְשֶׁל״ג בָּעֹמֶר חָל בְּיוֹם שִׁשִּׁי – גַּם הַסְּפָרַדִּים מִסְתַּפְּרִים בּוֹ לִכְבוֹד שַׁבָּת. וְגַם לְנָשִׁים וּלְבָנוֹת אָסוּר לְהִסְתַּפֵּר בִּימֵי הָאֲבֵלוּת.
💛 אַהֲבַת הַשֵּׁם: כָּל עֵדָה שָׁמְרָה עַל הַמָּסֹרֶת שֶׁלָּהּ בִּמְסִירוּת. כָּאן רוֹאִים כַּמָּה עַם יִשְׂרָאֵל זוֹכֵר אֶת הַצַּעַר – וְגַם יוֹדֵעַ לִשְׂמֹחַ בַּזְּמַן הַנָּכוֹן.
📖 פֶּרֶק ג׳, הֲלָכוֹת ג׳–ד׳, ו׳; פֶּרֶק ה׳, הֲלָכָה א׳

✂️לֹא מִסְתַּפְּרִים

בִּימֵי הָאֲבֵלוּת שֶׁבַּסְּפִירָה לֹא מִסְתַּפְּרִים, וְגַם אֶת הַיְּלָדִים הַקְּטַנִּים אֵין מְסַפְּרִים. רַק לְצֹרֶךְ גָּדוֹל, כְּדֵי לִמְנֹעַ מֵהֶם צַעַר, מֻתָּר. וּמִי שֶׁהַשֵּׂעָר גּוֹרֵם לוֹ כְּאֵב רֹאשׁ אוֹ פְּצָעִים – רַשַּׁאי לְהִסְתַּפֵּר.
לִכְבוֹד בְּרִית מִילָה מֻתָּר לְבַעֲלֵי הַשִּׂמְחָה – אֲבִי הַבֵּן, הַסַּנְדָּק וְהַמּוֹהֵל – לְהִסְתַּפֵּר.
💛 אַהֲבַת הַשֵּׁם: הַשֵּׂעָר הָאָרֹךְ מַזְכִּיר לָנוּ בְּכָל יוֹם שֶׁאֵלֶּה יְמֵי צַעַר. וְדַוְקָא מִתּוֹךְ הַצַּעַר אָנוּ מִתְחַזְּקִים בְּאַהֲבַת חֲבֵרִים.
📖 פֶּרֶק ג׳, הֲלָכָה ו׳

✂️📜 לְשׁוֹן חֲכָמִים: אִסּוּר תִּסְפּוֹרֶת בַּסְּפִירָה

״כָּתְבוּ הָרִאשׁוֹנִים שֶׁאֵין לְהִסְתַּפֵּר בִּימֵי סְפִירַת הָעֹמֶר... לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז... עַד בֹּקֶר יוֹם ל״ג בָּעֹמֶר״.
📖 אִסּוּר תִּסְפּוֹרֶת – מִנְהַג אֲבֵלוּת שֶׁלֹּא לְהִסְתַּפֵּר בְּרֹב יְמֵי סְפִירַת הָעֹמֶר, עַד ל״ג בָּעֹמֶר.
💛 אַהֲבַת הַשֵּׁם: בְּתוֹך הַשִּׂמְחָה שֶׁל סְפִירַת הָעֹמֶר, נוֹהֲגִים גַּם קְצָת אֲבֵלוּת עַל תַּלְמִידֵי רַבִּי עֲקִיבָא שֶׁנִּפְטְרוּ – ה׳ מְלַמֵּד אוֹתָנוּ לִזְכּוֹר גַּם אֶת הַצַּעַר שֶׁבְּתוֹך הַדֶּרֶך לְמַתַּן תּוֹרָה. 🧒 וְאַתָּה? אִם צָרִיך תִּסְפּוֹרֶת בִּימֵי הַסְּפִירָה – שְׁאַל אֶת אַבָּא אוֹ הָרַב אֵיך נוֹהֲגִים בְּמִשְׁפַּחְתְּכֶם.
📖 פֶּרֶק ג׳, הֲלָכָה ו׳

🎵רִקּוּדִים וּכְלֵי נְגִינָה

בִּימֵי הָאֲבֵלוּת שֶׁבַּסְּפִירָה אֵין עוֹרְכִים רִקּוּדִים וּמְחוֹלוֹת שֶׁל רְשׁוּת, וְלֹא מְנַגְּנִים וְלֹא שׁוֹמְעִים כְּלֵי זֶמֶר. אֲבָל בִּסְעוּדַת מִצְוָה – בְּרִית מִילָה, פִּדְיוֹן הַבֵּן, סִיּוּם מַסֶּכֶת, וּבַר מִצְוָה בַּיּוֹם שֶׁהַנַּעַר נִכְנָס לְמִצְווֹת – מֻתָּר לָשִׁיר וְלִרְקֹד כָּרָגִיל.
מַכְנִיסִים סֵפֶר תּוֹרָה לְבֵית הַכְּנֶסֶת בְּשִׁירָה וּבְרִקּוּדִים, כִּי זוֹ שִׂמְחָה שֶׁל מִצְוָה. וּבְחֹל הַמּוֹעֵד שֶׁל פֶּסַח מְנַגְּנִים וְרוֹקְדִים, כִּי מִצְוָה לִשְׂמֹחַ בּוֹ.
💛 אַהֲבַת הַשֵּׁם: שִׂמְחָה שֶׁל מִצְוָה אֵינָהּ נִדְחֵית. גַּם בִּימֵי הַצַּעַר, כְּשֶׁאָנוּ שְׂמֵחִים יַחַד עִם הַתּוֹרָה – אָנוּ מְתַקְּנִים אֶת מָה שֶׁהֶחְסִירוּ הַתַּלְמִידִים.
📖 פֶּרֶק ג׳, הֲלָכוֹת ח׳–ט׳

🎧מוּזִיקָה, שֶׁהֶחֱיָנוּ וְטִיּוּלִים

שִׁירִים בְּלִי כְּלֵי נְגִינָה מֻתָּר לִשְׁמֹעַ. אֲבָל שִׁירִים עִם כְּלֵי נְגִינָה מִמַּכְשִׁיר שֶׁבַּבַּיִת – גְּדוֹלֵי הַפּוֹסְקִים, הָרַב מֹשֶׁה פַיינְשְׁטֵיין וְהָרַב עוֹבַדְיָה יוֹסֵף, אוֹסְרִים, כְּמוֹ נְגִינָה חַיָּה. הָרַב מֶלָמֵד כּוֹתֵב שֶׁהָרוֹצֶה לְהָקֵל רַשַּׁאי לִשְׁמֹעַ שִׁירִים רְגִילִים אוֹ עֲצוּבִים – לֹא שִׁירֵי שִׂמְחָה, וְלֹא בְּקוֹל רָם.
מֻתָּר לְבָרֵךְ "שֶׁהֶחֱיָנוּ" עַל פְּרִי חָדָשׁ אוֹ עַל בֶּגֶד חָדָשׁ, וּמֻתָּר לְטַיֵּל. אֲבָל טִיּוּל שֶׁל בֵּית סֵפֶר, שֶׁיֵּשׁ בּוֹ שִׂמְחָה גְּדוֹלָה, מוּטָב לְקַיֵּם אַחֲרֵי ל״ג בָּעֹמֶר. טִיּוּל חִנּוּכִי – מֻתָּר.
💛 אַהֲבַת הַשֵּׁם: נִמְנָעִים מִשִּׂמְחָה מְיֻחֶדֶת, וְלֹא מִכָּל הֲנָאָה. הַתּוֹרָה אֵינָהּ רוֹצָה שֶׁנִּהְיֶה עֲצוּבִים – רַק שֶׁנִּזְכֹּר.
📖 פֶּרֶק ג׳, הֲלָכוֹת י׳–י״א

🧠 בְּדֹק אֶת עַצְמְךָ

לְמִנְהַג הַסְּפָרַדִּים שֶׁנּוֹהֲגִים אֲבֵלוּת עַד ל״ד, מָה מֻתָּר בְּיוֹם ל״ג בָּעֹמֶר?
💡 לְמִנְהָג זֶה, בְּל״ג בָּעֹמֶר שָׁרִים וְרוֹקְדִים לִכְבוֹד הִלּוּלַת רַבִּי שִׁמְעוֹן, אֲבָל אֵין מִסְתַּפְּרִים וְאֵין מִתְחַתְּנִים עַד הַבֹּקֶר שֶׁל ל״ד. אִם ל״ג חָל בְּיוֹם שִׁשִּׁי – מִסְתַּפְּרִים לִכְבוֹד שַׁבָּת.
לְמִנְהַג הָאַשְׁכְּנַזִּים, מָתַי כְּבָר מֻתָּר לְהִסְתַּפֵּר?
💡 לְמִנְהַג הָאַשְׁכְּנַזִּים, בְּיוֹם ל״ג בָּעֹמֶר מֻתָּר לְהִסְתַּפֵּר. בְּלֵיל ל״ג נוֹהֲגִים לְכַתְּחִלָּה לְהַחְמִיר, וּבִשְׁעַת הַצֹּרֶךְ אֶפְשָׁר לְהָקֵל.
בֵּית הַסֵּפֶר רוֹצֶה לָצֵאת לְטִיּוּל שָׁנָתִי שָׂמֵחַ בִּימֵי הַסְּפִירָה. מָה עָדִיף?
💡 טִיּוּלִים מֻתָּרִים בַּסְּפִירָה. אֲבָל טִיּוּל שֶׁל בֵּית סֵפֶר יֵשׁ בּוֹ שִׂמְחָה גְּדוֹלָה, וּמוּטָב לְקַיְּמוֹ אַחֲרֵי ל״ג בָּעֹמֶר. טִיּוּל חִנּוּכִי מֻתָּר לְכַתְּחִלָּה.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

בּוֹנֶה וְסוֹתֵר בַּבַּיִת וּבַכֵּלִים

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק ט״ו

🏠בּוֹנֶה וְסוֹתֵר

מְלֶאכֶת בּוֹנֶה כּוֹלֶלֶת כָּל חִבּוּר קָבוּעַ לַבַּיִת אוֹ לַקַּרְקַע: אָסוּר לִתְקֹעַ מַסְמֵר בַּקִּיר, לְהַבְרִיג נוּרָה לַמְּנוֹרָה אוֹ לְהָסִיר חַלּוֹן. אֲבָל לִתְלוֹת תְּמוּנָה עַל מַסְמֵר שֶׁכְּבָר קָבוּעַ בַּקִּיר – מֻתָּר. וְגַם לִפְתֹּחַ וְלִסְגֹּר דְּלָתוֹת, חַלּוֹנוֹת וּבְרָזִים – מֻתָּר, כִּי כָּךְ מִשְׁתַּמְּשִׁים בָּהֶם.
אָסוּר לַעֲשׂוֹת כַּדּוּר שֶׁלֶג אוֹ בֻּבַּת שֶׁלֶג. וְנָהֲגוּ יִשְׂרָאֵל שֶׁלֹּא לְהִשְׁתַּמֵּשׁ בְּמִטְרִיָּה בְּשַׁבָּת, כִּי הִיא כְּמוֹ אֹהֶל.
💛 אַהֲבַת הַשֵּׁם: כָּל הַשָּׁבוּעַ אָנוּ בּוֹנִים אֶת הָעוֹלָם, כְּמוֹ שֶׁבָּנוּ אֶת הַמִּשְׁכָּן. בְּשַׁבָּת אָנוּ מַפְסִיקִים לִבְנוֹת – וְנִזְכָּרִים שֶׁאֶת הַבַּיִת הָאֲמִתִּי בּוֹנֶה ה׳.
📖 פֶּרֶק ט״ו, הֲלָכוֹת א׳, ג׳, ה׳

🔧תִּקּוּן כֵּלִים

אָסוּר לְתַקֵּן כְּלִי בְּשַׁבָּת: לִתְקֹעַ יָדִית לַפַּטִּישׁ, לְחַבֵּר רֶגֶל שֶׁל כִּסֵּא שֶׁנִּתְפָּרְקָה, אוֹ לְיַשֵּׁר כַּפִּית שֶׁהִתְעַקְּמָה – זוֹ מְלֶאכֶת "מַכֶּה בְּפַטִּישׁ", הַגִּמּוּר הָאַחֲרוֹן שֶׁל כְּלִי. אֲבָל מֻתָּר לִפְתֹּחַ וְלִסְגֹּר מִכְסֶה שֶׁל צִנְצֶנֶת, כִּי כָּךְ מִשְׁתַּמְּשִׁים בּוֹ. וְגַם פְּקַק פְּלַסְטִיק שֶׁל בַּקְבּוּק חָדָשׁ – מֻתָּר לִפְתֹּחַ, כִּי גַּם לִפְנֵי הַפְּתִיחָה הוּא כְּבָר הָיָה פְּקָק, וְגַם רֹב הַמַּחְמִירִים מַסְכִּימִים לְכָךְ.
כַּדּוּר אוֹ מִזְרַן אֲוִיר שֶׁכְּבָר נֻפְּחוּ פַּעַם אַחַת – מֻתָּר לְנַפֵּחַ שׁוּב, גַּם בְּמַשְׁאֵבָה יָדָנִית. אֲבָל בַּלּוֹן – אָסוּר לְנַפֵּחַ, שֶׁמָּא יִקְשְׁרוּ אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת אָנוּ מְקַבְּלִים אֶת הָעוֹלָם כְּמוֹ שֶׁהוּא, גַּם עִם כַּפִּית עֲקֻמָּה. הַמְּנוּחָה הַזֹּאת מְלַמֶּדֶת אוֹתָנוּ שֶׁלֹּא הַכֹּל תָּלוּי בָּנוּ – ה׳ מַנְהִיג אֶת עוֹלָמוֹ.
📖 פֶּרֶק ט״ו, הֲלָכוֹת א׳, ו׳, ח׳, י״ג

🪀מִשְׂחָקִים עַל הַקַּרְקַע

אָסוּר לְיַשֵּׁר קַרְקַע – לְמַלֵּא גֻּמָּה אוֹ לְהָסִיר גַּבְשׁוּשִׁית – כִּי זוֹ מְלֶאכֶת בּוֹנֶה. לָכֵן אָסוּר לְשַׂחֵק בְּגֻלּוֹת עַל הָאֲדָמָה, שֶׁמָּא יַשְׁווּ גֻּמּוֹת, וְאָסוּר לְטַאטֵא חָצֵר שֶׁאֵינָהּ מְרֻצֶּפֶת. עַל הָרִצְפָּה שֶׁבְּתוֹךְ הַבַּיִת – מֻתָּר לְשַׂחֵק, וּמֻתָּר לְטַאטֵא אוֹתָהּ.
בְּאַרְגַּז חוֹל עִם חוֹל דַּק וְיָבֵשׁ – מֻתָּר לְשַׂחֵק. אֲבָל בְּחוֹל לַח – אָסוּר, וְאָסוּר לְהַרְטִיב אֶת הַחוֹל.
💛 אַהֲבַת הַשֵּׁם: גַּם בַּמִּשְׂחָק אֶפְשָׁר לִזְכֹּר אֶת הַשַּׁבָּת. יֶלֶד שֶׁבּוֹחֵר מִשְׂחָק מֻתָּר בִּמְקוֹם מִשְׂחָק אָסוּר – עוֹשֶׂה נַחַת רוּחַ לַה׳, וְשָׂמֵחַ בַּשַּׁבָּת בְּלֵב שָׁלֵם.
📖 פֶּרֶק ט״ו, הֲלָכוֹת ב׳, ט׳

⛺אֹהֶל

עֲשִׂיַּת אֹהֶל – גַּג שֶׁמֵּגֵן מִשֶּׁמֶשׁ וּמִגֶּשֶׁם – הִיא תּוֹלָדָה שֶׁל בּוֹנֶה, וַחֲכָמִים אָסְרוּ גַּם אֹהֶל זְמַנִּי. אֲבָל הִתִּירוּ לַעֲשׂוֹתוֹ בְּשִׁנּוּי הַסֵּדֶר: קֹדֶם מַחְזִיקִים אֶת הַגַּג בָּאֲוִיר, וְאַחַר כָּךְ מַכְנִיסִים מִתַּחְתָּיו אֶת הַקִּירוֹת אוֹ הָעַמּוּדִים. וּכְשֶׁמְּפָרְקִים – מוֹצִיאִים קֹדֶם אֶת הַקִּירוֹת, וְאַחַר כָּךְ אֶת הַגַּג.
יְלָדִים שֶׁרוֹצִים אֹהֶל מִשְּׂמִיכָה עַל כִּסְאוֹת – מַחְזִיקִים קֹדֶם אֶת הַשְּׂמִיכָה בָּאֲוִיר, וְאַחַר כָּךְ מַכְנִיסִים אֶת הַכִּסְאוֹת. עֲגָלַת יְלָדִים, כִּסֵּא וְשֻׁלְחָן מִתְקַפְּלִים – מֻתָּר לִפְתֹּחַ.
💛 אַהֲבַת הַשֵּׁם: בַּמִּדְבָּר יָשְׁבוּ יִשְׂרָאֵל בְּאֹהָלִים, וְה׳ שָׁמַר עֲלֵיהֶם בַּעֲנָנֵי הַכָּבוֹד. בְּשַׁבָּת אָנוּ זוֹכְרִים שֶׁהַהֲגַנָּה הָאֲמִתִּית שֶׁלָּנוּ בָּאָה מִן הַשָּׁמַיִם.
📖 פֶּרֶק ט״ו, הֲלָכוֹת ד׳–ה׳

🍰חוֹתְכִים עוּגָה – לֹא קֵיסָם

מְלֶאכֶת ״מְחַתֵּךְ״ הִיא חִתּוּךְ מְדֻיָּק בְּמִדָּה, כְּדֵי לַעֲשׂוֹת מִשֶּׁהוּ חָדָשׁ – כְּמוֹ עוֹר לְבֶגֶד אוֹ קְרָשִׁים לְבַיִת. הִיא אֵינָהּ חָלָה עַל מַאֲכָלִים: מֻתָּר לִפְרֹס עוּגָה לַחֲתִיכוֹת שָׁווֹת, וּלְשַׂרְטֵט חֲתָכִים עַל קְלִפַּת תַּפּוּז כְּדֵי לְקַלֵּף אוֹתוֹ.
אָסוּר לַעֲשׂוֹת קֵיסָם לַשִּׁנַּיִם מֵעָנָף קָשֶׁה. מִקַּשׁ אוֹ מִתֶּבֶן – מֻתָּר, כִּי הֵם רְאוּיִים לְמַאֲכַל בְּהֵמָה.
💛 אַהֲבַת הַשֵּׁם: עוּגָה שֶׁמְּחַלְּקִים בְּשָׁוֶה לְכָל בְּנֵי הַבַּיִת – הִיא חֵלֶק מֵעֹנֶג הַשַּׁבָּת וּמִן הָאַהֲבָה שֶׁבַּמִּשְׁפָּחָה.
📖 פֶּרֶק ט״ו, הֲלָכָה י׳

🥛פּוֹתְחִים שַׂקִּית חָלָב

מֻתָּר בְּשַׁבָּת לַעֲשׂוֹת חוֹר בְּשַׂקִּית חָלָב כְּדֵי לִמְזֹג, לִפְתֹּחַ שַׂקִּית סֻכָּר אוֹ סֻכָּרִיּוֹת, וְלִקְרֹעַ עֲטִיפָה שֶׁל וָפְלִים וְשׁוֹקוֹלָד – כִּי הָעֲטִיפָה נִזְרֶקֶת, וְהִיא כְּמוֹ קְלִפָּה שֶׁל פְּרִי.
יֵשׁ מַחְמִירִים לִקְרֹעַ אֶת הַשַּׂקִּית לְגַמְרֵי וּלְהַעֲבִיר אֶת הֶחָלָב לִכְלִי אַחֵר, אֲבָל הַהֲלָכָה כַּמְּקִלִּים. וּמִשְׁתַּדְּלִים לֹא לִקְרֹעַ אוֹתִיּוֹת.
💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לָנוּ אֶת הַשַּׁבָּת לְעֹנֶג – גַּם הַשּׁוֹקוֹלָד שֶׁל שַׁבָּת הוּא חֵלֶק מִכְּבוֹד הַשַּׁבָּת.
📖 פֶּרֶק ט״ו, הֲלָכָה י״ב

🧠 בְּדֹק אֶת עַצְמְךָ

שְׁנֵי אַחִים קְטַנִּים רוֹצִים לִבְנוֹת בַּסָּלוֹן אֹהֶל מִשְּׂמִיכָה וּמִכִּסְאוֹת. אֵיךְ יַעֲשׂוּ?
💡 חֲכָמִים אָסְרוּ לְהָקִים אֹהֶל זְמַנִּי כְּסֵדֶר הָרָגִיל – קֹדֶם קִירוֹת וְאַחַר כָּךְ גַּג. בְּשִׁנּוּי הַסֵּדֶר – קֹדֶם גַּג בָּאֲוִיר וְאַחַר כָּךְ קִירוֹת – הִתִּירוּ.
הַכַּדּוּר שֶׁל דָּנִי, שֶׁנֻּפַּח כְּבָר פְּעָמִים רַבּוֹת, הִתְרוֹקֵן בְּשַׁבָּת. מָה הַדִּין?
💡 כַּדּוּר שֶׁכְּבָר נֻפַּח פַּעַם אַחַת הוּא כְּבָר כְּלִי, וְנִפּוּחַ נוֹסָף אֵינוֹ עוֹשֶׂה אוֹתוֹ כְּלִי. בַּלּוֹן – אָסוּר לְנַפֵּחַ, שֶׁמָּא יִקְשְׁרוּ אוֹתוֹ.
בְּאֶמְצַע הַסְּעוּדָה הִתְעַקְּמָה מְעַט כַּפִּית. מָה הַדִּין?
💡 יִשּׁוּר כְּלִי שֶׁהִתְעַקֵּם הוּא תִּקּוּן שֶׁל כְּלִי – מְלֶאכֶת מַכֶּה בְּפַטִּישׁ, גַּם כְּשֶׁעוֹשִׂים זֹאת בַּיָּד. אֲבָל לְהִשְׁתַּמֵּשׁ בַּכַּפִּית כְּמוֹ שֶׁהִיא – מֻתָּר.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

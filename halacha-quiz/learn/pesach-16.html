לֵיל הַסֵּדֶר

🍷קַדֵּשׁ וְאַרְבַּע כּוֹסוֹת

פּוֹתְחִים אֶת הַסֵּדֶר בְּקִדּוּשׁ, וְהוּא הַכּוֹס הָרִאשׁוֹנָה מֵאַרְבַּע כּוֹסוֹת. מוֹזְגִים כּוֹס לְכָל אֶחָד מֵהַמְּסֻבִּים, וְכָל אֶחָד שׁוֹתֶה. תִּקְּנוּ אַרְבַּע כּוֹסוֹת לְשִׂמְחָה וּלְחֵרוּת, כְּנֶגֶד אַרְבַּע לְשׁוֹנוֹת שֶׁל גְּאֻלָּה – וְכָל חֵלֶק בַּהַגָּדָה נֶאֱמָר עַל כּוֹס.
לֹא מְקַדְּשִׁים לִפְנֵי צֵאת הַכּוֹכָבִים, כִּי אַרְבַּע הַכּוֹסוֹת שַׁיָּכוֹת לַלַּיְלָה. בְּסוֹף הַקִּדּוּשׁ מְבָרְכִים ״שֶׁהֶחֱיָנוּ״. וּמִי שֶׁשָּׁתָה אַרְבַּע כּוֹסוֹת בְּרֶצֶף – כְּאִלּוּ שָׁתָה כּוֹס אַחַת.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים בָּחֲרוּ לְקַדֵּשׁ עַל יַיִן מְשַׂמֵּחַ, לְלַמֵּד שֶׁהַקְּדֻשָּׁה מִתְגַּלָּה גַּם בַּחַיִּים הַשְּׂמֵחִים. אֶפְשָׁר לִהְיוֹת קָדוֹשׁ – וְגַם לִשְׂמֹחַ.
📖 פֶּרֶק ט״ז, הֲלָכוֹת ב׳, ה׳–ו׳

⏳בֵּין קִדּוּשׁ לְמַגִּיד

אֶת הַשֻּׁלְחָן עוֹרְכִים מֵרֹאשׁ, כְּדֵי לְקַדֵּשׁ מִיָּד אַחֲרֵי הַתְּפִלָּה, כְּשֶׁהַקְּטַנִּים עוֹד עֵרִים – אֲבָל לֹא לִפְנֵי צֵאת הַכּוֹכָבִים. מִי שֶׁשָּׁכַח ״שֶׁהֶחֱיָנוּ״ בַּקִּדּוּשׁ – מְבָרֵךְ כְּשֶׁנִּזְכָּר, כָּל עוֹד לֹא נִגְמַר הַחַג.
בֵּין כּוֹס רִאשׁוֹנָה לַשְּׁנִיָּה טוֹב שֶׁלֹּא לִשְׁתּוֹת עוֹד יַיִן, אֲבָל מִיץ מֻתָּר. בְּכוֹר שֶׁצָּם, אוֹ מִי שֶׁרָעֵב מְאוֹד, יָכוֹל לֶאֱכֹל אַחֲרֵי הַקִּדּוּשׁ מְעַט תַּפּוּחֵי אֲדָמָה, בֵּיצָה אוֹ פֵּרוֹת. מִשֶּׁהִתְחִילוּ אֶת הַהַגָּדָה – לֹא אוֹכְלִים וְלֹא שׁוֹתִים.
💛 אַהֲבַת הַשֵּׁם: כָּל הַהֲכָנוֹת נוֹעֲדוּ לָתֵת לַיְּלָדִים אֶת הַזְּמַן הַיָּקָר, כְּשֶׁהֵם עוֹד עֵרִים. הַסֵּדֶר הַטּוֹב בְּיוֹתֵר הוּא סֵדֶר שֶׁכָּל הַמִּשְׁפָּחָה שׁוּתָפָה בּוֹ.
📖 פֶּרֶק ט״ז, הֲלָכוֹת ב׳, ה׳, י״ג

🍽️הַקְּעָרָה

בַּקְּעָרָה מַנִּיחִים: שָׁלֹשׁ מַצּוֹת, מָרוֹר (חַסָּה אוֹ חֲזֶרֶת), זְרוֹעַ זֵכֶר לְקָרְבַּן פֶּסַח, בֵּיצָה זֵכֶר לְקָרְבַּן חֲגִיגָה, כַּרְפַּס עִם מֵי מֶלַח אוֹ חֹמֶץ, וַחֲרֹסֶת – זֵכֶר לַטִּיט שֶׁעָשׂוּ אֲבוֹתֵינוּ בְּמִצְרַיִם. הַיַּיִן אֵינוֹ בַּקְּעָרָה, כִּי הוּא מַשְׁקֶה.
הַזְּרוֹעַ רוֹמֶזֶת שֶׁנִּגְאַלְנוּ ״בִּזְרוֹעַ נְטוּיָה״, וְצוֹלִים אוֹתָהּ כְּמוֹ שֶׁצָּלוּ אֶת הַפֶּסַח. לְמִנְהַג הַסְּפָרַדִּים לוֹקְחִים רֶגֶל קִדְמִית שֶׁל בְּהֵמָה, וּלְמִנְהַג הָאַשְׁכְּנַזִּים – כָּנָף שֶׁל עוֹף.
💛 אַהֲבַת הַשֵּׁם: כָּל מַאֲכָל בַּקְּעָרָה מְסַפֵּר חֵלֶק מֵהַסִּפּוּר. הַבֵּיצָה מְנַחֶמֶת אוֹתָנוּ שֶׁבְּקָרוֹב יִבָּנֶה בֵּית הַמִּקְדָּשׁ, וְנַקְרִיב שׁוּב אֶת הַפֶּסַח וְהַחֲגִיגָה.
📖 פֶּרֶק ט״ז, הֲלָכָה ג׳

🙌מַגְבִּיהִים וּמְכַסִּים

כָּל זְמַן הַהַגָּדָה הַמַּצּוֹת מְגֻלּוֹת, כִּי הִיא ״לֶחֶם עֹנִי״ – לֶחֶם שֶׁעוֹנִים עָלָיו דְּבָרִים הַרְבֵּה. אֲבָל כְּשֶׁמַּגְבִּיהִים אֶת הַכּוֹס – בְּ״וְהִיא שֶׁעָמְדָה״ וּבִ״לְפִיכָךְ״ – מְכַסִּים אֶת הַמַּצּוֹת, שֶׁלֹּא נִרְאֶה כְּמִי שֶׁמְּכַבֵּד אֶת הַיַּיִן יוֹתֵר מֵהַמַּצָּה.
בְּ״מַצָּה זוֹ״ מַגְבִּיהִים אֶת הַמַּצָּה, וּבְ״מָרוֹר זֶה״ – אֶת הַמָּרוֹר. בְּ״פֶסַח זֶה״ לֹא מַגְבִּיהִים אֶת הַזְּרוֹעַ, כִּי הִיא רַק זֵכֶר, וְשֶׁלֹּא נִרְאֶה כְּמַקְרִיבִים קָרְבָּן בַּחוּץ. וְאֶת הַהַגָּדָה אוֹמְרִים בְּלִי הֲסִבָּה.
💛 אַהֲבַת הַשֵּׁם: גַּם לְמַאֲכָלִים יֵשׁ כָּבוֹד. כְּשֶׁאָנוּ שׁוֹמְרִים עַל כְּבוֹד הַמַּצָּה, אָנוּ מְלַמְּדִים אֶת עַצְמֵנוּ לְכַבֵּד כָּל מִצְוָה.
📖 פֶּרֶק ט״ז, הֲלָכוֹת ג׳, י״ט

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

הַכַּרְפַּס: לְמִנְהַג הַסְּפָרַדִּים לוֹקְחִים סֶלֶרִי. יֵשׁ אַשְׁכְּנַזִּים שֶׁלּוֹקְחִים פֶּטְרוֹזִילְיָה, וְרֹב הָאַשְׁכְּנַזִּים לוֹקְחִים תַּפּוּחֵי אֲדָמָה מְבֻשָּׁלִים. הַזְּרוֹעַ: לְמִנְהַג הַסְּפָרַדִּים – רֶגֶל קִדְמִית שֶׁל בְּהֵמָה, וּלְמִנְהַג הָאַשְׁכְּנַזִּים – כָּנָף שֶׁל עוֹף. וְיֵשׁ גַּם מַחֲלֹקוֹת שֶׁאֵינָן לְפִי עֵדוֹת: שִׁעוּר הָרְבִיעִית, וְהַדֶּרֶךְ לְהַטִּיף יַיִן בְּעֶשֶׂר הַמַּכּוֹת. וּבְמִיץ עֲנָבִים לְאַרְבַּע כּוֹסוֹת: רַבִּים מֵהַפּוֹסְקִים מַתִּירִים, וְהָרַב מֶלָמֵד מְהַדֵּר בְּיַיִן מְשַׂמֵּחַ.
אֵין חוֹבָה לְהָנִיחַ קְעָרָה לִפְנֵי כָּל אֶחָד – מַסְפִּיק לִפְנֵי עוֹרֵךְ הַסֵּדֶר, וְיֵשׁ נוֹהֲגִים לְהָנִיחַ קְעָרָה לִפְנֵי כָּל בַּעַל מִשְׁפָּחָה. גַּם בְּסִדּוּר הַמַּאֲכָלִים בַּקְּעָרָה יֵשׁ כַּמָּה מִנְהָגִים, וּבַהַגָּדוֹת רַבּוֹת מֻדְפָּס צִיּוּר שֶׁל הַקְּעָרָה.
💛 אַהֲבַת הַשֵּׁם: כָּל מִשְׁפָּחָה מַמְשִׁיכָה בְּמִנְהַג אֲבוֹתֶיהָ. הַקְּעָרוֹת נִרְאוֹת קְצָת שׁוֹנוֹת – אֲבָל בְּכֻלָּן אוֹתוֹ סִפּוּר, וְאוֹתָהּ אַהֲבָה לַה׳ שֶׁגְּאָלָנוּ.
📖 פֶּרֶק ט״ז, הֲלָכוֹת ג׳–ד׳, ט״ו

🍽️🔑 מוּשָּׂג: הַקְּעָרָה

״כָּל מַאֲכָל נוֹעָד לְהַזְכִּיר וּלְהַדְגִּישׁ רַעְיוֹן מְסֻיָּם, וַאֲנַחְנוּ צְרִיכִים שֶׁכֹּל הַמַּאֲכָלִים יִהְיוּ לְפָנֵינוּ כְּדֵי לְבַטֵּא אֶת עִנְיָנוֹ הַמְּיֻחָד שֶׁל לֵיל הַסֵּדֶר״.
📖 הַקְּעָרָה – הַכְּלִי שֶׁמַּנִּיחִים בּוֹ בְּלֵיל הַסֵּדֶר אֶת הַמַּצּוֹת, הַמָּרוֹר, הַחֲרֹסֶת, הַכַּרְפַּס, הַזְּרוֹעַ וְהַבֵּיצָה.
💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לָנוּ סֵדֶר שָׁלֵם, שֶׁבּוֹ כֹּל מַאֲכָל קָטָן מְלַמֵּד רַעְיוֹן גָּדוֹל עַל הַגְּאֻלָּה. 🧒 וְאַתָּה? עֲזֹר לְסַדֵּר אֶת הַקְּעָרָה, וּשְׁאַל מָה פֵּרוּשׁ כֹּל מַאֲכָל בָּהּ.
📖 פֶּרֶק ט״ז, הֲלָכָה ג׳

🍗🔑 מוּשָּׂג: הַזְּרוֹעַ

״תִּקְּנוּ חֲכָמִים לְהַנִּיחַ עַל הַשֻּׁלְחָן תַּבְשִׁיל זֵכֶר לְקָרְבַּן פֶּסַח... וְנוֹהֲגִים שֶׁהַתַּבְשִׁיל יִהְיֶה זְרוֹעַ, לִרְמֹז לְכָךְ שֶׁנִּגְאַלְנוּ בִּ׳זְרוֹעַ נְטוּיָה׳״.
📖 הַזְּרוֹעַ – תַּבְשִׁיל צָלוּי בַּקְּעָרָה, זֵכֶר לְקָרְבַּן הַפֶּסַח שֶׁהָיָה נֶאֱכָל בְּעֵת שֶׁבֵּית הַמִּקְדָּשׁ הָיָה קַיָּם.
💛 אַהֲבַת הַשֵּׁם: ה׳ גָּאַל אוֹתָנוּ בְּיָד חֲזָקָה וּבִזְרוֹעַ נְטוּיָה – לֹא בְּסֵתֶר, אֶלָּא בְּגִלּוּי גָּדוֹל לְעֵינֵי הָעוֹלָם. 🧒 וְאַתָּה? כְּשֶׁאַתָּה רוֹאֶה אֶת הַזְּרוֹעַ בַּקְּעָרָה, תִּזְכֹּר שֶׁה׳ פּוֹעֵל בְּגָלוּי בְּחַיֵּינוּ.
📖 פֶּרֶק ט״ז, הֲלָכָה ג׳

🌰🔑 מוּשָּׂג: חֲרֹסֶת

״מַנִּיחִים חֲרֹסֶת, שֶׁהִיא רוֹמֶזֶת לַטִּיט שֶׁהָיוּ מְכִינִים אֲבוֹתֵינוּ כְּשֶׁהָיוּ מְשֻׁעְבָּדִים בְּמִצְרַיִם, וּבָהּ טוֹבְלִים אֶת הַמָּרוֹר לִפְנֵי אֲכִילָתוֹ״.
📖 חֲרֹסֶת – תַּבְשִׁיל פֵּרוֹת וֶאֱגוֹזִים מְעוּכִים, זֵכֶר לַטִּיט שֶׁל עֲבוֹדַת הַפֶּרֶךְ בְּמִצְרַיִם.
💛 אַהֲבַת הַשֵּׁם: גַּם בַּתּוֹךְ הַזֵּכֶר לַקֹּשִׁי, ה׳ נוֹתֵן לָנוּ מַאֲכָל מָתוֹק – סִימָן שֶׁהַקֹּשִׁי הֶעָבָר לְטוֹבָה. 🧒 וְאַתָּה? כְּשֶׁאַתָּה טוֹבֵל מָרוֹר בַּחֲרֹסֶת, תִּזְכֹּר שֶׁגַּם קֹשִׁי יָכוֹל לְהֵפָּתֵר בְּמָתוֹק.

🧠 בְּדֹק אֶת עַצְמְךָ

בְּאֶמְצַע הַהַגָּדָה רוֹאֶה אַבָּא שֶׁהַקְּטַנִּים מַתְחִילִים לְהֵרָדֵם. מָה יַעֲשֶׂה?
💡 כְּדֵי לָצֵאת יְדֵי מִצְוַת הַסִּפּוּר צָרִיךְ לְפָחוֹת לוֹמַר אוֹ לִשְׁמֹעַ פֶּסַח, מַצָּה וּמָרוֹר וּמַשְׁמָעוּתָם. לָכֵן הוֹרִים שֶׁרוֹאִים שֶׁהַיְּלָדִים נִרְדָּמִים – אוֹמְרִים לָהֶם קֹדֶם אֶת שְׁלֹשֶׁת הַדְּבָרִים הָאֵלֶּה.
בְּ״וְהִיא שֶׁעָמְדָה״ מַגְבִּיהִים אֶת הַכּוֹס. לָמָּה מְכַסִּים אָז אֶת הַמַּצּוֹת?
💡 הַמַּצָּה חֲשׁוּבָה מֵהַיַּיִן. כְּשֶׁמַּקְדִּימִים אֶת הַכּוֹס וּמַגְבִּיהִים אוֹתָהּ, מְכַסִּים אֶת הַמַּצָּה – כְּמוֹ שֶׁמְּכַסִּים אֶת הַחַלּוֹת בְּקִדּוּשׁ שֶׁל שַׁבָּת.
בְּ״פֶסַח זֶה״ אֵין מַגְבִּיהִים אֶת הַזְּרוֹעַ. מַדּוּעַ?
💡 הַזְּרוֹעַ אֵינָהּ בְּשַׂר קָרְבַּן פֶּסַח אֶלָּא זֵכֶר בִּלְבַד, וְאִם יַגְבִּיהוּ אוֹתָהּ – זֶה יֵרָאֶה כְּמַקְרִיבִים קָדָשִׁים בַּחוּץ. אֶת הַמַּצָּה וְאֶת הַמָּרוֹר מַגְבִּיהִים.

הַתְּפִלּוֹת שֶׁאַחֲרֵי הָעֲמִידָה

🌸 תְּפִלַּת בָּנוֹת · פֶּרֶק י״ז

🎁מוֹסִיפוֹת מֵאַהֲבָה

נָשִׁים פְּטוּרוֹת מִכָּל מָה שֶׁהַגְּבָרִים אוֹמְרִים אַחֲרֵי הָעֲמִידָה: תַּחֲנוּן, אַשְׁרֵי, לַמְנַצֵּחַ, קְדֻשָּׁה דְּסִדְרָא, שִׁיר שֶׁל יוֹם, פִּטּוּם הַקְּטֹרֶת וְ״עָלֵינוּ לְשַׁבֵּחַ״. גַּם לַגְּבָרִים זֶה מִנְהָג שֶׁנַּעֲשָׂה כְּחוֹבָה, וְהַנָּשִׁים לֹא נָהֲגוּ כָּךְ. הָרוֹצָה לְהַדֵּר – רַשָּׁאִית לוֹמַר אוֹתָם.
בַּת שֶׁמִּתְנַדֶּבֶת לְקַיֵּם מִצְוָה שֶׁהִיא פְּטוּרָה מִמֶּנָּה – יֵשׁ לָהּ שָׂכָר, אֲבָל ״גָּדוֹל מְצֻוֶּה וְעוֹשֶׂה מִמִּי שֶׁאֵינוֹ מְצֻוֶּה וְעוֹשֶׂה״, כִּי עַל הַמְצֻוֶּה הַיֵּצֶר הָרָע מִתְאַמֵּץ יוֹתֵר לְהַפְרִיעַ.
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁפְּטוּרָה וּבְכָל זֹאת בּוֹחֶרֶת לְהוֹסִיף שִׁיר שֶׁל שֶׁבַח, מַרְאָה לַה׳ שֶׁהִיא רוֹצָה לִהְיוֹת קְרוֹבָה אֵלָיו.
📖 פֶּרֶק י״ז, הֲלָכָה א׳; פֶּרֶק ב׳, הֲלָכָה ח׳

🙏תַּחֲנוּנִים וּשְׁלֹשׁ עֶשְׂרֵה מִדּוֹת

מִי שֶׁרוֹצָה לוֹמַר תַּחֲנוּנִים – רָאוּי שֶׁלֹּא תַּפְסִיק בְּשִׂיחָה בֵּין הָעֲמִידָה לַתַּחֲנוּנִים, כִּי כְּשֶׁהֵם צְמוּדִים לָעֲמִידָה הֵם מִתְקַבְּלִים יוֹתֵר. בְּשֵׁנִי וַחֲמִישִׁי מוֹסִיפִים תַּחֲנוּנִים, כִּי הֵם יְמֵי רָצוֹן.
אֲמִירַת י״ג מִדּוֹת רַחֲמִים הִיא דָּבָר שֶׁבִּקְדֻשָּׁה, וְצָרִיךְ מִנְיָן כְּדֵי לְאָמְרָן. לָכֵן בַּת שֶׁמִּתְפַּלֶּלֶת בְּלִי מִנְיָן וְאוֹמֶרֶת תַּחֲנוּנִים – מְדַלֶּגֶת עַל י״ג מִדּוֹת.
💛 אַהֲבַת הַשֵּׁם: בְּי״ג מִדּוֹת רַחֲמִים יֵשׁ סְגֻלָּה גְּדוֹלָה לִמְחִילַת עָווֹן: כְּשֶׁאֲנַחְנוּ דְּבוּקִים בַּה׳, הַחֲטָאִים נַעֲשִׂים קְטַנִּים לְיַד הָאַהֲבָה הַגְּדוֹלָה.
📖 פֶּרֶק י״ז, הֲלָכָה ה׳

🔀וִדּוּי וּשְׁלֹשׁ עֶשְׂרֵה מִדּוֹת – שְׁנֵי מִנְהָגִים

לְמִנְהַג סְפָרַד אוֹמְרִים לִפְנֵי נְפִילַת אַפַּיִם וִדּוּי וְי״ג מִדּוֹת רַחֲמִים בְּכָל יוֹם. לְמִנְהַג אַשְׁכְּנַז וְתֵימָן (בַּלַּדִי) מוֹסִיפִים אוֹתָם רַק בְּשֵׁנִי וּבַחֲמִישִׁי.
וּבִנְפִילַת אַפַּיִם: בְּנֹסַח סְפָרַדִּי אוֹמְרִים אֶת מִזְמוֹר כ״ה בִּתְהִלִּים, וּבְנֻסְחֵי אַשְׁכְּנַז וּסְפָרַד־חֲסִידִי – אֶת מִזְמוֹר ו׳.
💛 אַהֲבַת הַשֵּׁם: שְׁנֵי הַמִּנְהָגִים כַּהֲלָכָה, וְכָל אַחַת נוֹהֶגֶת כְּמִנְהַג אֲבוֹתֶיהָ. וּבְכֻלָּם מְבַקְּשִׁים: ״ה׳, ה׳, אֵל רַחוּם וְחַנּוּן״.
📖 פֶּרֶק י״ז, הֲלָכָה ה׳

🧠 בְּדֹק אֶת עַצְמְךָ

מִיכַל סִיְּמָה אֶת תְּפִלַּת הָעֲמִידָה. מָה הַדִּין לְגַבֵּי תַּחֲנוּן, שִׁיר שֶׁל יוֹם וְ״עָלֵינוּ״?
💡 נָשִׁים פְּטוּרוֹת מִכָּל מָה שֶׁאוֹמְרִים אַחֲרֵי הָעֲמִידָה, כִּי גַּם לַגְּבָרִים זֶה מִנְהָג שֶׁנַּעֲשָׂה חוֹבָה, וְהַנָּשִׁים לֹא נָהֲגוּ כָּךְ. הָרוֹצָה לְהַדֵּר – רַשָּׁאִית.
אֲבִיגַיִל מִתְפַּלֶּלֶת לְבַד בַּבַּיִת וְרוֹצָה לוֹמַר תַּחֲנוּנִים. עַל מָה הִיא מְדַלֶּגֶת?
💡 י״ג מִדּוֹת רַחֲמִים הֵן דָּבָר שֶׁבִּקְדֻשָּׁה, וְצָרִיךְ מִנְיָן כְּדֵי לְאָמְרָן. לָכֵן מִי שֶׁמִּתְפַּלֶּלֶת בְּלִי מִנְיָן מְדַלֶּגֶת עֲלֵיהֶן, וְאֶת שְׁאָר הַתַּחֲנוּנִים אוֹמֶרֶת.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

תְּחוּמֵי שַׁבָּת

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק ל׳

🚶תְּחוּם שַׁבָּת

חֲכָמִים קָבְעוּ תְּחוּם שֶׁבְּתוֹכוֹ מֻתָּר לְהִתְהַלֵּךְ בְּשַׁבָּת: הָעִיר כֻּלָּהּ, וְעוֹד אַלְפַּיִם אַמָּה – כְּ־912 מֶטֶר – מִחוּצָה לָהּ לְכָל צַד. מִי שֶׁשָּׁבַת בַּשָּׂדֶה, מוֹדְדִים לוֹ אַלְפַּיִם אַמָּה מִן הַמָּקוֹם שֶׁבּוֹ הוּא נִמְצָא.
מִי שֶׁצָּרִיךְ לָלֶכֶת לִדְבַר מִצְוָה מֵעֵבֶר לַתְּחוּם, כְּמוֹ שִׁעוּר תּוֹרָה, יָכוֹל לְהַנִּיחַ "עֵרוּב תְּחוּמִין".
💛 אַהֲבַת הַשֵּׁם: בְּכָל הַשָּׁבוּעַ אֲנַחְנוּ נוֹדְדִים וּמְחַפְּשִׂים. בְּשַׁבָּת נִשְׁאָרִים בַּמָּקוֹם, וּמַרְגִּישִׁים שֶׁכָּל מָה שֶׁצָּרִיךְ – כְּבָר כָּאן, אֵצֶל ה׳.
📖 פֶּרֶק ל׳, הֲלָכָה א׳

🗺️יָצָא מִן הַתְּחוּם

מִי שֶׁיָּצָא בְּשַׁבָּת מִחוּץ לַתְּחוּם – אִבֵּד אֶת תְּחוּמוֹ, וְיֵשׁ לוֹ רַק אַרְבַּע אַמּוֹת. אֲבָל מִי שֶׁיָּצָא מִן הַתְּחוּם כְּדֵי לְהַצִּיל נְפָשׁוֹת – חֲכָמִים תִּקְּנוּ שֶׁאַחֲרֵי הַהַצָּלָה יֵשׁ לוֹ אַלְפַּיִם אַמָּה לְכָל רוּחַ.
בְּעֵרוּב תְּחוּמִין מַנִּיחִים לִפְנֵי שַׁבָּת מָזוֹן שֶׁל שְׁתֵּי סְעוּדוֹת. מַה שֶּׁמַּרְוִיחִים לְצַד אֶחָד – מַפְסִידִים לַצַּד הַשֵּׁנִי.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים דָּאֲגוּ לַמַּצִּילִים, כְּדֵי שֶׁלֹּא יְהַסְּסוּ לָצֵאת לְהַצִּיל. מִי שֶׁעוֹזֵר לַאֲחֵרִים – ה׳ דּוֹאֵג לוֹ.
📖 פֶּרֶק ל׳, הֲלָכוֹת י״א–י״ב

🧠 בְּדֹק אֶת עַצְמְךָ

מִתְנַדֵּב יָצָא בְּשַׁבָּת מִחוּץ לַתְּחוּם כְּדֵי לְהַצִּיל אָדָם. מָה דִּינוֹ אַחֲרֵי הַהַצָּלָה?
💡 מִי שֶׁיָּצָא מִן הַתְּחוּם סְתָם – יֵשׁ לוֹ רַק אַרְבַּע אַמּוֹת. אֲבָל לַמַּצִּיל תִּקְּנוּ חֲכָמִים אַלְפַּיִם אַמָּה, כְּדֵי שֶׁאֲנָשִׁים לֹא יְהַסְּסוּ לָצֵאת לְהַצִּיל.

ארבעת המינים

🌿 הִלְכוֹת סֻכּוֹת · פֶּרֶק ד׳

🌿אַרְבָּעָה מִינִים

״פְּרִי עֵץ הָדָר״ – אֶתְרוֹג; ״כַּפֹּת תְּמָרִים״ – לוּלָב; ״עֲנַף עֵץ עָבֹת״ – הֲדַס; ״עַרְבֵי נָחַל״ – עֲרָבָה. נוֹטְלִים אֶתְרוֹג אֶחָד, לוּלָב אֶחָד, שְׁלֹשָׁה הֲדַסִּים וּשְׁתֵּי עֲרָבוֹת. הַלּוּלָב הוּא הַגָּבוֹהַּ שֶׁבַּמִּינִים, וְלָכֵן מְבָרְכִים ״עַל נְטִילַת לוּלָב״.
מִן הַתּוֹרָה נוֹטְלִים לוּלָב רַק בַּיּוֹם הָרִאשׁוֹן, וּבַמִּקְדָּשׁ – שִׁבְעָה יָמִים. אַחֲרֵי הַחֻרְבָּן תִּקֵּן רַבָּן יוֹחָנָן בֶּן זַכַּאי לִטֹּל בְּכָל מָקוֹם שִׁבְעָה יָמִים, זֵכֶר לַמִּקְדָּשׁ.
💛 אַהֲבַת הַשֵּׁם: אַחֲרֵי שֶׁאָסַפְנוּ אֶת הַיְּבוּל, נוֹטְלִים אַרְבָּעָה מִינִים כְּאוֹת תּוֹדָה לַה׳ – וּכְמוֹ דֶּגֶל שֶׁל נִצָּחוֹן, אַחֲרֵי הַתְּשׁוּבָה שֶׁל הַיָּמִים הַנּוֹרָאִים.
📖 פֶּרֶק ד׳, הֲלָכָה א׳; פֶּרֶק א׳, הֲלָכָה ג׳

🤝כֻּלָּם יַחַד

אַרְבַּעַת הַמִּינִים מְעַכְּבִים זֶה אֶת זֶה: אִם חָסֵר מִין אֶחָד – אֵין מְקַיְּמִים אֶת הַמִּצְוָה. לְכַתְּחִלָּה אוֹגְדִים אֶת הַלּוּלָב עִם הַהֲדַסִּים וְהָעֲרָבוֹת, וְנוֹטְלִים אֶת כֻּלָּם יַחַד.
הָאֶתְרוֹג – יֵשׁ בּוֹ טַעַם וְרֵיחַ, כְּמוֹ יְהוּדִי שֶׁיֵּשׁ בּוֹ תּוֹרָה וּמַעֲשִׂים טוֹבִים. הַלּוּלָב – טַעַם בְּלִי רֵיחַ; הַהֲדַס – רֵיחַ בְּלִי טַעַם; וְהָעֲרָבָה – לֹא טַעַם וְלֹא רֵיחַ.
💛 אַהֲבַת הַשֵּׁם: אָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא: ״יִקָּשְׁרוּ כֻּלָּם אֲגֻדָּה אַחַת״. כְּשֶׁכָּל סוּגֵי הַיְּהוּדִים מִתְאַחֲדִים יַחַד – ה׳ שָׂמֵחַ בָּנוּ.
📖 פֶּרֶק ד׳, הֲלָכָה ב׳

🍃הֲדַס

״עֲנַף עֵץ עָבֹת״ – הֲדַס שֶׁעָלָיו גְּדֵלִים בִּשְׁלָשׁוֹת, שְׁלֹשָׁה עָלִים בְּגֹבַהּ אֶחָד, כְּמוֹ קְלִיעָה. אִם שְׁנַיִם יוֹצְאִים יַחַד וְהַשְּׁלִישִׁי לְמַעְלָה אוֹ לְמַטָּה – זֶה ״הֲדַס שׁוֹטֶה״, וְהוּא פָּסוּל.
אֹרֶךְ כָּל הֲדַס – לְפָחוֹת שְׁלֹשָׁה טְפָחִים (כ־24 ס״מ). הֲדַס שֶׁעָלָיו הִתְיַבְּשׁוּ לְגַמְרֵי, עַד שֶׁהֵם נִפְרָכִים בַּצִּפֹּרֶן וּלְבָנִים – פָּסוּל.
💛 אַהֲבַת הַשֵּׁם: הַהֲדַס רוֹמֵז לְמַעֲשִׂים טוֹבִים, שֶׁכְּמוֹ רֵיחַ טוֹב מִתְפַּשְּׁטִים סָבִיב. הַצַּדִּיקִים נִקְרָאִים הֲדַסִּים.
📖 פֶּרֶק ד׳, הֲלָכוֹת ג׳, ח׳

🌱עֲרָבָה

לָעֲרָבָה שְׁלֹשָׁה סִימָנִים: הֶעָלֶה מָשׁוּךְ לְאָרְכּוֹ כְּמוֹ נַחַל, שְׂפַת הֶעָלֶה חֲלָקָה, וְהֶעָנָף אָדֹם קְצָת. יֵשׁ צֶמַח דּוֹמֶה – צַפְצָפָה – שֶׁבִּשְׂפַת עָלֶיהָ יֵשׁ שִׁנַּיִם כְּמוֹ מַסּוֹר, וְהִיא פְּסוּלָה.
הָעֲרָבָה מִתְיַבֶּשֶׁת מַהֵר. עֲרָבָה שֶׁרֹב עָלֶיהָ נָשְׁרוּ, אוֹ שֶׁהִתְיַבְּשׁוּ וְהִלְבִּינוּ – פְּסוּלָה. לָכֵן שׁוֹמְרִים אוֹתָהּ בְּמַגֶּבֶת לַחָה אוֹ בְּנַרְתִּיק אָטוּם, וְיֵשׁ מְהַדְּרִים לְהַחְלִיף אוֹתָהּ בְּמֶשֶׁךְ הַחַג.
💛 אַהֲבַת הַשֵּׁם: לָעֲרָבָה אֵין טַעַם וְאֵין רֵיחַ, אֲבָל יֵשׁ בָּהּ כֹּחַ צְמִיחָה עָצוּם. גַּם מִן הַמִּשְׁפָּחוֹת הַפְּשׁוּטוֹת צָמְחוּ גְּדוֹלֵי יִשְׂרָאֵל.
📖 פֶּרֶק ד׳, הֲלָכוֹת ג׳, ט׳; פֶּרֶק ה׳, הֲלָכָה ז׳

🍋אֶתְרוֹג

״פְּרִי עֵץ הָדָר״ הוּא אֶתְרוֹג – לֹא לִימוֹן, וְלֹא אֶתְרוֹג שֶׁגָּדַל עַל עֵץ מֻרְכָּב. אֶתְרוֹג קָטָן מִנֶּפַח בֵּיצָה – פָּסוּל. וְגַם אֶתְרוֹג שָׁחֹר, אוֹ אֶתְרוֹג שֶׁצּוּרָתוֹ מְשֻׁנָּה לְגַמְרֵי, כְּמוֹ כַּדּוּר עָגֹל – פָּסוּל.
כְּתָמִים רְגִילִים, צְהֻבִּים אוֹ חוּמִים, אֵינָם פּוֹסְלִים – כָּךְ טִבְעָם שֶׁל אֶתְרוֹגִים. וְגַם אֶתְרוֹג בְּלִי פִּטָּם כָּשֵׁר לְכַתְּחִלָּה.
💛 אַהֲבַת הַשֵּׁם: הָאֶתְרוֹג – יֵשׁ בּוֹ טַעַם וְרֵיחַ, וְהוּא רוֹמֵז לִשְׁלֵמוּת. לָכֵן מְהַדְּרִים בְּיָפְיוֹ יוֹתֵר מִשְּׁאָר הַמִּינִים.
📖 פֶּרֶק ד׳, הֲלָכוֹת ג׳, י׳–י״ב

🌴לֵב אֶחָד

הַלּוּלָב רוֹמֵז לַתּוֹרָה וּלְתַלְמִידֵי הַחֲכָמִים. עָלָיו צוֹמְחִים לִשְׁנֵי צְדָדִים, אֲבָל כֻּלָּם צְמוּדִים לְשִׁדְרָה אַחַת, וְהוּא יָשָׁר וּמְכֻוָּן לְמַטָּרָה אַחַת. כָּךְ בַּתּוֹרָה יֵשׁ דֵּעוֹת שׁוֹנוֹת – וְהַשֹּׁרֶשׁ שֶׁל כֻּלָּן אֶחָד.
לָכֵן מְבָרְכִים ״עַל נְטִילַת לוּלָב״ – כִּי הַלּוּלָב הוּא הַגָּבוֹהַּ שֶׁבַּמִּינִים, כְּמוֹ הַתּוֹרָה, שֶׁאֵין לְמַעְלָה מִמֶּנָּה.
💛 אַהֲבַת הַשֵּׁם: ״מָה תָּמָר זֶה אֵין לוֹ אֶלָּא לֵב אֶחָד – אַף יִשְׂרָאֵל אֵין לָהֶם אֶלָּא לֵב אֶחָד לַאֲבִיהֶם שֶׁבַּשָּׁמַיִם״.
📖 פֶּרֶק ד׳, הֲלָכָה ג׳

💎הִדּוּר מִצְוָה

״זֶה אֵלִי וְאַנְוֵהוּ״ – מִצְוָה לְהַדֵּר וּלְהוֹסִיף עַד שְׁלִישׁ מִמְּחִיר הַמִּצְוָה. אִם יֵשׁ סֶט בְּשִׁשִּׁים שְׁקָלִים, סֶט יָפֶה יוֹתֵר בִּשְׁמוֹנִים וְסֶט יָפֶה עוֹד יוֹתֵר בְּמֵאָה – מִצְוַת הַהִדּוּר הִיא לִקְנוֹת אֶת הַסֶּט שֶׁל שְׁמוֹנִים. וּמִי שֶׁמּוֹסִיף יוֹתֵר – ה׳ יְשַׁלֵּם שְׂכָרוֹ.
מִי שֶׁמַּצָּבוֹ דָּחוּק – אֵין עָלָיו מִצְוָה לְהוֹסִיף שְׁלִישׁ. וְאֵין צֹרֶךְ לִהְיוֹת בְּמֶתַח: רֹב הַמִּינִים בַּשּׁוּק כְּשֵׁרִים, וְרַק שִׁנּוּי מַשְׁמָעוּתִי פּוֹסֵל.
💛 אַהֲבַת הַשֵּׁם: ה׳ קִבֵּל אֶת קָרְבָּנוֹ שֶׁל הֶבֶל, שֶׁהֵבִיא מִן הַמֻּבְחָר. כְּשֶׁאָנוּ בּוֹחֲרִים מִין יָפֶה, אָנוּ מַרְאִים לַה׳ שֶׁהַמִּצְוָה שֶׁלּוֹ חֲשׁוּבָה לָנוּ.
📖 פֶּרֶק ד׳, הֲלָכוֹת ד׳, י״ד

🌴לוּלָב כָּשֵׁר

עָלֵי הַלּוּלָב צוֹמְחִים מִשְּׁנֵי צְדָדָיו וּמְכַסִּים אֶת הַשִּׁדְרָה, וּלְכַתְּחִלָּה הֵם צְמוּדִים וּסְגוּרִים. לוּלָב שֶׁהִתְעַקֵּם מְאוֹד כְּמוֹ מַגָּל – פָּסוּל, וְלוּלָב שֶׁרֹב עָלָיו הִתְיַבְּשׁוּ וְהִלְבִּינוּ – פָּסוּל.
אֹרֶךְ הַשִּׁדְרָה – לְפָחוֹת אַרְבָּעָה טְפָחִים (כ־32 ס״מ). וּמֵהִדּוּרֵי הַלּוּלָב – שֶׁיִּהְיֶה יָשָׁר.
💛 אַהֲבַת הַשֵּׁם: הַלּוּלָב הַיָּשָׁר מְלַמֵּד אוֹתָנוּ לָלֶכֶת בְּדֶרֶךְ יְשָׁרָה, בְּלִי לִפְנוֹת לִשְׁנֵי כִּוּוּנִים.
📖 פֶּרֶק ד׳, הֲלָכוֹת ה׳, ז׳

🚫שָׁלֵם וְלֹא גָּזוּל

בַּיּוֹם הָרִאשׁוֹן הַמִּינִים צְרִיכִים לִהְיוֹת שְׁלֵמִים. אֶתְרוֹג שֶׁנִּקַּב וְנֶחְסַר מִמֶּנּוּ חֵלֶק, אוֹ לוּלָב שֶׁנִּפְתְּחָה רֹב הַ״תְּיוֹמֶת״ שֶׁל הֶעָלֶה הָאֶמְצָעִי הָעֶלְיוֹן – פְּסוּלִים בַּיּוֹם הָרִאשׁוֹן, וּכְשֵׁרִים בִּשְׁאָר הַיָּמִים.
מִין גָּזוּל אוֹ גָּנוּב – פָּסוּל כָּל שִׁבְעַת הַיָּמִים, אֲפִלּוּ אִם הוּא יָפֶה וּמְהֻדָּר, כִּי זוֹ ״מִצְוָה הַבָּאָה בַּעֲבֵרָה״. וְלוּלָב שֶׁל אָדָם שֶׁאֵינוֹ נִמְצָא – מֻתָּר לִטֹּל מִן הַיּוֹם הַשֵּׁנִי, בְּלִי לְהוֹצִיא אוֹתוֹ מִמְּקוֹמוֹ, אִם אֵין יָדוּעַ שֶׁהוּא מַקְפִּיד.
💛 אַהֲבַת הַשֵּׁם: ה׳ אֵינוֹ רוֹצֶה מִצְוָה שֶׁנַּעֲשֵׂית עַל חֶשְׁבּוֹן אַחֵר. מִצְוָה יָפָה בֶּאֱמֶת הִיא מִצְוָה יְשָׁרָה.
📖 פֶּרֶק ד׳, הֲלָכוֹת ו׳, י״א, י״ג

📅הַיּוֹם הָרִאשׁוֹן – ״לָכֶם״

״וּלְקַחְתֶּם לָכֶם בַּיּוֹם הָרִאשׁוֹן״ – בַּיּוֹם הָרִאשׁוֹן הַלּוּלָב צָרִיךְ לִהְיוֹת שֶׁל מִי שֶׁמְּקַיֵּם בּוֹ אֶת הַמִּצְוָה, וְלֹא שָׁאוּל. לָכֵן בַּעַל הַלּוּלָב נוֹתֵן אוֹתוֹ לַחֲבֵרוֹ ״בְּמַתָּנָה עַל מְנַת לְהַחֲזִיר״. בִּשְׁאָר הַיָּמִים אֶפְשָׁר לְקַיֵּם אֶת הַמִּצְוָה גַּם בְּלוּלָב שָׁאוּל.
קָטָן יָכוֹל לְקַבֵּל מַתָּנָה, אֲבָל אֵינוֹ יָכוֹל לָתֵת מַתָּנָה. לָכֵן בַּיּוֹם הָרִאשׁוֹן נוֹתְנִים לוֹ אֶת הַלּוּלָב רַק אַחֲרֵי שֶׁכָּל הַגְּדוֹלִים קִיְּמוּ בּוֹ אֶת הַמִּצְוָה.
💛 אַהֲבַת הַשֵּׁם: הַמִּצְוָה צְרִיכָה לִהְיוֹת ״שֶׁלָּנוּ״ בֶּאֱמֶת. וְכַמָּה יָפֶה שֶׁיְּהוּדִים נוֹתְנִים זֶה לָזֶה מַתָּנוֹת, כְּדֵי שֶׁכֻּלָּם יְקַיְּמוּ מִצְוָה.
📖 פֶּרֶק ד׳, הֲלָכָה י״ג

🧠 בְּדֹק אֶת עַצְמְךָ

לְאֵלִי אֵין לוּלָב, וְאַבָּא שֶׁל חֲבֵרוֹ רוֹצֶה לָתֵת לוֹ לְבָרֵךְ בַּיּוֹם הָרִאשׁוֹן. מָה נָכוֹן?
💡 בַּיּוֹם הָרִאשׁוֹן ״לָכֶם״ – מִשֶּׁלָּכֶם, וּבְלוּלָב שָׁאוּל אֵין יוֹצְאִים. בְּמַתָּנָה עַל מְנַת לְהַחֲזִיר – הַלּוּלָב שֶׁל אֵלִי, וְהוּא מַחְזִיר אוֹתוֹ אַחַר כָּךְ בְּמַתָּנָה.
בַּיּוֹם הָרִאשׁוֹן, מָתַי נוֹתְנִים אֶת הַלּוּלָב לְיֶלֶד קָטָן?
💡 קָטָן יָכוֹל לְקַבֵּל מַתָּנָה, אֲבָל אֵינוֹ יָכוֹל לְהַחֲזִיר אוֹתָהּ בְּמַתָּנָה. אִם יִתְּנוּ לוֹ אֶת הַלּוּלָב לִפְנֵי הַגְּדוֹלִים – הַלּוּלָב יִהְיֶה שֶׁלּוֹ, וְהַגְּדוֹלִים לֹא יוּכְלוּ לָצֵאת בּוֹ.
בַּיּוֹם הַשְּׁלִישִׁי שֶׁל הֶחָג רָאָה נֹעַם לוּלָב בְּבֵית הַכְּנֶסֶת, וּבַעֲלָיו אֵינוֹ שָׁם. מָה הַדִּין?
💡 מִן הַיּוֹם הַשֵּׁנִי אֶפְשָׁר לָצֵאת בְּלוּלָב שָׁאוּל, וּמִן הַסְּתָם נוֹחַ לְאָדָם שֶׁעוֹשִׂים מִצְוָה בַּחֲפָצָיו. אֲבָל לֹא מוֹצִיאִים אֶת הַלּוּלָב מִמְּקוֹמוֹ, וְשׁוֹמְרִים עָלָיו מְאוֹד. וְאִם יָדוּעַ שֶׁבַּעֲלָיו מַקְפִּיד – אָסוּר.

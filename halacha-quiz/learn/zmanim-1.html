רֹאשׁ חֹדֶשׁ

📅 רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת · פֶּרֶק א׳

🌙הַלְּבָנָה מְסַמֶּנֶת אֶת הַחֹדֶשׁ

הַשָּׁנָה נִקְבַּעַת לְפִי הַשֶּׁמֶשׁ, וְהַחֹדֶשׁ – לְפִי הַלְּבָנָה. בִּתְחִלַּת הַחֹדֶשׁ הַלְּבָנָה נִרְאֵית כְּקַו דַּק, וְהִיא הוֹלֶכֶת וְגָדֵלָה עַד שֶׁבְּאֶמְצַע הַחֹדֶשׁ הִיא עִגּוּל שָׁלֵם. אַחַר כָּךְ הִיא הוֹלֶכֶת וּמִתְמַעֶטֶת, עַד שֶׁהִיא נֶעֱלֶמֶת לְיוֹם בְּעֵרֶךְ. כְּשֶׁהִיא חוֹזֶרֶת כְּקַו דַּק – זֶה הַסִּימָן לְחֹדֶשׁ חָדָשׁ.
מַחֲזוֹר הַלְּבָנָה הוּא כְּעֶשְׂרִים וְתִשְׁעָה יָמִים וָחֵצִי. לָכֵן יֵשׁ חֹדֶשׁ שֶׁל 29 יָמִים, "חֹדֶשׁ חָסֵר", וְיֵשׁ חֹדֶשׁ שֶׁל 30 יוֹם, "חֹדֶשׁ מָלֵא".
💛 אַהֲבַת הַשֵּׁם: "יְהִי מְאֹרֹת בִּרְקִיעַ הַשָּׁמַיִם... וְהָיוּ לְאֹתֹת וּלְמוֹעֲדִים". ה׳ תָּלָה בַּשָּׁמַיִם שָׁעוֹן גָּדוֹל, וּבְכָל לַיְלָה שֶׁאָנוּ מַבִּיטִים בַּלְּבָנָה אָנוּ זוֹכְרִים שֶׁהוּא מְסַדֵּר לָנוּ אֶת הַזְּמַנִּים.
📖 פֶּרֶק א׳, הֲלָכָה א׳

⚖️יִשְׂרָאֵל מְקַדְּשִׁים אֶת הַחֹדֶשׁ

הַלְּבָנָה לְבַדָּהּ אֵינָהּ מַתְחִילָה אֶת הַחֹדֶשׁ. בִּימֵי בֵּית הַדִּין הַגָּדוֹל בָּאוּ עֵדִים שֶׁרָאוּ אֶת הַלְּבָנָה הַחֲדָשָׁה, וְעַל פִּי עֵדוּתָם קִדֵּשׁ בֵּית הַדִּין אֶת הַחֹדֶשׁ. זוֹ הַמִּצְוָה הָרִאשׁוֹנָה שֶׁנִּצְטַוּוּ יִשְׂרָאֵל בַּתּוֹרָה: "הַחֹדֶשׁ הַזֶּה לָכֶם".
קְבִיעַת הַחֹדֶשׁ כָּל כָּךְ חֲשׁוּבָה – כִּי כָּל הַחַגִּים תְּלוּיִים בָּהּ – עַד שֶׁהַתּוֹרָה הִתִּירָה לָעֵדִים לְחַלֵּל שַׁבָּת כְּדֵי לָלֶכֶת לִירוּשָׁלַיִם וּלְהָעִיד.
💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לָנוּ, לְעַם יִשְׂרָאֵל, לִהְיוֹת שֻׁתָּפִים בִּקְבִיעַת הַזְּמַן. לָכֵן אוֹמְרִים בַּתְּפִלָּה "מְקַדֵּשׁ יִשְׂרָאֵל וְרָאשֵׁי חֳדָשִׁים" – ה׳ מְקַדֵּשׁ אֶת יִשְׂרָאֵל, וְהֵם מְקַדְּשִׁים אֶת הַחֳדָשִׁים.
📖 פֶּרֶק א׳, הֲלָכוֹת א׳–ב׳

✨"מְקַדֵּשׁ יִשְׂרָאֵל וְרָאשֵׁי חֳדָשִׁים"

בְּסוֹף הַבְּרָכָה הָאֶמְצָעִית שֶׁל מוּסָף חוֹתְמִים "מְקַדֵּשׁ יִשְׂרָאֵל וְרָאשֵׁי חֳדָשִׁים": ה׳ קִדֵּשׁ אֶת יִשְׂרָאֵל, וְיִשְׂרָאֵל מְקַדְּשִׁים אֶת הַחֳדָשִׁים. הַקְּדֻשָּׁה שֶׁבַּזְּמַן מִתְגַּלָּה בְּכֹחָם שֶׁל יִשְׂרָאֵל.
חֹדֶשׁ שֶׁל 29 יָמִים נִקְרָא "חֹדֶשׁ חָסֵר", וְחֹדֶשׁ שֶׁל 30 יוֹם נִקְרָא "חֹדֶשׁ מָלֵא". וְהַנָּשִׁים, שֶׁלֹּא חָטְאוּ בָּעֵגֶל, נוֹהֲגוֹת לְכַבֵּד אֶת רֹאשׁ הַחֹדֶשׁ וּלְהִמָּנַע בּוֹ מִקְּצָת מְלָאכוֹת.
💛 אַהֲבַת הַשֵּׁם: לֹא מְשַׁנֶּה אִם הַחֹדֶשׁ חָסֵר אוֹ מָלֵא – כָּל חֹדֶשׁ הוּא מַתָּנָה מֵה׳, וְאָנוּ מְמַלְּאִים אוֹתוֹ בְּמִצְווֹת וּבְמַעֲשִׂים טוֹבִים.
📖 פֶּרֶק א׳, הֲלָכוֹת א׳–ב׳, ז׳, י״ד

🌑🔑 מוּשָּׂג: מוֹלָד הַלְּבָנָה

״בִּתְחִלַּת הַחֹדֶשׁ הַלְּבָנָה נִרְאֵית לְעֵינֵינוּ קְטַנָּה מְאֹד, כְּקַו דַּק... וְזֶה הַסִּימָן לִתְחִלַּת הַחֹדֶשׁ״.
📖 מוֹלָד – הָרֶגַע שֶׁבּוֹ הַלְּבָנָה מִתְחַדֶּשֶׁת וּמַתְחִיל חֹדֶשׁ חָדָשׁ.
💛 אַהֲבַת הַשֵּׁם: ה׳ קָבַע בַּבְּרִיאָה זְמַנִּים קְבוּעִים, ״לְאֹתוֹת וּלְמוֹעֲדִים״ – כָּל חֹדֶשׁ מַתְחִיל מֵחָדָשׁ, כְּמוֹ הֶזְדַּמְּנוּת נוֹסֶפֶת. 🧒 וְאַתָּה? כְּשֶׁאַתָּה רוֹאֶה לְבָנָה דַּקָּה בַּשָּׁמַיִם – זֶה הַזְּמַן לְהַתְחִיל מַשֶּׁהוּ טוֹב מֵחָדָשׁ.
📖 פֶּרֶק א׳, הֲלָכָה א׳; הֲלָכָה י״ח

🌗🔑 מוּשָּׂג: חֹדֶשׁ חָסֵר וְחֹדֶשׁ מָלֵא

״כַּאֲשֶׁר הַחֹדֶשׁ נִמְשָׁך עֶשְׂרִים וְתִשְׁעָה יָמִים הוּא נִקְרָא 'חֹדֶשׁ חָסֵר', וְכַאֲשֶׁר הוּא נִמְשָׁך שְׁלֹשִׁים יוֹם הוּא נִקְרָא 'חֹדֶשׁ מָלֵא'״.
📖 חֹדֶשׁ חָסֵר וְחֹדֶשׁ מָלֵא – חֹדֶשׁ שֶׁל כ״ט יוֹם נִקְרָא חָסֵר, וְשֶׁל ל׳ יוֹם נִקְרָא מָלֵא.
💛 אַהֲבַת הַשֵּׁם: מַחְזוֹר הַלְּבָנָה שֶׁה׳ בָּרָא אֵינוֹ מִסְתַּדֵּר בְּדִיּוּק בְּמִסְפָּרִים עֲגוֹלִים – וּבְכָל זֹאת סִדֵּר ה׳ אֶת הָעוֹלָם בְּחָכְמָה מֻפְלָאָה. 🧒 וְאַתָּה? בַּפַּעַם הַבָּאָה שֶׁרֹאשׁ חֹדֶשׁ חָל יוֹמַיִם – תֵּדַע שֶׁהַחֹדֶשׁ הַקּוֹדֵם הָיָה מָלֵא.
📖 פֶּרֶק א׳, הֲלָכָה א׳

👳🔑 מוּשָּׂג: קִדּוּשׁ הַחֹדֶשׁ

״צְרִיכִים לָבוֹא בִּפְנֵיכֶם עֵדִים, וּלְהָעִיד כִּי רָאוּ אֶת הַלְּבָנָה בְּחִדּוּשָׁהּ, וְעַל סֶמֶךְ עֵדוּתָם תְּקַדְּשׁוּ אֶת הַחֹדֶשׁ״.
📖 קִדּוּשׁ הַחֹדֶשׁ – בֵּית הַדִּין, עַל פִּי עֵדִים שֶׁרָאוּ אֶת הַלְּבָנָה, הוּא שֶׁקּוֹבֵעַ אֶת תְּחִלַּת הַחֹדֶשׁ.
💛 אַהֲבַת הַשֵּׁם: לֹא הַלְּבָנָה בִּלְבַד קוֹבַעַת – ״יִשְׂרָאֵל הֵם שֶׁמְּקַדְּשִׁים אֶת הַחֳדָשִׁים״. ה׳ נָתַן בְּיָדֵינוּ שֻׁתָּפוּת בִּקְדֻשַּׁת הַזְּמַן. 🧒 וְאַתָּה? כְּשֶׁשּׁוֹמְעִים ״מְקַדֵּשׁ יִשְׂרָאֵל וְרָאשֵׁי חֳדָשִׁים״ בַּתְּפִלָּה – תֵּדַע שֶׁזֶּה גַּם עַל שִׁמְךָ.
📖 פֶּרֶק א׳, הֲלָכָה ב׳

🏛️הַלּוּחַ שֶׁל הִלֵּל הַנָּשִׂיא

בִּימֵי הָרוֹמָאִים גָּבְרוּ הַגְּזֵרוֹת, וְהָיָה קָשֶׁה לְקַדֵּשׁ חֳדָשִׁים בְּבֵית הַדִּין שֶׁבְּאֶרֶץ יִשְׂרָאֵל. לָכֵן הִלֵּל הַשֵּׁנִי, הַנָּשִׂיא, וּבֵית דִּינוֹ חִשְּׁבוּ אֶת הַחֳדָשִׁים וְאֶת הַשָּׁנִים, וְקִדְּשׁוּ אוֹתָם עַד סוֹף כָּל הַדּוֹרוֹת. מֵאָז אָנוּ מוֹנִים לְפִי הַלּוּחַ הָעִבְרִי שֶׁהִתְקִין.
זֶה הָיָה בִּשְׁנַת ד׳ אֲלָפִים קי״ט לִבְרִיאַת הָעוֹלָם. וְאָנוּ מִתְפַּלְּלִים שֶׁנִּזְכֶּה בִּמְהֵרָה לַחֲזֹר וּלְקַדֵּשׁ חֳדָשִׁים בְּבֵית הַדִּין שֶׁבִּירוּשָׁלַיִם.
💛 אַהֲבַת הַשֵּׁם: גַּם בַּזְּמַנִּים הַקָּשִׁים דָּאֲגוּ חַכְמֵי יִשְׂרָאֵל שֶׁהַחַגִּים לֹא יִשָּׁכְחוּ. בִּזְכוּתָם כָּל יִשְׂרָאֵל, בְּכָל מָקוֹם, יוֹדְעִים מָתַי חָל כָּל חַג.
📖 פֶּרֶק א׳, הֲלָכָה ג׳

📅🔑 מוּשָּׂג: תּוֹלְדוֹת קִדּוּשׁ הֶחֳדָשִׁים

״הִלֵּל הַשֵּׁנִי... עָמַד הוּא וּבֵית דִּינוֹ וְחִשְּׁבוּ אֶת הֶחֳדָשִׁים וְהַשָּׁנִים, וְקִדְּשׁוּ אוֹתָם עַד סוֹף כָּל הַדּוֹרוֹת״.
📖 חֶשְׁבּוֹן הַלּוּחַ – מֵאָז הִלֵּל הַשֵּׁנִי, קוֹבְעִים אֶת הֶחֳדָשִׁים לְפִי חִשּׁוּב קָבוּעַ, לֹא לְפִי עֵדִים.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁחָשְׁשׁוּ שֶׁלֹּא יִהְיֶה עוֹד בֵּית דִּין, קִדֵּשׁ הִלֵּל אֶת כָּל הַחֳדָשִׁים עַד סוֹף הַדּוֹרוֹת – כְּדֵי שֶׁעַם יִשְׂרָאֵל לֹא יִשָּׁאֵר בְּלִי לוּחַ. 🧒 וְאַתָּה? תֵּדַע שֶׁהַלּוּחַ שֶׁתָּלוּי בְּבֵיתְכֶם מַמְשִׁיךְ חֻקָּה שֶׁל אֶלֶף וְשֵׁשׁ מֵאוֹת שָׁנָה.
📖 פֶּרֶק א׳, הֲלָכָה ג׳

📅יוֹם אֶחָד אוֹ יוֹמַיִם

כְּשֶׁהַחֹדֶשׁ הַקּוֹדֵם חָסֵר (29 יָמִים), רֹאשׁ חֹדֶשׁ הוּא יוֹם אֶחָד. כְּשֶׁהוּא מָלֵא (30 יוֹם), רֹאשׁ חֹדֶשׁ נִמְשָׁךְ יוֹמַיִם: יוֹם ל׳ שֶׁל הַחֹדֶשׁ הַקּוֹדֵם, וְיוֹם א׳ שֶׁל הַחֹדֶשׁ הֶחָדָשׁ.
עִקַּר רֹאשׁ חֹדֶשׁ הוּא הַיּוֹם הַשֵּׁנִי, אֲבָל גַּם בַּיּוֹם הָרִאשׁוֹן נוֹהֲגִים בְּכָל דִּינֵי רֹאשׁ חֹדֶשׁ: מוּסָף, הַלֵּל וְ"יַעֲלֶה וְיָבוֹא".
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁרֹאשׁ חֹדֶשׁ אָרֹךְ, הַשִּׂמְחָה כְּפוּלָה. כָּל רֹאשׁ חֹדֶשׁ הוּא הִזְדַּמְּנוּת לְהַתְחִיל מֵחָדָשׁ וּלְהוֹסִיף עוֹד קְצָת אוֹר בַּחַיִּים.

🌑לָמָּה הַלְּבָנָה קְטַנָּה?

בַּתְּחִלָּה בָּרָא ה׳ שְׁנֵי מְאוֹרוֹת גְּדוֹלִים. הַלְּבָנָה טָעֲנָה: אֵיךְ יִשְׁתַּמְּשׁוּ שְׁנֵי מְלָכִים בְּכֶתֶר אֶחָד? אָמַר לָהּ ה׳: "לְכִי וּמַעֲטִי אֶת עַצְמֵךְ". לָכֵן בְּרֹאשׁ חֹדֶשׁ הִקְרִיבוּ שָׂעִיר "לְחַטָּאת לַה׳" – כַּפָּרָה עַל מִעוּט הַלְּבָנָה.

📜קְרִיאַת הַתּוֹרָה וּמוּסָף

בְּרֹאשׁ חֹדֶשׁ עוֹלִים לַתּוֹרָה אַרְבָּעָה עוֹלִים, וְקוֹרְאִים עַל קָרְבַּן הַתָּמִיד וְעַל קָרְבְּנוֹת רֹאשׁ חֹדֶשׁ. אַחַר כָּךְ מִתְפַּלְּלִים מוּסָף, וְחוֹתְמִים בּוֹ: "מְקַדֵּשׁ יִשְׂרָאֵל וְרָאשֵׁי חֳדָשִׁים".
חוֹלְצִים אֶת הַתְּפִלִּין לִפְנֵי מוּסָף, אַחֲרֵי הַקַּדִּישׁ, כִּי תְּפִלַּת מוּסָף הִיא "אוֹת" כְּמוֹ יוֹם טוֹב. וְאוֹמְרִים בְּשַׁחֲרִית "בָּרְכִי נַפְשִׁי", שֶׁכָּתוּב בּוֹ: "עָשָׂה יָרֵחַ לְמוֹעֲדִים".

🌕מָתַי מְבָרְכִים עַל הַלְּבָנָה?

נוֹהֲגִים לְבָרֵךְ בְּמוֹצָאֵי שַׁבָּת. לְמִנְהַג הָאַשְׁכְּנַזִּים וּמָרוֹקוֹ – אַחֲרֵי שֶׁעָבְרוּ שְׁלֹשָׁה יָמִים שְׁלֵמִים מִן הַמּוֹלָד. לְמִנְהַג הַסְּפָרַדִּים וְהַחֲסִידִים – אַחֲרֵי הַיּוֹם הַשְּׁבִיעִי לַחֹדֶשׁ. אֶפְשָׁר לְבָרֵךְ עַד סוֹף לֵיל ט״ו, כִּי אַחַר כָּךְ הַלְּבָנָה מַתְחִילָה לְהִתְמַעֵט.
לֹא מְבָרְכִים לִפְנֵי תִּשְׁעָה בְּאָב וְלִפְנֵי יוֹם הַכִּפּוּרִים, כִּי צָרִיךְ לְבָרֵךְ בְּשִׂמְחָה. בְּמוֹצָאֵי יוֹם הַכִּפּוּרִים מְבָרְכִים מִיָּד, וּבְמוֹצָאֵי תִּשְׁעָה בְּאָב נָכוֹן לְחַכּוֹת עַד שֶׁיֹּאכְלוּ וְיִשְׁתּוּ.
💛 אַהֲבַת הַשֵּׁם: הַלְּבָנָה, שֶׁתָּמִיד חוֹזֶרֶת וְגָדֵלָה, מַזְכִּירָה לָנוּ אֶת נֵצַח יִשְׂרָאֵל. לָכֵן אוֹמְרִים בְּשִׂמְחָה: "דָּוִד מֶלֶךְ יִשְׂרָאֵל חַי וְקַיָּם".

👑"דָּוִד מֶלֶךְ יִשְׂרָאֵל חַי וְקַיָּם"

הַלְּבָנָה דּוֹמָה לְעַם יִשְׂרָאֵל: יֵשׁ לָהּ יְרִידוֹת וַעֲלִיּוֹת, אֲבָל הִיא תָּמִיד חוֹזֶרֶת. דָּוִד הַמֶּלֶךְ לִמֵּד אוֹתָנוּ לַהֲפֹךְ כָּל נְפִילָה לַעֲלִיָּה, וְלָכֵן מַלְכוּתוֹ נִמְשְׁלָה לַלְּבָנָה, וְאוֹמְרִים בְּקִדּוּשׁ לְבָנָה: "דָּוִד מֶלֶךְ יִשְׂרָאֵל חַי וְקַיָּם".
בַּגְּאֻלָּה הַשְּׁלֵמָה יִהְיֶה אוֹר הַלְּבָנָה כְּאוֹר הַחַמָּה. וְיֵשׁ שֶׁמּוֹסִיפִים בַּקָּשָׁה: "לְמַלֹּאת פְּגִימַת הַלְּבָנָה, וְלֹא יִהְיֶה בָּהּ שׁוּם מִעוּט".
💛 אַהֲבַת הַשֵּׁם: עַם יִשְׂרָאֵל חַי וְקַיָּם, כִּי הַקֶּשֶׁר שֶׁלּוֹ לַה׳ נִצְחִי. בְּכָל פַּעַם שֶׁאַתָּה רוֹאֶה לְבָנָה חֲדָשָׁה – זְכֹר שֶׁגַּם אַתָּה תָּמִיד יָכוֹל לַחֲזֹר וְלִגְדֹּל.

🌕קִדּוּשׁ לְבָנָה

פַּעַם בְּחֹדֶשׁ מְבָרְכִים עַל הַלְּבָנָה הַמִּתְחַדֶּשֶׁת. מְבָרְכִים בַּלַּיְלָה, כְּשֶׁרוֹאִים אֶת הַלְּבָנָה וְנֶהֱנִים מֵאוֹרָהּ. יוֹצְאִים מִן הַבַּיִת אֶל תַּחַת כִּפַּת הַשָּׁמַיִם, וְאוֹמְרִים אֶת הַבְּרָכָה בַּעֲמִידָה – כְּמוֹ מִי שֶׁיּוֹצֵא לְקַבֵּל פְּנֵי מֶלֶךְ.
נוֹהֲגִים לְבָרֵךְ בְּמוֹצָאֵי שַׁבָּת, בְּשִׂמְחָה וּבְבִגְדֵי שַׁבָּת, וּבְמִנְיָן. כְּשֶׁעֲנָנִים מְכַסִּים אֶת הַלְּבָנָה וְאִי אֶפְשָׁר לֵהָנוֹת מֵאוֹרָהּ – אֵין מְבָרְכִים.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים אָמְרוּ שֶׁבִּרְכַּת הַלְּבָנָה הִיא כְּמוֹ קַבָּלַת פְּנֵי הַשְּׁכִינָה. לָכֵן אַחֲרֶיהָ שָׁרִים וְרוֹקְדִים בְּשִׂמְחָה.

🧠 בְּדֹק אֶת עַצְמְךָ

מִמָּתַי אָנוּ קוֹבְעִים אֶת הַחֳדָשִׁים לְפִי לוּחַ קָבוּעַ?
💡 כְּשֶׁגָּבְרוּ גְּזֵרוֹת הָרוֹמָאִים, חִשֵּׁב הִלֵּל הַנָּשִׂיא עִם בֵּית דִּינוֹ אֶת הַחֳדָשִׁים, וְקִדְּשׁוּ אוֹתָם עַד סוֹף כָּל הַדּוֹרוֹת. לִפְנֵי כֵן קִדְּשׁוּ חֳדָשִׁים עַל פִּי עֵדִים.
לָמָּה הָיוּ מַקְרִיבִים בְּרֹאשׁ חֹדֶשׁ שָׂעִיר "לְחַטָּאת לַה׳"?
💡 הַלְּבָנָה רָצְתָה לִמְלֹךְ לְבַדָּהּ, וְה׳ אָמַר לָהּ: "לְכִי וּמַעֲטִי אֶת עַצְמֵךְ". עַל כָּךְ אָמַר ה׳: "הָבִיאוּ כַּפָּרָה עָלַי", וְלָכֵן הַשָּׂעִיר הוּא "לְחַטָּאת לַה׳".
אַבָּא שָׁכַח "יַעֲלֶה וְיָבוֹא" בְּעַרְבִית שֶׁל רֹאשׁ חֹדֶשׁ, וְסִיֵּם אֶת בִּרְכַּת "רְצֵה". מָה עוֹשֶׂה?
💡 מִי שֶׁשָּׁכַח בְּעַרְבִית וְסִיֵּם אֶת בִּרְכַּת "רְצֵה" – לֹא חוֹזֵר, כִּי בֵּית הַדִּין לֹא קִדֵּשׁ חֳדָשִׁים בַּלַּיְלָה. בְּשַׁחֲרִית וּבְמִנְחָה – חוֹזְרִים.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

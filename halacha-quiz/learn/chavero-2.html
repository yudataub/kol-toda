אֲמִירַת אֱמֶת

🤝 בֵּין אָדָם לַחֲבֵרוֹ · פֶּרֶק ב׳

✅הַחוֹתָם שֶׁל ה׳ – אֱמֶת

הָעוֹלָם עוֹמֵד עַל מִדַּת הָאֱמֶת. חֲזַ״ל אָמְרוּ: "חוֹתָמוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא אֱמֶת". וְהַתּוֹרָה מְצַוָּה: "מִדְּבַר שֶׁקֶר תִּרְחָק" – אָסוּר לְסַפֵּר מָה שֶׁקָּרָה, אוֹ מָה שֶׁאֲנַחְנוּ מְתַכְנְנִים, בְּשִׁנּוּי פְּרָטִים.
גַּם שֶׁקֶר קָטָן שֶׁלֹּא מַזִּיק לְאַף אֶחָד – כְּמוֹ לְהוֹסִיף פְּרָטִים לְסִפּוּר – אָסוּר. וְכָל שֶׁהַשֶּׁקֶר מַזִּיק יוֹתֵר, הוּא חָמוּר יוֹתֵר.
💛 אַהֲבַת הַשֵּׁם: "וְהָלַכְתָּ בִּדְרָכָיו" – ה׳ יוֹדֵעַ אֶת הַמְּצִיאוּת כְּמוֹ שֶׁהִיא, וְלָכֵן חוֹתָמוֹ אֱמֶת. כָּל פַּעַם שֶׁאֲנַחְנוּ אוֹמְרִים אֱמֶת, אֲנַחְנוּ דְּבֵקִים בְּמִדָּתוֹ וּמִתְקָרְבִים אֵלָיו.
📖 פֶּרֶק ב׳, הֲלָכוֹת א׳–ב׳

🎭שְׁנֵי סוּגֵי שֶׁקֶר

יֵשׁ שֶׁקֶר שֶׁפּוֹגֵעַ בְּאַחֵר – גּוֹרֵם לוֹ הֶפְסֵד אוֹ צַעַר, כְּמוֹ מִי שֶׁמַּכְחִישׁ שֶׁלָּוָה כֶּסֶף, אוֹ מְסַכְסֵךְ בֵּין חֲבֵרִים. וְיֵשׁ שֶׁקֶר שֶׁבָּא רַק כְּדֵי לִמְשֹׁךְ תְּשׂוּמַת לֵב, כְּשֶׁמּוֹסִיפִים לְסִפּוּר פְּרָטִים מְדֻמְיָנִים. שְׁנֵיהֶם אֲסוּרִים.
לְשֶׁקֶר שֶׁמַּזִּיק לְאַחֵר אֵין שׁוּם הֶתֵּר, לְעוֹלָם. וְהַשַּׁקְרָן מַזִּיק קֹדֶם כֹּל לְעַצְמוֹ: הוּא מִתְרַגֵּל לִחְיוֹת בְּדִמְיוֹן, וּמִתְרַחֵק מֵהַהֲבָנָה הָאֲמִתִּית שֶׁל הָעוֹלָם.
💛 אַהֲבַת הַשֵּׁם: חֲזַ״ל אָמְרוּ שֶׁהַשַּׁקְרָנִים אֵינָם זוֹכִים לִרְאוֹת פְּנֵי שְׁכִינָה. מִי שֶׁדָּבֵק בָּאֱמֶת, גַּם כְּשֶׁהִיא עוֹלָה לוֹ בְּמַשֶּׁהוּ, זוֹכֶה לֶאֱמוּנָה חֲזָקָה יוֹתֵר בַּה׳.
📖 פֶּרֶק ב׳, הֲלָכוֹת ג׳–ד׳

🚫רְחוֹקִים מִשֶּׁקֶר

"מִדְּבַר שֶׁקֶר תִּרְחָק" – לֹא רַק לֹא לְשַׁקֵּר, אֶלָּא גַּם לֹא לְהֵרָאוֹת כְּמִי שֶׁעוֹמֵד לְשַׁקֵּר. תַּלְמִיד שֶׁרַבּוֹ מְבַקֵּשׁ שֶׁיָּעִיד עַל הַלְוָאָה שֶׁלֹּא רָאָה – אָסוּר לוֹ, גַּם אִם הוּא בָּטוּחַ שֶׁרַבּוֹ צוֹדֵק. וַאֲפִלּוּ לָבוֹא לְבֵית הַדִּין וְלַעֲמֹד שָׁם כְּאִלּוּ יָעִיד – אָסוּר.
גַּם מִי שֶׁצּוֹדֵק אָסוּר לוֹ לְהַשִּׂיג אֶת מָה שֶׁמַּגִּיעַ לוֹ בְּעֶזְרַת שֶׁקֶר. עָדִיף לְהַפְסִיד – וְלֹא לְשַׁקֵּר.
💛 אַהֲבַת הַשֵּׁם: "שְׂפַת אֱמֶת תִּכּוֹן לָעַד" – דּוֹבֵר הָאֱמֶת קָשׁוּר לַמְּצִיאוּת שֶׁה׳ בָּרָא, וְלָכֵן הוּא יַצִּיב וְחָזָק. גַּם אִם הִפְסִיד לְרֶגַע, בַּסּוֹף יֵצֵא נִשְׂכָּר.
📖 פֶּרֶק ב׳, הֲלָכוֹת ה׳–ו׳

🤝עוֹמְדִים בַּדִּבּוּר

מִי שֶׁאָמַר לְחָבֵר שֶׁיִּתֵּן לוֹ מַתָּנָה קְטַנָּה, אוֹ שֶׁיָּבוֹא לַעֲזֹר לוֹ בָּעֶרֶב – צָרִיךְ לְהִתְאַמֵּץ לְקַיֵּם. מִי שֶׁלֹּא מְקַיֵּם נִקְרָא "מְחֻסַּר אֲמָנָה" – אָדָם שֶׁאִי אֶפְשָׁר לִסְמֹךְ עָלָיו.
וְאִם כְּבָר כְּשֶׁאָמַר לֹא הִתְכַּוֵּן לְקַיֵּם – זֶה חָמוּר יוֹתֵר: "אֶחָד בַּפֶּה וְאֶחָד בַּלֵּב". אֲבָל מִי שֶׁהִבְטִיחַ מַתָּנָה גְּדוֹלָה – כֻּלָּם מְבִינִים שֶׁהִתְכַּוֵּן לָתֵת רַק אִם יוּכַל.
💛 אַהֲבַת הַשֵּׁם: הַדִּבּוּר הוּא הַמַּתָּנָה הַמְּיֻחֶדֶת שֶׁה׳ נָתַן לָאָדָם. לָכֵן "כָּל הַמְקַיֵּם אֶת דְּבָרוֹ – רוּחַ חֲכָמִים נוֹחָה הֵימֶנּוּ".
📖 פֶּרֶק ב׳, הֲלָכָה ט׳

🧭רַב סָפְרָא וְהַהַחְלָטָה שֶׁבַּלֵּב

רַב סָפְרָא רָצָה לִמְכֹּר חֵפֶץ. בָּא קוֹנֶה בְּשָׁעָה שֶׁקָּרָא קְרִיאַת שְׁמַע וְהִצִּיעַ סְכוּם. רַב סָפְרָא לֹא עָנָה, וְהַקּוֹנֶה חָשַׁב שֶׁזֶּה מְעַט וְהוֹסִיף. אַחֲרֵי קְרִיאַת שְׁמַע אָמַר לוֹ: "תֵּן לִי אֶת הַסְּכוּם הָרִאשׁוֹן, כִּי כְּבָר אָז הִסְכַּמְתִּי בְּלִבִּי".
זוֹ מִדַּת חֲסִידוּת: לְקַיֵּם גַּם הַחְלָטָה שֶׁבַּלֵּב. מִן הַדִּין רַק הַדִּבּוּר מְחַיֵּב – וּמִי שֶׁלֹּא מְקַיֵּם אֶת דִּבּוּרוֹ נִקְרָא מְחֻסַּר אֲמָנָה.
💛 אַהֲבַת הַשֵּׁם: רַב סָפְרָא הִפְסִיד הַרְבֵּה זְהוּבִים, אֲבָל זָכָה לְהִדָּבֵק בְּמִדּוֹתָיו שֶׁל ה׳ וּלְהַכִּיר טוֹב יוֹתֵר אֶת הָאֱמֶת.
📖 פֶּרֶק ב׳, הֲלָכוֹת ט׳, י״ב

🤝עוֹמְדִים בַּדִּבּוּר – גַּם כְּשֶׁמַּצִּיעִים יוֹתֵר

מִי שֶׁסִּכֵּם עִם קוֹנֶה שֶׁיִּמְכֹּר לוֹ חֵפֶץ בִּמְחִיר מְסֻיָּם, וְאַחַר כָּךְ בָּא קוֹנֶה אַחֵר וּמַצִּיעַ יוֹתֵר – צָרִיךְ לִמְכֹּר לָרִאשׁוֹן בַּמְּחִיר שֶׁסִּכְּמוּ. הַסִּכּוּם מְחַיֵּב, אַף שֶׁהָיָה רַק בְּדִבּוּר.
בֵּית הַדִּין אֵינוֹ יָכוֹל לְהַכְרִיחַ אוֹתוֹ, כִּי לֹא נַעֲשָׂה קִנְיָן וְלֹא נֶחְתַּם שְׁטָר – אֲבָל חֲכָמִים קוֹרְאִים לְמִי שֶׁחוֹזֵר בּוֹ ״מְחֻסַּר אֲמָנָה״. רַק אִם אָמְרוּ בְּפֵרוּשׁ שֶׁהַסִּכּוּם אֵינוֹ מְחַיֵּב – מֻתָּר לְשַׁנּוֹת.
💛 אַהֲבַת הַשֵּׁם: חוֹתָמוֹ שֶׁל ה׳ אֱמֶת. כְּשֶׁאֲנִי עוֹמֵד בְּדִבּוּרִי גַּם כְּשֶׁזֶּה עוֹלֶה לִי כֶּסֶף – אֲנִי הוֹלֵךְ בִּדְרָכָיו.
📖 פֶּרֶק ב׳, הֲלָכָה י״א

🕊️מְשַׁנִּים מִפְּנֵי הַשָּׁלוֹם

לְמַעַן הַשָּׁלוֹם מֻתָּר וְאַף מִצְוָה לְשַׁנּוֹת מֵהָאֱמֶת. כְּשֶׁשָּׂרָה אָמְרָה שֶׁגַּם אַבְרָהָם זָקֵן, ה׳ סִפֵּר לְאַבְרָהָם רַק שֶׁאָמְרָה שֶׁהִיא זְקֵנָה – כְּדֵי שֶׁלֹּא יִפָּגַע. וַאֲחֵי יוֹסֵף אָמְרוּ לוֹ שֶׁאֲבִיהֶם צִוָּה לִסְלֹחַ לָהֶם, כְּדֵי שֶׁיִּהְיֶה שָׁלוֹם.
וּבְכָל זֹאת, לְכַתְּחִלָּה עָדִיף לְהַטּוֹת אֶת הַשִּׂיחָה לְנוֹשֵׂא אַחֵר. וּמְשַׁנִּים רַק כַּמָּה שֶׁצָּרִיךְ – וְרַק כְּשֶׁהַשִּׁנּוּי לֹא גּוֹרֵם נֵזֶק אוֹ בּוּשָׁה לְאַף אֶחָד.
💛 אַהֲבַת הַשֵּׁם: בְּעֹמֶק הַנְּשָׁמָה כָּל בְּנֵי הָאָדָם רוֹצִים לֶאֱהֹב זֶה אֶת זֶה, כִּי בּוֹרֵא אֶחָד בְּרָאָם. לָכֵן הַשָּׁלוֹם הוּא אֱמֶת עֲמֻקָּה עוֹד יוֹתֵר.
📖 פֶּרֶק ב׳, הֲלָכוֹת י״ג–י״ד

⚖️רַק לְמַעַן שָׁלוֹם אֲמִתִּי

אַהֲרֹן הַכֹּהֵן הָיָה הוֹלֵךְ אֶל כָּל אֶחָד מִשְּׁנֵי הַמִּתְקוֹטְטִים וְאוֹמֵר לוֹ: חֲבֵרְךָ מִצְטַעֵר וּמִתְבַּיֵּשׁ עַל מָה שֶׁעָשָׂה לְךָ. כְּשֶׁנִּפְגְּשׁוּ – הִתְחַבְּקוּ. אֲבָל מְשַׁנִּים רַק כְּשֶׁבָּרוּר שֶׁיֵּצֵא מִזֶּה שָׁלוֹם, וְרַק כְּשֶׁהַשֶּׁקֶר לֹא מַזִּיק לְאַף אֶחָד.
אִם מִישֶׁהוּ הִזִּיק לַחֲבֵרוֹ אוֹ עוֹמֵד לְרַמּוֹת אוֹתוֹ – אָסוּר לִשְׁתֹּק בְּשֵׁם הַשָּׁלוֹם, אֶלָּא חַיָּבִים לְהַזְהִיר. וּמִי שֶׁכְּבָר נִכְשַׁל בְּשֶׁקֶר – לֹא יְשַׁנֶּה אֲפִלּוּ לְמַעַן הַשָּׁלוֹם, כְּדֵי שֶׁלֹּא יִתְרַגֵּל לְהַתִּיר לְעַצְמוֹ.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁמֻּתָּר לְשַׁנּוֹת, מְשַׁנִּים רַק מָה שֶׁהֶכְרֵחִי. כָּךְ שׁוֹמְרִים עַל הָאֱמֶת – הַחוֹתָם שֶׁל ה׳ – גַּם בְּתוֹךְ מַעֲשֵׂה הַשָּׁלוֹם.
📖 פֶּרֶק ב׳, הֲלָכוֹת י״ג, ט״ו

🏃רוֹדְפִים שָׁלוֹם

מִצְוָה לַעֲשׂוֹת שָׁלוֹם בֵּין אָדָם לַחֲבֵרוֹ, בֵּין מִשְׁפָּחָה לְמִשְׁפָּחָה וּבֵין מְדִינָה לִמְדִינָה. אֶת רֹב הַמִּצְווֹת מְקַיְּמִים כְּשֶׁהֵן מַגִּיעוֹת לְיָדֵנוּ, אֲבָל אַחֲרֵי הַשָּׁלוֹם צָרִיךְ לִרְדֹּף: "בַּקֵּשׁ שָׁלוֹם וְרָדְפֵהוּ".
אַהֲרֹן הַכֹּהֵן הָיָה אוֹהֵב שָׁלוֹם וְרוֹדֵף שָׁלוֹם, וְלָכֵן כְּשֶׁנִּפְטַר בָּכוּ עָלָיו כָּל בֵּית יִשְׂרָאֵל שְׁלוֹשִׁים יוֹם.
💛 אַהֲבַת הַשֵּׁם: כָּל הַבְּרִיּוֹת בְּרוּאוֹת בִּידֵי בּוֹרֵא אֶחָד. כְּשֶׁאֲנַחְנוּ מְקָרְבִים לְבָבוֹת, אֲנַחְנוּ מְגַלִּים בָּעוֹלָם אֶת הָאַחְדוּת שֶׁל ה׳.
📖 פֶּרֶק ב׳, הֲלָכוֹת י״ד–ט״ו

🧑‍🏫אֱמֶת לַמּוֹרֶה

לִפְעָמִים מֻתָּר לְשַׁנּוֹת מֵהָאֱמֶת כְּדֵי לְהִמָּנַע מִבּוּשָׁה. אֲבָל לַמּוֹרֶה תַּלְמִיד חַיָּב לוֹמַר אֱמֶת, כִּי הַמּוֹרֶה צָרִיךְ לָדַעַת אֵיפֹה הַתַּלְמִיד עוֹמֵד, כְּדֵי לַעֲזֹר לוֹ לְהִתְקַדֵּם. לָכֵן אִם הַמּוֹרֶה שׁוֹאֵל "חָזַרְתָּ לַמִּבְחָן?" – עוֹנִים אֶת הָאֱמֶת.
וְאִם הַתַּלְמִיד נָבוֹךְ מְאוֹד מוּל כָּל הַכִּתָּה – יִתָּכֵן שֶׁמֻּתָּר לוֹ לְטַשְׁטֵשׁ שָׁם, אֲבָל אַחַר כָּךְ יָבוֹא אֶל הַמּוֹרֶה וְיֹאמַר לוֹ אֶת הָאֱמֶת.
💛 אַהֲבַת הַשֵּׁם: הַמּוֹרֶה וְהַתַּלְמִיד עוֹבְדִים יַחַד, כְּמוֹ שֻׁתָּפִים. אֱמֶת בֵּינֵיהֶם בּוֹנָה אֵמוּן – וְכָךְ אֶפְשָׁר לִלְמֹד אֶת תּוֹרַת ה׳ בֶּאֱמֶת.

🎒מַחְמָאָה לְחָבֵר

חָבֵר קָנָה דָּבָר וְשׁוֹאֵל מָה דַּעְתְּךָ, וְהַדָּבָר לֹא מוֹצֵא חֵן בְּעֵינֶיךָ. אֵין מִצְוָה לוֹמַר לוֹ אֶת כָּל הָאֱמֶת – הוּא כְּבָר קָנָה, וְרַק תְּצַעֵר אוֹתוֹ. עָדִיף לֹא לְדַבֵּר עַל זֶה, וְאִם הוּא שׁוֹאֵל – מְשַׁבְּחִים קְצָת כְּדֵי לְשַׂמֵּחַ אוֹתוֹ.
וְאִם אֶפְשָׁר לִמְצֹא בַּדָּבָר מַעֲלָה אֲמִתִּית – עָדִיף לְהִתְרַכֵּז בָּהּ. וְכָךְ פָּסְקוּ כְּבֵית הִלֵּל: בַּחֲתֻנָּה רוֹקְדִים לִפְנֵי הַכַּלָּה וְשָׁרִים "כַּלָּה נָאָה וַחֲסוּדָה".
💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁנַּרְבֶּה שָׁלוֹם וְאַהֲבָה בֵּין הַבְּרִיּוֹת. מִלָּה טוֹבָה בַּזְּמַן הַנָּכוֹן מְשַׂמַּחַת אֶת לֵב הֶחָבֵר.

📝לֹא מַעְתִּיקִים בַּמִּבְחָן

אָסוּר לְהַעְתִּיק בַּמִּבְחָן. זוֹ גְּנֵבַת דַּעַת: הַמּוֹרֶה חוֹשֵׁב שֶׁהַתַּלְמִיד יוֹדֵעַ, וּבֶאֱמֶת הוּא לֹא יוֹדֵעַ. וְזֶה גַּם הֵפֶךְ מִדַּת הָאֱמֶת, שֶׁבָּהּ אֲנַחְנוּ מְצֻוִּים לָלֶכֶת.
הָאֱמֶת יְקָרָה יוֹתֵר מֵהַצִּיּוּן. מִי שֶׁלֹּא מַעְתִּיק לוֹמֵד לִסְמֹךְ עַל הַיְּדִיעוֹת וְהַמַּאֲמַצִּים שֶׁלּוֹ, וּבַסּוֹף יַצְלִיחַ יוֹתֵר. וּמִי שֶׁמַּעְתִּיק – בַּסּוֹף מְגַלֶּה שֶׁהוּא רִמָּה בְּעִקָּר אֶת עַצְמוֹ.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי כּוֹתֵב רַק מָה שֶׁאֲנִי יוֹדֵעַ, אֲנִי הוֹלֵךְ בְּדַרְכֵי ה׳ – שֶׁחוֹתָמוֹ אֱמֶת. מַאֲמָץ אֲמִתִּי קָטָן שָׁוֶה יוֹתֵר מִצִּיּוּן גָּבוֹהַּ שֶׁל שֶׁקֶר.

🧠 בְּדֹק אֶת עַצְמְךָ

חָבֵר רוֹצֶה לַעֲשׂוֹת שָׁלוֹם בֵּין שְׁנֵי יְלָדִים שֶׁרָבוּ. מָתַי מֻתָּר לוֹ לְשַׁנּוֹת מֵהָאֱמֶת?
💡 אַהֲרֹן הַכֹּהֵן שִׁנָּה כְּדֵי לְהַשְׁכִּין שָׁלוֹם. אֲבָל אִם הַשֶּׁקֶר יַזִּיק לְמִישֶׁהוּ, אוֹ שֶׁלֹּא בָּרוּר שֶׁיָּבִיא שָׁלוֹם – אָסוּר לְשַׁנּוֹת.
שׁוֹאֲלִים אֶת יוֹסֵף אִם הוּא יוֹדֵעַ הֵיטֵב אֶת כָּל הַמַּסֶּכֶת, וְהוּא יוֹדֵעַ. לָמָּה מֻתָּר לוֹ לוֹמַר שֶׁלֹּא?
💡 חֲכָמִים הִתִּירוּ לְשַׁנּוֹת בְּעִנְיְנֵי הַמַּסֶּכֶת מִשּׁוּם עֲנָוָה, שֶׁלֹּא יְסַפֵּר בְּעַצְמוֹ עַל יְדִיעוֹתָיו. וְגַם כָּאן עָדִיף לְדַבֵּר בְּלָשׁוֹן שֶׁאֶפְשָׁר לְהָבִין בִּשְׁתֵּי דְּרָכִים, וְלֹא בְּשֶׁקֶר גָּמוּר.
חָבֵר בָּטוּחַ שֶׁאָבִי לָקַח לוֹ עֵט, וּמְבַקֵּשׁ שֶׁתַּגִּיד לַמּוֹרֶה שֶׁרָאִיתָ. אֲבָל לֹא רָאִיתָ. מָה עוֹשִׂים?
💡 "מִדְּבַר שֶׁקֶר תִּרְחָק" – אָסוּר לְהָעִיד עַל מָה שֶׁלֹּא רָאִינוּ, גַּם כְּשֶׁבְּטוּחִים שֶׁהֶחָבֵר צוֹדֵק. וַאֲפִלּוּ לְהֵרָאוֹת כְּמִי שֶׁעוֹמֵד לְשַׁקֵּר – אָסוּר.

הַטִּפּוּל בַּגּוּף

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק י״ד

✂️צִפָּרְנַיִם וְשֵׂעָר

אָסוּר לִגְזֹז צִפָּרְנַיִם אוֹ שְׂעָרוֹת, וְגַם לִכְסֹס צִפָּרְנַיִם בַּשִּׁנַּיִם אָסוּר. אָסוּר לְהִסְתָּרֵק בְּמַסְרֵק, כִּי נִתְלָשׁוֹת שְׂעָרוֹת – וְהַמַּסְרֵק מֻקְצֶה. אֲבָל מֻתָּר לְסַדֵּר אֶת הַשֵּׂעָר בְּמִבְרֶשֶׁת רַכָּה, שֶׁאֵינָהּ תּוֹלֶשֶׁת שְׂעָרוֹת, וְטוֹב לְיַחֵד אוֹתָהּ לְשַׁבָּת.
קוֹץ שֶׁנִּתְקַע בָּאֶצְבַּע – מֻתָּר לְהוֹצִיא, וְנִזְהָרִים שֶׁלֹּא יֵצֵא דָּם. וְעוֹר שֶׁמִּתְקַלֵּף בַּשְּׂפָתַיִם – אָסוּר לִתְלֹשׁ, גַּם בַּשִּׁנַּיִם.
💛 אַהֲבַת הַשֵּׁם: גַּם הַגּוּף שֶׁלָּנוּ שׁוֹבֵת בְּשַׁבָּת. אֶת הַטִּפּוּחַ עוֹשִׂים בְּיוֹם שִׁשִּׁי, וּבְשַׁבָּת בָּאִים לִפְנֵי ה׳ מוּכָנִים וּמְסֻדָּרִים – כְּמוֹ לִפְגִישָׁה עִם מֶלֶךְ.
📖 פֶּרֶק י״ד, הֲלָכוֹת ב׳–ג׳

💇שֵׂעָר, אִפּוּר וּבֹשֶׂם

חֲכָמִים אָסְרוּ לִקְלֹעַ צַמָּה בְּשַׁבָּת, כִּי הַקְּלִיעָה דּוֹמָה לִמְלֶאכֶת בּוֹנֶה, וְגַם לְפָרֵק צַמָּה אָסוּר, כִּי זֶה דּוֹמֶה לְסוֹתֵר. אֲבָל לֶאֱסֹף אֶת הַשֵּׂעָר בְּגוּמִיָּה – מֻתָּר, וּמֻתָּר לַעֲשׂוֹת שְׁבִיל בָּאֶמְצַע הַשֵּׂעָר בַּיָּדַיִם, וְלֹא בְּמַסְרֵק אוֹ בְּמִבְרֶשֶׁת. אָסוּר לְקַבֵּעַ תִּסְרֹקֶת בִּסְפְּרֵי אוֹ בְּשֶׁמֶן, אֲבָל לְהַתִּיז בֹּשֶׂם עַל הַשֵּׂעָר וְעַל הַגּוּף – מֻתָּר.
אִפּוּר אָסוּר בְּשַׁבָּת: לֹא צוֹבְעִים סְבִיב הָעֵינַיִם, לֹא אֶת הַלְּחָיַיִם וְלֹא אֶת הַשְּׂפָתַיִם – גַּם לֹא בִּשְׂפָתוֹן שָׁקוּף שֶׁנּוֹתֵן בָּרָק. גַּם לַכָּה לַצִּפָּרְנַיִם אֲסוּרָה, אֲפִלּוּ שְׁקוּפָה, כִּי גַּם הַבָּרָק נֶחְשָׁב צֶבַע. לְהָסִיר אִפּוּר בְּמַיִם אוֹ בְּצֶמֶר גֶּפֶן יָבֵשׁ – מֻתָּר.
💛 אַהֲבַת הַשֵּׁם: לִפְנֵי שַׁבָּת מִתְכּוֹנְנִים וּמִתְיַפִּים לִכְבוֹדָהּ, וּבְשַׁבָּת עַצְמָהּ נָחִים וּשְׂמֵחִים בַּיֹּפִי הַטִּבְעִי שֶׁה׳ נָתַן לָנוּ, הַמְעֻטָּר בְּבִגְדֵי שַׁבָּת וּבְתַכְשִׁיטִים.
📖 פֶּרֶק י״ד, הֲלָכוֹת ג׳–ד׳

🧴מְמָרֵחַ וְסַבּוֹן

מְלֶאכֶת מְמָרֵחַ הִיא מְרִיחָה וְהַחְלָקָה שֶׁל חֹמֶר סָמִיךְ. לָכֵן אָסוּר לִמְרֹחַ קְרֶם אוֹ מִשְׁחָה עַל הָעוֹר – גַּם כְּשֶׁרוֹצִים שֶׁיִּבָּלַע. אֲבָל שֶׁמֶן, וְתַכְשִׁיר נוֹזְלִי שֶׁמִּתְפַּשֵּׁט כְּשֶׁמַּנִּיחִים אוֹתוֹ עַל מִשְׁטָח – מֻתָּר. בְּסַבּוֹן מוּצָק אוֹ סָמִיךְ רֹב יִשְׂרָאֵל נוֹהֲגִים שֶׁלֹּא לְהִשְׁתַּמֵּשׁ, וְסַבּוֹן נוֹזְלִי – מֻתָּר.
מַגְבּוֹנִים לַחִים – מֻתָּר לְהִשְׁתַּמֵּשׁ בָּהֶם. וּבְמִשְׁחַת שִׁנַּיִם – נָכוֹן שֶׁלֹּא לְהִשְׁתַּמֵּשׁ, וּבִשְׁעַת הַצֹּרֶךְ אֶפְשָׁר לְהָקֵל.
💛 אַהֲבַת הַשֵּׁם: גַּם בַּטִּפּוּל בַּגּוּף אֶפְשָׁר לְהַרְגִּישׁ שֶׁשַּׁבָּת הַיּוֹם. כְּשֶׁאָנוּ מְשַׁנִּים מֵהַהֶרְגֵּלִים שֶׁל חֹל, הַגּוּף וְהַנְּשָׁמָה יַחַד מַרְגִּישִׁים אֶת קְדֻשַּׁת הַיּוֹם.
📖 פֶּרֶק י״ד, הֲלָכוֹת ה׳–ז׳

🪥רְחִיצָה וְצִחְצוּחַ

מֻתָּר לִרְחֹץ יָדַיִם בְּסַבּוֹן נוֹזְלִי, וּמֻתָּר לְצַחְצֵחַ שִׁנַּיִם בְּמִבְרֶשֶׁת – רַק אִם כִּמְעַט וַדַּאי שֶׁיֵּצֵא דָּם מִן הַחֲנִיכַיִם, אָסוּר. חֲכָמִים גָּזְרוּ שֶׁלֹּא לִרְחֹץ אֶת רֹב הַגּוּף בְּמַיִם חַמִּים, אֲבָל אֶת הַפָּנִים וְהַיָּדַיִם – מֻתָּר לִרְחֹץ גַּם בְּחַמִּים.
שְׂחִיָּה – אֲסוּרָה בְּשַׁבָּת, גַּם בִּבְרֵכָה. וּמִי שֶׁמִּתְרַחֵץ – אֵינוֹ סוֹחֵט אֶת הַשֵּׂעָר בַּיָּדַיִם, אֲבָל מֻתָּר לְנַגֵּב אוֹתוֹ בְּמַגֶּבֶת.
💛 אַהֲבַת הַשֵּׁם: הַזְּמַן הַפָּנוּי שֶׁל שַׁבָּת יָקָר מְאוֹד. בִּמְקוֹם בִּלּוּיִים שֶׁל חֹל, אָנוּ מַקְדִּישִׁים אוֹתוֹ לַתּוֹרָה, לַתְּפִלָּה וְלַמִּשְׁפָּחָה – וְזֶה הָעֹנֶג הָאֲמִתִּי.
📖 פֶּרֶק י״ד, הֲלָכוֹת ו׳–ט׳

🔀עוֹד דֵּעוֹת וּמִנְהָגִים

סַבּוֹן מוּצָק: לְדַעַת הַרְבֵּה פּוֹסְקִים אֵין בּוֹ אִסּוּר מֵעִקַּר הַדִּין, אֲבָל רֹב יִשְׂרָאֵל נָהֲגוּ לְהַחְמִיר וְלֹא לְהִשְׁתַּמֵּשׁ בּוֹ, וְהַנּוֹהֲגִים לְהָקֵל – יֵשׁ לָהֶם עַל מָה לִסְמֹךְ. בֹּץ עַל הַנַּעֲלַיִם: לְפִי הַשֻּׁלְחָן עָרוּךְ לֹא מְקַנְּחִים אוֹתוֹ בָּאֲדָמָה, שֶׁמָּא יַשְׁווּ גֻּמּוֹת, וְהָרַמָּ״א מַתִּיר. הָרוֹצֶה לְהָקֵל – רַשַּׁאי, וְטוֹב לְהַחְמִיר. עַל מְגָרֶדֶת אוֹ עַל מַרְצֶפֶת – מֻתָּר לְכֻלָּם.
פַּחִית שְׁתִיָּה: יֵשׁ אוֹסְרִים לִפְתֹּחַ אֶת הַלְּשׁוֹנִית, וְרַבִּים מַתִּירִים. הָרוֹצֶה לְהָקֵל – רַשַּׁאי, וְהָרוֹצֶה לְהַחְמִיר – מוֹזֵג לְכוֹס וְלֹא שׁוֹתֶה מִן הַפַּחִית. פְּקַק מַתֶּכֶת שֶׁל בַּקְבּוּק יַיִן: לְדַעַת רֹב הַפּוֹסְקִים מֻתָּר לִפְתֹּחַ, וְיֵשׁ אוֹסְרִים – וְטוֹב לִפְתֹּחַ מֵעֶרֶב שַׁבָּת.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁיֵּשׁ כַּמָּה דֵּעוֹת, כָּל מִשְׁפָּחָה הוֹלֶכֶת בְּדֶרֶךְ רַבּוֹתֶיהָ, וְאֵינָהּ מְזַלְזֶלֶת בַּדֶּרֶךְ שֶׁל חֲבֵרָתָהּ. כָּךְ לוֹמְדִים לְכַבֵּד אֶת הַתּוֹרָה וְאֶת כָּל הַשּׁוֹמְרִים אוֹתָהּ.
📖 פֶּרֶק י״ד, הֲלָכָה ו׳; פֶּרֶק ט״ו, הֲלָכוֹת ב׳, י״ג–י״ד

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

מִן הַדִּין אָסוּר לִרְחֹץ אֶת רֹב הַגּוּף בְּמַיִם חַמִּים בְּשַׁבָּת, אֲבָל בְּמַיִם קָרִים אוֹ פּוֹשְׁרִים – מֻתָּר. הָאַשְׁכְּנַזִּים נָהֲגוּ לְהַחְמִיר וְלֹא לְהִתְרַחֵץ גַּם בְּמַיִם קָרִים, מִשּׁוּם כַּמָּה חֲשָׁשׁוֹת: שֶׁמָּא יִסְחֲטוּ אֶת הַשֵּׂעָר בַּיָּדַיִם, שֶׁמָּא יְטַלְטְלוּ אֶת הַמַּיִם שֶׁעַל גּוּפָם בִּרְשׁוּת הָרַבִּים, וְשֶׁמָּא יָבוֹאוּ לִשְׂחוֹת.
בְּיוֹם חַם, אוֹ לְמִי שֶׁהִזִּיעַ, מֻתָּר לְכֻלָּם לְהִתְרַחֵץ בְּמַיִם קָרִים אוֹ פּוֹשְׁרִים. וּבְלִי צֹרֶךְ – הַמִּנְהָג הָרוֹוֵחַ שֶׁלֹּא לְהִתְרַחֵץ בְּשַׁבָּת. וְלִשְׂחוֹת – אָסוּר לְכֻלָּם.
💛 אַהֲבַת הַשֵּׁם: כָּל עֵדָה שׁוֹמֶרֶת אֶת הַשַּׁבָּת בַּדֶּרֶךְ שֶׁקִּבְּלָה מֵרַבּוֹתֶיהָ, וּשְׁתֵּי הַדְּרָכִים הֵן תּוֹרָה. כְּשֶׁאָנוּ מְכַבְּדִים אֶת הַמִּנְהָג שֶׁל חָבֵר מֵעֵדָה אַחֶרֶת, אָנוּ מַרְבִּים אַהֲבָה בֵּין בָּנָיו שֶׁל ה׳.
📖 פֶּרֶק י״ד, הֲלָכוֹת ח׳–ט׳

🧠 בְּדֹק אֶת עַצְמְךָ

לְאוּרִי יֵשׁ עוֹר יָבֵשׁ בַּיָּדַיִם. מָה מֻתָּר לוֹ לִמְרֹחַ בְּשַׁבָּת?
💡 מְרִיחַת חֹמֶר סָמִיךְ עַל הָעוֹר הִיא מְמָרֵחַ – גַּם כְּשֶׁרוֹצִים שֶׁיִּבָּלַע, כִּי חֵלֶק נִשְׁאָר וּמַחְלִיק אֶת הָעוֹר. בְּשֶׁמֶן וּבְתַכְשִׁיר נוֹזְלִי – אֵין מְמָרֵחַ.
לִכְבוֹד סְעוּדַת שַׁבָּת רוֹצָה רִבְקָה לְסַדֵּר אֶת שְׂעָרָהּ. מָה מֻתָּר לָהּ לַעֲשׂוֹת בְּשַׁבָּת עַצְמָהּ?
💡 חֲכָמִים אָסְרוּ לִקְלֹעַ צַמָּה וּלְפָרֵק אוֹתָהּ, וְאָסוּר לְקַבֵּעַ תִּסְרֹקֶת בִּסְפְּרֵי. שְׁבִיל מֻתָּר לַעֲשׂוֹת רַק בַּיָּדַיִם, כִּי הַמַּסְרֵק תּוֹלֵשׁ שְׂעָרוֹת. לֶאֱסֹף אֶת הַשֵּׂעָר בְּגוּמִיָּה – מֻתָּר, כִּי אֵין בָּזֶה בִּנְיָן.
יַעֲלָה רוֹצָה לִמְרֹחַ בְּשַׁבָּת לַכָּה שְׁקוּפָה עַל צִפָּרְנֶיהָ, בְּלִי שׁוּם צֶבַע. מָה הַדִּין?
💡 לַכָּה, שְׂפָתוֹן וְצֶבַע סְבִיב הָעֵינַיִם – הֵם צְבִיעָה, וַחֲכָמִים אָסְרוּ גַּם צֶבַע שֶׁאֵינוֹ מִתְקַיֵּם זְמַן רַב. גַּם לַכָּה שְׁקוּפָה אֲסוּרָה, כִּי הַבָּרָק נֶחְשָׁב צֶבַע. לִפְנֵי שַׁבָּת – מֻתָּר וְאַף רָאוּי לְהִתְיַפּוֹת לִכְבוֹדָהּ.

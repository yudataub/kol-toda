מִצְוַת הַלְוָאָה

🤝 בֵּין אָדָם לַחֲבֵרוֹ · פֶּרֶק ה׳

🤝מִצְוָה לְהַלְווֹת

מִצְוַת עֲשֵׂה מִן הַתּוֹרָה לְהַלְווֹת כֶּסֶף לְמִי שֶׁצָּרִיךְ, שֶׁנֶּאֱמַר: "אִם כֶּסֶף תַּלְוֶה אֶת עַמִּי". הַמִּצְוָה מֻטֶּלֶת עַל כָּל אֶחָד – אִישׁ וְאִשָּׁה, עָשִׁיר וְעָנִי. וּמַלְוִים לֹא רַק לֶעָנִי – גַּם לְעָשִׁיר שֶׁצָּרִיךְ.
חֲכָמִים אָמְרוּ שֶׁהַהַלְוָאָה גְּדוֹלָה אֲפִלּוּ מִן הַצְּדָקָה: מִי שֶׁלּוֹוֶה אֵינוֹ מִתְבַּיֵּשׁ, כִּי הוּא יַחֲזִיר אֶת הַכֶּסֶף, וְהַמְּצוּקָה שֶׁלּוֹ לֹא מִתְגַּלָּה לְכֻלָּם.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאַתָּה מַלְוֶה לְחָבֵר, אַתָּה עוֹזֵר לוֹ לַעֲמֹד בְּכָבוֹד. ה׳ אוֹהֵב אֶת מִי שֶׁשּׁוֹמֵר עַל הַכָּבוֹד שֶׁל אַחֵר – וְכָךְ אַתָּה הוֹלֵךְ בִּדְרָכָיו.
📖 פֶּרֶק ה׳, הֲלָכוֹת א׳, ו׳

⏰מַחֲזִירִים – וּמַלְוִים שׁוּב

אִם לֹא קָבְעוּ מָתַי לְהַחֲזִיר, הַלּוֹוֶה מַחֲזִיר אֶת הַהַלְוָאָה אַחֲרֵי שְׁלוֹשִׁים יוֹם. וְאִם הֶחֱזִיר וּבִקֵּשׁ שׁוּב – מַלְוִים לוֹ שׁוּב, אֲפִלּוּ מֵאָה פְּעָמִים, בְּשִׂמְחָה וּבְחִבָּה.
הֶחָפֵץ חַיִּים מָשַׁל מָשָׁל: בַּעַל חֲנוּת לֹא כּוֹעֵס עַל קוֹנֶה שֶׁחוֹזֵר שׁוּב וְשׁוּב, כִּי בְּכָל קְנִיָּה הוּא מַרְוִיחַ. כָּךְ בְּכָל הַלְוָאָה – מַרְוִיחִים עוֹד מִצְוָה.
💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לָנוּ חַיִּים וּבְרָכָה בְּכָל יוֹם מֵחָדָשׁ, וְאֵינוֹ אוֹמֵר "כְּבָר נָתַתִּי". כְּשֶׁאָנוּ עוֹזְרִים שׁוּב וְשׁוּב, אָנוּ דּוֹמִים לוֹ.
📖 פֶּרֶק ה׳, הֲלָכָה ג׳

💸בְּלִי רִבִּית

אָסוּר לְהַלְווֹת לִיהוּדִי בְּרִבִּית – כְּלוֹמַר, לְבַקֵּשׁ שֶׁיַּחֲזִיר יוֹתֵר מִמָּה שֶׁקִּבֵּל. הַלְוָאָה בְּלִי רִבִּית בָּאָה לַעֲזֹר לֶחָבֵר, וְהַלְוָאָה בְּרִבִּית מְנַצֶּלֶת אֶת הַמְּצוּקָה שֶׁלּוֹ כְּדֵי לְהִתְעַשֵּׁר.
לֹא רַק הַמַּלְוֶה עוֹבֵר עַל הָאִסּוּר – גַּם הַלּוֹוֶה, הָעֵדִים וְכָל מִי שֶׁמִּשְׁתַּתֵּף בַּהַלְוָאָה הַזּוֹ.
💛 אַהֲבַת הַשֵּׁם: יָצָאנוּ מִמִּצְרַיִם וְלָמַדְנוּ שֶׁהַכֶּסֶף אֵינוֹ הָעִקָּר. מִי שֶׁמַּלְוֶה בְּלִי רֶוַח מַרְאֶה שֶׁהַחֶסֶד חָשׁוּב לוֹ יוֹתֵר מִן הָרֶוַח – כְּמוֹ שֶׁה׳ לִמֵּד אוֹתָנוּ.
📖 פֶּרֶק ה׳, הֲלָכָה ד׳

🛍️הַכֶּסֶף שֶׁל הַהַלְוָאָה

אָסוּר לַלּוֹוֶה לְבַזְבֵּז אֶת כֶּסֶף הַהַלְוָאָה עַל מוֹתָרוֹת, שֶׁמָּא לֹא יוּכַל לְהַחֲזִיר. וְגַם אִם הוּא בָּטוּחַ שֶׁיַּחֲזִיר – לֹא יִקְנֶה בּוֹ דָּבָר מְיֻתָּר בְּלִי רְשׁוּת מְפֹרֶשֶׁת מִן הַמַּלְוֶה.
לְמִי שֶׁיָּדוּעַ כְּבַזְבְּזָן אוֹ כְּמִי שֶׁאֵינוֹ מַחֲזִיר – לֹא מַלְוִים, אֶלָּא עוֹזְרִים לוֹ בִּצְדָקָה. אֲבָל לֹא חוֹשְׁדִים בְּכֻלָּם: רֹב הָאֲנָשִׁים נֶאֱמָנִים וּמַחֲזִירִים.
💛 אַהֲבַת הַשֵּׁם: הֶחָפֵץ חַיִּים כָּתַב: מִי שֶׁמְּבַקְשִׁים מִמֶּנּוּ הַלְוָאָה לֹא קִבֵּל עֹנֶשׁ – אֶלָּא זְכוּת גְּדוֹלָה. ה׳ שָׁלַח אֵלָיו הִזְדַּמְּנוּת לַעֲשׂוֹת חֶסֶד.
📖 פֶּרֶק ה׳, הֲלָכוֹת ז׳–ט׳

📝כּוֹתְבִים – אוֹ מוֹחֲלִים

גַּם אָדָם טוֹב עָלוּל לִשְׁכֹּחַ שֶׁלָּוָה. לָכֵן מַלְוִים בִּפְנֵי עֵדִים, אוֹ כּוֹתְבִים אֶת הַהַלְוָאָה, אוֹ לוֹקְחִים מַשְׁכּוֹן. וּבְהַלְוָאָה קְטַנָּה אֶפְשָׁר גַּם לְהַחְלִיט בַּלֵּב: "אִם יִשְׁכַּח – אֲנִי מוֹחֵל לוֹ בְּלֵב שָׁלֵם".
וְהַלּוֹוֶה חַיָּב לְהַחֲזִיר! עַל מִי שֶׁאֵינוֹ מַחֲזִיר נֶאֱמַר: "לֹוֶה רָשָׁע וְלֹא יְשַׁלֵּם".
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁכּוֹתְבִים אוֹ מוֹחֲלִים מֵרֹאשׁ, שׁוֹמְרִים עַל הַשָּׁלוֹם בֵּין הַחֲבֵרִים. ה׳ רוֹצֶה שֶׁהַחֶסֶד יָבִיא אַהֲבָה – וְלֹא רִיב.
📖 פֶּרֶק ה׳, הֲלָכָה ח׳

⚖️כְּשֶׁהַכְּלָלִים מִתְנַגְּשִׁים

שְׁלוֹשָׁה כְּלָלִים, לְפִי הַסֵּדֶר: א׳ – מִי שֶׁצָּרִיךְ יוֹתֵר. ב׳ – קָרוֹב מִשְׁפָּחָה. ג׳ – שָׁכֵן וּבֶן הָעִיר. כְּשֶׁהֵם מִתְנַגְּשִׁים, הַכְּלָל הָרִאשׁוֹן גּוֹבֵר עַל הַשֵּׁנִי, וְהַשֵּׁנִי עַל הַשְּׁלִישִׁי. וּבִקְנִיָּה וּבִמְכִירָה – מַקְדִּימִים יְהוּדִי לְגוֹי, גַּם אִם הַגּוֹי מַצִּיעַ יוֹתֵר.
בְּדֶרֶךְ כְּלָל עָדִיף לְהַלְווֹת לְכַמָּה אֲנָשִׁים סְכוּמִים קְטַנִּים מֵאֲשֶׁר לְאֶחָד סְכוּם גָּדוֹל, כִּי בְּכָל הַלְוָאָה יֵשׁ מִצְוָה. אֲבָל מִי שֶׁעָלוּל לְהִתְמוֹטֵט בְּלִי הַהַלְוָאָה הַגְּדוֹלָה – קוֹדֵם לְכֻלָּם.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה מְלַמֶּדֶת אוֹתָנוּ לְסַדֵּר אֶת הַחֶסֶד בְּחָכְמָה. כָּךְ הָעֶזְרָה מַגִּיעָה לְמִי שֶׁהֲכִי צָרִיךְ – וְאִישׁ אֵינוֹ נִשְׁכָּח.
📖 פֶּרֶק ה׳, הֲלָכוֹת י׳, י״ב–י״ג

⚖️מִי קוֹדֵם?

כְּשֶׁאִי אֶפְשָׁר לַעֲזֹר לְכֻלָּם, יֵשׁ סֵדֶר: מִי שֶׁצָּרִיךְ יוֹתֵר – קוֹדֵם. לָכֵן עָנִי קוֹדֵם לְעָשִׁיר, וּמִי שֶׁצָּרִיךְ אֹכֶל קוֹדֵם לְמִי שֶׁצָּרִיךְ בְּגָדִים. אַחַר כָּךְ קְרוֹבֵי מִשְׁפָּחָה, וְאַחֲרֵיהֶם הַשְּׁכֵנִים וּבְנֵי הָעִיר.
"עֲנִיֵּי עִירְךָ קוֹדְמִים" לַעֲנִיֵּי עִיר אַחֶרֶת, וַעֲנִיֵּי אֶרֶץ יִשְׂרָאֵל קוֹדְמִים לַעֲנִיֵּי חוּץ לָאָרֶץ. וְגַם חָבֵר טוֹב נֶחְשָׁב כְּמוֹ קָרוֹב.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁכָּל אֶחָד דּוֹאֵג קֹדֶם לַקְּרוֹבִים אֵלָיו, אַף עָנִי לֹא נִשְׁאָר לְבַד. כָּךְ, צַעַד אַחַר צַעַד, כָּל עַם יִשְׂרָאֵל נַעֲשֶׂה מִשְׁפָּחָה אַחַת.
📖 פֶּרֶק ה׳, הֲלָכוֹת י״א, י״ג; פֶּרֶק ו׳, הֲלָכָה ה׳

🏦קֻפַּת גְּמִילוּת חֶסֶד

טוֹב לְהַפְרִישׁ בְּכָל חֹדֶשׁ סְכוּם קָבוּעַ לְהַלְוָאוֹת, וְכָךְ כְּשֶׁיְּבַקְשׁוּ מִמְּךָ – הַכֶּסֶף מוּכָן. רַבִּים מִתְאַרְגְּנִים יַחַד וּמְקִימִים קֻפַּת גְּמִילוּת חֶסֶד, וְכָל חֲבוּרָה בּוֹחֶרֶת לְמִי הִיא עוֹזֶרֶת.
מִי שֶׁלּוֹקֵחַ לְעַצְמוֹ מִן הַקֻּפָּה הַזּוֹ – חַיָּב לְהַחֲזִיר הַכֹּל, כִּי זֶה כֶּסֶף שֶׁל מִצְוָה. וְהֶחָפֵץ חַיִּים כָּתַב שֶׁרָאוּי שֶׁבְּכָל עִיר, וְאַף בְּכָל שְׁכוּנָה, תִּהְיֶה קֻפָּה כָּזוֹ.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁכֻּלָּם נוֹתְנִים מְעַט יַחַד, נוֹצֶרֶת קֻפָּה גְּדוֹלָה שֶׁמַּצִּילָה מִשְׁפָּחוֹת. זוֹ אַחְדוּת שֶׁל עַם שֶׁאוֹהֵב חֶסֶד, כְּמוֹ אַבְרָהָם אָבִינוּ.
📖 פֶּרֶק ה׳, הֲלָכָה י״ד

🧠 בְּדֹק אֶת עַצְמְךָ

רוֹנִי לָוָה מֵחֲבֵרוֹ כֶּסֶף כְּדֵי לִקְנוֹת סְפָרִים לְבֵית הַסֵּפֶר. בַּחֲנוּת הוּא רָאָה מִשְׂחָק יָקָר. מָה הַדִּין?
💡 גַּם מִי שֶׁבָּטוּחַ שֶׁיַּחֲזִיר – לֹא יִקְנֶה בְּכֶסֶף הַהַלְוָאָה מוֹתָרוֹת בְּלִי רְשׁוּת מְפֹרֶשֶׁת. הַמַּלְוֶה נִכְנַס לְסִכּוּן כְּדֵי לַעֲזֹר בְּצֹרֶךְ אֲמִתִּי, וְלֹא כְּדֵי לְמַמֵּן דְּבָרִים מְיֻתָּרִים.
לְאַבָּא יֵשׁ כֶּסֶף לְהַלְוָאָה אַחַת. מְבַקְשִׁים מִמֶּנּוּ: דּוֹד שֶׁגָּר בַּשְּׁכוּנָה וְאֵינוֹ עָנִי, וְעָנִי זָר מֵעִיר אַחֶרֶת. לְמִי יַלְוֶה?
💡 הַכְּלָל הָרִאשׁוֹן – מִי שֶׁצָּרִיךְ יוֹתֵר – גּוֹבֵר עַל קִרְבַת מִשְׁפָּחָה וְעַל קִרְבַת מָקוֹם. מַלְוִים קֹדֶם לֶעָנִי, וְאִם יִשָּׁאֵר כֶּסֶף – לַדּוֹד.
שְׁמוּאֵל מוֹכֵר אֶת הָאוֹפַנַּיִם הַיְּשָׁנִים שֶׁלּוֹ. קוֹנֶה יְהוּדִי וְקוֹנֶה גּוֹי רוֹצִים אוֹתָם, וְהַגּוֹי מַצִּיעַ קְצָת יוֹתֵר. מָה עָדִיף?
💡 "וְכִי תִמְכְּרוּ מִמְכָּר לַעֲמִיתֶךָ" – חֲכָמִים לָמְדוּ שֶׁבִּמְכִירָה וּבִקְנִיָּה מַקְדִּימִים אֶת הַיְּהוּדִי, גַּם כְּשֶׁהַגּוֹי מַצִּיעַ יוֹתֵר.

עשרת ימי תשובה

🎺 רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים · פֶּרֶק ה׳

🔍חֶשְׁבּוֹן נֶפֶשׁ

בַּעֲשֶׂרֶת הַיָּמִים שֶׁבֵּין רֹאשׁ הַשָּׁנָה לְיוֹם הַכִּפּוּרִים ה׳ קָרוֹב לְכָל קוֹרְאָיו, וְהַתְּשׁוּבָה מִתְקַבֶּלֶת מִיָּד. לָכֵן רָאוּי לְפַשְׁפֵּשׁ בַּמַּעֲשִׂים וּלְתַקֵּן. מִנְהַג חֲסִידִים לַעֲשׂוֹת חֶשְׁבּוֹן נֶפֶשׁ בְּכָל לַיְלָה לִפְנֵי הַשֵּׁנָה, וּבַיָּמִים הָאֵלֶּה רָאוּי לְכָל אָדָם לִנְהֹג כָּךְ.
בַּיָּמִים הָאֵלֶּה נִזְהָרִים מִכָּל אִסּוּר, וּמַרְבִּים בְּתוֹרָה, בִּתְפִלָּה, בִּצְדָקָה וּבְמַעֲשִׂים טוֹבִים.
💛 אַהֲבַת הַשֵּׁם: בַּיָּמִים הָאֵלֶּה אֲפִלּוּ יָחִיד שֶׁעוֹשֶׂה תְּשׁוּבָה – ה׳ מְקַבֵּל אוֹתוֹ כְּמוֹ צִבּוּר שָׁלֵם. הוּא מְחַכֶּה לְכָל אֶחָד מֵאִתָּנוּ.
📖 פֶּרֶק ה׳, הֲלָכָה א׳

⚖️חֶצְיוֹ זַכַּאי וְחֶצְיוֹ חַיָּב

חֲכָמִים אָמְרוּ: יִרְאֶה אָדָם אֶת עַצְמוֹ כְּאִלּוּ חֶצְיוֹ זַכַּאי וְחֶצְיוֹ חַיָּב, וְגַם אֶת כָּל הָעוֹלָם. עָשָׂה מִצְוָה אַחַת – אַשְׁרָיו, שֶׁהִכְרִיעַ אֶת עַצְמוֹ וְאֶת כָּל הָעוֹלָם לְכַף זְכוּת.
יֵשׁ נוֹהֲגִים בַּיָּמִים הָאֵלֶּה לְדַקְדֵּק יוֹתֵר בַּהֲלָכָה, כְּמוֹ לֶאֱכֹל בָּשָׂר ״גְּלַאט״ וּלְהַקְפִּיד עַל ״מַיִם אַחֲרוֹנִים״.
💛 אַהֲבַת הַשֵּׁם: לְכָל מַעֲשֶׂה קָטָן שֶׁלְּךָ יֵשׁ כֹּחַ גָּדוֹל. מִצְוָה אַחַת שֶׁלְּךָ יְכוֹלָה לְהָבִיא טוֹבָה לְכָל הָעוֹלָם!
📖 פֶּרֶק ה׳, הֲלָכָה א׳

👑שִׁנּוּיִים בַּתְּפִלָּה

בַּעֲשֶׂרֶת יְמֵי תְּשׁוּבָה אוֹמְרִים בַּעֲמִידָה ״הַמֶּלֶךְ הַקָּדוֹשׁ״ וְ״הַמֶּלֶךְ הַמִּשְׁפָּט״. וּמוֹסִיפִים אַרְבַּע תּוֹסָפוֹת: ״זָכְרֵנוּ לְחַיִּים״, ״מִי כָמוֹךָ״, ״וּכְתֹב לְחַיִּים״ וּ״בְסֵפֶר חַיִּים״.
מִי שֶׁאָמַר ״הָאֵל הַקָּדוֹשׁ״ וְלֹא תִּקֵּן מִיָּד – חוֹזֵר לְרֹאשׁ הַתְּפִלָּה. אֲבָל מִי שֶׁשָּׁכַח אַחַת מֵאַרְבַּע הַתּוֹסָפוֹת – אֵינוֹ חוֹזֵר.
💛 אַהֲבַת הַשֵּׁם: הָעוֹמֵד בִּתְפִלָּה עוֹמֵד לִפְנֵי הַמֶּלֶךְ. בַּיָּמִים הָאֵלֶּה, כְּשֶׁה׳ דָּן אֶת עוֹלָמוֹ, אֲנַחְנוּ מַרְגִּישִׁים אֶת מַלְכוּתוֹ קְרוֹבָה מְאוֹד.
📖 פֶּרֶק ה׳, הֲלָכָה ב׳

🕊️״שׁוּבָה יִשְׂרָאֵל״

הַהַפְטָרָה שֶׁל שַׁבַּת שׁוּבָה פּוֹתַחַת בַּפָּסוּק: ״שׁוּבָה יִשְׂרָאֵל עַד ה׳ אֱלֹהֶיךָ, כִּי כָשַׁלְתָּ בַּעֲוֹנֶךָ״. בַּיָּמִים הָאֵלֶּה מוֹסִיפִים בַּעֲמִידָה ״זָכְרֵנוּ לְחַיִּים״, וּמַרְבִּים בִּצְדָקָה – ״וּצְדָקָה תַּצִּיל מִמָּוֶת״.
״עֲבֵרוֹת שֶׁבֵּין אָדָם לַחֲבֵרוֹ – אֵין יוֹם הַכִּפּוּרִים מְכַפֵּר, עַד שֶׁיְּרַצֶּה אֶת חֲבֵרוֹ״.
💛 אַהֲבַת הַשֵּׁם: הַנָּבִיא לֹא אוֹמֵר ״שׁוּבָה אֶל ה׳״ אֶלָּא ״עַד ה׳״ – כְּאִלּוּ ה׳ מְחַכֶּה לָנוּ קָרוֹב מְאוֹד, וְצָרִיךְ רַק לְהַגִּיעַ עָדָיו.
📖 פֶּרֶק ה׳, הֲלָכוֹת ב׳–ד׳, ז׳

📚תּוֹרָה וּצְדָקָה

לִמּוּד תּוֹרָה הוּא יְסוֹד גָּדוֹל בִּתְשׁוּבָה: ״הַתַּלְמוּד מֵבִיא לִידֵי מַעֲשֶׂה״. רָאוּי לִלְמֹד בְּכָל יוֹם מְעַט בְּסֵפֶר מוּסָר, וּבְמִיּוּחָד בֶּאֱלוּל וּבַעֲשֶׂרֶת יְמֵי תְּשׁוּבָה. וְנָהֲגוּ יִשְׂרָאֵל לְהַרְבּוֹת בַּיָּמִים הָאֵלֶּה בִּצְדָקָה: ״וּצְדָקָה תַּצִּיל מִמָּוֶת״.
מִי שֶׁהָיָה רָגִיל לִלְמֹד דַּף אֶחָד – יִלְמַד שְׁנַיִם. וּמִי שֶׁאֵינוֹ יוֹדֵעַ לִלְמֹד – יַעֲסֹק בִּצְדָקָה.
💛 אַהֲבַת הַשֵּׁם: הַלִּמּוּד מְחַבֵּר אוֹתָנוּ לָרָצוֹן שֶׁל ה׳, וְהַצְּדָקָה מְלַמֶּדֶת אוֹתָנוּ לִהְיוֹת טוֹבִים כָּמוֹהוּ.
📖 פֶּרֶק ה׳, הֲלָכָה ג׳

🤝סְלִיחָה מֵחָבֵר

״עֲבֵרוֹת שֶׁבֵּין אָדָם לַמָּקוֹם – יוֹם הַכִּפּוּרִים מְכַפֵּר. עֲבֵרוֹת שֶׁבֵּין אָדָם לַחֲבֵרוֹ – אֵין יוֹם הַכִּפּוּרִים מְכַפֵּר, עַד שֶׁיְּרַצֶּה אֶת חֲבֵרוֹ״. מִי שֶׁפָּגַע בְּחָבֵר – יְפַיֵּס אוֹתוֹ מַהֵר, וּלְפָחוֹת לִפְנֵי יוֹם הַכִּפּוּרִים.
קֹדֶם מְבַקְּשִׁים סְלִיחָה מִן הֶחָבֵר וּמְשַׁלְּמִים לוֹ אִם הִזִּיקוּ, וְרַק אַחַר כָּךְ מִתְוַדִּים לִפְנֵי ה׳.
💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אֶת כָּל אֶחָד מִבָּנָיו. מִי שֶׁפּוֹגֵעַ בְּאָח – צָרִיךְ קֹדֶם לְהַשְׁלִים אִתּוֹ, וְרַק אָז הוּא חוֹזֵר בֶּאֱמֶת אֶל הָאַבָּא.
📖 פֶּרֶק ה׳, הֲלָכָה ד׳

📢לְתַקֵּן לָשׁוֹן הָרַע

מִי שֶׁדִּבֵּר בִּגְנוּת חֲבֵרוֹ – צָרִיךְ לְדַבֵּר בְּשִׁבְחוֹ לִפְנֵי כָּל מִי שֶׁשָּׁמַע אֶת הַגְּנוּת, וּלְהַסְבִּיר שֶׁלֹּא הִכִּיר אֶת כָּל הַתְּמוּנָה. וּמִי שֶׁהֶעֱלִיב חָבֵר בִּפְנֵי רַבִּים – נָכוֹן שֶׁיְּבַקֵּשׁ מִמֶּנּוּ סְלִיחָה בִּפְנֵי רַבִּים.
מִי שֶׁפָּגַע בְּרַבּוֹ – הוֹלֵךְ אֵלָיו אֲפִלּוּ אֶלֶף פְּעָמִים, עַד שֶׁיִּמְחַל לוֹ.
💛 אַהֲבַת הַשֵּׁם: הַמִּלִּים שֶׁלָּנוּ חֲזָקוֹת. כְּמוֹ שֶׁהֵן יְכוֹלוֹת לִפְגֹּעַ, הֵן יְכוֹלוֹת גַּם לְרַפֵּא.
📖 פֶּרֶק ה׳, הֲלָכוֹת ד׳–ה׳

💛לִסְלֹחַ בְּלֵב שָׁלֵם

רָאוּי לַנִּפְגָּע שֶׁלֹּא יִהְיֶה אַכְזָרִי, וְיִסְלַח לְמִי שֶׁמְּבַקֵּשׁ מִמֶּנּוּ סְלִיחָה. חֲכָמִים אָמְרוּ: ״כָּל הַמַּעֲבִיר עַל מִדּוֹתָיו – מַעֲבִירִים לוֹ עַל כָּל פְּשָׁעָיו״.
אִם הֶחָבֵר לֹא סָלַח, הַפּוֹגֵעַ חוֹזֵר אֵלָיו עִם שְׁלֹשָׁה אֲנָשִׁים, עַד שָׁלֹשׁ פְּעָמִים. אִם גַּם אָז לֹא סָלַח – הוּא הַחוֹטֵא.
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁמּוֹחֵל לַאֲחֵרִים – גַּם בַּשָּׁמַיִם מוֹחֲלִים לוֹ. סְלִיחָה לְחָבֵר הִיא מַתָּנָה שֶׁחוֹזֶרֶת אֵלֵינוּ.
📖 פֶּרֶק ה׳, הֲלָכָה ה׳

🗣️אֵיךְ מְבַקְּשִׁים סְלִיחָה

הַמְבַקֵּשׁ סְלִיחָה מְפָרֵט אֶת הַחֵטְא. אֲבָל אִם הַפֵּרוּט יְצַעֵר וִיבַיֵּשׁ אֶת הֶחָבֵר – עָדִיף שֶׁלֹּא לְפָרֵט. לְכַתְּחִלָּה מְבַקְּשִׁים סְלִיחָה בְּעַצְמֵנוּ, וּמִי שֶׁמִּתְבַּיֵּשׁ – יָכוֹל לִשְׁלֹחַ שָׁלִיחַ.
יֵשׁ שֶׁמְּבַקְּשִׁים לִפְנֵי יוֹם הַכִּפּוּרִים סְלִיחָה כְּלָלִית מִכָּל מַכָּרֵיהֶם – וְזֶה כִּמְעַט לֹא מוֹעִיל. הָעִקָּר: לַעֲשׂוֹת חֶשְׁבּוֹן נֶפֶשׁ, לְהִזָּכֵר בְּמִי בֶּאֱמֶת פָּגַעְנוּ – וּמִמֶּנּוּ לְבַקֵּשׁ סְלִיחָה.
💛 אַהֲבַת הַשֵּׁם: סְלִיחָה אֲמִתִּית יוֹצֵאת מִן הַלֵּב וּמְכֻוֶּנֶת לְאָדָם אֶחָד מְסֻיָּם. כָּךְ גַּם ה׳ אוֹהֵב כָּל אֶחָד בְּאֹפֶן אִישִׁי.
📖 פֶּרֶק ה׳, הֲלָכָה ה׳

🙏אָבִינוּ מַלְכֵּנוּ

בַּעֲשֶׂרֶת יְמֵי תְּשׁוּבָה אוֹמְרִים ״אָבִינוּ מַלְכֵּנוּ״ בְּשַׁחֲרִית וּבְמִנְחָה, אַחֲרֵי חֲזָרַת הַשַּׁ״ץ. אוֹמְרִים אוֹתָהּ בַּעֲמִידָה, וְרַבִּים פּוֹתְחִים אֶת אֲרוֹן הַקֹּדֶשׁ.
פַּעַם לֹא יָרְדוּ גְּשָׁמִים. רַק כְּשֶׁיָּרַד רַבִּי עֲקִיבָא לִפְנֵי הַתֵּבָה וְאָמַר: ״אָבִינוּ מַלְכֵּנוּ, אֵין לָנוּ מֶלֶךְ אֶלָּא אַתָּה; אָבִינוּ מַלְכֵּנוּ, לְמַעַנְךָ רַחֵם עָלֵינוּ״ – יָרְדוּ גְּשָׁמִים.
💛 אַהֲבַת הַשֵּׁם: אֲנַחְנוּ פּוֹנִים אֶל ה׳ בִּשְׁתֵּי דְּרָכִים יַחַד: כְּבָנִים אֶל אֲבִיהֶם, וְכַעֲבָדִים אֶל מַלְכָּם.
📖 פֶּרֶק ה׳, הֲלָכָה ו׳

🕯️שַׁבַּת שׁוּבָה

הַשַּׁבָּת שֶׁבְּתוֹךְ עֲשֶׂרֶת יְמֵי תְּשׁוּבָה נִקְרֵאת ״שַׁבַּת שׁוּבָה״, עַל שֵׁם הַהַפְטָרָה: ״שׁוּבָה יִשְׂרָאֵל עַד ה׳ אֱלֹהֶיךָ״. בַּשַּׁבָּת הַזֹּאת רַב הַמָּקוֹם דּוֹרֵשׁ וּמְעוֹרֵר אֶת הָעָם לִתְשׁוּבָה.

💡 צָרִיךְ לְבַטֵּל אֶת הַהַשְׁפָּעָה הָרָעָה שֶׁל הַדְּבָרִים – לְסַפֵּר בְּשִׁבְחוֹ לִפְנֵי מִי שֶׁשָּׁמְעוּ. וְאֶת הַוִּדּוּי לִפְנֵי ה׳ אוֹמְרִים רַק אַחֲרֵי שֶׁמְּפַיְּסִים אֶת הֶחָבֵר.
יְהוּדָה בִּקֵּשׁ סְלִיחָה מֵחֲבֵרוֹ שָׁלֹשׁ פְּעָמִים, בְּכָל פַּעַם עִם שְׁלֹשָׁה אֲנָשִׁים – וְהֶחָבֵר לֹא סָלַח. מָה הַדִּין?
💡 אַחֲרֵי שָׁלֹשׁ פְּעָמִים עִם שְׁלֹשָׁה אֲנָשִׁים – יְהוּדָה כְּבָר אֵינוֹ צָרִיךְ לְהִתְאַמֵּץ, וּמִי שֶׁלֹּא מָחַל הוּא הַחוֹטֵא. רַק מִי שֶׁפָּגַע בְּרַבּוֹ הוֹלֵךְ אֲפִלּוּ אֶלֶף פְּעָמִים.

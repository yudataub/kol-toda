שְׁרָצִים

🍽️ הִלְכוֹת כַּשְׁרוּת · פֶּרֶק כ״ג

🐛אִסּוּר שְׁרָצִים

הַתּוֹרָה אָסְרָה לֶאֱכֹל שְׁרָצִים: שִׁרְצֵי הַמַּיִם, שִׁרְצֵי הָאָרֶץ – כְּמוֹ נְמָלִים וְתוֹלָעִים – וְשִׁרְצֵי הָעוֹף, כְּמוֹ זְבוּבִים וְיַתּוּשִׁים. הָאִסּוּר חָמוּר מְאוֹד: הָאוֹכֵל אֲפִלּוּ חֶרֶק קָטָן אֶחָד, שָׁלֵם – עוֹבֵר עַל אִסּוּר תּוֹרָה.
מִן הָאִסּוּר הַזֶּה לָמַדְנוּ גַּם "אַל תְּשַׁקְּצוּ": אָסוּר לֶאֱכֹל מַאֲכָלִים מְאוּסִים, אוֹ לֶאֱכֹל בְּדֶרֶךְ מַגְעִילָה.
💛 אַהֲבַת הַשֵּׁם: אָמַר ה׳: אִלּוּ הֶעֱלֵיתִי אֶת יִשְׂרָאֵל מִמִּצְרַיִם רַק כְּדֵי שֶׁלֹּא יִטַּמְּאוּ בִּשְׁרָצִים – דַּיִּי. הַהִזָּהֲרוּת מִשְּׁרָצִים הִיא חֵלֶק מִן הַחֵרוּת וְהַקְּדֻשָּׁה שֶׁל עַם יִשְׂרָאֵל.
📖 פֶּרֶק כ״ג, הֲלָכוֹת א׳–ב׳

🍲בְּרִיָּה אֵינָהּ בְּטֵלָה

מְעַט אִסּוּר שֶׁהִתְעָרֵב בְּפִי שִׁשִּׁים הֶתֵּר – בְּדֶרֶךְ כְּלָל בָּטֵל. אֲבָל חֶרֶק שָׁלֵם הוּא "בְּרִיָּה", וְאֵינוֹ בָּטֵל אֲפִלּוּ בְּפִי אֶלֶף. זְבוּב שֶׁנָּפַל לְסִיר תַּבְשִׁיל וְלֹא מָצְאוּ אוֹתוֹ – הַתַּבְשִׁיל אָסוּר.
מָרָק צָלוּל – אֶפְשָׁר לְסַנֵּן בְּמִסְנֶנֶת, וְהַמָּרָק הַמְסֻנָּן מֻתָּר. חֲתִיכוֹת בָּשָׂר וְתַפּוּחֵי אֲדָמָה – שׁוֹטְפִים, וְהֵן מֻתָּרוֹת. חֶרֶק שֶׁנִּמְעַךְ אוֹ שֶׁחָסֵר מִמֶּנּוּ חֵלֶק – בָּטֵל בְּשִׁשִּׁים.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה מַחְשִׁיבָה אֲפִלּוּ זְבוּב קָטָן כִּבְרִיָּה שְׁלֵמָה. כָּל בְּרִיָּה שֶׁבָּרָא ה׳ חֲשׁוּבָה – וְכָל שֶׁכֵּן כָּל יְהוּדִי וִיהוּדִי.
📖 פֶּרֶק כ״ג, הֲלָכוֹת א׳, ד׳

✨קְדוֹשִׁים כִּי קָדוֹשׁ אֲנִי

שְׁלֹשָׁה סוּגֵי שְׁרָצִים יֵשׁ: שֶׁל הַמַּיִם, שֶׁל הָאָרֶץ וְשֶׁל הָעוֹף. הַתּוֹרָה חָזְרָה עַל הָאִסּוּר פְּעָמִים רַבּוֹת, וְהוֹסִיפָה: "אַל תְּשַׁקְּצוּ אֶת נַפְשֹׁתֵיכֶם" – אַל תַּעֲשׂוּ אֶת נַפְשְׁכֶם מְאוּסָה.
חֶרֶק שָׁלֵם הוּא בְּרִיָּה – וּבְרִיָּה אֵינָהּ בְּטֵלָה, אֲפִלּוּ בְּפִי אֶלֶף.
💛 אַהֲבַת הַשֵּׁם: מַאֲכָל אָסוּר מְטַמְטֵם אֶת הַלֵּב, וּשְׁמִירָה מִמֶּנּוּ פּוֹתַחַת אוֹתוֹ. מִי שֶׁנִּזְהָר בַּמַּאֲכָלִים, לִבּוֹ פָּתוּחַ לַתּוֹרָה וְלַקְּדֻשָּׁה.
📖 פֶּרֶק כ״ג, הֲלָכוֹת א׳–ב׳, ד׳

🍎חֶרֶק בְּתוֹךְ פְּרִי

חֶרֶק שֶׁגָּדַל בְּתוֹךְ פְּרִי קָטוּף, אוֹ בְּתוֹךְ קֶמַח, וְלֹא יָצָא מִמֶּנּוּ – אֵין עָלָיו אִסּוּר שֶׁרֶץ, כִּי הוּא נֶחְשָׁב חֵלֶק מִן הַפְּרִי. מִן הָרֶגַע שֶׁיָּצָא הַחוּצָה – הוּא נֶאֱסָר.
וּבְכָל זֹאת, כְּשֶׁרוֹאִים אוֹתוֹ וְהוּא מְעוֹרֵר גֹּעַל – לֹא אוֹכְלִים אוֹתוֹ, מִשּׁוּם "אַל תְּשַׁקְּצוּ".
💛 אַהֲבַת הַשֵּׁם: הַהֲלָכָה מְדַיֶּקֶת מְאוֹד, וְאֵינָהּ אוֹסֶרֶת יוֹתֵר מִן הַצֹּרֶךְ. מִי שֶׁלּוֹמֵד אֶת הַפְּרָטִים, שׁוֹמֵר מִצְווֹת בְּשִׂמְחָה וּבְלִי פַּחַד.
📖 פֶּרֶק כ״ג, הֲלָכוֹת ב׳–ג׳, ט״ו

🔎מָתַי חַיָּבִים לִבְדֹּק?

מַאֲכָל שֶׁבְּדֶרֶךְ כְּלָל יֵשׁ בּוֹ חֲרָקִים – חוֹבָה לִבְדֹּק. מַאֲכָל שֶׁרַק לִפְעָמִים יֵשׁ בּוֹ – צָרִיךְ בְּדִיקָה, וּבְדִיעֲבַד הוֹלְכִים אַחַר הָרֹב. וּמַאֲכָל שֶׁכִּמְעַט אַף פַּעַם אֵין בּוֹ – אוֹכְלִים בְּלִי בְּדִיקָה, אֶלָּא אִם רוֹאִים רָקָב, קוּרִים אוֹ חוֹרִים.
אִם בַּאֲרִיזָה שֶׁל מַאֲכָל שֶׁנֶּחְשָׁב נָקִי נִמְצְאוּ שְׁלֹשָׁה חֲרָקִים – כָּל הָאֲרִיזָה צְרִיכָה בְּדִיקָה.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה סוֹמֶכֶת עַל הָאָדָם שֶׁיַּעֲרִיךְ בְּיֹשֶׁר אֶת הַמַּצָּב שֶׁלְּפָנָיו. זוֹ אַחְרָיוּת – וְגַם אֵמוּן גָּדוֹל שֶׁה׳ נוֹתֵן בָּנוּ.
📖 פֶּרֶק כ״ג, הֲלָכָה ה׳

👁️חֲרָקִים זְעִירִים

הַכֹּל מַסְכִּימִים: חֲרָקִים זְעִירִים שֶׁעֵין הָאָדָם אֵינָהּ יְכוֹלָה לִרְאוֹת – אֵין צֹרֶךְ לְחַפֵּשׂ, כִּי הַתּוֹרָה נִתְּנָה לְפִי רְאִיַּת הָאָדָם. הַמַּחֲלֹקֶת הִיא עַל חֲרָקִים קְטַנִּים מְאוֹד שֶׁקָּשֶׁה לִרְאוֹת, כְּמוֹ בְּתוּת שָׂדֶה, בִּבְרוֹקוֹלִי וּבִכְרוּבִית. לְפִי עִקַּר הַהֲלָכָה – מָה שֶׁאָדָם רָגִיל אֵינוֹ רוֹאֶה, אֵין צֹרֶךְ לְחַפֵּשׂ.
דֶּרֶךְ הָאֶמְצַע: בַּתּוּת חוֹתְכִים אֶת הַחֵלֶק הַיָּרֹק, וְאֶת הַתּוּת, הַבְּרוֹקוֹלִי וְהַכְּרוּבִית מַשְׁרִים בְּמַיִם עִם חֹמֶר מְמַסֵּס וְשׁוֹטְפִים הֵיטֵב. הַמַּחְמִירִים מְקַלְּפִים, זוֹרְקִים אֶת הַתִּפְרַחַת אוֹ קוֹנִים מִגִּדּוּל מְיֻחָד.
💛 אַהֲבַת הַשֵּׁם: יֵשׁ מְהַדְּרִים, וְיֵשׁ נוֹהֲגִים לְפִי עִקַּר הַדִּין – וְכֻלָּם מִתְכַּוְּנִים לַעֲשׂוֹת אֶת רְצוֹן ה׳. אָנוּ מְכַבְּדִים כָּל מִשְׁפָּחָה וְאֶת רַבּוֹתֶיהָ, וְאֵין מְזַלְזְלִים בְּאַף אֶחָד.
📖 פֶּרֶק כ״ג, הֲלָכוֹת ו׳–ח׳, י״א–י״ב

🥬יְרָקוֹת עָלִים

בְּחַסָּה, כְּרוּב, פֶּטְרוֹזִילְיָה וְנַעְנַע יְכוֹלִים לְהִסְתַּתֵּר חֲרָקִים. לְפִי עִקַּר הַהֲלָכָה, כְּדַעַת הָרַב מֶלָמֵד, מַסְפִּיק לִשְׁטֹף הֵיטֵב וּלְהִסְתַּכֵּל בְּמַבָּט רָגִיל, וְאִם רוֹאִים חֶרֶק – מְסִירִים אוֹתוֹ. כְּדֵי שֶׁהַמַּיִם יַגִּיעוּ לְכָל מָקוֹם, מְפָרְקִים אֶת הֶעָלִים.
דֶּרֶךְ הָאֶמְצַע שֶׁהָרַב מֶלָמֵד מַמְלִיץ עָלֶיהָ: מַשְׁרִים אֶת הֶעָלִים כְּאַרְבַּע דַּקּוֹת בְּמַיִם עִם מֶלַח, חֹמֶץ אוֹ סַבּוֹן, וְשׁוֹטְפִים בְּמַיִם זוֹרְמִים. וְרַבָּנִים רַבִּים בַּדּוֹרוֹת הָאַחֲרוֹנִים מוֹרִים לְהַחְמִיר: לִבְדֹּק כָּל עָלֶה מוּל הָאוֹר, אוֹ לִקְנוֹת יְרָקוֹת מִגִּדּוּל מְיֻחָד.
💛 אַהֲבַת הַשֵּׁם: הַבְּדִיקָה הִיא מִצְוָה שֶׁאֶפְשָׁר לְקַיֵּם כָּל יוֹם בַּמִּטְבָּח. כְּשֶׁאָנוּ שׁוֹטְפִים חַסָּה בִּתְשׂוּמַת לֵב, אָנוּ שׁוֹמְרִים עַל הַקְּדֻשָּׁה שֶׁל גּוּפֵנוּ וְנַפְשֵׁנוּ.
📖 פֶּרֶק כ״ג, הֲלָכוֹת ח׳–י׳

🔀יְרָקוֹת עָלִים – שְׁתֵּי שִׁיטוֹת, וּשְׁתֵּיהֶן כַּהֲלָכָה

בְּחַסָּה, בִּכְרוּב וּבְפֶטְרוֹזִילְיָה מִסְתַּתְּרִים לִפְעָמִים חֲרָקִים זְעִירִים, שֶׁצִּבְעָם כְּצֶבַע הֶעָלֶה. לְפִי עִקַּר הַהֲלָכָה, כְּדַעַת הָרַב מֶלָמֵד – שׁוֹטְפִים הֵיטֵב, וּמַה שֶּׁאָדָם רָגִיל אֵינוֹ רוֹאֶה – אֵין צֹרֶךְ לְחַפֵּשׂ. וְטוֹב לְהַשְׁרוֹת אֶת הֶעָלִים בְּמַיִם עִם מֶלַח אוֹ חֹמֶץ.
רַבָּנִים רַבִּים בַּדּוֹרוֹת הָאַחֲרוֹנִים מוֹרִים לְהַחְמִיר: בּוֹדְקִים כָּל עָלֶה מוּל אוֹר הַשֶּׁמֶשׁ אוֹ מוּל שֻׁלְחַן אוֹר, אוֹ קוֹנִים יְרָקוֹת שֶׁגָּדְלוּ בַּחֲמָמוֹת מְיֻחָדוֹת. כָּל בַּיִת נוֹהֵג כְּפִי שֶׁהוֹרוּ לוֹ רַבּוֹתָיו.
💛 אַהֲבַת הַשֵּׁם: שְׁתֵּי הַשִּׁיטוֹת רוֹצוֹת אוֹתוֹ דָּבָר: לֹא לְהִכָּשֵׁל אֲפִלּוּ בְּחֶרֶק קָטָן אֶחָד, מִתּוֹךְ אַהֲבָה לַמִּצְווֹת שֶׁל ה׳.
📖 פֶּרֶק כ״ג, הֲלָכוֹת ח׳–י׳

🌴תְּמָרִים וּפֵרוֹת יְבֵשִׁים

יֵשׁ פֵּרוֹת שֶׁלִּפְעָמִים יֵשׁ בָּהֶם חֲרָקִים, כְּמוֹ תְּאֵנִים, תְּמָרִים, גּוֹיָאבוֹת וּמִשְׁמְשִׁים – פּוֹתְחִים אוֹתָם וּבוֹדְקִים לִפְנֵי שֶׁאוֹכְלִים. וּבְכָל פְּרִי שֶׁיֵּשׁ בּוֹ רָקָב – בּוֹדְקִים סְבִיב הָרָקָב.
צִמּוּקִים וּפֵרוֹת יְבֵשִׁים שֶׁיֵּשׁ בָּאֲרִיזָה שֶׁלָּהֶם קוּרִים אוֹ אֲבָקָה – כַּנִּרְאֶה יֵשׁ בָּהֶם חֲרָקִים. בּוֹדְקִים הֵיטֵב, וְלִפְעָמִים עָדִיף לִזְרֹק, כִּי הַבְּדִיקָה קָשָׁה.
💛 אַהֲבַת הַשֵּׁם: רֶגַע קָטָן שֶׁל בְּדִיקָה לִפְנֵי הָאֲכִילָה הוּא רֶגַע שֶׁל זִכָּרוֹן לַה׳. כָּךְ גַּם תָּמָר מָתוֹק נַעֲשֶׂה מִצְוָה.
📖 פֶּרֶק כ״ג, הֲלָכָה י״ג

🌾אֹרֶז, קִטְנִיּוֹת וְקֶמַח

אֹרֶז, קִטְנִיּוֹת וְקֶמַח בַּאֲרִיזָה סְגוּרָה שֶׁל חֶבְרָה אֲמִינָה, מֵחֲנוּת נְקִיָּה – אֵין צֹרֶךְ לִבְדֹּק. אֲבָל אִם רוֹאִים בָּאֲרִיזָה קוּרִים, פֵּרוּרִים אוֹ אֲבָקָה, אוֹ שֶׁקָּנוּ בְּתִפְזֹרֶת – בּוֹדְקִים עַל מִשְׁטָח.
קֶמַח שֶׁעָמַד פָּתוּחַ יְמָמָה – וּבְיוֹם חַם אֲפִלּוּ כַּמָּה שָׁעוֹת – צָרִיךְ לְנַפּוֹת בְּנָפָה. מִי שֶׁרוֹצֶה שֶׁהַקֶּמַח לֹא יַתְלִיעַ, שׁוֹמֵר אוֹתוֹ בַּמְּקָרֵר.
💛 אַהֲבַת הַשֵּׁם: בְּחַסְדֵי ה׳, הַקֶּמַח בִּימֵינוּ נָקִי הַרְבֵּה יוֹתֵר מִבֶּעָבָר. מִי שֶׁעוֹזֵר בַּאֲפִיַּת חַלּוֹת לְשַׁבָּת וְשׁוֹמֵר עַל הַקֶּמַח, מְכַבֵּד גַּם אֶת הַשַּׁבָּת וְגַם אֶת הַכַּשְׁרוּת.
📖 פֶּרֶק כ״ג, הֲלָכוֹת י״ד–ט״ז

🧠 בְּדֹק אֶת עַצְמְךָ

זְבוּב נָפַל לְסִיר גָּדוֹל שֶׁל מְרַק עוֹף צָלוּל, וְלֹא מָצְאוּ אוֹתוֹ. מָה הַדִּין?
💡 זְבוּב שָׁלֵם הוּא בְּרִיָּה, וְאֵינוֹ בָּטֵל. אֲבָל מָרָק צָלוּל אֶפְשָׁר לְסַנֵּן בְּמִסְנֶנֶת שֶׁזְּבוּב אֵינוֹ עוֹבֵר בָּהּ, וְהַמָּרָק מֻתָּר.
אֵילוּ חֲרָקִים הַכֹּל מַסְכִּימִים שֶׁאֵין צֹרֶךְ לְחַפֵּשׂ אוֹתָם?
💡 הַתּוֹרָה נִתְּנָה לְפִי רְאִיַּת הָאָדָם, וּמָה שֶׁאִי אֶפְשָׁר לִרְאוֹת – אֵין צֹרֶךְ לְחַפֵּשׂ. עַל חֲרָקִים שֶׁקָּשֶׁה מְאוֹד לִרְאוֹת יֵשׁ מַחֲלֹקֶת, וְרַבָּנִים רַבִּים מוֹרִים לְהַחְמִיר וְלִבְדֹּק גַּם אוֹתָם.
בְּקֻפְסַת פֵּרוֹת, שֶׁבְּדֶרֶךְ כְּלָל אֵין בָּהֶם חֲרָקִים, מָצְאוּ שְׁלֹשָׁה חֲרָקִים. מָה עַכְשָׁו?
💡 מַאֲכָל שֶׁנֶּחְשָׁב נָקִי, וְנִמְצְאוּ בּוֹ שְׁלֹשָׁה חֲרָקִים – כָּל מָה שֶׁבְּאוֹתָהּ אֲרִיזָה נֶחְשָׁב עַכְשָׁו כְּמַאֲכָל שֶׁיֵּשׁ בּוֹ חֲרָקִים, וְאָסוּר לְאָכְלוֹ בְּלִי בְּדִיקָה.

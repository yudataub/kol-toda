הַשְׁכָּמַת הַבֹּקֶר

🌸 תְּפִלַּת בָּנוֹת · פֶּרֶק ד׳

🦁🔑 מוּשָּׂג: לָקוּם כָּארי

חֲכָמִים אָמְרוּ: ״יִתְגַּבֵּר כָּאֲרִי לַעֲמוֹד בַּבֹּקֶר לַעֲבוֹדַת בּוֹרְאוֹ, שֶׁיְּהֵא הוּא מְעוֹרֵר הַשַּׁחַר״.
אַרְיֵה הוּא סֵמֶל לְמִי שֶׁמַּכִּיר בְּעֶרְכּוֹ וּמִתְמוֹדֵד בִּגְבוּרָה עִם מַכְשוֹלִים. אִשָּׁה שֶׁיֵּשׁ לָהּ יֵעוּד בְּחַיֶּיהָ קָמָה בַּבֹּקֶר בְּחֵשֶׁק וּבִזְרִיזוּת.
💛 אַהֲבַת הַשֵּׁם: הָאֱמוּנָה נוֹתֶנֶת תַּכְלִית לַחַיִּים: אִם ה׳ בָּחַר לָתֵת לָנוּ חַיִּים, יֵשׁ לָהֶם עֵרֶךְ גָּדוֹל, וְשָׁוֶה לָקוּם בְּשִׂמְחָה לִקְרַאת יוֹם חָדָש שֶׁל עֲבוֹדָתוֹ.
📖 פֶּרֶק ד׳, הֲלָכָה א׳

🙏🔑 מוּשָּׂג: מוֹדָה אֲנִי

כָּתְבוּ הָאַחֲרוֹנִים, שֶׁטוֹב לוֹמַר מִיָּד אַחַר הַקִּימָה מִן הַשֵּׁנָה: ״מוֹדָה אֲנִי לְפָנֶיךָ מֶלֶךְ חַי וְקַיָּם שֶׁהֶחֱזַרְתָּ בִּי נִשְׁמָתִי בְּחֶמְלָה, רַבָּה אֱמוּנָתֶךָ״.
הָאֱמוּנָה נוֹתֶנֶת לָאָדָם תַּכְלִית בְּחַיָּיו: אִם הַקָּדוֹשׁ בָּרוּךְ הוּא בָּחַר לָתֵת לָהּ חַיִּים, יֵשׁ לָהֶם עֵרֶךְ גָּדוֹל.
💛 אַהֲבַת הַשֵּׁם: בְּעֶשֶׂר מִלִּים קְטַנּוֹת, עוֹד לִפְנֵי שֶׁקּוֹרְאִים שֵׁם מָלֵא לַה׳, אוֹמְרִים לוֹ תּוֹדָה עַל רֶגַע הַחַיִּים הֶחָדָש.
📖 פֶּרֶק ד׳, הֲלָכָה א׳

🪞צְנִיעוּת – גַּם לְבַד

גַּם כְּשֶׁנִּמְצָאִים לְבַד בַּבַּיִת, לֹא אוֹמְרִים ״מִי רוֹאֶה אוֹתִי?״, כִּי ״מְלֹא כָל הָאָרֶץ כְּבוֹדוֹ״. לָכֵן בָּנוֹת מִתְלַבְּשׁוֹת בַּבַּיִת לְפִי כְּלָלֵי הַצְּנִיעוּת – שַׁרְווּלִים עַד הַמַּרְפֵּק וַחֲצָאִית עַד אַחֲרֵי הַבֶּרֶךְ – גַּם כְּשֶׁאֵין שָׁם אַף אֶחָד. וּבָנִים מְכַסִּים לְבַד אֶת מָה שֶׁרְגִילִים לְכַסּוֹת לִפְנֵי בְּנֵי הַמִּשְׁפָּחָה.
גַּם בַּפְּנִימִיָּה, שֶׁכֻּלָּן שָׁם בָּנוֹת, לֹא שׁוֹהוֹת בַּחֶדֶר בִּלְבוּשׁ שֶׁאֵינוֹ צָנוּעַ. אֶת הַבְּגָדִים הַתַּחְתּוֹנִים מַחְלִיפוֹת בַּשֵּׁרוּתִים, בַּמִּקְלַחַת אוֹ מִתַּחַת לְכִסּוּי.
💛 אַהֲבַת הַשֵּׁם: הַצְּנִיעוּת מַדְגִּישָׁה אֶת הַיֹּפִי הַפְּנִימִי שֶׁל הַנְּשָׁמָה וְשׁוֹמֶרֶת עָלָיו. אָנוּ מִתְכַּסִּים מִכְּבוֹד ה׳ וּמִכְּבוֹד צֶלֶם אֱלֹקִים שֶׁבָּנוּ.
📖 פֶּרֶק ד׳, הֲלָכָה ב׳

🧠 בְּדֹק אֶת עַצְמְךָ

רוּת לְבַד בַּבַּיִת, וְאֵין שׁוּם אָדָם אַחֵר. אֵיךְ הִיא מִתְלַבֶּשֶׁת?
💡 גַּם לְבַד לֹא אוֹמְרִים ״מִי רוֹאֶה אוֹתִי?״, כִּי הַקָּדוֹשׁ בָּרוּךְ הוּא מְלֹא כָל הָאָרֶץ כְּבוֹדוֹ. לָכֵן גַּם בַּבַּיִת מִתְלַבְּשִׁים לְפִי כְּלָלֵי הַצְּנִיעוּת – מִכְּבוֹד שָׁמַיִם וּמִכְּבוֹד צֶלֶם אֱלֹקִים שֶׁבָּנוּ.

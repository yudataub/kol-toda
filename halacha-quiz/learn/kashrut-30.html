חָלָב וּמוּצָרָיו

🍽️ הִלְכוֹת כַּשְׁרוּת · פֶּרֶק ל׳

🐄חָלָב טָהוֹר

חָלָב טָהוֹר בָּא מִבְּהֵמָה אוֹ חַיָּה טְהוֹרָה, שֶׁבְּשָׂרָהּ כָּשֵׁר – כְּמוֹ פָּרָה וְעֵז. חֲלֵב אָתוֹן, סוּסָה, גְּמַלָּה וַחֲזִירָה – טָמֵא. זֶה הַכְּלָל: "הַיּוֹצֵא מִן הַטָּמֵא – טָמֵא, וְהַיּוֹצֵא מִן הַטָּהוֹר – טָהוֹר".
שְׁנֵי הֶבְדֵּלִים יֵשׁ בֵּינֵיהֶם: חָלָב טָהוֹר יָכוֹל לְהִתְגַּבֵּן לִגְבִינָה, וְחָלָב טָמֵא – לֹא. וְצֶבַע הֶחָלָב הַטָּהוֹר לָבָן, וְשֶׁל הַטָּמֵא – צְהַבְהַב. וְחָלָב שֶׁל בְּהֵמָה טְרֵפָה – אָסוּר כְּמוֹ בְּשָׂרָהּ.
💛 אַהֲבַת הַשֵּׁם: כָּל כּוֹס חָלָב שֶׁאָנוּ שׁוֹתִים בַּבֹּקֶר הִיא מַתָּנָה מִן ה׳. כְּשֶׁאָנוּ בּוֹדְקִים שֶׁהִיא כְּשֵׁרָה, אָנוּ מְקַבְּלִים אֶת הַמַּתָּנָה בַּדֶּרֶךְ שֶׁה׳ רוֹצֶה.
📖 פֶּרֶק ל׳, הֲלָכָה א׳

🥛חֲלֵב יִשְׂרָאֵל

חֲכָמִים אָסְרוּ חָלָב שֶׁחָלַב גּוֹי, שֶׁמָּא עֵרֵב בּוֹ חָלָב שֶׁל בְּהֵמָה טְמֵאָה. אֲבָל כְּשֶׁיְּהוּדִי מַשְׁגִּיחַ עַל הַחֲלִיבָה – הֶחָלָב כָּשֵׁר. לֹא צָרִיךְ לִרְאוֹת אֶת כָּל הַחֲלִיבָה: מַסְפִּיק שֶׁהַגּוֹי יוֹדֵעַ שֶׁהַיְּהוּדִי יָכוֹל לְהִכָּנֵס וְלִרְאוֹת אוֹתוֹ בְּכָל רֶגַע.
עַל חָלָב שֶׁל חֲבָרוֹת גְּדוֹלוֹת וּמְפֻקָּחוֹת – נֶחְלְקוּ הַפּוֹסְקִים (רְאוּ בְּרָמָה ג׳). בְּאֶרֶץ יִשְׂרָאֵל, בְּכָל הַהֶכְשֵׁרִים שֶׁל הָרַבָּנֻיּוֹת מְהַדְּרִים שֶׁלֹּא לְהִשְׁתַּמֵּשׁ בַּחֲלֵב נָכְרִים. וְהַיּוֹם אֶפְשָׁר לְהַשְׁגִּיחַ גַּם בְּמַצְלֵמוֹת.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים הִרְחִיקוּ אוֹתָנוּ מִסָּפֵק – אֲפִלּוּ קָטָן מְאוֹד. יִשְׂרָאֵל קְדוֹשִׁים מְהַדְּרִים, כְּשֶׁאֶפְשָׁר, לֶאֱכֹל רַק מָה שֶׁכָּשֵׁר בְּלִי שׁוּם סָפֵק.
📖 פֶּרֶק ל׳, הֲלָכוֹת ב׳, ו׳

🧀גְּבִינַת גּוֹיִם

חֲכָמִים אָסְרוּ גַּם גְּבִינָה שֶׁהֵכִין גּוֹי – אֲפִלּוּ כְּשֶׁהֶחָלָב כָּשֵׁר. הַטַּעַם: כְּדֵי לְהָכִין גְּבִינָה צָרִיךְ חֹמֶר מְיֻחָד שֶׁמְּגַבֵּן אֶת הֶחָלָב, וְהַגּוֹיִם הָיוּ לוֹקְחִים אוֹתוֹ מֵעוֹר קֵבָה שֶׁל בְּהֵמָה שֶׁלֹּא נִשְׁחֲטָה כַּהֲלָכָה – נְבֵלָה.
הַיּוֹם מְגַבְּנִים גְּבִינוֹת בְּחֹמָרִים מֵחַיְדַּקִּים. וּבְכָל זֹאת, גְּבִינָה שֶׁגּוֹי הֵכִין אֲסוּרָה – כִּי חֲכָמִים גָּזְרוּ עַל כָּל הַגְּבִינוֹת שֶׁל גּוֹיִם.
💛 אַהֲבַת הַשֵּׁם: גְּזֵרָה שֶׁל חֲכָמִים עוֹמֶדֶת גַּם כְּשֶׁהַטַּעַם שֶׁלָּהּ נִרְאֶה רָחוֹק. אָנוּ שׁוֹמְעִים בְּקוֹל חֲכָמֵינוּ, כִּי הֵם רָאוּ רָחוֹק מֵאִתָּנוּ.
📖 פֶּרֶק ל׳, הֲלָכָה ז׳

🐮חֲלִיבָה בְּשַׁבָּת

חֲלִיבָה בְּשַׁבָּת אֲסוּרָה מִן הַתּוֹרָה. אֲבָל כְּדֵי שֶׁהַפָּרוֹת לֹא יִצְטַעֲרוּ, מֻתָּר לְבַקֵּשׁ מִגּוֹי שֶׁיַּחֲלֹב אוֹתָן – מִשּׁוּם צַעַר בַּעֲלֵי חַיִּים. הֶחָלָב מֻקְצֶה כָּל הַשַּׁבָּת, וְאַחֲרֵי שַׁבָּת מֻתָּר. הַיּוֹם חוֹלְבִים בִּמְכוֹנוֹת, וּבְשַׁבָּת מַפְעִילִים אוֹתָן בְּ"גְרָמָא".
חָלָב שֶׁיְּהוּדִי חָלַב בְּחִלּוּל שַׁבָּת – אָסוּר. וְרָאוּי לְהַעֲדִיף חָלָב מִמְּשָׁקִים שֶׁשּׁוֹמְרִים שַׁבָּת – כִּי הוּא כָּשֵׁר בְּלִי סָפֵק, וְכָךְ מְעוֹדְדִים שְׁמִירַת שַׁבָּת.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה דּוֹאֶגֶת גַּם לַפָּרוֹת! חֲכָמִים הֵקֵלּוּ בִּגְזֵרָתָם כְּדֵי שֶׁבַּעֲלֵי הַחַיִּים לֹא יִסְבְּלוּ – וְכָךְ שׁוֹמְרִים גַּם עַל הַשַּׁבָּת וְגַם עַל הָרַחֲמִים.
📖 פֶּרֶק ל׳, הֲלָכָה י״א

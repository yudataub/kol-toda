צומות החורבן

📅 רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת · פֶּרֶק ו׳

🕍אַרְבָּעָה צוֹמוֹת

אַחֲרֵי חֻרְבַּן בֵּית הַמִּקְדָּשׁ הָרִאשׁוֹן תִּקְּנוּ הַנְּבִיאִים אַרְבָּעָה צוֹמוֹת: עֲשָׂרָה בְּטֵבֵת, שִׁבְעָה עָשָׂר בְּתַמּוּז, תִּשְׁעָה בְּאָב וְצוֹם גְּדַלְיָה. הַצּוֹמוֹת מְעוֹרְרִים אוֹתָנוּ לְהִצְטַעֵר עַל הַחֻרְבָּן, לַחֲזֹר בִּתְשׁוּבָה וּלְתַקֵּן אֶת הַמַּעֲשִׂים שֶׁגָּרְמוּ לוֹ.
בִּימֵי בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי הָפְכוּ הַצּוֹמוֹת לְיָמִים שְׂמֵחִים, כְּדִבְרֵי זְכַרְיָה הַנָּבִיא. כְּשֶׁנֶּחְרַב הַבַּיִת הַשֵּׁנִי – חָזְרוּ לָצוּם.
💛 אַהֲבַת הַשֵּׁם: הַצּוֹם אֵינוֹ רַק צַעַר. הוּא קְרִיאָה לָשׁוּב אֶל ה׳ – וְהַבְטָחָה שֶׁיּוֹם אֶחָד הוּא יֵהָפֵךְ לְחַג שָׂמֵחַ.
📖 פֶּרֶק ו׳, הֲלָכָה א׳

🕊️"לְשָׂשׂוֹן וּלְשִׂמְחָה"

זְכַרְיָה הַנָּבִיא אָמַר: "צוֹם הָרְבִיעִי וְצוֹם הַחֲמִישִׁי וְצוֹם הַשְּׁבִיעִי וְצוֹם הָעֲשִׂירִי יִהְיֶה לְבֵית יְהוּדָה לְשָׂשׂוֹן וּלְשִׂמְחָה... וְהָאֱמֶת וְהַשָּׁלוֹם אֱהָבוּ". כְּשֶׁיִּבָּנֶה בֵּית הַמִּקְדָּשׁ, יְמֵי הַצּוֹם יֵהָפְכוּ לְיָמִים טוֹבִים.
בְּתַעֲנִית צִבּוּר קוֹרְאִים בַּתּוֹרָה עַל כַּפָּרַת חֵטְא הָעֵגֶל – כְּמוֹ שֶׁה׳ כִּפֵּר אָז, כָּךְ יְכַפֵּר גַּם עָלֵינוּ.
💛 אַהֲבַת הַשֵּׁם: "וְהָאֱמֶת וְהַשָּׁלוֹם אֱהָבוּ" – הַדֶּרֶךְ לַגְּאֻלָּה עוֹבֶרֶת בֶּאֱמֶת וּבְשָׁלוֹם בֵּינֵינוּ. כָּל מַעֲשֶׂה שֶׁל אַהֲבָה מְקָרֵב אֶת הַיּוֹם הַשָּׂמֵחַ.
📖 פֶּרֶק ו׳, הֲלָכָה א׳; פֶּרֶק ז׳, הֲלָכָה י״א

🧱עֲשָׂרָה בְּטֵבֵת

בַּעֲשָׂרָה בְּטֵבֵת בָּא נְבוּכַדְנֶאצַּר מֶלֶךְ בָּבֶל לָצוּר עַל יְרוּשָׁלַיִם, וּמִשָּׁם הִתְחִיל הַחֻרְבָּן. לַצּוֹם הַזֶּה נִצְטָרְפוּ עוֹד שְׁנֵי מְאֹרָעוֹת עֲצוּבִים: מוֹת עֶזְרָא הַסּוֹפֵר בְּט׳ בְּטֵבֵת, וְתַרְגּוּם הַתּוֹרָה לִיוָנִית בְּח׳ בְּטֵבֵת.
הָרַבָּנוּת הָרָאשִׁית קָבְעָה אֶת הַיּוֹם הַזֶּה לְיוֹם הַקַּדִּישׁ הַכְּלָלִי לְכָל קְדוֹשֵׁי הַשּׁוֹאָה שֶׁלֹּא נוֹדַע יוֹם פְּטִירָתָם.
💛 אַהֲבַת הַשֵּׁם: עַל הַמָּצוֹר עוֹנִים בְּבִנְיַן יְרוּשָׁלַיִם וְהָאָרֶץ, וְעַל מוֹת עֶזְרָא – בְּהַגְדָּלַת הַתּוֹרָה. כָּךְ הַצַּעַר נַעֲשֶׂה כֹּחַ לִבְנוֹת.
📖 פֶּרֶק ו׳, הֲלָכָה ב׳

📜שִׁבְעָה עָשָׂר בְּתַמּוּז

חֲמִשָּׁה דְּבָרִים רָעִים אֵרְעוּ בְּי״ז בְּתַמּוּז: נִשְׁתַּבְּרוּ הַלּוּחוֹת, בָּטַל קָרְבַּן הַתָּמִיד, הֻבְקְעָה חוֹמַת יְרוּשָׁלַיִם, אַפּוֹסְטְמוֹס שָׂרַף אֶת הַתּוֹרָה, וְהֶעֱמִידוּ צֶלֶם בַּהֵיכָל. הַצּוֹם נִקְבַּע בְּעִקָּר עַל הַבְקָעַת הַחוֹמָה.
כְּשָׁלֹשׁ שָׁנִים צָרוּ הָרוֹמָאִים עַל יְרוּשָׁלַיִם וְלֹא יָכְלוּ לָהּ. בִּגְלַל שִׂנְאַת חִנָּם וּמִלְחֶמֶת אַחִים נֶחְלְשׁוּ הַמְּגִנִּים, וּבְי״ז בְּתַמּוּז בָּקְעוּ הָרוֹמָאִים אֶת הַחוֹמָה.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁמֹּשֶׁה רָאָה אֶת הָעֵגֶל, שָׁבַר אֶת הַלּוּחוֹת. אֲבָל ה׳ סָלַח וְנָתַן לוּחוֹת שְׁנִיִּים – וְגַם עָלֵינוּ הוּא יְרַחֵם וְיִבְנֶה אֶת בֵּית הַמִּקְדָּשׁ.
📖 פֶּרֶק ו׳, הֲלָכָה ג׳; פֶּרֶק ז׳, הֲלָכָה י״א

🧱שִׂנְאַת חִנָּם

בִּגְלַל שִׂנְאַת חִנָּם וּמִלְחֶמֶת אַחִים נֶחְלְשׁוּ מְגִנֵּי יְרוּשָׁלַיִם, וְהָרוֹמָאִים בָּקְעוּ אֶת הַחוֹמָה בְּי״ז בְּתַמּוּז. אַחֲרֵי שְׁלֹשָׁה שָׁבוּעוֹת שֶׁל קְרָבוֹת, בְּתִשְׁעָה בְּאָב, נִשְׂרַף בֵּית הַמִּקְדָּשׁ.
גְּדַלְיָה בֶּן אֲחִיקָם, מוֹשֵׁל הַיְּהוּדִים בָּאָרֶץ, נִרְצַח בִּידֵי יְהוּדִי – וְעַל כָּךְ צָמִים בְּצוֹם גְּדַלְיָה.
💛 אַהֲבַת הַשֵּׁם: אִם שִׂנְאַת חִנָּם הֶחְרִיבָה – אַהֲבָה בֵּין אַחִים תִּבְנֶה. כָּל מִלָּה טוֹבָה לְחָבֵר הִיא עוֹד אֶבֶן בְּבִנְיַן יְרוּשָׁלַיִם.
📖 פֶּרֶק ו׳, הֲלָכוֹת ג׳, ה׳

🏚️תִּשְׁעָה בְּאָב וְחֵטְא הַמְּרַגְּלִים

חֲמִשָּׁה דְּבָרִים אֵרְעוּ בְּתִשְׁעָה בְּאָב: נִגְזַר עַל דּוֹר הַמִּדְבָּר שֶׁלֹּא יִכָּנְסוּ לָאָרֶץ, חָרַב בֵּית הַמִּקְדָּשׁ הָרִאשׁוֹן וְהַשֵּׁנִי, נִלְכְּדָה בֵּיתָר, וְנֶחְרְשָׁה הָעִיר. הַכֹּל הִתְחִיל בְּלֵיל תִּשְׁעָה בְּאָב, כְּשֶׁבָּכוּ יִשְׂרָאֵל עַל דִּבְרֵי הַמְּרַגְּלִים וּמָאֲסוּ בָּאָרֶץ.
אָמַר ה׳: "אַתֶּם בְּכִיתֶם בְּכִיָּה שֶׁל חִנָּם, וַאֲנִי אֶקְבַּע לָכֶם בְּכִיָּה לְדוֹרוֹת". רַק יְהוֹשֻׁעַ וְכָלֵב, שֶׁאָמְרוּ "טוֹבָה הָאָרֶץ מְאֹד מְאֹד", נִכְנְסוּ לָאָרֶץ.
💛 אַהֲבַת הַשֵּׁם: הַתִּקּוּן לְחֵטְא הַמְּרַגְּלִים הוּא לֶאֱהֹב אֶת אֶרֶץ יִשְׂרָאֵל וּלְהַאֲמִין שֶׁה׳ אִתָּנוּ – כְּמוֹ יְהוֹשֻׁעַ וְכָלֵב.
📖 פֶּרֶק ו׳, הֲלָכָה ד׳

🕯️צוֹם גְּדַלְיָה

אַחֲרֵי הַחֻרְבָּן מִנָּה מֶלֶךְ בָּבֶל אֶת גְּדַלְיָה בֶּן אֲחִיקָם לְמוֹשֵׁל עַל הַיְּהוּדִים שֶׁנִּשְׁאֲרוּ בָּאָרֶץ. יִשְׁמָעֵאל בֶּן נְתַנְיָה רָצַח אוֹתוֹ, וּבְכָךְ כָּבְתָה הַגַּחֶלֶת הָאַחֲרוֹנָה שֶׁל שִׁלְטוֹן יִשְׂרָאֵל בָּאָרֶץ. צָמִים עַל כָּךְ בִּשְׁלֹשָׁה בְּתִשְׁרֵי.
חֲכָמִים לָמְדוּ מִכָּאן: שְׁקוּלָה מִיתָתָם שֶׁל צַדִּיקִים כִּשְׂרֵפַת בֵּית אֱלֹהֵינוּ.
💛 אַהֲבַת הַשֵּׁם: בְּכָל דּוֹר יֵשׁ צַדִּיקִים שֶׁמַּחְזִיקִים אֶת הָעָם. הַצּוֹם מְלַמֵּד אוֹתָנוּ כַּמָּה יָקָר כָּל צַדִּיק לִפְנֵי ה׳.
📖 פֶּרֶק ו׳, הֲלָכָה ה׳

🧠 בְּדֹק אֶת עַצְמְךָ

בְּאֵיזֶה מְאֹרָע הִתְחִיל הַצַּעַר שֶׁל תִּשְׁעָה בְּאָב?
💡 בְּלֵיל תִּשְׁעָה בְּאָב בָּכוּ יִשְׂרָאֵל עַל דִּבְרֵי הַמְּרַגְּלִים. אָמַר ה׳: "אַתֶּם בְּכִיתֶם בְּכִיָּה שֶׁל חִנָּם, וַאֲנִי אֶקְבַּע לָכֶם בְּכִיָּה לְדוֹרוֹת" – וּבְאוֹתוֹ יוֹם נֶחְרְבוּ שְׁנֵי בָּתֵּי הַמִּקְדָּשׁ.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

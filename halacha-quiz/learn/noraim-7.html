הלכות יום הכיפורים

🎺 רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים · פֶּרֶק ז׳

🤍בְּגָדִים לְבָנִים

בְּיוֹם הַכִּפּוּרִים אֵין סְעוּדוֹת, וּמְכַבְּדִים אֶת הַיּוֹם בִּבְגָדִים נְקִיִּים. רַבִּים לוֹבְשִׁים לָבָן, כְּמוֹ מַלְאֲכֵי הַשָּׁרֵת. גְּבָרִים רַבִּים מִיּוֹצְאֵי אַשְׁכְּנַז לוֹבְשִׁים ״קִיטְל״ – בֶּגֶד לָבָן שֶׁמַּזְכִּיר גַּם מַלְאָכִים וְגַם תַּכְרִיכִים, כְּדֵי לְהִתְעוֹרֵר לִתְשׁוּבָה.
מְנַקִּים אֶת הַבַּיִת וּפוֹרְשִׂים מַפָּה נָאָה, וּמַדְלִיקִים אֶת כָּל הָאוֹרוֹת בְּבֵית הַכְּנֶסֶת לִכְבוֹד הַיּוֹם: ״בָּאֻרִים כַּבְּדוּ ה׳״. גַּם נָשִׁים רַבּוֹת לוֹבְשׁוֹת לָבָן, וְגַם מִי שֶׁלֹּא – אֵינָהּ מִתְקַשֶּׁטֶת בְּתַכְשִׁיטִים, מִפְּנֵי אֵימַת יוֹם הַדִּין.
💛 אַהֲבַת הַשֵּׁם: בַּיּוֹם הַזֶּה אֲנַחְנוּ לֹא נִמְשָׁכִים אַחֲרֵי הַגּוּף, וּנְקִיִּים מֵחֵטְא – כְּמוֹ מַלְאָכִים לְבָנִים לִפְנֵי ה׳.
📖 פֶּרֶק ז׳, הֲלָכָה א׳

🗣️הַוִּדּוּי

מִצְוָה עַל כָּל אָדָם לָשׁוּב וּלְהִתְוַדּוֹת עַל חֲטָאָיו בְּיוֹם הַכִּפּוּרִים. אוֹמְרִים ״אָשַׁמְנוּ, בָּגַדְנוּ...״ לְפִי סֵדֶר הָאוֹתִיּוֹת, וְ״עַל חֵטְא״. מִתְוַדִּים בַּעֲמִידָה, כּוֹפְפִים מְעַט אֶת הָרֹאשׁ, וּמַכִּים בָּאֶגְרוֹף עַל הֶחָזֶה – ״הַלֵּב שֶׁלִּי גָּרַם לַחֵטְא״.
הַוִּדּוּי בִּלְשׁוֹן רַבִּים – ״חָטָאנוּ״ – כִּי כָּל יִשְׂרָאֵל עֲרֵבִים זֶה לָזֶה, וְכָל אֶחָד מִתְוַדֶּה בְּשֵׁם כָּל הָעָם, כְּמוֹ הַכֹּהֵן הַגָּדוֹל.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאוֹמְרִים אֶת הַחֵטְא בַּפֶּה, הַתְּשׁוּבָה נִגְמֶרֶת. הַמִּלִּים הוֹפְכוֹת אֶת הַחֲרָטָה שֶׁבַּלֵּב לְהַחְלָטָה אֲמִתִּית.
📖 פֶּרֶק ז׳, הֲלָכוֹת ג׳–ד׳

🕯️נֵרוֹת יוֹם הַכִּפּוּרִים

מַדְלִיקִים נֵרוֹת לִכְבוֹד יוֹם הַכִּפּוּרִים, כְּמוֹ בְּשַׁבָּת, וּמְבָרְכִים: ״לְהַדְלִיק נֵר שֶׁל יוֹם הַכִּפּוּרִים״. כְּשֶׁיּוֹם הַכִּפּוּרִים חָל בְּשַׁבָּת: ״לְהַדְלִיק נֵר שֶׁל שַׁבָּת וְשֶׁל יוֹם הַכִּפּוּרִים״. הַנֵּרוֹת מְכַבְּדִים אֶת הַיּוֹם וּמְבִיאִים שְׁלוֹם בַּיִת.
נָשִׁים מְקַבְּלוֹת אֶת קְדֻשַּׁת הַיּוֹם בְּהַדְלָקַת הַנֵּרוֹת, וּמְבָרְכוֹת אַחֲרֵיהֶן ״שֶׁהֶחֱיָנוּ״ – וּמֵאָז אֲסוּרוֹת בְּכָל אִסּוּרֵי הַיּוֹם. מִי שֶׁצְּרִיכָה לִנְסֹעַ לְבֵית הַכְּנֶסֶת – מְכַוֶּנֶת שֶׁלֹּא לְקַבֵּל אֶת הַיּוֹם בַּהַדְלָקָה. וּמִי שֶׁאֵינָהּ בָּאָה לְבֵית הַכְּנֶסֶת – אוֹמֶרֶת לְפָחוֹת וִדּוּי אֶחָד.
💛 אַהֲבַת הַשֵּׁם: הָאוֹר שֶׁל הַנֵּרוֹת מַכְנִיס אֶת קְדֻשַּׁת הַיּוֹם אֶל הַבַּיִת: ״עַל כֵּן בָּאֻרִים כַּבְּדוּ ה׳״.
📖 פֶּרֶק ז׳, הֲלָכוֹת ה׳, ט׳

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

לִפְנֵי כְּנִיסַת יוֹם הַכִּפּוּרִים אוֹמְרִים תְּפִלָּה שֶׁיֵּשׁ בָּהּ וִדּוּי, כְּדֵי לְהִכָּנֵס לַיּוֹם מִתּוֹךְ תְּשׁוּבָה: הָאַשְׁכְּנַזִּים אוֹמְרִים ״תְּפִלָּה זַכָּה״, וְהַסְּפָרַדִּים אוֹמְרִים אֶת הַפִּיּוּט ״לְךָ אֵלִי תְּשׁוּקָתִי״. וְעוֹד מַחֲלֹקֶת: נַעֲלַיִם שֶׁאֵינָן מֵעוֹר – רֹב הַפּוֹסְקִים מַתִּירִים, וְהָרַב מֶלָמֵד אוֹסֵר כָּל נַעַל נוֹחָה שֶׁהוֹלְכִים בָּהּ בַּחוּץ.
הַכְּרִיעָה הָרְבִיעִית בְּמוּסָף: הָאַשְׁכְּנַזִּים כּוֹרְעִים בְּ״עָלֵינוּ לְשַׁבֵּחַ״ שֶׁלִּפְנֵי סֵדֶר הָעֲבוֹדָה, וְהַסְּפָרַדִּים – כְּשֶׁמַּזְכִּירִים אֶת הַשָּׂעִיר שֶׁעָלָה עָלָיו הַגּוֹרָל לַה׳. וְהַתֵּימָנִים אֵינָם כּוֹרְעִים בְּמוּסָף.
💛 אַהֲבַת הַשֵּׁם: כָּל הַמִּנְהָגִים נוֹסְדוּ עַל יְדֵי גְּדוֹלֵי יִשְׂרָאֵל. כָּל אֶחָד שׁוֹמֵר בְּאַהֲבָה עַל מִנְהַג אֲבוֹתָיו, וְכֻלָּנוּ עוֹמְדִים יַחַד לִפְנֵי ה׳ בַּיּוֹם הַקָּדוֹשׁ.
📖 פֶּרֶק ז׳, הֲלָכוֹת ה׳, י״ד; פֶּרֶק ט׳, הֲלָכָה ה׳

📜כָּל נִדְרֵי וּ״בָּרוּךְ שֵׁם״

בִּפְתִיחַת הַיּוֹם מוֹצִיאִים סִפְרֵי תוֹרָה, וְהַחַזָּן אוֹמֵר: ״אָנוּ מַתִּירִין לְהִתְפַּלֵּל עִם הָעֲבַרְיָנִים״ – כִּי בְּיוֹם הַכִּפּוּרִים כָּל יְהוּדִי מוּזְמָן לְהִתְפַּלֵּל יַחַד. אַחַר כָּךְ אוֹמְרִים ״כָּל נִדְרֵי״. מְקַבְּלִים אֶת קְדֻשַּׁת הַיּוֹם לִפְנֵי הַשְּׁקִיעָה, וְהַגַּבַּאי מַכְרִיז עַל כָּךְ.
בְּכָל הַשָּׁנָה אוֹמְרִים ״בָּרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ לְעוֹלָם וָעֶד״ בְּלַחַשׁ, וּבְיוֹם הַכִּפּוּרִים – בְּקוֹל רָם. כִּי בַּיּוֹם הַזֶּה אֲנַחְנוּ נְקִיִּים כְּמַלְאֲכֵי הַשָּׁרֵת, שֶׁמְּשַׁבְּחִים כָּךְ אֶת ה׳.
💛 אַהֲבַת הַשֵּׁם: ״כָּל תַּעֲנִית שֶׁאֵין בָּהּ מִפּוֹשְׁעֵי יִשְׂרָאֵל – אֵינָהּ תַּעֲנִית״. לְכָל יְהוּדִי יֵשׁ נִיצוֹץ מְיֻחָד, וּכְשֶׁכֻּלָּם יַחַד – כָּל יִשְׂרָאֵל שְׁלֵמִים.
📖 פֶּרֶק ז׳, הֲלָכוֹת י׳–י״ב

🔓נְעִילָה וּתְקִיעַת שׁוֹפָר

תְּפִלַּת נְעִילָה הִיא כְּנֶגֶד נְעִילַת שַׁעֲרֵי שָׁמַיִם, וּפוֹתְחִים בָּהּ אֶת אֲרוֹן הַקֹּדֶשׁ. בְּסוֹפָהּ אוֹמְרִים יַחַד: ״שְׁמַע יִשְׂרָאֵל״, ״בָּרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ״, וְשֶׁבַע פְּעָמִים ״ה׳ הוּא הָאֱלֹהִים״. אַחַר כָּךְ תּוֹקְעִים בַּשּׁוֹפָר.
״הַכֹּל הוֹלֵךְ אַחַר הַחִתּוּם״. גַּם מִי שֶׁנֶּחֱלַשׁ מִן הַצּוֹם – מִתְאַמֵּץ מְאוֹד בִּתְפִלַּת נְעִילָה.
💛 אַהֲבַת הַשֵּׁם: אַחֲרֵי הַשּׁוֹפָר כָּל יִשְׂרָאֵל יוֹדְעִים שֶׁה׳ אוֹהֵב אוֹתָם וְקִבֵּל אֶת תְּשׁוּבָתָם, וְשִׂמְחָה גְּדוֹלָה מְמַלֵּאת אֶת הַלֵּב: ״לְשָׁנָה הַבָּאָה בִּירוּשָׁלַיִם הַבְּנוּיָה!״
📖 פֶּרֶק ז׳, הֲלָכוֹת י״ז–י״ח

✍️״חָתְמֵנוּ״

בְּכָל הַיָּמִים הַנּוֹרָאִים מְבַקְּשִׁים ״כָּתְבֵנוּ בְּסֵפֶר חַיִּים״. בִּתְפִלַּת נְעִילָה, לִקְרַאת סוֹף הַיּוֹם, מְבַקְּשִׁים בִּלְשׁוֹן חֲתִימָה: ״חָתְמֵנוּ״ – כִּי זֶה זְמַן חִתּוּם הַדִּין. וּמִי שֶׁטָּעָה וְאָמַר ״כָּתְבֵנוּ״ – יָצָא.
בְּסוֹף נְעִילָה אוֹמְרִים שֶׁבַע פְּעָמִים ״ה׳ הוּא הָאֱלֹהִים״ – כְּנֶגֶד הַשְּׁכִינָה שֶׁעוֹלָה מֵעַל שִׁבְעָה רְקִיעִים, אַחֲרֵי שֶׁהִתְקָרְבָה אֵלֵינוּ בַּיָּמִים הַנּוֹרָאִים.
💛 אַהֲבַת הַשֵּׁם: בְּסוֹף הַיּוֹם הַכִּי קָדוֹשׁ אֲנַחְנוּ חוֹזְרִים אֶל הַיְּסוֹד שֶׁל הַכֹּל – הָאֱמוּנָה בַּה׳. וּכְכָל שֶׁהָאֱמוּנָה חֲזָקָה יוֹתֵר, הַשָּׁנָה טוֹבָה יוֹתֵר.
📖 פֶּרֶק ז׳, הֲלָכוֹת י״ז–י״ח

🍷הַבְדָּלָה בְּמוֹצָאֵי יוֹם הַכִּפּוּרִים

גַּם אַחֲרֵי צֵאת הַכּוֹכָבִים לֹא אוֹכְלִים עַד שֶׁמַּבְדִּילִים עַל כּוֹס יַיִן. מְבָרְכִים עַל הַגֶּפֶן, עַל הַנֵּר וְעַל הַהַבְדָּלָה – בְּלִי בְּשָׂמִים. אֶת בִּרְכַּת הָאֵשׁ מְבָרְכִים עַל נֵר שֶׁדָּלַק כָּל יוֹם הַכִּפּוּרִים, וְלָכֵן מַדְלִיקִים לִפְנֵי הַצּוֹם נֵר נְשָׁמָה.
אַחֲרֵי הַהַבְדָּלָה אוֹכְלִים בְּשִׂמְחָה. וַחֲסִידִים מְמַהֲרִים לְהַתְחִיל לִבְנוֹת אֶת הַסֻּכָּה – לָצֵאת מִמִּצְוָה לְמִצְוָה.
💛 אַהֲבַת הַשֵּׁם: בְּמוֹצָאֵי יוֹם הַכִּפּוּרִים יוֹצֵאת בַּת קוֹל: ״לֵךְ אֱכֹל בְּשִׂמְחָה לַחְמֶךָ... כִּי כְבָר רָצָה הָאֱלֹהִים אֶת מַעֲשֶׂיךָ״.
📖 פֶּרֶק ז׳, הֲלָכָה כ׳

🧠 בְּדֹק אֶת עַצְמְךָ

בְּיוֹם הַכִּפּוּרִים רוֹצֶה מֹשֶׁה לְהָרִיחַ עֲנַף הֲדַס רֵיחָנִי. מָה הַדִּין?
💡 שֶׁלֹּא כְּמוֹ תִּשְׁעָה בְּאָב, יוֹם הַכִּפּוּרִים הוּא יוֹם טוֹב – וּמְרִיחִים בְּשָׂמִים וּמְבָרְכִים עֲלֵיהֶם. וּבְהַבְדָּלָה שֶׁל מוֹצָאֵי יוֹם הַכִּפּוּרִים דַּוְקָא אֵין מְבָרְכִים עַל בְּשָׂמִים.
לָמָּה אוֹמְרִים ״בָּרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ לְעוֹלָם וָעֶד״ בְּקוֹל רָם דַּוְקָא בְּיוֹם הַכִּפּוּרִים?
💡 מֹשֶׁה רַבֵּנוּ שָׁמַע אֶת הַמַּלְאָכִים מְשַׁבְּחִים כָּךְ, וְלִמֵּד אֶת יִשְׂרָאֵל לוֹמַר זֹאת בְּלַחַשׁ. רַק בְּיוֹם הַכִּפּוּרִים, כְּשֶׁאֲנַחְנוּ נְקִיִּים כְּמַלְאָכִים – אוֹמְרִים בְּקוֹל.
עֶרֶב יוֹם הַכִּפּוּרִים חָל בְּיוֹם חֹל. מָה הַבְּרָכָה עַל הַנֵּרוֹת?
💡 מְבָרְכִים ״לְהַדְלִיק נֵר שֶׁל יוֹם הַכִּפּוּרִים״, וְהַנָּשִׁים מְקַבְּלוֹת אֶת הַיּוֹם וּמְבָרְכוֹת ״שֶׁהֶחֱיָנוּ״. גַּם בְּלִי סְעוּדָה, הַנֵּרוֹת מְכַבְּדִים אֶת הַיּוֹם וּמְבִיאִים שְׁלוֹם בַּיִת.

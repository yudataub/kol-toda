בִּטּוּל חָמֵץ וּבִעוּרוֹ

🍷 הִלְכוֹת פֶּסַח · פֶּרֶק ה׳

🗣️בִּטּוּל בַּלַּיְלָה וּבַבֹּקֶר

אַחֲרֵי הַבְּדִיקָה מְבַטְּלִים אֶת הֶחָמֵץ שֶׁלֹּא מָצָאנוּ. בְּנֹסַח אַשְׁכְּנַז, בְּעִבְרִית: ״כָּל חָמֵץ וּשְׂאוֹר שֶׁיֵּשׁ בִּרְשׁוּתִי, שֶׁלֹּא רְאִיתִיו וְשֶׁלֹּא בִּעַרְתִּיו, יִבָּטֵל וִיהֵא הֶפְקֵר כַּעֲפַר הָאָרֶץ״. אֶת הֶחָמֵץ שֶׁשּׁוֹמְרִים לָאֲרוּחוֹת וְלַשְּׂרֵפָה לֹא מְבַטְּלִים בַּלַּיְלָה. בַּבֹּקֶר, אַחֲרֵי הַשְּׂרֵפָה, מְבַטְּלִים אֶת הַכֹּל: ״שֶׁרְאִיתִיו וְשֶׁלֹּא רְאִיתִיו״.
אֶת הַבִּטּוּל שֶׁל הַבֹּקֶר אוֹמְרִים עַד סוֹף הַשָּׁעָה הַחֲמִישִׁית. מֵהַשָּׁעָה הַשִּׁשִּׁית הֶחָמֵץ אָסוּר בַּהֲנָאָה, וּכְבָר אִי אֶפְשָׁר לְבַטֵּל אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: בַּבִּטּוּל אֲנַחְנוּ אוֹמְרִים: הֶחָמֵץ שֶׁלִּי שָׁוֶה בְּעֵינַי כְּמוֹ עָפָר. מָה שֶׁה׳ אָסַר – כְּבָר לֹא חָשׁוּב לִי.
📖 פֶּרֶק ה׳, הֲלָכוֹת א׳, ד׳

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

בְּנֹסַח הַבִּטּוּל יֵשׁ שְׁנֵי מִנְהָגִים. לְנֹסַח אַשְׁכְּנַז מַזְכִּירִים ״חֲמִירָא וַחֲמִיעָא״ – חָמֵץ וּשְׂאוֹר, וּמַזְכִּירִים גַּם בִּטּוּל וְגַם הֶפְקֵר. לְנֹסַח סְפָרַד מַזְכִּירִים רַק חָמֵץ, כִּי הוּא כּוֹלֵל גַּם אֶת הַשְּׂאוֹר, וְרַק בִּטּוּל, כִּי מָה שֶׁמְּבֻטָּל מִמֵּילָא מֻפְקָר.
לָמָּה בְּנֹסַח אַשְׁכְּנַז שְׁנֵיהֶם? יֵשׁ שְׁנֵי פֵּרוּשִׁים לַבִּטּוּל: לְרַשִׁ״י וְלָרַמְבַּ״ן – הוּא הוֹפֵךְ אֶת הֶחָמֵץ לְעָפָר; לְבַעֲלֵי הַתּוֹסָפוֹת – הוּא מַפְקִיר אוֹתוֹ. נֹסַח אַשְׁכְּנַז מִתְחַשֵּׁב בִּשְׁנֵי הַפֵּרוּשִׁים.
💛 אַהֲבַת הַשֵּׁם: שְׁנֵי הַנֻּסְחָאוֹת בְּנוּיִים עַל דִּבְרֵי גְּדוֹלֵי יִשְׂרָאֵל, וְכָל אֶחָד שׁוֹמֵר עַל מִנְהַג אֲבוֹתָיו. וּבִשְׁנֵיהֶם אֲנַחְנוּ אוֹמְרִים יַחַד: הֶחָמֵץ – כַּעֲפַר הָאָרֶץ.
📖 פֶּרֶק ה׳, הֲלָכוֹת א׳–ב׳

🗯️📜 לְשׁוֹן חֲכָמִים: בִּטּוּל חָמֵץ

״כָּל חֲמִירָא וַחֲמִיעָא דְּאִכָּא בִרְשׁוּתִי, דְּלָא חֲזִיתֵיהּ וּדְלָא בַעֲרְתֵּיהּ, לִבְטַל וְלֶהֱוֵי הֶפְקֵר כְּעַפְרָא דְּאַרְעָא״.
📖 בִּטּוּל חָמֵץ – אֲמִירָה בַּלֵּב וּבַפֶּה שֶׁהֶחָמֵץ שֶׁלָּנוּ חָשׁוּב כֶּעָפָר, וְאֵינֶנּוּ שֶׁלָּנוּ יוֹתֵר.
💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁהַלֵּב וְהַפֶּה יְדַבְּרוּ אוֹתוֹ דָּבָר, וְלֹא מַסְפִּיק לוֹמַר מִלִּים בְּלִי לְהָבִין אוֹתָן. 🧒 וְאַתָּה? כְּשֶׁמְּבַטְּלִים חָמֵץ, הַקְשֵׁב לְמַשְׁמָעוּת הַמִּלִּים, בְּכָל שָׂפָה שֶׁתִּרְצֶה.
📖 פֶּרֶק ה׳, הֲלָכוֹת א׳–ב׳

💬לְהָבִין וּלְהִתְכַּוֵּן

נֹסַח הַבִּטּוּל נִכְתַּב בַּאֲרָמִית, כִּי פַּעַם רֹב הָעָם הֵבִינוּ אֲרָמִית. אֶפְשָׁר לוֹמַר אוֹתוֹ בְּעִבְרִית אוֹ בְּכָל שָׂפָה – הָעִקָּר לְהָבִין. מִי שֶׁאָמַר אוֹתוֹ וְחָשַׁב שֶׁזּוֹ תְּפִלָּה לִקְרַאת הֶחָג – לֹא בִּטֵּל אֶת חֲמֵצוֹ.
הַבִּטּוּל צָרִיךְ לִהְיוֹת בְּלֵב שָׁלֵם: לֹא לַחְשֹׁב לְהִשְׁתַּמֵּשׁ בֶּחָמֵץ גַּם אַחֲרֵי פֶּסַח. לְכַתְּחִלָּה אוֹמְרִים אוֹתוֹ בַּפֶּה, וּבַעַל הֶחָמֵץ מְבַטֵּל בְּעַצְמוֹ.
💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה אֶת הַלֵּב. מִלִּים בְּלִי כַּוָּנָה הֵן כְּמוֹ גּוּף בְּלִי נְשָׁמָה.
📖 פֶּרֶק ה׳, הֲלָכוֹת ב׳–ג׳

🔥שְׂרֵפַת הֶחָמֵץ

בַּבֹּקֶר מְבַעֲרִים אֶת הֶחָמֵץ שֶׁנִּמְצָא בַּבְּדִיקָה וְאֶת מָה שֶׁנִּשְׁאַר מֵאֲרוּחַת הַבֹּקֶר. מִן הַדִּין אֶפְשָׁר לְבַעֵר גַּם בְּדֶרֶךְ אַחֶרֶת – לְפוֹרֵר וּלְפַזֵּר לָרוּחַ אוֹ לִזְרֹק לַיָּם. אֲבָל יִשְׂרָאֵל קְדוֹשִׁים נָהֲגוּ לְהַדֵּר וְלִשְׂרֹף, כִּי אֵין דָּבָר שֶׁמְּבַעֵר אֶת הֶחָמֵץ מִן הָעוֹלָם יוֹתֵר מֵהַשְּׂרֵפָה.
לָכֵן מִשְׁתַּדְּלִים שֶׁיִּהְיֶה בִּרְשׁוּתֵנוּ חָמֵץ בְּעֶרֶב פֶּסַח – כְּדֵי לְקַיֵּם בּוֹ אֶת מִצְוַת הַשְׁבָּתַת הֶחָמֵץ לְכָל הַדֵּעוֹת.
💛 אַהֲבַת הַשֵּׁם: הֶחָמֵץ רוֹמֵז לְיֵצֶר הָרָע. כְּשֶׁאֲנַחְנוּ רוֹאִים אוֹתוֹ נִשְׂרָף, אֲנַחְנוּ מְבַקְּשִׁים מֵה׳ שֶׁיַּעֲזֹר לָנוּ לְבַעֵר גַּם אֶת הָרַע שֶׁבַּלֵּב.
📖 פֶּרֶק ה׳, הֲלָכָה ד׳; פֶּרֶק ג׳, הֲלָכָה ה׳

🔎חָמֵץ שֶׁנִּמְצָא בְּפֶסַח

מִי שֶׁמָּצָא חָמֵץ בְּפֶסַח – מְבַעֵר אוֹתוֹ מִיָּד, וְלֹא מְבָרֵךְ, כִּי הַבְּרָכָה שֶׁלִּפְנֵי הַבְּדִיקָה חָלָה עַל כָּל הֶחָמֵץ שֶׁלּוֹ. אִם מָצָא אוֹתוֹ בְּשַׁבָּת אוֹ בְּיוֹם טוֹב (וּכְבָר בִּטֵּל אֶת חֲמֵצוֹ) – הֶחָמֵץ מֻקְצֶה: מְכַסִּים אוֹתוֹ בִּכְלִי, וְשׂוֹרְפִים בְּמוֹצָאֵי הֶחָג.
מִי שֶׁלֹּא בָּדַק בְּלֵיל י״ד – בּוֹדֵק בַּיּוֹם בִּבְרָכָה. וְאִם לֹא בָּדַק לִפְנֵי פֶּסַח – בּוֹדֵק בְּפֶסַח בִּבְרָכָה.
💛 אַהֲבַת הַשֵּׁם: גַּם מִי שֶׁפִּסְפֵּס – יָכוֹל לְתַקֵּן. הַתּוֹרָה תָּמִיד פּוֹתַחַת לָנוּ דֶּלֶת לַחֲזֹר וּלְקַיֵּם.
📖 פֶּרֶק ה׳, הֲלָכָה ו׳; פֶּרֶק ד׳, הֲלָכָה י״ב

🧠 בְּדֹק אֶת עַצְמְךָ

לָמָּה מְבַטְּלִים שׁוּב בַּבֹּקֶר, אִם כְּבָר בִּטְּלוּ בַּלַּיְלָה?
💡 בַּלַּיְלָה אִי אֶפְשָׁר לְבַטֵּל אֶת הֶחָמֵץ שֶׁשּׁוֹמְרִים לָאֲרוּחוֹת וְלַשְּׂרֵפָה – הוּא חָשׁוּב לָנוּ. בַּבֹּקֶר מְבַטְּלִים אֶת הַכֹּל, לְמִקְרֶה שֶׁמַּשֶּׁהוּ נִשְׁכַּח.
עַד מָתַי אֶפְשָׁר לוֹמַר אֶת הַבִּטּוּל שֶׁל הַבֹּקֶר?
💡 מֵהַשָּׁעָה הַשִּׁשִּׁית הֶחָמֵץ אָסוּר בַּהֲנָאָה, וְאֵינוֹ נֶחְשָׁב יוֹתֵר שֶׁלָּנוּ – וְלָכֵן כְּבָר אִי אֶפְשָׁר לְבַטֵּל אוֹתוֹ, אֶלָּא רַק לְבַעֵר אוֹתוֹ בְּפֹעַל.
יוֹסִי אָמַר ״כָּל חֲמִירָא״, וְחָשַׁב שֶׁזּוֹ תְּפִלָּה לִשְׁלוֹם הֶחָג. מָה הַדִּין?
💡 הַבִּטּוּל הוּא הַחְלָטָה בַּלֵּב שֶׁהֶחָמֵץ כְּמוֹ עָפָר. מִי שֶׁלֹּא הֵבִין מָה הוּא אוֹמֵר – לֹא בִּטֵּל. לָכֵן אֶפְשָׁר וְרָצוּי לוֹמַר אוֹתוֹ בְּשָׂפָה שֶׁמְּבִינִים.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

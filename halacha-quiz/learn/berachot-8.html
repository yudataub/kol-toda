בִּרְכַּת הַפֵּרוֹת וְשֶׁהַכֹּל

🍎 הִלְכוֹת בְּרָכוֹת · פֶּרֶק ח׳

🍎פְּרִי הָעֵץ וּפְרִי הָאֲדָמָה

עַל פֵּרוֹת הָאִילָן מְבָרְכִים "בּוֹרֵא פְּרִי הָעֵץ": עֲנָבִים, רִמּוֹנִים, תַּפּוּחִים, אֲגָסִים, אֱגוֹזִים וּשְׁקֵדִים. עַל פֵּרוֹת הָאָרֶץ מְבָרְכִים "בּוֹרֵא פְּרִי הָאֲדָמָה": תִּירָס, אֲפוּנָה, עַגְבָנִיּוֹת וּמְלָפְפוֹנִים.
עֵץ הוּא צֶמַח שֶׁהַגֶּזַע שֶׁלּוֹ נִשְׁאָר שָׁנִים רַבּוֹת. גַּם שִׂיחַ נָמוּךְ כָּזֶה הוּא עֵץ – וְלָכֵן עַל אוּכְמָנִיּוֹת מְבָרְכִים "הָעֵץ".
💛 אַהֲבַת הַשֵּׁם: הָעֵץ גָּדֵל לְאַט, שָׁנִים רַבּוֹת, וּפֵרוֹתָיו מְשֻׁבָּחִים. כָּךְ גַּם אֲנַחְנוּ גְּדֵלִים בַּתּוֹרָה לְאַט לְאַט, וְהַשֵּׁם מְחַכֶּה בְּאַהֲבָה לַפֵּרוֹת הַטּוֹבִים שֶׁלָּנוּ.
📖 פֶּרֶק ח׳, הֲלָכוֹת א׳–ב׳

🍌בָּנָנָה – פְּרִי הָאֲדָמָה

הַבָּנָנָה וְהָאֲנָנָס גְּבוֹהִים כְּמוֹ עֵץ, אֲבָל בְּסוֹף כָּל עוֹנָה הַגֶּזַע שֶׁלָּהֶם נוֹבֵל. לָכֵן פָּסְקוּ לְבָרֵךְ עֲלֵיהֶם "הָאֲדָמָה" – וְכָךְ גַּם עַל פַּפָּאיָה.
כְּשֶׁיֵּשׁ סָפֵק אִם פְּרִי הוּא שֶׁל עֵץ אוֹ שֶׁל אֲדָמָה – מְבָרְכִים "הָאֲדָמָה", כִּי גַּם הָעֵץ גָּדֵל מִן הָאֲדָמָה.
💛 אַהֲבַת הַשֵּׁם: כָּל פְּרִי, מִן הָעֵץ אוֹ מִן הָאֲדָמָה, הוּא מַתָּנָה שֶׁהַשֵּׁם הִכִּין לָנוּ. כְּשֶׁאָנוּ מְדַיְּקִים בַּבְּרָכָה, אָנוּ מַרְאִים שֶׁאָנוּ שָׂמִים לֵב לְכָל מַתָּנָה שֶׁלּוֹ.
📖 פֶּרֶק ח׳, הֲלָכָה ב׳

🔄טָעִיתִי בַּבְּרָכָה!

מִי שֶׁבֵּרַךְ "הָאֲדָמָה" עַל פְּרִי עֵץ – יָצָא, כִּי גַּם הָעֵץ גָּדֵל מִן הָאֲדָמָה. אֲבָל מִי שֶׁבֵּרַךְ "הָעֵץ" עַל פְּרִי אֲדָמָה – לֹא יָצָא, וְחוֹזֵר וּמְבָרֵךְ "הָאֲדָמָה".
מִי שֶׁבֵּרַךְ "שֶׁהַכֹּל" עַל כָּל מַאֲכָל – יָצָא. אֲבָל לְכַתְּחִלָּה מְבָרְכִים עַל כָּל מַאֲכָל אֶת בִּרְכָתוֹ, וּמִי שֶׁלֹּא יוֹדֵעַ – שׁוֹאֵל רַב.
💛 אַהֲבַת הַשֵּׁם: הַשֵּׁם שָׂמֵחַ כְּשֶׁאָנוּ מִשְׁתַּדְּלִים לְדַיֵּק. וְגַם כְּשֶׁאָנוּ טוֹעִים, הוּא נָתַן לָנוּ דֶּרֶךְ לְתַקֵּן – כִּי הוּא רוֹצֶה בַּתּוֹדָה שֶׁלָּנוּ.
📖 פֶּרֶק ח׳, הֲלָכוֹת ב׳, ד׳

🔑מִלּוֹת מַפְתֵּחַ שֶׁכְּדַאי לִזְכֹּר

בָּרָמָה הָאַחֲרוֹנָה תִּבְנֶה בְּעַצְמְךָ אֶת הַתְּשׁוּבָה, אוֹת אַחַר אוֹת. מִסְפַּר הַמִּשְׁבָּצוֹת מְגַלֶּה כַּמָּה אוֹתִיּוֹת יֵשׁ בַּתְּשׁוּבָה.
כְּשֶׁיֵּשׁ סָפֵק וְאֵין אֶת מִי לִשְׁאֹל – עַל פְּרִי שֶׁאוּלַי שֶׁל עֵץ וְאוּלַי שֶׁל אֲדָמָה מְבָרְכִים "הָאֲדָמָה", וְעַל מַאֲכָל אַחֵר – "שֶׁהַכֹּל".
💛 אַהֲבַת הַשֵּׁם: "שֶׁהַכֹּל נִהְיָה בִּדְבָרוֹ" הִיא בְּרָכָה כְּלָלִית: הִיא מַזְכִּירָה שֶׁהָעוֹלָם כֻּלּוֹ נִבְרָא בְּדִבּוּר אֶחָד שֶׁל הַשֵּׁם.
📖 פֶּרֶק ח׳, הֲלָכוֹת ב׳, ד׳

🥚עַל מָה מְבָרְכִים "שֶׁהַכֹּל"?

עַל מַאֲכָל שֶׁלֹּא גָּדֵל מִן הָאֲדָמָה: בָּשָׂר, עוֹף, דָּגִים, בֵּיצִים, חָלָב וּגְבִינָה. גַּם עַל מַיִם שֶׁשּׁוֹתִים לַצָּמָא, וְעַל פִּטְרִיּוֹת – כִּי הֵן לֹא יוֹנְקוֹת אֶת מְזוֹנָן מִן הָאֲדָמָה.
עַל מִיץ פֵּרוֹת, כְּמוֹ מִיץ תַּפּוּזִים, וְעַל קָפֶה וָתֵה – "שֶׁהַכֹּל", כִּי הַפְּרִי הָפַךְ לְמַשְׁקֶה. רַק הָעֲנָבִים עוֹלִים לְמַעֲלָה וְנַעֲשִׂים יַיִן, שֶׁבִּרְכָתוֹ "הַגֶּפֶן".
💛 אַהֲבַת הַשֵּׁם: "שֶׁהַכֹּל נִהְיָה בִּדְבָרוֹ" – הַשֵּׁם בָּרָא אֶת הָעוֹלָם כֻּלּוֹ בְּדִבּוּר. בְּכָל לְגִימַת מַיִם אֲנַחְנוּ נִזְכָּרִים שֶׁהַכֹּל בָּא מִמֶּנּוּ.
📖 פֶּרֶק ח׳, הֲלָכוֹת ג׳, ט״ו, י״ז

🍏קְלִפּוֹת, גַּרְעִינִים וּמִיץ בַּצַּלַּחַת

קְלִפַּת תַּפּוּחַ אוֹ מְלָפְפוֹן הִיא חֵלֶק מִן הַפְּרִי, וּמְבָרְכִים עָלֶיהָ כְּמוֹ עַל הַפְּרִי. עַל גַּרְעִינִים שְׁחֹרִים שֶׁל חַמָּנִיּוֹת – "הָאֲדָמָה". עַל צַבָּר טָעִים – "הָעֵץ".
הָאוֹכֵל אֶשְׁכּוֹלִית וּבֵרַךְ "הָעֵץ" – יָכוֹל לִשְׁתּוֹת אַחַר כָּךְ אֶת הַמִּיץ שֶׁנִּשְׁאַר בַּצַּלַּחַת, כִּי הַמִּיץ טָפֵל לַפְּרִי.
💛 אַהֲבַת הַשֵּׁם: הַשֵּׁם בָּרָא כָּל פְּרִי עַד הַפְּרָט הַקָּטָן – הַקְּלִפָּה, הַגַּרְעִין וְהַמִּיץ. מִי שֶׁמִּתְבּוֹנֵן בַּפְּרָטִים, מְגַלֶּה כַּמָּה חָכְמָה וְאַהֲבָה יֵשׁ בָּהֶם.
📖 פֶּרֶק ח׳, הֲלָכוֹת ה׳–ו׳, ט״ו

🍫חָתוּךְ – אוֹ הִשְׁתַּנָּה לְגַמְרֵי?

פְּרִי שֶׁחָתְכוּ אוֹ גֵּרְדוּ – בִּרְכָתוֹ נִשְׁאֶרֶת: עַל סָלָט יְרָקוֹת וְעַל גֶּזֶר מְגֻרָּד – "הָאֲדָמָה". גַּם עַל צִ׳יפְּס וּפִירֶה מִתַּפּוּחֵי אֲדָמָה – "הָאֲדָמָה".
מַאֲכָל שֶׁהִשְׁתַּנָּה כָּל כָּךְ עַד שֶׁלֹּא רוֹאִים מֵאֵיזֶה פְּרִי הוּא – "שֶׁהַכֹּל": שׁוֹקוֹלָד, סֻכָּר וְסֻכָּרִיּוֹת. בְּמַאֲכָלִים כְּמוֹ קוֹרְנְפְלֶקְס יֵשׁ מִנְהָגִים שׁוֹנִים – שׁוֹאֲלִים רַב.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁלֹּא רוֹאִים מֵאֵיפֹה בָּא הַמַּאֲכָל, אֲנַחְנוּ יוֹדְעִים: הַכֹּל בָּא מֵהַשֵּׁם. הַבְּרָכָה פּוֹקַחַת אֶת הָעֵינַיִם לִרְאוֹת אֶת הַטּוֹב שֶׁלּוֹ בְּכָל מָקוֹם.
📖 פֶּרֶק ח׳, הֲלָכוֹת ח׳, י׳–י״ב

🔀שְׁתֵּי דֵּעוֹת – וּמָה נוֹהֲגִים

פּוֹפְּקוֹרְן: רַבִּים כָּתְבוּ לְבָרֵךְ "הָאֲדָמָה", כִּי הַתִּירָס רַק הִתְנַפֵּחַ – וְהַמִּנְהָג הָרוֹוֵחַ "שֶׁהַכֹּל". פְּרִיכִיּוֹת אֹרֶז: יֵשׁ אוֹמְרִים "הָאֲדָמָה" וְיֵשׁ אוֹמְרִים "שֶׁהַכֹּל" – וְהַמִּנְהָג הָרוֹוֵחַ "שֶׁהַכֹּל". מָרָק צָלוּל: לְדַעַת רֹב הָרִאשׁוֹנִים "הָאֲדָמָה", וּמֵחֲמַת הַסָּפֵק נוֹהֲגִים "שֶׁהַכֹּל". קוֹרְנְפְלֶקְס: לְפִי הָרַב מֶלָמֵד "שֶׁהַכֹּל", וְיֵשׁ פּוֹסְקִים "הָאֲדָמָה".
יֶלֶד שְׂמָאלִי שֶׁמְּבָרֵךְ: יֵשׁ אוֹמְרִים שֶׁיַּחֲזִיק אֶת הַמַּאֲכָל בְּיַד שְׂמֹאל, הַיָּד הַחֲזָקָה שֶׁלּוֹ; וּלְמִנְהַג הַמְּקֻבָּלִים – בְּיַד יָמִין. כָּל אֶחָד רַשַּׁאי לִבְחֹר אֶת מִנְהָגוֹ.
💛 אַהֲבַת הַשֵּׁם: כָּל הַמִּנְהָגִים נוֹסְדוּ עַל יְדֵי גְּדוֹלֵי תּוֹרָה. כָּל אֶחָד שׁוֹמֵר עַל מִנְהַג מִשְׁפַּחְתּוֹ בְּאַהֲבָה, וְכֻלָּנוּ עוֹבְדִים אֶת אוֹתוֹ ה׳.
📖 פֶּרֶק ח׳, הֲלָכוֹת י״ב–י״ג, ט״ז; פֶּרֶק ט׳, הֲלָכָה ד׳

🥔כְּדֶרֶךְ אֲכִילָתוֹ

אוֹכְלִים פְּרִי כְּמוֹ שֶׁרְגִילִים לְאָכְלוֹ – מְבָרְכִים אֶת בִּרְכָתוֹ. תַּפּוּחַ אֲדָמָה רְגִילִים לֶאֱכֹל מְבֻשָּׁל, וְלָכֵן עַל תַּפּוּחַ אֲדָמָה מְבֻשָּׁל – "הָאֲדָמָה", וְעַל תַּפּוּחַ אֲדָמָה חַי – "שֶׁהַכֹּל".
תַּפּוּחִים וַאֲגָסִים אוֹכְלִים גַּם חַיִּים וְגַם מְבֻשָּׁלִים – וּבִשְׁנֵי הַמַּצָּבִים מְבָרְכִים "הָעֵץ".
💛 אַהֲבַת הַשֵּׁם: הַשֵּׁם נָתַן לָנוּ פֵּרוֹת וְגַם חָכְמָה לְהָכִין מֵהֶם מַאֲכָלִים טְעִימִים. הַבְּרָכָה מוֹדָה לוֹ עַל הַפְּרִי כְּמוֹ שֶׁהוּא מַגִּיעַ אֵלֵינוּ, בְּדֶרֶךְ הַטּוֹבָה בְּיוֹתֵר.
📖 פֶּרֶק ח׳, הֲלָכָה י״ד

🧠 בְּדֹק אֶת עַצְמְךָ

אָרִי בֵּרֵךְ "בּוֹרֵא פְּרִי הָעֵץ" עַל מְלָפְפוֹן, וְאָז נִזְכַּר שֶׁטָּעָה. מָה עָלָיו לַעֲשׂוֹת?
💡 פְּרִי הָאֲדָמָה לֹא גָּדֵל עַל עֵץ, וְלָכֵן "הָעֵץ" עָלָיו אֵינָהּ בְּרָכָה נְכוֹנָה. רַק בַּמִּקְרֶה הַהָפוּךְ – "הָאֲדָמָה" עַל פְּרִי עֵץ – יוֹצְאִים, כִּי גַּם הָעֵץ גָּדֵל מִן הָאֲדָמָה.
דָּנִי בֵּרֵךְ "הָעֵץ" וְאָכַל אֶשְׁכּוֹלִית. בַּסּוֹף נִשְׁאַר מִיץ בַּצַּלַּחַת, וְהוּא רוֹצֶה לִשְׁתּוֹת אוֹתוֹ. מָה הַדִּין?
💡 מִיץ שֶׁנִּסְחַט לְבַד – "שֶׁהַכֹּל". אֲבָל מִיץ שֶׁיָּצָא מֵהַפְּרִי בִּזְמַן הָאֲכִילָה טָפֵל לַפְּרִי, וְנִפְטָר בַּבְּרָכָה שֶׁבֵּרַךְ עַל הָאֶשְׁכּוֹלִית.
לַאֲרוּחַת צָהֳרַיִם אָכְלָה הֲדַסָּה צַלַּחַת אֹרֶז מְבֻשָּׁל. מָה תְּבָרֵךְ לְפָנָיו וְאַחֲרָיו?
💡 עַל אֹרֶז מְבֻשָּׁל קָבְעוּ חֲכָמִים "מְזוֹנוֹת", כִּי הוּא מַזִּין כְּמוֹ מִינֵי דָּגָן. אֲבָל הוּא אֵינוֹ מִשִּׁבְעַת הַמִּינִים, וְלָכֵן אַחֲרָיו מְבָרְכִים "בּוֹרֵא נְפָשׁוֹת" – וְלֹא "עַל הַמִּחְיָה".

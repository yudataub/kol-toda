הַדְלָקַת נֵרוֹת חֲנֻכָּה

🕎 הִלְכוֹת חֲנֻכָּה · פֶּרֶק י״ב

🕯️מִצְוָה מִדִּבְרֵי חֲכָמִים – ״וְצִוָּנוּ״

הַדְלָקַת נֵר חֲנֻכָּה הִיא מִצְוָה שֶׁחֲכָמִים תִּקְּנוּ, וּבְכָל זֹאת מְבָרְכִים ״וְצִוָּנוּ״. כִּי הַתּוֹרָה נָתְנָה לַחֲכָמִים סַמְכוּת לְתַקֵּן מִצְווֹת, כְּמוֹ שֶׁנֶּאֱמַר: ״לֹא תָסוּר מִן הַדָּבָר אֲשֶׁר יַגִּידוּ לְךָ״.
גַּם נָשִׁים חַיָּבוֹת בַּמִּצְוָה, כִּי גַּם הֵן הָיוּ שֻׁתָּפוֹת בַּנֵּס. אִשָּׁה נְשׂוּאָה נוֹהֶגֶת לָצֵאת בְּהַדְלָקַת בַּעְלָהּ.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנַחְנוּ שׁוֹמְעִים לַחֲכָמִים, אֲנַחְנוּ שׁוֹמְעִים לַה׳ עַצְמוֹ. הַתּוֹרָה שֶׁבְּעַל פֶּה הִיא חֵלֶק מֵהַמַּתָּנָה שֶׁה׳ נָתַן לָנוּ.
📖 פֶּרֶק י״ב, הֲלָכָה א׳

🪙גַּם הֶעָנִי מַדְלִיק

מִצְוַת נֵר חֲנֻכָּה גְּדוֹלָה מְאוֹד, כִּי יֵשׁ בָּהּ פִּרְסוּם הַנֵּס. לָכֵן גַּם עָנִי שֶׁאֵין לוֹ כֶּסֶף לְנֵרוֹת צָרִיךְ לְבַקֵּשׁ מֵאֲחֵרִים אוֹ לִמְכֹּר אֲפִלּוּ אֶת בִּגְדוֹ – וּמַסְפִּיק שֶׁיַּדְלִיק נֵר אֶחָד בְּכָל לַיְלָה.
גַּם יְהוּדִי שֶׁגָּר לְבַד בְּמָקוֹם שֶׁאַף אֶחָד לֹא עוֹבֵר בּוֹ – מַדְלִיק נֵרוֹת, כְּדֵי לִזְכֹּר בְּעַצְמוֹ אֶת הַנֵּס.
💛 אַהֲבַת הַשֵּׁם: נֵר קָטָן אֶחָד שֶׁל עָנִי יָקָר מְאוֹד לִפְנֵי ה׳. אֵין יְהוּדִי שֶׁנִּשְׁאָר בְּלִי חֵלֶק בָּאוֹר שֶׁל חֲנֻכָּה.
📖 פֶּרֶק י״ב, הֲלָכָה א׳

🕯️שָׁלֹשׁ מַדְרֵגוֹת

הַחוֹבָה: נֵר אֶחָד בְּכָל לַיְלָה לְכָל הַבַּיִת. ״מְהַדְּרִין״: נֵר לְכָל אֶחָד מִבְּנֵי הַבַּיִת. ״מְהַדְּרִין מִן הַמְּהַדְּרִין״: לְפִי מִסְפַּר הַיָּמִים. כָּל יִשְׂרָאֵל נוֹהֲגִים כְּ״מְהַדְּרִין מִן הַמְּהַדְּרִין״.
בֵּית שַׁמַּאי אָמְרוּ: מַתְחִילִים בִּשְׁמוֹנָה וּפוֹחֲתִים. בֵּית הִלֵּל אָמְרוּ: מַתְחִילִים בְּאֶחָד וּמוֹסִיפִים – וְכָךְ נוֹהֲגִים. בְּסַךְ הַכֹּל מַדְלִיקִים ל״ו נֵרוֹת.
💛 אַהֲבַת הַשֵּׁם: בְּכָל יוֹם הַנֵּס גָּדַל, וּבְכָל יוֹם אֲנַחְנוּ מוֹסִיפִים אוֹר. ״מַעֲלִין בַּקֹּדֶשׁ״ – עוֹלִים בִּקְדֻשָּׁה, וְלֹא יוֹרְדִים.
📖 פֶּרֶק י״ב, הֲלָכָה ב׳

🔢בֵּית הִלֵּל וְהַשַּׁמָּשׁ

נוֹהֲגִים כְּבֵית הִלֵּל: בַּלַּיְלָה הָרִאשׁוֹן נֵר אֶחָד, וּמוֹסִיפִים עַד שְׁמוֹנָה. לְיַד הַנֵּרוֹת מַדְלִיקִים שַׁמָּשׁ, כְּדֵי שֶׁאִם צְרִיכִים אוֹר – יִשְׁתַּמְּשׁוּ בְּאוֹרוֹ.
בְּבֵית הַכְּנֶסֶת מַנִּיחִים אֶת הַחֲנֻכִּיָּה בַּכֹּתֶל הַדְּרוֹמִי, כְּמוֹ מְקוֹם הַמְּנוֹרָה בַּמִּקְדָּשׁ.
💛 אַהֲבַת הַשֵּׁם: הַשַּׁמָּשׁ מְשָׁרֵת אֶת כָּל הַנֵּרוֹת וּמַדְלִיק אוֹתָם, אֲבָל עוֹמֵד בַּצַּד. כָּךְ גַּם אֲנַחְנוּ יְכוֹלִים לְהַדְלִיק אוֹר אֵצֶל אֲחֵרִים – בַּעֲנָוָה.
📖 פֶּרֶק י״ב, הֲלָכוֹת ב׳, י״א, י״ד

🔀מִי מַדְלִיק? שְׁנֵי מִנְהָגִים

לְמִנְהַג הָאַשְׁכְּנַזִּים, כָּל אֶחָד מִבְּנֵי הַבַּיִת מַדְלִיק בִּבְרָכָה חֲנֻכִּיָּה מִשֶּׁלּוֹ, וּמַרְחִיקִים אֶת הַחֲנֻכִּיּוֹת זוֹ מִזּוֹ. גַּם יְלָדִים מִגִּיל חִנּוּךְ, בְּעֵרֶךְ מִגִּיל שֵׁשׁ, מַדְלִיקִים בִּבְרָכָה.
לְמִנְהַג הַסְּפָרַדִּים, אֶחָד – גְּדוֹל הַבַּיִת – מַדְלִיק לְכֻלָּם, כְּדֵי שֶׁיִּהְיֶה בָּרוּר כַּמָּה נֵרוֹת יֵשׁ, כְּמִסְפַּר הַיָּמִים.
💛 אַהֲבַת הַשֵּׁם: שְׁנֵי הַמִּנְהָגִים בְּנוּיִים עַל אַהֲבָה לַמִּצְוָה: זֶה רוֹצֶה שֶׁכֻּלָּם יִשְׁתַּתְּפוּ, וְזֶה רוֹצֶה שֶׁהַנֵּס יִתְפַּרְסֵם בְּיוֹתֵר. וְכָל אֶחָד שׁוֹמֵר עַל מִנְהַג אֲבוֹתָיו.
📖 פֶּרֶק י״ב, הֲלָכוֹת ג׳–ד׳

🧒יֶלֶד סְפָרַדִּי וְהַחֲנֻכִּיָּה שֶׁלּוֹ

בְּבַיִת סְפָרַדִּי, אִם הַיְּלָדִים רוֹצִים לְהַדְלִיק – אֶפְשָׁר לָתֵת לָהֶם חֲנֻכִּיָּה מִשֶּׁלָּהֶם, וּמַרְחִיקִים אוֹתָהּ מֵהַחֲנֻכִּיָּה שֶׁל אַבָּא. לְמִנְהַג רַבִּים מֵהַסְּפָרַדִּים לֹא מְבָרְכִים.
הָרַב מָרְדְּכַי אֵלִיָּהוּ זַצַ״ל הוֹרָה שֶׁיְּלָדִים עַד גִּיל בַּר מִצְוָה רַשָּׁאִים לְבָרֵךְ.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁאַבָּא מַדְלִיק לְכֻלָּם, הַלֵּב שֶׁל הַיֶּלֶד רוֹצֶה לְהִשְׁתַּתֵּף. הָרָצוֹן הַזֶּה לְהַדְלִיק הוּא בְּעַצְמוֹ אוֹר גָּדוֹל.
📖 פֶּרֶק י״ב, הֲלָכָה ג׳

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

מִי מַדְלִיק: לְמִנְהַג הַסְּפָרַדִּים, גְּדוֹל הַבַּיִת מַדְלִיק לְכֻלָּם, כְּדֵי שֶׁמִּסְפַּר הַנֵּרוֹת יַרְאֶה בְּדִיּוּק אֶת מִסְפַּר הַיָּמִים. לְמִנְהַג הָאַשְׁכְּנַזִּים, כָּל אֶחָד מִבְּנֵי הַבַּיִת מַדְלִיק חֲנֻכִּיָּה מִשֶּׁלּוֹ בִּבְרָכָה.
נֻסַּח הַבְּרָכָה: לְמִנְהַג הָאַשְׁכְּנַזִּים ״לְהַדְלִיק נֵר שֶׁל חֲנֻכָּה״, וּבְנֻסַּח סְפָרַד ״לְהַדְלִיק נֵר חֲנֻכָּה״.
💛 אַהֲבַת הַשֵּׁם: שְׁנֵי הַמִּנְהָגִים בְּנוּיִים עַל דִּבְרֵי גְּדוֹלֵי הַפּוֹסְקִים, וְכָל אֶחָד שׁוֹמֵר עַל מִנְהַג אֲבוֹתָיו. בְּכָל בַּיִת בְּיִשְׂרָאֵל – אוֹתוֹ אוֹר, וְאוֹתָהּ אַהֲבָה לַה׳.
📖 פֶּרֶק י״ב, הֲלָכוֹת ג׳–ה׳

🔀עוֹד דֵּעוֹת – וְכָל אַחַת בְּדֶרֶךְ הַהֲלָכָה

יֶלֶד בְּבַיִת סְפָרַדִּי שֶׁמַּדְלִיק חֲנֻכִּיָּה מִשֶּׁלּוֹ: לְמִנְהַג רַבִּים מֵהַסְּפָרַדִּים – בְּלִי בְּרָכָה. הָרַב מָרְדְּכַי אֵלִיָּהוּ הוֹרָה שֶׁיְּלָדִים עַד בַּר מִצְוָה רַשָּׁאִים לְבָרֵךְ. נֵרוֹת שֶׁכָּבוּ בָּרוּחַ: לְדַעַת רֹב הַפּוֹסְקִים מַדְלִיקִים שׁוּב בִּבְרָכָה, וּלְפִי הָרַב מֶלָמֵד – בְּלִי בְּרָכָה.
חֲנֻכִּיָּה בַּחֲתֻנָּה אוֹ בִּמְסִבָּה: לְדַעַת רַבִּים מֵרַבָּנֵי דּוֹרֵנוּ – מַדְלִיקִים בְּלִי בְּרָכָה, כִּי רַק בְּבֵית הַכְּנֶסֶת נָהֲגוּ לְבָרֵךְ. לְדַעַת כַּמָּה פּוֹסְקִים – מֻתָּר לְבָרֵךְ, כִּי הַמַּטָּרָה הִיא פִּרְסוּם הַנֵּס.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁיֵּשׁ שְׁתֵּי דֵּעוֹת, כָּל אֶחָד שׁוֹאֵל אֶת הָרַב שֶׁלּוֹ וְהוֹלֵךְ בְּדַרְכּוֹ – וּמְכַבֵּד אֶת מִי שֶׁנּוֹהֵג אַחֶרֶת.
📖 פֶּרֶק י״ב, הֲלָכוֹת ג׳, ט׳, ט״ו

👧בָּנוֹת וְנֵר חֲנֻכָּה

לְמִנְהַג אַשְׁכְּנַז, שֶׁבּוֹ כָּל אֶחָד מַדְלִיק, בְּהַרְבֵּה בָּתִּים בָּנוֹת לֹא נָהֲגוּ לְהַדְלִיק. אֲבָל בַּת שֶׁרוֹצָה – רַשָּׁאִית לְהַדְלִיק בִּבְרָכָה. וְטוֹב לְעוֹדֵד בָּנוֹת מִגִּיל חִנּוּךְ לְהַדְלִיק, לְפָחוֹת עַד בַּת מִצְוָה.
הַהַדְלָקָה מְקַשֶּׁרֶת אֶת הַבָּנוֹת לַתּוֹרָה וְלַמִּצְווֹת. וְאִם יִרְצוּ לְהַדְלִיק גַּם אַחֲרֵי בַּת מִצְוָה – תָּבוֹא עֲלֵיהֶן בְּרָכָה.
💛 אַהֲבַת הַשֵּׁם: גַּם הַנָּשִׁים הָיוּ שֻׁתָּפוֹת בַּנֵּס. כְּשֶׁבַּת מַדְלִיקָה נֵר מִשֶּׁלָּהּ, הִיא מַרְגִּישָׁה שֶׁגַּם הָאוֹר הַזֶּה שֶׁלָּהּ – מַתָּנָה מֵה׳.
📖 פֶּרֶק י״ב, הֲלָכָה ד׳

🙌הַבְּרָכוֹת

מְבָרְכִים ״לְהַדְלִיק נֵר שֶׁל חֲנֻכָּה״ – וּבְנֻסַּח סְפָרַד: ״לְהַדְלִיק נֵר חֲנֻכָּה״ – וְ״שֶׁעָשָׂה נִסִּים לַאֲבוֹתֵינוּ״. בַּלַּיְלָה הָרִאשׁוֹן מוֹסִיפִים ״שֶׁהֶחֱיָנוּ״. מִיָּד אַחֲרֵי הַבְּרָכוֹת מַדְלִיקִים, בְּלִי לְדַבֵּר בָּאֶמְצַע.
אַחֲרֵי הַנֵּר הָרִאשׁוֹן מַתְחִילִים לוֹמַר ״הַנֵּרוֹת הַלָּלוּ״, וּמַמְשִׁיכִים לְהַדְלִיק. וּמַדְלִיקִים כָּל נֵר הֵיטֵב, עַד שֶׁהָאֵשׁ אוֹחֶזֶת בָּרֹב הַפְּתִילָה.
💛 אַהֲבַת הַשֵּׁם: הַבְּרָכָה הָרִאשׁוֹנָה עַל הַמִּצְוָה, וְהַשְּׁנִיָּה – תּוֹדָה לַה׳ עַל הַנִּסִּים. כָּךְ הַהַדְלָקָה הוֹפֶכֶת לְשִׂיחָה שֶׁל אַהֲבָה עִם ה׳.
📖 פֶּרֶק י״ב, הֲלָכָה ה׳

👨‍👦כָּל הַמִּשְׁפָּחָה מִתְאַסֶּפֶת

מִשְׁתַּדְּלִים שֶׁכָּל בְּנֵי הַבַּיִת יִתְאַסְּפוּ לְהַדְלָקַת הַנֵּרוֹת: יִשְׁמְעוּ אֶת הַבְּרָכוֹת, יַעֲנוּ ״אָמֵן״ וְיִרְאוּ אֶת הַנֵּרוֹת. זֶה כָּבוֹד לַמִּצְוָה וּפִרְסוּם לַנֵּס.
מִי שֶׁלֹּא מַדְלִיק בְּעַצְמוֹ – כְּמוֹ בְּנֵי בַּיִת שֶׁיּוֹצְאִים בְּהַדְלָקַת אַבָּא – צָרִיךְ לִשְׁמֹעַ אֶת הַבְּרָכוֹת, וְכָךְ הוּא מִשְׁתַּתֵּף בַּתּוֹדָה עַל הַנֵּס.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁכָּל הַמִּשְׁפָּחָה עוֹמֶדֶת יַחַד לְיַד הָאוֹר וְעוֹנָה ״אָמֵן״, הַבַּיִת כֻּלּוֹ מוֹדֶה לַה׳ בְּקוֹל אֶחָד.
📖 פֶּרֶק י״ב, הֲלָכָה ו׳

🫒שֶׁמֶן זַיִת וְנֵרוֹת

כָּל הַשְּׁמָנִים וְהַפְּתִילוֹת כְּשֵׁרִים לְנֵר חֲנֻכָּה. נֵרוֹת שַׁעֲוָה אוֹ פָּרָפִין מְאִירִים יָפֶה, וְעוֹד יוֹתֵר מְהֻדָּר שֶׁמֶן זַיִת: הָאוֹר שֶׁלּוֹ צָלוּל, וְהוּא מַזְכִּיר אֶת נֵס פַּךְ הַשֶּׁמֶן.
צָרִיךְ שֶׁיִּהְיֶה בַּנֵּר מַסְפִּיק כְּדֵי לִדְלֹק חֲצִי שָׁעָה. נֵר קָטָן שֶׁיִּדְלַק פָּחוֹת – מַדְלִיקִים בְּלִי בְּרָכָה.
💛 אַהֲבַת הַשֵּׁם: כְּכָל שֶׁהַנֵּר מֵאִיר יוֹתֵר, הַנֵּס מִתְפַּרְסֵם יוֹתֵר. אֲנַחְנוּ בּוֹחֲרִים בַּיָּפֶה בְּיוֹתֵר – כִּי זוֹ מַתָּנָה לַה׳.
📖 פֶּרֶק י״ב, הֲלָכָה ז׳

🪔נֵר חֲנֻכָּה וְנֵר שַׁבָּת

נֵר שַׁבָּת נוֹעַד לְהָאִיר. אִם לֹא יִדְלַק יָפֶה, יֵשׁ חֲשָׁשׁ שֶׁיַּטֶּה אוֹתוֹ כְּדֵי לְתַקֵּן וִיחַלֵּל שַׁבָּת, וְלָכֵן לֹא מַדְלִיקִים אוֹתוֹ בְּכָל שֶׁמֶן. בְּנֵר חֲנֻכָּה אָסוּר לְהִשְׁתַּמֵּשׁ – וְלָכֵן כָּל שֶׁמֶן וּפְתִילָה שֶׁדּוֹלְקִים חֲצִי שָׁעָה כְּשֵׁרִים.
וְכֵן בְּחַשְׁמַל: לְנֵר שַׁבָּת אֶפְשָׁר בִּשְׁעַת הַצֹּרֶךְ לְהִשְׁתַּמֵּשׁ בְּנוּרָה, כִּי הוּא לְהָאִיר. נֵר חֲנֻכָּה – לֹא, כִּי הוּא זֵכֶר לַנֵּס.
💛 אַהֲבַת הַשֵּׁם: כָּל מִצְוָה יֵשׁ לָהּ תַּפְקִיד מְיֻחָד. כְּשֶׁמְּבִינִים לָמָּה – הַמִּצְוָה נִכְנֶסֶת לַלֵּב.
📖 פֶּרֶק י״ב, הֲלָכוֹת ז׳–ח׳

💡נוּרוֹת חַשְׁמַל

לְדַעַת רֹב הַפּוֹסְקִים לֹא יוֹצְאִים יְדֵי חוֹבָה בְּנוּרוֹת חַשְׁמַל: אֵין בָּהֶן פְּתִילָה וְשֶׁמֶן, וְנֵר חֲנֻכָּה צָרִיךְ לִהְיוֹת דּוֹמֶה לְנֵרוֹת הַמִּקְדָּשׁ. מִי שֶׁאֵין לוֹ שׁוּם נֵר כָּשֵׁר – יַדְלִיק נוּרוֹת בְּלִי בְּרָכָה.
חֲנֻכִּיּוֹת חַשְׁמַל גְּדוֹלוֹת בִּמְקוֹמוֹת צִבּוּרִיִּים – לֹא מְקַיְּמִים בָּהֶן אֶת הַמִּצְוָה, אֲבָל יֵשׁ בָּהֶן מַעֲלָה: הֵן מַזְכִּירוֹת לָרַבִּים אֶת הַנֵּס.
💛 אַהֲבַת הַשֵּׁם: נֵר חֲנֻכָּה הוּא לֹא סְתָם אוֹר – הוּא זֵכֶר לַמְּנוֹרָה שֶׁבַּמִּקְדָּשׁ. לָכֵן מַדְלִיקִים אֵשׁ אֲמִתִּית, כְּמוֹ הַכֹּהֲנִים.
📖 פֶּרֶק י״ב, הֲלָכָה ח׳

🔥הַהַדְלָקָה הִיא הַמִּצְוָה

הַמִּצְוָה הִיא לְהַדְלִיק. לָכֵן אִם הַנֵּרוֹת יָכְלוּ לִדְלֹק חֲצִי שָׁעָה, וּמִישֶׁהוּ הִפִּיל אוֹתָם בְּטָעוּת וְהֵם כָּבוּ – כְּבָר יָצָא. אֲבָל רָאוּי לְהַחְמִיר וּלְהַדְלִיק שׁוּב. וּמִי שֶׁהִדְלִיק בְּמָקוֹם שֶׁיֵּשׁ רוּחַ, וְהַנֵּרוֹת כָּבוּ – לֹא יָצָא. לְדַעַת רֹב הַפּוֹסְקִים מַדְלִיק שׁוּב בִּבְרָכָה, וְהָרַב מֶלָמֵד כּוֹתֵב שֶׁבִּגְלַל הַסָּפֵק מַדְלִיק שׁוּב בְּלִי בְּרָכָה.
נֵר שֶׁדָּלַק עוֹד מִלִּפְנֵי כֵן, לֹא לְשֵׁם מִצְוָה – מְכַבִּים אוֹתוֹ וּמַדְלִיקִים מֵחָדָשׁ לְשֵׁם מִצְוָה. וּמַדְלִיקִים בַּמָּקוֹם שֶׁבּוֹ הַנֵּרוֹת יַעַמְדוּ.
💛 אַהֲבַת הַשֵּׁם: ה׳ מְחַכֶּה לַמַּעֲשֶׂה שֶׁלָּנוּ. בָּרֶגַע שֶׁאָנוּ מַדְלִיקִים לְשֵׁם מִצְוָה, קִיַּמְנוּ אֶת רְצוֹנוֹ.
📖 פֶּרֶק י״ב, הֲלָכָה ט׳

🚫לֹא מִשְׁתַּמְּשִׁים בָּאוֹר

אָסוּר לְהִשְׁתַּמֵּשׁ בְּאוֹר נֵרוֹת חֲנֻכָּה – לֹא לִסְפֹּר כֶּסֶף לְאוֹרָם, וַאֲפִלּוּ לֹא לִלְמֹד בְּסֵפֶר. הֵם מְקֻדָּשִׁים לַמִּצְוָה, כְּמוֹ הַנֵּרוֹת בִּמְנוֹרַת הַמִּקְדָּשׁ, וְהֵם נוֹעֲדוּ לְפִרְסוּם הַנֵּס.
שֶׁמֶן וּפְתִילוֹת שֶׁנִּשְׁאֲרוּ אַחֲרֵי הַהַדְלָקָה – לֹא מִשְׁתַּמְּשִׁים בָּהֶם לְדָבָר אַחֵר. מָה שֶׁנִּשְׁאַר בַּחֲבִילָה אוֹ בַּבַּקְבּוּק – מֻתָּר לְכָל שִׁמּוּשׁ.
💛 אַהֲבַת הַשֵּׁם: יֵשׁ אוֹר שֶׁהוּא רַק לַה׳. כְּשֶׁאֲנַחְנוּ מִסְתַּכְּלִים בַּנֵּרוֹת וְלֹא מִשְׁתַּמְּשִׁים בָּהֶם, אֲנַחְנוּ לוֹמְדִים לְהִתְבּוֹנֵן – וּלְהוֹדוֹת.
📖 פֶּרֶק י״ב, הֲלָכָה י׳

🕎הַחֲנֻכִּיָּה וְהַשַּׁמָּשׁ

מַשְׁאִירִים רֶוַח בֵּין הַנֵּרוֹת, כְּדֵי שֶׁכָּל נֵר יֵרָאֶה לְבַד, וְשֶׁהַלֶּהָבוֹת לֹא יִתְאַחֲדוּ כְּמוֹ מְדוּרָה. לְכַתְּחִלָּה מַעֲמִידִים אֶת הַנֵּרוֹת בְּשׁוּרָה שָׁוָה, כְּדֵי שֶׁיִּרְאוּ אֶת מִסְפָּרָם.
מַדְלִיקִים גַּם שַׁמָּשׁ – נֵר שֶׁמֻּתָּר לְהִשְׁתַּמֵּשׁ לְאוֹרוֹ. מַנִּיחִים אוֹתוֹ בְּגֹבַהּ אַחֵר אוֹ בְּרִחוּק, וְנָהֲגוּ לְהַנִּיחַ אוֹתוֹ גָּבוֹהַּ יוֹתֵר.
💛 אַהֲבַת הַשֵּׁם: כָּל נֵר עוֹמֵד לְבַד וְנִרְאֶה בִּפְנֵי עַצְמוֹ – כְּמוֹ כָּל יוֹם שֶׁל הַנֵּס, וּכְמוֹ כָּל יְהוּדִי, שֶׁיֵּשׁ לוֹ אוֹר מְיֻחָד מִשֶּׁלּוֹ.
📖 פֶּרֶק י״ב, הֲלָכָה י״א

↩️מֵאֵיפֹה מַתְחִילִים?

בַּלַּיְלָה הָרִאשׁוֹן קוֹבְעִים אֶת הַנֵּר בַּצַּד הַיְּמָנִי שֶׁל הַחֲנֻכִּיָּה (יְמִין הַמַּדְלִיק). בְּכָל לַיְלָה מוֹסִיפִים נֵר מִשְּׂמֹאל, מַדְלִיקִים קֹדֶם אֶת הַנֵּר הֶחָדָשׁ, וְאַחַר כָּךְ פּוֹנִים יָמִינָה.
טוֹב לַעֲמֹד בִּזְמַן הַבְּרָכוֹת לְצַד שְׂמֹאל שֶׁל הַחֲנֻכִּיָּה, כָּךְ שֶׁהַנֵּר הַקָּרוֹב הוּא הַנֵּר שֶׁמַּדְלִיקִים רִאשׁוֹן.
💛 אַהֲבַת הַשֵּׁם: מַתְחִילִים בַּנֵּר הֶחָדָשׁ, כִּי הוּא מְסַפֵּר שֶׁהַנֵּס גָּדַל עוֹד יוֹם. כָּל יוֹם חָדָשׁ הוּא עוֹד הִזְדַּמְּנוּת לְהוֹסִיף אוֹר.
📖 פֶּרֶק י״ב, הֲלָכָה י״ב

⏰לִפְנֵי הַהַדְלָקָה

בַּחֲצִי הַשָּׁעָה שֶׁלִּפְנֵי זְמַן הַהַדְלָקָה אָסוּר לְהַתְחִיל לֶאֱכֹל אֲרוּחָה, אֲפִלּוּ קַלָּה, שֶׁמָּא יִשְׁכַּח לְהַדְלִיק. אֲבָל פֵּרוֹת וִירָקוֹת מֻתָּר לֶאֱכֹל בְּלִי הַגְבָּלָה, וְלֶחֶם אוֹ מְזוֹנוֹת – עַד כַּבֵּיצָה. גַּם לֹא הוֹלְכִים לִישֹׁן וְלֹא מַתְחִילִים עֲבוֹדָה אֲרֻכָּה.
כְּשֶׁמַּגִּיעַ הַזְּמַן מִזְדָּרְזִים לְהַדְלִיק, וַאֲפִלּוּ לִלְמֹד תּוֹרָה אָסוּר אָז – חוּץ מִשִּׁעוּר קָבוּעַ, שֶׁבְּסוֹפוֹ מַזְכִּירִים לְכֻלָּם לְהַדְלִיק.
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁאוֹהֵב מִצְוָה לֹא מְחַכֶּה לָהּ ״כְּשֶׁיִּהְיֶה לוֹ זְמַן״. הוּא מְכִין אֶת עַצְמוֹ וְרָץ אֵלֶיהָ בְּשִׂמְחָה.
📖 פֶּרֶק י״ב, הֲלָכָה י״ג

🕍בְּבֵית הַכְּנֶסֶת

נָהֲגוּ לְהַדְלִיק בְּבֵית הַכְּנֶסֶת בְּכָל הַבְּרָכוֹת, כְּדֵי לְפַרְסֵם אֶת הַנֵּס, וּמַנִּיחִים אֶת הַחֲנֻכִּיָּה בַּכֹּתֶל הַדְּרוֹמִי, כְּמוֹ הַמְּנוֹרָה בַּמִּקְדָּשׁ. מַדְלִיקִים לִפְנֵי עַרְבִית, כְּשֶׁיֵּשׁ שָׁם הַרְבֵּה אֲנָשִׁים, וּצְרִיכִים לִהְיוֹת שָׁם לְפָחוֹת עֲשָׂרָה.
הַהַדְלָקָה בְּבֵית הַכְּנֶסֶת לֹא פּוֹטֶרֶת – גַּם מִי שֶׁהִדְלִיק שָׁם חוֹזֵר וּמַדְלִיק בְּבֵיתוֹ בִּבְרָכָה. בְּעֶרֶב שַׁבָּת מַדְלִיקִים לִפְנֵי מִנְחָה, וּבְמוֹצָאֵי שַׁבָּת – אַחֲרֵי עַרְבִית.
💛 אַהֲבַת הַשֵּׁם: בֵּית הַכְּנֶסֶת הוּא ״מִקְדָּשׁ מְעַט״. כְּשֶׁהַחֲנֻכִּיָּה דּוֹלֶקֶת בּוֹ, אֲנַחְנוּ מַרְגִּישִׁים קְצָת כְּמוֹ בַּמִּקְדָּשׁ שֶׁל הַחַשְׁמוֹנָאִים.

🎉חֲנֻכִּיָּה בִּמְסִבָּה

רַבִּים מַדְלִיקִים חֲנֻכִּיָּה בְּכָל מָקוֹם שֶׁמִּתְאַסְּפִים בּוֹ אֲנָשִׁים, כְּמוֹ חֲתֻנָּה אוֹ מְסִבַּת חֲנֻכָּה. לְדַעַת רַבִּים מֵרַבָּנֵי דּוֹרֵנוּ לֹא מְבָרְכִים שָׁם, וּלְדַעַת כַּמָּה פּוֹסְקִים מֻתָּר לְבָרֵךְ, וּמִי שֶׁרוֹצֶה לִסְמֹךְ עֲלֵיהֶם – רַשַּׁאי.
אִם יֵשׁ שָׁם אֲנָשִׁים שֶׁאוּלַי לֹא הִדְלִיקוּ בְּבֵיתָם, חָשׁוּב לְהַדְלִיק בִּבְרָכָה, כְּדֵי שֶׁיִּשְׁמְעוּ וְיִלְמְדוּ. וּכְדַאי לְכַבֵּד בְּהַדְלָקָה יְהוּדִי שֶׁאֵינוֹ רָגִיל בְּמִצְווֹת.
💛 אַהֲבַת הַשֵּׁם: הַמִּצְווֹת שַׁיָּכוֹת לְכָל יִשְׂרָאֵל. כְּשֶׁאָנוּ מְכַבְּדִים יְהוּדִי לְהַדְלִיק, אֲנַחְנוּ מַדְלִיקִים גַּם אֶת הַנֵּר שֶׁבְּלִבּוֹ.

🧠 בְּדֹק אֶת עַצְמְךָ

לְעָנִי אֵין כֶּסֶף לִקְנוֹת נֵרוֹת חֲנֻכָּה. מָה הַדִּין?
💡 בִּגְלַל פִּרְסוּם הַנֵּס, גַּם עָנִי צָרִיךְ לְבַקֵּשׁ אוֹ לִמְכֹּר אֶת בִּגְדוֹ כְּדֵי לִקְנוֹת נֵר. אֲבָל הוּא לֹא צָרִיךְ לְהַדֵּר – מַסְפִּיק נֵר אֶחָד בְּכָל לַיְלָה.
נֵר חֲנֻכָּה הִיא מִצְוָה מִדִּבְרֵי חֲכָמִים. אֵיךְ אֶפְשָׁר לְבָרֵךְ עָלֶיהָ ״וְצִוָּנוּ״?
💡 הַתּוֹרָה אָמְרָה: ״לֹא תָסוּר מִן הַדָּבָר אֲשֶׁר יַגִּידוּ לְךָ״ – וְכָךְ נָתְנָה לַחֲכָמִים סַמְכוּת לְתַקֵּן מִצְווֹת. לָכֵן גַּם עַל מִצְוָה שֶׁלָּהֶם אוֹמְרִים ״וְצִוָּנוּ״.
הַאִם גַּם נָשִׁים חַיָּבוֹת בְּמִצְוַת נֵר חֲנֻכָּה?
💡 בְּדֶרֶךְ כְּלָל נָשִׁים פְּטוּרוֹת מִמִּצְווֹת שֶׁתְּלוּיוֹת בִּזְמַן, אֲבָל כָּאן גַּם הֵן הָיוּ שֻׁתָּפוֹת בַּנֵּס, וְלָכֵן חַיָּבוֹת. אִשָּׁה נְשׂוּאָה נוֹהֶגֶת לָצֵאת בְּהַדְלָקַת בַּעְלָהּ.

דִּינֵי קָטָן

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק כ״ד

🧒מִצְוַת חִנּוּךְ

מִצְוָה לְלַמֵּד אֶת הַיְּלָדִים תּוֹרָה וּלְהַרְגִּיל אוֹתָם לְקַיֵּם מִצְווֹת. מַתְחִילִים כְּשֶׁהַיֶּלֶד מֵבִין אֶת הַמִּצְוָה וְיָכוֹל לְקַיְּמָהּ: צִיצִית – כְּשֶׁהוּא יוֹדֵעַ לְהִתְעַטֵּף, תְּפִלִּין – לִקְרַאת בַּר הַמִּצְוָה. בְּגִיל שֵׁשׁ־שֶׁבַע לוֹמְדִים לְקַיֵּם אֶת רֹב הַמִּצְווֹת כָּרָאוּי.
מִגִּיל שָׁלֹשׁ, כְּשֶׁהַיֶּלֶד מֵבִין שֶׁיֵּשׁ דָּבָר אָסוּר, מַרְחִיקִים אוֹתוֹ מֵאִסּוּרִים. וְאָסוּר לְאִישׁ גָּדוֹל לְהַכְשִׁיל יֶלֶד – אֲפִלּוּ תִּינוֹק – בְּאִסּוּר.
💛 אַהֲבַת הַשֵּׁם: הַחִנּוּךְ צָרִיךְ לִהְיוֹת בְּדֶרֶךְ שֶׁמִּתְקַבֶּלֶת עַל הַלֵּב. שְׁנוֹת הַיַּלְדוּת הֵן זְמַן לְהִתְאַמֵּן וּלְהִתְקַדֵּם, צַעַד אַחַר צַעַד, עַד שֶׁמְּקַיְּמִים הַכֹּל בְּשִׂמְחָה.
📖 פֶּרֶק כ״ד, הֲלָכוֹת א׳–ג׳

💡יֶלֶד לֹא עוֹשֶׂה מְלָאכָה בִּשְׁבִילֵנוּ

הַתּוֹרָה אוֹמֶרֶת: "לֹא תַעֲשֶׂה כָל מְלָאכָה אַתָּה וּבִנְךָ וּבִתֶּךָ". לָכֵן אִם הָאוֹר כָּבָה, וְיֶלֶד קָטָן רוֹצֶה לְהַדְלִיק כְּדֵי לַעֲזֹר לַהוֹרִים – חוֹבָה לַעֲצֹר אוֹתוֹ. וְכָל שֶׁכֵּן שֶׁאָסוּר לוֹמַר לוֹ לְהַדְלִיק.
יֶלֶד שֶׁכִּבָּה בְּטָעוּת אֶת הָאוֹר – לֹא כּוֹעֲסִים עָלָיו, כְּדֵי שֶׁלֹּא יָבִין שֶׁהוּא צָרִיךְ לְהַדְלִיק שׁוּב.
💛 אַהֲבַת הַשֵּׁם: הַשַּׁבָּת שֶׁל הַיְּלָדִים קְדוֹשָׁה כְּמוֹ שֶׁל הַגְּדוֹלִים. כְּשֶׁאַבָּא עוֹצֵר אֶת בְּנוֹ בַּעֲדִינוּת, הוּא מְלַמֵּד אוֹתוֹ שֶׁהַשַּׁבָּת חֲשׁוּבָה יוֹתֵר מִכָּל נוֹחוּת.
📖 פֶּרֶק כ״ד, הֲלָכוֹת ב׳, ד׳

📚"וְלִמַּדְתֶּם אֹתָם אֶת בְּנֵיכֶם"

מִצְוַת הַחִנּוּךְ מֻטֶּלֶת עַל הָאָב, וְגַם לָאֵם יֵשׁ מִצְוָה לְחַנֵּךְ אֶת יְלָדֶיהָ לְתוֹרָה וּלְמִצְווֹת. וְאִם יֶלֶד עוֹמֵד לִפְגֹּעַ בְּאָדָם אוֹ לְהַזִּיק לְרֶכֶשׁ – חוֹבָה עַל כָּל אֶחָד לַעֲצֹר אוֹתוֹ.
טוֹב שֶׁהַהוֹרִים יִלְמְדוּ עִם יְלָדֵיהֶם בְּשַׁבָּת, וְטוֹב שֶׁבְּכָל קְהִלָּה יִהְיוּ שִׁעוּרִים רַבִּים לִילָדִים בְּשַׁבָּת.
💛 אַהֲבַת הַשֵּׁם: כְּכָל שֶׁהַיֶּלֶד מִתְקָרֵב לְבַר הַמִּצְוָה, כָּךְ הוּא מַקְדִּישׁ יוֹתֵר מִן הַשַּׁבָּת לַתּוֹרָה. כָּךְ הוּא מְגַלֶּה שֶׁהַתּוֹרָה הִיא הַמַּתָּנָה הֲכִי מְתוּקָה שֶׁל הַשַּׁבָּת.
📖 פֶּרֶק כ״ד, הֲלָכוֹת ג׳, ז׳

🤒יֶלֶד שֶׁצָּרִיךְ מְאוֹד

יֶלֶד קָטָן שֶׁצָּרִיךְ מְאוֹד דָּבָר מְסֻיָּם דִּינוֹ כְּחוֹלֶה. לְמָשָׁל, אִם אֵין לוֹ מָה לֶאֱכֹל, מֻתָּר לְבַקֵּשׁ מִגּוֹי לְבַשֵּׁל בִּשְׁבִילוֹ. וִילָדִים שֶׁמְּפַחֲדִים מְאוֹד מִן הַחֹשֶׁךְ – מֻתָּר לְבַקֵּשׁ מִגּוֹי לְהַדְלִיק לָהֶם אוֹר.
יֶלֶד שֶׁעוֹד לֹא בַּר מִצְוָה, וְסוֹבֵל מִכְּאֵב אוֹ מִחוּשׁ – מֻתָּר לָתֵת לוֹ תְּרוּפוֹת בְּשַׁבָּת.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה דּוֹאֶגֶת לַקְּטַנִּים בְּאַהֲבָה, כְּמוֹ אַבָּא רַחְמָן. הַשַּׁבָּת לֹא נִתְּנָה כְּדֵי שֶׁיֶּלֶד יִסְבֹּל.
📖 פֶּרֶק כ״ד, הֲלָכָה ו׳; פֶּרֶק כ״ה, הֲלָכָה ה׳

🧩מִשְׂחָקִים מֻתָּרִים

יְלָדִים רַשָּׁאִים לְשַׂחֵק בְּשַׁבָּת: שַׁחְמָט, דַּמְקָה, מִשְׂחֲקֵי זִכָּרוֹן, קֻבִּיָּה וּסְבִיבוֹן. פָּאזֶל – לְרֹב פּוֹסְקֵי זְמַנֵּנוּ אָסוּר, וְיֵשׁ מַתִּירִים, וְהָרַב מֶלָמֵד כּוֹתֵב שֶׁהָרוֹצִים לְהָקֵל לִילָדִים – רַשָּׁאִים. לֶגוֹ – לְדַעַת רַבִּים מֻתָּר, וְיֵשׁ מַחְמִירִים. אֲבָל מִשְׂחָק שֶׁיֵּשׁ בּוֹ כְּתִיבָה, גְּזִירָה אוֹ הַדְבָּקָה – אָסוּר, וְגַם מִשְׂחָק שֶׁהַמְּנַצֵּחַ זוֹכֶה בּוֹ בְּכֶסֶף אוֹ בְּאֹכֶל.
פְּלַסְטֶלִינָה – אֲסוּרָה, כִּי מוֹרְחִים בָּהּ וְעוֹשִׂים צוּרוֹת, וְהִיא מֻקְצֶה. אֹהֶל מִשְּׂמִיכָה – מוֹתְחִים קֹדֶם אֶת הַשְּׂמִיכָה בָּאֲוִיר, וְרַק אַחַר כָּךְ מַכְנִיסִים תַּחְתֶּיהָ אֶת הַכִּסְאוֹת.
💛 אַהֲבַת הַשֵּׁם: מֻתָּר לְשַׂחֵק – וְטוֹב לִזְכֹּר שֶׁהַשַּׁבָּת נִתְּנָה בְּעִקָּר לַתּוֹרָה. אַבָּא שֶׁלּוֹמֵד עִם בְּנוֹ בְּשַׁבָּת מְקַיֵּם "וְלִמַּדְתֶּם אֹתָם אֶת בְּנֵיכֶם".
📖 פֶּרֶק כ״ד, הֲלָכָה ז׳; פֶּרֶק י״ח, הֲלָכָה ד׳; פֶּרֶק ט״ו, הֲלָכָה ז׳

🧵מִשְׂחֲקֵי יְצִירָה

מִשְׂחָק שֶׁיֵּשׁ בּוֹ כְּתִיבָה, הַדְבָּקָה, גְּזִירָה אוֹ קְלִיעָה – אָסוּר בְּשַׁבָּת, וְלָכֵן לֹא קוֹלְעִים חוּטִים. אֲבָל מֻתָּר לְהַשְׁחִיל חֲרוּזִים לְמַחֲרֹזֶת שֶׁל מִשְׂחָק, שֶׁאֵינָהּ עוֹמֶדֶת לְהִתְקַיֵּם – וּבִתְנַאי שֶׁלֹּא קוֹשְׁרִים קֶשֶׁר בְּסוֹף הַחוּט, אֶלָּא רַק עֲנִיבָה.
קֻבִּיּוֹת אוֹ קְלָפִים שֶׁל שְׁנֵי מִשְׂחָקִים שֶׁהִתְעָרְבְבוּ – לֹא מְמַיְּנִים, כִּי זוֹ מְלֶאכֶת בּוֹרֵר. אֲבָל מֻתָּר לְהוֹצִיא מִן הַתַּעֲרֹבֶת אֶת הַסּוּג שֶׁרוֹצִים לְשַׂחֵק בּוֹ עַכְשָׁו.
💛 אַהֲבַת הַשֵּׁם: גַּם בְּמִשְׂחָק אֶפְשָׁר לִזְכֹּר שֶׁשַּׁבָּת הַיּוֹם. כְּשֶׁאָנוּ מְשַׂחֲקִים בְּדֶרֶךְ שֶׁל שַׁבָּת, הַשִּׂמְחָה שֶׁלָּנוּ נַעֲשֵׂית חֵלֶק מֵעֹנֶג הַשַּׁבָּת.
📖 פֶּרֶק כ״ד, הֲלָכָה ז׳

🔀דֵּעוֹת בְּמִשְׂחֲקֵי יְלָדִים

פָּאזֶל: לְרֹב פּוֹסְקֵי זְמַנֵּנוּ אָסוּר לְהַרְכִּיב פָּאזֶל בְּשַׁבָּת, כִּי גַּם צוּרָה שֶׁיֵּשׁ לָהּ מַשְׁמָעוּת נֶחְשֶׁבֶת כְּתִיבָה. וְיֵשׁ מַתִּירִים, כִּי הַכֹּל כְּבָר מְצֻיָּר, וְרַק מְקָרְבִים אֶת הַחֲלָקִים. הָרַב מֶלָמֵד כּוֹתֵב שֶׁהָרוֹצִים לְהָקֵל לִילָדִים – רַשָּׁאִים, וְלִמְבֻגָּרִים נָכוֹן לְהַחְמִיר. לֶגוֹ: לְדַעַת רַבִּים אֵין בּוֹ בּוֹנֶה וְסוֹתֵר, כִּי הַחִבּוּר רָפוּי, וְיֵשׁ מַחְמִירִים.
קוֹרְקִינֶט: יֵשׁ מַתִּירִים לִילָדִים, כְּמוֹ שֶׁמֻּתָּר לָהֶם לָרוּץ. וְיֵשׁ אוֹסְרִים, כִּי זוֹ רִיצָה בְּמַכְשִׁיר, כְּמוֹ אוֹפַנַּיִם. הָרַב מֶלָמֵד כּוֹתֵב שֶׁנָּכוֹן לְהַחְמִיר, וּבְדִיעֲבַד יֵשׁ לַמְּקִלִּים עַל מָה לִסְמֹךְ.
💛 אַהֲבַת הַשֵּׁם: כָּל הַפּוֹסְקִים רוֹצִים שֶׁהַיְּלָדִים יִשְׂמְחוּ בַּשַּׁבָּת וְגַם יַרְגִּישׁוּ אֶת קְדֻשָּׁתָהּ. כָּל מִשְׁפָּחָה שׁוֹמַעַת לְרַבּוֹתֶיהָ – וְכֻלָּם מְכַבְּדִים זֶה אֶת זֶה.
📖 פֶּרֶק כ״ד, הֲלָכוֹת ז׳, י׳; פֶּרֶק י״ח, הֲלָכָה ד׳; פֶּרֶק ט״ו, הֲלָכָה ז׳

🎈צַעֲצוּעִים בְּשַׁבָּת

מֻתָּר לִמְתֹּחַ קְפִיץ שֶׁל מְכוֹנִית צַעֲצוּעַ, בִּתְנַאי שֶׁאֵינָהּ מְצַפְצֶפֶת וְאֵינָהּ מַדְלִיקָה נוּרוֹת. צַעֲצוּעִים שֶׁפּוֹעֲלִים עַל סוֹלְלוֹת – אֲסוּרִים. וְלֹא מְנַפְּחִים בָּלוֹן, שֶׁמָּא יִקְשְׁרוּ אוֹתוֹ.
צַעֲצוּעִים שֶׁל כְּלֵי נְגִינָה – תֹּף, חֲצוֹצְרָה, פְּסַנְתֵּר וְרַעֲשָׁן – אֲסוּרִים וְהֵם מֻקְצֶה. לְתִינוֹק מֻתָּר לָתֵת צַעֲצוּעַ שֶׁמַּשְׁמִיעַ קוֹל כְּשֶׁמְּנַעְנְעִים אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: גַּם בִּזְמַן הַמִּשְׂחָק אֲנַחְנוּ זוֹכְרִים אֶת הַשַּׁבָּת. כָּךְ הַשַּׁבָּת לֹא נִשְׁאֶרֶת רַק בְּבֵית הַכְּנֶסֶת – הִיא נִכְנֶסֶת גַּם לַחֶדֶר שֶׁל הַיְּלָדִים.
📖 פֶּרֶק כ״ד, הֲלָכָה ח׳

⚽כַּדּוּר וְאוֹפַנַּיִם

כַּדּוּרֶגֶל, כַּדּוּרְסַל וְטֶנִיס אֲסוּרִים גַּם לִילָדִים, כִּי הֵם עֻבְדִּין דְּחֹל. אֲבָל בְּכַדּוּר שֶׁל יְלָדִים קְטַנִּים מֻתָּר לְשַׂחֵק – בַּבַּיִת אוֹ עַל רִצְפָּה מְרֻצֶּפֶת, וְלֹא עַל אֲדָמָה אוֹ דֶּשֶׁא, שֶׁמָּא יַשְׁווּ גֻּמּוֹת. גַּם טֶנִיס שֻׁלְחָן בַּבַּיִת – מֻתָּר.
אוֹפַנַּיִם אֲסוּרִים, גַּם עִם גַּלְגַּלֵּי עֵזֶר. תְּלַת אוֹפַן שֶׁל קְטַנִּים – מֻתָּר. עַל קוֹרְקִינֶט יֵשׁ מַתִּירִים וְיֵשׁ אוֹסְרִים, וְהָרַב מֶלָמֵד כּוֹתֵב שֶׁנָּכוֹן לְהַחְמִיר.
💛 אַהֲבַת הַשֵּׁם: הַשַּׁבָּת מַזְמִינָה אוֹתָנוּ לְמִשְׂחָקִים שְׁקֵטִים וּפְשׁוּטִים. כָּךְ הִיא נִשְׁאֶרֶת יוֹם מְיֻחָד, שֶׁל מְנוּחָה וְתוֹרָה.
📖 פֶּרֶק כ״ד, הֲלָכוֹת ט׳–י׳

🧠 בְּדֹק אֶת עַצְמְךָ

אָחִיו הַקָּטָן שֶׁל דָּוִד הִדְלִיק בְּטָעוּת מְנוֹרָה בַּחֶדֶר, כְּדֵי לְשַׂחֵק לְעַצְמוֹ. הַאִם מֻתָּר לְדָוִד לִקְרֹא לְאוֹרָהּ?
💡 קָטָן שֶׁעָשָׂה מְלָאכָה לְצֹרֶךְ עַצְמוֹ – מֻתָּר לַגָּדוֹל לֵהָנוֹת מִמֶּנָּה. רַק כְּשֶׁהִדְלִיק בִּשְׁבִיל גָּדוֹל – אָסוּר.
בְּשַׁבָּת אַחַר הַצָּהֳרַיִם רוֹצָה שִׁירָה לִקְלֹעַ צְמִיד מֵחוּטִים צִבְעוֹנִיִּים, כְּמוֹ בִּימוֹת הַחֹל. מָה הַדִּין?
💡 כָּל מִשְׂחָק שֶׁיֵּשׁ בּוֹ כְּתִיבָה, הַדְבָּקָה, גְּזִירָה אוֹ קְלִיעָה – אָסוּר בְּשַׁבָּת, גַּם לִילָדִים. אֶת הַצְּמִיד אֶפְשָׁר לִקְלֹעַ בְּמוֹצָאֵי שַׁבָּת.
מִיכַל מַשְׁחִילָה בְּשַׁבָּת חֲרוּזִים עַל חוּט, לְמַחֲרֹזֶת שֶׁל מִשְׂחָק. אֵיךְ תִּסְגֹּר אֶת הַמַּחֲרֹזֶת?
💡 מֻתָּר לַעֲשׂוֹת מַחֲרֹזֶת שֶׁל מִשְׂחָק שֶׁאֵינָהּ עוֹמֶדֶת לְהִתְקַיֵּם, וּבִתְנַאי שֶׁלֹּא קוֹשְׁרִים קֶשֶׁר בְּסוֹף הַחוּט, אֶלָּא סוֹגְרִים בַּעֲנִיבָה. גַּם הַדְבָּקָה אֲסוּרָה בְּשַׁבָּת.

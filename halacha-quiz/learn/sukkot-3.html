יְשִׁיבָה בַּסֻּכָּה

🏠תֵּשְׁבוּ – כְּעֵין תָּדוּרוּ

מִצְוָה לָגוּר בַּסֻּכָּה שִׁבְעָה יָמִים, כְּמוֹ שֶׁגָּרִים בַּבַּיִת כָּל הַשָּׁנָה. מַעֲבִירִים לַסֻּכָּה אֶת הַשֻּׁלְחָן וְהַכִּסְאוֹת הַיָּפִים, וְאֶת הַמִּטָּה וְהַמַּצָּעִים הַטּוֹבִים. בַּחַג הַסֻּכָּה הִיא הַבַּיִת הַקָּבוּעַ, וְהַבַּיִת – אַרְעַי.
סְעוּדוֹת וְשֵׁינָה – חוֹבָה בַּסֻּכָּה. לִמּוּד תּוֹרָה, קְרִיאָה בְּסֵפֶר וְשִׂיחָה עִם חֲבֵרִים – מִצְוָה בַּסֻּכָּה. וְגַם מִשְׂחָק שַׁחְמָט אוֹ שִׂיחָה בַּטֶּלֶפוֹן – טוֹב שֶׁיִּהְיוּ בַּסֻּכָּה, וְאָז יֵשׁ בָּהֶם מִצְוָה.
💛 אַהֲבַת הַשֵּׁם: בַּסֻּכָּה גַּם הָאֲכִילָה, הַשֵּׁינָה וְהַשִּׂיחָה הוֹפְכוֹת לְמִצְוָה. כָּל רֶגַע בַּסֻּכָּה הוּא רֶגַע שֶׁל קִרְבָה לַה׳.

👧נָשִׁים וּבָנוֹת בַּסֻּכָּה

נָשִׁים פְּטוּרוֹת מִמִּצְוַת הַסֻּכָּה, כִּי הִיא מִצְוַת עֲשֵׂה שֶׁהַזְּמַן גְּרָמָהּ. אֲבָל אִשָּׁה אוֹ בַּת שֶׁיּוֹשֶׁבֶת בַּסֻּכָּה – מְקַיֶּמֶת מִצְוָה.
הַאִם הֵן מְבָרְכוֹת ״לֵישֵׁב בַּסֻּכָּה״? לְמִנְהַג הָאַשְׁכְּנַזִּים וְחֵלֶק מִן הַסְּפָרַדִּים – מְבָרְכוֹת כְּמוֹ הַגְּבָרִים. לְמִנְהַג רֹב הַסְּפָרַדִּים – אֵינָן מְבָרְכוֹת, כִּי אֵינָן חַיָּבוֹת לָשֶׁבֶת בַּסֻּכָּה.
💛 אַהֲבַת הַשֵּׁם: כָּל מִי שֶׁיּוֹשֵׁב בְּצֵל הַסֻּכָּה זוֹכֶה לְמִצְוָה. כְּשֶׁכָּל הַמִּשְׁפָּחָה יוֹשֶׁבֶת יַחַד בַּסֻּכָּה – הַשִּׂמְחָה שֶׁל הַחַג שְׁלֵמָה.

🧽כְּבוֹד הַסֻּכָּה

נוֹהֲגִים בַּסֻּכָּה כְּמוֹ בַּחֶדֶר הַיָּפֶה וְהַמְכֻבָּד שֶׁבַּבַּיִת. לָכֵן לֹא שׁוֹטְפִים בָּהּ כֵּלִים, לֹא מַחְלִיפִים בָּהּ חִתּוּל לְתִינוֹק, וְלֹא מַנִּיחִים בָּהּ דְּלִי אוֹ פַּח לִשְׁיָרֵי אֹכֶל.
אַחֲרֵי הָאֲרוּחָה מְפַנִּים מַהֵר אֶת הַצַּלָּחוֹת וְאֶת הַסַּכּוּ״ם. אֶת כּוֹסוֹת הַשְּׁתִיָּה מֻתָּר לְהַשְׁאִיר, כִּי הֵן אֵינָן מְלֻכְלָכוֹת כָּל כָּךְ.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאָנוּ שׁוֹמְרִים עַל הַסֻּכָּה נְקִיָּה וּמְכֻבֶּדֶת, אָנוּ מְכַבְּדִים אֶת הַמִּצְוָה – וְאֶת ה׳ שֶׁצִּוָּה עָלֶיהָ.

🏡סֻכָּתוֹ קֶבַע וּבֵיתוֹ אַרְעַי

כָּל שִׁבְעַת הַיָּמִים עוֹשֶׂה אָדָם אֶת סֻכָּתוֹ קֶבַע וְאֶת בֵּיתוֹ אַרְעַי. בַּחַג הַבַּיִת מְשַׁמֵּשׁ רַק כְּמִטְבָּח וּמַחְסָן, שֶׁעוֹזְרִים לַסֻּכָּה.
לִמּוּד תּוֹרָה רָגִיל טוֹב לִלְמֹד בַּסֻּכָּה. לִמּוּד עָמֹק וּמְיַגֵּעַ, אוֹ לִמּוּד שֶׁצָּרִיךְ בּוֹ סְפָרִים רַבִּים – אֶפְשָׁר לִלְמֹד גַּם בְּבֵית הַמִּדְרָשׁ.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁהַסֻּכָּה הוֹפֶכֶת לַבַּיִת הָאֲמִתִּי, אָנוּ מְגַלִּים שֶׁהַבַּיִת הָאֲמִתִּי שֶׁלָּנוּ הוּא בְּצֵל ה׳.
📖 פֶּרֶק ג׳, הֲלָכָה ב׳

🍞הַלַּיְלָה הָרִאשׁוֹן

בַּלַּיְלָה הָרִאשׁוֹן שֶׁל סֻכּוֹת חוֹבָה לֶאֱכֹל לֶחֶם בַּסֻּכָּה – כְּמוֹ שֶׁבְּלֵיל הַסֵּדֶר חוֹבָה לֶאֱכֹל מַצָּה. אוֹכְלִים אַחֲרֵי צֵאת הַכּוֹכָבִים, וְלִפְנֵי חֲצוֹת הַלַּיְלָה.
כְּדֵי לֶאֱכֹל בְּתֵאָבוֹן, לֹא אוֹכְלִים דְּבָרִים מַשְׂבִּיעִים כְּשָׁלֹשׁ שָׁעוֹת לִפְנֵי הַשְּׁקִיעָה בְּעֶרֶב הַחַג. וְלִפְנֵי שֶׁאוֹכְלִים מְכַוְּנִים: אֲנִי מְקַיֵּם מִצְוַת ה׳, זֵכֶר לִיצִיאַת מִצְרַיִם וּלְעַנְנֵי הַכָּבוֹד.
💛 אַהֲבַת הַשֵּׁם: הַלַּיְלָה הָרִאשׁוֹן מֵנִיחַ אֶת הַיְּסוֹד לְכָל הַחַג. הַפְּרוּסָה הָרִאשׁוֹנָה בַּסֻּכָּה הִיא כְּנִיסָה שְׂמֵחָה אֶל צֵל הָאֱמוּנָה.
📖 פֶּרֶק ג׳, הֲלָכָה ג׳

🍞כַּמָּה לֶחֶם בַּלַּיְלָה הָרִאשׁוֹן

בַּלַּיְלָה הָרִאשׁוֹן מְדַקְדְּקִים כְּמוֹ בַּמַּצָּה שֶׁל לֵיל הַסֵּדֶר: אוֹכְלִים לְפָחוֹת כַּחֲצִי בֵּיצָה לֶחֶם, וְטוֹב לֶאֱכֹל יוֹתֵר מִבֵּיצָה שְׁלֵמָה. אוֹכְלִים בְּנַחַת וּבִרְצִיפוּת, בְּלִי לְהַפְסִיק יוֹתֵר מִכְּשֵׁשׁ אוֹ שֶׁבַע דַּקּוֹת.
צָרִיךְ לֶאֱכֹל לִפְנֵי חֲצוֹת הַלַּיְלָה. וּמִי שֶׁלֹּא הִסְפִּיק – אוֹכֵל עַד עֲלוֹת הַשַּׁחַר.
💛 אַהֲבַת הַשֵּׁם: הַדִּקְדּוּק בַּשִּׁעוּרִים מַרְאֶה כַּמָּה הַמִּצְוָה חֲשׁוּבָה לָנוּ. אָנוּ רוֹצִים לְקַיֵּם אוֹתָהּ בִּשְׁלֵמוּת.

🌧️גֶּשֶׁם בַּלַּיְלָה הָרִאשׁוֹן

אִם יוֹרֵד גֶּשֶׁם בַּלַּיְלָה הָרִאשׁוֹן, נָכוֹן לְהַמְתִּין שָׁעָה אוֹ שְׁעָתַיִם – אוּלַי הַגֶּשֶׁם יִפָּסֵק. אִם הוּא מַמְשִׁיךְ, מְקַדְּשִׁים בַּסֻּכָּה, מְבָרְכִים ״שֶׁהֶחֱיָנוּ״ וְאוֹכְלִים כְּזַיִת לֶחֶם – אֲבָל לֹא מְבָרְכִים ״לֵישֵׁב בַּסֻּכָּה״.
לָמָּה? יֵשׁ אוֹמְרִים שֶׁבַּלַּיְלָה הָרִאשׁוֹן גַּם מִצְטַעֵר חַיָּב לֶאֱכֹל בַּסֻּכָּה, וְיֵשׁ אוֹמְרִים שֶׁהוּא פָּטוּר – וְחוֹשְׁשִׁים לִשְׁתֵּי הַדֵּעוֹת. וְאִם הַגֶּשֶׁם נִפְסָק לִפְנֵי חֲצוֹת – חוֹזְרִים, אוֹכְלִים לֶחֶם וּמְבָרְכִים ״לֵישֵׁב בַּסֻּכָּה״.
💛 אַהֲבַת הַשֵּׁם: גַּם בַּגֶּשֶׁם אָנוּ מְחַפְּשִׂים דֶּרֶךְ לְקַיֵּם אֶת הַמִּצְוָה כְּכָל הָאֶפְשָׁר. אַהֲבָה אֲמִתִּית אֵינָהּ מְוַתֶּרֶת מַהֵר.

🍎סְעוּדָה אוֹ נִשְׁנוּשׁ

אֲכִילַת קֶבַע – אֲכִילָה שֶׁאוֹכְלִים כְּדֵי לִשְׂבֹּעַ – חוֹבָה בַּסֻּכָּה. לֶחֶם, עוּגוֹת, אִטְרִיּוֹת וּפְתִיתִים, יוֹתֵר מִנֶּפַח שֶׁל בֵּיצָה – אוֹכְלִים רַק בַּסֻּכָּה. אֲבָל פֵּרוֹת, מַיִם וּמִיצִים – מֻתָּר לֶאֱכֹל וְלִשְׁתּוֹת גַּם בַּחוּץ.
הַמְהַדְּרִים אוֹכְלִים בַּסֻּכָּה גַּם נִשְׁנוּשִׁים, וְאֵינָם שׁוֹתִים בַּחוּץ אֲפִלּוּ מַיִם – אֲבָל זוֹ אֵינָהּ חוֹבָה.
💛 אַהֲבַת הַשֵּׁם: הַמִּצְוָה אֵינָהּ רַק חוֹבָה – הִיא הִזְדַּמְּנוּת. כָּל כּוֹס מַיִם שֶׁשּׁוֹתִים בַּסֻּכָּה מוֹסִיפָה עוֹד רֶגַע שֶׁל מִצְוָה.

🍽️מָה נֶחְשָׁב אֲכִילַת קֶבַע

מִינֵי דָּגָן – לֶחֶם, עוּגוֹת וְאִטְרִיּוֹת – יוֹתֵר מִנֶּפַח שֶׁל בֵּיצָה, הֵם אֲכִילַת קֶבַע, וְאוֹכְלִים אוֹתָם רַק בַּסֻּכָּה. כְּנֶפַח בֵּיצָה וּפָחוֹת – אֲכִילַת אַרְעַי. מְעַט בָּשָׂר, דָּגִים אוֹ גְּבִינָה – מֻתָּר בַּחוּץ, אֲבָל אֲרוּחָה שֶׁשְּׂבֵעִים מִמֶּנָּה – רַק בַּסֻּכָּה.
בְּאֶמְצַע הַסְּעוּדָה, מִי שֶׁיּוֹצֵא לַבַּיִת לְהָבִיא דָּבָר – אֵינוֹ שׁוֹתֶה שָׁם אֲפִלּוּ מְעַט מַיִם, וְאֵינוֹ מַמְשִׁיךְ לִבְלֹעַ אֶת מָה שֶׁהִתְחִיל לֶאֱכֹל בַּסֻּכָּה. כָּל הַסְּעוּדָה – בַּסֻּכָּה.
💛 אַהֲבַת הַשֵּׁם: הַסְּעוּדָה כֻּלָּהּ, מִן הַבְּרָכָה וְעַד בִּרְכַּת הַמָּזוֹן, הִיא מִצְוָה אַחַת שְׁלֵמָה בְּצֵל הַסֻּכָּה.
📖 פֶּרֶק ג׳, הֲלָכָה ד׳

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

עַל מָה מְבָרְכִים ״לֵישֵׁב בַּסֻּכָּה״? הַתֵּימָנִים מְבָרְכִים בְּכָל פַּעַם שֶׁנִּכְנָסִים לַסֻּכָּה, וּשְׁאָר הָעֵדוֹת מְבָרְכִים עַל הָאֲכִילָה. לְמִנְהַג הָאַשְׁכְּנַזִּים – עַל לֶחֶם אוֹ מְזוֹנוֹת יוֹתֵר מִכְּבֵיצָה, וּמִי שֶׁלֹּא יֹאכַל כָּל כָּךְ – מְבָרֵךְ גַּם עַל טְעִימָה שֶׁל מְזוֹנוֹת אוֹ יַיִן. לְמִנְהַג הַסְּפָרַדִּים – עַל לֶחֶם יוֹתֵר מִכְּבֵיצָה, וְעַל עוּגוֹת וּשְׁאָר מְזוֹנוֹת – רַק כְּשֶׁאוֹכְלִים כְּנֶפַח אַרְבַּע בֵּיצִים.
סֵדֶר הַבְּרָכוֹת: הָאַשְׁכְּנַזִּים וְחֵלֶק מִן הַסְּפָרַדִּים מְבָרְכִים קֹדֶם עַל הַמַּאֲכָל, וְאַחַר כָּךְ ״לֵישֵׁב בַּסֻּכָּה״, בִּישִׁיבָה. חֵלֶק מִן הַסְּפָרַדִּים מְבָרְכִים קֹדֶם ״לֵישֵׁב בַּסֻּכָּה״ בַּעֲמִידָה, וְאַחַר כָּךְ מִתְיַשְּׁבִים וּמְבָרְכִים עַל הַמַּאֲכָל. נָכוֹן שֶׁכָּל אֶחָד יַמְשִׁיךְ בְּמִנְהַג אֲבוֹתָיו.
💛 אַהֲבַת הַשֵּׁם: כָּל הַמִּנְהָגִים נוֹסְדוּ עַל יְדֵי גְּדוֹלֵי תּוֹרָה. כָּל אֶחָד שׁוֹמֵר עַל מִנְהַג מִשְׁפַּחְתּוֹ בְּאַהֲבָה – וְכֻלָּנוּ מְבָרְכִים אֶת אוֹתוֹ ה׳, שֶׁצִּוָּנוּ לֵישֵׁב בַּסֻּכָּה.

🚪יְצִיאָה וַחֲזָרָה

😴שֵׁינָה בַּסֻּכָּה

🤒חוֹלֶה פָּטוּר

חוֹלֶה פָּטוּר מִן הַסֻּכָּה – לֹא רַק חוֹלֶה מְסֻכָּן, אֶלָּא גַּם מִי שֶׁכּוֹאֵב לוֹ הָרֹאשׁ וְקָשֶׁה לוֹ לָשֶׁבֶת בַּסֻּכָּה. וְאִם הוּא צָרִיךְ מִישֶׁהוּ שֶׁיַּעֲזֹר לוֹ – גַּם הָעוֹזֵר פָּטוּר.

״הָעוֹסֵק בְּמִצְוָה פָּטוּר מִן הַמִּצְוָה״ – כִּי הַטִּרְחָה לְחַפֵּשׂ סֻכָּה עֲלוּלָה לִפְגֹּעַ בַּמִּצְוָה שֶׁהוּא עָסוּק בָּהּ. אֲבָל אִם יֵשׁ סֻכָּה קְרוֹבָה וְנוֹחָה – בַּשָּׁעוֹת הַפְּנוּיוֹת אוֹכְלִים וִישֵׁנִים בָּהּ.
סְעוּדוֹת שֶׁל מִצְוָה – בְּרִית מִילָה, פִּדְיוֹן הַבֵּן, בַּר מִצְוָה וְסִיּוּם מַסֶּכֶת – עוֹרְכִים בַּסֻּכָּה.
💛 אַהֲבַת הַשֵּׁם: כָּל הַמִּצְווֹת בָּאוֹת מֵאוֹתוֹ ה׳. מִי שֶׁעָסוּק בְּמִצְוָה אַחַת – ה׳ אֵינוֹ דּוֹרֵשׁ מִמֶּנּוּ לַעֲזֹב אוֹתָהּ.
📖 פֶּרֶק ג׳, הֲלָכוֹת י״ב–י״ג

🚌חַיָּלִים, נוֹסְעִים וְטִיּוּלִים

״הָעוֹסֵק בְּמִצְוָה פָּטוּר מִן הַמִּצְוָה״: מִי שֶׁמְּלַוֶּה חוֹלֶה בְּבֵית הַחוֹלִים, אוֹ חַיָּל שֶׁעוֹסֵק בִּשְׁמִירָה וְאֵין לוֹ זְמַן פָּנוּי – פָּטוּר מִן הַסֻּכָּה. וְהַמְּפַקְּדִים צְרִיכִים לִדְאֹג לְסֻכָּה נוֹחָה לַחַיָּלִים.
טִיּוּל: יֵשׁ פּוֹסְקִים שֶׁמַּתִּירִים לְמִי שֶׁיּוֹצֵא לְטִיּוּל לֶאֱכֹל אֲרוּחַת קֶבַע בְּלִי סֻכָּה, כְּמוֹ שֶׁבְּכָל הַשָּׁנָה מְטַיְּלִים אֵינָם אוֹכְלִים דַּוְקָא בְּבַיִת. הָרַב מֶלָמֵד כּוֹתֵב שֶׁלְּמַעֲשֶׂה אֵין לְהָקֵל: מְתַכְנְנִים לֶאֱכֹל אֶת הָאֲרוּחוֹת בַּסֻּכָּה, וּבַטִּיּוּל אוֹכְלִים פֵּרוֹת, יְרָקוֹת וּמְעַט מְזוֹנוֹת. אֲבָל מִי שֶׁנּוֹסֵעַ לְמִצְוָה, כְּמוֹ לְהִתְפַּלֵּל בַּכֹּתֶל הַמַּעֲרָבִי, וְקָשֶׁה לוֹ לִמְצֹא סֻכָּה – רַשַּׁאי לֶאֱכֹל אֲרוּחָה בַּחוּץ.
💛 אַהֲבַת הַשֵּׁם: יְמֵי חֹל הַמּוֹעֵד קְדוֹשִׁים, וְנוֹעֲדוּ לְתוֹרָה, לִתְפִלָּה וְלִסְעוּדוֹת שֶׁל מִצְוָה. מִי שֶׁמַּקְדִּישׁ אוֹתָם לַה׳ – אָהוּב וְחָבִיב לְפָנָיו.
📖 פֶּרֶק ג׳, הֲלָכוֹת י״ג–י״ד

🧠 בְּדֹק אֶת עַצְמְךָ

בַּלַּיְלָה הָרִאשׁוֹן יוֹרֵד גֶּשֶׁם חָזָק. מָה נָכוֹן לַעֲשׂוֹת?
💡 נָכוֹן לְהַמְתִּין, אוּלַי הַגֶּשֶׁם יִפָּסֵק. אִם לֹא – מְקַדְּשִׁים בַּסֻּכָּה וְאוֹכְלִים כְּזַיִת לֶחֶם, לְדַעַת הָאוֹמְרִים שֶׁגַּם מִצְטַעֵר חַיָּב בַּלַּיְלָה הָרִאשׁוֹן. וְלֹא מְבָרְכִים ״לֵישֵׁב בַּסֻּכָּה״, כְּדֵי לָחוּשׁ לַדֵּעָה שֶׁהוּא פָּטוּר.
כַּמָּה לֶחֶם צָרִיךְ לֶאֱכֹל בַּסֻּכָּה בַּלַּיְלָה הָרִאשׁוֹן?
💡 מְדַקְדְּקִים כְּמוֹ בַּמַּצָּה שֶׁל לֵיל הַסֵּדֶר: לְפָחוֹת כַּחֲצִי בֵּיצָה, וְטוֹב יוֹתֵר מִבֵּיצָה. אוֹכְלִים בְּנַחַת, בִּרְצִיפוּת, וְלִפְנֵי חֲצוֹת הַלַּיְלָה.
בְּחֹל הַמּוֹעֵד יְהוֹנָתָן רוֹצֶה לֶאֱכֹל בַּמִּטְבָּח. מָה מֻתָּר לוֹ לֶאֱכֹל שָׁם?
💡 מִינֵי דָּגָן יוֹתֵר מִכְּבֵיצָה – גַּם אִטְרִיּוֹת וְגַם עוּגָה – הֵם אֲכִילַת קֶבַע. כְּבֵיצָה וּפָחוֹת – אֲכִילַת אַרְעַי, וּמֻתָּר בַּחוּץ. וַאֲרוּחָה שֶׁשְּׂבֵעִים מִמֶּנָּה, גַּם מִבָּשָׂר – רַק בַּסֻּכָּה.

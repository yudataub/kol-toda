צַעַר בַּעֲלֵי חַיִּים

🍽️ הִלְכוֹת כַּשְׁרוּת · פֶּרֶק ט״ו

🐈רַחֲמִים עַל כָּל בַּעֲלֵי הַחַיִּים

מִצְוָה לְרַחֵם עַל בַּעֲלֵי הַחַיִּים, וְאָסוּר לְצַעֵר אוֹתָם. מֻתָּר לְהִשְׁתַּמֵּשׁ בָּהֶם לַעֲבוֹדָה וְלַאֲכִילָה, אֲבָל לְהַכּוֹת בַּעַל חַיִּים לְשֵׁם הֲנָאָה, אוֹ מִתּוֹךְ כַּעַס – אָסוּר מִן הַתּוֹרָה.
אֶת חַיּוֹת הַהֶפְקֵר, כְּמוֹ חָתוּל רְחוֹב, אֵין חוֹבָה לְהַאֲכִיל. אֲבָל מִי שֶׁמְּרַחֵם עֲלֵיהֶן וְנוֹתֵן לָהֶן מִשְּׁיָרֵי הָאֹכֶל – מְקַיֵּם מִצְוָה.
💛 אַהֲבַת הַשֵּׁם: עַל ה׳ נֶאֱמַר: "וְרַחֲמָיו עַל כָּל מַעֲשָׂיו". כְּשֶׁאָנוּ מְרַחֲמִים עַל חָתוּל רָעֵב, אָנוּ הוֹלְכִים בִּדְרָכָיו שֶׁל ה׳.
📖 פֶּרֶק ט״ו, הֲלָכוֹת א׳, ו׳, ט״ו

💗וְרַחֲמָיו עַל כָּל מַעֲשָׂיו

לְכַתְּחִלָּה אֵין לְצַעֵר שׁוּם בַּעַל חַיִּים, גַּם לֹא אֶת הַקָּטָן בְּיוֹתֵר. כְּכָל שֶׁבַּעַל הַחַיִּים מְפֻתָּח יוֹתֵר – כְּמוֹ הַכֶּלֶב – הוּא מַרְגִּישׁ יוֹתֵר צַעַר, וְצָרִיךְ לְהִזָּהֵר בּוֹ יוֹתֵר. וּבַעֲלֵי חַיִּים שֶׁעוֹזְרִים לָנוּ – רָאוּי לְרַחֵם עֲלֵיהֶם עוֹד יוֹתֵר.
עַל שִׁלּוּחַ הַקֵּן הִבְטִיחָה הַתּוֹרָה: "לְמַעַן יִיטַב לָךְ וְהַאֲרַכְתָּ יָמִים" – מִי שֶׁמְּרַחֵם עַל הַבְּרִיּוֹת, ה׳ מְרַחֵם עָלָיו.
💛 אַהֲבַת הַשֵּׁם: כָּל מָה שֶׁבָּרָא ה׳ בְּעוֹלָמוֹ – לֹא בָּרָא דָּבָר אֶחָד לְבַטָּלָה. כָּל בַּעַל חַיִּים מְשַׁבֵּחַ אֶת ה׳ בְּדַרְכּוֹ, כְּמוֹ שֶׁמּוּבָא בְּפֶרֶק שִׁירָה.
📖 פֶּרֶק ט״ו, הֲלָכוֹת א׳, ט״ו; פֶּרֶק ט״ז, הֲלָכָה ב׳

🫏עוֹזְרִים לַבְּהֵמָה

מִי שֶׁרוֹאֶה חֲמוֹר שֶׁנָּפַל תַּחַת מַשָּׂא כָּבֵד – מִצְוָה לַעֲזֹר לְהוֹרִיד אֶת הַמַּשָּׂא, אֲפִלּוּ אִם הַחֲמוֹר שֶׁל אָדָם שֶׁאֲנַחְנוּ בְּרִיב אִתּוֹ. וּמִי שֶׁמַּעֲמִיס עַל בְּהֶמְתּוֹ יוֹתֵר מִכֹּחָהּ – עָתִיד לָתֵת עַל כָּךְ אֶת הַדִּין.
"לֹא תַחְסֹם שׁוֹר בְּדִישׁוֹ" – בְּהֵמָה שֶׁעוֹבֶדֶת בְּאֹכֶל, מַרְשִׁים לָהּ לֶאֱכֹל מִמֶּנּוּ. לֹא רָאוּי שֶׁהִיא תַּעֲבֹד בִּשְׁבִילֵנוּ וְלֹא תֵּהָנֶה מֵעֲבוֹדָתָהּ.
💛 אַהֲבַת הַשֵּׁם: בִּלְעָם הִכָּה אֶת אֲתוֹנוֹ בְּלִי סִבָּה, וְה׳ הִקְפִּיד עָלָיו: "עַל מָה הִכִּיתָ אֶת אֲתֹנְךָ?" ה׳ רוֹאֶה גַּם אֶת צַעֲרָהּ שֶׁל אָתוֹן, וְרוֹצֶה שֶׁנִּהְיֶה רַחֲמָנִים.
📖 פֶּרֶק ט״ו, הֲלָכוֹת ב׳–ג׳

⚖️פְּרִיקָה לִפְנֵי טְעִינָה

כְּשֶׁשְּׁנֵי אֲנָשִׁים צְרִיכִים עֶזְרָה – אֶחָד לְהוֹרִיד מַשָּׂא מֵחֲמוֹר שֶׁנָּפַל, וְאֶחָד לְהַעֲמִיס מַשָּׂא – עוֹזְרִים קֹדֶם בִּפְרִיקָה, כְּדֵי לְהָקֵל עַל צַעֲרוֹ שֶׁל הַחֲמוֹר. בִּפְרִיקָה עוֹזְרִים גַּם בְּלִי תַּשְׁלוּם, וּבִטְעִינָה אֵין חוֹבָה לַעֲזֹר בְּחִנָּם.
וְאִם הַטְּעִינָה הִיא שֶׁל אָדָם שֶׁשּׂוֹנֵא אוֹתָנוּ – מַקְדִּימִים לַעֲזֹר לוֹ, כְּדֵי לְהִתְגַּבֵּר עַל הַיֵּצֶר הָרַע, וְאוּלַי יַחְזְרוּ לֶאֱהֹב זֶה אֶת זֶה.
💛 אַהֲבַת הַשֵּׁם: הָרַחֲמִים עַל בַּעֲלֵי חַיִּים וְהָאַהֲבָה לִבְנֵי אָדָם הוֹלְכִים יַחַד. מִי שֶׁעוֹזֵר לַחֲמוֹר שֶׁל מִי שֶׁרָב אִתּוֹ, מְקָרֵב גַּם אֶת הַלְּבָבוֹת.
📖 פֶּרֶק ט״ו, הֲלָכָה ב׳

🥣קֹדֶם הַבְּהֵמָה

מִי שֶׁיֵּשׁ לוֹ בַּעֲלֵי חַיִּים – חַיָּב לִדְאֹג לִמְזוֹנָם, וְאָסוּר לוֹ לֶאֱכֹל לִפְנֵי שֶׁנָּתַן לָהֶם לֶאֱכֹל. כָּךְ לָמְדוּ חֲכָמִים מִן הַפָּסוּק: "וְנָתַתִּי עֵשֶׂב בְּשָׂדְךָ לִבְהֶמְתֶּךָ", וְרַק אַחַר כָּךְ "וְאָכַלְתָּ וְשָׂבָעְתָּ".
לֹא מַכְנִיסִים בַּעַל חַיִּים הַבַּיְתָה בְּלִי לָדַעַת שֶׁאֶפְשָׁר לָתֵת לוֹ מָזוֹן כָּרָאוּי, מָקוֹם מַתְאִים וְנִקָּיוֹן.
💛 אַהֲבַת הַשֵּׁם: מֹשֶׁה רַבֵּנוּ וְדָוִד הַמֶּלֶךְ הָיוּ רוֹעֵי צֹאן, וְדָאֲגוּ לַצֹּאן בְּרַחֲמִים. בִּזְכוּת זֶה בָּחַר בָּהֶם ה׳ לִהְיוֹת מַנְהִיגֵי יִשְׂרָאֵל.
📖 פֶּרֶק ט״ו, הֲלָכוֹת ד׳, י״ג

🦜חַיּוֹת מַחְמָד

מֻתָּר לְגַדֵּל חַיּוֹת מַחְמָד, כְּמוֹ כֶּלֶב, חָתוּל וְתֻכִּי. חֲכָמִים שִׁבְּחוּ אֶת נֶאֱמָנוּתוֹ שֶׁל הַכֶּלֶב וְאֶת צְנִיעוּתוֹ שֶׁל הֶחָתוּל. כְּשֶׁחַיַּת הַמַּחְמָד חוֹלָה – מְטַפְּלִים בָּהּ, וְכֶלֶב שֶׁמַּפְחִיד אֶת הָאוֹרְחִים – שׁוֹלְחִים לְאִלּוּף.
הַחוֹבָה לָתֵת לָהֶם לֶאֱכֹל לִפְנֵינוּ נוֹגַעַת בְּעִקָּר לְבַעֲלֵי חַיִּים שֶׁאֵינָם יְכוֹלִים לִמְצֹא אֹכֶל לְבַד, כְּמוֹ תֻּכִּי בַּכְּלוּב אוֹ תַּרְנְגוֹלוֹת בַּלּוּל. לְכֶלֶב וּלְחָתוּל שֶׁמִּסְתּוֹבְבִים בַּחוּץ – זוֹ מִדַּת חֲסִידוּת.
💛 אַהֲבַת הַשֵּׁם: הַכֶּלֶב נֶאֱמָן לִבְעָלָיו וְשׁוֹמֵר עַל הַבַּיִת, וְרָאוּי לְהַחֲזִיר לוֹ טוֹבָה. מִי שֶׁמַּכִּיר טוֹבָה אֲפִלּוּ לְכֶלֶב, יַכִּיר טוֹבָה גַּם לְהוֹרָיו וְלַה׳.
📖 פֶּרֶק ט״ו, הֲלָכוֹת ד׳, י״ג

🎣צַיִד, דַּיִג וּמוֹפָעִים אַכְזָרִיִּים

לֹא רָאוּי לָצוּד חַיּוֹת וְצִפּוֹרִים לְשֵׁם סְפּוֹרְט וְשַׁעֲשׁוּעַ. הַצַּיָּדִים שֶׁנִּזְכְּרוּ בַּתּוֹרָה, נִמְרוֹד וְעֵשָׂו, הָיוּ רְשָׁעִים שֶׁנֶּהֱנוּ מִן הָאַכְזָרִיּוּת. לָדוּג דָּגִים בְּחַכָּה כְּדֵי לֶאֱכֹל אוֹתָם – מֻתָּר. אֲבָל לָדוּג לְשֵׁם הֲנָאָה וּלְהַחֲזִיר אֶת הַדָּגִים לַמַּיִם – עָדִיף שֶׁלֹּא.
אָסוּר לִצְפּוֹת בְּמִלְחֲמוֹת שְׁוָרִים, וְאָסוּר לְאַלֵּף תַּרְנְגוֹלִים לְהִלָּחֵם זֶה בָּזֶה. מִי שֶׁקּוֹנֶה כַּרְטִיס לְמוֹפָע כָּזֶה נַעֲשֶׂה שֻׁתָּף לַעֲבֵרָה.
💛 אַהֲבַת הַשֵּׁם: "אַשְׁרֵי הָאִישׁ אֲשֶׁר לֹא הָלַךְ בַּעֲצַת רְשָׁעִים" – כָּךְ אָמְרוּ חֲכָמִים עַל מִי שֶׁלֹּא הָלַךְ לְמוֹפָעִים אַכְזָרִיִּים. עַם יִשְׂרָאֵל רַחֲמָנִים, וְהַשִּׂמְחָה שֶׁלָּנוּ אֵינָהּ בָּאָה מִצַּעַר שֶׁל אַחֵר.
📖 פֶּרֶק ט״ו, הֲלָכוֹת ו׳, י״ב

🐕גַּם כְּשֶׁמֻּתָּר – מְרַחֲמִים

יֵשׁ דְּבָרִים שֶׁמְּצַעֲרִים מְעַט בַּעַל חַיִּים, אֲבָל נַעֲשִׂים לְתוֹעֶלֶת הָאָדָם – כְּמוֹ לַחְתֹּךְ לַכֶּלֶב אֶת הַזָּנָב אוֹ אֶת הָאָזְנַיִם כְּדֵי שֶׁיֵּרָאֶה יָפֶה. מֵעִקַּר הַדִּין זֶה מֻתָּר, כִּי אֵין זוֹ אַכְזָרִיּוּת. אֲבָל מִצַּד מִדַּת הָרַחֲמָנוּת – עָדִיף שֶׁלֹּא לְצַעֵר אוֹתוֹ.
הַכְּלָל: כְּכָל שֶׁהַצַּעַר שֶׁל בַּעַל הַחַיִּים גָּדוֹל יוֹתֵר, וְהַצֹּרֶךְ שֶׁל הָאָדָם קָטָן יוֹתֵר – כָּךְ רָאוּי יוֹתֵר לְהִמָּנַע. מֻתָּר לְאַלֵּף כֶּלֶב בְּמַכּוֹת קַלּוֹת, וּמִמַּכּוֹת קָשׁוֹת – מִדַּת חֲסִידוּת לְהִמָּנַע.
💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁנִּהְיֶה רַחֲמָנִים גַּם בְּמָקוֹם שֶׁהַהֲלָכָה מַתִּירָה. מִי שֶׁמְּוַתֵּר עַל מְעַט יֹפִי כְּדֵי לֹא לְצַעֵר כֶּלֶב – הוֹלֵךְ בִּדְרָכָיו שֶׁל ה׳, שֶׁרַחֲמָיו עַל כָּל מַעֲשָׂיו.
📖 פֶּרֶק ט״ו, הֲלָכָה ח׳

🐕כֶּלֶב מַפְחִיד וּמַזִּיקִים

אָסוּר לְגַדֵּל כֶּלֶב שֶׁמַּפְחִיד אֶת הָעוֹבְרִים בָּרְחוֹב, אוֹ שֶׁנּוֹבֵחַ וּמַפְרִיעַ לַשְּׁכֵנִים. גַּם כֶּלֶב מְאֻלָּף – אִם הָעוֹבְרִים אֵינָם יוֹדְעִים זֹאת וּמְפַחֲדִים מִמֶּנּוּ – אָסוּר. רַק כְּשֶׁהוּא קָשׁוּר הֵיטֵב אוֹ מְגֻדָּר, וְכֻלָּם רוֹאִים שֶׁאֵינוֹ יָכוֹל לִתְקֹף – מֻתָּר.
נָחָשׁ אוֹ עַקְרָב בְּמָקוֹם שֶׁהוֹלְכִים בּוֹ אֲנָשִׁים – מִצְוָה לְהָרְגוֹ, כְּדֵי שֶׁלֹּא יַזִּיק, וְיֶלֶד קוֹרֵא מִיָּד לְאָדָם מְבֻגָּר. וּמֻתָּר לְהָרֹג עַכְבָּרִים, נְמָלִים וּזְבוּבִים שֶׁנִּכְנְסוּ לַבַּיִת.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה דּוֹאֶגֶת גַּם לְבַעֲלֵי הַחַיִּים וְגַם לַשְּׁכֵנִים. חַיַּת מַחְמָד צְרִיכָה לְהָבִיא שִׂמְחָה – וְלֹא פַּחַד.
📖 פֶּרֶק ט״ו, הֲלָכָה י״א

🧠 בְּדֹק אֶת עַצְמְךָ

שְׁנֵי חֲבֵרִים מְבַקְּשִׁים עֶזְרָה בְּאוֹתוֹ רֶגַע: אֶחָד – לְהוֹרִיד מַשָּׂא מֵחֲמוֹר שֶׁנָּפַל, וְהַשֵּׁנִי – לְהַעֲמִיס מַשָּׂא עַל חֲמוֹרוֹ. לְמִי עוֹזְרִים קֹדֶם?
💡 פְּרִיקָה קוֹדֶמֶת לִטְעִינָה, כִּי הַחֲמוֹר שֶׁנָּפַל סוֹבֵל עַכְשָׁו. וּבִפְרִיקָה עוֹזְרִים גַּם בְּחִנָּם, כְּדֵי לְהַצִּיל אֶת הַחֲמוֹר מִצַּעַר.
לַשָּׁכֵן יֵשׁ כֶּלֶב גָּדוֹל וְטוֹב, אֲבָל הוּא נוֹבֵחַ בְּקוֹל עַל כָּל מִי שֶׁעוֹבֵר בַּשְּׁבִיל, וְהַיְּלָדִים מְפַחֲדִים. מָה הַדִּין?
💡 אָסוּר לְגַדֵּל כֶּלֶב שֶׁמַּפְחִיד אֶת הָעוֹבְרִים, גַּם אִם הוּא מְאֻלָּף. כְּשֶׁכֻּלָּם רוֹאִים שֶׁהוּא קָשׁוּר אוֹ מְגֻדָּר – אֵין מִמָּה לְפַחֵד, וּמֻתָּר.
אֵיזֶה מִן הַבִּלּוּיִים הָאֵלֶּה מֻתָּר לִיהוּדִי?
💡 דַּיִג לְשֵׁם אֲכִילָה הוּא צֹרֶךְ אֲמִתִּי, וְהַדָּגִים מַרְגִּישִׁים פָּחוֹת צַעַר. צַיִד לְשֵׁם סְפּוֹרְט אֵינוֹ רָאוּי, וּמוֹפָעִים אַכְזָרִיִּים – אֲסוּרִים.

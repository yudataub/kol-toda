חַשְׁמַל וּמַכְשִׁירָיו

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק י״ז

💡חַשְׁמַל

הַדְלָקַת נוּרָה חַשְׁמַלִּית אוֹ גּוּף חִמּוּם אֲסוּרָה מִן הַתּוֹרָה, כִּי הַחוּט שֶׁבָּהֶם מִתְלַהֵט וּבוֹעֵר. כִּבּוּי נוּרָה – אָסוּר מִדִּבְרֵי חֲכָמִים. גַּם שְׁאָר הַמַּכְשִׁירִים הַחַשְׁמַלִּיִּים – טֶלֶפוֹן, מְאַוְרֵר, מַזְגָן, פַּעֲמוֹן וּמַחְשֵׁב – אָסוּר לְהַפְעִיל, וְיֵשׁ לָחוּשׁ לְדַעַת הָאוֹמְרִים שֶׁהָאִסּוּר מִן הַתּוֹרָה.
גַּם לְהַפְעִיל מַכְשִׁיר בְּשַׁלָּט – אָסוּר, כְּמוֹ בִּלְחִיצָה עַל כַּפְתּוֹר.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת הַבַּיִת שָׁקֵט מִצִּלְצוּלִים וּמִמַּסָּכִים. בַּשֶּׁקֶט הַזֶּה אֶפְשָׁר סוֹף סוֹף לִשְׁמֹעַ זֶה אֶת זֶה – וְלִשְׁמֹעַ אֶת קוֹלוֹ שֶׁל ה׳ בַּלֵּב.
📖 פֶּרֶק י״ז, הֲלָכוֹת א׳–ב׳

📞מַכְשִׁירִים שֶׁמַּשְׁמִיעִים קוֹל

אָסוּר לְדַבֵּר בְּטֶלֶפוֹן אוֹ בְּרַמְקוֹל בְּשַׁבָּת – גַּם אִם הִפְעִילוּ אוֹתָם לִפְנֵי שַׁבָּת, כִּי הַדִּבּוּר מַגְבִּיר אֶת הַזֶּרֶם שֶׁבָּהֶם. אֲבָל מִי שֶׁשּׁוֹמֵעַ קָשֶׁה – מֻתָּר לוֹ לְהִשְׁתַּמֵּשׁ בְּמַכְשִׁיר שְׁמִיעָה שֶׁהִפְעִיל לִפְנֵי שַׁבָּת, בְּלִי לְהַגְבִּיר וּבְלִי לְהַנְמִיךְ אוֹתוֹ.
שָׁעוֹן מְעוֹרֵר חַשְׁמַלִּי שֶׁהִתְחִיל לְצַלְצֵל – אָסוּר לְהַפְסִיק אֶת הַצִּלְצוּל. אֶפְשָׁר לַעֲטֹף אוֹתוֹ בִּשְׂמִיכוֹת וּלְהַנִּיחַ אוֹתוֹ בְּמָקוֹם שֶׁלֹּא יַפְרִיעַ.
💛 אַהֲבַת הַשֵּׁם: הַקּוֹל הֶחָשׁוּב בְּשַׁבָּת הוּא הַקּוֹל הַחַי: שִׁירֵי שַׁבָּת, דִּבְרֵי תּוֹרָה וְשִׂיחָה בַּמִּשְׁפָּחָה. כְּשֶׁהַמַּכְשִׁירִים שׁוֹתְקִים, הַלֵּב פָּנוּי לִשְׁמֹעַ אֶת הַקּוֹלוֹת הָאֵלֶּה.
📖 פֶּרֶק י״ז, הֲלָכוֹת ג׳, י״ז

🛗מַעֲלִית שַׁבָּת

אָסוּר לִלְחֹץ עַל כַּפְתּוֹרֵי מַעֲלִית בְּשַׁבָּת. בְּמַעֲלִית שַׁבָּת – שֶׁעוֹצֶרֶת מֵעַצְמָהּ בַּקּוֹמוֹת לְפִי תָּכְנִית שֶׁהֻפְעֲלָה מִלִּפְנֵי שַׁבָּת – נֶחְלְקוּ הַפּוֹסְקִים. הָעִקָּר לַהֲלָכָה כַּמַּקִּילִים, שֶׁמֻּתָּר לְהִשְׁתַּמֵּשׁ בָּהּ, וְהַמַּחְמִירִים – תָּבוֹא עֲלֵיהֶם בְּרָכָה.
מִי שֶׁמִּשְׁתַּמֵּשׁ בְּמַעֲלִית שַׁבָּת – לֹא נִכְנָס וְלֹא יוֹצֵא כְּשֶׁהַדֶּלֶת עוֹמֶדֶת לְהִסָּגֵר, כְּדֵי שֶׁלֹּא יִגְרֹם לָהּ לְהִפָּתַח שׁוּב.
💛 אַהֲבַת הַשֵּׁם: הַמַּעֲלִית שֶׁל שַׁבָּת אִטִּית יוֹתֵר, וְזֶה בְּסֵדֶר. בְּשַׁבָּת לֹא מְמַהֲרִים – הַזְּמַן עַצְמוֹ נַעֲשֶׂה רָגוּעַ וְקָדוֹשׁ.
📖 פֶּרֶק י״ז, הֲלָכָה ד׳

🔀דֵּעוֹת וּמִנְהָגִים בְּחַשְׁמַל

שְׁעוֹן שַׁבָּת: יֵשׁ אוֹסְרִים לִדְחוֹת בְּשַׁבָּת אֶת פְּעֻלָּתוֹ, וְהַמִּנְהָג הָרוֹוֵחַ לְהָקֵל. מַעֲלִית שַׁבָּת: יֵשׁ אוֹסְרִים לְהִשְׁתַּמֵּשׁ בָּהּ, יֵשׁ מַתִּירִים לַעֲלוֹת וְלֹא לָרֶדֶת, וְהָרַב מֶלָמֵד פּוֹסֵק כַּמְּקִלִּים – וְהַמַּחְמִירִים תָּבוֹא עֲלֵיהֶם בְּרָכָה. שָׁעוֹן שֶׁנִּטְעָן מִתְּנוּעוֹת הַיָּד אוֹ מִן הָאוֹר: יֵשׁ מַחְמִירִים שֶׁלֹּא לַעֲנֹד אוֹתוֹ, וְיֵשׁ מְקִלִּים, אִם הוּא יָכוֹל לַעֲבֹד עוֹד כַּמָּה יָמִים בְּלִי טְעִינָה.
פַּעֲמוֹן חַשְׁמַלִּי בַּכְּנִיסָה לְבִנְיָן: כְּשֶׁאֵין שׁוּם דֶּרֶךְ אַחֶרֶת שֶׁיִּשְׁמְעוּ אוֹתוֹ, יֵשׁ מַתִּירִים לִלְחֹץ עָלָיו בְּשִׁנּוּי, וְיֵשׁ אוֹסְרִים גַּם בְּשִׁנּוּי. הָרוֹצֶה לִסְמֹךְ עַל הַמְּקִלִּים – רַשַּׁאי, רַק בִּשְׁעַת הַדְּחָק וּפַעַם אַחַת.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁהַפּוֹסְקִים חֲלוּקִים, כֻּלָּם רוֹצִים דָּבָר אֶחָד: לִשְׁמֹר עַל קְדֻשַּׁת הַשַּׁבָּת. כָּל מִשְׁפָּחָה הוֹלֶכֶת בְּדֶרֶךְ רַבּוֹתֶיהָ, וְכֻלָּם יַחַד מְכַבְּדִים אֶת הַיּוֹם.
📖 פֶּרֶק י״ז, הֲלָכוֹת ד׳, ו׳, י״ג, י״ז

⏰שְׁעוֹן שַׁבָּת

לִפְנֵי שַׁבָּת מֻתָּר לְכַוֵּן שְׁעוֹן שַׁבָּת, שֶׁיַּדְלִיק וִיכַבֶּה אֶת הָאוֹר אוֹ מַכְשִׁירִים בַּשָּׁעוֹת הָרְצוּיוֹת. בְּשַׁבָּת עַצְמָהּ יֵשׁ אוֹסְרִים לְהָזִיז אֶת הַשָּׁעוֹן, וְהַמִּנְהָג הָרוֹוֵחַ לְהָקֵל וְלִדְחוֹת אֶת פְּעֻלָּתוֹ, כְּדֵי שֶׁהַמַּצָּב הַקַּיָּם יִמָּשֵׁךְ זְמַן רַב יוֹתֵר. אֲבָל לְהַקְדִּים אֶת הַהַדְלָקָה אוֹ אֶת הַכִּבּוּי – אָסוּר לְכֻלָּם.
לְמָשָׁל: הָאוֹר אָמוּר לִכְבּוֹת בְּאַחַת עֶשְׂרֵה – לְפִי הַמִּנְהָג הָרוֹוֵחַ מֻתָּר לִדְחוֹת לִשְׁתֵּים עֶשְׂרֵה, אֲבָל אָסוּר לְהַקְדִּים לְעֶשֶׂר.
💛 אַהֲבַת הַשֵּׁם: שְׁעוֹן שַׁבָּת הוּא דֻּגְמָה לַהֲכָנָה נְבוֹנָה: מְתַכְנְנִים מֵרֹאשׁ, וּבְשַׁבָּת נֶהֱנִים. מִי שֶׁחוֹשֵׁב עַל הַשַּׁבָּת כְּבָר בְּיוֹם שִׁשִּׁי מַרְאֶה כַּמָּה הִיא יְקָרָה לוֹ.
📖 פֶּרֶק י״ז, הֲלָכָה ו׳

🧊מְקָרֵר וּמַזְגָן

בְּרֹב הַמְּקָרְרִים נִדְלֶקֶת נוּרָה כְּשֶׁפּוֹתְחִים אֶת הַדֶּלֶת – וְלָכֵן מְנַתְּקִים אוֹתָהּ לִפְנֵי שַׁבָּת. אִם דָּאֲגוּ שֶׁפְּתִיחַת הַמְּקָרֵר לֹא תַּדְלִיק וְלֹא תְּכַבֶּה דָּבָר – מֻתָּר לִפְתֹּחַ אוֹתוֹ, גַּם כְּשֶׁהַמָּנוֹעַ אֵינוֹ עוֹבֵד. וּמֻתָּר לְהִכָּנֵס לְחֶדֶר שֶׁפּוֹעֵל בּוֹ מַזְגָן.
מִי שֶׁשָּׁכַח לְנַתֵּק אֶת הַנּוּרָה שֶׁבַּמְּקָרֵר – אָסוּר לוֹ לִפְתֹּחַ וְלִסְגֹּר אוֹתוֹ. וְתַנּוּר אֲפִיָּה שֶׁפּוֹעֵל בְּ"מַצַּב שַׁבָּת" – מֻתָּר לִפְתֹּחַ לְכָל הַדֵּעוֹת.
💛 אַהֲבַת הַשֵּׁם: הָרְשִׁימָה שֶׁל יוֹם שִׁשִּׁי – לְנַתֵּק נוּרָה, לְכַוֵּן שָׁעוֹן – הִיא חֵלֶק מִכְּבוֹד הַשַּׁבָּת. כָּל הֲכָנָה קְטַנָּה הִיא כְּמוֹ פֶּרַח שֶׁאָנוּ מְבִיאִים לִכְבוֹד הַמַּלְכָּה.
📖 פֶּרֶק י״ז, הֲלָכוֹת ח׳–ט׳

🚪דְּלָתוֹת וְחַיְשָׁנִים

אָסוּר לְהִתְקָרֵב לְדֶלֶת חַשְׁמַלִּית שֶׁנִּפְתַּחַת מֵאֵלֶיהָ, כִּי הַמִּתְקָרֵב מַפְעִיל אוֹתָהּ. גַּם לְחֶדֶר שֶׁהָאוֹר נִדְלָק בּוֹ מֵעַצְמוֹ כְּשֶׁנִּכְנָסִים – אָסוּר לְהִכָּנֵס. אֲבָל כְּשֶׁגּוֹי נִגָּשׁ לַדֶּלֶת כְּדֵי לְהִכָּנֵס בְּעַצְמוֹ – מֻתָּר לְהִצְטָרֵף אֵלָיו וּלְהִכָּנֵס.
מֻתָּר לַעֲבֹר בָּרְחוֹב לְיַד מַצְלֵמָה אוֹ חַיְשָׁן, כִּי הָעוֹבֵר אֵינוֹ רוֹצֶה בִּפְעֻלָּתָם. אֲבָל פָּנָס שֶׁנִּדְלָק בַּשְּׁבִיל שֶׁלִּפְנֵי הַבַּיִת – צָרִיךְ לְנַתֵּק לִפְנֵי שַׁבָּת.
💛 אַהֲבַת הַשֵּׁם: בָּעוֹלָם הַמָּלֵא מַכְשִׁירִים, הַשַּׁבָּת מְבַקֶּשֶׁת מֵאִתָּנוּ לִהְיוֹת עֵרִים. מִי שֶׁשָּׂם לֵב לְכָל דֶּלֶת וּלְכָל אוֹר – מַרְאֶה שֶׁהַשַּׁבָּת חַיָּה בְּתוֹכוֹ בְּכָל צַעַד.
📖 פֶּרֶק י״ז, הֲלָכוֹת י״א–י״ד

⌚שְׁעוֹן יָד בְּשַׁבָּת

מֻתָּר לַעֲנֹד שְׁעוֹן יָד אֶלֶקְטְרוֹנִי שֶׁרוֹאִים בּוֹ אֶת הַשָּׁעָה בְּלִי שׁוּם פְּעֻלָּה חַשְׁמַלִּית – גַּם אִם יֵשׁ בּוֹ לַחְצַן תְּאוּרָה אוֹ שְׁעוֹן עֶצֶר, כִּי עִקַּר שִׁמּוּשׁוֹ לְהֶתֵּר. אֲבָל שָׁעוֹן שֶׁצָּרִיךְ לִלְחֹץ עַל כַּפְתּוֹר כְּדֵי לִרְאוֹת בּוֹ אֶת הַשָּׁעָה – אָסוּר לַעֲנֹד, וְהוּא מֻקְצֶה. וְגַם שָׁעוֹן שֶׁהַמָּסָךְ שֶׁלּוֹ כָּבֶה, וְנִדְלָק כְּשֶׁמַּזִּיזִים אֶת הַיָּד – אָסוּר לַעֲנֹד בְּשַׁבָּת.
מִי שֶׁיּוֹדֵעַ בְּעַצְמוֹ שֶׁאוּלַי יִלְחַץ בְּשַׁבָּת עַל הַכַּפְתּוֹרִים – לֹא יַעֲנֹד שָׁעוֹן כָּזֶה. וְעַל שָׁעוֹן שֶׁנִּטְעָן מִתְּנוּעוֹת הַיָּד אוֹ מִן הָאוֹר – יֵשׁ מַחֲלֹקֶת, וְנִלְמַד עָלֶיהָ בָּרָמָה הַבָּאָה.
💛 אַהֲבַת הַשֵּׁם: הַזְּמַן שֶׁל שַׁבָּת שׁוֹנֶה: אֲנַחְנוּ לֹא רָצִים אַחֲרָיו, אֶלָּא נֶהֱנִים מִמֶּנּוּ. גַּם כְּשֶׁאָנוּ מַבִּיטִים בַּשָּׁעוֹן, אָנוּ זוֹכְרִים שֶׁכָּל רֶגַע בְּשַׁבָּת הוּא מַתָּנָה מֵה׳.
📖 פֶּרֶק י״ז, הֲלָכָה י״ז

🧠 בְּדֹק אֶת עַצְמְךָ

הַשָּׁעוֹן הַמְּעוֹרֵר הַחַשְׁמַלִּי הִתְחִיל לְצַלְצֵל בְּשַׁבָּת בַּבֹּקֶר, וְלֹא מַפְסִיק. מָה יַעֲשֶׂה יְהוּדָה?
💡 הַפְסָקַת הַצִּלְצוּל הִיא נִתּוּק שֶׁל פְּעֻלָּה חַשְׁמַלִּית, וַאֲסוּרָה – גַּם בְּיָדֵי אָח קָטָן. אֲבָל מֻתָּר לַעֲטֹף אֶת הַשָּׁעוֹן בִּשְׂמִיכוֹת, כְּדֵי שֶׁהָרַעַשׁ לֹא יַפְרִיעַ.
אַבָּא וּבָנָיו מִתְאָרְחִים בְּשַׁבָּת בְּמָלוֹן, וּבַכְּנִיסָה יֵשׁ דֶּלֶת חַשְׁמַלִּית. גּוֹי נִגָּשׁ לַדֶּלֶת כְּדֵי לְהִכָּנֵס. מָה מֻתָּר לָהֶם?
💡 הַגּוֹי פּוֹתֵחַ אֶת הַדֶּלֶת בִּשְׁבִיל עַצְמוֹ, וּמֻתָּר לְהִצְטָרֵף אֵלָיו. אֲבָל הַמִּתְקָרֵב בְּעַצְמוֹ מַפְעִיל אֶת הַדֶּלֶת, וְאַחֲרֵי יְהוּדִי שֶׁחִלֵּל שַׁבָּת – אָסוּר לְהִכָּנֵס.
לְסָבָא יֵשׁ מַכְשִׁיר שְׁמִיעָה חַשְׁמַלִּי. מָה הַדִּין בְּשַׁבָּת?
💡 מַכְשִׁיר שְׁמִיעָה שֶׁהֻפְעַל לִפְנֵי שַׁבָּת – מֻתָּר, כִּי הַקּוֹל שֶׁל הַמְּדַבְּרִים מַפְעִיל אוֹתוֹ רַק בַּעֲקִיפִין, וְאֵין בּוֹ הַשְׁמָעַת קוֹל לָרַבִּים. אֲבָל לְהַגְבִּיר, לְהַנְמִיךְ אוֹ לְהַדְלִיק – אָסוּר.

מִצְוַת הַשּׁוֹפָר

🎺 רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים · פֶּרֶק ד׳

🎺מִצְוָה לִשְׁמֹעַ

מִצְוַת עֲשֵׂה לִשְׁמֹעַ תְּרוּעַת שׁוֹפָר בְּרֹאשׁ הַשָּׁנָה: ״יוֹם תְּרוּעָה יִהְיֶה לָכֶם״. לִפְנֵי הַתְּקִיעוֹת מְבָרְכִים ״לִשְׁמֹעַ קוֹל שׁוֹפָר״ וְ״שֶׁהֶחֱיָנוּ״. הַמִּצְוָה הִיא לִשְׁמֹעַ, וְלֹא לִתְקֹעַ.
צָרִיךְ לִשְׁמֹעַ אֶת קוֹל הַשּׁוֹפָר עַצְמוֹ. מִי שֶׁשָּׁמַע רַק הֵד, אוֹ קוֹל מְעֻרְבָּב עִם הֵד – לֹא יָצָא.
💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לָנוּ מִצְוָה שֶׁל הַקְשָׁבָה. כְּשֶׁאֲנַחְנוּ שׁוֹמְעִים אֶת הַשּׁוֹפָר, אֲנַחְנוּ פּוֹתְחִים אֶת הַלֵּב לִשְׁמֹעַ אֶת מָה שֶׁה׳ רוֹצֶה מֵאִתָּנוּ.
📖 פֶּרֶק ד׳, הֲלָכוֹת א׳, ה׳

🔊שְׁלֹשָׁה קוֹלוֹת

תְּקִיעָה – קוֹל אָרֹךְ וּפָשׁוּט, שֶׁל שִׂמְחָה. שְׁבָרִים – שְׁלֹשָׁה קוֹלוֹת בֵּינוֹנִיִּים, כְּמוֹ אֲנָחָה. תְּרוּעָה – קוֹלוֹת קְצָרִים וּרְצוּפִים, כְּמוֹ בֶּכִי. לִפְנֵי כָּל תְּרוּעָה וְאַחֲרֶיהָ תּוֹקְעִים תְּקִיעָה.
רַבִּי אַבָּהוּ תִּקֵּן לִתְקֹעַ אֶת כָּל סוּגֵי הַתְּרוּעָה: שָׁלֹשׁ פְּעָמִים תשר״ת, שָׁלֹשׁ פְּעָמִים תש״ת וְשָׁלֹשׁ פְּעָמִים תר״ת – בְּיַחַד שְׁלֹשִׁים תְּקִיעוֹת.
💛 אַהֲבַת הַשֵּׁם: גַּם הַבֶּכִי שֶׁל הַתְּרוּעָה עָטוּף בְּשִׂמְחָה שֶׁל תְּקִיעָה. גַּם הַדִּין וְהַקְּשָׁיִים מְכֻוָּנִים מִן הַשָּׁמַיִם לְטוֹבָתֵנוּ.
📖 פֶּרֶק ד׳, הֲלָכוֹת א׳–ב׳

💔תְּקִיעָה, תְּרוּעָה, תְּקִיעָה

הַתְּקִיעָה הָרִאשׁוֹנָה – הַנְּשָׁמָה הַטְּהוֹרָה, כְּמוֹ יֶלֶד שֶׁעוֹד לֹא חָטָא. הַתְּרוּעָה – הַקְּשָׁיִים וְהַנְּפִילוֹת שֶׁבַּחַיִּים, כַּאֲנָחָה וְכִבְכִי. וְהַתְּקִיעָה הָאַחֲרוֹנָה – הַיֹּשֶׁר שֶׁחוֹזֵר אַחֲרֵי הַתְּשׁוּבָה.
כָּל יְחִידָה כָּזוֹ מְבַטֵּאת תְּחוּם אַחֵר בַּחַיִּים: הַהַתְחָלָה הַטּוֹבָה, הַמַּשְׁבֵּר שֶׁבָּאֶמְצַע, וְהַתִּקּוּן שֶׁבַּסּוֹף.
💛 אַהֲבַת הַשֵּׁם: כָּל אֶחָד מֵאִתָּנוּ נוֹפֵל לִפְעָמִים – וְהַשּׁוֹפָר מְבַשֵּׂר: אֶפְשָׁר לַחֲזֹר, וּלִהְיוֹת טוֹבִים עוֹד יוֹתֵר מִקֹּדֶם.
📖 פֶּרֶק ד׳, הֲלָכָה א׳

💯מֵאָה תְּקִיעוֹת

כִּמְעַט בְּכָל הַקְּהִלּוֹת נוֹהֲגִים הַיּוֹם לִתְקֹעַ מֵאָה תְּקִיעוֹת, בְּעִקְבוֹת הָאֲרִ״י הַקָּדוֹשׁ: שְׁלֹשִׁים לִפְנֵי מוּסָף, שְׁלֹשִׁים בַּחֲזָרַת הַשַּׁ״ץ שֶׁל מוּסָף – עֶשֶׂר אַחֲרֵי כָּל אַחַת מִשָּׁלֹשׁ הַבְּרָכוֹת – וְעוֹד אַרְבָּעִים בְּהֶמְשֵׁךְ הַתְּפִלָּה.
מִן הַתּוֹרָה חַיָּבִים לִשְׁמֹעַ תֵּשַׁע תְּקִיעוֹת, וּבִזְכוּת תַּקָּנַת רַבִּי אַבָּהוּ חַיָּבִים הַיּוֹם שְׁלֹשִׁים. מֵאָה – זֶה מִנְהַג כָּל יִשְׂרָאֵל.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנַחְנוּ מוֹסִיפִים יוֹתֵר מִמָּה שֶׁחַיָּבִים, אֲנַחְנוּ מַרְאִים כַּמָּה הַמִּצְוָה חֲבִיבָה עָלֵינוּ.
📖 פֶּרֶק ד׳, הֲלָכוֹת ב׳, ד׳

🎼לָמָּה שְׁלֹשִׁים?

הַתְּרוּעָה הִיא קוֹל שֶׁל צַעַר, וּבְמֶשֶׁךְ הַשָּׁנִים הָיָה סָפֵק אֵיךְ הִיא נִשְׁמַעַת: כַּאֲנָחוֹת (שְׁבָרִים), כְּבֶכִי (תְּרוּעָה), אוֹ שְׁנֵיהֶם יַחַד. רַבִּי אַבָּהוּ תִּקֵּן בְּעִירוֹ קֵיסָרִי לִתְקֹעַ אֶת שְׁלָשְׁתָּם, וְכָל הַקְּהִלּוֹת קִבְּלוּ אֶת תַּקָּנָתוֹ.
הַסֵּדֶר: שָׁלֹשׁ פְּעָמִים תשר״ת (תְּקִיעָה, שְׁבָרִים, תְּרוּעָה, תְּקִיעָה), שָׁלֹשׁ פְּעָמִים תש״ת וְשָׁלֹשׁ פְּעָמִים תר״ת. מִן הַתּוֹרָה – תֵּשַׁע תְּקִיעוֹת, וְהַיּוֹם חַיָּבִים שְׁלֹשִׁים.
💛 אַהֲבַת הַשֵּׁם: כְּמוֹ אָדָם שֶׁמִּצְטַעֵר – מַתְחִיל בַּאֲנָחָה וּמַמְשִׁיךְ בְּבֶכִי. ה׳ שׁוֹמֵעַ כָּל סוּג שֶׁל קוֹל שֶׁיּוֹצֵא מִן הַלֵּב.
📖 פֶּרֶק ד׳, הֲלָכָה ב׳

🪑תְּקִיעוֹת דִּמְיֻשָּׁב

הַתְּקִיעוֹת שֶׁלִּפְנֵי מוּסָף נִקְרָאוֹת ״תְּקִיעוֹת דִּמְיֻשָּׁב״, כִּי חֲכָמִים הִתִּירוּ לַשּׁוֹמְעִים לָשֶׁבֶת בָּהֶן. מִנְהַג הַסְּפָרַדִּים וְהַתֵּימָנִים – לָשֶׁבֶת. מִנְהַג הָאַשְׁכְּנַזִּים – לַעֲמֹד, כִּי בָּהֶן כְּבָר יוֹצְאִים יְדֵי חוֹבָה.
תּוֹקְעִים גַּם לִפְנֵי מוּסָף וְגַם בְּתוֹכוֹ – ״כְּדֵי לְעַרְבֵּב אֶת הַשָּׂטָן״: כְּשֶׁהוּא רוֹאֶה שֶׁיִּשְׂרָאֵל מְחַבְּבִים אֶת הַמִּצְווֹת, נִסְתָּמוֹת טַעֲנוֹתָיו.
💛 אַהֲבַת הַשֵּׁם: הַתְּקִיעוֹת מְקַשְּׁרוֹת אוֹתָנוּ לְאָבִינוּ שֶׁבַּשָּׁמַיִם – וּמִי שֶׁקָּשׁוּר לְאַבָּא, שׁוּם מְקַטְרֵג לֹא יָכוֹל לוֹ.
📖 פֶּרֶק ד׳, הֲלָכָה ג׳

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

תְּקִיעוֹת דִּמְיֻשָּׁב: הָאַשְׁכְּנַזִּים עוֹמְדִים, וְהַסְּפָרַדִּים יוֹשְׁבִים. ״שֶׁהֶחֱיָנוּ״ עַל הַשּׁוֹפָר בַּיּוֹם הַשֵּׁנִי: הָאַשְׁכְּנַזִּים מְבָרְכִים, וְטוֹב שֶׁהַתּוֹקֵעַ יִלְבַּשׁ בֶּגֶד חָדָשׁ; הַסְּפָרַדִּים אֵינָם מְבָרְכִים. בִּרְכַּת נָשִׁים עַל הַשּׁוֹפָר: לְמִנְהַג הָאַשְׁכְּנַזִּים מְבָרְכוֹת, וּלְמִנְהַג הַסְּפָרַדִּים אֵינָן מְבָרְכוֹת.
תְּקִיעוֹת בִּתְפִלַּת הַלַּחַשׁ שֶׁל מוּסָף: הַסְּפָרַדִּים וְהַחֲסִידִים תּוֹקְעִים גַּם בַּלַּחַשׁ, וְהַמִּתְפַּלְּלִים מִתְפַּלְּלִים בַּקֶּצֶב שֶׁל הַתּוֹקֵעַ. הָאַשְׁכְּנַזִּים תּוֹקְעִים רַק בַּחֲזָרַת הַשַּׁ״ץ, וְאֶת הַתְּקִיעוֹת הַחֲסֵרוֹת מַשְׁלִימִים אַחֲרֵי הַתְּפִלָּה.
💛 אַהֲבַת הַשֵּׁם: כָּל הַמִּנְהָגִים נוֹסְדוּ עַל יְדֵי גְּדוֹלֵי יִשְׂרָאֵל. כָּל אֶחָד שׁוֹמֵר בְּאַהֲבָה עַל מִנְהַג אֲבוֹתָיו, וְכֻלָּנוּ שׁוֹמְעִים אֶת אוֹתוֹ קוֹל שׁוֹפָר.
📖 פֶּרֶק ד׳, הֲלָכוֹת ג׳–ה׳; פֶּרֶק ג׳, הֲלָכָה ט׳; פֶּרֶק ד׳, הֲלָכָה ו׳

🎯מִצְווֹת צְרִיכוֹת כַּוָּנָה

מִי שֶׁמִּתְאַמֵּן לִתְקֹעַ וְיָצְאוּ לוֹ תְּקִיעוֹת כְּשֵׁרוֹת – לֹא יָצָא. מִי שֶׁשָּׁמַע מִבֵּיתוֹ תְּקִיעוֹת מִבֵּית הַכְּנֶסֶת וְלֹא כִּוֵּן לְקַיֵּם אֶת הַמִּצְוָה – לֹא יָצָא. הַתּוֹקֵעַ מְכַוֵּן לְהוֹצִיא אֶת כָּל הַשּׁוֹמְעִים, גַּם מִי שֶׁעוֹמֵד בַּחוּץ.
בְּדִיעֲבַד מַסְפִּיקָה ״כַּוָּנָה רְדוּמָה״: אִם יִשְׁאֲלוּ אֶת הַשּׁוֹמֵעַ לָמָּה בָּא, יַעֲנֶה – כְּדֵי לְקַיֵּם אֶת הַמִּצְוָה.
💛 אַהֲבַת הַשֵּׁם: הַמִּצְוָה הִיא לֹא רַק קוֹל בָּאָזְנַיִם – הִיא פְּגִישָׁה שֶׁל הַלֵּב עִם ה׳. לָכֵן חָשׁוּב לָדַעַת לָמָּה בָּאנוּ.
📖 פֶּרֶק ד׳, הֲלָכָה ה׳

🧒יְלָדִים וְשׁוֹפָר

יֶלֶד שֶׁהִגִּיעַ לְגִיל חִנּוּךְ – כְּשֶׁהוּא מֵבִין שֶׁמִּצְוָה לִשְׁמֹעַ תשר״ת, תש״ת וְתר״ת – מִצְוָה לְחַנְּכוֹ לִשְׁמֹעַ שׁוֹפָר. גַּם יֶלֶד קָטָן יוֹתֵר, שֶׁמַּצְלִיחַ לַעֲמֹד בְּשֶׁקֶט – טוֹב לַהֲבִיאוֹ, שֶׁיִּסְפֹּג זִכְרוֹנוֹת שֶׁל קְדֻשָּׁה.
יֶלֶד שֶׁאֵינוֹ מַצְלִיחַ לִשְׁתֹּק – אָסוּר לַהֲבִיאוֹ לְבֵית הַכְּנֶסֶת בִּזְמַן הַתְּקִיעוֹת, כְּדֵי שֶׁלֹּא יַפְרִיעַ. וְאַחֲרֵי הַתְּקִיעוֹת מְעוֹדְדִים יְלָדִים לִתְקֹעַ כָּל הַיּוֹם וּלְהִתְאַמֵּן – רַק לֹא כְּשֶׁאֲנָשִׁים יְשֵׁנִים.
💛 אַהֲבַת הַשֵּׁם: הַיֶּלֶד שֶׁמִּתְאַמֵּן הַיּוֹם בַּשּׁוֹפָר הוּא אוּלַי הַתּוֹקֵעַ שֶׁל בֵּית הַכְּנֶסֶת בְּעוֹד כַּמָּה שָׁנִים!

תּוֹקְעִים בַּיּוֹם: ״יוֹם תְּרוּעָה״. הַזְּמַן מַתְחִיל בְּהָנֵץ הַחַמָּה, וּבְדִיעֲבַד – מֵעֲלוֹת הַשַּׁחַר. נוֹהֲגִים לְהַחֲזִיק אֶת הַשּׁוֹפָר כְּשֶׁהַצַּד הָרָחָב פּוֹנֶה לְמַעְלָה, כְּמוֹ שֶׁנֶּאֱמַר: ״עָלָה אֱלֹהִים בִּתְרוּעָה״.
כְּשֶׁרֹאשׁ הַשָּׁנָה חָל בְּשַׁבָּת – לֹא תּוֹקְעִים בַּשּׁוֹפָר. חֲכָמִים חָשְׁשׁוּ שֶׁמָּא מִישֶׁהוּ יִשָּׂא אֶת הַשּׁוֹפָר בִּרְשׁוּת הָרַבִּים, כְּדֵי לִלְמֹד מִמִּי שֶׁבָּקִי לִתְקֹעַ.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת – הַשַּׁבָּת עַצְמָהּ מַשְׁלִימָה אֶת מָה שֶׁהַשּׁוֹפָר נוֹתֵן. ״זִכְרוֹן תְּרוּעָה״ – זוֹכְרִים אֶת הַתְּרוּעָה גַּם בְּלִי לִתְקֹעַ.

🎺נָשִׁים וּבָנוֹת וְהַשּׁוֹפָר

נָשִׁים פְּטוּרוֹת מִמִּצְוַת הַשּׁוֹפָר, כִּי הִיא מִצְוַת עֲשֵׂה שֶׁהַזְּמַן גְּרָמָהּ. אֲבָל מִי שֶׁשּׁוֹמַעַת – מְקַיֶּמֶת מִצְוָה וּמְקַבֶּלֶת שָׂכָר, וְרֹב נְשׁוֹת יִשְׂרָאֵל נוֹהֲגוֹת לִשְׁמֹעַ שׁוֹפָר.
הַבְּרָכָה: לְמִנְהַג הַסְּפָרַדִּים – אִשָּׁה שֶׁתּוֹקַעַת לְעַצְמָהּ אֵינָהּ מְבָרֶכֶת, וְגַם גֶּבֶר שֶׁתּוֹקֵעַ לְנָשִׁים אֵינוֹ מְבָרֵךְ. לְמִנְהַג הָאַשְׁכְּנַזִּים – אִשָּׁה מְבָרֶכֶת, וּכְשֶׁגֶּבֶר תּוֹקֵעַ לְנָשִׁים, אַחַת מֵהֶן מְבָרֶכֶת בִּשְׁבִיל כֻּלָּן.
💛 אַהֲבַת הַשֵּׁם: ה׳ שָׂמֵחַ בְּכָל מִי שֶׁבָּאָה לְקַיֵּם מִצְוָה מֵאַהֲבָה, גַּם כְּשֶׁאֵינָהּ חַיֶּבֶת בָּהּ.

🤫שֶׁקֶט גָּמוּר

צָרִיךְ לִשְׁמֹעַ כָּל תְּקִיעָה מִתְּחִלָּתָהּ וְעַד סוֹפָהּ. לָכֵן כָּל הַקָּהָל שׁוֹמֵר עַל שֶׁקֶט גָּמוּר, וּמִי שֶׁצָּרִיךְ לְהִשְׁתַּעֵל – מִתְאַפֵּק עַד סוֹף הַתְּקִיעוֹת.
הַבְּרָכָה שֶׁלִּפְנֵי הַתְּקִיעוֹת חָלָה גַּם עַל הַתְּקִיעוֹת שֶׁבְּמוּסָף, וְלָכֵן לֹא מְדַבְּרִים עַד שֶׁמִּסְתַּיְּמוֹת הַתְּקִיעוֹת שֶׁל מוּסָף.
💛 אַהֲבַת הַשֵּׁם: הַשֶּׁקֶט הוּא כָּבוֹד לַמֶּלֶךְ. כְּשֶׁכֻּלָּם שׁוֹתְקִים יַחַד, שׁוֹמְעִים הֵיטֵב אֶת קוֹל הַשּׁוֹפָר – וְאֶת קוֹל הַלֵּב.

🍰קִדּוּשׁ לִפְנֵי הַתְּקִיעוֹת

יֵשׁ נוֹהֲגִים, בְּעִקָּר בַּיְּשִׁיבוֹת שֶׁמַּאֲרִיכִים בִּתְפִלָּה, לְקַדֵּשׁ וְלִטְעֹם מַשֶּׁהוּ אַחֲרֵי שַׁחֲרִית, כְּדֵי לְהִתְחַזֵּק לְמוּסָף. אָסוּר לִקְבֹּעַ סְעוּדָה לִפְנֵי מִצְוָה, אֲבָל טְעִימָה קְטַנָּה – מֻתֶּרֶת.
לְכַתְּחִלָּה אוֹכְלִים כְּבֵיצָה מְזוֹנוֹת וְלֹא יוֹתֵר, כְּדֵי שֶׁלֹּא יֵחָשֵׁב סְעוּדָה. פֵּרוֹת וּמַאֲכָלִים קַלִּים – אֶפְשָׁר מְעַט יוֹתֵר, אֲבָל לֹא הַרְבֵּה, כְּדֵי לֹא לְהִתְעַיֵּף.
💛 אַהֲבַת הַשֵּׁם: הַגּוּף שֶׁלָּנוּ הוּא כְּלִי לַעֲבוֹדַת ה׳. מְחַזְּקִים אוֹתוֹ בְּמִדָּה – כְּדֵי שֶׁנּוּכַל לְהִתְפַּלֵּל בְּכַוָּנָה וּבְשִׂמְחָה.
📖 פֶּרֶק ד׳, הֲלָכָה ח׳

🕯️רֹאשׁ הַשָּׁנָה בְּשַׁבָּת

מִן הַתּוֹרָה תּוֹקְעִים גַּם בְּשַׁבָּת, אֲבָל חֲכָמִים אָסְרוּ: כֻּלָּם חַיָּבִים בַּשּׁוֹפָר, וְלֹא כֻּלָּם בְּקִיאִים בְּאִסּוּר טִלְטוּל, וְיֵשׁ חֲשָׁשׁ שֶׁמִּישֶׁהוּ יִשָּׂא אֶת שׁוֹפָרוֹ לְמִי שֶׁיְּלַמֵּד אוֹתוֹ, וְיַעֲבֹר עַל אִסּוּר שַׁבָּת.
בְּבֵית הַמִּקְדָּשׁ תָּקְעוּ גַּם בְּשַׁבָּת. אַחֲרֵי הַחֻרְבָּן תִּקֵּן רַבָּן יוֹחָנָן בֶּן זַכַּאי לִתְקֹעַ בִּמְקוֹם בֵּית הַדִּין הַגָּדוֹל. הַיּוֹם אֵין בֵּית דִּין כָּזֶה, וְלָכֵן לֹא תּוֹקְעִים בְּשׁוּם מָקוֹם.
💛 אַהֲבַת הַשֵּׁם: ״זִכְרוֹן תְּרוּעָה״ – בְּשַׁבָּת זוֹכְרִים אֶת הַתְּרוּעָה, וּקְדֻשַּׁת הַשַּׁבָּת עַצְמָהּ פּוֹעֶלֶת אֶת מָה שֶׁהַשּׁוֹפָר פּוֹעֵל.

⏱️אֹרֶךְ הַקּוֹלוֹת

תְּרוּעָה – תִּשְׁעָה קוֹלוֹת קְצָרִים (״טְרוּמִיטִין״), וּבְדִיעֲבַד שְׁלֹשָׁה. כָּל שֶׁבֶר – בְּאֹרֶךְ שְׁלֹשָׁה טְרוּמִיטִין. הַתְּקִיעָה צְרִיכָה לִהְיוֹת לְפָחוֹת כְּאֹרֶךְ הַתְּרוּעָה שֶׁל אוֹתוֹ סֵדֶר. מִי שֶׁרוֹצֶה לָצֵאת לְכָל הַדֵּעוֹת – מַאֲרִיךְ בַּתְּקִיעוֹת שֶׁל תשר״ת חָמֵשׁ שְׁנִיּוֹת, וּבְתש״ת וְתר״ת – שְׁתֵּי שְׁנִיּוֹת וָחֵצִי.
בֵּין הַתְּקִיעָה לַשְּׁבָרִים אוֹ לַתְּרוּעָה מַפְסִיקִים כְּדֵי נְשִׁימָה – כִּי הַתְּקִיעָה הִיא שִׂמְחָה, וְהַתְּרוּעָה הִיא בֶּכִי, וְלֹא מְחַבְּרִים בֵּינֵיהֶם.
💛 אַהֲבַת הַשֵּׁם: גַּם בְּרֶגַע שֶׁל בֶּכִי, יֵשׁ גְּבוּל וְסֵדֶר. ה׳ מְלַמֵּד אוֹתָנוּ לְהִצְטַעֵר בְּמִדָּה – וְלַחֲזֹר לְשִׂמְחָה.
📖 פֶּרֶק ד׳, הֲלָכוֹת י״א–י״ג

🔁טָעוּת בַּתְּקִיעוֹת

״יְחִידָה״ הִיא תשר״ת אֶחָד, תש״ת אֶחָד אוֹ תר״ת אֶחָד. תּוֹקֵעַ שֶׁהִתְבַּלְבֵּל וְתָקַע בְּכַוָּנָה קוֹל זָר בְּאֶמְצַע הַיְּחִידָה – חוֹזֵר לַתְּקִיעָה הָרִאשׁוֹנָה שֶׁל אוֹתָהּ יְחִידָה. אֲבָל לְעוֹלָם אֵינוֹ חוֹזֵר לִתְחִלַּת כָּל הַסֵּדֶר.
מִי שֶׁנִּסָּה לִתְקֹעַ כָּרָאוּי, וְיָצָא לוֹ בְּלִי כַּוָּנָה קוֹל לֹא נָכוֹן – נוֹשֵׁם וְתוֹקֵעַ שׁוּב רַק אֶת הַקּוֹל שֶׁהָיָה צָרִיךְ, וְאֵינוֹ חוֹזֵר לִתְחִלַּת הַיְּחִידָה.
💛 אַהֲבַת הַשֵּׁם: טָעוּת אֵינָהּ סוֹף הָעוֹלָם. חוֹזְרִים בְּדִיּוּק לַמָּקוֹם הַנָּכוֹן וּמַמְשִׁיכִים – כָּךְ גַּם בִּתְשׁוּבָה.

🐏אֵיזֶה שׁוֹפָר?

שׁוֹפָר הוּא קֶרֶן חֲלוּלָה שֶׁל בְּהֵמָה אוֹ חַיָּה. מִצְוָה שֶׁיִּהְיֶה כָּפוּף, וּמִצְוָה מִן הַמֻּבְחָר – שׁוֹפָר כָּפוּף שֶׁל אַיִל. קֶרֶן שֶׁל פָּרָה פְּסוּלָה, כִּי הִיא נִקְרֵאת ״קֶרֶן״ וְלֹא ״שׁוֹפָר״.
שׁוֹפָר שֶׁהִדְבִּיקוּ לוֹ חֲתִיכָה – אֲפִלּוּ קְטַנָּה – פָּסוּל. אֲבָל מֻתָּר לַחְקֹק עָלָיו צוּרוֹת, וְאִם קִצְּרוּ אוֹתוֹ וְנִשְׁאַר טֶפַח – כָּשֵׁר.
💛 אַהֲבַת הַשֵּׁם: ה׳ אָמַר: ״תִּקְעוּ לְפָנַי בְּשׁוֹפָר שֶׁל אַיִל, כְּדֵי שֶׁאֶזְכֹּר לָכֶם עֲקֵדַת יִצְחָק, וּמַעֲלֶה אֲנִי עֲלֵיכֶם כְּאִלּוּ עֲקַדְתֶּם עַצְמְכֶם לְפָנַי״.
📖 פֶּרֶק ד׳, הֲלָכוֹת ט״ו–ט״ז

🐏זֵכֶר לַעֲקֵדָה

רַבִּי אַבָּהוּ אָמַר: לָמָּה תּוֹקְעִים בְּשׁוֹפָר שֶׁל אַיִל? אָמַר ה׳: ״תִּקְעוּ לְפָנַי בְּשׁוֹפָר שֶׁל אַיִל, כְּדֵי שֶׁאֶזְכֹּר לָכֶם עֲקֵדַת יִצְחָק בֶּן אַבְרָהָם״. וּמִצְוָה שֶׁהַשּׁוֹפָר יִהְיֶה כָּפוּף – סִימָן שֶׁאֲנַחְנוּ כּוֹפְפִים אֶת לִבֵּנוּ לִפְנֵי ה׳.
אִם יֵשׁ לִבְחֹר בֵּין שׁוֹפָר כָּפוּף שֶׁאֵינוֹ שֶׁל אַיִל, לְבֵין שׁוֹפָר יָשָׁר שֶׁל אַיִל – עָדִיף הַכָּפוּף, כִּי הוּא דִּין בַּמִּשְׁנָה.
💛 אַהֲבַת הַשֵּׁם: ״מַעֲלֶה אֲנִי עֲלֵיכֶם כְּאִלּוּ עֲקַדְתֶּם עַצְמְכֶם לְפָנַי״. כְּשֶׁאֲנַחְנוּ מְוַתְּרִים בִּשְׁבִיל ה׳, אֲנַחְנוּ מַמְשִׁיכִים אֶת דַּרְכָּם שֶׁל אַבְרָהָם וְיִצְחָק.
📖 פֶּרֶק ד׳, הֲלָכָה ט״ו

🧠 בְּדֹק אֶת עַצְמְךָ

לָמָּה תּוֹקְעִים הַיּוֹם שְׁלֹשִׁים תְּקִיעוֹת, וְלֹא רַק תֵּשַׁע כְּמוֹ מִן הַתּוֹרָה?
💡 הָיוּ קְהִלּוֹת שֶׁהֵרִיעוּ כַּאֲנָחוֹת, הָיוּ שֶׁהֵרִיעוּ כְּבֶכִי, וְהָיוּ שֶׁעָשׂוּ אֶת שְׁנֵיהֶם. רַבִּי אַבָּהוּ תִּקֵּן לִתְקֹעַ אֶת כָּל הַסּוּגִים – וְכָךְ הִגַּעְנוּ לִשְׁלֹשִׁים.
מָה סֵדֶר שְׁלֹשִׁים הַתְּקִיעוֹת שֶׁלִּפְנֵי מוּסָף?
💡 מַתְחִילִים בְּתשר״ת – הַתְּרוּעָה שֶׁיֵּשׁ בָּהּ אֲנָחָה וּבֶכִי, מַמְשִׁיכִים בְּתש״ת וּמְסַיְּמִים בְּתר״ת. בְּדִיעֲבַד, מִי שֶׁשִּׁנָּה אֶת הַסֵּדֶר – יָצָא.
הַתּוֹקֵעַ הִתְבַּלְבֵּל, וְתָקַע תְּרוּעָה בֵּין הַתְּקִיעָה לַשְּׁבָרִים. מָה עוֹשֶׂה?
💡 מִי שֶׁתָּקַע בְּכַוָּנָה קוֹל זָר בְּאֶמְצַע הַיְּחִידָה – פָּסַל אוֹתָהּ, וְחוֹזֵר לַתְּקִיעָה הָרִאשׁוֹנָה שֶׁלָּהּ. כָּל יְחִידָה עוֹמֶדֶת בִּפְנֵי עַצְמָהּ, וְלָכֵן לְעוֹלָם לֹא חוֹזְרִים לִתְחִלַּת הַסֵּדֶר.

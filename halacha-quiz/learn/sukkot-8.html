הַקְהֵל

🌿 הִלְכוֹת סֻכּוֹת · פֶּרֶק ח׳

🎺מִצְוַת הַקְהֵל

פַּעַם בְּשֶׁבַע שָׁנִים, בְּחַג הַסֻּכּוֹת שֶׁאַחֲרֵי שְׁנַת הַשְּׁמִטָּה, הָיוּ מַקְהִילִים אֶת כָּל יִשְׂרָאֵל בַּמִּקְדָּשׁ – אֲנָשִׁים, נָשִׁים וְטַף – וְהַמֶּלֶךְ קָרָא לִפְנֵיהֶם בַּתּוֹרָה. גַּם חֲכָמִים גְּדוֹלִים שֶׁיָּדְעוּ אֶת כָּל הַתּוֹרָה הָיוּ חַיָּבִים לָבוֹא.
הַמַּעֲמָד הִתְקַיֵּם בְּמוֹצָאֵי הַיּוֹם הָרִאשׁוֹן שֶׁל הַחַג. מִי שֶׁפָּטוּר מֵעֲלִיָּה לָרֶגֶל פָּטוּר גַּם מֵהַקְהֵל – חוּץ מִן הַנָּשִׁים וְהַטַּף, שֶׁחַיָּבִים. וּכְשֶׁאֵין מֶלֶךְ – אֵין הַמִּצְוָה מִתְבַּטֶּלֶת, וְקוֹרֵא הַגָּדוֹל שֶׁבְּיִשְׂרָאֵל, כְּמוֹ נָשִׂיא, כֹּהֵן גָּדוֹל אוֹ גְּדוֹל הַדּוֹר.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁכָּל הָעָם, מִן הַמֶּלֶךְ וְעַד הַיֶּלֶד הַקָּטָן, מִתְאַסֵּף לִשְׁמֹעַ תּוֹרָה – כֻּלָּם מְבִינִים שֶׁהַתּוֹרָה הִיא עִקַּר חַיֵּינוּ וְתִפְאַרְתֵּנוּ.
📖 פֶּרֶק ח׳, הֲלָכוֹת א׳, ג׳

🎺כְּמוֹ בְּהַר סִינַי

מַעֲמַד הַקְהֵל הָיָה כְּמוֹ מַעֲמַד הַר סִינַי: כָּל יִשְׂרָאֵל יַחַד, וְכָל אֶחָד מַרְגִּישׁ כְּאִלּוּ עַכְשָׁיו הוּא מְקַבֵּל אֶת הַתּוֹרָה מִפִּי ה׳. הַמֶּלֶךְ הוּא רַק שָׁלִיחַ, לְהַשְׁמִיעַ אֶת דִּבְרֵי ה׳.
גַּם מִי שֶׁעָמַד רָחוֹק וְלֹא שָׁמַע הֵיטֵב – הָיָה צָרִיךְ לְכַוֵּן אֶת לִבּוֹ לַקְּרִיאָה. וְהַטַּף שֶׁבָּאוּ קִבְּלוּ חֵשֶׁק גָּדוֹל לִלְמֹד תּוֹרָה וּלְקַיֵּם מִצְווֹת.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁיֶּלֶד רוֹאֶה שֶׁכָּל הָעָם, וְגַם הַמֶּלֶךְ, מִתְאַסְּפִים בִּשְׁבִיל הַתּוֹרָה – הוּא מֵבִין שֶׁאֵין דָּבָר יָקָר מִמֶּנָּה.
📖 פֶּרֶק ח׳, הֲלָכוֹת א׳, ג׳

📖מָה קָרָא הַמֶּלֶךְ

הַמֶּלֶךְ קָרָא בִּלְשׁוֹן הַקֹּדֶשׁ מִתְּחִלַּת סֵפֶר דְּבָרִים וְעַד סוֹף ״שְׁמַע״, וְאַחַר כָּךְ ״וְהָיָה אִם שָׁמֹעַ״, פָּרָשִׁיּוֹת הַמַּעַשְׂרוֹת, פָּרָשַׁת הַמֶּלֶךְ, וְהַבְּרָכוֹת וְהַקְּלָלוֹת. גַּם מִי שֶׁלֹּא הֵבִין לְשׁוֹן הַקֹּדֶשׁ הָיָה חַיָּב לִשְׁמֹעַ.
הַכֹּהֲנִים תָּקְעוּ בַּחֲצוֹצְרוֹת בְּכָל יְרוּשָׁלַיִם. בָּנוּ בִּימָה גְּדוֹלָה שֶׁל עֵץ, וְסֵפֶר הַתּוֹרָה עָבַר מִיָּד לְיָד – מֵחַזַּן הַכְּנֶסֶת וְעַד הַכֹּהֵן הַגָּדוֹל – עַד הַמֶּלֶךְ, שֶׁקִּבֵּל אוֹתוֹ בַּעֲמִידָה.
💛 אַהֲבַת הַשֵּׁם: הַמֶּלֶךְ, הַנִּכְבָּד שֶׁבַּכֹּל, עוֹמֵד לִפְנֵי הַתּוֹרָה. כָּךְ כֻּלָּם לוֹמְדִים: אֵין דָּבָר חָשׁוּב מִן הַתּוֹרָה.
📖 פֶּרֶק ח׳, הֲלָכוֹת ב׳, ד׳

🕍זֵכֶר לְהַקְהֵל

הַיּוֹם אֵין מִצְוָה לְקַיֵּם הַקְהֵל, כִּי הִיא קְשׁוּרָה לָעֲלִיָּה לָרֶגֶל לְבֵית הַמִּקְדָּשׁ. אֲבָל גְּדוֹלֵי הָרַבָּנִים עוֹרְרוּ לַעֲשׂוֹת ״זֵכֶר לְהַקְהֵל״, וּמִשְּׁנַת תש״ו – בְּכָל מוֹצָאֵי שְׁמִטָּה, בַּיּוֹם הָרִאשׁוֹן שֶׁל חֹל הַמּוֹעֵד – מִתְקַיֵּם בַּכֹּתֶל הַמַּעֲרָבִי מַעֲמָד גָּדוֹל, בְּהַשְׁתַּתְּפוּת הָרַבָּנִים הָרָאשִׁיִּים.
״צִיּוֹן הִיא דֹּרֵשׁ אֵין לָהּ״ – מִכָּאן לָמְדוּ שֶׁצָּרִיךְ לִדְרֹשׁ אֶת צִיּוֹן וּלְהַזְכִּיר אוֹתָהּ. וְגַם שִׂמְחַת תּוֹרָה, שֶׁבָּהּ כָּל הַקָּהָל וְהַטַּף מִתְאַסְּפִים לִכְבוֹד הַתּוֹרָה, דּוֹמָה מְעַט לְהַקְהֵל.
💛 אַהֲבַת הַשֵּׁם: בְּכָל פַּעַם שֶׁאָנוּ זוֹכְרִים אֶת הַמִּקְדָּשׁ, אָנוּ מְקָרְבִים אֶת בִּנְיָנוֹ. עַם יִשְׂרָאֵל חוֹזֵר לְאַרְצוֹ, וּמִתְאַסֵּף שׁוּב לִשְׁמֹעַ תּוֹרָה.
📖 פֶּרֶק ח׳, הֲלָכָה ה׳; פֶּרֶק ז׳, הֲלָכָה ח׳

🧠 בְּדֹק אֶת עַצְמְךָ

מִי הָיָה חַיָּב לָבוֹא לְמַעֲמַד הַקְהֵל?
💡 ״הַקְהֵל אֶת הָעָם, הָאֲנָשִׁים וְהַנָּשִׁים וְהַטַּף״. גַּם חֲכָמִים שֶׁיָּדְעוּ אֶת כָּל הַתּוֹרָה, וְגַם מִי שֶׁלֹּא הֵבִין לְשׁוֹן הַקֹּדֶשׁ – הָיוּ חַיָּבִים, כִּי זֶה כְּמוֹ מַעֲמַד הַר סִינַי.
בִּימֵי הַמִּקְדָּשׁ, אִם אֵין מֶלֶךְ בְּיִשְׂרָאֵל – הַאִם מִצְוַת הַקְהֵל מִתְבַּטֶּלֶת?
💡 חֲכָמִים תִּקְּנוּ שֶׁהַמֶּלֶךְ יִקְרָא, כְּדֵי לְכַבֵּד אֶת הַמַּעֲמָד. אֲבָל כְּשֶׁאֵין מֶלֶךְ – אֵין הַמִּצְוָה מִתְבַּטֶּלֶת, וְקוֹרֵא הַגָּדוֹל שֶׁבְּיִשְׂרָאֵל: נָשִׂיא, כֹּהֵן גָּדוֹל אוֹ גְּדוֹל הַדּוֹר.
מִי שֶׁפָּטוּר מֵעֲלִיָּה לָרֶגֶל – פָּטוּר גַּם מֵהַקְהֵל. מָה דִּין הַנָּשִׁים?
💡 ״הַקְהֵל אֶת הָעָם, הָאֲנָשִׁים וְהַנָּשִׁים וְהַטַּף״. כָּל הַפָּטוּר מֵעֲלִיָּה לָרֶגֶל – כְּמוֹ הַזָּקֵן וְהַחוֹלֶה שֶׁאֵינָם יְכוֹלִים לַעֲלוֹת – פָּטוּר גַּם מֵהַקְהֵל, חוּץ מִן הַנָּשִׁים וְהַטַּף: הֵם חַיָּבִים בְּהַקְהֵל, אַף שֶׁנָּשִׁים פְּטוּרוֹת מֵעֲלִיָּה לָרֶגֶל.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

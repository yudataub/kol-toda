נְטִילַת יָדַיִם שַׁחֲרִית

🙏 הִלְכוֹת תְּפִלָּה · פֶּרֶק ח׳

💧נְטִילַת יָדַיִם שֶׁל שַׁחֲרִית

נוֹטְלִים כָּל יָד שָׁלֹשׁ פְּעָמִים לְסֵרוּגִין: יָמִין, שְׂמֹאל, יָמִין, שְׂמֹאל, יָמִין, שְׂמֹאל.
לִפְנֵי הַנְּטִילָה לֹא נוֹגְעִים בַּפֶּה, בָּאַף, בָּעֵינַיִם וּבָאָזְנַיִם – וְגַם לֹא בְּאֹכֶל.
💛 אַהֲבַת הַשֵּׁם: כָּל בֹּקֶר אֲנַחְנוּ כְּמוֹ בְּרִיאָה חֲדָשָׁה. בַּנְּטִילָה אֲנַחְנוּ מְקַדְּשִׁים אֶת הַיָּדַיִם לַעֲבוֹדַת הַשֵּׁם בַּיּוֹם הֶחָדָשׁ.
📖 פֶּרֶק ח׳, הֲלָכוֹת א׳–ב׳

🔀עוֹד מִנְהָגִים שֶׁל הַבֹּקֶר

אַרְבַּע אַמּוֹת: לְפִי הַזֹּהַר וְהַמְּקֻבָּלִים – וְכָךְ פָּסְקוּ הַחִידָ״א, הַמִּשְׁנָה בְּרוּרָה וְהַבֶּן אִישׁ חַי – לֹא הוֹלְכִים יוֹתֵר מֵאַרְבַּע אַמּוֹת (כִּשְׁנֵי מֶטֶר) לִפְנֵי הַנְּטִילָה, וּמְכִינִים מַיִם לְיַד הַמִּטָּה. יֵשׁ פּוֹסְקִים שֶׁמְּקִלִּים, כִּי כָּל הַבַּיִת נֶחְשָׁב כְּאַרְבַּע אַמּוֹת, אוֹ כִּי הָרוּחַ הָרָעָה נֶחְלְשָׁה בְּיָמֵינוּ – וְכָךְ הַמִּנְהָג הָרוֹוֵחַ.
מִי שֶׁהָיָה עֵר כָּל הַלַּיְלָה, כְּמוֹ בְּלֵיל שָׁבוּעוֹת: לְמִנְהַג הָאַשְׁכְּנַזִּים הוֹלֵךְ לַשֵּׁרוּתִים וְנוֹטֵל יָדַיִם בִּבְרָכָה, וּלְמִנְהַג הַסְּפָרַדִּים נוֹטֵל בְּלִי בְּרָכָה. וְהַבָּנוֹת מְבָרְכוֹת בְּבִרְכוֹת הַשַּׁחַר ״שֶׁעָשַׂנִי כִּרְצוֹנוֹ״.
💛 אַהֲבַת הַשֵּׁם: כָּל מִנְהָג נוֹסַד עַל יְדֵי גְּדוֹלֵי תּוֹרָה. וּלְכָל הַדֵּעוֹת נוֹטְלִים כָּל יָד שָׁלֹשׁ פְּעָמִים, וְלֹא נוֹגְעִים לִפְנֵי הַנְּטִילָה בַּפֶּה, בָּאַף, בָּעֵינַיִם וּבָאָזְנַיִם.
📖 פֶּרֶק ח׳, הֲלָכוֹת א׳, ג׳; פֶּרֶק ט׳, הֲלָכוֹת א׳, ו׳

🫗📜 לְשׁוֹן חֲכָמִים: נַטְלָה

״לוֹקְחִים תְּחִלָּה אֶת הַנַּטְלָה עִם הַמַּיִם בְּיַד יָמִין״.
📖 נַטְלָה – מִלְּשׁוֹן ״לִטֹּל״: כְּלִי עִם שְׁתֵּי יָדִיּוֹת, שֶׁבּוֹ נוֹטְלִים יָדַיִם.
💛 אַהֲבַת הַשֵּׁם: ״בְּכָל בֹּקֶר נַעֲשֶׂה הָאָדָם כְּמִי שֶׁנִּבְרָא מֵחָדָשׁ״ – ה׳ נוֹתֵן לְךָ הַתְחָלָה חֲדָשָׁה. 🧒 וְאַתָּה? הָכֵן נַטְלָה וּמַיִם לְיַד הַמִּטָּה, כְּדֵי לִטֹּל מִיָּד כְּשֶׁאַתָּה קָם.
📖 פֶּרֶק ח׳, הֲלָכוֹת א׳–ג׳

🔁📜 לְשׁוֹן חֲכָמִים: לְסֵרוּגִין

״נוֹטְלִים כָּל יָד שָׁלֹשׁ פְּעָמִים לְסֵרוּגִין״.
📖 לְסֵרוּגִין – בְּתוֹרוֹת, פַּעַם זוֹ וּפַעַם זוֹ: יָמִין, שְׂמֹאל, יָמִין, שְׂמֹאל, יָמִין, שְׂמֹאל.
💛 אַהֲבַת הַשֵּׁם: מַתְחִילִים בְּיַד יָמִין, ״הָרוֹמֶזֶת לְמִדַּת הַחֶסֶד״ – הַבֹּקֶר שֶׁלְּךָ מַתְחִיל בְּחֶסֶד. 🧒 וְאַתָּה? כְּשֶׁאַתָּה נוֹטֵל – תַּגִּיד בַּלֵּב: תּוֹדָה עַל יוֹם חָדָשׁ.
📖 פֶּרֶק ח׳, הֲלָכָה ב׳

📏🔑 מוּשָּׂג: אַרְבַּע אַמּוֹת

״צָרִיךְ לְהִזָּהֵר שֶׁלֹּא לָלֶכֶת יוֹתֵר מֵאַרְבַּע אַמּוֹת לִפְנֵי נְטִילַת הַיָּדַיִם שַׁחֲרִית, וְלָכֵן צָרִיךְ לְהָכִין מַיִם לִפְנֵי הַשֵּׁנָה, וּלְהַנִּיחָם לְיַד הַמִּטָּה״. וְיֵשׁ מְקִלִּים, ״כִּי כָּל הַבַּיִת נֶחְשָׁב כְּאַרְבַּע אַמּוֹת״.
📖 אַמָּה – מִדַּת אֹרֶךְ, מֵהַמַּרְפֵּק עַד קְצֵה הָאֶצְבַּע. אַרְבַּע אַמּוֹת – כִּשְׁנֵי מֶטֶר.
💛 אַהֲבַת הַשֵּׁם: כְּבָר בְּרֶגַע הַקִּימָה, לִפְנֵי שֶׁעוֹשִׂים כָּל דָּבָר אַחֵר, מִתְכּוֹנְנִים לַעֲבוֹדַת ה׳. 🧒 וְאַתָּה? הָכֵן בָּעֶרֶב נַטְלָה וּקְעָרָה לְיַד הַמִּטָּה.
📖 פֶּרֶק ח׳, הֲלָכָה ג׳

🙌מָתַי מְבָרְכִים "עַל נְטִילַת יָדַיִם"?

בְּכָל מִצְוָה מְבָרְכִים לִפְנֵי הַמַּעֲשֶׂה. אֲבָל כָּאן הַיָּדַיִם עוֹד לֹא נְקִיּוֹת, וְלָכֵן:
מְבָרְכִים מִיָּד אַחֲרֵי הַנְּטִילָה – לִפְנֵי שֶׁמְּנַגְּבִים אֶת הַיָּדַיִם.
💛 אַהֲבַת הַשֵּׁם: אֶת הַבְּרָכָה לַה׳ אוֹמְרִים בְּיָדַיִם נְקִיּוֹת – כִּי לַמֶּלֶךְ הַגָּדוֹל בָּאִים בְּכָבוֹד.
📖 פֶּרֶק ח׳, הֲלָכָה ד׳

🧒גַּם הַקְּטַנִּים נוֹטְלִים

מִצְוָה לְהַרְגִּיל אֶת הַיְּלָדִים וְהַיְּלָדוֹת לִטֹּל יָדַיִם שָׁלֹשׁ פְּעָמִים אַחֲרֵי הַשֵּׁנָה, מִגִּיל חִנּוּךְ – כְּשֶׁהֵם כְּבָר מְבִינִים אֵיךְ נוֹטְלִים. וּמִגִּיל מִצְווֹת זוֹ חוֹבָה: לְבָנִים מִגִּיל שְׁלֹשׁ עֶשְׂרֵה, וּלְבָנוֹת מִגִּיל שְׁתֵּים עֶשְׂרֵה.
מִי שֶׁנָּגַע בְּפֵרוֹת לִפְנֵי הַנְּטִילָה – שׁוֹטֵף אוֹתָם שָׁלֹשׁ פְּעָמִים בְּמַיִם. וּמַאֲכָל שֶׁאִי אֶפְשָׁר לִשְׁטֹף – לְפִי רֹב הַפּוֹסְקִים מֻתָּר בְּדִיעֲבַד לְאָכְלוֹ, וְלֹא זוֹרְקִים אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: יֶלֶד שֶׁנּוֹטֵל יָדַיִם בַּבֹּקֶר מַתְחִיל אֶת הַיּוֹם בִּקְדֻשָּׁה, וּמִתְרַגֵּל מִקָּטְנוֹ לְהִתְכּוֹנֵן לַעֲבוֹדַת ה׳.
📖 פֶּרֶק ח׳, הֲלָכוֹת ז׳–ח׳

🧠 בְּדֹק אֶת עַצְמְךָ

תָּמָר בַּת שֵׁשׁ, וּכְבָר מְבִינָה אֵיךְ נוֹטְלִים יָדַיִם. מָה עָלֶיהָ לַעֲשׂוֹת כְּשֶׁהִיא קָמָה בַּבֹּקֶר?
💡 מִצְוָה לְהַרְגִּיל אֶת הַיְּלָדִים וְהַיְּלָדוֹת מִגִּיל חִנּוּךְ לִטֹּל יָדַיִם שָׁלֹשׁ פְּעָמִים אַחֲרֵי הַשֵּׁנָה. מִגִּיל מִצְווֹת – שְׁתֵּים עֶשְׂרֵה לְבַת וּשְׁלֹשׁ עֶשְׂרֵה לְבֵן – זוֹ כְּבָר חוֹבָה.
יְהוֹנָתָן קָם בַּבֹּקֶר וְלָקַח תַּפּוּחַ מֵהַקְּעָרָה לִפְנֵי שֶׁנָּטַל יָדַיִם. מָה יַעֲשֶׂה עִם הַתַּפּוּחַ?
💡 פֵּרוֹת שֶׁנָּגְעוּ בָּהֶם לִפְנֵי הַנְּטִילָה שׁוֹטְפִים שָׁלֹשׁ פְּעָמִים – כְּמוֹ שֶׁנּוֹטְלִים כָּל יָד שָׁלֹשׁ פְּעָמִים. וּמַאֲכָל שֶׁאִי אֶפְשָׁר לִשְׁטֹף – לְפִי רֹב הַפּוֹסְקִים מֻתָּר בְּדִיעֲבַד, וְלֹא זוֹרְקִים אוֹתוֹ.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

הַזְכָּרַת גְּשָׁמִים וּבַקָּשָׁתָם

🌸 תְּפִלַּת בָּנוֹת · פֶּרֶק י״ג

🛠️טָעֲתָה? יֵשׁ תִּקּוּן

בַּת שֶׁהוֹסִיפָה תְּפִלָּה שֶׁאֵינָהּ חַיֶּבֶת בָּהּ – לְמָשָׁל מִנְחָה אוֹ עַרְבִית – וְטָעֲתָה בָּהּ, חוֹזֶרֶת וּמִתְפַּלֶּלֶת כַּהֲלָכָה. כִּי מִשֶּׁהִתְחִילָה, עָלֶיהָ לְהִתְפַּלֵּל כְּפִי שֶׁתִּקְּנוּ חֲכָמִים. מִסְתַּפֶּקֶת אִם הִתְפַּלְּלָה הַיּוֹם אֶת הַתְּפִלָּה שֶׁהִיא רְגִילָה בָּהּ? כָּל עוֹד לֹא עָבַר הַזְּמַן – מִתְפַּלֶּלֶת שׁוּב, וּמַתְנָה בְּלִבָּהּ: אִם כְּבָר הִתְפַּלַּלְתִּי – הֲרֵי זוֹ תְּפִלַּת נְדָבָה.
מִי שֶׁרְגִילָה בְּשַׁחֲרִית וּבְמִנְחָה וְשָׁכְחָה שַׁחֲרִית – מִתְפַּלֶּלֶת מִנְחָה פַּעֲמַיִם: הָרִאשׁוֹנָה מִנְחָה, וְהַשְּׁנִיָּה תַּשְׁלוּמִין. תַּשְׁלוּמִין יֵשׁ רַק לְמִי שֶׁשָּׁכְחָה אוֹ נֶאֶנְסָה, וְלֹא לְמִי שֶׁוִּתְּרָה בְּכַוָּנָה. וּמִי שֶׁמִּתְפַּלֶּלֶת רַק תְּפִלָּה אַחַת בַּיּוֹם וְעָבַר זְמַן שַׁחֲרִית – מִתְפַּלֶּלֶת מִנְחָה.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁטּוֹעִים, אֲנַחְנוּ לֹא מְוַתְּרִים עַל הַשִּׂיחָה עִם ה׳ – מְתַקְּנִים, וּמַמְשִׁיכִים בְּאַהֲבָה.
📖 פֶּרֶק י״ג, הֲלָכוֹת א׳, ה׳–ו׳

🔀הַפְסָקָה אֲרֻכָּה – שְׁנֵי מִנְהָגִים

מִי שֶׁהִפְסִיקָה בְּאֶמְצַע הָעֲמִידָה, בְּדִבּוּר אוֹ בִּשְׁתִיקָה, הַפְסָקָה אֲרֻכָּה – כַּזְּמַן שֶׁהָיְתָה יְכוֹלָה לְסַיֵּם בּוֹ אֶת כָּל הַתְּפִלָּה: לְפִי הַשֻּׁלְחָן עָרוּךְ, וְכָךְ מִנְהַג הַסְּפָרַדִּים – בְּכָל מִקְרֶה חוֹזֶרֶת לְהַתְחָלַת הַתְּפִלָּה. לְפִי הָרַמָּ״א, וְכָךְ מִנְהַג הָאַשְׁכְּנַזִּים – רַק אִם הִפְסִיקָה מֵחֲמַת אֹנֶס חוֹזֶרֶת לַהַתְחָלָה, וְאִם הִפְסִיקָה בִּרְצוֹנָהּ – מַמְשִׁיכָה מִן הַמָּקוֹם שֶׁפָּסְקָה.
וּבְהַפְסָקָה קְצָרָה – לְכָל הַמִּנְהָגִים מַמְשִׁיכָה מִן הַמָּקוֹם שֶׁפָּסְקָה.
💛 אַהֲבַת הַשֵּׁם: שְׁנֵי הַמִּנְהָגִים כַּהֲלָכָה, וְכָל אַחַת נוֹהֶגֶת כְּמִנְהַג אֲבוֹתֶיהָ.
📖 פֶּרֶק י״ג, הֲלָכָה ב׳

🌧️טָעֲתָה בְּ״מַשִּׁיב הָרוּחַ״?

בַּחֹרֶף שָׁכְחָה לוֹמַר ״מַשִּׁיב הָרוּחַ וּמוֹרִיד הַגֶּשֶׁם״, אֲבָל אָמְרָה ״מוֹרִיד הַטָּל״ – אֵינָהּ חוֹזֶרֶת, כִּי שִׁבְּחָה אֶת ה׳ עַל הַמַּיִם. וְאִם לֹא אָמְרָה גַּם טַל – חוֹזֶרֶת. וּבַקַּיִץ אָמְרָה בְּטָעוּת ״מוֹרִיד הַגֶּשֶׁם״ – חוֹזֶרֶת, כִּי בַּקַּיִץ אֵין זֶה שֶׁבַח.
שָׁכְחָה בַּחֹרֶף לְבַקֵּשׁ גֶּשֶׁם בְּבִרְכַּת הַשָּׁנִים, וְנִזְכְּרָה לִפְנֵי ״שׁוֹמֵעַ תְּפִלָּה״ – מְבַקֶּשֶׁת אֶת הַגֶּשֶׁם בְּ״שׁוֹמֵעַ תְּפִלָּה״, וּבָזֶה תִּקְּנָה. וְאִם כְּבָר עָבְרָה אֶת ״שׁוֹמֵעַ תְּפִלָּה״ – חוֹזֶרֶת לְבִרְכַּת הַשָּׁנִים, וּמִשָּׁם מַמְשִׁיכָה כַּסֵּדֶר.
💛 אַהֲבַת הַשֵּׁם: בְּ״שׁוֹמֵעַ תְּפִלָּה״ אֶפְשָׁר לְבַקֵּשׁ כָּל בַּקָּשָׁה. ה׳ שׁוֹמֵעַ אוֹתָנוּ גַּם כְּשֶׁאֲנַחְנוּ מְתַקְּנִים בְּאֶמְצַע הַדֶּרֶךְ.
📖 פֶּרֶק י״ג, הֲלָכָה ח׳

🔀תִּשְׁעִים פַּעַם – שְׁנֵי נֻסְחָאוֹת

הַטָּעוּת הַשְּׁכִיחָה בְּיוֹתֵר הִיא בְּהַזְכָּרַת גְּשָׁמִים וּבְבַקָּשָׁתָם, כִּי הַלָּשׁוֹן רְגִילָה לַנֹּסַח הַקּוֹדֵם. הָעֵצָה: בְּיוֹם הַהַחְלָפָה חוֹזְרִים עַל הַנֹּסַח הֶחָדָשׁ תִּשְׁעִים פַּעַם, וְאָז גַּם אִם מִסְתַּפְּקִים אִם אָמְרוּ נָכוֹן – אֵין צֹרֶךְ לַחֲזֹר. בְּלֵיל ז׳ בְּמַרְחֶשְׁוָן: לְמִנְהַג הַסְּפָרַדִּים אוֹמְרִים ״רוֹפֵא חוֹלֵי עַמּוֹ יִשְׂרָאֵל. בָּרֵךְ עָלֵינוּ״, וּלְמִנְהַג הָאַשְׁכְּנַזִּים ״וְאֶת כָּל מִינֵי תְבוּאָתָהּ לְטוֹבָה, וְתֵן טַל וּמָטָר לִבְרָכָה״.
בְּמוּסַף שֶׁל יוֹם רִאשׁוֹן שֶׁל פֶּסַח אוֹמְרִים תִּשְׁעִים פַּעַם: ״מְחַיֵּה מֵתִים אַתָּה רַב לְהוֹשִׁיעַ, מוֹרִיד הַטָּל״.
💛 אַהֲבַת הַשֵּׁם: שְׁנֵי הַנֻּסָּחִים כַּהֲלָכָה, וְכָל אַחַת נוֹהֶגֶת כְּמִנְהַג אֲבוֹתֶיהָ. וּבִשְׁנֵיהֶם אֲנַחְנוּ מְבַקְּשִׁים מֵה׳ גֶּשֶׁם שֶׁל בְּרָכָה לְאַרְצֵנוּ.
📖 פֶּרֶק י״ג, הֲלָכָה ט׳

🧠 בְּדֹק אֶת עַצְמְךָ

בְּאֶמְצַע ״רְפָאֵנוּ״ נִזְכְּרָה שִׁירָה שֶׁדִּלְּגָה עַל בִּרְכַּת ״סְלַח לָנוּ״. מָה תַּעֲשֶׂה?
💡 מִי שֶׁדִּלְּגָה עַל אַחַת הַבְּרָכוֹת הָאֶמְצָעִיּוֹת – חוֹזֶרֶת אֶל הַבְּרָכָה שֶׁדִּלְּגָה, וּמִמֶּנָּה מַמְשִׁיכָה כַּסֵּדֶר, גַּם אִם הִיא חוֹזֶרֶת עַל ״גְּאָלֵנוּ״ שֶׁכְּבָר אָמְרָה. וְאִם כְּבָר סִיְּמָה אֶת הַתְּפִלָּה – חוֹזֶרֶת וּמִתְפַּלֶּלֶת.
חֹדֶשׁ כִּסְלֵו. בַּבְּרָכָה הַשְּׁנִיָּה שֶׁל הָעֲמִידָה אָמְרָה דִּינָה ״מוֹרִיד הַטָּל״, וְשָׁכְחָה ״מַשִּׁיב הָרוּחַ וּמוֹרִיד הַגֶּשֶׁם״. מָה הַדִּין?
💡 מִי שֶׁשָּׁכְחָה בַּחֹרֶף ״מַשִּׁיב הָרוּחַ וּמוֹרִיד הַגֶּשֶׁם״ אֲבָל אָמְרָה ״מוֹרִיד הַטָּל״ – אֵינָהּ חוֹזֶרֶת, כִּי אָמְרָה שֶׁבַח שֶׁקָּשׁוּר לַמַּיִם. וְאִם לֹא אָמְרָה גַּם אֶת הַטַּל – חוֹזֶרֶת.

חַלָּה

🍽️ הִלְכוֹת כַּשְׁרוּת · פֶּרֶק י״א

🍞מַפְרִישִׁים חַלָּה

"רֵאשִׁית עֲרִסֹתֵכֶם חַלָּה תָּרִימוּ תְרוּמָה" – מִבָּצֵק שֶׁל חֲמֵשֶׁת מִינֵי דָּגָן (חִטָּה, שְׂעוֹרָה, כֻּסֶּמֶת, שִׁבֹּלֶת שׁוּעָל וְשִׁיפוֹן) מַפְרִישִׁים חַלָּה. מַפְרִישִׁים אַחֲרֵי הַלִּישָׁה, כְּשֶׁכְּבָר יֵשׁ בָּצֵק – וְלֹא מִן הַקֶּמַח.
הַיּוֹם מַפְרִישִׁים חֲתִיכָה קְטַנָּה, וְשׂוֹרְפִים אוֹתָהּ אוֹ עוֹטְפִים וּמַנִּיחִים בַּפַּח. בָּצֵק מִקֶּמַח אֹרֶז אוֹ תִּירָס – פָּטוּר מֵחַלָּה. וְנֹסַח הַבְּרָכָה שׁוֹנֶה בֵּין הָעֵדוֹת: "לְהַפְרִישׁ חַלָּה", "לְהַפְרִישׁ חַלָּה תְּרוּמָה" אוֹ "לְהַפְרִישׁ חַלָּה מִן הָעִסָּה".
💛 אַהֲבַת הַשֵּׁם: בִּזְמַן הַמִּקְדָּשׁ נִתְּנָה הַחַלָּה לַכֹּהֵן – "לְהָנִיחַ בְּרָכָה אֶל בֵּיתֶךָ". גַּם הַיּוֹם, כְּשֶׁמַּפְרִישִׁים חַלָּה, הַלֶּחֶם שֶׁלָּנוּ מִתְקַשֵּׁר לַקְּדֻשָּׁה וּבְרָכָה נִמְשֶׁכֶת לַבַּיִת.
📖 פֶּרֶק י״א, הֲלָכוֹת א׳–ב׳, ד׳–ה׳, ט׳

🍞חַלָּה – מִצְוָה שֶׁל נָשִׁים וּבָנוֹת

מִצְוַת הַפְרָשַׁת חַלָּה שַׁיֶּכֶת לִגְבָרִים וּלְנָשִׁים כְּאֶחָד. אֲבָל אִם אִישׁ וְאִשָּׁה לָשׁוּ יַחַד אֶת הַבָּצֵק – זְכוּתָהּ שֶׁל הָאִשָּׁה קוֹדֶמֶת, כִּי הִיא אַחְרָאִית יוֹתֵר עַל עִנְיְנֵי הַבַּיִת. וַחֲכָמִים אָמְרוּ שֶׁהַמִּצְוָה הַזֹּאת שַׁיֶּכֶת בִּמְיֻחָד לָאִשָּׁה.
מֵהַזְּמַן שֶׁהַבָּנוֹת וְהַבָּנִים מַגִּיעִים לְמִצְווֹת – הֵם רַשָּׁאִים לְהַפְרִישׁ חַלָּה בְּעַצְמָם. וְיֵשׁ נָשִׁים שֶׁנּוֹהֲגוֹת לְהַדֵּר וְלֶאֱפוֹת בְּעֶרֶב שַׁבָּת בָּצֵק גָּדוֹל, שֶׁחַיָּב בְּחַלָּה.
💛 אַהֲבַת הַשֵּׁם: פַּעַם הָיְתָה כָּל אִשָּׁה נוֹתֶנֶת בְּכָל יוֹם אֶת הַחַלָּה לִשְׁכֵנָתָהּ הַכֹּהֶנֶת, וְכָךְ הַלֶּחֶם שֶׁל כָּל בַּיִת הִתְקַשֵּׁר לַקְּדֻשָּׁה. גַּם הַיּוֹם, הַפְרָשַׁת חַלָּה מַמְשִׁיכָה בְּרָכָה לַבַּיִת.
📖 פֶּרֶק י״א, הֲלָכוֹת א׳–ב׳, ד׳–ה׳

🍞🔑 מוּשָּׂג: חַלָּה

״בְּבֹאֲכֶם אֶל הָאָרֶץ... וְהָיָה בַּאֲכָלְכֶם מִלֶּחֶם הָאָרֶץ תָּרִימוּ תְרוּמָה לַה׳. רֵאשִׁית עֲרִסֹתֵכֶם חַלָּה תָּרִימוּ תְרוּמָה״.
📖 חַלָּה – חֵלֶק קָטָן מֵהַבָּצֵק שֶׁתּוֹרְמִים לַכֹּהֵן, אַחֲרֵי שֶׁלָּשִׁים אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: כְּבָר בַּבָּצֵק שֶׁבַּקְּעָרָה זוֹכְרִים שֶׁה׳ נָתַן לָנוּ אֶת הַלֶּחֶם, וְתוֹרְמִים מִמֶּנּוּ רֵאשִׁית. 🧒 וְאַתָּה? כְּשֶׁאִמָּא אוֹפָה חַלּוֹת לְשַׁבָּת – שְׁאַל אוֹתָהּ אִם אַתָּה יָכוֹל לִרְאוֹת אֵיךְ מַפְרִישִׁים חַלָּה.
📖 פֶּרֶק י״א, הֲלָכָה א׳

🔀נֻסְחָאוֹת שׁוֹנִים לַבְּרָכָה – וְכֻלָּם כַּהֲלָכָה

לִפְנֵי הַפְרָשַׁת חַלָּה מְבָרְכִים, וּבְכָל עֵדָה סִיּוּם הַבְּרָכָה מְעַט שׁוֹנֶה: "לְהַפְרִישׁ חַלָּה" – יוֹצְאֵי תֵּימָן וְחֵלֶק מִיּוֹצְאֵי אַשְׁכְּנַז; "לְהַפְרִישׁ חַלָּה תְּרוּמָה" – עֲדוֹת הַמִּזְרָח; "לְהַפְרִישׁ חַלָּה מִן הָעִסָּה" – רַבִּים מִיּוֹצְאֵי אַשְׁכְּנַז וּצְפוֹן אַפְרִיקָה.
בְּכָל הַנֻּסְחָאוֹת הַבְּרָכָה פּוֹתַחַת כָּךְ: "בָּרוּךְ אַתָּה ה׳ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, אֲשֶׁר קִדְּשָׁנוּ בְּמִצְוֹתָיו וְצִוָּנוּ".
💛 אַהֲבַת הַשֵּׁם: הַנֻּסְחָאוֹת שׁוֹנִים, אֲבָל הַמִּצְוָה אַחַת: לְקַשֵּׁר אֶת הַלֶּחֶם שֶׁלָּנוּ לַקְּדֻשָּׁה, וּלְהַמְשִׁיךְ בְּרָכָה לַבַּיִת.
📖 פֶּרֶק י״א, הֲלָכוֹת ד׳–ה׳

🍞חַלָּה – מָתַי וּמֵאֵיזֶה בָּצֵק

כָּל זְמַן שֶׁלֹּא הִפְרִישׁוּ חַלָּה – הַלֶּחֶם אוֹ הָעוּגוֹת אֲסוּרִים בַּאֲכִילָה. הַיּוֹם אֶת הַחַלָּה שׂוֹרְפִים אוֹ טוֹמְנִים, וּמַפְרִישִׁים כַּמּוּת כָּלְשֶׁהִי. וְיֵשׁ נָשִׁים שֶׁנּוֹהֲגוֹת לְהַדֵּר וְלֶאֱפוֹת בְּעֶרֶב שַׁבָּת כְּשִׁעוּר חַלָּה.
בָּצֵק לְסֻפְגָּנִיּוֹת אוֹ לִקְרֶפְּלַךְ, שֶׁמְּבַשְּׁלִים אוֹ מְטַגְּנִים בְּשֶׁמֶן עָמֹק – נוֹהֲגִים לְהַפְרִישׁ מִמֶּנּוּ חַלָּה בְּלִי בְּרָכָה, כְּדֵי לָצֵאת יְדֵי כָּל הַדֵּעוֹת. וּמִי שֶׁאוֹפֶה מִמֶּנּוּ עוּגִיָּה קְטַנָּה – מַפְרִישׁ בִּבְרָכָה. וּפַנְקֵייק, בְּלִינְצֶ׳ס וּוָפֶל בֶּלְגִי – חַיָּבִים בְּחַלָּה כְּשֶׁיֵּשׁ בָּהֶם שִׁעוּר.
💛 אַהֲבַת הַשֵּׁם: בְּהַפְרָשַׁת הַחַלָּה הַלֶּחֶם שֶׁלָּנוּ נַעֲשֶׂה קָשׁוּר לַקֹּדֶשׁ, וּבְרָכָה נִמְשֶׁכֶת לַבַּיִת.
📖 פֶּרֶק י״א, הֲלָכוֹת ד׳, י״א–י״ב

🥐מֵאֵילּוּ מַאֲפִים מַפְרִישִׁים חַלָּה?

חַלָּה מַפְרִישִׁים מִבָּצֵק שֶׁל לֶחֶם – וּבִכְלַל זֶה כָּל מַאֲפֵה דָּגָן: עוּגוֹת, עוּגִיּוֹת, קְרֶקֶרִים, בֵּייגֶלֶךְ וּבַקְלָאוָה. וְגַם מַלַּוַּאח וּמוּפְלֶטָה שֶׁנֶּאֱפִים בְּמַחֲבַת, וְכֻבָּאנֶה וְגַ׳חְנוּן שֶׁנֶּאֱפִים בְּסִיר. אֲבָל תַּבְשִׁילֵי דָּגָן, כְּמוֹ פְּתִיתִים וְאִטְרִיּוֹת – אֵינָם לֶחֶם, וּפְטוּרִים מֵחַלָּה.
בְּצֵקִים שֶׁמַּקְפִּידִים שֶׁלֹּא יִתְעָרְבוּ – אֵינָם מִצְטָרְפִים לְשִׁעוּר חַלָּה. לָכֵן גַּנֶּנֶת שֶׁאוֹפָה עִם הַיְּלָדִים חַלּוֹת לְשַׁבָּת, וְכָל יֶלֶד לָשׁ לְעַצְמוֹ אֶת הַבָּצֵק שֶׁלּוֹ וּמַדְבִּיקִים עָלָיו אֶת שְׁמוֹ – הַחַלּוֹת שֶׁלָּהֶם אֵינָן מִצְטָרְפוֹת.
💛 אַהֲבַת הַשֵּׁם: ״וְהָיָה בַּאֲכָלְכֶם מִלֶּחֶם הָאָרֶץ״ – בְּכָל מַאֲפֶה שֶׁיּוֹצֵא מִן הַתַּנּוּר יֵשׁ הִזְדַּמְּנוּת לָתֵת לַה׳ אֶת הָרֵאשִׁית.
📖 פֶּרֶק י״א, הֲלָכוֹת ח׳, י׳

🧠 בְּדֹק אֶת עַצְמְךָ

מִשְׁפָּחָה רוֹצָה לֶאֱפוֹת עִסָּה גְּדוֹלָה שֶׁחַיֶּבֶת בְּחַלָּה, אֲבָל לְחַלֵּק אוֹתָהּ לִשְׁתַּיִם כְּדֵי שֶׁלֹּא לְהַפְרִישׁ. מָה הַדִּין?
💡 מִי שֶׁאֵינוֹ רוֹצֶה לֶאֱפוֹת הַרְבֵּה – אֵינוֹ חַיָּב. אֲבָל מִי שֶׁמִּתְכַּוֵּן לֶאֱפוֹת עִסָּה שֶׁחַיֶּבֶת בְּחַלָּה – אָסוּר לוֹ לְחַלֵּק אוֹתָהּ כְּדֵי לְהִפָּטֵר, כִּי בַּמִּצְוָה אָנוּ מִתְקַשְּׁרִים לָרַעְיוֹן הַגָּדוֹל שֶׁלָּהּ.
אִמָּא וְאַבָּא לָשׁוּ יַחַד בָּצֵק גָּדוֹל לְחַלּוֹת שַׁבָּת. מִי זוֹכֶה לְהַפְרִישׁ אֶת הַחַלָּה?
💡 הַמִּצְוָה שַׁיֶּכֶת לִגְבָרִים וּלְנָשִׁים, אֲבָל כְּשֶׁלָּשׁוּ יַחַד – זְכוּת הָאִשָּׁה קוֹדֶמֶת, כִּי הִיא אַחְרָאִית יוֹתֵר עַל עִנְיְנֵי הַבַּיִת. בַּת וּבֵן שֶׁהִגִּיעוּ לְמִצְווֹת רַשָּׁאִים לְהַפְרִישׁ בְּעַצְמָם.
סָבְתָא מְכִינָה גַ׳חְנוּן בְּסִיר, מִכַּמּוּת גְּדוֹלָה שֶׁל קֶמַח. מָה הַדִּין?
💡 מַאֲפֶה הוּא מָה שֶׁנֶּאֱפֶה בְּלִי לִהְיוֹת בְּתוֹךְ נוֹזְלִים – גַּם בְּמַחֲבַת אוֹ בְּסִיר עִם מְעַט שֶׁמֶן. לָכֵן מַפְרִישִׁים חַלָּה בִּבְרָכָה מִמַּלַּוַּאח וּמוּפְלֶטָה שֶׁנֶּאֱפִים בְּמַחֲבַת, וּמִכֻּבָּאנֶה וְגַ׳חְנוּן שֶׁנֶּאֱפִים בְּסִיר – כְּשֶׁיֵּשׁ בָּהֶם שִׁעוּר.

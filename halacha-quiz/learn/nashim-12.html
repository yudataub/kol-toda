תְּפִלַּת הָעֲמִידָה

🌸 תְּפִלַּת בָּנוֹת · פֶּרֶק י״ב

🔀שָׁלֹשׁ פְּסִיעוֹת לִפְנֵי הָעֲמִידָה – שְׁתֵּי דֵּעוֹת

כְּדֵי לְהַרְאוֹת שֶׁאֲנַחְנוּ רוֹצִים לְהִתְקָרֵב לַה׳, נוֹהֲגִים לִפְסֹעַ שָׁלֹשׁ פְּסִיעוֹת קָדִימָה לִפְנֵי הָעֲמִידָה, וּמְסַיְּמִים אוֹתָן לִפְנֵי ״ה׳ שְׂפָתַי תִּפְתָּח״. מִי שֶׁכְּבָר הָלְכָה לִמְקוֹם הַתְּפִלָּה – לְדַעַת הָאֵלִיָּה רַבָּה אֵינָהּ צְרִיכָה לִפְסֹעַ שׁוּב. וְיֵשׁ אוֹמְרִים (בֶּן אִישׁ חַי, כַּף הַחַיִּים): אִם עָבַר זְמַן מֵאָז שֶׁהִגִּיעָה – רָאוּי שֶׁתַּחֲזֹר וְתִפְסַע.
לֹא פּוֹסְעִים שָׁלֹשׁ לְאָחוֹר וּמִיָּד שָׁלֹשׁ קָדִימָה. מִי שֶׁרוֹצָה לְהַדֵּר – פּוֹסַעַת לְאָחוֹר זְמַן מָה לִפְנֵי הַתְּפִלָּה, מַמְתִּינָה מְעַט, וְאָז פּוֹסַעַת שָׁלֹשׁ פְּסִיעוֹת קָדִימָה.
💛 אַהֲבַת הַשֵּׁם: שְׁתֵּי הַדֵּעוֹת מְלַמְּדוֹת אוֹתוֹ דָּבָר: הַתְּפִלָּה הִיא הִתְקָרְבוּת. כָּל פְּסִיעָה קְטַנָּה הִיא כְּמוֹ לוֹמַר לַה׳: ״אֲנִי בָּאָה אֵלֶיךָ״.
📖 פֶּרֶק י״ב, הֲלָכָה א׳

🧭פָּנִים לִירוּשָׁלַיִם

בַּעֲמִידָה פּוֹנִים לְכִוּוּן יְרוּשָׁלַיִם, וְהַלֵּב – אֶל מְקוֹם הַמִּקְדָּשׁ וּבֵית קֹדֶשׁ הַקֳּדָשִׁים. בְּבֵית הַכְּנֶסֶת מַעֲמִידִים אֶת אֲרוֹן הַקֹּדֶשׁ בַּצַּד שֶׁפּוֹנֶה לִירוּשָׁלַיִם. אֲבָל הָעִקָּר הוּא יְרוּשָׁלַיִם, וְלָכֵן מִי שֶׁעוֹמֶדֶת בְּעֶזְרַת הַנָּשִׁים פּוֹנָה לִירוּשָׁלַיִם, וְלֹא דַּוְקָא לָאָרוֹן.
בִּרְחָבַת הַכֹּתֶל, בְּעֶזְרַת הַנָּשִׁים, פּוֹנִים בַּאֲלַכְסוֹן שְׂמֹאלָה – לְכִוּוּן מְקוֹם הַמִּקְדָּשׁ. מִי שֶׁאֵינָהּ יוֹדַעַת אֵיפֹה יְרוּשָׁלַיִם – מִתְפַּלֶּלֶת לְכָל כִּוּוּן וּמְכַוֶּנֶת אֶת לִבָּהּ לְאָבִיהָ שֶׁבַּשָּׁמַיִם.
💛 אַהֲבַת הַשֵּׁם: בְּכָל הָעוֹלָם, כָּל יְהוּדִי וִיהוּדִיָּה פּוֹנִים בַּתְּפִלָּה אֶל אוֹתוֹ מָקוֹם, שֶׁבּוֹ בָּחַר ה׳ לְהַשְׁרוֹת אֶת שְׁכִינָתוֹ. כָּל עַם יִשְׂרָאֵל – לֵב אֶחָד הַפּוֹנֶה אֶל אָבִינוּ.
📖 פֶּרֶק י״ב, הֲלָכָה ב׳

🤫📜 לְשׁוֹן חֲכָמִים: רַק שְׂפָתֶיהָ נָּעוֹת

״וְחַנָּה הִיא מְדַבֶּרֶת עַל לִבָּהּ, רַק שְׂפָתֶיהָ נָּעוֹת וְקוֹלָהּ לֹא יִשָּׁמֵעַ״ – ״מִכָּאן לַמִּתְפַּלֵּל שֶׁיַּחְתֹּךְ בִּשְׂפָתָיו... מִכָּאן שֶׁאָסוּר לְהַגְבִּיהַּ קוֹלוֹ בִּתְפִלָּתוֹ״.
📖 נָעוֹת – זָזוֹת. הַשְּׂפָתַיִם אוֹמְרוֹת אֶת הַמִּלִּים בְּלַחַשׁ, וְאִישׁ אֵינוֹ שׁוֹמֵעַ. לֹא בְּקוֹל רָם, וְגַם לֹא רַק בַּמַּחֲשָׁבָה.
💛 אַהֲבַת הַשֵּׁם: הַתְּפִלָּה בָּאָה ״לְבַטֵּא לִפְנֵי ה׳ אֶת הַשְּׁאִיפוֹת הָעֲמֻקּוֹת שֶׁל הַנְּשָׁמָה״ – כְּמוֹ סוֹד אָהוּב בֵּינֵךְ לְבֵין ה׳. 🧒 וְאַתְּ? בַּעֲמִידָה – לַחֲשִׁי כָּל מִלָּה בִּשְׂפָתַיִךְ, שֶׁרַק אַתְּ תִּשְׁמְעִי.
📖 פֶּרֶק י״ב, הֲלָכָה ו׳

⏱️אַחֲרֵי שָׁלֹשׁ הַפְּסִיעוֹת

בְּסוֹף הָעֲמִידָה פּוֹסְעִים שָׁלֹשׁ פְּסִיעוֹת לְאָחוֹר – נִפְרָדִים מִן הַמֶּלֶךְ. מִי שֶׁחוֹזֶרֶת מִיָּד לַמָּקוֹם מַרְאָה שֶׁלֹּא הֵבִינָה שֶׁעָמְדָה לִפְנֵי הַמֶּלֶךְ וְנִפְרְדָה מִמֶּנּוּ. וְאֵין שׁוּם טַעַם לְהִתְרוֹמֵם כְּמוֹ בַּקְּדֻשָּׁה כְּשֶׁחוֹזְרִים.
הַמִּנְהָג הַנָּכוֹן: מַמְתִּינִים בַּמָּקוֹם כַּחֲצִי דַּקָּה עַד דַּקָּה, וְאַחַר כָּךְ חוֹזְרִים. בִּשְׁעַת הַצֹּרֶךְ, כְּמוֹ כְּשֶׁעוֹמְדִים בְּמַעֲבָר – כַּמָּה שְׁנִיּוֹת. בְּמִנְיָן מַמְתִּינִים עַד שֶׁהַחַזָּן מַגִּיעַ לַקְּדֻשָּׁה, וּלְפָחוֹת עַד שֶׁהוּא מַתְחִיל אֶת הַחֲזָרָה.
💛 אַהֲבַת הַשֵּׁם: הַפְּרִידָה מִן הַמֶּלֶךְ חֲשׁוּבָה כְּמוֹ הַכְּנִיסָה אֵלָיו. כְּשֶׁאֲנַחְנוּ לֹא מְמַהֲרִים, אֲנַחְנוּ מַרְאִים כַּמָּה הַשִּׂיחָה עִם ה׳ יְקָרָה לָנוּ.
📖 פֶּרֶק י״ב, הֲלָכָה י״ב

🚌תְּפִלָּה בַּנְּסִיעָה

בַּת שֶׁנּוֹסַעַת בְּאוֹטוֹבּוּס אוֹ בְּרַכֶּבֶת, וְהִגִּיעַ זְמַן הַתְּפִלָּה – אִם הִיא יְכוֹלָה לַעֲמֹד וּלְכַוֵּן, טוֹב שֶׁתַּעֲמֹד. וְאִם הָעֲמִידָה מַפְרִיעָה לָהּ לְכַוֵּן – בִּגְלַל הַטַּלְטֵלוֹת, אוֹ כִּי הִיא מִתְבַּיֶּשֶׁת – יוֹשֶׁבֶת וּמִתְפַּלֶּלֶת.
גַּם בִּישִׁיבָה מַצְמִידָה אֶת הָרַגְלַיִם, מִשְׁתַּדֶּלֶת לִפְנוֹת לִירוּשָׁלַיִם, וּבִמְקוֹם הַכְּרִיעוֹת – קָמָה לְרֶגַע וְכוֹרַעַת, אִם הִיא יְכוֹלָה. וּבִמְכוֹנִית שֶׁמִּישֶׁהוּ אַחֵר נוֹהֵג בָּהּ, וְהוּא מְמַהֵר – מִתְפַּלֶּלֶת בִּישִׁיבָה, וְלֹא מְבַקֶּשֶׁת לַעֲצֹר, כְּדֵי שֶׁלֹּא תְּמַהֵר בִּתְפִלָּתָהּ.
💛 אַהֲבַת הַשֵּׁם: ה׳ נִמְצָא אִתָּנוּ גַּם בַּדֶּרֶךְ. אֶפְשָׁר לַעֲמֹד לְפָנָיו בְּכָל מָקוֹם – הָעִקָּר שֶׁהַלֵּב פּוֹנֶה אֵלָיו.
📖 פֶּרֶק י״ב, הֲלָכָה י״ג

🧠 בְּדֹק אֶת עַצְמְךָ

יָעֵל עוֹמֶדֶת בְּעֶזְרַת הַנָּשִׁים, וַאֲרוֹן הַקֹּדֶשׁ אֵינוֹ מְכֻוָּן בְּדִיּוּק לִירוּשָׁלַיִם. לְאָן הִיא פּוֹנָה בָּעֲמִידָה?
💡 הָעִקָּר בַּעֲמִידָה הוּא לִפְנוֹת לִירוּשָׁלַיִם. הָאָרוֹן עוֹמֵד בְּדֶרֶךְ כְּלָל בַּצַּד שֶׁל יְרוּשָׁלַיִם, אֲבָל כְּשֶׁהוּא לֹא מְכֻוָּן – פּוֹנִים לִירוּשָׁלַיִם. וּמִי שֶׁעוֹמֶדֶת בְּעֶזְרַת הַנָּשִׁים פּוֹנָה לִירוּשָׁלַיִם, וְלֹא דַּוְקָא לָאָרוֹן.
בִּרְחָבַת הַכֹּתֶל הַמַּעֲרָבִי, בְּעֶזְרַת הַנָּשִׁים – לְאָן פּוֹנִים בִּתְפִלַּת הָעֲמִידָה?
💡 מִי שֶׁעוֹמֶדֶת בִּירוּשָׁלַיִם פּוֹנָה אֶל מְקוֹם הַמִּקְדָּשׁ. בְּעֶזְרַת הַנָּשִׁים שֶׁבַּכֹּתֶל, מְקוֹם הַמִּקְדָּשׁ נִמְצָא בַּאֲלַכְסוֹן שְׂמֹאלָה – וְלָכֵן פּוֹנִים לְשָׁם.
אֲבִיגַיִל סִיְּמָה עֲמִידָה וּפָסְעָה שָׁלֹשׁ פְּסִיעוֹת לְאָחוֹר. הִיא רוֹצָה לַחֲזֹר לִמְקוֹמָהּ. מָה נָכוֹן?
💡 לֹא חוֹזְרִים מִיָּד – כְּאִלּוּ לֹא הֵבַנּוּ שֶׁנִּפְרַדְנוּ מִן הַמֶּלֶךְ. מַמְתִּינִים כַּחֲצִי דַּקָּה עַד דַּקָּה, וּבִשְׁעַת הַצֹּרֶךְ כַּמָּה שְׁנִיּוֹת. וּלְהִתְרוֹמֵם כְּמוֹ בַּקְּדֻשָּׁה – אֵין בָּזֶה שׁוּם טַעַם.

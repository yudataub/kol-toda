הִלְכוֹת קִדּוּשׁ

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק ו׳

🍷זוֹכְרִים אֶת הַשַּׁבָּת בַּקִּדּוּשׁ

בַּקִּדּוּשׁ אָנוּ מְקַיְּמִים אֶת מִצְוַת ״זָכוֹר״ וְזוֹכְרִים שְׁנֵי יְסוֹדוֹת: שֶׁה׳ בָּרָא אֶת הָעוֹלָם, וְשֶׁהוֹצִיא אוֹתָנוּ מִמִּצְרַיִם. אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה קָבְעוּ אֶת נֹסַח הַקִּדּוּשׁ, וְתִקְּנוּ לְאָמְרוֹ עַל כּוֹס יַיִן, סָמוּךְ לַסְּעוּדָה. לִפְנֵי הַבְּרָכָה נוֹהֲגִים לוֹמַר ״וַיְכֻלּוּ״.
גְּבָרִים וְנָשִׁים חַיָּבִים בַּקִּדּוּשׁ בְּאֹפֶן שָׁוֶה. יֶלֶד אוֹ יַלְדָּה שֶׁעוֹד לֹא הִגִּיעוּ לְמִצְווֹת חַיָּבִים רַק מִדִּבְרֵי חֲכָמִים, וְלָכֵן אֵינָם יְכוֹלִים לְהוֹצִיא גְּדוֹלִים.
💛 אַהֲבַת הַשֵּׁם: הַקִּדּוּשׁ הוּא כְּמוֹ הַכְרָזָה חֲגִיגִית: ה׳ בָּרָא אֶת הָעוֹלָם, וְהוּא בָּחַר בָּנוּ בְּאַהֲבָה. וְאוֹמְרִים אוֹתוֹ עַל יַיִן מְשַׂמֵּחַ – כִּי קְדֻשַּׁת הַשַּׁבָּת מִתְגַּלָּה גַּם בְּשִׂמְחָה.
📖 פֶּרֶק ו׳, הֲלָכוֹת א׳–ג׳

👩נָשִׁים וּבָנוֹת בַּקִּדּוּשׁ וּבַסְּעוּדוֹת

נָשִׁים חַיָּבוֹת בַּקִּדּוּשׁ בְּדִיּוּק כְּמוֹ גְּבָרִים, כִּי ״זָכוֹר״ וְ״שָׁמוֹר״ בְּדִבּוּר אֶחָד נֶאֶמְרוּ – וְלָכֵן אִשָּׁה אוֹ בַּת שֶׁהִגִּיעָה לְמִצְווֹת יְכוֹלָה לְקַדֵּשׁ וּלְהוֹצִיא גַּם גְּבָרִים. אִשָּׁה שֶׁקִּבְּלָה שַׁבָּת בְּהַדְלָקַת הַנֵּרוֹת – אָסוּר לָהּ לֶאֱכֹל וְלִשְׁתּוֹת עַד הַקִּדּוּשׁ. וְאִשָּׁה שֶׁמִּתְפַּלֶּלֶת שַׁחֲרִית – מֻתָּר לָהּ לִשְׁתּוֹת לִפְנֵי הַתְּפִלָּה, כְּמוֹ הַגְּבָרִים.
גַּם נָשִׁים וּבָנוֹת חַיָּבוֹת בְּשָׁלֹשׁ סְעוּדוֹת וּבְלֶחֶם מִשְׁנֶה – כִּי גַּם הֵן הָיוּ בְּנֵס הַמָּן. וְגַם סְעוּדַת מְלַוֶּה מַלְכָּה שַׁיֶּכֶת לִגְבָרִים וּלְנָשִׁים כְּאֶחָד. בְּדֶרֶךְ כְּלָל הֵן יוֹצְאוֹת בִּבְרָכָה שֶׁל הַמְבָרֵךְ עַל שְׁתֵּי הַחַלּוֹת.
💛 אַהֲבַת הַשֵּׁם: הַשַּׁבָּת הִיא שֶׁל כָּל הַמִּשְׁפָּחָה. כְּשֶׁכֻּלָּם – בָּנִים וּבָנוֹת, אַבָּא וְאִמָּא – זוֹכְרִים אֶת הַשַּׁבָּת וּמִתְעַנְּגִים בָּהּ יַחַד, כָּל הַבַּיִת מִתְמַלֵּא קְדֻשָּׁה וְשִׂמְחָה.
📖 פֶּרֶק ו׳, הֲלָכוֹת א׳, ט׳; פֶּרֶק ז׳, הֲלָכוֹת ב׳–ג׳, ח׳

🌞קִדּוּשׁ הַיּוֹם וּסְעוּדַת הַיּוֹם

חֲכָמִים תִּקְּנוּ גַּם קִדּוּשׁ בַּבֹּקֶר, לִכְבוֹד הַיּוֹם: אוֹמְרִים פְּסוּקִים עַל הַשַּׁבָּת, וּמְבָרְכִים ״בּוֹרֵא פְּרִי הַגָּפֶן״. מִי שֶׁלֹּא קִדֵּשׁ בַּלַּיְלָה – אוֹמֵר בַּבֹּקֶר אֶת בִּרְכַּת הַקִּדּוּשׁ שֶׁל הַלַּיְלָה, בְּלִי ״וַיְכֻלּוּ״.
קִדּוּשׁ הַלַּיְלָה חָשׁוּב יוֹתֵר, אֲבָל סְעוּדַת הַיּוֹם חֲשׁוּבָה מִסְּעוּדַת הַלַּיְלָה – וְלָכֵן מַאֲכָלִים טוֹבִים, כְּמוֹ יַיִן וּפֵרוֹת, שׁוֹמְרִים לַסְּעוּדָה שֶׁל הַיּוֹם.
💛 אַהֲבַת הַשֵּׁם: הַקִּדּוּשׁ בַּבֹּקֶר הוֹפֵךְ אֶת הָאֲרוּחָה לִסְעוּדָה חֲשׁוּבָה. כְּשֶׁשּׁוֹמְרִים אֶת הַטּוֹב בְּיוֹתֵר לְיוֹם הַשַּׁבָּת עַצְמוֹ – מַרְאִים שֶׁהַשַּׁבָּת כֻּלָּהּ יְקָרָה, מִתְּחִלָּתָהּ וְעַד סוֹפָהּ.
📖 פֶּרֶק ו׳, הֲלָכָה ב׳; פֶּרֶק ז׳, הֲלָכָה ד׳

🍇עַל מָה מְקַדְּשִׁים?

לְכַתְּחִלָּה מְקַדְּשִׁים עַל יַיִן מְשֻׁבָּח, כִּי הוּא הַמַּשְׁקֶה הֶחָשׁוּב – מְזִין וּמְשַׂמֵּחַ. אֶפְשָׁר לְקַדֵּשׁ גַּם עַל מִיץ עֲנָבִים וְעַל יַיִן לָבָן. יַיִן שֶׁעָמַד מְגֻלֶּה כַּמָּה שָׁעוֹת, אוֹ שֶׁרֵיחוֹ רַע – פָּסוּל לְקִדּוּשׁ.
מִי שֶׁאֵין לוֹ יַיִן – בַּלַּיְלָה מְקַדֵּשׁ עַל הַפַּת. בַּיּוֹם אֶפְשָׁר לְקַדֵּשׁ גַּם עַל מַשְׁקֶה חָשׁוּב שֶׁיֵּשׁ בּוֹ אַלְכּוֹהוֹל, כְּמוֹ בִּירָה.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים אָמְרוּ: ״כָּל הַמְקַדֵּשׁ עַל הַיַּיִן בְּלֵילֵי שַׁבָּתוֹת – מַאֲרִיכִים לוֹ יָמָיו״. כְּשֶׁמְּבִיאִים לַקִּדּוּשׁ אֶת הַמְשֻׁבָּח בְּיוֹתֵר, מְכַבְּדִים אֶת הַמֶּלֶךְ.
📖 פֶּרֶק ו׳, הֲלָכוֹת ג׳–ד׳

🏆כּוֹס שֶׁל בְּרָכָה

הַכּוֹס צְרִיכָה לְהַחֲזִיק לְפָחוֹת ״רְבִיעִית״, לִהְיוֹת שְׁלֵמָה וּנְקִיָּה, וּמְמַלְּאִים אוֹתָהּ. אוֹחֲזִים אוֹתָהּ בְּיַד יָמִין, מַגְבִּיהִים טֶפַח מֵעַל הַשֻּׁלְחָן וּמַבִּיטִים בַּכּוֹס, כְּדֵי לֹא לְהַסִּיחַ אֶת הַדַּעַת.
אַחֲרֵי הַבְּרָכָה שׁוֹתֶה הַמְקַדֵּשׁ ״מְלֹא לוּגְמָיו״ – כַּמּוּת שֶׁמְּמַלֵּאת לֶחִי אַחַת. יַיִן שֶׁכְּבָר שָׁתוּ מִמֶּנּוּ נִקְרָא ״פָּגוּם״, וְלֹא מְקַדְּשִׁים עָלָיו לְכַתְּחִלָּה.
💛 אַהֲבַת הַשֵּׁם: בִּכְלִי נָאֶה וּמָלֵא מַרְאִים כַּמָּה הַבְּרָכָה חֲשׁוּבָה לָנוּ. כָּל פְּרָט קָטָן בַּכּוֹס הוּא דֶּרֶךְ לְהַגִּיד לַה׳: לִכְבוֹדְךָ הֵכַנּוּ אֶת הַטּוֹב בְּיוֹתֵר.
📖 פֶּרֶק ו׳, הֲלָכוֹת ה׳–ו׳

🥂שׁוֹמְעִים, שׁוֹתְקִים וְטוֹעֲמִים

מִי שֶׁשּׁוֹמֵעַ קִדּוּשׁ יוֹצֵא יְדֵי חוֹבָה גַּם בְּלִי לִשְׁתּוֹת, וּמִצְוָה מִן הַמֻּבְחָר שֶׁכֻּלָּם יִטְעֲמוּ מִן הַיַּיִן. הַשּׁוֹמְעִים שׁוֹתְקִים עַד שֶׁהַמְקַדֵּשׁ שׁוֹתֶה ״מְלֹא לוּגְמָיו״.
לֹא מוֹזְגִים מִכּוֹס שֶׁשָּׁתוּ מִמֶּנָּה – הַיַּיִן ״פָּגוּם״. מוֹסִיפִים קֹדֶם מְעַט יַיִן מִן הַבַּקְבּוּק, וְאָז הוּא מְתֻקָּן. דֶּרֶךְ טוֹבָה אַחֶרֶת: לִמְזֹג לְכֻלָּם כּוֹסוֹת עוֹד לִפְנֵי הַקִּדּוּשׁ.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁכָּל הַמִּשְׁפָּחָה מַקְשִׁיבָה בְּשֶׁקֶט וְטוֹעֶמֶת מֵאוֹתוֹ יַיִן, כֻּלָּם מִתְחַבְּרִים לִבְרָכָה אַחַת – וְהַשַּׁבָּת נִכְנֶסֶת לַלֵּב שֶׁל כֻּלָּם.
📖 פֶּרֶק ו׳, הֲלָכָה ז׳

🥖לֶחֶם מִשְׁנֶה

בְּכָל סְעוּדַת שַׁבָּת בּוֹצְעִים עַל שְׁתֵּי כִּכָּרוֹת שְׁלֵמוֹת, זֵכֶר לַמָּן הַכָּפוּל שֶׁיָּרַד בְּיוֹם שִׁשִּׁי. הַמְבָרֵךְ אוֹחֵז אֶת שְׁתֵּיהֶן בִּשְׁעַת ״הַמּוֹצִיא״, וְהַמִּנְהָג הָרוֹוֵחַ לִבְצֹעַ אַחַת. לֹא מְסִירִים אֶת הַתָּוִית שֶׁעַל הַחַלָּה לִפְנֵי הַבְּצִיעָה, כְּדֵי שֶׁתִּשָּׁאֵר שְׁלֵמָה.
בִּשְׁעַת הַקִּדּוּשׁ מְכַסִּים אֶת הַחַלּוֹת: בְּדֶרֶךְ כְּלָל מְבָרְכִים עַל הַלֶּחֶם לִפְנֵי הַיַּיִן, וְעַכְשָׁו מַקְדִּימִים אֶת הַיַּיִן. וְעוֹד – זֵכֶר לַמָּן, שֶׁהָיָה מְכֻסֶּה בְּטַל מִלְמַעְלָה וּמִלְּמַטָּה.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת הַכֹּל כָּפוּל: ״זָכוֹר״ וְ״שָׁמוֹר״, שְׁנֵי נֵרוֹת וּשְׁתֵּי חַלּוֹת. הַלֶּחֶם הַכָּפוּל מַזְכִּיר שֶׁה׳ דּוֹאֵג לָנוּ גַּם לְשַׁבָּת – וְאֵין מָה לִדְאֹג.
📖 פֶּרֶק ו׳, הֲלָכָה ח׳; פֶּרֶק ז׳, הֲלָכָה ג׳

🔀עוֹמְדִים אוֹ יוֹשְׁבִים? שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

בְּקִדּוּשׁ לֵיל שַׁבָּת יֵשׁ שְׁנֵי מִנְהָגִים: יוֹצְאֵי סְפָרַד, וְגַם חֲסִידִים מִיּוֹצְאֵי אַשְׁכְּנַז, עוֹמְדִים – כְּמוֹ עֵדִים הַמְּעִידִים עַל בְּרִיאַת הָעוֹלָם, שֶׁעוֹמְדִים בִּשְׁעַת עֵדוּתָם. יוֹצְאֵי אַשְׁכְּנַז יוֹשְׁבִים – כְּדֵי שֶׁהַקִּדּוּשׁ יֵאָמֵר מַמָּשׁ בִּמְקוֹם הַסְּעוּדָה, וְכָל הַשּׁוֹמְעִים יִהְיוּ קְבוּעִים יַחַד עִם הַמְקַדֵּשׁ. וְיֵשׁ אַשְׁכְּנַזִּים שֶׁעוֹמְדִים רַק בְּ״וַיְכֻלּוּ״ וְאָז מִתְיַשְּׁבִים. בְּקִדּוּשׁ הַיּוֹם רֹב יִשְׂרָאֵל יוֹשְׁבִים.
אַף שֶׁגַּם אִשָּׁה יְכוֹלָה לְקַדֵּשׁ, נָהֲגוּ שֶׁהָאַבָּא מְקַדֵּשׁ בִּשְׁבִיל כָּל הַמִּשְׁפָּחָה. וּכְשֶׁכַּמָּה מִשְׁפָּחוֹת אוֹכְלוֹת יַחַד – טוֹב שֶׁאֶחָד יְקַדֵּשׁ לְכֻלָּם: ״בְּרָב עָם הַדְרַת מֶלֶךְ״.
💛 אַהֲבַת הַשֵּׁם: כָּל מִשְׁפָּחָה מְקַדֶּשֶׁת לְפִי מִנְהַג אֲבוֹתֶיהָ – וְכֻלָּם מְעִידִים יַחַד אוֹתָהּ עֵדוּת: ה׳ בָּרָא אֶת הָעוֹלָם, וְהַשַּׁבָּת הִיא הָאוֹת שֶׁלּוֹ.
📖 פֶּרֶק ו׳, הֲלָכָה ח׳

🤫לֹא טוֹעֲמִים לִפְנֵי הַקִּדּוּשׁ

מֵרֶגַע שֶׁנִּכְנְסָה הַשַּׁבָּת אָסוּר לֶאֱכֹל וְלִשְׁתּוֹת עַד הַקִּדּוּשׁ – אֲפִלּוּ מַיִם. אֲבָל מֻתָּר לִשְׁטֹף אֶת הַפֶּה אוֹ לִבְלֹעַ תְּרוּפָה. גַּם בְּשַׁבָּת בַּבֹּקֶר, אַחֲרֵי תְּפִלַּת שַׁחֲרִית, לֹא אוֹכְלִים לִפְנֵי הַקִּדּוּשׁ.
יֶלֶד קָטָן – מַרְגִּילִים אוֹתוֹ לְחַכּוֹת לַקִּדּוּשׁ, אֲבָל אִם הוּא רָעֵב אוֹ צָמֵא – מֻתָּר לָתֵת לוֹ לֶאֱכֹל וְלִשְׁתּוֹת לִפְנֵי הַקִּדּוּשׁ.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁמְּחַכִּים עִם הָאֹכֶל עַד הַקִּדּוּשׁ, מַרְאִים שֶׁהַשַּׁבָּת קוֹדֶמֶת לַבֶּטֶן. הַסְּעוּדָה מַתְחִילָה בִּקְדֻשָּׁה – וְכָל הָאֲכִילָה נַעֲשֵׂית מִצְוָה.
📖 פֶּרֶק ו׳, הֲלָכָה ט׳

📍קִדּוּשׁ בִּמְקוֹם סְעוּדָה

״וְקָרָאתָ לַשַּׁבָּת עֹנֶג״ – בַּמָּקוֹם שֶׁבּוֹ מִתְעַנְּגִים בַּסְּעוּדָה, שָׁם אוֹמְרִים אֶת הַקִּדּוּשׁ. לָכֵן צָרִיךְ לֶאֱכֹל בִּמְקוֹם הַקִּדּוּשׁ לֶחֶם, אוֹ לְפָחוֹת כְּזַיִת מְזוֹנוֹת. מִי שֶׁלֹּא אָכַל שָׁם – לֹא יָצָא, וְצָרִיךְ לְקַדֵּשׁ שׁוּב בַּמָּקוֹם שֶׁבּוֹ יֹאכַל.
מִי שֶׁשָּׁמַע קִדּוּשׁ בְּבֵית הַכְּנֶסֶת וְרַק שָׁתָה מְעַט מִיץ וְטָעַם פֵּרוּר – לֹא יָצָא, וְגַם נִמְצָא שֶׁאָכַל לִפְנֵי קִדּוּשׁ.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת הַקְּדֻשָּׁה וְהַהֲנָאָה הוֹלְכוֹת יַחַד: הַקִּדּוּשׁ וְהַסְּעוּדָה מַשְׁלִימִים זֶה אֶת זֶה. כָּךְ גַּם הָאֲכִילָה שֶׁלָּנוּ נַעֲשֵׂית חֵלֶק מִן הַמִּצְוָה.
📖 פֶּרֶק ו׳, הֲלָכָה י׳

🧠 בְּדֹק אֶת עַצְמְךָ

אַבָּא סִיֵּם אֶת בִּרְכַּת הַקִּדּוּשׁ. שְׁמוּאֵל רוֹצֶה לְסַפֵּר מָה לָמַד בַּחֵדֶר. מֵאֵיזֶה רֶגַע מֻתָּר לוֹ לְדַבֵּר?
💡 הַקִּדּוּשׁ נִגְמָר רַק כְּשֶׁהַמְקַדֵּשׁ שׁוֹתֶה ״מְלֹא לוּגְמָיו״. עַד אָז הַשּׁוֹמְעִים שׁוֹתְקִים, וְאַחַר כָּךְ אֶפְשָׁר לְסַפֵּר.
אַבָּא שָׁתָה מִכּוֹס הַקִּדּוּשׁ וְרוֹצֶה לִמְזֹג מִמֶּנָּה לַיְלָדִים. מָה יַעֲשֶׂה לִפְנֵי שֶׁהוּא מוֹזֵג?
💡 יַיִן שֶׁשָּׁתוּ מִמֶּנּוּ נִקְרָא ״פָּגוּם״. כְּשֶׁמּוֹסִיפִים לְתוֹכוֹ יַיִן מִן הַבַּקְבּוּק – הוּא נַעֲשֶׂה מְתֻקָּן, וְאֶפְשָׁר לִמְזֹג מִמֶּנּוּ לְכֻלָּם.
אַבָּא שֶׁל יוֹנָתָן שָׁמַע קִדּוּשׁ בְּבֵית הַכְּנֶסֶת, שָׁתָה מְעַט מִיץ וְטָעַם פֵּרוּר עוּגָה. מָה הַדִּין?
💡 צָרִיךְ לְקַדֵּשׁ בִּמְקוֹם סְעוּדָה – לֶאֱכֹל שָׁם לְפָחוֹת כְּזַיִת לֶחֶם אוֹ מְזוֹנוֹת. מִי שֶׁטָּעַם פָּחוֹת – לֹא יָצָא, וִיקַדֵּשׁ שׁוּב בַּבַּיִת לִפְנֵי הַסְּעוּדָה.

קְרִיאַת שְׁמַע

🙏 הִלְכוֹת תְּפִלָּה · פֶּרֶק ט״ו

👑"שְׁמַע יִשְׂרָאֵל" – מַמְלִיכִים אֶת ה׳

בַּפָּסוּק הָרִאשׁוֹן שֶׁל קְרִיאַת שְׁמַע אֲנַחְנוּ מְקַבְּלִים עָלֵינוּ עֹל מַלְכוּת שָׁמַיִם: ה׳ הוּא הַמֶּלֶךְ הַיָּחִיד שֶׁל כָּל הָעוֹלָם. לָכֵן מְכַוְּנִים בּוֹ מְאוֹד.
מְכַסִּים אֶת הָעֵינַיִם בְּיַד יָמִין, כְּדֵי לֹא לִרְאוֹת דָּבָר שֶׁיַּפְרִיעַ. וּמִי שֶׁהוֹלֵךְ בַּדֶּרֶךְ – עוֹמֵד בִּמְקוֹמוֹ בַּפָּסוּק הָרִאשׁוֹן.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁסּוֹגְרִים אֶת הָעֵינַיִם, נִשְׁאָרִים רֶגַע לְבַד עִם ה׳. זֶה רֶגַע שֶׁל קִרְבָה – אוֹמְרִים לוֹ שֶׁהוּא הַמֶּלֶךְ שֶׁלָּנוּ וְשֶׁאֲנַחְנוּ שֶׁלּוֹ.
📖 פֶּרֶק ט״ו, הֲלָכוֹת א׳, ו׳, י׳

📜שָׁלֹשׁ פָּרָשִׁיּוֹת – וְלָמָּה בַּסֵּדֶר הַזֶּה

"שְׁמַע" – קַבָּלַת עֹל מַלְכוּת שָׁמַיִם וְאַהֲבַת ה׳. "וְהָיָה אִם שָׁמֹעַ" – קַבָּלַת עֹל מִצְווֹת. "וַיֹּאמֶר" – מִצְוַת צִיצִית, שֶׁמַּזְכִּירָה אֶת כָּל הַמִּצְווֹת, וּבְסוֹפָהּ יְצִיאַת מִצְרַיִם.
קֹדֶם מְקַבְּלִים אֶת ה׳ כְּמֶלֶךְ, וְאַחַר כָּךְ אֶת מִצְווֹתָיו. וְאֶת "וַיֹּאמֶר" קוֹרְאִים גַּם בַּלַּיְלָה – כִּי זוֹכְרִים יְצִיאַת מִצְרַיִם גַּם בַּלַּיְלָה.
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁאוֹהֵב אֶת הַמֶּלֶךְ, שָׂמֵחַ לַעֲשׂוֹת אֶת רְצוֹנוֹ. לָכֵן קֹדֶם מַמְלִיכִים אֶת ה׳ בְּאַהֲבָה, וְהַמִּצְווֹת בָּאוֹת מִתּוֹךְ הָאַהֲבָה הַזֹּאת.
📖 פֶּרֶק ט״ו, הֲלָכוֹת א׳–ב׳

🔑מִלּוֹת מַפְתֵּחַ שֶׁכְּדַאי לִזְכֹּר

בָּרָמָה הָאַחֲרוֹנָה תִּבְנֶה בְּעַצְמְךָ אֶת הַתְּשׁוּבָה, אוֹת אַחַר אוֹת. מִסְפַּר הַמִּשְׁבָּצוֹת מְגַלֶּה כַּמָּה אוֹתִיּוֹת יֵשׁ בַּתְּשׁוּבָה.
כָּתוּב "וּבְשָׁכְבְּךָ וּבְקוּמֶךָ": קוֹרְאִים שְׁמַע בַּלַּיְלָה, וּבַבֹּקֶר – בְּשָׁלֹשׁ הַשָּׁעוֹת הָרִאשׁוֹנוֹת שֶׁל הַיּוֹם, הַזְּמַן שֶׁבּוֹ אֲנָשִׁים קָמִים מִשְּׁנָתָם.
💛 אַהֲבַת הַשֵּׁם: הַמִּלִּים הָרִאשׁוֹנוֹת שֶׁל הַבֹּקֶר וְהָאַחֲרוֹנוֹת שֶׁל הַלַּיְלָה שַׁיָּכוֹת לַה׳. כָּךְ כָּל הַיּוֹם שֶׁלָּנוּ עָטוּף בָּאַהֲבָה שֶׁלּוֹ.
📖 פֶּרֶק ט״ו, הֲלָכוֹת א׳–ב׳

💙מָה אוֹמְרִים בִּ״שְׁמַע יִשְׂרָאֵל״?

בְּפָרָשַׁת שְׁמַע שְׁלֹשָׁה חֲלָקִים: הָאֱמוּנָה – ״ה׳ אֶחָד״, וְאֵין כֹּחַ אַחֵר מִלְּבַדּוֹ; הָאַהֲבָה – ״וְאָהַבְתָּ אֵת ה׳ אֱלֹהֶיךָ בְּכָל לְבָבְךָ״; וְאֵיךְ לֹא שׁוֹכְחִים – קוֹרְאִים בַּבֹּקֶר וּבָעֶרֶב, מְלַמְּדִים אֶת הַיְּלָדִים, מַנִּיחִים תְּפִלִּין וְקוֹבְעִים מְזוּזָה.
״בְּכָל לְבָבְךָ״ – בַּיֵּצֶר הַטּוֹב וְגַם בַּיֵּצֶר הָרַע: גַּם אֶת הַכֹּחוֹת הַקָּשִׁים שֶׁלָּנוּ אֶפְשָׁר לְהַפְנוֹת לַעֲבוֹדַת ה׳.
💛 אַהֲבַת הַשֵּׁם: פָּרָשַׁת שְׁמַע מְלַוָּה אוֹתָנוּ תָּמִיד: בַּלֵּב – בַּקְּרִיאָה, בַּגּוּף – בַּתְּפִלִּין, וּבַבַּיִת – בַּמְּזוּזָה. כָּךְ אַהֲבַת ה׳ אִתָּנוּ בְּכָל מָקוֹם.
📖 פֶּרֶק ט״ו, הֲלָכָה ג׳

🤫"בָּרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ"

לִפְנֵי פְּטִירָתוֹ שָׁאַל יַעֲקֹב אָבִינוּ אֶת בָּנָיו אִם כֻּלָּם נֶאֱמָנִים לַה׳. עָנוּ כֻּלָּם: "שְׁמַע יִשְׂרָאֵל ה׳ אֱלֹהֵינוּ ה׳ אֶחָד". וְיַעֲקֹב עָנָה: "בָּרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ לְעוֹלָם וָעֶד".
הַמִּשְׁפָּט הַזֶּה לֹא כָּתוּב בַּתּוֹרָה, וְלָכֵן אוֹמְרִים אוֹתוֹ מִיָּד אַחֲרֵי הַפָּסוּק הָרִאשׁוֹן – בְּלַחַשׁ.
💛 אַהֲבַת הַשֵּׁם: יַעֲקֹב שָׂמַח כָּל כָּךְ שֶׁכָּל בָּנָיו אוֹהֲבִים אֶת ה׳. גַּם הַיּוֹם, כְּשֶׁאֲנַחְנוּ אוֹמְרִים "שְׁמַע", אֲנַחְנוּ מְשַׂמְּחִים אֶת אָבִינוּ – וְאֶת ה׳.
📖 פֶּרֶק ט״ו, הֲלָכָה ז׳

🔍קוֹרְאִים כְּמוֹ מִכְתָּב מֵהַמֶּלֶךְ

קוֹרְאִים קְרִיאַת שְׁמַע בְּכַוָּנָה, כְּאִלּוּ קוֹרְאִים עַכְשָׁו מִכְתָּב מֵהַמֶּלֶךְ. מְדַקְדְּקִים בְּכָל אוֹת, וְלֹא בּוֹלְעִים מִלִּים.
אוֹמְרִים בַּפֶּה וּמַשְׁמִיעִים לָאֹזֶן. מִי שֶׁרַק חָשַׁב אֶת הַמִּלִּים בַּלֵּב וְלֹא אָמַר אוֹתָן – לֹא יָצָא.
💛 אַהֲבַת הַשֵּׁם: מִכְתָּב מֵאַבָּא אָהוּב קוֹרְאִים לְאַט, מִלָּה אַחַר מִלָּה. כָּךְ גַּם אֶת דִּבְרֵי ה׳ – כִּי כָּל מִלָּה מִמֶּנּוּ יְקָרָה לָנוּ.
📖 פֶּרֶק ט״ו, הֲלָכָה ט׳

🧠 בְּדֹק אֶת עַצְמְךָ

צִיצִית הִיא מִצְוָה שֶׁל יוֹם. אִם כָּךְ, לָמָּה קוֹרְאִים אֶת פָּרָשַׁת "וַיֹּאמֶר" גַּם בַּלַּיְלָה?
💡 מִצְוָה לִזְכֹּר אֶת יְצִיאַת מִצְרַיִם בַּיּוֹם וּבַלַּיְלָה, וּפָרָשַׁת "וַיֹּאמֶר" מִסְתַּיֶּמֶת בָּהּ. עֹל מַלְכוּת שָׁמַיִם – בַּפָּרָשָׁה הָרִאשׁוֹנָה.
לָמָּה פָּרָשַׁת "שְׁמַע" בָּאָה לִפְנֵי פָּרָשַׁת "וְהָיָה אִם שָׁמֹעַ"?
💡 קֹדֶם מְקַבְּלִים אֶת ה׳ כְּמֶלֶךְ, וְאַחַר כָּךְ אֶת מִצְווֹתָיו. שָׂכָר וָעֹנֶשׁ – בְּ"וְהָיָה אִם שָׁמֹעַ", וְצִיצִית – בְּ"וַיֹּאמֶר".
אֵיךְ מְלַמֶּדֶת אוֹתָנוּ פָּרָשַׁת שְׁמַע לֹא לִשְׁכֹּחַ אֶת הָאֱמוּנָה בַּה׳?
💡 גַּם מִי שֶׁלָּמַד הֵיטֵב עָלוּל לִשְׁכֹּחַ בְּתוֹךְ טִרְדוֹת הַיּוֹם. לָכֵן הַתּוֹרָה צִוְּתָה לִקְרֹא שְׁמַע בַּבֹּקֶר וּבָעֶרֶב, לְהַנִּיחַ אֶת הַפָּרָשָׁה בַּתְּפִלִּין וְלִקְבֹּעַ אוֹתָהּ בַּמְּזוּזָה.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

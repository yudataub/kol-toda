טַעֲמֵי מִצְווֹת הַנָּשִׁים

🌸 תְּפִלַּת בָּנוֹת · פֶּרֶק ג׳

🏡לָמָּה פְּטוּרוֹת?

הַטַּעַם הַפָּשׁוּט שֶׁנָּשִׁים פְּטוּרוֹת מִמִּצְווֹת עֲשֵׂה שֶׁהַזְּמַן גְּרָמָן: כְּדֵי שֶׁיּוּכְלוּ לִבְנוֹת אֶת הַבַּיִת וְאֶת הַמִּשְׁפָּחָה. גִּדּוּל הַיְּלָדִים וְחִנּוּכָם דּוֹרְשִׁים מְסִירוּת בְּכָל שְׁעוֹת הַיּוֹם וְהַלַּיְלָה, וְאִי אֶפְשָׁר תָּמִיד לַעֲצֹר בִּזְמַן קָבוּעַ.
מִכָּאן לוֹמְדִים כַּמָּה גָּדוֹל עֶרְכָּהּ שֶׁל הַמִּשְׁפָּחָה, שֶׁבִּשְׁבִילָהּ פָּטְרָה הַתּוֹרָה אֶת הַנָּשִׁים מִמִּצְווֹת אֵלּוּ וּמִתַּלְמוּד תּוֹרָה בְּעִיּוּן.
💛 אַהֲבַת הַשֵּׁם: הַבַּיִת הַיְּהוּדִי הוּא כְּמוֹ מִקְדָּשׁ קָטָן. מִי שֶׁבּוֹנָה אוֹתוֹ בְּאַהֲבָה – עוֹבֶדֶת אֶת ה׳ בְּכָל רֶגַע.
📖 פֶּרֶק ג׳, הֲלָכָה ב׳

🌸נָשִׁים צַדְקָנִיּוֹת

בָּאֵרוּעִים הַגְּדוֹלִים שֶׁל עַם יִשְׂרָאֵל נִתְגַּלְּתָה מַעֲלַת הַנָּשִׁים: ״בִּזְכוּת נָשִׁים צַדְקָנִיּוֹת יָצְאוּ יִשְׂרָאֵל מִמִּצְרַיִם״. בְּמַתַּן תּוֹרָה פָּנוּ קֹדֶם אֶל הַנָּשִׁים. וְהַנָּשִׁים לֹא חָטְאוּ בְּחֵטְא הָעֵגֶל וְלֹא בְּחֵטְא הַמְרַגְּלִים.
וְעַל הֶעָתִיד אָמְרוּ חֲכָמִים: ״אֵין הַדּוֹרוֹת נִגְאָלִים אֶלָּא בִּזְכוּת נָשִׁים צַדְקָנִיּוֹת שֶׁבַּדּוֹר״.
💛 אַהֲבַת הַשֵּׁם: הָאֱמוּנָה שֶׁל הַנָּשִׁים הָיְתָה חֲזָקָה גַּם בְּשָׁעוֹת קָשׁוֹת. כָּל בַּת יְכוֹלָה לִהְיוֹת חֻלְיָה בַּשַּׁרְשֶׁרֶת הַזֹּאת שֶׁל אֱמוּנָה וְאַהֲבַת ה׳.
📖 פֶּרֶק ג׳, הֲלָכָה ה׳

💗תְּפִלָּה מֵהַלֵּב

בִּתְפִלַּת בָּנוֹת בּוֹלֵט הַצַּד הָאִישִׁי. מִכֵּיוָן שֶׁבַּת אֵינָהּ צְרִיכָה לְהִתְפַּלֵּל בְּמִנְיָן, הִיא יְכוֹלָה לִבְחֹר זְמַן שֶׁבּוֹ תּוּכַל לְכַוֵּן יוֹתֵר, וּלְהִתְפַּלֵּל בַּקֶּצֶב שֶׁלָּהּ.
בַּת שֶׁאֵינָהּ יוֹדַעַת לְהִתְפַּלֵּל בַּנֻּסָּח – תִּתְפַּלֵּל בִּלְשׁוֹנָהּ כְּפִי יְכָלְתָּהּ. וְהִיא אֵינָהּ חַיֶּבֶת לִשְׁמֹעַ חֲזָרַת הַשַּׁ״ץ.
💛 אַהֲבַת הַשֵּׁם: ״וְחַנָּה הִיא מְדַבֶּרֶת עַל לִבָּהּ״ – עִקַּר תְּפִלַּת הַנָּשִׁים הוּא בַּקָּשַׁת הָרַחֲמִים. ה׳ שׁוֹמֵעַ כָּל מִלָּה שֶׁיּוֹצֵאת מִלֵּב אוֹהֵב.
📖 פֶּרֶק ג׳, הֲלָכָה ט׳

🧠 בְּדֹק אֶת עַצְמְךָ

לָמָּה, לְפִי הַטַּעַם הַפָּשׁוּט, נָשִׁים פְּטוּרוֹת מִמִּצְווֹת עֲשֵׂה שֶׁהַזְּמַן גְּרָמָן?
💡 הָאַחְרָיוּת לִבְנוֹת אֶת הַמִּשְׁפָּחָה דּוֹרֶשֶׁת מְסִירוּת בְּכָל שְׁעוֹת הַיּוֹם, וְלָכֵן פָּטְרָה הַתּוֹרָה אֶת הַנָּשִׁים מִמִּצְווֹת שֶׁצָּרִיךְ לַעֲצֹר בִּשְׁבִילָן בִּזְמַן קָבוּעַ. מִכָּאן לוֹמְדִים כַּמָּה גָּדוֹל עֶרְכָּהּ שֶׁל הַמִּשְׁפָּחָה. מֵאוֹתוֹ טַעַם נָשִׁים פְּטוּרוֹת גַּם מִתַּלְמוּד תּוֹרָה בְּעִיּוּן.
בַּת אֵינָהּ חַיֶּבֶת לְהִתְפַּלֵּל בְּמִנְיָן. אֵיזוֹ מַעֲלָה יֵשׁ בָּזֶה לִתְפִלָּתָהּ?
💡 בִּתְפִלַּת בָּנוֹת בּוֹלֵט הַצַּד הָאִישִׁי: הִיא יְכוֹלָה לִבְחֹר זְמַן שֶׁבּוֹ תְּכַוֵּן יוֹתֵר, וְהַקֶּצֶב שֶׁלָּהּ אֵינוֹ תָּלוּי בַּצִּבּוּר. אֲבָל אֶת הָעֲמִידָה הִיא מִתְפַּלֶּלֶת גַּם בְּיָמֵי חֹל, וַחֲזָרַת הַשַּׁ״ץ אֵינָהּ בִּמְקוֹם תְּפִלָּתָהּ.

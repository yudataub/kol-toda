פְּרָזִים וּמֻקָּפִים

🎭 הִלְכוֹת פּוּרִים · פֶּרֶק י״ז

🏯🔑 מוּשָּׂג: שׁוּשָׁן פּוּרִים

״בַּכְּרַכִּים הַמֻּקָּפִים חוֹמָה מִימוֹת יְהוֹשֻׁעַ בִּן נוּן וּבְשׁוּשָׁן הַבִּירָה – בְּיוֹם ט״ו בַּאֲדָר״.
📖 שׁוּשָׁן פּוּרִים – יוֹם ט״ו בַּאֲדָר, שֶׁבּוֹ חוֹגְגִים פּוּרִים בְּעָרִים מֻקָּפוֹת חוֹמָה, כְּמוֹ בִּירוּשָׁלַיִם.
💛 אַהֲבַת הַשֵּׁם: 💛 בְּשׁוּשָׁן הַבִּירָה עַצְמָהּ נִלְחֲמוּ הַיְּהוּדִים יוֹם נוֹסָף, וְלָכֵן גַּם עָרִים מֻקָּפוֹת חוֹמָה חוֹגְגוֹת יוֹם נוֹסָף. 🧒 וְאַתָּה? אִם תְּבַקֵּר בִּירוּשָׁלַיִם בְּפוּרִים – תִּזְכֹּר לַחְגֹּג יוֹם אֶחָד אַחֲרֵי כֻּלָּם.
📖 פֶּרֶק י״ז, הֲלָכָה א׳

🧱📜 לְשׁוֹן חֲכָמִים: פָּרוּז וּמֻקָּף

״בְּכָל הַמְּקוֹמוֹת עוֹשִׂים אֶת הַפּוּרִים בְּיוֹם י״ד בַּאֲדָר, וּבַכְּרַכִּים הַמֻּקָּפִים חוֹמָה מִימוֹת יְהוֹשֻׁעַ בִּן נוּן – בְּיוֹם ט״ו״.
📖 פָּרוּז – גָּר בְּעִיר בְּלִי חוֹמָה עַתִּיקָה, חוֹגֵג בְּי״ד. מֻקָּף – גָּר בְּעִיר מֻקֶּפֶת חוֹמָה, חוֹגֵג בְּט״ו.
💛 אַהֲבַת הַשֵּׁם: 💛 בְּכָל מָקוֹם שֶׁבּוֹ יְהוּדִים גָּרִים – בִּכְרַךְ גָּדוֹל אוֹ בִּכְפָר קָטָן – ה׳ קָבַע לָהֶם זְמַן שִׂמְחָה מְדֻיָּק. 🧒 וְאַתָּה? תִּבְדֹּק יַחַד עִם הַהוֹרִים הַאִם הָעִיר שֶׁלְּךָ חוֹגֶגֶת בְּי״ד אוֹ בְּט״ו.
📖 פֶּרֶק י״ז, הֲלָכָה א׳

🏙️יְרוּשָׁלַיִם וּמְקוֹמוֹת שֶׁל סָפֵק

"כְּרַךְ וְכָל הַסָּמוּךְ לוֹ" – לָכֵן לֹא רַק בָּעִיר הָעַתִּיקָה, אֶלָּא בְּכָל שְׁכוּנוֹת יְרוּשָׁלַיִם הַסְּמוּכוֹת זוֹ לָזוֹ חוֹגְגִים בְּט״ו. יֵשׁ עָרִים שֶׁיֵּשׁ עֲלֵיהֶן סָפֵק אִם הָיוּ מֻקָּפוֹת חוֹמָה בִּימֵי יְהוֹשֻׁעַ – כְּמוֹ טְבֶרְיָה, חֶבְרוֹן, צְפַת, עַכּוֹ וְלוֹד. בְּכֻלָּן קוֹרְאִים אֶת הַמְּגִלָּה בְּי״ד בִּבְרָכָה.
בְּרֹב הַמְּקוֹמוֹת הַמְּסֻפָּקִים נוֹהֲגִים פּוּרִים רַק בְּי״ד. בִּמְקוֹמוֹת שֶׁהַסָּפֵק חָזָק, כְּמוֹ טְבֶרְיָה וְחֶבְרוֹן, רַבִּים קוֹרְאִים גַּם בְּט״ו – בְּלִי בְּרָכָה.
💛 אַהֲבַת הַשֵּׁם: בְּעֶזְרַת ה׳ יְרוּשָׁלַיִם הוֹלֶכֶת וְנִבְנֵית, וְהַשְּׁכוּנוֹת הָרְחוֹקוֹת נַעֲשׂוֹת סְמוּכוֹת לָעִיר. כָּל בַּיִת חָדָשׁ בָּהּ הוּא חֵלֶק מִן הַגְּאֻלָּה.
📖 פֶּרֶק י״ז, הֲלָכוֹת ב׳–ג׳

🚌פָּרוּז בֶּן יוֹמוֹ

הַחִיּוּב נִקְבָּע לְפִי הַמָּקוֹם שֶׁבּוֹ נִמְצָא הָאָדָם בְּיוֹם הַפּוּרִים, וְלֹא לְפִי מְקוֹם דִּירָתוֹ הַקְּבוּעָה. הַזְּמַן הַקּוֹבֵעַ הוּא עֲלוֹת הַשַּׁחַר. יְרוּשַׁלְמִי שֶׁבָּא לְעִיר אַחֶרֶת בְּלֵיל י״ד וְנִשְׁאַר שָׁם עַד אַחֲרֵי עֲלוֹת הַשַּׁחַר – חוֹגֵג כְּמוֹ אַנְשֵׁי הַפְּרָזִים, בְּי״ד.
אֲבָל אִם הִגִּיעַ לָעִיר הָאַחֶרֶת רַק בְּיוֹם י״ד, אַחֲרֵי עֲלוֹת הַשַּׁחַר, וְחָזַר לִירוּשָׁלַיִם – דִּינוֹ כִּירוּשַׁלְמִי, וְהוּא חוֹגֵג בְּט״ו.
💛 אַהֲבַת הַשֵּׁם: בְּכָל מָקוֹם שֶׁיְּהוּדִי נִמְצָא – שָׁם הוּא שָׂמֵחַ עִם הַצִּבּוּר שֶׁסְּבִיבוֹ. הַשִּׂמְחָה שֶׁלָּנוּ הִיא תָּמִיד יַחַד, עִם כְּלַל יִשְׂרָאֵל.
📖 פֶּרֶק י״ז, הֲלָכָה ד׳

📅פּוּרִים הַמְשֻׁלָּשׁ

י״ד בַּאֲדָר לְעוֹלָם אֵינוֹ חָל בְּשַׁבָּת, אֲבָל ט״ו יָכוֹל לָחוּל בְּשַׁבָּת. אָז הַפּוּרִים שֶׁל יְרוּשָׁלַיִם מִתְחַלֵּק לִשְׁלוֹשָׁה יָמִים: בְּיוֹם שִׁשִּׁי – קְרִיאַת הַמְּגִלָּה וּמַתָּנוֹת לָאֶבְיוֹנִים; בְּשַׁבָּת – "עַל הַנִּסִּים" וּקְרִיאַת הַתּוֹרָה שֶׁל פּוּרִים; וּבְיוֹם רִאשׁוֹן – הַסְּעוּדָה וּמִשְׁלוֹחַ הַמָּנוֹת.
לָמָּה לֹא קוֹרְאִים מְגִלָּה בְּשַׁבָּת? חֲכָמִים גָּזְרוּ, שֶׁמָּא יְטַלְטֵל אֶת הַמְּגִלָּה בִּרְשׁוּת הָרַבִּים. וּמַתָּנוֹת לָאֶבְיוֹנִים נוֹתְנִים כְּבָר בְּיוֹם שִׁשִּׁי, כִּי עֵינֵיהֶם שֶׁל הָעֲנִיִּים נְשׂוּאוֹת לְמִקְרָא מְגִלָּה.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁהַשַּׁבָּת וּפוּרִים נִפְגָּשִׁים, הַהֲלָכָה דּוֹאֶגֶת שֶׁהֶעָנִי לֹא יִצְטַעֵר. ה׳ חוֹשֵׁב קֹדֶם כֹּל עַל הַחַלָּשִׁים – וְכָךְ גַּם אֲנַחְנוּ.
📖 פֶּרֶק י״ז, הֲלָכָה ה׳

3️⃣🔑 מוּשָּׂג: פּוּרִים הַמְּשֻׁלָּשׁ

״כַּאֲשֶׁר חָל ט״ו בַּאֲדָר בְּשַׁבָּת, נִקְרָא אוֹתוֹ פּוּרִים ׳פּוּרִים הַמְּשֻׁלָּשׁ׳, מִפְּנֵי שֶׁדִּינֵי הַפּוּרִים שֶׁלּוֹ מִתְחַלְּקִים לִשְׁלֹשָׁה יָמִים״.
📖 פּוּרִים הַמְּשֻׁלָּשׁ – בִּירוּשָׁלַיִם: קוֹרְאִים מְגִלָּה בְּשִׁשִּׁי, אוֹמְרִים ״עַל הַנִּסִּים״ בְּשַׁבָּת, וְעוֹשִׂים סְעוּדָה בְּרִאשׁוֹן.
💛 אַהֲבַת הַשֵּׁם: 💛 גַּם כְּשֶׁהַשִּׂמְחָה צְרִיכָה לְהִתְפַּזֵּר עַל פְּנֵי שְׁלֹשָׁה יָמִים – הִיא לֹא פָּחוֹת שְׁלֵמָה. 🧒 וְאַתָּה? אִם תִּהְיֶה בִּירוּשָׁלַיִם בְּפוּרִים מְשֻׁלָּשׁ – תִּשְׂמַח שְׁלֹשָׁה יָמִים בְּרֶצֶף.
📖 פֶּרֶק י״ז, הֲלָכָה ה׳

🧠 בְּדֹק אֶת עַצְמְךָ

אֵלִיָּהוּ מִירוּשָׁלַיִם נָסַע לְסַבָּא בְּפֶתַח תִּקְוָה לְלֵיל י״ד, וְנִשְׁאַר שָׁם עַד אַחֲרֵי עֲלוֹת הַשַּׁחַר. מָתַי הוּא חוֹגֵג?
💡 הַחִיּוּב נִקְבָּע לְפִי הַמָּקוֹם שֶׁבּוֹ נִמְצָא בַּעֲלוֹת הַשַּׁחַר שֶׁל י״ד. מִי שֶׁבָּא לַפְּרָזִים בַּלַּיְלָה וְנִשְׁאַר עַד אַחֲרֵי עֲלוֹת הַשַּׁחַר – חוֹגֵג בְּי״ד, גַּם אִם יַחֲזֹר אַחַר כָּךְ לִירוּשָׁלַיִם.
ט״ו בַּאֲדָר חָל בְּשַׁבָּת. מָתַי עוֹשִׂים בִּירוּשָׁלַיִם אֶת סְעוּדַת הַפּוּרִים?
💡 אֶת הַמְּגִלָּה לֹא מְאַחֲרִים אַחֲרֵי זְמַן הַנֵּס, וְלָכֵן קוֹרְאִים בְּשִׁשִּׁי. אֲבָל אֶת הַסְּעוּדָה אֶפְשָׁר לַעֲשׂוֹת רַק אַחֲרֵי שֶׁהִגִּיעַ יוֹם הַנֵּס, וּבְשַׁבָּת לֹא עוֹשִׂים אוֹתָהּ – לָכֵן הִיא בְּיוֹם רִאשׁוֹן, עִם מִשְׁלוֹחַ הַמָּנוֹת.
📜 לְשׁוֹן חֲכָמִים: מָתַי חוֹגְגִים פּוּרִים בְּי״ד, וּמָתַי בְּט״ו?
💡 בְּכָל הַמְּקוֹמוֹת עוֹשִׂים אֶת הַפּוּרִים בְּיוֹם י״ד, וּבַכְּרַכִּים הַמֻּקָּפִים חוֹמָה מִימוֹת יְהוֹשֻׁעַ בִּן נוּן – בְּיוֹם ט״ו.

🎵שְׁלֹשֶׁת הַשָּׁבוּעוֹת

מִי״ז בְּתַמּוּז וְעַד תִּשְׁעָה בְּאָב הֵם "בֵּין הַמְּצָרִים" – יְמֵי צַעַר. בָּהֶם לֹא רוֹקְדִים, לֹא מְנַגְּנִים, וְלֹא מְבָרְכִים "שֶׁהֶחֱיָנוּ". לָכֵן לֹא אוֹכְלִים פְּרִי חָדָשׁ וְלֹא קוֹנִים בֶּגֶד חָדָשׁ שֶׁמְּבָרְכִים עָלָיו "שֶׁהֶחֱיָנוּ".
דְּבָרִים שֶׁאֵין מְבָרְכִים עֲלֵיהֶם "שֶׁהֶחֱיָנוּ", כְּמוֹ גַּרְבַּיִם אוֹ גּוּפִיּוֹת, מֻתָּר לִקְנוֹת עַד סוֹף תַּמּוּז. וּמֻתָּר לְטַיֵּל עַד רֹאשׁ חֹדֶשׁ אָב, בְּמִשְׁנֶה זְהִירוּת.
💛 אַהֲבַת הַשֵּׁם: אֵיךְ נֹאמַר "שֶׁהֶחֱיָנוּ... לַזְּמַן הַזֶּה" בִּימֵי הַחֻרְבָּן? אָנוּ מְחַכִּים לְבָרֵךְ אוֹתָהּ בְּשִׂמְחָה שְׁלֵמָה, כְּשֶׁיִּבָּנֶה הַמִּקְדָּשׁ.
📖 פֶּרֶק ח׳, הֲלָכוֹת א׳–ב׳, ו׳–ז׳

✂️תִּסְפֹּרֶת בְּבֵין הַמְּצָרִים

חֲכָמִים תִּקְּנוּ שֶׁלֹּא לְהִסְתַּפֵּר בַּשָּׁבוּעַ שֶׁחָל בּוֹ תִּשְׁעָה בְּאָב, וְכָךְ נוֹהֲגִים רַבִּים מִן הַסְּפָרַדִּים. הָאַשְׁכְּנַזִּים, וְחֵלֶק מִן הַסְּפָרַדִּים, כְּמוֹ יוֹצְאֵי מָרוֹקוֹ וְגִ׳רְבָּה, מַחְמִירִים שֶׁלֹּא לְהִסְתַּפֵּר בְּכָל שְׁלֹשֶׁת הַשָּׁבוּעוֹת.
בַּשָּׁבוּעַ שֶׁחָל בּוֹ תִּשְׁעָה בְּאָב אָסוּר לְסַפֵּר גַּם יְלָדִים קְטַנִּים, כְּדֵי לְחַנְּכָם לְהִתְאַבֵּל. לִפְנֵי כֵן, בִּשְׁעַת הַצֹּרֶךְ, מְקִלִּים לַקְּטַנִּים. לְהִסְתָּרֵק מֻתָּר גַּם בַּשָּׁבוּעַ שֶׁחָל בּוֹ.
💛 אַהֲבַת הַשֵּׁם: הַשֵּׂעָר שֶׁלֹּא סֻפַּר מְסַפֵּר לְכֻלָּם שֶׁאָנוּ מִתְאַבְּלִים עַל יְרוּשָׁלַיִם. כָּךְ גַּם הַגּוּף שֶׁלָּנוּ זוֹכֵר אֶת הַמִּקְדָּשׁ.

״שְׁלֹשֶׁת הַשָּׁבוּעוֹת... הֵם יָמִים שֶׁל צַעַר, שֶׁעֲלֵיהֶם נֶאֱמַר: 'כָּל רוֹדְפֶיהָ הִשִּׂיגוּהָ בֵּין הַמְּצָרִים'״.
📖 בֵּין הַמְּצָרִים – שְׁלֹשֶׁת הַשָּׁבוּעוֹת שֶׁבֵּין י״ז בְּתַמּוּז לְתִשְׁעָה בְּאָב, יָמִים שֶׁל צַעַר וּמְצוּקָה.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים הִמְלִיצוּ לִנְקוֹט מִשְׁנֵה זְהִירוּת בַּיָּמִים הַמּוּעָדִים לְפֻרְעָנוּת – ה׳ רוֹצֶה שֶׁנִּשְׁמוֹר עַל עַצְמֵנוּ בְּתוֹך הָאֲבֵלוּת. 🧒 וְאַתָּה? בִּשְׁלֹשֶׁת הַשָּׁבוּעוֹת – הֱיֵה קְצָת יוֹתֵר זָהִיר, וְהִמָּנַע מִשִּׁמְחָה גְּדוֹלָה בְּלִי צוֹרֶך.

🎉בַּר מִצְוָה בְּבֵין הַמְּצָרִים

מֻתָּר לָשִׁיר שִׁירֵי שִׂמְחָה בִּסְעוּדַת מִצְוָה, כְּמוֹ בְּרִית מִילָה, פִּדְיוֹן הַבֵּן וְשֶׁבַע בְּרָכוֹת. גַּם שִׂמְחַת בַּר מִצְוָה וּבַת מִצְוָה מֻתֶּרֶת – בִּתְנַאי שֶׁעוֹרְכִים אוֹתָהּ בַּיּוֹם שֶׁהִגִּיעוּ לְמִצְווֹת.
מִשֶּׁנִּכְנַס חֹדֶשׁ אָב אֵין מַזְמִינִים תִּזְמֹרֶת לְשׁוּם שִׂמְחָה, וְאֵין מַשְׁמִיעִים שִׁירֵי שִׂמְחָה מִמַּכְשִׁיר. רַק אֶת שִׁירֵי הַמִּצְוָה מֻתָּר לָשִׁיר בַּפֶּה, וְאַף לִרְקֹד מְעַט בְּמַעְגָּל.
💛 אַהֲבַת הַשֵּׁם: שִׂמְחַת מִצְוָה הִיא שִׂמְחָה שֶׁל קְדֻשָּׁה. גַּם בִּימֵי הָאֵבֶל, מִצְוָה מְבִיאָה אוֹר.
📖 פֶּרֶק ח׳, הֲלָכָה ג׳

🎵שִׁירִים בִּשְׁלֹשֶׁת הַשָּׁבוּעוֹת

הָרַב מֶלָמֵד מְחַלֵּק אֶת הַשִּׁירִים לִשְׁלֹשָׁה: שִׁירֵי שִׂמְחָה, כְּמוֹ שִׁירֵי חֲתֻנָּה; שִׁירִים רְגִילִים – לֹא שְׂמֵחִים בִּמְיֻחָד וְלֹא עֲצוּבִים; וְשִׁירִים עֲצוּבִים, עַל הַחֻרְבָּן. מִתְּחִלַּת שְׁלֹשֶׁת הַשָּׁבוּעוֹת לֹא שׁוֹמְעִים שִׁירֵי שִׂמְחָה. מֵרֹאשׁ חֹדֶשׁ אָב – גַּם לֹא שִׁירִים רְגִילִים, וְרַק שִׁירִים עֲצוּבִים.
גַּם שִׁיר שֶׁמֻּתָּר לִשְׁמֹעַ – לֹא שׁוֹמְעִים בְּקוֹל רָם, כִּי הָעָצְמָה הוֹפֶכֶת אוֹתוֹ לְשִׁיר שֶׁל שִׂמְחָה. וְשִׁירִים עֲצוּבִים עַל הַחֻרְבָּן – יֵשׁ בָּהֶם מַעֲלָה, כִּי הֵם מְעוֹרְרִים אֶת הַלֵּב לְהִתְאַבֵּל.
💛 אַהֲבַת הַשֵּׁם: הַשִּׁירִים שֶׁאָנוּ שׁוֹמְעִים מְכַוְּנִים אֶת הַלֵּב. בִּימֵי בֵּין הַמְּצָרִים אֲנַחְנוּ בּוֹחֲרִים שִׁירִים שֶׁמְּעוֹרְרִים גַּעְגּוּעִים לְבֵית הַמִּקְדָּשׁ.
📖 פֶּרֶק ח׳, הֲלָכוֹת ד׳–ה׳

🍇📜 לְשׁוֹן חֲכָמִים: שֶׁהֶחֱיָנוּ בַּשָּׁבוּעוֹת

״נָהֲגוּ יִשְׂרָאֵל... נִמְנָעִים בָּהֶם מֵרִקּוּדִים וּמְחוֹלוֹת, וְנִמְנָעִים מִלְּבָרֵךְ בָּהֶם 'שֶׁהֶחֱיָנוּ'״.
📖 שֶׁהֶחֱיָנוּ – בְּרָכָה עַל שִׂמְחָה חֲדָשָׁה, כְּגוֹן פְּרִי חָדָשׁ אוֹ בֶּגֶד חָדָשׁ. נִמְנָעִים מִמֶּנָּה בִּשְׁלֹשֶׁת הַשָּׁבוּעוֹת.
💛 אַהֲבַת הַשֵּׁם: בְּיָמִים שֶׁל אֲבֵלוּת מְמַעֲטִים בְּשִׂמְחָה חֲדָשָׁה, כְּדֵי שֶׁנִּזְכּוֹר אֶת הֶחֶסְרוֹן – אֲבָל ה׳ מַבְטִיח שֶׁהַשִּׂמְחָה תָּשׁוּב בִּמְהֵרָה. 🧒 וְאַתָּה? דְּחֵה פְּרִי חָדָשׁ אוֹ בֶּגֶד חָדָשׁ לְאַחַר תִּשְׁעָה בְּאָב, וּתְבָרֵך עָלָיו בְּשִׂמְחָה כְּפוּלָה.
📖 פֶּרֶק ח׳, הֲלָכָה ז׳

🍇״שֶׁהֶחֱיָנוּ״ בְּבֵין הַמְּצָרִים

בְּבֵין הַמְּצָרִים לֹא מְבָרְכִים ״שֶׁהֶחֱיָנוּ״ עַל פְּרִי חָדָשׁ בִּימֵי הַחֹל. מִי שֶׁיֵּשׁ לוֹ פְּרִי חָדָשׁ – דּוֹחֶה אוֹתוֹ לְשַׁבָּת, וּמְבָרֵךְ עָלָיו בְּשַׁבָּת. וְאִם הַפְּרִי עָלוּל לְהִתְקַלְקֵל עַד שַׁבָּת – אוֹכֵל מִיָּד וּמְבָרֵךְ.
מִי שֶׁהִזְדַּמְּנָה לוֹ מִצְוָה שֶׁמְּבָרְכִים עָלֶיהָ ״שֶׁהֶחֱיָנוּ״, כְּמוֹ פִּדְיוֹן הַבֵּן – מְבָרֵךְ. וְגַם הָרוֹאֶה חָבֵר אָהוּב אַחֲרֵי שְׁלֹשִׁים יוֹם – מְבָרֵךְ, כְּדֵי שֶׁלֹּא יַפְסִיד אֶת הַבְּרָכָה.
💛 אַהֲבַת הַשֵּׁם: גַּם בִּימֵי הָאֵבֶל, ה׳ שׁוֹלֵחַ לָנוּ שְׂמָחוֹת קְטַנּוֹת. הַשַּׁבָּת נוֹתֶנֶת לָנוּ מָקוֹם לְהוֹדוֹת עֲלֵיהֶן.

💍📜 לְשׁוֹן חֲכָמִים: נִשּׂוּאִין בֵּין הַמְּצָרִים

״נָהֲגוּ בְּרֹב קְהִלּוֹת יִשְׂרָאֵל שֶׁלֹּא לָשֵׂאת אִשָּׁה בִּשְׁלֹשֶׁת הַשָּׁבוּעוֹת... מִפְּנֵי שֶׁרָאוּי שֶׁהַנִּשּׂוּאִין יֵעָרְכוּ בְּסִימָן טוֹב וּבְמַזָּל טוֹב״.
📖 נִשּׂוּאִין – חֲתוּנָה. בִּשְׁלֹשֶׁת הַשָּׁבוּעוֹת נָהֲגוּ שֶׁלֹּא לְקַיֵּם נִשּׂוּאִין, מִפְּנֵי שֶׁאֵלּוּ יְמֵי פֻּרְעָנוּת.

🥩תִּשְׁעַת הַיָּמִים

"מִשֶּׁנִּכְנַס אָב מְמַעֲטִים בְּשִׂמְחָה". לָכֵן נָהֲגוּ שֶׁלֹּא לֶאֱכֹל בָּשָׂר וְלֹא לִשְׁתּוֹת יַיִן, שֶׁהֵם מְשַׂמְּחִים. לְמִנְהַג הָאַשְׁכְּנַזִּים – מֵרֹאשׁ חֹדֶשׁ אָב, כּוֹלֵל רֹאשׁ חֹדֶשׁ. לְמִנְהַג רֹב הַסְּפָרַדִּים – מִסּוֹף יוֹם רֹאשׁ חֹדֶשׁ.
גַּם בְּשַׂר עוֹף אָסוּר, וְגַם מִיץ עֲנָבִים. דָּגִים מֻתָּרִים. בְּשַׁבָּת אוֹכְלִים בָּשָׂר וְשׁוֹתִים יַיִן כָּרָגִיל, וְכֵן בִּסְעוּדַת מִצְוָה, כְּמוֹ בְּרִית מִילָה אוֹ סִיּוּם מַסֶּכֶת.

👕כְּבִיסָה וּרְחִיצָה

בִּגְדֵי תִּינוֹקוֹת שֶׁמִּתְלַכְלְכִים – מֻתָּר לְכַבֵּס. בִּרְחִיצָה: בְּאַשְׁכְּנַז נָהֲגוּ שֶׁלֹּא לְהִתְרַחֵץ כְּלָל בְּתִשְׁעַת הַיָּמִים, אֲפִלּוּ בְּקָרִים, וְהַסְּפָרַדִּים – לֹא בְּחַמִּים בַּשָּׁבוּעַ שֶׁחָל בּוֹ. הָרַב מֶלָמֵד כּוֹתֵב שֶׁכַּיּוֹם מִי שֶׁמִּצְטַעֵר אִם לֹא יִתְרַחֵץ – רַשַּׁאי לְהִתְרַחֵץ בְּפוֹשְׁרִים, לְשֵׁם נִקָּיוֹן וְלֹא לְתַעֲנוּג.

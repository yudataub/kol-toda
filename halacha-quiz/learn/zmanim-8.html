מִנְהֲגֵי שְׁלֹשֶׁת הַשָּׁבוּעוֹת

📅 רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת · פֶּרֶק ח׳

🎵שְׁלֹשֶׁת הַשָּׁבוּעוֹת

מִי״ז בְּתַמּוּז וְעַד תִּשְׁעָה בְּאָב הֵם "בֵּין הַמְּצָרִים" – יְמֵי צַעַר. בָּהֶם לֹא רוֹקְדִים, לֹא מְנַגְּנִים, וְלֹא מְבָרְכִים "שֶׁהֶחֱיָנוּ". לָכֵן לֹא אוֹכְלִים פְּרִי חָדָשׁ וְלֹא קוֹנִים בֶּגֶד חָדָשׁ שֶׁמְּבָרְכִים עָלָיו "שֶׁהֶחֱיָנוּ".
דְּבָרִים שֶׁאֵין מְבָרְכִים עֲלֵיהֶם "שֶׁהֶחֱיָנוּ", כְּמוֹ גַּרְבַּיִם אוֹ גּוּפִיּוֹת, מֻתָּר לִקְנוֹת עַד סוֹף תַּמּוּז. וּמֻתָּר לְטַיֵּל עַד רֹאשׁ חֹדֶשׁ אָב, בְּמִשְׁנֶה זְהִירוּת.
💛 אַהֲבַת הַשֵּׁם: אֵיךְ נֹאמַר "שֶׁהֶחֱיָנוּ... לַזְּמַן הַזֶּה" בִּימֵי הַחֻרְבָּן? אָנוּ מְחַכִּים לְבָרֵךְ אוֹתָהּ בְּשִׂמְחָה שְׁלֵמָה, כְּשֶׁיִּבָּנֶה הַמִּקְדָּשׁ.
📖 פֶּרֶק ח׳, הֲלָכוֹת א׳–ב׳, ו׳–ז׳

✂️תִּסְפֹּרֶת בְּבֵין הַמְּצָרִים

חֲכָמִים תִּקְּנוּ שֶׁלֹּא לְהִסְתַּפֵּר בַּשָּׁבוּעַ שֶׁחָל בּוֹ תִּשְׁעָה בְּאָב, וְכָךְ נוֹהֲגִים רַבִּים מִן הַסְּפָרַדִּים. הָאַשְׁכְּנַזִּים, וְחֵלֶק מִן הַסְּפָרַדִּים, כְּמוֹ יוֹצְאֵי מָרוֹקוֹ וְגִ׳רְבָּה, מַחְמִירִים שֶׁלֹּא לְהִסְתַּפֵּר בְּכָל שְׁלֹשֶׁת הַשָּׁבוּעוֹת.
בַּשָּׁבוּעַ שֶׁחָל בּוֹ תִּשְׁעָה בְּאָב אָסוּר לְסַפֵּר גַּם יְלָדִים קְטַנִּים, כְּדֵי לְחַנְּכָם לְהִתְאַבֵּל. לִפְנֵי כֵן, בִּשְׁעַת הַצֹּרֶךְ, מְקִלִּים לַקְּטַנִּים. לְהִסְתָּרֵק מֻתָּר גַּם בַּשָּׁבוּעַ שֶׁחָל בּוֹ.
💛 אַהֲבַת הַשֵּׁם: הַשֵּׂעָר שֶׁלֹּא סֻפַּר מְסַפֵּר לְכֻלָּם שֶׁאָנוּ מִתְאַבְּלִים עַל יְרוּשָׁלַיִם. כָּךְ גַּם הַגּוּף שֶׁלָּנוּ זוֹכֵר אֶת הַמִּקְדָּשׁ.
📖 פֶּרֶק ח׳, הֲלָכוֹת א׳, י׳

🥩תִּשְׁעַת הַיָּמִים

"מִשֶּׁנִּכְנַס אָב מְמַעֲטִים בְּשִׂמְחָה". לָכֵן נָהֲגוּ שֶׁלֹּא לֶאֱכֹל בָּשָׂר וְלֹא לִשְׁתּוֹת יַיִן, שֶׁהֵם מְשַׂמְּחִים. לְמִנְהַג הָאַשְׁכְּנַזִּים – מֵרֹאשׁ חֹדֶשׁ אָב, כּוֹלֵל רֹאשׁ חֹדֶשׁ. לְמִנְהַג רֹב הַסְּפָרַדִּים – מִסּוֹף יוֹם רֹאשׁ חֹדֶשׁ.
גַּם בְּשַׂר עוֹף אָסוּר, וְגַם מִיץ עֲנָבִים. דָּגִים מֻתָּרִים. בְּשַׁבָּת אוֹכְלִים בָּשָׂר וְשׁוֹתִים יַיִן כָּרָגִיל, וְכֵן בִּסְעוּדַת מִצְוָה, כְּמוֹ בְּרִית מִילָה אוֹ סִיּוּם מַסֶּכֶת.
💛 אַהֲבַת הַשֵּׁם: הַבָּשָׂר וְהַיַּיִן מַזְכִּירִים אֶת הַקָּרְבָּנוֹת וְהַנְּסָכִים שֶׁעַל הַמִּזְבֵּחַ. כְּשֶׁאָנוּ מְוַתְּרִים עֲלֵיהֶם, אָנוּ זוֹכְרִים מָה חָסֵר לָנוּ.
📖 פֶּרֶק ח׳, הֲלָכוֹת י״ב–ט״ו

🛒מִשֶּׁנִּכְנַס אָב

בְּתִשְׁעַת הַיָּמִים אֵין קוֹנִים דְּבָרִים מְשַׂמְּחִים – תַּכְשִׁיטִים, בְּגָדִים, רָהִיטִים וּמְכוֹנִית – וְאֵין צוֹבְעִים אֶת הַבַּיִת אוֹ בּוֹנִים לְנוֹי. אֵין תּוֹפְרִים וְאֵין סוֹרְגִים בְּגָדִים אוֹ כִּפּוֹת חֲדָשׁוֹת. אֲבָל צָרְכֵי מִצְוָה, כְּמוֹ תְּפִלִּין, מֻתָּר לִקְנוֹת.
סְעוּדַת סִיּוּם מַסֶּכֶת עִם בָּשָׂר – רַק לְמִי שֶׁהִגִּיעַ לַסִּיּוּם לְפִי סֵדֶר לִמּוּדוֹ. לֹא מְתַכְנְנִים סִיּוּם בִּמְיֻחָד כְּדֵי לְהַתִּיר בָּשָׂר.
💛 אַהֲבַת הַשֵּׁם: בְּתִשְׁעַת הַיָּמִים אָנוּ מְמַעֲטִים בְּשִׂמְחָה שֶׁל דְּבָרִים – כְּדֵי לְהַשְׁאִיר מָקוֹם לַגַּעְגּוּעַ לַשִּׂמְחָה הָאֲמִתִּית שֶׁל בֵּית הַמִּקְדָּשׁ.
📖 פֶּרֶק ח׳, הֲלָכוֹת י״ב, ט״ו–י״ח

👕כְּבִיסָה וּרְחִיצָה

חֲכָמִים אָסְרוּ לְכַבֵּס וְלִלְבֹּשׁ בְּגָדִים מְכֻבָּסִים בַּשָּׁבוּעַ שֶׁחָל בּוֹ תִּשְׁעָה בְּאָב, וְכָךְ נוֹהֲגִים הַסְּפָרַדִּים. הָאַשְׁכְּנַזִּים מַחְמִירִים בָּזֶה מֵרֹאשׁ חֹדֶשׁ אָב, וְרַק לִכְבוֹד שַׁבַּת חֲזוֹן לוֹבְשִׁים בִּגְדֵי שַׁבָּת. לָכֵן מְכִינִים מֵרֹאשׁ כַּמָּה בְּגָדִים: לוֹבְשִׁים כָּל אֶחָד כְּשָׁעָה, וְהֵם כְּבָר אֵינָם "מְכֻבָּסִים".
בִּגְדֵי תִּינוֹקוֹת שֶׁמִּתְלַכְלְכִים – מֻתָּר לְכַבֵּס. בִּרְחִיצָה: בְּאַשְׁכְּנַז נָהֲגוּ שֶׁלֹּא לְהִתְרַחֵץ כְּלָל בְּתִשְׁעַת הַיָּמִים, אֲפִלּוּ בְּקָרִים, וְהַסְּפָרַדִּים – לֹא בְּחַמִּים בַּשָּׁבוּעַ שֶׁחָל בּוֹ. הָרַב מֶלָמֵד כּוֹתֵב שֶׁכַּיּוֹם מִי שֶׁמִּצְטַעֵר אִם לֹא יִתְרַחֵץ – רַשַּׁאי לְהִתְרַחֵץ בְּפוֹשְׁרִים, לְשֵׁם נִקָּיוֹן וְלֹא לְתַעֲנוּג.
💛 אַהֲבַת הַשֵּׁם: הָאָבֵל מַפְסִיק לְפַנֵּק אֶת עַצְמוֹ מִתּוֹךְ צַעַר. כָּךְ אָנוּ מַרְאִים שֶׁהַחֻרְבָּן נוֹגֵעַ לְלִבֵּנוּ.
📖 פֶּרֶק ח׳, הֲלָכוֹת י״ט–כ״א

🧠 בְּדֹק אֶת עַצְמְךָ

בַּשָּׁבוּעַ שֶׁחָל בּוֹ תִּשְׁעָה בְּאָב – מָה הַדִּין בְּתִסְפֹּרֶת שֶׁל יְלָדִים קְטַנִּים?
💡 בַּשָּׁבוּעַ שֶׁחָל בּוֹ אֵין מְסַפְּרִים גַּם קְטַנִּים שֶׁהִגִּיעוּ לְחִנּוּךְ, וְנוֹהֲגִים שֶׁלֹּא לְסַפֵּר אֲפִלּוּ קְטַנִּים יוֹתֵר – כְּדֵי לְבַטֵּא אֶת עָגְמַת הַנֶּפֶשׁ שֶׁל הַיָּמִים הָאֵלֶּה.
אֵיךְ מִתְכּוֹנְנִים לַיָּמִים שֶׁבָּהֶם אָסוּר לִלְבֹּשׁ בְּגָדִים מְכֻבָּסִים?
💡 לוֹבְשִׁים לִפְנֵי זְמַן הָאִסּוּר כַּמָּה בְּגָדִים, כָּל אֶחָד כְּשָׁעָה, וְאָז הֵם כְּבָר אֵינָם נֶחְשָׁבִים מְכֻבָּסִים. אָסוּר לְכַבֵּס בַּיָּמִים הָאֵלֶּה – גַּם לֹא עַל יְדֵי כּוֹבֵס גּוֹי.
בְּתִשְׁעַת הַיָּמִים רוֹצֶה אַבָּא לַעֲשׂוֹת סִיּוּם מַסֶּכֶת עִם בָּשָׂר. מָתַי זֶה נָכוֹן?
💡 סִיּוּם מַסֶּכֶת הוּא סְעוּדַת מִצְוָה, וּמֻתָּר בָּהּ בָּשָׂר. אֲבָל אֵין מְתַכְנְנִים סִיּוּם בִּמְיֻחָד כְּדֵי לֶאֱכֹל בָּשָׂר – זֶה מְבַטֵּל בְּכַוָּנָה אֶת אֲבֵלוּת הַחֻרְבָּן.

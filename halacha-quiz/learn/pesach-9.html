🍷 הִלְכוֹת פֶּסַח · פֶּרֶק ט׳

🫘מָה הֵן קִטְנִיּוֹת?

קִטְנִיּוֹת הֵן גַּרְגְּרִים כְּמוֹ אֹרֶז, תִּירָס, חוּמוּס, שְׁעוּעִית, עֲדָשִׁים, אֲפוּנָה, סוֹיָה וְשֻׁמְשֻׁם. הֵן אֵינָן חָמֵץ מִן הַתּוֹרָה. גַּם מוּצָרִים שֶׁעֲשׂוּיִים מֵהֶן – כְּמוֹ קוֹרְנְפְלֶקְס וּפְרִיכִיּוֹת אֹרֶז – כְּלוּלִים בְּמִנְהַג הַקִּטְנִיּוֹת.
לְמִנְהַג הָאַשְׁכְּנַזִּים – לֹא אוֹכְלִים קִטְנִיּוֹת בְּפֶסַח. לְמִנְהַג רֹב הַסְּפָרַדִּים – אוֹכְלִים, וּבוֹרְרִים אוֹתָן הֵיטֵב שֶׁלֹּא יִתְעָרְבוּ בָּהֶן גַּרְגְּרֵי דָּגָן.
💛 אַהֲבַת הַשֵּׁם: כָּל מִשְׁפָּחָה מַמְשִׁיכָה בְּמִנְהַג אֲבוֹתֶיהָ. שְׁנֵי הַמִּנְהָגִים יוֹצְאִים מֵאוֹתָהּ אַהֲבָה לְמִצְווֹת הַפֶּסַח.
📖 פֶּרֶק ט׳, הֲלָכוֹת א׳–ב׳, ד׳

🌾לָמָּה הָאַשְׁכְּנַזִּים לֹא אוֹכְלִים קִטְנִיּוֹת?

לִפְנֵי כִּשְׁבַע מֵאוֹת שָׁנָה הִתְחִילוּ בְּאַשְׁכְּנַז לְהַחְמִיר. הַטְּעָמִים: מְבַשְּׁלִים קִטְנִיּוֹת כְּמוֹ דָּגָן וְעוֹשִׂים מֵהֶן קֶמַח, וְחָשְׁשׁוּ שֶׁיִּטְעוּ בֵּינֵיהֶם; וְגַרְגְּרֵי חִטָּה וּשְׂעוֹרָה מִתְעָרְבִים לִפְעָמִים בַּקִּטְנִיּוֹת – בַּשָּׂדוֹת וּבַמַּחְסָנִים – וְיַחְמִיצוּ בַּבִּשּׁוּל.
רֹב הַסְּפָרַדִּים אוֹכְלִים קִטְנִיּוֹת וְאֹרֶז, וּבוֹרְרִים הֵיטֵב. יֵשׁ קְהִלּוֹת סְפָרַדִּיּוֹת שֶׁנָּהֲגוּ לֹא לֶאֱכֹל אֹרֶז – כְּמוֹ בְּאִיזְמִיר, וְרַבִּים בְּמָרוֹקוֹ וּבְבַגְדָּד. מִי שֶׁאוֹכֵל אֹרֶז – קוֹנֶה עִם הַשְׁגָּחָה לְפֶסַח, אוֹ בּוֹדֵק אוֹתוֹ הֵיטֵב שָׁלֹשׁ פְּעָמִים.
💛 אַהֲבַת הַשֵּׁם: כָּל מִשְׁפָּחָה מַמְשִׁיכָה בְּמִנְהַג אֲבוֹתֶיהָ. וּמִי שֶׁאֵינוֹ יוֹדֵעַ מָה לַעֲשׂוֹת – שׁוֹאֵל שְׁאֵלַת חָכָם.
📖 פֶּרֶק ט׳, הֲלָכוֹת א׳–ב׳

🏡מִשְׁפָּחָה עִם שְׁנֵי מִנְהָגִים

אִשָּׁה מִמִּשְׁפָּחָה שֶׁלֹּא אוֹכֶלֶת קִטְנִיּוֹת, שֶׁהִתְחַתְּנָה עִם בַּעַל מִמִּשְׁפָּחָה שֶׁאוֹכֶלֶת – נוֹהֶגֶת כְּמִנְהַג בַּעְלָהּ, וּמֻתָּר לָהּ לֶאֱכֹל קִטְנִיּוֹת בְּלִי הַתָּרַת נְדָרִים. כִּי אִי אֶפְשָׁר שֶׁבְּנֵי זוּג יֵשְׁבוּ תָּמִיד לְיַד שֻׁלְחָן אֶחָד, וּמָה שֶׁמֻּתָּר לָזֶה – אָסוּר לָזֶה.
כָּךְ גַּם מִי שֶׁעוֹבֵר לָגוּר לְתָמִיד בְּמָקוֹם שֶׁנּוֹהֲגִים בּוֹ אַחֶרֶת – נוֹהֵג כְּמִנְהַג הַמָּקוֹם הֶחָדָשׁ.
💛 אַהֲבַת הַשֵּׁם: הַבַּיִת הַיְּהוּדִי הוּא שֻׁלְחָן אֶחָד וְלֵב אֶחָד. הַשָּׁלוֹם בַּבַּיִת חָשׁוּב לַה׳ מְאוֹד.
📖 פֶּרֶק ט׳, הֲלָכָה ג׳

🥔מָה אֵינוֹ קִטְנִיּוֹת?

קֶמַח תַּפּוּחֵי אֲדָמָה מֻתָּר בְּפֶסַח: כְּשֶׁהִתְחִיל הַמִּנְהָג לֹא הָיוּ תַּפּוּחֵי אֲדָמָה בְּאֵירוֹפָּה, וְרַק מָה שֶׁגְּדוֹלֵי אַשְׁכְּנַז נָהֲגוּ לֶאֱסֹר נִכְלָל בַּמִּנְהָג. גַּם קִינוֹאָה מֻתֶּרֶת לַהֲלָכָה, אִם בּוֹרְרִים אוֹתָהּ הֵיטֵב, וְיֵשׁ מַחְמִירִים.
בְּבוֹטְנִים יֵשׁ מִנְהָגִים שׁוֹנִים: בִּירוּשָׁלַיִם וּבִמְקוֹמוֹת רַבִּים נִמְנְעוּ מֵהֶם, וּבְלִיטָא אָכְלוּ. מִי שֶׁאֵינוֹ יוֹדֵעַ מָה נָהֲגוּ בְּמִשְׁפַּחְתּוֹ – רַשַּׁאי לֶאֱכֹל.
💛 אַהֲבַת הַשֵּׁם: מִנְהָג שֶׁל עַם יִשְׂרָאֵל שׁוֹמְרִים כְּמוֹ שֶׁהוּא – לֹא מְקַטִּינִים אוֹתוֹ וְלֹא מוֹסִיפִים עָלָיו. כָּךְ מְכַבְּדִים אֶת חַכְמֵי הַדּוֹרוֹת.
📖 פֶּרֶק ט׳, הֲלָכָה ד׳

🐣קִטְנִיּוֹת בַּבַּיִת

גַּם מִי שֶׁאֵינוֹ אוֹכֵל קִטְנִיּוֹת רַשַּׁאי לְהַשְׁאִיר אוֹתָן בַּבַּיִת בְּפֶסַח, לְהַאֲכִיל בָּהֶן בַּעֲלֵי חַיִּים, וּלְבַשֵּׁל אוֹתָן לְמִי שֶׁאוֹכֵל קִטְנִיּוֹת. הַמִּנְהָג אֵינוֹ חָמוּר כְּמוֹ אִסּוּר חָמֵץ.
חוֹלֶה שֶׁצָּרִיךְ קִטְנִיּוֹת – מֻתָּר לוֹ לֶאֱכֹל אוֹתָן, גַּם כְּשֶׁאֵין סַכָּנָה. וְגַם לְתִינוֹקוֹת שֶׁצְּרִיכִים – מֻתָּר. בּוֹרְרִים אוֹתָן הֵיטֵב, וְרָאוּי לְהַקְצוֹת לָהֶם כֵּלִים מְיֻחָדִים.
💛 אַהֲבַת הַשֵּׁם: הַמִּנְהָג חָשׁוּב – וְהַדְּאָגָה לַחַלָּשִׁים חֲשׁוּבָה עוֹד יוֹתֵר. כָּךְ אֲבוֹתֵינוּ שָׁמְרוּ עַל הַמִּנְהָג בְּאַהֲבָה וּבְרַחֲמִים.
📖 פֶּרֶק ט׳, הֲלָכוֹת ה׳, ז׳

🧠 בְּדֹק אֶת עַצְמְךָ

מָה אֶחָד הַטְּעָמִים לְמִנְהַג הָאַשְׁכְּנַזִּים שֶׁלֹּא לֶאֱכֹל קִטְנִיּוֹת?
💡 בַּשָּׂדוֹת וּבַמַּחְסָנִים מִתְעָרְבִים בַּקִּטְנִיּוֹת גַּרְגְּרֵי חִטָּה, וְהֵם עֲלוּלִים לְהַחְמִיץ בַּבִּשּׁוּל. קִטְנִיּוֹת עַצְמָן אֵינָן חָמֵץ מִן הַתּוֹרָה, וּבִימֵי הַתַּלְמוּד אָכְלוּ אֹרֶז בְּפֶסַח.
מִשְׁפָּחָה סְפָרַדִּית אוֹכֶלֶת אֹרֶז בְּפֶסַח. עַל מָה עָלֶיהָ לְהַקְפִּיד?
💡 הָאֹרֶז נֶאֱרָז הַיּוֹם בְּמָקוֹם שֶׁאוֹרְזִים קְמָחִים וְסֹלֶת, וְלָכֵן הָאוֹכְלִים אֹרֶז קוֹנִים אוֹתוֹ בְּהַשְׁגָּחָה מְיֻחֶדֶת לְפֶסַח, אוֹ בּוֹדְקִים אוֹתוֹ הֵיטֵב שָׁלֹשׁ פְּעָמִים.
אִמָּא בָּאָה מִמִּשְׁפָּחָה שֶׁלֹּא אוֹכֶלֶת קִטְנִיּוֹת, וְאַבָּא מִמִּשְׁפָּחָה שֶׁאוֹכֶלֶת. מָה הַדִּין?
💡 אִשָּׁה נוֹהֶגֶת כְּמִנְהֲגֵי בַּעְלָהּ, כְּמוֹ מִי שֶׁעָבַר לָגוּר לְתָמִיד בְּמָקוֹם חָדָשׁ. כָּךְ כָּל הַמִּשְׁפָּחָה יוֹשֶׁבֶת סְבִיב שֻׁלְחָן אֶחָד, וְאֵין צֹרֶךְ בְּהַתָּרַת נְדָרִים.

🎭 הִלְכוֹת פּוּרִים · פֶּרֶק י״ד

😊מִשֶּׁנִּכְנַס אֲדָר מַרְבִּים בְּשִׂמְחָה

אָמְרוּ חֲכָמִים: "מִשֶּׁנִּכְנַס אָב מְמַעֲטִין בְּשִׂמְחָה, מִשֶּׁנִּכְנַס אֲדָר מַרְבִּים בְּשִׂמְחָה". אֵין מִקְרֶה בָּעוֹלָם, וּלְכָל זְמַן יֵשׁ אֹפִי מִשֶּׁלּוֹ. נֵס פּוּרִים קָרָה בְּאֶמְצַע אֲדָר – סִימָן שֶׁיֵּשׁ לַחֹדֶשׁ הַזֶּה סְגֻלָּה לַהֲפֹךְ אֶת הָרַע לְטוֹב.
שִׂמְחָה רְגִילָה הִיא עַל הַטּוֹב. אֲבָל כְּשֶׁגַּם הָרַע מִתְהַפֵּךְ לְטוֹב – הַשִּׂמְחָה גְּדוֹלָה וּשְׁלֵמָה. כָּךְ הָיָה בְּפוּרִים.
💛 אַהֲבַת הַשֵּׁם: ה׳ מַנְהִיג אֶת הָעוֹלָם, וְכָל מָה שֶׁקּוֹרֶה – אֲפִלּוּ הָרַע – יִתְהַפֵּךְ בַּסּוֹף לְטוֹבָה. כְּכָל שֶׁנַּרְבֶּה בֶּאֱמוּנָה וּבְתוֹרָה, נְקָרֵב אֶת הַגְּאֻלָּה וְתִרְבֶּה הַשִּׂמְחָה בָּעוֹלָם.
📖 פֶּרֶק י״ד, הֲלָכָה א׳

📜אַרְבַּע פָּרָשִׁיּוֹת

חֲכָמִים תִּקְּנוּ לִקְרֹא אַרְבַּע פָּרָשִׁיּוֹת מְיֻחָדוֹת: שְׁקָלִים, זָכוֹר, פָּרָה וְהַחֹדֶשׁ. קוֹרְאִים אוֹתָן בְּשַׁבְּתוֹת חֹדֶשׁ אֲדָר, וְאֶת הָרִאשׁוֹנָה – בְּדֶרֶךְ כְּלָל בַּשַּׁבָּת שֶׁלִּפְנֵי אֲדָר. בְּכָל שַׁבָּת כָּזוֹ מוֹצִיאִים שְׁנֵי סִפְרֵי תוֹרָה: בָּרִאשׁוֹן קוֹרְאִים אֶת פָּרָשַׁת הַשָּׁבוּעַ, וּבַשֵּׁנִי – אֶת הַפָּרָשָׁה הַמְּיֻחֶדֶת לַמַּפְטִיר.
אֶת פָּרָשַׁת זָכוֹר קוֹרְאִים בַּשַּׁבָּת שֶׁלִּפְנֵי פּוּרִים, כְּדֵי לִסְמֹךְ אֶת זְכִירַת עֲמָלֵק לְפוּרִים – שֶׁבּוֹ נִמְחָה הָמָן, שֶׁהָיָה מִזֶּרַע עֲמָלֵק.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁאֵין בֵּית מִקְדָּשׁ, אֲנַחְנוּ קוֹרְאִים עָלָיו וּמִתְגַּעְגְּעִים אֵלָיו. כָּל קְרִיאָה כָּזוֹ הִיא תְּפִלָּה קְטַנָּה שֶׁנִּזְכֶּה לִבְנוֹתוֹ בִּמְהֵרָה.
📖 פֶּרֶק י״ד, הֲלָכָה ב׳

📚לָמָּה אַרְבַּע פָּרָשִׁיּוֹת?

פָּרָשַׁת שְׁקָלִים מַזְכִּירָה לָתֵת מַחֲצִית הַשֶּׁקֶל, כִּי מִנִּיסָן קָנוּ אֶת קָרְבְּנוֹת הַצִּבּוּר מִן הַתְּרוּמָה הַחֲדָשָׁה. פָּרָשַׁת זָכוֹר – לִזְכֹּר אֶת עֲמָלֵק לִפְנֵי פּוּרִים. פָּרָשַׁת פָּרָה – לְהִטַּהֵר לִקְרַאת קָרְבַּן הַפֶּסַח. פָּרָשַׁת הַחֹדֶשׁ – עַל קִדּוּשׁ הַחֹדֶשׁ וּמִצְווֹת הַפֶּסַח, כִּי נִיסָן רִאשׁוֹן לֶחֳדָשִׁים.
כְּשֶׁרֹאשׁ חֹדֶשׁ אֲדָר אוֹ נִיסָן חָל בְּשַׁבָּת – מוֹצִיאִים שְׁלוֹשָׁה סִפְרֵי תוֹרָה: פָּרָשַׁת הַשָּׁבוּעַ, רֹאשׁ חֹדֶשׁ, וְהַפָּרָשָׁה הַמְּיֻחֶדֶת. לְרֹב הַפּוֹסְקִים קְרִיאַת זָכוֹר הִיא מִן הַתּוֹרָה, וְלָכֵן מְדַקְדְּקִים בָּהּ יוֹתֵר מִכֻּלָּן.
💛 אַהֲבַת הַשֵּׁם: הַפָּרָשִׁיּוֹת מְכִינוֹת אוֹתָנוּ צַעַד אַחַר צַעַד: מִזִּכָּרוֹן וּמִלְחָמָה בָּרַע – אֶל טָהֳרָה וּגְאֻלָּה. כָּךְ ה׳ מוֹבִיל אוֹתָנוּ בְּאַהֲבָה מִפּוּרִים לְפֶסַח.
📖 פֶּרֶק י״ד, הֲלָכָה ב׳

🗓️שָׁנָה מְעֻבֶּרֶת וּפוּרִים קָטָן

הַחֳדָשִׁים נִקְבָּעִים לְפִי הַלְּבָנָה, וְהַשָּׁנִים לְפִי הַחַמָּה – כִּי פֶּסַח צָרִיךְ לִהְיוֹת תָּמִיד בָּאָבִיב. לָכֵן לִפְעָמִים מוֹסִיפִים חֹדֶשׁ אֲדָר שֵׁנִי. בְּשָׁנָה מְעֻבֶּרֶת חוֹגְגִים פּוּרִים וְקוֹרְאִים אֶת אַרְבַּע הַפָּרָשִׁיּוֹת בַּאֲדָר הַשֵּׁנִי – כְּדֵי לִסְמֹךְ אֶת גְּאֻלַּת פּוּרִים לִגְאֻלַּת מִצְרַיִם.
גַּם בַּאֲדָר הָרִאשׁוֹן יֵשׁ חֲגִיגִיּוּת: בְּי״ד וּבְט״ו בּוֹ אֵין מִתְעַנִּים, אֵין מַסְפִּידִים וְאֵין נוֹפְלִים אַפַּיִם. וְנָכוֹן לְהַרְבּוֹת בִּסְעוּדָה בְּי״ד בַּאֲדָר הָרִאשׁוֹן – "פּוּרִים קָטָן".
💛 אַהֲבַת הַשֵּׁם: הַשִּׂמְחָה מַתְחִילָה כְּבָר מֵרֹאשׁ חֹדֶשׁ אֲדָר הָרִאשׁוֹן. בְּשָׁנָה כָּזוֹ זוֹכִים לִשְׁנֵי חֳדָשִׁים שֶׁל שִׂמְחָה – מַתָּנָה מֵה׳.
📖 פֶּרֶק י״ד, הֲלָכָה ג׳

⚔️מָה עָשָׂה לָנוּ עֲמָלֵק

מִיָּד אַחֲרֵי יְצִיאַת מִצְרַיִם, בְּלִי שׁוּם סִבָּה, בָּא עֲמָלֵק וְתָקַף אֶת הָעֲבָדִים שֶׁיָּצְאוּ לְחֵרוּת. עֲמָלֵק שׂוֹנֵא אֶת יִשְׂרָאֵל, אֶת הַתּוֹרָה וְאֶת דֶּרֶךְ הַחֶסֶד וְהָאֱמֶת. שָׁלֹשׁ מִצְווֹת בַּתּוֹרָה עוֹסְקוֹת בּוֹ: לִזְכֹּר אֶת מַעֲשָׂיו, לֹא לִשְׁכֹּחַ אוֹתָם, וְלִמְחוֹת אֶת זִכְרוֹ.
יְהוּדִי רַחֲמָן מִטִּבְעוֹ, וְהָיָה נוֹטֶה לִסְלֹחַ. לָכֵן הַתּוֹרָה מְצַוָּה לִזְכֹּר: יֵשׁ רֹעַ בָּעוֹלָם, וְצָרִיךְ לְהִלָּחֵם בּוֹ.
💛 אַהֲבַת הַשֵּׁם: "כִּי יָד עַל כֵּס יָ־הּ" – ה׳ נִשְׁבַּע שֶׁאֵין שְׁמוֹ שָׁלֵם וְאֵין כִּסְאוֹ שָׁלֵם עַד שֶׁיִּמָּחֶה עֲמָלֵק. כְּשֶׁאָנוּ זוֹכְרִים, אָנוּ עוֹמְדִים לְצַד ה׳ וּמְחַכִּים לַיּוֹם שֶׁהַטּוֹב יְנַצֵּחַ.
📖 פֶּרֶק י״ד, הֲלָכָה ד׳

⚔️זָכוֹר וְלֹא תִּשְׁכָּח

"זָכוֹר אֵת אֲשֶׁר עָשָׂה לְךָ עֲמָלֵק" – מִצְוַת עֲשֵׂה. "לֹא תִּשְׁכָּח" – מִצְוַת לֹא תַּעֲשֶׂה. "תִּמְחֶה אֶת זֵכֶר עֲמָלֵק" – מִצְוַת עֲשֵׂה. הָמָן הָאֲגָגִי, מִזַּרְעוֹ שֶׁל אֲגַג מֶלֶךְ עֲמָלֵק, רָצָה לְהַשְׁמִיד אֶת כָּל יִשְׂרָאֵל – וְה׳ הִצִּילָנוּ.
אַחֲרֵי הַנֵּס הִתְעוֹרְרוּ בְּיִשְׂרָאֵל הַכֹּחוֹת לַעֲלוֹת לְאֶרֶץ יִשְׂרָאֵל, לְיַשְּׁבָהּ וְלִבְנוֹת אֶת בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי.
💛 אַהֲבַת הַשֵּׁם: הַזִּכָּרוֹן אֵינוֹ רַק עַל הָרַע – הוּא גַּם עַל הַהַצָּלָה. כְּשֶׁאָנוּ זוֹכְרִים אֶת עֲמָלֵק, אָנוּ זוֹכְרִים גַּם כַּמָּה ה׳ אוֹהֵב אוֹתָנוּ וְשׁוֹמֵר עָלֵינוּ בְּכָל דּוֹר.
📖 פֶּרֶק י״ד, הֲלָכוֹת ד׳–ה׳; פֶּרֶק ט״ו, הֲלָכָה א׳

👑שָׁאוּל וַאֲגַג

שְׁמוּאֵל הַנָּבִיא צִוָּה אֶת שָׁאוּל הַמֶּלֶךְ לְהִלָּחֵם בַּעֲמָלֵק. אֲבָל שָׁאוּל חָמַל עַל אֲגַג מֶלֶךְ עֲמָלֵק וְעַל מֵיטַב הַצֹּאן, וְלָכֵן הֶעֱבִיר ה׳ מִמֶּנּוּ אֶת הַמַּלְכוּת וּנְתָנָהּ לְדָוִד.
סִפְּרוּ חֲזַ״ל: בַּזְּמַן שֶׁשָּׁאוּל הִתְמַהְמֵהַּ בַּהֲרִיגַת אֲגַג, נִמְשַׁךְ זַרְעוֹ – וּמִמֶּנּוּ נוֹלַד לְבַסּוֹף הָמָן הָאֲגָגִי.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁה׳ מְצַוֶּה – הוּא יוֹדֵעַ מָה טוֹב לָנוּ יוֹתֵר מִמֶּנּוּ. גַּם כְּשֶׁנִּדְמֶה לָנוּ שֶׁאָנוּ יוֹדְעִים טוֹב יוֹתֵר, אָנוּ סוֹמְכִים עָלָיו בְּאַהֲבָה.
📖 פֶּרֶק י״ד, הֲלָכָה ה׳

👂פָּרָשַׁת זָכוֹר

פַּעַם אַחַת בְּשָׁנָה קוֹרְאִים אֶת פָּרָשַׁת זָכוֹר, וּבָזֶה מְקַיְּמִים אֶת מִצְוַת הַתּוֹרָה לִזְכֹּר אֶת עֲמָלֵק. לֹא צָרִיךְ שֶׁכָּל אֶחָד יִקְרָא בְּעַצְמוֹ: הַשּׁוֹמֵעַ מִן הַקּוֹרֵא בַּתּוֹרָה יוֹצֵא יְדֵי חוֹבָה. צָרִיךְ לְכַוֵּן לְקַיֵּם אֶת הַמִּצְוָה, וְטוֹב שֶׁהַגַּבַּאי יַכְרִיז עַל כָּךְ לִפְנֵי הַקְּרִיאָה.
מִי שֶׁנִּמְצָא בְּמָקוֹם שֶׁאֵין בּוֹ מִנְיָן – יִקְרָא בְּסֵפֶר תּוֹרָה אִם יֵשׁ. וְאִם אֵין סֵפֶר תּוֹרָה – יִקְרָא אֶת הַפָּרָשָׁה מִתּוֹךְ חֻמָּשׁ אוֹ סִדּוּר.
💛 אַהֲבַת הַשֵּׁם: לְכַתְּחִלָּה כָּל אֶחָד שׁוֹמֵעַ בַּמִּבְטָא שֶׁל מִשְׁפַּחְתּוֹ, אֲבָל כָּל הָעֵדוֹת – סְפָרַדִּים, אַשְׁכְּנַזִּים וְתֵימָנִים – יוֹצְאוֹת בְּכָל הַנֻּסְחָאוֹת. כֻּלָּנוּ עַם אֶחָד שֶׁזּוֹכֵר יַחַד.
📖 פֶּרֶק י״ד, הֲלָכָה ו׳

📖נָשִׁים וּבָנוֹת וּפָרָשַׁת זָכוֹר

לְדַעַת רֹב הַפּוֹסְקִים נָשִׁים פְּטוּרוֹת מִזְּכִירַת עֲמָלֵק, כִּי הַזְּכִירָה קְשׁוּרָה לַמִּלְחָמָה בּוֹ, וְנָשִׁים אֵינָן מְצֻוּוֹת לְהִלָּחֵם. וְיֵשׁ אוֹמְרִים שֶׁגַּם הֵן חַיָּבוֹת, כִּי הֵן מְסַיְּעוֹת לַלּוֹחֲמִים. לְמַעֲשֶׂה נָשִׁים פְּטוּרוֹת, אֲבָל לְכַתְּחִלָּה טוֹב שֶׁיִּשְׁמְעוּ אֶת פָּרָשַׁת זָכוֹר – וְכָךְ נוֹהֲגוֹת נָשִׁים רַבּוֹת.
מִי שֶׁקָּשֶׁה לָהּ לָבוֹא לְבֵית הַכְּנֶסֶת וְרוֹצָה לְקַיֵּם אֶת הַמִּצְוָה – תִּקְרָא בְּעַצְמָהּ אֶת הַפָּרָשָׁה מִתּוֹךְ חֻמָּשׁ. וְיֵשׁ מְקוֹמוֹת שֶׁקּוֹרְאִים לַנָּשִׁים מִסֵּפֶר תּוֹרָה, גַּם בְּלִי מִנְיָן.
💛 אַהֲבַת הַשֵּׁם: גַּם מִי שֶׁפְּטוּרָה מִמִּצְוָה יְכוֹלָה לְקַיֵּם אוֹתָהּ מֵרָצוֹן וּמֵאַהֲבָה. כָּךְ כָּל בַּת יִשְׂרָאֵל שׁוֹמֶרֶת עַל הַזִּכָּרוֹן שֶׁל עַמָּהּ.

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

כַּמָּה נוֹתְנִים זֵכֶר לְמַחֲצִית הַשֶּׁקֶל? יֵשׁ נוֹהֲגִים לָתֵת חֲצִי מִן הַמַּטְבֵּעַ הַמְּקֻבָּל – הַיּוֹם חֲצִי שֶׁקֶל. וְיֵשׁ נוֹהֲגִים לָתֵת שְׁלוֹשָׁה חֲצָאִים, כְּנֶגֶד שָׁלֹשׁ פְּעָמִים "תְּרוּמָה" (רַמָּ״א). וְיֵשׁ נוֹהֲגִים לָתֵת אֶת שְׁוִי מַחֲצִית הַשֶּׁקֶל בְּכֶסֶף טָהוֹר. כָּל הַמִּנְהָגִים כְּשֵׁרִים.
וּבְפָרָשַׁת זָכוֹר: לְדַעַת רֹב הַפּוֹסְקִים נָשִׁים פְּטוּרוֹת, וְיֵשׁ אוֹמְרִים שֶׁחַיָּבוֹת. לָכֵן לְכַתְּחִלָּה טוֹב שֶׁיִּשְׁמְעוּ, וְכָךְ נוֹהֲגוֹת נָשִׁים רַבּוֹת.
💛 אַהֲבַת הַשֵּׁם: "הַמּוֹסִיף בִּצְדָקָה – תָּבוֹא עָלָיו בְּרָכָה". כָּל אֶחָד שׁוֹמֵר עַל מִנְהַג מִשְׁפַּחְתּוֹ בְּאַהֲבָה, וְכֻלָּנוּ נוֹתְנִים לְאוֹתוֹ ה׳.

🕯️תַּעֲנִית אֶסְתֵּר

מִימֵי הַגְּאוֹנִים נוֹהֲגִים כָּל יִשְׂרָאֵל לְהִתְעַנּוֹת בְּי״ג בַּאֲדָר – זֵכֶר לַצּוֹם שֶׁל אֶסְתֵּר לִפְנֵי שֶׁנִּכְנְסָה אֶל הַמֶּלֶךְ, וְזֵכֶר לַתַּעֲנִית שֶׁל יִשְׂרָאֵל בְּי״ג בַּאֲדָר בְּאוֹתָהּ שָׁנָה, בַּיּוֹם שֶׁבּוֹ נִלְחֲמוּ בְּשׂוֹנְאֵיהֶם.
הַיְּהוּדִים שֶׁלֹּא הָיוּ צְרִיכִים לְהִלָּחֵם – חָזְרוּ בִּתְשׁוּבָה וְצָמוּ, כִּי בְּעֵת צָרָה מִתְעוֹרְרִים לִתְשׁוּבָה.
💛 אַהֲבַת הַשֵּׁם: גַּם הַיּוֹם יֵשׁ שׂוֹנְאִים שֶׁרוֹצִים לְהַשְׁמִידֵנוּ, וְאָנוּ זְקוּקִים לְרַחֲמֵי שָׁמַיִם. הַתַּעֲנִית וְהַתְּשׁוּבָה מְקָרְבוֹת אוֹתָנוּ אֶל אָבִינוּ שֶׁבַּשָּׁמַיִם.
📖 פֶּרֶק י״ד, הֲלָכָה ט׳

🕯️דִּינֵי תַּעֲנִית אֶסְתֵּר

הַצּוֹמוֹת הַקַּלִּים נִתְקְנוּ עַל יְדֵי חֲכָמִים, וְתַעֲנִית אֶסְתֵּר נִקְבְּעָה עַל פִּי מִנְהָגָם שֶׁל יִשְׂרָאֵל. לָכֵן דִּינָהּ קַל מִשְּׁאָר הַצּוֹמוֹת הַקַּלִּים, אֲבָל לְמַעֲשֶׂה כִּמְעַט אֵין הֶבְדֵּל בֵּינֵיהֶם. בְּמִנְחָה שֶׁל תַּעֲנִית אֶסְתֵּר אֵין נוֹפְלִים אַפַּיִם, כִּי זֶה עֶרֶב פּוּרִים.
כְּשֶׁי״ג בַּאֲדָר חָל בְּשַׁבָּת – מַקְדִּימִים אֶת הַתַּעֲנִית לְיוֹם חֲמִישִׁי. אָז הִיא אֵינָהּ עֶרֶב פּוּרִים, וּמִתְפַּלְּלִים מִנְחָה כְּמוֹ בְּכָל הַצּוֹמוֹת.
💛 אַהֲבַת הַשֵּׁם: אֵין תְּשׁוּבָה גְּדוֹלָה יוֹתֵר מִן הַתְּשׁוּבָה שֶׁעַל יְדֵי תַּעֲנִית – הִיא מְזַכֶּכֶת אֶת הַגּוּף וּמַחֲזִירָה אֶת הָרוּחַ לַמָּקוֹם הַמֶּרְכָּזִי. כָּךְ אָנוּ נִכְנָסִים לְפוּרִים קְרוֹבִים יוֹתֵר לַה׳.

🪙זֵכֶר לְמַחֲצִית הַשֶּׁקֶל

בִּזְמַן הַמִּקְדָּשׁ נָתַן כָּל אֶחָד מַחֲצִית הַשֶּׁקֶל לְקָרְבְּנוֹת הַצִּבּוּר. הַיּוֹם נוֹהֲגִים לָתֵת צְדָקָה בַּאֲדָר "זֵכֶר לְמַחֲצִית הַשֶּׁקֶל". הַזְּמַן הַטּוֹב – לִפְנֵי תְּפִלַּת מִנְחָה שֶׁל תַּעֲנִית אֶסְתֵּר, כְּדֵי שֶׁהַצְּדָקָה תִּצְטָרֵף לַתַּעֲנִית.
הַיּוֹם נוֹהֲגִים לָתֵת לְפָחוֹת חֲצִי שֶׁקֶל עֲבוּר כָּל אֶחָד מִבְּנֵי הַבַּיִת – גְּבָרִים וְנָשִׁים, וְגַם עֲבוּר הַיְּלָדִים הַקְּטַנִּים.
💛 אַהֲבַת הַשֵּׁם: כָּל אֶחָד נָתַן אוֹתוֹ סְכוּם, הֶעָשִׁיר וְהֶעָנִי – וְכָךְ הָיָה לְכֻלָּם חֵלֶק שָׁוֶה בַּקָּרְבָּנוֹת. לִפְנֵי ה׳ כֻּלָּנוּ חֲשׁוּבִים וְשׁוֹתְפִים.
📖 פֶּרֶק י״ד, הֲלָכָה י׳

🪙מִנְהֲגֵי מַחֲצִית הַשֶּׁקֶל

יֵשׁ נוֹהֲגִים לָתֵת חֲצִי מִן הַמַּטְבֵּעַ הַמְּקֻבָּל בַּמָּקוֹם. יֵשׁ נוֹהֲגִים לָתֵת שְׁלוֹשָׁה חֲצָאִים – כְּנֶגֶד שָׁלֹשׁ פְּעָמִים שֶׁנֶּאֱמַר "תְּרוּמָה" בְּפָרָשַׁת שְׁקָלִים. וְיֵשׁ נוֹהֲגִים לָתֵת אֶת שְׁוִי מַחֲצִית הַשֶּׁקֶל – כֶּסֶף טָהוֹר בְּמִשְׁקַל קָרוֹב לְעֶשֶׂר גְּרָם. כָּל הַמִּנְהָגִים כְּשֵׁרִים.
אֵין לָתֵת צְדָקָה זוֹ מִכַּסְפֵּי מַעֲשֵׂר, כִּי אָסוּר לְקַיֵּם מִצְווֹת וּמִנְהָגִים שֶׁאָדָם חַיָּב בָּהֶם מִכֶּסֶף שֶׁשַּׁיָּךְ לַצְּדָקָה.
💛 אַהֲבַת הַשֵּׁם: "הַמּוֹסִיף בִּצְדָקָה – תָּבוֹא עָלָיו בְּרָכָה". כְּשֶׁאָנוּ נוֹתְנִים בְּלֵב טוֹב, אָנוּ מִתְדַּמִּים לַה׳ שֶׁנּוֹתֵן לְכֻלָּם בְּחֶסֶד.

🧠 בְּדֹק אֶת עַצְמְךָ

לָמָּה קוֹרְאִים אֶת פָּרָשַׁת שְׁקָלִים כְּחֹדֶשׁ לִפְנֵי נִיסָן?
💡 מִנִּיסָן קָנוּ אֶת קָרְבְּנוֹת הַצִּבּוּר מִן הַתְּרוּמָה הַחֲדָשָׁה, וְלָכֵן הִזְכִּירוּ חֹדֶשׁ לִפְנֵי כֵן לְהָבִיא אֶת מַחֲצִית הַשֶּׁקֶל. טָהֳרָה – זוֹ פָּרָה, עֲמָלֵק – זָכוֹר, וְקִדּוּשׁ הַחֹדֶשׁ – הַחֹדֶשׁ.
רֹאשׁ חֹדֶשׁ אֲדָר חָל בְּשַׁבָּת. כַּמָּה סִפְרֵי תוֹרָה מוֹצִיאִים, וּמָה קוֹרְאִים בָּהֶם?
💡 בָּרִאשׁוֹן – פָּרָשַׁת הַשָּׁבוּעַ, בַּשֵּׁנִי – עִנְיַן רֹאשׁ חֹדֶשׁ, וּבַשְּׁלִישִׁי – פָּרָשַׁת שְׁקָלִים. כָּךְ גַּם בְּרֹאשׁ חֹדֶשׁ נִיסָן, עִם פָּרָשַׁת הַחֹדֶשׁ.
בְּשָׁנָה מְעֻבֶּרֶת יֵשׁ שְׁנֵי חָדְשֵׁי אֲדָר. מָתַי חוֹגְגִים אֶת פּוּרִים?
💡 פּוּרִים בַּאֲדָר הַשֵּׁנִי, כְּדֵי לִסְמֹךְ גְּאֻלָּה לִגְאֻלָּה. וְהַפּוּרִים הַקָּטָן הוּא דַּוְקָא בְּי״ד בַּאֲדָר הָרִאשׁוֹן.

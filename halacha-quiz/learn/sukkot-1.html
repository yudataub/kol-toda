חַג הַסֻּכּוֹת

🌿 הִלְכוֹת סֻכּוֹת · פֶּרֶק א׳

🌾שְׁלֹשָׁה שֵׁמוֹת לֶחָג

לֶחָג שְׁלֹשָׁה שֵׁמוֹת: חַג הַסֻּכּוֹת – עַל שֵׁם מִצְוַת הַסֻּכָּה; חַג הָאָסִיף – כִּי בּוֹ מְסַיְּמִים לֶאֱסֹף אֶת הַתְּבוּאָה וְהַפֵּרוֹת מִן הַשָּׂדוֹת; וְ״הֶחָג״ – כִּי הוּא הַשָּׂמֵחַ וְהֶחָגִיגִי שֶׁבַּחַגִּים.
בַּתּוֹרָה נִכְתְּבָה בְּחַג הַסֻּכּוֹת שִׂמְחָה שָׁלֹשׁ פְּעָמִים. אַחֲרֵי שֶׁאָסַפְנוּ אֶת כָּל הַיְּבוּל, וְאַחֲרֵי הַתְּשׁוּבָה וְהַכַּפָּרָה שֶׁל הַיָּמִים הַנּוֹרָאִים – הַשִּׂמְחָה שְׁלֵמָה.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁהַמַּחְסָנִים מְלֵאִים, הַחַג מְלַמֵּד אוֹתָנוּ לִשְׂמֹחַ וּלְהוֹדוֹת לַה׳ עַל כָּל הַטּוֹב שֶׁנָּתַן לָנוּ בְּמֶשֶׁךְ הַשָּׁנָה.
📖 פֶּרֶק א׳, הֲלָכוֹת א׳–ב׳, ח׳

🌾🔑 מוּשָּׂג: חַג הָאָסִיף

״חַג הַסֻּכּוֹת נִקְרָא גַּם ׳חַג הָאָסִיף׳... בָּעוֹנָה שֶׁבָּהּ מְסַיְּמִים לֶאֱסֹף אֶת הַתְּבוּאָה וְהַפֵּרוֹת״.
📖 אָסִיף – מִלְּשׁוֹן ״לֶאֱסֹף״: הַזְּמַן שֶׁבּוֹ מַכְנִיסִים אֶת כָּל יְבוּל הַשָּׂדֶה הַבַּיְתָה.
💛 אַהֲבַת הַשֵּׁם: חַג הָאָסִיף הוּא חֲגִיגַת הַתּוֹדָה עַל כֹּל מַה שֶּׁה׳ נָתַן לָנוּ בַּשָּׁנָה שֶׁחָלְפָה. 🧒 וְאַתָּה? בֶּחָג תַּגִּיד תּוֹדָה עַל דָּבָר טוֹב אֶחָד שֶׁקָּרָה לְךָ הַשָּׁנָה.
📖 פֶּרֶק א׳, הֲלָכָה ב׳

🏆דֶּגֶל שֶׁל נִצָּחוֹן

מָשָׁל: שְׁנַיִם נִכְנְסוּ לְמִשְׁפָּט, וְלֹא יָדְעוּ מִי נִצַּח – עַד שֶׁאֶחָד יָצָא וְהֵנִיף אֶת רָמְחוֹ. כָּךְ יִשְׂרָאֵל: אַחֲרֵי יְמֵי הַדִּין, כְּשֶׁהֵם יוֹצְאִים וְלוּלָבֵיהֶם וְאֶתְרוֹגֵיהֶם בִּידֵיהֶם – יוֹדְעִים שֶׁהֵם נִצְּחוּ בַּדִּין.
וְהַנִּעְנוּעִים לְאַרְבַּע רוּחוֹת הֵם גַּם תְּפִלָּה: שֶׁה׳ יַעֲצֹר רוּחוֹת רָעוֹת וּטְלָלִים רָעִים, וִיקַבֵּץ אֶת יִשְׂרָאֵל מֵאַרְבַּע כַּנְפוֹת הָאָרֶץ.
💛 אַהֲבַת הַשֵּׁם: אַחֲרֵי הַתְּשׁוּבָה שֶׁל הַיָּמִים הַנּוֹרָאִים, אָנוּ בָּאִים לֶחָג בְּשִׂמְחָה – כְּבָנִים אֲהוּבִים שֶׁאֲבִיהֶם סָלַח לָהֶם.
📖 פֶּרֶק א׳, הֲלָכָה ג׳; פֶּרֶק ה׳, הֲלָכוֹת ד׳–ה׳

☁️זֵכֶר לְעַנְנֵי הַכָּבוֹד

״לְמַעַן יֵדְעוּ דֹרֹתֵיכֶם כִּי בַסֻּכּוֹת הוֹשַׁבְתִּי אֶת בְּנֵי יִשְׂרָאֵל״. לְדַעַת רַבִּי אֱלִיעֶזֶר – הַסֻּכָּה זֵכֶר לְעַנְנֵי הַכָּבוֹד שֶׁהֵגֵנּוּ עַל יִשְׂרָאֵל בַּמִּדְבָּר. לְדַעַת רַבִּי עֲקִיבָא – זֵכֶר לַסֻּכּוֹת מַמָּשׁ שֶׁעָשׂוּ. לַהֲלָכָה זוֹכְרִים בַּסֻּכָּה אֶת עַנְנֵי הַכָּבוֹד וְאֶת יְצִיאַת מִצְרַיִם.
לָמָּה יוֹשְׁבִים בַּסֻּכָּה בַּסְּתָו, וְלֹא בָּאָבִיב שֶׁבּוֹ יָצָאנוּ מִמִּצְרַיִם? כִּי בָּאָבִיב רַבִּים בּוֹנִים סֻכּוֹת לַהֲנָאָתָם. בַּסְּתָו רוֹאִים שֶׁיּוֹשְׁבִים בַּסֻּכָּה רַק בִּשְׁבִיל הַמִּצְוָה.
💛 אַהֲבַת הַשֵּׁם: בַּמִּדְבָּר ה׳ לֹא רַק נָתַן לָנוּ מָן וּמַיִם – הוּא פָּרַשׂ עָלֵינוּ אֶת עֲנָנָיו מִכָּל הַצְּדָדִים. כְּשֶׁאָנוּ יוֹשְׁבִים בַּסֻּכָּה, אָנוּ מַרְגִּישִׁים אֶת הָאַהֲבָה שֶׁלּוֹ סְבִיבֵנוּ.
📖 פֶּרֶק א׳, הֲלָכוֹת ד׳–ה׳

☁️🔑 מוּשָּׂג: עַנְנֵי הַכָּבוֹד

״לְמַעַן יֵדְעוּ דֹרֹתֵיכֶם כִּי בַסֻּכּוֹת הוֹשַׁבְתִּי אֶת בְּנֵי יִשְׂרָאֵל בְּהוֹצִיאִי אוֹתָם מֵאֶרֶץ מִצְרָיִם, אֲנִי ה׳ אֱלוֹהֵיכֶם״.
📖 עַנְנֵי הַכָּבוֹד – עֲנָנִים שֶׁל כְּבוֹד ה׳ שֶׁהֵגֵנּוּ עַל יִשְׂרָאֵל בַּמִּדְבָּר, מֵאַרְבַּע רוּחוֹת וּמִלְּמַעְלָה וּמִלְּמַטָּה.
💛 אַהֲבַת הַשֵּׁם: ה׳ לֹא רַק נָתַן לָנוּ מָן וּמַיִם בַּמִּדְבָּר – הוּא פָּרַשׂ עָלֵינוּ אֶת עֲנָנָיו מִכָּל צַד. 🧒 וְאַתָּה? כְּשֶׁאַתָּה בַּסֻּכָּה – תִּזְכֹּר שֶׁגַּם אַתָּה מֻקָּף בְּאַהֲבָתוֹ שֶׁל ה׳.
📖 פֶּרֶק א׳, הֲלָכָה ד׳

🛖דִּירַת אַרְעַי

הַסֻּכָּה הִיא דִּירַת אַרְעַי. גָּבְהָהּ לְפָחוֹת עֲשָׂרָה טְפָחִים (כ־80 ס״מ) וְרָחְבָּהּ לְפָחוֹת שִׁבְעָה טְפָחִים (כ־56 ס״מ), כְּדֵי שֶׁאָדָם יוּכַל לָשֶׁבֶת וְלֶאֱכֹל בָּהּ. וְאִם הִיא גְּבוֹהָה מֵעֶשְׂרִים אַמָּה (כ־9 מֶטֶר) – פְּסוּלָה, כִּי סְכָךְ גָּבוֹהַּ כָּל כָּךְ צָרִיךְ בִּנְיַן קֶבַע.
סֻכָּה כְּשֵׁרָה גַּם עַל עֲגָלָה, עַל מְכוֹנִית אוֹ עַל סְפִינָה – אִם הִיא עוֹמֶדֶת בְּרוּחַ רְגִילָה. וְהַסֻּכָּה פְּטוּרָה מִמְּזוּזָה, כִּי הִיא דִּירַת אַרְעַי.
💛 אַהֲבַת הַשֵּׁם: הַבַּיִת הֶחָזָק עָלוּל לְהַשְׁכִּיחַ שֶׁה׳ הוּא הַשּׁוֹמֵר עָלֵינוּ. בַּסֻּכָּה אָנוּ זוֹכְרִים: ״אִם ה׳ לֹא יִבְנֶה בַיִת – שָׁוְא עָמְלוּ בוֹנָיו בּוֹ״.
📖 פֶּרֶק א׳, הֲלָכָה ו׳; פֶּרֶק ב׳, הֲלָכָה א׳

🫂כָּל יִשְׂרָאֵל בְּסֻכָּה אַחַת

אָמְרוּ חֲכָמִים: ״רְאוּיִים כָּל יִשְׂרָאֵל לֵישֵׁב בְּסֻכָּה אַחַת״. בְּחַג הָאָסִיף נֶאֱסָפוֹת יַחַד כָּל הַמַּעֲלוֹת הַטּוֹבוֹת שֶׁל כָּל הַיְּהוּדִים, וְכָךְ מִתְגַּלָּה הָאַחְדוּת.
עוֹד אָמְרוּ חֲכָמִים: מִי שֶׁמְּקַיֵּם מִצְוַת סֻכָּה בָּעוֹלָם הַזֶּה – הַקָּדוֹשׁ בָּרוּךְ הוּא יָגֵן עָלָיו לֶעָתִיד לָבוֹא.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאָנוּ יוֹשְׁבִים בַּסֻּכָּה, ה׳ פּוֹרֵשׂ עָלֵינוּ אֶת סֻכַּת שְׁלוֹמוֹ – עַל כָּל יִשְׂרָאֵל יַחַד.
📖 פֶּרֶק א׳, הֲלָכָה ח׳

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

בְּסִדּוּרִים רַבִּים יֵשׁ נֻסָּח לְהַזְמָנַת הָאוּשְׁפִּיזִין, אוֹרֵחַ אֶחָד בְּכָל יוֹם. לְמִנְהַג הַסְּפָרַדִּים וְהַחֲסִידִים הַסֵּדֶר: אַבְרָהָם, יִצְחָק, יַעֲקֹב, מֹשֶׁה, אַהֲרֹן, יוֹסֵף וְדָוִד. לְמִנְהַג הָאַשְׁכְּנַזִּים: אַבְרָהָם, יִצְחָק, יַעֲקֹב, יוֹסֵף, מֹשֶׁה, אַהֲרֹן וְדָוִד.
מְגִלַּת קֹהֶלֶת: לְמִנְהַג הָאַשְׁכְּנַזִּים קוֹרְאִים אוֹתָהּ בְּשַׁבָּת שֶׁחָלָה בְּחַג הַסֻּכּוֹת, בִּתְפִלַּת שַׁחֲרִית לִפְנֵי קְרִיאַת הַתּוֹרָה. רֹב הַסְּפָרַדִּים לֹא נָהֲגוּ לִקְרֹא קֹהֶלֶת בְּסֻכּוֹת, וְהַתֵּימָנִים קוֹרְאִים חֲלָקִים מִמֶּנָּה.
💛 אַהֲבַת הַשֵּׁם: קֹהֶלֶת מְלַמֶּדֶת שֶׁהַשִּׂמְחָה הָאֲמִתִּית הִיא שִׂמְחָה שֶׁל מִצְוָה: ״אֶת הָאֱלֹהִים יְרָא וְאֶת מִצְוֹתָיו שְׁמוֹר״. כָּל עֵדָה שׁוֹמֶרֶת עַל מִנְהָגָהּ, וְכֻלָּנוּ שְׂמֵחִים לִפְנֵי אוֹתוֹ ה׳.
📖 פֶּרֶק א׳, הֲלָכוֹת ט׳, י״ד

💧שִׂמְחַת בֵּית הַשּׁוֹאֵבָה

בְּכָל הַשָּׁנָה נִסְּכוּ עַל הַמִּזְבֵּחַ יַיִן, וְרַק בְּסֻכּוֹת נִסְּכוּ גַּם מַיִם מִמַּעְיַן הַשִּׁילוֹחַ. עַל שְׁאִיבַת הַמַּיִם הָיְתָה בַּמִּקְדָּשׁ שִׂמְחָה גְּדוֹלָה כָּל הַלַּיְלָה, עִם נְגִינָה וְרִקּוּדִים. חַכְמֵי יִשְׂרָאֵל וְהַחֲסִידִים רָקְדוּ, וְכָל הָעָם – אֲנָשִׁים וְנָשִׁים – בָּאוּ לִרְאוֹת וְלִשְׁמֹעַ. כְּדֵי שֶׁלֹּא תִּהְיֶה קַלּוּת רֹאשׁ, בָּנוּ מִרְפָּסוֹת סְבִיב עֶזְרַת הַנָּשִׁים: הַנָּשִׁים עָמְדוּ לְמַעְלָה, הַגְּבָרִים לְמַטָּה, וְהָרִקּוּדִים בָּאֶמְצַע. אָמְרוּ חֲכָמִים: ״מִי שֶׁלֹּא רָאָה שִׂמְחַת בֵּית הַשּׁוֹאֵבָה – לֹא רָאָה שִׂמְחָה מִיָּמָיו״.
רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל הָיָה זוֹרֵק שְׁמוֹנֶה אֲבוּקוֹת שֶׁל אֵשׁ וְתוֹפֵס אוֹתָן. וְהַיּוֹם עוֹרְכִים בְּלֵילוֹת חֹל הַמּוֹעֵד שִׂמְחַת בֵּית הַשּׁוֹאֵבָה – זֵכֶר לַמִּקְדָּשׁ.
💛 אַהֲבַת הַשֵּׁם: מִתּוֹךְ שִׂמְחַת הַמִּצְוָה זָכוּ גְּדוֹלֵי יִשְׂרָאֵל לְרוּחַ הַקֹּדֶשׁ. כְּשֶׁשְּׂמֵחִים בַּה׳ בְּכָל הַלֵּב – מִתְקָרְבִים אֵלָיו.

💧🔑 מוּשָּׂג: נִסּוּךְ הַמַּיִם

״רַק בְּחַג הַסֻּכּוֹת נוֹסְפָה מִצְוָה מְיֻחֶדֶת – לְנַסֵּךְ יַחַד עִם הַיַּיִן גַּם מַיִם״.
📖 נִסּוּךְ – יְצִיקָה עַל גַּבֵּי הַמִּזְבֵּחַ. נִסּוּךְ הַמַּיִם הָיָה רַק בְּחַג הַסֻּכּוֹת, לְבַקֵּשׁ גִּשְׁמֵי בְּרָכָה.
💛 אַהֲבַת הַשֵּׁם: הַמַּיִם מְקַיְּמִים אֶת כֻּלָּם – עֲשָׂבִים, חַיּוֹת וּבְנֵי אָדָם, בְּלִי יוֹצֵא מִן הַכְּלָל. 🧒 וְאַתָּה? כְּשֶׁאַתָּה שׁוֹתֶה מַיִם – תַּגִּיד תּוֹדָה לַה׳ עַל הַחֶסֶד הַגָּדוֹל הַזֶּה.

🐂שִׁבְעִים פָּרִים

בְּשִׁבְעַת יְמֵי הַחַג הִקְרִיבוּ שִׁבְעִים פָּרִים, כְּנֶגֶד שִׁבְעִים אֻמּוֹת הָעוֹלָם: בַּיּוֹם הָרִאשׁוֹן שְׁלֹשָׁה עָשָׂר, בַּשֵּׁנִי שְׁנֵים עָשָׂר, וְכָךְ פּוֹחֲתִים עַד שִׁבְעָה בַּיּוֹם הַשְּׁבִיעִי. מִתּוֹךְ אַהֲבָה הִקְרִיבוּ עֲבוּרָם, כְּדֵי לְהוֹסִיף לָהֶם שֶׁפַע וּבְרָכָה.
וּבִשְׁמִינִי עֲצֶרֶת מַקְרִיבִים פַּר אֶחָד – כְּנֶגֶד אֻמָּה יְחִידָה, יִשְׂרָאֵל. כְּמוֹ מֶלֶךְ שֶׁאַחֲרֵי סְעוּדָה גְּדוֹלָה אָמַר לְאוֹהֲבוֹ: עֲשֵׂה לִי סְעוּדָה קְטַנָּה, כְּדֵי שֶׁאֵהָנֶה מִמְּךָ.
💛 אַהֲבַת הַשֵּׁם: לֶעָתִיד לָבוֹא גַּם הַגּוֹיִים יַעֲלוּ לִירוּשָׁלַיִם לָחֹג אֶת חַג הַסֻּכּוֹת. וּבְכָל זֹאת, לַה׳ יֵשׁ אַהֲבָה מְיֻחֶדֶת לְיִשְׂרָאֵל – כְּמוֹ לְאוֹהֵב קָרוֹב.
📖 פֶּרֶק א׳, הֲלָכָה י״ג

🍲אוּשְׁפִּיזִין – אוֹרְחִים בַּסֻּכָּה

מִצְוָה מִן הַתּוֹרָה לְשַׁתֵּף בְּשִׂמְחַת הַחַג אֶת הָעֲנִיִּים וְהַבּוֹדְדִים, וּלְהַזְמִין אוֹתָם לַסֻּכָּה. וּבַזֹּהַר נֶאֱמַר שֶׁמַּזְמִינִים לַסֻּכָּה גַּם ״אוּשְׁפִּיזִין עִלָּאִין״ – שִׁבְעָה צַדִּיקִים, מֵאַבְרָהָם אָבִינוּ וְעַד דָּוִד הַמֶּלֶךְ.
הַצַּדִּיקִים בָּאִים רַק לְמִי שֶׁנּוֹתֵן גַּם לָעֲנִיִּים אֶת חֶלְקָם. וּבַדּוֹר שֶׁלָּנוּ מִצְוָה גְּדוֹלָה לְהַזְמִין לַסֻּכָּה גַּם אֲנָשִׁים עֲצוּבִים וּבוֹדְדִים.
💛 אַהֲבַת הַשֵּׁם: אַבְרָהָם אָבִינוּ אָהַב לְהַכְנִיס אוֹרְחִים. כְּשֶׁאָנוּ מְשַׂמְּחִים אָדָם בּוֹדֵד, אַבְרָהָם וְכָל הַצַּדִּיקִים שְׂמֵחִים לָשֶׁבֶת אִתָּנוּ.
📖 פֶּרֶק א׳, הֲלָכָה י״ד

🧠 בְּדֹק אֶת עַצְמְךָ

מָה הָיָה מְיֻחָד בַּנִּסּוּךְ עַל הַמִּזְבֵּחַ בְּחַג הַסֻּכּוֹת?
💡 בְּכָל הַשָּׁנָה נִסְּכוּ רַק יַיִן. בְּסֻכּוֹת, עִם קָרְבַּן הַתָּמִיד שֶׁל שַׁחַר, נִסְּכוּ גַּם מַיִם מִמַּעְיַן הַשִּׁילוֹחַ – וְעַל שְׁאִיבַת הַמַּיִם הָיְתָה שִׂמְחַת בֵּית הַשּׁוֹאֵבָה. בְּסֻכּוֹת נִדּוֹנִים עַל הַמַּיִם.
כְּנֶגֶד מָה הִקְרִיבוּ שִׁבְעִים פָּרִים בְּחַג הַסֻּכּוֹת?
💡 שִׁבְעִים הַפָּרִים הֵם כְּנֶגֶד שִׁבְעִים אֻמּוֹת – מִתּוֹךְ אַהֲבָה, כְּדֵי לְהוֹסִיף לָהֶם שֶׁפַע וּבְרָכָה. וּבִשְׁמִינִי עֲצֶרֶת מַקְרִיבִים פַּר אֶחָד, כְּנֶגֶד יִשְׂרָאֵל.
בְּשִׂמְחַת בֵּית הַשּׁוֹאֵבָה בַּמִּקְדָּשׁ – אֵיפֹה עָמְדוּ הַנָּשִׁים?
💡 כָּל הָעָם, אֲנָשִׁים וְנָשִׁים, בָּאוּ לִרְאוֹת אֶת הָרִקּוּדִים וְלִשְׁמֹעַ אֶת הַנְּגִינָה. בַּתְּחִלָּה עָמְדוּ הַנָּשִׁים בְּעֶזְרַת הַנָּשִׁים וְהַגְּבָרִים רָחוֹק יוֹתֵר. כְּשֶׁרָאוּ חֲכָמִים שֶׁיֵּשׁ קַלּוּת רֹאשׁ, בָּנוּ מִרְפָּסוֹת: הַנָּשִׁים עָמְדוּ לְמַעְלָה, הַגְּבָרִים לְמַטָּה, וְהָרִקּוּדִים בָּאֶמְצַע.

מְלָאכוֹת שֶׁבַּצּוֹמֵחַ

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק י״ט

🧺בֶּחָצֵר וּבַשְּׁבִיל

אֲדָמָה בֶּחָצֵר לֹא מְטַאטְאִים, וְלֹא בּוֹעֲטִים בְּחוֹל, שֶׁלֹּא יַשְׁווּ גֻּמּוֹת. רִצְפָּה מְרֻצֶּפֶת – מֻתָּר לְטַאטֵא. כִּסֵּא מֻתָּר לִגְרֹר, כִּי לֹא בָּטוּחַ שֶׁיַּעֲשֶׂה חָרִיץ. פֵּרוֹת שֶׁהִתְפַּזְּרוּ בֶּחָצֵר, אֶחָד כָּאן וְאֶחָד שָׁם – לֹא אוֹסְפִים לְסַל, כִּי זֶה מַעֲשֶׂה שֶׁל חֹל, אֲבָל מֻתָּר לְהָרִים מְעַט וְלֶאֱכֹל.
פֵּרוֹת שֶׁהִתְפַּזְּרוּ בְּתוֹךְ הַבַּיִת – מֻתָּר לֶאֱסֹף, כִּי זֶה אֵינוֹ דּוֹמֶה לִמְלֶאכֶת מְעַמֵּר שֶׁבַּשָּׂדֶה.
💛 אַהֲבַת הַשֵּׁם: אֲפִלּוּ הַתְּנוּעוֹת הַקְּטַנּוֹת שֶׁלָּנוּ שׁוֹנוֹת בְּשַׁבָּת. כָּךְ כָּל רֶגַע בַּיּוֹם הַזֶּה מַזְכִּיר לָנוּ שֶׁאֲנַחְנוּ בְּתוֹךְ זְמַן קָדוֹשׁ.
📖 פֶּרֶק י״ט, הֲלָכוֹת א׳–ב׳, י״א

🌱לֹא זוֹרְעִים וְלֹא מַשְׁקִים

בְּשַׁבָּת לֹא זוֹרְעִים וְלֹא מַשְׁקִים צְמָחִים, כִּי הַמַּיִם עוֹזְרִים לַצֶּמַח לִגְדֹּל. אָסוּר לָשִׂים גַּרְעִין אָבוֹקָדוֹ בְּכוֹס מַיִם כְּדֵי שֶׁיַּצְמִיחַ שָׁרָשִׁים, וְאָסוּר לִזְרֹק זְרָעִים עַל אֲדָמָה לַחָה. מִי שֶׁאוֹכֵל בַּגִּנָּה – לֹא נוֹטֵל יָדַיִם עַל הַצְּמָחִים.
אֶת הַמַּמְטֵרוֹת אֶפְשָׁר לִפְתֹּחַ אוֹ לְכַוֵּן בְּשָׁעוֹן לִפְנֵי שַׁבָּת, כִּי בְּשַׁבָּת עַצְמָהּ הַיְּהוּדִי אֵינוֹ עוֹשֶׂה דָּבָר.
💛 אַהֲבַת הַשֵּׁם: שִׁשָּׁה יָמִים אֲנַחְנוּ שֻׁתָּפִים שֶׁל ה׳ בִּבְנִיַּת הָעוֹלָם. בְּשַׁבָּת אֲנַחְנוּ עוֹצְרִים, וְרוֹאִים שֶׁהַכֹּל גָּדֵל בִּזְכוּתוֹ – גַּם בְּלִי שֶׁנִּגַּע.
📖 פֶּרֶק י״ט, הֲלָכוֹת ג׳–ד׳

🌳לֹא קוֹטְפִים וְלֹא מְטַפְּסִים

קְטִיפַת פְּרִי, עָלֶה אוֹ פֶּרַח הִיא מְלֶאכֶת קוֹצֵר. חֲכָמִים הוֹסִיפוּ גָּדֵר: לֹא מִשְׁתַּמְּשִׁים בְּעֵץ בְּשַׁבָּת – לֹא מְטַפְּסִים עָלָיו, לֹא נִשְׁעָנִים עָלָיו, לֹא מַנִּיחִים עָלָיו חֲפָצִים וְלֹא לוֹקְחִים מִמֶּנּוּ. כַּדּוּר שֶׁנִּתְקַע בְּעֵץ – לֹא מוֹרִידִים וְלֹא מְנַעְנְעִים אֶת הָעֵץ. גַּם בְּנַדְנֵדָה שֶׁקְּשׁוּרָה לְעֵץ לֹא מִתְנַדְנְדִים.
עַל דֶּשֶׁא מֻתָּר לָשֶׁבֶת וְלָלֶכֶת, כִּי חֲכָמִים גָּזְרוּ רַק עַל עֵצִים וְשִׂיחִים קָשִׁים. אֲבָל לֹא מְרִיחִים פְּרִי שֶׁעוֹד תָּלוּי עַל הָעֵץ, שֶׁמָּא יִקְטְפוּ אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: הָעֵץ הוּא מַתָּנָה מֵה׳. בְּשַׁבָּת אֲנַחְנוּ נוֹתְנִים לוֹ לִגְדֹּל בְּשֶׁקֶט, וְשׂוֹמְחִים בַּפֵּרוֹת שֶׁכְּבָר הֵכַנּוּ מֵעֶרֶב שַׁבָּת.
📖 פֶּרֶק י״ט, הֲלָכוֹת ו׳–ח׳

💐פְּרָחִים בָּאֲגַרְטָל וְעָצִיץ

פְּרָחִים שֶׁנִּקְטְפוּ לִפְנֵי שַׁבָּת אֵינָם מֻקְצֶה: מֻתָּר לְהָזִיז אֶת הָאֲגַרְטָל, לְהוֹצִיא פֶּרַח כְּדֵי לְהָרִיחַ וּלְהַחֲזִירוֹ לַמַּיִם, אִם הַפְּרָחִים כְּבָר פְּתוּחִים. אֲבָל פְּרָחִים שֶׁעוֹד לֹא נִפְתְּחוּ – לֹא שָׂמִים בְּמַיִם, כִּי הַמַּיִם יַעַזְרוּ לָהֶם לִפְרֹחַ. זֵר כָּזֶה שָׂמִים בַּאֲגַרְטָל בְּלִי מַיִם.
מֵעָצִיץ לֹא קוֹטְפִים וְאוֹתוֹ לֹא מַשְׁקִים. עָצִיץ שֶׁנָּפַל – לֹא מַחֲזִירִים אֵלָיו אֶת הָאֲדָמָה שֶׁנִּשְׁפְּכָה, כִּי זֶה מֵיטִיב עִם הַצֶּמַח, וְהָאֲדָמָה מֻקְצֶה.
💛 אַהֲבַת הַשֵּׁם: פֶּרַח יָפֶה עַל שֻׁלְחַן הַשַּׁבָּת מוֹסִיף כָּבוֹד לַשַּׁבָּת. כְּשֶׁאַתָּה מֵרִיחַ אוֹתוֹ, אַתָּה יָכוֹל לְבָרֵךְ אֶת ה׳ שֶׁבָּרָא רֵיחוֹת טוֹבִים.
📖 פֶּרֶק י״ט, הֲלָכוֹת ט׳–י׳

🧠 בְּדֹק אֶת עַצְמְךָ

תַּפּוּחִים שֶׁנָּשְׁרוּ לִפְנֵי שַׁבָּת מְפֻזָּרִים בְּכָל הֶחָצֵר, אֶחָד כָּאן וְאֶחָד שָׁם. מָה מֻתָּר לְאֵלִיָּהוּ לַעֲשׂוֹת?
💡 אִסּוּף פֵּרוֹת מְפֻזָּרִים בֶּחָצֵר נִרְאֶה כְּמַעֲשֶׂה שֶׁל חֹל, וְלָכֵן אָסוּר. לְהָרִים מְעַט וְלֶאֱכֹל – מֻתָּר. בְּתוֹךְ הַבַּיִת מֻתָּר גַּם לֶאֱסֹף.
בְּשַׁבָּת נָפַל עָצִיץ, וְחֵלֶק מִן הָאֲדָמָה שֶׁבּוֹ נִשְׁפַּךְ עַל הָרִצְפָּה. מָה נָכוֹן?
💡 הַחְזָרַת הָאֲדָמָה עוֹזֶרֶת לַצֶּמַח, וְזֶה כְּמוֹ חוֹרֵשׁ וְזוֹרֵעַ. וְחוּץ מִזֶּה הָאֲדָמָה עַצְמָהּ מֻקְצֶה, וְאָסוּר לְטַלְטֵל אוֹתָהּ.

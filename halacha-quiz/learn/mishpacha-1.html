👑כִּבּוּד הוֹרִים – כְּמוֹ כְּבוֹד ה׳

מִצְוַת עֲשֵׂה מִן הַתּוֹרָה לְכַבֵּד אֶת הָאָב וְאֶת הָאֵם וּלְיִרְאָה מֵהֶם: "כַּבֵּד אֶת אָבִיךָ וְאֶת אִמֶּךָ". חֲכָמִים אָמְרוּ שֶׁהַתּוֹרָה הִשְׁוְתָה אֶת כְּבוֹד הַהוֹרִים לִכְבוֹד ה׳: עַל הַהוֹרִים נֶאֱמַר "אִישׁ אִמּוֹ וְאָבִיו תִּירָאוּ", וְעַל ה׳ נֶאֱמַר "אֶת ה׳ אֱלֹהֶיךָ תִּירָא".
מִי שֶׁמְּכַבֵּד אֶת הוֹרָיו מִתְרַגֵּל לְהַכִּיר טוֹבָה – וּמִתּוֹךְ כָּךְ מַגִּיעַ לֶאֱמוּנָה בַּבּוֹרֵא, הַטּוֹב וְהַמֵּטִיב.
💛 אַהֲבַת הַשֵּׁם: "שְׁלֹשָׁה שֻׁתָּפִים יֵשׁ בָּאָדָם: הַקָּדוֹשׁ בָּרוּךְ הוּא, אָבִיו וְאִמּוֹ". כְּשֶׁאָדָם מְכַבֵּד אֶת הוֹרָיו, אוֹמֵר ה׳: "מַעֲלֶה אֲנִי עֲלֵיהֶם כְּאִלּוּ דַּרְתִּי בֵּינֵיהֶם וְכִבְּדוּנִי". כָּל כּוֹס מַיִם שֶׁאַתָּה מֵבִיא לְאַבָּא הִיא גַּם כָּבוֹד לַה׳.

🙏הַכָּרַת הַטּוֹב – הַדֶּרֶךְ לֶאֱמוּנָה

מַדּוּעַ הֻשְׁוָה כִּבּוּד הוֹרִים לִכְבוֹד שָׁמַיִם? מִי שֶׁאֵינוֹ מְכַבֵּד אֶת הוֹרָיו חוֹשֵׁב רַק עַל עַצְמוֹ, וְחוֹשֵׁב שֶׁהַכֹּל מַגִּיעַ לוֹ. הוּא לֹא שׁוֹאֵל מִי טִפֵּל בּוֹ בְּיַלְדוּתוֹ וּמִי דָּאַג לוֹ כְּשֶׁהָיָה חוֹלֶה – וְלָכֵן גַּם לֹא יַחְשֹׁב עַל הַבּוֹרֵא.
מִי שֶׁמְּכַבֵּד אֶת הוֹרָיו מִתְרַגֵּל לַחְשֹׁב עַל שָׁרְשֵׁי הַדְּבָרִים וּלְהַכִּיר תּוֹדָה, וּמִשָּׁם מַמְשִׁיךְ לֶאֱמוּנָה: "שְׂאוּ מָרוֹם עֵינֵיכֶם וּרְאוּ מִי בָרָא אֵלֶּה".
💛 אַהֲבַת הַשֵּׁם: מִי מוֹרִיד אֶת הַגֶּשֶׁם, מִי בָּרָא אֶת הַשֶּׁמֶשׁ, מִי נָפַח בָּנוּ רוּחַ חַיִּים? הַתּוֹדָה לַהוֹרִים הִיא הַמַּדְרֵגָה הָרִאשׁוֹנָה בַּסֻּלָּם שֶׁמַּגִּיעַ עַד הַתּוֹדָה לַה׳.

📜פְּסוּקִים שֶׁל כָּבוֹד

"כַּבֵּד אֶת אָבִיךָ וְאֶת אִמֶּךָ" – מִצְוַת הַכִּבּוּד. "אִישׁ אִמּוֹ וְאָבִיו תִּירָאוּ" – מִצְוַת הַמּוֹרָא. "מִפְּנֵי שֵׂיבָה תָּקוּם" – מִמֶּנּוּ לָמְדוּ לָקוּם לִפְנֵי הַהוֹרִים, שֶׁחֲשׁוּבִים לַיְּלָדִים יוֹתֵר מִכָּל זָקֵן וְחָכָם. וְדָמָא בֶּן נְתִינָה וִיתֵּר עַל מְכִירַת אֲבָנִים יְקָרוֹת לַחֹשֶׁן, כְּדֵי לֹא לְהָעִיר אֶת אָבִיו.
"שְׁלֹשָׁה שֻׁתָּפִים יֵשׁ בָּאָדָם: הַקָּדוֹשׁ בָּרוּךְ הוּא, אָבִיו וְאִמּוֹ". וְהוֹרִים שֶׁמָּחֲלוּ עַל כְּבוֹדָם – כְּבוֹדָם מָחוּל.
💛 אַהֲבַת הַשֵּׁם: כָּל פָּסוּק שֶׁאָנוּ לוֹמְדִים עַל כִּבּוּד הוֹרִים הוּא גַּם פָּסוּק עַל אַהֲבַת ה׳ – כִּי ה׳ הוּא הַשֻּׁתָּף הַשְּׁלִישִׁי בַּבַּיִת.
📖 פֶּרֶק א׳, הֲלָכוֹת א׳, ד׳–ה׳, ז׳

🪑🔑 מוּשָּׂג: מוֹרָא

״אִישׁ אִמּוֹ וְאָבִיו תִּירָאוּ״ – ״מוֹרָא: לֹא עוֹמֵד בִּמְקוֹמוֹ הַמְיֻחָד לוֹ, וְלֹא יוֹשֵׁב בִּמְקוֹמוֹ הַמְיֻחָד לוֹ, וְלֹא סוֹתֵר אֶת דְּבָרָיו״.
📖 מוֹרָא – מִלְּשׁוֹן ״תִּירָאוּ״: יִרְאַת כָּבוֹד. לֹא פַּחַד וּבֶהָלָה, אֶלָּא זְהִירוּת שֶׁלֹּא לִפְגֹּעַ בִּכְבוֹד הַהוֹרִים – לֹא לָשֶׁבֶת בִּמְקוֹמָם וְלֹא לִקְרֹא לָהֶם בִּשְׁמָם.
💛 אַהֲבַת הַשֵּׁם: ״הִשְׁוָה הַכָּתוּב״: ״אִישׁ אִמּוֹ וְאָבִיו תִּירָאוּ״ – וְ״אֶת ה׳ אֱלֹהֶיךָ תִּירָא״. מִי שֶׁלּוֹמֵד לְכַבֵּד אֶת הוֹרָיו, לוֹמֵד לְכַבֵּד גַּם אֶת ה׳. 🧒 וְאַתָּה? בַּסְּעוּדָה הַבָּאָה – הַשְׁאֵר אֶת הַכִּסֵּא שֶׁל אַבָּא פָּנוּי בִּשְׁבִילוֹ.
📖 פֶּרֶק א׳, הֲלָכוֹת א׳, ד׳, ו׳

👨‍👦📜 לְשׁוֹן חֲכָמִים: שְׁלוֹשָׁה שֻׁתָּפִים בָּאָדָם

״שְׁלוֹשָׁה שֻׁתָּפִים יֵשׁ בָּאָדָם, הַקָּדוֹשׁ בָּרוּךְ הוּא אָבִיו וְאִמּוֹ, בִּזְמַן שֶׁאָדָם מְכַבֵּד אֶת אָבִיו וְאִמּוֹ, אוֹמֵר הַקָּדוֹשׁ בָּרוּךְ הוּא: מַעֲלֶה אֲנִי עֲלֵיהֶם כְּאִלּוּ דַּרְתִּי בֵּינֵיהֶם וְכִבְּדוּנִי״.
📖 שֻׁתָּף – מִי שֶׁעוֹשֶׂה דָּבָר יַחַד עִם אַחֵר. כָּל אָדָם נִבְרָא בְּשֻׁתָּפוּת: אַבָּא וְאִמָּא מְבִיאִים אוֹתוֹ לָעוֹלָם וּמְגַדְּלִים אוֹתוֹ, וְה׳ ״נָפַח בּוֹ רוּחַ חַיִּים״.
💛 אַהֲבַת הַשֵּׁם: ״הִשְׁוָה הַכָּתוּב כִּבּוּד אָב וָאֵם לִכְבוֹד הַמָּקוֹם״ – מִי שֶׁמְּכַבֵּד אֶת הוֹרָיו, כְּאִלּוּ מְכַבֵּד אֶת ה׳ עַצְמוֹ. 🧒 וְאַתָּה? הַיּוֹם – אֱמֹר לְאַבָּא וּלְאִמָּא ״תּוֹדָה״ עַל דָּבָר אֶחָד שֶׁעָשׂוּ בִּשְׁבִילְךָ.

🙏🔑 מוּשָּׂג: הַכָּרַת הַטּוֹב

״מִי שֶׁמְּכַבֵּד אֶת הוֹרָיו, מֵכִין אֶת נַפְשׁוֹ לְמַחְשָׁבָה שֶׁל אֱמוּנָה, לְמַחְשָׁבָה שֶׁל הַכָּרַת תּוֹדָה... וּמִתּוֹךְ הַהִתְבּוֹנְנוּת בְּהוֹרָיו וְהַכָּרַת הַטּוֹב כְּלַפֵּיהֶם, יַמְשִׁיךְ הָלְאָה לֶאֱמוּנָה בְּבוֹרֵא הָעוֹלָם״.
📖 הַכָּרַת הַטּוֹב – לְהַכִּיר וְלִזְכֹּר אֶת הַטּוֹב שֶׁמִּישֶׁהוּ עָשָׂה לִי, וּלְהוֹדוֹת עָלָיו. מִי שֶׁ״חוֹשֵׁב שֶׁהַכֹּל מַגִּיעַ לוֹ״ – אֵינוֹ מַכִּיר טוֹבָה.
💛 אַהֲבַת הַשֵּׁם: מֵהַתּוֹדָה לַהוֹרִים מַגִּיעִים אֶל הַתּוֹדָה לַה׳: ״שְׂאוּ מָרוֹם עֵינֵיכֶם וּרְאוּ מִי בָרָא אֵלֶּה״. 🧒 וְאַתָּה? הַלַּיְלָה – חֲשֹׁב עַל שְׁלוֹשָׁה דְּבָרִים טוֹבִים שֶׁהוֹרֶיךָ עָשׂוּ לְךָ הַיּוֹם.

🔗חֻלְיָה בְּשַׁרְשֶׁרֶת הַדּוֹרוֹת

שְׁתֵּי מִצְווֹת גְּדוֹלוֹת יֵשׁ בַּתּוֹרָה בֵּין הוֹרִים לִילָדִים: כִּבּוּד הוֹרִים, שֶׁנִּזְכֶּרֶת בַּעֲשֶׂרֶת הַדִּבְּרוֹת, וְחִנּוּךְ הַיְּלָדִים. כִּבּוּד הוֹרִים קוֹשֵׁר אוֹתָנוּ אֶל כָּל הַדּוֹרוֹת שֶׁעָבְרוּ, וְחִנּוּךְ הַיְּלָדִים קוֹשֵׁר אֶת הַהוֹרִים אֶל כָּל הַדּוֹרוֹת שֶׁיָּבוֹאוּ.
כְּשֶׁאָנוּ מְכַבְּדִים אֶת הַהוֹרִים, אָנוּ לוֹמְדִים מֵהֶם אֶת כָּל הַחָכְמָה וְהַנִּסָּיוֹן שֶׁל הַדּוֹרוֹת שֶׁלְּפָנֵינוּ.
💛 אַהֲבַת הַשֵּׁם: הַחַיִּים שֶׁל כָּל אֶחָד מֵאִתָּנוּ קְצָרִים, אֲבָל דֶּרֶךְ הַשַּׁרְשֶׁרֶת הַזֹּאת אָנוּ מִתְחַבְּרִים אֶל הַנֶּצַח – אֶל עַם יִשְׂרָאֵל שֶׁל כָּל הַדּוֹרוֹת, שֶׁעוֹבֵד אֶת ה׳.

4️⃣אַרְבַּע מִצְווֹת

אַרְבַּע מִצְווֹת מִן הַתּוֹרָה עוֹסְקוֹת בַּיַּחַס לַהוֹרִים. שְׁתֵּי מִצְווֹת עֲשֵׂה: לְכַבֵּד אוֹתָם – לַעֲזֹר לָהֶם בְּכָל מָה שֶׁהֵם צְרִיכִים, וּלְיִרְאָה מֵהֶם – לֹא לִפְגֹּעַ בִּכְבוֹדָם. וּשְׁתֵּי מִצְווֹת לֹא תַּעֲשֶׂה: לֹא לְהַכּוֹת אוֹתָם, וְלֹא לְקַלֵּל אוֹתָם.
אָסוּר לְהַכּוֹת אוֹ לְקַלֵּל כָּל יְהוּדִי. אֲבָל כְּלַפֵּי אַבָּא וְאִמָּא הָאִסּוּר חָמוּר הַרְבֵּה יוֹתֵר, וְהַתּוֹרָה הֶחְמִירָה בּוֹ מְאֹד.
💛 אַהֲבַת הַשֵּׁם: הַיָּדַיִם וְהַפֶּה שֶׁלָּנוּ נִבְרְאוּ כְּדֵי לְהֵיטִיב. כְּשֶׁאָנוּ שׁוֹמְרִים אוֹתָם מִלִּפְגֹּעַ בַּהוֹרִים, גַּם בִּשְׁעַת כַּעַס, אָנוּ מְקַדְּשִׁים אוֹתָם לַה׳.
📖 פֶּרֶק א׳, הֲלָכָה ג׳

⚡מְקַלֵּל – חָמוּר מִמַּכֶּה

הַתּוֹרָה הֶעֱנִישָׁה אֶת מִי שֶׁמְּקַלֵּל אֶת הוֹרָיו בְּעֹנֶשׁ חָמוּר יוֹתֵר מִמִּי שֶׁמַּכֶּה אוֹתָם, וְאָסוּר לְקַלֵּל אוֹתָם אֲפִלּוּ אַחֲרֵי מוֹתָם.
בִּשְׁעַת כַּעַס אֲנָשִׁים נוֹטִים לְהִתְפָּרֵץ וּלְקַלֵּל, וְלָכֵן הֶחְמִירָה הַתּוֹרָה כְּדֵי לִמְנֹעַ זֹאת. וְעוֹד – הַמְקַלֵּל מַזְכִּיר אֶת שֵׁם ה׳ וּפוֹגֵעַ בּוֹ בִּכְבוֹד הוֹרָיו. וּבְפֹעַל, הָעֳנָשִׁים הַחֲמוּרִים שֶׁל בֵּית הַדִּין כִּמְעַט לֹא בֻּצְּעוּ – הֵם נוֹעֲדוּ לְהַרְאוֹת כַּמָּה הַחֵטְא חָמוּר.

🍽️🔑 מוּשָּׂג: מָקוֹם הַמְיֻחָד לָאָב

📛"אַבָּא" וְ"אִמָּא"

מִכְּלַל הַמּוֹרָא – לֹא קוֹרְאִים לַהוֹרִים בִּשְׁמָם, כְּמוֹ שֶׁקּוֹרְאִים לָהֶם חֲבֵרֵיהֶם, אֶלָּא "אַבָּא" וְ"אִמָּא". גַּם כְּשֶׁהֵם לֹא נִמְצָאִים – מַזְכִּירִים אוֹתָם בַּתּוֹאַר "אַבָּא" וְ"אִמָּא". אֲבָל מִי שֶׁשּׁוֹאֲלִים אוֹתוֹ "מַה שֵּׁם אָבִיךָ?" – רַשַּׁאי לַעֲנוֹת: "שֵׁם אָבִי מֹשֶׁה".
חָבֵר שֶׁשְּׁמוֹ כְּשֵׁם אָבִיךָ – לִפְנֵי אַבָּא לֹא קוֹרְאִים לוֹ בִּשְׁמוֹ, אֶלָּא פּוֹנִים אֵלָיו בְּדֶרֶךְ אַחֶרֶת. וְנֶכֶד שֶׁנִּקְרָא עַל שֵׁם סַבָּא שֶׁחַי – לְיַד סַבָּא קוֹרְאִים לוֹ בְּכִנּוּי חִבָּה, אֶלָּא אִם כֵּן סַבָּא הִסְכִּים.

📛שֵׁם שֶׁל אַבָּא – וְשֵׁם שֶׁל חָבֵר

חָבֵר שֶׁשְּׁמוֹ כְּשֵׁם אָבִיךָ: אִם זֶה שֵׁם רָגִיל, כְּמוֹ מֹשֶׁה אוֹ יַעֲקֹב – לִפְנֵי אַבָּא לֹא קוֹרְאִים לוֹ בִּשְׁמוֹ, וְשֶׁלֹּא בְּפָנָיו – מֻתָּר. אִם זֶה שֵׁם נָדִיר, כְּמוֹ קַלְמָן אוֹ יִפְרָח – גַּם שֶׁלֹּא בִּפְנֵי אַבָּא מְשַׁנִּים מְעַט אֶת הַשֵּׁם.
וּמִי שֶׁשּׁוֹאֲלִים אוֹתוֹ "מַה שֵּׁם אָבִיךָ?" – רַשַּׁאי לְכָל הַמִּנְהָגִים לַעֲנוֹת "שֵׁם אָבִי מֹשֶׁה".
💛 אַהֲבַת הַשֵּׁם: שֵׁם אֵינוֹ רַק מִלָּה – הוּא מְבַטֵּא אֶת הָאָדָם. כְּשֶׁאָנוּ נִזְהָרִים בַּשֵּׁם שֶׁל אַבָּא, אָנוּ לוֹמְדִים לְהִזָּהֵר גַּם בִּכְבוֹד שְׁמוֹ שֶׁל ה׳.

🔀"אָבִי מֹשֶׁה" – שְׁנֵי מִנְהָגִים, וּשְׁנֵיהֶם כַּהֲלָכָה

הַאִם מֻתָּר לוֹמַר אֶת שֵׁם הָאָב יַחַד עִם הַתּוֹאַר "אָבִי"? לְמִנְהַג יוֹצְאֵי סְפָרַד מֻתָּר לוֹמַר: "אָבִי מֹשֶׁה אָמַר כָּךְ", כִּי כְּבָר אָמַר "אָבִי", וּבָרוּר שֶׁהוּא מְכַבְּדוֹ. לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז לֹא אוֹמְרִים אֶת שֵׁם הָאָב כְּלָל, אֶלָּא "אָבִי אוֹמֵר כָּךְ".

😴לֹא מְעִירִים אֶת הַהוֹרִים

מִכְּלַל כִּבּוּד הוֹרִים – לֹא לְצַעֵר אוֹתָם, וּלְהָעִיר אָדָם מִשְּׁנָתוֹ זֶה צַעַר גָּדוֹל. דָּמָא בֶּן נְתִינָה לֹא הֵעִיר אֶת אָבִיו הַיָּשֵׁן, שֶׁהַמַּפְתֵּחַ הָיָה אֶצְלוֹ, וְהִפְסִיד מְכִירָה שֶׁל אֲבָנִים יְקָרוֹת לַחֹשֶׁן. לַשָּׁנָה הַבָּאָה נוֹלְדָה בְּעֶדְרוֹ פָּרָה אֲדֻמָּה.
אִם אַבָּא בִּקֵּשׁ "הָעֵר אוֹתִי לַתְּפִלָּה" – מְעִירִים אוֹתוֹ, וְזוֹ מִצְוָה. טֶלֶפוֹן "דָּחוּף" בִּזְמַן שֶׁאַבָּא יָשֵׁן – לֹא מְעִירִים, אֶלָּא אִם כֵּן בָּרוּר מָה הַדְּחִיפוּת וְשֶׁאַבָּא יִצְטַעֵר אִם לֹא יָעִירוּ אוֹתוֹ. וּכְשֶׁאֶפְשָׁר – עָדִיף שֶׁאָדָם אַחֵר יָעִיר.

⏰מָתַי מֻתָּר לְהָעִיר?

🧹עוֹזְרִים בַּבַּיִת

לֹא מַסְפִּיק לַעֲזֹר – צָרִיךְ לַעֲזֹר בְּסֵבֶר פָּנִים יָפוֹת. יֵשׁ בֵּן שֶׁעוֹזֵר לְהוֹרָיו בְּפָנִים זוֹעֲפוֹת וּבְמֹרַת רוּחַ, וְלֵב הַהוֹרִים נִשְׁבָּר. עַל כָּךְ אוֹמֵר הַפִּתְגָּם: "אָב אֶחָד מְפַרְנֵס עֲשָׂרָה בָּנִים וְשָׂמֵחַ, וַעֲשָׂרָה בָּנִים אֵינָם יְכוֹלִים לְפַרְנֵס אָב אֶחָד".

🤐שׁוֹתְקִים וְלֹא מַעֲלִיבִים

גַּם כְּשֶׁהַהוֹרִים מְבַיְּשִׁים אֶת בְּנָם, אֲפִלּוּ לִפְנֵי קָהָל – לֹא יַכְלִים אוֹתָם, אֶלָּא יִשְׁתֹּק וְיִירָא מִמֶּלֶךְ מַלְכֵי הַמְּלָכִים, שֶׁצִּוָּה אוֹתוֹ לִירֹא מֵהֶם. עִקַּר הַחִדּוּשׁ בַּמִּצְוָה הוּא שֶׁמְּכַבְּדִים אֶת הַהוֹרִים גַּם כְּשֶׁקָּשֶׁה.

🤐עַד הֵיכָן כִּבּוּד הוֹרִים

הַבֵּן לָבוּשׁ בִּגְדֵי חֲמוּדוֹת וְיוֹשֵׁב בְּרֹאשׁ הַקָּהָל, וּבָאוּ הוֹרָיו וּבִיְּשׁוּ אוֹתוֹ – לֹא יַכְלִים אוֹתָם, אֶלָּא יִשְׁתֹּק. עִקַּר הַחִדּוּשׁ בַּמִּצְוָה: גַּם כְּשֶׁהַהוֹרִים אֵינָם נוֹהֲגִים כָּרָאוּי – מְכַבְּדִים אוֹתָם.
וְאִם הַהוֹרִים גָּרְמוּ לוֹ נֵזֶק בְּכֶסֶף – אֵינוֹ צָרִיךְ לָשֵׂאת בּוֹ, וְיָכוֹל לִתְבֹּעַ אוֹתָם בְּבֵית דִּין. אֲבָל לְהַכְלִים וְלִכְעֹס כְּנֶגְדָּם – לֹא.

🔀לַעֲצֹר אֶת הַהוֹרִים? שְׁתֵּי דֵּעוֹת

הוֹרִים שֶׁבְּכַעֲסָם זָרְקוּ לַיָּם אֶת הָאַרְנָק שֶׁל בְּנָם – הַבֵּן לֹא יַכְלִים אוֹתָם וְלֹא יִכְעַס כְּנֶגְדָּם, וְאֶת הַנֵּזֶק יָכוֹל לִתְבֹּעַ בְּבֵית דִּין. וְאִם הוּא יָכוֹל לַעֲצֹר אוֹתָם לִפְנֵי שֶׁיִּזְרְקוּ? יֵשׁ אוֹמְרִים שֶׁעָדִיף שֶׁיַּעֲצֹר אוֹתָם בְּכֹחַ – בְּלִי לִפְגֹּעַ בָּהֶם, רַק לִבְלֹם. וְיֵשׁ אוֹמְרִים שֶׁלֹּא יַעֲצֹר, כִּי הוּא יָכוֹל לִתְבֹּעַ אוֹתָם אַחַר כָּךְ.

💗מְחִילָה – וּמִצְוָה

"אָב שֶׁמָּחַל עַל כְּבוֹדוֹ – כְּבוֹדוֹ מָחוּל". הוֹרִים שֶׁמְּוַתְּרִים לִבְנָם – הַבֵּן פָּטוּר, וְאִם לֹא יְכַבֵּד, לֹא תִּהְיֶה בְּיָדוֹ עֲבֵרָה. וְאִם יְכַבֵּד בְּכָל זֹאת – מְקַיֵּם מִצְוָה.
תְּלוּנוֹת וּדְרִישׁוֹת לֹא יְבִיאוּ כָּבוֹד. יְלָדִים אוֹהֲבִים לְכַבֵּד מֵרָצוֹן וּמֵאַהֲבָה, וְהַהִתְחַשְּׁבוּת בָּהֶם רַק מוֹסִיפָה לְכָבוֹד.

🌱רוֹאִים אֶת הַטּוֹב – וְאֶת הָאֱמֶת

כְּשֶׁאָדָם רוֹאֶה בְּהוֹרָיו דְּבָרִים טוֹבִים – הוּא מִתְחַבֵּר אֶל שָׁרָשָׁיו הַטּוֹבִים, וּמַמְשִׁיךְ אֶת חַיָּיו בְּטוֹב. וְאִם יֵשׁ בַּהוֹרִים חִסָּרוֹן – לֹא מִתְעַלְּמִים מִמֶּנּוּ, וְאִם יֹאמְרוּ לַעֲבֹר עֲבֵרָה – לֹא שׁוֹמְעִים לָהֶם. וּבְכָל זֹאת דָּנִים אוֹתָם לְכַף זְכוּת: אוּלַי לֹא לָמְדוּ מַסְפִּיק, אוֹ שֶׁקָּשֶׁה לָהֶם.

💡 רַבִּי יְהוּדָה הַנָּשִׂיא אָמַר: הַבֵּן יָרֵא יוֹתֵר מֵאָבִיו, וְלָכֵן הִקְדִּים ה׳ אֶת הָאֵם בְּמוֹרָא – שֶׁיִּירָא מִשְּׁנֵיהֶם בְּשָׁוֶה. וּבְכִבּוּד הִקְדִּים אֶת הָאָב, כִּי אֶת הָאֵם, שֶׁמְּשַׁדֶּלֶת אוֹתוֹ בִּדְבָרִים, הַבֵּן מְכַבֵּד יוֹתֵר מֵעַצְמוֹ.
לָמָּה הֶחְמִירָה הַתּוֹרָה בַּמְקַלֵּל אֶת הוֹרָיו יוֹתֵר מִן הַמַּכֶּה אוֹתָם?
💡 הָעָווֹן הַזֶּה מָצוּי יוֹתֵר, כִּי בִּשְׁעַת כַּעַס אֲנָשִׁים נוֹטִים לְקַלֵּל – וְלָכֵן הֶחְמִירָה הַתּוֹרָה כְּדֵי לִמְנֹעַ אוֹתָם. וְעוֹד, הַמְקַלֵּל מַזְכִּיר אֶת שֵׁם ה׳ כְּשֶׁהוּא פּוֹגֵעַ בְּהוֹרָיו.

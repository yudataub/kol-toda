תְּרוּמוֹת וּמַעַשְׂרוֹת

🍽️ הִלְכוֹת כַּשְׁרוּת · פֶּרֶק ז׳

🧺תְּרוּמוֹת וּמַעַשְׂרוֹת

מִפֵּרוֹת וּמִירָקוֹת שֶׁגָּדְלוּ בְּאֶרֶץ יִשְׂרָאֵל מַפְרִישִׁים תְּרוּמוֹת לַכֹּהֵן וּמַעַשְׂרוֹת לַלֵּוִי וְלֶעָנִי. כָּל זְמַן שֶׁלֹּא הִפְרִישׁוּ – הַפֵּרוֹת נִקְרָאִים "טֶבֶל", כְּלוֹמַר "טוֹב־לֹא", וְאָסוּר לֶאֱכֹל אוֹתָם.
הַיּוֹם כֻּלָּנוּ טְמֵאִים, וְהַכֹּהֲנִים אֵינָם יְכוֹלִים לֶאֱכֹל תְּרוּמָה. לָכֵן מַפְרִישִׁים מְעַט יוֹתֵר מֵאָחוּז אֶחָד, עוֹטְפִים אוֹתוֹ וּמַנִּיחִים בַּפַּח בְּדֶרֶךְ כָּבוֹד.
💛 אַהֲבַת הַשֵּׁם: בִּזְכוּת הַתְּרוּמוֹת וְהַמַּעַשְׂרוֹת יָכְלוּ הַכֹּהֲנִים וְהַלְוִיִּם לְלַמֵּד תּוֹרָה לְכָל יִשְׂרָאֵל. כָּךְ כָּל פְּרִי שֶׁל אֶרֶץ יִשְׂרָאֵל מִתְחַבֵּר לַקְּדֻשָּׁה.
📖 פֶּרֶק ז׳, הֲלָכָה א׳; פֶּרֶק ט׳, הֲלָכָה ה׳; פֶּרֶק י׳, הֲלָכָה א׳

🧺סֵדֶר הַתְּרוּמוֹת וְהַמַּעַשְׂרוֹת

קֹדֶם מַפְרִישִׁים "תְּרוּמָה גְּדוֹלָה" לַכֹּהֵן. אַחַר כָּךְ "מַעֲשֵׂר רִאשׁוֹן" – עֲשִׂירִית לַלֵּוִי, וְהַלֵּוִי נוֹתֵן מִמֶּנּוּ "תְּרוּמַת מַעֲשֵׂר" לַכֹּהֵן. וְאַחֲרֵי זֶה עוֹד מַעֲשֵׂר: "מַעֲשֵׂר שֵׁנִי" אוֹ "מַעֲשֵׂר עָנִי".
לַכֹּהֲנִים וְלַלְוִיִּם לֹא הָיְתָה נַחֲלָה בָּאָרֶץ. הֵם חָיוּ מִן הַמַּתָּנוֹת – כְּדֵי שֶׁיִּהְיוּ פְּנוּיִים לִלְמֹד וּלְלַמֵּד תּוֹרָה. וּבַעַל הַפֵּרוֹת בּוֹחֵר לְאֵיזֶה כֹּהֵן וּלְאֵיזֶה לֵוִי לָתֵת.
💛 אַהֲבַת הַשֵּׁם: כָּךְ נוֹצְרָה שֻׁתָּפוּת: הַחַקְלַאי עוֹבֵד בַּשָּׂדֶה, וְהַכֹּהֵן וְהַלֵּוִי מְלַמְּדִים תּוֹרָה. כֻּלָּם יַחַד בּוֹנִים עַם שֶׁרַגְלָיו נְטוּעוֹת בָּאָרֶץ וְרֹאשׁוֹ בַּשָּׁמַיִם.
📖 פֶּרֶק ז׳, הֲלָכוֹת ב׳–ד׳

🎁🔑 מוּשָּׂג: תְּרוּמָה גְּדוֹלָה

״כֹּל חֵלֶב יִצְהָר וְכָל חֵלֶב תִּירוֹשׁ וְדָגָן רֵאשִׁיתָם אֲשֶׁר יִתְּנוּ לַה׳ לְךָ נְתַתִּים״ – מַפְרִישִׁים רִאשׁוֹנָה אֶת הַ'תְּרוּמָה הַגְּדוֹלָה' וְנוֹתְנִים אוֹתָהּ לַכֹּהֵן.
📖 תְּרוּמָה גְּדוֹלָה – הַחֵלֶק הָרִאשׁוֹן מֵהַפֵּרוֹת, שֶׁנּוֹתְנִים לַכֹּהֵן.
💛 אַהֲבַת הַשֵּׁם: נוֹתְנִים לַכֹּהֵן דַּוְקָא אֶת הָרֵאשִׁית, כְּדֵי שֶׁיּוּכַל לְהִתְמַסֵּר לַעֲבוֹדַת הַקֹּדֶשׁ וּלְהוֹרָאַת הַתּוֹרָה. 🧒 וְאַתָּה? כְּשֶׁאַתָּה מְחַלֵּק מַשֶּׁהוּ טוֹב – תֵּן אֶת הַחֵלֶק הֲכִי יָפֶה קֹדֶם, וְלֹא אֶת הַשְּׁיָרַיִם.
📖 פֶּרֶק ז׳, הֲלָכָה ב׳

🔟🔑 מוּשָּׂג: מַעֲשֵׂר רִאשׁוֹן

״וְלִבְנֵי לֵוִי הִנֵּה נָתַתִּי כָּל מַעֲשֵׂר בְּיִשְׂרָאֵל לְנַחֲלָה, חֵלֶף עֲבֹדָתָם אֲשֶׁר הֵם עֹבְדִים אֶת עֲבֹדַת אֹהֶל מוֹעֵד״.
📖 מַעֲשֵׂר רִאשׁוֹן – עֲשִׂירִית מֵהַפֵּרוֹת, שֶׁנּוֹתְנִים לַלֵּוִי.
💛 אַהֲבַת הַשֵּׁם: הַלְּוִיִּים לֹא קִבְּלוּ חֵלֶק בָּאָרֶץ, כְּדֵי שֶׁיּוּכְלוּ לְהִתְמַסֵּר לְלַמֵּד תּוֹרָה לְכָל יִשְׂרָאֵל – וּמַעֲשֵׂר הָעָם הוּא שָׂכָרָם. 🧒 וְאַתָּה? תַּחְשֹׁב עַל מוֹרֶה אוֹ רַב שֶׁלִּמֵּד אוֹתְךָ, וְתַגִּיד לוֹ תּוֹדָה.
📖 פֶּרֶק ז׳, הֲלָכָה ב׳

🕍🔑 מוּשָּׂג: מַעֲשֵׂר שֵׁנִי

בְּשָׁנִים א׳ ב׳ ד׳ ה׳ לְמַחְזוֹר הַשְּׁמִטָּה, מַעֲשֵׂר נוֹסָף נִקְרָא 'מַעֲשֵׂר שֵׁנִי', ״וְהוּא נִשְׁאָר בְּיָדָיו שֶׁל בַּעַל הַפֵּרוֹת, וְעָלָיו לְהַעֲלוֹתוֹ לִירוּשָׁלַיִם וּלְאָכְלוֹ שָׁם בְּטָהֳרָה״.
📖 מַעֲשֵׂר שֵׁנִי – עֲשִׂירִית נוֹסֶפֶת מֵהַפֵּרוֹת, שֶׁמַּעֲלִים וְאוֹכְלִים בִּירוּשָׁלַיִם.
💛 אַהֲבַת הַשֵּׁם: ה׳ רָצָה שֶׁכָּל מִשְׁפָּחָה תַּגִּיעַ לִירוּשָׁלַיִם וְתִתְחַבֵּר לַקֹּדֶשׁ, לֹא רַק בַּחַגִּים אֶלָּא גַּם דֶּרֶךְ אֲכִילַת הַפֵּרוֹת שָׁם. 🧒 וְאַתָּה? כְּשֶׁאַתָּה מְבַקֵּר בִּירוּשָׁלַיִם – תַּחְשֹׁב כַּמָּה דּוֹרוֹת כְּבָר הֶעֱלוּ לְשָׁם אֶת פֵּרוֹת הָאָרֶץ.
📖 פֶּרֶק ז׳, הֲלָכָה ב׳

🤲🔑 מוּשָּׂג: מַעֲשֵׂר עָנִי

בִּשְׁנַת ג׳ וּבִשְׁנַת ו׳ לְמַחְזוֹר הַשְּׁמִטָּה, הַמַּעֲשֵׂר הַנּוֹסָף ״נִקְרָא 'מַעֲשֵׂר עָנִי' וְנִתָּן לָעֲנִיִּים״.
📖 מַעֲשֵׂר עָנִי – עֲשִׂירִית מֵהַפֵּרוֹת, שֶׁבִּשְׁנַת ג׳ וְו׳ נוֹתְנִים לָעֲנִיִּים בִּמְקוֹם מַעֲשֵׂר שֵׁנִי.
💛 אַהֲבַת הַשֵּׁם: בְּכָל מַחְזוֹר הַשְּׁמִטָּה יֵשׁ שָׁנִים שֶׁבָּהֶן הַחַקְלַאי נִזְכָּר בִּמְיֻחָד בָּעֲנִיִּים, לְהַשְׁווֹת אֶת כָּל חֶלְקֵי הָעָם. 🧒 וְאַתָּה? שְׁאַל אֶת הוֹרֶיךָ אֵיךְ הַמִּשְׁפָּחָה שֶׁלְּךָ עוֹזֶרֶת לַעֲנִיִּים.
📖 פֶּרֶק ז׳, הֲלָכָה ב׳

🎁📜 לְשׁוֹן חֲכָמִים: תְּרוּמַת מַעֲשֵׂר

״מִתּוֹךְ הַמַּעֲשֵׂר שֶׁנּוֹתְנִים לַלֵּוִי צָרִיךְ הַלֵּוִי לְהַפְרִישׁ מַעֲשֵׂר, שֶׁהוּא מַעֲשֵׂר מִן הַמַּעֲשֵׂר, וְלִתְּנוֹ לַכֹּהֵן״.
📖 תְּרוּמַת מַעֲשֵׂר – עֲשִׂירִית נוֹסֶפֶת, שֶׁהַלֵּוִי עַצְמוֹ מַפְרִישׁ מֵהַמַּעֲשֵׂר שֶׁקִּבֵּל, וְנוֹתֵן לַכֹּהֵן.
💛 אַהֲבַת הַשֵּׁם: גַּם הַלֵּוִי, שֶׁמְּקַבֵּל מַתָּנָה, זוֹכֵר לָתֵת הָלְאָה חֵלֶק מִמֶּנָּה. 🧒 וְאַתָּה? כְּשֶׁמְּקַבְּלִים מַתָּנָה – כֵּיף גַּם לַחְלֹק מִמֶּנָּה לְמִישֶׁהוּ אַחֵר.
📖 פֶּרֶק ז׳, הֲלָכָה ב׳

🏛️מַעֲשֵׂר שֵׁנִי, מַעֲשֵׂר עָנִי וּבִכּוּרִים

"מַעֲשֵׂר שֵׁנִי" הָיוּ מַעֲלִים לִירוּשָׁלַיִם וְאוֹכְלִים שָׁם בְּשִׂמְחָה, אוֹ פּוֹדִים בְּכֶסֶף וְקוֹנִים בּוֹ אֹכֶל בִּירוּשָׁלַיִם. בַּשָּׁנָה הַשְּׁלִישִׁית וּבַשָּׁנָה הַשִּׁשִּׁית שֶׁל הַשְּׁמִטָּה נוֹתְנִים בִּמְקוֹמוֹ "מַעֲשֵׂר עָנִי" לָעֲנִיִּים. הַיּוֹם פּוֹדִים אֶת הַמַּעֲשֵׂר הַשֵּׁנִי עַל פְּרוּטָה בְּמַטְבֵּעַ שֶׁמְּיַחֲדִים לְכָךְ.
"בִּכּוּרִים": מִי שֶׁרָאָה בְּשָׂדֵהוּ אֶת הַפֵּרוֹת הָרִאשׁוֹנִים מִשִּׁבְעַת הַמִּינִים, קָשַׁר עֲלֵיהֶם חוּט וְאָמַר "הֲרֵי אֵלּוּ בִּכּוּרִים". כְּשֶׁהִבְשִׁילוּ – הֶעֱלָה אוֹתָם לַכֹּהֵן בַּמִּקְדָּשׁ, בְּשִׁירָה וּבְשִׂמְחָה.
💛 אַהֲבַת הַשֵּׁם: "לְמַעַן תִּלְמַד לְיִרְאָה אֶת ה׳ אֱלֹהֶיךָ כָּל הַיָּמִים" – הַמַּעֲשֵׂר הֵבִיא מִשְׁפָּחוֹת לִירוּשָׁלַיִם וְלַתּוֹרָה. וְהַבִּכּוּרִים לִמְּדוּ לְהַקְדִּים לַה׳ אֶת הָרִאשׁוֹן וְהַטּוֹב בְּיוֹתֵר.
📖 פֶּרֶק ז׳, הֲלָכוֹת ט׳, י״א, ט״ו; פֶּרֶק ט׳, הֲלָכָה י״ב

🧠 בְּדֹק אֶת עַצְמְךָ

בַּעַל פַּרְדֵּס גָּמַר לֶאֱסֹף אֶת הַפֵּרוֹת. מָה מַפְרִישִׁים רִאשׁוֹן, וּמַדּוּעַ?
💡 קֹדֶם תְּרוּמָה גְּדוֹלָה לַכֹּהֵן, אַחַר כָּךְ מַעֲשֵׂר רִאשׁוֹן לַלֵּוִי, וְרַק אַחֲרָיו מַעֲשֵׂר שֵׁנִי אוֹ עָנִי. הַמַּתָּנוֹת נִתְּנוּ לַכֹּהֲנִים וְלַלְוִיִּם כְּדֵי שֶׁיִּהְיוּ פְּנוּיִים לְלַמֵּד תּוֹרָה.
מַה מְּיֻחָד בַּשָּׁנָה הַשְּׁלִישִׁית וּבַשָּׁנָה הַשִּׁשִּׁית שֶׁל מַחֲזוֹר הַשְּׁמִטָּה?
💡 בַּשָּׁנִים הָרִאשׁוֹנָה, הַשְּׁנִיָּה, הָרְבִיעִית וְהַחֲמִישִׁית מַפְרִישִׁים מַעֲשֵׂר שֵׁנִי, וּבַשְּׁלִישִׁית וּבַשִּׁשִּׁית – מַעֲשֵׂר עָנִי. בַּשָּׁנָה הַשְּׁבִיעִית הַפֵּרוֹת הֶפְקֵר וּפְטוּרִים.
מָה עָשָׂה חַקְלַאי כְּשֶׁרָאָה אֶת הַתְּאֵנִים הָרִאשׁוֹנוֹת מַתְחִילוֹת לִגְדֹּל בְּשָׂדֵהוּ?
💡 בִּכּוּרִים מְבִיאִים מִשִּׁבְעַת הַמִּינִים: מְסַמְּנִים בְּחוּט אֶת הַפֵּרוֹת הָרִאשׁוֹנִים, וּכְשֶׁהֵם מַבְשִׁילִים מַעֲלִים אוֹתָם בְּשִׂמְחָה לַכֹּהֵן בַּמִּקְדָּשׁ.

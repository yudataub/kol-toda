מַתְּנוֹת עֲנִיִּים

🍽️ הִלְכוֹת כַּשְׁרוּת · פֶּרֶק ו׳

🌾חָמֵשׁ מַתְּנוֹת עֲנִיִּים – לֶקֶט

הַתּוֹרָה צִוְּתָה לְשַׁתֵּף אֶת הָעֲנִיִּים בִּבְרָכַת הַשָּׂדֶה, בְּחָמֵשׁ מַתָּנוֹת: לֶקֶט, שִׁכְחָה, פֵּאָה, פֶּרֶט וְעוֹלְלוֹת. "לֶקֶט" – שִׁבֳּלִים שֶׁנָּפְלוּ בִּזְמַן הַקָּצִיר: אִם נָפְלוּ אַחַת אוֹ שְׁתַּיִם – מַשְׁאִירִים אוֹתָן לָעֲנִיִּים.
אִם נָפְלוּ שָׁלֹשׁ שִׁבֳּלִים יַחַד אוֹ יוֹתֵר – הֵן שֶׁל בַּעַל הַשָּׂדֶה.
💛 אַהֲבַת הַשֵּׁם: "לֶעָנִי וְלַגֵּר תַּעֲזֹב אֹתָם, אֲנִי ה׳ אֱלֹהֵיכֶם". חֲכָמִים אָמְרוּ שֶׁמִּי שֶׁנּוֹתֵן מַתְּנוֹת עֲנִיִּים כָּרָאוּי – כְּאִלּוּ בָּנָה אֶת בֵּית הַמִּקְדָּשׁ וְהִקְרִיב בּוֹ קָרְבָּנוֹת.
📖 פֶּרֶק ו׳, הֲלָכוֹת א׳–ב׳, ה׳

🌾🔑 מוּשָּׂג: פֵּאָה

״לֹא תְכַלֶּה פְּאַת שָׂדְךָ לִקְצֹר... לֶעָנִי וְלַגֵּר תַּעֲזֹב אֹתָם״.
📖 פֵּאָה – הַפִּנָּה שֶׁל הַשָּׂדֶה, שֶׁמַּשְׁאִירִים בִּלְתִּי קְצוּרָה לַעֲנִיִּים.
💛 אַהֲבַת הַשֵּׁם: הַחַקְלַאי מַשְׁאִיר חֵלֶק מִיְּבוּלוֹ לְאֵלֶּה שֶׁאֵין לָהֶם, וּבְכָךְ מְשַׁתֵּף אוֹתָם בְּבִרְכַּת ה׳. 🧒 וְאַתָּה? כְּשֶׁיֵּשׁ לְךָ הַרְבֵּה מִמַּשֶּׁהוּ – תַּשְׁאִיר חֵלֶק לַחֲבֵרִים.
📖 פֶּרֶק ו׳, הֲלָכָה א׳

👐🔑 מוּשָּׂג: חֲמֵשׁ מַתְּנוֹת עֲנִיִּים

״כְּלַל הַמַּתָּנוֹת הַלָּלוּ חָמֵשׁ: פֵּאָה, שִׁכְחָה, לֶקֶט, פֶּרֶט וְעוֹלְלוֹת״.
📖 חֲמֵשׁ מַתְּנוֹת עֲנִיִּים – חֲמִשָּׁה סוּגִים שֶׁל יְבוּל הַשָּׂדֶה וְהַכֶּרֶם, שֶׁהַחַקְלַאי מַשְׁאִיר לָעֲנִיִּים.
💛 אַהֲבַת הַשֵּׁם: חֲמֵשׁ דְּרָכִים שׁוֹנוֹת לִתֵּן לָעֲנִיִּים, בְּלִי שֶׁיִּצְטָרְכוּ לְבַקֵּשׁ וּבְלִי שֶׁהַנּוֹתֵן יֵדַע מִי לוֹקֵחַ. 🧒 וְאַתָּה? תַּחְשֹׁב עַל דֶּרֶךְ לָתֵת צְדָקָה בְּלִי שֶׁמִּישֶׁהוּ יֵדַע שֶׁזֶּה מִמְּךָ.
📖 פֶּרֶק ו׳, הֲלָכָה א׳

🌾🔑 מוּשָּׂג: לֶקֶט

״וְאִם יִפְּלוּ שִׁבֳּלִים בּוֹדְדוֹת בְּעֵת הַקָּצִיר, שֶׁהֵם הַלֶּקֶט... יַנִּיחֵם לָעֲנִיִּים״.
📖 לֶקֶט – שִׁבֳּלִים בּוֹדְדוֹת שֶׁנָּפְלוּ מִיַּד הַקּוֹצֵר, שֶׁמַּשְׁאִירִים לָעֲנִיִּים.
💛 אַהֲבַת הַשֵּׁם: גַּם דָּבָר קָטָן שֶׁנָּפַל בִּשְׁגָגָה שַׁיָּךְ לָעֲנִיִּים – ה׳ דּוֹאֵג גַּם לַפְּרָטִים הַקְּטַנִּים. 🧒 וְאַתָּה? כְּשֶׁמַּשֶּׁהוּ קָטָן נוֹפֵל וְלֹא חָסֵר לְךָ – תֵּן אוֹתוֹ לְמִי שֶׁצָּרִיךְ.
📖 פֶּרֶק ו׳, הֲלָכָה א׳

🤝לַעֲזֹר לֶעָנִי בְּכָבוֹד

הַתּוֹרָה מְלַמֶּדֶת לַעֲזֹר לָעֲנִיִּים בְּאַרְבַּע דְּרָכִים: מַתָּנוֹת שֶׁהֵם לוֹקְטִים בְּעַצְמָם בַּשָּׂדֶה, צְדָקָה, מַעֲשֵׂר עָנִי, וְשִׁתּוּף בַּשְּׂמָחוֹת. הַצְּדָקָה הַגְּדוֹלָה בְּיוֹתֵר – לַעֲזֹר לֶעָנִי לִמְצֹא עֲבוֹדָה. וְגַם עֲנִיִּים גּוֹיִים שֶׁבָּאוּ לִלְקֹט יַחַד עִם עֲנִיֵּי יִשְׂרָאֵל – נוֹתְנִים לָהֶם, מִפְּנֵי דַּרְכֵי שָׁלוֹם.
הַמַּתָּנוֹת הֵן לָעֲנִיִּים, "וְלֹא לָעוֹרְבִים וְלָעֲטַלֵּפִים". הַיּוֹם הַשָּׂדוֹת רְחוֹקִים, וְלֹא מִשְׁתַּלֵּם לָעֲנִיִּים לָבוֹא לִלְקֹט – וְלָכֵן כִּמְעַט אֵין מַשְׁאִירִים מַתָּנוֹת בַּשָּׂדוֹת. אֲבָל לוֹמְדִים מֵהֶן אֵיךְ לַעֲזֹר הַיּוֹם.
💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אֶת הֶעָנִי וְרוֹצֶה שֶׁיַּרְגִּישׁ חָשׁוּב וּמְכֻבָּד. כְּשֶׁאֲנַחְנוּ עוֹזְרִים בְּחָכְמָה וּבְכָבוֹד, אֲנַחְנוּ מַמְשִׁיכִים אֶת קְדֻשַּׁת הַמִּקְדָּשׁ לְכָל מָקוֹם.
📖 פֶּרֶק ו׳, הֲלָכוֹת ב׳, ח׳–ט׳

🌾פֵּאָה

"לֹא תְכַלֶּה פְּאַת שָׂדְךָ" – בְּסוֹף הַקָּצִיר מַשְׁאִירִים לָעֲנִיִּים חֵלֶק מִן הַשָּׂדֶה שֶׁלֹּא נִקְצַר. חֲכָמִים קָבְעוּ לְהַשְׁאִיר לְפָחוֹת אֶחָד מִשִּׁשִּׁים, וְיוֹתֵר כְּשֶׁיֵּשׁ עֲנִיִּים רַבִּים אוֹ בְּרָכָה גְּדוֹלָה.
הַפֵּאָה נִתֶּנֶת דַּוְקָא בַּסּוֹף: כְּדֵי שֶׁהָעֲנִיִּים יֵדְעוּ מָתַי לָבוֹא, וּכְדֵי שֶׁבַּעַל הַשָּׂדֶה לֹא יִתֵּן רַק לַחֲבֵרוֹ אוֹ רַק מִן הַגָּרוּעַ.
💛 אַהֲבַת הַשֵּׁם: בְּסוֹף הַקָּצִיר, כְּשֶׁהָעֲבוֹדָה נִגְמֶרֶת, קַל יוֹתֵר לִהְיוֹת נָדִיב. כָּךְ הַשָּׂדֶה מְסַיֵּם אֶת הָעוֹנָה בְּחֶסֶד.
📖 פֶּרֶק ו׳, הֲלָכָה ג׳

🌾שִׁכְחָה

חַקְלַאי שֶׁשָּׁכַח עֹמֶר תְּבוּאָה אוֹ פֵּרוֹת בַּשָּׂדֶה – לֹא חוֹזֵר לָקַחַת אוֹתָם, אֶלָּא מַשְׁאִיר אוֹתָם לָעֲנִיִּים: "וְשָׁכַחְתָּ עֹמֶר בַּשָּׂדֶה – לֹא תָשׁוּב לְקַחְתּוֹ".
זוֹ מִצְוָה מְיֻחֶדֶת, שֶׁמְּקַיְּמִים אוֹתָהּ בְּלִי לְהִתְכַּוֵּן – ה׳ מְזַמֵּן אוֹתָהּ. חָסִיד אֶחָד שָׁכַח עֹמֶר, וְשָׂמַח כָּל כָּךְ עַד שֶׁהֵבִיא קָרְבָּנוֹת לְתוֹדָה.
💛 אַהֲבַת הַשֵּׁם: "לְמַעַן יְבָרֶכְךָ ה׳ אֱלֹהֶיךָ בְּכֹל מַעֲשֵׂה יָדֶיךָ" – זוֹ הַהַבְטָחָה עַל הַשִּׁכְחָה. גַּם כְּשֶׁאֲנַחְנוּ שׁוֹכְחִים, ה׳ יָכוֹל לַהֲפֹךְ אֶת זֶה לְמִצְוָה.
📖 פֶּרֶק ו׳, הֲלָכָה ד׳

🍇פֶּרֶט וְעוֹלְלוֹת

בַּכֶּרֶם יֵשׁ עוֹד שְׁתֵּי מַתָּנוֹת. "פֶּרֶט" – עֲנָבִים בּוֹדְדִים שֶׁנָּפְלוּ בִּזְמַן הַבָּצִיר: אֶחָד אוֹ שְׁנַיִם – לָעֲנִיִּים. "עוֹלְלוֹת" – אֶשְׁכּוֹלוֹת קְטַנְטַנִּים, כְּמוֹ תִּינוֹק לְעֻמַּת אָדָם גָּדוֹל – מַשְׁאִירִים לָעֲנִיִּים.
בָּעֲנָבִים מַשְׁאִירִים לָעֲנִיִּים יוֹתֵר מִבְּכָל פְּרִי אַחֵר, כְּדֵי שֶׁגַּם הֵם יוּכְלוּ לִשְׂמֹחַ בְּיַיִן.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה רוֹצָה שֶׁהֶעָנִי יְקַבֵּל לֹא רַק לֶחֶם, אֶלָּא גַּם שִׂמְחָה. מִי שֶׁמְּשַׂמֵּחַ עָנִי – עוֹשֶׂה נַחַת רוּחַ לַה׳.
📖 פֶּרֶק ו׳, הֲלָכָה ו׳

⚖️כָּל הָעֲנִיִּים שָׁוִים

בַּעַל הַשָּׂדֶה פּוֹתֵחַ אֶת שָׂדֵהוּ לְכָל הָעֲנִיִּים, וְאָסוּר לוֹ לְהַעֲדִיף עָנִי אֶחָד עַל חֲבֵרוֹ. מִי שֶׁלֹּא נָתַן לְעָנִי לְהִכָּנֵס – גָּזַל אֶת חֶלְקוֹ. גַּם בַּעַל שָׂדֶה עָנִי חַיָּב לְהַשְׁאִיר מַתָּנוֹת לָעֲנִיִּים אֲחֵרִים.
הָעֲנִיִּים לוֹקְטִים אֶת הַמַּתָּנוֹת בְּעַצְמָם, וּבַעַל הַשָּׂדֶה לֹא קוֹטֵף בִּשְׁבִיל עָנִי מְסֻיָּם. חֲכָמִים קָבְעוּ שָׁלֹשׁ שָׁעוֹת בַּיּוֹם לִפְתִּיחַת הַשָּׂדוֹת, כְּדֵי שֶׁגַּם יְלָדִים וּזְקֵנִים יַסְפִּיקוּ לָבוֹא.
💛 אַהֲבַת הַשֵּׁם: בְּעֵינֵי ה׳ כָּל עָנִי חָשׁוּב. כְּשֶׁאֲנַחְנוּ נוֹתְנִים בְּיֹשֶׁר לְכֻלָּם, אֲנַחְנוּ הוֹלְכִים בִּדְרָכָיו שֶׁל ה׳, שֶׁזָּן אֶת כָּל הַבְּרִיּוֹת.
📖 פֶּרֶק ו׳, הֲלָכוֹת ז׳–ח׳

🧠 בְּדֹק אֶת עַצְמְךָ

בַּעַל שָׂדֶה רוֹצֶה שֶׁרַק בֶּן דּוֹדוֹ הֶעָנִי יִקַּח אֶת הַפֵּאָה. מַה הַדִּין?
💡 מַתְּנוֹת עֲנִיִּים שַׁיָּכוֹת לְכָל הָעֲנִיִּים, וּמִי שֶׁמּוֹנֵעַ מֵעָנִי לְהִכָּנֵס – גּוֹזֵל אֶת חֶלְקוֹ. וְגַם אָסוּר לִקְטֹף בִּשְׁבִיל עָנִי מְסֻיָּם – הָעֲנִיִּים לוֹקְטִים בְּעַצְמָם.
📜 לְשׁוֹן חֲכָמִים: כַּמָּה סוּגֵי מַתָּנוֹת לָעֲנִיִּים יֵשׁ בְּיֵבוּל הַשָּׂדֶה וְהַכֶּרֶם?
💡 חֲכָמִים מָנוּ חָמֵשׁ מַתָּנוֹת: פֵּאָה, שִׁכְחָה, לֶקֶט, פֶּרֶט וְעוֹלְלוֹת – כֻּלָּן נִשְׁאָרוֹת לָעֲנִיִּים.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

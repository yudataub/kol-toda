🌿הוֹשַׁעְנָא רַבָּה

הַיּוֹם הַשְּׁבִיעִי שֶׁל סֻכּוֹת נִקְרָא ״הוֹשַׁעְנָא רַבָּה״. בְּסֻכּוֹת נִדּוֹנִים עַל הַמַּיִם, וּבַיּוֹם הַזֶּה נֶחְתָּם הַדִּין – וְלָכֵן מַרְבִּים בִּתְפִלּוֹת ״הוֹשַׁע נָא״. בּוֹ מַקִּיפִים אֶת הַבִּימָה שֶׁבַע פְּעָמִים.
הַמִּנְהָג הָרוֹוֵחַ – לִלְמֹד תּוֹרָה בְּלֵיל הוֹשַׁעְנָא רַבָּה, וְיֵשׁ נוֹהֲגִים לְהִשָּׁאֵר עֵרִים כָּל הַלַּיְלָה. וּבַיּוֹם עוֹרְכִים סְעוּדָה חֲשׁוּבָה.
💛 אַהֲבַת הַשֵּׁם: גַּם אַחֲרֵי יוֹם הַכִּפּוּרִים, ה׳ מַמְשִׁיךְ לְחַכּוֹת לַתְּשׁוּבָה שֶׁלָּנוּ. בְּהוֹשַׁעְנָא רַבָּה עוֹד אֶפְשָׁר לְהֵיטִיב אֶת הַדִּין.

🗓️הַשָּׁלָב הָאַחֲרוֹן שֶׁל הַדִּין

שְׁלֹשָׁה זְמַנִּים לַדִּין: רֹאשׁ הַשָּׁנָה, יוֹם הַכִּפּוּרִים וְהוֹשַׁעְנָא רַבָּה. גְּזַר הַדִּין נֶחְתָּם בְּיוֹם הַכִּפּוּרִים, וּבְהוֹשַׁעְנָא רַבָּה נִכְתָּבִים הַ״פְּתָקִים״ וְנִמְסָרִים לַשְּׁלִיחִים. עַד אָז עוֹד אֶפְשָׁר לְשַׁנּוֹת לְטוֹבָה, וְלָכֵן נָכוֹן לַחֲזֹר בִּתְשׁוּבָה בַּיּוֹם הַזֶּה.
״אֵין מַיִם אֶלָּא תּוֹרָה״ – הַדִּין עַל הַמַּיִם כּוֹלֵל אֶת כָּל חַיֵּי הָאָדָם, בַּגּוּף וּבַנְּשָׁמָה. וְעַד סוֹף שְׁמִינִי עֲצֶרֶת עוֹד מוֹעִילָה הַתְּשׁוּבָה.
💛 אַהֲבַת הַשֵּׁם: ה׳ אֵינוֹ מְמַהֵר לִסְגֹּר אֶת הַדֶּלֶת. הוּא מְחַכֶּה לָנוּ עוֹד וָעוֹד, כְּמוֹ אַבָּא שֶׁרוֹצֶה רַק טוֹב לִבְנוֹ.

📜📜 לְשׁוֹן חֲכָמִים: הוֹשַׁעְנָא רַבָּה

״הַיּוֹם הַשְּׁבִיעִי שֶׁל סֻכּוֹת נִקְרָא ׳הוֹשַׁעְנָא רַבָּה׳... נֶחְתָּם הַדִּין, וְלָכֵן מַרְבִּים בִּתְפִלּוֹת ׳הוֹשַׁע נָא׳״.
📖 הוֹשַׁעְנָא רַבָּה – ״הוֹשַׁע נָא״ בְּרִבּוּי גָּדוֹל: הַיּוֹם הַשְּׁבִיעִי, שֶׁבּוֹ מַרְבִּים לְבַקֵּשׁ מֵה׳ יְשׁוּעָה.
💛 אַהֲבַת הַשֵּׁם: גַּם אַחֲרֵי יוֹם הַכִּפּוּרִים ה׳ מַמְשִׁיךְ לְחַכּוֹת לַתְּשׁוּבָה שֶׁלָּנוּ. 🧒 וְאַתָּה? בְּהוֹשַׁעְנָא רַבָּה – בַּקֵּשׁ מֵה׳ עוֹד פַּעַם אַחַת עַל דָּבָר שֶׁחָשׁוּב לְךָ.

🌾חִבּוּט עֲרָבָה

בְּהוֹשַׁעְנָא רַבָּה נוֹטְלִים עֲרָבוֹת – זֵכֶר לַמִּקְדָּשׁ – וְחוֹבְטִים בָּהֶן, כְּמוֹ שֶׁהִנְהִיגוּ הַנְּבִיאִים. נוֹהֲגִים, עַל פִּי הָאֲרִ״י, לִטֹּל חָמֵשׁ עֲרָבוֹת קְשׁוּרוֹת וְלַחְבֹּט בָּהֶן חָמֵשׁ פְּעָמִים עַל הַקַּרְקַע. זֶה מִנְהָג, וְאֵין מְבָרְכִים עָלָיו.
אֹרֶךְ הָעֲרָבָה – שְׁלֹשָׁה טְפָחִים, כְּמוֹ הָעֲרָבָה שֶׁבַּלּוּלָב. אֲבָל בָּעֲרָבָה שֶׁקְּשׁוּרָה לַלּוּלָב אֵין חוֹבְטִים.
💛 אַהֲבַת הַשֵּׁם: הָעֲרָבָה, שֶׁאֵין לָהּ טַעַם וְרֵיחַ, רוֹמֶזֶת לְבַעַל תְּשׁוּבָה שֶׁיּוֹדֵעַ שֶׁכָּל יְשׁוּעָתוֹ מֵה׳. ״בִּמְקוֹם שֶׁבַּעֲלֵי תְּשׁוּבָה עוֹמְדִים – צַדִּיקִים גְּמוּרִים אֵינָם עוֹמְדִים״.

🌿עֲרָבָה בַּמִּקְדָּשׁ

בַּמִּקְדָּשׁ הֵבִיאוּ עֲרָבוֹת אֲרֻכּוֹת וְזָקְפוּ אוֹתָן בְּצִדֵּי הַמִּזְבֵּחַ, וְרָאשֵׁיהֶן כְּפוּפִים עַל גַּבֵּי הַמִּזְבֵּחַ – הֲלָכָה לְמֹשֶׁה מִסִּינַי. הַנְּבִיאִים הִנְהִיגוּ לִטֹּל עֲרָבוֹת בְּכָל מָקוֹם וְלַחְבֹּט בָּהֶן, וְאַחֲרֵי הַחֻרְבָּן מַמְשִׁיכִים בַּיּוֹם הַשְּׁבִיעִי – זֵכֶר לַמִּקְדָּשׁ.
מִנְהַג הָעֲרָבָה חָשׁוּב כָּל כָּךְ, שֶׁהַלּוּחַ הָעִבְרִי נִקְבַּע כָּךְ שֶׁהוֹשַׁעְנָא רַבָּה לְעוֹלָם לֹא יָחוּל בְּשַׁבָּת – כְּדֵי שֶׁהַמִּנְהָג לֹא יִתְבַּטֵּל.
💛 אַהֲבַת הַשֵּׁם: הָעֲרָבָה הַפְּשׁוּטָה מְלַמֶּדֶת עֲנָוָה: אַחֲרֵי כָּל הַתְּפִלּוֹת, אָנוּ בָּאִים לִפְנֵי ה׳ וְיוֹדְעִים שֶׁכָּל יְשׁוּעָתֵנוּ מִמֶּנּוּ – וְלָכֵן הָעֲרָבוֹת נִקְרָאוֹת ״הוֹשַׁעְנוֹת״.

🌾🔑 מוּשָּׂג: חִבּוּט עֲרָבָה

״בְּהוֹשַׁעְנָא רַבָּה נוֹטְלִים עֲרָבוֹת... וְחוֹבְטִים בָּהֶן, כְּמוֹ שֶׁהִנְהִיגוּ הַנְּבִיאִים״.
📖 חִבּוּט – הַכָּאָה קַלָּה שֶׁל הָעֲרָבוֹת עַל הַקַּרְקַע, מִנְהָג מִימֵי הַנְּבִיאִים, בְּלִי בְּרָכָה.
💛 אַהֲבַת הַשֵּׁם: הָעֲרָבָה, שֶׁאֵין לָהּ טַעַם וְרֵיחַ, רוֹמֶזֶת לְבַעַל תְּשׁוּבָה הַיּוֹדֵעַ שֶׁכָּל יְשׁוּעָתוֹ מֵה׳. 🧒 וְאַתָּה? בְּחִבּוּט הָעֲרָבָה – תַּחְשֹׁב עַל דָּבָר אֶחָד שֶׁתְּשַׁפֵּר בַּשָּׁנָה הַבָּאָה.

🍂אַחֲרֵי הֶחָג

אַחֲרֵי הַחַג מֻתָּר לְהִשְׁתַּמֵּשׁ בָּאַרְבָּעָה מִינִים וּבַעֲצֵי הַסֻּכָּה, וּמֻתָּר לֶאֱכֹל אֶת הָאֶתְרוֹג. הֵם ״תַּשְׁמִישֵׁי מִצְוָה״, וְאֵינָם צְרִיכִים גְּנִיזָה – אֲבָל אָסוּר לְבַזּוֹת אוֹתָם. מַנִּיחִים אוֹתָם בְּצַד הֶחָצֵר, בַּמָּקוֹם שֶׁמַּנִּיחִים עֲנָפִים גְּזוּמִים, וְלֹא בְּאַשְׁפָּה מְטֻנֶּפֶת.
יֵשׁ נוֹהֲגִים לִשְׁמֹר אֶת הָעֲרָבוֹת עַד פֶּסַח, כְּדֵי לִשְׂרֹף בָּהֶן אֶת הֶחָמֵץ. וְלֹא פּוֹסְעִים עַל עֲצֵי הַסֻּכָּה דֶּרֶךְ בִּזָּיוֹן.
💛 אַהֲבַת הַשֵּׁם: מִצְוָה מַשְׁאִירָה חוֹתָם שֶׁל קְדֻשָּׁה. גַּם אַחֲרֵי שֶׁסִּיַּמְנוּ – אָנוּ מְכַבְּדִים אֶת מָה שֶׁשִּׁמֵּשׁ אוֹתָנוּ לַעֲבוֹדַת ה׳.

🏠נִפְרָדִים מִן הַסֻּכָּה

לֹא מְפָרְקִים אֶת הַסֻּכָּה עַד סוֹף הַחַג – גַּם לֹא אַחֲרֵי הַסְּעוּדָה שֶׁל הוֹשַׁעְנָא רַבָּה, כִּי עַד הָעֶרֶב עוֹד יֵשׁ מִצְוָה לָשֶׁבֶת בָּהּ. כְּשָׁעָתַיִם וָחֵצִי לִפְנֵי הַשְּׁקִיעָה מֻתָּר לְפַנּוֹת אֶת הָרָהִיטִים לִקְרַאת שְׁמִינִי עֲצֶרֶת.
רָאוּי לָשֶׁבֶת בַּסֻּכָּה בַּשָּׁעוֹת הָאַחֲרוֹנוֹת, וְיֵשׁ נוֹהֲגִים לְנַשֵּׁק אֶת הַסֻּכָּה כְּשֶׁנִּפְרָדִים מִמֶּנָּה. הַסְּכָךְ, הַדְּפָנוֹת וְהַקִּשּׁוּטִים מֻקְצִים עַד סוֹף הַיּוֹם הַשְּׁמִינִי.
💛 אַהֲבַת הַשֵּׁם: קָשֶׁה לְהִפָּרֵד מִמִּצְוָה שֶׁאָהַבְנוּ שִׁבְעָה יָמִים. הַנְּשִׁיקָה לַסֻּכָּה אוֹמֶרֶת: נִתְרָאֶה בַּשָּׁנָה הַבָּאָה!

🧠 בְּדֹק אֶת עַצְמְךָ

לָמָּה הוֹשַׁעְנָא רַבָּה לְעוֹלָם אֵינוֹ חָל בְּשַׁבָּת?
💡 מִנְהַג הָעֲרָבָה חָשׁוּב כָּל כָּךְ, שֶׁהַלּוּחַ הָעִבְרִי נִקְבַּע כָּךְ שֶׁהוֹשַׁעְנָא רַבָּה לֹא יָחוּל בְּשַׁבָּת. כָּךְ אֶפְשָׁר לִטֹּל עֲרָבוֹת, לְהִתְחַנֵּן וְלִזְעֹק עַל הַמַּיִם.
מָה מְיֻחָד בְּיוֹם הוֹשַׁעְנָא רַבָּה, לְפִי מָה שֶׁאָמְרוּ חֲכָמִים?
💡 שְׁלֹשָׁה זְמַנִּים לַדִּין: רֹאשׁ הַשָּׁנָה, יוֹם הַכִּפּוּרִים וְהוֹשַׁעְנָא רַבָּה. בְּהוֹשַׁעְנָא רַבָּה נִכְתָּבִים הַפְּתָקִים שֶׁל גְּזַר הַדִּין, וְעַד שֶׁהַשְּׁלִיחִים יוֹצְאִים – עוֹד אֶפְשָׁר לְהֵיטִיב אוֹתוֹ בִּתְשׁוּבָה.
חִבּוּט הָעֲרָבָה – הַאִם מְבָרְכִים עָלָיו?
💡 חֲכָמִים נֶחְלְקוּ אִם הַנְּבִיאִים קָבְעוּ אֶת הַחִבּוּט כְּתַקָּנָה שֶׁמְּבָרְכִים עָלֶיהָ אוֹ כְּמִנְהָג. לַהֲלָכָה – זֶה מִנְהָג, וְאֵין מְבָרְכִים עָלָיו. וּבָעֲרָבָה שֶׁקְּשׁוּרָה לַלּוּלָב אֵין חוֹבְטִים.

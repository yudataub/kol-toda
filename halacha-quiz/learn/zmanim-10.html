הִלְכוֹת תִּשְׁעָה בְּאָב

📅 רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת · פֶּרֶק י׳

⛔חֲמִשָּׁה עִנּוּיִים

צוֹם תִּשְׁעָה בְּאָב נִמְשָׁךְ יְמָמָה שְׁלֵמָה – מִשְּׁקִיעַת הַחַמָּה וְעַד צֵאת הַכּוֹכָבִים. אָסוּר בּוֹ בַּחֲמִשָּׁה דְּבָרִים, כְּמוֹ בְּיוֹם הַכִּפּוּרִים: אֲכִילָה וּשְׁתִיָּה, רְחִיצָה, סִיכָה, נְעִילַת הַסַּנְדָּל וְתַשְׁמִישׁ הַמִּטָּה.
בַּצּוֹמוֹת הָאֲחֵרִים אָסוּר רַק לֶאֱכֹל וְלִשְׁתּוֹת, וְרַק בַּיּוֹם. תִּשְׁעָה בְּאָב חָמוּר יוֹתֵר, כִּי בּוֹ נֶחְרְבוּ שְׁנֵי בָּתֵּי הַמִּקְדָּשׁ.
💛 אַהֲבַת הַשֵּׁם: בְּיוֹם הַכִּפּוּרִים מִתְעַנִּים וְלוֹבְשִׁים בְּגָדִים נָאִים, וּבְתִשְׁעָה בְּאָב מִתְעַנִּים וּמִתְאַבְּלִים. גַּם הָאֵבֶל הוּא אַהֲבָה – מִי שֶׁאוֹהֵב, מִתְגַּעְגֵּעַ.
📖 פֶּרֶק י׳, הֲלָכוֹת א׳–ב׳

🤒חוֹלִים, וְיוֹם הַכִּפּוּרִים

תִּשְׁעָה בְּאָב דּוֹמֶה לְיוֹם הַכִּפּוּרִים, אֲבָל קַל מִמֶּנּוּ: חוֹלֶה פָּטוּר מִן הַצּוֹם לְגַמְרֵי, וְאֵינוֹ צָרִיךְ לֶאֱכֹל פָּחוֹת פָּחוֹת מִכַּשִּׁעוּר. הוּא אוֹכֵל דְּבָרִים פְּשׁוּטִים, לֹא מַעֲדַנִּים. וּבְכָל סָפֵק – מְקִלִּים, כִּי הַצּוֹם מִדִּבְרֵי חֲכָמִים.
מִצַּד שֵׁנִי, תִּשְׁעָה בְּאָב חָמוּר מִיּוֹם הַכִּפּוּרִים, כִּי הוּא יוֹם אֵבֶל: יוֹשְׁבִים עַל הָאָרֶץ, לֹא אוֹמְרִים שָׁלוֹם, מַחְשִׁיכִים אֶת הָאוֹר וְלֹא לוֹמְדִים תּוֹרָה מְשַׂמַּחַת. בְּיוֹם הַכִּפּוּרִים לוֹבְשִׁים בְּגָדִים נָאִים וְשָׁרִים. וְאִשָּׁה שֶׁמְּכִינָה אֹכֶל לַיְּלָדִים אוֹ לְסוֹף הַצּוֹם רַשָּׁאִית לִשְׁטֹף מַאֲכָלִים וְכֵלִים, כִּי אֵין זוֹ רְחִיצָה שֶׁל תַּעֲנוּג.
💛 אַהֲבַת הַשֵּׁם: בְּיוֹם הַכִּפּוּרִים אָנוּ שְׂמֵחִים בַּכַּפָּרָה, וּבְתִשְׁעָה בְּאָב – מִתְגַּעְגְּעִים לְבֵית אָבִינוּ. שְׁנֵי הַיָּמִים מְקָרְבִים אוֹתָנוּ לַה׳.
📖 פֶּרֶק י׳, הֲלָכוֹת ב׳–ג׳, ה׳

🚫🔑 מוּשָּׂג: חֲמֵשֶׁת הָעִנּוּיִים

״בְּצוֹם תִּשְׁעָה בְּאָב אֲסוּרִים בְּכָל חֲמֵשֶׁת הָעִנּוּיִים, שֶׁהֵם: אֲכִילָה וּשְׁתִיָּה, רְחִיצָה, סִיכָה, נְעִילַת הַסַּנְדָּל, תַּשְׁמִישׁ הַמִּטָּה״.
📖 חֲמֵשֶׁת הָעִנּוּיִים – חֲמִשָּׁה אִסּוּרִים שֶׁל צוֹם תִּשְׁעָה בְּאָב, כְּמוֹ בְּיוֹם הַכִּפּוּרִים.
💛 אַהֲבַת הַשֵּׁם: תִּשְׁעָה בְּאָב דּוֹמֶה לְיוֹם הַכִּפּוּרִים בְּחֻמְרָתוֹ – ה׳ מְלַמֵּד אוֹתָנוּ שֶׁחֻרְבַּן בֵּית הַמִּקְדָּשׁ חָמוּר כְּמוֹ יוֹם הַדִּין. 🧒 וְאַתָּה? בְּתִשְׁעָה בְּאָב – שִׂים לֵב לְכָל אֶחָד מֵחֲמֵשֶׁת הָעִנּוּיִים, וְתָבִין כַּמָּה הַיּוֹם מְיֻחָד.
📖 פֶּרֶק י׳, הֲלָכָה ב׳

🚿בְּלִי רְחִיצָה וּבְלִי סִיכָה

בְּתִשְׁעָה בְּאָב אָסוּר לִרְחֹץ לְהַנָּאָה – לֹא בְּמַיִם חַמִּים וְלֹא בְּקָרִים, וַאֲפִלּוּ אֶצְבַּע אַחַת. אֲבָל מִי שֶׁהִתְלַכְלֵךְ בְּבֹץ, אוֹ שֶׁיָּצָא לוֹ דָּם מִן הָאַף – רוֹחֵץ אֶת הַמָּקוֹם הַמְלֻכְלָךְ, כִּי אֵינוֹ מִתְכַּוֵּן לְתַעֲנוּג.
בַּבֹּקֶר נוֹטְלִים יָדַיִם שָׁלֹשׁ פְּעָמִים – רַק עַד סוֹף קִשְׁרֵי הָאֶצְבָּעוֹת. אָסוּר לָסוּךְ קְרֵם אוֹ שֶׁמֶן לְתַעֲנוּג, אֲבָל מִשְׁחָה לִרְפוּאָה אוֹ נֶגֶד יַתּוּשִׁים – מֻתֶּרֶת.
💛 אַהֲבַת הַשֵּׁם: בַּיּוֹם הַזֶּה אָנוּ מְוַתְּרִים עַל הַנְּעִימוּת שֶׁל הַגּוּף, כְּדֵי שֶׁהַלֵּב יַרְגִּישׁ אֶת הַצַּעַר עַל בֵּית ה׳.
📖 פֶּרֶק י׳, הֲלָכוֹת ה׳–ז׳

🧴📜 לְשׁוֹן חֲכָמִים: סִיכָה

״אָסוּר לָסוּך אֲפִלּוּ מָקוֹם קָטָן שֶׁבַּגּוּף בְּשֶׁמֶן אוֹ מִשְׁחָה... אֵין מְרִיחִים בְּשָׂמִים בְּתִשְׁעָה בְּאָב, מִפְּנֵי שֶׁיֵּשׁ בְּזֶה תַּעֲנוּג״.
📖 סִיכָה – מְרִיחַת שֶׁמֶן, קְרֵם אוֹ בֹּשֶׂם לְשֵׁם תַּעֲנוּג. אֲסוּרָה בְּתִשְׁעָה בְּאָב, אֶחָד מֵחֲמֵשֶׁת הָעִנּוּיִים.
💛 אַהֲבַת הַשֵּׁם: בְּתִשְׁעָה בְּאָב מְבַטְּלִים אֲפִלּוּ תַּעֲנוּגִים קְטַנִּים, כְּדֵי שֶׁנַּרְגִּישׁ בַּגּוּף שֶׁזֶּה לֹא יוֹם רָגִיל. 🧒 וְאַתָּה? בְּתִשְׁעָה בְּאָב – דְּחֵה אֶת הַבֹּשֶׂם לְיוֹם אַחֵר; קְרֵם לִרְפוּאָה בִּלְבַד מֻתָּר.
📖 פֶּרֶק י׳, הֲלָכָה ז׳

👟בְּלִי נַעֲלַיִם

אָסוּר לִנְעֹל בְּתִשְׁעָה בְּאָב נַעֲלֵי עוֹר. בְּנַעֲלַיִם שֶׁאֵינָן מֵעוֹר – רֹב הַפּוֹסְקִים מַתִּירִים. הָרַב מֶלָמֵד כּוֹתֵב שֶׁכַּיּוֹם, כְּשֶׁעוֹשִׂים נַעֲלַיִם טוֹבוֹת מִכָּל חֹמֶר, כָּל נַעַל שֶׁהוֹלְכִים בָּהּ בַּחוּץ עַל אֲבָנִים – אֲסוּרָה. נַעֲלֵי בַּיִת פְּשׁוּטוֹת מִבַּד אוֹ מִגּוּמִי – מֻתָּרוֹת לְכָל הַדֵּעוֹת.
גַּם לְדַעַת הָרַב מֶלָמֵד, אֵין לִמְחוֹת בְּיַד הַסּוֹמְכִים עַל הַמַּתִּירִים. חוֹלֶה, אוֹ מִי שֶׁהוֹלֵךְ בְּמָקוֹם שֶׁיֵּשׁ בּוֹ עַקְרַבִּים אוֹ בֹּץ, רַשַּׁאי לִנְעֹל נַעֲלַיִם.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁמַּרְגִּישִׁים אֶת הָאֲבָנִים מִתַּחַת לָרַגְלַיִם, זוֹכְרִים אֶת אַבְנֵי יְרוּשָׁלַיִם הַחֲרֵבָה – וּמִתְפַּלְּלִים שֶׁתִּבָּנֶה.
📖 פֶּרֶק י׳, הֲלָכָה ח׳

🔀שְׁתֵּי שִׁיטוֹת – וּשְׁתֵּיהֶן כַּהֲלָכָה

יְשִׁיבָה עַל הָאָרֶץ: הָאַשְׁכְּנַזִּים – עַד חֲצוֹת; חֵלֶק מִן הַסְּפָרַדִּים – עַד מִנְחָה. "עֲנֵנוּ" בִּתְפִלַּת הַלַּחַשׁ: הָאַשְׁכְּנַזִּים – רַק בְּמִנְחָה; הַסְּפָרַדִּים – בְּכָל שָׁלֹשׁ הַתְּפִלּוֹת. בִּרְכַּת "שֶׁעָשָׂה לִי כָּל צָרְכִּי": הָאַשְׁכְּנַזִּים אוֹמְרִים אוֹתָהּ; רֹב הַסְּפָרַדִּים אֵינָם אוֹמְרִים, כִּי אֵין נוֹעֲלִים נַעֲלַיִם. בָּשָׂר בַּעֲשִׂירִי בְּאָב: הָאַשְׁכְּנַזִּים – עַד חֲצוֹת; חֵלֶק מִן הַסְּפָרַדִּים – כָּל הַיּוֹם.
נַעֲלַיִם שֶׁאֵינָן מֵעוֹר: רֹב הַפּוֹסְקִים – מֻתָּר; הָרַב מֶלָמֵד – אָסוּר, אִם הוֹלְכִים בָּהֶן בַּחוּץ. "בֹּקֶר טוֹב": רֹב הַפּוֹסְקִים – אָסוּר; יֵשׁ אוֹמְרִים – מֻתָּר. וְכָל אֶחָד מַמְשִׁיךְ בְּמִנְהַג אֲבוֹתָיו וְרַבּוֹתָיו.
💛 אַהֲבַת הַשֵּׁם: כָּל עֵדָה מִתְאַבֶּלֶת עַל יְרוּשָׁלַיִם בַּדֶּרֶךְ שֶׁקִּבְּלָה מֵרַבּוֹתֶיהָ, וּשְׁתֵּי הַדְּרָכִים הֵן תּוֹרָה. כְּשֶׁתִּבָּנֶה – נִשְׂמַח כֻּלָּנוּ יַחַד.
📖 פֶּרֶק י׳, הֲלָכוֹת ח׳, י״א, י״ג, ט״ז, י״ח–י״ט

🩴📜 לְשׁוֹן חֲכָמִים: נְעִילַת הַסַּנְדָּל

״אָסוּר בְּתִשְׁעָה בְּאָב לִנְעוֹל סַנְדָּל אוֹ נַעַל... אֲבָל מֻתָּר לָלֶכֶת בְּנַעֲלֵי בַּיִת מִבַּד אוֹ מִגּוּמִי פְּשׁוּטוֹת״.
📖 נְעִילַת הַסַּנְדָּל – אֶחָד מֵחֲמֵשֶׁת הָעִנּוּיִים: אֵין נוֹעֲלִים נַעֲלֵי עוֹר רְגִילוֹת בְּתִשְׁעָה בְּאָב.
💛 אַהֲבַת הַשֵּׁם: בְּיוֹם שֶׁבּוֹ נֶחְרַב בֵּית הַמִּקְדָּשׁ, מְבַטְּלִים אֲפִלּוּ נוֹחִיּוּת קְטַנָּה כְּמוֹ נַעֲלַיִם טוֹבוֹת, כְּדֵי שֶׁנַּרְגִּישׁ בַּגּוּף אֶת הַצַּעַר. 🧒 וְאַתָּה? בְּתִשְׁעָה בְּאָב – לְבַשׁ נַעֲלֵי בַּיִת פְּשׁוּטוֹת, וְתִזְכּוֹר עַל מָה אַתָּה מִתְאַבֵּל.
📖 פֶּרֶק י׳, הֲלָכָה ח׳

📖מָה לוֹמְדִים בְּתִשְׁעָה בְּאָב?

דִּבְרֵי תּוֹרָה מְשַׂמְּחִים – "פִּקּוּדֵי ה׳ יְשָׁרִים מְשַׂמְּחֵי לֵב" – וְלָכֵן בְּתִשְׁעָה בְּאָב אָסוּר לִלְמֹד תּוֹרָה, כְּמוֹ אָבֵל. מֻתָּר וּמִצְוָה לִלְמֹד דְּבָרִים עֲצוּבִים: מְגִלַּת אֵיכָה, סֵפֶר אִיּוֹב, נְבוּאוֹת הַחֻרְבָּן בְּיִרְמְיָהוּ, אַגָּדוֹת הַחֻרְבָּן וְהִלְכוֹת תִּשְׁעָה בְּאָב.
אָסוּר לִקְרֹא בְּתִשְׁעָה בְּאָב סְפָרִים מְרַתְּקִים וְעִתּוֹנִים, אֲבָל רָאוּי לִקְרֹא סִפְרֵי הִיסְטוֹרְיָה עַל הַחֻרְבָּן וְעַל צָרוֹת יִשְׂרָאֵל.
💛 אַהֲבַת הַשֵּׁם: דַּוְקָא בַּיּוֹם שֶׁאָסוּר לִלְמֹד, מַרְגִּישִׁים כַּמָּה הַתּוֹרָה מְשַׂמַּחַת אוֹתָנוּ – וְכַמָּה אָנוּ מְחַכִּים לַחֲזֹר אֵלֶיהָ.

🤐בְּלִי "שָׁלוֹם"

כְּמוֹ אָבֵל, גַּם בְּתִשְׁעָה בְּאָב אֵין אוֹמְרִים "שָׁלוֹם" אִישׁ לְרֵעֵהוּ. לְדַעַת רֹב הַפּוֹסְקִים גַּם "בֹּקֶר טוֹב" אָסוּר, וְיֵשׁ אוֹמְרִים שֶׁמֻּתָּר. מִי שֶׁאֵינוֹ יוֹדֵעַ וְאוֹמֵר לָנוּ שָׁלוֹם – עוֹנִים לוֹ בְּשָׂפָה רָפָה וּבְכֹבֶד רֹאשׁ, וְנִזְהָרִים שֶׁלֹּא לִפְגֹּעַ בּוֹ.
מֻתָּר לְבָרֵךְ "מַזָּל טוֹב" וְלִלְחֹץ יָד. אֵין שׁוֹלְחִים מַתָּנוֹת, אֲבָל צְדָקָה – מִצְוָה, כְּמוֹ לִשְׁלֹחַ אֹכֶל לַעֲנִיִּים לְסוֹף הַצּוֹם.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁאָנוּ לֹא אוֹמְרִים "שָׁלוֹם", אָנוּ נִזְהָרִים מְאוֹד בִּכְבוֹד הֶחָבֵר. הָאֵבֶל עַל הַמִּקְדָּשׁ אֵינוֹ בָּא עַל חֶשְׁבּוֹן אַהֲבַת יִשְׂרָאֵל.

🏛️הַכֹּתֶל, טִיּוּלִים וּמְלָאכָה

בְּתִשְׁעָה בְּאָב לֹא מְטַיְּלִים וְלֹא מְשׂוֹחֲחִים שִׂיחוֹת רֵעִים, כְּדֵי שֶׁלֹּא לְהַסִּיחַ אֶת הַדַּעַת מִן הָאֵבֶל. אֲבָל לָלֶכֶת לַכֹּתֶל הַמַּעֲרָבִי וּלְהִתְפַּלֵּל עַל בִּנְיַן בֵּית הַמִּקְדָּשׁ – זֶה תִּקּוּן גָּדוֹל לְתִשְׁעָה בְּאָב. כְּשֶׁפּוֹגְשִׁים שָׁם חֲבֵרִים – לֹא אוֹמְרִים שָׁלוֹם, וְאֶפְשָׁר לִלְחֹץ יָד בְּאַהֲבָה.
נָהֲגוּ כָּל יִשְׂרָאֵל שֶׁלֹּא לַעֲשׂוֹת מְלָאכָה בְּתִשְׁעָה בְּאָב עַד חֲצוֹת הַיּוֹם – מְלָאכוֹת שֶׁלּוֹקְחוֹת זְמַן וְטוֹרְדוֹת אֶת הַמַּחְשָׁבָה, כְּמוֹ תְּפִירָה, תִּקּוּנִים וּכְתִיבָה. הַדְלָקַת אוֹר וְכִבּוּיוֹ, אוֹ קְשִׁירָה – מֻתָּרִים.
💛 אַהֲבַת הַשֵּׁם: הָאֵבֶל עַל הַמִּקְדָּשׁ אֵינוֹ יֵאוּשׁ. בַּכֹּתֶל אָנוּ עוֹמְדִים מוּל מְקוֹם הַמִּקְדָּשׁ וּמְבַקְּשִׁים שֶׁיִּבָּנֶה בִּמְהֵרָה בְּיָמֵינוּ.
📖 פֶּרֶק י׳, הֲלָכוֹת י״ב, י״ד

🕯️אֵיכָה בַּחֹשֶׁךְ

בְּלֵיל תִּשְׁעָה בְּאָב, אַחֲרֵי עַרְבִית, קוֹרְאִים מְגִלַּת אֵיכָה. מַחְשִׁיכִים אֶת בֵּית הַכְּנֶסֶת וּמַשְׁאִירִים מְעַט אוֹר, מְסִירִים אֶת הַפָּרֹכֶת מִן הָאָרוֹן, וְיוֹשְׁבִים עַל הָרִצְפָּה כְּמוֹ אֲבֵלִים. מִתְפַּלְּלִים לְאַט, בְּקוֹל חַלָּשׁ.
בְּשַׁחֲרִית אוֹמְרִים קִינוֹת, וְרֹב הַקְּהִלּוֹת אֵינָן מִתְעַטְּפוֹת בְּטַלִּית וְאֵינָן מַנִּיחוֹת תְּפִלִּין. מַנִּיחִים אוֹתָם בְּמִנְחָה, כְּשֶׁמַּתְחִילָה קְצָת נֶחָמָה.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים סִפְּרוּ שֶׁכְּשֶׁנֶּחְרַב הַבַּיִת, ה׳ כִּבְיָכוֹל כִּבָּה אֶת הַפָּנָסִים שֶׁלּוֹ, כְּמוֹ מֶלֶךְ אָבֵל. גַּם ה׳ מִתְאַבֵּל אִתָּנוּ – וְהוּא גַּם יְנַחֵם אוֹתָנוּ.

🪑יְשִׁיבָה עַל הָאָרֶץ

נוֹהֲגִים לָשֶׁבֶת עַל הָאָרֶץ כְּמוֹ אֲבֵלִים. הָאַשְׁכְּנַזִּים וְחֵלֶק מִן הַסְּפָרַדִּים יוֹשְׁבִים כָּךְ עַד חֲצוֹת הַיּוֹם, וְיֵשׁ מִן הַסְּפָרַדִּים שֶׁמַּחְמִירִים עַד תְּפִלַּת מִנְחָה. מֻתָּר לָשֶׁבֶת עַל כָּרִית נְמוּכָה, וִישִׁיבָה עַל מַדְרֵגוֹת נֶחְשֶׁבֶת כִּישִׁיבָה עַל הַקַּרְקַע.
רַבִּים מוֹרִידִים אֶת הַמִּזְרָן לָרִצְפָּה לַשֵּׁינָה בְּלֵיל תִּשְׁעָה בְּאָב. זָקֵן, חוֹלֶה, וּמִי שֶׁסּוֹבֵל מִכְּאֵבֵי גַּב – רַשַּׁאִים לָשֶׁבֶת עַל כִּסֵּא רָגִיל.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁיּוֹשְׁבִים נָמוּךְ, הַלֵּב מַרְגִּישׁ שָׁפָל וְעָצוּב. וּמִתּוֹךְ הַשִּׁפְלוּת הַזֹּאת אָנוּ מִתְפַּלְּלִים שֶׁה׳ יְרוֹמֵם אֶת יְרוּשָׁלַיִם.

🕊️"קָרָא עָלַי מוֹעֵד"

בְּתִשְׁעָה בְּאָב לֹא אוֹמְרִים תַּחֲנוּן, כִּי הוּא נִקְרָא "מוֹעֵד": "קָרָא עָלַי מוֹעֵד לִשְׁבֹּר בַּחוּרָי". מוֹעֵד הוּא זְמַן מְיֻחָד: לֹא זָכִינוּ – נַעֲשָׂה יוֹם אֵבֶל; כְּשֶׁנִּזְכֶּה – יֵהָפֵךְ לְיוֹם טוֹב.
בְּמִנְחָה מַחֲזִירִים אֶת הַפָּרֹכֶת, מַנִּיחִים טַלִּית וּתְפִלִּין, וּמוֹסִיפִים "נַחֵם" בְּבִרְכַּת "בּוֹנֵה יְרוּשָׁלַיִם". וְיֵשׁ נָשִׁים שֶׁנּוֹהֲגוֹת לְנַקּוֹת אֶת הַבַּיִת אַחֲרֵי חֲצוֹת, מִתּוֹךְ צִפִּיָּה לַמָּשִׁיחַ, שֶׁנּוֹלַד בְּתִשְׁעָה בְּאָב.
💛 אַהֲבַת הַשֵּׁם: גַּם בְּתוֹךְ הָאֵבֶל אָנוּ מְחַכִּים לַגְּאֻלָּה. אַחֲרֵי חֲצוֹת כְּבָר מַתְחִילָה הַנֶּחָמָה – "מְנַחֵם צִיּוֹן וּבוֹנֵה יְרוּשָׁלַיִם".

🔥עֲשִׂירִי בְּאָב

בְּתִשְׁעָה בְּאָב לִפְנוֹת עֶרֶב הִצִּיתוּ אֶת בֵּית הַמִּקְדָּשׁ, וְהוּא נִשְׂרַף כָּל יוֹם י׳ בְּאָב. לָכֵן נָהֲגוּ שֶׁלֹּא לֶאֱכֹל בָּשָׂר וְלֹא לִשְׁתּוֹת יַיִן גַּם בַּעֲשִׂירִי בְּאָב: לְמִנְהַג הָאַשְׁכְּנַזִּים וְחֵלֶק מִן הַסְּפָרַדִּים – עַד חֲצוֹת הַיּוֹם. לְמִנְהַג חֵלֶק מִן הַסְּפָרַדִּים – כָּל הַיּוֹם.
לְדַעַת רֹב הָאַחֲרוֹנִים, עַד חֲצוֹת גַּם לֹא מְכַבְּסִים, לֹא מִסְתַּפְּרִים, לֹא שׁוֹמְעִים מוּזִיקָה שְׂמֵחָה וְלֹא מִתְרַחֲצִים בְּחַמִּים. וְאֶת קִדּוּשׁ הַלְּבָנָה דּוֹחִים לְאַחַר הַצּוֹם, כְּשֶׁאֶפְשָׁר לְבָרֵךְ בְּשִׂמְחָה.
💛 אַהֲבַת הַשֵּׁם: רַבִּי יוֹחָנָן אָמַר שֶׁהָיָה קוֹבֵעַ אֶת הַצּוֹם בַּעֲשִׂירִי, שֶׁבּוֹ נִשְׂרַף רֹב הַהֵיכָל. אֲבָל חֲכָמִים קָבְעוּ אוֹתוֹ בַּתְּשִׁיעִי, כִּי הַכֹּל הוֹלֵךְ אַחַר הַהַתְחָלָה.

📅תִּשְׁעָה בְּאָב שֶׁחָל בְּשַׁבָּת

כְּשֶׁתִּשְׁעָה בְּאָב חָל בְּשַׁבָּת, הַצּוֹם נִדְחֶה לְיוֹם רִאשׁוֹן. בַּשַּׁבָּת עַצְמָהּ אֵין מַרְאִים שׁוּם סִימַן אֲבֵלוּת, וְאוֹכְלִים וְשׁוֹתִים כְּמוֹ בְּכָל שַׁבָּת – וַאֲפִלּוּ כִּסְעוּדַת שְׁלֹמֹה בִּשְׁעָתוֹ.
בְּמוֹצָאֵי הַצּוֹם הַנִּדְחֶה, מִיָּד אַחֲרֵי צֵאת הַכּוֹכָבִים, מֻתָּר לְהִסְתַּפֵּר, לְכַבֵּס וּלְהִתְרַחֵץ בְּחַמִּים. וּבְבָשָׂר וְיַיִן – רַבִּים נִמְנָעִים בְּאוֹתוֹ עֶרֶב, וְיֵשׁ מְקִלִּים.
💛 אַהֲבַת הַשֵּׁם: הַשַּׁבָּת חֲזָקָה מִן הָאֵבֶל: גַּם בַּיּוֹם שֶׁבּוֹ נֶחְרַב הַבַּיִת, הַשַּׁבָּת מְבִיאָה שִׂמְחָה וּמְנוּחָה – טַעַם שֶׁל גְּאֻלָּה.
📖 פֶּרֶק י׳, הֲלָכָה כ׳

🔀בְּמוֹצָאֵי צוֹם נִדְחֶה – שְׁתֵּי שִׁיטוֹת

כְּשֶׁהַצּוֹם נִדְחָה לְיוֹם רִאשׁוֹן, מִיָּד אַחֲרָיו מֻתָּר לְהִסְתַּפֵּר, לְכַבֵּס וּלְהִתְרַחֵץ בְּחַמִּים. אֲבָל בְּבָשָׂר וְיַיִן נֶחְלְקוּ: רַבִּים נִמְנָעִים מֵהֶם בְּאוֹתוֹ עֶרֶב, כִּי לֹא רָאוּי לִשְׂמֹחַ בְּבָשָׂר וְיַיִן מִיָּד אַחֲרֵי הַצּוֹם. וְיֵשׁ מְקִלִּים לֶאֱכֹל בָּשָׂר מִיָּד בְּצֵאת הַצּוֹם.
וְזֶה שׁוֹנֶה מִשָּׁנָה רְגִילָה, שֶׁבָּהּ נִמְנָעִים מִבָּשָׂר וְיַיִן גַּם בַּעֲשִׂירִי בְּאָב.
💛 אַהֲבַת הַשֵּׁם: שְׁתֵּי הַשִּׁיטוֹת כַּהֲלָכָה. כָּל אֶחָד מַמְשִׁיךְ בְּמִנְהַג אֲבוֹתָיו, וְכֻלָּנוּ מְחַכִּים יַחַד לַיּוֹם שֶׁבּוֹ תִּשְׁעָה בְּאָב יִהְיֶה חַג.
📖 פֶּרֶק י׳, הֲלָכָה כ׳

🧒יְלָדִים בְּתִשְׁעָה בְּאָב

יְלָדִים אֵינָם צָמִים כָּל הַיּוֹם. מִגִּיל תֵּשַׁע נוֹהֲגִים לְחַנְּכָם לָצוּם כַּמָּה שָׁעוֹת לְפִי כֹּחָם, וְרַבִּים מְהַדְּרִים שֶׁיְּלָדִים מִגִּיל שֵׁשׁ לֹא יֹאכְלוּ וְלֹא יִשְׁתּוּ בַּלַּיְלָה. כְּשֶׁאוֹכְלִים – אוֹכְלִים מַאֲכָלִים פְּשׁוּטִים.
מִגִּיל שֵׁשׁ בְּעֵרֶךְ, כְּשֶׁהַיֶּלֶד מַתְחִיל לְהָבִין מַהוּ הַחֻרְבָּן, מְחַנְּכִים אוֹתוֹ שֶׁלֹּא לִנְעֹל נַעֲלַיִם שֶׁל חוּץ, וְשֶׁלֹּא לִרְחֹץ וְלָסוּךְ לְתַעֲנוּג. וְיֶלֶד רַשַּׁאי לִלְמֹד בְּעַצְמוֹ אֶת מַה שֶּׁמֻּתָּר לַגְּדוֹלִים.
💛 אַהֲבַת הַשֵּׁם: גַּם הַיְּלָדִים שֻׁתָּפִים בָּאֵבֶל עַל יְרוּשָׁלַיִם – וְהֵם גַּם יִהְיוּ שֻׁתָּפִים בַּשִּׂמְחָה הַגְּדוֹלָה כְּשֶׁתִּבָּנֶה.
📖 פֶּרֶק י׳, הֲלָכָה כ״א

🧠 בְּדֹק אֶת עַצְמְךָ

חוֹלֶה צָרִיךְ לֶאֱכֹל בְּתִשְׁעָה בְּאָב. אֵיךְ יֹאכַל?
💡 חוֹלֶה פָּטוּר מִצּוֹם תִּשְׁעָה בְּאָב לְגַמְרֵי, וְאֵינוֹ צָרִיךְ לֶאֱכֹל פָּחוֹת מִכַּשִּׁעוּר – זֶה רַק בְּיוֹם הַכִּפּוּרִים, שֶׁהוּא מִן הַתּוֹרָה. אֲבָל לֹא יְעַנֵּג אֶת עַצְמוֹ בְּמַעֲדַנִּים.
הַאִם רָאוּי לָלֶכֶת לַכֹּתֶל הַמַּעֲרָבִי בְּתִשְׁעָה בְּאָב?
💡 אֵין תִּקּוּן גָּדוֹל יוֹתֵר לְתִשְׁעָה בְּאָב מִלָּלֶכֶת לַכֹּתֶל וּלְהִתְפַּלֵּל עַל בִּנְיַן הַמִּקְדָּשׁ, וְהָאֲנָשִׁים הָרַבִּים מַגְדִּילִים אֶת כֹּחַ הַתְּפִלָּה. כְּשֶׁפּוֹגְשִׁים חֲבֵרִים – לֹא אוֹמְרִים שָׁלוֹם, אֲבָל לוֹחֲצִים יָד בְּאַהֲבָה.
מָה מֻתָּר לַעֲשׂוֹת בְּתִשְׁעָה בְּאָב לִפְנֵי חֲצוֹת הַיּוֹם?
💡 עַד חֲצוֹת לֹא עוֹשִׂים מְלָאכוֹת שֶׁלּוֹקְחוֹת זְמַן וְטוֹרְדוֹת אֶת הַמַּחְשָׁבָה – תִּקּוּנִים, תְּפִירָה וּכְתִיבָה. מְלָאכָה קְצָרָה, כְּמוֹ הַדְלָקַת אוֹר וְכִבּוּיוֹ, אֵינָהּ מַסִּיחָה אֶת הַדַּעַת וּמֻתֶּרֶת.

דָּם וְהַכְשָׁרַת הַבָּשָׂר

🍽️ הִלְכוֹת כַּשְׁרוּת · פֶּרֶק כ״ב

🚫לֹא אוֹכְלִים דָּם

הַתּוֹרָה הִתִּירָה לֶאֱכֹל בָּשָׂר, אֲבָל אָסְרָה אֶת הַדָּם: "רַק חֲזַק לְבִלְתִּי אֲכֹל הַדָּם, כִּי הַדָּם הוּא הַנָּפֶשׁ". הָאִסּוּר הוּא בְּדַם בְּהֵמָה, חַיָּה וְעוֹף.
כְּשֶׁמְּבַשְּׁלִים בָּשָׂר שֶׁלֹּא הֻכְשַׁר, הַדָּם יוֹצֵא מִמֶּנּוּ וְחוֹזֵר וְנִבְלָע בּוֹ – וְכָל הַבָּשָׂר נֶאֱסָר. לָכֵן מַכְשִׁירִים אֶת הַבָּשָׂר לִפְנֵי הַבִּשּׁוּל.
💛 אַהֲבַת הַשֵּׁם: הַדָּם נוֹעַד לְקַיֵּם אֶת חַיֵּי בַּעַל הַחַיִּים, וְלֹא לְמַאֲכַל הָאָדָם. מִי שֶׁנִּזְהָר מִן הַדָּם שׁוֹמֵר עַל לֵב רַחֲמָן וְעָדִין.
📖 פֶּרֶק כ״ב, הֲלָכוֹת א׳–ב׳

🧂הַדָּם הוּא הַנֶּפֶשׁ

הַבָּשָׂר מֻתָּר, וְהַדָּם אָסוּר – "כִּי הַדָּם הוּא הַנָּפֶשׁ". לָכֵן מַכְשִׁירִים אֶת הַבָּשָׂר בִּשְׁלֹשָׁה שְׁלַבִּים: הֲדָחָה, מְלִיחָה וַהֲדָחָה.
"הַיּוֹצֵא מִן הַטָּמֵא – טָמֵא": מָה שֶׁיּוֹצֵא מִבַּעַל חַיִּים טָמֵא – אָסוּר, חוּץ מִן הַדְּבַשׁ.
💛 אַהֲבַת הַשֵּׁם: בְּכָל מַאֲכָל שֶׁעַל הַשֻּׁלְחָן יֵשׁ הִזְדַּמְּנוּת לִזְכֹּר אֶת ה׳. כְּשֶׁאָנוּ שׁוֹמְרִים עַל דִּינֵי הַכַּשְׁרוּת, הָאֹכֶל נוֹתֵן לָנוּ כֹּחַ לַעֲבוֹדַת ה׳.
📖 פֶּרֶק כ״ב, הֲלָכוֹת א׳, ה׳; פֶּרֶק כ״ד, הֲלָכוֹת א׳, ד׳

🦷דַּם דָּגִים וְדַם אָדָם

אִסּוּר דָּם הוּא רַק בִּבְהֵמָה, חַיָּה וְעוֹף, שֶׁצְּרִיכִים שְׁחִיטָה. בְּדָגִים וּבַחֲגָבִים אֵין אִסּוּר דָּם, כִּי הֵם פָּחוֹת מְפֻתָּחִים וּמַרְגִּישִׁים פָּחוֹת צַעַר. אֲבָל דַּם דָּגִים שֶׁנֶּאֱסַף בִּכְלִי – אָסוּר לִשְׁתּוֹת, אֶלָּא אִם רוֹאִים בּוֹ קַשְׂקַשִּׂים, שֶׁלֹּא יִטְעוּ אֲחֵרִים.
דַּם אָדָם – חֲכָמִים אָסְרוּ. מִי שֶׁנָּשַׁךְ לֶחֶם אוֹ תַּפּוּחַ וְנִשְׁאַר עָלָיו דָּם מִן הַפֶּה – מְגָרֵד אֶת הַדָּם וּמַמְשִׁיךְ לֶאֱכֹל. דָּם שֶׁבֵּין הַשִּׁנַּיִם, אוֹ מֵאֶצְבַּע שֶׁנִּפְצְעָה – מֻתָּר לִמְצֹץ.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים חָשְׁבוּ גַּם עַל מָה שֶׁאֲחֵרִים רוֹאִים – "מַרְאִית עַיִן". יְהוּדִי שׁוֹמֵר שֶׁלֹּא יִלְמְדוּ מִמֶּנּוּ דָּבָר לֹא נָכוֹן, גַּם כְּשֶׁהוּא עַצְמוֹ יוֹדֵעַ שֶׁהַכֹּל בְּסֵדֶר.
📖 פֶּרֶק כ״ב, הֲלָכוֹת ב׳, י״ד

🧂מְלִיחַת הַבָּשָׂר

מַכְשִׁירִים בָּשָׂר בִּמְלִיחָה, בִּשְׁלֹשָׁה שְׁלַבִּים: הֲדָחָה – מַשְׁרִים אֶת הַבָּשָׂר בְּמַיִם כַּחֲצִי שָׁעָה; מְלִיחָה – מְפַזְּרִים עָלָיו מֶלַח גַּס מִכָּל הַצְּדָדִים, עַל מִשְׁטָח מְחֹרָר אוֹ מְשֻׁפָּע, כְּדֵי שֶׁהַדָּם יִזּוֹב; וַהֲדָחָה – שׁוֹטְפִים הֵיטֵב, עַד שֶׁלֹּא נִשְׁאָר עָלָיו דָּם וּמֶלַח.
לְכַתְּחִלָּה מַשְׁאִירִים אֶת הַבָּשָׂר בַּמֶּלַח שָׁעָה. אַחֲרֵי מְלִיחָה כַּדִּין, גַּם אִם יוֹצֵא מִן הַבָּשָׂר נוֹזֵל אָדֹם – הוּא אֵינוֹ נֶחְשָׁב דָּם.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה אֵינָהּ דּוֹרֶשֶׁת מֵאִתָּנוּ דָּבָר שֶׁבְּנֵי אָדָם אֵינָם יְכוֹלִים לַעֲשׂוֹת – "לֹא נִתְּנָה תּוֹרָה לְמַלְאֲכֵי הַשָּׁרֵת". מִי שֶׁעָשָׂה אֶת הַמְּלִיחָה כַּדִּין, אוֹכֵל בָּשָׂר כָּשֵׁר בְּלֵב שָׁלֵם.
📖 פֶּרֶק כ״ב, הֲלָכוֹת ג׳, ה׳–ז׳

🔥צְלִיָּה וְכָבֵד

דֶּרֶךְ נוֹסֶפֶת לְהַכְשִׁיר בָּשָׂר הִיא צְלִיָּה עַל הָאֵשׁ: הָאֵשׁ שׁוֹאֶבֶת אֶת הַדָּם וְשׂוֹרֶפֶת אוֹתוֹ. צוֹלִים עַד שֶׁהַבָּשָׂר רָאוּי לַאֲכִילָה, וְדוֹאֲגִים שֶׁלַּדָּם יִהְיֶה מָקוֹם לִזּוֹב. כֹּחַ הַצְּלִיָּה גָּדוֹל מִכֹּחַ הַמְּלִיחָה.
הַכָּבֵד מָלֵא דָּם, וְאֶת הַכָּבֵד מַכְשִׁירִים רַק בִּצְלִיָּה – וְלֹא בִּמְלִיחָה. אַחֲרֵי שֶׁצָּלוּ אוֹתוֹ כַּדִּין, מֻתָּר גַּם לְבַשֵּׁל אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לָנוּ דְּרָכִים שׁוֹנוֹת לְהַכְשִׁיר אֶת הָאֹכֶל, וְכָל אַחַת מַתְאִימָה לְמַאֲכָל אַחֵר. כָּךְ כָּל יְהוּדִי יָכוֹל לֶאֱכֹל בְּכַשְׁרוּת וּבְשִׂמְחָה.
📖 פֶּרֶק כ״ב, הֲלָכוֹת י׳–י״ב

🧠 בְּדֹק אֶת עַצְמְךָ

אוֹרִי נָשַׁךְ פְּרוּסַת לֶחֶם, וְרָאָה עָלֶיהָ מְעַט דָּם מִן הַחֲנִיכַיִם. מָה הוּא עוֹשֶׂה?
💡 חֲכָמִים אָסְרוּ דַּם אָדָם מִשּׁוּם מַרְאִית עַיִן. לָכֵן מְגָרְדִים אֶת הַדָּם שֶׁעַל הַלֶּחֶם, וְהַשְּׁאָר מֻתָּר. דָּם שֶׁבֵּין הַשִּׁנַּיִם – מֻתָּר לִמְצֹץ.
לָמָּה אָסוּר לְבַשֵּׁל בָּשָׂר שֶׁלֹּא הֻכְשַׁר?
💡 דָּם שֶׁבָּלוּעַ בַּבָּשָׂר כְּדֶרֶךְ טִבְעוֹ – מֻתָּר. אֲבָל בַּבִּשּׁוּל הַדָּם יוֹצֵא וְנִבְלָע שׁוּב, וְכָל הַבָּשָׂר נֶאֱסָר. לָכֵן מוֹלְחִים אוֹ צוֹלִים קֹדֶם.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

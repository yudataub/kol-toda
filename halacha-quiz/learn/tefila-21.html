נְפִילַת אַפַּיִם וְתַחֲנוּנִים

🙏 הִלְכוֹת תְּפִלָּה · פֶּרֶק כ״א

🙇נְפִילַת אַפַּיִם – לֵב פָּתוּחַ לִפְנֵי ה׳

מֹשֶׁה רַבֵּנוּ הִתְפַּלֵּל עַל יִשְׂרָאֵל בִּישִׁיבָה, בַּעֲמִידָה וּבִנְפִילַת אַפַּיִם. גַּם אֲנַחְנוּ: אַחֲרֵי שֶׁיָּשַׁבְנוּ וְעָמַדְנוּ בַּתְּפִלָּה, אוֹמְרִים תַּחֲנוּן – תְּפִלָּה שֶׁל עֲנָוָה וּשְׁפִיכַת הַלֵּב.
יוֹצְאֵי אַשְׁכְּנַז, וְגַם חֵלֶק מִיּוֹצְאֵי סְפָרַד, מַרְכִּינִים אֶת הָרֹאשׁ עַל הַזְּרוֹעַ. רַבִּים מִבְּנֵי עֲדוֹת הַמִּזְרָח, הַנּוֹהֲגִים כְּבֶן אִישׁ חַי, אוֹמְרִים אֶת הַמִּזְמוֹר בִּישִׁיבָה בְּלִי לִפֹּל עַל הַפָּנִים. כָּל אֶחָד – כְּמִנְהַג אֲבוֹתָיו.
💛 אַהֲבַת הַשֵּׁם: בְּתַחֲנוּן אֲנַחְנוּ אוֹמְרִים לַה׳: כָּל מָה שֶׁיֵּשׁ לִי – שֶׁלְּךָ. מִי שֶׁמַּרְגִּישׁ קָטָן לִפְנֵי ה׳ פּוֹתֵחַ אֶת הַלֵּב, וְה׳ קָרוֹב אֵלָיו.
📖 פֶּרֶק כ״א, הֲלָכוֹת א׳, ג׳

🔀מִנְהָגֵי תַּחֲנוּן וְקַדִּישׁ

נְפִילַת אַפַּיִם: יוֹצְאֵי אַשְׁכְּנַז וְחֵלֶק מִיּוֹצְאֵי סְפָרַד מַרְכִּינִים אֶת הָרֹאשׁ עַל הַזְּרוֹעַ. לְפִי הַשֻּׁלְחָן עָרוּךְ נוֹפְלִים תָּמִיד עַל יַד שְׂמֹאל, וּלְפִי הָרַמָּ״א – בְּשַׁחֲרִית, כְּשֶׁהַתְּפִלִּין עַל שְׂמֹאל, עַל יַד יָמִין. רַבִּים מִבְּנֵי עֲדוֹת הַמִּזְרָח, הַנּוֹהֲגִים כְּבֶן אִישׁ חַי, אֵינָם נוֹפְלִים כְּלָל.
בְּיוֹם שֶׁנַּעַר הִגִּיעַ לְמִצְווֹת – רַבִּים נָהֲגוּ שֶׁלֹּא לוֹמַר תַּחֲנוּן בַּמִּנְיָן שֶׁלּוֹ, וְיֵשׁ נוֹהֲגִים לוֹמַר. וּבַקַּדִּישׁ: לְמִנְהַג אַשְׁכְּנַז עוֹנִים עַד ״לְעָלַם וּלְעָלְמֵי עָלְמַיָּא״, וּלְמִנְהַג הַסְּפָרַדִּים – עַד ״דַּאֲמִירָן בְּעָלְמָא״.
💛 אַהֲבַת הַשֵּׁם: כָּל עֵדָה שׁוֹמֶרֶת עַל מִנְהָגָהּ בְּאַהֲבָה – וְכֻלָּנוּ יַחַד מְגַדְּלִים וּמְקַדְּשִׁים אֶת שְׁמוֹ הַגָּדוֹל שֶׁל ה׳.
📖 פֶּרֶק כ״א, הֲלָכוֹת ג׳–ד׳, ח׳; פֶּרֶק כ״ג, הֲלָכָה י״א

🌈שְׁלֹשׁ עֶשְׂרֵה מִדּוֹת רַחֲמִים

אַחֲרֵי חֵטְא הָעֵגֶל לִמֵּד ה׳ אֶת מֹשֶׁה: ״ה׳ ה׳ אֵ-ל רַחוּם וְחַנּוּן...״, וְאָמַר: כְּשֶׁיִּשְׂרָאֵל יֹאמְרוּ אֶת הַמִּדּוֹת הָאֵלֶּה – אֲנִי מוֹחֵל לָהֶם.
אוֹמְרִים אוֹתָן רַק בְּמִנְיָן שֶׁל עֲשָׂרָה, כִּי הֵן דָּבָר שֶׁבִּקְדֻשָּׁה. מִי שֶׁמִּתְפַּלֵּל לְבַד – לֹא אוֹמֵר אוֹתָן, אֶלָּא אִם כֵּן הוּא קוֹרֵא אוֹתָן בְּטַעֲמֵי הַמִּקְרָא.
💛 אַהֲבַת הַשֵּׁם: ה׳ לִמֵּד אוֹתָנוּ בְּעַצְמוֹ אֵיךְ לְבַקֵּשׁ מְחִילָה – כְּמוֹ אַבָּא שֶׁאוֹמֵר לַבֵּן: ״כָּכָה תְּבַקֵּשׁ, וַאֲנִי אֶסְלַח״.
📖 פֶּרֶק כ״א, הֲלָכָה ה׳

🎉יְמֵי שִׂמְחָה – בְּלִי תַּחֲנוּן

תַּחֲנוּן הוּא תְּפִלָּה שֶׁל צַעַר עַל הַחֻלְשׁוֹת וְהַחֶסְרוֹנוֹת שֶׁלָּנוּ. לָכֵן בְּיָמִים שֶׁל שִׂמְחָה לֹא אוֹמְרִים אוֹתוֹ: בְּשַׁבָּת, בְּיוֹם טוֹב וּבְרֹאשׁ חֹדֶשׁ.
גַּם בַּחֲנֻכָּה, בְּפוּרִים וּבְכָל חֹדֶשׁ נִיסָן – וּבְמִנְיָן שֶׁמִּתְפַּלֵּל בּוֹ חָתָן אוֹ אֲבִי הַבֵּן בְּיוֹם הַבְּרִית – אֵין אוֹמְרִים תַּחֲנוּן.
💛 אַהֲבַת הַשֵּׁם: בְּיוֹם שֶׁל שִׂמְחַת מִצְוָה, הַשִּׂמְחָה עַצְמָהּ מְקָרֶבֶת אוֹתָנוּ לַה׳. ה׳ רוֹצֶה שֶׁנַּעֲבֹד אוֹתוֹ גַּם בְּלֵב נִשְׁבָּר וְגַם בְּשִׂמְחָה.
📖 פֶּרֶק כ״א, הֲלָכוֹת ז׳–ח׳

🧠 בְּדֹק אֶת עַצְמְךָ

שִׁמְעוֹן מִתְפַּלֵּל שַׁחֲרִית לְבַד בַּבַּיִת, וְהִגִּיעַ לִשְׁלֹשׁ עֶשְׂרֵה הַמִּדּוֹת. מָה יַעֲשֶׂה?
💡 שְׁלֹשׁ עֶשְׂרֵה מִדּוֹת הֵן דָּבָר שֶׁבִּקְדֻשָּׁה, וְאוֹמְרִים אוֹתָן רַק בַּעֲשָׂרָה. הַמִּתְפַּלֵּל לְבַד יָכוֹל לִקְרֹא אוֹתָן בְּטַעֲמֵי הַמִּקְרָא, כְּמוֹ קוֹרֵא בַּתּוֹרָה.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

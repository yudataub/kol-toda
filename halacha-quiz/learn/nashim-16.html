קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ

🌸 תְּפִלַּת בָּנוֹת · פֶּרֶק ט״ז

👑שְׁמַע יִשְׂרָאֵל – בְּכָל יוֹם

קְרִיאַת שְׁמַע הִיא מִצְוָה שֶׁהַזְּמַן גְּרָמָהּ, וְלָכֵן בָּנוֹת פְּטוּרוֹת מִמֶּנָּה וּמִבִּרְכוֹתֶיהָ. אֲבָל בְּמִצְווֹת הָאֱמוּנָה בָּנוֹת חַיָּבוֹת כְּמוֹ בָּנִים. לָכֵן רָאוּי שֶׁכָּל בַּת תֹּאמַר בְּכָל יוֹם שְׁנֵי פְּסוּקִים: ״שְׁמַע יִשְׂרָאֵל״ וּ״בָרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ לְעוֹלָם וָעֶד״.
הָרוֹצָה לְהַדֵּר אוֹמֶרֶת אֶת כָּל שָׁלֹשׁ הַפָּרָשִׁיּוֹת וְאֶת בִּרְכַּת ״אֱמֶת וְיַצִּיב״. בְּבָתֵּי סֵפֶר רַבִּים נָהֲגוּ שֶׁהַבָּנוֹת אוֹמְרוֹת קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ.
💛 אַהֲבַת הַשֵּׁם: בִּשְׁנֵי הַפְּסוּקִים הָאֵלֶּה אָנוּ מְקַבְּלִים עָלֵינוּ עֹל מַלְכוּת שָׁמַיִם: ה׳ הוּא הַמֶּלֶךְ שֶׁלָּנוּ, וְהוּא אֶחָד. זוֹ הַהַצְהָרָה הֲכִי גְּדוֹלָה שֶׁל אֱמוּנָה, וְכָל בַּת אוֹמֶרֶת אוֹתָהּ בְּאַהֲבָה.
📖 פֶּרֶק ט״ז, הֲלָכָה א׳

📖״קָדוֹשׁ קָדוֹשׁ״ בְּלִי מִנְיָן

בְּבִרְכַּת ״יוֹצֵר אוֹר״ אוֹמְרִים אֶת פְּסוּקֵי הַקְּדֻשָּׁה. לְדַעַת רֹב הַפּוֹסְקִים זוֹ אֵינָהּ צְרִיכָה מִנְיָן, כִּי אֲנַחְנוּ רַק מְסַפְּרִים אֵיךְ הַמַּלְאָכִים מְקַדְּשִׁים אֶת ה׳. וּלְכַתְּחִלָּה, כְּדֵי לָצֵאת יְדֵי כָּל הַפּוֹסְקִים, בַּת שֶׁאוֹמֶרֶת אוֹתָהּ בְּלִי מִנְיָן אוֹמֶרֶת אֶת הַפְּסוּקִים בְּטַעֲמֵי הַמִּקְרָא, כְּמִי שֶׁקּוֹרֵאת בַּנָּבִיא.
לֹא צָרִיךְ לָדַעַת הֵיטֵב אֶת הַטְּעָמִים – מַסְפִּיק לוֹמַר מֵעֵין הַטְּעָמִים. וּמִי שֶׁאוֹמֶרֶת קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ – מַתְחִילָה עֲמִידָה מִיָּד אַחֲרֵי ״גָּאַל יִשְׂרָאֵל״.
💛 אַהֲבַת הַשֵּׁם: אָמְרוּ חֲכָמִים: כָּל הַסּוֹמֵךְ גְּאֻלָּה לִתְפִלָּה – הוּא בֶּן הָעוֹלָם הַבָּא. הַזִּכָּרוֹן שֶׁה׳ גָּאַל אוֹתָנוּ מִמִּצְרַיִם הוּא כְּמוֹ דְּפִיקָה עַל פֶּתַח הַמֶּלֶךְ – וּמִיָּד נִכְנָסִים לְדַבֵּר אִתּוֹ.
📖 פֶּרֶק ט״ז, הֲלָכָה י״ג

🛑מָתַי מַפְסִיקִים בִּקְרִיאַת שְׁמַע?

בְּאֶמְצַע קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ לֹא מְדַבְּרִים. אֲבָל מֻתָּר לְהַפְסִיק כְּדֵי לִמְנֹעַ נֵזֶק לַגּוּף אוֹ הֶפְסֵד שֶׁל כֶּסֶף – וְעָדִיף בֵּין הַפְּרָקִים אוֹ בֵּין הַבְּרָכוֹת.
מִי שֶׁרוֹאָה חֲבֵרָה עוֹשָׂה אִסּוּר – רוֹמֶזֶת לָהּ. וְאִם הִיא לֹא מְבִינָה מֵהָרֶמֶז, מַפְסִיקָה בְּדִבּוּר כְּדֵי לְהַפְרִישׁ אוֹתָהּ מֵהָאִסּוּר. אֲבָל בְּ״שְׁמַע יִשְׂרָאֵל״ וּבְ״בָרוּךְ שֵׁם״ לֹא מַפְסִיקִים כְּלָל, חוּץ מִפִּקּוּחַ נֶפֶשׁ.
💛 אַהֲבַת הַשֵּׁם: אִם מֻתָּר לְהַפְסִיק לִכְבוֹד בְּנֵי אָדָם – קַל וָחֹמֶר לִכְבוֹד שָׁמַיִם. לַעֲזֹר לַחֲבֵרָה לְהִמָּנַע מֵאִסּוּר זוֹ אַהֲבַת ה׳ וְאַהֲבַת הַחֲבֵרָה יַחַד.
📖 פֶּרֶק ט״ז, הֲלָכָה י״ד

🧠 בְּדֹק אֶת עַצְמְךָ

שָׂרָה אוֹמֶרֶת בַּבַּיִת, בְּלִי מִנְיָן, אֶת בִּרְכַּת ״יוֹצֵר אוֹר״, וּמַגִּיעָה לַפְּסוּקִים ״קָדוֹשׁ קָדוֹשׁ קָדוֹשׁ״. מָה כְּדַאי לָהּ לַעֲשׂוֹת לְכַתְּחִלָּה?
💡 לְדַעַת רֹב הַפּוֹסְקִים, הַקְּדֻשָּׁה שֶׁבְּ״יוֹצֵר״ אֵינָהּ צְרִיכָה מִנְיָן, כִּי רַק מְסַפְּרִים אֵיךְ הַמַּלְאָכִים מְקַדְּשִׁים. וּלְכַתְּחִלָּה, כְּדֵי לָצֵאת יְדֵי כֻּלָּם, אוֹמְרִים אוֹתָהּ בְּטַעֲמֵי הַמִּקְרָא – כְּמוֹ לִימוּד, וְזֶה מֻתָּר לְכָל הַדֵּעוֹת.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

🤍בָּרוּךְ דַּיַּן הָאֱמֶת

חֲכָמִים תִּקְּנוּ לָאֲבֵלִים לְבָרֵךְ ״בָּרוּךְ אַתָּה ה׳ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, דַּיַּן הָאֱמֶת״. וּמִי שֶׁשּׁוֹמֵעַ שֶׁיְּהוּדִי נִפְטַר, נוֹהֵג לוֹמַר ״בָּרוּךְ דַּיַּן הָאֱמֶת״.
בַּבְּרָכָה הַזֹּאת אָנוּ אוֹמְרִים: ה׳ מַנְהִיג אֶת הָעוֹלָם בְּצֶדֶק, וְהַכֹּל לְטוֹבָה – גַּם כְּשֶׁאֵינֶנּוּ מְבִינִים.
💛 אַהֲבַת הַשֵּׁם: גַּם בְּרֶגַע עָצוּב אֲנַחְנוּ לֹא לְבַד. אָנוּ סוֹמְכִים עַל ה׳, כְּמוֹ יֶלֶד שֶׁסּוֹמֵךְ עַל אָבִיו הָאוֹהֵב.
📖 פֶּרֶק י׳, הֲלָכָה א׳

😢מִצְטַעֲרִים – וּמַאֲמִינִים

מִצְוָה לְהִתְאַבֵּל עַל הַנִּפְטָר וּלְכַבֵּד אֶת זִכְרוֹ, וּמֻתָּר וְרָאוּי לִבְכּוֹת וּלְהִצְטַעֵר. אֲבָל אָסוּר לְהִתְאַבֵּל יוֹתֵר מִדַּי וְלִשְׁקֹעַ בְּיֵאוּשׁ.
מִי שֶׁמִּתְאַבֵּל יוֹתֵר מִדַּי מַרְאֶה כְּאִלּוּ אֵינוֹ מַאֲמִין שֶׁה׳ מַנְהִיג אֶת עוֹלָמוֹ בְּצֶדֶק וּלְטוֹבָה. הָאֲבֵלוּת מְשַׁלֶּבֶת צַעַר וֶאֱמוּנָה יַחַד.
💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אוֹתָנוּ גַּם בִּזְמַנִּים קָשִׁים. הָאֱמוּנָה מְלַוָּה אֶת הַצַּעַר, וּמְחַזֶּקֶת אוֹתָנוּ לְהַמְשִׁיךְ לִחְיוֹת וְלַעֲשׂוֹת טוֹב.
📖 פֶּרֶק י׳, הֲלָכָה א׳

🕊️שְׁתִיקָה שֶׁל כָּבוֹד

הָאָבֵל אָסוּר בִּשְׁאֵלַת שָׁלוֹם, כִּי אֵינוֹ שָׁרוּי בְּשָׁלוֹם. וּבַסְּעוּדָה הָרִאשׁוֹנָה מַגִּישִׁים לוֹ מַאֲכָלִים עֲגֻלִּים, שֶׁאֵין לָהֶם ״פֶּה״ – לִרְמֹז שֶׁיָּפָה הַשְּׁתִיקָה לָאֲבֵלִים וְלַמְנַחֲמִים.
״בָּרוּךְ דַּיַּן הָאֱמֶת״: גַּם בְּרֶגַע שֶׁל צַעַר אָנוּ מַאֲמִינִים שֶׁה׳ מַנְהִיג אֶת עוֹלָמוֹ בְּצֶדֶק, וְהַכֹּל לְטוֹבָה.
💛 אַהֲבַת הַשֵּׁם: הָאֱמוּנָה וְהַשְּׁתִיקָה הַמְּכֻבֶּדֶת עוֹזְרוֹת לַלֵּב לְהִתְנַחֵם, לְאַט לְאַט.
📖 פֶּרֶק י׳, הֲלָכוֹת א׳, ז׳

⚖️📜 לְשׁוֹן חֲכָמִים: בָּרוּךְ דַּיַּן הָאֱמֶת

״מְצֻוֶּה עַל הָאָדָם לְהִתְחַזֵּק בֶּאֱמוּנָה שֶׁה׳ שֶׁבְּיָדוֹ נֶפֶשׁ כָּל חַי, מַנְהִיג וְדָן אֶת עוֹלָמוֹ בְּצֶדֶק... לְשֵׁם כָּךְ תִּקְּנוּ חֲכָמִים לָאֲבֵלִים לְבָרֵךְ: ׳בָּרוּךְ... דַּיַּן הָאֱמֶת׳״.
📖 דַּיָּן – שׁוֹפֵט. דַּיַּן הָאֱמֶת – ה׳, שֶׁשּׁוֹפֵט אֶת הָעוֹלָם בְּצֶדֶק. ״וְאַף שֶׁלְּעִתִּים אֵינֶנּוּ מְבִינִים אֶת הַנְהָגָתוֹ, הִנְנוּ מַאֲמִינִים... שֶׁכָּל הַנְהָגָתוֹ בְּצֶדֶק וְהַכֹּל לְטוֹבָה״.
💛 אַהֲבַת הַשֵּׁם: ״הַכְּאֵב גָּדוֹל וְנוֹרָא... וּמֵאִידָךְ אָסוּר לִשְׁקֹעַ בְּיֵאוּשׁ״ – גַּם בָּרֶגַע הַקָּשֶׁה בְּיוֹתֵר אֲנַחְנוּ לֹא לְבַד: ה׳ מַנְהִיג אֶת הָעוֹלָם. 🧒 וְאַתָּה? כְּשֶׁקּוֹרֶה לְךָ מַשֶּׁהוּ לֹא נָעִים – אֱמֹר בַּלֵּב: ״ה׳ יוֹדֵעַ, וְהַכֹּל לְטוֹבָה״.
📖 פֶּרֶק י׳, הֲלָכָה א׳

🪑יוֹשְׁבִים שִׁבְעָה

שִׁבְעָה יָמִים הָאֲבֵלִים יוֹשְׁבִים עַל הָאָרֶץ, עַל מִזְרָן אוֹ עַל כִּסֵּא נָמוּךְ – וְלָכֵן קוֹרְאִים לָזֶה ״יְשִׁיבַת שִׁבְעָה״. מִי שֶׁחוֹשֵׁשׁ לְגַבּוֹ יָכוֹל לָשֶׁבֶת עַל כִּסֵּא גָּבוֹהַּ יוֹתֵר, אֲבָל נָמוּךְ מֵהָרָגִיל.
בִּימֵי הַשִּׁבְעָה הָאֲבֵלִים אֵינָם יוֹצְאִים מֵהַבַּיִת – וְלָכֵן אָנוּ בָּאִים אֲלֵיהֶם.
💛 אַהֲבַת הַשֵּׁם: הַיְּשִׁיבָה הַנְּמוּכָה מַרְאָה אֶת הַצַּעַר. וּכְשֶׁאָנוּ בָּאִים וְיוֹשְׁבִים לְיָדָם, הֵם מַרְגִּישִׁים שֶׁעַם יִשְׂרָאֵל כֻּלּוֹ אִתָּם.
📖 פֶּרֶק י׳, הֲלָכָה ז׳

🤫בְּשֶׁקֶט וּבְכֹבֶד רֹאשׁ

בְּבֵית הָאֲבֵלִים מְדַבְּרִים בְּקוֹל שָׁקֵט וּבְכֹבֶד רֹאשׁ, לִכְבוֹד הַנִּפְטָר. עָדִיף לִשְׁתֹּק מֵאֲשֶׁר לְדַבֵּר עַל דִּבְרֵי חֻלִּין וּצְחוֹק.
אֶת הָאָבֵל לֹא שׁוֹאֲלִים לִשְׁלוֹמוֹ, וְלֹא אוֹמְרִים לוֹ ״שָׁלוֹם״ – כִּי בַּיָּמִים הָאֵלֶּה הוּא אֵינוֹ שָׁרוּי בְּשָׁלוֹם. וְיֵשׁ שֶׁנִּמְנָעִים גַּם מִלְּחִיצַת יָד.
💛 אַהֲבַת הַשֵּׁם: לִפְעָמִים הַנֶּחָמָה הֲכִי גְּדוֹלָה הִיא לֹא מִלִּים, אֶלָּא לֵב שֶׁיּוֹשֵׁב לְיַד לֵב.
📖 פֶּרֶק י׳, הֲלָכוֹת ז׳–ח׳

🍲עוֹזְרִים לָאֲבֵלִים

אֶת הָאֲרוּחָה הָרִאשׁוֹנָה שֶׁל הָאֲבֵלִים מְכִינִים לָהֶם הַשְּׁכֵנִים אוֹ הַקְּרוֹבִים – כְּדֵי לְחַזֵּק אוֹתָם וּלְהַחְזִיר אוֹתָם לַחַיִּים. וְגַם בְּכָל יְמֵי הַשִּׁבְעָה נוֹהֲגִים לְהָכִין לָהֶם אֹכֶל.
רַבִּים מַגִּישִׁים בָּאֲרוּחָה הָרִאשׁוֹנָה מַאֲכָלִים עֲגֻלִּים, כְּמוֹ בֵּיצִים וַעֲדָשִׁים: ״גַּלְגַּל חוֹזֵר בָּעוֹלָם״ – וְעוֹד יָבוֹאוּ לָהֶם יָמִים טוֹבִים. וְלָעִגּוּל אֵין ״פֶּה״, לִרְמֹז שֶׁיָּפָה הַשְּׁתִיקָה.
💛 אַהֲבַת הַשֵּׁם: סִיר אֹכֶל חַם שֶׁמְּבִיאִים לָאֲבֵלִים הוּא חֶסֶד גָּדוֹל – וְכָל חֶסֶד שֶׁאָנוּ עוֹשִׂים מְשַׂמֵּחַ אֶת ה׳.
📖 פֶּרֶק י׳, הֲלָכָה ז׳

🕯️נֵר נְשָׁמָה וּתְפִלָּה

רַבִּים נוֹהֲגִים לְהַדְלִיק נֵר לְעִלּוּי נִשְׁמַת הַנִּפְטָר בְּכָל שִׁבְעַת הַיָּמִים. וּמִשְׁתַּדְּלִים לְהִתְפַּלֵּל בְּמִנְיָן בְּבֵית הָאֲבֵלִים, וּבִימֵי שֵׁנִי וַחֲמִישִׁי מְבִיאִים לְשָׁם סֵפֶר תּוֹרָה.
לְיַד הַתְּפִלּוֹת אוֹמְרִים דִּבְרֵי תּוֹרָה וְלוֹמְדִים מִשְׁנָיוֹת – כִּי ״מִשְׁנָה״ בְּאוֹתִיּוֹת ״נְשָׁמָה״.
💛 אַהֲבַת הַשֵּׁם: כָּל תְּפִלָּה וְכָל מִשְׁנָה שֶׁאוֹמְרִים שָׁם הֵן כְּמוֹ אוֹר שֶׁעוֹלֶה אֶל הַנְּשָׁמָה.
📖 פֶּרֶק י׳, הֲלָכָה ז׳

🗣️מָה אוֹמְרִים וּמָתַי

מֻתָּר לַמְנַחֲמִים לְהִתְעַנְיֵן מְעַט בְּחַיֵּי הָאָבֵל, אֲבָל לֹא לְהִגָּרֵר לְשִׂיחוֹת אֲרֻכּוֹת עַל עֲבוֹדָה וְעַל עִנְיְנֵי חֻלִּין. וּכְבָר אָמְרוּ חֲכָמִים: הַמְנַחֲמִים אֵינָם פּוֹתְחִים בְּדִבּוּר, עַד שֶׁהָאָבֵל מַתְחִיל לְדַבֵּר.
בָּאִים בְּשָׁעוֹת מְקֻבָּלוֹת, לֹא מַאֲרִיכִים כְּשֶׁהָאֲבֵלִים עֲיֵפִים, וּמַשְׁאִירִים מָקוֹם גַּם לִמְנַחֲמִים אֲחֵרִים.
💛 אַהֲבַת הַשֵּׁם: נִימוּס וְרָגִישׁוּת הֵם חֵלֶק מִן הַמִּצְוָה. הַמְנַחֵם הַטּוֹב חוֹשֵׁב עַל הָאָבֵל, וְלֹא עַל עַצְמוֹ.
📖 פֶּרֶק י׳, הֲלָכוֹת ז׳–ח׳

🏠מִצְוָה גְּדוֹלָה לְנַחֵם

מִצְוָה גְּדוֹלָה לְנַחֵם אֲבֵלִים. עִקַּר הַנֶּחָמָה: לְהִשְׁתַּתֵּף בְּצַעֲרָם, לְסַפֵּר בְּשִׁבְחוֹ שֶׁל הַנִּפְטָר, וּלְהִתְעוֹרֵר מִכָּךְ לְמַעֲשִׂים טוֹבִים.
בִּימֵי הַשִּׁבְעָה מְסַפְּרִים עַל הַצְּדָדִים הַטּוֹבִים שֶׁל הַנִּפְטָר. הֵם הָעִקָּר – הֵם הַנְּשָׁמָה שֶׁלּוֹ, וְהֵם נִשְׁאָרִים לָנֶצַח.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאָנוּ בָּאִים לְנַחֵם, אָנוּ הוֹלְכִים בְּדַרְכֵי ה׳: כְּמוֹ שֶׁה׳ נִחֵם אֶת יִצְחָק אַחֲרֵי פְּטִירַת אַבְרָהָם – כָּךְ גַּם אָנוּ מְנַחֲמִים.
📖 פֶּרֶק י׳, הֲלָכָה ח׳

💎הַצְּדָדִים הַטּוֹבִים

לֹא פֶּלֶא שֶׁבִּימֵי הַשִּׁבְעָה מִתְבָּרֵר שֶׁהַנִּפְטָר הָיָה טוֹב יוֹתֵר מִמָּה שֶׁהִכִּירוּ: בַּחַיִּים הָרְגִילִים אֵינֶנּוּ שָׂמִים לֵב לָעִקָּר הַטּוֹב. אֲבָל הָאֱמֶת הִיא שֶׁהַצְּדָדִים הַטּוֹבִים הֵם הָעִקָּר.
כְּשֶׁמְּסַפְּרִים עֲלֵיהֶם – מְרוֹמְמִים אֶת זִכְרוֹ שֶׁל הַנִּפְטָר, מַמְשִׁיכִים אֶת דַּרְכּוֹ הַטּוֹבָה, וּמוֹעִילִים לְעִלּוּי נִשְׁמָתוֹ.
💛 אַהֲבַת הַשֵּׁם: גַּם בְּכָל אָדָם חַי כְּדַאי לְחַפֵּשׂ אֶת הַטּוֹב – כִּי הוּא הָעִקָּר, וְהוּא הַנְּשָׁמָה הַקְּדוֹשָׁה שֶׁה׳ נָתַן בּוֹ.

📖 קַדִּישׁ – מִלְּשׁוֹן קְדֻשָּׁה: תְּפִלָּה בַּאֲרָמִית שֶׁאוֹמְרִים בַּצִּבּוּר, וּבָהּ מְקַדְּשִׁים אֶת שֵׁם ה׳. הַבֵּן מְקַדֵּשׁ שֵׁם שָׁמַיִם – וְהַזְּכוּת עוֹלָה לְהוֹרָיו.

📖 פֶּרֶק י׳, הֲלָכָה י״ב

📅יוֹם הַשָּׁנָה

יוֹם הַשָּׁנָה – ״יָארְצַייט״ – הוּא יוֹם הַפְּטִירָה שֶׁל הָאָב אוֹ הָאֵם, בְּכָל שָׁנָה. כְּשֶׁאֶפְשָׁר, הַבָּנִים וְהַבָּנוֹת עוֹלִים לַקֶּבֶר. וּבַיּוֹם הַזֶּה מוֹסִיפִים בְּתוֹרָה, בְּמַעֲשִׂים טוֹבִים, בִּצְדָקָה, בִּתְפִלָּה וּבְקַדִּישׁ לְעִלּוּי נִשְׁמָתָם.
רַבִּים נוֹהֲגִים לְהִתְעַנּוֹת בְּיוֹם הַשָּׁנָה לְשֵׁם תְּשׁוּבָה, וּמִי שֶׁהַתַּעֲנִית קָשָׁה לוֹ – תּוֹרֵם צְדָקָה בִּמְקוֹמָהּ. פַּעַם הָיוּ מַדְלִיקִים נֵרוֹת בְּבֵית הַכְּנֶסֶת כְּדֵי לְהוֹסִיף אוֹר בַּתְּפִלָּה. הַיּוֹם יֵשׁ חַשְׁמַל, וְלָכֵן הָעִקָּר הוּא לִתְרֹם לְבֵית הַכְּנֶסֶת אוֹ לַעֲנִיִּים, וְיֵשׁ מַדְלִיקִים נֵר נְשָׁמָה בַּבַּיִת לְזֵכֶר.
💛 אַהֲבַת הַשֵּׁם: בְּכָל שָׁנָה מֵחָדָשׁ אָנוּ מְחַבְּרִים אֶת הַזִּכָּרוֹן לְמַעֲשִׂים טוֹבִים – וְכָךְ הָאַהֲבָה לַהוֹרִים מַמְשִׁיכָה לִפְרֹחַ.
📖 פֶּרֶק י׳, הֲלָכָה י״ד

🌱הַהַשְׁפָּעָה מַמְשִׁיכָה

חֲכָמִים אָמְרוּ: חַיָּיו שֶׁל הַנִּפְטָר הִסְתַּיְּמוּ, אֲבָל הַהַשְׁפָּעָה שֶׁלּוֹ לֹא הִסְתַּיְּמָה. כְּשֶׁהַיְּלָדִים שֶׁהִשְׁאִיר מוֹסִיפִים מַעֲשִׂים טוֹבִים – הַחִנּוּךְ וְהַדֶּרֶךְ שֶׁלּוֹ מַמְשִׁיכִים לְהַשְׁפִּיעַ, וְהוּא זוֹכֶה לְשָׂכָר וְלַעֲלִיָּה.
לָכֵן בְּיוֹם הַשָּׁנָה הַיְּלָדִים מִתְעוֹרְרִים לִתְשׁוּבָה, וּמוֹסִיפִים בְּתוֹרָה, בִּצְדָקָה, בִּתְפִלָּה וּבְקַדִּישׁ לְעִלּוּי נִשְׁמַת הוֹרֵיהֶם.
💛 אַהֲבַת הַשֵּׁם: כָּל מִצְוָה שֶׁאָנוּ עוֹשִׂים הִיא גַּם מַתָּנָה לְכָל מִי שֶׁחִנֵּךְ אוֹתָנוּ – הַהוֹרִים, הַסָּבִים וְהַמּוֹרִים.
📖 פֶּרֶק י׳, הֲלָכָה י״ד

📅🔑 מוּשָּׂג: יָארְצַייט

״יוֹם הַשָּׁנָה (יָארְצַייט) הוּא יוֹם הַמִּיתָה שֶׁל הָאָב אוֹ הָאֵם בְּכָל שָׁנָה וְשָׁנָה״ – ״נוֹהֲגִים הַיְּלָדִים... לְהוֹסִיף בְּלִמּוּד תּוֹרָה וּבְמַעֲשִׂים טוֹבִים וּבְמַתַּן צְדָקָה וּבִתְפִלָּה וּבְקַדִּישׁ״.
📖 יָארְצַייט – בְּיִידִישׁ: ״זְמַן שָׁנָה״. יוֹם הַשָּׁנָה לַפְּטִירָה, שֶׁחוֹזֵר בְּכָל שָׁנָה בְּאוֹתוֹ תַּאֲרִיךְ.
💛 אַהֲבַת הַשֵּׁם: ״אִם הֵם מִתְעַלִּים בְּמַעֲשִׂים טוֹבִים, הֲרֵי שֶׁחִנּוּכוֹ וּמוֹרַשְׁתּוֹ מַמְשִׁיכִים לְהַשְׁפִּיעַ לְטוֹבָה״ – כָּל מִצְוָה שֶׁלְּךָ הִיא גַּם זְכוּת לְהוֹרֶיךָ. 🧒 וְאַתָּה? הַיּוֹם – עֲשֵׂה מִצְוָה אַחַת וֶאֱמֹר בַּלֵּב: ״זֶה בִּזְכוּת אַבָּא וְאִמָּא שֶׁחִנְּכוּ אוֹתִי״.
📖 פֶּרֶק י׳, הֲלָכָה י״ד

🕯️🔑 מוּשָּׂג: נֵר נְשָׁמָה

״בֶּעָבָר הָיוּ רְגִילִים לְהַדְלִיק נֵר נְשָׁמָה בְּבֵית הַכְּנֶסֶת כְּדֵי שֶׁיּוֹסִיף אוֹר בַּתְּפִלָּה לִכְבוֹד שָׁמַיִם... יֵשׁ נוֹהֲגִים לְהַדְלִיק נֵר נְשָׁמָה בַּבַּיִת, וְהָעִקָּר לָתֵת בִּמְקוֹם הַנֵּרוֹת תְּרוּמָה לְהַחְזָקַת בֵּית הַכְּנֶסֶת אוֹ לַעֲנִיִּים״.
📖 נֵר נְשָׁמָה – נֵר שֶׁמַּדְלִיקִים בְּיוֹם הַשָּׁנָה לְזֵכֶר הַנִּפְטָר. פַּעַם הִדְלִיקוּ אוֹתוֹ בְּבֵית הַכְּנֶסֶת, כְּדֵי שֶׁיּוֹסִיף אוֹר בַּתְּפִלָּה.
💛 אַהֲבַת הַשֵּׁם: נֵר מֵאִיר לַאֲחֵרִים – וְכָךְ גַּם צְדָקָה וּמַעֲשִׂים טוֹבִים מְאִירִים לִזְכוּת הַנְּשָׁמָה. 🧒 וְאַתָּה? בְּיוֹם הַזִּכָּרוֹן שֶׁל קָרוֹב – תֵּן מַטְבֵּעַ לִצְדָקָה לְעִלּוּי נִשְׁמָתוֹ.
📖 פֶּרֶק י׳, הֲלָכָה י״ד

🧠 בְּדֹק אֶת עַצְמְךָ

יָשַׁבְתָּ עִם אַבָּא בְּבֵית הָאֲבֵלִים, וְהָאָבֵל עֲדַיִן שׁוֹתֵק. מָה נָכוֹן לַעֲשׂוֹת?
💡 הַמְנַחֲמִים אֵינָם פּוֹתְחִים בְּדִבּוּר עַד שֶׁהָאָבֵל מַתְחִיל לְדַבֵּר. וְעָדִיף לִשְׁתֹּק מֵאֲשֶׁר לְדַבֵּר בְּעִנְיְנֵי חֻלִּין וּצְחוֹק.
בָּאתָ לְנַחֵם, וְהָאֲבֵלִים נִרְאִים עֲיֵפִים מְאוֹד. מָה עוֹשִׂים?
💡 הַמְנַחֲמִים נוֹהֲגִים בְּנִימוּס: בָּאִים בְּשָׁעוֹת מְקֻבָּלוֹת, לֹא מַאֲרִיכִים כְּשֶׁהָאֲבֵלִים עֲיֵפִים, וּמַשְׁאִירִים מָקוֹם לִמְנַחֲמִים אֲחֵרִים.
מָה עִקַּר הַנֶּחָמָה לָאֲבֵלִים?
💡 עִקַּר הַנֶּחָמָה: לְהִשְׁתַּתֵּף בְּצַעַר הָאֲבֵלִים, לְסַפֵּר בְּשִׁבְחוֹ שֶׁל הַנִּפְטָר וּלְהִתְעוֹרֵר לְמַעֲשִׂים טוֹבִים. מֻתָּר וְרָאוּי לְהִצְטַעֵר – רַק לֹא יוֹתֵר מִדַּי.

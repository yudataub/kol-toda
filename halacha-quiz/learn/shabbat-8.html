הַבְדָּלָה וּמוֹצָאֵי שַׁבָּת

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק ח׳

✨בֵּין קֹדֶשׁ לְחוֹל

כְּמוֹ שֶׁמְּקַבְּלִים אֶת הַשַּׁבָּת בְּקִדּוּשׁ, כָּךְ נִפְרָדִים מִמֶּנָּה בְּהַבְדָּלָה – בִּתְפִלָּה וְעַל הַכּוֹס. בַּתְּפִלָּה מַבְדִּילִים בַּבְּרָכָה הָרְבִיעִית שֶׁל הָעֲמִידָה, בַּקֶּטַע ״אַתָּה חוֹנַנְתָּנוּ״ – כִּי זוֹ בִּרְכַּת הַדַּעַת, וּבְלִי דַּעַת אִי אֶפְשָׁר לְהַבְחִין בֵּין קֹדֶשׁ לְחוֹל.
מִי שֶׁשָּׁכַח ״אַתָּה חוֹנַנְתָּנוּ״ בַּתְּפִלָּה – אֵינוֹ חוֹזֵר לְהִתְפַּלֵּל, וְיוֹצֵא בַּהַבְדָּלָה שֶׁעַל הַכּוֹס.
💛 אַהֲבַת הַשֵּׁם: בַּתְּחִלָּה, כְּשֶׁיִּשְׂרָאֵל הָיוּ עֲנִיִּים, תִּקְּנוּ לְהַבְדִּיל רַק בַּתְּפִלָּה, וּכְשֶׁהֶעֱשִׁירוּ – הוֹסִיפוּ אֶת הַכּוֹס. הַהַבְדָּלָה מְלַמֶּדֶת אוֹתָנוּ לְהַרְגִּישׁ מָה קָדוֹשׁ – וְלָקַחַת אֶת הַקְּדֻשָּׁה אִתָּנוּ אֶל הַשָּׁבוּעַ.
📖 פֶּרֶק ח׳, הֲלָכָה א׳

👩נָשִׁים וּבָנוֹת בַּהַבְדָּלָה

נָשִׁים חַיָּבוֹת בַּהַבְדָּלָה כְּמוֹ גְּבָרִים, כִּי הִיא קְשׁוּרָה לַקִּדּוּשׁ. יֵשׁ מִי שֶׁאוֹמֵר שֶׁהֵן פְּטוּרוֹת, וְלָכֵן נוֹהֲגוֹת לְכַתְּחִלָּה לִשְׁמֹעַ הַבְדָּלָה מִגֶּבֶר. אֲבָל כְּשֶׁאֵין גֶּבֶר – אִשָּׁה חַיֶּבֶת לְהַבְדִּיל לְעַצְמָהּ, בְּכָל אַרְבַּע הַבְּרָכוֹת. וְאִם הַגֶּבֶר כְּבָר הִבְדִּיל לְעַצְמוֹ – נָכוֹן שֶׁהִיא תַּבְדִּיל לְעַצְמָהּ.
נָשִׁים שֶׁאֵינָן מִתְפַּלְּלוֹת עַרְבִית יוֹצְאוֹת בַּהַבְדָּלָה שֶׁעַל הַכּוֹס. אֲבָל לִפְנֵי שֶׁעוֹשׂוֹת מְלָאכָה, כְּמוֹ הַדְלָקַת אוֹר, צְרִיכוֹת לוֹמַר: ״בָּרוּךְ הַמַּבְדִּיל בֵּין קֹדֶשׁ לְחוֹל״.
💛 אַהֲבַת הַשֵּׁם: הַהַבְדָּלָה הִיא פְּרֵדָה מִן הַשַּׁבָּת הָאֲהוּבָה. כָּל אֶחָד וְאַחַת מִבְּנֵי הַבַּיִת שׁוֹמְעִים, מְרִיחִים וְרוֹאִים אֶת הָאוֹר – וְלוֹקְחִים אִתָּם קְדֻשָּׁה לְכָל הַשָּׁבוּעַ.
📖 פֶּרֶק ח׳, הֲלָכוֹת א׳, ח׳

🍷אַרְבַּע בְּרָכוֹת

בַּהַבְדָּלָה עַל הַכּוֹס אוֹמְרִים פְּסוּקִים שֶׁל בְּרָכָה, וְאָז אַרְבַּע בְּרָכוֹת: ״בּוֹרֵא פְּרִי הַגָּפֶן״ עַל הַיַּיִן, בְּרָכָה עַל הַבְּשָׂמִים, ״בּוֹרֵא מְאוֹרֵי הָאֵשׁ״ עַל הַנֵּר, וּ״הַמַּבְדִּיל בֵּין קֹדֶשׁ לְחוֹל״. הַיַּיִן וְ״הַמַּבְדִּיל״ – הֵם הָעִקָּר.
כְּשֶׁאֵין יַיִן – אֶפְשָׁר לְהַבְדִּיל עַל ״חֲמַר מְדִינָה״: מַשְׁקֶה חָשׁוּב שֶׁשּׁוֹתִים בַּמְּדִינָה כְּמוֹ יַיִן, כְּגוֹן בִּירָה, וּמְבָרְכִים עָלָיו ״שֶׁהַכֹּל״. מַשְׁקֶה קַל כְּמוֹ קוֹלָה – אֵינוֹ מַתְאִים.
💛 אַהֲבַת הַשֵּׁם: סֵדֶר הַבְּרָכוֹת עוֹלֶה כְּמוֹ סֻלָּם: מִטַּעַם – לְרֵיחַ, מֵרֵיחַ – לִרְאִיָּה, וּמֵרְאִיָּה – לַשֵּׂכֶל שֶׁמַּבְדִּיל בֵּין קֹדֶשׁ לְחוֹל. כָּךְ הַשַּׁבָּת מְרוֹמֶמֶת אוֹתָנוּ – מִן הַגּוּף אֶל הַנְּשָׁמָה.
📖 פֶּרֶק ח׳, הֲלָכוֹת ב׳, ד׳

🔀שְׁנֵי מִנְהָגִים בַּהַבְדָּלָה – וּשְׁנֵיהֶם כַּהֲלָכָה

בְּבִרְכַּת הַבְּשָׂמִים: הָאַשְׁכְּנַזִּים מְבָרְכִים עַל כָּל סוּג ״בּוֹרֵא מִינֵי בְשָׂמִים״ – כְּדֵי שֶׁמִּי שֶׁאֵינוֹ בָּקִי בְּסוּגֵי הַבְּשָׂמִים לֹא יִטְעֶה, וּבְרָכָה זוֹ מוֹעִילָה לְכָל הַסּוּגִים. הַסְּפָרַדִּים מְבָרְכִים אֶת הַבְּרָכָה הַמְּיֻחֶדֶת לְכָל סוּג: עַל הֲדַס וְעַל רוֹזְמָרִין – ״בּוֹרֵא עֲצֵי בְשָׂמִים״.
גַּם לְגַבֵּי עֲמִידָה בַּהַבְדָּלָה יֵשׁ שְׁנֵי מִנְהָגִים: יֵשׁ נוֹהֲגִים, לְפִי הַשֻּׁלְחָן עָרוּךְ, לְהַבְדִּיל בִּישִׁיבָה – כְּדֵי שֶׁכָּל הַשּׁוֹמְעִים יִהְיוּ קְבוּעִים יַחַד עִם הַמַּבְדִּיל. וְיֵשׁ נוֹהֲגִים, לְפִי הָרָמָ״א, לְהַבְדִּיל בַּעֲמִידָה – לִכְבוֹד הַשַּׁבָּת שֶׁיּוֹצֵאת.
💛 אַהֲבַת הַשֵּׁם: שְׁנֵי הַמִּנְהָגִים יוֹצְאִים מֵאוֹתוֹ לֵב: לְהָשִׁיב אֶת הַנֶּפֶשׁ בְּרֵיחַ טוֹב, וּלְכַבֵּד אֶת הַשַּׁבָּת עַד הָרֶגַע הָאַחֲרוֹן. כָּל קְהִלָּה שׁוֹמֶרֶת עַל דֶּרֶךְ רַבּוֹתֶיהָ – וְכֻלָּן אֲהוּבוֹת לַה׳.
📖 פֶּרֶק ח׳, הֲלָכוֹת ג׳, ה׳

🌿בְּשָׂמִים

בְּמוֹצָאֵי שַׁבָּת הַנְּשָׁמָה הַיְתֵרָה עוֹזֶבֶת, וְהַנֶּפֶשׁ מִצְטַעֶרֶת – וּכְדֵי לְהָשִׁיב אֶת רוּחָהּ מְרִיחִים רֵיחַ טוֹב. כָּל הַשּׁוֹמְעִים מְרִיחִים, וְהַמַּבְדִּיל מְחַכֶּה לָהֶם. אֵין מְבָרְכִים עַל רֵיחַ שֶׁנּוֹעַד לְהַעֲבִיר רֵיחַ רַע, כְּמוֹ מְטַהֵר אֲוִיר בַּשֵּׁרוּתִים.
בְּנֹסַח הַבְּרָכָה יֵשׁ מִנְהָגִים: הָאַשְׁכְּנַזִּים מְבָרְכִים תָּמִיד ״בּוֹרֵא מִינֵי בְשָׂמִים״, וְהַסְּפָרַדִּים מְבָרְכִים אֶת הַבְּרָכָה הַמְּיֻחֶדֶת לְכָל סוּג – כְּמוֹ ״עֲצֵי בְשָׂמִים״ עַל הֲדַס.
💛 אַהֲבַת הַשֵּׁם: הָרֵיחַ הוּא הַחוּשׁ שֶׁהַנְּשָׁמָה נֶהֱנֵית מִמֶּנּוּ. גַּם מִי שֶׁאֵינוֹ מַרְגִּישׁ צַעַר בְּצֵאת הַשַּׁבָּת – הַבְּשָׂמִים מַזְכִּירִים לוֹ כַּמָּה הִיא הָיְתָה יְקָרָה.
📖 פֶּרֶק ח׳, הֲלָכָה ה׳

🔥בּוֹרֵא מְאוֹרֵי הָאֵשׁ

בְּמוֹצָאֵי שַׁבָּת נָתַן ה׳ דֵּעָה בְּאָדָם הָרִאשׁוֹן, וְהוּא הִקִּישׁ שְׁתֵּי אֲבָנִים וְיָצְאָה אֵשׁ – לְזֵכֶר זֶה מְבָרְכִים עַל הַנֵּר. לְכַתְּחִלָּה מְבָרְכִים עַל ״אֲבוּקָה״ – נֵר קָלוּעַ עִם כַּמָּה פְּתִילוֹת, וְאִם אֵין – אֶפְשָׁר עַל שְׁנֵי גַּפְרוּרִים.
נוֹהֲגִים לְהַבִּיט לְאוֹר הַנֵּר בַּצִּפָּרְנַיִם וּבְכַף הַיָּד. לֹא מְבָרְכִים עַל נֵר נְשָׁמָה, שֶׁהוּדְלַק לְכָבוֹד וְלֹא כְּדֵי לְהָאִיר. וְעַל נוּרַת חַשְׁמַל – רַבִּים אוֹמְרִים שֶׁאֵין מְבָרְכִים.
💛 אַהֲבַת הַשֵּׁם: גַּם הָאֵשׁ הִיא מַתָּנָה שֶׁל ה׳. אָדָם הָרִאשׁוֹן קִבֵּל אוֹתָהּ בְּמוֹצָאֵי הַשַּׁבָּת הָרִאשׁוֹנָה – וְגַם אֲנַחְנוּ מוֹדִים עָלֶיהָ, וְיוֹצְאִים אֶל הַשָּׁבוּעַ עִם אוֹר.
📖 פֶּרֶק ח׳, הֲלָכָה ו׳

📅מִי שֶׁלֹּא הִבְדִּיל

מִי שֶׁלֹּא הִבְדִּיל עַל הַכּוֹס בְּמוֹצָאֵי שַׁבָּת – מֵאֹנֶס, כְּמוֹ חַיָּל בְּמִבְצָע, אוֹ מִשִּׁכְחָה – יָכוֹל לְהַשְׁלִים. לְפִי רֹב הַפּוֹסְקִים – עַד סוֹף יוֹם שְׁלִישִׁי, כִּי שְׁלֹשֶׁת הַיָּמִים הָרִאשׁוֹנִים קְשׁוּרִים לַשַּׁבָּת שֶׁעָבְרָה. וְיֵשׁ נוֹהֲגִים לְהַשְׁלִים רַק בְּיוֹם רִאשׁוֹן, לִפְנֵי שֶׁאָכְלוּ.
בַּהַשְׁלָמָה מְבָרְכִים רַק עַל הַיַּיִן וְ״הַמַּבְדִּיל״. עַל הַבְּשָׂמִים וְעַל הַנֵּר מְבָרְכִים רַק בְּמוֹצָאֵי שַׁבָּת עַצְמוֹ.
💛 אַהֲבַת הַשֵּׁם: הַשַּׁבָּת מַשְׁפִּיעָה עוֹד יָמִים אַחֲרֶיהָ. גַּם בְּאֶמְצַע הַשָּׁבוּעַ אֶפְשָׁר לְהַרְגִּישׁ אֶת הַבְּרָכָה שֶׁלָּהּ – וּלְהִתְכּוֹנֵן כְּבָר לַשַּׁבָּת הַבָּאָה.
📖 פֶּרֶק ח׳, הֲלָכָה ז׳

🔀שְׁתֵּי שִׁיטוֹת בְּמוֹצָאֵי שַׁבָּת

מִי שֶׁלֹּא הִבְדִּיל בְּמוֹצָאֵי שַׁבָּת: לְפִי רֹב הַפּוֹסְקִים מַשְׁלִים עַל הַכּוֹס עַד סוֹף יוֹם שְׁלִישִׁי, גַּם אִם כְּבָר אָכַל. וְיֵשׁ נוֹהֲגִים לְהַשְׁלִים בִּבְרָכָה רַק בְּיוֹם רִאשׁוֹן, וּבִתְנַאי שֶׁלֹּא אָכַל דָּבָר מִמּוֹצָאֵי שַׁבָּת.
מִשְּׁקִיעַת הַחַמָּה וְעַד הַהַבְדָּלָה אָסוּר לֶאֱכֹל וְלִשְׁתּוֹת. מַיִם – לְפִי הַשֻּׁלְחָן עָרוּךְ מֻתָּר לִשְׁתּוֹת, כִּי אֵין לָהֶם חֲשִׁיבוּת; וְיֵשׁ מֵהָאַחֲרוֹנִים שֶׁאָסְרוּ גַּם מַיִם.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאָנוּ מַמְתִּינִים לַהַבְדָּלָה לִפְנֵי שֶׁאוֹכְלִים, אָנוּ מַרְאִים שֶׁהַשַּׁבָּת חֲשׁוּבָה לָנוּ עַד הָרֶגַע הָאַחֲרוֹן – וְנִפְרָדִים מִמֶּנָּה בְּכָבוֹד.
📖 פֶּרֶק ח׳, הֲלָכוֹת ז׳–ח׳

⏳מָתַי מֻתָּר לַעֲשׂוֹת מְלָאכָה – וּמָתַי לֶאֱכֹל?

גַּם בְּצֵאת הַשַּׁבָּת מוֹסִיפִים מִן הַחוֹל עַל הַקֹּדֶשׁ, וּמְחַכִּים כַּמָּה דַּקּוֹת אַחֲרֵי צֵאת הַכּוֹכָבִים. מְלָאכָה מֻתֶּרֶת רַק אַחֲרֵי שֶׁהִבְדִּילוּ בְּדִבּוּר – בַּתְּפִלָּה, אוֹ בַּאֲמִירַת ״בָּרוּךְ הַמַּבְדִּיל בֵּין קֹדֶשׁ לְחוֹל״. לֶאֱכֹל וְלִשְׁתּוֹת – רַק אַחֲרֵי הַהַבְדָּלָה עַל הַכּוֹס.
מִי שֶׁהִתְחִיל סְעוּדָה שְׁלִישִׁית לִפְנֵי הַשְּׁקִיעָה – מַמְשִׁיךְ לֶאֱכֹל גַּם אַחַר כָּךְ, וְאֵינוֹ צָרִיךְ לְהַפְסִיק בִּגְלַל הַהַבְדָּלָה.
💛 אַהֲבַת הַשֵּׁם: לֹא מְמַהֲרִים לִבְרֹחַ מִן הַשַּׁבָּת. רַק אַחֲרֵי שֶׁנִּפְרַדְנוּ מִמֶּנָּה בְּדִבּוּר וּבִבְרָכָה – חוֹזְרִים לַחוֹל, וְלוֹקְחִים אִתָּנוּ אֶת הָאוֹר שֶׁלָּהּ.
📖 פֶּרֶק ח׳, הֲלָכָה ח׳

🧠 בְּדֹק אֶת עַצְמְךָ

יָצְאוּ הַכּוֹכָבִים. אֵיתָן כְּבָר הִתְפַּלֵּל עַרְבִית וְאָמַר ״אַתָּה חוֹנַנְתָּנוּ״, אֲבָל עוֹד לֹא שָׁמַע הַבְדָּלָה עַל הַכּוֹס. מָה מֻתָּר לוֹ?
💡 אַחֲרֵי הַבְדָּלָה בְּדִבּוּר – בַּתְּפִלָּה – מֻתָּר לַעֲשׂוֹת מְלָאכָה. אֲבָל לֶאֱכֹל וְלִשְׁתּוֹת מֻתָּר רַק אַחֲרֵי הַהַבְדָּלָה עַל הַכּוֹס, מִפְּנֵי חֲשִׁיבוּתָהּ.
חַיָּל הָיָה בְּמִבְצָע בְּמוֹצָאֵי שַׁבָּת וְלֹא הִבְדִּיל. בְּיוֹם רִאשׁוֹן בַּבֹּקֶר, עוֹד לִפְנֵי שֶׁאָכַל, הוּא חוֹזֵר לַבָּסִיס. מָה יַעֲשֶׂה?
💡 מִי שֶׁלֹּא הִבְדִּיל – מַשְׁלִים אַחַר כָּךְ עַל הַכּוֹס. אֲבָל הַבְּשָׂמִים בָּאִים לְהָשִׁיב אֶת הַנֶּפֶשׁ בְּצֵאת הַשַּׁבָּת, וְהַנֵּר – לְזֵכֶר הָאֵשׁ שֶׁל מוֹצָאֵי שַׁבָּת; לָכֵן מְבָרְכִים עֲלֵיהֶם רַק בְּמוֹצָאֵי שַׁבָּת.
מוֹצָאֵי שַׁבָּת. שִׁירָה לֹא הִתְפַּלְּלָה עַרְבִית, וְרוֹצָה לְהַדְלִיק אוֹר לִפְנֵי שֶׁאַבָּא מַבְדִּיל. מָה תַּעֲשֶׂה?
💡 אַחֲרֵי צֵאת הַשַּׁבָּת אָסוּר לַעֲשׂוֹת מְלָאכָה עַד שֶׁמַּבְדִּילִים בְּדִבּוּר – בַּתְּפִלָּה, אוֹ בַּאֲמִירַת ״בָּרוּךְ הַמַּבְדִּיל בֵּין קֹדֶשׁ לְחוֹל״. מִי שֶׁלֹּא הִתְפַּלְּלָה עַרְבִית אוֹמֶרֶת אֶת הַמִּשְׁפָּט הַזֶּה, וְאָז מֻתָּר לָהּ לְהַדְלִיק. לֶאֱכֹל – רַק אַחֲרֵי הַהַבְדָּלָה עַל הַכּוֹס.

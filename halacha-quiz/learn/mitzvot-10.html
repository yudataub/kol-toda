הִלְכוֹת מְזוּזָה

🧵 צִיצִית, תְּפִלִּין וּמְזוּזָה · פֶּרֶק י׳

🏠מָה יֵשׁ בַּמְּזוּזָה?

בַּתּוֹרָה כָּתוּב פַּעֲמַיִם: "וּכְתַבְתָּם עַל מְזוּזוֹת בֵּיתֶךָ וּבִשְׁעָרֶיךָ". לָכֵן כּוֹתְבִים עַל קְלָף שְׁתֵּי פָּרָשִׁיּוֹת – שְׁמַע, וְהָיָה אִם שָׁמֹעַ – וְקוֹבְעִים אוֹתוֹ בְּפֶתַח הַבַּיִת. בַּתְּפִלִּין יֵשׁ אַרְבַּע פָּרָשִׁיּוֹת, וּבַמְּזוּזָה רַק שְׁתַּיִם. בְּמִצְוַת מְזוּזָה כֻּלָּם חַיָּבִים בְּשָׁוֶה.
בַּבַּיִת הַבָּטוּחַ קַל לִשְׁכֹּחַ אֶת ה׳. לָכֵן נוֹהֲגִים לָשִׂים יָד עַל הַמְּזוּזָה וּלְנַשֵּׁק אוֹתָהּ, כְּשֶׁיּוֹצְאִים וּכְשֶׁנִּכְנָסִים – כְּדֵי לְהִזָּכֵר בְּמָה שֶׁכָּתוּב בָּהּ.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים לָמְדוּ מִן הַתּוֹרָה: מִי שֶׁזָּהִיר בְּמִצְוַת מְזוּזָה – יַאֲרִיכוּ יָמָיו וִימֵי בָּנָיו. הַנְּשִׁיקָה לַמְּזוּזָה הִיא כְּמוֹ "שָׁלוֹם" חַם לַה׳, בְּכָל פַּעַם שֶׁאַתָּה יוֹצֵא אוֹ חוֹזֵר הַבַּיְתָה.
📖 פֶּרֶק י׳, הֲלָכוֹת א׳–ב׳

🏡מְזוּזָה – לְכֻלָּם בְּשָׁוֶה

תְּפִלִּין מַנִּיחִים רַק גְּבָרִים, אֲבָל בְּמִצְוַת מְזוּזָה כֻּלָּם חַיָּבִים בְּשָׁוֶה – אֲנָשִׁים וְנָשִׁים. לָכֵן אִשָּׁה שֶׁגָּרָה לְבַדָּהּ בְּבֵיתָהּ חַיֶּבֶת לִקְבֹּעַ בּוֹ מְזוּזָה, וּמְבָרֶכֶת "לִקְבֹּעַ מְזוּזָה".
אֶפְשָׁר לוֹמַר שֶׁהָאִשָּׁה שַׁיֶּכֶת לְמִצְוַת הַמְּזוּזָה עוֹד יוֹתֵר מֵהָאִישׁ: הִיא עִקַּר הַבַּיִת, וְעָלֶיהָ הָאַחְרָיוּת לִקְדֻשַּׁת הַבַּיִת. וּבַיִת שֶׁל שֻׁתָּפִים – כֻּלָּם אַחְרָאִים, וּכְשֶׁאֶחָד קוֹבֵעַ מְזוּזָה, כֻּלָּם יוֹצְאִים בָּהּ.
💛 אַהֲבַת הַשֵּׁם: הַמְּזוּזָה הוֹפֶכֶת אֶת הַבַּיִת כֻּלּוֹ לְמָקוֹם שֶׁל אֱמוּנָה. כָּל מִי שֶׁגָּר בּוֹ – גָּדוֹל וְקָטָן, אִישׁ וְאִשָּׁה – נִזְכָּר בְּכָל כְּנִיסָה שֶׁה׳ אֶחָד.
📖 פֶּרֶק י׳, הֲלָכָה ב׳

🚪מְזוּזָה בְּכָל חֶדֶר

לֹא רַק בַּפֶּתַח הָרָאשִׁי: גַּם בְּכָל חֶדֶר בַּבַּיִת קוֹבְעִים מְזוּזָה – בַּמִּטְבָּח, בַּסָּלוֹן וּבְחֶדֶר הַשֵּׁינָה. חֶדֶר שֶׁחֲסֵרָה בּוֹ מְזוּזָה – מִשְׁתַּדְּלִים לֹא לְהִשְׁתַּמֵּשׁ בּוֹ וְלֹא לִישֹׁן בּוֹ, עַד שֶׁקּוֹבְעִים בּוֹ מְזוּזָה.
בְּבֵית הַשִּׁמּוּשׁ וּבַמִּקְלַחַת לֹא קוֹבְעִים מְזוּזָה, כִּי הֵם אֵינָם דִּירַת כָּבוֹד.
💛 אַהֲבַת הַשֵּׁם: לְכָל חֶדֶר יֵשׁ תַּפְקִיד מִשֶּׁלּוֹ – לֶאֱכֹל, לִלְמֹד, לָנוּחַ. הַמְּזוּזָה בְּכָל פֶּתַח אוֹמֶרֶת: גַּם כָּאן ה׳ אִתָּנוּ, וְכָל מָה שֶׁנַּעֲשֶׂה בַּחֶדֶר הַזֶּה יָכוֹל לִהְיוֹת קָדוֹשׁ.
📖 פֶּרֶק י׳, הֲלָכָה ג׳

🫙בֵּית מְזוּזָה מַתְאִים

בַּפֶּתַח הָרָאשִׁי, בַּסָּלוֹן וּבַמִּטְבָּח עָדִיף בֵּית מְזוּזָה שָׁקוּף, שֶׁרוֹאִים דַּרְכּוֹ אֶת הַמְּזוּזָה. בְּחֶדֶר הַשֵּׁינָה, שֶׁמִּתְלַבְּשִׁים בּוֹ, שׁוֹמְרִים עַל כְּבוֹד הַמְּזוּזָה בִּשְׁנֵי כִּסּוּיִים – וּלְפָחוֹת אֶחָד מֵהֶם אָטוּם, שֶׁלֹּא רוֹאִים דַּרְכּוֹ אֶת הַקְּלָף.
טוֹב לַעֲטֹף קֹדֶם אֶת הַקְּלָף בְּנַיְלוֹן שָׁקוּף, שֶׁיִּשְׁמֹר עָלָיו מִלַּחוּת, וְאַחַר כָּךְ לְהַכְנִיס אוֹתוֹ לְבַיִת שֶׁאֵינוֹ שָׁקוּף.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁשּׁוֹמְרִים עַל כְּבוֹד הַמְּזוּזָה, שׁוֹמְרִים עַל כְּבוֹד שֵׁם ה׳ שֶׁכָּתוּב בָּהּ. הַזְּהִירוּת הַזֹּאת מְלַמֶּדֶת אוֹתָנוּ לְהִתְנַהֵג בְּכָבוֹד גַּם בְּתוֹךְ הַבַּיִת שֶׁלָּנוּ.
📖 פֶּרֶק י׳, הֲלָכָה ג׳

🐄רֶפֶת, חֶדֶר קָטָן וּפֶתַח בַּתִּקְרָה

פַּעַם גָּרוּ הַפָּרוֹת וְהַתַּרְנְגוֹלוֹת בְּחֶדֶר שֶׁהִשְׁתַּמְּשׁוּ בּוֹ גַּם לְצָרְכֵי הַבַּיִת, וְלָכֵן הָיָה חַיָּב בִּמְזוּזָה. רֶפֶת אוֹ לוּל שֶׁל הַיּוֹם, עִם הַרְבֵּה בְּהֵמוֹת וְרֵיחַ רַע, אֵינָם דִּירַת כָּבוֹד – וְלֹא קוֹבְעִים בָּהֶם מְזוּזָה.
חֶדֶר קָטָן מֵאַרְבַּע אַמּוֹת עַל אַרְבַּע אַמּוֹת (כְּמֶטֶר וְתִשְׁעִים עַל מֶטֶר וְתִשְׁעִים), פֶּתַח נָמוּךְ מִשְּׁמוֹנִים סֶנְטִימֶטֶר, וּפֶתַח שׁוֹכֵב בַּתִּקְרָה שֶׁעוֹלִים אֵלָיו בְּסֻלָּם – פְּטוּרִים מִמְּזוּזָה.
💛 אַהֲבַת הַשֵּׁם: הַהֲלָכָה מְדַיֶּקֶת: מְזוּזָה לְמָקוֹם שֶׁיֵּשׁ בּוֹ חַיִּים שֶׁל בַּיִת, וּבְכָבוֹד. כָּךְ אֲנַחְנוּ לוֹמְדִים לְהַעֲרִיךְ אֶת הַבַּיִת שֶׁלָּנוּ כְּמָקוֹם שֶׁרָאוּי לִשְׁכִינָה.
📖 פֶּרֶק י׳, הֲלָכוֹת ד׳, י״א–י״ב

🕍בֵּית כְּנֶסֶת וַחֲנוּת

מִן הַדִּין בֵּית כְּנֶסֶת פָּטוּר מִמְּזוּזָה, כִּי "בֵּיתֶךָ" הוּא מָקוֹם שֶׁגָּרִים בּוֹ – אוֹכְלִים וִישֵׁנִים. בְּבֵית מִדְרָשׁ נוֹהֲגִים לִקְבֹּעַ מְזוּזָה בְּלִי בְּרָכָה, וְהַיּוֹם קוֹבְעִים גַּם בְּבָתֵּי כְּנֶסֶת, כִּי לוֹמְדִים בָּהֶם שִׁעוּרֵי תּוֹרָה.
בַּחֲנוּת שֶׁנִּמְצָאִים בָּהּ כָּל יוֹם שָׁעוֹת רַבּוֹת – קוֹבְעִים מְזוּזָה.
💛 אַהֲבַת הַשֵּׁם: הַמְּזוּזָה נוֹעֲדָה בִּמְיֻחָד לַבַּיִת – לַמָּקוֹם שֶׁבּוֹ אוֹכְלִים, יְשֵׁנִים וְצוֹחֲקִים. ה׳ רוֹצֶה לִהְיוֹת אִתָּנוּ דַּוְקָא בַּחַיִּים הָרְגִילִים, וְלֹא רַק בְּבֵית הַכְּנֶסֶת.
📖 פֶּרֶק י׳, הֲלָכוֹת ה׳–ו׳

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

לְפִי רֹב הַפּוֹסְקִים קוֹבְעִים אֶת הַמְּזוּזָה זְקוּפָה. כָּךְ פָּסַק הַשֻּׁלְחָן עָרוּךְ, וְכָךְ נוֹהֲגִים הַסְּפָרַדִּים. יוֹצְאֵי אַשְׁכְּנַז, וְגַם יוֹצְאֵי מָרוֹקוֹ, קוֹבְעִים אוֹתָהּ בַּאֲלַכְסוֹן – רֹאשָׁהּ לְצַד פְּנִים הַבַּיִת – כְּדֵי לָצֵאת גַּם לְפִי שִׁיטַת רַבֵּנוּ תָּם. וְעוֹד מַחֲלֹקֶת: בַּחֲנוּת שֶׁנִּמְצָאִים בָּהּ כָּל יוֹם שָׁעוֹת רַבּוֹת קוֹבְעִים מְזוּזָה. יֵשׁ אוֹמְרִים שֶׁקּוֹבְעִים אוֹתָהּ בְּלִי בְּרָכָה, כִּי יֵשׁ פּוֹסְקִים שֶׁפּוֹטְרִים חֲנֻיּוֹת, וְיֵשׁ אוֹמְרִים שֶׁמְּבָרְכִים.
גַּם הָאַשְׁכְּנַזִּים מַטִּים אוֹתָהּ רַק מְעַט, כָּךְ שֶׁהִיא קְרוֹבָה יוֹתֵר לִזְקוּפָה מֵאֲשֶׁר לִשְׁכוּבָה.
💛 אַהֲבַת הַשֵּׁם: שְׁנֵי הַמִּנְהָגִים כַּהֲלָכָה, וְכָל אֶחָד נוֹהֵג כְּמִנְהַג אֲבוֹתָיו. זְקוּפָה אוֹ נוֹטָה – בְּכָל בַּיִת הַמְּזוּזָה אוֹמֶרֶת אוֹתוֹ דָּבָר: ״שְׁמַע יִשְׂרָאֵל, ה׳ אֱלֹהֵינוּ, ה׳ אֶחָד״.
📖 פֶּרֶק י׳, הֲלָכוֹת ו׳–ז׳

↗️זְקוּפָה אוֹ בַּאֲלַכְסוֹן?

לְפִי רֹב הַפּוֹסְקִים קוֹבְעִים אֶת הַמְּזוּזָה זְקוּפָה, וְכָךְ נוֹהֲגִים הַסְּפָרַדִּים. יוֹצְאֵי אַשְׁכְּנַז וּמָרוֹקוֹ קוֹבְעִים אוֹתָהּ בַּאֲלַכְסוֹן – רֹאשָׁהּ לְצַד פְּנִים הַבַּיִת – כְּדֵי לָצֵאת גַּם לְפִי שִׁיטַת רַבֵּנוּ תָּם. גַּם הֵם מַטִּים אוֹתָהּ רַק מְעַט, כָּךְ שֶׁהִיא קְרוֹבָה יוֹתֵר לִזְקוּפָה.
אַף אֶחָד אֵינוֹ קוֹבֵעַ מְזוּזָה שׁוֹכֶבֶת לְגַמְרֵי לָרֹחַב – לְפִי רֹב הַפּוֹסְקִים הִיא פְּסוּלָה.
💛 אַהֲבַת הַשֵּׁם: כָּל מִנְהָג שׁוֹמֵר עַל הַמְּזוּזָה בְּדֶרֶךְ אַחֶרֶת, וּשְׁנֵיהֶם מְכַוְּנִים לְקַיֵּם אֶת רְצוֹן ה׳ בְּדִיּוּק. כְּשֶׁאַתָּה רוֹאֶה מְזוּזָה שׁוֹנָה מִשֶּׁלְּךָ – זֶה סִימָן לְאַהֲבָה אַחַת בִּשְׁנֵי מִנְהָגִים.
📖 פֶּרֶק י׳, הֲלָכָה ז׳

🧳אוֹרֵחַ, שׂוֹכֵר וּמָלוֹן

הַמִּצְוָה מֻטֶּלֶת עַל בַּעַל הַבַּיִת – "מְזוּזוֹת בֵּיתֶךָ" – וְלֹא עַל הָאוֹרֵחַ. לָכֵן מֻתָּר לְאוֹרֵחַ לֶאֱכֹל וְלִישֹׁן בְּחֶדֶר בְּלִי מְזוּזָה. מִי שֶׁשּׂוֹכֵר דִּירָה נֶחְשָׁב כְּבַעַל הַבַּיִת: בְּאֶרֶץ יִשְׂרָאֵל הַשּׂוֹכֵר חַיָּב לִקְבֹּעַ מְזוּזָה מִיָּד, אֲפִלּוּ לְיוֹם אֶחָד, וּבְחוּץ לָאָרֶץ – כְּשֶׁשּׂוֹכֵר לְיוֹתֵר מִשְּׁלֹשִׁים יוֹם.
אוֹרֵחַ בְּמָלוֹן פָּטוּר שְׁלֹשִׁים יוֹם, גַּם בְּאֶרֶץ יִשְׂרָאֵל. הַחִיּוּב הַמְּיֻחָד שֶׁל שׂוֹכֵר דִּירָה בָּאָרֶץ בָּא כְּדֵי שֶׁיִּתְקַשֵּׁר לְבֵיתוֹ וְיִשְׁתַּקַּע בָּאָרֶץ – וּמִמָּלוֹן אַף אֶחָד לֹא עוֹשֶׂה בַּיִת לְתָמִיד.
💛 אַהֲבַת הַשֵּׁם: מְזוּזָה בַּבַּיִת בְּאֶרֶץ יִשְׂרָאֵל מְחַבֶּרֶת אוֹתָנוּ גַּם לַבַּיִת וְגַם לָאָרֶץ שֶׁה׳ נָתַן לָנוּ. כָּל מְזוּזָה חֲדָשָׁה הִיא עוֹד שֹׁרֶשׁ קָטָן שֶׁל עַם יִשְׂרָאֵל בְּאַדְמָתוֹ.
📖 פֶּרֶק י׳, הֲלָכוֹת ח׳–י׳

🚪הַפֶּתַח – בֵּין הַבַּיִת לַחוּץ

בַּבַּיִת אָנוּ חַיִּים עִם הַמִּשְׁפָּחָה, וּבַחוּץ לוֹמְדִים, עוֹבְדִים וּפוֹגְשִׁים רַעֲיוֹנוֹת חֲדָשִׁים. הַפֶּתַח הוּא נְקֻדַּת הַמַּעֲבָר בֵּין הַשְּׁנַיִם – וְשָׁם בְּדִיּוּק צִוְּתָה הַתּוֹרָה לִקְבֹּעַ מְזוּזָה.
כְּשֶׁנִּכְנָסִים – לֹא מַכְנִיסִים הַבַּיְתָה דְּבָרִים שֶׁאֵינָם רְאוּיִים לַבַּיִת. כְּשֶׁיּוֹצְאִים – זוֹכְרִים אֶת הַמְּזוּזָה, וְלֹא נִסְחָפִים אַחֲרֵי רַעֲיוֹנוֹת זָרִים.
💛 אַהֲבַת הַשֵּׁם: הַמְּזוּזָה הִיא כְּמוֹ שׁוֹמֵר אוֹהֵב בַּפֶּתַח: הִיא מְלַוָּה אוֹתְךָ הַחוּצָה עִם "שְׁמַע יִשְׂרָאֵל", וּמְקַבֶּלֶת אוֹתְךָ חֲזָרָה הַבַּיְתָה עִם אוֹתָהּ אֱמוּנָה.
📖 פֶּרֶק י׳, הֲלָכָה י״ב

🙏הַבְּרָכָה עַל הַמְּזוּזָה

מַנִּיחִים אֶת הַמְּזוּזָה בִּמְקוֹמָהּ, מְכִינִים מַסְמְרִים אוֹ דֶּבֶק, מְבָרְכִים "לִקְבֹּעַ מְזוּזָה" – וּמִיָּד קוֹבְעִים. מִי שֶׁקּוֹבֵעַ מְזוּזָה בְּבֵית חֲבֵרוֹ מְבָרֵךְ "עַל קְבִיעַת מְזוּזָה".
מְזוּזָה שֶׁהוּרְדָה לִבְדִיקָה שֶׁל כַּמָּה שָׁעוֹת – מְבָרְכִים כְּשֶׁמַּחֲזִירִים. בִּבְדִיקָה קְצָרָה יֵשׁ סָפֵק, וְהָעֵצָה: לְהַחֲלִיף אֶת מְקוֹמוֹת הַמְּזוּזוֹת – שֶׁל הַמִּטְבָּח לְחֶדֶר הַשֵּׁינָה וְשֶׁל חֶדֶר הַשֵּׁינָה לַמִּטְבָּח – וְאָז מְבָרְכִים לְכָל הַדֵּעוֹת.
💛 אַהֲבַת הַשֵּׁם: הַבְּרָכָה עוֹזֶרֶת לָנוּ לַעֲשׂוֹת אֶת הַמִּצְוָה בְּכַוָּנָה, וְלֹא מִתּוֹךְ הֶרְגֵּל. בְּרֶגַע שֶׁל קְבִיעַת מְזוּזָה, כָּל הַמִּשְׁפָּחָה יְכוֹלָה לַעֲצֹר וְלַחְשֹׁב: הַבַּיִת הַזֶּה שַׁיָּךְ לַה׳.
📖 פֶּרֶק י׳, הֲלָכוֹת י״ג–י״ד

🔍בּוֹדְקִים פַּעֲמַיִם בְּשֶׁבַע שָׁנִים

עִם הַזְּמַן הָאוֹתִיּוֹת עֲלוּלוֹת לְהִטַּשְׁטֵשׁ מֵהַלַּחוּת, אוֹ שֶׁהַדְּיוֹ מִתְפּוֹרֵר. לָכֵן בּוֹדְקִים מְזוּזָה שֶׁל בַּיִת פַּעֲמַיִם בְּשֶׁבַע שָׁנִים – בְּעֵרֶךְ כָּל שָׁלֹשׁ שָׁנִים וָחֵצִי. אֶת הַבְּדִיקָה הַזֹּאת כָּל אָדָם יָכוֹל לַעֲשׂוֹת בְּעַצְמוֹ, וּבְסָפֵק שׁוֹאֲלִים רַב.
אֶת הַבְּדִיקָה הַיְּסוֹדִית, אוֹת אַחַר אוֹת, עוֹשֶׂה מַגִּיהַּ מֻמְחֶה עוֹד לִפְנֵי הַקְּנִיָּה. לָכֵן קוֹנִים מְזוּזוֹת רַק מֵאָדָם יָשָׁר.
💛 אַהֲבַת הַשֵּׁם: כְּמוֹ שֶׁבּוֹדְקִים מִדֵּי פַּעַם שֶׁהַכֹּל בְּסֵדֶר אֵצֶל מִי שֶׁאוֹהֲבִים, כָּךְ בּוֹדְקִים אֶת הַמְּזוּזָה. הַשְּׁמִירָה הַזֹּאת אוֹמֶרֶת: הַמִּצְוָה שֶׁלִּי חַיָּה וַחֲשׁוּבָה לִי.
📖 פֶּרֶק י׳, הֲלָכָה י״ד

🔨קְבוּעָה וַחֲזָקָה

הַמְּזוּזָה צְרִיכָה לִהְיוֹת קְבוּעָה וִיצִיבָה, שֶׁלֹּא תִּפֹּל. קוֹבְעִים אוֹתָהּ בְּמַסְמְרִים אוֹ בְּבָרְגִים, וּבְפֶתַח שֶׁל מַתֶּכֶת אֶפְשָׁר גַּם בְּדֶבֶק חָזָק – הָעִקָּר שֶׁתִּהְיֶה מְחֻבֶּרֶת הֵיטֵב.
נְיַר דֶּבֶק לֹא מַחֲזִיק שָׁנִים. לְכַתְּחִלָּה מִשְׁתַּמְּשִׁים בַּדֶּבֶק הֶחָזָק בְּיוֹתֵר, וּבְדִיעֲבַד – נְיַר דֶּבֶק שֶׁיַּחֲזִיק לְפָחוֹת שְׁלֹשִׁים יוֹם.
💛 אַהֲבַת הַשֵּׁם: מְזוּזָה קְבוּעָה הִיא כְּמוֹ אֱמוּנָה יַצִּיבָה – לֹא נוֹפֶלֶת בְּכָל רוּחַ. כְּשֶׁאַתָּה קוֹבֵעַ אוֹתָהּ חָזָק, אַתָּה אוֹמֵר לַה׳: הַקֶּשֶׁר אִתְּךָ קָבוּעַ אֶצְלִי בַּבַּיִת.
📖 פֶּרֶק י׳, הֲלָכָה ט״ו

🚚עוֹבְרִים דִּירָה

מִי שֶׁעוֹזֵב דִּירָה – לֹא מוֹרִיד אֶת הַמְּזוּזוֹת, כְּדֵי שֶׁהַדָּרִים הַבָּאִים לֹא יִנָּזְקוּ. גַּם שׂוֹכֵר שֶׁקָּנָה אֶת הַמְּזוּזוֹת בְּעַצְמוֹ מַשְׁאִיר אוֹתָן. מֻתָּר לְבַקֵּשׁ עֲלֵיהֶן תַּשְׁלוּם מֵהַדַּיָּר הֶחָדָשׁ – וְגַם אִם הוּא מְסָרֵב לְשַׁלֵּם, מַשְׁאִירִים אוֹתָן.
אִם בַּעַל הַבַּיִת אֵינוֹ יְהוּדִי, אוֹ שֶׁעוֹמְדִים לַהֲרֹס אֶת הַבַּיִת – לוֹקְחִים אֶת הַמְּזוּזוֹת. וּמִי שֶׁיֵּשׁ לוֹ מְזוּזוֹת יְקָרוֹת מְאוֹד – מַחְלִיף אוֹתָן לִפְנֵי הַמַּעֲבָר בִּמְזוּזוֹת כְּשֵׁרוֹת רְגִילוֹת.
💛 אַהֲבַת הַשֵּׁם: הַדְּאָגָה לְמִי שֶׁיָּבוֹא אַחֲרֶיךָ – גַּם אִם אֵינְךָ מַכִּיר אוֹתוֹ – הִיא אַהֲבַת יִשְׂרָאֵל אֲמִתִּית. ה׳ אוֹהֵב מִי שֶׁחוֹשֵׁב עַל הַזּוּלָת.
📖 פֶּרֶק י׳, הֲלָכָה ט״ז

➡️צַד יָמִין, שְׁלִישׁ עֶלְיוֹן

קוֹבְעִים אֶת הַמְּזוּזָה בַּצַּד הַיְמָנִי שֶׁל מִי שֶׁנִּכְנָס לַבַּיִת אוֹ לַחֶדֶר. מְזוּזָה בַּצַּד הַשְּׂמָאלִי – לֹא יָצָא. מְחַלְּקִים אֶת גֹּבַהּ הַפֶּתַח לִשְׁלֹשָׁה חֲלָקִים, וְקוֹבְעִים בִּתְחִלַּת הַשְּׁלִישׁ הָעֶלְיוֹן. בְּדֶלֶת שֶׁל שְׁנֵי מֶטֶר – מֵעַל מֶטֶר וּשְׁלֹשִׁים וְאַרְבָּעָה סֶנְטִימֶטֶר.
אִם לֹא מָדְדוּ – עָדִיף לִטְעוֹת לְמַעְלָה וְלֹא לְמַטָּה, רַק לֹא בַּטֶּפַח הָעֶלְיוֹן שֶׁל הַפֶּתַח.
💛 אַהֲבַת הַשֵּׁם: כְּבָר בַּצַּעַד הָרִאשׁוֹן לְתוֹךְ הַבַּיִת, הַמְּזוּזָה פּוֹגֶשֶׁת אוֹתְךָ מִיָּמִין – כְּמוֹ חָבֵר טוֹב שֶׁמְּחַכֶּה לְיַד הַדֶּלֶת. הִיא מְקַבֶּלֶת אוֹתְךָ בְּכָל פַּעַם בְּ"שְׁמַע יִשְׂרָאֵל".
📖 פֶּרֶק י׳, הֲלָכָה י״ז

📐לָמָּה בַּשְּׁלִישׁ הָעֶלְיוֹן?

גַּם בָּאָדָם יֵשׁ שְׁלֹשָׁה חֲלָקִים: לְמַעְלָה הַשֵּׂכֶל, אַחֲרָיו הַלֵּב וְהָרֶגֶשׁ, וּלְבַסּוֹף הַמַּעֲשֶׂה. הַמְּזוּזָה בַּשְּׁלִישׁ הָעֶלְיוֹן שֶׁל הַפֶּתַח – כִּי רַעֲיוֹנוֹתֶיהָ נִקְבָּעִים קֹדֶם כֹּל בַּמַּחֲשָׁבָה.
וְהִיא בִּתְחִלַּת הַשְּׁלִישׁ, קְרוֹבָה לְמַטָּה – כְּדֵי שֶׁתַּשְׁפִּיעַ גַּם עַל הָרֶגֶשׁ וְעַל הַמַּעֲשִׂים.
💛 אַהֲבַת הַשֵּׁם: אֱמוּנָה שֶׁנִּשְׁאֶרֶת רַק בָּרֹאשׁ אֵינָהּ שְׁלֵמָה. הַמְּזוּזָה מְלַמֶּדֶת שֶׁהַמַּחֲשָׁבָה עַל ה׳ צְרִיכָה לָרֶדֶת אֶל הַלֵּב וְאֶל הַיָּדַיִם – וּלְהָפֵךְ לְמַעֲשִׂים טוֹבִים.
📖 פֶּרֶק י׳, הֲלָכָה י״ז

🧠 בְּדֹק אֶת עַצְמְךָ

שִׁמְשׁוֹן יָשֵׁן אֵצֶל חָבֵר, וּבְחֶדֶר הָאוֹרְחִים אֵין מְזוּזָה. מָה הַדִּין?
💡 "מְזוּזוֹת בֵּיתֶךָ" – הַחִיּוּב עַל בַּעַל הַבַּיִת אוֹ עַל הַשּׂוֹכֵר, וְלֹא עַל הָאוֹרֵחַ. לָכֵן מֻתָּר לָאוֹרֵחַ לֶאֱכֹל וְלִישֹׁן בְּחֶדֶר בְּלִי מְזוּזָה.
מִשְׁפָּחָה מִירוּשָׁלַיִם שָׂכְרָה דִּירָה בִּצְפַת לִשְׁלֹשָׁה יָמִים. הַאִם עֲלֵיהֶם לִקְבֹּעַ בָּהּ מְזוּזָה?
💡 בְּאֶרֶץ יִשְׂרָאֵל מִי שֶׁשּׂוֹכֵר דִּירָה, אֲפִלּוּ לְיוֹם אֶחָד, חַיָּב בִּמְזוּזָה מִיָּד – כְּדֵי שֶׁיִּתְקַשֵּׁר לַבַּיִת וְלָאָרֶץ. שְׁלֹשִׁים יוֹם הֵם לְחוּץ לָאָרֶץ וּלְמָלוֹן.
מִשְׁפַּחַת לֵוִי עוֹבֶרֶת דִּירָה, וּמִשְׁפָּחָה יְהוּדִית אַחֶרֶת תִּכָּנֵס לַדִּירָה הַיְּשָׁנָה. מָה יַעֲשׂוּ בַּמְּזוּזוֹת?
💡 אָסוּר לְהוֹרִיד אֶת הַמְּזוּזוֹת כְּשֶׁעוֹזְבִים, כְּדֵי שֶׁהַדָּרִים הַבָּאִים לֹא יִנָּזְקוּ. מֻתָּר לְבַקֵּשׁ מֵהֶם תַּשְׁלוּם – וְגַם אִם יְסָרְבוּ, מַשְׁאִירִים.

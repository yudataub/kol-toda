מַבְעִיר וּמְכַבֶּה

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק ט״ז

🔥לֹא תְבַעֲרוּ אֵשׁ

אֶת מְלֶאכֶת הַהַבְעָרָה הִזְכִּירָה הַתּוֹרָה בִּמְפֹרָשׁ: "לֹא תְבַעֲרוּ אֵשׁ בְּכֹל מֹשְׁבֹתֵיכֶם בְּיוֹם הַשַּׁבָּת". אָסוּר לְהַדְלִיק אֵשׁ בְּכָל דֶּרֶךְ – בְּגַפְרוּר, בְּמַצִּית אוֹ בְּחַשְׁמַל. וּבְשַׁבָּת גַּם לְהַגְבִּיר אֵשׁ שֶׁכְּבָר דּוֹלֶקֶת אָסוּר מִן הַתּוֹרָה – כְּמוֹ לְסוֹבֵב אֶת כַּפְתּוֹר הַגָּז.
סְוֶדֶר צֶמֶר אוֹ בֶּגֶד סִינְתֶטִי שֶׁיּוֹצְאִים מִמֶּנּוּ נִיצוֹצוֹת כְּשֶׁפּוֹשְׁטִים אוֹתוֹ – מֻתָּר לִלְבֹּשׁ, כִּי אֵין כַּוָּנָה לְהוֹצִיא אֵשׁ.
💛 אַהֲבַת הַשֵּׁם: בָּאֵשׁ יֵשׁ כֹּחַ עָצוּם – בְּעֶזְרָתָהּ בָּנָה הָאָדָם כֵּלִים וּמְכוֹנוֹת. בְּשַׁבָּת אָנוּ מַנִּיחִים אֶת הַכֹּחַ הַזֶּה בַּצַּד, וְזוֹכְרִים אֶת הַבּוֹרֵא שֶׁנָּתַן לָנוּ אוֹתוֹ וְהוֹצִיא אוֹתָנוּ מִמִּצְרַיִם.
📖 פֶּרֶק ט״ז, הֲלָכוֹת א׳–ב׳

🔥לָמָּה דַּוְקָא הָאֵשׁ?

כָּל מְלָאכוֹת הַשַּׁבָּת נִלְמְדוּ מִן הַמִּשְׁכָּן, וְאַף עַל פִּי כֵן הַתּוֹרָה הִזְכִּירָה בִּמְפֹרָשׁ אֶת הַהַבְעָרָה. רַבִּי נָתָן לִמֵּד: כְּדֵי לְלַמֵּד שֶׁכָּל מְלָאכָה הִיא אִסּוּר בִּפְנֵי עַצְמוֹ. וּמְלֶאכֶת מְכַבֶּה הָיְתָה בַּמִּשְׁכָּן כְּדֵי לַעֲשׂוֹת מִן הָעֵצִים הַבּוֹעֲרִים פֶּחָמִים, שֶׁבְּעֶזְרָתָם הִכִּינוּ צְבָעִים לַיְּרִיעוֹת.
גַּם מִי שֶׁמְּכַבֶּה נֵר כְּדֵי לַחֲסֹךְ שֶׁמֶן, אוֹ כִּי הָאוֹר מַפְרִיעַ לוֹ – עוֹבֵר עַל אִסּוּר.
💛 אַהֲבַת הַשֵּׁם: הָאֵשׁ מְסַמֶּלֶת אֶת כָּל הַכֹּחַ שֶׁל הָאָדָם לְשַׁנּוֹת אֶת הָעוֹלָם. כְּשֶׁאָנוּ מַרְפִּים מִמֶּנָּה בְּשַׁבָּת, אָנוּ מַכְרִיזִים: הָעוֹלָם שַׁיָּךְ לַה׳, וְאָנוּ רַק שׁוֹמְרִים עָלָיו.
📖 פֶּרֶק ט״ז, הֲלָכוֹת א׳, ה׳

🕯️נֵרוֹת שַׁבָּת

חֲכָמִים גָּזְרוּ שֶׁלֹּא לִקְרֹא לְבַד לְאוֹר נֵר שֶׁמֶן, שֶׁמָּא יַטֶּה אֶת הַנֵּר כְּדֵי שֶׁיָּאִיר יוֹתֵר. אֶפְשָׁר לִלְמֹד לְאוֹרוֹ עִם חָבֵר, וְלִשְׁמֹר זֶה עַל זֶה. לְאוֹר נֵר פָּרָפִין – לְדַעַת הַרְבֵּה פּוֹסְקִים מֻתָּר, וּלְאוֹר נוּרָה חַשְׁמַלִּית – מֻתָּר לִקְרֹא גַּם לְבַד.
כְּשֶׁהַנֵּרוֹת דּוֹלְקִים – לֹא פּוֹתְחִים מוּלָם חַלּוֹן אוֹ דֶּלֶת, שֶׁמָּא הָרוּחַ תְּכַבֶּה אוֹתָם. אֲבָל אִם רוּחַ מַתְחִילָה לִנְשֹׁב מֵחַלּוֹן פָּתוּחַ – מֻתָּר לִסְגֹּר אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: נֵרוֹת שַׁבָּת מְבִיאִים שָׁלוֹם וְאוֹר לַבַּיִת. כְּשֶׁאָנוּ שׁוֹמְרִים עֲלֵיהֶם, אָנוּ שׁוֹמְרִים גַּם עַל הָאוֹר הַפְּנִימִי שֶׁל הַשַּׁבָּת – הָאוֹר שֶׁל הַמִּשְׁפָּחָה וְשֶׁל הַתּוֹרָה.
📖 פֶּרֶק ט״ז, הֲלָכוֹת ג׳–ד׳

🚒שְׂרֵפָה בְּשַׁבָּת

כְּשֶׁיֵּשׁ סַכָּנָה לְחַיֵּי אָדָם – אֲפִלּוּ סָפֵק – מִצְוָה לְכַבּוֹת אֶת הָאֵשׁ וּלְהַזְעִיק מִיָּד כַּבָּאִים, כִּי פִּקּוּחַ נֶפֶשׁ דּוֹחֶה שַׁבָּת. בְּשָׁעָה כָּזוֹ לֹא שׁוֹאֲלִים רַב – מְמַהֲרִים לְהַצִּיל. שְׂרֵפָה גְּדוֹלָה בְּבֵית מְגוּרִים נֶחְשֶׁבֶת סָפֵק סַכָּנָה. אֲבָל כְּשֶׁבָּרוּר שֶׁאֵין סַכָּנָה לְאַף אָדָם, וְרַק רְכוּשׁ נִשְׂרָף – אָסוּר לְכַבּוֹת, כִּי הֶפְסֵד מָמוֹן אֵינוֹ דּוֹחֶה שַׁבָּת.
בְּמָקוֹם הֶפְסֵד מֻתָּר לִגְרֹם לְכִבּוּי בַּעֲקִיפִין – לְמָשָׁל לְהַנִּיחַ שַׂקִּיּוֹת מַיִם בְּמָקוֹם שֶׁהָאֵשׁ עוֹד לֹא הִגִּיעָה אֵלָיו. וְגַחֶלֶת בָּרְחוֹב שֶׁעֲלוּלָה לִפְצֹעַ אֲנָשִׁים וְאִי אֶפְשָׁר לְפַנּוֹת – מֻתָּר לְכַבּוֹת.
💛 אַהֲבַת הַשֵּׁם: חַיֵּי אָדָם יְקָרִים לַה׳ יוֹתֵר מִכָּל דָּבָר. הַשַּׁבָּת נִתְּנָה כְּדֵי שֶׁנִּחְיֶה בָּהּ – וְלָכֵן כְּשֶׁיֵּשׁ סַכָּנָה, דַּוְקָא הַהַצָּלָה הִיא רְצוֹן ה׳.
📖 פֶּרֶק ט״ז, הֲלָכוֹת ו׳–ח׳

🧠 בְּדֹק אֶת עַצְמְךָ

בְּשַׁבָּת פָּרְצָה שְׂרֵפָה גְּדוֹלָה בְּבִנְיַן מְגוּרִים. נִרְאֶה שֶׁכָּל הַדַּיָּרִים כְּבָר יָצְאוּ. מָה עוֹשִׂים?
💡 גַּם סָפֵק סַכָּנָה דּוֹחֶה שַׁבָּת, וּשְׂרֵפָה גְּדוֹלָה בְּבֵית מְגוּרִים נֶחְשֶׁבֶת סָפֵק סַכָּנָה. בְּשָׁעָה כָּזוֹ לֹא שׁוֹאֲלִים רַב – מְמַהֲרִים לְהַצִּיל וּלְהַזְעִיק כַּבָּאִים.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

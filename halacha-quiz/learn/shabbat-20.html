בַּעֲלֵי חַיִּים

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק כ׳

🐕גַּם הַבְּהֵמוֹת נָחוֹת

הַתּוֹרָה צִוְּתָה שֶׁגַּם בַּעֲלֵי הַחַיִּים יָנוּחוּ בְּשַׁבָּת: "לְמַעַן יָנוּחַ שׁוֹרְךָ וַחֲמֹרֶךָ". לָכֵן לֹא מַעֲבִידִים שׁוּם חַיָּה – לֹא חֲמוֹר שֶׁסּוֹחֵב מַשָּׂא, וְלֹא יוֹנֵי דֹּאַר. וְאֶת הַכֶּלֶב מֻתָּר לְהוֹצִיא לָרְחוֹב בִּרְצוּעָה שֶׁעַל צַוָּארוֹ.
נוֹתְנִים אֹכֶל וּמַיִם לְחַיּוֹת שֶׁתְּלוּיוֹת בָּנוּ, כְּמוֹ חָתוּל אוֹ תַּרְנְגוֹלוֹת, וְגַם לְכֶלֶב אוֹ לְחָתוּל רָעֵב שֶׁאֵין לוֹ בְּעָלִים. אֲבָל לִדְבוֹרִים וּלְיוֹנֵי שׁוֹבָךְ, שֶׁמִּסְתַּדְּרוֹת לְבַד, לֹא נוֹתְנִים.
💛 אַהֲבַת הַשֵּׁם: "וְרַחֲמָיו עַל כָּל מַעֲשָׂיו" – ה׳ מְרַחֵם עַל כָּל בְּרִיָּה. כְּשֶׁאַתָּה דּוֹאֵג לְחָתוּל רָעֵב, אַתָּה הוֹלֵךְ בִּדְרָכָיו שֶׁל ה׳.
📖 פֶּרֶק כ׳, הֲלָכוֹת א׳–ג׳

🐴רַחֲמִים עַל כָּל הַבְּרוּאִים

מִצְוַת הַשַּׁבָּת אֵינָהּ רַק שֶׁלָּנוּ: "וּבַיּוֹם הַשְּׁבִיעִי תִּשְׁבֹּת, לְמַעַן יָנוּחַ שׁוֹרְךָ וַחֲמֹרֶךָ". הָאִסּוּר הוּא בְּכָל בַּעֲלֵי הַחַיִּים, גַּם בְּעוֹפוֹת וּבְדָגִים. וְכָל מִי שֶׁגּוֹרֵם לַחֲמוֹר לָלֶכֶת עִם מַשָּׂא – אֲפִלּוּ בְּקוֹלוֹ – עוֹבֵר עַל אִסּוּר תּוֹרָה.
פָּרָה שֶׁלֹּא חוֹלְבִים אוֹתָהּ סוֹבֶלֶת מְאוֹד. לָכֵן, מִשּׁוּם צַעַר בַּעֲלֵי חַיִּים, הִתִּירוּ חֲכָמִים לְבַקֵּשׁ מִגּוֹי לַחְלֹב אוֹתָהּ בְּשַׁבָּת.
💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁגַּם הַשּׁוֹר וְהַחֲמוֹר יַרְגִּישׁוּ אֶת הַמְּנוּחָה. מִי שֶׁלּוֹמֵד לְרַחֵם עַל חַיָּה, לוֹמֵד לְרַחֵם עַל כָּל אָדָם – וְדוֹמֶה לְבוֹרְאוֹ.
📖 פֶּרֶק כ׳, הֲלָכוֹת א׳, ד׳

🐈חַיּוֹת מַחְמָד וְדָגֵי אַקְוַרְיוּם

בַּעֲלֵי חַיִּים הֵם מֻקְצֶה. אֲבָל חַיּוֹת מַחְמָד שֶׁבַּעֲלֵיהֶן רְגִילִים לְהָרִים אוֹתָן וּלְשַׂחֵק אִתָּן כָּל הַשָּׁבוּעַ – לְדַעַת הָרַב פַיינְשְׁטֵיין וְהָרַב אוֹיֶערְבַּאךְ אֵינָן מֻקְצֶה, וּמֻתָּר לְבַעֲלֵיהֶן לְהָרִים אוֹתָן. כָּךְ פּוֹסֵק הָרַב מֶלָמֵד. וּלְפִי יַלְקוּט יוֹסֵף וְאָרְחוֹת שַׁבָּת גַּם הֵן מֻקְצֶה: מֻתָּר רַק לִדְחֹף אוֹ לִגְרֹר אוֹתָן כְּדֵי לִמְנֹעַ מֵהֶן צַעַר, וְלֹא לְהָרִים.
דָּג שֶׁקָּפַץ מִן הָאַקְוַרְיוּם – מֻתָּר לְהַחֲזִיר אוֹתוֹ לַמַּיִם, מִשּׁוּם צַעַר בַּעֲלֵי חַיִּים. וְדָג שֶׁמֵּת בָּאַקְוַרְיוּם – מֻתָּר לְהוֹצִיא, כְּדֵי שֶׁלֹּא יַדְבִּיק אֶת הַדָּגִים הָאֲחֵרִים.
💛 אַהֲבַת הַשֵּׁם: ״וְרַחֲמָיו עַל כָּל מַעֲשָׂיו״ – גַּם בַּשַּׁבָּת, יוֹם הַמְּנוּחָה, אָנוּ דּוֹאֲגִים לַבְּרוּאִים הַקְּטַנִּים שֶׁה׳ הִפְקִיד בְּיָדֵינוּ.
📖 פֶּרֶק כ׳, הֲלָכָה ה׳

🔀חֲתוּל הַבַּיִת – שְׁתֵּי שִׁיטוֹת, וּשְׁתֵּיהֶן כַּהֲלָכָה

חַיַּת מַחְמָד שֶׁבַּעֲלֶיהָ מְשַׂחֲקִים אִתָּהּ כָּל הַשָּׁבוּעַ – הַאִם הִיא מֻקְצֶה בְּשַׁבָּת?
לְפִי הָרַב פַיינְשְׁטֵיין וְהָרַב אוֹיֶערְבַּאךְ – אֵינָהּ מֻקְצֶה, וּמֻתָּר לְהָרִים אוֹתָהּ. וּלְפִי יַלְקוּט יוֹסֵף וְאָרְחוֹת שַׁבָּת – הִיא מֻקְצֶה, וּמֻתָּר רַק לִדְחֹף אוֹ לִגְרֹר אוֹתָהּ כְּדֵי לִמְנֹעַ מִמֶּנָּה צַעַר.
💛 אַהֲבַת הַשֵּׁם: שְׁתֵּי הַשִּׁיטוֹת מְלַמְּדוֹת אוֹתָנוּ לְרַחֵם עַל בַּעֲלֵי הַחַיִּים, וְכָל אֶחָד שׁוֹאֵל אֶת רַבּוֹ וְהוֹלֵךְ בְּדֶרֶךְ אֲבוֹתָיו.
📖 פֶּרֶק כ׳, הֲלָכָה ה׳

🐦לֹא צָדִים

צֵידָה הִיא אַחַת מִן הַמְּלָאכוֹת: לִתְפֹּס חַיָּה אוֹ צִפּוֹר וְלִסְגֹּר אוֹתָהּ. צִפּוֹר שֶׁנִּכְנְסָה הַבַּיְתָה דֶּרֶךְ הַחַלּוֹן – לֹא סוֹגְרִים אַחֲרֶיהָ אֶת הַחַלּוֹן. רוֹצִים לִסְגֹּר? קֹדֶם מְגָרְשִׁים אוֹתָהּ הַחוּצָה. וּמִי שֶׁמַּאֲכִיל תֻּכִּי בַּכְּלוּב – נִזְהָר לֹא לִפְתֹּחַ אֶת הַכְּלוּב.
לְשַׁחְרֵר חַיָּה מִמַּלְכֹּדֶת – מֻתָּר. הָאִסּוּר הוּא לָצוּד, לֹא לְשַׁחְרֵר.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת אֲנַחְנוּ לֹא שׁוֹלְטִים עַל הַבְּרִיאָה, אֶלָּא נוֹתְנִים לָהּ לִהְיוֹת חָפְשִׁית. כָּךְ אֲנַחְנוּ זוֹכְרִים שֶׁהָעוֹלָם כֻּלּוֹ שַׁיָּךְ לַה׳.
📖 פֶּרֶק כ׳, הֲלָכוֹת ו׳–ז׳

🐜נְמָלִים, יַתּוּשִׁים וְעַקְרַבִּים

אָסוּר לַהֲרֹג בְּשַׁבָּת שׁוּם בַּעַל חַיִּים, אֲפִלּוּ נְמָלָה קְטַנָּה. נְמָלִים עַל הַשְּׁבִיל – מְדַלְּגִים מֵעֲלֵיהֶן. יַתּוּשׁ עוֹמֵד עַל הַיָּד – אֶפְשָׁר לְהַרְחִיק אוֹתוֹ, אֲבָל לֹא לִמְעֹךְ וְלֹא לַהֲרֹג. יַתּוּשִׁים בַּחֶדֶר – מֻתָּר לְרַסֵּס כְּדֵי לְגָרֵשׁ, לֹא עֲלֵיהֶם, וְחַלּוֹן פָּתוּחַ כְּדֵי שֶׁיִּבְרְחוּ.
נָחָשׁ אוֹ עַקְרָב שֶׁיָּכוֹל לְסַכֵּן חַיֵּי אָדָם – הוֹרְגִים גַּם בְּשַׁבָּת, כִּי פִּקּוּחַ נֶפֶשׁ דּוֹחֶה שַׁבָּת.
💛 אַהֲבַת הַשֵּׁם: הַזְּהִירוּת אֲפִלּוּ מִנְּמָלָה קְטַנָּה מְלַמֶּדֶת כַּמָּה הַחַיִּים יְקָרִים. וְהַחַיִּים שֶׁל אָדָם יְקָרִים מִכֻּלָּם – בִּשְׁבִילָם ה׳ צִוָּה לְהַצִּיל גַּם בְּשַׁבָּת.
📖 פֶּרֶק כ׳, הֲלָכוֹת ח׳, י׳

🩸לֹא מוֹצִיאִים דָּם בְּשַׁבָּת

אָסוּר מִן הַתּוֹרָה לְהוֹצִיא דָּם בְּשַׁבָּת – זוֹ מְלֶאכֶת חוֹבֵל. לָכֵן לֹא מְגָרְדִים פֶּצַע בְּאֹפֶן שֶׁיּוֹצִיא דָּם, וְלֹא מְצַחְצְחִים שִׁנַּיִם בְּאֹפֶן שֶׁיּוֹרִיד דָּם.
גַּם מַכָּה שֶׁעוֹשָׂה ״כְּחֹל״ מִתַּחַת לָעוֹר נֶחְשֶׁבֶת חוֹבֵל. וְכַמּוּבָן, לְהַכּוֹת אָסוּר גַּם בְּחֹל!
💛 אַהֲבַת הַשֵּׁם: ״הַדָּם הוּא הַנֶּפֶשׁ״. בְּשַׁבָּת, יוֹם שֶׁל חַיִּים וּבְרָכָה, שׁוֹמְרִים עַל כָּל טִפָּה שֶׁל חַיִּים.
📖 פֶּרֶק כ׳, הֲלָכָה ט׳

🧠 בְּדֹק אֶת עַצְמְךָ

יַתּוּשׁ נָחַת עַל זְרוֹעוֹ שֶׁל יוֹנָתָן בְּשַׁבָּת. מָה מֻתָּר לוֹ לַעֲשׂוֹת?
💡 עֲקִיצַת יַתּוּשׁ אֵינָהּ מְסֻכֶּנֶת, וְלָכֵן אָסוּר לְהָרְגוֹ. כְּדֵי לְהָסִיר צַעַר הִתִּירוּ לְהַרְחִיק אוֹתוֹ מִן הָעוֹר, וּבִלְבַד שֶׁלֹּא יִמְעַךְ אוֹתוֹ.
בְּשַׁבָּת יֵשׁ לְאוּרִי פֶּצַע קָטָן בַּבֶּרֶךְ, וְהוּא מְגָרֵד לוֹ מְאוֹד. מָה נָכוֹן?
💡 אָסוּר מִן הַתּוֹרָה לְהוֹצִיא דָּם בְּשַׁבָּת, כִּי ״הַדָּם הוּא הַנֶּפֶשׁ״. לָכֵן לֹא מְגָרְדִים פֶּצַע בְּאֹפֶן שֶׁיּוֹצִיא דָּם.
דַּג זָהָב קָפַץ בְּשַׁבָּת מִן הָאַקְוַרְיוּם אֶל הַשֻּׁלְחָן, וְהוּא מְפַרְפֵּר. מָה עוֹשִׂים?
💡 אַף שֶׁהַדָּג מֻקְצֶה, בִּשְׁעַת הַדְּחָק – כְּשֶׁאֵין פִּתְרוֹן אַחֵר – מֻתָּר לְטַלְטֵל מֻקְצֶה מִשּׁוּם צַעַר בַּעֲלֵי חַיִּים. לָכֵן מַחֲזִירִים אֶת הַדָּג לָאַקְוַרְיוּם, אִם יֵשׁ לְהַנִּיחַ שֶׁיִּחְיֶה.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

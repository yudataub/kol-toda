צִבְיוֹן הַשַּׁבָּת

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק כ״ב

👑חֲפָצֶיךָ וְחֶפְצֵי שָׁמַיִם

הַנָּבִיא יְשַׁעְיָהוּ אָמַר: "וְקָרָאתָ לַשַּׁבָּת עֹנֶג... וְכִבַּדְתּוֹ מֵעֲשׂוֹת דְּרָכֶיךָ מִמְּצוֹא חֶפְצְךָ". חֲכָמִים לָמְדוּ: "חֲפָצֶיךָ אֲסוּרִים, חֶפְצֵי שָׁמַיִם מֻתָּרִים". לָכֵן לְצֹרֶךְ מִצְוָה מֻתָּר לְדַבֵּר עַל דְּבָרִים שֶׁל חֹל – לְמָשָׁל לְהִתְחַיֵּב לָתֵת צְדָקָה, אוֹ לְהַכְרִיז עַל אֲבֵדָה.
מִי שֶׁשּׁוֹמֵר עַל צִבְיוֹן הַשַּׁבָּת זוֹכֶה לְשָׂכָר גָּדוֹל: "אָז תִּתְעַנַּג עַל ה׳".
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנַחְנוּ עוֹזְבִים בְּשַׁבָּת אֶת הָעִנְיָנִים שֶׁלָּנוּ, נִשְׁאָר מָקוֹם לָעִנְיָנִים שֶׁל ה׳. וּמִי שֶׁעוֹסֵק בְּחֶפְצֵי שָׁמַיִם – מִתְעַנֵּג עַל ה׳ בֶּאֱמֶת.
📖 פֶּרֶק כ״ב, הֲלָכוֹת א׳, י״א

🛒בְּלִי קְנִיּוֹת וּבְלִי מְחִירִים

בְּשַׁבָּת לֹא קוֹנִים וְלֹא מוֹכְרִים, וְגַם לֹא מַלְוִים. מִי שֶׁצָּרִיךְ מֵהַשָּׁכֵן מַשֶּׁהוּ לְשַׁבָּת – יְבַקֵּשׁ בִּלְשׁוֹן "הַשְׁאָלָה": "תַּשְׁאִיל לִי". וְלֹא קוֹרְאִים בְּשַׁבָּת מְחִירִים, פִּרְסֹמוֹת לִקְנִיּוֹת אוֹ חֶשְׁבּוֹנוֹת.
מֻתָּר לִקְרֹא בְּשַׁבָּת סִפְרֵי קֹדֶשׁ וְגַם סְפָרִים שֶׁל חָכְמָה וּמַדָּע. אֲבָל יְדִיעוֹת שֶׁגּוֹרְמוֹת לְעֶצֶב וּדְאָגָה – לֹא קוֹרְאִים.
💛 אַהֲבַת הַשֵּׁם: כָּל הַשָּׁבוּעַ הָרֹאשׁ מָלֵא בְּ"כַּמָּה זֶה עוֹלֶה". בְּשַׁבָּת אֲנַחְנוּ מְשַׁחְרְרִים אֶת הַלֵּב מִן הַדְּאָגָה, וּבוֹטְחִים בַּה׳ שֶׁזָּן אֶת כֻּלָּם.
📖 פֶּרֶק כ״ב, הֲלָכוֹת ב׳, ד׳, י״ב

🚲עֻבְדִּין דְּחֹל

מַעֲשִׂים שֶׁהֵם מַעֲשִׂים מֻבְהָקִים שֶׁל יוֹם חֹל אֲסוּרִים בְּשַׁבָּת, וְנִקְרָאִים "עֻבְדִּין דְּחֹל". לָכֵן לֹא רוֹכְבִים עַל אוֹפַנַּיִם, לֹא רָצִים לְשֵׁם הִתְעַמְּלוּת, וְלֹא מְשַׂחֲקִים כַּדּוּרֶגֶל, כַּדּוּרְסַל אוֹ טֶנִיס – גַּם לֹא יְלָדִים.
גַּם לִמְדֹּד וְלִשְׁקֹל בְּשַׁבָּת אָסוּר – לֹא אֶת הַגֹּבַהּ וְלֹא אֶת הַמִּשְׁקָל. לְצֹרֶךְ מִצְוָה אוֹ לְצֹרֶךְ חוֹלֶה, כְּמוֹ מְדִידַת חֹם – מֻתָּר.
💛 אַהֲבַת הַשֵּׁם: הַשַּׁבָּת אֵינָהּ סְתָם יוֹם חֹפֶשׁ. הִיא יוֹם קָדוֹשׁ, וְכָל מַה שֶּׁאֲנַחְנוּ עוֹשִׂים בּוֹ מַזְכִּיר לָנוּ שֶׁאֲנַחְנוּ עִם הַמֶּלֶךְ.
📖 פֶּרֶק כ״ב, הֲלָכוֹת ו׳, ח׳, י״ג

🚶הוֹלְכִים בְּנַחַת

הַנָּבִיא אָמַר: "וְכִבַּדְתּוֹ מֵעֲשׂוֹת דְּרָכֶיךָ" – שֶׁלֹּא תִּהְיֶה הַהֲלִיכָה שֶׁל שַׁבָּת כְּמוֹ שֶׁל חֹל. לָכֵן בְּשַׁבָּת לֹא רָצִים וְלֹא פּוֹסְעִים פְּסִיעוֹת גַּסּוֹת לְצָרְכֵי עַצְמֵנוּ. אֲבָל לִתְפִלָּה אוֹ לְשִׁעוּר תּוֹרָה – מִצְוָה לָרוּץ!
מֻתָּר לָרוּץ כְּדֵי לְהִנָּצֵל מִגֶּשֶׁם, לִקְפֹּץ מֵעַל שְׁלוּלִית, וּלְיְלָדִים מֻתָּר לְשַׂחֵק בְּמִשְׂחֲקֵי רִיצָה לַהֲנָאָתָם.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת הָעוֹלָם כְּאִלּוּ כְּבָר מְתֻקָּן, וְאֵין לְאָן לְמַהֵר. וּכְשֶׁאַתָּה רָץ לְבֵית הַכְּנֶסֶת, אַתָּה מַרְאֶה לַה׳ שֶׁאַתָּה מִשְׁתּוֹקֵק לִפְגֹּשׁ אוֹתוֹ.
📖 פֶּרֶק כ״ב, הֲלָכָה ז׳

🗣️דִּבּוּר שֶׁל שַׁבָּת

"וְדַבֵּר דָּבָר" – שֶׁלֹּא יִהְיֶה הַדִּבּוּר שֶׁל שַׁבָּת כְּמוֹ שֶׁל חֹל. לֹא אוֹמְרִים בְּשַׁבָּת "מָחָר אֶסַּע", "מָחָר אֶכְתֹּב" אוֹ "מָחָר אֶקְנֶה", כִּי אֵלֶּה דְּבָרִים שֶׁאָסוּר לַעֲשׂוֹת בְּשַׁבָּת. אֲבָל מֻתָּר לוֹמַר "מָחָר אֵלֵךְ לִירוּשָׁלַיִם", כִּי הֲלִיכָה מֻתֶּרֶת.
לַחְשֹׁב עַל כָּךְ – מֻתָּר: "דִּבּוּר אָסוּר, הִרְהוּר מֻתָּר". וּמֻתָּר לְסַפֵּר עַל מָה שֶׁכְּבָר עָשִׂינוּ.
💛 אַהֲבַת הַשֵּׁם: הַפֶּה שֶׁלָּנוּ יָכוֹל לִבְנוֹת אֲוִירָה. כְּשֶׁאָנוּ מְדַבְּרִים בְּשַׁבָּת עַל תּוֹרָה, עַל הַמִּשְׁפָּחָה וְעַל הַטּוֹב שֶׁה׳ נָתַן – הַשַּׁבָּת מִתְמַלֵּאת קְדֻשָּׁה.
📖 פֶּרֶק כ״ב, הֲלָכָה י׳

🗣️🔑 מוּשָּׂג: דִּבּוּר שֶׁל שַׁבָּת

״וְדַבֵּר דָּבָר״ – שֶׁלֹּא יִהְיֶה הַדִּבּוּר שֶׁל שַׁבָּת כְּמוֹ שֶׁל חֹל. לֹא אוֹמְרִים בְּשַׁבָּת ״מָחָר אֶסַּע״, ״מָחָר אֶכְתֹּב״ אוֹ ״מָחָר אֶקְנֶה״, כִּי אֵלֶּה דְּבָרִים שֶׁאָסוּר לַעֲשׂוֹת בְּשַׁבָּת.
📖 אֲבָל מֻתָּר לוֹמַר ״מָחָר אֵלֵךְ לִירוּשָׁלַיִם״, כִּי הֲלִיכָה מֻתֶּרֶת, וּמֻתָּר לְסַפֵּר עַל מָה שֶׁכְּבָר עָשִׂינוּ.
💛 אַהֲבַת הַשֵּׁם: הַפֶּה שֶׁלָּנוּ יָכוֹל לִבְנוֹת אֲוִירָה. כְּשֶׁאָנוּ מְדַבְּרִים בְּשַׁבָּת עַל תּוֹרָה וְעַל הַטּוֹב שֶׁה׳ נָתַן – הַשַּׁבָּת מִתְמַלֵּאת קְדֻשָּׁה. 🧒 וְאַתָּה? נַסֵּה לֹא לְדַבֵּר בְּשַׁבָּת עַל קְנִיּוֹת וְעַל תִּכְנוּנֵי הַשָּׁבוּעַ הַבָּא.
📖 פֶּרֶק כ״ב, הֲלָכָה י׳

🎲מִשְׂחָקִים וּמְחִיאוֹת כַּף

יֵשׁ פּוֹסְקִים שֶׁאוֹסְרִים עַל גְּדוֹלִים לְשַׂחֵק בְּשַׁבָּת, כְּמוֹ שַׁחְמָט, כִּי הַשַּׁבָּת נוֹעֲדָה לַתּוֹרָה, וְיֵשׁ מַתִּירִים. טוֹב לִגְדוֹלִים לְהַחְמִיר, וִילָדִים מֻתָּר לָהֶם לְשַׂחֵק. רִקּוּד וּמְחִיאוֹת כַּף בִּזְמַן שִׁירָה – לְדַעַת הַרְבֵּה פּוֹסְקִים אֲסוּרִים, וְיֵשׁ מְקִלִּים. וּלְכָל הַדֵּעוֹת מֻתָּר לִמְחֹא כַּף בְּשִׁנּוּי – בְּגַב הַיָּד עַל כַּף הַיָּד.
כַּדּוּרֶגֶל, כַּדּוּרְסַל וְטֶנִיס – אֲסוּרִים לְכֻלָּם, גַּם לִילָדִים, כִּי הֵם עֻבְדִּין דְּחֹל. וְאָסוּר לְשַׂחֵק עַל כֶּסֶף.
💛 אַהֲבַת הַשֵּׁם: הַזְּמַן שֶׁל שַׁבָּת יָקָר מְאוֹד. גַּם כְּשֶׁמֻּתָּר לְשַׂחֵק, כְּדַאי לִשְׁמֹר חֵלֶק גָּדוֹל מִן הַיּוֹם לְלִמּוּד תּוֹרָה – כִּי שָׁם מַרְגִּישִׁים אֶת ה׳ הֲכִי קָרוֹב.
📖 פֶּרֶק כ״ב, הֲלָכוֹת י״ג, י״ח

🔀דֵּעוֹת וּמִנְהָגִים בְּשִׁירָה וּבְקוֹלוֹת

רִקּוּד וּמְחִיאוֹת כַּף בִּזְמַן שִׁירָה: רַבִּים נוֹהֲגִים לִרְקֹד וְלִמְחֹא כַּף בְּשִׁירֵי שַׁבָּת. לְדַעַת הַרְבֵּה פּוֹסְקִים זֶה אָסוּר, כְּמוֹ שֶׁתִּקְּנוּ חֲכָמִים. וְיֵשׁ מְקִלִּים, כִּי לְדַעְתָּם הַגְּזֵרָה הָיְתָה בִּזְמַנָּם, כְּשֶׁהָיוּ רְגִילִים לְהָבִיא כְּלֵי נְגִינָה תּוֹךְ כְּדֵי רִקּוּד. בְּשִׂמְחַת תּוֹרָה – גַּם הַמַּחְמִירִים רוֹקְדִים.
פַּעֲמוֹן יָדָנִי אוֹ טַבַּעַת מַתֶּכֶת עַל הַדֶּלֶת, כְּדֵי שֶׁיָּבוֹאוּ לִפְתֹּחַ: לְפִי הַשֻּׁלְחָן עָרוּךְ מֻתָּר, כִּי אֵין כַּוָּנָה לְזֶמֶר, וְהָרַמָּ״א אוֹסֵר, כִּי זֶה דּוֹמֶה לִכְלִי נְגִינָה. אֲבָל לִדְפֹּק בַּיָּד עַל הַדֶּלֶת – מֻתָּר לְכֻלָּם.
💛 אַהֲבַת הַשֵּׁם: הַשִּׁירָה בְּשַׁבָּת מְעוֹרֶרֶת אֶת הַלֵּב לְאַהֲבַת ה׳. גַּם מִי שֶׁמַּחְמִיר וְגַם מִי שֶׁמֵּקֵל – כֻּלָּם שָׁרִים לִכְבוֹד הַשַּׁבָּת מִתּוֹךְ שִׂמְחָה.
📖 פֶּרֶק כ״ב, הֲלָכוֹת י״ג, י״ז–י״ח

💰שְׂכַר שַׁבָּת וַהֲכָנָה לְחֹל

אָסוּר לְקַבֵּל שָׂכָר עַל עֲבוֹדָה שֶׁנַּעֲשְׂתָה בְּשַׁבָּת – גַּם עַל עֲבוֹדָה מֻתֶּרֶת כְּמוֹ מֶלְצָרוּת, וְגַם אַחֲרֵי הַשַּׁבָּת. אֲבָל מִי שֶׁהִתְנַדֵּב, מֻתָּר לָתֵת לוֹ מַתָּנָה בְּמוֹצָאֵי שַׁבָּת. וְעוֹד: לֹא טוֹרְחִים בְּשַׁבָּת כְּדֵי לְהָכִין לְיוֹם חֹל – לֹא מְסַדְּרִים מִטּוֹת לַלַּיְלָה, וְלֹא שׁוֹטְפִים כֵּלִים שֶׁלֹּא יִצְטָרְכוּ הַיּוֹם.
דְּבָרִים קַלִּים שֶׁרְגִילִים לַעֲשׂוֹת תָּמִיד – מֻתָּר: לָשִׂים סִימָנִיָּה בַּסֵּפֶר, וּלְהַחֲזִיר לַמְּקָרֵר אֶת הָאֹכֶל שֶׁנִּשְׁאַר.
💛 אַהֲבַת הַשֵּׁם: הַשַּׁבָּת אֵינָהּ "הֲכָנָה" לְיוֹם רִאשׁוֹן – הִיא הַמַּטָּרָה עַצְמָהּ. כְּשֶׁאֵינְךָ מְנַצֵּל אוֹתָהּ לְחֹל, אַתָּה אוֹמֵר לַה׳: הַיּוֹם הַזֶּה כֻּלּוֹ שֶׁלְּךָ.
📖 פֶּרֶק כ״ב, הֲלָכוֹת י״ד–ט״ז

🎵בְּלִי כְּלֵי נְגִינָה

חֲכָמִים אָסְרוּ לְנַגֵּן בְּשַׁבָּת בִּכְלֵי נְגִינָה, שֶׁמָּא יְתַקְּנוּ אֶת הַכְּלִי. וְאָסוּר לִשְׁמֹעַ רַדְיוֹ אוֹ לִצְפּוֹת בְּטֶלֶוִיזְיָה, גַּם אִם הִדְלִיקוּ אוֹתָם לִפְנֵי שַׁבָּת – זֶה פּוֹגֵעַ בִּכְבוֹד הַשַּׁבָּת.
קוֹל שֶׁאֵינוֹ מוּזִיקָה – מֻתָּר: לִדְפֹּק בַּיָּד עַל הַדֶּלֶת כְּדֵי שֶׁיִּפְתְּחוּ, לִמְחֹא כַּף כְּדֵי לְהָעִיר חָבֵר, וְגַם לִשְׁרֹק בַּפֶּה.
💛 אַהֲבַת הַשֵּׁם: אֶת הַשִּׁירִים הֲכִי יָפִים שֶׁל שַׁבָּת שָׁרִים בַּפֶּה, סְבִיב הַשֻּׁלְחָן. זְמִירוֹת שֶׁל כָּל הַמִּשְׁפָּחָה הֵן הַמּוּזִיקָה שֶׁל אַהֲבַת ה׳.
📖 פֶּרֶק כ״ב, הֲלָכוֹת י״ז, י״ט

🧠 בְּדֹק אֶת עַצְמְךָ

מִיכָאֵל מְדַבֵּר עִם חָבֵר בְּשַׁבָּת. אֵיזֶה מִשְׁפָּט מֻתָּר לוֹ לְהַגִּיד?
💡 דִּבּוּר עַל מְלָאכָה אֲסוּרָה – טֶלֶפוֹן, קְנִיָּה, כְּתִיבָה – אָסוּר בְּשַׁבָּת. "אֲדַבֵּר" הוּא רַק רֶמֶז, וְהוּא מֻתָּר אַף שֶׁהֶחָבֵר מֵבִין שֶׁהַכַּוָּנָה לַטֶּלֶפוֹן.
אֲרִיאֵל עָבַד בְּשַׁבָּת כְּמֶלְצַר בִּסְעוּדָה, וְסִכְּמוּ מֵרֹאשׁ שֶׁיְּשַׁלְּמוּ לוֹ רַק עַל הַשַּׁבָּת. מָה הַדִּין?
💡 שְׂכַר שַׁבָּת אָסוּר גַּם עַל עֲבוֹדָה מֻתֶּרֶת, וְגַם בְּדִיעֲבַד. אִלּוּ סִכְּמוּ שֶׁיַּעֲבֹד גַּם בְּמוֹצָאֵי שַׁבָּת, הַשָּׂכָר הָיָה "מֻבְלָע" בִּשְׂכַר חֹל.
אַחֲרֵי סְעוּדָה שְׁלִישִׁית, דַּקּוֹת סְפוּרוֹת לִפְנֵי צֵאת הַשַּׁבָּת, כֻּלָּם יוֹצְאִים מִן הַחֶדֶר. מֻתָּר לְסַדֵּר עַכְשָׁו אֶת הַשֻּׁלְחָן?
💡 כָּל עוֹד הַסִּדּוּר מוֹעִיל לַשַּׁבָּת – מֻתָּר. אֲבָל כְּשֶׁהוּא מוֹעִיל רַק לְמוֹצָאֵי שַׁבָּת, זוֹ הֲכָנָה מִשַּׁבָּת לְחֹל, וַאֲסוּרָה.

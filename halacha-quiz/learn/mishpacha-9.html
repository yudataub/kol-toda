👶פִּדְיוֹן הַבֵּן

״קַדֶּשׁ לִי כָל בְּכוֹר״ – מִי שֶׁנּוֹלַד לוֹ בֵּן בְּכוֹר, פּוֹדֶה אוֹתוֹ אַחֲרֵי שְׁלֹשִׁים יוֹם: הוּא נוֹתֵן לַכֹּהֵן חֲמִשָּׁה שְׁקָלִים שֶׁל כֶּסֶף. נוֹתְנִים כֶּסֶף טָהוֹר, אוֹ חֵפֶץ שֶׁשָּׁוֶה כְּמוֹתוֹ – אֲבָל לֹא צֶ׳ק וְלֹא שְׁטָרוֹת.
רַק בֵּן בְּכוֹר צָרִיךְ פִּדְיוֹן, לֹא בַּת. לָכֵן מִשְׁפָּחָה שֶׁהַיַּלְדָּה הָרִאשׁוֹנָה שֶׁלָּהּ הִיא בַּת – פְּטוּרָה. זוֹ מִצְוָה נְדִירָה, וּמְקַיְּמִים אוֹתָהּ בְּהִדּוּר, עִם אוֹרְחִים רַבִּים וּסְעוּדָה גְּדוֹלָה.
💛 אַהֲבַת הַשֵּׁם: בְּכָל דָּבָר רִאשׁוֹן – הַפְּרִי הָרִאשׁוֹן, הַבֵּן הָרִאשׁוֹן – רוֹאִים יוֹתֵר אֶת יַד ה׳. הַבְּכוֹר מַזְכִּיר לָנוּ שֶׁכָּל יֶלֶד הוּא מַתָּנָה מִן הַשָּׁמַיִם.

🕍פֶּטֶר רֶחֶם וְשֵׁבֶט לֵוִי

לְעִנְיַן פִּדְיוֹן, בְּכוֹר הוּא ״פֶּטֶר רֶחֶם״ – הַבֵּן הָרִאשׁוֹן שֶׁל אִמּוֹ. לְעִנְיַן יְרֻשָּׁה, בְּכוֹר הוּא הַבֵּן הָרִאשׁוֹן שֶׁל אָבִיו, וְהוּא יוֹרֵשׁ פִּי שְׁנַיִם.

״יֵשׁ קְדֻשָּׁה בַּבֵּן הָרִאשׁוֹן שֶׁנּוֹלַד, עַד שֶׁאֵינוֹ יָכוֹל לִחְיוֹת חַיֵּי חֻלִּין רְגִילִים בָּעוֹלָם הַזֶּה, מִבְּלִי שֶׁנִּפְדֶּה אוֹתוֹ כְּמִצְוַת הַתּוֹרָה״.

🙋הַכֹּהֵן שׁוֹאֵל

🔁הַכֹּהֵן וְהַכֶּסֶף

רַבִּי טַרְפוֹן, שֶׁהָיָה כֹּהֵן, הָיָה מַחֲזִיר לִפְעָמִים אֶת כֶּסֶף הַפִּדְיוֹן – כַּנִּרְאֶה לְהוֹרִים עֲנִיִּים. מֻתָּר לַכֹּהֵן לְהַחְזִיר אֶת הַכֶּסֶף בְּמַתָּנָה, אֲבָל לֹא תָּמִיד – כְּדֵי שֶׁלֹּא כָּל הַהוֹרִים יָבוֹאוּ רַק אֵלָיו, וְהַכֹּהֲנִים הָאֲחֵרִים יְקֻפְּחוּ.
הָאָב צָרִיךְ לָתֵת בְּלֵב שָׁלֵם, וְלֹא לְחַכּוֹת שֶׁיַּחְזִירוּ לוֹ. פַּעַם הִסְתּוֹבֵב אָב לְיַד רַבִּי חֲנִינָא הַכֹּהֵן, כְּדֵי לִרְמֹז לוֹ שֶׁיַּחְזִיר – וְרַבִּי חֲנִינָא אָמַר לוֹ: ״לֹא טוֹב אַתָּה עוֹשֶׂה!״
💛 אַהֲבַת הַשֵּׁם: נְתִינָה אֲמִתִּית הִיא נְתִינָה בְּלִי ״אוּלַי יַחְזִירוּ לִי״. כָּךְ אָנוּ רוֹצִים לָתֵת לַה׳ – בְּלֵב שָׁלֵם.

🕍מִי פָּטוּר מִפִּדְיוֹן

בְּכוֹר שֶׁנּוֹלַד לְכֹהֵן אוֹ לְלֵוִי – פָּטוּר מִפִּדְיוֹן, כִּי הַלְוִיִּים בָּאוּ בִּמְקוֹם בְּכוֹרֵי יִשְׂרָאֵל. וּבְכוֹר שֶׁל יִשְׂרָאֵל נִפְדֶּה רַק אֵצֶל כֹּהֵן, לֹא אֵצֶל לֵוִי.
גַּם כְּשֶׁהָאָב יִשְׂרְאֵלִי, אִם הָאֵם הִיא בַּת כֹּהֵן אוֹ בַּת לֵוִי – הַבְּכוֹר פָּטוּר, כִּי הַבְּכוֹרָה לְפִדְיוֹן תְּלוּיָה בָּאֵם. אֲבָל הַבַּת שֶׁל בַּת הַכֹּהֵן, שֶׁאָבִיהָ יִשְׂרְאֵלִי, הִיא כְּבָר יִשְׂרְאֵלִית – וּבְנָהּ הַבְּכוֹר חַיָּב בְּפִדְיוֹן. כָּךְ לִפְעָמִים סַבָּא רַבָּא כֹּהֵן פּוֹדֶה אֶת הַנִּין שֶׁלּוֹ!
💛 אַהֲבַת הַשֵּׁם: כָּל שֵׁבֶט וְכָל מִשְׁפָּחָה בְּעַם יִשְׂרָאֵל קְשׁוּרִים זֶה לָזֶה, וְכֻלָּם יַחַד עוֹבְדִים אֶת ה׳.

🧠 בְּדֹק אֶת עַצְמְךָ

לְאַבָּא נוֹלַד בֵּן בְּכוֹר. בְּמָה מֻתָּר לוֹ לִפְדּוֹת אוֹתוֹ?
💡 אֶפְשָׁר לִפְדּוֹת בְּכֶסֶף טָהוֹר אוֹ בְּחֵפֶץ שֶׁשָּׁוֶה כְּמוֹתוֹ, כְּמוֹ פָּמוֹטִים אוֹ גְּבִיעֵי כֶּסֶף. אֲבָל לֹא בְּקַרְקַע, לֹא בְּצֶ׳ק וְלֹא בִּשְׁטָרוֹת.
הַבְּכוֹר נוֹלַד בְּיוֹם רִאשׁוֹן. בְּאֵיזֶה יוֹם יִהְיֶה הַפִּדְיוֹן?
💡 פּוֹדִים בַּיּוֹם הַשְּׁלֹשִׁים וְאֶחָד, וְסוֹפְרִים גַּם אֶת יוֹם הַלֵּדָה וְגַם אֶת יוֹם הַפִּדְיוֹן. לָכֵן הַפִּדְיוֹן תָּמִיד אַחֲרֵי אַרְבָּעָה שָׁבוּעוֹת וְיוֹמַיִם: נוֹלַד בְּיוֹם רִאשׁוֹן – נִפְדֶּה בְּיוֹם שְׁלִישִׁי.
אַבָּא יִשְׂרְאֵלִי, וְאִמָּא בַּת כֹּהֵן. נוֹלַד לָהֶם בֵּן בְּכוֹר. מָה הַדִּין?
💡 כָּתוּב ״פֶּטֶר כָּל רֶחֶם בִּבְנֵי יִשְׂרָאֵל״ – הַחִיּוּב תָּלוּי בָּאֵם, שֶׁתִּהְיֶה יִשְׂרְאֵלִית. בַּת כֹּהֵן אוֹ בַּת לֵוִי אֵינָהּ יִשְׂרְאֵלִית, וְלָכֵן הַבְּכוֹר שֶׁלָּהּ פָּטוּר.

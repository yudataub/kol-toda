כְּלָלֵי אִסּוּר חָמֵץ

🍷 הִלְכוֹת פֶּסַח · פֶּרֶק ב׳

⛔אַרְבַּע מִצְווֹת בְּחָמֵץ

שָׁלֹשׁ מִצְווֹת ״לֹא תַעֲשֶׂה״: לֹא לֶאֱכֹל חָמֵץ (וְגַם לֹא לֵהָנוֹת מִמֶּנּוּ), שֶׁלֹּא יֵרָאֶה חָמֵץ שֶׁלָּנוּ, וְשֶׁלֹּא יִמָּצֵא חָמֵץ שֶׁלָּנוּ. וּמִצְוַת עֲשֵׂה אַחַת: לְהַשְׁבִּית אֶת הֶחָמֵץ לִקְרַאת הַפֶּסַח.
אִסּוּר אֲכִילַת חָמֵץ בְּפֶסַח חָמוּר מְאוֹד: עָנְשׁוֹ כָּרֵת, וְלֹא מַלְקוּת כְּמוֹ רֹב אִסּוּרֵי הָאֲכִילָה.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה מְבַקֶּשֶׁת לְהִתְרַחֵק מֵהֶחָמֵץ לְגַמְרֵי. כְּמוֹ שֶׁמִּתְרַחֲקִים מִדָּבָר שֶׁמַּזִּיק, כָּךְ בְּפֶסַח אֲנַחְנוּ מְנַקִּים אֶת הַבַּיִת וְאֶת הַלֵּב לִכְבוֹד ה׳.
📖 פֶּרֶק ב׳, הֲלָכָה א׳

📜״לֹא יֵרָאֶה וְלֹא יִמָּצֵא״

״וְלֹא יֵרָאֶה לְךָ חָמֵץ, וְלֹא יֵרָאֶה לְךָ שְׂאֹר בְּכָל גְּבֻלֶךָ״, וְ״שִׁבְעַת יָמִים שְׂאֹר לֹא יִמָּצֵא בְּבָתֵּיכֶם״. עַל שְׁנֵי הָאִסּוּרִים הָאֵלֶּה עוֹבֵר מִי שֶׁיֵּשׁ בִּרְשׁוּתוֹ בְּפֶסַח חָמֵץ בְּשִׁעוּר כַּזַּיִת לְפָחוֹת. וּמִצְוַת הָעֲשֵׂה: ״תַּשְׁבִּיתוּ שְׂאֹר מִבָּתֵּיכֶם״.
הָאוֹכֵל חָמֵץ בְּפֶסַח בְּמֵזִיד – עָנְשׁוֹ כָּרֵת. וּבְמִצְוַת הַשְׁבָּתַת הֶחָמֵץ, כְּמוֹ בִּשְׁאָר מִצְווֹת הַפֶּסַח, אֲנָשִׁים וְנָשִׁים שָׁוִים.
💛 אַהֲבַת הַשֵּׁם: קֹדֶם מְפַנִּים מֵהַבַּיִת אֶת הֶחָמֵץ, שֶׁרוֹמֵז לְיֵצֶר הָרָע – וְאָז אֶפְשָׁר לְקַבֵּל אֶת הַקְּדֻשָּׁה שֶׁל הַמַּצָּה.
📖 פֶּרֶק ב׳, הֲלָכָה א׳; פֶּרֶק ג׳, הֲלָכָה א׳

🚫🔑 מוּשָּׂג: בַּל יֵרָאֶה וּבַל יִמָּצֵא

״וְלֹא יֵרָאֶה לְךָ חָמֵץ וְלֹא יֵרָאֶה לְךָ שְׂאֹר בְּכָל גְּבֻלֶךָ... שִׁבְעַת יָמִים שְׂאֹר לֹא יִמָּצֵא בְּבָתֵּיכֶם״.
📖 בַּל יֵרָאֶה וּבַל יִמָּצֵא – שְׁנֵי אִסּוּרִים מִן הַתּוֹרָה: שֶׁלֹּא יֵרָאֶה וְשֶׁלֹּא יִמָּצֵא חָמֵץ בִּרְשׁוּתֵנוּ בְּפֶסַח.
💛 אַהֲבַת הַשֵּׁם: ה׳ לֹא מְבַקֵּשׁ רַק שֶׁלֹּא נֹאכַל חָמֵץ, אֶלָּא שֶׁנִּתְרַחֵק מִמֶּנּוּ לְגַמְרֵי, כְּמוֹ מִי שֶׁמַּקְפִּיד עַל חֵפֶץ שֶׁל חָבֵר. 🧒 וְאַתָּה? אִם קִבַּלְתָּ עַל עַצְמְךָ לִשְׁמֹר עַל חֵפֶץ – שְׁמֹר עָלָיו הֵיטֵב, כְּאִלּוּ הוּא שֶׁלְּךָ.
📖 פֶּרֶק ב׳, הֲלָכָה א׳

⏰עַד מָתַי אוֹכְלִים חָמֵץ?

בְּעֶרֶב פֶּסַח, י״ד בְּנִיסָן, מְחַלְּקִים אֶת הַיּוֹם לִשְׁתֵּים עֶשְׂרֵה ״שָׁעוֹת זְמַנִּיּוֹת״. בְּאַרְבַּע הַשָּׁעוֹת הָרִאשׁוֹנוֹת מֻתָּר לֶאֱכֹל חָמֵץ. בַּשָּׁעָה הַחֲמִישִׁית אָסוּר לֶאֱכֹל, אֲבָל מֻתָּר לֵהָנוֹת: לְהַאֲכִיל בּוֹ בְּהֵמָה אוֹ לִמְכֹּר אוֹתוֹ לְגוֹי. מֵהַשָּׁעָה הַשִּׁשִּׁית – אָסוּר גַּם לֵהָנוֹת.
מֵחֲצוֹת הַיּוֹם הָאִסּוּר הוּא מִן הַתּוֹרָה. חֲכָמִים הוֹסִיפוּ עוֹד שָׁעָה לַהֲנָאָה וְעוֹד שְׁתֵּי שָׁעוֹת לַאֲכִילָה, כִּי בְּיוֹם מְעֻנָּן אֶפְשָׁר לִטְעוֹת בַּזְּמַן.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים בָּנוּ לָנוּ ״גָּדֵר״ כְּדֵי שֶׁלֹּא נִכָּשֵׁל. מִי שֶׁאוֹהֵב אֶת ה׳ שָׂמֵחַ לְהִתְרַחֵק מֵהָעֲבֵרָה עוֹד צַעַד.
📖 פֶּרֶק ב׳, הֲלָכָה ב׳; פֶּרֶק ג׳, הֲלָכָה ו׳

🍞מָה זֶה חָמֵץ?

חָמֵץ נוֹצָר מֵאֶחָד מֵחֲמֵשֶׁת מִינֵי דָּגָן – חִטָּה, שְׂעוֹרָה, שִׁבֹּלֶת שׁוּעָל, שִׁיפוֹן וְכֻסְּמִין – שֶׁנָּגְעוּ בְּמַיִם וְהֶחְמִיצוּ. מֵהֶם אוֹפִים לֶחֶם, עוּגוֹת וּפִתּוֹת. וְדַוְקָא מֵהֶם עוֹשִׂים גַּם אֶת הַמַּצָּה: לָשִׁים מַהֵר וּמַכְנִיסִים מִיָּד לַתַּנּוּר, לִפְנֵי שֶׁהַבָּצֵק מַסְפִּיק לְהַחְמִיץ.
אֹרֶז וְדֹחַן אֵינָם חָמֵץ מִן הַתּוֹרָה, גַּם אִם הֵם תּוֹפְחִים. (עַל מִנְהַג הַקִּטְנִיּוֹת נִלְמַד בְּחִידוֹן 3.)
💛 אַהֲבַת הַשֵּׁם: הַלֶּחֶם הוּא הַמַּאֲכָל הֶחָשׁוּב שֶׁל הָאָדָם, וְעָלָיו מְבָרְכִים ״הַמּוֹצִיא״. בְּפֶסַח אֲנַחְנוּ מְוַתְּרִים עָלָיו בְּשִׂמְחָה – כִּי ה׳ בִּקֵּשׁ.
📖 פֶּרֶק ב׳, הֲלָכָה ג׳

⏱️שְׁמוֹנֶה עֶשְׂרֵה דַּקּוֹת

כָּל זְמַן שֶׁלָּשִׁים אֶת הַבָּצֵק, הוּא לֹא מַחְמִיץ – אֲפִלּוּ יוֹם שָׁלֵם. אֲבָל בָּצֵק שֶׁל קֶמַח וּמַיִם שֶׁעוֹמֵד 18 דַּקּוֹת בְּלִי תְּנוּעָה – מַתְחִיל לְהַחְמִיץ. בְּמָקוֹם חַם הוּא מַחְמִיץ עוֹד יוֹתֵר מַהֵר.
בָּצֵק שֶׁנִּרְאוּ בּוֹ סְדָקִים – כְּבָר הֶחְמִיץ, גַּם אִם לֹא עָבְרוּ 18 דַּקּוֹת.
💛 אַהֲבַת הַשֵּׁם: בְּפֶסַח מְמַהֲרִים וּמְזָרְזִים, כְּמוֹ אֲבוֹתֵינוּ שֶׁיָּצְאוּ מִמִּצְרַיִם ״טֶרֶם יֶחְמָץ״. הַזְּרִיזוּת הִיא סִימָן שֶׁל אַהֲבָה לַמִּצְוָה.
📖 פֶּרֶק ב׳, הֲלָכוֹת ג׳–ד׳

🍞🔑 מוּשָּׂג: חָמֵץ

״חָמֵץ שֶׁאָסְרָה הַתּוֹרָה בְּפֶסַח הוּא אֶחָד מֵחֲמֵשֶׁת מִינֵי דָּגָן שֶׁנָּגְעוּ בְּמַיִם עַד שֶׁהֶחְמִיצוּ: חִטָּה, שְׂעוֹרָה, שִׁבֹּלֶת שׁוּעָל, שִׁיפוֹן וְכֻסְּמִין״.
📖 חָמֵץ – בָּצֵק מֵחֲמֵשֶׁת מִינֵי דָּגָן שֶׁהֶחְמִיץ בְּמַגָּעוֹ עִם מַיִם.
💛 אַהֲבַת הַשֵּׁם: הֶחָמֵץ רוֹמֵז לְיֵצֶר הַגַּאֲוָה, וְהַמַּצָּה לְמִדַּת הָעֲנָוָה – ה׳ רוֹצֶה שֶׁדַּוְקָא בְּפֶסַח נִזְכֹּר מִי בָּרָא אֶת הָעוֹלָם וְקָבַע אֶת כָּל יְסוֹדוֹתָיו. 🧒 וְאַתָּה? לִפְנֵי שֶׁאַתָּה מְחַפֵּשׁ חָמֵץ בַּחֶדֶר – תַּגִּיד בַּלֵּב: אֲנִי קָטָן וְעָנָו לִפְנֵי ה׳.
📖 פֶּרֶק ב׳, הֲלָכָה ג׳; פֶּרֶק א׳, הֲלָכָה ה׳

🫧🔑 מוּשָּׂג: שְׂאוֹר

״שְׁנֵי סוּגֵי חִמּוּץ: חָמֵץ וּשְׂאוֹר... שְׂאוֹר (כְּעֵין שְׁמָרִים) הוּא הַסּוּג הַשֵּׁנִי שֶׁל חָמֵץ, וְתַפְקִידוֹ לְזָרֵז וּלְהַשְׁבִּיחַ אֶת תַּהְלִיךְ הַהֶחְמָצָה״.
📖 שְׂאוֹר – כְּעֵין שְׁמָרִים, חֹמֶר שֶׁמְּזָרֵז אֶת הֶחְמָצַת הַבָּצֵק.
💛 אַהֲבַת הַשֵּׁם: אֲפִלּוּ חֹמֶר קָטָן יָכוֹל לְשַׁנּוֹת הַכֹּל – ה׳ רוֹצֶה שֶׁנִּשְׁמֹר גַּם מִדְּבָרִים קְטַנִּים שֶׁיּוֹצְרִים שִׁנּוּי גָּדוֹל לְרַע. 🧒 וְאַתָּה? שִׂים לֵב לְדָבָר קָטָן שֶׁיָּכוֹל לְהַשְׁפִּיעַ עָלֶיךָ, לְטוֹב אוֹ לְרַע.
📖 פֶּרֶק ב׳, הֲלָכָה ג׳

🐕חָמֵץ שֶׁהִתְקַלְקֵל

לֶחֶם שֶׁהִתְעַפֵּשׁ וְאָדָם כְּבָר לֹא יֹאכַל אוֹתוֹ – עֲדַיִן חָמֵץ, כִּי אֶפְשָׁר לְהַחְמִיץ בּוֹ בָּצֵק אַחֵר. רַק חָמֵץ שֶׁהִתְקַלְקֵל כָּל כָּךְ, עַד שֶׁאֲפִלּוּ כֶּלֶב לֹא יֹאכַל אוֹתוֹ – אֵינוֹ נֶחְשָׁב יוֹתֵר חָמֵץ.
אֹכֶל לִכְלָבִים וּלְחָתוּלִים שֶׁיֵּשׁ בּוֹ חָמֵץ – חוֹבָה לְבַעֵר, כִּי הֶחָמֵץ שֶׁבּוֹ הָיָה רָאוּי מִתְּחִלָּתוֹ לְמַאֲכַל אָדָם.
💛 אַהֲבַת הַשֵּׁם: גַּם פֵּרוּר יָשָׁן אֵינוֹ ״לֹא חָשׁוּב״ בְּפֶסַח. כָּךְ אֲנַחְנוּ לוֹמְדִים לְדַקְדֵּק בְּמִצְווֹת ה׳ גַּם בִּדְבָרִים קְטַנִּים.
📖 פֶּרֶק ב׳, הֲלָכָה ו׳

🧠 בְּדֹק אֶת עַצְמְךָ

בְּמוֹצָאֵי פֶּסַח רוֹצִים לִקְנוֹת עוּגִיּוֹת. אֵיזֶה חָמֵץ מֻתָּר?
💡 חָמֵץ שֶׁל גּוֹי שֶׁעָבַר עָלָיו הַפֶּסַח – מֻתָּר. אֲבָל חָמֵץ שֶׁל יְהוּדִי שֶׁהִשְׁאִיר אוֹתוֹ בִּרְשׁוּתוֹ בְּפֶסַח – חֲכָמִים אָסְרוּ אוֹתוֹ בַּאֲכִילָה וּבַהֲנָאָה.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

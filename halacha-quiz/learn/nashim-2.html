מִצְוַת תְּפִלָּה לְנָשִׁים

🌸 תְּפִלַּת בָּנוֹת · פֶּרֶק ב׳

🙏גַּם בָּנוֹת מִתְפַּלְּלוֹת

חֲכָמִים אָמְרוּ בַּמִּשְׁנָה שֶׁנָּשִׁים חַיָּבוֹת בִּתְפִלָּה. לְדַעַת רֹב הַפּוֹסְקִים, בַּת חַיֶּבֶת לְהִתְפַּלֵּל שְׁמוֹנֶה עֶשְׂרֵה בְּשַׁחֲרִית וּבְמִנְחָה, בְּדִיּוּק כְּמוֹ בֵּן. וְגַם בְּשַׁבָּתוֹת וּבְחַגִּים מִתְפַּלְּלוֹת שַׁחֲרִית וּמִנְחָה.
עַרְבִית הִיא תְּפִלַּת רְשׁוּת לְבָנוֹת. הַגְּבָרִים קִבְּלוּ עֲלֵיהֶם לְהִתְפַּלֵּל עַרְבִית כְּחוֹבָה, אֲבָל הַנָּשִׁים לֹא קִבְּלוּ זֹאת עֲלֵיהֶן.
💛 אַהֲבַת הַשֵּׁם: הַתְּפִלָּה נוֹעֲדָה לְבַקֵּשׁ רַחֲמִים מֵה׳, וְכֻלָּנוּ – בָּנִים וּבָנוֹת – צְרִיכִים אֶת רַחֲמָיו. לָכֵן כָּל בַּת מְדַבֶּרֶת עִם ה׳ בְּכָל יוֹם, כְּמוֹ בַּת שֶׁפּוֹנָה אֶל אָבִיהָ הָאוֹהֵב.
📖 פֶּרֶק ב׳, הֲלָכוֹת א׳–ב׳

🌙לִפְנֵי הַשֵּׁינָה

גַּם בָּנוֹת צְרִיכוֹת לוֹמַר קְרִיאַת שְׁמַע שֶׁעַל הַמִּטָּה, וּלְבָרֵךְ אֶת בִּרְכַּת ״הַמַּפִּיל״ לִפְנֵי הַשֵּׁינָה.
בַּבֹּקֶר פּוֹתְחוֹת אֶת הַיּוֹם בִּבְרָכוֹת וּבִתְפִלָּה, וּבַלַּיְלָה סוֹגְרוֹת אוֹתוֹ בִּקְרִיאַת שְׁמַע. כָּךְ כָּל הַיּוֹם עָטוּף בִּתְפִלָּה.
💛 אַהֲבַת הַשֵּׁם: לִפְנֵי שֶׁנִּרְדָּמִים מַפְקִידִים אֶת עַצְמֵנוּ בְּיַד ה׳, כְּמוֹ יֶלֶד קָטָן שֶׁנִּרְדָּם בְּשֶׁקֶט בִּזְרוֹעוֹת אָבִיו.
📖 פֶּרֶק ב׳, הֲלָכָה א׳

🌅גַּם הַבְּרָכוֹת – חוֹבָה

גַּם בַּת שֶׁמִּתְפַּלֶּלֶת שְׁמוֹנֶה עֶשְׂרֵה צְרִיכָה לְבָרֵךְ אֶת בִּרְכוֹת הַשַּׁחַר וְאֶת בִּרְכוֹת הַתּוֹרָה. מִי שֶׁלֹּא הִסְפִּיקָה לְהִתְפַּלֵּל שַׁחֲרִית – תִּתְפַּלֵּל מִנְחָה, וּבְדִיעֲבַד, אִם לֹא הִסְפִּיקָה גַּם מִנְחָה – תִּתְפַּלֵּל עַרְבִית.
לָמָּה בָּנוֹת חַיָּבוֹת בִּתְפִלָּה, אַף שֶׁהִיא תְּלוּיָה בִּזְמַן? כִּי הַתְּפִלָּה נוֹעֲדָה לְבַקֵּשׁ רַחֲמִים, וְגַם הַנָּשִׁים צְרִיכוֹת לְבַקֵּשׁ רַחֲמִים כְּמוֹ הַגְּבָרִים.
💛 אַהֲבַת הַשֵּׁם: בִּרְכוֹת הַבֹּקֶר הֵן הַתּוֹדָה הָרִאשׁוֹנָה שֶׁל הַיּוֹם, וְהַתְּפִלָּה הִיא הַבַּקָּשָׁה. כְּשֶׁמַּתְחִילִים אֶת הַיּוֹם בְּתוֹדָה וּבְבַקָּשָׁה, כָּל הַיּוֹם קָשׁוּר לַה׳.
📖 פֶּרֶק ב׳, הֲלָכוֹת א׳–ב׳, ה׳

🔀מַחֲלֹקֶת פּוֹסְקִים

מוּסָף: יֵשׁ אוֹמְרִים שֶׁנָּשִׁים חַיָּבוֹת בּוֹ כְּמוֹ בְּשַׁחֲרִית וּבְמִנְחָה (הַמָּגֵן גִּבּוֹרִים), וְיֵשׁ אוֹמְרִים שֶׁפְּטוּרוֹת, כִּי הוּא תָּלוּי בִּזְמַן (הַצְּלָ״ח). קְרִיאַת הַתּוֹרָה בְּשַׁבָּת: לְדַעַת הַמָּגֵן אַבְרָהָם נָשִׁים חַיָּבוֹת לִשְׁמֹעַ, וּלְדַעַת רֹב רֻבָּם שֶׁל הַפּוֹסְקִים – פְּטוּרוֹת. זְכִירַת יְצִיאַת מִצְרַיִם בְּכָל יוֹם: לְדַעַת הַמָּגֵן אַבְרָהָם חַיָּבוֹת, וְלָכֵן אוֹמְרוֹת ״אֱמֶת וְיַצִּיב״; לְדַעַת רֹב הַפּוֹסְקִים – פְּטוּרוֹת.
לַהֲלָכָה בְּכָל שְׁלָשְׁתָּן נָשִׁים פְּטוּרוֹת, וְהָרוֹצָה – מְקַיֶּמֶת מִצְוָה. בְּמוּסָף לְכַתְּחִלָּה רָאוּי לָחוּשׁ לַמְחַיְּבִים, וּבְרֹאשׁ הַשָּׁנָה וּבְיוֹם הַכִּפּוּרִים רָאוּי שֶׁכָּל אִשָּׁה תִּתְפַּלֵּל מוּסָף. טוֹב לִשְׁמֹעַ קְרִיאַת הַתּוֹרָה. וְאֶת יְצִיאַת מִצְרַיִם עָדִיף לִזְכֹּר בְּ״אֱמֶת וְיַצִּיב״.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁהַפּוֹסְקִים חֲלוּקִים, כֻּלָּם רוֹצִים שֶׁנִּתְקָרֵב לַה׳. מִי שֶׁמּוֹסִיפָה מִצְוָה מֵרָצוֹן – תָּבוֹא עָלֶיהָ בְּרָכָה.
📖 פֶּרֶק ב׳, הֲלָכוֹת א׳, ט׳–י׳; פֶּרֶק ט״ז, הֲלָכָה ג׳

⚖️שָׁלֹשׁ שִׁיטוֹת

כַּמָּה תְּפִלּוֹת חַיֶּבֶת בַּת לְהִתְפַּלֵּל? לְדַעַת רֹב הַפּוֹסְקִים – שַׁחֲרִית וּמִנְחָה. יֵשׁ אוֹמְרִים, לְפִי שִׁיטַת הָרַמְבַּ״ם, שֶׁהִיא חַיֶּבֶת רַק תְּפִלַּת עֲמִידָה אַחַת בְּכָל יוֹם, בְּאֶחָד מִזְּמַנֵּי הַתְּפִלָּה. וְיֵשׁ מְקִלִּים שֶׁאוֹמְרִים שֶׁהִיא יוֹצֵאת בְּבִרְכוֹת הַשַּׁחַר וּבְבִרְכוֹת הַתּוֹרָה, כִּי יֵשׁ בָּהֶן שֶׁבַח, בַּקָּשָׁה וְהוֹדָאָה.
לְמַעֲשֶׂה: לְכַתְּחִלָּה טוֹב לְהִתְפַּלֵּל שַׁחֲרִית וּמִנְחָה. מִי שֶׁמִּתְפַּלֶּלֶת תְּפִלָּה אַחַת – יָצְאָה, וְטוֹב שֶׁתִּהְיֶה שַׁחֲרִית, ״כְּדֵי לִפְתֹּחַ אֶת הַיּוֹם בִּתְפִלָּה״. וּלְהִסְתַּפֵּק בַּבְּרָכוֹת בִּלְבַד – ״אֵין רָאוּי לִנְהֹג כֵּן״.
💛 אַהֲבַת הַשֵּׁם: כָּל הַשִּׁיטוֹת מַסְכִּימוֹת שֶׁבַּת פּוֹנָה אֶל ה׳ בְּכָל יוֹם. הַמַּחֲלֹקֶת הִיא רַק כַּמָּה, וְהַלֵּב הָאוֹהֵב רוֹצֶה לְדַבֵּר עִם ה׳ כַּמָּה שֶׁיּוֹתֵר.
📖 פֶּרֶק ב׳, הֲלָכוֹת ג׳–ה׳

⏰מִצְווֹת שֶׁהַזְּמַן גְּרָמָן

״כָּל מִצְוַת עֲשֵׂה שֶׁהַזְּמַן גְּרָמָא – אֲנָשִׁים חַיָּבִין וְנָשִׁים פְּטוּרוֹת״. מִצְוָה כָּזוֹ צָרִיךְ לְקַיֵּם רַק בִּזְמַן מְסֻיָּם. לָכֵן בָּנוֹת פְּטוּרוֹת מִקְּרִיאַת שְׁמַע, מִתְּפִלִּין, מִצִּיצִית, מִסֻּכָּה, מִלּוּלָב, מִשּׁוֹפָר וּמִסְּפִירַת הָעֹמֶר.
בְּמִצְווֹת שֶׁאֵינָן תְּלוּיוֹת בִּזְמַן בָּנוֹת חַיָּבוֹת כְּמוֹ בָּנִים: מְזוּזָה, צְדָקָה, הַלְוָאָה לְמִי שֶׁצָּרִיךְ, וּתְרוּמוֹת וּמַעַשְׂרוֹת.
💛 אַהֲבַת הַשֵּׁם: כָּל מִצְוָה הִיא דֶּרֶךְ לְהִתְקָרֵב לַה׳. בָּנוֹת קוֹלְטוֹת אֶת הָאוֹר שֶׁל הַמִּצְווֹת גַּם כְּשֶׁהֵן פְּטוּרוֹת מֵהֶן, וּמְשַׂמְּחוֹת אֶת ה׳ בַּמִּצְווֹת הָרַבּוֹת שֶׁהֵן חַיָּבוֹת בָּהֶן.
📖 פֶּרֶק ב׳, הֲלָכָה ז׳

✅חַיָּבוֹת – אַף שֶׁתָּלוּי בִּזְמַן

יֵשׁ מִצְווֹת שֶׁתְּלוּיוֹת בִּזְמַן, וּבְכָל זֹאת לָמַדְנוּ מִן הַתּוֹרָה שֶׁנָּשִׁים חַיָּבוֹת בָּהֶן: אֲכִילַת מַצָּה בְּלֵיל הַסֵּדֶר, שִׂמְחַת הֶחָג וְקִדּוּשׁ שֶׁל שַׁבָּת. וּבְמִצְווֹת שֶׁתִּקְּנוּ חֲכָמִים בִּגְלַל נֵס בָּנוֹת חַיָּבוֹת, כִּי ״אַף הֵן הָיוּ בְּאוֹתוֹ הַנֵּס״: אַרְבַּע כּוֹסוֹת, מִקְרָא מְגִלָּה וְנֵרוֹת חֲנֻכָּה.
בְּכָל מִצְווֹת ״לֹא תַעֲשֶׂה״ בָּנוֹת חַיָּבוֹת כְּמוֹ בָּנִים, גַּם כְּשֶׁהֵן תְּלוּיוֹת בִּזְמַן. לְמָשָׁל: לֹא לֶאֱכֹל חָמֵץ בְּפֶסַח, וְלֹא לֶאֱכֹל וְלִשְׁתּוֹת בְּיוֹם הַכִּפּוּרִים.
💛 אַהֲבַת הַשֵּׁם: בְּלֵיל הַסֵּדֶר, בְּפוּרִים וּבַחֲנֻכָּה כָּל הַמִּשְׁפָּחָה מוֹדָה יַחַד לַה׳ עַל הַנִּסִּים שֶׁעָשָׂה לָנוּ. גַּם הַבָּנוֹת הָיוּ שָׁם, וְגַם הֵן מוֹדוֹת בְּשִׂמְחָה.
📖 פֶּרֶק ב׳, הֲלָכָה ז׳

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

הַאִם בַּת מְבָרֶכֶת עַל מִצְוָה שֶׁהַזְּמַן גְּרָמָהּ, כְּמוֹ סֻכָּה אוֹ לוּלָב? לְדַעַת הָרַמְבַּ״ם וְהַשֻּׁלְחָן עָרוּךְ – לֹא, כִּי אֵיךְ תֹּאמַר ״וְצִוָּנוּ״ כְּשֶׁלֹּא נִצְטַוְּתָה? וְכָךְ מִנְהַג רֹב יוֹצְאוֹת סְפָרַד. לְדַעַת רַבֵּנוּ תָּם וְהָרַמָּ״א – מְבָרֶכֶת, כִּי ״וְצִוָּנוּ״ פֵּרוּשׁוֹ כְּלַל יִשְׂרָאֵל, וְהִיא חֵלֶק מִכְּלַל יִשְׂרָאֵל. וְכָךְ מִנְהַג כָּל יוֹצְאוֹת אַשְׁכְּנַז. וְכָךְ גַּם בְּהַלֵּל: לְמִנְהַג אַשְׁכְּנַז רַשָּׁאִית לְבָרֵךְ, וּלְמִנְהַג סְפָרַד לֹא תְּבָרֵךְ.
בִּרְכוֹת שֶׁבַח שֶׁאֵין בָּהֶן ״וְצִוָּנוּ״, כְּמוֹ ״בָּרוּךְ שֶׁאָמַר״ וְ״יִשְׁתַּבַּח״: לְדַעַת הָרַב עוֹבַדְיָה יוֹסֵף, בְּבֵית סֵפֶר שֶׁלּוֹמְדוֹת בּוֹ אַשְׁכְּנַזִּיּוֹת וּסְפָרַדִּיּוֹת, מַדְרִיכִים אֶת הַסְּפָרַדִּיּוֹת לֹא לוֹמַר אֶת חֲתִימַת הַבְּרָכוֹת. לְדַעַת פּוֹסְקִים רַבִּים, וְכָךְ הַמִּנְהָג הָרוֹוֵחַ, גַּם הַסְּפָרַדִּיּוֹת מְבָרְכוֹת, כִּי אֵלֶּה בִּרְכוֹת שֶׁבַח וְהוֹדָיָה.
💛 אַהֲבַת הַשֵּׁם: כָּל הַמִּנְהָגִים נִשְׁעָנִים עַל גְּדוֹלֵי תּוֹרָה. כָּל בַּת שׁוֹמֶרֶת עַל מִנְהַג מִשְׁפַּחְתָּהּ בְּאַהֲבָה, וּמְכַבֶּדֶת אֶת חֲבֶרְתָּהּ שֶׁנּוֹהֶגֶת אַחֶרֶת.
📖 פֶּרֶק ב׳, הֲלָכוֹת ח׳–ט׳

🎺מוּסָף וְהַלֵּל

תְּפִלַּת מוּסָף נֶאֱמֶרֶת בְּשַׁבָּתוֹת, בְּרָאשֵׁי חֳדָשִׁים, בְּיָמִים טוֹבִים וּבְחֹל הַמּוֹעֵד, כְּנֶגֶד קָרְבְּנוֹת הַמּוּסָף. לַהֲלָכָה אֵין חוֹבָה לְבָנוֹת לְהִתְפַּלֵּל מוּסָף, וְהָרוֹצָה – רַשָּׁאִית וְיֵשׁ לָהּ זְכוּת. וּמֵאֲמִירַת הַלֵּל, לְכָל הַדֵּעוֹת, בָּנוֹת פְּטוּרוֹת, וְהָרוֹצָה לוֹמַר – תָּבוֹא עָלֶיהָ בְּרָכָה.
בְּרֹאשׁ הַשָּׁנָה וּבְיוֹם הַכִּפּוּרִים רָאוּי שֶׁכָּל אִשָּׁה תִּתְפַּלֵּל מוּסָף, כִּי בִּתְפִלַּת מוּסָף נִמְצֵאת עִקַּר בַּקָּשַׁת הָרַחֲמִים שֶׁל הַיָּמִים הַנּוֹרָאִים.
💛 אַהֲבַת הַשֵּׁם: בַּיָּמִים הַנּוֹרָאִים כָּל עַם יִשְׂרָאֵל מְבַקֵּשׁ רַחֲמִים יַחַד. כְּשֶׁגַּם הַבָּנוֹת מִצְטָרְפוֹת לִתְפִלַּת הַמּוּסָף, הַבַּקָּשָׁה שֶׁל כֻּלָּנוּ שְׁלֵמָה יוֹתֵר.

📜רוֹאִים אֶת הַתּוֹרָה

כְּשֶׁמַּגְבִּיהִים אֶת סֵפֶר הַתּוֹרָה, מִצְוָה עַל הַגְּבָרִים וְעַל הַנָּשִׁים לִרְאוֹת אֶת הַכְּתָב, לִכְרֹעַ וְלוֹמַר ״וְזֹאת הַתּוֹרָה אֲשֶׁר שָׂם מֹשֶׁה לִפְנֵי בְּנֵי יִשְׂרָאֵל״.
בַּת שֶׁיְּכוֹלָה – טוֹב שֶׁתִּשְׁמַע אֶת קְרִיאַת הַתּוֹרָה בְּשַׁבָּת. גַּם מִי שֶׁפְּטוּרָה, אִם הִיא שׁוֹמַעַת – מְקַיֶּמֶת מִצְוָה וְיֵשׁ לָהּ בָּזֶה זְכוּת.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאָנוּ רוֹאִים אֶת אוֹתִיּוֹת הַתּוֹרָה מוּרָמוֹת לְמַעְלָה, אָנוּ אוֹמְרִים בְּגַאֲוָה: זוֹ הַתּוֹרָה שֶׁלָּנוּ, שֶׁה׳ נָתַן לָנוּ בְּאַהֲבָה.
📖 פֶּרֶק ב׳, הֲלָכָה י׳

🧠 בְּדֹק אֶת עַצְמְךָ

שִׁירָה יְכוֹלָה לְהִתְפַּלֵּל הַיּוֹם רַק תְּפִלַּת עֲמִידָה אַחַת. אֵיזוֹ עָדִיף שֶׁתִּתְפַּלֵּל?
💡 מִי שֶׁמִּתְפַּלֶּלֶת תְּפִלָּה אַחַת בְּיוֹם – יָצְאָה, וְטוֹב שֶׁתִּהְיֶה זוֹ שַׁחֲרִית, כְּדֵי לִפְתֹּחַ אֶת הַיּוֹם בִּתְפִלָּה. לֹא הִסְפִּיקָה – תִּתְפַּלֵּל מִנְחָה, וּבְדִיעֲבַד עַרְבִית.
לָמָּה בָּנוֹת חַיָּבוֹת בִּתְפִלָּה, אַף שֶׁהִיא מִצְוָה הַתְּלוּיָה בִּזְמַן?
💡 הַתְּפִלָּה נוֹעֲדָה לְבַקֵּשׁ רַחֲמִים, וְגַם הַנָּשִׁים צְרִיכוֹת רַחֲמִים כְּמוֹ הַגְּבָרִים. לָכֵן תִּקְּנוּ חֲכָמִים אֶת הַתְּפִלּוֹת לִגְבָרִים וּלְנָשִׁים כְּאַחַת. ״אַף הֵן הָיוּ בְּאוֹתוֹ הַנֵּס״ הוּא הַטַּעַם לִמְגִלָּה וְלַחֲנֻכָּה, וְלֹא לִתְפִלָּה.
הֲדַסָּה אוֹמֶרֶת: ״אֲנִי מִסְתַּפֶּקֶת בְּבִרְכוֹת הַשַּׁחַר וּבְבִרְכוֹת הַתּוֹרָה, בְּלִי עֲמִידָה״. מָה כּוֹתֵב הָרַב מֶלָמֵד?
💡 יֵשׁ מְקִלִּים שֶׁבַּת יוֹצֵאת בְּבִרְכוֹת הַשַּׁחַר וְהַתּוֹרָה, וְלָכֵן יֵשׁ עַל מָה לִסְמֹךְ בְּדֹחַק. אֲבָל לְדַעַת רֻבָּם הַמַּכְרִיעַ שֶׁל הַפּוֹסְקִים חוֹבָה לְהִתְפַּלֵּל עֲמִידָה לְפָחוֹת פַּעַם אַחַת בְּיוֹם, וְלָכֵן אֵין רָאוּי לִנְהֹג כָּךְ.

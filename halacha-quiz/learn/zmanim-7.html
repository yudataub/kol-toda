דִּינֵי הַצּוֹמוֹת הַקַּלִּים

📅 רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת · פֶּרֶק ז׳

⏰צוֹם קַל – מָתַי וּמָה

בִּשְׁלֹשֶׁת הַצּוֹמוֹת הַקַּלִּים צָמִים מֵעֲלוֹת הַשַּׁחַר – כְּשֶׁהָאוֹר הָרִאשׁוֹן נִרְאֶה בַּמִּזְרָח – וְעַד צֵאת הַכּוֹכָבִים. אָסוּר בָּהֶם רַק לֶאֱכֹל וְלִשְׁתּוֹת. רְחִיצָה וּנְעִילַת נַעֲלַיִם – מֻתָּרוֹת.
תִּשְׁעָה בְּאָב חָמוּר יוֹתֵר: הוּא מַתְחִיל כְּבָר בָּעֶרֶב, וְאָסוּר בּוֹ גַּם לִרְחֹץ וְלִנְעֹל נַעֲלַיִם. וּכְשֶׁעֲשָׂרָה בְּטֵבֵת חָל בְּיוֹם שִׁשִּׁי – צָמִים עַד צֵאת הַכּוֹכָבִים, אַף שֶׁהַשַּׁבָּת כְּבָר נִכְנְסָה.
💛 אַהֲבַת הַשֵּׁם: צוֹם שֶׁל יוֹם אֶחָד מְחַזֵּק אֶת הַלֵּב. אָנוּ מְוַתְּרִים לְכַמָּה שָׁעוֹת עַל אֹכֶל – כְּדֵי לִזְכֹּר מָה בֶּאֱמֶת חָסֵר לָנוּ.
📖 פֶּרֶק ז׳, הֲלָכוֹת ב׳–ג׳

🚿מָה עוֹד נָכוֹן בְּצוֹם קַל

בַּצּוֹמוֹת הַקַּלִּים אָסוּר רַק לֶאֱכֹל וְלִשְׁתּוֹת. אֲבָל טוֹב לְהַחְמִיר שֶׁלֹּא לְהִתְרַחֵץ בְּמַיִם חַמִּים בִּזְמַן הַצּוֹם, וּלְנִקָּיוֹן מֻתָּר לְהִתְרַחֵץ בְּפוֹשְׁרִים.
נָכוֹן גַּם שֶׁלֹּא לְהִסְתַּפֵּר, לֹא לִשְׁמֹעַ מוּזִיקָה שְׂמֵחָה וְלֹא לַעֲרֹךְ קְנִיּוֹת שֶׁל שִׂמְחָה בִּזְמַן הַצּוֹם – כְּמוֹ בְּתִשְׁעַת הַיָּמִים.
💛 אַהֲבַת הַשֵּׁם: הַצּוֹם אֵינוֹ רַק בֶּטֶן רֵיקָה. כְּשֶׁגַּם הַלֵּב זוֹכֵר אֶת הַחֻרְבָּן – הַצּוֹם מְקָרֵב אוֹתָנוּ לַגְּאֻלָּה.
📖 פֶּרֶק ז׳, הֲלָכָה ב׳

💧לִפְנֵי הַצּוֹם וּבְתוֹכוֹ

מִי שֶׁהָלַךְ לִישֹׁן לִפְנֵי צוֹם קַל וְקָם לִפְנֵי עֲלוֹת הַשַּׁחַר – אָסוּר לוֹ לֶאֱכֹל, כִּי כְּבָר הֵסִיחַ דַּעְתּוֹ מִלֶּאֱכֹל. אִם הִתְנָה בְּלִבּוֹ לִפְנֵי הַשֵּׁינָה שֶׁיֹּאכַל אִם יָקוּם – מֻתָּר. בִּשְׁתִיָּה נֶחְלְקוּ: לְפִי הָרַמָּ״א מֻתָּר לִשְׁתּוֹת גַּם בְּלִי תְּנַאי, כִּי רַבִּים רְגִילִים לִשְׁתּוֹת כְּשֶׁקָּמִים. לְפִי הַשֻּׁלְחָן עָרוּךְ – רַק אִם הִתְנָה. לָכֵן הָרוֹצֶה לִשְׁתּוֹת יַתְנֶה זֹאת בְּלִבּוֹ.
בְּצוֹם קַל מַתִּיר הָרַב מֶלָמֵד לְמִי שֶׁחוֹשֵׁשׁ מֵרֵיחַ רַע לִשְׁטֹף אֶת הַפֶּה וּלְצַחְצֵחַ שִׁנַּיִם עִם מִשְׁחָה, וְיִזָּהֵר מְאוֹד שֶׁלֹּא לִבְלֹעַ. וְיֵשׁ פּוֹסְקִים רַבִּים שֶׁמַּחְמִירִים שֶׁלֹּא לִשְׁטֹף אֶת הַפֶּה בְּתַעֲנִית צִבּוּר. חוֹלֶה שֶׁצָּרִיךְ תְּרוּפָה – בּוֹלֵעַ אוֹתָהּ, וְאִם אֶפְשָׁר – בְּלִי מַיִם.
💛 אַהֲבַת הַשֵּׁם: הַהֲכָנָה לַצּוֹם הִיא חֵלֶק מִן הַצּוֹם. מִי שֶׁחוֹשֵׁב מֵרֹאשׁ, שׁוֹמֵר עַל הַהֲלָכָה וְגַם עַל עַצְמוֹ.
📖 פֶּרֶק ז׳, הֲלָכוֹת ד׳–ה׳, ז׳

🔀שְׁתֵּי שִׁיטוֹת – וּשְׁתֵּיהֶן כַּהֲלָכָה

בָּשָׂר וְיַיִן: הָאַשְׁכְּנַזִּים – מֵרֹאשׁ חֹדֶשׁ אָב; רֹב הַסְּפָרַדִּים – אַחֲרֵי רֹאשׁ חֹדֶשׁ. תִּסְפֹּרֶת: הָאַשְׁכְּנַזִּים – כָּל שְׁלֹשֶׁת הַשָּׁבוּעוֹת; רַבִּים מִן הַסְּפָרַדִּים – בַּשָּׁבוּעַ שֶׁחָל בּוֹ תִּשְׁעָה בְּאָב. כְּבִיסָה: הָאַשְׁכְּנַזִּים – מֵרֹאשׁ חֹדֶשׁ; הַסְּפָרַדִּים – בַּשָּׁבוּעַ שֶׁחָל בּוֹ. "עֲנֵנוּ" בִּתְפִלַּת הַלַּחַשׁ: הָאַשְׁכְּנַזִּים – רַק בְּמִנְחָה; הַסְּפָרַדִּים – גַּם בְּשַׁחֲרִית.
שְׁתִיָּה לִפְנֵי עֲלוֹת הַשַּׁחַר בְּצוֹם קַל: לְפִי הָרַמָּ״א – מֻתֶּרֶת גַּם בְּלִי תְּנַאי; לְפִי הַשֻּׁלְחָן עָרוּךְ – רַק אִם הִתְנָה. רְחִיצָה בְּתִשְׁעַת הַיָּמִים לְמִי שֶׁמִּצְטַעֵר: לְפִי הַמִּנְהָג הַקַּדְמוֹן – לֹא מִתְרַחֲצִים; לְפִי הָרַב מֶלָמֵד – מִתְרַחֲצִים בְּפוֹשְׁרִים. עוֹלֵי תֵּימָן אוֹכְלִים בָּשָׂר עַד הַסְּעוּדָה הַמַּפְסֶקֶת.
💛 אַהֲבַת הַשֵּׁם: כָּל עֵדָה מִתְאַבֶּלֶת עַל יְרוּשָׁלַיִם בַּדֶּרֶךְ שֶׁקִּבְּלָה מֵרַבּוֹתֶיהָ, וּשְׁתֵּי הַדְּרָכִים הֵן תּוֹרָה. וְכֻלָּנוּ מִתְפַּלְּלִים יַחַד עַל אוֹתוֹ בַּיִת.
📖 פֶּרֶק ז׳, הֲלָכוֹת ד׳–ה׳, י׳; פֶּרֶק ח׳, הֲלָכוֹת י׳, י״ג, י״ט, כ״א

🤒חוֹלֶה, וּמִי שֶׁשָּׁכַח

חוֹלֶה פָּטוּר מִן הַצּוֹמוֹת הַקַּלִּים, גַּם כְּשֶׁאֵין בּוֹ סַכָּנָה – לְמָשָׁל, מִי שֶׁיֵּשׁ לוֹ שַׁפַּעַת אוֹ חֹם גָּבוֹהַּ וְשׁוֹכֵב בַּמִּטָּה. הוּא אוֹכֵל וְשׁוֹתֶה כְּפִי צָרְכּוֹ, בְּלִי מַאַכְלֵי פִּנּוּק. אֲבָל חֻלְשָׁה וּכְאֵב רֹאשׁ שֶׁל צוֹם – אֵינָם מַחֲלָה.
מִי שֶׁשָּׁכַח וְאָכַל אוֹ שָׁתָה בַּצּוֹם – חַיָּב לְהַמְשִׁיךְ לָצוּם. אֵינוֹ צָרִיךְ לָצוּם יוֹם אַחֵר בִּמְקוֹמוֹ, וְטוֹב שֶׁיּוֹסִיף צְדָקָה וְלִמּוּד תּוֹרָה.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה דּוֹאֶגֶת לַחוֹלִים בְּרַחֲמִים. וּמִי שֶׁנִּכְשַׁל – אֵינוֹ מְוַתֵּר, אֶלָּא מַמְשִׁיךְ מֵאוֹתוֹ רֶגַע בְּכָל הַכֹּחַ.
📖 פֶּרֶק ז׳, הֲלָכוֹת ו׳–ז׳

🧒יְלָדִים בַּצּוֹם

קְטַנִּים שֶׁלֹּא הִגִּיעוּ לְגִיל מִצְווֹת פְּטוּרִים מִן הַצּוֹמוֹת שֶׁתִּקְּנוּ חֲכָמִים. רַבִּים נוֹהֲגִים לְחַנְּכָם לָצוּם כַּמָּה שָׁעוֹת לְפִי כֹּחָם – אֲבָל לֹא כָּל הַיּוֹם.
כְּשֶׁמַּאֲכִילִים קְטַנִּים בְּיוֹם הַצּוֹם, נוֹתְנִים לָהֶם מַאֲכָלִים פְּשׁוּטִים בִּלְבַד – כְּדֵי לְחַנְּכָם לְהִתְאַבֵּל עִם הַצִּבּוּר.
💛 אַהֲבַת הַשֵּׁם: גַּם יֶלֶד שֶׁאוֹכֵל יָכוֹל לְהִשְׁתַּתֵּף בְּצַעַר הַצִּבּוּר: לְוַתֵּר עַל מַמְתָּק, וּלְהִתְפַּלֵּל שֶׁבֵּית הַמִּקְדָּשׁ יִבָּנֶה.
📖 פֶּרֶק ז׳, הֲלָכָה ט׳

👶🔑 מוּשָּׂג: קְטַנִּים פְּטוּרִים מֵהַצּוֹם

״קְטַנִּים שֶׁלֹּא הִגִּיעוּ לְגִיל מִצְווֹת, פְּטוּרִים מִן הַצּוֹמוֹת שֶׁתִּקְּנוּ חֲכָמִים... רַבִּים נוֹהֲגִים לְחַנְּכָם לָצוּם מִסְפַּר שָׁעוֹת לְפִי כֹּחָם״.
📖 קְטַנִּים – יְלָדִים שֶׁלֹּא הִגִּיעוּ לְגִיל מִצְווֹת, פְּטוּרִים מִצּוֹמוֹת הֶחָכָמִים, וְנוֹהֲגִים לְחַנְּכָם קְצָת לְפִי כֹּחָם.
💛 אַהֲבַת הַשֵּׁם: ה׳ לֹא מְבַקֵּשׁ מִילָדִים קְטַנִּים לָצוּם כָּל הַיּוֹם – הוּא רוֹצֶה שֶׁיִּלְמְדוּ לְאַט לְאַט, כְּפִי כֹּחָם. 🧒 וְאַתָּה? אִם אַתָּה קָטָן – שְׁאַל אֶת אַבָּא וְאִמָּא כַּמָּה שָׁעוֹת מַתְאִים לָך לָצוּם.
📖 פֶּרֶק ז׳, הֲלָכָה ט׳

🙏עֲנֵנוּ וּקְרִיאַת "וַיְחַל"

בְּתַעֲנִית צִבּוּר מוֹסִיפִים בַּתְּפִלָּה "עֲנֵנוּ", וְקוֹרְאִים בַּתּוֹרָה "וַיְחַל מֹשֶׁה" – עַל כַּפָּרַת חֵטְא הָעֵגֶל. בִּתְפִלַּת הַלַּחַשׁ אוֹמְרִים "עֲנֵנוּ" בְּבִרְכַּת "שׁוֹמֵעַ תְּפִלָּה": לְמִנְהַג הָאַשְׁכְּנַזִּים – רַק בְּמִנְחָה, שֶׁמָּא יֵחָלֵשׁ וְיִשְׁבֹּר אֶת הַצּוֹם. לְמִנְהַג הַסְּפָרַדִּים – גַּם בְּשַׁחֲרִית.
קוֹרְאִים קְרִיאַת תַּעֲנִית, וְהַחַזָּן אוֹמֵר "עֲנֵנוּ" בַּחֲזָרָה, רַק כְּשֶׁיֵּשׁ לְפָחוֹת שִׁשָּׁה מִתְעַנִּים. וּמִי שֶׁאָכַל כַּשִּׁעוּר וְשָׁבַר אֶת הַצּוֹם – אֵינוֹ אוֹמֵר "עֲנֵנוּ".
💛 אַהֲבַת הַשֵּׁם: כְּמוֹ שֶׁה׳ סָלַח עַל חֵטְא הָעֵגֶל וְנָתַן לוּחוֹת שְׁנִיִּים – כָּךְ נְבַקֵּשׁ שֶׁיִּסְלַח לָנוּ וְיִבְנֶה אֶת בֵּית הַמִּקְדָּשׁ.
📖 פֶּרֶק ז׳, הֲלָכוֹת י׳–י״א

📣📜 לְשׁוֹן חֲכָמִים: תַּעֲנִית צִבּוּר

״תִּקְּנוּ חֲכָמִים לְהוֹסִיף... בְּרָכָה מְיֻחֶדֶת עַל הַצּוֹם – 'עֲנֵנוּ'. וְאוֹמְרִים אוֹתָהּ בִּתְנַאי שֶׁיֵּשׁ שָׁם לְפָחוֹת שִׁשָּׁה מִתְעַנִּים״.
📖 תַּעֲנִית צִבּוּר – צוֹם שֶׁכָּל הַקָּהָל שׁוֹמֵר יַחַד, וּבוֹ מוֹסִיפִים בַּתְּפִלָּה בְּרָכַת 'עֲנֵנוּ'.
💛 אַהֲבַת הַשֵּׁם: בְּתַעֲנִית צִבּוּר אֲנַחְנוּ לֹא מִתְעַנִּים לְבַד – כָּל הַקָּהָל פּוֹנֶה יַחַד אֶל ה׳ וּמְבַקֵּשׁ שֶׁיַּעֲנֵנוּ. 🧒 וְאַתָּה? בְּיוֹם תַּעֲנִית צִבּוּר – עֲנֵה 'אָמֵן' בְּכַוָּנָה לְבִרְכַּת 'עֲנֵנוּ'.
📖 פֶּרֶק ז׳, הֲלָכָה י׳

🧠 בְּדֹק אֶת עַצְמְךָ

אַתָּה רוֹצֶה לִשְׁתּוֹת אִם תִּתְעוֹרֵר לִפְנֵי עֲלוֹת הַשַּׁחַר בְּיוֹם צוֹם קַל. מָה תַּעֲשֶׂה?
💡 מִי שֶׁהָלַךְ לִישֹׁן כְּבָר הֵסִיחַ דַּעְתּוֹ מִלֶּאֱכֹל. לְפִי הָרַמָּ״א מֻתָּר לִשְׁתּוֹת גַּם בְּלִי תְּנַאי, וּלְפִי הַשֻּׁלְחָן עָרוּךְ – רַק אִם הִתְנָה. לָכֵן, כְּדֵי לָצֵאת יְדֵי כֻּלָּם, יַתְנֶה בְּלִבּוֹ לִפְנֵי הַשֵּׁינָה. וּבְדִיעֲבַד, מִי שֶׁהִתְעוֹרֵר צָמֵא – רַשַּׁאי לִשְׁתּוֹת.
לְפִי הָרַב מֶלָמֵד, מָה הַדִּין בְּצִחְצוּחַ שִׁנַּיִם בְּצוֹם קַל, לְמִי שֶׁחוֹשֵׁשׁ מֵרֵיחַ רַע?
💡 הָרַב מֶלָמֵד מַתִּיר בְּצוֹם קַל לִשְׁטֹף אֶת הַפֶּה וּלְצַחְצֵחַ שִׁנַּיִם עִם מִשְׁחָה, כִּי אֵינוֹ מִתְכַּוֵּן לִשְׁתּוֹת – וְנִזְהָרִים שֶׁלֹּא לִבְלֹעַ. יֵשׁ פּוֹסְקִים רַבִּים שֶׁמַּחְמִירִים בְּתַעֲנִית צִבּוּר, וְכָל אֶחָד יִשְׁאַל אֶת רַבּוֹ. בְּיוֹם הַכִּפּוּרִים אָסוּר.
לָמָּה נוֹהֲגִים הָאַשְׁכְּנַזִּים לוֹמַר "עֲנֵנוּ" בִּתְפִלַּת הַלַּחַשׁ רַק בְּמִנְחָה?
💡 בִּתְפִלַּת "עֲנֵנוּ" אוֹמְרִים "בְּיוֹם צוֹם תַּעֲנִיתֵנוּ". מִי שֶׁאָמַר זֹאת בְּשַׁחֲרִית וְשָׁבַר אַחַר כָּךְ אֶת הַצּוֹם – אָמַר שֶׁקֶר. מִי שֶׁצָּם עַד מִנְחָה – מִן הַסְּתָם יְסַיֵּם. הַסְּפָרַדִּים אוֹמְרִים "עֲנֵנוּ" גַּם בְּשַׁחֲרִית.

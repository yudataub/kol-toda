הִלְכוֹת שְׁמִירַת הַנֶּפֶשׁ

🤝 בֵּין אָדָם לַחֲבֵרוֹ · פֶּרֶק ט׳

🏠מַעֲקֶה לַגַּג

מִצְוָה מִן הַתּוֹרָה לִבְנוֹת מַעֲקֶה לְגַג שֶׁעוֹלִים אֵלָיו וּמִשְׁתַּמְּשִׁים בּוֹ: "וְעָשִׂיתָ מַעֲקֶה לְגַגֶּךָ, וְלֹא תָשִׂים דָּמִים בְּבֵיתֶךָ". מִי שֶׁמַּשְׁאִיר גַּג כָּזֶה בְּלִי מַעֲקֶה – מְבַטֵּל מִצְוַת עֲשֵׂה וְעוֹבֵר עַל לֹא תַעֲשֶׂה.
כְּשֶׁבּוֹנִים מַעֲקֶה לַגַּג מְבָרְכִים: "אֲשֶׁר קִדְּשָׁנוּ בְּמִצְוֹתָיו וְצִוָּנוּ לַעֲשׂוֹת מַעֲקֶה". וְגַג רְעָפִים מְשֻׁפָּע, שֶׁאֵין עוֹלִים עָלָיו – פָּטוּר.
💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לָנוּ אֶת הַחַיִּים בְּמַתָּנָה, וְהוּא רוֹצֶה שֶׁנִּשְׁמֹר עֲלֵיהֶם. גַּם בְּנִיַּת מַעֲקֶה הִיא עֲבוֹדַת ה׳ – וּמְבָרְכִים עָלֶיהָ כְּמוֹ עַל מִצְוָה.
📖 פֶּרֶק ט׳, הֲלָכוֹת א׳, ה׳

🧭מִן הַמַּעֲקֶה – לְכָל הַסַּכָּנוֹת

חֲכָמִים לָמְדוּ מִמִּצְוַת הַמַּעֲקֶה מִצְוָה כְּלָלִית: לְהִזָּהֵר מִכָּל הַסַּכָּנוֹת וּלְהָסִיר אוֹתָן. הַגַּג בְּלִי מַעֲקֶה הוּא רַק דֻּגְמָה – וְכָךְ גַּם כָּל מִכְשׁוֹל אַחֵר.
עַל בְּנִיַּת מַעֲקֶה לַגַּג מְבָרְכִים "לַעֲשׂוֹת מַעֲקֶה". עַל כִּסּוּי בּוֹר – לֹא מְבָרְכִים, כִּי שָׁם יֵשׁ רַק מִצְוַת לֹא תַעֲשֶׂה.
💛 אַהֲבַת הַשֵּׁם: כָּל פַּעַם שֶׁאָנוּ מַרְחִיקִים סַכָּנָה, אָנוּ שׁוֹמְרִים עַל מַתְּנַת הַחַיִּים. ה׳ שָׂמֵחַ בְּמִי שֶׁשּׁוֹמֵר עַל עַצְמוֹ וְעַל אֲחֵרִים.
📖 פֶּרֶק ט׳, הֲלָכוֹת א׳, ו׳; פֶּרֶק י׳, הֲלָכָה א׳

🪜🔑 מוּשָּׂג: מַעֲקֶה

״כִּי תִבְנֶה בַּיִת חָדָשׁ וְעָשִׂיתָ מַעֲקֶה לְגַגֶּךָ, וְלֹא תָשִׂים דָּמִים בְּבֵיתֶךָ״ (דְּבָרִים כ״ב, ח׳).
📖 מַעֲקֶה – גָּדֵר גְּבוֹהָה בִּשְׂפַת הַגַּג, שֶׁמּוֹנַעַת נְפִילָה.
💛 אַהֲבַת הַשֵּׁם: ה׳ דּוֹאֵג לָנוּ עוֹד לִפְנֵי שֶׁקָּרָה מַשֶּׁהוּ רַע, וְלָכֵן נָתַן מִצְווָה לִבְנוֹת מַעֲקֶה מֵרֹאשׁ. 🧒 וְאַתָּה? אַל תִּתְקָרֵב לִשְׂפַת גַּג אוֹ מִרְפֶּסֶת בְּלִי מַעֲקֶה גָּבוֹהַ.
📖 פֶּרֶק ט׳, הֲלָכָה א׳

🪜🔑 מוּשָּׂג: שְׁתֵּי מִצְווֹת בַּמַּעֲקֶה

״מִצְוַת בְּנִיַּת הַמַּעֲקֶה כּוֹלֶלֶת שְׁתֵּי מִצְווֹת: מִצְוַת עֲשֵׂה לִבְנוֹת אֶת הַמַּעֲקֶה, וּמִצְוַת לֹא תַעֲשֶׂה – ״וְלֹא תָשִׂים דָּמִים בְּבֵיתֶךָ״״.
📖 מִי שֶׁמַּשְׁאִיר גַּג בְּלֹא מַעֲקֶה – לֹא רַק מְבַטֵּל מִצְוָה, אֶלָּא גַּם עוֹבֵר עַל אִסּוּר.
💛 אַהֲבַת הַשֵּׁם: ה׳ לֹא מִסְתַּפֵּק בְּ״אַל תַּזִּיק״ – הוּא מְצַוֶּה אוֹתָנוּ מֵרֹאשׁ לִבְנוֹת הֲגָנָה. 🧒 וְאַתָּה? בְּבַיִת אוֹ בְּגִנָּה בְּלִי מַעֲקֶה – הַרְחֵק בְּעַצְמְךָ מֵהַקָּצֶה.
📖 פֶּרֶק ט׳, הֲלָכָה א׳

🧱מַעֲקֶה גָּבוֹהַּ וְחָזָק

הַמַּעֲקֶה צָרִיךְ לִהְיוֹת גָּבוֹהַּ עֲשָׂרָה טְפָחִים – לְפִי רֹב הַפּוֹסְקִים כִּשְׁמוֹנִים סֶנְטִימֶטֶר, וְיֵשׁ מַחְמִירִים שֶׁצָּרִיךְ מֶטֶר. וְהוּא צָרִיךְ לִהְיוֹת חָזָק, כָּךְ שֶׁאָדָם יוּכַל לְהִשָּׁעֵן עָלָיו וְלֹא לִפֹּל.
קִיר נָמוּךְ בְּגֹבַהּ שֶׁל סַפְסָל אֵינוֹ מַעֲקֶה, כִּי יְלָדִים עֲלוּלִים לַעֲלוֹת וְלַעֲמֹד עָלָיו. בּוֹנִים מֵעֵבֶר לוֹ מַעֲקֶה גָּבוֹהַּ.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה מְלַמֶּדֶת אוֹתָנוּ לַחְשֹׁב מֵרֹאשׁ עַל כָּל מִי שֶׁעָלוּל לְהִפָּגַע – גַּם עַל יֶלֶד קָטָן שֶׁמְּטַפֵּס. זוֹ אַהֲבָה שֶׁרוֹאָה אֶת הָאַחֵר.
📖 פֶּרֶק ט׳, הֲלָכוֹת ג׳, ה׳

🕳️בּוֹר מְכֻסֶּה, סֻלָּם תָּקִין

בּוֹר אוֹ בְּאֵר בֶּחָצֵר – מְכַסִּים אוֹתוֹ אוֹ בּוֹנִים סְבִיבוֹ מַעֲקֶה, גַּם אִם אֵין בּוֹ מַיִם. וְכֵן בְּחָצֵר שֶׁיֵּשׁ בָּהּ שְׁנֵי גְּבָהִים – בּוֹנִים מַעֲקֶה בֵּינֵיהֶם.
אָסוּר לְהַשְׁאִיר בַּבַּיִת סֻלָּם רָעוּעַ אוֹ מַכְשִׁיר חַשְׁמַל מְקֻלְקָל, כִּי מִי שֶׁלֹּא יוֹדֵעַ עָלוּל לְהִשְׁתַּמֵּשׁ בָּהֶם וּלְהִפָּגַע.
💛 אַהֲבַת הַשֵּׁם: "וְלֹא תָשִׂים דָּמִים בְּבֵיתֶךָ" – הַבַּיִת שֶׁלָּנוּ צָרִיךְ לִהְיוֹת מָקוֹם בָּטוּחַ לְכָל מִי שֶׁנִּכְנָס אֵלָיו. כָּךְ מְכַבְּדִים אֶת הַחַיִּים שֶׁה׳ נָתַן.
📖 פֶּרֶק ט׳, הֲלָכוֹת ג׳, ו׳–ז׳

🔀גֹּבַהּ הַמַּעֲקֶה – שְׁתֵּי שִׁיטוֹת

חֲכָמִים קָבְעוּ שֶׁהַמַּעֲקֶה צָרִיךְ לִהְיוֹת גָּבוֹהַּ עֲשָׂרָה טְפָחִים. אֲבָל נֶחְלְקוּ הָאַחֲרוֹנִים כַּמָּה הוּא טֶפַח: לְדַעַת רֹב הַפּוֹסְקִים – שְׁמוֹנָה סֶנְטִימֶטְרִים, וּמַסְפִּיק מַעֲקֶה שֶׁל שְׁמוֹנִים סֶנְטִימֶטְרִים.
לְדַעַת הַמַּחְמִירִים, וּבֵינֵיהֶם הַחֲזוֹן אִישׁ – טֶפַח הוּא עֲשָׂרָה סֶנְטִימֶטְרִים, וְהַמַּעֲקֶה צָרִיךְ לִהְיוֹת גָּבוֹהַּ מֶטֶר שָׁלֵם. וּלְכָל הַדֵּעוֹת – הוּא צָרִיךְ לִהְיוֹת חָזָק, שֶׁאֶפְשָׁר לְהִשָּׁעֵן עָלָיו.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה דּוֹאֶגֶת לַחַיִּים שֶׁלָּנוּ עַד הַסֶּנְטִימֶטֶר. מִי שֶׁבּוֹנֶה מַעֲקֶה כָּרָאוּי, שׁוֹמֵר עַל הַמַּתָּנָה הַגְּדוֹלָה שֶׁנָּתַן לָנוּ ה׳.
📖 פֶּרֶק ט׳, הֲלָכָה ג׳

🧴מְסִירִים כָּל סַכָּנָה

"הִשָּׁמֶר לְךָ וּשְׁמֹר נַפְשְׁךָ מְאֹד" – מִצְוָה לְהָסִיר כָּל מִכְשׁוֹל שֶׁיֵּשׁ בּוֹ סַכָּנָה. לָכֵן חֹמֶר רַעַל שׁוֹמְרִים בְּמָקוֹם שֶׁיְּלָדִים קְטַנִּים לֹא מַגִּיעִים אֵלָיו, וּלְעוֹלָם לֹא שָׂמִים אֶקוֹנוֹמִיקָה בְּבַקְבּוּק שֶׁשּׁוֹתִים מִמֶּנּוּ.
לֹא הוֹלְכִים לְיַד קִיר נָטוּי שֶׁעָלוּל לִפֹּל, וְלֹא עַל גֶּשֶׁר רָעוּעַ. וּמִי שֶׁמְּגַדֵּל כֶּלֶב מַפְחִיד – קוֹשֵׁר אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: שְׁמִירַת הַגּוּף הִיא שְׁמִירָה עַל הַכְּלִי שֶׁבּוֹ אָנוּ עוֹבְדִים אֶת ה׳. מִי שֶׁנִּזְהָר – יָכוֹל לְהַמְשִׁיךְ לִלְמֹד תּוֹרָה וְלַעֲשׂוֹת מִצְווֹת.
📖 פֶּרֶק ט׳, הֲלָכָה ז׳

📜הִשְׁתַּדְּלוּת וֶאֱמוּנָה

הָאֱמוּנָה אֵינָהּ אוֹמֶרֶת: "אַל תַּעֲשֶׂה כְּלוּם, וּסְמֹךְ עַל ה׳". הִיא אוֹמֶרֶת: עֲשֵׂה כָּל מָה שֶׁהַשֵּׂכֶל וְהַטֶּבַע דּוֹרְשִׁים – וְדַע שֶׁגַּם אָז אַתָּה צָרִיךְ לְעֶזְרַת ה׳. ה׳ מַנְהִיג אֶת הָעוֹלָם בְּדֶרֶךְ הַטֶּבַע, וְאֵין סוֹמְכִים עַל הַנֵּס.
אָסוּר לָלֶכֶת לְטִיּוּל בְּדֶרֶךְ מְסֻכֶּנֶת. וּבְדֶרֶךְ שֶׁיֵּשׁ בָּהּ קְצָת סַכָּנָה – רָאוּי לֹא לָלֶכֶת לְשֵׁם הֲנָאָה. וּמִי קוֹבֵעַ מָה מְסֻכָּן? אֲנָשִׁים בַּעֲלֵי נִסָּיוֹן, כְּמוֹ הָאַחְרָאִים לְכָךְ בַּמִּשְׁטָרָה אוֹ בִּיחִידוֹת הַחִלּוּץ.
💛 אַהֲבַת הַשֵּׁם: כָּךְ אָנוּ שֻׁתָּפִים שֶׁל ה׳: אָנוּ מִשְׁתַּדְּלִים, וְהוּא נוֹתֵן אֶת הַכֹּחַ וְהַבְּרָכָה. וְאַחֲרֵי הַהַצְלָחָה – זוֹכְרִים לְהוֹדוֹת לוֹ.
📖 פֶּרֶק ט׳, הֲלָכוֹת ז׳, י״ד

🚭לֹא מְעַשְּׁנִים

הַיּוֹם יָדוּעַ בְּבֵרוּר שֶׁהָעִשּׁוּן מְסֻכָּן מְאוֹד לַבְּרִיאוּת – וְלָכֵן אָסוּר מִן הַתּוֹרָה לְעַשֵּׁן, וְגַם נַרְגִּילָה. לִפְנֵי שֶׁמִּתְרַגְּלִים, קַל מְאוֹד לֹא לְהַתְחִיל; אַחֲרֵי שֶׁמִּתְמַכְּרִים – קָשֶׁה מְאוֹד לְהַפְסִיק.
גַּם מִי שֶׁכְּבָר מְעַשֵּׁן – מִצְוָה עָלָיו לְהַפְסִיק. וְהַנִּסָּיוֹן מַרְאֶה שֶׁרֹב הָאֲנָשִׁים מְסֻגָּלִים לְהַפְסִיק.
💛 אַהֲבַת הַשֵּׁם: "וְנִשְׁמַרְתֶּם מְאֹד לְנַפְשֹׁתֵיכֶם" – הַגּוּף שֶׁלָּנוּ הוּא פִּקָּדוֹן מֵאֵת ה׳. מִי שֶׁאוֹמֵר "לֹא" לַסִּיגַרְיָה, אוֹמֵר "כֵּן" לַחַיִּים שֶׁה׳ נָתַן לוֹ.
📖 פֶּרֶק ט׳, הֲלָכוֹת ח׳–ט׳, י״ג

🚬עָשָׁן שֶׁל אֲחֵרִים

אָסוּר לִגְרֹם עָשָׁן שֶׁמַּזִּיק לַאֲחֵרִים. בְּמָקוֹם צִבּוּרִי אֶפְשָׁר לִדְרֹשׁ מִמְּעַשֵּׁן לְהַפְסִיק לְעַשֵּׁן לְיָדְךָ – וְגַם אִם עִשֵּׁן שָׁם שָׁנִים רַבּוֹת, אֵין לוֹ זְכוּת לְהַזִּיק. וּבְבַיִת פְּרָטִי – בַּעַל הַבַּיִת קוֹבֵעַ.
אַבָּא שֶׁמְּבַקֵּשׁ מִבְּנוֹ לִקְנוֹת לוֹ סִיגַרְיוֹת: בְּעִקָּרוֹ אָסוּר, כִּי זוֹ עֶזְרָה לַעֲבֵרָה, וְכִבּוּד הוֹרִים אֵינוֹ דּוֹחֶה מִצְווֹת. אֲבָל אִם הַסֵּרוּב יִגְרֹם מְרִיבָה וְצַעַר גְּדוֹלִים בַּמִּשְׁפָּחָה – עָדִיף לִקְנוֹת, כְּדֵי לִשְׁמֹר עַל הָאַהֲבָה וְהַכָּבוֹד.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה דּוֹאֶגֶת גַּם לַבְּרִיאוּת וְגַם לַשָּׁלוֹם בַּבַּיִת. "דְּרָכֶיהָ דַרְכֵי נֹעַם" – גַּם כְּשֶׁצָּרִיךְ לִשְׁקֹל בְּחָכְמָה.

🚭לֹא מְעוֹדְדִים לְעַשֵּׁן

כֵּיוָן שֶׁהָעִשּׁוּן מְסֻכָּן לַבְּרִיאוּת, אָסוּר לְעוֹדֵד אֲחֵרִים לְעַשֵּׁן – זֶה ״לִפְנֵי עִוֵּר לֹא תִתֵּן מִכְשֹׁל״. לָכֵן אָסוּר גַּם לְפַרְסֵם סִיגַרְיוֹת, כִּי הַפִּרְסוּם מְפַתֶּה בְּעִקָּר צְעִירִים לְהַתְחִיל לְעַשֵּׁן וּלְהִתְמַכֵּר.
וְאֵין לוֹמַר ״אִם לֹא אֲנִי – יַעֲשׂוּ זֹאת אֲחֵרִים״. כָּל עִדּוּד מוֹסִיף, וְאַדְּרַבָּה – מִצְוָה לִמְחוֹת נֶגֶד זֶה.
💛 אַהֲבַת הַשֵּׁם: חֲבֵרוּת אֲמִתִּית שׁוֹמֶרֶת עַל הֶחָבֵר. מִי שֶׁאוֹהֵב אֶת חֲבֵרוֹ – מַרְחִיק אוֹתוֹ מִמָּה שֶׁיַּזִּיק לוֹ.

🧠 בְּדֹק אֶת עַצְמְךָ

עוֹדֵד אוֹמֵר: "ה׳ שׁוֹמֵר עָלַי, אָז אֲנִי לֹא צָרִיךְ קַסְדָּה כְּשֶׁאֲנִי רוֹכֵב עַל אוֹפַנַּיִם". מָה נָכוֹן?
💡 הָאֱמוּנָה אֵינָהּ אוֹמֶרֶת "אַל תַּעֲשֶׂה כְּלוּם". ה׳ מַנְהִיג אֶת הָעוֹלָם בְּדֶרֶךְ הַטֶּבַע, וְלָכֵן נִזְהָרִים כְּמוֹ שֶׁהַשֵּׂכֶל דּוֹרֵשׁ – וְיוֹדְעִים שֶׁגַּם אָז צְרִיכִים לְעֶזְרָתוֹ.
הַכִּתָּה מְתַכְנֶנֶת טִיּוּל בְּנַחַל. מִי קוֹבֵעַ אִם הַמַּסְלוּל מְסֻכָּן מִדַּי?
💡 דֶּרֶךְ מְסֻכֶּנֶת אָסוּר לְטַיֵּל בָּהּ, וּבְדֶרֶךְ שֶׁיֵּשׁ בָּהּ קְצָת סַכָּנָה רָאוּי לֹא לְטַיֵּל לְשֵׁם הֲנָאָה. אֶת זֶה קוֹבְעִים בַּעֲלֵי נִסָּיוֹן – אַנְשֵׁי הַמִּשְׁטָרָה וְהַחִלּוּץ.
בַּתַּחֲנָה, אִישׁ מְעַשֵּׁן לְיַד אַבָּא שֶׁל שָׁלוֹם. הַמְּעַשֵּׁן אוֹמֵר: "אֲנִי מְעַשֵּׁן כָּאן כְּבָר שָׁנִים". מָה הַדִּין?
💡 בְּמָקוֹם צִבּוּרִי אֶפְשָׁר לִדְרֹשׁ מִן הַמְּעַשֵּׁן לְהַפְסִיק. אֵין לְאָדָם "זְכוּת" לְהַזִּיק לִשְׁכֵנָיו, גַּם אִם עָשָׂה זֹאת שָׁנִים – וְהַיּוֹם יָדוּעַ שֶׁהָעָשָׁן מַזִּיק גַּם לְמִי שֶׁעוֹמֵד לְיָדוֹ.

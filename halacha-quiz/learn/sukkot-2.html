הִלְכוֹת סֻכָּה

🌿 הִלְכוֹת סֻכּוֹת · פֶּרֶק ב׳

🌿מִמָּה עוֹשִׂים סְכָךְ

הַסְּכָךְ הוּא עִקַּר הַסֻּכָּה, וְיֵשׁ לוֹ שְׁלֹשָׁה תְּנָאִים: שֶׁיִּגְדַּל מִן הַקַּרְקַע – כְּמוֹ עֵצִים, עֲנָפִים וְקָנִים; שֶׁיִּהְיֶה תָּלוּשׁ – וְלֹא עֲנָפִים שֶׁעוֹד מְחֻבָּרִים לָעֵץ; וְשֶׁלֹּא יִהְיֶה כְּלִי – לֹא כִּסְאוֹת, לֹא מִטּוֹת וְלֹא שִׁבְרֵי כֵּלִים.
בַּרְזֶל, פְּלַסְטִיק וְעוֹרוֹת – פְּסוּלִים, כִּי אֵינָם צוֹמְחִים. גַּם נְיָר וְקַרְטוֹן פְּסוּלִים, וְגַם פֵּרוֹת שֶׁבְּנֵי אָדָם אוֹכְלִים.
💛 אַהֲבַת הַשֵּׁם: הַסְּכָךְ הַפָּשׁוּט מִן הַטֶּבַע מַזְכִּיר לָנוּ שֶׁהַכֹּל בָּא מִיַּד ה׳ – מִן הָאֲדָמָה שֶׁהוּא בָּרָא.
📖 פֶּרֶק ב׳, הֲלָכָה ב׳

🪵נְסָרִים וּ״סְכָךְ לָנֶצַח״

תִּקְרַת בַּיִת מִקְּרָשִׁים פְּסוּלָה לַסֻּכָּה, כִּי הִיא קֶבַע. לָכֵן גָּזְרוּ חֲכָמִים שֶׁלֹּא לְסַכֵּךְ בִּנְסָרִים רְחָבִים, שֶׁנִּרְאִים כְּמוֹ תִּקְרָה, וְהַיּוֹם נוֹהֲגִים לְהַחְמִיר גַּם בִּנְסָרִים צָרִים יוֹתֵר. נְסָרִים שֶׁרָחְבָּם פָּחוֹת מִטֶּפַח (כ־7.5 ס״מ) כְּשֵׁרִים לְכָל הַדֵּעוֹת – אִם אֵין קוֹבְעִים אוֹתָם בְּמַסְמְרִים אוֹ בְּדֶבֶק.
״סְכָךְ לָנֶצַח״ – קָנִים אוֹ נְסָרִים דַּקִּים שֶׁמְּחֻבָּרִים בְּחוּטִים כְּמוֹ מַחְצֶלֶת – כָּשֵׁר, וְכָךְ נוֹהֲגִים. מַחְצֶלֶת שֶׁעָשׂוּ לְשֵׁם סְכָךְ – כְּשֵׁרָה, וּמַחְצֶלֶת שֶׁעָשׂוּ לִשְׁכִיבָה – פְּסוּלָה.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים שָׁמְרוּ שֶׁהַסֻּכָּה תִּשָּׁאֵר דִּירַת אַרְעַי אֲמִתִּית. כָּךְ הִיא מְלַמֶּדֶת אוֹתָנוּ בְּכָל שָׁנָה מֵחָדָשׁ לִבְטֹחַ בַּה׳ וְלֹא בַּקִּירוֹת.
📖 פֶּרֶק ב׳, הֲלָכוֹת ב׳, ד׳

☀️צִלָּתָהּ מְרֻבָּה מֵחַמָּתָהּ

הַסְּכָךְ צָרִיךְ לַחְסֹם אֶת רֹב קַרְנֵי הַשֶּׁמֶשׁ, כָּךְ שֶׁבַּסֻּכָּה יִהְיֶה יוֹתֵר צֵל מִשֶּׁמֶשׁ. מוֹדְדִים לְמַעְלָה, בַּסְּכָךְ עַצְמוֹ, וְלֹא לְפִי כִּתְמֵי הָאוֹר עַל הָרִצְפָּה.
לְכַתְּחִלָּה טוֹב שֶׁהַצֵּל יִהְיֶה רַב, וְעִם זֹאת יִרְאוּ דֶּרֶךְ הַסְּכָךְ כּוֹכָבִים בַּלַּיְלָה. סְכָךְ אָטוּם כָּל כָּךְ שֶׁגַּם גֶּשֶׁם אֵינוֹ עוֹבֵר בּוֹ – דּוֹמֶה לְגַג שֶׁל בַּיִת, וְיֵשׁ פּוֹסְלִים אוֹתוֹ.
💛 אַהֲבַת הַשֵּׁם: הַסְּכָךְ מֵגֵן עָלֵינוּ מִן הַשֶּׁמֶשׁ, וּבְכָל זֹאת אֵינוֹ סוֹגֵר אֶת הַשָּׁמַיִם. כָּךְ ה׳ שׁוֹמֵר עָלֵינוּ – וְאָנוּ מַבִּיטִים תָּמִיד לְמַעְלָה, אֵלָיו.
📖 פֶּרֶק ב׳, הֲלָכָה ג׳

🧱דְּפָנוֹת הַסֻּכָּה

כָּל חֹמֶר כָּשֵׁר לַדְּפָנוֹת – עֵץ, אֶבֶן, זְכוּכִית, פְּלַסְטִיק וַאֲפִלּוּ גֶּדֶר רֶשֶׁת – בִּתְנַאי שֶׁהֵן עוֹמְדוֹת בְּרוּחַ מְצוּיָה. לְכַתְּחִלָּה בּוֹנִים סֻכָּה מְהֻדֶּרֶת, עִם אַרְבַּע דְּפָנוֹת שְׁלֵמוֹת.
הַדֹּפֶן גְּבוֹהָה לְפָחוֹת עֲשָׂרָה טְפָחִים, וּמַתְחִילָה פָּחוֹת מִשְּׁלֹשָׁה טְפָחִים (כ־22 ס״מ) מֵעַל הָרִצְפָּה. חוּטִים חֲזָקִים אוֹ מוֹטוֹת שֶׁבֵּינֵיהֶם פָּחוֹת מִשְּׁלֹשָׁה טְפָחִים – נֶחְשָׁבִים דֹּפֶן. וְ״סֻכָּה לָנֶצַח״, שֶׁהַיְּרִיעוֹת שֶׁלָּהּ מְחֻבָּרוֹת הֵיטֵב מִכָּל הַצְּדָדִים: יֵשׁ מִפּוֹסְקֵי זְמַנֵּנוּ שֶׁפִּקְפְּקוּ בָּהּ, וְהָרַב מֶלָמֵד פּוֹסֵק שֶׁהִיא כְּשֵׁרָה, וְהַמְהַדְּרִים מוֹסִיפִים בָּהּ מוֹטוֹת.
💛 אַהֲבַת הַשֵּׁם: הַדְּפָנוֹת מַקִּיפוֹת אוֹתָנוּ מִכָּל הַצְּדָדִים, כְּמוֹ עַנְנֵי הַכָּבוֹד שֶׁהִקִּיפוּ אֶת יִשְׂרָאֵל מֵאַרְבַּע רוּחוֹת.
📖 פֶּרֶק ב׳, הֲלָכוֹת ו׳–ח׳

🌳תַּחַת כִּפַּת הַשָּׁמַיִם

הַסֻּכָּה צְרִיכָה לַעֲמֹד תַּחַת הַשָּׁמַיִם, כְּדֵי שֶׁרַק הַסְּכָךְ יְסַכֵּךְ עַל הַיּוֹשְׁבִים. סֻכָּה תַּחַת עֵץ אוֹ תַּחַת תִּקְרָה – פְּסוּלָה. וּמִי שֶׁפּוֹרֵשׂ סָדִין מִתַּחַת לַסְּכָךְ כְּדֵי לְהַרְבּוֹת צֵל – פָּסַל אֶת הַסֻּכָּה.
מֻתָּר לִבְנוֹת סֻכָּה לְיַד בִּנְיָן גָּבוֹהַּ שֶׁמַּסְתִּיר אֶת הַשֶּׁמֶשׁ, וְגַם מִתַּחַת לְחַבְלֵי כְּבִיסָה אוֹ חוּטֵי חַשְׁמַל דַּקִּים. וּמֻתָּר לָשֶׁבֶת בַּסֻּכָּה עִם כּוֹבַע גָּדוֹל עַל הָרֹאשׁ.
💛 אַהֲבַת הַשֵּׁם: שׁוּם דָּבָר אֵינוֹ מַפְרִיד בֵּינֵינוּ לְבֵין הַסְּכָךְ וְהַשָּׁמַיִם. כָּךְ גַּם בַּלֵּב – אָנוּ רוֹצִים שֶׁשּׁוּם דָּבָר לֹא יַפְרִיד בֵּינֵינוּ לְבֵין ה׳.
📖 פֶּרֶק ב׳, הֲלָכוֹת ט׳–י׳

🎨סֻכָּה נָאָה וּקְדוֹשָׁה

״זֶה אֵלִי וְאַנְוֵהוּ״ – מִצְוָה לִבְנוֹת סֻכָּה נָאָה, וּלְקַשֵּׁט אוֹתָהּ בְּשַׁרְשְׁרָאוֹת, בְּפֵרוֹת וּבִתְמוּנוֹת יָפוֹת. אֶת הַקִּשּׁוּטִים תּוֹלִים קָרוֹב לַסְּכָךְ – בְּתוֹךְ אַרְבָּעָה טְפָחִים (כ־30 ס״מ). וְנִזְהָרִים מִשְּׂרֵפָה: מַרְחִיקִים אֶת הַנּוּרוֹת מִן הַסְּכָךְ.
הַסֻּכָּה וְקִשּׁוּטֶיהָ קְדוֹשִׁים כָּל הַחַג – אָסוּר לָקַחַת מֵהֶם שׁוּם דָּבָר, אֲפִלּוּ קֵיסָם לַשִּׁנַּיִם. אֲבָל מֻתָּר לְהִשָּׁעֵן עַל הַדֹּפֶן וְלִתְלוֹת עָלֶיהָ חֲפָצִים, כְּמוֹ עַל קִיר שֶׁל בַּיִת.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאָנוּ מְקַשְּׁטִים אֶת הַסֻּכָּה בְּאַהֲבָה, אָנוּ מַרְאִים שֶׁהַמִּצְוָה יְקָרָה לָנוּ – כְּמוֹ שֶׁמְּקַשְּׁטִים אֶת הַבַּיִת לִקְרַאת אוֹרֵחַ חָשׁוּב.
📖 פֶּרֶק ב׳, הֲלָכוֹת י׳, ט״ו–ט״ז

🛏️בְּצֵל הַסְּכָךְ

מִצְוַת הַסֻּכָּה – לָשֶׁבֶת בְּצֵל הַסְּכָךְ. רֶוַח שֶׁל אֲוִיר בַּסְּכָךְ, שֶׁרָחְבּוֹ שְׁלֹשָׁה טְפָחִים (כ־22 ס״מ) – אָסוּר לָשֶׁבֶת תַּחְתָּיו. רֶוַח קָטָן מִזֶּה – בָּטֵל לַסֻּכָּה, וּבִלְבַד שֶׁרֹב הָרֹאשׁ וְרֹב הַגּוּף לֹא יִהְיוּ תַּחְתָּיו.
מִטַּת קוֹמָתַיִם: אִם בֵּין הַמִּטָּה הַתַּחְתּוֹנָה לָעֶלְיוֹנָה יֵשׁ עֲשָׂרָה טְפָחִים (כ־76 ס״מ) – הַיָּשֵׁן לְמַטָּה לֹא קִיֵּם מִצְוַת סֻכָּה. אִם יֵשׁ פָּחוֹת – קִיֵּם. וְכָךְ גַּם הַיָּשֵׁן מִתַּחַת לַשֻּׁלְחָן.
💛 אַהֲבַת הַשֵּׁם: הַסֻּכָּה נִקְרֵאת ״צֵל הָאֱמוּנָה״. אָנוּ רוֹצִים לִהְיוֹת כֻּלָּנוּ, מִן הָרֹאשׁ וְעַד הָרַגְלַיִם, מִתַּחַת לַצֵּל הַקָּדוֹשׁ הַזֶּה.
📖 פֶּרֶק ב׳, הֲלָכוֹת י׳–י״א; פֶּרֶק א׳, הֲלָכָה ו׳

🔨תַּעֲשֶׂה – וְלֹא מִן הֶעָשׂוּי

אֶת הַסֻּכָּה עוֹשִׂים עַל יְדֵי הֲנָחַת הַסְּכָךְ. לָכֵן בּוֹנִים קֹדֶם אֶת הַדְּפָנוֹת, וְרַק אַחַר כָּךְ מַנִּיחִים אֶת הַסְּכָךְ. מִי שֶׁהֵנִיחַ קֹדֶם סְכָךְ וְאַחַר כָּךְ בָּנָה דְּפָנוֹת – לְדַעַת פּוֹסְקִים רַבִּים הַסֻּכָּה פְּסוּלָה. וְגַגּוֹן שֶׁפּוֹרְשִׂים בְּשָׁעַת גֶּשֶׁם: כְּשֶׁהוּא פָּרוּשׂ – הַסֻּכָּה פְּסוּלָה, וּכְשֶׁמְּסִירִים אוֹתוֹ – הִיא חוֹזֶרֶת לִכְשֵׁרוּתָהּ.
סֻכָּה שֶׁנִּשְׁאֲרָה מִשָּׁנָה שֶׁעָבְרָה, אוֹ סֻכָּה שֶׁעָשׂוּ רוֹעִים לְצֵל – כְּשֵׁרָה. אֲבָל לְכַתְּחִלָּה מְחַדְּשִׁים בָּהּ דָּבָר לִכְבוֹד הַחַג: מוֹסִיפִים סְכָךְ טֶפַח עַל טֶפַח, אוֹ קָנֶה דַּק לְכָל אָרְכָּהּ.
💛 אַהֲבַת הַשֵּׁם: בְּכָל שָׁנָה אָנוּ בּוֹנִים מֵחָדָשׁ, כִּי הַמִּצְוָה חֲדָשָׁה וַאֲהוּבָה. כָּךְ הַסֻּכָּה שֶׁל הַשָּׁנָה נִבְנֵית בִּשְׁבִיל הַחַג שֶׁל הַשָּׁנָה.
📖 פֶּרֶק ב׳, הֲלָכוֹת י״ב–י״ג

🔨מִמִּצְוָה לְמִצְוָה

מִצְוָה לַעֲסֹק בְּהָכָנַת הַסֻּכָּה. חֲסִידִים וְאַנְשֵׁי מַעֲשֶׂה מַתְחִילִים לִבְנוֹת אוֹתָהּ מִיָּד בְּמוֹצָאֵי יוֹם הַכִּפּוּרִים, כְּדֵי לָצֵאת מִמִּצְוָה לְמִצְוָה, וּמִשְׁתַּדְּלִים לְסַיֵּם עַד לְמָחֳרָת.
מְבָרְכִים עַל הַסֻּכָּה כְּשֶׁיּוֹשְׁבִים בָּהּ, וְלֹא כְּשֶׁבּוֹנִים אוֹתָהּ. וּבְכָל זֹאת – גַּם בַּבְּנִיָּה יֵשׁ מִצְוָה.
💛 אַהֲבַת הַשֵּׁם: כָּל עָנָף שֶׁאָנוּ מַנִּיחִים הוּא חֵלֶק מִן הַמִּצְוָה. מִי שֶׁבּוֹנֶה בְּשִׂמְחָה – מַרְאֶה לַה׳ כַּמָּה הוּא מְחַכֶּה לֶחָג.

🤝סֻכָּה שֶׁל חָבֵר

מֻתָּר לְקַיֵּם אֶת הַמִּצְוָה בְּסֻכָּה שְׁאוּלָה. גַּם כְּשֶׁבַּעַל הַסֻּכָּה אֵינוֹ נִמְצָא – מֻתָּר לָשֶׁבֶת בָּהּ, כִּי נוֹחַ לִיהוּדִי שֶׁעוֹשִׂים מִצְוָה בְּמָמוֹנוֹ. אֲבָל אִם יָדוּעַ שֶׁהוּא מַקְפִּיד, אוֹ שֶׁהוּא עָלוּל לַחֲזֹר וּלְהִתְבַּיֵּשׁ לְהִכָּנֵס – אָסוּר בְּלִי רְשׁוּת.
מִי שֶׁהֵקִים סֻכָּה בְּשֶׁטַח שֶׁל אַחֵר בְּלִי רְשׁוּת, אוֹ בְּשֶׁטַח צִבּוּרִי כְּשֶׁהַצִּבּוּר מִתְנַגֵּד – לֹא יְבָרֵךְ בָּהּ, כִּי הַסֻּכָּה הוּקְמָה בְּאִסּוּר.
💛 אַהֲבַת הַשֵּׁם: ה׳ אֵינוֹ רוֹצֶה מִצְוָה שֶׁפּוֹגַעַת בְּאַחֵר. סֻכָּה כְּשֵׁרָה בֶּאֱמֶת הִיא סֻכָּה שֶׁבְּנוּיָה גַּם עַל יֹשֶׁר וְדֶרֶךְ אֶרֶץ.

🧠 בְּדֹק אֶת עַצְמְךָ

אַבָּא רוֹצֶה לְסַכֵּךְ בִּנְסָרִים. אֵילוּ נְסָרִים כְּשֵׁרִים לְכָל הַדֵּעוֹת?
💡 נְסָרִים שֶׁרָחְבָּם פָּחוֹת מִטֶּפַח כְּשֵׁרִים לְכָל הַדֵּעוֹת – אִם אֵין קוֹבְעִים אוֹתָם בְּמַסְמְרִים אוֹ בְּדֶבֶק. הַצֶּבַע אֵינוֹ פּוֹסֵל אֶת הַסְּכָךְ, אֲבָל גַּם אֵינוֹ מַכְשִׁיר נֶסֶר רָחָב.
הַסֻּכָּה שֶׁל מִשְׁפַּחַת לֵוִי נִשְׁאֲרָה עוֹמֶדֶת מִשָּׁנָה שֶׁעָבְרָה. מָה עוֹשִׂים לִפְנֵי הֶחָג?
💡 סֻכָּה יְשָׁנָה כְּשֵׁרָה מִן הַדִּין, אֲבָל הִיא לֹא נִבְנְתָה לְשֵׁם הַחַג שֶׁל הַשָּׁנָה. לָכֵן לְכַתְּחִלָּה מְחַדְּשִׁים בָּהּ דָּבָר בַּסְּכָךְ – טֶפַח עַל טֶפַח, אוֹ קָנֶה דַּק לְכָל אָרְכָּהּ.
נֹעַם מִהֵר: קֹדֶם הֵנִיחַ סְכָךְ עַל עַמּוּדִים, וְרַק אַחַר כָּךְ בָּנָה אֶת הַדְּפָנוֹת. מָה הַדִּין?
💡 ״תַּעֲשֶׂה – וְלֹא מִן הֶעָשׂוּי״: הַסֻּכָּה צְרִיכָה לְהֵעָשׂוֹת עַל יְדֵי הֲנָחַת הַסְּכָךְ. מִי שֶׁהֵנִיחַ קֹדֶם סְכָךְ – הַסֻּכָּה נַעֲשְׂתָה כְּשֶׁבָּנָה אֶת הַדְּפָנוֹת, וּלְדַעַת פּוֹסְקִים רַבִּים הִיא פְּסוּלָה.

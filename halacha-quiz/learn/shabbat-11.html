בּוֹרֵר

🥜שְׁלֹשָׁה תְּנָאִים בְּבוֹרֵר

בְּשַׁבָּת אָסוּר לִבְרֹר כְּדֶרֶךְ מְלָאכָה, אֲבָל מֻתָּר לִבְרֹר כְּדֶרֶךְ אֲכִילָה. כְּדֵי שֶׁזֶּה יִהְיֶה דֶּרֶךְ אֲכִילָה צְרִיכִים שְׁלֹשָׁה תְּנָאִים: לוֹקְחִים אֶת הָאֹכֶל מִתּוֹךְ הַפְּסֹלֶת וְלֹא אֶת הַפְּסֹלֶת; בּוֹרְרִים בַּיָּד וְלֹא בִּכְלִי שֶׁנּוֹעַד לְבֵרוּר; וְעוֹשִׂים זֹאת כְּדֵי לֶאֱכֹל עַכְשָׁו – לְאַלְתַּר.
בּוֹטְנִים בְּתוֹךְ קְלִפּוֹת: לוֹקְחִים אֶת הַבּוֹטְנִים וְאוֹכְלִים – מֻתָּר. מוֹצִיאִים אֶת הַקְּלִפּוֹת מִן הַקְּעָרָה – אִסּוּר תּוֹרָה.
💛 אַהֲבַת הַשֵּׁם: הָעוֹלָם מְעֹרָב טוֹב וָרַע, וּבִימֵי הַחֹל אָנוּ עֲמֵלִים לְהַפְרִיד בֵּינֵיהֶם. בְּשַׁבָּת אָנוּ פָּשׁוּט לוֹקְחִים אֶת הַטּוֹב וְנֶהֱנִים מִמֶּנּוּ – כְּמוֹ יְלָדִים שֶׁאַבָּא אוֹהֵב מַגִּישׁ לָהֶם מַתָּנָה מוּכָנָה.
📖 פֶּרֶק י״א, הֲלָכוֹת א׳–ב׳

🌾מְבָרְרִים אֶת הַטּוֹב מִן הָרַע

אַרְבַּע מְלָאכוֹת מַפְרִידוֹת בֵּין אֹכֶל לִפְסֹלֶת: דָּשׁ – מוֹצִיא אֶת הַגַּרְגְּרִים מִן הַשִּׁבֳּלִים; זוֹרֶה – מַעִיף אֶת הַמּוֹץ בָּרוּחַ; בּוֹרֵר – מוֹצִיא בַּיָּד אֲבָנִים וְגוּשֵׁי עָפָר; מְרַקֵּד – מְנַפֶּה אֶת הַקֶּמַח בְּנָפָה. כְּדֶרֶךְ מְלָאכָה – אָסוּר מִן הַתּוֹרָה, וּכְדֶרֶךְ אֲכִילָה – מֻתָּר.
גַּם בַּנֶּפֶשׁ שֶׁלָּנוּ טוֹב וָרַע מְעֹרָבִים, וְתַפְקִידֵנוּ לִבְרֹר אֶת הַטּוֹב. בִּימֵי הַחֹל עוֹבְדִים קָשֶׁה בַּבֵּרוּר, וּבְשַׁבָּת מִתְעַנְּגִים עַל הַטּוֹב.
💛 אַהֲבַת הַשֵּׁם: הַכֹּחַ שֶׁאָנוּ אוֹסְפִים בְּשַׁבָּת – מִן הָאֱמוּנָה וּמִן הַקְּדֻשָּׁה – עוֹזֵר לָנוּ לְהַבְחִין כָּל הַשָּׁבוּעַ בֵּין טוֹב לְרַע, וְלִבְחֹר בַּטּוֹב בְּאַהֲבָה.
📖 פֶּרֶק י״א, הֲלָכָה א׳

🥗שְׁנֵי מִינִים מְעֹרָבִים

גַּם שְׁנֵי מַאֲכָלִים טוֹבִים שֶׁהִתְעָרְבוּ – אָסוּר לְמַיֵּן אוֹתָם. מִי שֶׁרוֹצֶה רַק אֱגוֹזִים מִתּוֹךְ תַּעֲרֹבֶת שֶׁל אֱגוֹזִים וּשְׁקֵדִים – לוֹקֵחַ אֶת הָאֱגוֹזִים כְּדֵי לֶאֱכֹל עַכְשָׁו, וְלֹא מוֹצִיא אֶת הַשְּׁקֵדִים. כָּל מַאֲכָל שֶׁיֵּשׁ לוֹ טַעַם אַחֵר נֶחְשָׁב מִין אַחֵר.
אִסּוּר בּוֹרֵר קַיָּם רַק בְּתַעֲרֹבֶת. מַאֲכָלִים שֶׁמֻּנָּחִים זֶה לְיַד זֶה, וַחֲתִיכוֹת גְּדוֹלוֹת שֶׁל בָּשָׂר בְּתוֹךְ מָרָק – אֵינָם נֶחְשָׁבִים מְעֹרָבִים.
💛 אַהֲבַת הַשֵּׁם: גַּם בֵּין שְׁנֵי דְּבָרִים טוֹבִים צָרִיךְ לִבְחֹר נָכוֹן. הַבְּחִירָה הַנְּכוֹנָה בְּשַׁבָּת – לָקַחַת מָה שֶׁצָּרִיךְ עַכְשָׁו – מְלַמֶּדֶת אוֹתָנוּ לִחְיוֹת אֶת הָרֶגַע עִם ה׳.
📖 פֶּרֶק י״א, הֲלָכוֹת ג׳–ד׳

🥚לֹא מוֹצִיאִים פְּסֹלֶת

גַּם כְּשֶׁיֵּשׁ רַק מְעַט פְּסֹלֶת וְקַל לְהוֹצִיא אוֹתָהּ – אָסוּר לְהוֹצִיא אוֹתָהּ לְבַדָּהּ מִתּוֹךְ הָאֹכֶל. קְלִפַּת בֵּיצָה שֶׁנָּפְלָה לְסָלַט בֵּיצִים – מוֹצִיאִים יַחַד עִם מְעַט בֵּיצָה. וּמִי שֶׁאֵינוֹ אוֹהֵב בָּצָל שֶׁבַּסָּלָט – אוֹכֵל אֶת מָה שֶׁהוּא אוֹהֵב, וּמַשְׁאִיר אֶת הַבָּצָל בַּצַּלַּחַת.
חֶרֶק שֶׁנָּפַל לְכוֹס – מוֹצִיאִים אוֹתוֹ בְּכַף יַחַד עִם מְעַט מַשְׁקֶה. פֵּרוֹת מְלֻכְלָכִים מֻתָּר לִשְׁטֹף כְּדֵי לֶאֱכֹל עַכְשָׁו, אֲבָל לֹא לְהַשְׁרוֹת אוֹתָם בְּמַיִם כְּדֵי שֶׁהֶעָפָר יִשְׁקַע.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁמַּשְׁאִירִים אֶת הַפְּסֹלֶת בַּצַּד וְלוֹקְחִים רַק אֶת הַטּוֹב, הַלֵּב לוֹמֵד לְהִתְמַקֵּד בַּטּוֹב. זוֹ הַשַּׁבָּת: יוֹם שֶׁבּוֹ רוֹאִים אֶת הַטּוֹב שֶׁה׳ נָתַן, וּשְׂמֵחִים בּוֹ.
📖 פֶּרֶק י״א, הֲלָכוֹת ה׳, ט״ו

⏰לְאַלְתַּר – סָמוּךְ לַסְּעוּדָה

מֻתָּר לִבְרֹר אֹכֶל לִפְנֵי הַסְּעוּדָה בִּשְׁבִיל כָּל בְּנֵי הַבַּיִת וְהָאוֹרְחִים, כְּמוֹ שֶׁרְגִילִים לְהָכִין לִפְנֵי סְעוּדָה. גַּם אִם הַסְּעוּדָה תִּמָּשֵׁךְ שָׁלֹשׁ שָׁעוֹת, וְהַמַּאֲכָל יֻגַּשׁ רַק בְּסוֹפָהּ – זוֹ דֶּרֶךְ אֲכִילָה. אֲבָל לִבְרֹר שָׁעוֹת רַבּוֹת לִפְנֵי כֵן – אָסוּר מִן הַתּוֹרָה.
מִי שֶׁבָּרַר לַסְּעוּדָה הַקְּרוֹבָה, וּבַסּוֹף נִשְׁאַר לוֹ גַּם לַסְּעוּדָה הַבָּאָה – לֹא עָבַר עַל אִסּוּר, וּבִלְבַד שֶׁלֹּא עָשָׂה זֹאת בְּכַוָּנָה.
💛 אַהֲבַת הַשֵּׁם: הַהֲכָנָה לַסְּעוּדָה הִיא חֵלֶק מִכְּבוֹד הַשַּׁבָּת. כְּשֶׁמְּכִינִים בְּשִׂמְחָה לְכָל הַמִּשְׁפָּחָה וְלָאוֹרְחִים, הַשֻּׁלְחָן שֶׁל שַׁבָּת נַעֲשֶׂה מָקוֹם שֶׁל אַהֲבָה וְחֶסֶד.
📖 פֶּרֶק י״א, הֲלָכָה ו׳

🥄בַּיָּד, בְּכַף וּבְמַזְלֵג

כַּף וּמַזְלֵג אֵינָם כְּלֵי בֵּרוּר, אֶלָּא עוֹזְרִים לַיָּד – וּמֻתָּר לִבְרֹר בָּהֶם כְּדֶרֶךְ אֲכִילָה. אֲבָל מְסַנֶּנֶת, רֶשֶׁת אוֹ מַכְשִׁיר לְהוֹצָאַת גַּלְעִינִים – אֲסוּרִים. לָכֵן אֶת הַשֶּׁמֶן שֶׁבְּקֻפְסַת טוּנָה לֹא שׁוֹפְכִים, אֶלָּא מוֹצִיאִים אֶת הַטּוּנָה בְּכַף לְצַלַּחַת אַחֶרֶת. מְלָפְפוֹנִים חֲמוּצִים גְּדוֹלִים – מֻתָּר לִשְׁפֹּךְ מֵהֶם אֶת הַמַּיִם.
מִלְחִיָּה עִם גַּרְגְּרֵי אֹרֶז – מֻתָּר לְהִשְׁתַּמֵּשׁ בָּהּ, כִּי הִיא נוֹעֲדָה לְפַזֵּר מֶלַח וְלֹא לִבְרֹר. וְאֶת שַׂקִּית הַתֵּה מוֹצִיאִים מִן הַכּוֹס וּמַנִּיחִים מִיָּד בַּצַּד, בְּלִי לְהַחְזִיק אוֹתָהּ מֵעַל הַכּוֹס כְּדֵי שֶׁתְּטַפְטֵף.
💛 אַהֲבַת הַשֵּׁם: הַהֶבְדֵּל בֵּין כַּף לִמְסַנֶּנֶת נִרְאֶה קָטָן, אֲבָל הוּא הַהֶבְדֵּל בֵּין אֲכִילָה לִמְלָאכָה. מִי שֶׁשָּׂם לֵב לְהֶבְדֵּלִים כָּאֵלֶּה מְכַבֵּד אֶת הַשַּׁבָּת – וּמְכַבֵּד אֶת ה׳ שֶׁנָּתַן אוֹתָהּ.
📖 פֶּרֶק י״א, הֲלָכוֹת ז׳, י״ג–י״ד

🍊קְלִפּוֹת, גַּרְעִינִים וַעֲצָמוֹת

מֻתָּר לְקַלֵּף פְּרִי כְּדֵי לֶאֱכֹל אוֹתוֹ עַכְשָׁו – גַּם בְּסַכִּין, כִּי כָּךְ אוֹכְלִים. מֻתָּר לְפַצֵּחַ אֱגוֹזִים וּלְקַלֵּף בּוֹטְנִים, וְלִזְרֹק גַּרְעִין שֶׁל שָׁזִיף אוֹ תָּמָר. הָאוֹכֵל דָּג – מוֹצִיא אֶת הָעֲצָמוֹת תּוֹךְ כְּדֵי אֲכִילָה, וּמֻתָּר לְהָסִיר גַּרְעִינֵי אֲבַטִּיחַ סָמוּךְ לָאֲכִילָה.
אֲבָל לְקַלֵּף פֵּרוֹת כְּדֵי לֶאֱכֹל אוֹתָם רַק אַחֲרֵי זְמַן רַב – אָסוּר, כִּי זוֹ כְּבָר דֶּרֶךְ מְלָאכָה.
💛 אַהֲבַת הַשֵּׁם: ה׳ עָטַף אֶת הַפֵּרוֹת בִּקְלִפָּה שֶׁשּׁוֹמֶרֶת עֲלֵיהֶם עַד שֶׁנֹּאכַל אוֹתָם. כְּשֶׁאָנוּ מְקַלְּפִים פְּרִי וּמְבָרְכִים עָלָיו, אָנוּ פּוֹתְחִים מַתָּנָה שֶׁה׳ הֵכִין בִּשְׁבִילֵנוּ.
📖 פֶּרֶק י״א, הֲלָכוֹת ח׳–י׳, י״ח

🔀מַחְלוֹקוֹת בַּהֲכָנַת הָאֹכֶל – וְכָל אֶחָד כְּמִנְהָגוֹ

קַצֶּפֶת מִבֵּיצָה אוֹ מִשַּׁמֶּנֶת מְתוּקָה: לְדַעַת רַבִּים אָסוּר לְהַקְצִיף בְּשַׁבָּת, כִּי זֶה נִרְאֶה כַּהֲכָנָה לְבִשּׁוּל; וְיֵשׁ מְקִלִּים. קְלִיפַת תַּפּוּחַ, מְלָפְפוֹן אוֹ גֶּזֶר בְּמַקְלֵף: יֵשׁ אוֹמְרִים שֶׁמֻּתָּר, כִּי גַּם הַקְּלִפָּה רְאוּיָה לַאֲכִילָה; וְיֵשׁ אוֹמְרִים שֶׁהַמַּקְלֵף הוּא כְּלִי בֵּרוּר, וּמְקַלְּפִים רַק בְּסַכִּין, סָמוּךְ לָאֲכִילָה.
צְבִיעַת מַשְׁקִים: יֵשׁ מַחְמִירִים, וּלְדַעַת רֹב הַפּוֹסְקִים מֻתָּר – וּלְכַתְּחִלָּה שָׂמִים קֹדֶם אֶת הַתַּרְכִּיז. מְעִיכַת בָּנָנָה: יֵשׁ מַחְמִירִים וּמוֹעֲכִים רַק בְּכַף, וְהַהֲלָכָה כַּמַּתִּירִים לִמְעֹךְ גַּם בְּמַזְלֵג, סָמוּךְ לָאֲכִילָה.
💛 אַהֲבַת הַשֵּׁם: הַמַּחְלוֹקוֹת מְלַמְּדוֹת אוֹתָנוּ כַּמָּה חֲכָמֵינוּ דִּקְדְּקוּ בִּשְׁמִירַת הַשַּׁבָּת. כְּשֶׁאָנוּ שׁוֹמְרִים עַל מִנְהַג הַבַּיִת וּמְכַבְּדִים אֶת מִנְהַג הֶחָבֵר – הַשַּׁבָּת שֶׁלָּנוּ מְלֵאָה שָׁלוֹם.
📖 פֶּרֶק י״א, הֲלָכָה ח׳; פֶּרֶק י״ב, הֲלָכוֹת ב׳, י׳, י״ב

🍊מְסַנְּנִים מַשְׁקֶה בְּשַׁבָּת

מַשְׁקֶה שֶׁאֶפְשָׁר לִשְׁתּוֹת גַּם בְּלִי סִנּוּן – מֻתָּר לְסַנֵּן. לְמָשָׁל מִיץ תַּפּוּזִים עִם חֲתִיכוֹת פְּרִי, אוֹ מַיִם מִבֶּרֶז שֶׁיֵּשׁ עָלָיו מְסַנֶּנֶת.
מַשְׁקֶה שֶׁאִי אֶפְשָׁר לִשְׁתּוֹת בְּלִי סִנּוּן – אָסוּר לְסַנֵּן, כִּי זוֹ מְלֶאכֶת בּוֹרֵר.
💛 אַהֲבַת הַשֵּׁם: גַּם בַּפְּרָטִים הַקְּטַנִּים – כּוֹס מִיץ – אֲנַחְנוּ שׁוֹמְרִים עַל הַשַּׁבָּת, כִּי הִיא מַתָּנָה שֶׁל אַהֲבָה מֵה׳.
📖 פֶּרֶק י״א, הֲלָכָה י״א

🧸בּוֹרֵר בְּמִשְׂחָקִים וּבְכֵלִים

דִּינֵי בּוֹרֵר שַׁיָּכִים גַּם לְמִשְׂחָקִים, לְכֵלִים וְלִסְפָרִים. שְׁנֵי מִשְׂחָקֵי קֻבִּיּוֹת שֶׁהִתְעָרְבוּ – אָסוּר לְמַיֵּן, אֲבָל מֻתָּר לָקַחַת מִתּוֹכָם אֶת הַסּוּג שֶׁרוֹצִים לְשַׂחֵק בּוֹ עַכְשָׁו. סַכּוּ״ם מְעֹרָב – אָסוּר לְמַיֵּן, אֲבָל לִפְנֵי הַסְּעוּדָה מֻתָּר לְהַנִּיחַ לְיַד כָּל צַלַּחַת אֶת הַכֵּלִים הַמַּתְאִימִים.
סְפָרִים שֶׁעוֹמְדִים בַּמַּדָּף אֵינָם מְעֹרָבִים. וּמִי שֶׁלָּמַד בְּהַרְבֵּה סְפָרִים – מֻתָּר לוֹ לְהַחְזִיר כָּל סֵפֶר לִמְקוֹמוֹ בָּאָרוֹן.
💛 אַהֲבַת הַשֵּׁם: גַּם הַמִּשְׂחָק בְּשַׁבָּת יָכוֹל לִהְיוֹת חֵלֶק מִשְּׁמִירַת הַשַּׁבָּת. יֶלֶד שֶׁמְּשַׂחֵק וְזוֹכֵר אֶת הַהֲלָכָה – מְשַׂמֵּחַ אֶת ה׳ גַּם בִּשְׁעַת הַמִּשְׂחָק.
📖 פֶּרֶק י״א, הֲלָכָה ט״ז

🌰דָּשׁ וּמְפָרֵק

מְלֶאכֶת דָּשׁ מוֹצִיאָה אֶת הָאֹכֶל מִבֵּית הַגִּדּוּל שֶׁלּוֹ, כְּמוֹ גַּרְגְּרֵי חִטָּה מִן הַשִּׁבֳּלִים, אוֹ אֲפוּנָה וּשְׁעוּעִית מִתַּרְמִילִים יְבֵשִׁים. זוֹ מְלָאכָה שֶׁל שָׂדֶה וְשֶׁל מִפְעָל. אֲבָל אֱגוֹזִים וּבוֹטְנִים, שֶׁרְגִילִים לְפַצֵּחַ רַק בִּשְׁעַת הָאֲכִילָה – מֻתָּר לְפַצֵּחַ.
שְׁקֵדִים עִם קְלִפָּה קָשָׁה – מֻתָּר לְקַלֵּף. אֲבָל אִם נִשְׁאֲרָה עֲלֵיהֶם הַקְּלִפָּה הַיְּרֻקָּה – מְקַלְּפִים שָׁקֵד אֶחָד וְאוֹכְלִים אוֹתוֹ, וְאַחַר כָּךְ עוֹד אֶחָד.
💛 אַהֲבַת הַשֵּׁם: הַשָּׂדֶה וְהַמִּפְעָל שַׁיָּכִים לִימֵי הַחֹל, וְהַשֻּׁלְחָן הֶעָרוּךְ שַׁיָּךְ לְשַׁבָּת. כְּשֶׁאָנוּ עוֹזְבִים אֶת עֲבוֹדַת הַשָּׂדֶה, אָנוּ פְּנוּיִים לְהוֹדוֹת לַה׳ עַל הַיְּבוּל שֶׁהוּא נָתַן.
📖 פֶּרֶק י״א, הֲלָכוֹת י״ז–י״ח

🧠 בְּדֹק אֶת עַצְמְךָ

לִפְנֵי סְעוּדַת לֵיל שַׁבָּת אַבָּא מוֹצִיא שְׁקֵדִים מִתּוֹךְ תַּעֲרֹבֶת שֶׁל שְׁקֵדִים וּבוֹטְנִים, כְּדֵי לְהַגִּישׁ אוֹתָם בְּסוֹף הַסְּעוּדָה. מָה הַדִּין?
💡 מֻתָּר לְהָכִין אֶת כָּל הַמַּאֲכָלִים סָמוּךְ לִתְחִלַּת הַסְּעוּדָה, גַּם אִם יֻגְּשׁוּ בְּסוֹפָהּ – וּבִשְׁבִיל כָּל הַמְּסֻבִּים. אֲבָל אֶת הַבּוֹטְנִים, שֶׁהֵם הַפְּסֹלֶת כָּאן, אָסוּר לְהוֹצִיא.
נֹעַם פָּתַח קֻפְסַת טוּנָה בְּשֶׁמֶן, וְרוֹצֶה לְהַגִּישׁ לַסְּעוּדָה טוּנָה בְּלִי שֶׁמֶן. מָה יַעֲשֶׂה?
💡 הַטּוּנָה קְטַנָּה וּמְעֹרֶבֶת בַּשֶּׁמֶן. שְׁפִיכַת הַשֶּׁמֶן הִיא הוֹצָאַת פְּסֹלֶת, וּמְסַנֶּנֶת הִיא כְּלִי בֵּרוּר. כְּשֶׁלּוֹקְחִים אֶת הַטּוּנָה בְּכַף – לוֹקְחִים אֹכֶל מִתּוֹךְ פְּסֹלֶת.
לָמָּה מֻתָּר לְפַצֵּחַ אֱגוֹזִים בְּשַׁבָּת, וְזֶה אֵינוֹ נֶחְשָׁב מְלֶאכֶת דָּשׁ?
💡 דָּשׁ הִיא מְלָאכָה שֶׁל שָׂדֶה אוֹ מִפְעָל, כְּשֶׁמּוֹצִיאִים הַרְבֵּה אֹכֶל מִקְּלִפָּתוֹ לִקְרַאת מְכִירָה. אֱגוֹזִים מְפַצְּחִים רַק לִקְרַאת הָאֲכִילָה – וְזוֹ דֶּרֶךְ אֲכִילָה. אֲבָל אֲפוּנָה מִתַּרְמִילִים יְבֵשִׁים – זֶה כְּבָר דָּשׁ.

הִלְכוֹת צִיצִית

🧵 צִיצִית, תְּפִלִּין וּמְזוּזָה · פֶּרֶק ח׳

🧵הַבֶּגֶד שֶׁל עַם יִשְׂרָאֵל

לְכָל עַם יֵשׁ בֶּגֶד מְיֻחָד מִשֶּׁלּוֹ. הַבֶּגֶד שֶׁלָּנוּ הוּא הַטַּלִּית עִם אַרְבַּע הַכְּנָפוֹת וְהַצִּיצִיּוֹת – וְאוֹתוֹ בָּחַר בִּשְׁבִילֵנוּ ה׳. הַתּוֹרָה אָמְרָה: "עַל אַרְבַּע כַּנְפוֹת כְּסוּתְךָ" – לָכֵן רַק בֶּגֶד עִם אַרְבַּע פִּנּוֹת חַיָּב בְּצִיצִית. בֶּגֶד עָגֹל, אוֹ בֶּגֶד עִם שָׁלֹשׁ פִּנּוֹת – פָּטוּר.
לְחֻלְצָה, לְמִכְנָסַיִם וְלִמְעִיל אֵין אַרְבַּע כְּנָפוֹת. כְּדֵי לִזְכּוֹת בַּמִּצְוָה, קוֹנִים בֶּגֶד מְרֻבָּע מְיֻחָד – טַלִּית קָטָן – וְקוֹשְׁרִים בּוֹ צִיצִיּוֹת.
💛 אַהֲבַת הַשֵּׁם: הַכִּפָּה הִיא מִנְהָג, אֲבָל הַצִּיצִית הִיא מִצְוָה מְפֹרֶשֶׁת מִן הַתּוֹרָה. כְּשֶׁאַתָּה לוֹבֵשׁ צִיצִית, אַתָּה לָבוּשׁ בַּבֶּגֶד שֶׁהַמֶּלֶךְ בְּעַצְמוֹ בָּחַר לְעַמּוֹ – וְכֻלָּם יוֹדְעִים לְמִי אַתָּה שַׁיָּךְ.
📖 פֶּרֶק ח׳, הֲלָכוֹת א׳–ב׳

🌟צִיצִית לִפְנֵי כִּפָּה

יְהוּדִי אֶחָד, שֶׁרָצָה לְהִתְקָרֵב לַתּוֹרָה, אָמַר לְהָרַב צְבִי יְהוּדָה קוּק שֶׁקָּשֶׁה לוֹ לַחְבֹּשׁ כִּפָּה. הָרַב עָנָה לוֹ: לַכִּפָּה אֵין חֲשִׁיבוּת גְּדוֹלָה כָּל כָּךְ, אֲבָל לַצִּיצִית יֵשׁ חֲשִׁיבוּת מְרֻבָּה – כִּי הִיא מִצְוָה מִן הַתּוֹרָה.
הָאִישׁ הִסְתּוֹבֵב בִּירוּשָׁלַיִם עִם צִיצִיּוֹת בַּחוּץ וּבְלִי כִּפָּה – וְאַחַר כָּךְ הִתְקַדֵּם, וְגַם כִּסָּה אֶת רֹאשׁוֹ בְּכִפָּה.
💛 אַהֲבַת הַשֵּׁם: הָרַב יָדַע לְהַתְחִיל מִמָּה שֶׁהֲכִי חָשׁוּב. כְּשֶׁאָדָם מִתְחַבֵּר בְּאַהֲבָה לְמִצְוָה אַחַת מִן הַתּוֹרָה, הַלֵּב שֶׁלּוֹ נִפְתָּח – וְהוּא מַמְשִׁיךְ לְהִתְקָרֵב צַעַד אַחַר צַעַד.
📖 פֶּרֶק ח׳, הֲלָכָה א׳

🧵📜 לְשׁוֹן חֲכָמִים: הַבֶּגֶד הַלְּאֻמִּי

״הַבֶּגֶד הַלְּאֻמִּי שֶׁלָּנוּ הוּא הַטַּלִּית בַּעֲלַת אַרְבַּע הַכְּנָפוֹת וְהַצִּיצִיּוֹת... הַבֶּגֶד שֶׁלָּנוּ נִקְבַּע עַל יְדֵי הַקָּדוֹשׁ בָּרוּךְ הוּא״.
📖 לְאֻמִּי – שֶׁל הָעָם כֻּלּוֹ, הַמְּבַטֵּא אֶת זֶהוּתוֹ.
💛 אַהֲבַת הַשֵּׁם: 💛 כָּל עַם בָּחַר לְעַצְמוֹ בֶּגֶד מְיֻחָד, אֲבָל אֶת הַבֶּגֶד שֶׁלָּנוּ בָּחַר ה׳ בְּעַצְמוֹ – סִימָן שֶׁהוּא רוֹצֶה קֶשֶׁר אִתָּנוּ מִקָּרוֹב. 🧒 וְאַתָּה? לְבַשׁ טַלִּית קָטָן בְּגַאֲוָה, כְּמוֹ בֶּגֶד שֶׁה׳ בְּעַצְמוֹ בָּחַר בִּשְׁבִילְךָ.
📖 פֶּרֶק ח׳, הֲלָכָה א׳

✋חָמֵשׁ כְּנָפוֹת וְרַב קְטִינָא

בֶּגֶד עִם חָמֵשׁ כְּנָפוֹת חַיָּב בְּצִיצִית, כִּי בְּתוֹךְ הֶחָמֵשׁ יֵשׁ אַרְבַּע, וְהַכָּנָף הַנּוֹסֶפֶת אֵינָהּ גּוֹרַעַת. קוֹשְׁרִים אֶת הַצִּיצִיּוֹת בְּאַרְבַּע הַכְּנָפוֹת הָרְחוֹקוֹת זוֹ מִזּוֹ.
רַב קְטִינָא לָבַשׁ רַק בְּגָדִים בְּלִי אַרְבַּע כְּנָפוֹת. מַלְאָךְ אָמַר לוֹ: מִי שֶׁאֵין לוֹ בֶּגֶד כָּזֶה אָמְנָם אֵינוֹ עוֹבֵר עֲבֵרָה, אֲבָל בְּעֵת צָרָה – זְכוּת הַמִּצְוָה מַצִּילָה אֶת מִי שֶׁהִשְׁתַּדֵּל לְהִתְחַיֵּב בָּהּ.
💛 אַהֲבַת הַשֵּׁם: יֵשׁ מִי שֶׁעוֹשֶׂה רַק מָה שֶׁחַיָּבִים, וְיֵשׁ מִי שֶׁמְּחַפֵּשׂ אֵיךְ לִשְׂמֹחַ אֶת ה׳ עוֹד. מִי שֶׁלּוֹבֵשׁ צִיצִית בְּרָצוֹן, מְבַקֵּשׁ לִהְיוֹת קָרוֹב לַה׳ – וְהַקִּרְבָה הַזֹּאת שׁוֹמֶרֶת עָלָיו.
📖 פֶּרֶק ח׳, הֲלָכָה ב׳

🧵🔑 מוּשָּׂג: אַרְבַּע כְּנָפוֹת

״עַל אַרְבַּע כַּנְפוֹת כְּסוּתְךָ אֲשֶׁר תְּכַסֶּה בָּהּ״ (דְּבָרִים כ״ב, י״ב). ...בֶּגֶד עָגֹל שֶׁאֵין בּוֹ כְּנָפוֹת כְּלָל, אוֹ בֶּגֶד מְשֻׁלָּשׁ שֶׁיֵּשׁ בּוֹ שָׁלֹשׁ כְּנָפוֹת בִּלְבַד – פָּטוּר מִן הַצִּיצִית.
📖 כָּנָף – פִּנַּת הַבֶּגֶד. דַּוְקָא בֶּגֶד מְרֻבָּע, עִם אַרְבַּע פִּנּוֹת, חַיָּב בְּצִיצִית.
💛 אַהֲבַת הַשֵּׁם: 💛 ה׳ בָּחַר בִּשְׁבִילֵנוּ בֶּגֶד מְיֻחָד, כְּמוֹ שֶׁיֵּשׁ לְכָל עַם בֶּגֶד לְאֻמִּי מִשֶּׁלּוֹ. 🧒 וְאַתָּה? כְּשֶׁאַתָּה בּוֹחֵר טַלִּית קָטָן – שִׂים לֵב שֶׁיֵּשׁ לָהּ בֶּאֱמֶת אַרְבַּע כְּנָפוֹת.
📖 פֶּרֶק ח׳, הֲלָכָה ב׳

👁️רוֹאִים וְזוֹכְרִים

בְּפָרָשַׁת צִיצִית כָּתוּב: "וּרְאִיתֶם אֹתוֹ וּזְכַרְתֶּם אֶת כָּל מִצְוֹת ה׳". כְּשֶׁמִּתְעַטְּפִים בַּצִּיצִית מְכַוְּנִים שְׁתֵּי כַּוָּנוֹת: שֶׁאֲנַחְנוּ מְקַיְּמִים אֶת מִצְוַת ה׳, וְשֶׁהַצִּיצִית תַּזְכִּיר לָנוּ אֶת כָּל הַמִּצְווֹת.
הַמִּצְוָה הַזֹּאת נִמְצֵאת עַל הַבֶּגֶד שֶׁלְּךָ, וְלָכֵן רוֹאִים אוֹתָהּ כָּל הַיּוֹם – וְנִזְכָּרִים שׁוּב וָשׁוּב בְּמִי שֶׁצִּוָּה עָלֶיהָ.
💛 אַהֲבַת הַשֵּׁם: כְּמוֹ קֶשֶׁר עַל הָאֶצְבַּע שֶׁמַּזְכִּיר דָּבָר חָשׁוּב, הַצִּיצִית מַזְכִּירָה לְךָ אֶת ה׳ בְּאֶמְצַע הַמִּשְׂחָק, הַלִּמּוּד וְהָאֹכֶל. כָּךְ כָּל הַיּוֹם שֶׁלְּךָ נַעֲשֶׂה יוֹם שֶׁל קֶשֶׁר עִם ה׳.
📖 פֶּרֶק ח׳, הֲלָכוֹת ג׳–ד׳

🔢כָּל הַמִּצְווֹת בְּתוֹךְ הַצִּיצִית

רַשִׁ״י מְגַלֶּה רֶמֶז: הַמִּלָּה צִיצִית בְּגִימַטְרִיָּה הִיא שֵׁשׁ מֵאוֹת. מוֹסִיפִים שְׁמוֹנָה חוּטִים וַחֲמִשָּׁה קְשָׁרִים – וּמַגִּיעִים לְשֵׁשׁ מֵאוֹת וּשְׁלוֹשׁ עֶשְׂרֵה, כְּמִסְפַּר כָּל מִצְווֹת הַתּוֹרָה.
כָּךְ מִתְקַיֵּם הַפָּסוּק: "וּרְאִיתֶם אֹתוֹ וּזְכַרְתֶּם אֶת כָּל מִצְוֹת ה׳ וַעֲשִׂיתֶם אֹתָם".
💛 אַהֲבַת הַשֵּׁם: בְּתוֹךְ כַּמָּה חוּטִים לְבָנִים מִסְתַּתֶּרֶת כָּל הַתּוֹרָה כֻּלָּהּ. ה׳ נָתַן לָנוּ מַתָּנָה קְטַנָּה שֶׁאֶפְשָׁר לָקַחַת לְכָל מָקוֹם – וְהִיא מַזְכִּירָה אֶת כָּל הָאַהֲבָה הַגְּדוֹלָה שֶׁלּוֹ.
📖 פֶּרֶק ח׳, הֲלָכָה ג׳

🎯🔑 מוּשָּׂג: כַּוָּנָה

״יְכַוֵּן בְּהִתְעַטְּפוֹ, שֶׁצִּוָּנוּ הַקָּדוֹשׁ בָּרוּךְ הוּא לְהִתְעַטֵּף בּוֹ כְּדֵי שֶׁנִּזְכֹּר כָּל מִצְוֹתָיו לַעֲשׂוֹתָם״.
📖 כַּוָּנָה – לַחְשֹׁב בְּלֵב שֶׁמְּקַיְּמִים אֶת מִצְוַת ה׳ בִּשְׁעַת מַעֲשֵׂה הַמִּצְוָה.
💛 אַהֲבַת הַשֵּׁם: 💛 בְּצִיצִית וּבִתְפִלִּין הוֹסִיפָה הַתּוֹרָה כַּוָּנָה מְיֻחֶדֶת: לִזְכֹּר דֶּרֶךְ הַצִּיצִית אֶת כָּל מִצְוֹת ה׳. 🧒 וְאַתָּה? כְּשֶׁתִּתְעַטֵּף בְּטַלִּית קָטָן – חֲשֹׁב רֶגַע שֶׁה׳ צִוָּה עָלֶיהָ.
📖 פֶּרֶק ח׳, הֲלָכָה ג׳

🧥צִיצִיּוֹת בַּחוּץ אוֹ בִּפְנִים?

בַּשֻּׁלְחָן עָרוּךְ כָּתוּב שֶׁעִקַּר הַמִּצְוָה לִלְבֹּשׁ אֶת הַטַּלִּית הַקָּטָן כָּךְ שֶׁיִּרְאוּ אוֹתָהּ תָּמִיד. עַל פִּי הַקַּבָּלָה, הַבֶּגֶד עַצְמוֹ נִלְבָּשׁ מִתַּחַת לַחֻלְצָה. הַמָּגֵן אַבְרָהָם וְהַמִּשְׁנָה בְּרוּרָה כָּתְבוּ שֶׁאֶת הַחוּטִים מוֹצִיאִים הַחוּצָה, כְּדֵי שֶׁיֵּרָאוּ. הַסְּפָרַדִּים נָהֲגוּ לְהַשְׁאִיר גַּם אֶת הַצִּיצִיּוֹת בִּפְנִים.
בְּכָל הַמִּנְהָגִים הַמַּטָּרָה אַחַת: שֶׁהַצִּיצִית תַּזְכִּיר לָנוּ אֶת ה׳ וְאֶת כָּל מִצְווֹתָיו.
💛 אַהֲבַת הַשֵּׁם: הַהִסְתַּכְּלוּת עַל הַצִּיצִית הִיא כְּמוֹ מַבָּט בִּתְמוּנָה שֶׁל מִישֶׁהוּ אָהוּב – הִיא מְעוֹרֶרֶת אֶת הַלֵּב. כָּל מַבָּט כָּזֶה מַחְזִיר אוֹתְךָ אֶל ה׳ וְאֶל הַדֶּרֶךְ הַטּוֹבָה.

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

הַשֻּׁלְחָן עָרוּךְ כָּתַב שֶׁעִקַּר הַמִּצְוָה לִלְבֹּשׁ אֶת הַטַּלִּית הַקָּטָן עַל הַבְּגָדִים, כְּדֵי לִרְאוֹת אוֹתוֹ תָּמִיד. עַל פִּי הַקַּבָּלָה, מְקוֹם הַטַּלִּית הַקָּטָן מִתַּחַת לַבְּגָדִים. הַסְּפָרַדִּים נָהֲגוּ לְהַשְׁאִיר בִּפְנִים גַּם אֶת הַצִּיצִיּוֹת. הָאַשְׁכְּנַזִּים נוֹהֲגִים כְּמוֹ שֶׁכָּתְבוּ הַמָּגֵן אַבְרָהָם וְהַמִּשְׁנָה בְּרוּרָה: הַבֶּגֶד מִתַּחַת לַחֻלְצָה, וְאֶת הַחוּטִים מוֹצִיאִים הַחוּצָה – ״וּרְאִיתֶם אֹתוֹ״.
לְמִנְהַג הָאַשְׁכְּנַזִּים – הַצִּיצִיּוֹת בַּחוּץ. לְמִנְהַג הַסְּפָרַדִּים – הַצִּיצִיּוֹת בִּפְנִים.
💛 אַהֲבַת הַשֵּׁם: שְׁנֵי הַמִּנְהָגִים כַּהֲלָכָה, וְכָל אֶחָד נוֹהֵג כְּמִנְהַג אֲבוֹתָיו. וּבִשְׁנֵיהֶם הַצִּיצִית צְמוּדָה אֵלֵינוּ כָּל הַיּוֹם, וּמַזְכִּירָה לָנוּ אֶת ה׳ וְאֶת כָּל מִצְווֹתָיו.

🌍הַטַּלִּית וְהָעוֹלָם

הַטַּלִּית הַמְרֻבַּעַת רוֹמֶזֶת לָעוֹלָם הַגַּשְׁמִי עַל אַרְבַּע רוּחוֹתָיו, עִם כָּל הַכֹּחוֹת וְהָרְצוֹנוֹת הַחֲזָקִים שֶׁבּוֹ. חוּטֵי הַצִּיצִית רוֹמְזִים לְמִצְווֹת הַתּוֹרָה, וּבָהֶם אוֹחֲזִים אֶת הַטַּלִּית.
כְּמוֹ שֶׁתּוֹפְסִים אֶת הַטַּלִּית בַּצִּיצִיּוֹת, כָּךְ הַמִּצְווֹת מְלַמְּדוֹת אוֹתָנוּ לְהִשְׁתַּמֵּשׁ בְּכָל הַכֹּחוֹת שֶׁלָּנוּ בְּדֶרֶךְ טוֹבָה – בְּלִי לְאַבֵּד אוֹתָם, וּבְלִי שֶׁהֵם יִסְחֲפוּ אוֹתָנוּ.
💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לְךָ כֹּחוֹת גְּדוֹלִים: לָרוּץ, לִרְצוֹת, לֶאֱהֹב וּלְהַצְלִיחַ. הַמִּצְווֹת הֵן הַהֶגֶה שֶׁלּוֹ, שֶׁמְּכַוֵּן אֶת כָּל הַכֹּחוֹת הָאֵלֶּה לְטוֹבָתְךָ וּלְטוֹבַת הָעוֹלָם.

🧶מֵאֵיזֶה בַּד?

לְכָל הַדֵּעוֹת קוֹשְׁרִים צִיצִית בְּכָל בֶּגֶד מְרֻבָּע, מִכָּל סוּג שֶׁל בַּד. רַבִּים מְהַדְּרִים בְּטַלִּית שֶׁל צֶמֶר, כְּדֵי לְקַיֵּם מִצְוָה מִן הַתּוֹרָה לְכָל הַשִּׁיטוֹת. מִפִּשְׁתָּן לֹא עוֹשִׂים טַלִּית, וּבֶגֶד שֶׁל עוֹר פָּטוּר מִצִּיצִית.
עַל טַלִּית מִנַּיְלוֹן אוֹ מִפּוֹלִיאֶסְטֶר (כְּמוֹ צִיצִית רֶשֶׁת) יֵשׁ סָפֵק, וְלָכֵן לֹא מְבָרְכִים עָלֶיהָ. מִי שֶׁחַם לוֹ בַּקַּיִץ – יִבְחַר טַלִּית מִכֻּתְנָה, וִיבָרֵךְ.
💛 אַהֲבַת הַשֵּׁם: מִצְוָה שֶׁאֶפְשָׁר לְבָרֵךְ עָלֶיהָ הִיא הִזְדַּמְּנוּת לְהַגִּיד לַה׳ "קִדְּשָׁנוּ בְּמִצְוֹתָיו". לָכֵן כְּדַאי לִבְחֹר בֶּגֶד שֶׁמְּאַפְשֵׁר לָנוּ לְהוֹדוֹת לוֹ בְּכָל בֹּקֶר בְּלִי סָפֵק.

☀️צִיצִית – בַּיּוֹם

"וּרְאִיתֶם אֹתוֹ" – זְמַן הַצִּיצִית הוּא כְּשֶׁאֶפְשָׁר לִרְאוֹת, כְּלוֹמַר בַּיּוֹם. הָרִאשׁוֹנִים נֶחְלְקוּ: יֵשׁ שֶׁהוֹלְכִים לְפִי הַשָּׁעָה – יוֹם אוֹ לַיְלָה, וְיֵשׁ שֶׁהוֹלְכִים לְפִי סוּג הַבֶּגֶד – בֶּגֶד יוֹם אוֹ בֶּגֶד לַיְלָה. לְמַעֲשֶׂה חוֹשְׁשִׁים לִשְׁתֵּי הַדֵּעוֹת.
מִי שֶׁלּוֹבֵשׁ טַלִּית קָטָן בַּלַּיְלָה – הַצִּיצִיּוֹת צְרִיכוֹת לִהְיוֹת כְּשֵׁרוֹת, אֲבָל לֹא מְבָרְכִים עֲלֵיהֶן.
💛 אַהֲבַת הַשֵּׁם: הַבְּרָכָה עַל הַצִּיצִית בַּבֹּקֶר פּוֹתַחַת אֶת הַיּוֹם בְּמַחְשָׁבָה עַל ה׳. כְּשֶׁאַתָּה מִתְעַטֵּף בְּאוֹר הַיּוֹם, אַתָּה כְּאִלּוּ אוֹמֵר: כָּל מָה שֶׁאֶעֱשֶׂה הַיּוֹם – יִהְיֶה לְפָנֶיךָ.

🕰️נָשִׁים וּבָנוֹת וּמִצְוַת צִיצִית

נָשִׁים פְּטוּרוֹת מִמִּצְווֹת עֲשֵׂה שֶׁהַזְּמַן גְּרָמָן – מִצְווֹת שֶׁתְּלוּיוֹת בִּזְמַן, כְּמוֹ לוּלָב בְּסֻכּוֹת וְשׁוֹפָר בְּרֹאשׁ הַשָּׁנָה. מִצְוַת צִיצִית הִיא רַק בַּיּוֹם וְלֹא בַּלַּיְלָה, וְלָכֵן גַּם הִיא תְּלוּיָה בִּזְמַן – וְנָשִׁים פְּטוּרוֹת מִמֶּנָּה. נָשִׁים שֶׁרוֹצוֹת לְהִתְעַטֵּף בְּטַלִּית – רַשָּׁאִיּוֹת, בְּצִנְעָה.
בְּמִצְווֹת לֹא תַעֲשֶׂה אֵין שׁוּם הֶבְדֵּל: כָּל אִסּוּרֵי הַתּוֹרָה שָׁוִים לִגְבָרִים וּלְנָשִׁים. וּבְמִצְווֹת עֲשֵׂה שֶׁאֵינָן תְּלוּיוֹת בִּזְמַן – כְּמוֹ אֱמוּנָה בַּה׳, אַהֲבַת הָרֵעַ וְכִבּוּד הוֹרִים – נָשִׁים חַיָּבוֹת כְּמוֹ גְּבָרִים.
💛 אַהֲבַת הַשֵּׁם: יֵשׁ שֶׁכָּתְבוּ: הָאִישׁ מְחַכֶּה לִזְמַנִּים מְיֻחָדִים, וְהָאִשָּׁה מְרוֹמֶמֶת אֶת כָּל הַשִּׁגְרָה וּמְמַלֵּאת אוֹתָהּ בִּקְדֻשָּׁה. שְׁנֵי הַכֹּחוֹת יַחַד עוֹבְדִים אֶת ה׳ בִּשְׁלֵמוּת.
📖 פֶּרֶק ח׳, הֲלָכוֹת ח׳–ט׳

⏰🔑 מוּשָּׂג: מִצְוַת עֲשֵׂה שֶׁהַזְּמַן גְּרָמָא

״מִצְוַת הַצִּיצִית הִיא מִצְוַת עֲשֵׂה שֶׁהַזְּמַן גְּרָמָהּ, וְנָשִׁים פְּטוּרוֹת״.
📖 שֶׁהַזְּמַן גְּרָמָא – מִצְוָה שֶׁתְּלוּיָה בִּזְמַן מְסֻיָּם, לֹא בְּכָל עֵת.
💛 אַהֲבַת הַשֵּׁם: 💛 ה׳ נָתַן לִגְבָרִים וּלְנָשִׁים תַּפְקִידִים שׁוֹנִים בִּזְמַנֵּיהֶם, וּלְכָל אֶחָד דֶּרֶךְ מְיֻחֶדֶת לְהִתְקָרֵב אֵלָיו. 🧒 וְאַתָּה? לְמַד אֶת סִבַּת הַפְּטוֹר, וְלֹא תַּחְשֹׁב שֶׁמִּישֶׁהוּ ״פָּחוֹת חָשׁוּב״.

⚙️צִיצִיּוֹת שֶׁל מְכוֹנָה

אֶת חוּטֵי הַצִּיצִית צָרִיךְ לִטְווֹת לְשֵׁם מִצְוַת צִיצִית – "גְּדִלִים תַּעֲשֶׂה לָּךְ": לָךְ – לְשֵׁם חוֹבָתְךָ. לְדַעַת רֹב הָאַחֲרוֹנִים, אִם הִפְעִילוּ אֶת הַמְּכוֹנָה לְשֵׁם מִצְוָה – הַצִּיצִיּוֹת כְּשֵׁרוֹת, וּמְבָרְכִים עֲלֵיהֶן. וְיֵשׁ אוֹסְרִים, כִּי מְכוֹנָה אֵינָהּ יְכוֹלָה לְכַוֵּן.
קוֹנִים צִיצִיּוֹת מְכוֹנָה רַק עִם הַשְׁגָּחָה נֶאֱמָנָה. וְהַיּוֹם צִיצִיּוֹת עֲבוֹדַת יָד אֵינָן יְקָרוֹת, וְרָאוּי לִקְנוֹת אוֹתָן – כְּדֵי לְקַיֵּם אֶת הַמִּצְוָה לְכָל הַדֵּעוֹת.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאָנוּ מְהַדְּרִים בַּחוּטִים עַצְמָם, אָנוּ מַרְאִים שֶׁהַמִּצְוָה יְקָרָה לָנוּ – מֵהַחוּט הָרִאשׁוֹן וְעַד הַקֶּשֶׁר הָאַחֲרוֹן.

🪢📜 לְשׁוֹן חֲכָמִים: לְשֵׁם חוֹבְךָ

נֶאֱמַר ״גְּדִלִים תַּעֲשֶׂה לָּךְ״, דָּרְשׁוּ חֲכָמִים: ״לָךְ – לְשֵׁם חוֹבְךָ״.
📖 לְשֵׁם חוֹבְךָ – צָרִיךְ לִטְווֹת אֶת חוּטֵי הַצִּיצִית בְּכַוָּנָה מְפֹרֶשֶׁת שֶׁהֵם לְשֵׁם מִצְוַת צִיצִית.
💛 אַהֲבַת הַשֵּׁם: 💛 אֲפִלּוּ הַחוּטִים עַצְמָם צְרִיכִים לְהֵעָשׂוֹת מִתּוֹךְ כַּוָּנָה לְשֵׁם ה׳. 🧒 וְאַתָּה? כְּשֶׁאַתָּה קוֹנֶה טַלִּית קָטָן חֲדָשָׁה, שְׁאַל אִם הַצִּיצִיּוֹת נַעֲשׂוּ לְשֵׁם הַמִּצְוָה.

📏חוּטִים אֲרֻכִּים וּשְׁלֵמִים

חוּטֵי הַצִּיצִית צְרִיכִים לִהְיוֹת אֲרֻכִּים לְפָחוֹת 24 סֶנְטִימֶטֶר מִשְּׂפַת הַבֶּגֶד, יַחַד עִם הַקְּשָׁרִים. אִם נִקְרַע חוּט אֶחָד, אֲפִלּוּ עַד הַקְּשָׁרִים – הַצִּיצִית עֲדַיִן כְּשֵׁרָה. אִם נִקְרְעוּ שְׁנֵי חוּטִים וְנִשְׁאֲרוּ מֵהֶם 4 סֶנְטִימֶטֶר – גַּם כֵּן כְּשֵׁרָה.
חוּט שֶׁנֶּחְתַּךְ בַּשֹּׁרֶשׁ שֶׁלּוֹ, בֵּין הַקְּשָׁרִים לְבֵין הַבֶּגֶד – הַצִּיצִית פְּסוּלָה.
💛 אַהֲבַת הַשֵּׁם: כְּמוֹ שֶׁבּוֹדְקִים שֶׁהַמַּתָּנָה לְאָדָם אָהוּב שְׁלֵמָה וְיָפָה, כָּךְ מַקְפִּידִים שֶׁהַצִּיצִית תִּהְיֶה שְׁלֵמָה. מִי שֶׁמַּשְׁגִּיחַ עַל הַחוּטִים שֶׁלּוֹ, מַרְאֶה לַה׳ כַּמָּה הַמִּצְוָה יְקָרָה לוֹ.

״אִם חוּט נֶחְתַּךְ מֵעִקָּרוֹ, הַיְנוּ בֵּין מְקוֹם הַגְּדִיל לִמְקוֹם חִבּוּרוֹ לִכְנַף הַטַּלִּית – הַצִּיצִית פְּסוּלָה״.
📖 פְּסוּלָה – לֹא כְּשֵׁרָה לְקִיּוּם הַמִּצְוָה. צָרִיךְ לְהַחְלִיף אֶת הַצִּיצִית.
💛 אַהֲבַת הַשֵּׁם: 💛 ה׳ רוֹצֶה שֶׁהַמִּצְוָה תְּקֻיַּם בְּדִיּוּק, וְלָכֵן נָתַן לָנוּ סִימָנִים בְּרוּרִים לָדַעַת מָתַי הִיא כְּשֵׁרָה. 🧒 וְאַתָּה? אִם רָאִיתָ חוּט צִיצִית חָתוּךְ לְיַד הַבֶּגֶד – סַפֵּר לְאַבָּא, כְּדֵי לְהַחְלִיף אֶת הַטַּלִּית קָטָן.

🪢🔑 מוּשָּׂג: גְּדִיל

״אֹרֶךְ חוּטֵי הַצִּיצִית כּוֹלֵל אֶת הַגְּדִיל – הַחֵלֶק שֶׁיֵּשׁ בּוֹ קְשָׁרִים וּכְרִיכוֹת – וְהַחוּטִים הַפְּשׁוּטִים... וְטוֹב שֶׁשְּׁלִישׁ מֵהָאֹרֶךְ יִהְיֶה מִן הַגְּדִיל״.
📖 גְּדִיל – הַחֵלֶק הָעֶלְיוֹן שֶׁל חוּטֵי הַצִּיצִית, שֶׁקָּשׁוּר וְכָרוּךְ.
💛 אַהֲבַת הַשֵּׁם: 💛 אֲפִלּוּ הַיַּחַס בֵּין הַגְּדִיל לַחוּטִים הַפְּשׁוּטִים נִקְבַּע בְּדִיּוּק – ה׳ אוֹהֵב סֵדֶר וְדִיּוּק בְּכָל מִצְוָה קְטַנָּה. 🧒 וְאַתָּה? סְפֹר בְּעֵינֶיךָ אֶת הַקְּשָׁרִים בַּגְּדִיל שֶׁל הַצִּיצִית שֶׁלְּךָ.

🧠 בְּדֹק אֶת עַצְמְךָ

לְאֵיתָן יֵשׁ בֶּגֶד מְיֻחָד עִם חָמֵשׁ כְּנָפוֹת. מָה הַדִּין?
💡 בְּתוֹךְ חָמֵשׁ כְּנָפוֹת יֵשׁ גַּם אַרְבַּע, וְהַכָּנָף הַנּוֹסֶפֶת אֵינָהּ גּוֹרַעַת. קוֹשְׁרִים אַרְבַּע צִיצִיּוֹת בַּכְּנָפוֹת הָרְחוֹקוֹת זוֹ מִזּוֹ.
דָּנִי אוֹמֵר: אֵין לִי בֶּגֶד שֶׁל אַרְבַּע כְּנָפוֹת, אָז אֲנִי לֹא צָרִיךְ צִיצִית. מָה לָמְדוּ מֵרַב קְטִינָא?
💡 מִי שֶׁאֵין לוֹ בֶּגֶד מְרֻבָּע פָּטוּר, וְאֵין בְּיָדוֹ עֲבֵרָה. אֲבָל הַמַּלְאָךְ לִמֵּד אֶת רַב קְטִינָא: בְּעֵת צָרָה, זְכוּת הַמִּצְוָה מַצִּילָה אֶת מִי שֶׁהִשְׁתַּדֵּל לְהִתְחַיֵּב בָּהּ.
לָמָּה דַּוְקָא הַצִּיצִית יְכוֹלָה לְהַזְכִּיר לָנוּ אֶת כָּל הַמִּצְווֹת?
💡 הַצִּיצִית נִמְצֵאת עַל הַבֶּגֶד, וְלָכֵן אָדָם רוֹאֶה אוֹתָהּ בְּכָל עֵת, נִזְכָּר בְּמִי שֶׁצִּוָּה – וּמִתּוֹךְ כָּךְ זוֹכֵר אֶת כָּל הַמִּצְווֹת.

מַעֲשֵׂה שַׁבָּת וְלִפְנֵי עִוֵּר

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק כ״ו

⏳מְלָאכָה שֶׁנַּעֲשְׂתָה בְּשַׁבָּת

חֲכָמִים אָסְרוּ לֵהָנוֹת מִמְּלָאכָה שֶׁיְּהוּדִי עָשָׂה בְּשַׁבָּת. מִי שֶׁעָשָׂה מְלָאכָה בְּמֵזִיד – אָסוּר לוֹ לֵהָנוֹת מִמֶּנָּה לְעוֹלָם, וְלַאֲחֵרִים מֻתָּר בְּמוֹצָאֵי שַׁבָּת. מַאֲכָל שֶׁבִּשְּׁלוּ בְּשַׁבָּת – אָסוּר לֶאֱכֹל בְּשַׁבָּת, גַּם לֹא לְשֵׁם עֹנֶג שַׁבָּת.
מָה שֶׁאֶפְשָׁר הָיָה לַעֲשׂוֹת גַּם בְּלִי הַמְּלָאכָה – מֻתָּר. אִם הִדְלִיקוּ אוֹר בַּחֲדַר הַמַּדְרֵגוֹת, אֶפְשָׁר לַעֲלוֹת, כִּי גַּם בַּחֹשֶׁךְ עוֹלִים. אִם כִּבּוּ אוֹר – מֻתָּר לִישֹׁן שָׁם.
💛 אַהֲבַת הַשֵּׁם: אֵין זֶה רָאוּי לֵהָנוֹת בַּיּוֹם הַקָּדוֹשׁ מִמַּשֶּׁהוּ שֶׁנּוֹלַד מִתּוֹךְ חִלּוּל שַׁבָּת. הַהֲנָאָה הָאֲמִתִּית בְּשַׁבָּת בָּאָה מִן הַקְּדֻשָּׁה.
📖 פֶּרֶק כ״ו, הֲלָכוֹת א׳–ב׳, ז׳

🫱מִתְעַסֵּק וְקָטָן

מִי שֶׁעָבַר לְיַד הַקִּיר, נִשְׁעַן עָלָיו וּבְלִי שׁוּם כַּוָּנָה הִדְלִיק אוֹר – נִקְרָא "מִתְעַסֵּק", וּמֻתָּר לֵהָנוֹת מִן הָאוֹר. אֲבָל מִי שֶׁהִדְלִיק מִתּוֹךְ הֶרְגֵּל, כְּמוֹ בְּכָל יְמוֹת הַשָּׁבוּעַ – זֶה שׁוֹגֵג, וְאָסוּר לֵהָנוֹת.
קָטָן שֶׁהִדְלִיק אוֹר בִּשְׁבִיל עַצְמוֹ – מֻתָּר לַגְּדוֹלִים לֵהָנוֹת מִמֶּנּוּ. אִם הִדְלִיק בִּשְׁבִיל גָּדוֹל – אָסוּר.
💛 אַהֲבַת הַשֵּׁם: ה׳ יוֹדֵעַ מָה בַּלֵּב. הַהֲלָכָה מַבְחִינָה בֵּין מִי שֶׁהִתְכַּוֵּן לְמִי שֶׁלֹּא הִתְכַּוֵּן – וְלָכֵן כְּדַאי שֶׁכָּל מַעֲשֶׂה שֶׁלָּנוּ יִהְיֶה מְכֻוָּן לְטוֹב.
📖 פֶּרֶק כ״ו, הֲלָכוֹת א׳, ג׳

🚗לֹא עוֹזְרִים לַעֲבֹר עֲבֵרָה

הַתּוֹרָה אָמְרָה: "וְלִפְנֵי עִוֵּר לֹא תִתֵּן מִכְשֹׁל" – אָסוּר לַעֲזֹר לְאָדָם לַעֲבֹר עֲבֵרָה. לָכֵן לֹא נוֹתְנִים לְמִי שֶׁאֵינוֹ שׁוֹמֵר שַׁבָּת לְהִשְׁתַּמֵּשׁ בְּשַׁבָּת בַּמְּכוֹנִית אוֹ בָּרַדְיוֹ שֶׁלָּנוּ. וְנֶהָג שֶׁעוֹצֵר בְּשַׁבָּת וְשׁוֹאֵל עַל הַדֶּרֶךְ – לֹא מַדְרִיכִים אוֹתוֹ, וּמִתְנַצְּלִים בְּנִימוּס.
מֻתָּר לְהַזְמִין לְשַׁבָּת אוֹרֵחַ שֶׁאֵינוֹ שׁוֹמֵר שַׁבָּת, אִם מַצִּיעִים לוֹ בְּכֵנוּת מָקוֹם לִישֹׁן – כְּדֵי לְהַרְבּוֹת אַהֲבָה וְקֵרוּב.
💛 אַהֲבַת הַשֵּׁם: אֲנַחְנוּ אוֹהֲבִים כָּל יְהוּדִי, וְדַוְקָא מִתּוֹךְ אַהֲבָה לֹא נַעֲזֹר לוֹ לְחַלֵּל שַׁבָּת. אֶפְשָׁר לְסָרֵב בְּחִיּוּךְ – וּלְהַזְמִין אוֹתוֹ לְטַעֵם אֶת יֹפִי הַשַּׁבָּת.
📖 פֶּרֶק כ״ו, הֲלָכָה ט׳

🛡️"וְלִפְנֵי עִוֵּר"

הַתּוֹרָה אוֹסֶרֶת לָשִׂים מִכְשׁוֹל לִפְנֵי עִוֵּר – וַחֲכָמִים לָמְדוּ: אָסוּר לְהַכְשִׁיל אָדָם בַּעֲבֵרָה, וְאָסוּר לַעֲזֹר לְמִי שֶׁעוֹבֵר עֲבֵרָה. לָכֵן מְכוֹנַת מַשְׁקָאוֹת שֶׁרֹב הַקּוֹנִים בָּהּ יְהוּדִים – צָרִיךְ לְהַפְסִיק אֶת פְּעֻלָּתָהּ לִפְנֵי שַׁבָּת.
גַּם כְּשֶׁאֵין בְּיָדֵנוּ לַעֲצֹר אֶת הַחִלּוּל, אֲנַחְנוּ לֹא שֻׁתָּפִים בּוֹ – וּמְקָרְבִים בְּאַהֲבָה.
💛 אַהֲבַת הַשֵּׁם: כָּל יִשְׂרָאֵל עֲרֵבִים זֶה לָזֶה. כְּשֶׁאֲנַחְנוּ נִזְהָרִים לֹא לְהַכְשִׁיל אַחֵר, אֲנַחְנוּ שׁוֹמְרִים גַּם עָלָיו – כִּי הוּא אָחִינוּ.
📖 פֶּרֶק כ״ו, הֲלָכוֹת ט׳–י׳

🧠 בְּדֹק אֶת עַצְמְךָ

יְהוּדִי הִדְלִיק בְּשַׁבָּת אוֹר בַּחֲדַר הַמַּדְרֵגוֹת. מָה מֻתָּר לְנֹעַם לַעֲשׂוֹת?
💡 מָה שֶׁאֶפְשָׁר הָיָה לַעֲשׂוֹת גַּם בְּלִי הָאוֹר – מֻתָּר. אֲבָל לֹא רָצִים וְלֹא קוֹרְאִים בִּזְכוּת הָאוֹר, כִּי זוֹ הֲנָאָה מִמְּלֶאכֶת שַׁבָּת.
אָדָם בִּשֵּׁל בְּשַׁבָּת בְּמֵזִיד. מָה דִּין הָאֹכֶל?
💡 בְּשַׁבָּת אָסוּר לְכֻלָּם לֶאֱכֹל, גַּם לְשֵׁם עֹנֶג שַׁבָּת. בְּמוֹצָאֵי שַׁבָּת מֻתָּר לַאֲחֵרִים, וּמִי שֶׁבִּשֵּׁל בְּמֵזִיד – לֹא יֵהָנֶה מִזֶּה לְעוֹלָם.
בְּשַׁבָּת נִשְׁעַן אֵלִי עַל הַקִּיר, וּבְלִי לָשִׂים לֵב לָחַץ עַל הַמֶּתֶג וְהִדְלִיק אוֹר. מָה הַדִּין?
💡 מִי שֶׁעָשָׂה מְלָאכָה בְּלִי שׁוּם כַּוָּנָה הוּא "מִתְעַסֵּק", וּמֻתָּר לֵהָנוֹת. מִי שֶׁהִדְלִיק מִתּוֹךְ הֶרְגֵּל – הִתְכַּוֵּן לְהַדְלִיק, וְהוּא שׁוֹגֵג.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

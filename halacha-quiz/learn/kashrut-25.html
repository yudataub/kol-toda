בָּשָׂר בְּחָלָב

🍽️ הִלְכוֹת כַּשְׁרוּת · פֶּרֶק כ״ה

🐐לֹא תְבַשֵּׁל גְּדִי בַּחֲלֵב אִמּוֹ

שָׁלֹשׁ פְּעָמִים כָּתוּב בַּתּוֹרָה: "לֹא תְבַשֵּׁל גְּדִי בַּחֲלֵב אִמּוֹ". מִכָּאן לָמְדוּ שְׁלֹשָׁה אִסּוּרִים: אָסוּר לְבַשֵּׁל בָּשָׂר וְחָלָב יַחַד, אָסוּר לֶאֱכֹל אוֹתָם, וְאָסוּר לֵהָנוֹת מֵהֶם.
חֲכָמִים הוֹסִיפוּ: גַּם בְּלִי בִּשּׁוּל – אָסוּר לֶאֱכֹל בָּשָׂר וְחָלָב יַחַד, כְּמוֹ כָּרִיךְ שֶׁל נַקְנִיק וּגְבִינָה.
💛 אַהֲבַת הַשֵּׁם: הַבָּשָׂר לְבַדּוֹ כָּשֵׁר, וְהֶחָלָב לְבַדּוֹ כָּשֵׁר – רַק הַחִבּוּר בֵּינֵיהֶם אָסוּר. כְּשֶׁאָנוּ מַפְרִידִים בֵּינֵיהֶם, אָנוּ זוֹכְרִים אֶת ה׳ בְּכָל אֲרוּחָה.
📖 פֶּרֶק כ״ה, הֲלָכָה א׳

🔥אָסוּר גַּם בַּהֲנָאָה

בָּשָׂר וְחָלָב שֶׁהִתְבַּשְּׁלוּ יַחַד – אֲסוּרִים גַּם בַּהֲנָאָה. לָכֵן אָסוּר לִמְכֹּר אוֹתָם לְגוֹי, לָתֵת לוֹ אוֹתָם בְּמַתָּנָה, אוֹ לָשִׂים אוֹתָם לִפְנֵי חַיּוֹת הֶפְקֵר. קוֹבְרִים אוֹתָם, וְאִם קָשֶׁה – זוֹרְקִים לַפַּח.
גַּם לְבַשֵּׁל בָּשָׂר בְּחָלָב בִּשְׁבִיל גּוֹי – אָסוּר. בְּזֶה הָאִסּוּר הַזֶּה חָמוּר מִשְּׁאָר הַמַּאֲכָלִים הָאֲסוּרִים.
💛 אַהֲבַת הַשֵּׁם: בָּשָׂר וְחָלָב, כָּל אֶחָד בִּמְקוֹמוֹ, טוֹבִים וּכְשֵׁרִים. רַק הָעִרְבּוּב שֶׁלָּהֶם הוֹפֵךְ אוֹתָם לְאִסּוּר חָמוּר – וְכָךְ אָנוּ לוֹמְדִים לִשְׁמֹר עַל הַסֵּדֶר שֶׁקָּבַע ה׳.
📖 פֶּרֶק כ״ה, הֲלָכָה א׳

🧱גָּדֵר לַתּוֹרָה

מִן הַתּוֹרָה אָסוּר לְבַשֵּׁל בָּשָׂר בְּחָלָב, לֶאֱכֹל אוֹתָם וְלֵהָנוֹת מֵהֶם. חֲכָמִים עָשׂוּ גָּדֵר סְבִיב הָאִסּוּר: לֹא לֶאֱכֹל אוֹתָם יַחַד גַּם בְּלִי בִּשּׁוּל, לֹא לְהַנִּיחַ אוֹתָם עַל שֻׁלְחָן אֶחָד, לֹא לֶאֱפוֹת לֶחֶם חֲלָבִי, וּלְהַמְתִּין אַחֲרֵי בָּשָׂר.
בָּשָׂר וְחָלָב מְצוּיִים בְּכָל מִטְבָּח, וְקַל מְאוֹד לְעָרֵב אוֹתָם בְּטָעוּת. לָכֵן חִזְּקוּ חֲכָמִים אֶת הַגָּדֵר כָּל כָּךְ.
💛 אַהֲבַת הַשֵּׁם: גָּדֵר טוֹבָה אֵינָהּ עֹנֶשׁ – הִיא שְׁמִירָה. חֲכָמִים בָּנוּ אוֹתָהּ מֵאַהֲבָה, כְּדֵי שֶׁלֹּא נִכָּשֵׁל בְּאִסּוּר הַתּוֹרָה.
📖 פֶּרֶק כ״ה, הֲלָכוֹת א׳, ה׳

🍲🔑 מוּשָּׂג: בָּשָׂר בְּחָלָב

״לֹא תְבַשֵּׁל גְּדִי בַּחֲלֵב אִמּוֹ״ – אָסוּר לְבַשֵּׁל בָּשָׂר עִם חָלָב, וְאָסוּר גַּם לְאָכְלָם וְגַם לֵהָנוֹת מֵהֶם.
📖 בָּשָׂר בְּחָלָב – שְׁלֹשָׁה אִסּוּרִים בְּמִלָּה אַחַת: לֹא מְבַשְּׁלִים, לֹא אוֹכְלִים וְלֹא נֶהֱנִים מֵהַתַּעֲרֹבֶת.
💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה מְלַמֶּדֶת אוֹתָנוּ רַחֲמִים: לֹא לְעָרֵב אֶת בְּשַׂר הַגְּדִי עִם הֶחָלָב שֶׁנּוֹעַד לְגַדֵּל אוֹתוֹ. 🧒 וְאַתָּה? בְּבַיִת כָּשֵׁר יֵשׁ כֵּלִים נִפְרָדִים לְבָשָׂר וּלְחָלָב – שְׁאַל אֶת אִמָּא אֵיךְ מַכִּירִים בֵּינֵיהֶם.
📖 פֶּרֶק כ״ה, הֲלָכָה א׳

🍗גַּם עוֹף

מִן הַתּוֹרָה הָאִסּוּר הוּא בִּבְשַׂר בְּהֵמָה – שׁוֹר, כֶּבֶשׂ וְעֵז. חֲכָמִים אָסְרוּ גַּם בְּשַׂר עוֹף וּבְשַׂר חַיָּה עִם חָלָב. לָכֵן שְׁנִיצֶל עוֹף עִם גְּבִינָה – אָסוּר.
בְּכָל הַגְּזֵרוֹת הִשְׁווּ חֲכָמִים אֶת הָעוֹף לַבְּהֵמָה, כְּדֵי שֶׁלֹּא יָקֵלּוּ בְּעוֹף וְיָבוֹאוּ לְהָקֵל גַּם בִּבְהֵמָה.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים בָּנוּ גָּדֵר סְבִיב הַתּוֹרָה, כְּמוֹ גָּדֵר סְבִיב גַּן יָפֶה. מִי שֶׁשּׁוֹמֵר עַל הַגָּדֵר – שׁוֹמֵר עַל הַגַּן כֻּלּוֹ.
📖 פֶּרֶק כ״ה, הֲלָכוֹת ב׳, ה׳

🧀לֹא עַל שֻׁלְחָן אֶחָד

כְּשֶׁאוֹכְלִים בָּשָׂר, לֹא יִהְיֶה עַל הַשֻּׁלְחָן מַאֲכָל חֲלָבִי, וּכְשֶׁאוֹכְלִים חָלָב – לֹא יִהְיֶה עָלָיו בָּשָׂר, שֶׁמָּא יִטְעוּ וְיֹאכְלוּ אוֹתָם יַחַד. וּשְׁנֵי אֲנָשִׁים שֶׁמַּכִּירִים זֶה אֶת זֶה – לֹא יֹאכְלוּ זֶה בָּשָׂר וְזֶה חָלָב עַל שֻׁלְחָן אֶחָד, אֶלָּא אִם שָׂמוּ בֵּינֵיהֶם סִימָן, כְּמוֹ מַפָּה נִפְרֶדֶת.
לִפְנֵי שֶׁאוֹכְלִים בָּשָׂר עַל שֻׁלְחָן שֶׁאָכְלוּ עָלָיו חָלָב, אוֹ לְהֶפֶךְ – מְנַקִּים אֶת הַשֻּׁלְחָן מִן הַשְּׁיָרִים, אוֹ מַחְלִיפִים מַפָּה.
💛 אַהֲבַת הַשֵּׁם: סִימָן קָטָן עַל הַשֻּׁלְחָן שׁוֹמֵר עַל מִצְוָה גְּדוֹלָה. כָּךְ גַּם חֲבֵרִים טוֹבִים עוֹזְרִים זֶה לָזֶה לִשְׁמֹר עַל הַתּוֹרָה.
📖 פֶּרֶק כ״ה, הֲלָכוֹת ג׳, ו׳

🍞לֶחֶם חֲלָבִי

חֲכָמִים גָּזְרוּ שֶׁלֹּא לָלוּשׁ לֶחֶם בְּחָלָב אוֹ בְּשׁוּמַן בָּשָׂר, שֶׁמָּא יִשְׁכְּחוּ וְיֹאכְלוּ אוֹתוֹ עִם הַמִּין הַשֵּׁנִי. וְאִם לָשׁוּ בְּטָעוּת – הַלֶּחֶם אָסוּר.
מֻתָּר לְהָכִין מְעַט לֶחֶם חֲלָבִי לַסְּעוּדָה הַקְּרוֹבָה, אוֹ לֶחֶם בְּצוּרָה מְיֻחֶדֶת שֶׁבְּנֵי הַבַּיִת מַכִּירִים. עוּגוֹת – מֻתָּר לְהָכִין חֲלָבִיּוֹת, כִּי בְּדֶרֶךְ כְּלָל לֹא אוֹכְלִים אוֹתָן עִם בָּשָׂר.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים חָשְׁבוּ עַל כָּל פְּרוּסָה שֶׁתַּגִּיעַ לַשֻּׁלְחָן. הַדְּאָגָה שֶׁלָּהֶם הִיא דְּאָגָה שֶׁל אַהֲבָה – שֶׁאַף יְהוּדִי לֹא יִכָּשֵׁל.
📖 פֶּרֶק כ״ה, הֲלָכָה ד׳

🥗לֶחֶם וְסָלָטִים מֵאֲרוּחַת בָּשָׂר

לֶחֶם שֶׁהֻגַּשׁ בַּאֲרוּחַת בָּשָׂר, וְנָגְעוּ בּוֹ בְּיָדַיִם שְׁמֵנוֹת מִבָּשָׂר – לֹא מַגִּישִׁים אוֹתוֹ בַּאֲרוּחָה חֲלָבִית. וְכֵן סָלָט שֶׁלָּקְחוּ מִמֶּנּוּ בְּמַזְלְגוֹת שֶׁאָכְלוּ בָּהֶם בָּשָׂר. אִם נִזְהֲרוּ בָּזֶה – מֻתָּר.
בַּמְּקָרֵר מֻתָּר לָשִׂים בָּשָׂר וּגְבִינָה עַל אוֹתוֹ מַדָּף, אִם הֵם אֵינָם נוֹגְעִים זֶה בָּזֶה. אִם נָגְעוּ – שׁוֹטְפִים אֶת מְקוֹם הַנְּגִיעָה.
💛 אַהֲבַת הַשֵּׁם: שְׁמִירָה בְּעֵינַיִם פְּקוּחוֹת הִיא חֵלֶק מִן הָאַהֲבָה לַמִּצְווֹת. מִי שֶׁשָּׂם לֵב גַּם לִפְרוּסַת לֶחֶם, מַרְאֶה שֶׁכָּל פְּרָט בַּתּוֹרָה חָשׁוּב לוֹ.
📖 פֶּרֶק כ״ה, הֲלָכָה ו׳

🍽️כֵּלִים נִפְרָדִים

בְּמִטְבָּח כָּשֵׁר יֵשׁ סִירִים, צַלָּחוֹת וְסַכּוּ״ם לְבָשָׂר, וַאֲחֵרִים לְחָלָב. נָכוֹן שֶׁיִּהְיֶה בֵּינֵיהֶם הֶבְדֵּל בּוֹלֵט, כְּמוֹ צֶבַע אַחֵר, וּמָקוֹם קָבוּעַ לְכָל סוּג – כְּדֵי שֶׁלֹּא יִתְחַלְּפוּ.
גַּם מַאֲכָלִים קָרִים מַגִּישִׁים בַּכֵּלִים הַמַּתְאִימִים. רַבִּים נוֹהֲגִים בִּשְׁתֵּי מִלְחִיּוֹת – אַחַת לְבָשָׂר וְאַחַת לְחָלָב.
💛 אַהֲבַת הַשֵּׁם: הַסֵּדֶר בַּמִּטְבָּח הוּא חֵלֶק מִן הַקְּדֻשָּׁה שֶׁל הַבַּיִת. כָּל יֶלֶד שֶׁמַּחֲזִיר כַּף לַמְּגֵרָה הַנְּכוֹנָה – שׁוֹמֵר עַל הַכַּשְׁרוּת שֶׁל כָּל הַמִּשְׁפָּחָה.
📖 פֶּרֶק כ״ה, הֲלָכָה ז׳

🥕פַּרְוֶה

מַאֲכַל פַּרְוֶה אֵינוֹ בְּשָׂרִי וְאֵינוֹ חֲלָבִי – כְּמוֹ פֵּרוֹת, יְרָקוֹת וּבֵיצִים. אֶת מַאֲכְלֵי הַפַּרְוֶה אֶפְשָׁר לֶאֱכֹל גַּם עִם בָּשָׂר וְגַם עִם חָלָב.
קֶרֶשׁ חִתּוּךְ מֵעֵץ אוֹ מִפְּלַסְטִיק סוֹפֵג טְעָמִים, וְלָכֵן מְיַחֲדִים קֶרֶשׁ לְבָשָׂר וְקֶרֶשׁ לְחָלָב, וּמִי שֶׁצָּרִיךְ – גַּם קֶרֶשׁ לְפַרְוֶה.
💛 אַהֲבַת הַשֵּׁם: הַפַּרְוֶה מַזְכִּיר לָנוּ שֶׁיֵּשׁ הַרְבֵּה מַאֲכָלִים טוֹבִים שֶׁמַּתְאִימִים לְכָל אֲרוּחָה. ה׳ נָתַן לָנוּ שֶׁפַע שֶׁל פֵּרוֹת וִירָקוֹת – וְאָנוּ מְבָרְכִים וּמוֹדִים עֲלֵיהֶם.

🥄לֹא מְעָרְבְבִים כֵּלִים

גַּם כְּשֶׁמְּכִינִים מַאֲכָל פַּרְוֶה קַר, כְּמוֹ סָלָט, נָכוֹן לְהָכִין אוֹתוֹ בְּכֵלִים מִמִּין אֶחָד: בִּקְעָרָה בְּשָׂרִית – מְעָרְבְבִים בְּכַף בְּשָׂרִית, וּבִקְעָרָה חֲלָבִית – בְּכַף חֲלָבִית. כָּךְ שׁוֹמְרִים עַל הַהַפְרָדָה בֵּין כְּלֵי בָּשָׂר לִכְלֵי חָלָב.
בִּשְׁעַת הַצֹּרֶךְ – כְּשֶׁמְּמַהֲרִים וְאֵין כַּף בְּשָׂרִית שְׁטוּפָה – אֶפְשָׁר לְעָרְבֵּב אֶת הַפַּרְוֶה בְּכַף חֲלָבִית, אוֹ לְהֶפֶךְ. וּבְמִטְבָּחִים גְּדוֹלִים, כְּמוֹ בְּבֵית סֵפֶר אוֹ בְּאוּלָם, קוֹבְעִים כְּלָלִים מַחְמִירִים יוֹתֵר.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי שָׂם לֵב לְכָל כַּף וּלְכָל קְעָרָה, אֲנִי מַרְאֶה שֶׁהַמִּצְווֹת שֶׁל ה׳ חֲשׁוּבוֹת לִי גַּם בַּפְּרָטִים הַקְּטַנִּים.

🥄טָעִיתִי בַּכַּף!

מִי שֶׁעֵרֵב תַּבְשִׁיל בְּשָׂרִי רוֹתֵחַ בְּכַף מַתֶּכֶת חֲלָבִית נְקִיָּה – בְּדִיעֲבַד הַתַּבְשִׁיל כָּשֵׁר, אֲבָל הַכַּף צְרִיכָה הַכְשָׁרָה בְּהַגְעָלָה. וְכֵן מִי שֶׁחָתַךְ בָּשָׂר רוֹתֵחַ בְּסַכִּין חֲלָבִית נְקִיָּה.
אִם הַבָּשָׂר הָיָה קַר – מַסְפִּיק לְנַקּוֹת הֵיטֵב אֶת הַסַּכִּין. וּבְכָל טָעוּת – לֹא מַסְתִּירִים, אֶלָּא מְסַפְּרִים לְאַבָּא וְשׁוֹאֲלִים רַב.
💛 אַהֲבַת הַשֵּׁם: כָּל אֶחָד טוֹעֶה לִפְעָמִים. מִי שֶׁמְּסַפֵּר עַל טָעוּתוֹ בְּיֹשֶׁר שׁוֹמֵר עַל הַכַּשְׁרוּת שֶׁל כָּל הַבַּיִת – וְה׳ אוֹהֵב אֶת הַיְּשָׁרִים.

🚰כִּיּוֹר וְשַׁיִשׁ

נוֹהֲגִים לְהַתְקִין שְׁנֵי כִּיּוֹרִים – אֶחָד לִכְלֵי חָלָב וְאֶחָד לִכְלֵי בָּשָׂר – וּלְיַחֵד צַד אֶחָד שֶׁל הַשַּׁיִשׁ לְחָלָב וְצַד שֵׁנִי לְבָשָׂר. נוֹהֲגִים גַּם בְּמַטְלִיּוֹת קִרְצוּף נִפְרָדוֹת.
בִּמְדִיחַ כֵּלִים מֻתָּר לִשְׁטֹף כְּלֵי בָּשָׂר וּכְלֵי חָלָב – אֲבָל לֹא יַחַד, אֶלָּא בָּזֶה אַחַר זֶה. וְיֵשׁ מְהַדְּרִים לְיַחֵד מֵדִיחַ לְכָל מִין. מִי שֶׁיֵּשׁ לוֹ כִּיּוֹר אֶחָד – מְנַקֶּה אוֹתוֹ הֵיטֵב בֵּין סוּג לְסוּג.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁכָּל דָּבָר בַּמִּטְבָּח נִמְצָא בִּמְקוֹמוֹ, קַל יוֹתֵר לִשְׁמֹר מִצְווֹת. הַסֵּדֶר הַזֶּה עוֹזֵר לָנוּ לַעֲשׂוֹת אֶת רְצוֹן ה׳ בְּשִׂמְחָה, בְּלִי טָעֻיּוֹת.

♨️תַּנּוּר שֶׁל בָּשָׂר וְשֶׁל חָלָב

רַבִּים נוֹהֲגִים לְיַחֵד בַּתַּנּוּר תָּא לִבְשָׂרִי וְתָא לַחֲלָבִי. הָרוֹצֶה לְהִשְׁתַּמֵּשׁ בְּאוֹתוֹ תָּא – מְיַחֵד תַּבְנִית לְכָל מִין, וּבֵין מִין לְמִין מְחַמֵּם אֶת הַתַּנּוּר כַּחֲצִי שָׁעָה בַּחֹם הַגָּבוֹהַּ בְּיוֹתֵר.
טָעוּ וְאָפוּ מַאֲכָל חֲלָבִי בְּתַנּוּר בְּשָׂרִי – בְּדִיעֲבַד הַמַּאֲכָל כָּשֵׁר, כִּי לֹא נִכְנַס בּוֹ טַעַם מַמָּשִׁי שֶׁל בָּשָׂר, וְרֵיחַ אֵינוֹ אוֹסֵר. וְאֶת הַתַּנּוּר מַכְשִׁירִים בְּחִמּוּם שֶׁל חֲצִי שָׁעָה.
💛 אַהֲבַת הַשֵּׁם: הַהַפְרָדָה בֵּין בָּשָׂר לְחָלָב מְלַוָּה אוֹתָנוּ בְּכָל פִּנָּה בַּמִּטְבָּח – וְכָל פִּנָּה כָּזוֹ הִיא מָקוֹם שֶׁל מִצְוָה.

📦מִיקְרוֹגַל אֶחָד

בְּאוֹתוֹ מִיקְרוֹגַל אֶפְשָׁר לְחַמֵּם מַאֲכָלִים חֲלָבִיִּים וּבְשָׂרִיִּים, אִם יֵשׁ הַפְרָדָה: לֹא מַנִּיחִים אֶת הָאֹכֶל יָשָׁר עַל הַצַּלַּחַת הַקְּבוּעָה שֶׁל הַמִּיקְרוֹגַל, אֶלָּא עַל צַלַּחַת בְּשָׂרִית אוֹ חֲלָבִית, וּמְכַסִּים בְּמִכְסֶה מְיֻחָד לְכָל סוּג.
אֶפְשָׁר גַּם לִקְבֹּעַ שֶׁהַמִּיקְרוֹגַל חֲלָבִי, וּכְשֶׁמְּחַמְּמִים בּוֹ בָּשָׂר – שָׂמִים צַלַּחַת נוֹסֶפֶת, וְעוֹטְפִים אֶת הָאֹכֶל בְּמִכְסֶה, בְּקֻפְסָה אוֹ בְּשַׂקִּית.
💛 אַהֲבַת הַשֵּׁם: הַהֲלָכָה מַתְאִימָה אֶת עַצְמָהּ גַּם לַמַּכְשִׁירִים הַחֲדָשִׁים שֶׁל יָמֵינוּ. הַתּוֹרָה חַיָּה בְּכָל דּוֹר – וּבְכָל מִטְבָּח.

🧠 בְּדֹק אֶת עַצְמְךָ

בְּטָעוּת הִתְבַּשְּׁלוּ קְצִיצוֹת בָּשָׂר בְּרֹטֶב שֶׁל שַׁמֶּנֶת. מָה עוֹשִׂים בַּתַּבְשִׁיל?
💡 בָּשָׂר וְחָלָב שֶׁהִתְבַּשְּׁלוּ – אֲסוּרִים בַּהֲנָאָה. גַּם מַתָּנָה לְגוֹי וְגַם אֹכֶל לְחַיּוֹת הֶפְקֵר נוֹתְנִים לָנוּ הֲנָאָה. לָכֵן קוֹבְרִים אוֹ זוֹרְקִים לַפַּח.
אַבָּא עֵרֵב תַּבְשִׁיל בָּשָׂר רוֹתֵחַ, וּבְטָעוּת עֵרֵב בְּכַף מַתֶּכֶת חֲלָבִית נְקִיָּה. מָה הַדִּין?
💡 כַּף מַתֶּכֶת נְקִיָּה אֵינָהּ מוֹסִיפָה טַעַם, וְלָכֵן בְּדִיעֲבַד הַתַּבְשִׁיל כָּשֵׁר. אֲבָל הַכַּף הָיְתָה פַּעַם בְּחָלָב רוֹתֵחַ וּפַעַם בְּבָשָׂר רוֹתֵחַ, וּצְרִיכָה הַגְעָלָה.
בַּבַּיִת יֵשׁ מִיקְרוֹגַל אֶחָד, וְרוֹצִים לְחַמֵּם בּוֹ קְצִיצוֹת עוֹף. אֵיךְ עוֹשִׂים זֹאת?
💡 בְּאוֹתוֹ מִיקְרוֹגַל אֶפְשָׁר לְחַמֵּם בָּשָׂר וְחָלָב, אִם מַפְרִידִים: צַלַּחַת מֵאוֹתוֹ סוּג, וּמִכְסֶה מְיֻחָד לְכָל סוּג – כְּדֵי שֶׁלֹּא יִכָּנְסוּ אֵדִים רַבִּים לָאֹכֶל.

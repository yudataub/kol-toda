תְּפִלָּה וְקִדּוּשׁ בְּשַׁבָּת

🌸 תְּפִלַּת בָּנוֹת · פֶּרֶק כ״ב

🔀פּוֹנִים לְמַעֲרָב – שְׁנֵי מִנְהָגִים

בְּקַבָּלַת שַׁבָּת נוֹהֲגִים לִפְנוֹת לְצַד מַעֲרָב, שֶׁשָּׁם הַחַמָּה שׁוֹקַעַת. לְמִנְהַג הַסְּפָרַדִּים – פּוֹנִים לְמַעֲרָב בַּאֲמִירַת ״מִזְמוֹר לְדָוִד״ וּבְ״לְכָה דוֹדִי״. לְמִנְהַג הָאַשְׁכְּנַזִּים – פּוֹנִים לְמַעֲרָב רַק בַּבַּיִת הָאַחֲרוֹן שֶׁל ״לְכָה דוֹדִי״, שֶׁבּוֹ אוֹמְרִים ״בּוֹאִי כַלָּה״.
יֵשׁ נוֹהֲגִים לִפְנוֹת לְצַד פֶּתַח בֵּית הַכְּנֶסֶת, גַּם כְּשֶׁאֵינוֹ בְּצַד מַעֲרָב. וְהַנָּשִׁים נוֹהֲגוֹת לְקַבֵּל אֶת הַשַּׁבָּת בְּהַדְלָקַת הַנֵּרוֹת.
💛 אַהֲבַת הַשֵּׁם: שְׁנֵי הַמִּנְהָגִים כַּהֲלָכָה. כֻּלָּנוּ פּוֹנִים יַחַד לְקַבֵּל אֶת פְּנֵי הַשַּׁבָּת – כְּמוֹ שֶׁיּוֹצְאִים לִקְרַאת כַּלָּה אֲהוּבָה.
📖 פֶּרֶק כ״ב, הֲלָכָה ב׳

🕯️🔑 מוּשָּׂג: תּוֹסֶפֶת שַׁבָּת

״כְּדֵי לְקַיֵּם אֶת מִצְוַת תּוֹסֶפֶת שַׁבָּת, עֲלֵיהֶם לְקַבֵּל אֶת הַשַּׁבָּת לִפְנֵי הַשְּׁקִיעָה... הַנָּשִׁים נוֹהֲגוֹת לְקַבֵּל אֶת הַשַּׁבָּת בְּהַדְלָקַת נֵרוֹת, וְעַל יְדֵי כָּךְ יֵשׁ לָהֶן תּוֹסֶפֶת שַׁבָּת לַמְהַדְּרִין״.
📖 תּוֹסֶפֶת – הוֹסָפָה. מוֹסִיפִים זְמַן מִן הַחֹל עַל הַשַּׁבָּת: מְקַבְּלִים אוֹתָהּ עוֹד לִפְנֵי הַשְּׁקִיעָה.
💛 אַהֲבַת הַשֵּׁם: יוֹצְאִים לִקְרַאת הַשַּׁבָּת מֻקְדָּם, כְּמוֹ לִקְרַאת אוֹרַחַת אֲהוּבָה – ״בּוֹאִי כַלָּה שַׁבָּת הַמַּלְכָּה״. 🧒 וְאַתְּ? בְּעֶרֶב שַׁבָּת – הִיִּי מוּכָנָה עוֹד לִפְנֵי הַשְּׁקִיעָה.
📖 פֶּרֶק כ״ב, הֲלָכָה ב׳

🕍תְּפִלַּת עֲמִידָה בְּשַׁבָּת

לְכַתְּחִלָּה טוֹב שֶׁבָּנוֹת יִתְפַּלְּלוּ עֲמִידָה שֶׁל שַׁחֲרִית וּמִנְחָה בְּכָל יוֹם, וּבְשַׁבָּת – בְּנֹסַח שֶׁל שַׁבָּת. וְגַם מִי שֶׁבִּימֵי הַחֹל מִסְתַּפֶּקֶת בְּבִרְכוֹת הַשַּׁחַר וּבִרְכוֹת הַתּוֹרָה – רָאוּי שֶׁבְּשַׁבָּת תִּתְפַּלֵּל עֲמִידָה שֶׁל שַׁחֲרִית, כִּי בְּשַׁבָּת יֵשׁ בְּדֶרֶךְ כְּלָל יוֹתֵר זְמַן.
נָשִׁים רַבּוֹת אַף מְהַדְּרוֹת לָלֶכֶת לְבֵית הַכְּנֶסֶת וּלְהִתְפַּלֵּל עִם הַצִּבּוּר בְּלֵיל שַׁבָּת וּבְשַׁחֲרִית.
💛 אַהֲבַת הַשֵּׁם: שַׁבָּת הִיא זְמַן שֶׁל מְנוּחָה – וְגַם זְמַן שֶׁל קִרְבָה. כְּשֶׁיֵּשׁ יוֹתֵר פְּנַאי, אֶפְשָׁר לַעֲמֹד לִפְנֵי ה׳ בְּלִי לְמַהֵר.
📖 פֶּרֶק כ״ב, הֲלָכָה ו׳

🔀מוּסָף וּקְרִיאַת הַתּוֹרָה – מַחֲלֹקֶת פּוֹסְקִים

מוּסָף: לְדַעַת הַמָּגֵן גִּבּוֹרִים נָשִׁים חַיָּבוֹת, כִּי גַּם בּוֹ מְבַקְּשִׁים רַחֲמִים, וְהוּא נִתְקַן לִכְבוֹד קְדֻשַּׁת הַיּוֹם – כְּמוֹ קִדּוּשׁ. לְדַעַת הַצְּלַ״ח פְּטוּרוֹת, כִּי הוּא תָּלוּי בִּזְמַן. לְמַעֲשֶׂה הֲלָכָה כַּמְּקִלִּים, וְהָרוֹצָה לְהִתְפַּלֵּל – רַשָּׁאִית וְיֵשׁ לָהּ זְכוּת. וּבְרֹאשׁ הַשָּׁנָה וּבְיוֹם הַכִּפּוּרִים רָאוּי לְכָל אִשָּׁה לְהִתְפַּלֵּל מוּסָף, כִּי בּוֹ עִקַּר בַּקָּשַׁת הָרַחֲמִים.
קְרִיאַת הַתּוֹרָה בְּשַׁבָּת: לְדַעַת הַמָּגֵן אַבְרָהָם נָשִׁים חַיָּבוֹת לִשְׁמֹעַ, כִּי גַּם הֵן צְרִיכוֹת לִשְׁמֹעַ אֶת כָּל הַתּוֹרָה בְּכָל שָׁנָה. לְדַעַת רֹב רֻבָּם שֶׁל הַפּוֹסְקִים פְּטוּרוֹת, וְכָךְ הֲלָכָה. וְהָרוֹצָה לְהַדֵּר – טוֹב שֶׁתִּשְׁמַע, וְיֵשׁ לָהּ בָּזֶה מִצְוָה.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁבַּת פְּטוּרָה, כָּל תְּפִלָּה נוֹסֶפֶת וְכָל פָּסוּק שֶׁהִיא שׁוֹמַעַת מִן הַתּוֹרָה – הֵם זְכוּת וּמַתָּנָה שֶׁהִיא נוֹתֶנֶת לַה׳ מֵרְצוֹנָהּ.
📖 פֶּרֶק כ״ב, הֲלָכָה ו׳; פֶּרֶק כ״ג, הֲלָכָה ב׳

⏰אִחַרְתְּ בְּשַׁבָּת?

בַּת שֶׁמַּגִּיעָה לְבֵית הַכְּנֶסֶת בְּשַׁבָּת בַּבֹּקֶר כְּשֶׁהַצִּבּוּר כְּבָר עוֹמֵד לְהִתְפַּלֵּל עֲמִידָה – מִצַּד הַדִּין רַשָּׁאִית לִבְחֹר, וְהָעִקָּר: אֵיךְ שֶׁתְּכַוֵּן יוֹתֵר. הָרַב מֶלָמֵד מַמְלִיץ: לוֹמַר בִּרְכוֹת הַשַּׁחַר וְהַתּוֹרָה, לְדַלֵּג עַל פְּסוּקֵי דְּזִמְרָה וְעַל קְרִיאַת שְׁמַע וּבִרְכוֹתֶיהָ, וּלְהִתְפַּלֵּל עֲמִידָה עִם הַצִּבּוּר.
כָּךְ הִיא זוֹכָה לַעֲנוֹת אָמֵן וּקְדֻשָּׁה בַּחֲזָרַת הַשַּׁ״ץ וְלִשְׁמֹעַ קְרִיאַת הַתּוֹרָה. וְאִם יֵשׁ לָהּ עוֹד זְמַן – טוֹב שֶׁתֹּאמַר גַּם ״שְׁמַע״ וְ״אֱמֶת וְיַצִּיב״ לִפְנֵי הָעֲמִידָה.
💛 אַהֲבַת הַשֵּׁם: עִקַּר הַחוֹבָה שֶׁל בַּת הִיא תְּפִלַּת הָעֲמִידָה – וּכְשֶׁהִיא מִתְפַּלֶּלֶת אוֹתָהּ עִם הַצִּבּוּר, הִיא עוֹשָׂה אוֹתָהּ בַּדֶּרֶךְ הֲכִי מְהֻדֶּרֶת.
📖 פֶּרֶק כ״ב, הֲלָכָה ז׳

🍷📜 לְשׁוֹן חֲכָמִים: זָכוֹר וְשָׁמוֹר בְּדִבּוּר אֶחָד

״מִצְוַת ׳זָכוֹר׳ הֻקְּשָׁה לְמִצְוַת ׳שָׁמוֹר׳, שֶׁשְּׁתֵיהֶן נֶאֶמְרוּ בְּדִבּוּר אֶחָד וְלָכֵן דִּינֵיהֶן שָׁוֶה״.
📖 ״זָכוֹר״ – לִזְכֹּר אֶת הַשַּׁבָּת, בַּקִּדּוּשׁ. ״שָׁמוֹר״ – לְהִמָּנַע מִמְּלָאכָה. בְּדִבּוּר אֶחָד – ה׳ אָמַר אֶת שְׁנֵיהֶם יַחַד, בְּבַת אַחַת.
💛 אַהֲבַת הַשֵּׁם: ״מִצְוַת ׳זָכוֹר׳ עִנְיָנָהּ לִזְכֹּר אֶת יְסוֹדוֹת הָאֱמוּנָה״ – בְּרִיאַת הָעוֹלָם וִיצִיאַת מִצְרַיִם. 🧒 וְאַתְּ? בַּקִּדּוּשׁ – הַקְשִׁיבִי לְכָל מִלָּה וְעִנִי ״אָמֵן״.
📖 פֶּרֶק כ״ב, הֲלָכָה ח׳

🕯️״בּוֹאִי כַלָּה״ – וּמְחַכִּים לַקִּדּוּשׁ

מִשֶּׁנִּכְנְסָה שַׁבָּת – אָסוּר לֶאֱכֹל אוֹ לִשְׁתּוֹת לִפְנֵי הַקִּדּוּשׁ, וַאֲפִלּוּ מַיִם. אֲבָל מֻתָּר לִשְׁטֹף אֶת הַפֶּה בְּמַיִם אוֹ לִבְלֹעַ תְּרוּפָה. מִי שֶׁהִדְלִיקָה נֵרוֹת – קִבְּלָה שַׁבָּת בַּהַדְלָקָה, וּמֵאָז אֵינָהּ שׁוֹתָה עַד הַקִּדּוּשׁ.
בַּת שֶׁאֵינָהּ מַדְלִיקָה נֵרוֹת מְקַבֶּלֶת עָלֶיהָ תּוֹסֶפֶת שַׁבָּת בַּאֲמִירַת ״בּוֹאִי כַלָּה שַׁבָּת הַמַּלְכָּה״, וּמֵאוֹתָהּ שָׁעָה אֲסוּרָה בַּאֲכִילָה וּבִשְׁתִיָּה עַד הַקִּדּוּשׁ.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים אָסְרוּ לֶאֱכֹל לִפְנֵי הַקִּדּוּשׁ כְּדֵי שֶׁנְּמַהֵר לְקַיֵּם אֶת מִצְוַת ״זָכוֹר אֶת יוֹם הַשַּׁבָּת לְקַדְּשׁוֹ״. הַשַּׁבָּת הִיא מַלְכָּה – וּמְקַבְּלִים אוֹתָהּ בְּכָבוֹד.
📖 פֶּרֶק כ״ב, הֲלָכָה ט׳

🧠 בְּדֹק אֶת עַצְמְךָ

בְּשַׁבָּת בַּבֹּקֶר הִגִּיעָה אֱלִישֶׁבַע לְבֵית הַכְּנֶסֶת, וְהַצִּבּוּר כְּבָר מַתְחִיל עֲמִידָה. מָה הָרַב מֶלָמֵד מַמְלִיץ?
💡 מִצַּד הַדִּין הִיא רַשָּׁאִית לִבְחֹר אֵיךְ שֶׁתְּכַוֵּן יוֹתֵר. אֲבָל הָרַב מֶלָמֵד מַמְלִיץ: לוֹמַר בִּרְכוֹת הַשַּׁחַר וְהַתּוֹרָה, לְדַלֵּג עַל פְּסוּקֵי דְּזִמְרָה וּקְרִיאַת שְׁמַע, וּלְהִתְפַּלֵּל עֲמִידָה עִם הַצִּבּוּר – וְאָז לַעֲנוֹת בַּחֲזָרָה וְלִשְׁמֹעַ קְרִיאַת הַתּוֹרָה.
בַּת שֶׁבִּימֵי הַחֹל מִסְתַּפֶּקֶת בְּבִרְכוֹת הַשַּׁחַר וּבִרְכוֹת הַתּוֹרָה – מָה רָאוּי שֶׁתַּעֲשֶׂה בְּשַׁבָּת?
💡 גַּם מִי שֶׁבְּחֹל מִסְתַּפֶּקֶת בַּבְּרָכוֹת – רָאוּי שֶׁבְּשַׁבָּת תִּתְפַּלֵּל עֲמִידָה שֶׁל שַׁחֲרִית, כִּי בְּשַׁבָּת יֵשׁ בְּדֶרֶךְ כְּלָל יוֹתֵר זְמַן פָּנוּי. וּמוּסָף – נָשִׁים אֵינָן חַיָּבוֹת בּוֹ לְפִי הַהֲלָכָה, וְהָרוֹצָה מִתְפַּלֶּלֶת.
📜 לְשׁוֹן חֲכָמִים: מָה פֵּרוּשׁ ״זָכוֹר וְשָׁמוֹר בְּדִבּוּר אֶחָד נֶאֶמְרוּ״?
💡 ״זָכוֹר״ וְ״שָׁמוֹר״ נֶאֶמְרוּ בְּבַת אַחַת – ״וְלָכֵן דִּינֵיהֶן שָׁוֶה״, וּבָנוֹת חַיָּבוֹת גַּם בְּקִדּוּשׁ.

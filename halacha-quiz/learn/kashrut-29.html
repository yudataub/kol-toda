יַיִן וּמַשְׁקָאוֹת שֶׁל גּוֹיִם

🍽️ הִלְכוֹת כַּשְׁרוּת · פֶּרֶק כ״ט

🍇לָמָּה דַּוְקָא יַיִן?

הַיַּיִן הוּא הַמַּשְׁקֶה הַחָשׁוּב וְהַמְשַׂמֵּחַ בְּיוֹתֵר. לָכֵן מְקַדְּשִׁים עָלָיו בְּשַׁבָּת וּבְחַג, וּמְבָרְכִים עָלָיו בְּבְרִית מִילָה וּבְכָל שִׂמְחָה שֶׁל מִצְוָה. אֲבָל דַּוְקָא מִשּׁוּם שֶׁכֹּחוֹ גָּדוֹל, הוּא מְקָרֵב לְבָבוֹת יוֹתֵר מִכָּל מַאֲכָל – וְלָכֵן חֲכָמִים הֶחְמִירוּ בּוֹ בְּיוֹתֵר.
יַיִן שֶׁהֵכִין גּוֹי אָסוּר. וְלֹא רַק זֶה: גַּם יַיִן שֶׁל יְהוּדִי שֶׁגּוֹי נָגַע בּוֹ – נֶאֱסָר. וְכָל זֶה נוֹגֵעַ גַּם לְמִיץ עֲנָבִים.
💛 אַהֲבַת הַשֵּׁם: לְכָל כֹּחַ גָּדוֹל יֵשׁ שְׁנֵי צְדָדִים. כְּשֶׁהַיַּיִן מְשַׁמֵּשׁ לְקִדּוּשׁ וּלְשִׂמְחָה שֶׁל מִצְוָה – הוּא מְקָרֵב אוֹתָנוּ לַה׳.
📖 פֶּרֶק כ״ט, הֲלָכָה א׳; פֶּרֶק כ״ז, הֲלָכָה ב׳

✋מִי שֶׁאֵינוֹ עוֹבֵד עֲבוֹדָה זָרָה

יַיִן שֶׁנָּגַע בּוֹ גּוֹי עוֹבֵד עֲבוֹדָה זָרָה – אָסוּר גַּם בַּהֲנָאָה: אָסוּר לְמָכְרוֹ וְאָסוּר לָתֵת אוֹתוֹ בְּמַתָּנָה. אֲבָל גּוֹי שֶׁאֵינוֹ עוֹבֵד עֲבוֹדָה זָרָה, כְּמוֹ גּוֹי חִילוֹנִי – הַיַּיִן שֶׁנָּגַע בּוֹ אָסוּר רַק בִּשְׁתִיָּה, וּמֻתָּר בַּהֲנָאָה.
אִם נָגַע בְּיַיִן לִפְנֵי שֶׁבִּשְּׁלוּ אוֹתוֹ – הַבִּשּׁוּל לֹא מַצִּיל אוֹתוֹ, כִּי כְּבָר נֶאֱסָר.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים קָבְעוּ אֶת הַדִּינִים בְּדִיּוּק לְפִי הַחֲשָׁשׁ: הַחֻמְרָה הַגְּדוֹלָה בְּמָקוֹם שֶׁל עֲבוֹדָה זָרָה, וְחֻמְרָה קְטַנָּה יוֹתֵר כְּדֵי לִשְׁמֹר עַל הַיִּחוּד שֶׁלָּנוּ.
📖 פֶּרֶק כ״ט, הֲלָכוֹת א׳–ב׳, ה׳

🍷🔑 מוּשָּׂג: יַיִן נֶסֶךְ

כֵּיוָן שֶׁהַיַּיִן ״מְשַׂמֵּחַ וּמְקָרֵב לְבָבוֹת״, הֶחְמִירוּ חֲכָמִים בְּיֵין גּוֹיִם חֻמְרָה יְתֵרָה, מֵחֲשָׁשׁ לְקִרְבָה יְתֵרָה בֵּין יִשְׂרָאֵל לַגּוֹיִם.
📖 יַיִן נֶסֶךְ – יַיִן שֶׁנָּגַע בּוֹ עוֹבֵד עֲבוֹדָה זָרָה בְּכַוָּנָה לְנַסְּכוֹ לֶאֱלִיל, אָסוּר אֲפִלּוּ בַּהֲנָאָה.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים דָּאֲגוּ שֶׁדַּוְקָא בְּדָבָר הַמְּקָרֵב לְבָבוֹת, כְּמוֹ הַיַּיִן, נִשָּׁאֵר קָרוֹב לְה׳ וְלֹא נִגָּרֵר אַחַר דְּרָכִים זָרוֹת. 🧒 וְאַתָּה? כְּשֶׁאַתָּה שׁוֹתֶה יַיִן בְּקִדּוּשׁ – תִּזְכֹּר שֶׁגַּם הוּא דֶּרֶךְ לְהִתְקָרֵב לַה׳.
📖 פֶּרֶק כ״ט, הֲלָכָה א׳

🔥מְבֻשָּׁל וְחֹמֶץ

יַיִן מְבֻשָּׁל אֵינוֹ נֶאֱסָר בְּמַגַּע גּוֹי, כִּי עוֹבְדֵי הָאֱלִילִים לֹא נִסְּכוּ יַיִן מְבֻשָּׁל לָאֱלִילִים שֶׁלָּהֶם. גַּם יַיִן שֶׁנַּעֲשָׂה חֹמֶץ – אֵינוֹ נֶאֱסָר בְּמַגַּע גּוֹי.
עַל מִיץ עֲנָבִים וְיַיִן מְפֻסְטָרִים, שֶׁחִמְּמוּ אוֹתָם כְּדֵי שֶׁיִּשָּׁמְרוּ – נֶחְלְקוּ פּוֹסְקֵי זְמַנֵּנוּ אִם הֵם כִּמְבֻשָּׁלִים (רְאוּ בְּרָמָה ג׳).
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁמְּבִינִים אֶת הַטַּעַם שֶׁל גְּזֵרָה, מְבִינִים גַּם אֵיפֹה הִיא נִגְמֶרֶת. הַתּוֹרָה לֹא מַחְמִירָה סְתָם – הִיא שׁוֹמֶרֶת עָלֵינוּ בְּחָכְמָה.
📖 פֶּרֶק כ״ט, הֲלָכוֹת ד׳–ה׳

🔀שְׁתֵּי שִׁיטוֹת – מְקִילִים וּמְהַדְּרִים

מִיץ עֲנָבִים וְיַיִן מְפֻסְטָרִים: יֵשׁ מַחְמִירִים, כִּי טַעְמָם לֹא הִשְׁתַּנָּה. וְיֵשׁ סוֹבְרִים שֶׁהֵם כִּמְבֻשָּׁלִים, כִּי הִגִּיעוּ לְחֹם שֶׁמְּבַשֵּׁל אוֹתָם. הַהֲלָכָה כַּמְּקִילִים, וְהָרוֹצִים לְהַדֵּר – מַחְמִירִים. אֲבָל בְּמָקוֹם שֶׁעָלוּל לִהְיוֹת עֶלְבּוֹן – גַּם הָרוֹצֶה לְהַדֵּר יְקַל.
חָלָב שֶׁגּוֹיִם חָלְבוּ בִּשְׁבִיל חֲבָרוֹת גְּדוֹלוֹת וּמְפֻקָּחוֹת: לְדַעַת רֹב הַפּוֹסְקִים אֵין בּוֹ אִסּוּר, כִּי אֵין חֲשָׁשׁ שֶׁיְּעָרְבוּ בּוֹ חָלָב טָמֵא. וְיֵשׁ מַחְמִירִים. לְמַעֲשֶׂה הָעִקָּר כַּמְּקִילִים, וְיִשְׂרָאֵל קְדוֹשִׁים מְהַדְּרִים כְּשֶׁאֵין בְּכָךְ קֹשִׁי רַב.
💛 אַהֲבַת הַשֵּׁם: גַּם הַמְּקִילִים וְגַם הַמְּהַדְּרִים הוֹלְכִים בְּדַרְכֵי הַתּוֹרָה. לֹא מְבַזִּים אֶת מִי שֶׁנּוֹהֵג אַחֶרֶת – וְלֹא מַעֲלִיבִים אִישׁ בִּשְׁבִיל הִדּוּר.
📖 פֶּרֶק כ״ט, הֲלָכָה ה׳; פֶּרֶק ל׳, הֲלָכוֹת ד׳, ו׳

🍾בַּקְבּוּק סָגוּר וְחָתוּם

גּוֹי שֶׁהֵזִיז אוֹ נִעֵר בַּקְבּוּק יַיִן סָגוּר – אֲפִלּוּ בִּפְקָק פָּשׁוּט – הַיַּיִן אֵינוֹ נֶאֱסָר, כִּי אֵין דֶּרֶךְ לְנַסֵּךְ בִּכְלִי סָגוּר. בְּבָתֵּי הַחֲרֹשֶׁת חוֹתְמִים אֶת הַבַּקְבּוּקִים בְּפְקָק וְעוֹד עֲטִיפָה – שְׁנֵי חוֹתָמוֹת.
מִי שֶׁפּוֹעֵל גּוֹי נִשְׁאָר לְבַד בְּבֵיתוֹ, וְאוּלַי יִשְׁתֶּה מִן הַיַּיִן – יַקְפִּיד שֶׁהַבַּקְבּוּקִים יִהְיוּ חֲתוּמִים כַּדִּין, אוֹ יִנְעַל אוֹתָם בְּאָרוֹן. יַיִן מְבֻשָּׁל – אֵין צֹרֶךְ לִשְׁמֹר.
💛 אַהֲבַת הַשֵּׁם: הַשְּׁמִירָה עַל הַכַּשְׁרוּת דּוֹרֶשֶׁת מַחֲשָׁבָה מֵרֹאשׁ. מִי שֶׁמְּתַכְנֵן – שׁוֹמֵר בְּקַלּוּת וּבְשִׂמְחָה.
📖 פֶּרֶק כ״ט, הֲלָכוֹת ז׳, ט׳

🍽️מֶלְצַר, וּמִי שֶׁנּוֹגֵעַ לְהַכְעִיס

כְּשֶׁהַמֶּלְצַר בְּשִׂמְחָה אֵינוֹ יְהוּדִי, הַמִּנְהָג הוּא שֶׁהַסּוֹעֲדִים פּוֹתְחִים אֶת הַבַּקְבּוּק וּמוֹזְגִים לְעַצְמָם, כִּי קָשֶׁה לְפַקֵּחַ שֶׁהוּא לֹא יִגַּע בַּיַּיִן. וּבְדִיעֲבַד, אִם הַמֶּלְצַר מָזַג לָהֶם – הַיַּיִן כָּשֵׁר, כָּל עוֹד לֹא רָאוּ אוֹתוֹ נוֹגֵעַ בַּיַּיִן.
גּוֹי שֶׁנָּגַע בְּיַיִן כְּדֵי לְהַכְעִיס אֶת הַיְּהוּדִי וּלְהַפְסִיד אֶת יֵינוֹ – לֹא גָּזְרוּ עַל זֶה חֲכָמִים, וְהַיַּיִן מֻתָּר. וְאַדְּרַבָּה: יִשְׁתֶּה מִמֶּנּוּ לְפָנָיו, שֶׁיֵּדַע שֶׁלֹּא הִצְלִיחַ.
💛 אַהֲבַת הַשֵּׁם: הַהֲלָכָה לֹא נוֹתֶנֶת לְמִי שֶׁרוֹצֶה לְהַזִּיק לִשְׁלֹט בָּנוּ. אָנוּ שׁוֹמְרִים מִצְווֹת מִתּוֹךְ שִׂמְחָה – לֹא מִתּוֹךְ פַּחַד.
📖 פֶּרֶק כ״ט, הֲלָכוֹת ח׳, י׳

🧠 בְּדֹק אֶת עַצְמְךָ

לָמָּה הֶחְמִירוּ חֲכָמִים בְּיַיִן יוֹתֵר מִבִּשְׁאָר מַאַכְלֵי גּוֹיִם?
💡 כֹּחַ הַיַּיִן גָּדוֹל: הוּא מְשַׂמֵּחַ וּמְקָרֵב לְבָבוֹת. לָכֵן הוּא מַתְאִים לְשִׂמְחָה שֶׁל מִצְוָה – וּמִשּׁוּם כָּךְ גַּם הֶחְמִירוּ בּוֹ, כְּדֵי שֶׁלֹּא יְקָרֵב אוֹתָנוּ לְהִתְבּוֹלְלוּת.
פּוֹעֵל גּוֹי חִילוֹנִי, שֶׁאֵינוֹ עוֹבֵד עֲבוֹדָה זָרָה, נָגַע בְּיַיִן שֶׁל יְהוּדִי. מָה דִּין הַיַּיִן?
💡 דִּין גּוֹי שֶׁאֵינוֹ עוֹבֵד עֲבוֹדָה זָרָה קַל בְּדַרְגָּה אַחַת: הַיַּיִן אָסוּר בִּשְׁתִיָּה וּמֻתָּר בַּהֲנָאָה. וּבִשּׁוּל אַחֲרֵי הַנְּגִיעָה לֹא מַצִּיל אוֹתוֹ.
לָמָּה יַיִן מְבֻשָּׁל אֵינוֹ נֶאֱסָר כְּשֶׁגּוֹי נוֹגֵעַ בּוֹ?
💡 הַגְּזֵרָה נִקְבְּעָה עַל יַיִן שֶׁרָאוּי לְנִסּוּךְ. יַיִן מְבֻשָּׁל לֹא נִסְּכוּ לָאֱלִילִים, וְלָכֵן מַגַּע גּוֹי אֵינוֹ אוֹסֵר אוֹתוֹ.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

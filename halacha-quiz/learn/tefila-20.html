בִּרְכַּת כֹּהֲנִים

🙏 הִלְכוֹת תְּפִלָּה · פֶּרֶק כ׳

✋בִּרְכַּת כֹּהֲנִים

מִצְוָה מִן הַתּוֹרָה שֶׁהַכֹּהֲנִים יְבָרְכוּ אֶת יִשְׂרָאֵל בְּכָל יוֹם: "יְבָרֶכְךָ ה׳ וְיִשְׁמְרֶךָ…". גַּם מִי שֶׁעוֹמֵד מוּל הַכֹּהֲנִים בִּשְׁתִיקָה וּבְכַוָּנָה וְעוֹנֶה אָמֵן – שֻׁתָּף בְּקִיּוּם הַמִּצְוָה.
בִּזְמַן הַבְּרָכָה עוֹמְדִים מוּל הַכֹּהֲנִים וּמְכַוְּנִים לַבְּרָכָה, וְלֹא מַבִּיטִים בַּכֹּהֲנִים וְלֹא בְּשׁוּם דָּבָר אַחֵר – כְּדֵי לֹא לְהַסִּיחַ אֶת הַדַּעַת.
💛 אַהֲבַת הַשֵּׁם: בְּכָל יוֹם מֵחָדָשׁ אָנוּ עוֹמְדִים לְקַבֵּל בְּרָכָה, וְזוֹכְרִים שֶׁה׳ הוּא הַמְבָרֵךְ אֶת עַמּוֹ בְּאַהֲבָה. כְּכָל שֶׁנִּרְצֶה יוֹתֵר אֶת הַבְּרָכָה – כָּךְ נִהְיֶה מוּכָנִים יוֹתֵר לְקַבֵּל אוֹתָהּ.
📖 פֶּרֶק כ׳, הֲלָכוֹת א׳–ב׳

🌟ה׳ הוּא הַמְבָרֵךְ

הַמִּצְוָה לְבָרֵךְ חָלָה עַל כָּל הַכֹּהֲנִים – צַדִּיקִים וְגַם מִי שֶׁחוֹטֵא. כִּי בֶּאֱמֶת ה׳ הוּא הַמְבָרֵךְ אֶת עַמּוֹ, וְהַכֹּהֲנִים רַק מְבַטְּאִים אֶת הַבְּרָכָה בִּשְׂפָתָם. הַבְּרָכָה לֹא תְּלוּיָה בְּצִדְקוּתוֹ שֶׁל הַכֹּהֵן, אֶלָּא בִּרְצוֹן ה׳ לְבָרֵךְ אֶת יִשְׂרָאֵל.
כֹּהֵן שֶׁקָּרְאוּ לוֹ לַעֲלוֹת לַדּוּכָן וְסֵרֵב – נֶחְשָׁב לוֹ כְּאִלּוּ בִּטֵּל שָׁלֹשׁ מִצְווֹת, כִּי ה׳ מִתְאַוֶּה לְבָרֵךְ אֶת יִשְׂרָאֵל. וְאִם כְּבָר בֵּרֵךְ הַיּוֹם וְקָרְאוּ לוֹ בְּמִנְיָן אַחֵר – מִצְוָה שֶׁיַּעֲלֶה שׁוּב.
💛 אַהֲבַת הַשֵּׁם: "וְשָׂמוּ אֶת שְׁמִי עַל בְּנֵי יִשְׂרָאֵל וַאֲנִי אֲבָרְכֵם". ה׳ עַצְמוֹ רוֹצֶה לְבָרֵךְ אוֹתָנוּ – וְהוּא מְחַכֶּה שֶׁנַּעֲמֹד מוּלוֹ וּנְקַבֵּל.
📖 פֶּרֶק כ׳, הֲלָכוֹת א׳, י׳

🪜🔑 מוּשָּׂג: דּוּכָן

״כָּל כֹּהֵן שֶׁקְּרָאוּ לוֹ לַעֲלוֹת לַדּוּכָן לְבָרֵךְ אֶת יִשְׂרָאֵל״.
דּוּכָן – הַמָּקוֹם שֶׁלִּפְנֵי הָאָרוֹן, שֶׁעָלָיו עוֹלִים הַכֹּהֲנִים לְבָרֵךְ אֶת הָעָם.
💛 אַהֲבַת הַשֵּׁם: ה׳ ״מִתְאַוֶּה לְבָרֵךְ אֶת יִשְׂרָאֵל״, וְלָכֵן יֵשׁ לְכָךְ מָקוֹם קָבוּעַ.
📖 פֶּרֶק כ׳, הֲלָכָה א׳

🤲📜 לְשׁוֹן חֲכָמִים: נְשִׂיאַת כַּפַּיִם

״אֵין מְבָרְכִין אֶלָּא בִּלְשׁוֹן הַקֹּדֶשׁ וּבַעֲמִידָה וּבִנְשִׂיאַת כַּפַּיִם וּבְקוֹל רָם״.
📖 נְשִׂיאָה – הֲרָמָה. כַּפַּיִם – יָדַיִם: הַכֹּהֲנִים מְרִימִים אֶת יְדֵיהֶם וּמְבָרְכִים.
💛 אַהֲבַת הַשֵּׁם: ״מִלְּשׁוֹן הַפְּסוּקִים מַשְׁמָע שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא מִתְאַוֶּה לְבָרֵךְ אֶת יִשְׂרָאֵל״ – ה׳ רוֹצֶה לְבָרֵךְ אוֹתְךָ. 🧒 וְאַתָּה? בְּבִרְכַּת כֹּהֲנִים – עֲמֹד מוּל הַכֹּהֲנִים, שְׁתֹק, וַעֲנֵה ״אָמֵן״ בְּכַוָּנָה.
📖 פֶּרֶק כ׳, הֲלָכוֹת א׳, ד׳

🪜📜 לְשׁוֹן חֲכָמִים: דּוּכָן

״כָּל כֹּהֵן שֶׁקְּרָאוּ לוֹ לַעֲלוֹת לַדּוּכָן לְבָרֵךְ אֶת יִשְׂרָאֵל וְסֵרֵב... נֶחְשָׁב לוֹ הַדָּבָר כְּאִלּוּ בִּטֵּל שָׁלֹשׁ מִצְווֹת מִן הַתּוֹרָה״.
📖 דּוּכָן – הַמָּקוֹם שֶׁלִּפְנֵי אֲרוֹן הַקֹּדֶשׁ, שֶׁעָלָיו עוֹלִים הַכֹּהֲנִים לְבָרֵךְ.
💛 אַהֲבַת הַשֵּׁם: כָּל כָּךְ חָשׁוּב לַה׳ שֶׁהַבְּרָכָה תַּגִּיעַ אֵלֵינוּ – עַד שֶׁכֹּהֵן שֶׁלֹּא עוֹלֶה נֶחְשָׁב כְּמִי שֶׁבִּטֵּל שָׁלֹשׁ מִצְווֹת. 🧒 וְאַתָּה? כְּשֶׁהַכֹּהֲנִים עוֹלִים לַדּוּכָן – עֲמֹד מוּלָם וְהַקְשֵׁב.
📖 פֶּרֶק כ׳, הֲלָכָה א׳

📍אֵיפֹה עוֹמְדִים?

הַמִּתְבָּרְכִים עוֹמְדִים מוּל הַכֹּהֲנִים, פָּנִים מוּל פָּנִים. מִי שֶׁעוֹמֵד מֵאֲחוֹרֵי הַכֹּהֲנִים – אֵינוֹ בַּבְּרָכָה, וּמִי שֶׁעוֹמֵד מַמָּשׁ בַּצַּד – יִפְנֶה אֲלֵיהֶם. גַּם כְּשֶׁאֲנָשִׁים גְּבוֹהִים אוֹ עַמּוּד עוֹמְדִים בֵּינְךָ לְבֵין הַכֹּהֲנִים, אַתָּה בַּבְּרָכָה. אֲבָל מִי שֶׁמַּפְנֶה אֶת גַּבּוֹ – אֵינוֹ בַּבְּרָכָה.
נָשִׁים וִילָדִים שֶׁאֵינָם צְרִיכִים לָבוֹא לְבֵית הַכְּנֶסֶת, וְגַם מִי שֶׁלֹּא יָכוֹל הָיָה לָבוֹא – נִכְלָלִים בַּבְּרָכָה שֶׁל כָּל יִשְׂרָאֵל. רַק מִי שֶׁיָּכוֹל לָבוֹא וּמִתְרַשֵּׁל – אֵינוֹ בַּבְּרָכָה.
💛 אַהֲבַת הַשֵּׁם: הַבְּרָכָה נֶאֱמֶרֶת כְּמוֹ אָדָם שֶׁמְּדַבֵּר עִם חֲבֵרוֹ – פָּנִים אֶל פָּנִים. ה׳ פּוֹנֶה אֵלֵינוּ בְּאַהֲבָה, וְגַם אֲנַחְנוּ פּוֹנִים אֵלָיו.
📖 פֶּרֶק כ׳, הֲלָכָה ג׳

📜אַרְבַּע הֲלָכוֹת מִן הַתּוֹרָה

בְּבִרְכַּת כֹּהֲנִים יֵשׁ אַרְבַּע הֲלָכוֹת מִן הַתּוֹרָה: מְבָרְכִים בִּלְשׁוֹן הַקֹּדֶשׁ, בַּעֲמִידָה, בִּנְשִׂיאַת כַּפַּיִם וּבְקוֹל רָם. קְרִיאַת שְׁמַע וּתְפִלָּה אֶפְשָׁר לוֹמַר גַּם בְּשָׂפָה אַחֶרֶת, אֲבָל בִּרְכַּת כֹּהֲנִים – דַּוְקָא בִּלְשׁוֹן הַקֹּדֶשׁ, כְּמוֹ שֶׁנֶּאֱמַר "כֹּה תְבָרֲכוּ".
בִּרְכַּת כֹּהֲנִים הִיא בְּרָכָה שֶׁבָּאָה מִן ה׳ אֵלֵינוּ, וּבִלְשׁוֹן הַקֹּדֶשׁ ה׳ מְגַלֶּה אֶת רְצוֹנוֹ. לָכֵן הַכֹּהֲנִים אוֹמְרִים אוֹתָהּ בְּדִיּוּק כְּמוֹ שֶׁהִיא כְּתוּבָה בַּתּוֹרָה.
💛 אַהֲבַת הַשֵּׁם: רַבִּי נַחְמָן מִבְּרֶסְלֶב הִסְבִּיר: הַיָּדַיִם סְמוּכוֹת לַלֵּב. כְּשֶׁהַכֹּהֲנִים פּוֹרְשִׂים יְדֵיהֶם, הֵם מְבַטְּאִים אֶת אַהֲבַת הַלֵּב לְעַם יִשְׂרָאֵל.
📖 פֶּרֶק כ׳, הֲלָכָה ד׳

🌅בְּרָכָה שֶׁל יוֹם

בְּבֵית הַמִּקְדָּשׁ הַכֹּהֲנִים בֵּרְכוּ אֶת הָעָם אַחֲרֵי עֲבוֹדַת הַקָּרְבָּנוֹת, וְהַקָּרְבָּנוֹת קְרֵבִים רַק בַּיּוֹם. לָכֵן אֵין בִּרְכַּת כֹּהֲנִים בִּתְפִלַּת עַרְבִית. וּבְמִנְחָה לֹא מְבָרְכִים, כִּי זֶה אַחֲרֵי הָאֲרוּחָה וְאוּלַי הַכֹּהֵן שָׁתָה יַיִן. אֲבָל בִּנְעִילָה וּבְמִנְחָה שֶׁל תַּעֲנִית צִבּוּר – מְבָרְכִים.
הַכֹּהֲנִים מַתְחִילִים לָלֶכֶת לַדּוּכָן בְּבִרְכַּת "רְצֵה", שֶׁהִיא הַבְּרָכָה עַל הַחְזָרַת הָעֲבוֹדָה לְבֵית הַמִּקְדָּשׁ.
💛 אַהֲבַת הַשֵּׁם: הַתְּפִלָּה הִיא בִּמְקוֹם הַקָּרְבָּנוֹת. וּכְמוֹ שֶׁאַחֲרֵי הַקָּרְבָּנוֹת זָכוּ לִבְרָכָה, כָּךְ אַחֲרֵי שֶׁאָנוּ מִתְקָרְבִים לַה׳ בִּתְפִלָּה – אָנוּ רְאוּיִים לְקַבֵּל אֶת בִּרְכָתוֹ.
📖 פֶּרֶק כ׳, הֲלָכָה ה׳

💧הַכֹּהֲנִים מִתְכּוֹנְנִים

לִפְנֵי בִּרְכַּת כֹּהֲנִים הַכֹּהֲנִים נוֹטְלִים יְדֵיהֶם עַד הַפֶּרֶק – הַמָּקוֹם שֶׁבּוֹ כַּף הַיָּד מִתְחַבֶּרֶת לַזְּרוֹעַ. נוֹהֲגִים שֶׁלֵּוִי יוֹצֵק אֶת הַמַּיִם עַל יְדֵיהֶם. אִם אֵין לֵוִי – רָצוּי שֶׁבְּכוֹר יִטֹּל אֶת יְדֵיהֶם, וְאִם אֵין גַּם בְּכוֹר – הַכֹּהֵן נוֹטֵל בְּעַצְמוֹ. עַל הַנְּטִילָה הַזּוֹ לֹא מְבָרְכִים.
רַבָּן יוֹחָנָן בֶּן זַכַּאי תִּקֵּן שֶׁהַכֹּהֲנִים לֹא יַעֲלוּ לַדּוּכָן בְּנַעֲלַיִם אוֹ בְּסַנְדָּלִים, מִשּׁוּם כְּבוֹד הַצִּבּוּר. בְּמָקוֹם שֶׁלֹּא נָהֲגוּ לַעֲמֹד יְחֵפִים לִפְנֵי אֲנָשִׁים מְכֻבָּדִים – עוֹלִים בְּגַרְבַּיִם.
💛 אַהֲבַת הַשֵּׁם: "שְׂאוּ יְדֵכֶם קֹדֶשׁ וּבָרְכוּ אֶת ה׳". הַלֵּוִי שֶׁיּוֹצֵק מַיִם מוֹסִיף קְדֻשָּׁה עַל קְדֻשָּׁה, וְכָךְ יוֹצֵאת הַבְּרָכָה מִיָּדַיִם קְדוֹשׁוֹת.
📖 פֶּרֶק כ׳, הֲלָכוֹת ו׳, ח׳

📣"כֹּהֲנִים!"

הַכֹּהֲנִים לֹא מַתְחִילִים לְבָרֵךְ עַד שֶׁהַחַזָּן אוֹ הַגַּבַּאי קוֹרֵא "כֹּהֲנִים", כְּמוֹ שֶׁנֶּאֱמַר לְמֹשֶׁה: "אָמוֹר לָהֶם". וְאִם עוֹלֶה רַק כֹּהֵן אֶחָד – לֹא קוֹרְאִים "כֹּהֲנִים". גַּם אֶת הַבְּרָכָה עַצְמָהּ הַחַזָּן מַקְרִיא מִלָּה מִלָּה, וְהַכֹּהֲנִים חוֹזְרִים אַחֲרָיו.
הַחַזָּן מַקְרִיא בְּקוֹל בָּרוּר, מְעַט חַלָּשׁ מִקּוֹל הַכֹּהֲנִים. וְהַקָּהָל עוֹנֶה "אָמֵן" אַחֲרֵי כָּל פָּסוּק.
💛 אַהֲבַת הַשֵּׁם: קֹדֶם אֲנַחְנוּ מְבַקְּשִׁים – "כֹּהֲנִים!" – וְרַק אַחַר כָּךְ ה׳ מַשְׁפִּיעַ. כְּשֶׁאָנוּ מִתְעוֹרְרִים מִלְּמַטָּה לִרְצוֹת אֶת הַבְּרָכָה, מִתְעוֹרֵר רָצוֹן מִלְמַעְלָה לָתֵת לָנוּ אוֹתָהּ.
📖 פֶּרֶק כ׳, הֲלָכָה ז׳

💛לְבָרֵךְ בְּאַהֲבָה

הַכֹּהֲנִים מְבָרְכִים: "אֲשֶׁר קִדְּשָׁנוּ בִּקְדֻשָּׁתוֹ שֶׁל אַהֲרֹן וְצִוָּנוּ לְבָרֵךְ אֶת עַמּוֹ יִשְׂרָאֵל בְּאַהֲבָה". לָכֵן כֹּהֵן שֶׁשּׂוֹנֵא אֶת הַצִּבּוּר, וַאֲפִלּוּ אָדָם אֶחָד מֵהַצִּבּוּר, אָסוּר לוֹ לְבָרֵךְ – עַד שֶׁיִּמְחֶה אֶת הַשִּׂנְאָה מִלִּבּוֹ, אוֹ שֶׁיֵּצֵא לִפְנֵי בִּרְכַּת כֹּהֲנִים.
אַהֲרֹן הַכֹּהֵן הָיָה "אוֹהֵב שָׁלוֹם וְרוֹדֵף שָׁלוֹם, אוֹהֵב אֶת הַבְּרִיּוֹת וּמְקָרְבָן לַתּוֹרָה". הַכֹּהֲנִים תַּפְקִידָם לְלַמֵּד תּוֹרָה וְלִהְיוֹת אַנְשֵׁי חֶסֶד וְשָׁלוֹם.
💛 אַהֲבַת הַשֵּׁם: בְּרָכָה אֲמִתִּית יוֹצֵאת רַק מִלֵּב אוֹהֵב. כְּשֶׁאָנוּ אוֹהֲבִים זֶה אֶת זֶה, אָנוּ פּוֹתְחִים שַׁעַר לִבְרָכָה שֶׁל ה׳.
📖 פֶּרֶק כ׳, הֲלָכָה ט׳

🧠 בְּדֹק אֶת עַצְמְךָ

קְרִיאַת שְׁמַע אֶפְשָׁר לוֹמַר בְּכָל שָׂפָה. וּבִרְכַּת כֹּהֲנִים?
💡 "כֹּה תְבָרֲכוּ" – בַּלָּשׁוֹן הַזּוֹ שֶׁכְּתוּבָה בַּתּוֹרָה. בִּרְכַּת כֹּהֲנִים בָּאָה מִן ה׳, וְלָכֵן אוֹמְרִים אוֹתָהּ בְּדִיּוּק בִּלְשׁוֹן הַקֹּדֶשׁ.
בַּמִּנְיָן הַיּוֹם יֵשׁ רַק כֹּהֵן אֶחָד. מָה הַדִּין?
💡 "אָמוֹר לָהֶם" – לְשׁוֹן רַבִּים, וְלָכֵן קוֹרְאִים "כֹּהֲנִים" רַק כְּשֶׁיֵּשׁ לְפָחוֹת שְׁנַיִם. כֹּהֵן אֶחָד עוֹלֶה וּמְבָרֵךְ בְּלִי הַקְּרִיאָה.
כֹּהֵן כָּעַס מְאוֹד עַל אָדָם אֶחָד בְּבֵית הַכְּנֶסֶת וְשׂוֹנֵא אוֹתוֹ. מָה עָלָיו לַעֲשׂוֹת?
💡 הַכֹּהֲנִים מְבָרְכִים "בְּאַהֲבָה", וְכֹהֵן שֶׁשּׂוֹנֵא אֲפִלּוּ אָדָם אֶחָד – אָסוּר לוֹ לְבָרֵךְ. עָלָיו לִמְחוֹת אֶת הַשִּׂנְאָה מִלִּבּוֹ, אוֹ לָצֵאת לִפְנֵי בִּרְכַּת כֹּהֲנִים.

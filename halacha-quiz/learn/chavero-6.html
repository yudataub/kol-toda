הִלְכוֹת צְדָקָה

🤝 בֵּין אָדָם לַחֲבֵרוֹ · פֶּרֶק ו׳

🪙מִצְוַת הַצְּדָקָה

מִצְוָה לָתֵת צְדָקָה לְכָל יְהוּדִי שֶׁצָּרִיךְ. חֲכָמִים אָמְרוּ: "שְׁקוּלָה צְדָקָה כְּנֶגֶד כָּל הַמִּצְווֹת". וּמִי שֶׁיָּכוֹל לָתֵת וְאֵינוֹ נוֹתֵן – עוֹבֵר גַּם עַל "לֹא תְאַמֵּץ אֶת לְבָבְךָ".
הַמַּעֲלָה הַגְּבוֹהָה בְּיוֹתֵר: לַעֲזֹר לֶעָנִי לַעֲמֹד עַל רַגְלָיו – לִמְצֹא לוֹ עֲבוֹדָה אוֹ לְלַמֵּד אוֹתוֹ מִקְצוֹעַ. וְכָל צְדָקָה – כְּכָל שֶׁהִיא נִתֶּנֶת בְּדֶרֶךְ מְכֻבֶּדֶת יוֹתֵר, כָּךְ הִיא מְעֻלָּה יוֹתֵר.
💛 אַהֲבַת הַשֵּׁם: הַנּוֹתֵן צְדָקָה דּוֹמֶה לְהַקָּדוֹשׁ בָּרוּךְ הוּא, שֶׁזָּן וּמְפַרְנֵס אֶת כָּל הַבְּרוּאִים. כָּל מַטְבֵּעַ קָטָן מְתַקֵּן אֶת הָעוֹלָם וּמְקָרֵב אֶת הַגְּאֻלָּה.
📖 פֶּרֶק ו׳, הֲלָכָה א׳

🪜🔑 מוּשָּׂג: מַעְלַת הַצְּדָקָה הַגְּבוֹהָה

״הַמַּעֲלָה הַגְּבוֹהָה בַּצְּדָקָה הִיא לְסַיֵּעַ לֶעָנִי לַעְמֹד עַל רַגְלָיו בְּכֹחוֹת עַצְמוֹ... הָעוֹזֵר לֶעָנִי לִמְצֹא עֲבוֹדָה מְקַיֵּם אֶת הַצְּדָקָה בַּדֶּרֶךְ הַמְעֻלָּה בְּיוֹתֵר״.
📖 הַמַּעֲלָה הַגְּבוֹהָה – לֹא רַק לָתֵת כֶּסֶף, אֶלָּא לְעֲזֹר לֶעָנִי לַעְמֹד בְּעַצְמוֹ.
💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁהֶעָנִי יִשָּׁאֵר בְּכָבוֹד, וְלֹא יִצְטָרֵךְ לְבַקֵּשׁ יוֹתֵר. 🧒 וְאַתָּה? עֲזֹר לֶחָבֵר לִלְמֹד מַשֶּׁהוּ שֶׁהוּא לֹא יוֹדֵעַ – זֶה נוֹתֵן לוֹ כֹּחַ לְהִסְתַּדֵּר בְּעַצְמוֹ.
📖 פֶּרֶק ו׳, הֲלָכָה א׳

🪜🔑 מוּשָּׂג: לֹא כָּל הָעֵזְרוֹת שָׁווֹת

״וְכַאֲשֶׁר אֵין אֶפְשָׁרוּת לַעְזֹר לֶעָנִי לַעְמֹד עַל רַגְלָיו... אֲזַי הַמִּצְוָה לִדְאֹג לְקִיּוּמוֹ. וְכָל שֶׁנּוֹתְנִים לוֹ אֶת הָעֶזְרָה בְּדֶרֶךְ מְכֻבָּדָה יוֹתֵר – כֵּן הַצְּדָקָה מְעֻלָּה יוֹתֵר״.
📖 אֲפִלּוּ בֵּין נוֹתְנֵי צְדָקָה יֵשׁ דַּרְגּוֹת – כֵּן מְכֻבָּד יוֹתֵר, כֵּן גָּבוֹהַ יוֹתֵר.
💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה לֹא רַק שֶׁהֶעָנִי יְקַבֵּל, אֶלָּא שֶׁהוּא יְקַבֵּל בְּכָבוֹד. 🧒 וְאַתָּה? כְּשֶׁאַתָּה נוֹתֵן – תֵּן בְּצִנְעָה, בְּלִי לְבַיֵּשׁ.
📖 פֶּרֶק ו׳, הֲלָכָה ב׳

🤲צְדָקָה בְּחָכְמָה

הַדֶּרֶךְ הַנְּכוֹנָה הִיא לָתֵת צְדָקָה דֶּרֶךְ גַּבָּאֵי צְדָקָה חֲכָמִים וַהֲגוּנִים. הֵם מַכִּירִים אֶת הָעֲנִיִּים, יוֹדְעִים מִי בֶּאֱמֶת צָרִיךְ, וְעוֹזְרִים בְּדֶרֶךְ שֶׁמְּעוֹדֶדֶת לַעֲבֹד.
וְעָנִי שֶׁמִּתְבַּיֵּשׁ לְקַבֵּל עֶזְרָה – מְפַיְּסִים אוֹתוֹ בִּדְבָרִים: הַכֹּל מִן הַשָּׁמַיִם, וְגַלְגַּל חוֹזֵר בָּעוֹלָם.
💛 אַהֲבַת הַשֵּׁם: חֶסֶד אֲמִתִּי הוּא לֹא רַק לָתֵת כֶּסֶף – אֶלָּא לַחְשֹׁב מָה בֶּאֱמֶת טוֹב לָאַחֵר. כָּךְ ה׳ עוֹשֶׂה עִמָּנוּ חֶסֶד: בְּחָכְמָה וּבְאַהֲבָה.
📖 פֶּרֶק ו׳, הֲלָכָה ג׳

🚪עָנִי שֶׁדּוֹפֵק בַּדֶּלֶת

לְעָנִי שֶׁעוֹבֵר מִבַּיִת לְבַיִת וּמְבַקֵּשׁ צְדָקָה, לֹא נוֹתְנִים מַתָּנָה גְּדוֹלָה – אֲבָל גַּם לֹא שׁוֹלְחִים אוֹתוֹ בְּלִי כְּלוּם. נוֹתְנִים לוֹ מַתָּנָה קְטַנָּה, כְּמוֹ מַטְבֵּעַ.
אֶת עִקַּר הַמַּעֲשֵׂר נוֹתְנִים לְפִי שִׁקּוּל דַּעַת – לָעֲנִיִּים הַקְּרוֹבִים וְהַשְּׁכֵנִים, וּלְקֻפַּת הַצְּדָקָה שֶׁל הַמָּקוֹם.
💛 אַהֲבַת הַשֵּׁם: גַּם מַטְבֵּעַ קָטָן, כְּשֶׁהוּא נִתָּן בְּחִיּוּךְ, אוֹמֵר לֶעָנִי: אַתָּה לֹא לְבַד. כָּךְ אָנוּ פּוֹתְחִים אֶת הַיָּד – וְגַם אֶת הַלֵּב.
📖 פֶּרֶק ו׳, הֲלָכָה ד׳

📏כַּמָּה נוֹתְנִים?

מִדָּה בֵּינוֹנִית – לָתֵת מַעֲשֵׂר, עֲשִׂירִית מִן הָרְוָחִים. מִצְוָה מִן הַמֻּבְחָר – לָתֵת חֲמִישִׁית. וּמִי שֶׁנּוֹתֵן פָּחוֹת מִמַּעֲשֵׂר – נוֹתֵן בְּעַיִן רָעָה.
חֲכָמִים דָּרְשׁוּ: "עַשֵּׂר תְּעַשֵּׂר" – עַשֵּׂר בִּשְׁבִיל שֶׁתִּתְעַשֵּׁר. וּבַשֻּׁלְחָן עָרוּךְ נִפְסַק: "לְעוֹלָם אֵין אָדָם מַעֲנִי מִן הַצְּדָקָה".
💛 אַהֲבַת הַשֵּׁם: הַנּוֹתֵן אֵינוֹ מַפְסִיד. ה׳ שׁוֹלֵחַ בְּרָכָה בְּמַעֲשֵׂה יָדָיו, וְאָז הוּא יָכוֹל לָתֵת עוֹד – וְלִהְיוֹת שֻׁתָּף בְּתִקּוּן הָעוֹלָם.
📖 פֶּרֶק ו׳, הֲלָכוֹת ז׳–ח׳

💰🔑 מוּשָּׂג: מַעְשֵׂר כְּסָפִים

״מִדָּה בֵּינוֹנִית לָתֵת מַעְשֵׂר מִן הָרְוָוחִים... וְהַנּוֹתֵן פָּחוֹת, מְקַיֵּם אֶת מִצְוַת הַצְּדָקָה כְּמִדַּת עַיִן רָעָה״.
📖 מַעְשֵׂר כְּסָפִים – לֶתֵת עֲשִׂירִית מֵהַכֶּסֶף שֶׁמַּרְוִיחִים לִצְדָקָה.
💛 אַהֲבַת הַשֵּׁם: מִי שֶׁמַּפְרִיש מַעְשֵׂר קוֹבֵעַ שֶׁהַכֶּסֶף שֶׁלּוֹ הוּא גַּם מַתָּנָה מֵה׳, לֹא רַק שֶׁלּוֹ. 🧒 וְאַתָּה? קִבַּלְתָּ דְּמֵי כִּיס? הַפְרֵש מִמֶּנּוּ חֵלֶק קָטָן לִצְדָקָה.
📖 פֶּרֶק ו׳, הֲלָכָה ז׳

💰מַעֲשֵׂר מִדְּמֵי הַכִּיס

יֶלֶד אוֹ יַלְדָּה שֶׁמְּקַבְּלִים דְּמֵי כִּיס לְשִׁמּוּשׁ חָפְשִׁי – מִתְחַנְּכִים לְהַפְרִישׁ מֵהֶם מַעֲשֵׂר, אֶחָד מִכָּל עֲשָׂרָה. וְכָךְ גַּם מִכֶּסֶף שֶׁקִּבְּלוּ בְּמַתָּנָה, אַף שֶׁהַהוֹרִים כְּבָר עִשְּׂרוּ אוֹתוֹ.
כֶּסֶף שֶׁהַהוֹרִים נָתְנוּ לְדָבָר מְסֻיָּם, כְּמוֹ נְסִיעָה אוֹ בֶּגֶד – אֵין צָרִיךְ לְעַשֵּׂר. אֲבָל אִם נִשְׁאַר עֹדֶף, וְהַהוֹרִים מַרְשִׁים לְהִשְׁתַּמֵּשׁ בּוֹ כִּרְצוֹנְךָ – מְעַשְּׂרִים אֶת הָעֹדֶף.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאַתָּה שָׂם בַּקֻּפָּה חֵלֶק מִן הַכֶּסֶף שֶׁלְּךָ, אַתָּה אוֹמֵר: הַכֹּל מֵאֵת ה׳, וַאֲנִי שָׂמֵחַ לְחַלֵּק. זוֹ הֲכָנָה נִפְלָאָה לְחַיִּים שֶׁל נְתִינָה.
📖 פֶּרֶק ו׳, הֲלָכָה ט׳

🧾מִמָּה מְעַשְּׂרִים?

מַעֲשֵׂר כְּסָפִים מַפְרִישִׁים מִכָּל כֶּסֶף שֶׁמַּגִּיעַ לָאָדָם – מִשְׂכֹּרֶת, יְרֻשָּׁה אוֹ מַתָּנָה. אֲבָל רַק מִכֶּסֶף, וְלֹא מִמַּתָּנָה שֶׁל חֵפֶץ. וְכֶסֶף שֶׁנִּתַּן לְדָבָר מְסֻיָּם – מְעַשְּׂרִים רַק מִן הָעֹדֶף שֶׁמֻּתָּר לְהִשְׁתַּמֵּשׁ בּוֹ כִּרְצוֹנְךָ.
יֶלֶד שֶׁמְּקַבֵּל דְּמֵי כִּיס כְּבָר אַחְרַאי עַל הַכֶּסֶף שֶׁלּוֹ – וְלָכֵן הוּא מִתְחַנֵּךְ לְעַשֵּׂר, גַּם אִם הַהוֹרִים כְּבָר עִשְּׂרוּ אֶת הַכֶּסֶף לִפְנֵי שֶׁנָּתְנוּ לוֹ.
💛 אַהֲבַת הַשֵּׁם: הַמַּעֲשֵׂר מְלַמֵּד שֶׁכָּל שֶׁקֶל שֶׁמַּגִּיעַ אֵלֵינוּ הוּא מַתָּנָה מִשָּׁמַיִם. מִי שֶׁמִּתְרַגֵּל לְכָךְ מִקָּטְנוּ – גָּדֵל לִהְיוֹת יְהוּדִי שֶׁלִּבּוֹ פָּתוּחַ.
📖 פֶּרֶק ו׳, הֲלָכָה ט׳

📒חֶשְׁבּוֹן הַמַּעֲשֵׂר

טוֹב לְחַשֵּׁב אֶת הַמַּעֲשֵׂר בְּדִיּוּק וְלִרְשֹׁם אוֹתוֹ בְּמַחְבֶּרֶת אוֹ בְּפִנְקָס. מִי שֶׁחַיָּב כֶּסֶף לַחֲבֵרוֹ, וְאֵינוֹ יָכוֹל לְשַׁלֵּם גַּם אֶת הַחוֹב וְגַם אֶת הַמַּעֲשֵׂר – מְשַׁלֵּם קֹדֶם אֶת הַחוֹב, וְרוֹשֵׁם כַּמָּה הוּא חַיָּב לַמַּעֲשֵׂר.
בְּכֶסֶף הַמַּעֲשֵׂר לֹא קוֹנִים דָּבָר שֶׁאַתָּה עַצְמְךָ חַיָּב בּוֹ – כְּמוֹ תְּפִלִּין לְעַצְמְךָ, אוֹ מַתָּנוֹת לָאֶבְיוֹנִים שֶׁחוֹבָה לָתֵת בְּפוּרִים. וּלְכַתְּחִלָּה נוֹתְנִים אוֹתוֹ לָעֲנִיִּים וּלְלוֹמְדֵי תּוֹרָה.
💛 אַהֲבַת הַשֵּׁם: הַמַּעֲשֵׂר אֵינוֹ סְתָם הוֹצָאָה – הוּא הַחֵלֶק שֶׁה׳ הִפְקִיד אֶצְלְךָ בִּשְׁבִיל אֲחֵרִים. מִי שֶׁשּׁוֹמֵר עָלָיו בְּדִיּוּק, מַרְאֶה שֶׁהוּא נֶאֱמָן לַה׳.
📖 פֶּרֶק ו׳, הֲלָכוֹת י׳–י״ב

🧠 בְּדֹק אֶת עַצְמְךָ

אַבָּא נָתַן לְאֵלִי עֶשְׂרִים שְׁקָלִים לִנְסִיעוֹת, וְנִשְׁאֲרוּ לוֹ חֲמִשָּׁה. אַבָּא אָמַר: "הָעֹדֶף שֶׁלְּךָ". מִמָּה מְעַשְּׂרִים?
💡 כֶּסֶף שֶׁנִּתַּן לִנְסִיעוֹת אֵינוֹ לְשִׁמּוּשׁ חָפְשִׁי, וְאֵין מְעַשְּׂרִים אוֹתוֹ. אֲבָל הָעֹדֶף – שֶׁאַבָּא הִרְשָׁה לְהִשְׁתַּמֵּשׁ בּוֹ כִּרְצוֹנוֹ – חַיָּב בְּמַעֲשֵׂר, גַּם אִם אַבָּא כְּבָר עִשֵּׂר.
דָּוִד חַיָּב לַחֲבֵרוֹ כֶּסֶף, וְגַם הִגִּיעַ הַזְּמַן לָתֵת מַעֲשֵׂר. אֵין לוֹ מַסְפִּיק לִשְׁנֵיהֶם. מָה יַעֲשֶׂה?
💡 מִי שֶׁחַיָּב לַחֲבֵרוֹ – מְשַׁלֵּם קֹדֶם אֶת הַחוֹב, וְדוֹחֶה אֶת הַמַּעֲשֵׂר. הוּא רוֹשֵׁם בְּפִנְקָס כַּמָּה הוּא חַיָּב, וּמְשַׁלֵּם הַכֹּל כְּשֶׁיּוּכַל. לֹא לוֹוִים כְּדֵי לָתֵת מַעֲשֵׂר.
לִקְרַאת בַּר הַמִּצְוָה רוֹצֶה נֹעַם לִקְנוֹת לְעַצְמוֹ תְּפִלִּין בְּכֶסֶף הַמַּעֲשֵׂר שֶׁחָסַךְ. מָה הַדִּין?
💡 אָסוּר לְשַׁלֵּם מִכֶּסֶף הַמַּעֲשֵׂר מִצְוָה שֶׁאָדָם חַיָּב בָּהּ בְּעַצְמוֹ, כְּמוֹ תְּפִלִּין לְעַצְמוֹ. אֲבָל לִקְנוֹת תְּפִלִּין לְאָדָם עָנִי – מֻתָּר.

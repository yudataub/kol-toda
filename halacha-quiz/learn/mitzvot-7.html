הִלְכוֹת כִּפָּה

🧵 צִיצִית, תְּפִלִּין וּמְזוּזָה · פֶּרֶק ז׳

🧢הַשְּׁכִינָה מֵעַל רֹאשִׁי

רַב הוּנָא לֹא הָלַךְ אַרְבַּע אַמּוֹת בְּרֹאשׁ מְגֻלֶּה, כִּי אָמַר: הַשְּׁכִינָה לְמַעְלָה מֵרֹאשִׁי. בַּהַתְחָלָה זֶה הָיָה מִנְהַג חֲסִידוּת שֶׁל חֲכָמִים, וּבְמֶשֶׁךְ הַזְּמַן כָּל עַם יִשְׂרָאֵל קִבֵּל אוֹתוֹ, וְהוּא נַעֲשָׂה מִנְהָג מְחַיֵּב: לֹא הוֹלְכִים יוֹתֵר מֵאַרְבַּע אַמּוֹת – כִּשְׁנֵי מֶטֶר – בְּלִי כִּפָּה.
כְּשֶׁיֵּשׁ רוּחַ חֲזָקָה וְהַכִּפָּה לֹא נִשְׁאֶרֶת עַל הָרֹאשׁ – מַחֲזִיקִים אוֹתָהּ בַּיָּד עַל הָרֹאשׁ.
💛 אַהֲבַת הַשֵּׁם: הַכִּפָּה מַזְכִּירָה לְךָ בְּכָל רֶגַע שֶׁיֵּשׁ מִישֶׁהוּ מֵעָלֶיךָ – אָבִינוּ שֶׁבַּשָּׁמַיִם, שֶׁרוֹאֶה אוֹתְךָ וְשׁוֹמֵר עָלֶיךָ. מִי שֶׁזּוֹכֵר שֶׁה׳ קָרוֹב אֵלָיו, הוֹלֵךְ בָּעוֹלָם בְּבִטָּחוֹן וּבְשִׂמְחָה.
📖 פֶּרֶק ז׳, הֲלָכָה א׳

🌴רַב נַחְמָן וְהַתְּמָרִים

כְּשֶׁרַב נַחְמָן בַּר יִצְחָק הָיָה תִּינוֹק, אָמְרוּ לְאִמּוֹ שֶׁהוּא עָלוּל לִהְיוֹת גַּנָּב. הִיא הִקְפִּידָה שֶׁיְּכַסֶּה תָּמִיד אֶת רֹאשׁוֹ, כְּדֵי שֶׁתִּהְיֶה עָלָיו יִרְאַת שָׁמַיִם, וְשֶׁיִּתְפַּלֵּל שֶׁהַיֵּצֶר הָרָע לֹא יִשְׁלֹט בּוֹ. פַּעַם אַחַת נָפַל הַכִּסּוּי מֵרֹאשׁוֹ, הוּא רָאָה תְּמָרִים עַל עֵץ שֶׁאֵינוֹ שֶׁלּוֹ – וְלֹא הִצְלִיחַ לְהִתְגַּבֵּר.
אָז הֵבִין רַב נַחְמָן: כִּסּוּי הָרֹאשׁ מַזְכִּיר לָאָדָם שֶׁיֵּשׁ מִי שֶׁמֵּעָלָיו, וְנוֹתֵן לוֹ כֹּחַ לְהִשָּׁמֵר מִן הַחֵטְא.
💛 אַהֲבַת הַשֵּׁם: גַּם לְאָדָם גָּדוֹל יֵשׁ יֵצֶר חָזָק. הַכִּפָּה הִיא כְּמוֹ חִבּוּק קָטָן מִלְמַעְלָה, שֶׁאוֹמֵר לְךָ: ה׳ אִתְּךָ, וְהוּא נוֹתֵן לְךָ כֹּחַ לִבְחֹר בַּטּוֹב.
📖 פֶּרֶק ז׳, הֲלָכָה א׳

🙇🔑 מוּשָּׂג: מוֹרָא שָׁמַיִם

״הִדְרִיכָה הָאֵם אֶת בְּנָהּ שֶׁיַּקְפִּיד תָּמִיד לַחְבֹּשׁ כִּפָּה לְרֹאשׁוֹ, כְּדֵי שֶׁיִּהְיֶה עָלָיו מוֹרָא שָׁמַיִם״.
📖 מוֹרָא שָׁמַיִם – יִרְאַת שָׁמַיִם: לְהַרְגִּישׁ שֶׁה׳ מֵעָלֵינוּ וְרוֹאֶה אוֹתָנוּ.
💛 אַהֲבַת הַשֵּׁם: ״הַכִּפָּה כְּאִלּוּ תּוֹחֶמֶת אֶת גְּבוּלוֹ שֶׁל הָאָדָם, וּמַזְכִּירָה לוֹ שֶׁיֵּשׁ מִי שֶׁהוּא מֵעָלָיו״. 🧒 וְאַתָּה? כְּשֶׁאַתָּה מְסַדֵּר אֶת הַכִּפָּה – תִּזְכֹּר: ה׳ מֵעָלַי.
📖 פֶּרֶק ז׳, הֲלָכָה א׳

✨🔑 מוּשָּׂג: מִנְהַג חֲסִידוּת

״חֲבִישַׁת הַכִּפָּה הִתְחִילָה כְּמִנְהַג חֲסִידוּת... בְּמֶשֶׁךְ הַזְּמַן הִתְפַּשֵּׁט הַמִּנְהָג בְּכָל יִשְׂרָאֵל... וְהָפַךְ לִהְיוֹת מִנְהָג מְחַיֵּב״.
📖 מִנְהַג חֲסִידוּת – הִדּוּר שֶׁל מִי שֶׁמּוֹסִיף מֵאַהֲבָה, יוֹתֵר מִמָּה שֶׁחַיָּבִים.
💛 אַהֲבַת הַשֵּׁם: מָה שֶׁהִתְחִיל אֵצֶל חֲסִידִים מְעַטִּים – הָפַךְ לְמִנְהַג כָּל יִשְׂרָאֵל. 🧒 וְאַתָּה? בְּחַר מִנְהַג חֲסִידוּת קָטָן אֶחָד וְקַיֵּם אוֹתוֹ.
📖 פֶּרֶק ז׳, הֲלָכָה א׳

👀כִּפָּה שֶׁרוֹאִים מִכָּל צַד

לְפִי רֹב הַפּוֹסְקִים אֵין חוֹבָה שֶׁהַכִּפָּה תְּכַסֶּה אֶת רֹב הָרֹאשׁ – הָעִקָּר שֶׁהָרֹאשׁ נֶחְשָׁב מְכֻסֶּה. אֲבָל רַבִּים כָּתְבוּ שֶׁהַכִּפָּה צְרִיכָה לְהֵרָאוֹת מִכָּל צִדֵּי הָרֹאשׁ, כָּךְ שֶׁמִּכָּל צַד שֶׁיַּבִּיטוּ – יִרְאוּ שֶׁאַתָּה חוֹבֵשׁ כִּפָּה.
הַיּוֹם הַכִּפָּה הִיא סִמָּן שֶׁל שׁוֹמְרֵי מִצְווֹת. מִי שֶׁחוֹבֵשׁ כִּפָּה שֶׁרוֹאִים הֵיטֵב – מְקַדֵּשׁ שֵׁם שָׁמַיִם.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנָשִׁים רוֹאִים יֶלֶד שֶׁלֹּא מִתְבַּיֵּשׁ לִהְיוֹת יְהוּדִי שׁוֹמֵר תּוֹרָה, שְׁמוֹ שֶׁל ה׳ מִתְגַּדֵּל בָּעוֹלָם. זוֹ דֶּרֶךְ פְּשׁוּטָה לְהַרְאוֹת לַה׳ שֶׁאַתָּה גֵּאֶה לִהְיוֹת שֶׁלּוֹ.
📖 פֶּרֶק ז׳, הֲלָכָה ג׳

🔀עוֹד מַחֲלֹקוֹת – וְכֻלָּן לְשֵׁם שָׁמַיִם

טַלִּית מִכֻּתְנָה: לְכָל הַדֵּעוֹת קוֹשְׁרִים בָּהּ צִיצִית. לְפִי הַשֻּׁלְחָן עָרוּךְ – הַחִיּוּב בָּהּ מִדִּבְרֵי חֲכָמִים, וּמִן הַתּוֹרָה רַק בְּצֶמֶר; לְפִי הָרַמָּ״א – כָּל בֶּגֶד חַיָּב מִן הַתּוֹרָה. נָשִׁים שֶׁמִּתְעַטְּפוֹת בְּטַלִּית: לְמִנְהַג אַשְׁכְּנַז – מְבָרְכוֹת, וּלְמִנְהַג סְפָרַד – לֹא מְבָרְכוֹת.
גֹּדֶל הַכִּפָּה: לְדַעַת רַבִּי שְׁלֹמֹה קְלוּגֶר, הַכִּפָּה צְרִיכָה לְכַסּוֹת אֶת רֹב הָרֹאשׁ. מַרְבִּית הַפּוֹסְקִים חָלְקוּ עָלָיו: אֵין חוֹבָה, וְהָעִקָּר שֶׁהָרֹאשׁ נֶחְשָׁב מְכֻסֶּה. וְהָרוֹצֶה לְהַחְמִיר – חוֹבֵשׁ כִּפָּה גְּדוֹלָה.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁהַפּוֹסְקִים חֲלוּקִים, כֻּלָּם רוֹצִים לְקַיֵּם אֶת רְצוֹן ה׳ בִּשְׁלֵמוּת. מִי שֶׁמְּהַדֵּר כְּדֵי לָצֵאת לְכָל הַשִּׁיטוֹת – מַרְאֶה אַהֲבָה לַמִּצְוָה.
📖 פֶּרֶק ז׳, הֲלָכָה ג׳; פֶּרֶק ח׳, הֲלָכוֹת ו׳, ח׳

🙌שְׁתֵּי דְּרָגוֹת בַּכִּפָּה

לָלֶכֶת עִם כִּפָּה כָּל הַיּוֹם – זֶה מִנְהַג חֲסִידוּת שֶׁכָּל יִשְׂרָאֵל קִבְּלוּ עֲלֵיהֶם. אֲבָל כְּשֶׁנִּכְנָסִים לְבֵית הַכְּנֶסֶת, מְבָרְכִים, מִתְפַּלְּלִים אוֹ אוֹמְרִים פְּסוּקִים – יֵשׁ חוֹבָה מִדִּבְרֵי חֲכָמִים לְכַסּוֹת אֶת הָרֹאשׁ.
מִי שֶׁכִּפָּתוֹ עָפָה יָכוֹל לָלֶכֶת כְּשֶׁיָּדוֹ עַל רֹאשׁוֹ. אֲבָל כְּדֵי לְבָרֵךְ אוֹ לְהִתְפַּלֵּל יָדוֹ שֶׁלּוֹ לֹא מוֹעִילָה, כִּי הַגּוּף לֹא מְכַסֶּה אֶת עַצְמוֹ – רַק יַד חֲבֵרוֹ שֶׁמֻּנַּחַת עַל רֹאשׁוֹ.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁמְּדַבְּרִים עִם מֶלֶךְ, מַקְפִּידִים יוֹתֵר עַל הַהוֹפָעָה. הַבְּרָכָה וְהַתְּפִלָּה הֵן דִּבּוּר יָשִׁיר עִם ה׳, וְלָכֵן חֲכָמִים חִיְּבוּ לְכַסּוֹת אֶת הָרֹאשׁ דַּוְקָא בָּרְגָעִים הַקְּדוֹשִׁים הָאֵלֶּה.
📖 פֶּרֶק ז׳, הֲלָכָה ד׳

🧠 בְּדֹק אֶת עַצְמְךָ

הַכִּפָּה שֶׁל נֹעַם נָפְלָה לִשְׁלוּלִית, וְהוּא רוֹצֶה לְבָרֵךְ עַל תַּפּוּחַ. מָה יַעֲשֶׂה?
💡 לִבְרָכָה יֵשׁ חוֹבָה מִדִּבְרֵי חֲכָמִים לְכַסּוֹת אֶת הָרֹאשׁ. יָדוֹ שֶׁל אָדָם לֹא מְכַסָּה אֶת גּוּפוֹ שֶׁלּוֹ, אֲבָל יַד חֲבֵרוֹ נֶחְשֶׁבֶת כִּסּוּי.

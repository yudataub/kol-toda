הִלְכוֹת סְפִירַת הָעֹמֶר

📅 רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת · פֶּרֶק ב׳

🌾מִצְוַת סְפִירַת הָעֹמֶר

מִצְוָה לִסְפֹּר אַרְבָּעִים וְתִשְׁעָה יָמִים, שֶׁהֵם שִׁבְעָה שָׁבוּעוֹת. מַתְחִילִים בְּלֵיל ט״ז בְּנִיסָן, בְּמוֹצָאֵי הַיּוֹם הָרִאשׁוֹן שֶׁל פֶּסַח – הַלַּיְלָה שֶׁבּוֹ קָצְרוּ אֶת הָעֹמֶר בִּימֵי הַמִּקְדָּשׁ. בְּסוֹף הַסְּפִירָה מַגִּיעַ חַג הַשָּׁבוּעוֹת.
זוֹ מִצְוָה עַל כָּל אֶחָד מִיִּשְׂרָאֵל, וְכָל אֶחָד צָרִיךְ לִסְפֹּר בְּפִיו בְּעַצְמוֹ. אֶת הַבְּרָכָה אֶפְשָׁר לִשְׁמֹעַ מִן הַחַזָּן.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁיָּצְאוּ מִמִּצְרַיִם, סָפְרוּ יִשְׂרָאֵל מֵרֹב חִבָּה כָּל יוֹם: "הֲרֵי עָבַר יוֹם רִאשׁוֹן, הֲרֵי עָבַר יוֹם שֵׁנִי" – כָּל כָּךְ חִכּוּ לַתּוֹרָה. גַּם אָנוּ סוֹפְרִים בְּגַעְגּוּעִים לְמַתַּן תּוֹרָה.
📖 פֶּרֶק ב׳, הֲלָכָה א׳

📖"וּסְפַרְתֶּם לָכֶם"

חַג הַשָּׁבוּעוֹת מְיֻחָד: אֵין לוֹ תַּאֲרִיךְ קָבוּעַ בַּחֹדֶשׁ, אֶלָּא הוּא בָּא אַחֲרֵי שֶׁסּוֹפְרִים שִׁבְעָה שָׁבוּעוֹת – וְלָכֵן נִקְרָא "חַג הַשָּׁבוּעוֹת". הַסְּפִירָה הִיא הַסֻּלָּם שֶׁמְּחַבֵּר אֶת יְצִיאַת מִצְרַיִם לְמַתַּן תּוֹרָה.
בְּכָל יוֹם סוֹפְרִים יָמִים וְשָׁבוּעוֹת: "הַיּוֹם עֲשָׂרָה יָמִים, שֶׁהֵם שָׁבוּעַ אֶחָד וּשְׁלֹשָׁה יָמִים".
💛 אַהֲבַת הַשֵּׁם: בְּפֶסַח ה׳ בָּחַר בָּנוּ, וּבְשָׁבוּעוֹת נָתַן לָנוּ אֶת תּוֹרָתוֹ – כְּמוֹ שֶׁאוֹמְרִים בְּבִרְכַּת הַתּוֹרָה: "אֲשֶׁר בָּחַר בָּנוּ מִכָּל הָעַמִּים וְנָתַן לָנוּ אֶת תּוֹרָתוֹ".
📖 פֶּרֶק ב׳, הֲלָכוֹת א׳–ג׳

🔢יָמִים וְשָׁבוּעוֹת

לִפְנֵי הַסְּפִירָה מְבָרְכִים "עַל סְפִירַת הָעֹמֶר", וְסוֹפְרִים בַּעֲמִידָה. מַזְכִּירִים גַּם אֶת הַיָּמִים וְגַם אֶת הַשָּׁבוּעוֹת. לְמָשָׁל, בַּיּוֹם הָעֲשִׂירִי אוֹמְרִים: "הַיּוֹם עֲשָׂרָה יָמִים, שֶׁהֵם שָׁבוּעַ אֶחָד וּשְׁלֹשָׁה יָמִים לָעֹמֶר".
יֵשׁ אוֹמְרִים "לָעֹמֶר" וְיֵשׁ אוֹמְרִים "בָּעֹמֶר" – וּבְכָל הַנֻּסָּחִים יוֹצְאִים. הָעִקָּר הוּא הַבְּרָכָה וְהַסְּפִירָה.
💛 אַהֲבַת הַשֵּׁם: שִׁבְעָה הוּא מִסְפָּר שָׁלֵם, כְּמוֹ שִׁבְעַת יְמֵי הַבְּרִיאָה. בְּשִׁבְעָה שָׁבוּעוֹת כָּל צַד בָּאֹפִי שֶׁלָּנוּ מִזְדַּכֵּךְ וּמִתְכּוֹנֵן לְקַבֵּל אֶת הַתּוֹרָה.
📖 פֶּרֶק ב׳, הֲלָכָה ג׳

🕰️בֵּין הַשְּׁמָשׁוֹת, בַּר מִצְוָה וְנָשִׁים

לְדַעַת רֹב הַפּוֹסְקִים, בַּזְּמַן הַזֶּה מִצְוַת הַסְּפִירָה מִדִּבְרֵי חֲכָמִים, וְלָכֵן מִי שֶׁסָּפַר בֵּין הַשְּׁקִיעָה לְצֵאת הַכּוֹכָבִים – יָצָא. אֲבָל נָכוֹן לַחֲזֹר וְלִסְפֹּר אַחֲרֵי צֵאת הַכּוֹכָבִים בְּלִי בְּרָכָה, כְּדֵי לָצֵאת גַּם לְדַעַת הָאוֹמְרִים שֶׁהַמִּצְוָה מִן הַתּוֹרָה.
נַעַר שֶׁנַּעֲשָׂה בַּר מִצְוָה בְּאֶמְצַע הַסְּפִירָה, וְסָפַר בְּכָל הַיָּמִים – מַמְשִׁיךְ לִסְפֹּר בִּבְרָכָה, וְכֵן הַמִּנְהָג הָרוֹוֵחַ. נָשִׁים וּבָנוֹת פְּטוּרוֹת מִן הַסְּפִירָה, כִּי זוֹ מִצְוַת עֲשֵׂה שֶׁהַזְּמַן גְּרָמָהּ – וְהָרוֹצָה לִסְפֹּר, יֵשׁ לָהּ מִצְוָה. לְמִנְהַג הָאַשְׁכְּנַזִּים רַשָּׁאִית לְבָרֵךְ, אִם הִיא יוֹדַעַת שֶׁאִם תִּשְׁכַּח יוֹם – תַּמְשִׁיךְ בְּלִי בְּרָכָה. לְמִנְהַג הַסְּפָרַדִּים – אֵינָהּ מְבָרֶכֶת.
💛 אַהֲבַת הַשֵּׁם: הַסְּפִירָה שֶׁל יֶלֶד אֵינָהּ מִשְׂחָק. כְּשֶׁהוּא סוֹפֵר בְּכָל יוֹם, סְפִירָתוֹ תְּמִימָה – וְהִיא מַמְשִׁיכָה גַּם כְּשֶׁהוּא נַעֲשֶׂה גָּדוֹל.
📖 פֶּרֶק ב׳, הֲלָכוֹת ד׳, ח׳–ט׳

🌌סוֹפְרִים בַּלַּיְלָה

סוֹפְרִים בַּלַּיְלָה, כִּי הַתּוֹרָה אָמְרָה "שֶׁבַע שַׁבָּתוֹת תְּמִימֹת" – שְׁלֵמוֹת, וְהַיְּמָמָה מַתְחִילָה בַּלַּיְלָה. מִצְוָה לִסְפֹּר בִּתְחִלַּת הַלַּיְלָה, אֲבָל קֹדֶם מִתְפַּלְּלִים עַרְבִית, כִּי מִצְוָה תְּדִירָה קוֹדֶמֶת.
מִי שֶׁרָגִיל לְהִתְפַּלֵּל עַרְבִית מְאֻחָר – סוֹפֵר אַחֲרֵי תְּפִלָּתוֹ הַקְּבוּעָה, בַּצִּבּוּר, כְּדֵי שֶׁלֹּא יִשְׁכַּח וְלֹא יִטְעֶה.
💛 אַהֲבַת הַשֵּׁם: כָּל לַיְלָה בַּסְּפִירָה הוּא עוֹד שְׁלַב בַּסֻּלָּם שֶׁמִּפֶּסַח לְשָׁבוּעוֹת. מִי שֶׁסּוֹפֵר בִּזְמַנּוֹ אוֹמֵר לַה׳: אֲנִי מְחַכֶּה לַתּוֹרָה שֶׁלְּךָ.
📖 פֶּרֶק ב׳, הֲלָכוֹת ה׳–ו׳

😟שָׁכַחְתִּי לִסְפֹּר!

מִי שֶׁשָּׁכַח לִסְפֹּר בַּלַּיְלָה – סוֹפֵר בַּיּוֹם שֶׁאַחֲרָיו בְּלִי בְּרָכָה, וּמִן הַלַּיְלָה הַבָּא מַמְשִׁיךְ לִסְפֹּר בִּבְרָכָה. אֲבָל מִי שֶׁשָּׁכַח יוֹם שָׁלֵם, לַיְלָה וָיוֹם – מַמְשִׁיךְ לִסְפֹּר בְּכָל הַיָּמִים הַבָּאִים בְּלִי בְּרָכָה.
מִי שֶׁשָּׁכַח יוֹם שָׁלֵם יִשְׁמַע אֶת הַבְּרָכָה מִן הַחַזָּן, וְיִתְכַּוֵּן לָצֵאת בָּהּ. וּמִי שֶׁיֵּשׁ לוֹ רַק סָפֵק אִם שָׁכַח – מַמְשִׁיךְ בִּבְרָכָה.
💛 אַהֲבַת הַשֵּׁם: הַסְּפִירָה הִיא שַׁרְשֶׁרֶת שֶׁמְּחַבֶּרֶת אֶת פֶּסַח לְשָׁבוּעוֹת. כָּל יוֹם שֶׁאָנוּ זוֹכְרִים לִסְפֹּר מוֹסִיף עוֹד חֻלְיָה לַשַּׁרְשֶׁרֶת.
📖 פֶּרֶק ב׳, הֲלָכוֹת ו׳–ח׳

🔀שְׁתֵּי שִׁיטוֹת – וּשְׁתֵּיהֶן כַּהֲלָכָה

עַד מָתַי לֹא מִסְתַּפְּרִים בַּסְּפִירָה? לְמִנְהַג הָאַשְׁכְּנַזִּים – עַד ל״ג בָּעֹמֶר, וּבְיוֹם ל״ג עַצְמוֹ כְּבָר מֻתָּר. לְמִנְהַג רַבִּים מִן הַסְּפָרַדִּים – עַד הַבֹּקֶר שֶׁל ל״ד בָּעֹמֶר, וּבְיוֹם ל״ג עֲדַיִן לֹא מִסְתַּפְּרִים (אֲבָל שָׁרִים וְרוֹקְדִים). סְפִירַת הָעֹמֶר שֶׁל נָשִׁים: לְמִנְהַג הָאַשְׁכְּנַזִּים – הָרוֹצָה סוֹפֶרֶת בִּבְרָכָה; לְמִנְהַג הַסְּפָרַדִּים – סוֹפֶרֶת בְּלִי בְּרָכָה. מוּזִיקָה מִמַּכְשִׁיר בַּבַּיִת: הָרַב פַיינְשְׁטֵיין וְהָרַב עוֹבַדְיָה – אָסוּר; הָרַב מֶלָמֵד – הָרוֹצֶה מֵקֵל בְּשִׁירִים רְגִילִים.
יֵשׁ קְהִלּוֹת סְפָרַדִּיּוֹת שֶׁמְּסַיְּמוֹת אֶת הָאֲבֵלוּת כְּבָר בְּל״ג בָּעֹמֶר. וְהַנּוֹהֲגִים לְפִי הָאֲרִ״י אֵינָם מִסְתַּפְּרִים עַד עֶרֶב חַג הַשָּׁבוּעוֹת.
💛 אַהֲבַת הַשֵּׁם: כָּל עֵדָה שׁוֹמֶרֶת אֶת הַמִּנְהָג שֶׁקִּבְּלָה מֵרַבּוֹתֶיהָ, וּשְׁתֵּי הַדְּרָכִים הֵן תּוֹרָה. דַּוְקָא בִּימֵי הַסְּפִירָה, כְּשֶׁאָנוּ מְכַבְּדִים אֶת מִנְהַג הֶחָבֵר – אָנוּ מְתַקְּנִים אֶת מָה שֶׁחָסַר לְתַלְמִידֵי רַבִּי עֲקִיבָא.
📖 פֶּרֶק ב׳, הֲלָכָה ט׳; פֶּרֶק ג׳, הֲלָכוֹת ג׳–ד׳, ו׳, י׳

🤫"כַּמָּה הַיּוֹם לָעֹמֶר?"

אִם חָבֵר שׁוֹאֵל אוֹתְךָ בָּעֶרֶב כַּמָּה הַיּוֹם לָעֹמֶר, וְעוֹד לֹא סָפַרְתָּ – אַל תַּעֲנֶה "הַיּוֹם כָּךְ וְכָךְ", כִּי אָז כְּבָר יָצָאתָ יְדֵי הַסְּפִירָה, וְלֹא תּוּכַל לְבָרֵךְ. עֲנֵה: "אֶתְמוֹל הָיָה כָּךְ וְכָךְ".
לִפְנֵי הַבְּרָכָה חוֹשְׁבִים כַּמָּה הַיּוֹם. מִי שֶׁמִּסְתַּפֵּק אִם הַיּוֹם תִּשְׁעָה אוֹ עֲשָׂרָה, וְאֵין אֶת מִי לִשְׁאֹל – אוֹמֵר אֶת שְׁתֵּי הַסְּפִירוֹת בְּלִי בְּרָכָה.
💛 אַהֲבַת הַשֵּׁם: הַדִּקְדּוּק הַקָּטָן הַזֶּה מַרְאֶה כַּמָּה אָנוּ אוֹהֲבִים אֶת הַמִּצְוָה: אָנוּ רוֹצִים לְקַיֵּם אוֹתָהּ כְּמוֹ שֶׁצָּרִיךְ, עִם בְּרָכָה וּבְשִׂמְחָה.
📖 פֶּרֶק ב׳, הֲלָכָה י׳

🧠 בְּדֹק אֶת עַצְמְךָ

סָפַרְתָּ בֵּין הַשְּׁקִיעָה לְצֵאת הַכּוֹכָבִים. מָה נָכוֹן לַעֲשׂוֹת?
💡 לְרֹב הַפּוֹסְקִים יָצָא, כִּי הַסְּפִירָה הַיּוֹם מִדִּבְרֵי חֲכָמִים. אֲבָל נָכוֹן לַחֲזֹר וְלִסְפֹּר אַחֲרֵי צֵאת הַכּוֹכָבִים, בְּלִי בְּרָכָה – כְּדֵי לָצֵאת גַּם לְדַעַת הָאוֹמְרִים שֶׁהִיא מִן הַתּוֹרָה.
אָרִי נַעֲשָׂה בַּר מִצְוָה בַּיּוֹם הָעֶשְׂרִים לָעֹמֶר, וְסָפַר בְּכָל הַיָּמִים. אֵיךְ יִסְפֹּר מֵעַתָּה?
💡 לְדַעַת רֹב הַפּוֹסְקִים, קָטָן שֶׁהִקְפִּיד לִסְפֹּר בְּכָל הַיָּמִים – סְפִירָתוֹ תְּמִימָה, וְהוּא מַמְשִׁיךְ לִסְפֹּר בִּבְרָכָה. וְכֵן הַמִּנְהָג הָרוֹוֵחַ.
חָבֵר שׁוֹאֵל אוֹתְךָ בָּעֶרֶב "כַּמָּה הַיּוֹם לָעֹמֶר?", וְעוֹד לֹא סָפַרְתָּ. מָה תַּעֲנֶה?
💡 מִי שֶׁעוֹנֶה "הַיּוֹם כָּךְ וְכָךְ" כְּבָר יָצָא יְדֵי הַסְּפִירָה, וְלֹא יוּכַל לְבָרֵךְ. לָכֵן עוֹנִים: "אֶתְמוֹל הָיָה כָּךְ וְכָךְ".

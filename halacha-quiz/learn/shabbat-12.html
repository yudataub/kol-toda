הֲכָנַת מַאֲכָלִים

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק י״ב

🔪טוֹחֵן

טְחִינָה הוֹפֶכֶת דָּבָר שָׁלֵם לְקֶמַח אוֹ לַחֲתִיכוֹת קְטַנּוֹת, וְהִיא אֲסוּרָה בְּשַׁבָּת – כְּמוֹ טְחִינַת פּוֹלֵי קָפֶה אוֹ פִּרוּר גּוּשׁ אֲדָמָה. אֲבָל מָה שֶּׁכְּבָר נִטְחַן וְאַחַר כָּךְ נַעֲשָׂה גּוּשׁ – כְּמוֹ לֶחֶם, עוּגִיּוֹת, שׁוֹקוֹלָד וְסֻכָּר – מֻתָּר לְפוֹרֵר.
סָלָט מְכִינִים סָמוּךְ לַסְּעוּדָה, וְלֹא בַּחֲתִיכוֹת קְטַנּוֹת בִּמְיֻחָד. בָּנָנָה וְאָבוֹקָדוֹ מֻתָּר לִמְעֹךְ בְּמַזְלֵג לִפְנֵי הָאֲכִילָה. וּבְפוּמְפִּיָּה (מְגָרֶדֶת) – אָסוּר לְהִשְׁתַּמֵּשׁ.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת אָנוּ לֹא יוֹצְרִים דְּבָרִים חֲדָשִׁים, וְאָנוּ נֶהֱנִים מִמָּה שֶׁכְּבָר מוּכָן. כָּךְ אָנוּ מַרְאִים שֶׁאֲנַחְנוּ סוֹמְכִים עַל ה׳, שֶׁנָּתַן לָנוּ כָּל מָה שֶׁצָּרִיךְ.
📖 פֶּרֶק י״ב, הֲלָכוֹת א׳–ב׳

🥣לָשׁ

לִישָׁה הוֹפֶכֶת אַבְקָה וְנוֹזֵל לְעִסָּה אַחַת, וְלָכֵן אָסוּר אֲפִלּוּ לִשְׁפֹּךְ מַיִם עַל קֶמַח. כְּשֶׁהַנּוֹזֵל נִשְׁאָר נוֹזֵל – כְּמוֹ קָפֶה וְסֻכָּר בְּמַיִם – אֵין לִישָׁה. גַּם כְּשֶׁהַחֲתִיכוֹת מִתְפַּזְּרוֹת וְאֵינָן נִדְבָּקוֹת לְגוּשׁ – כְּמוֹ גְּרָנוֹלָה בִּגְבִינָה – מֻתָּר. עִסָּה רַכָּה שֶׁנִּשְׁפֶּכֶת מֻתָּר לְהָכִין בְּשִׁנּוּי: מַחְלִיפִים אֶת סֵדֶר נְתִינַת הַחֳמָרִים, וּמְעַרְבְּבִים בְּקַוִּים יְשָׁרִים וְלֹא בְּסִבּוּב.
אָסוּר לְהָכִין ג׳ֶלִי אוֹ פִּירֶה מֵאַבְקָה, כִּי בַּסּוֹף הֵם נַעֲשִׂים עִסָּה עָבָה. סָלַט בֵּיצִים מֻתָּר לְהָכִין סָמוּךְ לָאֲכִילָה, וּמְעַרְבְּבִים אוֹתוֹ בְּשִׁנּוּי.
💛 אַהֲבַת הַשֵּׁם: הַשִּׁנּוּי הַקָּטָן בַּיָּד מַזְכִּיר לַלֵּב: הַיּוֹם שַׁבָּת. כָּךְ גַּם בְּתוֹךְ הַמִּטְבָּח אָנוּ חַיִּים אֶת הַשַּׁבָּת בְּכָל תְּנוּעָה, וְלֹא שׁוֹכְחִים לְרֶגַע לִפְנֵי מִי אָנוּ עוֹמְדִים.
📖 פֶּרֶק י״ב, הֲלָכוֹת ג׳–ד׳, ו׳–ז׳

🍋סְחִיטַת פֵּרוֹת

הַסּוֹחֵט פְּרִי כְּדֵי לְהוֹצִיא מִמֶּנּוּ מִיץ לִשְׁתִיָּה – עוֹבֵר עַל מְלֶאכֶת "מְפָרֵק", שֶׁהִיא תּוֹלָדָה שֶׁל דָּשׁ. אֲבָל לִסְחֹט לִימוֹן עַל סָלָט אוֹ עַל דָּג – מֻתָּר, כִּי הַמִּיץ עוֹבֵר מִמַּאֲכָל לְמַאֲכָל, וְאֵינוֹ נַעֲשֶׂה מַשְׁקֶה.
לִימוֹנָדָה מְכִינִים כָּךְ: סוֹחֲטִים אֶת הַלִּימוֹן לְתוֹךְ הַסֻּכָּר, וְרַק אַחַר כָּךְ מוֹסִיפִים מַיִם. אֶשְׁכּוֹלִית מֻתָּר לֶאֱכֹל בְּכַפִּית, וְגַם לִשְׁתּוֹת אֶת הַמִּיץ שֶׁנִּשְׁאַר בַּצַּלַּחַת.
💛 אַהֲבַת הַשֵּׁם: הַהֲלָכָה מוֹצֵאת דֶּרֶךְ שֶׁבָּהּ אֶפְשָׁר גַּם לֵהָנוֹת מִטַּעַם הַלִּימוֹן וְגַם לִשְׁמֹר שַׁבָּת. ה׳ רוֹצֶה שֶׁנִּתְעַנֵּג בְּשַׁבָּת – בְּדֶרֶךְ שֶׁמְּקָרֶבֶת אוֹתָנוּ אֵלָיו.
📖 פֶּרֶק י״ב, הֲלָכָה ח׳

🧊קֶרַח, מֶלַח וְצֶבַע

אָסוּר לְרַסֵּק קֶרַח אוֹ שֶׁלֶג כְּדֵי לַעֲשׂוֹת מֵהֶם מַיִם. אֲבָל מֻתָּר לָשִׂים קֻבִּיּוֹת קֶרַח בְּתוֹךְ מַשְׁקֶה, וְהֵן נָמַסּוֹת מֵעַצְמָן. גַּם מֻתָּר לָלֶכֶת עַל שֶׁלֶג. וְצֶבַע בְּמַאֲכָלִים אֵינוֹ אָסוּר – מֻתָּר לָתֵת כּוּרְכּוּם בְּתַבְשִׁיל, אַף שֶׁהוּא צוֹבֵעַ אוֹתוֹ בְּצָהֹב.
מְלִיחַת יְרָקוֹת כְּמוֹ מְלָפְפוֹן וּצְנוֹן דּוֹמָה לִכְבִישָׁה. לָכֵן טוֹבְלִים חֲתִיכָה אַחַת בְּמֶלַח וְאוֹכְלִים אוֹתָהּ מִיָּד. בְּסָלָט שֶׁיֵּשׁ בּוֹ שֶׁמֶן וְתַבְלִינִים – מֻתָּר לָתֵת מֶלַח.
💛 אַהֲבַת הַשֵּׁם: גַּם מִקֻּבִּיַּת קֶרַח אֶפְשָׁר לִלְמֹד: אֲנַחְנוּ לֹא עוֹשִׂים, אֲבָל ה׳ עוֹשֶׂה. הַקֶּרַח נָמֵס מֵעַצְמוֹ – וְאָנוּ נֶהֱנִים מִן הָעוֹלָם שֶׁה׳ מַנְהִיג, בְּלִי לְהִתְאַמֵּץ.
📖 פֶּרֶק י״ב, הֲלָכוֹת ט׳–י׳, י״ב

🎂אוֹתִיּוֹת עַל עוּגָה

אָסוּר לִכְתֹּב אוֹתִיּוֹת אוֹ לְצַיֵּר צוּרוֹת עַל עוּגָה בִּקְרֶם אוֹ בְּסֻכָּרִיּוֹת. עוּגָה שֶׁיֵּשׁ עָלֶיהָ אוֹתִיּוֹת אוֹ צוּרוֹת – חוֹתְכִים בֵּין הָאוֹתִיּוֹת, וְלֹא בְּתוֹכָן. וְאוֹתִיּוֹת שֶׁעֲשׂוּיוֹת מִן הַבָּצֵק עַצְמוֹ, כְּמוֹ בְּבִיסְקְוִיטִים – מֻתָּר לַחְתֹּךְ וְלִשְׁבֹּר.
אָסוּר לַעֲשׂוֹת גְּבִינָה מֵחָלָב בְּשַׁבָּת. אֲבָל לִמְרֹחַ מִמְרָח עַל פְּרוּסָה – מֻתָּר.
💛 אַהֲבַת הַשֵּׁם: אוֹתִיּוֹת הֵן דָּבָר חָשׁוּב: בָּהֶן נִכְתְּבָה הַתּוֹרָה. גַּם עַל עוּגָה אָנוּ נִזְהָרִים בָּהֶן – וְכָךְ לוֹמְדִים לְכַבֵּד כָּל אוֹת וְכָל מִלָּה.
📖 פֶּרֶק י״ב, הֲלָכָה י״א

🔀קָרוּשׁ וְקֶרַח – שְׁנֵי מִנְהָגִים, וּשְׁנֵיהֶם כַּהֲלָכָה

רֹטֶב קָרוּשׁ שֶׁמִּתְחַמֵּם נַעֲשֶׂה נוֹזֵל. לְפִי רֹב הַפּוֹסְקִים זֶה מֻתָּר, כִּי הַחֹם מַמִּיס אוֹתוֹ וְלֹא הַיָּדַיִם – וְכָךְ מִנְהַג הַסְּפָרַדִּים. לְמִנְהַג הָאַשְׁכְּנַזִּים, לְכַתְּחִלָּה נִזְהָרִים שֶׁלֹּא לַהֲפֹךְ מַאֲכָל קָרוּשׁ לְנוֹזֵל עַל יְדֵי חִמּוּם, וּבִשְׁעַת הַצֹּרֶךְ מְקִלִּים. וְאִם הָרֹטֶב טָפֵל לָאֹכֶל הַמּוּצָק – גַּם הָאַשְׁכְּנַזִּים מְחַמְּמִים אוֹתוֹ לְכַתְּחִלָּה.
הֲכָנַת קֶרַח בַּמַּקְפִּיא בְּשַׁבָּת: לְמִנְהַג הַסְּפָרַדִּים – מֻתֶּרֶת. לְמִנְהַג הָאַשְׁכְּנַזִּים יֵשׁ מַחְמִירִים לְכַתְּחִלָּה, וּבִשְׁעַת הַצֹּרֶךְ, כְּמוֹ בְּיוֹם חַם – מֻתָּר; וְיֵשׁ פּוֹסְקִים אַשְׁכְּנַזִּים שֶׁמְּקִלִּים גַּם בְּיוֹם שֶׁאֵינוֹ חַם.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁהַמִּנְהָגִים שׁוֹנִים, כֻּלָּם שׁוֹאֲפִים לְאוֹתוֹ דָּבָר: לִשְׁמֹר עַל הַשַּׁבָּת וּלְהִתְעַנֵּג בָּהּ. מִי שֶׁמְּכַבֵּד אֶת מִנְהַג חֲבֵרוֹ – מַרְבֶּה שָׁלוֹם בְּיִשְׂרָאֵל.
📖 פֶּרֶק י״ב, הֲלָכָה י״ב; פֶּרֶק י׳, הֲלָכָה ה׳

🧠 בְּדֹק אֶת עַצְמְךָ

אַבְקַת ג׳ֶלִי עִם מַיִם הִיא בְּהַתְחָלָה נוֹזֵל. לָמָּה בְּכָל זֹאת אָסוּר לְהָכִין ג׳ֶלִי בְּשַׁבָּת?
💡 הַמְּעָרֵב מַיִם בְּאַבְקַת ג׳ֶלִי עוֹבֵר עַל מְלֶאכֶת לָשׁ, כִּי בַּסּוֹף הַתַּעֲרֹבֶת מִתְקַשָּׁה. צֶבַע בְּמַאֲכָלִים אֵינוֹ אָסוּר, וְקָפֶה אוֹ סֻכָּר בְּמַיִם – מֻתָּרִים.
בְּשַׁבָּת חַמָּה רוֹצֶה אֲרִיאֵל לְקָרֵר אֶת הַמַּיִם שֶׁבַּכּוֹס. מָה מֻתָּר?
💡 רִסּוּק קֶרַח אוֹ שֶׁלֶג בַּיָּדַיִם כְּדֵי לַעֲשׂוֹת מֵהֶם מַיִם אָסוּר, כִּי הוּא דּוֹמֶה לִסְחִיטַת פֵּרוֹת. כְּשֶׁהַקֶּרַח נָמֵס מֵעַצְמוֹ – אֵין עוֹשִׂים מַעֲשֶׂה, וּמֻתָּר.
שְׁלֹמֹה מֵכִין סָלַט בֵּיצִים לִסְעוּדַת שַׁבָּת. אֵיךְ יַעֲשֶׂה זֹאת?
💡 סָלַט בֵּיצִים נַעֲשֶׂה כְּמוֹ עִסָּה אַחַת, וְלָכֵן מְכִינִים אוֹתוֹ סָמוּךְ לָאֲכִילָה וּמְעַרְבְּבִים בְּשִׁנּוּי. רִסּוּק בְּמַזְלֵג – מֻתָּר, אֲבָל פּוּמְפִּיָּה נוֹעֲדָה לִטְחִינָה.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

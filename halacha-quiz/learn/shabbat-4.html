הַדְלָקַת נֵרוֹת שַׁבָּת

🕯️ הִלְכוֹת שַׁבָּת · פֶּרֶק ד׳

🕯️לָמָּה מַדְלִיקִים נֵרוֹת?

חֲכָמִים תִּקְּנוּ לְהַדְלִיק נֵר לִכְבוֹד שַׁבָּת, מִשְּׁלֹשָׁה טְעָמִים: לִכְבוֹד הַשַּׁבָּת – כִּי אֵין סְעוּדָה חֲשׁוּבָה בַּחֹשֶׁךְ; לְעֹנֶג – כִּי מִי שֶׁאֵינוֹ רוֹאֶה אֶת הָאֹכֶל אֵינוֹ נֶהֱנֶה מִמֶּנּוּ; וּלְשָׁלוֹם בַּיִת – כְּדֵי שֶׁלֹּא נִתָּקֵל בַּחֹשֶׁךְ וְנִכְעַס.
מִי שֶׁיֵּשׁ לוֹ מְעַט כֶּסֶף – קוֹנֶה קֹדֶם לֶחֶם, אַחַר כָּךְ נֵרוֹת, וְרַק אַחַר כָּךְ יַיִן. כִּי אֶפְשָׁר לְקַדֵּשׁ עַל הַלֶּחֶם, אֲבָל בְּלִי נֵר אֵין כָּבוֹד וְעֹנֶג.
💛 אַהֲבַת הַשֵּׁם: בַּחֹשֶׁךְ הַכֹּל נִרְאֶה מְבֻלְבָּל, וּכְשֶׁמַּדְלִיקִים נֵר – רוֹאִים שֶׁכָּל דָּבָר בִּמְקוֹמוֹ. כָּךְ הַשַּׁבָּת מוֹסִיפָה בָּעוֹלָם אוֹר שֶׁל אֱמוּנָה, וּמְגַלָּה שֶׁה׳ מְכַוֵּן הַכֹּל לְטוֹבָה.
📖 פֶּרֶק ד׳, הֲלָכָה א׳

🏠אֵיפֹה מַדְלִיקִים, וּמִי?

עִקַּר הַמִּצְוָה – לְהַדְלִיק בִּמְקוֹם הַסְּעוּדָה, וְהַנֵּרוֹת צְרִיכִים לִדְלֹק לְפָחוֹת עַד סוֹף הָאֲרוּחָה. הַמִּצְוָה שַׁיֶּכֶת לְכָל יִשְׂרָאֵל, אֲבָל הָאֵם קוֹדֶמֶת בָּהּ, כִּי הִיא אַחְרָאִית עַל הַבַּיִת – וּבְהַדְלָקָתָהּ יוֹצְאִים כָּל בְּנֵי הַבַּיִת.
אִם אִמָּא מִתְעַכֶּבֶת וְהַשְּׁקִיעָה קְרֵבָה – אַבָּא אוֹ אֶחָד הַיְּלָדִים מַדְלִיק, וְלֹא נִכְנָסִים לְסָפֵק חִלּוּל שַׁבָּת.
💛 אַהֲבַת הַשֵּׁם: אוֹר הַנֵּרוֹת הוּא אוֹר שֶׁל שָׁלוֹם בַּבַּיִת. חֲכָמִים אָמְרוּ: מִי שֶׁמְּהַדֵּר בְּנֵרוֹת שַׁבָּת זוֹכֶה לְבָנִים תַּלְמִידֵי חֲכָמִים – שֶׁמּוֹסִיפִים אוֹר שֶׁל תּוֹרָה בָּעוֹלָם.
📖 פֶּרֶק ד׳, הֲלָכוֹת א׳–ב׳

🔀מִנְהָגִים בַּנֵּרוֹת וּבָעֲלִיָּה לַתּוֹרָה – וְכֻלָּם כַּהֲלָכָה

אוֹרַחַת שֶׁמַּדְלִיקָה לְיַד בַּעֲלַת הַבַּיִת: לְפִי הַשֻּׁלְחָן עָרוּךְ, וְכָךְ נוֹהֲגִים בְּחֵלֶק מִקְּהִלּוֹת הַסְּפָרַדִּים – מַדְלִיקָה בְּלִי בְּרָכָה. לְפִי הָרָמָ״א וְרֹב הַפּוֹסְקִים, וְכָךְ נוֹהֲגִים כָּל הָאַשְׁכְּנַזִּים וְחֵלֶק מֵהַסְּפָרַדִּים, כְּמוֹ עוֹלֵי מָרוֹקוֹ – מַדְלִיקָה בִּבְרָכָה, כִּי כָּל נֵר מוֹסִיף אוֹר. יֵשׁ נוֹהֲגִים שֶׁגַּם הַבָּנוֹת שֶׁהִגִּיעוּ לְחִנּוּךְ מַדְלִיקוֹת נֵרוֹת בִּבְרָכָה, כְּמוֹ חֲסִידֵי חַבַּ״ד; וּלְדַעַת רֹב הַפּוֹסְקִים רַק הָאֵם מַדְלִיקָה.
יֶלֶד שֶׁלֹּא הִגִּיעַ לְמִצְווֹת וְיוֹדֵעַ לְמִי מְבָרְכִים: יֵשׁ מֵהַסְּפָרַדִּים שֶׁמַּעֲלִים אוֹתוֹ לַתּוֹרָה, וּמִנְהַג אַשְׁכְּנַז וְחֵלֶק מֵהַסְּפָרַדִּים שֶׁלֹּא לְהַעֲלוֹתוֹ. יֵשׁ שֶׁמַּעֲלִים אוֹתוֹ רַק לִשְׁבִיעִי, וְהַתֵּימָנִים – לְשִׁשִּׁי.
💛 אַהֲבַת הַשֵּׁם: ״רָאוּי לְכָל אִשָּׁה לְהַמְשִׁיךְ בְּמִנְהַג מִשְׁפַּחְתָּהּ״ – וְכָךְ כָּל אֶחָד וְאַחַת. כְּשֶׁכָּל מִשְׁפָּחָה שׁוֹמֶרֶת עַל מִנְהָגָהּ, כֻּלָּנוּ מְאִירִים יַחַד אֶת הַשַּׁבָּת.
📖 פֶּרֶק ד׳, הֲלָכוֹת ב׳, ו׳; פֶּרֶק ה׳, הֲלָכָה ו׳

✌️שְׁנֵי נֵרוֹת – זָכוֹר וְשָׁמוֹר

מִן הַדִּין מַסְפִּיק נֵר אֶחָד, וְנָהֲגוּ לְהַדְלִיק שְׁנַיִם – כְּנֶגֶד ״זָכוֹר״ וְ״שָׁמוֹר״. יֵשׁ מוֹסִיפִים נֵר לְכָל אֶחָד מִבְּנֵי הַבַּיִת. הַשֶּׁמֶן הַמְשֻׁבָּח בְּיוֹתֵר הוּא שֶׁמֶן זַיִת, שֶׁאוֹרוֹ צָלוּל, וְהַיּוֹם רַבִּים מַדְלִיקִים נֵרוֹת שֶׁאוֹרָם חָזָק וְיַצִּיב.
בִּזְמַן הַבְּרָכָה יֵשׁ שְׁנֵי מִנְהָגִים: לְפִי הַשֻּׁלְחָן עָרוּךְ מְבָרְכוֹת לִפְנֵי הַהַדְלָקָה; וְיוֹצְאוֹת אַשְׁכְּנַז, וְגַם רַבּוֹת מִיּוֹצְאוֹת סְפָרַד, מַדְלִיקוֹת קֹדֶם, מְכַסּוֹת אֶת הָעֵינַיִם וּמְבָרְכוֹת. כָּל אַחַת נוֹהֶגֶת כְּמִנְהַג מִשְׁפַּחַת אִמָּהּ. גֶּבֶר שֶׁמַּדְלִיק – מְבָרֵךְ לִפְנֵי הַהַדְלָקָה.
💛 אַהֲבַת הַשֵּׁם: שְׁנֵי הַנֵּרוֹת מַזְכִּירִים אֶת שְׁתֵּי הַפָּנִים שֶׁל הַשַּׁבָּת: הַזְּכִירָה וְהַשְּׁמִירָה. כְּמוֹ פְּתִילָה רַכָּה וְשֶׁמֶן זַךְ שֶׁיַּחַד יוֹצֵר אוֹר – כָּךְ מִשְׁפָּחָה מְאֻחֶדֶת מְאִירָה יַחַד.
📖 פֶּרֶק ד׳, הֲלָכוֹת ג׳–ד׳

👧בָּנוֹת וְנָשִׁים בַּנֵּרוֹת וּבַתּוֹרָה

אִשָּׁה שֶׁשָּׁכְחָה לְהַדְלִיק נֵרוֹת, וְהַבַּיִת הָיָה חָשׁוּךְ – נָהֲגוּ שֶׁמֵּאָז הִיא מוֹסִיפָה בְּכָל שַׁבָּת נֵר אֶחָד. וְאִם דָּלַק אוֹר חַשְׁמַל – אֵינָהּ צְרִיכָה לְהוֹסִיף. בִּפְנִימִיַּת בָּנוֹת אַחַת מַדְלִיקָה בַּחֲדַר הָאֹכֶל בִּשְׁבִיל כֻּלָּן, וּבָנוֹת שֶׁרוֹצוֹת לִזְכּוֹת בַּמִּצְוָה רַשָּׁאִיּוֹת לְהַדְלִיק בַּחֲדָרִים בִּבְרָכָה – אִם מֻתָּר שָׁם לְהַדְלִיק.
גַּם לְנָשִׁים וּלְבָנוֹת מִצְוָה לִלְמֹד תּוֹרָה בְּשַׁבָּת – הַלִּמּוּד הוּא חֵלֶק מֵעֹנֶג הַשַּׁבָּת, וְרָאוּי לִקְבֹּעַ לִמּוּד בַּהֲלָכָה וּבֶאֱמוּנָה. נָשִׁים חַיָּבוֹת בְּבִרְכוֹת הַשַּׁחַר וּבַעֲמִידָה שֶׁל שַׁחֲרִית וּמִנְחָה, וּפְטוּרוֹת מִתְּפִלָּה בְּמִנְיָן; וּמִי שֶׁיְּכוֹלָה – טוֹב שֶׁתָּבוֹא לְבֵית הַכְּנֶסֶת בְּשַׁבָּת.
💛 אַהֲבַת הַשֵּׁם: הַמִּצְוָה לְהַקְבִּיל פְּנֵי רַב בְּשַׁבָּת נִלְמֶדֶת מִן הָאִשָּׁה הַשּׁוּנַמִּית. הָרַב צְבִי יְהוּדָה קוּק הִסְבִּיר: בַּיַּחַס הַכְּלָלִי אֶל הַתּוֹרָה וְאֶל לוֹמְדֶיהָ – הַנָּשִׁים עוֹלוֹת עַל הַגְּבָרִים.
📖 פֶּרֶק ד׳, הֲלָכוֹת ג׳, ח׳; פֶּרֶק ה׳, הֲלָכוֹת ה׳, י״ד–ט״ו

🔀שְׁנֵי מִנְהָגִים בְּבִרְכַּת הַנֵּרוֹת – וּשְׁנֵיהֶם כַּהֲלָכָה

לְפִי הַשֻּׁלְחָן עָרוּךְ, וְכָךְ נוֹהֲגוֹת חֵלֶק מִיּוֹצְאוֹת סְפָרַד, מְבָרְכִים לִפְנֵי הַהַדְלָקָה – כְּמוֹ בְּכָל מִצְוָה, שֶׁמְּבָרְכִים עָלֶיהָ לִפְנֵי שֶׁעוֹשִׂים אוֹתָהּ. כָּל יוֹצְאוֹת אַשְׁכְּנַז, כְּדִבְרֵי הָרָמָ״א, וְגַם יוֹצְאוֹת סְפָרַד רַבּוֹת, כְּמוֹ בְּמָרוֹקוֹ וּבְבָבֶל, מַדְלִיקוֹת קֹדֶם וְאַחַר כָּךְ מְבָרְכוֹת – כִּי בִּזְמַן הַבְּרָכָה הֵן מְקַבְּלוֹת אֶת הַשַּׁבָּת, וְאַחֲרֵי קַבָּלַת הַשַּׁבָּת כְּבָר אִי אֶפְשָׁר לְהַדְלִיק.
הַמְבָרְכוֹת אַחֲרֵי הַהַדְלָקָה מְכַסּוֹת אֶת הָעֵינַיִם בִּשְׁעַת הַבְּרָכָה, וְרַק אַחֲרֶיהָ מִסְתַּכְּלוֹת בָּאוֹר – כְּדֵי שֶׁהַבְּרָכָה תִּהְיֶה לִפְנֵי הֲנָאַת הַמִּצְוָה. כָּל אִשָּׁה נוֹהֶגֶת כְּמִנְהַג מִשְׁפַּחַת אִמָּהּ.
💛 אַהֲבַת הַשֵּׁם: שְׁנֵי הַמִּנְהָגִים יוֹצְאִים מֵאוֹתָהּ אַהֲבָה: לְקַבֵּל אֶת הַשַּׁבָּת בִּבְרָכָה וּבְאוֹר. כְּשֶׁאָנוּ מְכַבְּדִים אֶת מִנְהַג הַבַּיִת שֶׁל הֶחָבֵר – הָאוֹר שֶׁל כֻּלָּנוּ גָּדֵל.
📖 פֶּרֶק ד׳, הֲלָכָה ד׳

💡נֵר אוֹ חַשְׁמַל?

בִּשְׁעַת הַצֹּרֶךְ אֶפְשָׁר לְקַיֵּם אֶת הַמִּצְוָה בְּנוּרַת לַהַט חַשְׁמַלִּית, כִּי חוּט הַמַּתֶּכֶת הַבּוֹעֵר הוּא כְּמוֹ פְּתִילָה. אֲבָל עָדִיף נֵר – כְּדֵי לָצֵאת לְדַעַת כֻּלָּם, וְכִי בְּנֵר נִכָּר שֶׁזּוֹ הַדְלָקָה מְיֻחֶדֶת לִכְבוֹד שַׁבָּת.
לְכַתְּחִלָּה מְכַבִּים אֶת הָאוֹר הַחַשְׁמַלִּי כַּמָּה דַּקּוֹת לִפְנֵי הַהַדְלָקָה, וּבִשְׁעַת הַהַדְלָקָה מַדְלִיקִים אוֹתוֹ שׁוּב יַחַד עִם הַנֵּרוֹת – וְהַבְּרָכָה חָלָה עַל שְׁנֵיהֶם.
💛 אַהֲבַת הַשֵּׁם: כָּל אוֹר שֶׁמּוֹסִיפִים לִכְבוֹד שַׁבָּת הוּא אוֹר שֶׁל מִצְוָה. כְּשֶׁמַּדְלִיקִים בְּכַוָּנָה – גַּם הַמְּנוֹרָה הָרְגִילָה שֶׁבַּבַּיִת נַעֲשֵׂית חֵלֶק מִכְּבוֹד הַשַּׁבָּת.
📖 פֶּרֶק ד׳, הֲלָכָה ה׳

🧳אוֹרְחִים, יְשִׁיבוֹת וְחַיָּלִים

לְמִצְוַת הַנֵּרוֹת שְׁנֵי חֲלָקִים: שֶׁיִּהְיֶה אוֹר בִּמְקוֹם הַסְּעוּדָה, וְשֶׁכָּל אָדָם יְקַיֵּם אֶת הַמִּצְוָה. אוֹרֵחַ שֶׁאוֹכֵל וְיָשֵׁן אֵצֶל מִשְׁפָּחָה – יוֹצֵא בְּהַדְלָקַת בַּעֲלַת הַבַּיִת, כְּמוֹ בְּנֵי הַבַּיִת. וּמִי שֶׁאוֹכֵל אֶת סְעוּדַת הַלַּיְלָה בְּחֶדֶר מִשֶּׁלּוֹ – מַדְלִיק שָׁם בִּבְרָכָה.
בִּישִׁיבָה וּבַצָּבָא כֻּלָּם כְּמוֹ מִשְׁפָּחָה אַחַת גְּדוֹלָה: אֶחָד מַדְלִיק בִּבְרָכָה בַּחֲדַר הָאֹכֶל בִּשְׁבִיל כֻּלָּם, וְדוֹאֲגִים שֶׁיַּגִּיעַ מְעַט אוֹר גַּם לַחֲדָרִים.
💛 אַהֲבַת הַשֵּׁם: הַנֵּר שֶׁאֶחָד מַדְלִיק מֵאִיר לְכֻלָּם. כָּךְ עַם יִשְׂרָאֵל: כְּשֶׁאֲנַחְנוּ יַחַד, מִצְוָה שֶׁל אֶחָד מְאִירָה לְכָל הַחֲבֵרִים.
📖 פֶּרֶק ד׳, הֲלָכוֹת ז׳–ח׳

🧠 בְּדֹק אֶת עַצְמְךָ

הַמְּנוֹרָה בַּסָּלוֹן דּוֹלֶקֶת עוֹד מֵהַצָּהֳרַיִם, וְעוֹד מְעַט מַדְלִיקִים נֵרוֹת שַׁבָּת. מָה הַמְּהֻדָּר?
💡 לְכַתְּחִלָּה מְכַבִּים אֶת הַחַשְׁמַל לִפְנֵי הַהַדְלָקָה, וּמַדְלִיקִים אוֹתוֹ שׁוּב יַחַד עִם הַנֵּרוֹת – כָּךְ נִכָּר שֶׁהָאוֹר לִכְבוֹד שַׁבָּת. וּמִן הַדִּין אֶפְשָׁר לְבָרֵךְ גַּם כְּשֶׁהַחַשְׁמַל דּוֹלֵק, כִּי הַנֵּרוֹת מוֹסִיפִים אוֹר.
יוֹסֵף, בָּחוּר רַוָּק, מִתְאָרֵחַ לְשַׁבָּת אֵצֶל מִשְׁפָּחָה – אוֹכֵל וְיָשֵׁן אֶצְלָם. הַאִם יַדְלִיק נֵרוֹת?
💡 כְּמוֹ שֶׁבְּנֵי הַבַּיִת יוֹצְאִים בְּהַדְלָקַת בַּעֲלַת הַבַּיִת, כָּךְ גַּם הָאוֹרֵחַ – וּכְבָר יֵשׁ אוֹר בִּמְקוֹם הַסְּעוּדָה. מִצְוַת הַנֵּרוֹת שַׁיֶּכֶת לְכֻלָּם, גַּם לִגְבָרִים.
בִּפְנִימִיַּת בָּנוֹת אַחַת הַבָּנוֹת מַדְלִיקָה נֵרוֹת בַּחֲדַר הָאֹכֶל בִּשְׁבִיל כֻּלָּן. מָה עִם שְׁאָר הַבָּנוֹת?
💡 בַּהַדְלָקָה שֶׁבַּחֲדַר הָאֹכֶל כֻּלָּן יוֹצְאוֹת. אֲבָל בָּנוֹת שֶׁרוֹצוֹת לִזְכּוֹת בַּמִּצְוָה רַשָּׁאִיּוֹת לְהַדְלִיק בַּחֲדָרִים בִּבְרָכָה. וְאִם אוֹסְרִים שָׁם לְהַדְלִיק מֵחֲשַׁשׁ שְׂרֵפָה – כֻּלָּן יוֹצְאוֹת בַּהַדְלָקָה שֶׁבַּחֲדַר הָאֹכֶל.

בִּרְכַּת הַמּוֹצִיא

🍎 הִלְכוֹת בְּרָכוֹת · פֶּרֶק ג׳

🌾לָמָּה לַלֶּחֶם בְּרָכָה מְיֻחֶדֶת?

הַלֶּחֶם עָשׂוּי מִדָּגָן שֶׁצּוֹמֵחַ בָּאֲדָמָה, וְהָיָה מַתְאִים לְבָרֵךְ עָלָיו "בּוֹרֵא פְּרִי הָאֲדָמָה". אֲבָל הוּא עִקַּר מַאֲכָלוֹ שֶׁל הָאָדָם, וְלָכֵן תִּקְּנוּ לוֹ חֲכָמִים בְּרָכָה מְיֻחֶדֶת. וְכַמָּה עֲבוֹדָה יֵשׁ בּוֹ: חוֹרְשִׁים, זוֹרְעִים, קוֹצְרִים, דָּשִׁים, טוֹחֲנִים, לָשִׁים וְאוֹפִים!
אוֹמְרִים "הַמּוֹצִיא לֶחֶם מִן הָאָרֶץ" – כְּדֵי שֶׁלֹּא נַחְשֹׁב שֶׁאֲנַחְנוּ לְבַד הוֹצֵאנוּ אֶת הַלֶּחֶם. הָאֲדָמָה, הַדָּגָן וְהַשֵּׂכֶל לֶאֱפוֹת – הַכֹּל מִמֶּנּוּ.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁאָנוּ עוֹבְדִים קָשֶׁה, ה׳ הוּא שֶׁנּוֹתֵן אֶת הַכֹּחַ וְאֶת הַחָכְמָה. כְּשֶׁזּוֹכְרִים אוֹתוֹ, כָּל הַכִּשְׁרוֹנוֹת שֶׁלָּנוּ הוֹפְכִים לְטוֹבָה, וַאֲנַחְנוּ נַעֲשִׂים שֻׁתָּפִים שֶׁלּוֹ בְּבִנְיַן הָעוֹלָם.
📖 פֶּרֶק ג׳, הֲלָכָה א׳

🤲עֶשֶׂר אֶצְבָּעוֹת עַל הַלֶּחֶם

עַל כָּל מַאֲכָל טוֹב לְבָרֵךְ כְּשֶׁהוּא אָחוּז בְּיַד יָמִין. אֶת הַלֶּחֶם, מֵרֹב חֲשִׁיבוּתוֹ, טוֹב לֶאֱחֹז בְּשָׁעַת הַבְּרָכָה בִּשְׁתֵּי הַיָּדַיִם – בְּעֶשֶׂר אֶצְבָּעוֹת.
עֶשֶׂר הָאֶצְבָּעוֹת רוֹמְזוֹת לְעֶשֶׂר מִצְוֹת שֶׁבַּהֲכָנַת הַלֶּחֶם, כְּמוֹ: לֶקֶט, שִׁכְחָה וּפֵאָה לָעֲנִיִּים, תְּרוּמוֹת וּמַעַשְׂרוֹת, וְהַפְרָשַׁת חַלָּה מִן הַבָּצֵק.
💛 אַהֲבַת הַשֵּׁם: כָּל פְּרוּסָה מְחֻבֶּרֶת לְמִצְוֹת שֶׁל חֶסֶד וּנְתִינָה. כְּשֶׁאַתָּה אוֹחֵז בָּהּ בְּעֶשֶׂר אֶצְבָּעוֹת, אַתָּה כְּאִלּוּ מְחַבֵּק אֶת כָּל הַטּוֹב שֶׁה׳ שָׂם בְּתוֹכָהּ.
📖 פֶּרֶק ג׳, הֲלָכָה ב׳

🥖עַל אֵיזֶה לֶחֶם מְבָרְכִים?

מִי שֶׁיֵּשׁ לְפָנָיו כַּמָּה לְחָמִים מְבָרֵךְ עַל הֶחָשׁוּב שֶׁבָּהֶם, כְּדֵי שֶׁהַשֶּׁבַח לַה׳ יִהְיֶה הַיָּפֶה בְּיוֹתֵר. לֶחֶם חִטָּה קוֹדֵם לְלֶחֶם שְׂעוֹרָה – הוּא מֻקְדָּם בְּשִׁבְעַת הַמִּינִים, וְגַם טַעֲמוֹ מְשֻׁבָּח יוֹתֵר.
לַחְמָנִיָּה שְׁלֵמָה קוֹדֶמֶת לְלֶחֶם גָּדוֹל שֶׁכְּבָר חָתְכוּ מִמֶּנּוּ – הַשְּׁלֵמוּת חֲשׁוּבָה מִן הַגֹּדֶל. וּכְשֶׁשְּׁנֵי הַלְּחָמִים שָׁוִים – מְבָרְכִים עַל הַגָּדוֹל.
💛 אַהֲבַת הַשֵּׁם: לַה׳ נוֹתְנִים אֶת הַמֻּבְחָר. כְּמוֹ שֶׁבּוֹחֲרִים אֶת הַמַּתָּנָה הַיָּפָה בְּיוֹתֵר לְמִי שֶׁאוֹהֲבִים, כָּךְ בּוֹחֲרִים אֶת הַלֶּחֶם הֶחָשׁוּב לַבְּרָכָה.
📖 פֶּרֶק ג׳, הֲלָכָה ב׳

🔪בְּצִיעָה – וְטוֹעֲמִים מִיָּד

בְּלֶחֶם גָּדוֹל מַתְחִילִים לַחְתֹּךְ לִפְנֵי הַבְּרָכָה, אֲבָל מַשְׁאִירִים אוֹתוֹ מְחֻבָּר. לַחְמָנִיָּה אוֹ פִּתָּה – מְבָרְכִים כְּשֶׁהִיא שְׁלֵמָה, וּבוֹצְעִים אַחֲרֵי הַבְּרָכָה. וּבְשַׁבָּת לֹא חוֹתְכִים אֶת הַחַלָּה לִפְנֵי הַבְּרָכָה.
מִיָּד אַחֲרֵי "הַמּוֹצִיא" טוֹעֲמִים מִן הַלֶּחֶם. מִי שֶׁדִּבֵּר לִפְנֵי שֶׁטָּעַם, אֲפִלּוּ מִלָּה אַחַת, צָרִיךְ לְבָרֵךְ שׁוּב.
💛 אַהֲבַת הַשֵּׁם: הַבְּרָכָה וְהָאֲכִילָה הֵן כְּמוֹ מִשְׁפָּט אֶחָד: "תּוֹדָה, ה׳" – וּמִיָּד נֶהֱנִים מֵהַמַּתָּנָה. כָּךְ הַתּוֹדָה שֶׁלָּנוּ חַיָּה וַאֲמִתִּית.
📖 פֶּרֶק ג׳, הֲלָכָה ג׳

🗣️דִּבּוּר שֶׁל צֹרֶךְ, וְחַלּוֹת שֶׁל שַׁבָּת

מִי שֶׁבֵּרֵךְ "הַמּוֹצִיא", וְלִפְנֵי שֶׁטָּעַם בִּקֵּשׁ מֶלַח אוֹ סַכִּין – לֹא נָהַג כָּרָאוּי, אֲבָל אֵינוֹ חוֹזֵר לְבָרֵךְ, כִּי דִּבֵּר עַל צָרְכֵי הָאֲכִילָה. בְּשַׁבָּת מְבָרְכִים עַל שְׁתֵּי חַלּוֹת שְׁלֵמוֹת, וְלָכֵן לֹא מַתְחִילִים לַחְתֹּךְ לִפְנֵי הַבְּרָכָה.
בִּסְעוּדָה גְּדוֹלָה, כְּמוֹ בַּר מִצְוָה, בְּכָל שֻׁלְחָן מְבָרֵךְ אֶחָד בִּשְׁבִיל מִי שֶׁנָּטְלוּ אִתּוֹ – כְּדֵי שֶׁלֹּא יִהְיֶה הֶפְסֵק גָּדוֹל מִדַּי בֵּין הַנְּטִילָה לַבְּרָכָה.
💛 אַהֲבַת הַשֵּׁם: הַהֲלָכָה שׁוֹמֶרֶת שֶׁהַבְּרָכָה תִּהְיֶה צְמוּדָה לַהֲנָאָה. כְּמוֹ שֶׁאוֹמְרִים תּוֹדָה לְאַבָּא בָּרֶגַע שֶׁמְּקַבְּלִים מִמֶּנּוּ, כָּךְ אוֹמְרִים תּוֹדָה לַה׳ – וּמִיָּד טוֹעֲמִים.
📖 פֶּרֶק ג׳, הֲלָכוֹת ג׳–ד׳

👨‍👦אֶחָד מְבָרֵךְ בִּשְׁבִיל כֻּלָּם

כְּשֶׁאוֹכְלִים יַחַד, טוֹב שֶׁאֶחָד יְבָרֵךְ "הַמּוֹצִיא" בְּקוֹל בִּשְׁבִיל כֻּלָּם, כִּי בְּרָכָה שֶׁל רַבִּים מְכֻבֶּדֶת יוֹתֵר. בְּשַׁבָּת מְחַכִּים שֶׁכֻּלָּם יִגְמְרוּ לִטֹּל יָדַיִם, וְאָז בַּעַל הַבַּיִת מְבָרֵךְ וְכֻלָּם עוֹנִים "אָמֵן".
מִי שֶׁיָּצָא בַּבְּרָכָה שֶׁל אַחֵר – לֹא מְדַבֵּר עַד שֶׁהוּא עַצְמוֹ טוֹעֵם מִן הַלֶּחֶם.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁכָּל הַמִּשְׁפָּחָה עוֹנָה "אָמֵן" יַחַד, ה׳ שׁוֹמֵעַ תּוֹדָה אַחַת גְּדוֹלָה מִכֻּלָּם. הָאַחְדוּת סְבִיב הַשֻּׁלְחָן מְשַׂמַּחַת אֶת אָבִינוּ שֶׁבַּשָּׁמַיִם.
📖 פֶּרֶק ג׳, הֲלָכָה ד׳

🧂מֶלַח עַל הַשֻּׁלְחָן

פַּעַם אָפוּ לֶחֶם בְּלִי מֶלַח, וְלָכֵן טָבְלוּ אוֹתוֹ בְּמֶלַח, כְּדֵי שֶׁהַפְּרוּסָה שֶׁאַחֲרֵי הַבְּרָכָה תִּהְיֶה טְעִימָה. הַיּוֹם הַלֶּחֶם מָלוּחַ מֵעַצְמוֹ – וּבְכָל זֹאת רַבִּים מְהַדְּרִים לִטְבֹּל אוֹתוֹ בְּמֶלַח.
מַנִּיחִים מֶלַח עַל הַשֻּׁלְחָן, כִּי הַשֻּׁלְחָן דּוֹמֶה לַמִּזְבֵּחַ: "עַל כָּל קָרְבָּנְךָ תַּקְרִיב מֶלַח".
💛 אַהֲבַת הַשֵּׁם: הָאֹכֶל שֶׁלָּנוּ נוֹתֵן לָנוּ כֹּחַ לַעֲבֹד אֶת ה׳, כְּמוֹ קָרְבָּן. כָּךְ אֲפִלּוּ אֲרוּחַת עֶרֶב רְגִילָה הוֹפֶכֶת לַעֲבוֹדַת ה׳ מְלֵאָה בְּאַהֲבָה.
📖 פֶּרֶק ג׳, הֲלָכָה ה׳

🔀שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה

הַיּוֹם הַלֶּחֶם אָפוּי עִם מֶלַח, וּמֵעִקַּר הַדִּין אֵין צֹרֶךְ לִטְבֹּל אוֹתוֹ. אֲבָל רַבִּים מְהַדְּרִים לִטְבֹּל אוֹתוֹ בְּמֶלַח אוֹ בְּסָלָט, וְיֵשׁ נוֹהֲגִים עַל פִּי הַסּוֹד לִטְבֹּל דַּוְקָא בְּמֶלַח – בְּעִקָּר בְּשַׁבָּתוֹת וּבְיָמִים טוֹבִים. וְעַל עוּגָה בַּסְּעוּדָה – כָּל אֶחָד כְּפִי שֶׁקָּבַע לְעַצְמוֹ.
רוֹצִים לְהַמְשִׁיךְ אֶת הַסְּעוּדָה בְּמָקוֹם אַחֵר? לְמִנְהַג הָאַשְׁכְּנַזִּים וְחֵלֶק מִיּוֹצְאֵי סְפָרַד – מְכַוְּנִים לְכָךְ בִּשְׁעַת "הַמּוֹצִיא". לְמִנְהַג חֵלֶק מִיּוֹצְאֵי סְפָרַד – מְבָרְכִים קֹדֶם בִּרְכַּת הַמָּזוֹן, אוֹ מְבָרְכִים "הַמּוֹצִיא" לְיַד פֶּתַח הַבַּיִת.
💛 אַהֲבַת הַשֵּׁם: כָּל הַמִּנְהָגִים נוֹסְדוּ עַל יְדֵי גְּדוֹלֵי תּוֹרָה. כָּל אֶחָד שׁוֹמֵר עַל מִנְהַג מִשְׁפַּחְתּוֹ בְּאַהֲבָה, וְכֻלָּנוּ עוֹבְדִים אֶת אוֹתוֹ ה׳.
📖 פֶּרֶק ג׳, הֲלָכוֹת ה׳, ח׳, י״א

🍗מָה "הַמּוֹצִיא" פּוֹטֵר?

בִּרְכַּת "הַמּוֹצִיא" פּוֹטֶרֶת אֶת כָּל מָה שֶׁאוֹכְלִים בַּסְּעוּדָה כְּדֵי לִשְׂבֹּעַ: בָּשָׂר, דָּגִים, אֹרֶז, תַּפּוּחֵי אֲדָמָה וְסָלָט – וְגַם מַיִם וּמִיץ. כְּדֵי שֶׁיִּפָּטְרוּ גַּם כְּשֶׁאוֹכְלִים אוֹתָם בְּלִי לֶחֶם, אוֹכְלִים קֹדֶם כְּזַיִת לֶחֶם.
קִנּוּחַ – כְּמוֹ פֵּרוֹת, אֲבַטִּיחַ אוֹ גְּלִידָה – צָרִיךְ בְּרָכָה רִאשׁוֹנָה מִשֶּׁלּוֹ. וְגַם עַל יַיִן מְבָרְכִים בְּנִפְרָד.
💛 אַהֲבַת הַשֵּׁם: "הַמּוֹצִיא" הִיא כְּמוֹ תּוֹדָה גְּדוֹלָה עַל כָּל הָאֲרוּחָה. וְעַל הַקִּנּוּחַ הַמָּתוֹק מוֹסִיפִים עוֹד תּוֹדָה – כִּי כָּל תַּעֲנוּג קָטָן הוּא עוֹד חֶסֶד מֵה׳.
📖 פֶּרֶק ג׳, הֲלָכָה ו׳

🍉פֵּרוֹת וְקִנּוּחִים בַּסְּעוּדָה

פֵּרוֹת וִירָקוֹת שֶׁאוֹכְלִים עִם הַמָּנוֹת הָעִקָּרִיּוֹת – כְּמוֹ צִמּוּקִים בָּאֹרֶז אוֹ סָלָט גֶּזֶר לְיַד הַבָּשָׂר – נִפְטָרִים בְּ"הַמּוֹצִיא". אֲבָל מָה שֶׁבָּא לְטַעַם טוֹב וְלֹא לִשְׂבִיעָה – כְּמוֹ קוֹמְפּוֹט בַּסּוֹף, אוֹ מֶלוֹן בַּהַתְחָלָה – צָרִיךְ בְּרָכָה רִאשׁוֹנָה. חֲמוּצִים בַּהַתְחָלָה פּוֹתְחִים אֶת הַתֵּאָבוֹן, וְהֵם חֵלֶק מֵהַסְּעוּדָה.
עַל קִנּוּחִים מְבָרְכִים רַק לִפְנֵיהֶם וְלֹא אַחֲרֵיהֶם – כִּי בִּרְכַּת הַמָּזוֹן פּוֹטֶרֶת אֶת כָּל מָה שֶׁנֶּאֱכַל בַּסְּעוּדָה.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים מְלַמְּדִים אוֹתָנוּ לָשִׂים לֵב לַמַּטָּרָה שֶׁל כָּל מַאֲכָל. מִי שֶׁשָּׂם לֵב מַרְגִּישׁ אֶת הַחֶסֶד שֶׁל ה׳ בְּכָל שָׁלָב שֶׁל הָאֲרוּחָה.
📖 פֶּרֶק ג׳, הֲלָכוֹת ו׳–ז׳, ט׳

🍰עוּגָה וְסֻפְגָּנִיּוֹת בְּסוֹף הַסְּעוּדָה

עַל עוּגָה שֶׁמַּגִּישִׁים לְקִנּוּחַ בַּסְּעוּדָה נֶחְלְקוּ: יֵשׁ אוֹמְרִים שֶׁ"הַמּוֹצִיא" פּוֹטֶרֶת אוֹתָהּ, כִּי הִיא דּוֹמָה לְלֶחֶם וּמַשְׂבִּיעָה. וְיֵשׁ אוֹמְרִים שֶׁמְּבָרְכִים עָלֶיהָ "מְזוֹנוֹת", כִּי הִיא קִנּוּחַ. כָּל אֶחָד מַחְלִיט אֵיךְ הוּא נוֹהֵג, וְנוֹהֵג כָּךְ תָּמִיד.
מִי שֶׁעוֹד לֹא הֶחְלִיט – לֹא מְבָרֵךְ "מְזוֹנוֹת", כִּי סְפֵק בְּרָכוֹת לְהָקֵל. עַל סֻפְגָּנִיּוֹת בְּסוֹף הַסְּעוּדָה לֹא מְבָרְכִים, וְעָדִיף לֶאֱכֹל אוֹתָן מִחוּץ לַסְּעוּדָה.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁיֵּשׁ סָפֵק, אֶפְשָׁר לִבְחֹר דֶּרֶךְ בְּרוּרָה וּלְהַתְמִיד בָּהּ – וְכָךְ כָּל בְּרָכָה נֶאֱמֶרֶת בְּלֵב שָׁלֵם וּבְשִׂמְחָה.
📖 פֶּרֶק ג׳, הֲלָכָה ח׳

⏳עַד מָתַי הַבְּרָכָה מוֹעִילָה?

"הַמּוֹצִיא" פּוֹטֵר אֶת כָּל הַסְּעוּדָה, גַּם אִם הִיא נִמְשֶׁכֶת שָׁעוֹת. מִי שֶׁהִפְסִיק כְּדֵי לְהִתְפַּלֵּל, אוֹ הִתְנַמְנֵם לְיַד הַשֻּׁלְחָן – מַמְשִׁיךְ לֶאֱכֹל בְּלִי בְּרָכָה. אֲבָל מִי שֶׁהָלַךְ לִישֹׁן עַל מִטָּתוֹ – נוֹטֵל יָדַיִם וּמְבָרֵךְ שׁוּב.
מַאֲכָל שֶׁהֵבִיאוּ פִּתְאֹם בְּאֶמְצַע הַסְּעוּדָה, וְלֹא חָשַׁבְתָּ עָלָיו בִּשְׁעַת "הַמּוֹצִיא" – צָרִיךְ בְּרָכָה מִשֶּׁלּוֹ.
💛 אַהֲבַת הַשֵּׁם: הַבְּרָכָה שֶׁבִּתְחִלַּת הַסְּעוּדָה מְלַוָּה אוֹתָנוּ לְאֹרֶךְ כָּל הָאֲרוּחָה. וּכְשֶׁמַּגִּיעַ דָּבָר חָדָשׁ וּמַפְתִּיעַ – אוֹמְרִים עָלָיו תּוֹדָה חֲדָשָׁה לַה׳.
📖 פֶּרֶק ג׳, הֲלָכָה י׳

🚶יוֹצְאִים בְּאֶמְצַע הַסְּעוּדָה?

לְכַתְּחִלָּה לֹא יוֹצְאִים מֵהַבַּיִת בְּאֶמְצַע הַסְּעוּדָה, כִּי יֵשׁ אוֹמְרִים שֶׁהַיְּצִיאָה מַפְסִיקָה אֶת הַסְּעוּדָה. אִם אֶחָד מִן הָאוֹכְלִים נִשְׁאָר בַּבַּיִת – מֻתָּר לָצֵאת כְּשֶׁיֵּשׁ צֹרֶךְ, לְמָשָׁל לְקַבֵּל פְּנֵי אוֹרְחִים.
הוֹלְכֵי דְּרָכִים, שֶׁאֵין לָהֶם מָקוֹם קָבוּעַ, רַשָּׁאִים לְהַתְחִיל לֶאֱכֹל בְּמָקוֹם אֶחָד וּלְהַמְשִׁיךְ לֶאֱכֹל בַּדֶּרֶךְ.
💛 אַהֲבַת הַשֵּׁם: הַסְּעוּדָה הִיא זְמַן מְיֻחָד שֶׁל הוֹדָאָה לַה׳, וְלֹא קוֹטְעִים אוֹתָהּ סְתָם. וּמִי שֶׁנִּמְצָא בַּדֶּרֶךְ – ה׳ מְלַוֶּה אוֹתוֹ וְאֶת הַבְּרָכָה שֶׁלּוֹ לְכָל מָקוֹם.
📖 פֶּרֶק ג׳, הֲלָכָה י״א

🥨לֹא מְנַשְׁנְשִׁים לִפְנֵי הַסְּעוּדָה

מִי שֶׁעוֹמֵד לֶאֱכֹל סְעוּדָה, לֹא אוֹכֵל וְלֹא שׁוֹתֶה לִפְנֶיהָ. אֶת מָה שֶׁאָכַל קֹדֶם הָיָה "הַמּוֹצִיא" פּוֹטֵר – וְכָךְ גָּרַם לִבְרָכָה שֶׁאֵינָהּ צְרִיכָה, וְגַם נִכְנַס לְסָפֵק אִם לְבָרֵךְ אַחֲרֶיהָ בְּרָכָה אַחֲרוֹנָה.
מִי שֶׁמֻּכְרָח לֶאֱכֹל מַשֶּׁהוּ לִפְנֵי הַסְּעוּדָה – יְבָרֵךְ אַחֲרָיו בְּרָכָה אַחֲרוֹנָה, וְיַמְתִּין רֶבַע עַד חֲצִי שָׁעָה אוֹ יֵצֵא לְטִיּוּל קָצָר, לִפְנֵי שֶׁיַּתְחִיל אֶת הַסְּעוּדָה.
💛 אַהֲבַת הַשֵּׁם: כָּל בְּרָכָה הִיא דָּבָר יָקָר, וְלֹא מוֹסִיפִים בְּרָכוֹת שֶׁלֹּא לְצֹרֶךְ. מִי שֶׁמְּכַבֵּד אֶת שֵׁם ה׳ מְתַכְנֵן אֶת הָאֲכִילָה שֶׁלּוֹ בְּחָכְמָה.
📖 פֶּרֶק ג׳, הֲלָכָה י״ב

🧠 בְּדֹק אֶת עַצְמְךָ

שְׁמוּאֵל בֵּרֵךְ "הַמּוֹצִיא", וְלִפְנֵי שֶׁטָּעַם אָמַר לְאָחִיו: "תַּעֲבִיר לִי אֶת הַמֶּלַח". מָה הַדִּין?
💡 דִּבּוּר בֵּין הַבְּרָכָה לָאֲכִילָה מַפְסִיק – אֲבָל דִּבּוּר עַל צָרְכֵי הָאֲכִילָה, כְּמוֹ מֶלַח אוֹ סַכִּין, אֵינוֹ הֶפְסֵק. בְּכָל זֹאת, לְכַתְּחִלָּה טוֹעֲמִים קֹדֶם וּמְדַבְּרִים אַחַר כָּךְ.
בְּיוֹם חֹל מַתְחִילִים לַחְתֹּךְ לֶחֶם גָּדוֹל לִפְנֵי הַבְּרָכָה. לָמָּה בְּשַׁבָּת לֹא חוֹתְכִים אֶת הַחַלָּה לִפְנֵי הַבְּרָכָה?
💡 בְּשַׁבָּת מִצְוָה לְבָרֵךְ עַל שְׁתֵּי חַלּוֹת שְׁלֵמוֹת. מִי שֶׁמַּתְחִיל לַחְתֹּךְ – הַחַלָּה כְּבָר לֹא שְׁלֵמָה. (לַחְמָנִיָּה וּפִתָּה מְבָרְכִים שְׁלֵמוֹת מִסִּבָּה אַחֶרֶת: קַל לִבְצֹעַ אוֹתָן מַהֵר.)
בִּסְעוּדַת בַּר מִצְוָה גְּדוֹלָה נָטְלוּ אוֹרְחִים רַבִּים יָדַיִם. אֵיךְ עָדִיף לְבָרֵךְ "הַמּוֹצִיא"?
💡 כְּשֶׁאוֹכְלִים יַחַד, טוֹב שֶׁאֶחָד יְבָרֵךְ לָאֲחֵרִים. אֲבָל בִּסְעוּדָה גְּדוֹלָה הַהַמְתָּנָה לְכֻלָּם תִּגְרֹם הֶפְסֵק אָרֹךְ מִדַּי בֵּין הַנְּטִילָה לַבְּרָכָה – וְלָכֵן מְבָרְכִים בְּכָל שֻׁלְחָן.

📜אַרְבַּע פָּרָשִׁיּוֹת

בַּתְּפִלִּין יֵשׁ אַרְבַּע פָּרָשִׁיּוֹת מִן הַתּוֹרָה, וּבְכָל אַחַת מֵהֶן כְּתוּבָה מִצְוַת הַתְּפִלִּין: קַדֶּשׁ, וְהָיָה כִּי יְבִיאֲךָ, שְׁמַע, וְהָיָה אִם שָׁמֹעַ. בַּשְּׁתַּיִם הָרִאשׁוֹנוֹת – יְצִיאַת מִצְרַיִם וּקְדֻשַּׁת יִשְׂרָאֵל. בַּשְּׁתַּיִם הָאַחֲרוֹנוֹת – אַחְדוּת ה׳, אַהֲבַת ה׳ וְלִמּוּד הַתּוֹרָה לַבָּנִים.
בְּאַרְבַּע הַפָּרָשִׁיּוֹת הָאֵלֶּה כְּתוּבִים עִקְּרֵי הָאֱמוּנָה שֶׁל עַם יִשְׂרָאֵל.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁמַּנִּיחִים תְּפִלִּין, קוֹשְׁרִים עַל הַגּוּף אֶת הַדְּבָרִים הֲכִי חֲשׁוּבִים שֶׁה׳ אָמַר לָנוּ. זֶה כְּמוֹ לְהַחֲזִיק קָרוֹב לַלֵּב מִכְתָּב אָהוּב מֵאַבָּא – וְלֹא לְהִפָּרֵד מִמֶּנּוּ.

📦בַּיִת אֶחָד וְאַרְבָּעָה בָּתִּים

אֶת הַפָּרָשִׁיּוֹת מַכְנִיסִים לְבַיִת קָטָן וּמְרֻבָּע, עָשׂוּי מֵעוֹר שֶׁל בְּהֵמָה טְהוֹרָה. בַּתְּפִלִּין שֶׁל יָד כָּל אַרְבַּע הַפָּרָשִׁיּוֹת כְּתוּבוֹת עַל קְלָף אֶחָד, בְּבַיִת אֶחָד. בַּתְּפִלִּין שֶׁל רֹאשׁ יֵשׁ אַרְבָּעָה בָּתִּים קְטַנִּים, וּבְכָל אֶחָד – פָּרָשָׁה עַל קְלָף נִפְרָד.
אוּלַי יֵשׁ בָּזֶה רֶמֶז: בָּרֹאשׁ יֵשׁ מָקוֹם לְהַרְבֵּה מַחֲשָׁבוֹת שׁוֹנוֹת, אֲבָל כְּשֶׁמַּגִּיעִים לְמַעֲשֶׂה – יֵשׁ הֲלָכָה אַחַת.
💛 אַהֲבַת הַשֵּׁם: מֻתָּר לִשְׁאֹל, לַחְשֹׁב וּלְהִתְלַבֵּט – זֶה הָרֹאשׁ. אֲבָל כְּשֶׁבָּא הַזְּמַן לַעֲשׂוֹת, מִי שֶׁאוֹהֵב אֶת ה׳ הוֹלֵךְ בַּדֶּרֶךְ הָאַחַת שֶׁהוּא צִוָּה.

⬛הֲלָכָה לְמֹשֶׁה מִסִּינַי

רֹב הַדִּינִים שֶׁל עֲשִׂיַּת הַתְּפִלִּין נִמְסְרוּ לְמֹשֶׁה בְּסִינַי: הַבָּתִּים מְרֻבָּעִים, עַל הַתְּפִלִּין שֶׁל רֹאשׁ יֵשׁ צוּרַת הָאוֹת שִׁי״ן, הֵן תְּפוּרוֹת בְּגִידִים, וְיֵשׁ לָהֶן מַעְבַּרְתָּא – מַעֲבָר שֶׁבּוֹ נִכְנֶסֶת הָרְצוּעָה.
הָרְצוּעוֹת שְׁחֹרוֹת, וְהַקֶּשֶׁר שֶׁלָּהֶן בְּצוּרַת הָאוֹת דָּלֶת.
💛 אַהֲבַת הַשֵּׁם: אֶת הַפְּרָטִים הָאֵלֶּה לִמֵּד ה׳ אֶת מֹשֶׁה, וּמֵאָז הֵם עוֹבְרִים מֵאָב לְבֵן. כְּשֶׁתַּנִּיחַ תְּפִלִּין, תַּנִּיחַ בְּדִיּוּק כְּמוֹ אֲבוֹתֶיךָ – שַׁרְשֶׁרֶת שֶׁל אַהֲבָה שֶׁמַּגִּיעָה עַד הַר סִינַי.

📦🔑 מוּשָּׂג: בַּיִת

״עוֹשִׂים בֵּית קָטָן וּמְרֻבָּע מֵעוֹר שֶׁל בְּהֵמָה טְהוֹרָה... וּבִתְפִלִּין שֶׁל רֹאשׁ עוֹשִׂים בַּיִת שֶׁמִּתְחַלֵּק לְאַרְבָּעָה בָּתִּים קְטַנִּים, וְכוֹתְבִים אֶת הַפָּרָשִׁיּוֹת עַל אַרְבָּעָה קְלָפִים נִפְרָדִים״.
📖 בַּיִת – הַתָּא הַקָּטָן מֵעוֹר שֶׁבּוֹ מַנִּיחִים אֶת הַקְּלָף עִם הַפָּרָשָׁה.
💛 אַהֲבַת הַשֵּׁם: 💛 בָּרֹאשׁ יֵשׁ מָקוֹם לְהַרְבֵּה מַחֲשָׁבוֹת שׁוֹנוֹת, וְלָכֵן אַרְבָּעָה בָּתִּים נִפְרָדִים – אֲבָל בִּתְפִלִּין שֶׁל יָד, בְּבַיִת אֶחָד, יֵשׁ הֲלָכָה אַחַת. 🧒 וְאַתָּה? כְּשֶׁתִּרְאֶה תְּפִלִּין שֶׁל רֹאשׁ – סְפֹר בְּעֵינֶיךָ אֶת אַרְבַּעַת הַבָּתִּים.

⚫📜 לְשׁוֹן חֲכָמִים: רְצוּעָה

״וְהָרְצוּעוֹת יִהְיוּ שְׁחֹרוֹת, וְהַקֶּשֶׁר שֶׁלָּהֶם כְּעֵין ד׳״.
📖 רְצוּעָה – הָרְצוּעָה הַשְּׁחֹרָה שֶׁקּוֹשְׁרִים בָּהּ אֶת הַתְּפִלִּין לַיָּד וְלָרֹאשׁ.
💛 אַהֲבַת הַשֵּׁם: 💛 כָּל פְּרָט בַּתְּפִלִּין – הַצֶּבַע, הַקֶּשֶׁר, הַצּוּרָה – נִמְסַר לְמֹשֶׁה בְּסִינַי, כְּדֵי שֶׁנֵּדַע שֶׁה׳ בְּעַצְמוֹ קָבַע אֶת הַמִּצְוָה הַזֹּאת בְּדִיּוּק. 🧒 וְאַתָּה? לִפְנֵי שֶׁתָּנִיחַ תְּפִלִּין, בְּדֹק שֶׁהָרְצוּעוֹת שְׁלֵמוֹת וּשְׁחֹרוֹת.

📜📜 לְשׁוֹן חֲכָמִים: קְלָף

״הַפָּרָשִׁיּוֹת שֶׁבַּתְּפִלִּין וּבַמְּזוּזָה כְּתוּבוֹת עַל קְלָף, הֶעָשׂוּי מֵעוֹר שֶׁל בְּהֵמָה טְהוֹרָה״.

💛 אַהֲבַת הַשֵּׁם: 💛 הַפָּרָשִׁיּוֹת כְּתוּבוֹת עַל קְלָף מֵעוֹר בְּהֵמָה, כְּדֵי לְרַמֵּז שֶׁהָרַעֲיוֹנוֹת הָעֶלְיוֹנִים שֶׁל הַתּוֹרָה חוֹדְרִים עַד הַחֹמֶר הַגַּשְׁמִי. 🧒 וְאַתָּה? כְּשֶׁתִּרְאֶה סוֹפֵר סְתָ״ם כּוֹתֵב, שְׁאַל אוֹתוֹ עַל הַקְּלָף שֶׁהוּא כּוֹתֵב עָלָיו.

🛡️שׁוֹמְרִים עַל כְּבוֹד הַתְּפִלִּין

הַתְּפִלִּין קְדוֹשׁוֹת, וְלָכֵן נִזְהָרִים מְאוֹד בִּכְבוֹדָן. לֹא נִכְנָסִים בָּהֶן לְבֵית הַכִּסֵּא, וְלֹא אוֹכְלִים בָּהֶן סְעוּדָה קְבוּעָה. מִי שֶׁהַתְּפִלִּין בְּתוֹךְ הַתִּיק שֶׁלּוֹ – לֹא יֵשֵׁב עַל הַתִּיק, אֶלָּא אִם כֵּן הוּא יוֹדֵעַ שֶׁהֵן בַּצַּד הַשֵּׁנִי.
מִן הַתּוֹרָה אֶפְשָׁר לְהָנִיחַ תְּפִלִּין גַּם בַּלַּיְלָה, אֲבָל חֲכָמִים אָסְרוּ – שֶׁמָּא יֵרָדֵם בָּהֶן, וְהַתְּפִלִּין יְבֻזּוּ.
💛 אַהֲבַת הַשֵּׁם: אֶת מָה שֶׁיָּקָר לָנוּ אֲנַחְנוּ שׁוֹמְרִים בִּזְהִירוּת. הַכָּבוֹד שֶׁאַתָּה נוֹתֵן לַתְּפִלִּין הוּא כָּבוֹד לַה׳, שֶׁשְּׁמוֹ כָּתוּב בָּהֶן.

🌙שָׁכַח כָּל הַיּוֹם

מִי שֶׁשָּׁכַח לְהָנִיחַ תְּפִלִּין כָּל הַיּוֹם – לֹא יַנִּיחַ אוֹתָן בַּלַּיְלָה, כִּי חֲכָמִים אָסְרוּ זֹאת. אֲבָל אִם נִזְכַּר בֵּין הַשְּׁמָשׁוֹת, כְּשֶׁעוֹד לֹא חָשְׁכָה מַמָּשׁ – יַנִּיחַ אוֹתָן בְּלִי בְּרָכָה.
בֵּין הַשְּׁמָשׁוֹת הוּא זְמַן שֶׁל סָפֵק – אוּלַי עוֹד יוֹם וְאוּלַי כְּבָר לַיְלָה. לָכֵן מַנִּיחִים, אֲבָל לֹא מְבָרְכִים.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁשּׁוֹכְחִים, לֹא מְוַתְּרִים עַל מָה שֶׁעוֹד אֶפְשָׁר לַעֲשׂוֹת. וּמָחָר בַּבֹּקֶר אֶפְשָׁר לְהַתְחִיל מֵחָדָשׁ – ה׳ שָׂמֵחַ בְּכָל צַעַד שֶׁלְּךָ אֵלָיו.

✋מְמַשְׁמְשִׁים בַּתְּפִלִּין

כָּל זְמַן שֶׁהַתְּפִלִּין עָלֶיךָ, אָסוּר לְהַסִּיחַ מֵהֶן אֶת הַדַּעַת. לָכֵן נוֹגְעִים בָּהֶן בַּיָּד מִפַּעַם לְפַעַם – וְזֶה נִקְרָא מִשְׁמוּשׁ. קֹדֶם נוֹגְעִים בַּתְּפִלִּין שֶׁל יָד, וְאַחַר כָּךְ בַּתְּפִלִּין שֶׁל רֹאשׁ.
לָמָּה קֹדֶם שֶׁל יָד? כִּי הִיא קְרוֹבָה יוֹתֵר לַיָּד הַנּוֹגַעַת, וְ"אֵין מַעֲבִירִין עַל הַמִּצְווֹת" – לֹא מְדַלְּגִים עַל מִצְוָה שֶׁפּוֹגְשִׁים רִאשׁוֹנָה.
💛 אַהֲבַת הַשֵּׁם: כְּשֶׁמִּישֶׁהוּ אָהוּב יוֹשֵׁב לְיָדְךָ, אַתָּה לֹא שׁוֹכֵחַ שֶׁהוּא שָׁם. הַמִּשְׁמוּשׁ הוּא כְּמוֹ לְהַגִּיד לַה׳ בְּשֶׁקֶט: אֲנִי זוֹכֵר שֶׁאַתָּה אִתִּי.

💭מָה זֶה הֶסַּח הַדַּעַת?

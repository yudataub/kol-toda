דיני המקום והזמן

🕎 הִלְכוֹת חֲנֻכָּה · פֶּרֶק י״ג

🚪עַל פֶּתַח הַבַּיִת מִבַּחוּץ

חֲכָמִים אָמְרוּ: ״נֵר חֲנֻכָּה מִצְוָה לְהַנִּיחוֹ עַל פֶּתַח בֵּיתוֹ מִבַּחוּץ״, כְּדֵי שֶׁהָעוֹבְרִים בָּרְחוֹב יִרְאוּ וְיִתְפַּרְסֵם הַנֵּס. מִי שֶׁגָּר בַּקּוֹמָה הַשְּׁנִיָּה – מַנִּיחַ בַּחַלּוֹן שֶׁפּוֹנֶה לָרְחוֹב. וּבִזְמַן סַכָּנָה מֵהַגּוֹיִים – מַנִּיחַ עַל הַשֻּׁלְחָן בְּתוֹךְ הַבַּיִת.
מַדְלִיקִים בְּצַד שְׂמֹאל שֶׁל הַפֶּתַח, וְכָךְ מִי שֶׁנִּכְנָס מֻקָּף בְּמִצְווֹת: הַמְּזוּזָה מִיָּמִין, וְהַנֵּר מִשְּׂמֹאל.
💛 אַהֲבַת הַשֵּׁם: הַבַּיִת הַיְּהוּדִי מְחַבֵּק אֶת מִי שֶׁנִּכְנָס בּוֹ בְּמִצְווֹת מִשְּׁנֵי הַצְּדָדִים. כָּךְ אוֹהֲבִים אֶת ה׳ – בַּכְּנִיסָה, בַּיְּצִיאָה וּבְכָל מָקוֹם.
📖 פֶּרֶק י״ג, הֲלָכָה א׳

📏בְּאֵיזֶה גֹּבַהּ?

לְכַתְּחִלָּה מַנִּיחִים אֶת הַחֲנֻכִּיָּה כָּךְ שֶׁהַלֶּהָבָה תִּהְיֶה בֵּין שְׁלֹשָׁה לַעֲשָׂרָה טְפָחִים – בְּעֵרֶךְ בֵּין 23 לְ־76 ס״מ. גָּבוֹהַּ יוֹתֵר – יַחְשְׁבוּ שֶׁהַנֵּר נוֹעַד לְהָאִיר אֶת הַכְּנִיסָה. עַל הָרִצְפָּה – יֵרָאֶה כְּאִלּוּ הִנִּיחוּ אוֹתוֹ שָׁם רַק לְרֶגַע.
בְּדִיעֲבַד, גַּם מִי שֶׁהִדְלִיק נָמוּךְ אוֹ גָּבוֹהַּ מִזֶּה – יָצָא. אֲבָל בְּפֶתַח גָּבוֹהַּ מְאוֹד, מֵעַל עֶשְׂרִים אַמָּה (כְּתִשְׁעָה מֶטֶר) – לֹא יָצָא, כִּי אֲנָשִׁים לֹא מִסְתַּכְּלִים כָּל כָּךְ לְמַעְלָה.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים חָשְׁבוּ גַּם עַל הָעֵינַיִם שֶׁל הָעוֹבְרִים בָּרְחוֹב: שֶׁיִּרְאוּ וְיָבִינוּ מִיָּד שֶׁזֶּה נֵר שֶׁל מִצְוָה.
📖 פֶּרֶק י״ג, הֲלָכָה א׳

🏠בְּבַיִת פְּרָטִי

הַיּוֹם, נֵרוֹת שֶׁמַּדְלִיקִים מִחוּץ לַפֶּתַח עֲלוּלִים לִכְבּוֹת בָּרוּחַ. מִי שֶׁרוֹצֶה לְהַדֵּר בְּכָל הַהִדּוּרִים – קוֹנֶה תֵּבַת זְכוּכִית וּמַדְלִיק בָּהּ מִחוּץ לַפֶּתַח, בְּצַד שְׂמֹאל. וּמִי שֶׁלֹּא – מַדְלִיק בְּתוֹךְ הַבַּיִת, וּבְחַלּוֹן שֶׁפּוֹנֶה לָרְחוֹב הוּא מְהַדֵּר בְּפִרְסוּם הַנֵּס.
כְּשֶׁהַפֶּתַח מֻסְתָּר מֵהָרְחוֹב וְהַחַלּוֹן נִרְאֶה יוֹתֵר: יֵשׁ אוֹמְרִים שֶׁעָדִיף הַפֶּתַח וְיֵשׁ אוֹמְרִים הַחַלּוֹן. הָרַב מֶלָמֵד כּוֹתֵב שֶׁהַחַלּוֹן מְהֻדָּר יוֹתֵר, וְגַם בַּפֶּתַח יֵשׁ מַעֲלָה.
💛 אַהֲבַת הַשֵּׁם: אֲנַחְנוּ לֹא מַסְתִּירִים אֶת הָאוֹר. אֲנַחְנוּ שְׂמֵחִים שֶׁכָּל הָרְחוֹב יִרְאֶה כַּמָּה טוֹב ה׳ עָשָׂה לָנוּ.
📖 פֶּרֶק י״ג, הֲלָכָה ב׳

🏢בְּבִנְיַן דִּירוֹת

רֹב הָאֲנָשִׁים גָּרִים הַיּוֹם בְּבִנְיְנֵי קוֹמוֹת. הַמָּקוֹם הַטּוֹב בְּיוֹתֵר הוּא חַלּוֹן אוֹ מִרְפֶּסֶת שֶׁפּוֹנִים לָרְחוֹב. לֹא לוֹקְחִים חֲנֻכִּיָּה עִם גַּב, כִּי הַגַּב מַסְתִּיר אֶת הַנֵּרוֹת מִצַּד אֶחָד.
יֵשׁ סוֹבְרִים שֶׁעָדִיף לְהַדְלִיק עַל פֶּתַח הַדִּירָה שֶׁפּוֹנֶה לַחֲדַר הַמַּדְרֵגוֹת, כְּדֵי שֶׁהַנִּכְנָס יִהְיֶה מֻקָּף בְּמִצְווֹת – וְגַם כָּךְ יוֹצְאִים יְדֵי חוֹבָה. הָרַב מֶלָמֵד כּוֹתֵב שֶׁטּוֹב יוֹתֵר בַּחַלּוֹן, כְּדֵי לְפַרְסֵם אֶת הַנֵּס, גַּם אִם הוּא גָּבוֹהַּ מֵעֲשָׂרָה טְפָחִים, וְגַם בְּקוֹמָה גְּבוֹהָה אִם רוֹאִים מֵהָרְחוֹב.
💛 אַהֲבַת הַשֵּׁם: כָּל חַלּוֹן מוּאָר בַּבִּנְיָן הוּא כְּמוֹ קוֹל קָטָן שֶׁאוֹמֵר: ״כָּאן גָּרִים יְהוּדִים שֶׁמּוֹדִים לַה׳״.
📖 פֶּרֶק י״ג, הֲלָכָה ג׳

🔀יֵשׁ נוֹהֲגִים כָּךְ וְיֵשׁ נוֹהֲגִים כָּךְ

מוֹצָאֵי שַׁבַּת חֲנֻכָּה: רַבִּים נוֹהֲגִים לְהַבְדִּיל קֹדֶם, וְיֵשׁ נוֹהֲגִים לְהַדְלִיק נֵר חֲנֻכָּה קֹדֶם. זְמַן הַהַדְלָקָה: יֵשׁ נוֹהֲגִים לְהַדְלִיק בַּשְּׁקִיעָה, וְהַמִּנְהָג הָרוֹוֵחַ – בְּצֵאת הַכּוֹכָבִים.
בְּבִנְיַן דִּירוֹת: יֵשׁ סוֹבְרִים שֶׁעָדִיף פֶּתַח הַדִּירָה, שֶׁיִּהְיֶה הַנִּכְנָס מֻקָּף בְּמִצְווֹת. הָרַב מֶלָמֵד כּוֹתֵב שֶׁעָדִיף הַחַלּוֹן לָרְחוֹב, שֶׁשָּׁם הַנֵּס מִתְפַּרְסֵם יוֹתֵר.
💛 אַהֲבַת הַשֵּׁם: לְכָל מִנְהָג יֵשׁ טַעַם טוֹב וּמָקוֹם בַּהֲלָכָה. כְּשֶׁאָנוּ מְכַבְּדִים אֶת מִי שֶׁנּוֹהֵג אַחֶרֶת, אֲנַחְנוּ מוֹסִיפִים אוֹר שֶׁל אַהֲבַת יִשְׂרָאֵל.
📖 פֶּרֶק י״ג, הֲלָכוֹת ג׳–ה׳

🌃מָתַי מַדְלִיקִים?

חֲכָמִים קָבְעוּ לְהַדְלִיק ״מִשֶּׁתִּשְׁקַע הַחַמָּה וְעַד שֶׁתִּכְלֶה רֶגֶל מִן הַשּׁוּק״ – כְּשֶׁכְּבָר חָשׁוּךְ, וְהָרְחוֹבוֹת עוֹד מְלֵאִים אֲנָשִׁים. נֶחְלְקוּ הָרִאשׁוֹנִים: יֵשׁ אוֹמְרִים מִתְּחִלַּת הַשְּׁקִיעָה, וְיֵשׁ אוֹמְרִים מִסּוֹפָהּ – צֵאת הַכּוֹכָבִים. יֵשׁ נוֹהֲגִים לְהַדְלִיק בַּשְּׁקִיעָה, וְהַמִּנְהָג הָרוֹוֵחַ – בְּצֵאת הַכּוֹכָבִים, כְּעֶשְׂרִים דַּקּוֹת אַחֲרֵי הַשְּׁקִיעָה.
מִי שֶׁמִּתְפַּלֵּל עַרְבִית בְּצֵאת הַכּוֹכָבִים – מִתְפַּלֵּל קֹדֶם, וּמְמַהֵר הַבַּיְתָה לְהַדְלִיק. וְטוֹב לְהַכִין נֵרוֹת שֶׁיִּדְלְקוּ שְׁעָתַיִם וְיוֹתֵר, כִּי הַיּוֹם אֲנָשִׁים מִסְתּוֹבְבִים בָּרְחוֹב עַד מְאֻחָר.
💛 אַהֲבַת הַשֵּׁם: בְּדִיּוּק כְּשֶׁהַחֹשֶׁךְ יוֹרֵד, אֲנַחְנוּ מַדְלִיקִים. הַזְּמַן הַזֶּה הוּא הַזְּמַן שֶׁבּוֹ הָאוֹר שֶׁלָּנוּ נִרְאֶה הֲכִי טוֹב.
📖 פֶּרֶק י״ג, הֲלָכָה ד׳

🕯️עֶרֶב שַׁבַּת חֲנֻכָּה

בְּעֶרֶב שַׁבָּת אִי אֶפְשָׁר לְהַדְלִיק אַחֲרֵי הַשְּׁקִיעָה, כִּי הַשַּׁבָּת כְּבָר נִכְנְסָה. לָכֵן מַדְלִיקִים נֵרוֹת חֲנֻכָּה לִפְנֵי נֵרוֹת שַׁבָּת.
עִקַּר פִּרְסוּם הַנֵּס הוּא בַּלַּיְלָה, וְלָכֵן מַקְפִּידִים שֶׁיִּהְיֶה מַסְפִּיק שֶׁמֶן, כְּדֵי שֶׁהַנֵּרוֹת יִדְלְקוּ עַד חֲצִי שָׁעָה אַחֲרֵי צֵאת הַכּוֹכָבִים. וְטוֹב לְהִתְפַּלֵּל קֹדֶם מִנְחָה בְּצִבּוּר.
💛 אַהֲבַת הַשֵּׁם: בְּשַׁבַּת חֲנֻכָּה שְׁנֵי אוֹרוֹת נִפְגָּשִׁים: אוֹר הַשַּׁבָּת וְאוֹר הַנֵּס. הַבַּיִת מָלֵא קְדֻשָּׁה וְשִׂמְחָה כִּפְלַיִם.
📖 פֶּרֶק י״ג, הֲלָכָה ה׳

✨מוֹצָאֵי שַׁבַּת חֲנֻכָּה

בְּמוֹצָאֵי שַׁבָּת מִתְפַּלְּלִים עַרְבִית, וְאַחַר כָּךְ מַדְלִיקִים. רַבִּים נוֹהֲגִים לְהַבְדִּיל קֹדֶם, כִּי הַהַבְדָּלָה מְסַיֶּמֶת אֶת הַשַּׁבָּת. וְיֵשׁ נוֹהֲגִים לְהַדְלִיק קֹדֶם נֵר חֲנֻכָּה, כְּדֵי לְהַדְלִיק קָרוֹב לְצֵאת הַכּוֹכָבִים וּלְהַאֲרִיךְ אֶת הַשַּׁבָּת.
מִי שֶׁמַּדְלִיק לִפְנֵי הַהַבְדָּלָה – אוֹמֵר קֹדֶם ״אַתָּה חוֹנַנְתָּנוּ״ בַּתְּפִלָּה, אוֹ ״בָּרוּךְ הַמַּבְדִּיל בֵּין קֹדֶשׁ לְחֹל״. לִשְׁנֵי הַמִּנְהָגִים יֵשׁ מָקוֹם בַּהֲלָכָה.
💛 אַהֲבַת הַשֵּׁם: הַשַּׁבָּת נִפְרֶדֶת מֵאִתָּנוּ – וְהַנֵּרוֹת שֶׁל חֲנֻכָּה מַמְשִׁיכִים אֶת הָאוֹר אֶל הַשָּׁבוּעַ הֶחָדָשׁ.
📖 פֶּרֶק י״ג, הֲלָכָה ה׳

⏰מִי שֶׁמְּאַחֵר לַחֲזֹר

הַזְּמַן הַמְּהֻדָּר הוּא מִצֵּאת הַכּוֹכָבִים וְעַד חֲצִי שָׁעָה אַחֲרָיו. מִי שֶׁקָּשֶׁה לוֹ לַחֲזֹר הַבַּיְתָה בַּזְּמַן – רַשַּׁאי לְהַדְלִיק מְאֻחָר יוֹתֵר, וּמִשְׁתַּדֵּל לֹא לְאַחֵר אַחֲרֵי הַשָּׁעָה תֵּשַׁע. עַד שֶׁמַּדְלִיק – לֹא אוֹכֵל אֲרוּחָה.
בְּבֵית כְּנֶסֶת שֶׁיֵּשׁ בּוֹ שִׁעוּר תּוֹרָה קָבוּעַ אַחֲרֵי עַרְבִית – עָדִיף לְקַיֵּם אֶת הַשִּׁעוּר, וּבְסוֹפוֹ לְהַזְכִּיר לְכֻלָּם לְהַדְלִיק.
💛 אַהֲבַת הַשֵּׁם: גַּם מִי שֶׁעָסוּק מְאוֹד שׁוֹמֵר לַמִּצְוָה מָקוֹם בַּלֵּב. וְלִמּוּד תּוֹרָה – הוּא הָאוֹר הֲכִי גָּדוֹל.
📖 פֶּרֶק י״ג, הֲלָכָה ו׳

💡 אַחֲרֵי תֵּשַׁע מַדְלִיקִים כָּל הַלַּיְלָה עַד עֲלוֹת הַשַּׁחַר. מְבָרְכִים רַק אִם מִישֶׁהוּ יִרְאֶה אֶת הַנֵּרוֹת – וְכָאן כֻּלָּם יְשֵׁנִים וְהָרְחוֹב רֵיק, וְלָכֵן בְּלִי בְּרָכָה.
יוֹסֵף שָׁכַח לְגַמְרֵי לְהַדְלִיק בַּלַּיְלָה הַשְּׁלִישִׁי. כַּמָּה נֵרוֹת יַדְלִיק בַּלַּיְלָה הָרְבִיעִי?
💡 מִי שֶׁלֹּא הִדְלִיק כָּל הַלַּיְלָה הִפְסִיד אֶת הַמִּצְוָה שֶׁל אוֹתוֹ לַיְלָה בִּלְבַד. לְמָחֳרָת מַדְלִיק כְּמוֹ כָּל הָאֲנָשִׁים – אַרְבָּעָה נֵרוֹת.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

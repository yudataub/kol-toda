בִּרְכוֹת הָרְאִיָּה

🍎 הִלְכוֹת בְּרָכוֹת · פֶּרֶק ט״ו

🏞️נוֹפִים מַרְשִׁימִים

עַל יָם, אֲגַם, נָהָר גָּדוֹל, הַר גָּבוֹהַּ וּמִדְבָּר מְבָרְכִים ״עוֹשֶׂה מַעֲשֵׂה בְרֵאשִׁית״ – אִם לֹא רָאִינוּ אוֹתָם שְׁלוֹשִׁים יוֹם. עַל אֲגַם שֶׁבְּנֵי אָדָם יָצְרוּ בְּסֶכֶר לֹא מְבָרְכִים, כִּי הַבְּרָכָה הִיא עַל מַעֲשֵׂה הַבּוֹרֵא. וְעַל נוֹף בִּתְמוּנָה אוֹ בְּסֶרֶט – לֹא מְבָרְכִים.
מִי שֶׁגָּר לְיַד הַיָּם וְיָכוֹל לִרְאוֹת אוֹתוֹ תָּמִיד – לֹא מְבָרֵךְ עָלָיו. בְּטִיּוּל מְבָרְכִים; בִּנְסִיעָה רְגִילָה – רַק אִם שָׂם לֵב וְהִתְפַּעֵל.
💛 אַהֲבַת הַשֵּׁם: הַיָּם וְהֶהָרִים מְסַפְּרִים עַל גְּדֻלַּת הַבּוֹרֵא. כְּשֶׁאָנוּ עוֹצְרִים וּמְבָרְכִים, הַנּוֹף הַיָּפֶה נַעֲשֶׂה שִׁיר תּוֹדָה לַה׳.
📖 פֶּרֶק ט״ו, הֲלָכוֹת א׳–ה׳

🔭רוֹאִים בָּעֵינַיִם

בִּרְכוֹת הָרְאִיָּה מְבָרְכִים עַל מַרְאֶה מְיֻחָד, שֶׁמְּעוֹרֵר הִתְפַּעֲלוּת אֵצֶל רֹב הָאֲנָשִׁים – אִם לֹא רָאִינוּ אוֹתוֹ שְׁלֹשִׁים יוֹם. הָרוֹאֶה בְּמִשְׁקָפַיִם – כְּרוֹאֶה רָגִיל. אֲבָל בִּתְמוּנָה, בְּסֶרֶט אוֹ בְּטֶלֶסְקוֹפּ – לֹא מְבָרְכִים.
מְבָרְכִים בִּזְמַן הָרְאִיָּה, אוֹ לְכָל הַמְאֻחָר תּוֹךְ כְּדֵי דִּבּוּר אַחֲרֶיהָ. וּכְשֶׁכַּמָּה אֲנָשִׁים רוֹאִים יַחַד – נָכוֹן שֶׁכָּל אֶחָד יְבָרֵךְ לְעַצְמוֹ, וּלְמִי שֶׁאֵינוֹ יוֹדֵעַ – אֶחָד מְבָרֵךְ בִּשְׁבִילוֹ.
💛 אַהֲבַת הַשֵּׁם: אֲנָשִׁים יוֹצְאִים לְטַיֵּל כְּדֵי לִרְאוֹת בְּעֵינֵיהֶם – כִּי רַק כָּךְ הַלֵּב מִתְרַגֵּשׁ בֶּאֱמֶת. מִתּוֹךְ הַהִתְרַגְּשׁוּת הַזּוֹ אֲנַחְנוּ פּוֹגְשִׁים אֶת הַבּוֹרֵא.
📖 פֶּרֶק ט״ו, הֲלָכָה א׳

🔀עוֹד שִׁיטוֹת – וְכֻלָּן כַּהֲלָכָה

הַקֶּשֶׁת: יֵשׁ אוֹמְרִים שֶׁהָרוֹאֶה קֶשֶׁת מְבָרֵךְ, אֲבָל לֹא מְסַפֵּר לַחֲבֵרוֹ, כִּי אֵין רָאוּי לְהִסְתַּכֵּל בָּהּ הַרְבֵּה. וְיֵשׁ אוֹמְרִים לְהֵפֶךְ – שֶׁנָּכוֹן לְסַפֵּר, כְּדֵי שֶׁגַּם הוּא יִזְכֹּר אֶת הַבְּרִית וְיִתְעוֹרֵר לִתְשׁוּבָה, וְכָךְ נוֹהֲגִים. בִּרְכַּת הָאִילָנוֹת בְּשַׁבָּת: יֵשׁ נוֹהֲגִים שֶׁלֹּא לְבָרֵךְ, וּלְדַעַת רֹב הַפּוֹסְקִים אֵין סִבָּה לְהִמָּנַע.
הַיָּם הַתִּיכוֹן: לְפִי הַשֻּׁלְחָן עָרוּךְ מְבָרְכִים עָלָיו "שֶׁעָשָׂה אֶת הַיָּם הַגָּדוֹל". לְדַעַת הַרְבֵּה פּוֹסְקִים – רַק עַל הָאוֹקְיָנוֹס, וְעַל הַיָּם הַתִּיכוֹן "עוֹשֶׂה מַעֲשֵׂה בְרֵאשִׁית", וְכָךְ נוֹהֲגִים.
💛 אַהֲבַת הַשֵּׁם: כָּל הַשִּׁיטוֹת רוֹצוֹת לְכַבֵּד אֶת ה׳ וְאֶת הַבְּרִיאָה שֶׁלּוֹ. מִי שֶׁשּׁוֹמֵר עַל מִנְהָגוֹ וּמְכַבֵּד אֶת מִנְהַג חֲבֵרוֹ – מוֹסִיף שָׁלוֹם בָּעוֹלָם.
📖 פֶּרֶק ט״ו, הֲלָכוֹת ב׳, ח׳, י׳

🏞️כַּמָּה נוֹפִים בְּיוֹם אֶחָד

רְאִיָּה שֶׁל הַר אֶחָד אֵינָהּ פּוֹטֶרֶת הַר אַחֵר. לָכֵן מִי שֶׁרוֹאֶה בְּיוֹם אֶחָד אֶת הַיָּם הַתִּיכוֹן, אֶת הַחֶרְמוֹן וְאֶת הַכִּנֶּרֶת – מְבָרֵךְ עַל כָּל אֶחָד בִּנְפָרָד.
אֲבָל הָרוֹאֶה כַּמָּה נוֹפִים יַחַד, כְּמוֹ הַכִּנֶּרֶת וְהַר אַרְבֵּל – מְבָרֵךְ עַל כֻּלָּם בְּרָכָה אַחַת. וְכָךְ גַּם הָרִים אוֹ גְּבָעוֹת דּוֹמִים בְּאֵזוֹר אֶחָד.
💛 אַהֲבַת הַשֵּׁם: כָּל נוֹף הוּא יְצִירָה חֲדָשָׁה שֶׁל ה׳. בְּכָל מָקוֹם שֶׁמַּגִּיעִים אֵלָיו – יֵשׁ סִבָּה חֲדָשָׁה לְהוֹדוֹת.
📖 פֶּרֶק ט״ו, הֲלָכָה ג׳

⛈️בְּרָקִים וּרְעָמִים

עַל בְּרָקִים וּרְעָמִים מְבָרְכִים מִיָּד, תּוֹךְ כְּדֵי דִּבּוּר. נוֹהֲגִים לְבָרֵךְ עַל הַבָּרָק ״עוֹשֶׂה מַעֲשֵׂה בְרֵאשִׁית״, וְעַל הָרַעַם, שֶׁקּוֹלוֹ מַפְחִיד יוֹתֵר – ״שֶׁכֹּחוֹ וּגְבוּרָתוֹ מָלֵא עוֹלָם״.
מְבָרְכִים פַּעַם אַחַת בַּיּוֹם עַל אוֹתָם עֲנָנִים. רַק בְּיוֹם חָדָשׁ, אוֹ כְּשֶׁהַשָּׁמַיִם הִתְבַּהֲרוּ וּבָאוּ עֲנָנִים חֲדָשִׁים – מְבָרְכִים שׁוּב.
💛 אַהֲבַת הַשֵּׁם: חֲכָמִים אָמְרוּ: ״לֹא נִבְרְאוּ רְעָמִים אֶלָּא לִפְשֹׁט עַקְמוּמִית שֶׁבַּלֵּב״. גַּם הַסְּעָרָה הַגְּדוֹלָה הִיא מַעֲשֵׂה ה׳, וְהִיא מְעוֹרֶרֶת אוֹתָנוּ לַחֲזֹר אֵלָיו.
📖 פֶּרֶק ט״ו, הֲלָכוֹת ו׳–ז׳

🌋רְעִידַת אֲדָמָה, כּוֹכָב וּסְעָרָה

עַל רְעִידַת אֲדָמָה – אֲפִלּוּ קַלָּה – עַל כּוֹכָב שָׁבִיט אוֹ כּוֹכָב נוֹפֵל, וְעַל רוּחַ סְעָרָה חֲזָקָה, מְבָרְכִים ״עוֹשֶׂה מַעֲשֵׂה בְרֵאשִׁית״ אוֹ ״שֶׁכֹּחוֹ וּגְבוּרָתוֹ מָלֵא עוֹלָם״. עַל הַמַּפְחִיד יוֹתֵר – ״שֶׁכֹּחוֹ וּגְבוּרָתוֹ״.
מְבָרְכִים מִיָּד, תּוֹךְ כְּדֵי דִּבּוּר. מִי שֶׁאִחֵר – הִפְסִיד אֶת הַבְּרָכָה, וּמְחַכֶּה אוּלַי לְבָרָק אוֹ לְרַעַם נוֹסָף.
💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁהָעוֹלָם רוֹעֵד, אָנוּ יוֹדְעִים שֶׁהַכֹּל בְּיַד ה׳. הַבְּרָכָה מַחְלִיפָה אֶת הַפַּחַד בֶּאֱמוּנָה.
📖 פֶּרֶק ט״ו, הֲלָכוֹת ו׳–ז׳

🌸בִּרְכַּת הָאִילָנוֹת

בְּחֹדֶשׁ נִיסָן, כְּשֶׁרוֹאִים עֵצֵי פְּרִי פּוֹרְחִים, מְבָרְכִים: ״שֶׁלֹּא חִסַּר בְּעוֹלָמוֹ כְּלוּם, וּבָרָא בּוֹ בְּרִיּוֹת טוֹבוֹת וְאִילָנוֹת טוֹבִים לֵהָנוֹת בָּהֶם בְּנֵי אָדָם״.
מְבָרְכִים רַק עַל עֵץ פְּרִי בִּזְמַן הַפְּרִיחָה, וְלֹא אַחֲרֵי שֶׁהַפֵּרוֹת גָּדְלוּ. לְכַתְּחִלָּה – עַל שְׁנֵי עֵצִים לְפָחוֹת. מִי שֶׁלֹּא הִסְפִּיק בְּנִיסָן – יָכוֹל לְבָרֵךְ בְּאִיָּר.
💛 אַהֲבַת הַשֵּׁם: הָעֵצִים עָמְדוּ יְבֵשִׁים כָּל הַחֹרֶף, וְה׳ הֶחֱיָה אוֹתָם שׁוּב. הוּא נָתַן לָנוּ פֵּרוֹת טְעִימִים – אֲפִלּוּ שֶׁאֶפְשָׁר לִחְיוֹת גַּם בִּלְעֲדֵיהֶם – מִתּוֹךְ אַהֲבָה.
📖 פֶּרֶק ט״ו, הֲלָכָה ח׳

🌧️תּוֹדָה עַל הַגֶּשֶׁם

הַגְּשָׁמִים הֵם שֶׁפַע שֶׁל חַיִּים מֵה׳ – לָאָדָם, לַבְּהֵמוֹת וְלַצְּמָחִים. בְּשָׁנָה שֶׁהַגְּשָׁמִים נֶעֶצְרוּ וְאַחַר כָּךְ הִתְחִילוּ לָרֶדֶת, תִּקְּנוּ חֲכָמִים לְהוֹדוֹת לַה׳ בִּבְרָכָה מְיֻחֶדֶת.
אֶת בִּרְכַּת הַהוֹדָאָה אוֹמְרִים רַק כְּשֶׁיּוֹרֵד גֶּשֶׁם חָזָק – כְּשֶׁהַטִּפּוֹת קוֹפְצוֹת מִן הַשְּׁלוּלִיּוֹת ״כְּחָתָן לִקְרַאת כַּלָּה״.
💛 אַהֲבַת הַשֵּׁם: ״וְאִלּוּ פִינוּ מָלֵא שִׁירָה כַּיָּם... אֵין אֲנַחְנוּ מַסְפִּיקִים לְהוֹדוֹת לְךָ״ – עַל כָּל טִפָּה וְטִפָּה.
📖 פֶּרֶק ט״ו, הֲלָכָה ט׳

🌈קֶשֶׁת בֶּעָנָן

הָרוֹאֶה קֶשֶׁת מְבָרֵךְ: ״זוֹכֵר הַבְּרִית וְנֶאֱמָן בִּבְרִיתוֹ וְקַיָּם בְּמַאֲמָרוֹ״ – עַל הַבְּרִית שֶׁכָּרַת ה׳ עִם נֹחַ, שֶׁלֹּא יִהְיֶה עוֹד מַבּוּל.
עַל כָּל קֶשֶׁת מְבָרְכִים מֵחָדָשׁ, גַּם בְּתוֹךְ שְׁלוֹשִׁים יוֹם – כִּי הַקֶּשֶׁת הַקּוֹדֶמֶת כְּבָר חָלְפָה. מַבִּיטִים בָּהּ רַק מְעַט, כְּדֵי לְבָרֵךְ.
💛 אַהֲבַת הַשֵּׁם: הַקֶּשֶׁת הִיא סִימָן שֶׁה׳ זוֹכֵר אוֹתָנוּ וּמְרַחֵם עַל הָעוֹלָם. גַּם כְּשֶׁיֵּשׁ עֲנָנִים, הַבְּרִית שֶׁלּוֹ אִתָּנוּ קַיֶּמֶת.
📖 פֶּרֶק ט״ו, הֲלָכָה י׳

🌈📜 לְשׁוֹן חֲכָמִים: זוֹכֵר הַבְּרִית

הָרוֹאֶה אֶת הַקֶּשֶׁת בֶּעָנָן, מְבָרֵךְ: ״בָּרוּךְ... זוֹכֵר הַבְּרִית וְנֶאֱמָן בִּבְרִיתוֹ וְקַיָּם בְּמַאֲמָרוֹ״.
📖 זוֹכֵר הַבְּרִית – ה׳ זוֹכֵר אֶת הַבְּרִית עִם נֹחַ, שֶׁלֹּא יָבוֹא עוֹד מַבּוּל עַל הָעוֹלָם.
💛 אַהֲבַת הַשֵּׁם: הַקֶּשֶׁת מַזְכִּירָה לָנוּ שֶׁה׳ שׁוֹמֵר עַל הַבְּטָחָתוֹ לָעוֹלָם, וּבִזְכוּת הַבְּרִית הוּא מְקַיֵּם אֶת הָעוֹלָם. 🧒 וְאַתָּה? כְּשֶׁתִּרְאֶה קֶשֶׁת – תַּגִּיד אֶת הַבְּרָכָה, וְלֹא תִּסְתַּכֵּל בָּהּ הַרְבֵּה.

🧠 בְּדֹק אֶת עַצְמְךָ

אֲרִיאֵל גָּר עַל חוֹף הַיָּם הַתִּיכוֹן. הַחֹדֶשׁ הָיָה עָסוּק, וְלֹא הִבִּיט בַּיָּם שְׁלוֹשִׁים יוֹם. כְּשֶׁיִּרְאֶה אוֹתוֹ – מָה הַדִּין?
💡 מִי שֶׁגָּר לְיַד הַיָּם יָכוֹל לִרְאוֹת אוֹתוֹ בְּקַלּוּת, וְאֵין לוֹ חִדּוּשׁ בָּרְאִיָּה – גַּם לֹא מֵחוֹף אַחֵר. רַק אִם עָזַב אֶת מְקוֹמוֹ לִשְׁלוֹשִׁים יוֹם וְחָזַר – יְבָרֵךְ.
עַל אֵיזֶה מָקוֹם לֹא מְבָרְכִים ״עוֹשֶׂה מַעֲשֵׂה בְרֵאשִׁית״?
💡 הַבְּרָכָה הִיא שֶׁבַח לְמַעֲשֵׂה הַבּוֹרֵא, וְלֹא לְמַעֲשֵׂה בְּנֵי אָדָם. לָכֵן עַל אֲגַם שֶׁנּוֹצַר מִסֶּכֶר לֹא מְבָרְכִים, גַּם אִם הוּא עֲנָק. עַל יָם הַמֶּלַח, הַכִּנֶּרֶת וְהַר תָּבוֹר – מְבָרְכִים.
בְּאֶמְצַע סוּפָה רָאָה עוֹבַדְיָה בָּרָק וּבֵרַךְ. אַחַר כָּךְ רָאָה עוֹד הַרְבֵּה בְּרָקִים מֵאוֹתָם עֲנָנִים. מָה יַעֲשֶׂה?
💡 כָּל הַבְּרָקִים שֶׁל אוֹתָהּ מַעֲרֶכֶת עֲנָנִים יוֹצְאִים בִּבְרָכָה אַחַת. לְמָחֳרָת, אוֹ כְּשֶׁהַשָּׁמַיִם הִתְבַּהֲרוּ וּבָאוּ עֲנָנִים חֲדָשִׁים – מְבָרְכִים שׁוּב.
📖 לִקְרִיאַת הַפֶּרֶק בַּמָּקוֹר – פְּנִינֵי הֲלָכָה

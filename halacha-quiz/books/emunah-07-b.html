הִנֵּה לֹא יָנוּם וְלֹא יִישָׁן שׁוֹמֵר יִשְׂרָאֵל

🌱 נַחְשׁוֹן בַּיָּם · רָמָה ב׳

נַחְשׁוֹן בַּיָּם – מַחֲשָׁבוֹת שֶׁל סָפֵק

רָמָה ב׳ – הַצַּעַד שֶׁל הָאֱמוּנָה

מָה חָשְׁבוּ בְּנֵי יִשְׂרָאֵל עַל שְׂפַת הַיָּם, מַדּוּעַ נַחְשׁוֹן קָפַץ, וְאֵיךְ אוֹמְרִים ׳ה׳ אִתִּי׳ בִּשְׁעַת פַּחַד

📖 אֱמוּנָה · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
🏃

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹשֶׂה צַעַד שֶׁל אֱמוּנָה, וְה׳ עוֹזֵר 💛

📖 אֱמוּנָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִדְרַשׁ חֲזַ״ל · שְׁמוֹת י״ד כ״ב

רָמָה ב׳ – הַצַּעַד שֶׁל הָאֱמוּנָה

🏃 נַחְשׁוֹן בֶּן עַמִּינָדָב

כְּשֶׁבְּנֵי יִשְׂרָאֵל פָּחֲדוּ, נַחְשׁוֹן בֶּן עַמִּינָדָב קָפַץ וְיָרַד אֶל הַיָּם רִאשׁוֹן, וְאַחֲרָיו בָּאוּ כָּל יִשְׂרָאֵל.

חֲזַ״ל לוֹמְדִים מִן הַפָּסוּק ׳בְּתוֹךְ הַיָּם בַּיַּבָּשָׁה׳ שֶׁהַיָּם לֹא נִקְרַע עַד שֶׁהֵם הִגִּיעוּ אֵלָיו עַד הַחוֹטֶם, וְרַק אָז נַעֲשָׂה יַבָּשָׁה.

ה׳ הִמְתִּין לַצַּעַד שֶׁל נַחְשׁוֹן.

הַנֵּס בָּא אַחֲרֵי הַצַּעַד שֶׁל הָאֱמוּנָה.

1
💭

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא נִבְהָל מִמַּחֲשָׁבוֹת, אֲנִי בּוֹחֵר לְהַאֲמִין 💛

📖 אֱמוּנָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – הַצַּעַד שֶׁל הָאֱמוּנָה

💭 מַחֲשָׁבוֹת שֶׁל סָפֵק

עַל שְׂפַת הַיָּם, אַחֲרֵי כָּל הַמַּכּוֹת, בְּנֵי יִשְׂרָאֵל חָשְׁשׁוּ: ׳אוּלַי ה׳ שָׁכַח אוֹתָנוּ, אוּלַי הוּא לֹא אוֹהֵב אוֹתָנוּ, אוּלַי מֹשֶׁה הִמְצִיא הַכֹּל׳.

הָרַב אוֹמֵר: כָּל מִי שֶׁיִּבְדֹּק אֶת עַצְמוֹ יִרְאֶה שֶׁהַמַּחֲשָׁבוֹת הָאֵלֶּה עוֹלוֹת גַּם אֶצְלוֹ כָּל פַּעַם שֶׁיֵּשׁ נִסָּיוֹן.

לָכֵן אֵין לְהִבָּהֵל מֵהֶן, אֶלָּא לִבְחֹר מָה לַעֲשׂוֹת אִתָּן.

מַחֲשָׁבוֹת סָפֵק בָּאוֹת לְכֻלָּם בַּנִּסָּיוֹן.

2
😌

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִזְדַּכֵּר בַּפְּסוּקִים בִּשְׁעַת פַּחַד 💛

📖 אֱמוּנָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים קכ״א ד׳ · תְּהִלִּים צ״א י״ד

רָמָה ב׳ – הַצַּעַד שֶׁל הָאֱמוּנָה

😌 קַל לְהַאֲמִין בְּלִי נִסָּיוֹן

הָרַב מְלַמֵּד: קַל מְאֹד לָאָדָם לְהַאֲמִין בְּדְבָרִים כָּל עוֹד הוּא לֹא עוֹמֵד בַּנִּסָּיוֹן שֶׁלּוֹ. אֲבָל כְּשֶׁמַּגִּיעַ נִסָּיוֹן קָשֶׁה, קָשֶׁה לוֹ מְאֹד לְהַאֲמִין שֶׁה׳ כָּאן.

הָעֵצָה: לְהִזָּכֵר בַּפְּסוּקִים, כְּמוֹ ׳הִנֵּה לֹא יָנוּם וְלֹא יִישָׁן שׁוֹמֵר יִשְׂרָאֵל׳, וּבַפָּסוּק ׳כִּי בִי חָשַׁק וַאֲפַלְּטֵהוּ אֲשַׂגְּבֵהוּ כִּי יָדַע שְׁמִי׳.

הָאֱמוּנָה נִבְחֶנֶת בְּשָׁעָה קָשָׁה.

3
⭐

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹחֵר לַחֲשֹׁב אֱמוּנָה, וְשָׂמֵחַ 💛

📖 אֱמוּנָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – הַצַּעַד שֶׁל הָאֱמוּנָה

⭐ הַמַּחֲשָׁבָה נֶחְשֶׁבֶת

הָרַב אוֹמֵר: כָּל מַחֲשָׁבָה שֶׁהָאָדָם חוֹשֵׁב בִּזְמַן שֶׁל נִסָּיוֹן תַּעֲמֹד לוֹ, וְיְקַבֵּל שָׂכָר גָּדוֹל עַל שֶׁחָשַׁב אֱמוּנָה וְלֹא סָפֵק.

וְעוֹד: מִי שֶׁחוֹשֵׁב כָּךְ מְאֻשָּׁר גַּם בָּעוֹלָם הַזֶּה.

כְּלוֹמַר, מַחֲשָׁבָה אַחַת טוֹבָה בִּשְׁעַת פַּחַד נוֹתֶנֶת גַּם שִׂמְחָה.

מַחֲשָׁבָה שֶׁל אֱמוּנָה נוֹתֶנֶת שָׂכָר וְשִׂמְחָה.

4
🌟

💛 אַהֲבַת הַשֵּׁם: בְּכָל פַּחַד אֲנִי אוֹמֵר: ה׳ אִתִּי 💛

📖 אֱמוּנָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – הַצַּעַד שֶׁל הָאֱמוּנָה

🌟 גַּם אֲנִי עַל שְׂפַת יָם

כָּל אֶחָד עוֹמֵד לִפְעָמִים עַל שְׂפַת יָם: מִבְחָן, בְּדִיקָה אֵצֶל הָרוֹפֵא, מָקוֹם חָדָשׁ, מַשִּׂימָה קָשָׁה.

מָה אוֹמְרִים לְעַצְמֵנוּ? כְּמוֹ נַחְשׁוֹן: ׳ה׳ אִתִּי׳. וְאָז עוֹשִׂים אֶת הַצַּעַד.

וְהַנֵּס יָבוֹא אַחֲרֵי הַצַּעַד.

בְּכָל פַּחַד אֶפְשָׁר לִהְיוֹת נַחְשׁוֹן.

5
💡

רָמָה ב׳ – הַצַּעַד שֶׁל הָאֱמוּנָה

מָה לוֹקְחִים מִרָמָה ב׳?

מִי שֶׁחוֹשֵׁב מַחֲשָׁבָה שֶׁל אֱמוּנָה בִּשְׁעַת קֹשִׁי, כְּמוֹ נַחְשׁוֹן, מְקָרֵב אֶת עַצְמוֹ לַה׳. ה׳ רוֹאֶה כָּל צַעַד וְכָל מַחֲשָׁבָה, וְאוֹהֵב אוֹתָם.

6
❤️
הִנֵּה לֹא יָנוּם וְלֹא יִישָׁן שׁוֹמֵר יִשְׂרָאֵל

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי הִרְגַּשְׁתִּי כְּמוֹ עַל שְׂפַת יָם, וּמָה חָשַׁבְתִּי אָז?
  • אֵיזֶה מִשְׁפָּט שֶׁל אֱמוּנָה אֶזְכֹּר בַּפַּעַם הַבָּאָה שֶׁאֶפְחַד?
  • אֶבְחַר הַשָּׁבוּעַ מַצָּב אֶחָד מַפְחִיד, אֹמַר לְעַצְמִי ׳ה׳ אִתִּי׳, אֶעֱשֶׂה צַעַד אֶחָד וְאֶרְשֹׁם מָה קָרָה.

רָמָה א׳ – קָמִים כַּאֲרִי

⏰ הַצִּלְצוּל בַּבֹּקֶר

הַשָּׁעוֹן מְצַלְצֵל. אֲבָל הַגּוּף כָּבֵד, וְהַמִּטָּה חַמָּה וְנוֹחָה.

בְּקִצּוּר שׁוּלְחָן עָרוּךְ כָּתוּב שֶׁהַיֵּצֶר הָרָע מִתְחַכֵּם וּמְשַׁכְנֵעַ: בַּחֹרֶף הוּא אוֹמֵר: ׳אֵיךְ תָּקוּם עַכְשָׁו? הַקֹּר גָּדוֹל!׳ וּבַקַּיִץ: ׳אֵיךְ תָּקוּם? עוֹד לֹא שָׂבַעְתָּ מִשֵּׁנָה!׳.

הָעֵצָה: לָקוּם מִיָּד, לִפְנֵי שֶׁהוּא מַסְפִּיק לְהַשְׁכִּיעַ אוֹתָנוּ.

קָמִים מִיָּד, לִפְנֵי שֶׁהַתֵּרוּצִים מַסְפִּיקִים לְדַבֵּר.

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שׁוּלְחָן עָרוּךְ אוֹרַח חַיִּים א׳ א׳

רָמָה א׳ – קָמִים כַּאֲרִי

🦁 יִתְגַּבֵּר כַּאֲרִי

בְּתְחִלַּת הַשֻּׁלְחָן עָרוּךְ כָּתוּב: ׳יִתְגַּבֵּר כַּאֲרִי לַעֲמֹד בַּבֹּקֶר לַעֲבוֹדַת בּוֹרְאוֹ, שֶׁיְּהֵא הוּא מְעוֹרֵר אֶת הַשַּׁחַר׳.

הַכַּוָּנָה: לֹא מְחַכִּים שֶׁהַשַּׁחַר יָעִיר אוֹתִי – אֲנִי מְעוֹרֵר אֶת הַשַּׁחַר. אֲנִי מִתְגַּבֵּר כְּמוֹ אֲרִי וְקָם בְּעַצְמִי, בִּרְצוֹנִי.

לֹא הַשַּׁחַר מְעוֹרֵר אוֹתִי – אֲנִי מְעוֹרֵר אֶת הַשַּׁחַר.

2
💰

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹחֵר שֶׁעֲבוֹדַת ה׳ תִּהְיֶה הָעִנְיָן הֲכִי חָשׁוּב לִי 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, קִצּוּר שׁוּלְחָן עָרוּךְ א׳ ג׳

רָמָה א׳ – קָמִים כַּאֲרִי

💰 אִם הָיוּ קוֹרְאִים לִי לְעֵסֶק

קִצּוּר שׁוּלְחָן עָרוּךְ נוֹתֵן מָשָׁל: אִם הָיָה אָדָם קוֹרֵא לְךָ לְעֵסֶק שֶׁתַּרְוִיחַ בּוֹ כֶּסֶף, אוֹ לִגְבּוֹת חוֹב שֶׁחַיָּבִים לְךָ, אוֹ לְהַצִּיל אֶת הָרְכוּשׁ שֶׁלְּךָ מֵהֶפְסֵד – הָיִיתָ קָם מִיָּד.

אֵין שׁוּם בְּעָיָה לָקוּם כְּשֶׁחוֹשְׁבִים שֶׁזֶּה חָשׁוּב. הַשְּׁאֵלָה הִיא רַק: מָה חָשׁוּב לִי יוֹתֵר?

כְּשֶׁחָשׁוּב לָנוּ – קָמִים מִיָּד.

3
👑

💛 אַהֲבַת הַשֵּׁם: אֲנִי קָם בַּבֹּקֶר לִפְנֵי ה׳, שֶׁאוֹהֵב אוֹתִי וְקוֹרֵא לִי 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, קִצּוּר שׁוּלְחָן עָרוּךְ א׳ ג׳

רָמָה א׳ – קָמִים כַּאֲרִי

👑 מֶלֶךְ מַלְכֵי הַמְּלָכִים

וְעוֹד: אִם הָיָה לוֹ לָלֶכֶת לַעֲבוֹדַת מֶלֶךְ בָּשָׂר וָדָם, הוּא הָיָה קָם בִּזְרִיזוּת וְלֹא מִתְרַשֵּׁל, פֶּן יַעֲלִילוּ עָלָיו אוֹ כְּדֵי לִמְצֹא חֵן בְּעֵינֵי הַמֶּלֶךְ.

עַל אַחַת כַּמָּה וְכַמָּה כְּשֶׁקּוֹרְאִים לָנוּ לַעֲבוֹדַת מֶלֶךְ מַלְכֵי הַמְּלָכִים, הַקָּדוֹשׁ בָּרוּךְ הוּא! לָכֵן צָרִיךְ לָקוּם בִּמְהִירוּת וּבִזְרִיזוּת.

קָמִים בִּשְׁבִיל מֶלֶךְ מַלְכֵי הַמְּלָכִים.

🌄

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְהַשְׁכִּים בַּבֹּקֶר כְּמוֹ אַבְרָהָם אָבִינוּ 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּרֵאשִׁית י״ט כ״ז · כ״א י״ד · כ״ב ג׳

רָמָה א׳ – קָמִים כַּאֲרִי

🌄 וַיַּשְׁכֵּם אַבְרָהָם

הַתּוֹרָה מְסַפֶּרֶת עַל אַבְרָהָם שׁוּב וָשׁוּב: ׳וַיַּשְׁכֵּם אַבְרָהָם בַּבֹּקֶר׳. הוּא הִשְׁכִּים כְּדֵי לַעֲמֹד בַּמָּקוֹם שֶׁעָמַד בּוֹ לִפְנֵי ה׳, וְכֵן לִשְׁלֹחַ אֶת יִשְׁמָעֵאל, וְכֵן לָלֶכֶת לָעֲקֵדָה.

מִכָּאן לוֹמְדִים: זְרִיזִין מַקְדִּימִין לַמִּצְוֺת. מִצְוָה שֶׁיֵּשׁ לָהּ זְמַן – עוֹשִׂים בִּתְחִלַּת הַזְּמַן.

אַבְרָהָם הִשְׁכִּים – וְאָנוּ לוֹמְדִים מִמֶּנּוּ.

💛 אַהֲבַת הַשֵּׁם: אֲנִי הוֹלֵךְ לִישֹׁן בַּזְּמַן כְּדֵי לְהַשְׁכִּים לַעֲבוֹדַת ה׳ 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשְׁלֵי ו׳ ט׳ (הַגְּרָ״א)

רָמָה א׳ – קָמִים כַּאֲרִי

🌙 מַתְחִילִים מִן הָעֶרֶב

הַגְּרָ״א מְבָאֵר אֶת הַפָּסוּק: ׳עַד מָתַי עָצֵל תִּשְׁכָּב, מָתַי תָּקוּם מִשְּׁנָתֶךָ׳. הֶעָצֵל מִתְעַצֵּל לָלֶכֶת לִישֹׁן בַּזְּמַן, וְלָכֵן קָשֶׁה לוֹ לָקוּם בַּזְּמַן.

הַזָּרִיז הוֹלֵךְ לִישֹׁן מֻקְדָּם, כְּדֵי שֶׁיּוּכַל לְהַשְׁכִּים לָקוּם. הַקִּימָה בַּבֹּקֶר מַתְחִילָה מֵהָעֶרֶב.

מִי שֶׁהוֹלֵךְ לִישֹׁן בַּזְּמַן – קָם בְּקַלּוּת.

6
🕰️

💛 אַהֲבַת הַשֵּׁם: אֲנִי קָם בַּזְּמַן כְּדֵי לְהַגִּיעַ לִתְפִלָּה עִם הַצִּבּוּר 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשְׁנָה בְּרוּרָה א׳ ט׳

רָמָה א׳ – קָמִים כַּאֲרִי

🕰️ מָתַי קָמִים?

הַמִּשְׁנָה בְּרוּרָה אוֹמֶרֶת שֶׁעַל כָּל פָּנִים לֹא מְאַחֲרִים אֶת זְמַן הַתְּפִלָּה שֶׁהַצִּבּוּר מִתְפַּלֵּל בּוֹ.

לָכֵן קָמִים מֻקְדָּם מַסְפִּיק כְּדֵי לְהָכִין אֶת עַצְמֵנוּ, לְנַקּוֹת אֶת הַגּוּף, וְלָלֶכֶת לְבֵית הַכְּנֶסֶת לְהִתְפַּלֵּל עִם הַצִּבּוּר. כָּךְ לֹא מְאַחֲרִים לַתְּפִלָּה.

קָמִים בַּזְּמַן כְּדֵי לְהַסְפִּיק לְהִתְכּוֹנֵן וּלְהִתְפַּלֵּל עִם הַצִּבּוּר.

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, קִצּוּר שׁוּלְחָן עָרוּךְ א׳ · אַרְחוֹת חַיִּים

רָמָה א׳ – קָמִים כַּאֲרִי

😌 גַּם שֵׁנָה חֲשׁוּבָה

זְרִיזוּת אֵינָהּ אוֹמֶרֶת לְוַתֵּר עַל שֵׁנָה שֶׁהַגּוּף צָרִיךְ. אִם אָדָם מַאֲרִיךְ לִשְׁכַּב מִשּׁוּם בְּרִיאוּת – וְלֹא כְּדֵי לֵהָנוֹת וּלְהִתְפַּנֵּק – מֻתָּר.

וְאִם בִּגְלַל שֵׁנָה קְצָרָה מִדַּי הוּא מִתְנַמְנֵם בַּתְּפִלָּה, אָז יִישַׁן כָּל צָרְכּוֹ.

הַמַּטָּרָה הִיא לַעֲבֹד אֶת ה׳ בְּעֵרָנוּת, לֹא לְהִתְעַיֵּף.

זְרִיזוּת הִיא לֹא לְוַתֵּר עַל שֵׁנָה, אֶלָּא לְהַשְׁכִּים בַּזְּמַן.

💡

רָמָה א׳ – קָמִים כַּאֲרִי

מָה לוֹקְחִים מִרָמָה א׳?

כְּשֶׁאֲנִי קָם בִּזְרִיזוּת בַּבֹּקֶר, אֲנִי אוֹמֵר לַה׳: אַתָּה חָשׁוּב לִי יוֹתֵר מִן הַמִּטָּה. וְה׳ שֶׁאוֹהֵב אוֹתִי קוֹרֵא לִי, וַאֲנִי עוֹנֶה לוֹ בְּשִׂמְחָה.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָה הַתֵּרוּץ שֶׁאֲנִי אוֹמֵר לְעַצְמִי הֲכִי הַרְבֵּה כְּשֶׁצָּרִיךְ לָקוּם בַּבֹּקֶר?
  • אִם הָיוּ קוֹרְאִים לִי לְדָבָר שֶׁאֲנִי מְאֹד רוֹצֶה – הָיִיתִי קָם מִיָּד. מָה עוֹזֵר לִי לָשִׂים לֵב שֶׁעֲבוֹדַת ה׳ חֲשׁוּבָה גַּם כֵּן?
  • הַשָּׁבוּעַ אֲנִי עוֹשֶׂה ׳שָׁבוּעַ שֶׁל קִימָה מְהִירָה׳: קָם מִיָּד, אוֹמֵר ׳מוֹדֶה אֲנִי׳, וּמְסַמֵּן וִי בִּטַבְלָה.

וְאָהַבְתָּ – לֶאֱהֹב אֶת ה׳ גַּם כְּשֶׁקָּשֶׁה

לָמָּה לֹא מַסְפִּיק לְהַאֲמִין, אֵיךְ מַגִּיעִים לְאַהֲבָה, וְאֵיךְ שְׂמֵחִים גַּם בְּיוֹם קָשֶׁה

📖 פְּסוּקִים ד׳–ה׳ · קֻנְטְרֵס ׳וֶאֱמוּנָתְךָ בַּלֵּילוֹת׳

🎯 לַחִידוֹן

רָמָה ב׳ – אַהֲבָה גַּם בַּקֹּשִׁי

📜 הַגְּמָרָא: לְקַבֵּל בְּשִׂמְחָה

הַגְּמָרָא שׁוֹאֶלֶת עַל הַמִּשְׁנָה: עַל בְּשׂוֹרָה טוֹבָה מְבָרְכִים ׳הַטּוֹב וְהַמֵּטִיב׳, וְעַל בְּשׂוֹרָה רָעָה מְבָרְכִים ׳בָּרוּךְ דַּיַּן הָאֱמֶת׳. אִם כֵּן, מָה פֵּרוּשׁ ׳כְּשֵׁם שֶׁמְּבָרֵךְ עַל הַטּוֹבָה׳?

עוֹנֶה רָבָא: ׳לְקַבּוּלִינְהוּ בְּשִׂמְחָה׳ – הַבְּרָכָה שׁוֹנָה, אֲבָל אֶת הָרָעָה מְקַבְּלִים בְּשִׂמְחָה, כְּמוֹ אֶת הַטּוֹבָה.

וְהַשֻּׁלְחָן עָרוּךְ פּוֹסֵק: מְבָרְכִים עַל הָרָעָה ׳בְּדַעַת שְׁלֵמָה וּבְנֶפֶשׁ חֲפֵצָה׳.

הַבְּרָכָה שׁוֹנָה – אֲבָל הַשִּׂמְחָה אוֹתָהּ שִׂמְחָה.

1

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁקָּשֶׁה, אֲנִי יָכוֹל לָשִׁיר לַה׳ 💛

רָמָה ב׳ – אַהֲבָה גַּם בַּקֹּשִׁי

🎵 פְּסוּקִים שֶׁל שִׂמְחָה – גַּם בַּקֹּשִׁי

הַגְּמָרָא מְבִיאָה פְּסוּקִים שֶׁמְּלַמְּדִים לְהוֹדוֹת עַל הַכֹּל:

׳חֶסֶד וּמִשְׁפָּט אָשִׁירָה׳ – גַּם עַל הַחֶסֶד וְגַם עַל הַדִּין אֲנִי שָׁר.

׳כּוֹס יְשׁוּעוֹת אֶשָּׂא... צָרָה וְיָגוֹן אֶמְצָא – וּבְשֵׁם ה׳ אֶקְרָא׳.

׳ה׳ נָתַן וַה׳ לָקָח, יְהִי שֵׁם ה׳ מְבֹרָךְ׳ – כָּךְ אָמַר אִיּוֹב.

וְרַבִּי עֲקִיבָא לִמֵּד לוֹמַר תָּמִיד: ׳כָּל דְּעָבִיד רַחֲמָנָא לְטַב עָבִיד׳ – כָּל מָה שֶׁה׳ עוֹשֶׂה, לְטוֹבָה הוּא עוֹשֶׂה.

׳כָּל מָה שֶׁה׳ עוֹשֶׂה – לְטוֹבָה הוּא עוֹשֶׂה׳: מִשְׁפָּט שֶׁאוֹמְרִים תָּמִיד.

2

💛 אַהֲבַת הַשֵּׁם: כָּל קֹשִׁי הוּא הִזְדַּמְּנוּת שֶׁה׳ נוֹתֵן לִי לְהַרְאוֹת לוֹ שֶׁאֲנִי מַאֲמִין 💛

📖 פָּרָשַׁת שְׁמַע – דְּבָרִים ו׳, פֶּרֶק ו׳, פָּסוּק ה׳ (הַטּוּר וְהַשֻּׁלְחָן עָרוּךְ)

רָמָה ב׳ – אַהֲבָה גַּם בַּקֹּשִׁי

🥇 שְׂמֵחִים בַּמִּצְווֹת, לֹא בַּצַּעַר

אֵיךְ אֶפְשָׁר לִשְׂמֹחַ בְּדָבָר קָשֶׁה? הַטּוּר, וְאַחֲרָיו הַשֻּׁלְחָן עָרוּךְ (סִימָן רכ״ב), מְחַדְּשִׁים: הַשִּׂמְחָה הִיא לֹא בַּצַּעַר עַצְמוֹ, אֶלָּא בְּכָךְ שֶׁבַּקֹּשִׁי זוֹכִים לְקַיֵּם מִצְווֹת גְּדוֹלוֹת: ׳אָנֹכִי׳, ׳לֹא יִהְיֶה לְךָ׳, ׳שְׁמַע יִשְׂרָאֵל׳ וְ׳וְאָהַבְתָּ׳.

וְהַנָּבִיא חֲבַקּוּק הֶעֱמִיד אֶת כָּל הַמִּצְווֹת עַל אַחַת: ׳וְצַדִּיק בֶּאֱמוּנָתוֹ יִחְיֶה׳ – הַשָּׂכָר עַל מִצְוַת הָאֱמוּנָה גָּדוֹל מִכֻּלָּן.

בְּיוֹם קָשֶׁה זוֹכִים לְמִצְווֹת הָאֱמוּנָה וְהָאַהֲבָה – וְעַל זֶה שְׂמֵחִים.

3

💛 אַהֲבַת הַשֵּׁם: בְּיוֹם קָשֶׁה אֲנִי כְּמוֹ חַיָּל גִּבּוֹר שֶׁמַּרְאֶה לַמֶּלֶךְ כַּמָּה הוּא אוֹהֵב אוֹתוֹ 💛

📖 פָּרָשַׁת שְׁמַע – דְּבָרִים ו׳, פֶּרֶק ו׳, פָּסוּק ה׳ (מְסִלַּת יְשָׁרִים)

רָמָה ב׳ – אַהֲבָה גַּם בַּקֹּשִׁי

🔄 ׳וְנַהֲפוֹךְ הוּא׳ – כָּל הַשָּׁנָה

הַמִּצְוָה הַגְּדוֹלָה בְּיוֹתֵר הִיא לַחְשֹׁב: אֲנִי אוֹהֵב אֶת ה׳, וַה׳ אוֹהֵב אוֹתִי. כְּשֶׁטּוֹב, זֶה קַל. כְּשֶׁקָּשֶׁה, זֶה קָשֶׁה הַרְבֵּה יוֹתֵר – וְלָכֵן דַּוְקָא אָז הַמִּצְוָה גְּדוֹלָה יוֹתֵר.

כָּךְ יֵשׁ ׳וְנַהֲפוֹךְ הוּא׳ כָּל הַשָּׁנָה, לֹא רַק בְּפוּרִים: כְּשֶׁרַע לְךָ – זֶה הַזְּמַן הַטּוֹב בְּיוֹתֵר לַעֲבֹד אֶת ה׳.

מְסִלַּת יְשָׁרִים: כְּמוֹ שַׂר צָבָא גִּבּוֹר שֶׁבּוֹחֵר דַּוְקָא בַּמִּלְחָמָה הַחֲזָקָה, כְּדֵי לְהַרְאוֹת אֶת כֹּחוֹ. וְכָל אוֹהֵב שָׂמֵחַ כְּשֶׁיֵּשׁ לוֹ הִזְדַּמְּנוּת לְהַרְאוֹת כַּמָּה הוּא אוֹהֵב.

כְּשֶׁקָּשֶׁה לַחְשֹׁב שֶׁה׳ אוֹהֵב אוֹתִי – דַּוְקָא אָז הַמִּצְוָה הֲכִי גְּדוֹלָה.

4

💛 אַהֲבַת הַשֵּׁם: כָּל מִצְוָה שֶׁאֲנִי עוֹשֶׂה הִיא עָנָף, וְהָאַהֲבָה הִיא הַשֹּׁרֶשׁ 💛

📖 פָּרָשַׁת שְׁמַע – דְּבָרִים ו׳, פֶּרֶק ו׳, פָּסוּק ה׳ (אִבְּן עֶזְרָא)

רָמָה ב׳ – אַהֲבָה גַּם בַּקֹּשִׁי

🌳 הַשֹּׁרֶשׁ שֶׁל כָּל הַמִּצְווֹת

לֹא רַק חוֹבוֹת הַלְּבָבוֹת. גַּם אִבְּן עֶזְרָא כּוֹתֵב שֶׁשֹּׁרֶשׁ כָּל הַמִּצְווֹת הוּא שֶׁהָאָדָם יֹאהַב אֶת ה׳ בְּכָל נַפְשׁוֹ וְיִדְבַּק בּוֹ.

וְסֵפֶר הָעִקָּרִים מוֹסִיף: אַהֲבַת ה׳ הִיא תַּכְלִית כָּל הַתּוֹרָה, כִּי הִיא מוֹשֶׁכֶת אֶת אַהֲבַת ה׳ אֶל הָאָדָם.

וְזֶה פֵּרוּשׁ ׳לְדָבְקָה בוֹ׳: לֵב שָׁלֵם וְאַהֲבָה נֶאֱמָנָה.

כְּשֶׁאֲנִי אוֹהֵב אֶת ה׳ – גַּם ה׳ מַרְעִיף עָלַי אֶת אַהֲבָתוֹ.

5

💛 אַהֲבַת הַשֵּׁם: לִפְנֵי ׳שְׁמַע׳ אֲנִי נִזְכָּר בְּכָל הַטּוֹב שֶׁה׳ עָשָׂה לִי הַיּוֹם 💛

📖 פָּרָשַׁת שְׁמַע – דְּבָרִים ו׳, פֶּרֶק ו׳, פָּסוּק ה׳ (הֶחָפֵץ חַיִּים)

רָמָה ב׳ – אַהֲבָה גַּם בַּקֹּשִׁי

⏰ עַכְשָׁו – וְלֹא אַחַר כָּךְ

הֶחָפֵץ חַיִּים מְדַקְדֵּק: מִי שֶׁקּוֹרֵא שְׁמַע בְּלִי תְּפִלִּין וְחוֹשֵׁב לְהָנִיחַ אוֹתָן אַחַר כָּךְ – לֹא מוֹעִיל, כִּי צָרִיךְ אוֹתָן עַכְשָׁו, כְּשֶׁאוֹמֵר ׳וּקְשַׁרְתָּם׳. כָּךְ גַּם ׳וְאָהַבְתָּ׳: הָאַהֲבָה צְרִיכָה לִהְיוֹת בַּלֵּב בְּאוֹתוֹ רֶגַע.

עֵצָה: לִפְנֵי שְׁמַע מִתְבּוֹנְנִים רֶגַע בַּחֲסָדִים שֶׁל ה׳ וּבְכָךְ שֶׁהוּא מַאֲרִיךְ אַפּוֹ עָלֵינוּ – וְהָאַהֲבָה נִשְׁאֶרֶת גַּם כְּשֶׁמַּגִּיעִים לְ׳וְאָהַבְתָּ׳.

בְּדֶרֶךְ כְּלָל אֶפְשָׁר לְקַיֵּם אֶת הַמִּצְוָה בְּכָל זְמַן; אֲבָל כְּשֶׁקּוֹרֶה דָּבָר קָשֶׁה – חוֹבָה לְעוֹרֵר אֶת הָאַהֲבָה.

אוֹמְרִים ׳וְאָהַבְתָּ׳ – אוֹהֲבִים עַכְשָׁו. וּבְשָׁעָה קָשָׁה – חוֹבָה.

רָמָה ב׳ – אַהֲבָה גַּם בַּקֹּשִׁי

מָה לוֹקְחִים מִרָמָה ב׳?

ה׳ לֹא נוֹתֵן לָנוּ קֹשִׁי כְּדֵי לְצַעֵר אוֹתָנוּ, אֶלָּא כְּמוֹ אַבָּא אוֹהֵב – וּבְכָל קֹשִׁי הוּא נוֹתֵן לָנוּ הִזְדַּמְּנוּת לְקַיֵּם אֶת הַמִּצְווֹת הַגְּדוֹלוֹת בְּיוֹתֵר: אֱמוּנָה וְאַהֲבָה. מִי שֶׁאוֹמֵר בְּיוֹם קָשֶׁה ׳כָּל מָה שֶׁה׳ עוֹשֶׂה – לְטוֹבָה׳, עוֹשֶׂה נַחַת רוּחַ גְּדוֹלָה לְאָבִיו שֶׁבַּשָּׁמַיִם, וּמְקַיֵּם בֶּאֱמֶת ׳וְאָהַבְתָּ אֵת ה׳ אֱלֹהֶיךָ׳.

וְאָהַבְתָּ אֵת ה׳ אֱלֹהֶיךָ בְּכָל לְבָבְךָ וּבְכָל נַפְשְׁךָ וּבְכָל מְאֹדֶךָ

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָה ה׳ נָתַן לִי הַיּוֹם – מֵהָעֵינַיִם שֶׁרוֹאוֹת וְעַד הָאֹכֶל שֶׁעַל הַשֻּׁלְחָן – שֶׁבִּגְלָלוֹ אֲנִי יָכוֹל לֶאֱהֹב אוֹתוֹ?
  • כְּשֶׁמַּשֶּׁהוּ לֹא הוֹלֵךְ לִי, אֵילוּ שְׁתֵּי מַחֲשָׁבוֹת אֲנִי יָכוֹל לַחְשֹׁב, כְּדֵי לְקַיֵּם ׳שְׁמַע׳ וְ׳וְאָהַבְתָּ׳?
  • מָה הָיָה קָשֶׁה לִי הַשָּׁבוּעַ – וְאֵיךְ אֲקַבֵּל דָּבָר כָּזֶה בַּפַּעַם הַבָּאָה, כְּמוֹ שַׂר צָבָא שֶׁשָּׂמֵחַ לְהַרְאוֹת אֶת אַהֲבָתוֹ?

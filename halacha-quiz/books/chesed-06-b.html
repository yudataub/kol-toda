רָמָה ב׳ – חֶסֶד שֶׁל חָסִיד

📋 רַחֲמִים אוֹ חֶסֶד?

מִתּוֹךְ ׳נְתִיבוֹת הַחַיִּים׳: רַחֲמִים הֵם הֲטָבָה הֶכְרֵחִית שֶׁבָּאָה מֵעַצְמָהּ, בְּלִי מַחֲשָׁבָה, וְרַק כְּשֶׁרוֹאִים אֶת הַמִּקְרֶה. חֶסֶד הוּא הֲטָבָה רְצוֹנִית שֶׁבָּאָה מִמַּחֲשָׁבָה, וְנוֹתֵן גַּם כְּשֶׁלֹּא רוֹאִים.

בְּחֶסֶד הַנּוֹתֵן מְנַסֶּה לָרֶדֶת לְסוֹף דַּעְתּוֹ שֶׁל הַמְקַבֵּל: מָה מַפְרִיעַ לוֹ? לָכֵן יָבוֹא לְבַקֵּר חוֹלֶה בַּשָּׁעָה שֶׁנּוֹחַ לוֹ.

וְעוֹד: רַחֲמִים לִפְעָמִים נוֹתְנִים כְּדֵי לְהָקֵל עַל הַלֵּב שֶׁלִּי, וְחֶסֶד רַק בִּשְׁבִיל שֶׁיִּהְיֶה טוֹב לַשֵּׁנִי.

חֶסֶד בָּא מִמַּחֲשָׁבָה עַל טוֹבַת הַמְקַבֵּל.

רָמָה ב׳ – חֶסֶד שֶׁל חָסִיד

🚪 אִיּוֹב וְאַבְרָהָם

אָבוֹת דְּרַבִּי נָתָן: אִיּוֹב עָשָׂה אַרְבָּעָה פְּתָחִים לְבֵיתוֹ, כְּדֵי שֶׁהָעֲנִיִּים לֹא יִצְטַעֲרוּ לְהַקִּיף אֶת הַבַּיִת. אֲפִילוּ כָּךְ, ה׳ אָמַר לוֹ: עֲדַיִן לֹא הִגַּעְתָּ לַחֲצִי שִׁעוּר שֶׁל אַבְרָהָם.

אִיּוֹב יָשַׁב בְּבֵיתוֹ וְנָתַן לְמִי שֶׁבָּא כְּפִי הֶרְגֵּלוֹ. אַבְרָהָם יָצָא לַדְּרָכִים, וְנָתַן לְכָל אֶחָד גַּם מַה שֶּׁאֵינוֹ רָגִיל. הוּא אַף בָּנָה בָּתִּים גְּדוֹלִים עַל הַדְּרָכִים וְהִנִּיחַ בָּהֶם אֹכֶל וּמַשְׁקֶה, וְכָל הַבָּא אָכַל, שָׁתָה וּבֵרַךְ.

לָמָּה זֶה קָשֶׁה לִשְׁאֹל מֵעַצְמִי? כִּי אֶצְלִי זֶה חֶסֶד: לֹא רַק לַעֲנוֹת, אֶלָּא לָצֵאת.

הַחֶסֶד שֶׁל אַבְרָהָם יוֹצֵא לְחַפֵּשׂ, וְנוֹתֵן גַּם מַה שֶּׁלֹּא רָגִיל.

רָמָה ב׳ – חֶסֶד שֶׁל חָסִיד

💞 נָתוֹן תִּתֵּן לוֹ

׳נָתוֹן תִּתֵּן לוֹ וְלֹא יֵרַע לְבָבְךָ בְּתִתְּךָ לוֹ׳. הַשִּׁעוּר לוֹמֵד: תֵּן פַּעֲמַיִם – אַחַת בִּשְׁבִיל מִדַּת הָרַחֲמִים, וְאַחַת בִּשְׁבִיל מִדַּת הַחֶסֶד.

הַנְּתִינָה הַשְּׁנִיָּה צְרִיכָה לִהְיוֹת אַחֲרֵי שֶׁהַלֵּב כְּבָר רָגוּעַ וְאֵין חֶסְרוֹן שֶׁמַּפְרִיעַ לוֹ.

וְהַפָּסוּק מוֹסִיף: ׳כִּי בִּגְלַל הַדָּבָר הַזֶּה יְבָרֶכְךָ ה׳׳ – מִי שֶׁנּוֹתֵן בְּחֶסֶד, זוֹכֶה גַּם הוּא לְחֶסֶד.

נוֹתְנִים גַּם מֵרַחֲמִים וְגַם מֵחֶסֶד, בְּלֵב רָגוּעַ.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לַעֲשׂוֹת נַחַת רוּחַ לַחָבֵר וְלַה׳ 💛

📖 חֶסֶד – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְסִלַּת יְשָׁרִים, הַקְדָּמָה

רָמָה ב׳ – חֶסֶד שֶׁל חָסִיד

🏅 צַדִּיק וְחָסִיד

הַמְּסִלַּת יְשָׁרִים מְסַבֵּר: צַדִּיק עוֹשֶׂה מַה שֶּׁצָּרִיךְ, וְחָסִיד חוֹשֵׁב מַה יִּהְיֶה נַחַת רוּחַ לַה׳.

לַעֲשׂוֹת מַה שֶּׁצָּרִיךְ – זֶה כְּמוֹ רַחֲמִים. לַעֲשׂוֹת נַחַת רוּחַ – זֶה חֶסֶד.

כָּל עוֹד אֲנִי נוֹתֵן מִתּוֹךְ רַחֲמִים, אֵין לִי סִבָּה לָתֵת לַה׳, כִּי עַל ה׳ אֲנִי לֹא מְרַחֵם. אֲבָל מִי שֶׁמִּתְרַגֵּל לַעֲשׂוֹת נַחַת רוּחַ לַחָבֵר, יִרְצֶה לַעֲשׂוֹת נַחַת רוּחַ גַּם לַה׳, וְזוֹ אַהֲבַת ה׳.

חֶסֶד מְלַמֵּד לַעֲשׂוֹת נַחַת רוּחַ, וְכָךְ בָּאִים לְאַהֲבַת ה׳.

רָמָה ב׳ – חֶסֶד שֶׁל חָסִיד

🍞 חֵן, חֶסֶד וְרַחֲמִים

בְּבִרְכַּת הַמָּזוֹן אוֹמְרִים: ׳הוּא הַזָּן אֶת הָעוֹלָם… בְּחֵן בְּחֶסֶד וּבְרַחֲמִים׳. בְּחֵן – לְצַדִּיק גָּדוֹל, בְּשִׂמְחָה עֲצוּמָה. בְּחֶסֶד – לְמִי שֶׁהִתְנַהֲגוּתוֹ רְאוּיָה. בְּרַחֲמִים – גַּם לְמִי שֶׁלֹּא רָאוּי לְפִי מַעֲשָׂיו, ה׳ זָן אוֹתוֹ.

וּבְ׳בּוֹרֵא נְפָשׁוֹת׳: לֶחֶם וּמַיִם שָׁוִים לְכֻלָּם, וְהַמַּמְתַּקִּים שׁוֹנִים לְכָל אֶחָד לְפִי מָה שֶׁהוּא אוֹהֵב.

ה׳ זָן אֶת כֻּלָּם, וְגַם נֶהֱנֶה לְהַפְתִּיעַ אוֹתָנוּ בְּמַמְתַּקִּים.

מָה לוֹקְחִים מִרָמָה ב׳?

מִי שֶׁמִּתְרַגֵּל לָתֵת מִתּוֹךְ חֶסֶד, לוֹמֵד לַעֲשׂוֹת נַחַת רוּחַ. וּמִי שֶׁרָגִיל לַעֲשׂוֹת נַחַת רוּחַ לַחָבֵר, יִרְצֶה גַּם לְשַׂמֵּחַ אֶת ה׳ וְיֶאֱהַב אוֹתוֹ.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה עָשִׂיתִי הַשָּׁבוּעַ מִתּוֹךְ רַחֲמִים, וּמָה עָשִׂיתִי מִתּוֹךְ חֶסֶד?
  • מִי חָבֵר שֶׁאֵינִי רוֹאֶה, וְאֶפְשָׁר לְהִתְקַשֵּׁר אֵלָיו וְלִשְׁאֹל מַה שְּׁלוֹמוֹ?

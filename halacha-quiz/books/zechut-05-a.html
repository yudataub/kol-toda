🪞 כְּמוֹ שֶׁאֲנִי דָּן – כָּךְ דָּנִים אוֹתִי

דָּנִי הִפִּיל סֵפֶר שֶׁל חָבֵר, וְכֻלָּם הִסְתַּכְּלוּ עָלָיו. דָּנִי מְקַוֶּה שֶׁיַּחְשְׁבוּ: ׳זֶה קָרָה בְּטָעוּת׳.

הַבַּעַל שֵׁם טוֹב לִמֵּד: מִשָּׁמַיִם דָּנִים אֶת הָאָדָם כְּמוֹ שֶׁהוּא דָּן אֶת הַבְּרִיּוֹת.

מִי שֶׁדָּן אֶת חֲבֵרוֹ לְטוֹבָה, דָּנִים גַּם אוֹתוֹ לְטוֹבָה. וְזֶה הַשִּׁעוּר שֶׁלָּנוּ הַיּוֹם.

כְּמוֹ שֶׁאֲנִי דָּן אֶת הֶחָבֵר – כָּךְ דָּנִים אוֹתִי.

🐑

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִשְׁתַּדֵּל לֹא לִשְׁפֹּט מַהֵר, כִּי גַּם אֲנִי יָכוֹל לִטְעוֹת 💛

📖 דָּן לְכַף זְכוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁמוּאֵל ב׳ י״ב א׳–ד׳

רָמָה א׳ – כְּמוֹ שֶׁאֲנִי דָּן

🐑 הַמָּשָׁל שֶׁל נָתָן הַנָּבִיא

נָתָן הַנָּבִיא בָּא אֶל דָּוִד הַמֶּלֶךְ וְסִפֵּר לוֹ מָשָׁל: ׳שְׁנֵי אֲנָשִׁים הָיוּ בְּעִיר אַחַת, אֶחָד עָשִׁיר וְאֶחָד רָאשׁ׳.

לֶעָשִׁיר הָיוּ צֹאן וּבָקָר הַרְבֵּה מְאֹד. וְלָרָשׁ הָיְתָה כִּבְשָׂה אַחַת קְטַנָּה, שֶׁגָּדְלָה עִמּוֹ וְעִם בָּנָיו.

בָּא אוֹרֵחַ אֶל הֶעָשִׁיר, וְהוּא לָקַח אֶת כִּבְשַׂת הָרָשׁ.

בַּמָּשָׁל: לֶעָשִׁיר הָיוּ הַרְבֵּה צֹאן, וְלָרָשׁ רַק כִּבְשָׂה אַחַת.

2
☝️

💛 אַהֲבַת הַשֵּׁם: לִפְנֵי שֶׁאֲנִי אוֹמֵר ׳הוּא אָשֵׁם׳, אֲנִי שׁוֹאֵל: אֵיךְ הָיִיתִי רוֹצֶה שֶׁיִּדּוֹנוּ אוֹתִי? 💛

📖 דָּן לְכַף זְכוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁמוּאֵל ב׳ י״ב ה׳–ז׳

רָמָה א׳ – כְּמוֹ שֶׁאֲנִי דָּן

☝️ ׳אַתָּה הָאִישׁ׳

דָּוִד שָׁמַע אֶת הַמָּשָׁל וְכָעַס מְאֹד עַל הֶעָשִׁיר: הוּא יְשַׁלֵּם אֶת הַכִּבְשָׂה אַרְבַּעְתַּיִם!

אָז אָמַר לוֹ נָתָן שְׁתֵּי מִלִּים: ׳אַתָּה הָאִישׁ׳. הַמָּשָׁל הָיָה עַל דָּוִד עַצְמוֹ.

דָּוִד דָּן בְּחֻמְרָה אֶת הָאִישׁ שֶׁבַּמָּשָׁל, וּבְאוֹתָהּ מִדָּה נִדּוֹן הוּא עַצְמוֹ. כָּךְ לָמְדוּ שֶׁמִּי שֶׁדָּן אַחֵר, דָּנִים אוֹתוֹ בְּאוֹתָהּ מִדָּה.

מִי שֶׁשּׁוֹפֵט בְּחֻמְרָה, נִשְׁפָּט בְּחֻמְרָה.

3
🏛️

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת בְּהֵיכַל הַזְּכוּת, קָרוֹב לַה׳ 💛

📖 דָּן לְכַף זְכוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הַזֹּהַר · בַּעַל הַתּוֹלְדוֹת

רָמָה א׳ – כְּמוֹ שֶׁאֲנִי דָּן

🏛️ שְׁנֵי הֵיכָלוֹת

בַּזֹּהַר כָּתוּב שֶׁיֵּשׁ שְׁנֵי הֵיכָלוֹת. בְּהֵיכַל הַזְּכוּת דָּנִים אֶת הָאָדָם לִזְכוּת, אֲפִילוּ אִם הוּא רָשָׁע. בְּהֵיכַל הַחוֹבָה דָּנִים לְחוֹבָה.

וְאֵיזֶה הֵיכָל נִבְחָר בִּשְׁבִילִי? זֶה תָּלוּי בְּאֵיךְ אֲנִי מִתְנַהֵג.

מִי שֶׁדָּן אֲחֵרִים לִזְכוּת, נִדּוֹן בְּהֵיכַל הַזְּכוּת.

אֲנִי בּוֹחֵר בְּהֵיכַל הַזְּכוּת כְּשֶׁאֲנִי דָּן אֲחֵרִים לִזְכוּת.

4
🧭

💛 אַהֲבַת הַשֵּׁם: בָּעֶרֶב אֲנִי סוֹפֵר כַּמָּה פְּעָמִים דַּנְתִּי חָבֵר לְטוֹבָה, וּמוֹדֶה לַה׳ 💛

📖 דָּן לְכַף זְכוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בַּעַל הַתּוֹלְדוֹת

רָמָה א׳ – כְּמוֹ שֶׁאֲנִי דָּן

🧭 הַסִּימָן שֶׁל הַלֵּב

בַּעַל הַתּוֹלְדוֹת כָּתַב: אָדָם יָכוֹל לָדַעַת בְּאֵיזוֹ מַדְרֵגָה הוּא נִמְצָא.

אִם הוּא דָּן אֶת חֲבֵרוֹ לִזְכוּת – זֶה סִימָן שֶׁהוּא עַכְשָׁו בְּהֵיכַל הַקְּדֻשָּׁה.

לָכֵן בָּעֶרֶב אֶפְשָׁר לִשְׁאֹל אֶת עַצְמִי: ׳הַיּוֹם, כַּמָּה פְּעָמִים דַּנְתִּי חָבֵר לִזְכוּת?׳

דָּנְתִּי חָבֵר לִזְכוּת – סִימָן טוֹב עַל הַלֵּב שֶׁלִּי.

🌤️

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹמֵר עָלַי כְּמוֹ צֵל טוֹב, וַאֲנִי רוֹצֶה לִהְיוֹת צֵל טוֹב לַחֲבֵרִים 💛

📖 דָּן לְכַף זְכוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים קכ״א ה׳

רָמָה א׳ – כְּמוֹ שֶׁאֲנִי דָּן

🌤️ ׳ה׳ צִלְּךָ׳

׳ה׳ שֹׁמְרֶךָ, ה׳ צִלְּךָ׳ – כְּמוֹ שֶׁהַצֵּל זָז בְּדִיּוּק כְּמוֹ שֶׁאֲנִי זָז, כָּךְ ה׳ מִתְנַהֵג אִתִּי כְּמוֹ שֶׁאֲנִי מִתְנַהֵג.

כְּמוֹ הַצֵּל – כָּךְ ה׳ נוֹהֵג עִם הָאָדָם כְּמוֹ שֶׁהוּא נוֹהֵג.

6
🕊️

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁחָבֵר פּוֹגֵעַ בִּי בְּלִי כַּוָּנָה, אֲנִי מוֹחֵל לוֹ בְּלֵב שָׁלֵם 💛

📖 דָּן לְכַף זְכוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, חֲזַ״ל

רָמָה א׳ – כְּמוֹ שֶׁאֲנִי דָּן

🕊️ מִי שֶׁלֹּא מַקְפִּיד

כֻּלָּנוּ רוֹצִים שֶׁיִּמְחֲלוּ לָנוּ עַל הַטָּעֻיּוֹת שֶׁלָּנוּ. חֲזַ״ל גִּלּוּ דֶּרֶךְ: הַמַּעֲבִיר עַל מִדּוֹתָיו – מַעֲבִירִים לוֹ.

זֶה מִי שֶׁלֹּא מַקְפִּיד עַל חָבֵר שֶׁצִּעֵר אוֹתוֹ. הוּא לֹא נוֹשֵׂא טִינָה, וְחוֹשֵׁב: ׳אוּלַי הוּא לֹא הִתְכַּוֵּן׳.

וְכָךְ גַּם ה׳ לֹא מַקְפִּיד עָלָיו.

אֲנִי לֹא מַקְפִּיד עַל חָבֵר, וְכָךְ מַעֲבִירִים גַּם לִי.

🎫

💛 אַהֲבַת הַשֵּׁם: אֲנִי נוֹתֵן לַחֲבֵרִים אֶת אוֹתוֹ הֶסְבֵּר טוֹב שֶׁאֲנִי רוֹצֶה לְעַצְמִי 💛

📖 דָּן לְכַף זְכוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎫 הַכַּרְטִיס הַנָּחְמָד

הַיּוֹם נַעֲשֶׂה תַּרְגִּיל: כָּל אֶחָד זוֹכֵר דָּבָר מֵבִישׁ שֶׁעָשָׂה, כְּמוֹ שֶׁהִפִּיל מַשֶּׁהוּ אוֹ שָׁכַח לְהָבִיא.

אַחַר כָּךְ כּוֹתְבִים עַל כַּרְטִיס אֶת הַהֶסְבֵּר הַטּוֹב שֶׁהָיִינוּ רוֹצִים שֶׁיַּחְשְׁבוּ.

כָּךְ אֲנִי לוֹמֵד: מָה שֶׁאֲנִי רוֹצֶה שֶׁיַּחְשְׁבוּ עָלַי – אֲנִי חוֹשֵׁב עַל אֲחֵרִים.

אֲנִי כּוֹתֵב הֶסְבֵּר טוֹב לְמַעֲשֶׂה מֵבִישׁ, וְחוֹשֵׁב כָּךְ גַּם עַל חָבֵר.

מָה לוֹקְחִים מִרָמָה א׳?

כְּשֶׁאֲנִי דָּן אֶת חֲבֵרִי לִזְכוּת, אֲנִי מְלַמֵּד אֶת עַצְמִי אֵיךְ ה׳ יִדּוֹן אוֹתִי. הוּא אוֹהֵב אוֹתִי כְּמוֹ אַבָּא, וְשָׂמֵחַ כְּשֶׁאֲנִי מוֹצֵא טוֹב בְּבָנָיו.

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי דָּנוּ אוֹתִי לִזְכוּת, וּמָה הִרְגַּשְׁתִּי?
  • מִי הַחָבֵר שֶׁאֲנִי קָשֶׁה כְּלַפָּיו בַּלֵּב, וּמָה הֶסְבֵּר טוֹב אֶפְשָׁר לִמְצֹא לוֹ?
  • אֶכְתֹּב כַּרְטִיס נָחְמָד: הֶסְבֵּר טוֹב לְמַעֲשֶׂה מֵבִישׁ שֶׁעָשִׂיתִי, וְאֶתֵּן לְחָבֵר הֶסְבֵּר כָּזֶה.

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים קי״ח כ״ה · שִׁעוּר ׳לְמִי אֲנַחְנוּ רוֹקְדִים׳

רָמָה ב׳ – שִׂמְחָה שֶׁזּוֹכֶרֶת

🙏 בַּקָּשָׁה בְּתוֹךְ הַשִּׂמְחָה

לִפְנֵי הַהַקָּפוֹת אוֹמְרִים: ׳אָנָּא ה׳ הוֹשִׁיעָה נָּא, אָנָּא ה׳ הַצְלִיחָה נָּא׳ (תְּהִלִּים קי״ח), ׳עוֹזֵר דַּלִּים׳, ׳עֲנֵנוּ בְיוֹם קָרְאֵנוּ׳. מַדּוּעַ הַשָּׁעָה הַשְּׂמֵחָה נִפְתַּחַת בִּלְשׁוֹן שֶׁל צוּקָה וְעֶזְרָה?

הַשִּׁעוּר מַקְשִׁיב דַּוְקָא לַצֵּרוּף שֶׁל שִׂמְחָה וּתְחִנָּה. בְּיוֹם שֶׁבּוֹ אָדָם מַרְגִּישׁ קָרוֹב אֶל ה׳, הוּא לֹא צָרִיךְ לְהַסְתִּיר אֶת מַה שֶּׁחָסֵר לוֹ.

הַמַּעְגָּל אֵינוֹ מַכְרִיחַ אוֹתָנוּ לִשְׁכֹּחַ אֶת הַדַּלִּים, וּבְוַדַּאי אֵינוֹ מְבַטֵּל אֶת צְרָכֵיהֶם. שִׂמְחָה אֲמִתִּית מַרְחִיבָה אֶת הַלֵּב עַד שֶׁהוּא זוֹכֵר גַּם אוֹתָם.

הוֹשִׁיעָה נָּא אֵינוֹ מַפְרִיעַ לַהַקָּפָה, הוּא מְגַלֶּה אֶל מִי פּוֹנִים.

🕺

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ בַּקֶּשֶׁר עִם ה׳, לֹא רַק בְּסִיּוּם 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׁעוּר ׳לְמִי אֲנַחְנוּ רוֹקְדִים׳

רָמָה ב׳ – שִׂמְחָה שֶׁזּוֹכֶרֶת

🕺 הָרִקּוּד לִפְנֵי הַסִּיּוּם

בְּסִיּוּם מַסֶּכֶת חוֹגְגִים אַחֲרֵי הַלִּמּוּד. בְּשִׂמְחַת תּוֹרָה הַתְּמוּנָה אַחֶרֶת: רוֹקְדִים בַּלַּיְלָה וְשׁוּב בַּבֹּקֶר לִפְנֵי שֶׁקּוֹרְאִים אֶת הַפְּסוּקִים הָאַחֲרוֹנִים. רַק אַחַר כָּךְ מְסַיְּמִים אֶת הַתּוֹרָה וּמַתְחִילִים אוֹתָהּ מֵחָדָשׁ.

אִם הַשִּׂמְחָה הָיְתָה רַק עַל עֶצֶם הַסִּיּוּם, לָמָּה הִיא מַתְחִילָה לִפְנָיו?

הַשִּׁעוּר מַצִּיעַ: סִיּוּם הַתּוֹרָה הוּא בִּטּוּי חָשׁוּב לַשִּׂמְחָה, אֲבָל לֹא כָּל הַיְּסוֹד שֶׁלָּהּ.

הַשִּׂמְחָה רְחָבָה מִשִּׂמְחָה עַל סִיּוּם הַקְּרִיאָה בִּלְבָד.

2
📜

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי לוֹמֵד תּוֹרָה, אֲנִי קָרוֹב אֶל ה׳ עַצְמוֹ 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִדְרָשׁ (וַיִּקְרָא רַבָּה) · זֹהַר ג׳ ע״ג א׳ · שִׂמְחַת תּוֹרָה (הָרַב אֱלִיָּהוּ דִּיסְקִין)

רָמָה ב׳ – שִׂמְחָה שֶׁזּוֹכֶרֶת

📜 בְּךָ – בְּתוֹרָתְךָ

בַּמִּדְרָשׁ שָׁאַל רַבִּי אָבִין: ׳אֵין אָנוּ יוֹדְעִים בַּמֶּה לִשְׂמֹחַ – בַּיּוֹם אִם בְּהַקָּדוֹשׁ בָּרוּךְ הוּא?׳ בָּא שְׁלֹמֹה וּפֵרֵשׁ: ׳נָגִילָה וְנִשְׂמְחָה בְּךָ – בְּתוֹרָתְךָ, בִּישׁוּעָתְךָ׳.

רַבִּי יִצְחָק הוֹסִיף: ׳בְּךָ – בְּעֶשְׂרִים וּשְׁתַּיִם אוֹתִיּוֹת שֶׁכָּתַבְתָּ לָנוּ בְּתוֹרָתְךָ׳.

הַזֹּהַר אוֹמֵר: ׳אוֹרַיְיתָא וְקוּדְשָׁא בְּרִיךְ הוּא חַד הוּא׳. כְּשֶׁאָדָם עוֹסֵק בַּתּוֹרָה כְּדֵי לָדַעַת מָה דְּבַר ה׳ – הוּא דָּבֵק בְּה׳ עַצְמוֹ.

לִמּוּד תּוֹרָה הוּא דְּבֵקוּת בְּה׳ עַצְמוֹ.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִשְׂמֹחַ בְּה׳ עַצְמוֹ 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דְּבָרִים ט״ז ט״ו · סֵפֶר הַחִנּוּךְ תקכ״ד · שִׂמְחַת תּוֹרָה (הָרַב אֱלִיָּהוּ דִּיסְקִין)

רָמָה ב׳ – שִׂמְחָה שֶׁזּוֹכֶרֶת

🏆 שִׂמְחָה בְּלִי אֶמְצָעִים

בְּסֻכּוֹת יֵשׁ אֶמְצָעִים לַשִּׂמְחָה: סֻכָּה, לוּלָב וְאֶתְרוֹג. בִּשְׁמִינִי עֲצֶרֶת אֵין אֶת כָּל אֵלֶּה, וְהַתּוֹרָה אוֹמֶרֶת: ׳וְהָיִיתָ אַךְ שָׂמֵחַ׳.

לְפִי הַגְּאוֹן מִוִּילְנָא, הָ׳אַךְ׳ מְלַמֵּד שֶׁבַּיּוֹם הַזֶּה נִשְׁאֶרֶת רַק מִצְוַת שִׂמְחָה.

זֶה לֹא חֶסְרוֹן, אֶלָּא מַעֲלָה: הִגַּעְנוּ לַפִּסְגָּה, וּכְבָר לֹא צְרִיכִים אֶמְצָעִים כְּדֵי לִשְׂמֹחַ בְּה׳ – שׂוֹמְחִים בְּה׳ עַצְמוֹ. כְּפִי שֶׁכָּתַב הַחִנּוּךְ: ׳יוֹם שְׁמִינִי עֲצֶרֶת כֻּלּוֹ לְשֵׁם ה׳׳.

בִּשְׁמִינִי עֲצֶרֶת שׂוֹמְחִים בְּה׳ עַצְמוֹ, בְּלִי אֶמְצָעִים.

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹקֵחַ אִתִּי לַחֹרֶף אֶת ה׳ וְאֶת הַתּוֹרָה 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַשִׁ״י, וַיִּקְרָא כ״ג ל״ו · שִׂמְחַת תּוֹרָה (הָרַב אֱלִיָּהוּ דִּיסְקִין)

רָמָה ב׳ – שִׂמְחָה שֶׁזּוֹכֶרֶת

🍽️ קָשֶׁה עָלַי פְּרֵדַתְכֶם

רַשִׁ״י מְבָאֵר ׳עֲצֶרֶת הִיא׳: ׳עָצַרְתִּי אֶתְכֶם אֶצְלִי, כְּמֶלֶךְ שֶׁזִּמֵּן אֶת בָּנָיו לִסְעוּדָה כָּךְ וְכָךְ יָמִים. כֵּיוָן שֶׁהִגִּיעַ זְמַנָּן לְהִפָּטֵר אָמַר: בָּנַי, בְּבַקָּשָׁה מִכֶּם, עַכְּבוּ עִמִּי עוֹד יוֹם אֶחָד, קָשֶׁה עָלַי פְּרֵדַתְכֶם׳.

מִי שֶׁשָּׁאַל: מָה הִרְוַחְנוּ בְּיוֹם נוֹסָף? הֲרֵי אַחַר כָּךְ קָשֶׁה יוֹתֵר לְהִפָּרֵד.

הַתְּשׁוּבָה: אֶת הַסֻּכָּה אֲנַחְנוּ מַכְנִיסִים לַמַּחְסָן, אֲבָל ה׳ וְתוֹרָתוֹ לֹא הוֹלְכִים לַמַּחְסָן. הֵם נִשְׁאָרִים אִתָּנוּ כָּל לֵילוֹת הַחֹרֶף.

מַה שֶּׁנִּשְׁאָר לַחֹרֶף: ה׳ וְהַתּוֹרָה.

💛 אַהֲבַת הַשֵּׁם: גַּם לִי קָשֶׁה לְהִפָּרֵד מֵה׳ וּמִשִּׂמְחַת הֶחָג 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׂמְחַת תּוֹרָה (הָרַב אֱלִיָּהוּ דִּיסְקִין)

רָמָה ב׳ – שִׂמְחָה שֶׁזּוֹכֶרֶת

🌙 הַקָּפוֹת שְׁנִיּוֹת

בְּשִׂמְחַת תּוֹרָה אָמַר ה׳ ׳קָשֶׁה עָלַי פְּרֵדַתְכֶם׳. בְּמוֹצָאֵי הֶחָג אוֹמֵר עַם יִשְׂרָאֵל לַקָּדוֹשׁ בָּרוּךְ הוּא: ׳רִבּוֹנוֹ שֶׁל עוֹלָם, קָשֶׁה עָלֵינוּ לְהִפָּרֵד מִמְּךָ – עוֹד רִקּוּד אֶחָד!׳

זֶה הַטַּעַם לַמִּנְהָג שֶׁל הַקָּפוֹת שְׁנִיּוֹת בְּמוֹצָאֵי שִׂמְחַת תּוֹרָה.

הָאַהֲבָה הִיא מִשְּׁנֵי הַצְּדָדִים: ה׳ אוֹהֵב אוֹתָנוּ וְרוֹצֶה שֶׁנִּשָּׁאֵר, וַאֲנַחְנוּ אוֹהֲבִים אוֹתוֹ וְלֹא רוֹצִים לָלֶכֶת.

אַהֲבָה מִשְּׁנֵי הַצְּדָדִים: ׳קָשֶׁה עָלֵינוּ לְהִפָּרֵד׳.

רָמָה ב׳ – שִׂמְחָה שֶׁזּוֹכֶרֶת

מָה לוֹקְחִים מִרָמָה ב׳?

הַשִּׂמְחָה הַגְּדוֹלָה בְּיוֹתֵר הִיא שִׂמְחָה בְּה׳ עַצְמוֹ. ה׳ מַזְמִין אוֹתָנוּ ׳עַכְּבוּ עִמִּי׳, וַאֲנַחְנוּ עוֹנִים: קָשֶׁה לָנוּ לְהִפָּרֵד. זוֹ אַהֲבָה הַדָּדִית שֶׁנִּשְׁאֶרֶת כָּל הַשָּׁנָה.

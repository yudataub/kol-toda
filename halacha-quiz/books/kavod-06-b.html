רָמָה ב׳ – שִׁבְעִים וּשְׁנַיִם פְּסוּקִים

📖 קָרְבָּן אֶחָד, שְׁתֵּים עֶשְׂרֵה פָּרָשִׁיּוֹת

בַּיּוֹם הָרִאשׁוֹן: ׳וַיְהִי הַמַּקְרִיב בַּיּוֹם הָרִאשׁוֹן אֶת קָרְבָּנוֹ נַחְשׁוֹן בֶּן עַמִּינָדָב לְמַטֵּה יְהוּדָה׳. בַּיּוֹם הַשֵּׁנִי: ׳הִקְרִיב נְתַנְאֵל בֶּן צוּעָר נְשִׂיא יִשָּׂשכָר׳. וְכָךְ הַלְאָה, עַד הַנָּשִׂיא הַשְּׁנֵים עָשָׂר.

הַתּוֹרָה כּוֹתֶבֶת שִׁבְעִים וּשְׁנַיִם פְּסוּקִים (בַּמִּדְבָּר ז׳ י״ב–פ״ג) עַל קָרְבָּן שָׁוֶה.

רַבֵּנוּ בַּחְיֵי שׁוֹאֵל: הֲרֵי ׳כַּמָּה גּוּפֵי תּוֹרָה תְּלוּיִין בְּתֵבָה אַחַת אוֹ בְּאוֹת אַחַת׳, וְכָאן שְׁתֵּים עֶשְׂרֵה פָּרָשִׁיּוֹת שֶׁל עִנְיָן אֶחָד.

שִׁבְעִים וּשְׁנַיִם פְּסוּקִים עַל קָרְבָּן שֶׁחוֹזֵר שְׁנֵים עָשָׂר פְּעָמִים.

רָמָה ב׳ – שִׁבְעִים וּשְׁנַיִם פְּסוּקִים

👑 כִּי מְכַבְּדַי אֲכַבֵּד

תְּשׁוּבָתוֹ שֶׁל רַבֵּנוּ בַּחְיֵי: ׳הַקָּדוֹשׁ בָּרוּךְ הוּא חוֹלֵק כָּבוֹד לִירֵאָיו וּמְכַבֵּד מְכַבְּדָיו, כָּאָמוּר כִּי מְכַבְּדַי אֲכַבֵּד׳.

אִלּוּ הִזְכִּירָה הַתּוֹרָה רַק אֶת נַחְשׁוֹן וְכָתְבָה ׳וְכֵן הִקְרִיבוּ כָּל הַנְּשִׂיאִים׳, הָיָה בָּזֶה קִצּוּר בִּכְבוֹד הָאֲחֵרִים.

וְהַנְּשִׂיאִים הִקְרִיבוּ לִכְבוֹד ה׳, וְלָכֵן ה׳ כִּבֵּד כָּל אֶחָד מֵהֶם.

ה׳ מְכַבֵּד מְכַבְּדָיו, וְלָכֵן כָּל נָשִׂיא קִבֵּל פָּרָשָׁה.

💛 אַהֲבַת הַשֵּׁם: ה׳ מַכִּיר אֶת הַמַּחֲשָׁבָה הָאִישִׁית שֶׁלִּי, וְגַם אֶת הַמַּעֲשֶׂה 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַבֵּנוּ בַּחְיֵי, בַּמִּדְבָּר ז׳ פ״ד

רָמָה ב׳ – שִׁבְעִים וּשְׁנַיִם פְּסוּקִים

🎯 לְכָל אֶחָד טַעַם מִשֶּׁלּוֹ

עוֹד תְּשׁוּבָה: כָּל נָשִׂיא חָשַׁב בְּקָרְבָּנוֹ טַעַם בִּפְנֵי עַצְמוֹ, שֶׁלֹּא כְּטַעַם חֲבֵרוֹ. הַטְּעָמִים הָיוּ חֲלוּקִים, כִּי כָּל אֶחָד מֵהֶם רָמַז לַשֵּׁבֶט שֶׁלּוֹ.

לָכֵן בָּאוּ הַפָּרָשִׁיּוֹת חֲלוּקוֹת, פָּרָשָׁה בִּפְנֵי עַצְמָהּ לְכָל אֶחָד.

גַּם כְּשֶׁהַמַּעֲשֶׂה נִרְאֶה אוֹתוֹ דָּבָר, מֵאֲחוֹרָיו עוֹמֶדֶת מַחֲשָׁבָה אִישִׁית.

מַעֲשֶׂה שָׁוֶה, אֲבָל מַחֲשָׁבָה אִישִׁית לְכָל נָשִׂיא.

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַבֵּנוּ בַּחְיֵי, בַּמִּדְבָּר ז׳ פ״ד

רָמָה ב׳ – שִׁבְעִים וּשְׁנַיִם פְּסוּקִים

⚖️ שָׁוִים בַּזְּכוּת

רַבֵּנוּ בַּחְיֵי מוֹסִיף: ׳כְּשֵׁם שֶׁהָיוּ שָׁוִים בְּקָרְבָּנָם, בָּעֵצָה וּבְשִׁעוּרֵי הַמִּשְׁקָל, כָּךְ הָיוּ שָׁוִין בַּזְּכוּת. וּלְפִי שֶׁכְּבוֹד כָּל אֶחָד שָׁוֶה בְּעֵינֵי הַמָּקוֹם, תַּאֲרִיךְ הַתּוֹרָה בְּסִפּוּר כָּל אֶחָד מֵהֶן בְּשָׁוֶה׳.

הַתּוֹרָה אֵינָהּ מְקַצֶּרֶת בַּנָּשִׂיא הָאַחֲרוֹן וְאֵינָהּ מַאֲרִיכָה רַק בָּרִאשׁוֹן.

כָּבוֹד כָּל אֶחָד שָׁוֶה בְּעֵינֵי הַמָּקוֹם, וְלָכֵן סִפּוּר שָׁוֶה לְכָל אֶחָד.

💛 אַהֲבַת הַשֵּׁם: ה׳ שָׂמֵחַ בְּכָל מַעֲשֶׂה טוֹב שֶׁלִּי, גַּם אִם טָעִיתִי בַּעָבָר 💛

רָמָה ב׳ – שִׁבְעִים וּשְׁנַיִם פְּסוּקִים

⏳ לְפִי הַמַּעֲשֶׂה שֶׁבְּאוֹתָהּ שָׁעָה

אַף עַל פִּי שֶׁגָּלוּי לִפְנֵי ה׳ שֶׁהַנְּשִׂיאִים עֲתִידִים לִטְעוֹת אַחַר כָּךְ בְּעִנְיָן אַחֵר, ׳אֵין הַקָּדוֹשׁ בָּרוּךְ הוּא דָּן אֶת הָאָדָם אֶלָּא לְפִי מַעֲשָׂיו שֶׁבְּאוֹתָהּ שָׁעָה׳.

הוּא חָלַק לָהֶם כָּבוֹד, כִּי הִקְרִיבוּ לִכְבוֹדוֹ.

לָמַדְנוּ: מְכַבְּדִים אָדָם עַל מַה שֶּׁהוּא עוֹשֶׂה עַכְשָׁו, וְלֹא מַזְכִּירִים לוֹ אֶת הָעָבָר.

מְכַבְּדִים אָדָם עַל הַמַּעֲשֶׂה הַטּוֹב שֶׁלּוֹ עַכְשָׁו.

🌿 כָּבוֹד לַבְּרִיּוֹת וְכָבוֹד לַבּוֹרֵא

מִכָּאן לוֹמְדִים: נִלְמַד מִכָּאן כַּמָּה חָשׁוּב לְבוֹרֵא עוֹלָם כְּבוֹד הָאָדָם, וּמִמֵּילָא כַּמָּה חָשׁוּב שֶׁנְּכַבֵּד אָנוּ אֶת בָּנָיו הַיְּקָרִים.

בַּמִּשְׁנָה בְּאָבוֹת: ׳אֵיזֶהוּ מְכֻבָּד – הַמְכַבֵּד אֶת הַבְּרִיּוֹת׳, וְרוֹאִים שֶׁלְכַבֵּד אֶת הַבְּרִיּוֹת נֶחְשָׁב לְכַבֵּד אֶת ה׳.

מְכַבְּדִים אֶת הַבְּרִיּוֹת – מְכַבְּדִים אֶת ה׳.

מָה לוֹקְחִים מִרָמָה ב׳?

הַתּוֹרָה לֹא קִצְּרָה בִּכְבוֹד אַף נָשִׂיא, כִּי כְּבוֹד כָּל אֶחָד שָׁוֶה בְּעֵינֵי ה׳. מִי שֶׁמְּכַבֵּד אֶת הַבְּרִיּוֹת מְכַבֵּד אֶת ה׳, וה׳ אוֹהֵב וּמְכַבֵּד אוֹתוֹ בַּחֲזָרָה.

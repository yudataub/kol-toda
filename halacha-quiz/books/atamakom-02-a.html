🏫 מִכִּתָּה בְּבֵית סֵפֶר

בִּתְחִלַּת שְׁנַת תש״ו (1946) הוּקַם בְּבֵית הַסֵּפֶר ׳בְּנֵה׳, בְּבֵית צִינוֹבֶר בִּרְחוֹב יְהוּדָה הַנָּשִׂיא, בֵּית כְּנֶסֶת שֶׁל ׳בְּנֵי עֲקִיבָא׳ – רַק לְשַׁבָּתוֹת. אֶת הַמִּנְיָן הֵקִים הַמַּדְרִיךְ אַבְרָהָם שִׁילֶר.

אַחֲרֵי שְׁלוֹשָׁה חֳדָשִׁים הִצְטָרְפוּ אֵלָיו חַבְרֵי ׳הַפּוֹעֵל הַמִּזְרָחִי׳. הָיָה לָהֶם סֵפֶר תּוֹרָה אֶחָד וְכַמָּה סִדּוּרִים.

בֵּית כְּנֶסֶת נוֹלַד בְּכִתָּה אַחַת בְּבֵית הַסֵּפֶר.

כַּחֲמִשִּׁים אִישׁ הִתְפַּלְּלוּ שָׁם, רֹבָּם פּוֹעֲלֵי ׳אָתָא׳, הַצֶּבַע וּמִפְעָלֵי הַמִּפְרָץ. הַגַּבַּאי ר׳ מֹשֶׁה שִׁילֶר סִדֵּר כָּל שִׁשִּׁי אֶת הַחֶדֶר וְאֶת הַכִּסְאוֹת, וּבְמוֹצָאֵי שַׁבָּת הֶחְזִיר אוֹתוֹ לְכִתָּה.

פּוֹעֲלִים הָפְכוּ כִּתָּה לְבֵית כְּנֶסֶת כָּל שַׁבָּת.

אֶבֶן הַפִּנָּה הוּנְחָה בְּג׳ בְּכִסְלֵו תשי״א (12.11.1950), בְּשָׁעָה חָמֵשׁ אַחַר הַצָּהֳרַיִם, בְּהִשְׁתַּתְּפוּת הָרַבָּנִים הָרָאשִׁיִּים, שָׂרִים וְרָאשֵׁי הַתְּנוּעָה.

תָּכְנִית גְּדוֹלָה לְבֵית כְּנֶסֶת בְּשֵׁשׁ מֵאוֹת מְקוֹמוֹת.

בִּשְׁנַת תשט״ז הוֹחְלְטָה בְּנִיַּת הַקּוֹמָה הַשְּׁנִיָּה, בְּכֶסֶף מֵעִזְבּוֹנָם שֶׁל פִּישֶׁר וְצִינוֹבֶר, שֶׁהִשִּׂיג ר׳ בּוֹיִם. הַתִּיכוֹן הַדָּתִי עָבַר אֵלֶיהָ זְמַנִּית.

מִשְּׁנַת תשכ״ג פּוֹעֵל סְנִיף ׳בְּנֵי עֲקִיבָא׳ בַּקּוֹמָה הַשְּׁנִיָּה עַד הַיּוֹם. עַל הַגַּג עוֹמֵדת חֲנֻכִּיָּה שֶׁתָּרַם ר׳ אַרְבֶּסְפֶלְד.

הַקּוֹמָה הַשְּׁנִיָּה הִגִּיעָה אַחֲרֵי שָׁנִים שֶׁל חִסָּכוֹן.

רָמָה א׳ – מָה רוֹאִים בְּבֵית הַפּוֹעֵל הַמִּזְרָחִי

מָה לוֹקְחִים מִרָמָה א׳?

מִנְיָן בְּכִתָּה הָפַךְ לְבֵית כְּנֶסֶת עַל שֵׁם מְיַסֵּד הָעִיר. כָּךְ ה׳ מְצַמִּיחַ בַּיִת מִהַתְחָלָה קְטַנָּה – וְזֶה סִימָן לְאַהֲבָתוֹ אֶת עַמּוֹ.

❤️
זֶה־הַשַּׁעַר לַה׳ צַדִּיקִים יָבֹאוּ בוֹ

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מִי עָזַר לְבֵית הַכְּנֶסֶת שֶׁלִּי בְּלִי שֶׁאַף אֶחָד יֵדַע?
  • מָה אֲנִי יָכוֹל לַעֲשׂוֹת כְּדֵי שֶׁהַמִּנְיָן בְּבֵית הַכְּנֶסֶת שֶׁלָּנוּ יַמְשִׁיךְ לִהְיוֹת?
  • הַשָּׁבוּעַ אֶעֱמֹד מוּל שֶׁלֶט שֶׁל בֵּית כְּנֶסֶת, אֶקְרָא עַל מִי הוּא נִקְרָא, וְאֶשְׁאַל מִי הָיָה.

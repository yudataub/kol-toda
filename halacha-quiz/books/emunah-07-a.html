🏊 עַל קְצֵה הַמַּצְלֵף

אֲבִיעָד עָמַד עַל קְצֵה הַמַּצְלֵף הַגָּבוֹהַּ בַּבְּרֵכָה. הַלֵּב דָּפַק חָזָק, וְהַמַּחֲשָׁבוֹת אָמְרוּ: ׳אוּלַי אֶפֹּל? אוּלַי כְּדַאי לַחֲזֹר?׳

גַּם בִּשְׁעַת פַּחַד, הַמַּחֲשָׁבוֹת שֶׁאֲנִי חוֹשֵׁב חֲשׁוּבוֹת מְאֹד. הַשִּׁעוּר עַל אֱמוּנָה מְלַמֵּד אוֹתָנוּ אֵיךְ.

וּבִשְׁבִיל זֶה נִפְגַּשׁ עִם נַחְשׁוֹן בֶּן עַמִּינָדָב.

בִּזְמַן פַּחַד, הַמַּחֲשָׁבָה שֶׁלִּי חֲשׁוּבָה מְאֹד.

📖 אֱמוּנָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁמוֹת י״ד י׳, י״ג

רָמָה א׳ – ה׳ אִתִּי

🌊 עַל שְׂפַת הַיָּם

בְּנֵי יִשְׂרָאֵל יָצְאוּ מִמִּצְרַיִם וְהִגִּיעוּ לְיַם סוּף. פִּתְאוֹם רָאוּ שֶׁהַמִּצְרִים רוֹדְפִים אַחֲרֵיהֶם. הַכָּתוּב אוֹמֵר: ׳וַיִּירְאוּ מְאֹד וַיִּצְעֲקוּ בְנֵי יִשְׂרָאֵל אֶל ה׳׳.

מֹשֶׁה אָמַר לָעָם: ׳אַל תִּירָאוּ, הִתְיַצְּבוּ וּרְאוּ אֶת יְשׁוּעַת ה׳׳.

בְּאוֹתוֹ רֶגַע הָיָה צָרִיךְ לְהַאֲמִין שֶׁה׳ כָּאן.

בְּשָׁעָה קָשָׁה אֶפְשָׁר לִצְעֹק אֶל ה׳ וּלְהַאֲמִין.

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁיֵּשׁ לִי סָפֵק, אֲנִי בּוֹחֵר לְהַאֲמִין בַּה׳ 💛

📖 אֱמוּנָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – ה׳ אִתִּי

💭 מַחֲשָׁבוֹת שֶׁל סָפֵק

הָרַב מְלַמֵּד: עַל שְׂפַת הַיָּם הִתְחִילוּ לָבוֹא גַּם מַחֲשָׁבוֹת כָּאֵלֶּה: ׳אוּלַי ה׳ שָׁכַח אוֹתָנוּ? אוּלַי הוּא לֹא אוֹהֵב אוֹתָנוּ? אוּלַי מֹשֶׁה הִמְצִיא הַכֹּל?׳

וְזֶה אַחֲרֵי שֶׁרָאוּ אֶת כָּל הַמַּכּוֹת וְהַנִּסִּים!

כָּל אֶחָד מַכִּיר מַחֲשָׁבוֹת כָּאֵלֶּה כְּשֶׁיֵּשׁ נִסָּיוֹן, וְזֶה נוֹרְמָלִי. הַשְּׁאֵלָה: מָה עוֹשִׂים אִתָּן?

מַחֲשָׁבוֹת סָפֵק בָּאוֹת לְכֻלָּם, וְחָשׁוּב מָה עוֹשִׂים אִתָּן.

🏃

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְהַאֲמִין כְּמוֹ נַחְשׁוֹן 💛

📖 אֱמוּנָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִדְרַשׁ חֲזַ״ל · שְׁמוֹת י״ד כ״ב

רָמָה א׳ – ה׳ אִתִּי

🏃 נַחְשׁוֹן קוֹפֵץ

בְּנֵי יִשְׂרָאֵל פָּחֲדוּ לָבוֹא אֶל הַיָּם, עַד שֶׁנַחְשׁוֹן בֶּן עַמִּינָדָב קָפַץ וְיָרַד רִאשׁוֹן אֶל הַיָּם. אַחֲרָיו נִכְנְסוּ כָּל יִשְׂרָאֵל.

נַחְשׁוֹן לֹא רָאָה עֲדַיִן אֶת הַיַּבָּשָׁה. הוּא יָרַד אֶל הַמַּיִם מִתּוֹךְ אֱמוּנָה.

הַמַּיִם עָלוּ עָלָיו, עַד הַחוֹטֶם.

נַחְשׁוֹן הֶאֱמִין וְקָפַץ, עוֹד לִפְנֵי שֶׁרָאָה אֶת הַנֵּס.

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹשֶׂה צַעַד שֶׁל אֱמוּנָה, וְה׳ עוֹזֵר לִי 💛

📖 אֱמוּנָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִדְרַשׁ חֲזַ״ל · שְׁמוֹת י״ד כ״ב

רָמָה א׳ – ה׳ אִתִּי

🏝️ בְּתוֹךְ הַיָּם בַּיַּבָּשָׁה

הַפָּסוּק אוֹמֵר: ׳וַיָּבֹאוּ בְנֵי יִשְׂרָאֵל בְּתוֹךְ הַיָּם בַּיַּבָּשָׁה׳. אֲבָל אִם הָיָה יַם – לָמָּה יַבָּשָׁה? וְאִם הָיְתָה יַבָּשָׁה – לָמָּה בְּתוֹךְ הַיָּם?

חֲזַ״ל מְלַמְּדִים: הַיָּם לֹא נִקְרַע עַד שֶׁהֵם נִכְנְסוּ אֵלָיו, עַד הַחוֹטֶם, וְרַק אַחַר כָּךְ נַעֲשָׂה יַבָּשָׁה.

ה׳ עוֹזֵר אַחֲרֵי שֶׁעוֹשִׂים צַעַד שֶׁל אֱמוּנָה.

הַנֵּס בָּא אַחֲרֵי הַצַּעַד שֶׁל נַחְשׁוֹן.

📖 אֱמוּנָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – ה׳ אִתִּי

😌 קַל לְהַאֲמִין בְּלִי נִסָּיוֹן

הָרַב אוֹמֵר: קַל מְאֹד לְהַאֲמִין כָּל עוֹד לֹא עוֹמְדִים בַּנִּסָּיוֹן. אֲבָל כְּשֶׁמַּגִּיעַ נִסָּיוֹן קָשֶׁה, פִּתְאוֹם קָשֶׁה לְהַאֲמִין שֶׁה׳ כָּאן.

הַיָּם הָיָה נִסָּיוֹן כָּזֶה. מִי שֶׁמַּאֲמִין בַּנִּסָּיוֹן – הוּא כְּמוֹ נַחְשׁוֹן.

הָאֱמוּנָה שֶׁלָּנוּ נִבְחֶנֶת דַּוְקָא בִּשְׁעַת הַנִּסָּיוֹן.

⭐ כָּל מַחֲשָׁבָה נֶחְשֶׁבֶת

הָרַב אוֹמֵר דָּבָר מְיֻחָד: כָּל מַחֲשָׁבָה שֶׁאֲנִי חוֹשֵׁב בִּזְמַן נִסָּיוֹן – ה׳ רוֹאֶה אוֹתָהּ, וְהִיא נֶחְשֶׁבֶת לִי.

אִם חָשַׁבְתִּי ׳ה׳ אִתִּי׳ בִּשְׁעַת הַפַּחַד, זֶה שָׁוֶה הַרְבֵּה. וּמִי שֶׁחוֹשֵׁב כָּךְ, מַרְגִּישׁ גַּם שִׂמְחָה.

מַחֲשָׁבָה שֶׁל אֱמוּנָה בִּשְׁעַת קֹשִׁי שָׁוָה הַרְבֵּה.

יֵשׁ מִשְׁפָּט שֶׁאֶפְשָׁר לוֹמַר כְּשֶׁמְּפַחֲדִים: ׳הִנֵּה לֹא יָנוּם וְלֹא יִישָׁן שׁוֹמֵר יִשְׂרָאֵל׳. ה׳ לֹא נִרְדָּם, וְהוּא שׁוֹמֵר עָלֵינוּ כָּל הַזְּמַן.

אֶפְשָׁר לְהוֹסִיף: ׳ה׳ שֹׁמְרֶךָ, ה׳ צִלְּךָ עַל יַד יְמִינֶךָ׳.

כְּמוֹ נַחְשׁוֹן, אֲנִי אוֹמֵר בַּלֵּב: ׳ה׳ אִתִּי׳, וְעוֹשֶׂה אֶת הַצַּעַד. זוֹ אֱמוּנָה.

בְּכָל פַּחַד אֲנִי יָכוֹל לִהְיוֹת נַחְשׁוֹן.

מָה לוֹקְחִים מִרָמָה א׳?

נַחְשׁוֹן קָפַץ כִּי הֶאֱמִין שֶׁה׳ אִתּוֹ. כְּשֶׁאֲנִי מְפַחֵד וְאוֹמֵר ׳ה׳ אִתִּי׳, אֲנִי מַרְגִּישׁ שֶׁה׳ קָרוֹב אֵלַי וְאוֹהֵב אוֹתִי.

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי הִרְגַּשְׁתִּי כְּמוֹ עַל שְׂפַת יָם, וּמָה חָשַׁבְתִּי אָז?

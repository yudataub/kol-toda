🎁 מַתָּנָה מֵהַמֶּלֶךְ

דִּמְיְנוּ שֶׁשָּׁלִיחַ שֶׁל הַמֶּלֶךְ דּוֹפֵק בַּדֶּלֶת וּמוֹסֵר לָכֶם חֲבִילָה עִם מַתָּנָה. מִי לֹא יִשְׂמַח?

בְּסֵפֶר חֲרֵדִים כָּתוּב שֶׁכָּל מִצְוָה וּמִצְוָה הִיא בְּדִיּוּק כָּךְ: ׳דּוֹרוֹן׳ – מַתָּנָה שֶׁשָּׁלַח לָנוּ הַמֶּלֶךְ, הַקָּדוֹשׁ בָּרוּךְ הוּא.

כָּל מִצְוָה הִיא דּוֹרוֹן, מַתָּנָה מֵהַמֶּלֶךְ.

1

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ כִּי ה׳ שָׁלַח לִי מַתָּנָה 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סֵפֶר חֲרֵדִים (רַבִּי אֶלְעָזָר אַזְכָּרִי)

רָמָה א׳ – מַתָּנָה מֵהַמֶּלֶךְ

👑 מַהוּ דּוֹרוֹן?

׳דּוֹרוֹן׳ הוּא מַתָּנָה שֶׁנּוֹתְנִים לְאָדָם חָשׁוּב. סֵפֶר חֲרֵדִים מְלַמֵּד: ׳דּוֹרָנָא הוּא דְּשָׁדַר לֵיהּ קוּדְשָׁא בְּרִיךְ הוּא׳ – זֶה דּוֹרוֹן שֶׁשָּׁלַח לוֹ הַקָּדוֹשׁ בָּרוּךְ הוּא.

מִי שֶׁמְּקַבֵּל מַתָּנָה מֵהַמֶּלֶךְ מַרְגִּישׁ חָשׁוּב וְשָׂמֵחַ. כָּךְ גַּם אֲנִי כְּשֶׁאֲנִי עוֹשֶׂה מִצְוָה.

דּוֹרוֹן = מַתָּנָה שֶׁנּוֹתְנִים לְאָדָם חָשׁוּב.

2

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹחֵר לִרְאוֹת בְּכָל מִצְוָה הִזְדַּמְּנוּת וּמַתָּנָה 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׁעוּר ׳שִׂמְחָה כְּמוֹ גּוֹל׳

💡 עֹל אוֹ מַתָּנָה?

אִם אָדָם מַרְגִּישׁ שֶׁהַמִּצְווֹת הֵן עֹל שֶׁמֻּטָּל עָלָיו, הוּא לֹא יִשְׂמַח. אֲבָל אִם הוּא מַרְגִּישׁ שֶׁכָּל מִצְוָה הִיא הִזְדַּמְּנוּת פָּז שֶׁה׳ זִכָּה אוֹתוֹ בָּהּ, הַשִּׂמְחָה בָּאָה מֵעַצְמָהּ.

הַיְּסוֹד לְשִׂמְחָה בְּמִצְווֹת: לְהַרְגִּישׁ שֶׁהַמִּצְוָה אֵינָהּ בַּקָּשָׁה שֶׁלָּנוּ, אֶלָּא מַתָּנָה שֶׁקִּבַּלְנוּ.

מִצְוָה אֵינָהּ בַּקָּשָׁה, הִיא מַתָּנָה.

3

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ גַּם בְּמִצְווֹת קְטַנּוֹת 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הִתְעוֹרְרוּת לַעֲבֹד עַל מִצְוַת הַשִּׂמְחָה

רָמָה א׳ – מַתָּנָה מֵהַמֶּלֶךְ

🥤 גַּם מִצְוָה קְטַנָּה

לֹא רַק אַחֲרֵי שְׁעוֹת שֶׁל לִמּוּד צָרִיךְ לִשְׂמֹחַ. גַּם אַחֲרֵי שֶׁנָּתַנּוּ כּוֹס מַיִם לְאָח קָטָן, אוֹ אַחֲרֵי שֶׁבֵּרַכְנוּ בִּרְכַּת ׳שֶׁהַכֹּל׳, אוֹ אַחֲרֵי עֶזְרָה קְטַנָּה בַּבַּיִת.

׳וְכָל מִצְוָה וּמִצְוָה שֶׁתִּזְדַּמֵּן לוֹ׳ – כָּל אַחַת מֵהֶן דּוֹרוֹן.

גַּם מִצְוָה קְטַנָּה הִיא דּוֹרוֹן.

4

💛 אַהֲבַת הַשֵּׁם: תּוֹדָה, ה׳, עַל הַיָּדַיִם וְעַל הַמִּצְוָה 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הִתְעוֹרְרוּת לַעֲבֹד עַל מִצְוַת הַשִּׂמְחָה

רָמָה א׳ – מַתָּנָה מֵהַמֶּלֶךְ

🙌 מִי נָתַן לִי יָדַיִם?

כְּשֶׁאֲנִי עוֹשֶׂה מִצְוָה, אֶפְשָׁר לַחְשֹׁב: מִי נָתַן לִי יָדַיִם לַעֲשׂוֹת אוֹתָהּ? מִי נָתַן לִי רַגְלַיִם לָלֶכֶת אֵלֶיהָ? וּמִי נָתַן לִי בִּכְלָל אֶת הַמִּצְוָה הַנֶּהְדֶּרֶת הַזֹּאת?

כְּשֶׁחוֹשְׁבִים כָּךְ, מְבִינִים שֶׁזֶּה לֹא דָּבָר ״מוּבָן מֵאֵלָיו״ אֶלָּא מַתָּנָה.

לֹא מוּבָן מֵאֵלָיו – מַתָּנָה שֶׁנִּתְּנָה לִי.

5

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְחַדֵּשׁ אֶת הַשִּׂמְחָה שֶׁלִּי בְּכָל פַּעַם 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׁעוּר ׳שִׂמְחָה כְּמוֹ גּוֹל׳

רָמָה א׳ – מַתָּנָה מֵהַמֶּלֶךְ

🔁 לֹא לְהִתְרַגֵּל

מִי שֶׁעוֹשֶׂה מִצְווֹת מִיּוֹם שֶׁנּוֹלַד עָלוּל לְהַרְגִּישׁ שֶׁזֶּה שִׁגְרָה, עוֹד דָּבָר שֶׁעוֹשִׂים. מִי שֶׁרַק הִתְחִיל מַרְגִּישׁ בְּכָל פַּעַם חִדּוּשׁ גָּדוֹל.

לָכֵן מִי שֶׁגָּדַל בְּבַיִת עִם מִצְווֹת צָרִיךְ לְחַדֵּשׁ בְּעַצְמוֹ אֶת הַהַרְגָּשָׁה. וְאִם יַצְלִיחַ, שְׂכָרוֹ גָּדוֹל, כִּי קָשֶׁה לוֹ יוֹתֵר.

מְחַדְּשִׁים אֶת הַהַרְגָּשָׁה, וְלֹא נִשְׁאָרִים בְּשִׁגְרָה.

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ שֶׁאֲנִי יָכוֹל לִשְׂמֹחַ בְּמִצְווֹתֶיךָ 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הִתְעוֹרְרוּת לַעֲבֹד עַל מִצְוַת הַשִּׂמְחָה

🌟 שָׂמֵחַ שֶׁאֲנִי יָכוֹל לִשְׂמֹחַ

יֵשׁ שִׁיטָה יָפָה: אֶפְשָׁר לִשְׂמֹחַ עַל זֶה שֶׁזָּכִיתִי לִשְׂמֹחַ. לֹא קַל לִשְׂמֹחַ בְּמִצְווֹת, וְאִם אֲנִי מַצְלִיחַ אֲפִילּוּ קְצָת – זוֹ זְכוּת.

כָּךְ הַשִּׂמְחָה מוֹלִידָה עוֹד שִׂמְחָה.

שָׂמֵחַ עַל כָּךְ שֶׁזָּכִיתִי לִשְׂמֹחַ.

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹשֶׂה אֶת הַמִּצְוָה בְּחִיּוּךְ, כִּי זֶה דּוֹרוֹן 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

✨ הַמֶּלֶךְ שָׁלַח לִי

לִפְנֵי מִצְוָה אַחַת אֲנִי יָכוֹל לַעֲצֹר וְלוֹמַר בַּלֵּב: ׳הַמֶּלֶךְ שָׁלַח לִי אֶת זֶה!׳ אַחַר כָּךְ אֲנִי עוֹשֶׂה אֶת הַמִּצְוָה בְּחִיּוּךְ.

ּכֵּן אֶפְשָׁר לִבְחֹר מִצְוָה ׳קְטַנָּה׳ אַחַת בְּכָל יוֹם וְלִרְאוֹת אוֹתָהּ כְּמַתָּנָה.

אוֹמְרִים בַּלֵּב: הַמֶּלֶךְ שָׁלַח לִי אֶת זֶה.

מָה לוֹקְחִים מִרָמָה א׳?

כָּל מִצְוָה הִיא מַתָּנָה שֶׁה׳ שׁוֹלֵחַ לִי. כְּשֶׁאֲנִי מְקַבֵּל אוֹתָהּ בְּחִיּוּךְ, אֲנִי מַרְגִּישׁ כַּמָּה הוּא אוֹהֵב אוֹתִי, וְאַהֲבָתִי אֵלָיו גְּדֵלָה.

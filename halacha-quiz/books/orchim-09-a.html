רָמָה א׳ – פֵּרוֹת בָּעוֹלָם הַזֶּה

🍽️ הַסְּעוּדָה וְהַבְּשׂוֹרָה

אַבְרָהָם וְשָׂרָה הֵאֱכִילוּ אוֹרְחִים שֶׁעָבְרוּ בַּמִּדְבָּר. בְּאוֹתָהּ סְעוּדָה אָמַר לוֹ אֶחָד הָאוֹרְחִים: ׳שׁוֹב אָשׁוּב אֵלֶיךָ כָּעֵת חַיָּה, וְהִנֵּה בֵן לְשָׂרָה אִשְׁתֶּךָ׳.

הַבְּשׂוֹרָה הַטּוֹבָה הִגִּיעָה בְּאוֹתָהּ סְעוּדָה מַמָּשׁ.

הַמִּצְוָה וְהַפְּרִי בָּאוּ כִּמְעַט בְּיַחַד.

הַבְּשׂוֹרָה הַטּוֹבָה בָּאָה בְּאוֹתָהּ סְעוּדָה שֶׁל הַכְנָסַת אוֹרְחִים.

רָמָה א׳ – פֵּרוֹת בָּעוֹלָם הַזֶּה

🌳 פֵּרוֹת וְקֶרֶן

חֲכָמִים מָנוּ אֶת הַכְנָסַת אוֹרְחִים בֵּין הַדְּבָרִים ׳שֶׁאָדָם אוֹכֵל פֵּרוֹתֵיהֶן בָּעוֹלָם הַזֶּה וְהַקֶּרֶן קַיֶּמֶת לוֹ לָעוֹלָם הַבָּא׳.

כְּמוֹ עֵץ: אוֹכְלִים אֶת הַפֵּרוֹת עַכְשָׁיו, וְהָעֵץ עַצְמוֹ נִשְׁאָר.

בְּמִצְוָה הַזֹּאת רוֹאִים פֵּרוֹת כְּבָר כָּאן.

אוֹכְלִים אֶת הַפֵּרוֹת בָּעוֹלָם הַזֶּה, וְהַקֶּרֶן נִשְׁמֶרֶת לָעוֹלָם הַבָּא.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַאֲמִין שֶׁמַּעֲשֶׂה טוֹב אֶחָד יָכוֹל לְהָגֵן עָלַי 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סֵפֶר הַבְּרִית · בְּרֵאשִׁית י״ט א׳

רָמָה א׳ – פֵּרוֹת בָּעוֹלָם הַזֶּה

🚪 לוֹט שֶׁקָּם

לוֹט יָשַׁב בְּשַׁעַר סְדֹם. כְּשֶׁרָאָה שְׁנֵי אוֹרְחִים, הוּא קָם לִקְרָאתָם וְהִכְנִיס אוֹתָם לְבֵיתוֹ.

כְּשֶׁסְּדֹם הָפְכָה, לוֹט נִצַּל. הַהַכְנָסָה שֶׁל הָאוֹרְחִים הָיְתָה הַפֶּתַח לְהַצָּלָה שֶׁלּוֹ.

מִצְוָה אַחַת, כְּשֶׁעוֹשִׂים אוֹתָהּ מֵהַלֵּב, יְכוֹלָה לְהַגֵּן.

לוֹט נִצַּל בִּזְכוּת שֶׁקָּם לִקְרַאת הָאוֹרְחִים וְהִכְנִיס אוֹתָם.

🛏️

💛 אַהֲבַת הַשֵּׁם: אֲנִי יָכוֹל לַעֲשׂוֹת חֶסֶד קָטָן, וְה׳ מַחֲזִיר בְּרָכָה גְּדוֹלָה 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְלָכִים ב׳ ד׳ · סֵפֶר הַבְּרִית

רָמָה א׳ – פֵּרוֹת בָּעוֹלָם הַזֶּה

🛏️ הָעֲלִיָּה הַקְּטַנָּה

הַשּׁוּנַמִּית הֵכִינָה לֶאֱלִישָׁע הַנָּבִיא עֲלִיָּה קְטַנָּה, חֶדֶר מְיֻחָד בַּבַּיִת, כְּדֵי שֶׁיִּהְיֶה לוֹ מָקוֹם לְהִתְאָרֵחַ.

בִּזְכוּת הַחֶדֶר הַקָּטָן הַזֶּה, זָכְתָה לְבֵן.

גַּם חֶסֶד קָטָן, כְּמוֹ חֶדֶר אֶחָד בַּבַּיִת, יָכוֹל לְהָבִיא בְּרָכָה גְּדוֹלָה.

חֶדֶר קָטָן לְאוֹרֵחַ הֵבִיא לַשּׁוּנַמִּית בְּרָכָה גְּדוֹלָה.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַאֲמִין שֶׁה׳ דּוֹאֵג לְמִי שֶׁדּוֹאֵג לַאֲחֵרִים 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְלָכִים א׳ י״ז · סֵפֶר הַבְּרִית

רָמָה א׳ – פֵּרוֹת בָּעוֹלָם הַזֶּה

🍞 בִּשְׁנַת רָעָב

בִּשְׁנַת רָעָב הִתְאָרֵחַ אֵלִיָּהוּ הַנָּבִיא אֵצֶל אַלְמָנָה בְּצָרְפַת.

בִּזְכוּת שֶׁאֵרְחָה אוֹתוֹ הִיא הִתְפַּרְנְסָה גַּם בְּזְמַן הָרָעָב.

דַּוְקָא כְּשֶׁקָּשֶׁה, אֲנָשִׁים שֶׁפּוֹתְחִים אֶת הַלֵּב לַאֲחֵרִים מְקַבְּלִים עֶזְרָה מֵה׳.

הָאַלְמָנָה מִצָּרְפַת הִתְפַּרְנְסָה בִּזְכוּת שֶׁאֵרְחָה אֶת אֵלִיָּהוּ.

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סֵפֶר הַבְּרִית · חֶפֶץ חַיִּים

רָמָה א׳ – פֵּרוֹת בָּעוֹלָם הַזֶּה

💌 מִכְתַּב אַהֲבָה קָטָן

מִי שֶׁמְּחַכֶּה לִישׁוּעָה – בְּפַרְנָסָה, בִּבְרִיאוּת, בְּשִׂמְחָה – יִפְתַּח אֶת הַבַּיִת. לֹא כְּדֵי ׳לִקְנוֹת׳ יְשׁוּעָה, אֶלָּא כִּי הַדֶּלֶת הַפְּתוּחָה הִיא כְּלִי לִבְרָכָה.

ה׳ מַרְאֶה לָנוּ פֵּרוֹת כְּבָר כָּאן, כְּדֵי שֶׁנַּרְגִּישׁ שֶׁהוּא שׁוֹמֵעַ וְרוֹאֶה.

כָּל יְשׁוּעָה כָּזוֹ הִיא ׳מִכְתַּב אַהֲבָה קָטָן׳ מִמֶּנּוּ.

הַדֶּלֶת הַפְּתוּחָה הִיא כְּלִי לִבְרָכָה, לֹא דֶּרֶךְ לִקְנוֹת יְשׁוּעָה.

👶

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְהַכְנִיס אוֹרְחִים בְּשִׂמְחָה כְּמוֹ אַבְרָהָם אָבִינוּ 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – פֵּרוֹת בָּעוֹלָם הַזֶּה

👶 קָרָא לוֹ אַבְרָהָם

מְסֻפָּר עַל רַב שֶׁהִתְאָרֵחַ אֶצְלוֹ אוֹרֵחַ בְּלִי שֶׁחִכָּה לוֹ. הָרַב קִבֵּל אוֹתוֹ בְּשִׂמְחָה גְּדוֹלָה וּבִזְרִיזוּת, כְּמוֹ אַבְרָהָם אָבִינוּ.

תִּשְׁעָה חֳדָשִׁים אַחַר כָּךְ נוֹלַד לוֹ בֵּן. מָה קָרָא לוֹ? אַבְרָהָם, לְזֵכֶר הַמַּעֲשֶׂה הַנֶּהְדָּר שֶׁל אַבְרָהָם אָבִינוּ שֶׁהִשְׁפִּיעַ עָלָיו כָּל כָּךְ.

הָרַב קָרָא לִבְנוֹ אַבְרָהָם לְזֵכֶר הַכְנָסַת הָאוֹרְחִים.

🏠

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַזְמִין בְּרָכָה לַבַּיִת כְּשֶׁאֲנִי פּוֹתֵחַ אוֹתוֹ לַאֲחֵרִים 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, חֶפֶץ חַיִּים, אַהֲבַת חֶסֶד · חִדּוּשֵׁי וַיֵּרָא

רָמָה א׳ – פֵּרוֹת בָּעוֹלָם הַזֶּה

🏠 בַּיִת פָּתוּחַ, בְּרָכָה נִכְנֶסֶת

מִי שֶׁמְּפַנֶּה בְּבֵיתוֹ מָקוֹם לַאֲחֵרִים, זוֹכֶה שֶׁה׳ יְמַלֵּא אֶת בֵּיתוֹ בְּבְרָכָה. זֶה מִדָּה כְּנֶגֶד מִדָּה: אֲנִי נוֹתֵן מָקוֹם, וְה׳ נוֹתֵן לִי.

הַבַּיִת הַפָּתוּחַ הוּא בַּיִת שֶׁהַבְּרָכָה נִכְנֶסֶת אֵלָיו.

גַּם אוֹרֵחַ אֶחָד בְּשַׁבָּת הוּא פֶּתַח לְשֶׁפַע.

בַּיִת שֶׁמַּכְנִיס אוֹרְחִים הוּא בַּיִת שֶׁמַּזְמִין בְּרָכָה.

💡

רָמָה א׳ – פֵּרוֹת בָּעוֹלָם הַזֶּה

מָה לוֹקְחִים מִרָמָה א׳?

ה׳ מַרְאֶה לָנוּ פֵּרוֹת שֶׁל חֶסֶד כְּבָר בָּעוֹלָם הַזֶּה, כְּדֵי שֶׁנַּרְגִּישׁ שֶׁהוּא שׁוֹמֵעַ וְרוֹאֶה. כָּל בְּרָכָה כָּזוֹ הִיא סִימָן לְאַהֲבָתוֹ אֵלֵינוּ, וְהִיא מְקָרֶבֶת אוֹתָנוּ אֵלָיו.

❤️
אֵלּוּ דְבָרִים שֶׁאָדָם אוֹכֵל פֵּרוֹתֵיהֶם בָּעוֹלָם הַזֶּה וְהַקֶּרֶן קַיֶּמֶת לוֹ לָעוֹלָם הַבָּא

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי קָרָה לִי מַשֶּׁהוּ טוֹב אַחֲרֵי שֶׁעָשִׂיתִי חֶסֶד, וּמָה הִרְגַּשְׁתִּי?
  • הַאִם אֲנִי פּוֹתֵחַ אֶת הַלֵּב לַאֲחֵרִים מֵאַהֲבָה, אוֹ רַק כְּשֶׁאֲנִי מְקַוֶּה לְקַבֵּל?
  • הַשָּׁבוּעַ אֶכְתֹּב טַבְלַת פְּרָס שֶׁל מִי הִכְנִיס אוֹרְחִים וּמָה קִבֵּל, וְאוֹסִיף שׁוּרָה: מִי הָאוֹרֵחַ הַבָּא שֶׁאֲנִי מַזְמִין.

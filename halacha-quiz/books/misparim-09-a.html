בְּבּוּאֶנּוֹס אַיְרֶס, בִּירַת אַרְגֶּנְטִינָה, הָיְתָה כַּלָּה צְעִירָה בְּלִי דִּירָה. לַהוֹרִים הָיָה רַק סְכוּם קָטָן, וְאִי אֶפְשָׁר הָיָה לִקְנוֹת בּוֹ דִּירָה. דּוֹדָתָהּ אָמְרָה לָהּ בֶּאֱמוּנָה תְּמִימָה: ׳תַּתְחִילִי אַרְבָּעִים יוֹם שֶׁל מִזְמוֹר לְתוֹדָה׳.

הַכַּלָּה כָּתְבָה עַל דַּף אַרְבָּעָה עָשָׂר תְּנָאִים שֶׁהִיא רוֹצָה בַּדִּירָה, וּבְכָל יוֹם הוֹדְתָה לַה׳ עַל דְּבָרִים טוֹבִים מֵהַחַיִּים שֶׁלָּהּ. עוֹד לִפְנֵי שֶׁנִּגְמְרוּ אַרְבָּעִים הַיָּמִים מָצְאוּ דִּירָה בַּסְּכוּם הַקָּטָן, וְכָל אַרְבָּעָה עָשָׂר הַתְּנָאִים הִתְקַיְּמוּ בָּהּ.

עוֹד לִפְנֵי שֶׁנִּגְמְרוּ אַרְבָּעִים יוֹם שֶׁל הוֹדָיָה, הַיְשׁוּעָה בָּאָה.

8
🎯

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה לַה׳ עַל הַכֹּחַ לְהִשְׁתַּפֵּר כָּל יוֹם 💛

📖 סוֹדָם שֶׁל הַמִּסְפָּרִים בַּתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, צֶעטִיל קָטָן ט״ז · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – אַרְבָּעִים יוֹם וְעוֹלָם חָדָשׁ

🎯 אַרְבָּעִים יוֹם שֶׁל הֶרְגֵּל טוֹב

רַבִּי אֱלִימֶלֶךְ מִלִּיזֶ׳נְסְק לָמַד: מִי שֶׁטִּבְעוֹ עַקְשָׁנִי, יִשְׁבֹּר אֶת טִבְעוֹ אַרְבָּעִים יוֹם רְצוּפִים. וּמִי שֶׁקָּשֶׁה לוֹ לְהַתְמִיד בַּלִּמּוּד, יַרְגִּיל אֶת עַצְמוֹ לִלְמֹד יוֹתֵר מֵהֶרְגֵּלוֹ אַרְבָּעִים יוֹם.

הַשִּׁעוּר מוֹסִיף: כָּל דָּבָר טוֹב שֶׁעוֹשִׂים אַרְבָּעִים יוֹם יֵשׁ לוֹ כֹּחַ מְיֻחָד לִהְיוֹת קָבוּעַ וְנֶחְקָק בַּנֶּפֶשׁ.

אַרְבָּעִים יוֹם רְצוּפִים, וְהֶרְגֵּל טוֹב נִקְבָּע.

9
💡

רָמָה א׳ – אַרְבָּעִים יוֹם וְעוֹלָם חָדָשׁ

מָה לוֹקְחִים מִרָמָה א׳?

ה׳ נוֹתֵן לָנוּ הַתְחָלוֹת חֲדָשׁוֹת: אַרְבָּעִים יוֹם וְהָעוֹלָם חָדָשׁ, אַרְבָּעִים יוֹם וְהָאָדָם חָדָשׁ. כְּשֶׁאֲנִי מוֹדֶה לוֹ, אֲנִי מַרְגִּישׁ אֶת אַהֲבָתוֹ שֶׁמְּחַדֶּשֶׁת אוֹתִי.

10
❤️
וַיְהִי הַגֶּשֶׁם עַל־הָאָרֶץ אַרְבָּעִים יוֹם וְאַרְבָּעִים לָיְלָה

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָה בַּחַיִּים שֶׁלִּי הוּא מַעֲבָר מִמַּצָּב אֶחָד לְמַצָּב חָדָשׁ, וּמָה צָרִיךְ כְּדֵי לַעֲבֹר אוֹתוֹ?
  • אֵיזֶה הֶרְגֵּל טוֹב הָיִיתִי רוֹצֶה לְהַתְחִיל, וְאֵיךְ אֶפְשָׁר לְהַמְשִׁיךְ בּוֹ כָּל יוֹם?
  • אֲנִי בּוֹחֵר הֶרְגֵּל טוֹב אֶחָד, וְעוֹשֶׂה אוֹתוֹ אַרְבָּעִים יוֹם רְצוּפִים, וְכָל יוֹם אוֹמֵר גַּם תּוֹדָה אַחַת לַה׳.

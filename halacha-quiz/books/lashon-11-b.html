רָמָה ב׳ – הָאֱמֶת וְהַמַּסְקָנָה

🔬 מָה הָיָה הַחֵטְא?

הָרַמְבַּ״ן שׁוֹאֵל: מֹשֶׁה שָׁלַח אוֹתָם לְדַוֵּחַ מַה שֶׁרָאוּ – אָז מָה פִּשְׁעָם? הוּא עוֹנֶה: בְּעִנְיַן הַפֵּרוֹת, הָעָם וְהֶעָרִים הֵם אָמְרוּ אֱמֶת. אֲבָל רִשְׁעָם בַּמִּלָּה ׳אֶפֶס׳, שֶׁמּוֹרָה עַל דָּבָר שֶׁאִי אֶפְשָׁר בְּשׁוּם אֹפֶן.

מֻתָּר הָיָה לָהֶם לְסַפֵּר שֶׁיֵּשׁ גִּבּוֹרִים וְעָרִים בְּצוּרוֹת. אָסוּר הָיָה לָהֶם לְהַסִּיק מִזֶּה שֶׁלֹּא נַצְלִיחַ לִכְבֹּשׁ אֶת הָאָרֶץ.

הַחֵטְא הוּא הַמַּסְקָנָה שֶׁהֵם הוֹצִיאוּ מִן הָאֱמֶת.

הַחֵטְא אֵינוֹ בָּעוּבְדּוֹת, אֶלָּא בַּמַּסְקָנָה: ׳אֶפֶס׳ וְלָכֵן לֹא נַצְלִיחַ.

רָמָה ב׳ – הָאֱמֶת וְהַמַּסְקָנָה

✂️ שְׁנֵי חֲלָקִים בַּדִּבָּה

הַתּוֹרָה מְחַלֶּקֶת אֶת הַחֵטְא לִשְׁנַיִם. בַּחֵלֶק הָרִאשׁוֹן הֵם אָמְרוּ ׳אֶפֶס׳ וְהִתְיָאֲשׁוּ מֵהַנִּצָּחוֹן. בַּחֵלֶק הַשֵּׁנִי כָּתוּב: ׳וַיּוֹצִיאוּ דִּבַּת הָאָרֶץ׳ – ׳אֶרֶץ אֹכֶלֶת יוֹשְׁבֶיהָ הִוא, וְכָל הָעָם אֲשֶׁר רָאִינוּ בְתוֹכָהּ אַנְשֵׁי מִדּוֹת׳.

הָרַמְבַּ״ן וְהַמַּלְבִּ״ים מֵסְבִּירִים: הֵם פֵּרְשׁוּ שֶׁהָאֲוִיר וְהַפֵּרוֹת בָּאָרֶץ חֲזָקִים כָּל כָּךְ, שֶׁרַק עֲנָקִים מִתְקַיְּמִים בָּהּ, וַאֲנָשִׁים רְגִילִים לֹא יוּכְלוּ לָגוּר שָׁם.

זֶה פֵּרוּשׁ שֶׁלָּהֶם – וְלֹא מַה שֶׁרָאוּ.

דִּבַּת הָאָרֶץ הִיא הַפֵּרוּשׁ הָרַע שֶׁנָּתְנוּ לְמַה שֶׁרָאוּ.

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׁעוּרֵי דִּבַּת הָאָרֶץ · הָרַב אֱלִיָּהוּ דִּיסְקִין

רָמָה ב׳ – הָאֱמֶת וְהַמַּסְקָנָה

🎒 אֵין לָהֶם רַשִׁ״י

הָרַב אֱלִיָּהוּ דִּיסְקִין מֵבִיא שְׁאֵלָה בְּשֵׁם הַמַּשְׁגִּיחַ רַבִּי דָּן סֶגַּל: אֶצְלֵנוּ יֵשׁ רַשִׁ״י שֶׁמְּסַפֵּר שֶׁהָאֵבֶל בֶּעָרִים הָיָה לְטוֹבָה, אֲבָל לַמְרַגְּלִים לֹא הָיָה רַשִׁ״י. מַה בִּקְשׁוּ מֵהֶם?

הַתְּשׁוּבָה: הֵם כְּבָר רָאוּ אֶת כָּל הַחֲסָדִים – קְרִיעַת יַם סוּף, הַמָּן, הָעֲנָנִים וְהַבְּאֵר. כְּשֶׁאָדָם מַכִּיר כָּל כָּךְ הַרְבֵּה טוֹבוֹת, הוּא יוֹדֵעַ שֶׁלֹּא יָבִיא אוֹתוֹ ה׳ לְאֶרֶץ מְסֻכֶּנֶת.

גַּם הַמַּסְקָנָה ׳הַמּוּזָרָה׳ שֶׁיֵּשׁ לָהּ הֶסְבֵּר טוֹב, יְשָׁרָה יוֹתֵר מֵהַמַּחֲשָׁבָה שֶׁה׳ מַכְשִׁיל אוֹתָנוּ.

מִי שֶׁזּוֹכֵר אֶת הַחֲסָדִים, מְחַפֵּשׂ הֶסְבֵּר טוֹב לַמַּצָּב.

רָמָה ב׳ – הָאֱמֶת וְהַמַּסְקָנָה

🧠 אֲנָשִׁים נִרְגָּנִים

בְּסֵפֶר דְּבָרִים כָּתוּב עַל הָעָם: ׳וַתֵּרָגְנוּ בְאָהֳלֵיכֶם וַתֹּאמְרוּ בְּשִׂנְאַת ה׳ אֹתָנוּ הוֹצִיאָנוּ מֵאֶרֶץ מִצְרָיִם׳. מַה זֶה נִרְגָּן? רַבֵּנוּ יוֹנָה כּוֹתֵב: הַנִּרְגָּן דָּן כָּל דָּבָר לְחוֹבָה וְלֹא לִזְכוּת, וּפְעָמִים הוּא חוֹשֵׁב אֲפִלּוּ עַל חֲסָדֵי ה׳ שֶׁהֵם נָקָם.

אֶצְל מֹשֶׁה אוֹתָן עוּבְדּוֹת הָיוּ סִבָּה לִבְטֹחַ, וְאֶצְל הַנִּרְגָּנִים – סִבָּה לְהִתְיָאֵשׁ.

הַהֶבְדֵּל אֵינוֹ בְּמָה שֶׁרוֹאִים אֶלָּא בְּאֵיךְ מְפָרְשִׁים.

אוֹתָן עוּבְדּוֹת: אֶחָד רוֹאֶה אַהֲבָה וְאֶחָד רוֹאֶה שִׂנְאָה, כִּי הַפֵּרוּשׁ שׁוֹנֶה.

🕊️ יְהוֹשֻׁעַ וְכָלֵב

יְהוֹשֻׁעַ וְכָלֵב רָאוּ אֶת אוֹתָהּ אֶרֶץ, וְאָמְרוּ: ׳טוֹבָה הָאָרֶץ מְאֹד מְאֹד. אִם חָפֵץ בָּנוּ ה׳ וְהֵבִיא אֹתָנוּ אֶל הָאָרֶץ הַזֹּאת…׳. כָּלֵב אָמַר: ׳עָלֹה נַעֲלֶה וְיָרַשְׁנוּ אֹתָהּ כִּי יָכוֹל נוּכַל לָהּ׳.

הֵם לֹא הִסְתִּירוּ שֶׁיֵּשׁ קֹשִׁי. הֵם אָמְרוּ אוֹתוֹ עִם אֱמוּנָה: ה׳ אִתָּנוּ.

עֲשָׂרָה מְרַגְּלִים הוֹצִיאוּ דִּבָּה, וּשְׁנַיִם תִּקְּנוּ אֶת הַמַּבָּט. אֶפְשָׁר לְסַפֵּר אֶת אוֹתָהּ הָאֱמֶת בִּשְׁתֵּי דְּרָכִים.

אוֹתָהּ אֱמֶת, בִּשְׁתֵּי דְּרָכִים: אֶחָד מְיַאֵשׁ וְאֶחָד מְחַזֵּק.

🧱 אֱמֶת אֵינָהּ הֶתֵּר

בַּהֲלָכָה הָרִאשׁוֹנָה שֶׁל הֶחָפֵץ חַיִּים כָּתוּב שֶׁלָּשׁוֹן הָרָע אָסוּר גַּם כְּשֶׁהוּא אֱמֶת. כָּךְ גַּם בְּדִבַּת הָאָרֶץ: דִּבּוּר שֶׁגּוֹרֵם לַאֲנָשִׁים לֶאֱהֹב אֶת הָאָרֶץ פָּחוֹת – זֶה הָאִסּוּר, וְלֹא שׁוֹאֲלִים אִם הוּא נָכוֹן.

מִי שֶׁבָּא לְסַפֵּר עַל קֹשִׁי יָכוֹל לִשְׁאֹל אֶת עַצְמוֹ: הַאִם אֲנִי מְסַפֵּר כְּדֵי לְתַקֵּן, אוֹ רַק מוֹסִיף עוֹד סִבָּה לְהִתְרַחֵק?

הַשְּׁאֵלָה הִיא לֹא ׳הַאִם זֶה נָכוֹן׳ אֶלָּא ׳מָה זֶה עוֹשֶׂה׳.

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְרַגְּלִים לֹא הָיוּ צְרִיכִים לְשַׁקֵּר כְּדֵי לִפְגֹּעַ: הַמַּסְקָנָה הָרָעָה הָיְתָה מַסְפִּיקָה. מִי שֶׁזּוֹכֵר שֶׁה׳ אוֹהֵב אוֹתָנוּ וְנָתַן לָנוּ אֶרֶץ טוֹבָה, רוֹאֶה בַּקֹּשִׁי הִזְדַּמְּנוּת לִבְטֹחַ בּוֹ וּלְהוֹדוֹת לוֹ.

🤔 חוֹשְׁבִים עַל זֶה

  • אֵיפֹה הַשָּׁבוּעַ סִפַּרְתִּי דָּבָר נָכוֹן, אֲבָל הוּא גָּרַם לַשּׁוֹמֵעַ לְהִתְרַחֵק, וְאֵיךְ אֲסַפֵּר אוֹתוֹ בַּפַּעַם הַבָּאָה?

  • אֲצַיֵּר אוֹתוֹ מַרְאֶה פַּעֲמַיִם – בְּעַיִן טוֹבָה וּבְעַיִן רָעָה – וְאֶרְשֹׁם מָה הִרְגַּשְׁתִּי בְּכָל צִיּוּר.

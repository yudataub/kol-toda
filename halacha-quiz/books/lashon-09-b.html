רָמָה ב׳ – לָשׁוֹן חֲכָמִים מַרְפֵּא

😤 נִרְגָּנוּת – הַקּוֹץ הַגָּדוֹל

נִרְגָּן הוּא מִי שֶׁמִּתְלוֹנֵן כָּל הַזְּמַן וְרוֹאֶה רַק מַה שֶּׁחָסֵר. בַּמִּדְבָּר עַם יִשְׂרָאֵל לֹא הִתְבּוֹנֵן בַּחֶסֶד הַגָּדוֹל שֶׁה׳ רָצָה לַעֲשׂוֹת: לְהַכְנִיס אוֹתָם לָאָרֶץ. בִּמְקוֹם לְהוֹדוֹת – הֵם הִתְלוֹנְנוּ.

אִם נִרְגָּנוּת גּוֹרֶמֶת לְכָךְ, כַּמָּה גְּדוֹלָה הַהוֹדָיָה! כְּשֶׁאָדָם מוֹצִיא מִפִּיו מִלִּים שֶׁל הוֹדָיָה, הוּא מַשְׁפִּיעַ טוֹב עַל כָּל הַסְּבִיבָה שֶׁלּוֹ.

נִרְגָּנוּת רוֹאָה רַק אֶת הָרַע, וְהוֹדָיָה הִיא הַפֶּרַח שֶׁמַּחֲלִיף אוֹתָהּ.

רָמָה ב׳ – לָשׁוֹן חֲכָמִים מַרְפֵּא

⚔️ חֶרֶב אוֹ מַרְפֵּא

בַּמִּשְׁלֵי כָּתוּב: ׳יֵשׁ בּוֹטֶה כְּמַדְקְרוֹת חָרֶב וּלְשׁוֹן חֲכָמִים מַרְפֵּא׳. מִלִּים חַדּוֹת דּוֹקְרוֹת כְּמוֹ קוֹצִים אוֹ כְּמוֹ חֶרֶב, אֲבָל לְשׁוֹן שֶׁל חֲכָמִים מְרַפְּאָה.

וְעוֹד כָּתוּב שָׁם: ׳מַרְפֵּא לָשׁוֹן עֵץ חַיִּים׳. לָשׁוֹן טוֹבָה הִיא כְּמוֹ עֵץ שֶׁל חַיִּים: הִיא נוֹתֶנֶת כֹּחַ וְחַיִּים.

אֶפְשָׁר לִבְחֹר: לִדְקֹר אוֹ לְרַפֵּא.

מִלִּים יְכוֹלוֹת לִדְקֹר כְּמוֹ חֶרֶב, וִיכוֹלוֹת לְרַפֵּא כְּמוֹ עֵץ חַיִּים.

💛 אַהֲבַת הַשֵּׁם: אֲנִי שׁוֹמֵר עַל הַכָּבוֹד שֶׁל כָּל חָבֵר, כִּי כָּל אָדָם נִבְרָא בְּצֶלֶם ה׳ 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בָּבָא מְצִיעָא נ״ח ב׳

רָמָה ב׳ – לָשׁוֹן חֲכָמִים מַרְפֵּא

😳 הַלְבָּנַת פָּנִים

חֲכָמִים אָמְרוּ: הַמַּלְבִּין פְּנֵי חֲבֵרוֹ בָּרַבִּים – כְּאִלּוּ שׁוֹפֵךְ דָּמִים.

אֶחָד הַחֲכָמִים, רַב נַחְמָן בַּר יִצְחָק, הִסְכִּים וְהִסְבִּיר: רוֹאִים אֶת הַפָּנִים הָאֲדֻמּוֹת נֶעֶלְמוֹת, וְהַפָּנִים נַעֲשׂוֹת לְבָנוֹת.

לָכֵן כְּשֶׁחָבֵר טוֹעֶה אוֹ נוֹפֵל לִפְנֵי כֻּלָּם, לֹא צוֹחֲקִים וְלֹא מַצְבִּיעִים. עוֹזְרִים לוֹ בְּשֶׁקֶט.

בּוּשָׁה בָּרַבִּים כּוֹאֶבֶת כָּל כָּךְ, עַד שֶׁחֲכָמִים הִשְׁווּ אוֹתָהּ לִשְׁפִיכוּת דָּמִים.

🎯

💛 אַהֲבַת הַשֵּׁם: אֲנִי נוֹתֵן לַחֲבֵרִים שֵׁם טוֹב, כִּי כָּל אֶחָד יָקָר לַה׳ 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – לָשׁוֹן חֲכָמִים מַרְפֵּא

🎯 עַל הַמַּעֲשֶׂה, לֹא עַל הַיֶּלֶד

שֵׁם גְּנַאי כְּמוֹ ׳שַׁקְרָן׳, ׳טִפֵּשׁ׳ אוֹ ׳שָׁמֵן׳ נִדְבָּק בַּלֵּב וְיָכוֹל לְהִשָּׁאֵר שָׁנִים. הַיֶּלֶד מַתְחִיל לַחְשֹׁב: ׳אוּלַי זֶה בֶּאֱמֶת אֲנִי?׳

לָכֵן הַכְּלָל: אִם צָרִיךְ לוֹמַר מַשֶּׁהוּ, מְדַבְּרִים עַל הַמַּעֲשֶׂה, לֹא עַל הַיֶּלֶד.

וְהַדֶּרֶךְ הַטּוֹבָה בְּיוֹתֵר: לִשְׁתֹּק, אוֹ לְעוֹדֵד: ׳גַּם אֲנִי טָעִיתִי פַּעַם׳.

מִלָּה עַל הַמַּעֲשֶׂה אֶפְשָׁר לְתַקֵּן, וְשֵׁם עַל הַיֶּלֶד נִשְׁאָר.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְחַפֵּשׂ אֶת הָרֶגַע הַנָּכוֹן לְשַׂמֵּחַ אֶת הַלֵּב שֶׁל חָבֵר 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשְׁלֵי כ״ה י״א · י״ב כ״ה

רָמָה ב׳ – לָשׁוֹן חֲכָמִים מַרְפֵּא

🍏 תַּפּוּחֵי זָהָב

בַּמִּשְׁלֵי כָּתוּב: ׳תַּפּוּחֵי זָהָב בְּמַשְׂכִּיּוֹת כָּסֶף – דָּבָר דָּבֻר עַל אָפְנָיו׳. מִלָּה טוֹבָה הִיא כְּמוֹ תַּכְשִׁיט יָקָר, וְצָרִיךְ לוֹמַר אוֹתָהּ בַּזְּמַן הַנָּכוֹן, בַּמָּקוֹם הַנָּכוֹן וּבַדֶּרֶךְ הַנְּכוֹנָה.

וְעוֹד כָּתוּב: ׳דְּאָגָה בְלֶב אִישׁ יַשְׁחֶנָּה וְדָבָר טוֹב יְשַׂמְּחֶנָּה׳ – הַדְּאָגָה מַכְבִּידָה עַל הַלֵּב, וְדָבָר טוֹב מְשַׂמֵּחַ אוֹתוֹ.

מִלָּה טוֹבָה בִּזְמַנָּהּ מְשַׂמַּחַת לֵב דּוֹאֵג.

💛 אַהֲבַת הַשֵּׁם: אֲנִי זוֹרֵעַ מִלִּים טוֹבוֹת, וְה׳ מַצְמִיחַ מֵהֶן שִׂמְחָה 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – לָשׁוֹן חֲכָמִים מַרְפֵּא

🌱 זְרָעִים שֶׁל מִלִּים

מִלִּים הֵן כְּמוֹ זְרָעִים. אִם זוֹרְעִים מִלִּים טוֹבוֹת כְּמוֹ ׳אַתָּה חָבֵר נֶהְדָּר׳, יִצְמְחוּ פְּרָחִים שֶׁל חֲבֵרוּת וְשִׂמְחָה. אִם זוֹרְעִים מִלִּים רָעוֹת כְּמוֹ ׳אַתָּה לֹא תַּצְלִיחַ׳, יִצְמְחוּ קוֹצִים שֶׁל רִיב וְעֶצֶב.

כְּשֶׁרוֹאִים אֶת הַטּוֹב בַּחָבֵר וּמְדַבְּרִים עָלָיו, עוֹזְרִים לַטּוֹב לִגְדֹּל.

בִּמְקוֹם שֶׁאֵין מָה לוֹמַר טוֹב – שׁוֹתְקִים. וּבִמְקוֹם שֶׁיֵּשׁ – אוֹמְרִים בְּשִׂמְחָה.

מִלִּים טוֹבוֹת הֵן זְרָעִים שֶׁל פְּרָחִים.

מָה לוֹקְחִים מִרָמָה ב׳?

פֶּה שֶׁנּוֹתֵן פְּרָחִים הוּא פֶּה שֶׁמּוֹדֶה וְשָׂמֵחַ. כְּשֶׁאֲנִי מְחַפֵּשׂ אֶת הַטּוֹב בַּאֲחֵרִים וְאוֹמֵר אוֹתוֹ, אֲנִי מוֹדֶה לַה׳ עַל הַבְּרִיּוֹת שֶׁבָּרָא, וְאַהֲבָתִי אֵלָיו גְּדֵלָה.

  • אֵיזוֹ מִלָּה ׳קוֹץ׳ אָמַרְתִּי אוֹ שָׁמַעְתִּי הַשָּׁבוּעַ, וּבְאֵיזוֹ מִלָּה ׳פֶּרַח׳ אֶפְשָׁר לְהַחֲלִיף אוֹתָהּ?
  • מִי יוּכַל לִשְׂמֹחַ הַיּוֹם מִמִּלָּה טוֹבָה שֶׁלִּי, וּמָה אֹמַר לוֹ?
  • אֹמַר מַחְמָאָה אֲמִתִּית אַחַת לְחָבֵר בַּכִּתָּה, אוֹדֶה לְאַבָּא אוֹ לְאִמָּא עַל דָּבָר שֶׁעָשׂוּ – וּמָחָר אֲסַפֵּר לַכִּתָּה אֵיךְ זֶה הִרְגִּישׁ.

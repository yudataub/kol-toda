רָמָה ב׳ – אַרְבַּע הַתַּחֲנוֹת

🏋️ לָמָּה צָרִיךְ אִמּוּן?

חֲזַ״ל לִמְּדוּ: טוֹב הַמַּלְבִּין שִׁנַּיִם לַחֲבֵרוֹ יוֹתֵר מִמַּשְׁקֵהוּ חָלָב. כְּלוֹמַר, חִיּוּךְ שֶׁמַּרְאֶה לֶחָבֵר שֶׁשְּׂמֵחִים בּוֹ שָׁוֶה יוֹתֵר מִכּוֹס חָלָב.

אֲבָל כְּדֵי לָתֵת חִיּוּךְ אֲמִתִּי צָרִיךְ גַּם אֶת הַיְּכֹלֶת: פָּנִים רְפוּיוֹת, שְׁרִירִים חֲזָקִים וְלֵב רָגוּעַ.

לָכֵן בַּשִּׁעוּר הַזֶּה לוֹמְדִים לְהִתְאַמֵּן בְּחִיּוּךְ.

חִיּוּךְ הוּא מִצְוָה, וּמִצְוָה אֲנַחְנוּ מִתְאַמְּנִים לְקַיֵּם.

1

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא מְוַתֵּר עַל הַחִיּוּךְ בָּאֶמְצַע, אֲפִלּוּ כְּשֶׁזֶּה מְאַמֵּץ 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סֵפֶר הַחִיּוּךְ · מֵאִיר גּוּאֶטָה וְיוֹסִי עִמָּנוּאֵל

רָמָה ב׳ – אַרְבַּע הַתַּחֲנוֹת

🪞 תַּחֲנָה רִאשׁוֹנָה: שְׁרִירֵי הַחִיּוּךְ

מֵאִיר גּוּאֶטָה מַסְבִּיר שֶׁהַפָּנִים הֵם שְׁרִירִים, וּבְלִי שִׁמּוּשׁ הֵם נֶחֱלָשִׁים.

בְּ׳הַחִיּוּךְ הַקָּבוּעַ׳ מַחְזִיקִים חִיּוּךְ רָחָב 60 שְׁנִיּוֹת מוּל מַרְאָה: אַחֲרֵי 15 שְׁנִיּוֹת מוּזָר, אַחֲרֵי 30 רַעַד קַל בַּלְּחָיַיִם, אַחֲרֵי 45 מַאֲמָץ. בְּאוֹתוֹ רֶגַע הַמֹּחַ מֵבִין שֶׁאֲנַחְנוּ רְצִינִיִּים.

יוֹסִי עִמָּנוּאֵל מְסַפֵּר שֶׁגַּם חִיּוּךְ שֶׁמַּתְחִילִים בְּכַוָּנָה, כְּשֶׁמַּחְזִיקִים אוֹתוֹ דַּקָּה, מַשְׂמִיחַ אֶת הַלֵּב. לַמִּתְקַדְּמִים: עִפָּרוֹן נָקִי לָרֹחַב בֵּין הַשִּׁנַּיִם.

60 שְׁנִיּוֹת שֶׁל חִיּוּךְ מוּל מַרְאָה מַתְחִילוֹת לְהַשְׂמִיחַ אֶת הַלֵּב.

2

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְשַׁחְרֵר מֶתַח לִפְנֵי שֶׁאֲנִי מְדַבֵּר 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סֵפֶר הַחִיּוּךְ · מַרְסִיָּה בְּרִיר

רָמָה ב׳ – אַרְבַּע הַתַּחֲנוֹת

🧘 תַּחֲנָה שְׁנִיָּה: שִׁחְרוּר הַמֶּתַח

לִפְעָמִים לֹא הַשְּׁרִירִים חַלָּשִׁים אֶלָּא הַגּוּף לָחוּץ. אָז הַכְּתֵפַיִם מִתְכַּוְּצוֹת וְהַלֶּסֶת נִנְעֶלֶת, וְחִיּוּךְ בְּכֹחַ הוּא כְּמוֹ פְּתִיחַת דֶּלֶת נְעוּלָה.

מַרְסִיָּה בְּרִיר מַצִּיעָה שְׁנֵי תַּרְגִּילִים: שִׁחְרוּר הַלֶּסֶת (עִסּוּי מִתַּחַת לָאָזְנַיִם וּפִהוּק אָרֹךְ) וְהַצְלָבוֹת (בֶּרֶךְ נוֹגַעַת בְּמַרְפֵּק הַצַּד הַשֵּׁנִי, דַּקָּה לְאַט).

אַחֲרֵיהֶם הַפָּנִים מִתְרַכְּכוֹת, וְהַחִיּוּךְ הָאֲמִתִּי שֶׁבָּא מֵהָעֵינַיִם יָכוֹל לַעֲלוֹת.

קֹדֶם מְשַׁחְרְרִים אֶת הַמֶּתַח, וְאָז הַחִיּוּךְ עוֹלֶה בְּקַלּוּת.

3

💛 אַהֲבַת הַשֵּׁם: אֲנִי דָּן אֶת חֲבֵרִי לְכַף זְכוּת וּמוֹצֵא אֶת הַיַּהֲלוֹם 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סֵפֶר הַחִיּוּךְ, פֶּרֶק י״ד

רָמָה ב׳ – אַרְבַּע הַתַּחֲנוֹת

💎 תַּחֲנָה שְׁלִישִׁית: הַיַּהֲלוֹם

הַסִּבָּה הָעִקָּרִית שֶׁאֲנַחְנוּ לֹא מְחַיְּכִים הִיא הַפֵּרוּשׁ שֶׁאֲנַחְנוּ נוֹתְנִים לַמַּצָּב. חָבֵר נִתְקַע בִּי בַּמִּסְדְּרוֹן, וְהַפֵּרוּשׁ הָאוֹטוֹמָטִי הוּא ׳הוּא עָשָׂה אֶת זֶה בְּכַוָּנָה!׳

בְּתַרְגִּיל הַיַּהֲלוֹם עוֹצְרִים, נוֹשְׁמִים, וּמְסַפְּרִים סִפּוּר אַחֵר: אוּלַי הוּא מִהֵר וְלֹא רָאָה אוֹתִי. אָח קָטָן שָׁפַךְ שׁוֹקוֹ? בָּרוּךְ ה׳ שֶׁיֵּשׁ לִי אָח בָּרִיא, וְשׁוֹקוֹ, וְשָׁטִיחַ.

בַּהַתְחָלָה זֶה מַרְגִּישׁ מְאֻלָּץ, אֲבָל אַחֲרֵי אִמּוּן הַמֹּחַ לוֹמֵד לַחְפֹּשׂ אֶת הַטּוֹב לְבַד. זֶה הַמַּעֲשֶׂה שֶׁל דָּן לְכַף זְכוּת, כְּמוֹ שֶׁלָּמַדְנוּ אֵצֶל יוֹסֵף.

לְלַטֵּשׁ אֶת הַסִּפּוּר: לְחַפֵּשׂ פֵּרוּשׁ טוֹב שֶׁגּוֹרֵם לְחַיֵּךְ.

4

💛 אַהֲבַת הַשֵּׁם: אֲנִי פּוֹתֵחַ אֶת הַיּוֹם בְּ׳מוֹדֶה אֲנִי׳ וּבְחִיּוּךְ 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, ׳מוֹדֶה אֲנִי׳ · סֵפֶר הַחִיּוּךְ

רָמָה ב׳ – אַרְבַּע הַתַּחֲנוֹת

☀️ תַּחֲנָה רְבִיעִית: הַתְחָלָה בְּחִיּוּךְ

בַּשְּׁנִיּוֹת הָרִאשׁוֹנוֹת אַחֲרֵי הַהִתְעוֹרְרוּת הַלֵּב פָּתוּחַ בְּמִיחוּד. לָכֵן:

1. כְּשֶׁהַשָּׁעוֹן מְצַלְצֵל לֹא קוֹפְצִים מֵהַמִּטָּה.

2. מוֹתְחִים חִיּוּךְ, עוֹד לִפְנֵי הַמַּחֲשָׁבָה הָרִאשׁוֹנָה.

3. אוֹמְרִים ׳מוֹדֶה אֲנִי׳ מִתּוֹךְ הַחִיּוּךְ.

4. חֲצִי דַּקָּה מְדַמְיְנִים אֶת הַיּוֹם כְּיוֹם מֻצְלָח: מְחַיְּכִים לַחֲבֵרִים וּמַצְלִיחִים בַּמְּשִׂימוֹת.

מִי שֶׁמַּתְחִיל בַּחִיּוּךְ יוֹצֵר לְעַצְמוֹ תְּנוּפָה טוֹבָה לְכָל הַיּוֹם.

בֹּקֶר שֶׁמַּתְחִיל בְּחִיּוּךְ נוֹתֵן תְּנוּפָה לְכָל הַיּוֹם.

🔁 אַחֲרֵי הַמַּעֲשִׂים נִמְשָׁכִים הַלְּבָבוֹת

אוּלַי מִישֶׁהוּ יֹאמַר: ׳אִם אֲנִי מְחַיֵּךְ בִּשְׁבִיל הָאִמּוּן, אֲנִי צָבוּעַ׳. הַיֵּצֶר הָרַע אוֹהֵב לִלְחֹשׁ אֶת זֶה.

אֲבָל סֵפֶר הַחִנּוּךְ מְלַמֵּד: אַחֲרֵי הַמַּעֲשִׂים נִמְשָׁכִים הַלְּבָבוֹת. הַמַּעֲשֶׂה הַחִיצוֹנִי מְעוֹרֵר אֶת הַפְּנִימִי.

וְהַתְּשׁוּבָה לַיֵּצֶר הִיא בְּחִיּוּךְ: ׳אֲנִי לֹא צָבוּעַ, אֲנִי מִתְאַמֵּן. אֲנִי בּוֹנֶה אֶת הַכֵּלִים שֶׁיַּחֲזִיקוּ אֶת הָאוֹר.׳

מִתְאַמְּנִים בְּחִיּוּךְ לֹא מִתּוֹךְ צְבִיעוּת אֶלָּא כְּדֵי לִבְנוֹת כֵּלִים לָאוֹר.

מָה לוֹקְחִים מִרָמָה ב׳?

🤔 חוֹשְׁבִים עַל זֶה

  • אֵיזוֹ מֵאַרְבַּע הַתַּחֲנוֹת הַכִּי קַלָּה לִי, וְאֵיזוֹ הַכִּי קָשָׁה?
  • מָה קָרָה לִי הַשָּׁבוּעַ שֶׁאֶפְשָׁר ׳לְלַטֵּשׁ׳ לְיַהֲלוֹם, כְּמוֹ אָח שֶׁשָּׁפַךְ שׁוֹקוֹ אוֹ חָבֵר שֶׁנִּתְקַע בִּי?
  • אֲנִי מַתְחִיל אֶתְגָּר 21 יָמִים: בַּבֹּקֶר ׳מוֹדֶה אֲנִי׳ בְּחִיּוּךְ, דַּקָּה חִיּוּךְ מוּל הַמַּרְאָה, וּבִרְגַע שֶׁל כַּעַס שִׁחְרוּר לֶסֶת וְיַהֲלוֹם. מַדְפִּיס טַבְלָה, מוֹצֵא שֻׁתָּף לְאִמּוּן וּמְסַמֵּן ✔ בְּכָל יוֹם.

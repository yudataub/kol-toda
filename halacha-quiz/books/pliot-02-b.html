רָמָה ב׳ – מַה אוֹמְרִים הַמְּפָרְשִׁים

🔍 מָה הַחָכְמָה? שָׁלוֹשׁ תְּשׁוּבוֹת

הַמְּפָרְשִׁים שׁוֹאֲלִים: הֲרֵי כָּל הָעוֹלָם נִבְרָא בְּחָכְמָה – ׳כֻּלָּם בְּחָכְמָה עָשִׂיתָ׳ – אָז לָמָּה מַדְגִּישִׁים אֶת הָאָדָם? יֵשׁ שָׁלוֹשׁ תְּשׁוּבוֹת.

רִאשׁוֹנָה: בָּאָדָם יֵשׁ חָכְמָה נִפְלָאָה מְיֻחֶדֶת. שְׁנִיָּה (הַמַּהֲרַשׁ״א): ה׳ נָתַן לָאָדָם גַּם חָכְמָה בָּרֹאשׁ שֶׁאֵין לַבְּהֵמָה, כְּדֵי שֶׁיִּתְבּוֹנֵן. שְׁלִישִׁית (הַתּוֹסָפוֹת): ה׳ הֵכִין אֶת הַמָּזוֹן לִפְנֵי שֶׁבָּרָא אֶת הָאָדָם.

כֻּלָּן אֱמֶת, וְכֻלָּן סִבָּה לְהוֹדָיָה.

בְּחָכְמָה: חָכְמָה בַּגּוּף, חָכְמָה בָּרֹאשׁ, וּמָזוֹן שֶׁהוּכַן מֵרֹאשׁ.

רָמָה ב׳ – מַה אוֹמְרִים הַמְּפָרְשִׁים

🎁 חָכְמָה אוֹ חֶסֶד?

רַבִּי יְחֶזְקֵאל מִקּוֹזְמִיר שָׁאַל: אִם ה׳ הֵכִין אֹכֶל לִפְנֵי שֶׁבָּרָא אֶת הָאָדָם, מַדּוּעַ זֶה נִקְרָא ׳חָכְמָה׳ וְלֹא ׳חֶסֶד׳?

וְהַתְּשׁוּבָה: אִם הָאָדָם הָיָה נִבְרָא קֹדֶם, הָיָה צָרִיךְ לִהְיוֹת רָאוּי לְקַבֵּל אֹכֶל. עַכְשָׁו הָאֹכֶל כְּבָר קַיָּם, וְ׳זֶה נֶהֱנֶה וְזֶה אֵינוֹ חָסֵר׳. גַּם מִי שֶׁעוֹד לֹא הִסְפִּיק לִהְיוֹת רָאוּי יָכוֹל לְקַבֵּל.

זוֹ חָכְמָה שֶׁל מִי שֶׁרוֹאֶה מֵרֹאשׁ מָה הָאָדָם יִצְטָרֵךְ.

הָאֹכֶל הוּכַן מֵרֹאשׁ, וְגַם מִי שֶׁעוֹד לֹא רָאוּי יָכוֹל לְקַבֵּל.

רָמָה ב׳ – מַה אוֹמְרִים הַמְּפָרְשִׁים

🎈 רַשִׁ״י וְהַתּוֹסָפוֹת עַל הַנָּאד

גַּם הַתּוֹסָפוֹת מְסַפְּרִים עַל הַנָּאד: אָדָם עוֹשֶׂה נָאד, וַאֲפִלּוּ נֶקֶב אֶחָד קָטָן כְּמוֹ שֶׁל מַחַט מַסְפִּיק כְּדֵי שֶׁהָרוּחַ תֵּצֵא וְהוּא לֹא יוּכַל לִשְׁמֹר בּוֹ יַיִן. וְה׳ בָּרָא נְקָבִים נְקָבִים בָּאָדָם – וְהָרוּחַ נִשְׁמֶרֶת!

יֵשׁ הֶבְדֵּל בֵּין רַשִׁ״י לַתּוֹסָפוֹת: לְפִי רַשִׁ״י הַחָכְמָה הִיא הַנְּקָבִים הָרַבִּים עַצְמָם. לְפִי הַתּוֹסָפוֹת הַחָכְמָה הִיא בְּעֶצֶם יְצִירַת הָאָדָם, וְהַנָּאד הוּא הַפֶּלֶא.

שְׁנֵיהֶם מַסְכִּימִים: הַגּוּף שֶׁלָּנוּ הוּא פֶּלֶא גָּדוֹל.

נָאד עִם נֶקֶב אֶחָד מִתְרוֹקֵן, וְהַגּוּף שׁוֹמֵר עַל הָרוּחַ.

🌸 לָמָּה פַּעֲמַיִם?

בַּבְּרָכָה אוֹמְרִים פַּעֲמַיִם ׳נְקָבִים נְקָבִים׳ וּפַעֲמַיִם ׳חֲלוּלִים חֲלוּלִים׳. לָמָּה? הַלְּבוּשׁ מְבָאֵר: הָעִקָּר הוּא הַפֶּה, שֶׁמַּכְנִיס אֹכֶל, וְהַפְּתָחִים שֶׁדַּרְכָּם הַגּוּף מְנַקֶּה אֶת עַצְמוֹ. אֲבָל אוֹמְרִים פַּעֲמַיִם כְּדֵי לְכַלֵּל גַּם אֶת שְׁאָר הַנְּקָבִים, כְּמוֹ הָאַף וְהָאָזְנַיִם, וְכֵן אֶת שְׁאָר הָאֵיבָרִים הַחֲלוּלִים.

וְיֵשׁ בֵּאוּר יָפֶה נוֹסָף: כָּל הַגּוּף מָלֵא נְקָבִים דַּקִּים שֶׁאִי אֶפְשָׁר לִרְאוֹת, וְגַם עֲלֵיהֶם מוֹדִים.

פַּעֲמַיִם – כְּדֵי לְהוֹדוֹת גַּם עַל כָּל שְׁאָר הַפְּתָחִים וְהָאֵיבָרִים.

📖 נִפְלְאוֹת הַבְּרִיאָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בֵּית יוֹסֵף וְטוּר אוֹרַח חַיִּים ו׳

רָמָה ב׳ – מַה אוֹמְרִים הַמְּפָרְשִׁים

🔤 חֲלָלִים אוֹ חֲלוּלִים?

בַּגְּמָרָא כָּתוּב ׳חֲלָלִים׳, וְכָךְ גּוֹרֵס רַשִׁ״י. אֲבָל הָרִי״ף, הָרַמְבַּ״ם, הַטּוּר וְהַשֻּׁלְחָן עָרוּךְ גּוֹרְסִים ׳חֲלוּלִים׳. מַדּוּעַ?

סִבָּה אַחַת: הַטּוּר מוֹצֵא בְּ׳חֲלוּלִים חֲלוּלִים׳ רֶמֶז לְ-248 הָאֵיבָרִים. סִבָּה שְׁנִיָּה: ׳חֲלָלִים׳ הוּא גַּם לָשׁוֹן שֶׁל עֶצֶב, וְהַבְּרָכָה הִיא שִׁיר שֶׁל חַיִּים.

וְיֵשׁ שְׁאֵלָה מַצְחִיקָה: אֵיךְ אֶפְשָׁר ׳לִבְרֹא חָלָל׳? חָלָל הוּא ׳כְּלוּם׳, כְּמוֹ הַחוֹר שֶׁבְּתוֹךְ הַבֵּיגֶלֶה! כְּמוֹ בַּסִּפּוּר עַל אַנְשֵׁי חֶלֶם שֶׁרָצוּ לִקְנוֹת רַק אֶת הַחוֹר. הַתְּשׁוּבָה: מִי שֶׁבּוֹרֵא אֵבֶר, בּוֹרֵא גַּם אֶת הַמָּקוֹם הַפָּנוּי שֶׁבּוֹ.

חֲלוּלִים: רֶמֶז ל-248, וְלָשׁוֹן שֶׁל חַיִּים.

🌿 לָמָּה לִדְאֹג?

הָרַב יוֹנָתָן אַיְיבֶּשִׁיץ, בַּסֵּפֶר ׳יַעְרוֹת דְּבַשׁ׳, מְלַמֵּד: מִי שֶׁמִּתְבּוֹנֵן בַּגּוּף – בַּגִּידִים, בָּעֲצָמוֹת וּבָעוֹרְקִים, זֶה רַךְ וְזֶה קָשֶׁה, הַכֹּל בְּמִזְגוֹ הַמְּמֻצָּע – מַכִּיר אֶת אַהֲבַת ה׳ וַחֲכְמָתוֹ.

אָז לִבּוֹ מִתְנַקֶּה, וְהוּא אוֹמֵר: הֲרֵי ה׳ הֵכִין לָאָדָם אֶת כָּל צָרְכּוֹ – לָמָּה לִדְאֹג?

כְּבָר לָמַדְנוּ שֶׁהִתְבּוֹנְנוּת מְבִיאָה אַהֲבָה. כָּאן רוֹאִים שֶׁהִיא מְבִיאָה גַּם שֶׁקֶט בַּלֵּב.

הִתְבּוֹנְנוּת בַּגּוּף מְבִיאָה אַהֲבָה, שֶׁקֶט וּבִטָּחוֹן.

מָה לוֹקְחִים מִרָמָה ב׳?

בְּרָכַת ׳אֲשֶׁר יָצַר׳ הִיא הַזְמָנָה לְהִסְתַּכֵּל עַל הַגּוּף שֶׁלָּנוּ בְּעֵינַיִם חֲדָשׁוֹת. מִי שֶׁמְּבִין שֶׁהַכֹּל בָּנוּי בְּחָכְמָה, מַרְגִּישׁ שֶׁה׳ אוֹהֵב אוֹתוֹ וְדוֹאֵג לוֹ מִן הַרֶּגַע הָרִאשׁוֹן.

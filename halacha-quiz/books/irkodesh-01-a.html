בְּהַר ה׳ יֵרָאֶה

🗺️ שַׁעַר הַשֵּׁם · רָמָה א׳

יִרְאֶה וְשָׁלֵם

רָמָה א׳ – שֵׁם מִשְּׁנֵי שֵׁמוֹת

לָמָּה לִירוּשָׁלַיִם יֵשׁ שֵׁם שֶׁנּוֹלַד מִשְּׁנֵי שֵׁמוֹת, וּמָה הוּא מְלַמֵּד אוֹתָנוּ עַל יִרְאָה וְעַל שָׁלוֹם

📖 יְרוּשָׁלַיִם – עִיר הַקֹּדֶשׁ · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
⛰️

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ שֶׁיֵּשׁ בָּעוֹלָם מָקוֹם מְיֻחָד שֶׁה׳ בָּחַר בּוֹ 💛

📖 יְרוּשָׁלַיִם – עִיר הַקֹּדֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּרֵאשִׁית כ״ב י״ד

רָמָה א׳ – שֵׁם מִשְּׁנֵי שֵׁמוֹת

⛰️ אַבְרָהָם עַל הָהָר

אַבְרָהָם אָבִינוּ עָלָה עַל הַר בְּאֶרֶץ מוֹרִיָּה, וְשָׁם עָבַר רֶגַע גָּדוֹל בְּחַיָּיו.

כְּשֶׁהַכֹּל הִסְתַּיֵּם, הוּא עָמַד וְנָתַן שֵׁם לַמָּקוֹם. הַשֵּׁם הָיָה: ׳ה׳ יִרְאֶה׳.

אַחַר כָּךְ הוּא הוֹסִיף שֶׁבְּכָל הַדּוֹרוֹת יֹאמְרוּ: ׳בְּהַר ה׳ יֵרָאֶה׳.

אַבְרָהָם קָרָא לַמָּקוֹם: ה׳ יִרְאֶה.

1
🔭

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַרְגִּישׁ שֶׁה׳ מוֹצֵא לְכָל דָּבָר חָשׁוּב אֶת הַמָּקוֹם הַנָּכוֹן 💛

📖 יְרוּשָׁלַיִם – עִיר הַקֹּדֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַשִׁ״י בְּרֵאשִׁית כ״ב י״ד

רָמָה א׳ – שֵׁם מִשְּׁנֵי שֵׁמוֹת

🔭 מָה פֵּרוּשׁ ׳יִרְאֶה׳?

רַשִׁ״י מְבָאֵר: ה׳ יִבְחַר אֶת הַמָּקוֹם הַזֶּה וְיַרְאֶה אוֹתוֹ לְעַמּוֹ.

בַּמָּקוֹם הַזֶּה תִּשְׁרֶה הַשְּׁכִינָה, וְיַקְרִיבוּ בּוֹ קָרְבָּנוֹת.

וְכָל הַדּוֹרוֹת הַבָּאִים יֹאמְרוּ: בְּהַר הַזֶּה ה׳ מִתְגַּלֶּה לְעַמּוֹ.

ה׳ יִבְחַר וְיַרְאֶה אֶת הַמָּקוֹם הַקָּדוֹשׁ.

2
👑

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹהֵב שֶׁבַּשֵּׁם שֶׁל הָעִיר שֶׁלָּנוּ יֵשׁ גַּם מִלָּה שֶׁל שָׁלוֹם 💛

📖 יְרוּשָׁלַיִם – עִיר הַקֹּדֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּרֵאשִׁית י״ד י״ח

רָמָה א׳ – שֵׁם מִשְּׁנֵי שֵׁמוֹת

👑 מֶלֶךְ שָׁלֵם

בִּימֵי אַבְרָהָם הָיָה מֶלֶךְ שְׁמוֹ מַלְכִּי צֶדֶק, מֶלֶךְ שָׁלֵם.

חֲכָמִים מְלַמְּדִים שֶׁזֶּה שֵׁם בֶּן נֹחַ, צַדִּיק גָּדוֹל.

הוּא קָרָא לַמָּקוֹם ׳שָׁלֵם׳, מִלָּה שֶׁקְּרוֹבָה לַמִּלָּה שָׁלוֹם.

שֵׁם קָרָא לַמָּקוֹם: שָׁלֵם.

3
🤝

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד מִזֶּה שֶׁאֶפְשָׁר לְכַבֵּד שְׁנֵי אֲנָשִׁים בְּבַת אַחַת 💛

📖 יְרוּשָׁלַיִם – עִיר הַקֹּדֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הַמִּדְרָשׁ

רָמָה א׳ – שֵׁם מִשְּׁנֵי שֵׁמוֹת

🤝 שְׁנֵי צַדִּיקִים וּשְׁנֵי שֵׁמוֹת

אַבְרָהָם אָמַר ׳יִרְאֶה׳, וְשֵׁם אָמַר ׳שָׁלֵם׳.

הַמִּדְרָשׁ מְסַפֵּר: אִם ה׳ יִקְרָא לַמָּקוֹם ׳יִרְאֶה׳, שֵׁם יִצְטַעֵר. וְאִם יִקְרָא ׳שָׁלֵם׳, אַבְרָהָם יִצְטַעֵר.

מָה עָשָׂה ה׳? חִבֵּר אֶת שְׁנֵי הַשֵּׁמוֹת וְקָרָא: יְרוּשָׁלַיִם.

ה׳ חִבֵּר אֶת שְׁנֵי הַשֵּׁמוֹת לְשֵׁם אֶחָד.

4
🔗

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי אוֹמֵר יְרוּשָׁלַיִם, אֲנִי מַרְגִּישׁ שֶׁאֲנִי אוֹמֵר שְׁנֵי שֵׁמוֹת יְקָרִים 💛

📖 יְרוּשָׁלַיִם – עִיר הַקֹּדֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרֹא״שׁ בְּרֵאשִׁית כ״ב י״ד

רָמָה א׳ – שֵׁם מִשְּׁנֵי שֵׁמוֹת

🔗 שֵׁם אֶחָד מִשְּׁנַיִם

הַמִּלָּה יְרוּשָׁלַיִם בְּנוּיָה מִשְּׁנֵי חֲלָקִים: יִרְאֶה וְשָׁלֵם.

לְפִי הָרֹא״שׁ, דָּוִד הַמֶּלֶךְ קָרָא לָעִיר יְרוּשָׁלַיִם, וְלָקַח אֶת הַלָּשׁוֹן שֶׁל שְׁנֵיהֶם.

כָּךְ הַשֵּׁם מַזְכִּיר לָנוּ בְּכָל פַּעַם שֶׁנֹּאמַר אוֹתוֹ אֶת אַבְרָהָם וְאֶת שֵׁם.

יְרוּשָׁלַיִם הוּא שֵׁם מְחֻבָּר: יִרְאֶה וְשָׁלֵם.

5
👁️

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי מִתְפַּלֵּל בְּשֶׁקֶט וּבְכָבוֹד, אֲנִי מַרְגִּישׁ שֶׁה׳ קָרוֹב אֵלַי 💛

📖 יְרוּשָׁלַיִם – עִיר הַקֹּדֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תּוֹסָפוֹת

רָמָה א׳ – שֵׁם מִשְּׁנֵי שֵׁמוֹת

👁️ יִרְאָה – בֵּין אָדָם לַמָּקוֹם

הַחֵלֶק הָרִאשׁוֹן הוּא יִרְאָה: לִזְכֹּר שֶׁה׳ קָרוֹב אֵלֵינוּ, וְלִנְהֹג בְּכָבוֹד.

זֶה מָה שֶׁיֵּשׁ בֵּין אָדָם לַמָּקוֹם, כְּלוֹמַר בֵּינִי לְבֵין ה׳.

הַתּוֹסָפוֹת מְלַמְּדוֹת שֶׁמִּי שֶׁהָיָה רוֹאֶה אֶת הַמָּקוֹם הַקָּדוֹשׁ הָיָה מַרְגִּישׁ יִרְאַת שָׁמַיִם.

יִרְאָה הִיא הַקֶּשֶׁר שֶׁבֵּין הָאָדָם לְבֵין ה׳.

6
🕊️

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ כְּשֶׁאֲנִי מַצְלִיחַ לַעֲשׂוֹת שָׁלוֹם בֵּין שְׁנֵי חֲבֵרִים 💛

📖 יְרוּשָׁלַיִם – עִיר הַקֹּדֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שֵׁם הָעִיר

רָמָה א׳ – שֵׁם מִשְּׁנֵי שֵׁמוֹת

🕊️ שָׁלוֹם – בֵּין אָדָם לַחֲבֵרוֹ

הַחֵלֶק הַשֵּׁנִי הוּא שָׁלוֹם: לִהְיוֹת בְּשָׁלוֹם עִם כָּל אֶחָד.

זֶה מָה שֶׁיֵּשׁ בֵּין אָדָם לַחֲבֵרוֹ.

בְּמָקוֹם שֶׁיֵּשׁ בּוֹ שָׁלוֹם, ה׳ שָׂמֵחַ לָשֶׁבֶת.

שָׁלוֹם הוּא מָה שֶׁיֵּשׁ בֵּין אָדָם לַחֲבֵרוֹ.

7
🏗️

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת גַּם הָאִישׁ שֶׁמַּבִּיט לְמַעְלָה וְגַם הָאִישׁ שֶׁעוֹזֵר לַחֲבֵרוֹ 💛

📖 יְרוּשָׁלַיִם – עִיר הַקֹּדֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מָשָׁל

רָמָה א׳ – שֵׁם מִשְּׁנֵי שֵׁמוֹת

🏗️ שְׁנֵי אֲנָשִׁים וְחוֹמָה

דִּמְיְנוּ שְׁנֵי אֲנָשִׁים שֶׁבּוֹנִים חוֹמָה.

הָאֶחָד מַבִּיט לְמַעְלָה וְלֹא שׁוֹכֵחַ מִי נָתַן לוֹ אֶת הַכֹּחַ: זוֹ יִרְאָה.

הַשֵּׁנִי אוֹחֵז בְּיָדוֹ שֶׁל חֲבֵרוֹ וְעוֹזֵר לוֹ לְהַעֲלוֹת אֶת הָאֲבָנִים: זֶה שָׁלוֹם.

בְּלִי שְׁנֵיהֶם הַחוֹמָה לֹא תַּעֲמֹד.

חוֹמָה חֲזָקָה צְרִיכָה גַּם יִרְאָה וְגַם שָׁלוֹם.

8
💡

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹחֵר הַיּוֹם לְהוֹסִיף קְצָת יִרְאָה וּקְצָת שָׁלוֹם 💛

📖 יְרוּשָׁלַיִם – עִיר הַקֹּדֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, יְרוּשָׁלַיִם

רָמָה א׳ – שֵׁם מִשְּׁנֵי שֵׁמוֹת

💡 מָה לוֹקְחִים מִן הַשֵּׁם

הַשֵּׁם יְרוּשָׁלַיִם הוּא שִׁעוּר קָטָן.

מִי שֶׁרוֹצֶה לִבְנוֹת בַּיִת טוֹב, כִּתָּה טוֹבָה אוֹ עִיר טוֹבָה, צָרִיךְ שְׁנֵי דְּבָרִים: לִזְכֹּר אֶת ה׳, וְלִהְיוֹת בְּשָׁלוֹם עִם כֻּלָּם.

אֲנִי יָכוֹל לְהַתְחִיל הַיּוֹם, בַּכִּתָּה שֶׁלִּי.

עִיר טוֹבָה בְּנוּיָה מִיִּרְאָה וּמִשָּׁלוֹם.

9
💡

רָמָה א׳ – שֵׁם מִשְּׁנֵי שֵׁמוֹת

מָה לוֹקְחִים מִרָמָה א׳?

הַשֵּׁם יְרוּשָׁלַיִם מְלַמֵּד אוֹתִי שֶׁה׳ אוֹהֵב כְּשֶׁשְּׁנֵי אֲנָשִׁים טוֹבִים מְחַבְּרִים בֵּינֵיהֶם. כָּל פַּעַם שֶׁאֲנִי אוֹמֵר אֶת שֵׁם הָעִיר, אֲנִי אוֹמֵר תּוֹדָה עַל הַמָּקוֹם הַזֶּה, וְעַל הָאַהֲבָה שֶׁבֵּין בְּנֵי אָדָם.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי הִרְגַּשְׁתִּי שֶׁה׳ קָרוֹב אֵלַי, וְאֵיפֹה זֶה קָרָה?
  • עִם מִי בַּכִּתָּה אוֹ בַּבַּיִת קָשֶׁה לִי לִהְיוֹת בְּשָׁלוֹם, וּמָה אֶפְשָׁר לַעֲשׂוֹת?
  • הַיּוֹם אֲחַבֵּר עִם חֲבֵרִים שֵׁם לַכִּתָּה מִשְּׁתֵּי מִדּוֹת שֶׁאָנוּ צְרִיכִים, וְנֶחְלֹט עַל מַעֲשֶׂה אֶחָד שֶׁמְּקַיֵּם אוֹתוֹ.

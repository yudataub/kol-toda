🧥 מְעִיל לֶחָבֵר

יוֹם קַר בַּחוּץ. יֶלֶד רוֹאֶה חָבֵר שֶׁשָּׁכַח אֶת הַמְּעִיל וְרוֹעֵד מִקֹּר, וּמוֹשִׁיט לוֹ אֶת הַצָּעִיף שֶׁלּוֹ.

שְׁאֵלָה גְּדוֹלָה שָׁאֲלוּ חֲכָמִים: אֵיךְ אֶפְשָׁר לָלֶכֶת אַחֲרֵי ה׳? הֲרֵי הוּא גָּדוֹל וְנֶעֱלָם מֵאִתָּנוּ.

הַתְּשׁוּבָה: הוֹלְכִים בְּדַרְכֵי ה׳ כְּשֶׁעוֹשִׂים מַעֲשֵׂי חֶסֶד כָּמוֹהוּ.

הוֹלְכִים אַחֲרֵי ה׳ כְּשֶׁעוֹשִׂים חֶסֶד כָּמוֹהוּ.

רָמָה א׳ – לָלֶכֶת אַחֲרֵי ה׳

🚶 אַחֲרֵי ה׳ תֵּלֵכוּ

בַּתּוֹרָה כָּתוּב: ׳אַחֲרֵי ה׳ אֱלֹהֵיכֶם תֵּלֵכוּ׳. אֲבָל אֵיךְ אָדָם יָכוֹל לָלֶכֶת אַחֲרָיו? הֲרֵי כָּתוּב גַּם: ׳כִּי ה׳ אֱלֹהֶיךָ אֵשׁ אוֹכְלָה הוּא׳.

הַגְּמָרָא עוֹנָה: לָלֶכֶת אַחַר מִדּוֹתָיו שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא, כְּלוֹמַר לַעֲשׂוֹת מַה שֶּׁהוּא עוֹשֶׂה.

וְהִיא נוֹתֶנֶת אַרְבָּעָה דּוּגְמָאוֹת: הוּא מַלְבִּישׁ, מְבַקֵּר חוֹלִים, מְנַחֵם אֲבֵלִים וְקוֹבֵר מֵתִים.

לָלֶכֶת אַחֲרֵי ה׳ = לָלֶכֶת אַחַר מִדּוֹתָיו.

רָמָה א׳ – לָלֶכֶת אַחֲרֵי ה׳

👕 הַחֶסֶד הָרִאשׁוֹן

הַחֶסֶד הָרִאשׁוֹן בַּתּוֹרָה: ׳וַיַּעַשׂ ה׳ אֱלֹהִים לְאָדָם וּלְאִשְׁתּוֹ כׇּתְנוֹת עוֹר וַיַּלְבִּשֵׁם׳. ה׳ הִלְבִּישׁ אֶת מִי שֶׁהָיָה עָרוֹם.

מִכָּאן לָמְדוּ חֲכָמִים: גַּם אָנוּ מַלְבִּישִׁים מִי שֶׁצָּרִיךְ, כְּמוֹ מְעִיל לְחָבֵר שֶׁקַּר לוֹ.

זֶה הַחֶסֶד שֶׁהַתּוֹרָה בּוֹחֶרֶת לִפְתֹּחַ בּוֹ.

הַתּוֹרָה פּוֹתַחַת בְּחֶסֶד: ה׳ הִלְבִּישׁ אֶת אָדָם.

📖 חֶסֶד – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּרֵאשִׁית י״ח א׳ · סוֹטָה י״ד א׳

🛏️ בִּקּוּר חוֹלִים

בְּאֵלֹנֵי מַמְרֵא כָּתוּב: ׳וַיֵּרָא אֵלָיו ה׳ בְּאֵלֹנֵי מַמְרֵא׳. חֲכָמִים לָמְדוּ מִכָּאן שֶׁה׳ בִּקֵּר חוֹלֶה, וְגַם אָנוּ מְבַקְּרִים.

מִי שֶׁחָבֵר שֶׁלּוֹ חוֹלֶה, יָכוֹל לָבוֹא, לְהִתְקַשֵּׁר אוֹ לְצַיֵּר לוֹ צִיּוּר.

וְיֵשׁ כְּלָל: מְבַקְּרִים בַּשָּׁעָה שֶׁנּוֹחַ לֶחָבֵר הַחוֹלֶה, וְלֹא בַּשָּׁעָה שֶׁנּוֹחַ לִי.

מְבַקְּרִים חוֹלֶה בַּשָּׁעָה שֶׁנּוֹחַ לוֹ.

רָמָה א׳ – לָלֶכֶת אַחֲרֵי ה׳

🤗 נִחוּם אֲבֵלִים

אַחֲרֵי מוֹת אַבְרָהָם כָּתוּב: ׳וַיְבָרֶךְ אֱלֹהִים אֶת יִצְחָק בְּנוֹ׳. חֲכָמִים לָמְדוּ שֶׁה׳ נִחֵם אֶת יִצְחָק שֶׁהָיָה אָבֵל.

כְּשֶׁחָבֵר עָצוּב מְאֹד כִּי קָרָה לוֹ דָּבָר קָשֶׁה, אֶפְשָׁר לָשֶׁבֶת לְיָדוֹ, לְהַקְשִׁיב וְלוֹמַר מִלָּה טוֹבָה.

זֶה הַחֶסֶד: לֹא לְהַשְׁאִיר אָדָם לְבַדּוֹ בַּצַּעַר.

מְנַחֲמִים: יוֹשְׁבִים לְיַד מִי שֶׁעָצוּב וּמַקְשִׁיבִים.

📖 חֶסֶד – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דְּבָרִים ל״ד ו׳ · רַשִׁ״י

רָמָה א׳ – לָלֶכֶת אַחֲרֵי ה׳

⛰️ הַחֶסֶד הָאַחֲרוֹן

בְּסוֹף הַתּוֹרָה כָּתוּב: ׳וַיִּקְבֹּר אֹתוֹ בַּגַּי׳. רַשִׁ״י מְסַבֵּר שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא בִּכְבוֹדוֹ וּבְעַצְמוֹ קָבַר אֶת מֹשֶׁה רַבֵּנוּ.

וְלֹא יָדַע אִישׁ אֶת מְקוֹם הַקֶּבֶר, וּבְכָל זֹאת ה׳ עָשָׂה אֶת הַחֶסֶד.

הַחֶסֶד הָאַחֲרוֹן בַּתּוֹרָה הוּא קְבוּרָה, וְהוּא נִקְרָא חֶסֶד שֶׁל אֱמֶת.

הַתּוֹרָה מְסַיֶּמֶת בְּחֶסֶד: ה׳ קָבַר אֶת מֹשֶׁה.

רָמָה א׳ – לָלֶכֶת אַחֲרֵי ה׳

🔁 תְּחִלָּה וְסוֹף

רַבִּי שִׂמְלַאי דָּרַשׁ: הַתּוֹרָה תְּחִלָּתָהּ חֶסֶד וְסוֹפָהּ חֶסֶד. הִיא פּוֹתַחַת בַּהַלְבָּשָׁה וּמְסַיֶּמֶת בִּקְבוּרָה.

הַמַּהֲרַ״ל כָּתַב שֶׁהַהַתְחָלָה וְהַסּוֹף הֵם עִקַּר הַדָּבָר. כְּשֶׁסֵּפֶר נִפְתָּח וְנִגְמָר בְּחֶסֶד, כָּל מַה שֶּׁבֵּינֵיהֶם מֵאִיר.

כָּךְ גַּם הַיּוֹם שֶׁלִּי: כְּדַאי שֶׁיִּהְיֶה מָלֵא בְּחֶסֶד מֵהַבֹּקֶר עַד הָעֶרֶב.

תּוֹרָה: מַתְחִילָה בְּחֶסֶד וְנִגְמֶרֶת בְּחֶסֶד.

📖 חֶסֶד – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב · סוֹטָה י״ד א׳

רָמָה א׳ – לָלֶכֶת אַחֲרֵי ה׳

💛 חֶסֶד גַּם לְמִי שֶׁטָּעָה

אָדָם הָרִאשׁוֹן בְּדִיּוּק חָטָא, וּבְכָל זֹאת ה׳ הִלְבִּישׁ אוֹתוֹ. לָמָּה דַּוְקָא הַלְבָּשָׁה נִקְרֵאת ׳הַחֶסֶד הָרִאשׁוֹן׳? כִּי חֶסֶד אֱמִתִּי נוֹתְנִים גַּם לְמִי שֶׁלֹּא מַגִּיעַ לוֹ.

ה׳ לֹא חִכָּה שֶׁאָדָם יְתַקֵּן אֶת הַכֹּל. הוּא בָּא וְנָתַן.

כְּשֶׁחָבֵר טָעָה וְגַם הוּא צָרִיךְ עֶזְרָה, אֶפְשָׁר לַעֲזֹר גַּם לוֹ.

חֶסֶד אֱמִתִּי נוֹתְנִים גַּם לְמִי שֶׁטָּעָה.

מָה לוֹקְחִים מִרָמָה א׳?

ה׳ מַתְחִיל אֶת הַתּוֹרָה בְּחֶסֶד וּמְסַיֵּם אוֹתָהּ בְּחֶסֶד. כְּשֶׁאֲנִי מֵיטִיב לַאֲחֵרִים, אֲנִי הוֹלֵךְ בְּדַרְכָּיו וּמִתְקָרֵב אֵלָיו, וְאַהֲבָתִי אֵלָיו גְּדֵלָה.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה הַחֶסֶד שֶׁעָשִׂיתִי הַשָּׁבוּעַ בְּדִיּוּק כְּמוֹ ה׳ – לְהַלְבִּישׁ, לְבַקֵּר אוֹ לְנַחֵם?

  • אֶבְחַר מַדְרֵגָה אַחַת שֶׁל חֶסֶד, אֶעֱשֶׂה אוֹתָהּ בַּשָּׁעָה שֶׁנּוֹחַ לַמְקַבֵּל, וְאֶרְשֹׁם מִי הָיָה וּמָה הִרְגַּשְׁתִּי.

🪨 אַבְרָהָם קוֹנֶה

אַבְרָהָם אָבִינוּ רָצָה מָקוֹם קָבוּעַ לְמִשְׁפַּחְתּוֹ: מְעָרַת הַמַּכְפֵּלָה בְּחֶבְרוֹן.

הוּא לֹא לָקַח אוֹתָהּ בְּחִנָּם. הוּא שָׁקַל לְעֶפְרוֹן אַרְבַּע מֵאוֹת שֶׁקֶל כֶּסֶף, כֶּסֶף מָלֵא, וְכָךְ הַמְּעָרָה נַעֲשֵׂית שֶׁלּוֹ.

אַבְרָהָם קָנָה אֶת מְעָרַת הַמַּכְפֵּלָה בְּכֶסֶף מָלֵא.

רָמָה א׳ – מִקְנֶה, גְּזֵלָה וַחֲזָרָה

💰 לָמָּה קָנָה?

לָמָּה אַבְרָהָם קָנָה וְלֹא קִבֵּל מַתָּנָה? אוֹמְרִים: כְּדֵי שֶׁלֹּא יֹאמְרוּ שֶׁהַמָּקוֹם אֵינוֹ שֶׁלּוֹ.

וְיֵשׁ עוֹד סִבָּה: דָּבָר שֶׁמַּשְׁקִיעִים בּוֹ, נִשְׁאָר. וּמַה שֶּׁבָּא בְּחִנָּם, שָׁוֶה חִנָּם.

מַה שֶּׁמַּשְׁקִיעִים בּוֹ נִשְׁאָר.

📖 אַהֲבַת אֶרֶץ יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הַמְּעָרָה שֶׁחָזְרָה

רָמָה א׳ – מִקְנֶה, גְּזֵלָה וַחֲזָרָה

❤️ כֶּסֶף מִלְּשׁוֹן כִּסּוּפִים

כְּשֶׁאָדָם מְשַׁלֵּם עַל דָּבָר, הוּא אוֹהֵב אוֹתוֹ. וְאוֹמְרִים בַּשִּׁעוּר: כֶּסֶף הוּא מִלְּשׁוֹן כִּסּוּפִים.

אֶת אֶרֶץ יִשְׂרָאֵל לֹא רַק יוֹרְשִׁים וְגָרִים בָּהּ, אֶלָּא גַּם אוֹהֲבִים אוֹתָהּ מַמָּשׁ. כֻּלָּנוּ מַכִּירִים אֶת הַתְּמוּנָה הַמְּפֻרְסֶמֶת שֶׁל עוֹלֵי תֵּימָן שֶׁיָּרְדוּ מִן הַמָּטוֹס וְנִשְּׁקוּ אֶת הָאֲדָמָה.

כֶּסֶף = כִּסּוּפִים: מִי שֶׁמְּשַׁלֵּם, אוֹהֵב.

📖 אַהֲבַת אֶרֶץ יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הַמְּעָרָה שֶׁחָזְרָה

רָמָה א׳ – מִקְנֶה, גְּזֵלָה וַחֲזָרָה

🐄 הָעֵגֶל שֶׁבָּרַח

הַמִּדְרָשׁ מְסַפֵּר: כְּשֶׁאַבְרָהָם קִבֵּל אוֹרְחִים בְּאֵלוֹנֵי מַמְרֵא, בֶּן בָּקָר בָּרַח לוֹ. אַבְרָהָם רָץ אַחֲרָיו עַד שֶׁהִגִּיעַ אֶל הַמְּעָרָה.

שָׁם הֵרִיחַ רֵיחַ גַּן עֵדֶן. כָּךְ הִתְחִילָה הָאַהֲבָה שֶׁלּוֹ לַמָּקוֹם הַזֶּה.

אַבְרָהָם הִכִּיר אֶת הַמְּעָרָה כְּשֶׁרָץ אַחֲרֵי בֶּן בָּקָר.

רָמָה א׳ – מִקְנֶה, גְּזֵלָה וַחֲזָרָה

🚪 שֶׁבַע מֵאוֹת שָׁנָה בַּחוּץ

בְּמֶשֶׁךְ כְּשֶׁבַע מֵאוֹת שָׁנָה יְהוּדִי לֹא יָכוֹל הָיָה לְהִכָּנֵס אֶל הַמְּעָרָה. הָיָה מֻתָּר לַעֲמֹד בַּחוּץ וּלְהִתְפַּלֵּל מִבַּחוּץ, אֲבָל לֹא לְהִכָּנֵס.

כְּאִלּוּ הַמְּעָרָה שֶׁאַבְרָהָם קָנָה הָיְתָה גְּזוּלָה מִמֶּנּוּ.

שֶׁבַע מֵאוֹת שָׁנָה עָמְדוּ בַּחוּץ וְלֹא נִכְנְסוּ.

📖 אַהֲבַת אֶרֶץ יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הַמְּעָרָה שֶׁחָזְרָה

רָמָה א׳ – מִקְנֶה, גְּזֵלָה וַחֲזָרָה

🪜 הַמַּדְרֵגָה הַשְּׁבִיעִית

מִחוּץ לַמִּבְנֶה הָיוּ מַדְרֵגוֹת שֶׁעָלוּ לְכִוּוּן הַמְּעָרָה, וִיהוּדִים הֻתְּרוּ לַעֲלוֹת רַק עַד הַמַּדְרֵגָה הַשְּׁבִיעִית.

בִּשְׁנַת תרצ״ה בָּא לְבַקֵּר הָאַדְמוֹ״ר מִגּוּר, בַּעַל ׳אַמְרֵי אֱמֶת׳. הוּא לֹא הִשְׁלִים עִם זֶה, וְהִמְשִׁיךְ לַעֲלוֹת עַד הַמַּדְרֵגָה הָאַחַת עֶשְׂרֵה, וְשָׁם הִתְפַּלֵּל.

הָאַדְמוֹ״ר מִגּוּר עָלָה מֵעֵבֶר לַמַּדְרֵגָה הַשְּׁבִיעִית.

📖 אַהֲבַת אֶרֶץ יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הַמְּעָרָה שֶׁחָזְרָה

🕳️ פֶּתַח קָטָן בַּקִּיר

סוֹפֵר צָרְפָתִי שֶׁבִּקֵּר בְּחֶבְרוֹן בִּשְׁנַת 1894 כָּתַב מָה רָאָה. בַּקִּיר, קָרוֹב לָאָרֶץ, הָיָה פֶּתַח קָטָן. יְהוּדִים הֻתְּרוּ לָשִׂים בּוֹ אֶת הָרֹאשׁ, כְּשֶׁהֵם זוֹחֲלִים, וּלְנַשֵּׁק אֶת הָאֲבָנִים.

בָּעֶרֶב בָּאוּ עוֹלֵי רֶגֶל עֲנִיִּים, עִרְבְּבוּ דִּמְעוֹת בִּתְפִלָּתָם וְנִסּוּ לְהַצְמִיד שְׂפָתַיִם אֶל הַמָּקוֹם.

גַּם דֶּרֶךְ פֶּתַח קָטָן, הַיְּהוּדִים בָּאוּ לְנַשֵּׁק אֶת הָאֲבָנִים.

🎉 הַמְּעָרָה חָזְרָה

בִּשְׁנַת תשכ״ז, אַחֲרֵי מִלְחֶמֶת שֵׁשֶׁת הַיָּמִים, הַמְּעָרָה חָזְרָה לְיַד יִשְׂרָאֵל.

הַיּוֹם אֶפְשָׁר לְהִכָּנֵס אֵלֶיהָ כִּמְעַט מָתַי שֶׁרוֹצִים. מַה שֶּׁנִּגְזַל מֵאַבְרָהָם, וְהוּא קָנָה בְּכֶסֶף מָלֵא, חָזַר אֵלֵינוּ.

הַמְּעָרָה שֶׁאַבְרָהָם קָנָה חָזְרָה לְעַם יִשְׂרָאֵל.

מָה לוֹקְחִים מִרָמָה א׳?

אַבְרָהָם אָבִינוּ קָנָה אֶת הַמְּעָרָה בְּכֶסֶף מָלֵא, וְדּוֹרוֹת שְׁלֵמִים כָּסְפוּ אֵלֶיהָ. כְּשֶׁאֲנִי נִזְכָּר כַּמָּה זֶה הָיָה קָשֶׁה, אֲנִי מוֹדֶה לַה׳ עוֹד יוֹתֵר עַל הַטּוֹבָה, וְכָךְ אַהֲבָתִי אֵלָיו גְּדֵלָה.

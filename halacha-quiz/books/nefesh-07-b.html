רָמָה ב׳ – שְׁתֵּי דְּרָכִים

⚖️ שְׁנֵי גְּדוֹלִים, שְׁתֵּי דְּרָכִים

שַׁמַּאי וְהִלֵּל הָיוּ שְׁנֵי גְּדוֹלֵי הַדּוֹר. שַׁמַּאי הָיָה קַפְּדָן – מַחְמִיר וּמְדַיֵּק – וְלֹא מִקְּרִירוּת, אֶלָּא מֵאַהֲבָה עֲצוּמָה לַתּוֹרָה, שֶׁקָּשָׁה לוֹ לְוַתֵּר עַל קִיּוּמָהּ.

הִלֵּל הָיָה עַנְוְתָן וְסַבְלָן. וְאַף עַל פִּי כֵן, שְׁנֵיהֶם הָיוּ צַדִּיקִים גְּדוֹלִים שֶׁאָהֲבוּ אֶת ה׳ וְאֶת הַתּוֹרָה.

וְגַם שַׁמַּאי, כְּמוֹ שֶׁלָּמַדְנוּ בַּמִּשְׁנָה, הָיָה אוֹמֵר: ׳הֱוֵי מְקַבֵּל אֶת כָּל הָאָדָם בְּסֵבֶר פָּנִים יָפוֹת׳.

שְׁנֵיהֶם צַדִּיקִים; שַׁמַּאי קַפְּדָן מֵאַהֲבָה, וְהִלֵּל עַנְוְתָן.

🔤 הַגֵּר הָרִאשׁוֹן – מָה הַלֶּקַח?

הַגֵּר בִּקֵּשׁ רַק תּוֹרָה שֶׁבִּכְתָב. שַׁמַּאי רָאָה סַכָּנָה אֲמִתִּית: גֵּר כָּזֶה יְפָרֵשׁ אֶת הַתּוֹרָה לְפִי הֲבָנָתוֹ הָאִישִׁית, בְּלִי לָדַעַת אֶת הַמָּסֹרֶת, וְזֶה עָלוּל לַהֲבִיא לִפְרוּדִים בָּעָם. אֲבָל הוּא דָּחָה אוֹתוֹ בְּכַעַס, בְּלִי לְהַסְבִּיר.

הִלֵּל הֶעֱמִיד אוֹתוֹ בְּחֻוְיָה: לִמֵּד אוֹתִיּוֹת, וּלְמָחֳרָת הָפַךְ אֶת שְׁמוֹתֵיהֶן. כְּשֶׁהַגֵּר הִתְפַּלֵּא, שָׁאַל אוֹתוֹ: ׳אֵיךְ אַתָּה יוֹדֵעַ שֶׁהָאוֹתִיּוֹת מֵאֶתְמוֹל נְכוֹנוֹת? אַתָּה סוֹמֵךְ עָלַי׳.

אִם סוֹמְכִים עַל הַמָּסֹרֶת לְגַבֵּי שֵׁם הָאוֹתִיּוֹת, חַיָּבִים לִסְמֹךְ גַּם עַל הַתּוֹרָה שֶׁבְּעַל פֶּה שֶׁעָבְרָה מִדּוֹר לְדוֹר.

בְּלִי מָסֹרֶת אֵין לָדַעַת שׁוּם דָּבָר – הִלֵּל הֶרְאָה זֹאת בְּחֻוְיָה.

רָמָה ב׳ – שְׁתֵּי דְּרָכִים

🧱 בִּנְיָן וּכְלָל

הַגֵּר הַשֵּׁנִי בִּקֵּשׁ לִלְמֹד אֶת כָּל הַתּוֹרָה עַל רֶגֶל אַחַת. שַׁמַּאי כָּעַס, כִּי הוּא הֵבִין שֶׁהַגֵּר חוֹשֵׁב שֶׁתּוֹרָה אֶפְשָׁר לְקַבֵּל מַהֵר וּבְלִי עָמָל. הוּא הֶרְאָה לוֹ מַקֵּל שֶׁל בַּנַּאי: בִּנְיָן לֹא בּוֹנִים בְּחָמֵשׁ דַּקּוֹת, שׁוּרָה אַחַר שׁוּרָה, וְכָךְ גַּם הַתּוֹרָה – בִּנְיָן רוּחָנִי שֶׁבּוֹנִים שָׁנִים רַבּוֹת.

הִלֵּל לֹא שִׁנָּה אֶת הַבַּקָּשָׁה, אֶלָּא עָנָה לָהּ מַמָּשׁ: ׳מַה שֶּׁשָּׂנוּי עָלֶיךָ אַל תַּעֲשֶׂה לַחֲבֵרְךָ – זוֹ כָּל הַתּוֹרָה כֻּלָּהּ, וְאִידַּךְ פֵּרוּשָׁהּ הוּא, זִיל גְּמוֹר׳.

הַשְּׁאָר הוּא פֵּרוּשׁ וְצָרִיךְ לִלְמֹד אוֹתוֹ: הִלֵּל פָּתַח לַגֵּר שַׁעַר לְהַמְשִׁיךְ וְלִלְמֹד, בְּלִי הַרְגָּשָׁה שֶׁדּוֹחִים אוֹתוֹ.

הִלֵּל עָנָה לַבַּקָּשָׁה וְנָתַן כְּלָל, וּבִקֵּשׁ לִלְמֹד אֶת הַשְּׁאָר.

רָמָה ב׳ – שְׁתֵּי דְּרָכִים

👑 הַגֵּר הַשְּׁלִישִׁי – הַלִּמּוּד מְגַלֶּה

הַגֵּר הִתְלַהֵב מִבִּגְדֵי הַכֹּהֵן הַגָּדוֹל וְרָצָה לְהִתְגַּיֵּר כְּדֵי לִהְיוֹת כֹּהֵן גָּדוֹל. שַׁמַּאי רָאָה בַּבַּקָּשָׁה שִׁטְחִיּוּת, וְדָחָה אוֹתוֹ.

הִלֵּל לֹא בִּזָּה אוֹתוֹ עַל הַהִתְלַהֲבוּת הַתְּמִימָה. הוּא אָמַר: ׳בּוֹא נִלְמַד תּוֹרָה וְנִרְאֶה׳, וְלָמְדוּ יַחַד, יוֹם אַחַר יוֹם, בְּלִי לָמֵהַר.

כְּשֶׁהִגִּיעוּ בְּחֻמָּשׁ בַּמִּדְבָּר לַפָּסוּק ׳וְהַזָּר הַקָּרֵב יוּמָת׳, הֵבִין הַגֵּר בְּעַצְמוֹ שֶׁלֹּא יוּכַל לִהְיוֹת כֹּהֵן גָּדוֹל – וְלֹא כָּעַס וְלֹא הִתְיָאֵשׁ. הוּא אָמַר: ׳טָעַמְתִּי כַּמָּה הַתּוֹרָה יָפָה, וַאֲנִי רוֹצֶה לְהִשָּׁאֵר׳.

הִלֵּל לֹא שִׁקֵּר, אֶלָּא נָתַן לַגֵּר לִטְעֹם וּלְגַלּוֹת בְּעַצְמוֹ.

רָמָה ב׳ – שְׁתֵּי דְּרָכִים

🕊️ עַנְוְתָן כְּהִלֵּל – וְלֹא קַפְּדָן בְּחִנָּם

שְׁלוֹשָׁה גֵּרִים נִדְחוּ בְּכַעַס, וּבְסוֹף הִתְקַבְּלוּ בְּסַבְלָנוּת וְנִשְׁאֲרוּ. חֲכָמִים אָמְרוּ: ׳לְעוֹלָם יְהֵא אָדָם עַנְוְתָן כְּהִלֵּל וְלֹא קַפְּדָן כְּשַׁמַּאי׳.

שִׂימוּ לֵב לַנִּסּוּחַ: לֹא נֶאֱמַר ׳יִהְיֶה אָדָם הִלֵּל׳, אֶלָּא רַק ׳עַנְוְתָן כְּהִלֵּל׳. לֹא מְבַקְּשִׁים מִכָּל אָדָם לִהְיוֹת גָּדוֹל בַּתּוֹרָה, אֶלָּא לִרְכֹּשׁ אֶת הַסַּבְלָנוּת שֶׁלּוֹ כְּלַפֵּי אֲחֵרִים.

וְשַׁמַּאי לֹא הָיָה רַע: הוּא רָצָה לִשְׁמֹר עַל כְּבוֹדָהּ הָאֲמִתִּי שֶׁל הַתּוֹרָה, שֶׁלֹּא תֵּרָאֶה כְּמִשְׂחָק. אֲבָל הַדֶּרֶךְ הַסַּבְלָנִית מְקָרֶבֶת אֶת הָאֲנָשִׁים.

הַדֶּרֶךְ הַסַּבְלָנִית מְקָרֶבֶת; וְהַמִּדָּה שֶׁל הִלֵּל אֶפְשָׁר לִרְכֹּשׁ.

מָה לוֹקְחִים מִרָמָה ב׳?

הִלֵּל לִמֵּד שֶׁסַּבְלָנוּת וְסֵבֶר פָּנִים יָפוֹת פּוֹתְחִים לְבָבוֹת לַתּוֹרָה וְלָאֱמוּנָה. כָּל פַּעַם שֶׁאֲנִי מְקַבֵּל מִישֶׁהוּ בְּסַבְלָנוּת, אֲנִי מִתְקָרֵב אֶל ה׳ שֶׁמְּקַבֵּל אוֹתִי בְּאַהֲבָה.

שָׁלוֹשׁ הַשְׁגָּחוֹת בַּמִּזְמוֹרִים הָאַחֲרוֹנִים שֶׁל תְּהִלִּים: עָלַי, עַל עַם יִשְׂרָאֵל וְעַל כָּל הַבְּרִיאָה

🎯 לַחִידוֹן

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה לְמִי שֶׁעוֹזֵר לִי, וְיוֹדֵעַ שֶׁהַתְּשׁוּעָה בָּאָה מֵה׳ 💛

📖 פִּרְקֵי הַתְּפִלָּה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים קמ״ו, ג׳

רָמָה ב׳ – בִּטָּחוֹן וְהַשְׁגָּחָה

🆘 תְּשׁוּעָה וְעֶזְרָה – לֹא אוֹתוֹ דָּבָר

בַּפָּסוּק ׳בְּבֶן אָדָם שֶׁאֵין לוֹ תְשׁוּעָה׳ יֵשׁ מִלָּה חֲשׁוּבָה: תְּשׁוּעָה.

הַהֶבְדֵּל בֵּין תְּשׁוּעָה לְעֶזְרָה: תְּשׁוּעָה הִיא כְּשֶׁהַפִּתְרוֹן מַתְחִיל וְנִגְמָר בְּלִי שֶׁאָדָם יָכוֹל לַעֲשׂוֹת אוֹתוֹ – וְאָז חַיָּבִים עֶזְרָה מִן הַשָּׁמַיִם. עֶזְרָה הִיא כְּשֶׁאָדָם עוֹשֶׂה אֶת הַדָּבָר בְּסִיּוּעַ שֶׁל מִישֶׁהוּ.

בְּנֵי אָדָם יְכוֹלִים לְסַיֵּעַ, אֲבָל אֶת הַתְּשׁוּעָה הַגְּדוֹלָה – רַק ה׳ נוֹתֵן.

בֶּן אָדָם יָכוֹל לַעֲזֹר, אֲבָל אֶת הַתְּשׁוּעָה נוֹתֵן רַק ה׳.

1

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי רָעֵב וּמְבָרֵךְ, אֲנִי יוֹדֵעַ שֶׁה׳ הוּא שֶׁנּוֹתֵן אֶת הָאֹכֶל 💛

📖 פִּרְקֵי הַתְּפִלָּה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים קמ״ו, ז׳

רָמָה ב׳ – בִּטָּחוֹן וְהַשְׁגָּחָה

🍞 לֶחֶם לָרְעֵבִים וּמִשְׁפָּט לָעֲשׁוּקִים

׳עֹשֶׂה מִשְׁפָּט לָעֲשׁוּקִים׳ – כְּשֶׁעָשׁוּ לְאָדָם מִשְׁפָּט שֶׁלֹּא כַּדִּין, ה׳ בְּעַצְמוֹ עוֹשֶׂה אֶת הַמִּשְׁפָּט שֶׁלּוֹ.

׳נֹתֵן לֶחֶם לָרְעֵבִים׳ – לִפְעָמִים ה׳ מַמְתִּין עַד שֶׁהָאָדָם רָעֵב, כְּדֵי שֶׁיִּתְפַּלֵּל טוֹב עַל מְזוֹנוֹתָיו.

וְכָךְ גַּם הָרְעָבוֹן לְפָנִים הוּא הִזְדַּמְּנוּת: מִתּוֹךְ הַצֹּרֶךְ בָּאָה תְּפִלָּה אֲמִתִּית.

הַצֹּרֶךְ מְחַבֵּר אוֹתָנוּ לַתְּפִלָּה, וַה׳ נוֹתֵן אֶת הַלֶּחֶם בִּזְמַנּוֹ.

2

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְהַרְגִּישׁ אֶת עֶזְרַת ה׳, וְלָכֵן אֲנִי בּוֹטֵחַ בּוֹ 💛

📖 פִּרְקֵי הַתְּפִלָּה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים קמ״ו, ה׳ (הַחִידָ״א)

רָמָה ב׳ – בִּטָּחוֹן וְהַשְׁגָּחָה

📜 הַחִידָ״א: בִּזְכוּת הַבִּטָּחוֹן

יֵשׁ כָּאן שְׁאֵלָה: בְּדֶרֶךְ כְּלָל קֹדֶם בּוֹטְחִים בַּה׳ וְאַחַר כָּךְ הוּא עוֹזֵר. אֲבָל הַפָּסוּק אוֹמֵר ׳אַשְׁרֵי שֶׁאֵל יַעֲקֹב בְּעֶזְרוֹ, שִׂבְרוֹ עַל ה׳ אֱלֹהָיו׳ – קֹדֶם הָעֶזְרָה, וְאַחַר כָּךְ הַבִּטָּחוֹן!

הַחִידָ״א מְתָרֵץ: אַשְׁרֵי מִי שֶׁזָּכָה שֶׁהָאֵל בְּעֶזְרוֹ, וּבַמֶּה זָכָה לָזֶה? בִּזְכוּת שֶׁבִּטְחוֹנוֹ הָיָה עַל ה׳ אֱלֹהָיו.

כְּלוֹמַר, הַבִּטָּחוֹן הוּא הַסִּבָּה, וְהָעֶזְרָה הִיא הַשָּׂכָר.

הַבִּטָּחוֹן בַּה׳ הוּא מַה שֶּׁגּוֹרֵם לְעֶזְרָה שֶׁלּוֹ.

3

💛 אַהֲבַת הַשֵּׁם: אֲנִי חֵלֶק מֵעַם יִשְׂרָאֵל, וְה׳ דּוֹאֵג לָנוּ יַחַד 💛

📖 פִּרְקֵי הַתְּפִלָּה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים קמ״ז, ב׳–ג׳, י״ט

רָמָה ב׳ – בִּטָּחוֹן וְהַשְׁגָּחָה

🇮🇱 הַמִּזְמוֹר עַל עַם יִשְׂרָאֵל

הַמִּזְמוֹר הַשֵּׁנִי (קמ״ז) מְדַבֵּר עַל הַשְׁגָּחַת ה׳ עַל עַם יִשְׂרָאֵל כְּצִבּוּר: ׳בּוֹנֵה יְרוּשָׁלַיִם ה׳, נִדְחֵי יִשְׂרָאֵל יְכַנֵּס׳.

הוּא גַּם מְרַפֵּא אֶת הַלֵּב: ׳הָרוֹפֵא לִשְׁבוּרֵי לֵב וּמְחַבֵּשׁ לְעַצְּבוֹתָם׳.

וְהוּא גָּדוֹל מְאֹד וְגַם קָרוֹב: ׳מוֹנֶה מִסְפָּר לַכּוֹכָבִים, לְכֻלָּם שֵׁמוֹת יִקְרָא׳. אִם הוּא יוֹדֵעַ שֵׁם לְכָל כּוֹכָב – קַל לְהָבִין שֶׁהוּא מַכִּיר גַּם אוֹתָנוּ.

וּבַסּוֹף: ׳מַגִּיד דְּבָרָיו לְיַעֲקֹב, חֻקָּיו וּמִשְׁפָּטָיו לְיִשְׂרָאֵל׳ – אֶת הַתּוֹרָה נָתַן לְעַמּוֹ.

ה׳ מְכַנֵּס אֶת עַמּוֹ, מְרַפֵּא לֵב שָׁבוּר, וְנָתַן לָנוּ תּוֹרָה.

4

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְחַכֶּה לְחַסְדּוֹ שֶׁל ה׳ בְּשִׂמְחָה וּבְבִטָּחוֹן 💛

📖 פִּרְקֵי הַתְּפִלָּה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים קמ״ז, ט׳–י״א (אֲרוֹמִמְךָ)

רָמָה ב׳ – בִּטָּחוֹן וְהַשְׁגָּחָה

🐦 לָמָּה ה׳ לֹא מַכִּין לָנוּ אֹכֶל בַּפֶּתַח?

הַמִּזְמוֹר אוֹמֵר שֶׁה׳ ׳נוֹתֵן לִבְהֵמָה לַחְמָהּ, לִבְנֵי עֹרֵב אֲשֶׁר יִקְרָאוּ׳ – הַחַיּוֹת לֹא צְרִיכוֹת לִטְרֹחַ הַרְבֵּה.

אָז לָמָּה לֹא מַכִּין ה׳ גַּם לָנוּ אֹכֶל לְיַד הַדֶּלֶת? בְּסֵפֶר ׳אֲרוֹמִמְךָ׳ מְבָאֲרִים שֶׁזֶּה בְּגַלְלוֹ שֶׁ׳רוֹצֶה ה׳ אֶת יְרֵאָיו, אֶת הַמְיַחֲלִים לְחַסְדּוֹ׳.

ה׳ רוֹצֶה לִרְאוֹת אֶת מִי שֶׁמְּצַפֶּה לְחַסְדּוֹ וּבוֹטֵחַ בּוֹ.

ה׳ רוֹצֶה שֶׁנְּצַפֶּה לְחַסְדּוֹ וְנִבְטַח בּוֹ.

5

💛 אַהֲבַת הַשֵּׁם: הַשֶּׁמֶשׁ, הַיָּרֵחַ וַאֲנִי – כֻּלָּנוּ מְהַלְלִים אֶת ה׳ יַחַד 💛

📖 פִּרְקֵי הַתְּפִלָּה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים קמ״ח, א׳–י״ג

רָמָה ב׳ – בִּטָּחוֹן וְהַשְׁגָּחָה

🌍 הַמִּזְמוֹר עַל הַבְּרִיאָה כֻּלָּהּ

הַמִּזְמוֹר הַשְּׁלִישִׁי (קמ״ח) קוֹרֵא לַכֹּל לְהַלֵּל: ׳הַלְלוּ אֶת ה׳ מִן הַשָּׁמַיִם׳ – ׳הַלְלוּהוּ שֶׁמֶשׁ וְיָרֵחַ, הַלְלוּהוּ כָּל כּוֹכְבֵי אוֹר׳.

וְאַחַר כָּךְ ׳הַלְלוּ אֶת ה׳ מִן הָאָרֶץ׳: תַּנִּינִים, אֵשׁ וּבָרָד, שֶׁלֶג, הָרִים וַאֲרָזִים, חַיָּה וּבְהֵמָה, רֶמֶשׂ וְצִפּוֹר – וְגַם בְּנֵי אָדָם: מַלְכֵי אֶרֶץ, בַּחוּרִים, וּזְקֵנִים עִם נְעָרִים.

לָמָּה? ׳כִּי הוּא צִוָּה וְנִבְרָאוּ, וַיַּעֲמִידֵם לָעַד, חָק נָתַן וְלֹא יַעֲבוֹר׳ – הֵם שׁוֹמְרִים עַל הַסֵּדֶר שֶׁה׳ קָבַע.

כָּל הַבְּרִיאָה מְהַלֶּלֶת אֶת ה׳ בַּסֵּדֶר שֶׁהוּא קָבַע לָהּ.

מָה לוֹקְחִים מִרָמָה ב׳?

ה׳ רוֹצֶה שֶׁנִּבְטַח בּוֹ וּנְיַחֵל לְחַסְדּוֹ, וְהוּא עוֹזֵר לְמִי שֶׁבּוֹטֵחַ. כְּשֶׁאֲנִי בּוֹטֵחַ בּוֹ כְּמוֹ יַעֲקֹב אָבִינוּ, אֲנִי מַרְגִּישׁ שֶׁאֲנִי לֹא לְבַד, וְשֶׁגַּם הַשֶּׁמֶשׁ וְהַיָּרֵחַ מְהַלְלִים אוֹתוֹ יַחַד אִתִּי.

מָה הַהֶבְדֵּל בֵּין כָּבוֹד לְעֹנֶג, אֵיךְ מְכַבְּדִים בַּיָּדַיִם וּמִתְעַנְּגִים בַּלֵּב, וְלָמָּה הַפָּסוּק מַמְשִׁיךְ ׳אָז תִּתְעַנַּג עַל ה׳׳

🌅 אָז תִּתְעַנַּג עַל ה׳

הַפָּסוּק אוֹמֵר שֶׁמִּי שֶׁמְּכַבֵּד וּמְעַנֵּג אֶת הַשַּׁבָּת – ׳אָז תִּתְעַנַּג עַל ה׳׳.

לְפִי מְסִלַּת יְשָׁרִים זֶה תַּכְלִית הָאָדָם: ׳שֶׁהָאָדָם לֹא נִבְרָא אֶלָּא לְהִתְעַנֵּג עַל ה׳ וְלֵהָנוֹת מִזִּיו שְׁכִינָתוֹ׳.

בָּעוֹלָם הַזֶּה מִתְאַמְּנִים בָּזֶה, וְהַשְּׁלֵמוּת הַגְּדוֹלָה תִּהְיֶה בָּעוֹלָם הַבָּא.

הַשַּׁבָּת מְאַמֶּנֶת אוֹתָנוּ לְהִתְעַנֵּג עַל ה׳.

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹצֵר רֶגַע מִכָּל הַמְּרוּצָה וּמוֹדֶה לַה׳ 💛

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, יְשַׁעְיָהוּ נ״ח י״ג · תַּפְקִיד הַשַּׁבָּת

רָמָה ב׳ – מִכָּבוֹד וְעֹנֶג לְאַהֲבָה

🕊️ פּוֹרְשִׁים כְּדֵי לְהוֹדוֹת

הַפָּסוּק אוֹמֵר: ׳וְכִבַּדְתּוֹ מֵעֲשׂוֹת דְּרָכֶיךָ מִמְּצוֹא חֶפְצְךָ׳ – לִפְרֹשׁ קְצָת מֵעִנְיְנֵי הָעוֹלָם.

לָמָּה? כָּל עוֹד הָאָדָם שָׁקוּעַ בְּעִנְיְנֵי הָעוֹלָם הַזֶּה, קָשֶׁה לוֹ לְהוֹדוֹת וּלְהַלֵּל אֶת ה׳.

וְכָךְ אוֹמְרִים בַּתְּפִלָּה: ׳וּקְדוֹשִׁים בְּכָל יוֹם יְהַלְלוּךָ סֶּלָה׳.

כְּשֶׁפּוֹרְשִׁים קְצָת מֵהָעוֹלָם, קַל יוֹתֵר לְהוֹדוֹת.

💞

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁיִּהְיֶה לִי נָעִים לְדַבֵּר עִם ה׳ 💛

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְסִלַּת יְשָׁרִים, פֶּרֶק י״ט

רָמָה ב׳ – מִכָּבוֹד וְעֹנֶג לְאַהֲבָה

💞 אַהֲבָה שֶׁגַּם הַדִּבּוּר בָּהּ נָעִים

מְסִלַּת יְשָׁרִים מְלַמֵּד אֵיךְ מַגִּיעִים לְהִתְעַנֵּג עַל ה׳: דֶּרֶךְ הָאַהֲבָה.

מִי שֶׁאוֹהֵב מְאֹד אֶת בְּנוֹ הַיָּחִיד, נָעִים לוֹ אֲפִלּוּ רַק לְדַבֵּר עִמּוֹ. כָּךְ מִי שֶׁאוֹהֵב אֶת ה׳, נָעִים לוֹ לְהַזְכִּיר אֶת שְׁמוֹ, לְהַלֵּל אוֹתוֹ וְלִלְמֹד תּוֹרָה – זֶה שַׁעֲשׁוּעַ וְעֹנֶג מַמָּשׁ.

מִפֵּרוֹת הָאַהֲבָה הֵם הַשִּׂמְחָה וְהַדְּבֵקוּת בַּשֵּׁם.

מִי שֶׁאוֹהֵב אֶת ה׳, נָעִים לוֹ אֲפִלּוּ לְדַבֵּר אִתּוֹ.

💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת אֲנִי שָׁר וּמוֹדֶה, וְהַלֵּב שֶׁלִּי נִפְתָּח 💛

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַד״ק · תְּהִלִּים צ״ב

רָמָה ב׳ – מִכָּבוֹד וְעֹנֶג לְאַהֲבָה

🎵 מִשַּׁבָּת לְשִׁיר

הָרַד״ק מוֹסִיף: מִי שֶׁמְּשַׁנֶּה אֶת הַשַּׁבָּת מִשְּׁאָר הַיָּמִים לְטוֹב, יִזְכֹּר אֶת מַעֲשֵׂה בְּרֵאשִׁית וִישַׁבֵּחַ אֶת ה׳ בְּפִיו וּבְלִבּוֹ, וְנַפְשׁוֹ תִּתְעַנֵּג בּוֹ.

זֶה מַה שֶּׁאָמַר דָּוִד: ׳מִזְמוֹר שִׁיר לְיוֹם הַשַּׁבָּת, טוֹב לְהֹדוֹת לַה׳׳.

אִם בַּשַּׁבָּת עוֹזְבִים אֶת הָעִנְיָנִים הַגַּשְׁמִיִּים וּמַתְחִילִים לְהוֹדוֹת – מַגִּיעִים לְהִתְעַנֵּג עַל ה׳.

מוֹדִים לַה׳ בַּשַּׁבָּת – וּמַגִּיעִים לְהִתְעַנֵּג עָלָיו.

🤲

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹטֵחַ בַּה׳ שֶׁהַכֹּל מִמֶּנּוּ וּלְטוֹבָה 💛

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים ל״ז ג׳–ה׳

רָמָה ב׳ – מִכָּבוֹד וְעֹנֶג לְאַהֲבָה

🤲 בִּטָּחוֹן – דֶּרֶךְ נוֹסֶפֶת

דָּוִד הַמֶּלֶךְ מְלַמֵּד דֶּרֶךְ נוֹסֶפֶת: ׳בְּטַח בַּה׳ וַעֲשֵׂה טוֹב… וְהִתְעַנַּג עַל ה׳ וְיִתֶּן לְךָ מִשְׁאֲלֹת לִבֶּךָ׳.

לְשַׁבֵּחַ אֶת ה׳ אֶפְשָׁר רַק כְּשֶׁמַּאֲמִינִים שֶׁהַכֹּל לְטוֹבָה וְשֶׁכָּל מַה שֶּׁיֵּשׁ לָנוּ בָּא מֵאִתּוֹ.

מִי שֶׁמַּאֲמִין כָּךְ מְהַלֵּל – וּמִתּוֹךְ כָּךְ מִתְעַנֵּג עַל ה׳.

בִּטָּחוֹן וֶאֱמוּנָה מְבִיאִים לְהִתְעַנֵּג עַל ה׳.

💛 אַהֲבַת הַשֵּׁם: אֲנִי שׁוֹאֵל אֶת עַצְמִי: לְמִי זֶה לִכְבוֹד? 💛

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, קְנִיּוֹת לִכְבוֹד שַׁבָּת

רָמָה ב׳ – מִכָּבוֹד וְעֹנֶג לְאַהֲבָה

⚖️ כַּמָּה קוֹנִים וְלָמָּה

כְּשֶׁקּוֹנִים לִכְבוֹד שַׁבָּת – קוֹנִים כְּדֶרֶךְ סְעוּדָה חֲשׁוּבָה, וְלְפִי מַה שֶּׁיֵּשׁ בַּיָּד.

נוֹהֲגִים לֶאֱכֹל דָּגִים, מַאֲכָל חָשׁוּב לִכְבוֹד שַׁבָּת שֶׁרוֹמֵז גַּם לְפַרְנָסָה מִן הַשָּׁמַיִם.

מוּתָּר וְרָצוּי לְהוֹצִיא לִכְבוֹד שַׁבָּת, אֲבָל לֹא לִכְבוֹד הַבֶּטֶן.

הַהוֹצָאָה בִּשְׁבִיל הַשַּׁבָּת, וְלֹא בִּשְׁבִיל הַבֶּטֶן.

מָה לוֹקְחִים מִרָמָה ב׳?

כָּבוֹד וְעֹנֶג הֵם דֶּרֶךְ אֶל ׳אָז תִּתְעַנַּג עַל ה׳׳. מִי שֶׁפּוֹרֵשׁ קְצָת מִטִּרְדּוֹת הָעוֹלָם וּמוֹדֶה לַה׳, מַרְגִּישׁ אֶת אַהֲבָתוֹ וְאוֹהֵב אוֹתוֹ בַּחֲזָרָה.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה אֲנִי עוֹשֶׂה בְּעַצְמִי כְּדֵי לְכַבֵּד אֶת הַשַּׁבָּת, וּמָה גּוֹרֵם לִי לְהִתְעַנֵּג בָּהּ?
  • מָה הָעִקָּר אֶצְלִי בַּשַּׁבָּת – הָאֹכֶל, אוֹ הַשַּׁבָּת עַצְמָהּ?
  • בְּשַׁבָּת הַקְּרוֹבָה אֶבְחַר דָּבָר אֶחָד לְכַבֵּד בּוֹ אֶת הַשַּׁבָּת בְּעַצְמִי וְדָבָר אֶחָד לְהִתְעַנֵּג בּוֹ, וְאֶרְשֹׁם מָה הִרְגַּשְׁתִּי.

וְשַׁבְתָּ עַד ה׳ אֱלֹהֶיךָ וְשָׁמַעְתָּ בְקֹלוֹ

🌅 מְקָרְבִים אֶת הַגְּאֻלָּה · רָמָה ב׳

וְשַׁבְתָּ עַד ה׳

רָמָה ב׳ – הַפְּסוּקִים וְהַמְּקוֹרוֹת

פָּרָשַׁת נִצָּבִים שׁוֹזֶרֶת תְּשׁוּבָה וּגְאֻלָּה כְּמוֹ סֻלָּם, וְהַתְּשׁוּבָה הִיא חֲזָרָה בְּאַהֲבָה

📖 הַגְּאֻלָּה · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
📖

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹהֵב לִלְמֹד שֶׁה׳ מוֹלִיךְ אוֹתָנוּ בְּסֵדֶר וּבְרַכּוּת 💛

📖 הַגְּאֻלָּה, דְּבָרִים ל׳ א׳–ו׳

רָמָה ב׳ – הַפְּסוּקִים וְהַמְּקוֹרוֹת

📖 הַפָּרָשָׁה שֶׁשּׁוֹזֶרֶת

בְּסוֹף פָּרָשַׁת נִצָּבִים הַתּוֹרָה כּוֹתֶבֶת פְּסוּקֵי תְּשׁוּבָה וּפְסוּקֵי גְּאֻלָּה לְסֵרוּגִין: ׳וַהֲשֵׁבֹתָ אֶל לְבָבֶךָ׳, ׳וְשַׁבְתָּ עַד ה׳׳, ׳וְשָׁב ה׳ אֶת שְׁבוּתְךָ׳, ׳וּמָל ה׳ אֶת לְבָבְךָ׳.

הַסֵּדֶר הַזֶּה מְלַמֵּד שֶׁהַגְּאֻלָּה אֵינָהּ בָּאָה בְּבַת אַחַת, אֶלָּא כְּמוֹ סֻלָּם: תְּשׁוּבָה, גְּאֻלָּה, תְּשׁוּבָה, גְּאֻלָּה.

כָּל שָׁלָב מוֹלִיךְ אֶל הַשָּׁלָב הַבָּא.

הַתּוֹרָה שׁוֹזֶרֶת תְּשׁוּבָה וּגְאֻלָּה כְּמוֹ סֻלָּם.

1
💭

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לָתֵת לַלֵּב שֶׁלִּי רֶגַע שֶׁקֵּט לִפְנֵי כָּל צַעַד 💛

📖 הַגְּאֻלָּה, דְּבָרִים ל׳ א׳

רָמָה ב׳ – הַפְּסוּקִים וְהַמְּקוֹרוֹת

💭 ״וַהֲשֵׁבֹתָ״ וְלֹא ״וְשַׁבְתָּ״

בַּפָּסוּק הָרִאשׁוֹן לֹא כָּתוּב ׳וְשַׁבְתָּ׳, אֶלָּא ׳וַהֲשֵׁבֹתָ אֶל לְבָבֶךָ׳.

זֶה פֹּעַל אַחֵר: לְהָשִׁיב אֶל הַלֵּב אֶת מָה שֶׁעָבַר עָלֵינוּ, לְהָבִין שֶׁזֶּה מַשְׁמָעוּתִי.

זֶה צַעַד שֶׁקֵּט וּפְנִימִי, לִפְנֵי שֶׁעוֹשִׂים מַשֶּׁהוּ מִבַּחוּץ.

הַצַּעַד הָרִאשׁוֹן הוּא פְּנִימִי: לְהָשִׁיב אֶל הַלֵּב.

2
🚪

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ שֶׁגַּם מִי שֶׁעָמַד רַק בַּפֶּתַח כְּבָר קָרוֹב לַה׳ 💛

📖 הַגְּאֻלָּה, דְּבָרִים ל׳ ב׳

רָמָה ב׳ – הַפְּסוּקִים וְהַמְּקוֹרוֹת

🚪 ״עַד״ וְ״אֶל״

בַּפָּסוּק הַשֵּׁנִי כָּתוּב ׳וְשַׁבְתָּ עַד ה׳׳, וְכֵן בַּנָּבִיא: ׳שׁוּבָה יִשְׂרָאֵל עַד ה׳ אֱלֹהֶיךָ׳. אֲבָל בְּסוֹף הַפָּרָשָׁה כָּתוּב ׳כִּי תָשׁוּב אֶל ה׳׳.

מָשָׁל: ׳הִגַּעְתִּי עַד הַבִּנְיָן׳ וּ׳הִגַּעְתִּי אֶל הַבִּנְיָן׳. מִי שֶׁהִגִּיעַ עַד עוֹמֵד בַּחוּץ, לְיַד הַפֶּתַח. מִי שֶׁהִגִּיעַ אֶל נִכְנָס פְּנִימָה.

כָּךְ יֵשׁ שְׁתֵּי דַּרְגּוֹת: מַתְחִילִים בִּתְשׁוּבָה שֶׁמַּגִּיעָה עַד הַפֶּתַח, וְהַשְּׁלֵמָה נִכְנֶסֶת פְּנִימָה.

׳עַד׳ הוּא עַד הַפֶּתַח, ׳אֶל׳ הוּא פְּנִימָה.

3
🤝

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַרְגִּישׁ שֶׁאֲנִי לֹא שָׁב לְבַד, וְה׳ הוֹלֵךְ אִתִּי 💛

📖 הַגְּאֻלָּה, דְּבָרִים ל׳ ג׳, ו׳

רָמָה ב׳ – הַפְּסוּקִים וְהַמְּקוֹרוֹת

🤝 שְׁנֵי צְדָדִים שָׁבִים

אֶת הַמִּלָּה ׳שָׁב׳ הַתּוֹרָה כּוֹתֶבֶת גַּם עַל ה׳: ׳וְשָׁב ה׳ אֱלֹהֶיךָ אֶת שְׁבוּתְךָ וְרִחֲמֶךָ׳. הַמִּלָּה ׳שָׁב׳ פֵּרוּשָׁהּ חָזַר: אֶפְשָׁר לְהָבִין שֶׁה׳, כִּבְיָכוֹל, חוֹזֵר וּמַחְזִיר אוֹתָנוּ.

וּבַפָּסוּק עַל הַלֵּב: ׳וּמָל ה׳ אֱלֹהֶיךָ אֶת לְבָבְךָ לְאַהֲבָה אֶת ה׳׳.

יֵשׁ שָׁלָב שֶׁבּוֹ הַתְּשׁוּבָה בָּאָה מִשְּׁנֵי הַצְּדָדִים יַחַד: אֲנַחְנוּ שָׁבִים, וְה׳ פּוֹתֵחַ אֶת הַלֵּב.

בַּתְּשׁוּבָה שָׁבִים שְׁנֵי צְדָדִים: אֲנַחְנוּ וְה׳.

4
⚖️

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹהֵב שֶׁהַחֲכָמִים לוֹמְדִים יַחַד וְחוֹשְׁבִים בִּשְׁתֵּי דְּרָכִים 💛

📖 הַגְּאֻלָּה, סַנְהֶדְרִין צ״ז ע״ב

רָמָה ב׳ – הַפְּסוּקִים וְהַמְּקוֹרוֹת

⚖️ מַה בָּא קֹדֶם?

בַּגְּמָרָא נֶחְלְקוּ שְׁנֵי חֲכָמִים: רַבִּי אֱלִיעֶזֶר אָמַר: אִם יִשְׂרָאֵל עוֹשִׂים תְּשׁוּבָה נִגְאָלִים, וְהֵבִיא פָּסוּק: ׳שׁוּבוּ בָנִים שׁוֹבָבִים אֶרְפָּה מְשׁוּבֹתֵיכֶם׳.

רַבִּי יְהוֹשֻׁעַ הֵבִיא פָּסוּק אַחֵר: ׳חִנָּם נִמְכַּרְתֶּם וְלֹא בְכֶסֶף תִּגָּאֵלוּ׳, וְחָשַׁב שֶׁהַגְּאֻלָּה לֹא תְּלוּיָה רַק בָּזֶה.

שְׁנֵיהֶם מְדַבְּרִים עַל הַגְּאֻלָּה שֶׁה׳ רוֹצֶה בָּהּ, וְהַמַּחְלֹקֶת מְלַמֶּדֶת שֶׁאֶפְשָׁר לְהַבִּיט עַל הָעִנְיָן מִשְּׁתֵּי זָוִיּוֹת.

חֲכָמִים נֶחְלְקוּ אִם הַגְּאֻלָּה תְּלוּיָה בִּתְשׁוּבָה.

5
📜

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִתְרַגֵּשׁ שֶׁהַתּוֹרָה מַבְטִיחָה לָנוּ שֶׁנָּשׁוּב 💛

📖 הַגְּאֻלָּה, רַמְבַּ״ם תְּשׁוּבָה ז׳ ה׳

רָמָה ב׳ – הַפְּסוּקִים וְהַמְּקוֹרוֹת

📜 מָה פָּסַק הָרַמְבַּ״ם

הָרַמְבַּ״ם כּוֹתֵב: ׳כָּל הַנְּבִיאִים כֻּלָּן צִוּוּ עַל הַתְּשׁוּבָה, וְאֵין יִשְׂרָאֵל נִגְאָלִין אֶלָּא בִּתְשׁוּבָה׳.

וְהוּא מוֹסִיף: ׳וּכְבָר הִבְטִיחָה תּוֹרָה שֶׁסּוֹף יִשְׂרָאֵל לַעֲשׂוֹת תְּשׁוּבָה, וּמִיָּד הֵן נִגְאָלִין׳, וּמֵבִיא אֶת הַפְּסוּקִים ׳וְשַׁבְתָּ עַד ה׳׳ וְ׳וְשָׁב ה׳׳.

זֹאת הַבְטָחָה טוֹבָה: הַתְּשׁוּבָה בְּוַדַּאי תָּבוֹא, וְהִיא תְּקָרֵב אֶת הַגְּאֻלָּה.

הָרַמְבַּ״ם: הַתּוֹרָה הִבְטִיחָה תְּשׁוּבָה, וְהִיא מְקָרֶבֶת גְּאֻלָּה.

6
💡

רָמָה ב׳ – הַפְּסוּקִים וְהַמְּקוֹרוֹת

מָה לוֹקְחִים מִרָמָה ב׳?

הַתּוֹרָה מְלַמֶּדֶת שֶׁה׳ שָׁב אֵלֵינוּ בְּרַחֲמִים וְעוֹזֵר לַלֵּב לָשׁוּב. כְּשֶׁאֲנִי עוֹשֶׂה תְּשׁוּבָה מִתּוֹךְ אַהֲבָה, אֲנִי מַרְגִּישׁ אֶת אַהֲבַת ה׳ אֵלַי.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • אֵיפֹה אֲנִי עוֹמֵד הַיּוֹם עַל הַסֻּלָּם: בַּמַּחֲשָׁבָה, בַּדֶּרֶךְ אֶל הַפֶּתַח, אוֹ כְּבָר בִּפְנִים?
  • מָה הַצַּעַד הַקָּטָן שֶׁאֲנִי יָכוֹל לַעֲשׂוֹת הַשָּׁבוּעַ כְּדֵי לָשׁוּב אֶל ה׳ מִתּוֹךְ אַהֲבָה?
  • הַמְּשִׂימָה: לִבְנוֹת סֻלָּם קָטָן שֶׁל אַרְבָּעָה צְעָדִים, וְלַעֲשׂוֹת הַשָּׁבוּעַ אֶת הַצַּעַד הָרִאשׁוֹן.

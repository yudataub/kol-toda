🌱 הָאִישׁ הַמְכֻעָּר · רָמָה ב׳

לֵךְ לָאֻמָּן שֶׁעֲשָׂאַנִי

רָמָה ב׳ – קָנֶה וְאֶרֶז

הָאִישׁ הַמְכֻעָּר וְרַבִּי אֶלְעָזָר: גַּאֲוָה, בִּזָּיוֹן, בַּקָּשַׁת סְלִיחָה, וְרַךְ כְּקָנֶה

📖 כְּבוֹד הַבְּרִיּוֹת · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
🧐

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁהַתּוֹרָה שֶׁאֲנִי לוֹמֵד תְּקָרֵב אוֹתִי לַה׳ וּתְעַנֵּן אֶת לִבִּי 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תַּעֲנִית כ׳ א׳

רָמָה ב׳ – קָנֶה וְאֶרֶז

🧐 מַה בְּדִיּוּק הָיְתָה הַטָּעוּת?

הַגְּמָרָא מְסַפֶּרֶת שֶׁרַבִּי אֶלְעָזָר הָיָה ׳שָׂמֵחַ שִׂמְחָה גְּדוֹלָה׳, וּבְאוֹתָהּ שָׁעָה ׳הָיְתָה דַּעְתּוֹ גַּסָּה עָלָיו מִפְּנֵי שֶׁלָּמַד תּוֹרָה הַרְבֵּה׳.

כְּלוֹמַר: הַתּוֹרָה בְּעַצְמָהּ הִיא דָּבָר נִפְלָא, אֲבָל כְּשֶׁהַלֵּב מִתְמַלֵּא בָּהּ גַּאֲוָה, הוּא רוֹאֶה אֶת הַזּוּלָת קָטָן.

מִשָּׁם בָּאוּ שְׁתֵּי הַטָּעֻיּוֹת: לֹא הֶחְזִיר שָׁלוֹם, וְדִבֵּר בִּגְנַאי עַל הַמַּרְאֶה.

גַּאֲוָה גּוֹרֶמֶת לָנוּ לִרְאוֹת אֶת הַזּוּלָת קָטָן.

1
🏺

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹאֶה בְּכָל אָדָם אֶת מַעֲשֵׂה יְדֵי הָאֻמָּן הַגָּדוֹל, וְאוֹהֵב אוֹתוֹ 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תַּעֲנִית כ׳ א׳

רָמָה ב׳ – קָנֶה וְאֶרֶז

🏺 הַכְּלִי וְהָאֻמָּן

הָאִישׁ לֹא חָלַק עַל כָּךְ שֶׁהוּא נִרְאֶה אַחֶרֶת. הוּא אָמַר רַק: ׳לֵךְ וֶאֱמֹר לָאֻמָּן שֶׁעֲשָׂאַנִי: כַּמָּה מְכֻעָּר כְּלִי זֶה שֶׁעָשִׂיתָ׳.

כְּשֶׁאָדָם עוֹשֶׂה כְּלִי, הַכְּלִי הוּא מַעֲשֵׂה יָדָיו. מִי שֶׁמְּבַקֵּר אֶת הַכְּלִי – מְבַקֵּר אֶת הָאֻמָּן.

וְהָאָדָם הוּא מַעֲשֵׂה יָדָיו שֶׁל ה׳. מִי שֶׁלּוֹעֵג לוֹ, לוֹעֵג לְיוֹצְרוֹ.

הָאָדָם הוּא מַעֲשֵׂה יָדָיו שֶׁל ה׳.

2
⚖️

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִתְבּוֹנֵן שֶׁה׳ כָּל כָּךְ אוֹהֵב אֶת הָאָדָם, עַד שֶׁמִּי שֶׁפּוֹגֵעַ בּוֹ פּוֹגֵעַ גַּם בְּכָבוֹדוֹ 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, וַיִּקְרָא כ״ד, רַבֵּנוּ בַּחְיֵי

רָמָה ב׳ – קָנֶה וְאֶרֶז

⚖️ פְּגִיעָה בָּאָדָם וּפְגִיעָה בְּכָבוֹד שָׁמַיִם

הַתּוֹרָה בְּפָרָשַׁת אֱמֹר מְחַבֶּרֶת יַחַד אֶת דִּין הַמְקַלֵּל אֶת ה׳ וְאֶת דִּין הַמַּכֶּה אָדָם. הַמְפָרְשִׁים מְבִינִים שֶׁהִיא רוֹצָה לְלַמֵּד: אַל תַּחְשֹׁב שֶׁרַק לְחַלֵּל אֶת הַשֵּׁם חָמוּר. גַּם לִפְגֹּעַ בְּאָדָם חָמוּר.

הַסִּבָּה: ׳בְּצֶלֶם אֱלֹהִים עָשָׂה אֶת הָאָדָם׳.

וּלְכֵן הַסִּפּוּר שֶׁל רַבִּי אֶלְעָזָר נוֹגֵעַ לֹא רַק לְבֵין אָדָם לַחֲבֵרוֹ, אֶלָּא לְכִבּוּד ה׳.

פְּגִיעָה בָּאָדָם נוֹגַעַת בְּכָבוֹד ה׳.

3
🙇

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה לַה׳ עַל הָאֶפְשָׁרוּת לְתַקֵּן, וּמִתְפַּלֵּל שֶׁאֶזְכֶּה לְהִתְנַצֵּל בֶּאֱמֶת 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תַּעֲנִית כ׳ א׳

רָמָה ב׳ – קָנֶה וְאֶרֶז

🙇 הִתְנַצְּלוּת אֲמִתִּית

רַבִּי אֶלְעָזָר לֹא הִסְתַּפֵּק בִּ׳סְלִיחָה׳ קְצָרָה. הוּא יָרַד מִן הַחֲמוֹר, הִשְׁתַּטֵּחַ, וְאָמַר: ׳נַעֲנֵיתִי לְךָ, מְחַל לִי׳. הוּא הוֹרִיד אֶת עַצְמוֹ.

הָאִישׁ לֹא מָחַל מִיָּד, וְרַבִּי אֶלְעָזָר הָלַךְ אַחֲרָיו עַד הָעִיר. לָמַדְנוּ שֶׁהִתְנַצְּלוּת אֲמִתִּית לָעִתִּים דּוֹרֶשֶׁת סַבְלָנוּת.

וּבַסּוֹף: ׳בִּלְבַד שֶׁלֹּא יְהֵא רָגִיל לַעֲשׂוֹת כֵּן׳ – מְחִילָה צְרִיכָה גַּם הַחְלָטָה לֹא לַחְזֹר.

הִתְנַצְּלוּת אֲמִתִּית = הַכָּרָה, בַּקָּשָׁה, וְהַחְלָטָה לֹא לַחְזֹר.

4
🌿

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְבַקֵּשׁ מֵה׳ לִהְיוֹת רַךְ כְּקָנֶה וְלָשֶׁבֶת לְיַד מַיִם חַיִּים שֶׁל תּוֹרָה 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תַּעֲנִית כ׳ א׳

רָמָה ב׳ – קָנֶה וְאֶרֶז

🌿 רַךְ כְּקָנֶה

הַגְּמָרָא מְתָאֶרֶת: הַקָּנֶה עוֹמֵד בִּמְקוֹם מַיִם, גִּזְעוֹ מַחֲלִיף וְשָׁרָשָׁיו מְרֻבִּין. אֲפִלּוּ כָּל הָרוּחוֹת בָּאוֹת וְנוֹשְׁבוֹת בּוֹ – אֵין מְזִיזוֹת אוֹתוֹ, אֶלָּא הוֹלֵךְ וּבָא עִמָּהֶן. דָּמְמוּ הָרוּחוֹת – עָמַד הַקָּנֶה בִּמְקוֹמוֹ.

הָאֶרֶז אֵינוֹ עוֹמֵד לְיַד מַיִם, וְאֵין שָׁרָשָׁיו מְרֻבִּין. כְּשֶׁנּוֹשֶׁבֶת בּוֹ רוּחַ דְּרוֹמִית, עוֹקֶרֶת אוֹתוֹ וְהוֹפֶכֶת אוֹתוֹ עַל פָּנָיו.

מִי שֶׁרַךְ וְעָנָו מַחֲזִיק מַעֲמָד בְּכָל מַצָּב.

עֲנָוָה הִיא כּוֹחַ: הַקָּנֶה נִשְׁאָר עוֹמֵד, הָאֶרֶז נֶעֱקָר.

📣

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת כְּלִי לַתּוֹרָה וּלְחַנֵּךְ אֶת עַצְמִי מִטָּעֻיּוֹת 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תַּעֲנִית כ׳ א׳

רָמָה ב׳ – קָנֶה וְאֶרֶז

📣 גָּדוֹל שֶׁמּוֹדֶה בְּטָעוּת

מִיָּד אַחֲרֵי שֶׁהָאִישׁ מָחַל, רַבִּי אֶלְעָזָר נִכְנַס לְבֵית הַמִּדְרָשׁ וְדָרַשׁ אֶת הַדְּרָשָׁה עַל הַקָּנֶה וְהָאֶרֶז. הוּא לֹא הִסְתִּיר אֶת מַה שֶּׁקָּרָה, אֶלָּא הִפְנִים אֶת הַלֶּקַח וְלִמֵּד אוֹתוֹ לָרַבִּים.

גַּם הַקָּנֶה זָכָה: ׳לִיטּוֹל הֵימֶנּוּ קֻלְמוֹס לִכְתֹּב בּוֹ סֵפֶר תּוֹרָה׳.

מִי שֶׁלָּמַד לְהִתְכּוֹפֵף, נַעֲשֶׂה כְּלִי לַתּוֹרָה.

גְּדוֹלִים לֹא מִתְבַּיְּשִׁים לְלַמֵּד מִטָּעוּת.

💡

רָמָה ב׳ – קָנֶה וְאֶרֶז

מָה לוֹקְחִים מִרָמָה ב׳?

גַּם גְּדוֹלֵי יִשְׂרָאֵל טָעוּ וְתִקְּנוּ. הַתְּשׁוּבָה וְהָעֲנָוָה מְקָרְבוֹת אֶת הָאָדָם אֶל ה׳, שֶׁאוֹהֵב אֶת הַמְּבַקְּשִׁים סְלִיחָה.

  • מִי הוּא הָאֻמָּן שֶׁעָשָׂה אֶת כָּל אָדָם, וְאֵיךְ זֶה מְשַׁנֶּה אֶת הַיַּחַס שֶׁלִּי אֵלָיו?

אַל נָא תַעֲבֹר מֵעַל עַבְדֶּךָ

🌱 גְּדוֹלָה מֵהַשְּׁכִינָה · רָמָה ב׳

גְּדוֹלָה הַכְנָסַת אוֹרְחִים מִקַּבָּלַת פְּנֵי שְׁכִינָה

רָמָה ב׳ – צֶלֶם אֱלֹקִים וְהֲלִיכָה בְּדַרְכָּיו

אַבְרָהָם מְבַקֵּשׁ מֵה׳ לְהַמְתִּין וְרָץ אֶל הָאוֹרְחִים – מַדּוּעַ זוֹ הֲלִיכָה בְּדַרְכֵי ה׳, וּמָה רוֹאִים בְּכָל אוֹרֵחַ

📖 הַכְנָסַת אוֹרְחִים · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
⚖️

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא עוֹזֵב אֶת ה׳ כְּשֶׁאֲנִי עוֹזֵר, אֲנִי הוֹלֵךְ בְּדַרְכָּיו 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שַׁבָּת קכ״ז א׳ · שְׁלָ״ה פָּרָשַׁת וַיֵּרָא

רָמָה ב׳ – צֶלֶם אֱלֹקִים וְהֲלִיכָה בְּדַרְכָּיו

⚖️ גְּדוֹלָה מִקַּבָּלַת פְּנֵי שְׁכִינָה

הַגְּמָרָא (שַׁבָּת קכ״ז) לוֹמֶדֶת מִפְּסוּק ג׳: ׳גְּדוֹלָה הַכְנָסַת אוֹרְחִים מִקַּבָּלַת פְּנֵי שְׁכִינָה׳. אַבְרָהָם עָמַד בְּשִׂיחָה עִם ה׳, רָאָה אוֹרְחִים, בִּקֵּשׁ מִמֶּנּוּ לְהַמְתִּין, וְרָץ אֲלֵיהֶם.

מַה יָכוֹל לִהְיוֹת גָּדוֹל יוֹתֵר מִלַּעֲמֹד לִפְנֵי בּוֹרֵא עוֹלָם? הַבֵּיאוּר: אַבְרָהָם לֹא עָזַב אֶת ה׳, הוּא עָבַר מִלַּעֲמֹד לְפָנָיו לָלֶכֶת אַחֲרָיו. בְּקַבָּלַת פְּנֵי שְׁכִינָה עוֹמְדִים מוּל ה׳, וּבְהַכְנָסַת אוֹרְחִים אֲנַחְנוּ דּוֹמִים לוֹ: ׳מַה הוּא – אַף אַתָּה׳.

הַשְּׁלָ״ה הַקָּדוֹשׁ כּוֹתֵב שֶׁזֹּאת מִצְוַת עֲשֵׂה: ׳וְהָלַכְתָּ בִּדְרָכָיו׳.

הַכְנָסַת אוֹרְחִים הִיא הֲלִיכָה בְּדַרְכֵי ה׳, לֹא עֲזִיבָה שֶׁלּוֹ.

1
🪞

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִסְתַּכֵּל בְּאוֹרֵחַ וְרוֹאֶה בּוֹ צֶלֶם אֱלֹקִים 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מַהֲרַ״ל, נְתִיבוֹת עוֹלָם · אָבוֹת ג׳ י״ד

רָמָה ב׳ – צֶלֶם אֱלֹקִים וְהֲלִיכָה בְּדַרְכָּיו

🪞 צֶלֶם אֱלֹקִים שֶׁרוֹאִים בָּעֵינַיִם

מֵה רָאָה אַבְרָהָם בִּשְׁלוֹשָׁה עֲרָבִים מְאֻבָּקִים? הַמַּהֲרַ״ל: הַכְנָסַת אוֹרְחִים הִיא כְּבוֹד הָאָדָם שֶׁנִּבְרָא בְּצֶלֶם אֱלֹקִים. ׳חָבִיב אָדָם שֶׁנִּבְרָא בְצֶלֶם׳.

לָמָּה גְּדוֹלָה מִקַּבָּלַת פְּנֵי שְׁכִינָה? כִּי אֶת ה׳ אֵין רוֹאִים (׳כִּי לֹא יִרְאַנִי הָאָדָם וָחָי׳), אֲבָל אֶת הַצֶּלֶם שֶׁבָּאוֹרֵחַ רוֹאִים בָּעֵינַיִם, מְכַבְּדִים וּמְחַבְּקִים.

וְדַוְקָא ׳פָּנִים חֲדָשׁוֹת׳: אָדָם שֶׁבָּא לְבֵיתִי לָרִאשׁוֹנָה, כְּאִלּוּ פָּגַשְׁתִּי צֶלֶם אֱלֹקִים חָדָשׁ.

בַּאוֹרֵחַ רוֹאִים אֶת צֶלֶם הָאֱלֹקִים, וּמְכַבְּדִים אוֹתוֹ בְּעֵינַיִם וּבְלֵב.

2
📚

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹדֵק לִפְנֵי שֶׁאֲנִי קָם: מִי עוֹד יָכוֹל לַעֲזֹר כְּמוֹ שֶׁצָּרִיךְ 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, חָפֵץ חַיִּים, אַהֲבַת חֶסֶד ח״ג פֶּרֶק א׳ · בָּבָא מְצִיעָא פ״ו ב׳

רָמָה ב׳ – צֶלֶם אֱלֹקִים וְהֲלִיכָה בְּדַרְכָּיו

📚 מָתַי מַפְסִיקִים לִלְמֹד?

הֶחָפֵץ חַיִּים מְבָאֵר: הַכְנָסַת אוֹרְחִים גְּדוֹלָה גַּם מִלִּמּוּד תּוֹרָה, אֲבָל לֹא תָּמִיד מַפְסִיקִים. יֵשׁ שְׁלוֹשָׁה מַצָּבִים: א. אֵין אַחֵר שֶׁיְּאָרֵחַ – חוֹבָה לְהַפְסִיק וְלָלֶכֶת לָאוֹרְחִים. ב. יֵשׁ אַחֵר שֶׁיְּאָרֵחַ כָּמוֹנִי – אָסוּר לְהַפְסִיק. ג. יֵשׁ אַחֵר אֲבָל לֹא טוֹב כָּמוֹנִי – רְשׁוּת לְהַפְסִיק, וְיֵשׁ בְּכָךְ מַעֲלָה.

אַבְרָהָם הָיָה בַּמַּצָּב הַשְּׁלִישִׁי: הָיוּ בְּבֵיתוֹ אֲחֵרִים, אֲבָל אִישׁ לֹא כִּבֵּד אוֹרְחִים כְּמוֹתוֹ, וְלָכֵן הָלַךְ בְּעַצְמוֹ.

שְׁלוֹשָׁה מַצָּבִים: אֵין אַחֵר, יֵשׁ אַחֵר טוֹב כָּמוֹנִי, יֵשׁ אַחֵר פָּחוֹת טוֹב.

🤝

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִרְאוֹת בְּכָל מַעֲשֵׂה חֶסֶד אֶת רְצוֹן ה׳ 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַשִׁ״י, בְּרֵאשִׁית י״ח ג׳ · חִדּוּשֵׁי פָּרָשַׁת וַיֵּרָא

רָמָה ב׳ – צֶלֶם אֱלֹקִים וְהֲלִיכָה בְּדַרְכָּיו

🤝 אֲדֹנָי, אַל נָא תַעֲבֹר

בְּפָסוּק ג׳ כָּתוּב ׳אֲדֹנָי אִם נָא מָצָאתִי חֵן בְּעֵינֶיךָ אַל נָא תַעֲבֹר מֵעַל עַבְדֶּךָ׳. חֲכָמִים מְפָרְשִׁים שֶׁאַבְרָהָם פָּנָה אֶל ה׳ וְאָמַר: הַמְתֵּן לִי עַד שֶׁאֶכְנִיס אֶת הָאוֹרְחִים. (רַשִׁ״י מֵבִיא גַּם פֵּרוּשׁ שֶׁפָּנָה לְגָדוֹל שֶׁבָּאוֹרְחִים, אֲבָל כָּאן לוֹמְדִים אֶת הַפֵּרוּשׁ הַ׳שֶּׁפּוֹנֶה לַה׳׳.)

מִי שֶׁהוֹלֵךְ בְּדַרְכֵי ה׳ חַי בְּכָל לִבּוֹ שֶׁגַּם מַעֲשֵׂה הַחֶסֶד הוּא רְצוֹן ה׳. אֶצְלוֹ גַּם הַשְּׁכִינָה וְגַם הָאוֹרְחִים הֵם שֵׁם ה׳.

אַבְרָהָם בִּקֵּשׁ מִה׳ לְהַמְתִּין, כִּי הַחֶסֶד הוּא רְצוֹן ה׳.

🏗️

💛 אַהֲבַת הַשֵּׁם: הִזְדַּמְנוּת לְחֶסֶד הִיא עֲבוֹדַת ה׳ בְּעַצְמָהּ 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, חִדּוּשֵׁי פָּרָשַׁת וַיֵּרָא · בָּבָא מְצִיעָא פ״ו ב׳

רָמָה ב׳ – צֶלֶם אֱלֹקִים וְהֲלִיכָה בְּדַרְכָּיו

🏗️ יְסוֹד וְקוֹמוֹת

הֲרֵי ׳תַּלְמוּד תּוֹרָה כְּנֶגֶד כֻּלָּם׳, אֵיךְ הַכְנָסַת אוֹרְחִים גְּדוֹלָה מִמֶּנּוּ? כְּשֶׁמְּדֻבָּר בְּבִטּוּל קָבוּעַ שֶׁל הַתּוֹרָה, הַתּוֹרָה עֲדִיפָה. אֲבָל מִצְוָה חַד-פַּעֲמִית שֶׁבָּאָה לְיָדוֹ – ׳לְעוֹלָם יַעֲסֹק אָדָם בְּתוֹרָה וּבִגְמִילוּת חֲסָדִים׳.

בִּנְיָן גָּבוֹהַּ צָרִיךְ יְסוֹדוֹת חֲזָקִים, וְהַחֶסֶד הוּא הַיְּסוֹד שֶׁעָלָיו עוֹמְדוֹת כָּל הַקּוֹמוֹת.

כְּשֶׁבָּאָה הִזְדַּמְנוּת לְחֶסֶד, לֹא רוֹאִים בָּהּ הַפְרָעָה לַעֲבוֹדַת ה׳, אֶלָּא עֲבוֹדַת ה׳ עַצְמָהּ.

רָמָה ב׳ – צֶלֶם אֱלֹקִים וְהֲלִיכָה בְּדַרְכָּיו

מָה לוֹקְחִים מִרָמָה ב׳?

אַבְרָהָם לֹא עָזַב אֶת ה׳ כְּשֶׁרָץ אֶל הָאוֹרְחִים, הוּא הָלַךְ בְּדַרְכָּיו. כְּשֶׁאֲנִי מְכַבֵּד אָדָם, אֲנִי מְכַבֵּד אֶת צֶלֶם הָאֱלֹקִים שֶׁבּוֹ, וְאוֹהֵב אֶת מִי שֶׁבְּרָאוֹ.

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי הִרְגַּשְׁתִּי שֶׁלַּעֲזֹר לְאַחֵר הוּא גַּם עֲבוֹדַת ה׳?
  • מָה אֲנִי עוֹשֶׂה כְּשֶׁקּוֹרְאִים לִי מִמַּשֶּׁהוּ חָשׁוּב לַעֲזֹר – בְּלֵב שָׁלֵם אוֹ בְּלֵב חָצוּי?
  • הַיּוֹם אֶדַּמֶּה שֶׁכָּל אוֹרֵחַ הוּא בֶּן מֶלֶךְ, אָכִין לוֹ שְׁלוֹשָׁה דְּבָרִים, וְאֶרְשֹׁם מָה הִרְגַּשְׁתִּי.

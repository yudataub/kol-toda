⚖️ בִּשְׁנֵי הֲבָדִלִים

הַגְּמָרָא בְּסֻכָּה אוֹמֶרֶת שֶׁבִּשְׁלֹשָׁה דְּבָרִים גְּמִילוּת חֲסָדִים גְּדוֹלָה מִן הַצְּדָקָה. שְׁנַיִם מֵהֶם: ׳צְדָקָה בְּמָמוֹנוֹ, גְּמִילוּת חֲסָדִים בֵּין בְּגוּפוֹ בֵּין בְּמָמוֹנוֹ. צְדָקָה לַעֲנִיִּים, גְּמִילוּת חֲסָדִים בֵּין לַעֲנִיִּים בֵּין לָעֲשִׁירִים׳.

לְפִי הַפְּשָׁט זֶה הֶבְדֵּל שֶׁל כַּמּוּת: חֶסֶד אֶפְשָׁר לַעֲשׂוֹת בְּיוֹתֵר אֳפָנִים.

הֶבְדֵּל שֶׁל כַּמּוּת: חֶסֶד אֶפְשָׁר לַעֲשׂוֹת בְּיוֹתֵר אֳפָנִים.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁהַחֶסֶד שֶׁלִּי יָבוֹא מִלֵּב טוֹב 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרַב חַיִּים פְּרִידְלַנְדֶּר · נְתִיבוֹת עוֹלָם לַמַּהַרַ״ל

רָמָה ב׳ – מִצַּד הַנּוֹתֵן

❤️ הַמַּהַרַ״ל: הֶבְדֵּל שֶׁל אֵיכוּת

הָרַב חַיִּים פְּרִידְלַנְדֶּר מְבָאֵר אֶת דִּבְרֵי הַמַּהַרַ״ל: הַהֶבְדֵּל הוּא בָּאֹפֶן שֶׁל הַמַּעֲשֶׂה.

צְדָקָה בָּאָה לְמַלֵּא אֶת חִסְרוֹנוֹ שֶׁל הֶעָנִי. אִם הַמְקַבֵּל אֵינוֹ נִצְרָךְ, אֵין כָּאן צְדָקָה.

חֶסֶד בָּא מִלֵּב טוֹב שֶׁל הַנּוֹתֵן, שֶׁמְּחַפֵּשׂ אֵיךְ לְהַשְׁפִּיעַ מִטּוּבוֹ. לָכֵן אֶפְשָׁר לַעֲשׂוֹת חֶסֶד גַּם לְעָשִׁיר, כִּי הַחֶסֶד הוּא מִצַּד הַנּוֹתֵן.

צְדָקָה מִצַּד הַמְקַבֵּל, חֶסֶד מִצַּד הַנּוֹתֵן.

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂם לֵב לָאֹפֶן שֶׁבּוֹ אֲנִי עוֹזֵר 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרַב חַיִּים פְּרִידְלַנְדֶּר

רָמָה ב׳ – מִצַּד הַנּוֹתֵן

😊 בְּחֶסֶד חָשׁוּב אֵיךְ

מִכָּאן: סֵבֶר פָּנִים יָפוֹת הוּא חוֹבָה מֵעִקַּר מִדַּת הַחֶסֶד.

בְּעֲשִׂיַּת חֶסֶד חָשׁוּב אֵיךְ עוֹשִׂים אוֹתוֹ יוֹתֵר מִמָּה שֶׁעוֹשִׂים. הַלֵּב שֶׁמַּכְנִיסִים לַמַּעֲשֶׂה הוּא הָעִקָּר.

כְּמוֹ שֶׁבְּמִצְוֹת שֶׁבֵּין אָדָם לַמָּקוֹם הָעִקָּר הוּא כַּוָּנַת הַלֵּב, כָּךְ בֵּין אָדָם לַחֲבֵרוֹ: הַמְקַבֵּל צָרִיךְ לְהַרְגִּישׁ שֶׁאִכְפַּת לִי מִמֶּנּוּ.

בְּחֶסֶד חָשׁוּב אֵיךְ עוֹשִׂים, יוֹתֵר מִמָּה עוֹשִׂים.

💛 אַהֲבַת הַשֵּׁם: אֲנִי נוֹתֵן בְּשִׂמְחָה וּבְפָנִים טוֹבוֹת 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שֻׁלְחָן עָרוּךְ יוֹרֶה דֵּעָה רמ״ט ג׳

רָמָה ב׳ – מִצַּד הַנּוֹתֵן

💰 צְדָקָה בְּסֵבֶר פָּנִים יָפוֹת

בַּשֻּׁלְחָן עָרוּךְ כָּתוּב: ׳צָרִיךְ לָתֵת אֶת הַצְּדָקָה בְּסֵבֶר פָּנִים יָפוֹת, בְּשִׂמְחָה וּבְטוֹב לֵבָב, וּמִצְטַעֵר עִם הֶעָנִי בְּצַעֲרוֹ וּמְדַבֵּר לוֹ דִּבְרֵי תַּנְחוּמִין. וְאִם נְתָנָהּ בְּפָנִים זְעוּפוֹת וְרָעוֹת, הִפְסִיד זְכוּתוֹ׳.

אֲפִלּוּ בִּצְדָקָה, שֶׁבָּאָה ׳מִצַּד הַמְקַבֵּל׳, פָּנִים זְעוּפוֹת מַפְסִידוֹת אֶת הַזְּכוּת. כֶּסֶף לְבַדּוֹ אֵינוֹ מַסְפִּיק.

גַּם בִּצְדָקָה נוֹתְנִים בְּפָנִים יָפוֹת, וְאֶת הַזְּכוּת מַפְסִידִים בְּפָנִים זְעוּפוֹת.

💛 אַהֲבַת הַשֵּׁם: אֲנִי נוֹתֵן לַחֲבֵרִים תְּשׁוּמֶת לֵב, וְזֶה הַדָּבָר הֲכִי חָשׁוּב 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׂפְתֵי חַיִּים

רָמָה ב׳ – מִצַּד הַנּוֹתֵן

🎀 הַמַּתָּנָה הִיא הֶעָרָכָה

בְּסֵפֶר שִׂפְתֵי חַיִּים מוּסָבָר מַדּוּעַ הַנּוֹתֵן מַתָּנוֹת טוֹבוֹת בְּפָנִים כְּבוּשׁוֹת נֶחְשָׁב ׳כְּאִלּוּ לֹא נָתַן כְּלוּם׳: עִקַּר הַמַּתָּנָה הוּא לְבַטֵּא אֶת הַהֶעָרָכָה וְהַחִבָּה שֶׁיֵּשׁ לָאָדָם כְּלַפֵּי חֲבֵרוֹ.

מִי שֶׁנּוֹתֵן בְּלִי סֵבֶר פָּנִים יָפוֹת, הַמְקַבֵּל אֵינוֹ מַרְגִּישׁ שׁוּם הֶעָרָכָה.

תְּשׁוּמֶת לֵב הִיא הַמּוּצָר הֲכִי חָשׁוּב: לְהַרְאוֹת לַשֵּׁנִי שֶׁהוּא חָשׁוּב בְּעֵינַי, וְשֶׁאֲנִי חוֹשֵׁב עָלָיו.

תְּשׁוּמֶת לֵב הִיא הַמַּתָּנָה הַחֲשׁוּבָה בְּיוֹתֵר.

🌿 רַשְׁבִּ״י מַבִּיט פְּנִימָה

כְּשֶׁיָּצָא רַשְׁבִּ״י מֵהַמְּעָרָה, דַּעְתּוֹ לֹא הִתְיַשְּׁבָה עַד שֶׁרָאָה יְהוּדִי רָץ עִם שְׁנֵי הֲדַסִּים לִכְבוֹד שַׁבָּת. אָמַר: ׳רְאֵה כַּמָּה חֲבִיבִין מִצְוֹת עַל יִשְׂרָאֵל׳.

רַשְׁבִּ״י מַבִּיט תָּמִיד אֶל הַפְּנִימִיּוּת. מַעֲשֵׂה מִצְוָה לְבַדּוֹ לֹא הִשְׁקִיט אוֹתוֹ, אֲבָל חִבָּה לַמִּצְוָה כֵּן.

כָּךְ גַּם בְּחֶסֶד: הוּא דֶּרֶךְ לְהַרְאוֹת כַּמָּה אֲנִי מְחַבֵּב אֶת הַשֵּׁנִי.

מַעֲשֶׂה לְבַדּוֹ לֹא מַסְפִּיק, צָרִיךְ גַּם חִבָּה מֵהַלֵּב.

מָה לוֹקְחִים מִרָמָה ב׳?

מָה פֵּרוּשׁ ׳בְּכָל קָרְאֵנוּ אֵלָיו׳, אֵיךְ מַרְגִּישִׁים שֶׁה׳ קָרוֹב, וְלָמָּה כְּדַאי לִקְרֹא לוֹ כָּל יוֹם

📖 תְּפִלָּה מִן הַלֵּב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דְּבָרִים ד׳ ז׳ · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – קִרְבַת אֱלֹהִים

🧩 שְׁנֵי פֵּרוּשִׁים לַפָּסוּק

׳וּמִי גּוֹי גָּדוֹל אֲשֶׁר לוֹ אֱלֹהִים קְרֹבִים אֵלָיו, כַּה׳ אֱלֹהֵינוּ בְּכָל קָרְאֵנוּ אֵלָיו׳.

הַפְּשָׁט: אֶפְשָׁר לִקְרֹא לַה׳ וְהוּא שׁוֹמֵעַ בְּכָל פַּעַם שֶׁקּוֹרְאִים.

הַדְּרָשָׁה: אֵיךְ זוֹכֶה אָדָם שֶׁה׳ יִהְיֶה קָרוֹב אֵלָיו? אִם הוּא מִתְרַגֵּל לִקְרֹא לַה׳ כָּל הַזְּמַן.

וְעוֹד לוֹמְדִים מִכָּאן: יְהוּדִי צָרִיךְ לִחְיוֹת בִּתְחוּשָׁה שֶׁה׳ בֶּאֱמֶת קָרוֹב אֵלָיו.

פֵּרוּשׁ פָּשׁוּט, וּפֵרוּשׁ שֶׁל דְּרָשָׁה: מִתְרַגְּלִים לִקְרֹא וּמַרְגִּישִׁים קִרְבָה.

1
🔭

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹצֵר וּמִתְבּוֹנֵן, וְהַלֵּב שֶׁלִּי מַרְגִּישׁ אוֹתְךָ, ה׳ 💛

📖 תְּפִלָּה מִן הַלֵּב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְסִלַּת יְשָׁרִים, פֶּרֶק י״ט

רָמָה ב׳ – קִרְבַת אֱלֹהִים

🔭 לַעֲמֹד מַמָּשׁ לִפְנֵי ה׳

בִּמְסִלַּת יְשָׁרִים כָּתוּב שֶׁיֵּשׁ שְׁלוֹשָׁה דְּבָרִים לְהִסְתַּכֵּל בָּהֶם כְּשֶׁמִּתְפַּלְּלִים, וְהָרִאשׁוֹן הוּא שֶׁהָאָדָם עוֹמֵד מַמָּשׁ לִפְנֵי הַבּוֹרֵא, נוֹשֵׂא וְנוֹתֵן עִמּוֹ, אַף עַל פִּי שֶׁהָעַיִן לֹא רוֹאָה אוֹתוֹ.

זֶה הַדָּבָר הַקָּשֶׁה בְּיוֹתֵר לְצַיֵּר בַּלֵּב, כִּי הַחוּשִׁים לֹא עוֹזְרִים כְּלָל: לֹא רוֹאִים וְלֹא שׁוֹמְעִים.

אֲבָל מִי שֶׁבַּעַל שֵׂכֶל נָכוֹן, בִּמְעַט הִתְבּוֹנְנוּת וְשִׂימַת לֵב יָכוֹל לִקְבֹּעַ בְּלִבּוֹ אֶת הָאֱמֶת הַזֹּאת.

מִתְבּוֹנְנִים בְּשֵׂכֶל, וְאָז הַלֵּב מַרְגִּישׁ שֶׁעוֹמְדִים לִפְנֵי ה׳.

💎

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְחַפֵּשׂ אֶת הַטּוֹב הָאֲמִתִּי: קִרְבַת ה׳ 💛

📖 תְּפִלָּה מִן הַלֵּב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְסִלַּת יְשָׁרִים, פֶּרֶק א׳ · תְּהִלִּים ע״ג כ״ח

רָמָה ב׳ – קִרְבַת אֱלֹהִים

💎 קִרְבַת אֱלֹהִים – הַטּוֹב הָאֲמִתִּי

בִּמְסִלַּת יְשָׁרִים כָּתוּב שֶׁהַשְּׁלֵמוּת הָאֲמִתִּית הִיא הַדְּבֵקוּת בְּה׳ יִתְבָּרַךְ, וְזֶה מַה שֶּׁאָמַר דָּוִד הַמֶּלֶךְ: ׳וַאֲנִי קִרְבַת אֱלֹהִים לִי טוֹב׳.

רַק זֶה הוּא הַטּוֹב. וְכָל דָּבָר אַחֵר שֶׁבְּנֵי אָדָם חוֹשְׁבִים שֶׁהוּא טוֹב, הוּא בְּעֶצֶם הֶבֶל וָשָׁוְא.

עַל כֵּן לֹא מְדַבְּרִים עִם ה׳ רַק כְּשֶׁיֵּשׁ בְּעָיָה: הָעִקָּר הוּא שֶׁנִּזְכֶּה לִחְיוֹת קָרוֹב אֵלָיו.

קִרְבָה אֶל ה׳ הִיא הַטּוֹב הָאֲמִתִּי, וְהַשְּׁאָר הוּא הֶבֶל.

🧭

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת מִצַּדִּיקֶיךָ, ה׳, וּלְדַבֵּר אִתְּךָ כָּל יוֹם 💛

📖 תְּפִלָּה מִן הַלֵּב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שׁוּלְחָן עָרוּךְ אֹרַח חַיִּים א׳ א׳ (הָרֵמָ״א) · רַבִּי נַחְמָן

רָמָה ב׳ – קִרְבַת אֱלֹהִים

🧭 שִׁוִּיתִי ה׳ לְנֶגְדִּי תָּמִיד

הָרֵמָ״א כָּתַב בַּהַתְחָלָה שֶׁל שׁוּלְחָן עָרוּךְ: ׳שִׁוִּיתִי ה׳ לְנֶגְדִּי תָמִיד׳ – כְּלָל גָּדוֹל בַּתּוֹרָה וּמַעֲלָה גְּדוֹלָה שֶׁל הַצַּדִּיקִים. וְהַגְּרָ״א כָּתַב שֶׁזֹּאת כָּל מַעֲלַת הַצַּדִּיקִים.

אִם זֹאת כָּל כָּךְ גְּדוֹלָה, אֵיךְ זוֹכִים לָהּ? רַבִּי נַחְמָן מִבְּרַסְלֶב גִּלָּה עֵצָה: לְדַבֵּר עִם ה׳ בְּמִלִּים שֶׁלִּי.

עַל זֶה נִלְמַד בַּשִּׁעוּרִים הַבָּאִים בַּסִּדְרָה.

לִזְכֹּר שֶׁה׳ לְנֶגְדִּי זוֹ מַעֲלַת הַצַּדִּיקִים, וְדִבּוּר עִם ה׳ עוֹזֵר לָהּ.

🧵

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִחְיוֹת כָּל יוֹם עִם הַתְּחוּשָׁה שֶׁאַתָּה עִמָּדִי 💛

📖 תְּפִלָּה מִן הַלֵּב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – קִרְבַת אֱלֹהִים

🧵 לֹא רֶגַע אֶחָד – דֶּרֶךְ חַיִּים

הַשִּׁעוּר מַדְגִּישׁ: הַמַּטָּרָה הָעִקָּרִית אֵינָהּ רַק לִפְתֹּר אֶת כָּל הַצָּרוֹת (גַּם זֶה לֹא דָּבָר נוֹרָא), אֶלָּא שֶׁאָדָם יִזְכֶּה לִחְיוֹת קָרוֹב אֶל ה׳.

וְכֵן הַבִּטָּחוֹן: הוּא מִדָּה שֶׁבַּלֵּב כְּמוֹ כָּל הַמִּדּוֹת. זֶה לֹא רֶגַע אֶחָד, אֶלָּא הִתְנַהֲגוּת קְבוּעָה: חַיִּים בִּתְחוּשָׁה שֶׁה׳ דּוֹאֵג לְכָל צְרָכַי, וְשֶׁ׳אַתָּה עִמָּדִי׳.

קִרְבַת ה׳ וּבִטָּחוֹן הֵם דֶּרֶךְ חַיִּים קְבוּעָה, לֹא רֶגַע אֶחָד.

רָמָה ב׳ – קִרְבַת אֱלֹהִים

מָה לוֹקְחִים מִרָמָה ב׳?

ה׳ אִתָּנוּ בְּכָל קָרְאֵנוּ אֵלָיו. מִי שֶׁמִּתְרַגֵּל לִקְרֹא לוֹ וּלְהִתְבּוֹנֵן, מַרְגִּישׁ אֶת הַקִּרְבָה וְאוֹהֵב אֶת ה׳ יוֹתֵר.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי הִרְגַּשְׁתִּי הַשָּׁבוּעַ שֶׁה׳ קָרוֹב אֵלַי?
  • מָה אֹמַר לַה׳ בַּלֵּב כְּשֶׁאַרְגִּישׁ לְבַד אוֹ מְפֻחָד?
  • לִפְנֵי כָּל דָּבָר חָשׁוּב הַשָּׁבוּעַ – שִׁעוּרֵי בַּיִת, אֲרוּחָה אוֹ כְּנִיסָה לַכִּתָּה – אֹמַר בַּלֵּב ׳אַבָּא, אֲנִי כָּאן׳, וְאֶרְשֹׁם כַּמָּה פְּעָמִים הִצְלַחְתִּי.

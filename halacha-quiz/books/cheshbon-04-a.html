רָמָה א׳ – שׁוֹאֲלִים לָמָּה

🍎 אֲנִי אוֹכֵל תַּפּוּחַ – לָמָּה?

יֶלֶד אוֹכֵל תַּפּוּחַ וְשׁוֹאֲלִים אוֹתוֹ: ׳לָמָּה אַתָּה אוֹכֵל?׳

הוּא יָכוֹל לוֹמַר: ׳כִּי טָעִים׳. וְהוּא יָכוֹל לוֹמַר גַּם: ׳כְּדֵי שֶׁיִּהְיֶה לִי כֹּחַ לְהִתְפַּלֵּל וְלִלְמֹד׳.

הָאֲכִילָה אוֹתָהּ אֲכִילָה, אֲבָל הָעִנְיָן אַחֵר: הַפַּעַם הִיא חֵלֶק מִשֵּׁרוּת ה׳.

אוֹתָהּ אֲכִילָה, וְסִבָּה אַחֶרֶת: כְּדֵי שֶׁיִּהְיֶה לִי כֹּחַ.

🔑

💛 אַהֲבַת הַשֵּׁם: אֲנִי זוֹכֵר אוֹתְךָ, ה׳, בְּכָל מַה שֶׁאֲנִי עוֹשֶׂה 💛

📖 חֶשְׁבּוֹן הַנֶּפֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשְׁלֵי ג׳ ו׳

רָמָה א׳ – שׁוֹאֲלִים לָמָּה

🔑 ׳בְּכָל דְּרָכֶיךָ דָעֵהוּ׳

בַּתּוֹרָה כָּתוּב: ׳בְּכָל דְּרָכֶיךָ דָעֵהוּ׳ – בְּכָל מַעֲשֶׂה שֶׁאֲנִי עוֹשֶׂה, אֲנִי זוֹכֵר אֶת ה׳.

לֹא רַק כְּשֶׁאֲנִי מִתְפַּלֵּל, אֶלָּא גַּם כְּשֶׁאֲנִי אוֹכֵל, הוֹלֵךְ, מְשַׂחֵק וְיָשֵׁן.

וְהַפָּסוּק מַבְטִיחַ: ׳וְהוּא יְיַשֵּׁר אֹרְחֹתֶיךָ׳ – ה׳ יַדְרִיךְ אוֹתִי.

זוֹכְרִים אֶת ה׳ בְּכָל מַעֲשֶׂה, וְהוּא מַדְרִיךְ.

❓

💛 אַהֲבַת הַשֵּׁם: אֲנִי שׁוֹאֵל מָה אַתָּה רוֹצֶה מִמֶּנִּי עַכְשָׁו, ה׳ 💛

📖 חֶשְׁבּוֹן הַנֶּפֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – שׁוֹאֲלִים לָמָּה

❓ שְׁאֵלַת הַ׳לָּמָּה׳

לִפְנֵי שֶׁעוֹשִׂים פְּעֻלָּה אוֹמְרִים לְעַצְמֵנוּ שְׁתֵּי שְׁאֵלוֹת קְטַנּוֹת: ׳לָמָּה אֲנִי עוֹשֶׂה אֶת זֶה?׳ וְ׳הַאִם זֶה רְצוֹן ה׳ עַכְשָׁו?׳

לְמָשָׁל, לִפְנֵי שֶׁאֲנִי מְשַׂחֵק: הַאִם הָיָה לִי זְמַן לִלְמֹד וְלַעֲזֹר בַּבַּיִת?

הַשְּׁאֵלָה הַזּוֹ מְכַוֶּנֶת אוֹתִי לַדֶּרֶךְ הַטּוֹבָה.

לִפְנֵי כָּל מַעֲשֶׂה שׁוֹאֲלִים: לָמָּה, וְהַאִם זֶה רְצוֹן ה׳?

3
🛏️

💛 אַהֲבַת הַשֵּׁם: אֲנִי יָשֵׁן כְּדֵי שֶׁיִּהְיֶה לִי כֹּחַ לַעֲבוֹדָתְךָ 💛

📖 חֶשְׁבּוֹן הַנֶּפֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שֻׁלְחָן עָרוּךְ אוֹרַח חַיִּים רל״א

רָמָה א׳ – שׁוֹאֲלִים לָמָּה

🛏️ לָמָּה יְשֵׁנִים?

אֲנִי הוֹלֵךְ לִישֹׁן. לָמָּה? כִּי עָיֵף. אֲבָל אֶפְשָׁר לְכַוֵּן גַּם לָתֵת מְנוּחָה לַגּוּף, שֶׁיִּהְיֶה לִי כֹּחַ וּבְרִיאוּת לִלְמֹד וּלְהִתְפַּלֵּל מָחָר.

כָּךְ גַּם הַשֵּׁנָה נַעֲשֵׂית חֵלֶק מֵעֲבוֹדַת ה׳.

וְהַחֲכָמִים לִמְּדוּ: כָּל מַעֲשֶׂה, גַּם שֵׁנָה, אֶפְשָׁר לַעֲשׂוֹת לְשֵׁם שָׁמַיִם.

גַּם שֵׁנָה יְכוֹלָה לִהְיוֹת לְשֵׁם שָׁמַיִם.

💧

💛 אַהֲבַת הַשֵּׁם: אֲנִי שׁוֹתֶה בְּכַוָּנָה, וְה׳ שָׂמֵחַ בִּי 💛

📖 חֶשְׁבּוֹן הַנֶּפֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שֻׁלְחָן עָרוּךְ אוֹרַח חַיִּים רל״א

רָמָה א׳ – שׁוֹאֲלִים לָמָּה

💧 גַּם כְּשֶׁצָּמֵא וְרָעֵב

חַם בַּחוּץ וַאֲנִי צָמֵא מְאֹד. מוּתָּר לִשְׁתּוֹת! אֲבָל אֶפְשָׁר לִשְׁתּוֹת בְּכַוָּנָה.

חֲכָמֵינוּ לִמְּדוּ: אֲפִלּוּ אָדָם צָמֵא וְרָעֵב, כְּשֶׁהוּא אוֹכֵל וְשׁוֹתֶה כְּדֵי לַעֲבֹד אֶת בּוֹרְאוֹ, זֶה מְשֻׁבָּח.

אֲנִי שׁוֹתֶה, וְאוֹמֵר בְּלִבִּי: ׳אֲנִי שׁוֹתֶה כְּדֵי שֶׁיִּהְיֶה לִי כֹּחַ לַעֲשׂוֹת אֶת רְצוֹן ה׳׳.

גַּם כְּשֶׁצָּמֵאִים אֶפְשָׁר לִשְׁתּוֹת לְשֵׁם שָׁמַיִם.

5
🤖

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹשֶׂה מִצְווֹת בְּלֵב פָּתוּחַ 💛

📖 חֶשְׁבּוֹן הַנֶּפֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, יְשַׁעְיָהוּ כ״ט יג׳

רָמָה א׳ – שׁוֹאֲלִים לָמָּה

🤖 לֹא מִצְוָה מְלֻמָּדָה

רוֹבּוֹט עוֹשֶׂה פְּעֻלּוֹת בְּלִי לַחְשֹׁב. הוּא פָּשׁוּט מִתְרַגֵּל.

הַנָּבִיא יְשַׁעְיָהוּ הִזְהִיר מִ׳מִּצְוַת אֲנָשִׁים מְלֻמָּדָה׳: לַעֲשׂוֹת רַק מִתּוֹךְ הֶרְגֵּל, כְּשֶׁהַלֵּב רָחוֹק.

מִי שֶׁשּׁוֹאֵל ׳לָמָּה׳ לֹא נַעֲשֶׂה רוֹבּוֹט, וְהַלֵּב שֶׁלּוֹ נִכְנָס לַמַּעֲשֶׂה.

שׁוֹאֲלִים ׳לָמָּה׳ כְּדֵי לֹא לַעֲשׂוֹת מִתּוֹךְ הֶרְגֵּל.

⏸️

💛 אַהֲבַת הַשֵּׁם: אֲנִי נִשְׁאָר רָגוּעַ גַּם כְּשֶׁמַּפְרִיעִים לִי 💛

📖 חֶשְׁבּוֹן הַנֶּפֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – שׁוֹאֲלִים לָמָּה

⏸️ וְאִם מַפְרִיעִים לִי?

לִפְעָמִים בָּאֶמְצַע מַשֶּׁהוּ מַפְרִיעַ: אָח קוֹרֵא לִי, הַטֶּלֶפוֹן מְצַלְצֵל.

מִי שֶׁזּוֹכֵר שֶׁהוּא עוֹשֶׂה מַעֲשֶׂה מִפְּנֵי רְצוֹן ה׳, לֹא כּוֹעֵס עַל הַהַפְרָעָה, כִּי הוּא יוֹדֵעַ שֶׁגַּם זֶה הָרָצוֹן שֶׁל ה׳ עַכְשָׁו.

אֲנִי נוֹשֵׁם, מְחַיֵּךְ, וְשׁוֹאֵל: מָה ה׳ רוֹצֶה מִמֶּנִּי עַכְשָׁו?

לֹא כּוֹעֲסִים עַל הַפְרָעָה: גַּם הִיא רָצוֹן ה׳.

🔔

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹבֵד אוֹתְךָ, ה׳, תָּמִיד 💛

📖 חֶשְׁבּוֹן הַנֶּפֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שֻׁלְחָן עָרוּךְ אוֹרַח חַיִּים רל״א

רָמָה א׳ – שׁוֹאֲלִים לָמָּה

🔔 מַזְכִּירִים לְעַצְמֵנוּ

לֹא מַסְפִּיק לִשְׁאֹל ׳לָמָּה׳ רַק בַּהַתְחָלָה. כְּדַאי לְהַזְכִּיר לְעַצְמֵנוּ כַּמָּה פְּעָמִים בִּזְמַן הַמַּעֲשֶׂה: ׳אֲנִי עוֹשֶׂה אֶת זֶה כִּי ה׳ רוֹצֶה׳.

כָּךְ הַמַּעֲשֶׂה נִשְׁאָר קָרוֹב אֶל הַלֵּב.

וּמִי שֶׁנּוֹהֵג כָּךְ עוֹבֵד אֶת בּוֹרְאוֹ תָּמִיד.

מַזְכִּירִים לְעַצְמֵנוּ גַּם בְּאֶמְצַע הַמַּעֲשֶׂה.

💡

רָמָה א׳ – שׁוֹאֲלִים לָמָּה

מָה לוֹקְחִים מִרָמָה א׳?

כְּשֶׁאֲנִי שׁוֹאֵל ׳לָמָּה אֲנִי עוֹשֶׂה אֶת זֶה׳, אֲנִי מְחַבֵּר אֶת כָּל הַיּוֹם שֶׁלִּי אֶל ה׳. כָּךְ גַּם אֲרוּחָה וְשֵׁנָה נַעֲשׂוֹת דֶּרֶךְ לְהִתְקָרֵב אֵלָיו.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה הַסִּבָּה שֶׁאֲנִי אוֹכֵל אֶת אֲרוּחַת הַבֹּקֶר, וְאֵיךְ אֶפְשָׁר לְחַבֵּר אוֹתָהּ לַה׳?
  • בְּאֵיזֶה דָּבָר שֶׁאֲנִי עוֹשֶׂה כָּל יוֹם אֲנִי עוֹשֶׂה בְּלִי לַחְשֹׁב, וְאֵיךְ אֶשְׁאַל ׳לָמָּה׳?
  • הַיּוֹם אֲשַׂחֵק עִם מִישֶׁהוּ בְּמִשְׂחָק ׳הַסִּבָּה׳: אֹמַר ׳אֲנִי עוֹשֶׂה … כְּדֵי …׳ שְׁלוֹשָׁה פְּעָמִים בַּיּוֹם, וְאֶרְשֹׁם מָה הִרְגַּשְׁתִּי.

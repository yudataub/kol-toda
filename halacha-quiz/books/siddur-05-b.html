רָמָה ב׳ – עוֹמֶק הַמִּלִּים

📜 בִּרְכַּת הַשִּׁיר

הַגְּמָרָא בְּמַסֶּכֶת פְּסָחִים קוֹרֵאת לַ׳נִּשְׁמַת כָּל חַי׳ ׳בִּרְכַּת הַשִּׁיר׳, כִּי הִיא שִׁיר שֶׁל הוֹדָאָה לַה׳. בַּשַּׁבָּת אוֹמְרִים אוֹתָהּ אַחֲרֵי פְּסוּקֵי דְזִמְרָה.

הָרֹקֵחַ מוֹסִיף: אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה תִּקְּנוּ אוֹתָהּ כְּדֵי שֶׁנְּשַׁבֵּחַ אֶת ה׳ אַחֲרֵי הַשֵּׁנָה, עַל הַנְּשָׁמָה שֶׁחָזְרָה אֵלֵינוּ. אַחַר כָּךְ תִּקְּנוּ גַּם אֶת ׳אֱלֹהַי נְשָׁמָה שֶׁנָּתַתָּ בִּי׳ וְאֶת ׳הַמַּעֲבִיר שֵׁנָה מֵעֵינָי׳.

הַנִּשְׁמַת הִיא שִׁיר תּוֹדָה עַל כָּךְ שֶׁקַּמְנוּ וְהַנְּשָׁמָה חָזְרָה.

1

💛 אַהֲבַת הַשֵּׁם: גַּם בְּיוֹם קָשֶׁה אֲנִי זוֹכֵר שֶׁה׳ לֹא הִפְסִיק לִשְׁמֹר עָלַי 💛

📖 פִּרְקֵי הַתְּפִלָּה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּנֵי יִשָּׂשׂכָר · רַד״ק תְּהִלִּים קכ״א

רָמָה ב׳ – עוֹמֶק הַמִּלִּים

🔦 לֹא יָנוּם וְלֹא יִישָׁן – גַּם כְּשֶׁנִּדְמֶה אַחֶרֶת

הַ׳בְּנֵי יִשָּׂשׂכָר׳ מְבָאֵר אֶת שְׁתֵּי הַמִּלִּים: ׳לֹא יָנוּם׳ – גַּם כְּשֶׁנִּדְמֶה לָנוּ בִּשְׁעַת צָרָה שֶׁה׳ רַק מִתְנַמְנֵם, הוּא מַשְׁגִּיחַ עָלֵינוּ. ׳וְלֹא יִישָׁן׳ – גַּם כְּשֶׁה׳ נִסְתָּר מֵאִתָּנוּ וַאֲנַחְנוּ ׳יְשֵׁנִים׳ מִן הַמִּצְווֹת, הוּא לֹא מַפְסִיק לִשְׁמֹר עָלֵינוּ.

הָרַד״ק מוֹסִיף: ה׳ שׁוֹמֵר אֶת יִשְׂרָאֵל בְּכָל עֵת, גַּם בַּגָּלוּת, וְלֹא נוֹתֵן לָאוֹיְבִים לַעֲשׂוֹת בָּנוּ כָּלָה.

גַּם כְּשֶׁנִּדְמֶה שֶׁה׳ רָחוֹק – הוּא קָרוֹב, עֵר וְשׁוֹמֵר.

2

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי רוֹצֶה לִהְיוֹת טוֹב יוֹתֵר, אֲנִי יוֹדֵעַ שֶׁה׳ הוּא שֶׁמְּעוֹרֵר אוֹתִי 💛

📖 פִּרְקֵי הַתְּפִלָּה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סֵפֶר הַיָּשָׁר שַׁעַר ט׳ · דִּבְרֵי אֱלִימֶלֶךְ

רָמָה ב׳ – עוֹמֶק הַמִּלִּים

⏰ הַמְעוֹרֵר יְשֵׁנִים – גַּם בַּלֵּב

לֹא רַק שֵׁנָה שֶׁל לַיְלָה. סֵפֶר הַיָּשָׁר מְלַמֵּד: מִי שֶׁהִתְעוֹרֵר מִשְּׁנָתוֹ הָרוּחָנִית וְעָזַב מַעֲשִׂים לֹא טוֹבִים – זֶה סִימָן לְאַהֲבַת ה׳ אֵלָיו, כִּי הוּא מַחֲלִיף לוֹ לֵב אֶבֶן בְּלֵב בָּשָׂר.

ה׳דִּבְרֵי אֱלִימֶלֶךְ׳ מִגְּרָאדְזִיסְק הִתְכַּוֵּן, כְּשֶׁאָמַר ׳מְעוֹרֵר יְשֵׁנִים׳, שֶׁה׳ יְעוֹרֵר אֶת נִשְׁמַת יִשְׂרָאֵל שֶׁנִּרְדְּמָה מִן הַקְּדֻשָּׁה.

וְ׳הַמֵּקִיץ נִרְדָּמִים׳ – אֲפִלּוּ מִי שֶׁהוּא כְּמוֹ יָשֵׁן עָמֹק, ה׳ מֵקִיץ אוֹתוֹ בְּאַהֲבָה.

כְּשֶׁהַלֵּב מִתְעוֹרֵר לַחְזֹר לְה׳ – זוֹ מַתָּנָה שֶׁל אַהֲבָה.

3

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁמִּישֶׁהוּ עוֹזֵר לִי, אֲנִי אוֹמֵר תּוֹדָה לוֹ וְתוֹדָה לְךָ, ה׳ 💛

📖 פִּרְקֵי הַתְּפִלָּה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בֵּאוּר הָרַב שְׁטֵינְהוֹיז

רָמָה ב׳ – עוֹמֶק הַמִּלִּים

🙏 וּלְךָ לְבַדְּךָ אֲנַחְנוּ מוֹדִים

אַחֲרֵי כָּל הַמַּתָּנוֹת אוֹמְרִים: ׳וּלְךָ לְבַדְּךָ אֲנַחְנוּ מוֹדִים׳. הָרַב שְׁטֵינְהוֹיז מְבָאֵר: גַּם כְּשֶׁבְּנֵי אָדָם מֵיטִיבִים לָנוּ, נִזְכֹּר שֶׁהַכֹּל מִמְּךָ לְבַדְּךָ, וְנוֹדֶה לַה׳ עַל הַטּוֹבָה שֶׁקִּבַּלְנוּ עַל יְדֵיהֶם.

וְגַם נַכִּיר טוֹבָה לַשְּׁלִיחִים – כִּי ה׳ רָצָה לְזַכּוֹת אוֹתָנוּ לְהוֹדוֹת לָהֶם, וּבָחַר בָּהֶם לִהְיוֹת הַשְּׁלִיחִים שֶׁלּוֹ.

מוֹדִים לַה׳ וְגַם לַשָּׁלִיחַ שֶׁהוּא שָׁלַח.

4

💛 אַהֲבַת הַשֵּׁם: גַּם הַיּוֹם ה׳ זָן אוֹתִי וְשָׁמַר עָלַי, וְאֲנִי אוֹמֵר תּוֹדָה 💛

📖 פִּרְקֵי הַתְּפִלָּה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, נֻסַּח הַתְּפִלָּה · בֵּאוּר הָרַב שְׁטֵינְהוֹיז

רָמָה ב׳ – עוֹמֶק הַמִּלִּים

🌊 עַד הֵנָּה עֲזָרוּנוּ רַחֲמֶיךָ

הַנִּשְׁמַת מוֹנָה אֶת הַנִּסִּים: ׳מִמִּצְרַיִם גְּאַלְתָּנוּ, וּמִבֵּית עֲבָדִים פְּדִיתָנוּ, בְּרָעָב זַנְתָּנוּ וּבְשָׂבָע כִּלְכַּלְתָּנוּ, מֵחֶרֶב הִצַּלְתָּנוּ, וּמִדֶּבֶר מִלַּטְתָּנוּ׳. וְאַחַר כָּךְ: ׳עַד הֵנָּה עֲזָרוּנוּ רַחֲמֶיךָ, וְלֹא עֲזָבוּנוּ חֲסָדֶיךָ׳ – עַד הַיּוֹם הַזֶּה.

הָרַב שְׁטֵינְהוֹיז מְבָאֵר: מְסַפְּרִים עַל נִסֵּי מִצְרַיִם כְּדֵי לִזְכֹּר אֶת הַחֶסֶד וְהָרַחֲמִים שֶׁהִתְגַּלּוּ אָז, וּלְהַכִּיר אוֹתָם גַּם בַּחַיִּים שֶׁלָּנוּ.

הַנִּסִּים שֶׁל פַּעַם מְלַמְּדִים אוֹתָנוּ לְהַכִּיר אֶת הַחֶסֶד שֶׁל הַיּוֹם.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת יָשָׁר וְחָסִיד, וְלָשִׁיר לְךָ, ה׳, בַּלֵּב 💛

📖 פִּרְקֵי הַתְּפִלָּה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרֹקֵחַ · יִשְׂמַח יִשְׂרָאֵל

רָמָה ב׳ – עוֹמֶק הַמִּלִּים

🎼 מִי הֵם יְשָׁרִים, צַדִּיקִים, חֲסִידִים וּקְדוֹשִׁים

הָרֹקֵחַ מְפָרֵשׁ אֶת אַרְבַּע הַקְּבוּצוֹת שֶׁבַּסּוֹף: יְשָׁרִים – יְשָׁרִים בְּלִבָּם וּבֶאֱמֶת. צַדִּיקִים – עוֹשִׂים צֶדֶק וְלֹא לוֹקְחִים מֵאִישׁ דָּבָר. חֲסִידִים – מִי שֶׁשּׁוֹמֵעַ חֶרְפָּתוֹ וְאֵינוֹ מֵשִׁיב. קְדוֹשִׁים – אֵלֶּה שֶׁהֵם פְּרוּשִׁים.

׳לַיְשָׁרִים נָאוָה תְהִלָּה׳: ה׳ בְּרוֹב רַחֲמָיו מְקַבֵּל תְּשׁוּבָה, וְגַם מִי שֶׁהִתְעַקֵּם נַעֲשֶׂה יָשָׁר עַל יְדֵי תְּשׁוּבָה (יִשְׂמַח יִשְׂרָאֵל).

חָסִיד אֲמִתִּי יוֹדֵעַ לִשְׁמֹעַ וְלֹא לְהָשִׁיב.

מָה לוֹקְחִים מִרָמָה ב׳?

ה׳ לֹא יָשֵׁן וְלֹא מַפְסִיק לִשְׁמֹר עָלֵינוּ, וְהוּא מְעוֹרֵר אֶת הַלֵּב לַחְזֹר אֵלָיו בְּאַהֲבָה. מִי שֶׁמַּרְגִּישׁ אֶת הַחֶסֶד הַזֶּה מוֹדֶה לַה׳ וְגַם לְכָל מִי שֶׁעוֹזֵר לוֹ, וְשׁוֹמֵעַ וְלֹא מֵשִׁיב כְּדֶרֶךְ הַחֲסִידִים.

🤔 חוֹשְׁבִים עַל זֶה

  • אֵיזוֹ מַתָּנָה מֵה׳ – מֵהַפֶּה וְהָעֵינַיִם וְעַד הַיָּדַיִם וְהָרַגְלַיִם – אֲנִי רוֹצֶה לְהוֹדוֹת עָלֶיהָ הַיּוֹם?
  • מָתַי הִרְגַּשְׁתִּי שֶׁה׳ שׁוֹמֵר עָלַי וְלֹא נִרְדָּם, וְאֵיךְ אֲנִי יָכוֹל לִזְכֹּר אֶת זֶה בְּיוֹם קָשֶׁה?
  • מִי עָזַר לִי הַשָּׁבוּעַ, וְאֵיךְ אֲנִי אוֹמֵר תּוֹדָה גַּם לוֹ וְגַם לַה׳ שֶׁשָּׁלַח אוֹתוֹ?

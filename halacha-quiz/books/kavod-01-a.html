מַהוּ כָּבוֹד?

רָמָה א׳ – לְכַבֵּד מֵהַלֵּב

כָּבוֹד מַתְחִיל בַּלֵּב: כָּל אָדָם בֵּן שֶׁל מֶלֶךְ, וּבְכָל אֶחָד יֵשׁ מַעֲלָה שֶׁרָאוּי לְהַעֲרִיךְ

📖 כְּבוֹד הַבְּרִיּוֹת · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
🚪

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְכַבֵּד בֶּאֱמֶת, מִן הַלֵּב, כְּמוֹ שֶׁה׳ רוֹצֶה 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – לְכַבֵּד מֵהַלֵּב

🚪 פּוֹתְחִים דֶּלֶת – וְזֶה הַכֹּל?

דָּנִי פּוֹתֵחַ אֶת הַדֶּלֶת לְחָבֵר מֵהַכִּתָּה, אֲבָל בַּלֵּב הוּא חוֹשֵׁב: ׳הוּא לֹא חָשׁוּב בִּשְׁבִילִי׳. הַאִם דָּנִי נָתַן כָּבוֹד?

הַרְבֵּה אֲנָשִׁים חוֹשְׁבִים שֶׁכָּבוֹד הוּא רַק מַעֲשִׂים: לָקוּם, לִפְתֹּחַ דֶּלֶת, לְדַבֵּר יָפֶה.

אֲבָל כָּבוֹד אֲמִתִּי מַתְחִיל בַּלֵּב, בְּמַחֲשָׁבָה וּבְהַרְגָּשָׁה, וְאַחַר כָּךְ מַגִּיעִים גַּם הַמַּעֲשִׂים.

כָּבוֹד אֵינוֹ רַק מַעֲשִׂים – הוּא מַתְחִיל בַּלֵּב.

1
💭

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד לְהַעֲרִיךְ אֶת הַבְּרִיּוֹת שֶׁה׳ בָּרָא, וְכָךְ אֲנִי מִתְקָרֵב אֵלָיו 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – לְכַבֵּד מֵהַלֵּב

💭 כָּבוֹד פֵּרוּשׁוֹ – לְהַעֲרִיךְ

הַמּוֹרֶה נִכְנָס לַכִּתָּה, וּשְׁנֵי יְלָדִים קָמִים. הָאֶחָד קָם כִּי הוּא מַעֲרִיךְ אֶת הַמּוֹרֶה וּמֵבִין כַּמָּה הוּא חָשׁוּב. הַשֵּׁנִי קָם רַק כִּי כֻּלָּם קָמִים.

הַמַּעֲשֶׂה אוֹתוֹ מַעֲשֶׂה, אֲבָל הַכָּבוֹד הָאֲמִתִּי הוּא שֶׁל הָרִאשׁוֹן.

לְכַבֵּד אֶת הַשֵּׁנִי פֵּרוּשׁוֹ: לְהָבִין בַּלֵּב שֶׁיֵּשׁ בּוֹ עֵרֶךְ וּמַעֲלָה, וּלְהַעֲרִיךְ אוֹתוֹ.

לְכַבֵּד אֶת הַשֵּׁנִי = לְהַעֲרִיךְ אוֹתוֹ בַּלֵּב.

2
👑

💛 אַהֲבַת הַשֵּׁם: אֲנִי יוֹדֵעַ שֶׁכָּל אָדָם הוּא יְצִירָה שֶׁל ה׳, וְזֶה מוֹסִיף לְאַהֲבָתִי אֵלָיו 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁמוּאֵל א׳ ב׳ ל׳

רָמָה א׳ – לְכַבֵּד מֵהַלֵּב

👑 כָּל אָדָם – בֵּן שֶׁל מֶלֶךְ

דִּמְיְנוּ שֶׁבֵּן שֶׁל מֶלֶךְ הוֹלֵךְ בָּרְחוֹב בִּבְגָדִים פְּשׁוּטִים. מִי שֶׁיֵּדַע מִי אָבִיו, יְכַבֵּד אוֹתוֹ מִיָּד.

כָּל אָדָם הוּא בְּרִיָּה שֶׁל בּוֹרֵא הָעוֹלָם, כְּמוֹ בֵּן שֶׁל הַמֶּלֶךְ. לָכֵן יֵשׁ מִצְוָה לְכַבֵּד כָּל אָדָם.

כָּתוּב: ׳כִּי מְכַבְּדַי אֲכַבֵּד׳ – מִי שֶׁמְּכַבֵּד אֶת בְּרִיּוֹתָיו שֶׁל ה׳, מְכַבֵּד אוֹתוֹ.

כָּל אָדָם הוּא בֵּן שֶׁל מֶלֶךְ, וְלָכֵן מְכַבְּדִים אוֹתוֹ.

3
🧱

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְכַבֵּד גַּם מִי שֶׁטָּעָה, כִּי כֻּלָּנוּ יְלָדִים שֶׁל אָבִינוּ שֶׁבַּשָּׁמַיִם 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – לְכַבֵּד מֵהַלֵּב

🧱 הַיְּסוֹד שֶׁתָּמִיד קַיָּם

יֵשׁ שֶׁאוֹמְרִים: ׳אֲכַבֵּד אוֹתוֹ רַק אִם יַצְלִיחַ׳. אֲבָל אִם הַכָּבוֹד תָּלוּי רַק בְּמַה שֶׁהוּא עוֹשֶׂה, בַּיּוֹם שֶׁלֹּא יַצְלִיחַ לֹא יִשָּׁאֵר כָּבוֹד.

חָבֵר הִפִּיל אֶת הַמַּגָּשׁ בַּחֲדַר הָאֹכֶל וְכֻלָּם הִסְתַּכְּלוּ עָלָיו. הוּא עֲדַיִן בֵּן שֶׁל מֶלֶךְ!

הַיְּסוֹד הָרִאשׁוֹן שֶׁל הַכָּבוֹד קַיָּם תָּמִיד, גַּם כְּשֶׁטּוֹעִים וְגַם כְּשֶׁלֹּא מַצְלִיחִים.

הַכָּבוֹד לְכָל אָדָם לֹא תָּלוּי בְּהַצְלָחָה.

4
⭐

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַאֲמִין שֶׁה׳ שָׂם מַעֲלָה טוֹבָה בְּכָל אָדָם, וַאֲנִי אֲחַפֵּשׂ אוֹתָהּ 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אָבוֹת ד׳ ג׳

רָמָה א׳ – לְכַבֵּד מֵהַלֵּב

⭐ לְכָל אָדָם יֵשׁ שָׁעָה

בַּכִּתָּה יֵשׁ יֶלֶד שָׁקֵט שֶׁכִּמְעַט לֹא שׁוֹמְעִים אוֹתוֹ. אֲבָל כְּשֶׁמִּתְבּוֹנְנִים, רוֹאִים שֶׁהוּא מְצַיֵּר נֶהְדָּר וְתָמִיד זוֹכֵר לְהוֹדוֹת.

חֲכָמֵינוּ לִמְּדוּ: ׳אֵין לְךָ אָדָם שֶׁאֵין לוֹ שָׁעָה׳ – לְכָל אָדָם יֵשׁ רֶגַע שֶׁבּוֹ הוּא מַצְלִיחַ וּמֵאִיר.

זֶה הַיְּסוֹד הַשֵּׁנִי שֶׁל כָּבוֹד: מְכַבְּדִים אֶת הַמַּעֲלָה הַמְיֻחֶדֶת שֶׁיֵּשׁ בְּכָל אֶחָד.

בְּכָל אָדָם יֵשׁ מַעֲלָה, וְצָרִיךְ רַק לְחַפֵּשׂ אוֹתָהּ.

5
🔍

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְחַפֵּשׂ טוֹב בַּחֲבֵרִים, וְכָךְ אֲנִי לוֹמֵד לִרְאוֹת אֶת מַעֲשֵׂי ה׳ בָּעוֹלָם 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – לְכַבֵּד מֵהַלֵּב

🔍 אֵיךְ מוֹצְאִים מַעֲלָה?

בְּשָׁלוֹשׁ פְּעֻלּוֹת קְטַנּוֹת: מַסְתַּכְּלִים עַל הֶחָבֵר בְּלִי לִשְׁפֹּט אוֹתוֹ מֵרֹאשׁ, מְחַפְּשִׂים מַה טּוֹב בּוֹ, וְאוֹמְרִים לוֹ.

הַמַּעֲלָה יְכוֹלָה לִהְיוֹת קְטַנָּה: הוּא מַקְשִׁיב, הוּא מְחַיֵּךְ, הוּא עוֹזֵר.

כָּל אָדָם, אִם רַק מְחַפְּשִׂים, יֵשׁ לוֹ מַעֲלָה. מִי שֶׁמְּחַפֵּשׂ – מוֹצֵא.

מַסְתַּכְּלִים, מְחַפְּשִׂים מַעֲלָה, וְאוֹמְרִים אוֹתָהּ.

6
🌟

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַעֲרִיךְ אֶת ה׳ בְּלִבִּי, כִּי הוּא בָּרָא אֶת הָעוֹלָם וְטוֹב לַכֹּל 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – לְכַבֵּד מֵהַלֵּב

🌟 וּמָה עִם כָּבוֹד לַה׳?

אֶפְשָׁר לִשְׁאֹל: הַאִם חָסֵר לַה׳ מַשֶּׁהוּ? הוּא לֹא צָרִיךְ שֶׁנַּרְגִּישׁ לוֹ שֶׁהוּא חָשׁוּב.

לָכֵן כָּבוֹד אֵינוֹ ׳לָתֵת הַרְגָּשָׁה׳. כָּבוֹד לַה׳ הוּא לְהַכִּיר בַּשֵּׂכֶל וּבַלֵּב שֶׁהוּא בָּרָא אֶת הָעוֹלָם, שׁוֹלֵט בַּכֹּל וְטוֹב לַכֹּל.

גַּם כָּאן כָּבוֹד הוּא הֲבָנָה וְהַעֲרָכָה שֶׁבַּלֵּב.

כָּבוֹד הוּא הֲבָנָה וְהַעֲרָכָה שֶׁבַּלֵּב – גַּם כְּלַפֵּי ה׳.

📝

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה לַה׳ עַל הַמַּעֲלוֹת שֶׁבָּרָא בָּאֲנָשִׁים שֶׁסְּבִיבִי 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – לְכַבֵּד מֵהַלֵּב

📝 מָה לוֹקְחִים מֵהַשִּׁעוּר?

יֵשׁ שְׁנֵי יְסוֹדוֹת לְכָבוֹד: הָרִאשׁוֹן – כָּל אָדָם בֵּן שֶׁל מֶלֶךְ, תָּמִיד. הַשֵּׁנִי – לְכָל אָדָם יֵשׁ מַעֲלָה מְיֻחֶדֶת.

וְהַכָּבוֹד מַתְחִיל בַּלֵּב: מַעֲרִיכִים, וְרַק אַחַר כָּךְ מַגִּיעִים הַמַּעֲשִׂים.

הַיּוֹם נִבְחַר חָבֵר, נִמְצָא בּוֹ מַעֲלָה, וְנֹאמַר לוֹ אוֹתָהּ.

הַיּוֹם: מוֹצְאִים מַעֲלָה בְּחָבֵר וְאוֹמְרִים לוֹ.

רָמָה א׳ – לְכַבֵּד מֵהַלֵּב

מָה לוֹקְחִים מִרָמָה א׳?

כָּבוֹד אֲמִתִּי מַתְחִיל בַּלֵּב, וּבַלֵּב גַּם אַהֲבַת ה׳. מִי שֶׁמַּעֲרִיךְ אֶת בְּרִיּוֹתָיו שֶׁל ה׳, מַעֲרִיךְ גַּם אֶת הַבּוֹרֵא שֶׁבָּרָא אוֹתָם.

🤔 חוֹשְׁבִים עַל זֶה

  • מִי הָאָדָם שֶׁאֲנִי מְכַבֵּד בַּפֶּה אֲבָל לֹא בַּלֵּב, וְאֵיךְ אַעֲרִיךְ אוֹתוֹ בֶּאֱמֶת?
  • מָה הַמַּעֲלָה שֶׁיֵּשׁ בִּי שֶׁאָדָם אֶחָד כְּבָר הֶעֱרִיךְ?
  • אֶכְתֹּב מַעֲלָה אַחַת בְּחָבֵר וְאַרְאֶה לוֹ – וְאֶרְשֹׁם מָה הוּא הִרְגִּישׁ וּמָה הִרְגַּשְׁתִּי אֲנִי.

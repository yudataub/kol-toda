בֵּית הַמִּקְדָּשׁ וְרַבִּי יְהוֹשֻׁעַ, חֶסֶד שֶׁמּוֹצִיא אוֹתִי מֵ׳רַק אֲנִי׳, וּדְרָכִים לִקְנוֹת אַהֲבַת חֶסֶד

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹשֶׂה חֶסֶד מִן הַלֵּב, כִּי ה׳ מְבַקֵּשׁ מֵאִתָּנוּ לֵב טוֹב 💛

📖 חֶסֶד – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, יוֹמָא ט׳ ב׳ · אָבוֹת דְּרַבִּי נָתָן י״ג

רָמָה ב׳ – שְׁמוֹנֶה דְּרָכִים לֶאֱהֹב חֶסֶד

🏛️ הַבַּיִת הַשֵּׁנִי: חֶסֶד בְּלִי לֵב

הַגְּמָרָא אוֹמֶרֶת: בַּבַּיִת הַשֵּׁנִי עָסְקוּ בַּתּוֹרָה, בְּמִצְוֹת וּבִגְמִילוּת חֲסָדִים, וְהוּא חָרַב מִפְּנֵי שִׂנְאַת חִנָּם. אֵיךְ אֶפְשָׁר? אֲנָשִׁים שֶׁעוֹשִׂים חֶסֶד, וְשׂוֹנְאִים זֶה אֶת זֶה?

הַתְּשׁוּבָה: אֶפְשָׁר לַעֲשׂוֹת חֶסֶד בְּלִי שֶׁהַלֵּב יִשְׁתַּנֶּה — כְּשֶׁעוֹשִׂים כִּי צָרִיךְ עוֹד מִצְוָה, וְלֹא כִּי אֲכְפַּת מִן הָאָדָם שֶׁמּוּלִי. מַטָּרַת הַחֶסֶד אֵינָהּ רַק שֶׁלַּשֵּׁנִי יִהְיֶה טוֹב, אֶלָּא שֶׁאֲנִי אֶהְפֹּךְ לְבַעַל מִדּוֹת טוֹבוֹת וְאוֹהֵב יִשְׂרָאֵל.

וְלָכֵן: לִפְנֵי חֶסֶד שׁוֹאֲלִים — אֲנִי עוֹשֶׂה לְ׳סַמֵּן׳, אוֹ כִּי אֲכְפַּת לִי?

חֶסֶד צָרִיךְ לַהֲפֹךְ אוֹתִי לְאָדָם טוֹב, לֹא רַק לְמַלֵּא חוֹבָה.

💛 אַהֲבַת הַשֵּׁם: אֲנִי יוֹצֵא מֵעַצְמִי, וְכָךְ הַלֵּב שֶׁלִּי מִתְקָרֵב לַה׳ 💛

📖 חֶסֶד – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אָבוֹת דְּרַבִּי נָתָן ד׳ ה׳ · מִכְתָּב מֵאֵלִיָּהוּ ד׳

רָמָה ב׳ – שְׁמוֹנֶה דְּרָכִים לֶאֱהֹב חֶסֶד

🔥 לָמָּה הַחֶסֶד מְכַפֵּר?

הַגְּמָרָא שׁוֹאֶלֶת: אִם יֵשׁ לָנוּ כַּפָּרָה בְּגְמִילוּת חֲסָדִים, וְכֻלָּנוּ עוֹשִׂים חֶסֶד כָּל הַיּוֹם — אֵין לָנוּ עֲוֹנוֹת? וְלָמָּה הַחֶסֶד מְכַפֵּר כְּמוֹ מִזְבֵּחַ?

הַתְּשׁוּבָה: מַהוּת הַחֵטְא הִיא אָנוֹכִיּוּת וְחֶסְרוֹן דְּבֵקוּת. כְּשֶׁאָדָם עוֹשֶׂה חֶסֶד הוּא יוֹצֵא מֵעַצְמוֹ. מִי שֶׁמַּקְרִיב מֵעַצְמוֹ נֶחְשָׁב כְּמוֹ קָרְבָּן.

חֶסֶד מְכַפֵּר כִּי הוּא מוֹצִיא אוֹתִי מֵהָאָנוֹכִיּוּת.

🤲

💛 אַהֲבַת הַשֵּׁם: אֲנִי נוֹתֵן מִתּוֹךְ הַרְגָּשָׁה, כִּי ה׳ נוֹתֵן לִי מִתּוֹךְ אַהֲבָה 💛

📖 חֶסֶד – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – שְׁמוֹנֶה דְּרָכִים לֶאֱהֹב חֶסֶד

🤲 לֹא סְתָם נְתִינָה

אוּלַי מַסְפִּיק לָתֵת הַרְבֵּה? אֲבָל יֵשׁ יְלָדִים שֶׁנּוֹתְנִים הַרְבֵּה לַהוֹרִים — וְלֹא אוֹהֲבִים אוֹתָם. מָה חָסֵר?

צָרִיךְ לָשֵׂאת וְלָתֵת: לֹא סְתָם לָתֵת, אֶלָּא לָשֵׂאת בְּעֹל — לְהַרְגִּישׁ אֶת הַשֵּׁנִי. אָז הַחֶסֶד נַעֲשֶׂה אֶמְצָעִי לִקְנוֹת אַהֲבָה.

צָרִיךְ לָשֵׂאת וְלָתֵת – לֹא סְתָם לָתֵת.

💭

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִתְבּוֹנֵן כַּמָּה הַחֶסֶד שֶׁלִּי חָשׁוּב, וְשָׂמֵחַ בּוֹ 💛

📖 חֶסֶד – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – שְׁמוֹנֶה דְּרָכִים לֶאֱהֹב חֶסֶד

💭 דְּרָכִים נוֹסָפוֹת לֶאֱהֹב חֶסֶד

בְּשִׁעוּרִים הַקּוֹדְמִים לָמַדְנוּ לְצַיֵּר בַּלֵּב אֶת הַצַּעַר שֶׁל הַשֵּׁנִי. הִנֵּה עוֹד דֶּרֶךְ: כְּשֶׁחָבֵר עוֹד לֹא סוֹבֵל, אֲבָל אִם לֹא אֶעֱזֹר לוֹ הוּא יִסְבֹּל — לְדַמְיֵן כַּמָּה הוּא יִסְבֹּל אִם לֹא אֶעֱשֶׂה.

וְעוֹד: לְהִתְבּוֹנֵן כַּמָּה הַחֶסֶד שֶׁאֲנִי עוֹשֶׂה בַּבַּיִת מוֹעִיל, וּמָה הָיָה קוֹרֶה בִּלְעָדָיו. אָדָם נֶהֱנֶה כְּשֶׁהוּא מַרְגִּישׁ שֶׁמַּה שֶּׁהוּא עוֹשֶׂה חָשׁוּב — וְאוֹהֵב אֶת מַה שֶּׁנֶּהֱנֶה מִמֶּנּוּ.

מִתְבּוֹנְנִים כַּמָּה הַחֶסֶד חָשׁוּב, וְנֶהֱנִים מִמֶּנּוּ.

💛 אַהֲבַת הַשֵּׁם: אֲנִי חוֹזֵר עַל מַעֲשֵׂי חֶסֶד כְּדֵי לִקְנוֹת לֵב טוֹב, כְּמוֹ אַבְרָהָם אָבִינוּ 💛

📖 חֶסֶד – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַמְבַּ״ם הִלְכוֹת דֵּעוֹת א׳ ז׳

רָמָה ב׳ – שְׁמוֹנֶה דְּרָכִים לֶאֱהֹב חֶסֶד

🔁 יַעֲשֶׂה וִישַׁנֶּה וִישַׁלֵּשׁ

הָרַמְבַּ״ם מְלַמֵּד אֵיךְ קוֹנִים מִדָּה טוֹבָה: יַעֲשֶׂה, יְשַׁנֶּה וִישַׁלֵּשׁ בַּמַּעֲשִׂים, וְיַחֲזֹר עֲלֵיהֶם תָּמִיד, עַד שֶׁיִּהְיוּ קַלִּים עָלָיו וְיִקָּבְעוּ בַּנֶּפֶשׁ. זוֹ דֶּרֶךְ ה׳, שֶׁלִּמֵּד אַבְרָהָם אָבִינוּ.

וּמַה שֶּׁחָשׁוּב: לַחְשֹׁב בִּשְׁעַת הַחֶסֶד ׳אֲנִי רוֹצֶה לִקְנוֹת אֶת מִדַּת הַחֶסֶד׳. כָּךְ הַמַּעֲשֶׂה פּוֹעֵל עַל הַלֵּב.

וְעוֹד אֶפְשָׁר לִקְרֹא כַּמָּה חָשׁוּב לַעֲסֹק בַּחֶסֶד, כְּדֵי שֶׁיִּהְיֶה לָנוּ רָצוֹן לַעֲשׂוֹת אוֹתוֹ.

חוֹזְרִים עַל הַחֶסֶד, וְחוֹשְׁבִים: אֲנִי קוֹנֶה מִדָּה.

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹהֵב אֶת עַם יִשְׂרָאֵל, וְכָךְ אֲנִי אוֹהֵב אֶת ה׳ שֶׁבָּחַר בָּנוּ 💛

📖 חֶסֶד – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִיכָה ו׳ ח׳

רָמָה ב׳ – שְׁמוֹנֶה דְּרָכִים לֶאֱהֹב חֶסֶד

🌍 אַהֲבַת יִשְׂרָאֵל

בְּמִיכָה כָּתוּב: ׳הִגִּיד לְךָ אָדָם מַה טּוֹב וּמָה ה׳ דּוֹרֵשׁ מִמְּךָ, כִּי אִם עֲשׂוֹת מִשְׁפָּט וְאַהֲבַת חֶסֶד וְהַצְנֵעַ לֶכֶת עִם אֱלֹהֶיךָ׳.

דֶּרֶךְ נוֹסֶפֶת לֶאֱהֹב חֶסֶד: לִקְנוֹת אַהֲבַת יִשְׂרָאֵל. מִי שֶׁאוֹהֵב אֶת הָעָם, אוֹהֵב לַעֲשׂוֹת לוֹ חֶסֶד.

וְכָךְ כָּל הַדְּרָכִים חוֹזְרוֹת לִנְקֻדָּה אַחַת: לֵב טוֹב שֶׁאוֹהֵב.

מִי שֶׁאוֹהֵב אֶת יִשְׂרָאֵל – אוֹהֵב לַעֲשׂוֹת חֶסֶד.

מָה לוֹקְחִים מִרָמָה ב׳?

ה׳ חָפֵץ בְּחֶסֶד – אֲבָל חֶסֶד שֶׁבָּא מִן הַלֵּב. כְּשֶׁאֲנִי עוֹשֶׂה אוֹתוֹ שׁוּב וָשׁוּב, בִּזְרִיזוּת וּבְאֲכְפַּתִּיּוּת, הַלֵּב שֶׁלִּי הוֹפֵךְ לְלֵב אוֹהֵב וְדוֹמֶה יוֹתֵר לְה׳ הַמֵּיטִיב.

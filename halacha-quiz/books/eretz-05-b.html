רָמָה ב׳ – מַה לּוֹמְדִים מֵהַכִּסּוּפִים

⚖️ שְׁנֵי פֵּרוּשִׁים לְ׳הִסְתּוֹפֵף׳

בַּמִּדְרָשׁ חֲכָמִים מִסְתַּכְּלִים בַּפָּסוּק ׳בָּחַרְתִּי הִסְתּוֹפֵף בְּבֵית אֱלֹהַי׳. רַבִּי תַּנְחוּם בְּרַבִּי חֲנִילַאי וְרַב מַסְבִּירִים בִּשְׁנֵי אֳפָנִים.

אֶחָד אוֹמֵר: דָּוִד אָמַר: גַּם אִם יֵשׁ לִי אַרְמוֹן וְאוּלָם בְּחוּץ לָאָרֶץ, וּבָאָרֶץ אֵין לִי אֶלָּא סַף – בָּחַרְתִּי בַּסַּף. הָאַחֵר אוֹמֵר: אֲפִלּוּ רַק חֲרוּבִים בָּאָרֶץ.

בִּשְׁנֵי הַפֵּרוּשִׁים הָאָרֶץ חֲשׁוּבָה יוֹתֵר מִכָּל שָׁוֶה כֶּסֶף. הַמִּדְרָשׁ מְסַיֵּם: ׳אֶרֶץ חֶמְדָּה – אֶרֶץ שֶׁחָמְדוּ לָהּ הָאָבוֹת׳.

בַּמִּדְרָשׁ, דָּוִד מַעֲדִיף אֶת הַסַּף אוֹ אֶת הַחֲרוּבִים בָּאָרֶץ עַל אַרְמוֹן בַּחוּץ.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְחַבֵּב אֶת הָאָרֶץ בְּכָל הַלֵּב 💛

📖 אַהֲבַת אֶרֶץ יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים ק״ב ט״ו · רַמְבַּ״ן · רַבֵּנוּ יוֹנָה · סֵפֶר חֲרֵדִים

רָמָה ב׳ – מַה לּוֹמְדִים מֵהַכִּסּוּפִים

💬 רָצוּ, וְלֹא רָצוּ

הָרַמְבַּ״ן מַסְבִּיר שֶׁהַמִּלִּים ׳רָצוּ׳ וְ׳וַתִּרְצֵנִי׳ הֵן עִנְיָן שֶׁל חֵפֶץ וְחֵשֶׁק. לֹא רָצִים, אֶלָּא רוֹצִים בַּלֵּב.

רַבֵּנוּ יוֹנָה מַסְבִּיר אֶת ׳יְחֹנֵנוּ׳ לְפִי הַפָּסוּק ׳מְחוֹנֵן דַּלִּים׳: הַדַּלִּים מוֹצְאִים חֵן בְּעֵינָיו, וְהוּא נוֹתֵן לָהֶם בְּעַיִן יָפָה. כָּךְ עֲפַר הָאָרֶץ מוֹצֵא חֵן בְּעֵינֵיהֶם.

בַּעַל הַ׳חֲרֵדִים׳ לוֹמֵד מִכָּאן שֶׁצָּרִיךְ כָּל יְהוּדִי לְחַבֵּב אֶת הָאָרֶץ וְלָבוֹא אֵלֶיהָ בְּתַשְׁוּקָה גְּדוֹלָה, ׳כְּבֵן אֶל חֵיק אִמּוֹ׳.

רוֹצִים אֶת הָאָרֶץ בַּלֵּב, עַד שֶׁגַּם הֶעָפָר שֶׁלָּהּ חָבִיב.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁהַלֵּב שֶׁלִּי יִהְיֶה דָּבוּק בָּאָרֶץ 💛

📖 אַהֲבַת אֶרֶץ יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁמוּאֵל א׳ כ״ו י״ט

רָמָה ב׳ – מַה לּוֹמְדִים מֵהַכִּסּוּפִים

🧭 מֵהִסְתַּפֵּחַ בְּנַחֲלַת ה׳

כְּשֶׁדָּוִד הֻכְרַח לִבְרֹחַ מֵהָאָרֶץ, הוּא בָּחַר מִלָּה מְיֻחֶדֶת: ׳מֵהִסְתַּפֵּחַ בְּנַחֲלַת ה׳׳. ׳לְהִסְתַּפֵּחַ׳ פֵּרוּשׁוֹ לְהִצְטָרֵף וּלְהִתְחַבֵּר.

הוּא לֹא דִּבֵּר רַק עַל גּוּף שֶׁנִּמְצָא בְּמָקוֹם אֶחָד אוֹ בְּמָקוֹם אַחֵר, אֶלָּא עַל חִבּוּר שֶׁל הַלֵּב אֶל נַחֲלַת ה׳.

הַמִּדְרָשׁ הוֹסִיף שֶׁגַּם בַּמִּדְרָשׁ הַזֶּה לַמֵּדְנוּ עַל חֹמֶר הַקֶּשֶׁר: הַקֶּשֶׁר לָאָרֶץ הוּא לֹא רַק גֵּאוֹגְרַפִי.

דָּוִד הִרְגִּישׁ אֶת הַפְּרֵדָה מִן הָאָרֶץ כְּפֵרוּד מֵחִבּוּר שֶׁל הַלֵּב.

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד מֵהַכִּסּוּפִים שֶׁל מֹשֶׁה 💛

📖 אַהֲבַת אֶרֶץ יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דְּבָרִים ג׳ כ״ג–כ״ה · אִבְּן עֶזְרָא · דְּבָרִים רַבָּה

רָמָה ב׳ – מַה לּוֹמְדִים מֵהַכִּסּוּפִים

🕊️ חֲמֵשׁ מֵאוֹת וְחֲמֵשׁ עֶשְׂרֵה תְּפִלּוֹת

מֹשֶׁה רַבֵּנוּ לֹא נִכְנַס לָאָרֶץ, וַאֲפִלּוּ כָּךְ הוּא הִמְשִׁיךְ לְבַקֵּשׁ: ׳אֶעְבְּרָה נָּא וְאֶרְאֶה אֶת הָאָרֶץ הַטּוֹבָה׳. לְפִי הַמִּדְרָשׁ, הִתְפַּלֵּל חֲמֵשׁ מֵאוֹת וְחֲמֵשׁ עֶשְׂרֵה תְּפִלּוֹת.

מוּבָן שֶׁיֵּשׁ לִשְׁאֹל: לָמָּה הַתּוֹרָה מְסַפֶּרֶת אֶת זֶה בְּהַרְחָבָה? הָאִבְּן עֶזְרָא עוֹנֶה: טַעַם הַפָּרָשָׁה – לְחַבֵּב אֶת אֶרֶץ יִשְׂרָאֵל.

בְּכָל דּוֹר, הַקּוֹרְאִים רוֹאִים עַד כַּמָּה הָאָרֶץ הָיְתָה חֲבִיבָה אֵצֶל מֹשֶׁה, וְלוֹמְדִים לְחַבֵּב אוֹתָהּ גַּם הֵם.

הַתּוֹרָה מְסַפֶּרֶת עַל הַכִּסּוּפִים שֶׁל מֹשֶׁה כְּדֵי שֶׁנִּלְמַד לֶאֱהֹב.

📖 אַהֲבַת אֶרֶץ יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אִבְּן עֶזְרָא, דְּבָרִים ג׳ · אַהֲבַת אֶרֶץ חֶמְדָּה

רָמָה ב׳ – מַה לּוֹמְדִים מֵהַכִּסּוּפִים

💎 מִי שֶׁמַּכִּיר – מוֹדֶה

בַּמָּשָׁל שֶׁל הָאִבְּן עֶזְרָא, הַבֵּן לֹא יָדַע אֶת עֶרֶךְ הַפְּנִינִים. כְּשֶׁהַסּוֹחֵר הִתְפַּעֵל, הַבֵּן הִתְרַגֵּשׁ עַד דְּמָעוֹת, וּפִיו הִתְמַלֵּא שִׁיר וְשֶׁבַח עַל אָבִיו שֶׁהִנְחִיל לוֹ אוֹצָר כָּזֶה.

כָּךְ הַבָּנִים יָרְשׁוּ אֶת הָאָרֶץ וְלֹא תָּמִיד מַכִּירִים אֶת הַסְּגֻלָּה שֶׁלָּהּ. כְּשֶׁהֵם לוֹמְדִים תּוֹרָה וְרוֹאִים כַּמָּה תְּפִלּוֹת מֹשֶׁה הִתְפַּלֵּל, הֵם מַכִּירִים: ׳אֶרֶץ חֶמְדָּה טוֹבָה וּרְחָבָה׳.

וּמִתּוֹךְ הַהֶכֵּרוּת הַלֵּב נִרְגַּשׁ וּמוֹדֶה לַבּוֹרֵא עַל הַמַּתָּנָה.

הֶכֵּרוּת הָעֵרֶךְ מְעוֹרֶרֶת אַהֲבָה, וְהָאַהֲבָה מְעוֹרֶרֶת הוֹדָיָה.

רָמָה ב׳ – מַה לּוֹמְדִים מֵהַכִּסּוּפִים

🎁 אֵין בְּרָכָה עַל הַכִּסּוּפִים

הָרַב שְׁלֹמֹה זַלְמָן אוֹיֶרְבַּךְ אָמַר שֶׁיִּתָּכֵן שֶׁהַטַּעַם שֶׁאֵין מְבָרְכִים עַל מִצְוַת יִשּׁוּב הָאָרֶץ הוּא שֶׁלֹּא יֵרָאֶה כְּאִלּוּ זֶה עֹל. אֲנַחְנוּ לֹא יוֹשְׁבִים בָּאָרֶץ רַק כִּי צִוּוּ עָלֵינוּ.

אֶלָּא, אָמַר, אֲנַחְנוּ שְׂמֵחִים וּמִשְׁתּוֹקְקִים לָשֶׁבֶת עַל אַדְמַת הַקֹּדֶשׁ שֶׁל אֶרֶץ חֶמְדָּתֵנוּ.

הוּא הִזְכִּיר אֶת הַגְּר״א, שֶׁכָּתַב שֶׁ׳עֶלְיוֹנִים וְתַחְתּוֹנִים מִתְאַוִּים לָשֶׁכֹּן בָּהּ׳.

אֶת הָאָרֶץ אוֹהֲבִים מֵרָצוֹן וּמִגַּעְגּוּעִים, וְלֹא מִתּוֹךְ חוֹבָה בִּלְבַד.

מָה לוֹקְחִים מִרָמָה ב׳?

מִי שֶׁמַּכִּיר אֶת עֵרֶךְ הָאָרֶץ, מִתְרַגֵּשׁ וּמוֹדֶה. הַכִּסּוּפִים שֶׁל דָּוִד וּמֹשֶׁה לִמְּדוּ אוֹתָנוּ לֶאֱהֹב אֶת הָאָרֶץ, וְהָאַהֲבָה הַזֹּאת מְקָרֶבֶת אוֹתָנוּ לַה׳ שֶׁנָּתַן לָנוּ אוֹתָהּ.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה אֲנִי אוֹהֵב בָּאָרֶץ, גַּם דְּבָרִים פְּשׁוּטִים כְּמוֹ אֲבָנִים וְעָפָר?
  • מָתַי הִרְגַּשְׁתִּי גַּעְגּוּעִים לְמָקוֹם שֶׁאֲנִי אוֹהֵב, וְאֵיךְ זֶה דּוֹמֶה לְדָוִד וּלְמֹשֶׁה?
  • אֶכְתֹּב תְּפִלָּה קְצָרָה: מָה הָיִיתִי מְבַקֵּשׁ אִלּוּ הָיִיתִי רָחוֹק מֵהָאָרֶץ, וְאַקְרִיא אוֹתָהּ בַּבַּיִת.

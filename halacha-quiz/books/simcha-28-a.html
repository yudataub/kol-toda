יֶלֶד שָׁמַע שִׁיר בַּחֲתֻנָּה, וְכָל הַשָּׁבוּעַ הוּא מְזַמְזֵם אוֹתוֹ בַּדֶּרֶךְ לְבֵית הַסֵּפֶר. אֲבָל אֶת רְשִׁימַת הַמִּלִּים לַמִּבְחָן הוּא כְּבָר שָׁכַח.

שִׁיר נִשְׁאָר בַּלֵּב הַרְבֵּה יוֹתֵר מִמִּלִּים רְגִילוֹת. בַּשִּׁעוּר עַל כּוֹחָהּ שֶׁל שִׁירָה לוֹמְדִים לָמָּה זֶה כָּךְ, וּמָה אֶפְשָׁר לַעֲשׂוֹת עִם הַכֹּחַ הַזֶּה.

וְהַכֹּל מַתְחִיל בִּשְׁאֵלָה מִסֵּפֶר שׁוֹפְטִים.

שִׁיר נִשְׁאָר בַּלֵּב יוֹתֵר מִמִּלִּים רְגִילוֹת.

❓ מָה קָרָה בֵּין שְׁנֵי הַפְּסוּקִים?

בְּסֵפֶר שׁוֹפְטִים כָּתוּב פַּעֲמַיִם: ׳וַיֹּסִפוּ בְּנֵי יִשְׂרָאֵל לַעֲשׂוֹת הָרַע׳. הַמִּלָּה ׳וַיֹּסִפוּ׳ אוֹמֶרֶת: הוֹסִיפוּ עוֹד עַל מָה שֶׁקָּרָה קֹדֶם.

אֲבָל בְּפֶרֶק ו׳ כָּתוּב רַק: ׳וַיַּעֲשׂוּ בְּנֵי יִשְׂרָאֵל הָרַע׳, בְּלִי ׳וַיֹּסִפוּ׳, כְּאִלּוּ זוֹ הַתְחָלָה חֲדָשָׁה.

חֲזַ״ל שָׁאֲלוּ: מָה הִשְׁתַּנָּה? וְעָנוּ: בֵּין הַפְּסוּקִים, בְּפֶרֶק ה׳, נִמְצֵאת שִׁירַת דְּבוֹרָה.

בֵּין ׳וַיֹּסִפוּ׳ לְ׳וַיַּעֲשׂוּ׳ נִמְצֵאת שִׁירַת דְּבוֹרָה.

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שׁוֹחֵר טוֹב תְּהִלִּים י״ח · רש״י שׁוֹפְטִים ו׳ א׳

רָמָה א׳ – שִׁיר שֶׁנִּשְׁאָר בַּלֵּב

🌅 שִׁיר אַחֲרֵי נֵס

חֲזַ״ל אָמְרוּ: כָּל מִי שֶׁנַּעֲשָׂה לוֹ נֵס וְאוֹמֵר שִׁירָה – מוֹחֲלִים לוֹ עַל כָּל מָה שֶׁקָּרָה קֹדֶם, וְהוּא נַעֲשֶׂה בְּרִיָּה חֲדָשָׁה. כָּךְ לָמְדוּ מִשִּׁירַת דְּבוֹרָה.

ה׳ עָשָׂה לִבְנֵי יִשְׂרָאֵל נֵס וְהִצִּיל אוֹתָם, וְהֵם שָׁרוּ לוֹ שִׁיר תּוֹדָה. אַחֲרֵי הַשִּׁיר הֵם הִתְחִילוּ מֵחָדָשׁ.

זֶה כֹּחוֹ שֶׁל שִׁיר: הוּא לֹא רַק יָפֶה, הוּא פּוֹתֵחַ דַּף חָדָשׁ.

שִׁיר תּוֹדָה אַחֲרֵי נֵס פּוֹתֵחַ דַּף חָדָשׁ.

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רמב״ן דְּבָרִים ל״א י״ט

רָמָה א׳ – שִׁיר שֶׁנִּשְׁאָר בַּלֵּב

📜 שִׁיר עִם מְקוֹמוֹת רֵיקִים

שִׁיר נִכְתָּב בְּצוּרָה מְיֻחֶדֶת: יֵשׁ בּוֹ מְקוֹמוֹת רֵיקִים בֵּין חֵלֶק לְחֵלֶק, כְּמוֹ לְבֵנִים שֶׁמְּסֻדָּרוֹת עִם רְוָחִים. הַגְּמָרָא קוֹרֵאת לָזֶה ׳אֲרִיחַ עַל גַּבֵּי לְבֵנָה׳.

הָרַמְבַּ״ן מַסְבִּיר שֶׁבַּמְּקוֹמוֹת הָרֵיקִים מְזַמְּרִים נִגּוּן בְּלִי מִלִּים.

שִׁיר הוּא מִלִּים וְנִגּוּן יַחַד.

בַּמְּקוֹמוֹת הָרֵיקִים שֶׁבְּשִׁיר מְזַמְּרִים נִגּוּן.

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, כּוֹחָהּ שֶׁל שִׁירָה

רָמָה א׳ – שִׁיר שֶׁנִּשְׁאָר בַּלֵּב

🔁 שׁוּב וָשׁוּב

מִלִּים יְבֵשׁוֹת – אָדָם מִתְעַיֵּף מֵהֶן אַחֲרֵי כַּמָּה פְּעָמִים. אֲבָל שִׁיר אֶפְשָׁר לָשִׁיר שׁוּב וָשׁוּב, גַּם כְּשֶׁכְּבָר יוֹדְעִים אֶת הַמִּלִּים, כִּי נָעִים לָשִׁיר.

יֶלֶד אוֹהֵב אֶת הַשִּׁיר שֶׁלּוֹ וְשָׁר אוֹתוֹ עוֹד פַּעַם וְעוֹד פַּעַם, בִּלְתִּי מִתְעַיֵּף.

לָכֵן שִׁיר חַי הַרְבֵּה זְמַן בְּפִי הָאֲנָשִׁים.

שִׁיר אֶפְשָׁר לָשִׁיר שׁוּב וָשׁוּב וְלֹא מִתְעַיְּפִים.

רָמָה א׳ – שִׁיר שֶׁנִּשְׁאָר בַּלֵּב

❤️ שִׁיר מְדַבֵּר אֶל הַלֵּב

מִלִּים רְגִילוֹת מְדַבְּרוֹת אֶל הַשֵּׂכֶל וְעוֹזְרוֹת לָדַעַת. אֲבָל שִׁיר מְדַבֵּר אֶל הַלֵּב וְאֶל הָרְגָשׁוֹת.

לָכֵן שִׁיר מִתְפַּשֵּׁט: מִישֶׁהוּ מְזַמְזֵם, עוֹד אֶחָד שׁוֹמֵעַ וְשָׁר, וְכָךְ הוּא עוֹבֵר מֵאָדָם לְאָדָם וּמֵעִיר לְעִיר.

וְהוּא נִזְכָּר שָׁנִים רַבּוֹת.

שִׁיר פּוֹנֶה אֶל הַלֵּב, לָכֵן הוּא מִתְפַּשֵּׁט וְנִזְכָּר.

🌍 שִׁיר שֶׁהָלַךְ בְּכָל הָעָם

שִׁירָתָם שֶׁל דְּבוֹרָה וּבָרָק לֹא הָיְתָה שִׁיר חַד־פַּעֲמִי שֶׁנֶּאֱמַר בְּהִתְלַהֲבוּת וְנִשְׁכַּח. הִיא הִתְפַּרְסְמָה בְּכָל הַדּוֹר, וְשָׁרוּ אוֹתָהּ בְּכָל מָקוֹם בְּיִשְׂרָאֵל.

הַשִּׁיר הִכְנִיס אֶת הַנֵּס עָמֹק בַּלֵּב, וְכָךְ הַנֵּס נִזְכַּר שָׁנִים רַבּוֹת.

וּכְשֶׁזּוֹכְרִים אֶת הַטּוֹבוֹת שֶׁעָשָׂה לָנוּ ה׳, מִתְעוֹרֵר רָצוֹן לֶאֱהֹב אוֹתוֹ.

שִׁירַת דְּבוֹרָה הִתְפַּרְסְמָה וְנִשְׁמְעָה בְּכָל הָעָם.

🌱 בְּרִיָּה חֲדָשָׁה

אַחֲרֵי הַשִּׁיר בְּנֵי יִשְׂרָאֵל הָיוּ בֶּאֱמֶת בְּרִיָּה חֲדָשָׁה. הֵם לֹא הָיוּ צְרִיכִים לְהַמְשִׁיךְ בְּאוֹתָן טָעֻיּוֹת מִן הָעָבָר.

בְּדֶרֶךְ כְּלָל טָעוּת אַחַת גּוֹרֶרֶת טָעוּת אַחֶרֶת. אֲבָל כָּאן מָה שֶׁעָשׂוּ אֶתְמוֹל לֹא גָּרַר טָעוּת חֲדָשָׁה הַיּוֹם.

לָכֵן כָּתוּב לָנוּ: ׳שִׁירוּ לַה׳ שִׁיר חָדָשׁ׳.

מָה לוֹקְחִים מִרָמָה א׳?

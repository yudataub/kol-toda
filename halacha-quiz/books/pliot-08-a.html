💛 אַהֲבַת הַשֵּׁם: אֲנִי מַבִּיט בַּשָּׁמַיִם וְאוֹהֵב אֶת מִי שֶׁעָשָׂה אוֹתָם 💛

📖 נִפְלְאוֹת הַבְּרִיאָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, יְשַׁעְיָה מ׳ כ״ו · תְּהִלִּים ח׳ ד׳ · רַמְבַּ״ם

רָמָה א׳ – שָׁמַיִם שֶׁמְּדַבְּרִים

🔭 עֶרֶב אֶחָד שֶׁל הִסְתַּכְּלוּת

הַנָּבִיא יְשַׁעְיָהוּ קוֹרֵא לָנוּ: ׳שְׂאוּ מָרוֹם עֵינֵיכֶם וּרְאוּ מִי בָרָא אֵלֶּה׳. וְדָוִד הַמֶּלֶךְ אָמַר: ׳כִּי אֶרְאֶה שָׁמֶיךָ מַעֲשֵׂי אֶצְבְּעֹתֶיךָ, יָרֵחַ וְכוֹכָבִים אֲשֶׁר כּוֹנָנְתָּה׳.

הַמְּשִׂימָה: בְּעֶרֶב אֶחָד לָצֵאת וּלְהִסְתַּכֵּל בַּשָּׁמַיִם דַּקָּה אַחַת בְּשֶׁקֶט, וְלוֹמַר מִשְׁפָּט אֶחָד: ׳מָה רַבּוּ מַעֲשֶׂיךָ ה׳׳.

הָרַמְבַּ״ם מְלַמֵּד שֶׁהַהִסְתַּכְּלוּת בַּבְּרִיאָה מוֹסִיפָה אַהֲבָה לַבּוֹרֵא.

יוֹצְאִים לְהִסְתַּכֵּל בַּשָּׁמַיִם וְאוֹמְרִים: מָה רַבּוּ מַעֲשֶׂיךָ.

רָמָה א׳ – שָׁמַיִם שֶׁמְּדַבְּרִים

מָה לוֹקְחִים מִרָמָה א׳?

הַשָּׁמַיִם לֹא אוֹמְרִים מִלָּה, וְיֵשׁ בָּהֶם הַרְבֵּה אַהֲבָה: תְּכֵלֶת, אוֹר שֶׁמִּתְחַזֵּק לְאַט, אֲוִיר לִנְשִׁימָה. כְּשֶׁאֲנִי מַבִּיט, אֲנִי מַרְגִּישׁ כַּמָּה ה׳ אוֹהֵב אוֹתָנוּ.

הַשָּׁמַיִם מְסַפְּרִים כְּבוֹד אֵ-ל וּמַעֲשֵׂה יָדָיו מַגִּיד הָרָקִיעַ

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָה רָאִיתִי הַשָּׁבוּעַ בַּשָּׁמַיִם, וּמָה הִרְגַּשְׁתִּי?
  • אֵיזֶה דָּבָר שֶׁה׳ נָתַן לִי מִן הַשָּׁמַיִם (אוֹר, אֲוִיר, גֶּשֶׁם) רָגִיל לִי עַד כְּדֵי כָּךְ שֶׁאֲנִי לֹא שָׂם לֵב?
  • אֵצֵא בְּעֶרֶב אֶחָד לְהַבִּיט בְּכוֹכָבִים אוֹ בַּיָּרֵחַ דַּקָּה אַחַת, אֹמַר ׳מָה רַבּוּ מַעֲשֶׂיךָ ה׳׳ וְאֶרְשֹׁם מָה הִרְגַּשְׁתִּי.

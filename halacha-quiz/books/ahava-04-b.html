רָמָה ב׳ – הַכָּרַת הַטּוֹב בֶּאֱמֶת

🧐 הַכָּרַת הַטּוֹב – יוֹתֵר מִמִּלִּים

הַשְּׁאֵלָה: אִם הָאָדָם אָמַר אֶת הַמִּלִּים הַנְּכוֹנוֹת, לָמָּה הַנּוֹתֵן מַרְגִּישׁ שֶׁלֹּא הוֹדוּ לוֹ?

כִּי ׳הַכָּרַת הַטּוֹב׳ אֵינָהּ רַק מִלִּים וְלֹא רַק תְּחוּשַׁת חוֹבָה. הִיא הַכָּרָה מַמָּשִׁית בַּטּוֹב שֶׁל הַנּוֹתֵן: מַרְגִּישִׁים אֶת טוּב לִבּוֹ וְאֶת מִדּוֹתָיו הַטּוֹבוֹת, וּמִזֶּה גּוֹבֶרֶת הַהֶעֱרָכָה וְהָאַהֲבָה אֵלָיו.

קֶשֶׁר עַיִן הוּא הַבִּטּוּי לְכָךְ, וְהוּא מְגַלֶּה אֶת אֲמִתּוּת הַתּוֹדָה יוֹתֵר מִן הַמִּלִּים.

הַכָּרַת הַטּוֹב הִיא הַכָּרָה מַמָּשִׁית בְּטוּב לִבּוֹ שֶׁל הַנּוֹתֵן.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת אוֹרֵחַ טוֹב בָּעוֹלָם שֶׁה׳ בָּרָא 💛

📖 אַהֲבַת ה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּרָכוֹת נ״ח א׳

רָמָה ב׳ – הַכָּרַת הַטּוֹב בֶּאֱמֶת

🍽️ אוֹרֵחַ טוֹב וְאוֹרֵחַ רַע

הַגְּמָרָא מְסַפֶּרֶת עַל שְׁנֵי סוּגֵי אוֹרְחִים. הָאוֹרֵחַ הַטּוֹב אוֹמֵר: ׳כַּמָּה טְרָחוֹת טָרַח בַּעַל הַבַּיִת בִּשְׁבִילִי! כָּל מַה שֶּׁטָּרַח, לֹא טָרַח אֶלָּא בִּשְׁבִילִי׳.

הָאוֹרֵחַ הָרַע אוֹמֵר: ׳מָה טָרַח כָּאן? כָּל הַטִּרְחָה הָיְתָה בִּשְׁבִיל אִשְׁתּוֹ וּבָנָיו׳.

הָאוֹרֵחַ הַטּוֹב אוֹהֵב לְהוֹדוֹת וְלִרְאוֹת טוֹב, וְלָכֵן הוּא מַכְרִיעַ לְצַד הַטּוֹב. הוּא אוֹהֵב אֶת הַהוֹדָיָה, וְכָל טוֹבָה קְטַנָּה נֶחְשֶׁבֶת לוֹ גְּדוֹלָה.

הָאוֹרֵחַ הַטּוֹב אוֹהֵב לְהוֹדוֹת, וְרוֹאֶה אֶת הַטּוֹב.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹצֵא דֶּרֶךְ לְהוֹדוֹת: מִשְׁפָּט, כַּרְטִיס אוֹ צִיּוּר 💛

רָמָה ב׳ – הַכָּרַת הַטּוֹב בֶּאֱמֶת

✏️ מִמִּלָּה לְמִשְׁפָּט

הַפְּסִיכוֹתֶרָפִּיסְטִית לֵיאַת קֶרֶן מְלַמֶּדֶת: צָרִיךְ לְהַחְזִיר אֶת הָאַחְרָיוּת לַיֶּלֶד. קִבַּלְתָּ מַתָּנָה – תִּמְצָא דֶּרֶךְ לְהוֹדוֹת עָלֶיהָ, בִּכְרִטִיס אוֹ בְּצִיּוּר.

וְחָשׁוּב לְהַרְחִיב אֶת ׳תּוֹדָה׳ לְמִשְׁפָּט שָׁלֵם: עַל מָה אֲנִי מוֹדֶה, וּמָה עָשִׂיתִי בַּמַּתָּנָה.

כְּשֶׁהַיֶּלֶד יוֹדֵעַ שֶׁעָלָיו לוֹמַר מִשְׁפָּט, הוּא נַעֲשֶׂה מוּדָע יוֹתֵר לְחֲוָיוֹתָיו, וּמַעֲרִיךְ אֶת הַמַּתָּנָה יוֹתֵר.

מִשְׁפָּט שָׁלֵם גּוֹרֵם לָנוּ לָשִׂים לֵב וּלְהַעֲרִיךְ יוֹתֵר.

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹמֵר תּוֹדָה גַּם כְּשֶׁלֹּא בְּדִיּוּק מָה שֶׁרָצִיתִי 💛

📖 אַהֲבַת ה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, לְלַמֵּד אֶת הַיֶּלֶד לוֹמַר תּוֹדָה

רָמָה ב׳ – הַכָּרַת הַטּוֹב בֶּאֱמֶת

🎀 גַּם כְּשֶׁהַמַּתָּנָה לֹא מָצְאָה חֵן

אִם הַמַּתָּנָה לֹא מָצְאָה חֵן אוֹ שֶׁכְּבָר יֵשׁ לָנוּ כָּזוֹ, אוֹמְרִים תּוֹדָה בְּכָל זֹאת: הַתּוֹדָה נֶאֱמֶרֶת עַל עֶצֶם הַנְּתִינָה, כִּי מִישֶׁהוּ הִשְׁקִיעַ וְהֵכִין אוֹ קָנָה בִּשְׁבִילֵנוּ.

אֶפְשָׁר לוֹמַר מַשֶּׁהוּ נֶחְמָד, כְּמוֹ ׳רוֹאִים שֶׁהִיא מוּשְׁקַעַת מְאוֹד׳.

אִם צָרִיךְ לְהַחֲלִיף, מְבַשְּׂרִים בְּעֵדִינוּת וְעִם סִבָּה אוֹבְּיֶקְטִיבִית, כְּמוֹ מִדָּה לֹא נְכוֹנָה אוֹ צֶבַע שֶׁלֹּא מַתְאִים.

תּוֹדָה עַל עֶצֶם הַנְּתִינָה, וּבְעֵדִינוּת אֶפְשָׁר לְהַחֲלִיף.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְפַרְגֵּן לְמִי שֶׁעָזַר, וְכָךְ אֲנִי מְחַזֵּק אֶת הָאַהֲבָה 💛

רָמָה ב׳ – הַכָּרַת הַטּוֹב בֶּאֱמֶת

🌟 גַּם פִּרְגּוּן הוּא תּוֹדָה

בַּבַּיִת שֶׁיֵּשׁ בּוֹ חֲלוּקַת תַּפְקִידִים, הַיְּלָדִים יַשְׁלִימוּ אֶת מְשִׂימוֹתֵיהֶם בְּיוֹתֵר רָצוֹן אִם יַרְגִּישׁוּ שֶׁמַּעֲרִיכִים אֶת הַמַּאֲמָץ.

לְמָשָׁל: ׳יוֹנָתָן, תּוֹדָה שֶׁזָּרַקְתָּ אֶת הַזֶּבֶל בְּלִי שֶׁבִּקַּשְׁתִּי׳.

לֹא תָּמִיד מַתְאִים ׳תּוֹדָה׳, כְּשֶׁבִּקְשׁוּ אֶת הַדָּבָר הַרְבֵּה פְּעָמִים. אָז פִּרְגּוּן מַתְאִים יוֹתֵר: ׳אֵיזֶה כֵּיף לְהִכָּנֵס לַבַּיִת נָקִי!׳.

פִּרְגּוּן הוּא סוּג שֶׁל תּוֹדָה, כְּשֶׁ׳תּוֹדָה׳ רְגִילָה לֹא מַתְאִימָה.

💛 אַהֲבַת הַשֵּׁם: כָּל ׳תּוֹדָה׳ שֶׁאֲנִי אוֹמֵר מֵהַלֵּב מְקָרֶבֶת אוֹתִי אֶל ה׳ 💛

📖 אַהֲבַת ה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שַׁעֲרֵי הוֹדָיָה לָרַב לוּגַסִּי

רָמָה ב׳ – הַכָּרַת הַטּוֹב בֶּאֱמֶת

🧩 מֵהָאָדָם אֶל ה׳

כְּדֵי לְפַתֵּחַ אֶת הַנְּעִימוּת שֶׁל הַכָּרַת הַטּוֹב לַה׳, מְתַרְגְּלִים אוֹתָהּ בְּכָל הִזְדַּמְּנוּת – גַּם מוּל בְּנֵי אָדָם.

מִי שֶׁלֹּא מְעוֹרֵר הוֹדָיָה אֲמִתִּית כְּשֶׁמַּטִּיבִים לוֹ, מַגְדִּיל אֶת הַכְּפִיּוּת טוֹבָה שֶׁלּוֹ. מִי שֶׁמּוֹדֶה, מַרְגִּישׁ אֶת הַנְּעִימוּת שֶׁל הַהוֹדָיָה וּמַגְדִּיל אוֹתָהּ.

לָכֵן חָשׁוּב לְהוֹדוֹת ׳בְּפֶה מָלֵא וּבְלֵב נִרְגָּשׁ׳ עַל כָּל הֲטָבָה.

תּוֹדָה אֲמִתִּית לַבְּנֵי אָדָם מְחַזֶּקֶת אֶת הַכָּרַת הַטּוֹב לַה׳.

מָה לוֹקְחִים מִרָמָה ב׳?

מִי שֶׁמַּכִּיר אֶת הַטּוֹב שֶׁל הַנּוֹתֵן וּמַרְגִּישׁ אוֹתוֹ, אוֹמֵר תּוֹדָה שֶׁנִּכְנֶסֶת לַלֵּב. וְכָל תּוֹדָה כָּזוֹ לָאָדָם מְחַזֶּקֶת גַּם אֶת הַתּוֹדָה לַה׳, וְאֶת הָאַהֲבָה אֵלָיו.

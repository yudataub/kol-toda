📜

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד שֶׁכָּל אָדָם יָכוֹל לְבָרֵךְ, וה׳ שׁוֹמֵעַ 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְגִלָּה ט״ו א׳ · מַהַרְשָׁ״א

רָמָה ב׳ – כֹּחָהּ שֶׁל בְּרָכָה

📜 שְׁנֵי גְּדוֹלֵי הַדּוֹר

הַגְּמָרָא (מְגִלָּה ט״ו א׳): ׳לְעוֹלָם אַל תְּהִי בִּרְכַּת הֶדְיוֹט קַלָּה בְּעֵינֶיךָ, שֶׁהֲרֵי שְׁנֵי גְּדוֹלֵי הַדּוֹר בֵּרְכוּם שְׁנֵי הֶדְיוֹטוֹת וְנִתְקַיְּמָה בָּהֶם׳ – דָּוִד שֶׁבֵּרַךְ אוֹתוֹ אֲרַוְנָה, וְדָנִיֵּאל שֶׁבֵּרַךְ אוֹתוֹ דָּרְיָוֶשׁ.

הַמַּהַרְשָׁ״א שׁוֹאֵל: לָמָּה הַתּוֹרָה סִפְּרָה מָה אָמַר אֲרַוְנָה? לְלַמֵּד שֶׁגַּם דְּבָרָיו הִשְׁפִּיעוּ.

הַבְּרָכָה שֶׁל הֶדְיוֹט הִשְׁפִּיעָה גַּם עַל גְּדוֹלֵי הַדּוֹר.

1
👴

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְכַבֵּד זְקֵנִים וּמְבָרֵךְ אוֹתָם בְּאַהֲבָה 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רוּת רַבָּה ו׳ ב׳

רָמָה ב׳ – כֹּחָהּ שֶׁל בְּרָכָה

👴 לְבָרֵךְ אֶת הַזָּקֵן

בְּמִדְרַשׁ רוּת רַבָּה: ׳לְעוֹלָם אַל יִמְנַע אָדָם עַצְמוֹ מִלֵּילֵךְ אֵצֶל זָקֵן לְבָרֲכוֹ׳. בֹּעַז הָיָה זָקֵן, וְהִתְבָּרֵךְ בִּבְרָכַת נָעֳמִי וְרוּת.

אֶפְשָׁר לְבָרֵךְ גַּם אֶת מִי שֶׁגָּדוֹל מִמֶּנּוּ, וְאֶפְשָׁר לִבְקֹשׁ בְּרָכָה מִזְּקֵנִים.

מוּתָּר לָלֶכֶת לִזְקֵנִים וּלְבָרֵךְ אוֹתָם.

2
🤲

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹהֵב אֶת מִי שֶׁאֲנִי מְבָרֵךְ, וה׳ אוֹהֵב אוֹתִי 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, חֻלִּין מ״ט א׳ · בְּמִדְבַּר ו׳ כ״ד–כ״ו

רָמָה ב׳ – כֹּחָהּ שֶׁל בְּרָכָה

🤲 בְּאַהֲבָה

רַבִּי עֲקִיבָא לָמַד מִ׳וַאֲבָרְכָה מְבָרְכֶיךָ׳ אֶת בִּרְכַּת כֹּהֲנִים. הַשִּׁעוּר מַסְבִּיר: כְּדֵי שֶׁהַבְּרָכָה תִּתְקַבֵּל חַיָּב לֶאֱהֹב אֶת מִי שֶׁמְּבָרְכִים וּלְרָצוֹת שֶׁיִּהְיֶה לוֹ טוֹב.

לָכֵן הַכֹּהֲנִים מְבָרְכִים ׳בְּאַהֲבָה׳, וְכָךְ גַּם כָּל אָדָם.

בִּרְכַּת כֹּהֲנִים וְכָל בְּרָכָה צְרִיכָה אַהֲבָה.

3
🔢

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹחֵר עַיִן טוֹבָה וְלֵב טוֹב, כְּדֵי לְהַרְבּוֹת בְּרָכָה בָּעוֹלָם 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּמִדְבַּר כ״ב ו׳ · בְּרֵאשִׁית י״ב ב׳ · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – כֹּחָהּ שֶׁל בְּרָכָה

🔢 קַל וָחֹמֶר

בַּשִּׁעוּר לוֹמְדִים קַל וָחֹמֶר: אִם לְעַיִן רָעָה יֵשׁ כֹּחַ לְהַזִּיק, כְּמוֹ אֵצֶל בִּלְעָם – קַל וָחֹמֶר שֶׁלְּעַיִן טוֹבָה וּלְבִרְכַּת הַלֵּב יֵשׁ כֹּחַ גָּדוֹל יוֹתֵר.

לְבִלְעָם הָיְתָה עַיִן רָעָה, וְלָכֵן לֹא יָכוֹל לְבָרֵךְ מִן הַלֵּב. לְאַבְרָהָם הָיְתָה עַיִן טוֹבָה, וְכָתוּב: ׳וֶהְיֵה בְּרָכָה׳.

אִם עַיִן רָעָה מַזִּיקָה, קַל וָחֹמֶר שֶׁבְּרָכָה מוֹעִילָה.

4
🌟

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִתְבָּרֵךְ מֵה׳ וּמְבָרֵךְ אֶת הַזּוּלַת 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשְׁלֵי כ״ב ט׳ · בְּרֵאשִׁית י״ב ג׳ · חָתָם סוֹפֵר

רָמָה ב׳ – כֹּחָהּ שֶׁל בְּרָכָה

🌟 מִי שֶׁעֵינוֹ טוֹבָה מְבָרֵךְ

׳טוֹב עַיִן הוּא יְבֹרָךְ׳ – מִי שֶׁיֵּשׁ לוֹ עַיִן טוֹבָה יָכוֹל לְבָרֵךְ אֶת הַשֵּׁנִי.

ה׳ מְבָרֵךְ אֶת הַמְבָרֵךְ קֹדֶם, כְּמוֹ שֶׁכָּתוּב ׳וַאֲבָרְכָה מְבָרְכֶיךָ׳, כְּדֵי שֶׁיְּבָרֵךְ יוֹתֵר מִכָּל הַלֵּב. כָּךְ הַבְּרָכָה בָּאָה מִן הַלֵּב וּמַגִּיעָה אֶל הַלֵּב.

עַיִן טוֹבָה מְאַפְשֶׁרֶת לְבָרֵךְ, וה׳ מְבָרֵךְ קֹדֶם.

5
🙏

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְבַקֵּשׁ בְּרָכָה, וּמוֹדֶה לַה׳ עַל אֲנָשִׁים שֶׁאוֹהֲבִים אוֹתִי 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – כֹּחָהּ שֶׁל בְּרָכָה

🙏 לְבַקֵּשׁ בְּרָכָה

הַשִּׁעוּר מְלַמֵּד: אִם רוֹצִים שֶׁפַע שֶׁל בְּרָכָה, אֶפְשָׁר לְחַפֵּשׂ אָדָם שֶׁרוֹצֶה שֶׁיִּהְיֶה לָנוּ טוֹב וְלִבְקֹשׁ מִמֶּנּוּ שֶׁיִּתְפַּלֵּל עָלֵינוּ.

וְיֶשְׁנוֹ יְסוֹד גָּדוֹל: כָּל עוֹד נַאֲמִין שֶׁלַּדִּבּוּר שֶׁלָּנוּ יֵשׁ כֹּחַ, כָּךְ בֶּאֱמֶת יִהְיֶה לוֹ כֹּחַ. זוֹ דֶּרֶךְ יָפָה לְקָרֵב לֵבָבוֹת.

מְבַקְשִׁים בְּרָכָה מִמִּי שֶׁאוֹהֵב אוֹתָנוּ.

6
💡

רָמָה ב׳ – כֹּחָהּ שֶׁל בְּרָכָה

מָה לוֹקְחִים מִרָמָה ב׳?

כָּל בְּרָכָה שֶׁיּוֹצֵאת מִן הַלֵּב מִתְקַיֶּמֶת, גַּם אִם הִיא בָּאָה מֵאָדָם פָּשׁוּט. ה׳ נָתַן לָנוּ אֶת הַכֹּחַ הַזֶּה, וְהוּא שָׂמֵחַ בְּכָל בְּרָכָה שֶׁאֲנִי נוֹתֵן לַאֲחֵרִים.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • מִי הָאָדָם שֶׁהַבְּרָכָה שֶׁלּוֹ הִשְׂמִיחָה אוֹתִי, וְאֶת מִי אֲנִי יָכוֹל לְבָרֵךְ הַיּוֹם?
  • מָה קָשֶׁה לִי יוֹתֵר: לְבָרֵךְ מִכָּל הַלֵּב אוֹ לְקַבֵּל בְּרָכָה, וְלָמָּה?
  • אֲבָרֵךְ חָבֵר אֶחָד בְּבְרָכָה אַחַת מִכָּל הַלֵּב, וְאֶרְשֹׁם מָה הִרְגַּשְׁתִּי אֲנִי וּמָה הִרְגִּישׁ הוּא.

📜 גָּדוֹל הַשָּׁלוֹם

בַּמִּדְרָשׁ בַּמִּדְבָּר רַבָּה יֵשׁ כִּשְׁלוֹשִׁים אֲמִירוֹת שֶׁמַּתְחִילוֹת בִּמְלִים ׳גָּדוֹל הַשָּׁלוֹם׳. כָּל אַחַת מְלַמֶּדֶת עוֹד סִבָּה שֶׁהַשָּׁלוֹם חָשׁוּב.

לְמָשָׁל: הַנְּבִיאִים נָטְעוּ בְּפִי כֻּלָּם רַק אֶת הַמִּלָּה שָׁלוֹם. שָׁלוֹם נִתַּן לַעֲנָוִים, לְאוֹהֲבֵי תוֹרָה וּלְלוֹמְדֶיהָ, וּלְעוֹשֵׂי צְדָקָה.

וְעוֹד: שָׁלוֹם שָׁקוּל כְּנֶגֶד הַכֹּל.

הַמִּדְרָשׁ מוֹנֶה הַרְבֵּה סִבּוֹת לְ׳גָּדוֹל הַשָּׁלוֹם׳.

📖 שָׁלוֹם וְוִתּוּר – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, עוּקְצִין ג׳ י״ב · תְּהִלִּים כ״ט י״א

רָמָה ב׳ – כְּלִי מַחֲזִיק בְּרָכָה

🫙 כְּלִי מַחֲזִיק בְּרָכָה

רַבִּי שִׁמְעוֹן בֶּן חַלַּפְתָּא סִיֵּם בָּזֶה אֶת הַמִּשְׁנָיוֹת: ׳לֹא מָצָא הַקָּדוֹשׁ בָּרוּךְ הוּא כְּלִי מַחֲזִיק בְּרָכָה לְיִשְׂרָאֵל אֶלָּא הַשָּׁלוֹם׳, שֶׁנֶּאֱמַר: ׳ה׳ עֹז לְעַמּוֹ יִתֵּן, ה׳ יְבָרֵךְ אֶת עַמּוֹ בַּשָּׁלוֹם׳.

הַפֵּרוּשׁ: כָּל הַבְּרָכוֹת שֶׁבָּעוֹלָם אֵינָן מִתְקַיְּמוֹת אִם אֵין שָׁלוֹם.

מָשָׁל: אָדָם מַכְנִיס חֲפָצִים לְשַׂקִּית, וְהַכֹּל נוֹפֵל, כִּי יֵשׁ חוֹר. כָּךְ ה׳ מַשְׁפִּיעַ בְּרָכָה, אֲבָל אָדָם שֶׁאֵינוֹ שָׁלֵם דּוֹמֶה לְשַׂקִּית עִם חוֹר.

בְּרָכָה נִשְׁמֶרֶת רַק אֵצֶל מִי שֶׁשָּׁלֵם וּבְשָׁלוֹם.

🔏

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי מְבַקֵּשׁ שָׁלוֹם בַּתְּפִלָּה, אֲנִי מְבַקֵּשׁ שֶׁכָּל הַטּוֹב שֶׁה׳ נוֹתֵן יִשָּׁאֵר 💛

📖 שָׁלוֹם וְוִתּוּר – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בַּמִּדְבָּר רַבָּה י״א · נְתִיבוֹת עוֹלָם

רָמָה ב׳ – כְּלִי מַחֲזִיק בְּרָכָה

🔏 שָׁלוֹם – הַחוֹתָם

בִּרְכַּת כֹּהֲנִים מִסְתַּיֶּמֶת בְּ׳וְיָשֵׂם לְךָ שָׁלוֹם׳, וְחֲכָמִים לָמְדוּ: אֵין הַבְּרָכוֹת מוֹעִילוֹת כְּלוּם אֶלָּא אִם כֵּן יֵשׁ שָׁלוֹם עִמָּהֶן. גַּם רַבִּי אֶלְעָזָר הַקַּפָּר אָמַר: אֵין חוֹתֵם אֶת כָּל הַתְּפִלָּה אֶלָּא שָׁלוֹם.

שִׁשָּׁה סִדְרֵי מִשְׁנָה, שְׁמוֹנֶה עֶשְׂרֵה, קַדִּישׁ וּבִרְכַּת הַמָּזוֹן – כֻּלָּם מִסְתַּיְּמִים בְּשָׁלוֹם.

בְּסֵפֶר נְתִיבוֹת עוֹלָם כָּתוּב: אֵין קִיּוּם לַדָּבָר אֶלָּא בְּחוֹתָמוֹ, וְשָׁלוֹם נוֹתֵן קִיּוּם. בְּלִי שָׁלוֹם יֵשׁ שִׁנּוּי, וּבְשָׁלוֹם הַכֹּל עוֹמֵד.

שָׁלוֹם הוּא הַחוֹתָם שֶׁנּוֹתֵן קִיּוּם לַבְּרָכָה.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה לְה׳ עַל כָּל רֶגַע שֶׁל שָׁלוֹם בַּבַּיִת, בַּכִּתָּה וּבַחֶדֶר 💛

📖 שָׁלוֹם וְוִתּוּר – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּרֵאשִׁית כ״ח כ״א · מַלְאָכִי ב׳ ה׳ · יְשַׁעְיָהוּ ל״ב י״ח

רָמָה ב׳ – כְּלִי מַחֲזִיק בְּרָכָה

🏆 שָׁלוֹם הוּא הַשָּׂכָר

רַבִּים מִגְּדוֹלֵי הַתּוֹרָה קִבְּלוּ שָׁלוֹם בְּשָׂכָר. יַעֲקֹב בִּקֵּשׁ: ׳וְשַׁבְתִּי בְשָׁלוֹם אֶל בֵּית אָבִי׳. עַל אַהֲרֹן כָּתוּב: ׳בְּרִיתִי הָיְתָה אִתּוֹ הַחַיִּים וְהַשָּׁלוֹם׳.

וּלְיִשְׂרָאֵל מַבְטִיחַ ה׳: ׳וְנָתַתִּי שָׁלוֹם בָּאָרֶץ׳, וְגַם: ׳וְיָשַׁב עַמִּי בִּנְוֵה שָׁלוֹם׳.

מִי שֶׁשּׁוֹמֵר תּוֹרָה וּמִצְווֹת מְקַבֵּל שָׁלוֹם, וְשָׁלוֹם הוּא מַתָּנָה גְּדוֹלָה.

הַקָּדוֹשׁ בָּרוּךְ הוּא נוֹתֵן שָׁלוֹם כְּשָׂכָר לַצַּדִּיקִים.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַתְחִיל אֶת הַיּוֹם בְּשָׁלוֹם, כִּי כָּךְ אֲנִי מִתְקָרֵב לְה׳ 💛

רָמָה ב׳ – כְּלִי מַחֲזִיק בְּרָכָה

🧭 דֶּרֶךְ וּמַטָּרָה

הַשָּׁלוֹם הוּא גַּם כְּלִי שֶׁמַּחֲזִיק בְּרָכָה, וְגַם מַטָּרָה: ׳וְכָל נְתִיבוֹתֶיהָ שָׁלוֹם׳ וְכָל הַתּוֹרָה נִתְּנָה לַעֲשׂוֹת שָׁלוֹם.

שָׁלוֹם בֵּין אָדָם לַחֲבֵרוֹ הוּא כְּלִי לִבְרָכָה, וְשָׁלוֹם בֵּין אָדָם לַבּוֹרֵא הוּא מַטָּרָה.

וּמָה שֶׁהוּא מַטָּרָה בַּסּוֹף צָרִיךְ גַּם לְהַתְחִיל בּוֹ, כִּי אִם לֹא מַתְחִילִים, לֹא מַגִּיעִים.

הַשָּׁלוֹם הוּא גַּם הַדֶּרֶךְ וְגַם הַמַּטָּרָה.

🛍️

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְבָרֵךְ חֲבֵרִים וְסוֹלֵחַ לָהֶם, וְכָךְ אֲנִי מַרְגִּישׁ אֶת בִּרְכַּת ה׳ 💛

📖 שָׁלוֹם וְוִתּוּר – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – כְּלִי מַחֲזִיק בְּרָכָה

🛍️ שַׂקִּית בְּרָכוֹת בַּכִּתָּה

מְכִינִים בַּכִּתָּה שַׂקִּית בְּרָכוֹת: כָּל תַּלְמִיד כּוֹתֵב בְּרָכָה לְחָבֵר וְשָׂם בַּשַּׂקִּית.

כְּדֵי שֶׁהַשַּׂקִּית תַּחֲזִיק, צָרִיךְ לְתַקֵּן אֶת הַחוֹרִים: סְלִיחָה, פִּיּוּס, וִתּוּר קָטָן. כָּל תִּקּוּן הוּא תְּפִירָה שֶׁל שָׁלוֹם.

הַבְּרָכָה שֶׁאֲנִי מְאַחֵל לְחָבֵר מְחַזֶּקֶת גַּם אוֹתִי.

מְתַקְּנִים חוֹרִים בְּסְלִיחָה, וְכָךְ הַבְּרָכָה נִשְׁמֶרֶת.

מָה לוֹקְחִים מִרָמָה ב׳?

הַשָּׁלוֹם הוּא הַכְּלִי שֶׁשּׁוֹמֵר עַל כָּל הַטּוֹב שֶׁה׳ נוֹתֵן. כְּשֶׁאֲנִי מְתַקֵּן חוֹר שֶׁל רִיב וּמְבָרֵךְ חָבֵר, אֲנִי שׁוֹמֵר עַל הַבְּרָכָה, וְאַהֲבָתִי לַקָּדוֹשׁ בָּרוּךְ הוּא גְּדֵלָה.

  • מָה הַ״חוֹר״ הַקָּטָן שֶׁיֵּשׁ בֵּינִי לְבֵין חָבֵר, וְאֵיךְ אֲתַקֵּן אוֹתוֹ?

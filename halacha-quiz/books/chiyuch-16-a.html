רַבִּי קָלְמָן הִשְׁאִיר לָנוּ מֶסֶר חָשׁוּב: ׳אֲנָשִׁים מְחַכִּים שֶׁיִּקְרֶה מַשֶּׁהוּ נוֹרָא כְּדֵי לְהִתְעוֹרֵר עַל הַחַיִּים. לָמָּה לְחַכּוֹת כְּדֵי לְהַתְחִיל לִשְׂמֹחַ?׳

הוּא שָׁאַל אוֹתָנוּ: יֵשׁ לָכֶם רַגְלַיִם שֶׁהוֹלְכוֹת? אֲוִיר בָּרֵאוֹת? אָז לָמָּה לִבְכּוֹת עַל פְּקָק, וְלִכְעֹס כְּשֶׁהַמִּיץ נִשְׁפָּךְ?

לֹא מְחַכִּים לְדָבָר נוֹרָא כְּדֵי לְהַתְחִיל לִשְׂמֹחַ.

9

💛 אַהֲבַת הַשֵּׁם: תּוֹדָה, ה׳, עַל עוֹד יוֹם 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַבִּי קָלְמָן גּוֹלְדְשְׁמִידְט

רָמָה א׳ – אֲנִי בּוֹחֵר לִשְׂמֹחַ

🙏 תּוֹדָה עַל עוֹד יוֹם

הַחִיּוּךְ הוּא הַדֶּרֶךְ לוֹמַר לַבּוֹרֵא: ׳תּוֹדָה שֶׁנָּתַתָּ לִי עוֹד יוֹם׳.

כְּשֶׁאֲנִי מְחַיֵּךְ בַּבֹּקֶר, אֲנִי אוֹמֵר תּוֹדָה עַל הָרַגְלַיִם שֶׁהוֹלְכוֹת, עַל הָאֲוִיר שֶׁנּוֹשְׁמִים וְעַל הַיּוֹם הֶחָדָשׁ.

וְכָךְ גַּם דְּבָרִים קְטַנִּים כְּמוֹ מִיץ שֶׁנִּשְׁפַּךְ נִרְאִים פָּחוֹת גְּדוֹלִים.

חִיּוּךְ הוּא דֶּרֶךְ לוֹמַר תּוֹדָה עַל עוֹד יוֹם.

10
💡

רָמָה א׳ – אֲנִי בּוֹחֵר לִשְׂמֹחַ

מָה לוֹקְחִים מִרָמָה א׳?

הַשִּׂמְחָה הִיא בְּחִירָה, וְה׳ נוֹתֵן לָנוּ כֹּחַ לִבְחֹר אוֹתָהּ. כְּשֶׁאֲנִי מְחַיֵּךְ גַּם בְּיוֹם קָשֶׁה, אֲנִי מַרְגִּישׁ אֶת אַהֲבָתוֹ שֶׁל ה׳ שֶׁתָּמִיד אִתִּי.

11
❤️
לָמָּה לְחַכּוֹת לְמַשֶּׁהוּ נוֹרָא כְּדֵי לְהַתְחִיל לִשְׂמֹחַ?

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי הָיָה לִי יוֹם קָשֶׁה, וּמָה הָיָה קוֹרֶה אִם הָיִיתִי בּוֹחֵר לְחַיֵּךְ?
  • מַה פֵּרוּשׁ שֶׁרַבִּי קָלְמָן ׳לֹא צָחַק כִּי הָיָה לוֹ מַצְחִיק, אֶלָּא כִּי רָצָה לִחְיוֹת׳?
  • הַשָּׁבוּעַ אֲנִי בּוֹחֵר הֶסְכֵּם קָטָן שֶׁל לִמּוּד אוֹ חֶסֶד כָּל יוֹם, וְכָל פַּעַם שֶׁאֲנִי מִתְלוֹנֵן עַל דָּבָר קָטָן, אֲנִי נִזְכָּר בְּרַבִּי קָלְמָן וּמוֹצֵא בּוֹ מַשֶּׁהוּ מַצְחִיק.

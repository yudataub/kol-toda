חֲתֻנָּה בְּהַר סִינַי

רָמָה ב׳ – חֻפָּה וְהַר

לָמָּה מַתַּן תּוֹרָה הוּא כְּמוֹ חֲתֻנָּה, מָה הַחֻפָּה וְהַנֵּרוֹת וְהַכּוֹס מַזְכִּירִים מִסִּינַי, וְאֵיךְ מְקַשְּׁטִים אֶת הָעֶרֶב שֶׁל שָׁבוּעוֹת

📖 שָׁבוּעוֹת – עָמַדְנוּ בְּהַר סִינַי · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
📜

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ שֶׁאֲנִי חֵלֶק מִן הַחֲתֻנָּה הַגְּדוֹלָה הַזֹּאת 💛

📖 שָׁבוּעוֹת – עָמַדְנוּ בְּהַר סִינַי – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תַּעֲנִית כ״ו ב׳

רָמָה ב׳ – חֻפָּה וְהַר

📜 הַמִּשְׁנָה בְּסוֹף מַסֶּכֶת תַּעֲנִית

בַּמִּשְׁנָה כָּתוּב עַל הַפָּסוּק ״בְּיוֹם חֲתֻנָּתוֹ וּבְיוֹם שִׂמְחַת לִבּוֹ״: ״בְּיוֹם חֲתֻנָּתוֹ – זֶה מַתַּן תּוֹרָה.״

הַחֲתֻנָּה הִיא בֵּין ה׳ לְבֵין עַם יִשְׂרָאֵל, וְהַמַּתָּנָה הִיא הַתּוֹרָה.

בַּכְּתֻבָּה שֶׁל חַג הַשָּׁבוּעוֹת, שֶׁהִתְחַבֵּר בְּצָפַת רַבִּי יִשְׂרָאֵל נַגָּרָה, מְתֹאָר ה׳ שֶׁיּוֹרֵד אֶל הַר סִינַי בְּשִׂמְחָה.

מַתַּן תּוֹרָה הוּא יוֹם הַחֲתֻנָּה בֵּין ה׳ לְעַם יִשְׂרָאֵל.

1
🧍

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד מִנְהָגִים וּמַרְגִּישׁ אֶת הַקֶּשֶׁר לְסִינַי 💛

📖 שָׁבוּעוֹת – עָמַדְנוּ בְּהַר סִינַי – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָאֶלֶף לְךָ שְׁלֹמֹה · שְׁמוֹת י״ט

רָמָה ב׳ – חֻפָּה וְהַר

🧍 עוֹמְדִים בַּחֻפָּה

רַבִּי שְׁלֹמֹה קְלוּגֶר נִשְׁאַל בַּסֵּפֶר ״הָאֶלֶף לְךָ שְׁלֹמֹה״: אֶפְשָׁר לָשֶׁבֶת בִּשְׁעַת הַחֻפָּה?

הוּא עָנָה: בְּתִקּוּנֵי הַזֹּהַר כָּתוּב שֶׁכָּל עִנְיְנֵי הַחֻפָּה נִלְמָדִים מִמַּעֲמַד הַר סִינַי. וּבְסִינַי כָּתוּב ״וַיַּעַמְדוּ מֵרָחֹק״ וְ״וַיִּתְיַצְּבוּ בְּתַחְתִּית הָהָר״.

לָכֵן עוֹמְדִים גַּם בַּחֻפָּה. וּמוּסָף עוֹד: חֲכָמִים דָּרְשׁוּ שֶׁה׳ כָּפָה עֲלֵיהֶם אֶת הָהָר כְּמוֹ גִּיגִית, כְּמוֹ חֻפָּה.

הַחֻפָּה הִיא כְּמֵעֵין מַעֲמַד הַר סִינַי, וְלָכֵן עוֹמְדִים.

2
🥂

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא מִתְיָאֵשׁ, כִּי אֶפְשָׁר לָקוּם וְלִבְנוֹת מֵחָדָשׁ 💛

📖 שָׁבוּעוֹת – עָמַדְנוּ בְּהַר סִינַי – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁטַר הַתְּנָאִים שֶׁל מַתַּן תּוֹרָה

רָמָה ב׳ – חֻפָּה וְהַר

🥂 נֵרוֹת וְכוֹס שְׁבוּרָה

הַנֵּרוֹת בַּחֻפָּה הֵם, לְפִי מָה שֶׁאוֹמְרִים, זֵכֶר לַלַּפִּידִים שֶׁהָיוּ בְּסִינַי: ״וַיְהִי קֹלֹת וּבְרָקִים״.

שְׁבִירַת הַכּוֹס: הַפֵּרוּשׁ הַמֻּכָּר הוּא זֵכֶר לַחֻרְבָּן. פֵּרוּשׁ נוֹסָף: זֵכֶר לִשְׁבִירַת הַלּוּחוֹת.

בְּחֵטְא הָעֵגֶל נִשְׁבְּרוּ הַלּוּחוֹת, וְאַחַר כָּךְ נִתְּנוּ לוּחוֹת שְׁנִיִּים. אָמְרוּ לַחָתָן: גַּם אִם יִהְיֶה שֶׁבֶר, ״אַל תִּתְיָאֵשׁ״, אֶפְשָׁר לִבְנוֹת מֵחָדָשׁ, וְלִפְעָמִים אֶפְשָׁר לִבְנוֹת אֲפִלּוּ גָּדוֹל יוֹתֵר.

הַשֶּׁבֶר אֵינוֹ הַסּוֹף: אַחֲרָיו בּוֹנִים מֵחָדָשׁ.

❤️

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַרְגִּישׁ שֶׁה׳ מְחַכֶּה לִי בְּאַהֲבָה 💛

📖 שָׁבוּעוֹת – עָמַדְנוּ בְּהַר סִינַי – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָאוֹר הַחַיִּים, שְׁמוֹת י״ט

רָמָה ב׳ – חֻפָּה וְהַר

❤️ ״בְּיוֹם הַזֶּה בָּאוּ מִדְבַּר סִינַי״

הָאוֹר הַחַיִּים שׁוֹאֵל: לָמָּה הַתּוֹרָה כּוֹתֶבֶת קֹדֶם ״בְּיוֹם הַזֶּה בָּאוּ מִדְבַּר סִינַי״, וְרַק אַחַר כָּךְ ״וַיִּסְעוּ מֵרְפִידִים״? הָיָה צָרִיךְ לִכְתֹּב קֹדֶם אֶת הַנְּסִיעָה!

הוּא עוֹנֶה: ״הָאַהֲבָה מְקַלְקֶלֶת אֶת הַשּׁוּרָה״. כָּל הָעוֹלָמוֹת חִכּוּ לְיוֹם הַזֶּה, וּכְשֶׁהֶגִּיעוּ, הַכָּתוּב לֹא עָצַר כֹּחַ לְסַפֵּר אֶת סֵדֶר הַדְּבָרִים. הוּא הִקְדִּים אֶת הַבְּשׂוֹרָה.

בְּרֹאשׁ חֹדֶשׁ סִיוָן ה׳ הֶחְזִיק אֶת בְּנֵי יִשְׂרָאֵל בַּדֶּרֶךְ, כִּי אִם הָיוּ מַגִּיעִים מֻקְדָּם, הוּא לֹא הָיָה יָכוֹל לְהַמְתִּין מִלָּתֵת לָהֶם אֶת הַתּוֹרָה.

ה׳ שָׂמֵחַ כָּל כָּךְ בָּנוּ, שֶׁהוּא לֹא יָכוֹל לְהַמְתִּין לָתֵת לָנוּ אֶת הַתּוֹרָה.

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ לְקַשֵּׁט אֶת עַצְמִי בַּתּוֹרָה 💛

📖 שָׁבוּעוֹת – עָמַדְנוּ בְּהַר סִינַי – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הַכְּתֻבָּה שֶׁל חַג הַשָּׁבוּעוֹת

רָמָה ב׳ – חֻפָּה וְהַר

📚 עֶשְׂרִים וְאַרְבָּעָה קִשּׁוּטִים

חֲכָמִים אָמְרוּ: כְּמוֹ שֶׁהַכַּלָּה מְקֻשֶּׁטֶת בְּעֶשְׂרִים וְאַרְבָּעָה קִשּׁוּטִים, כָּךְ עַם יִשְׂרָאֵל צָרִיךְ לִהְיוֹת בָּקִי בְּעֶשְׂרִים וְאַרְבָּעָה סִפְרֵי הַתָּנַ״ךְ.

בַּזֹּהַר מוּבָא מִנְהַג תִּקּוּן לֵיל שָׁבוּעוֹת: לָמָּה עֵרִים בַּלַּיְלָה? כִּי בַּבֹּקֶר יֵשׁ חֲתֻנָּה, וְצָרִיךְ לְקַשֵּׁט.

לָכֵן קוֹרְאִים בַּלַּיְלָה מִכָּל סִפְרֵי הַתָּנַ״ךְ, וְעַם יִשְׂרָאֵל מַגִּיעַ מְקֻשָּׁט לַחֻפָּה.

בְּלֵיל שָׁבוּעוֹת מְקַשְּׁטִים בְּעֶשְׂרִים וְאַרְבָּעָה סִפְרֵי הַתָּנַ״ךְ.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְהַמְשִׁיךְ לִלְמֹד, גַּם כְּשֶׁקָּשֶׁה 💛

📖 שָׁבוּעוֹת – עָמַדְנוּ בְּהַר סִינַי – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִכְתַּב רַבִּי שְׁלֹמֹה אַלְקַבֵּץ, הַשְּׁל״ה

רָמָה ב׳ – חֻפָּה וְהַר

🕯️ מַעֲשֶׂה בְּרַבִּי יוֹסֵף קָארוֹ

בְּלֵיל שָׁבוּעוֹת, בְּחֶבְרַת הַלּוֹמְדִים שֶׁל רַבִּי יוֹסֵף קָארוֹ, שָׁמַע רַבִּי שְׁלֹמֹה אַלְקַבֵּץ וְגַם הַחֲבֵרִים קוֹל מְדַבֵּר מִפִּיו שֶׁל רַבִּי יוֹסֵף קָארוֹ, קוֹל חָזָק בְּחִתּוּךְ אוֹתִיּוֹת. כֻּלָּם נָפְלוּ עַל פְּנֵיהֶם מֵרֹב יִרְאָה.

הַקּוֹל אָמַר: ״אַשְׁרֵיכֶם בָּנַי, שׁוּבוּ אֶל לִמּוּדְכֶם וְאַל תַּפְסִיקוּ רֶגַע, וַעֲלוּ לְאֶרֶץ יִשְׂרָאֵל.״

אַחַר כָּךְ עָלוּ רַבִּי יוֹסֵף קָארוֹ וְרַבִּי שְׁלֹמֹה אַלְקַבֵּץ לִצְפַת. כָּךְ מַשְׁמִיעִים אֶת הַמַּעֲשֶׂה בְּמִכְתָּבוֹ שֶׁל רַבִּי שְׁלֹמֹה אַלְקַבֵּץ.

בְּלֵיל שָׁבוּעוֹת שָׁמְעוּ קוֹל שֶׁאָמַר: שׁוּבוּ אֶל לִמּוּדְכֶם וְאַל תַּפְסִיקוּ.

רָמָה ב׳ – חֻפָּה וְהַר

מָה לוֹקְחִים מִרָמָה ב׳?

הַחֻפָּה, הַנֵּרוֹת וְהַכּוֹס הַשְּׁבוּרָה מַזְכִּירִים לִי אֶת הַר סִינַי. חֲכָמִים לִמְּדוּ שֶׁמַּתַּן תּוֹרָה הוּא יוֹם חֲתֻנָּה שֶׁל שִׂמְחָה, וְשֶׁה׳ כָּל כָּךְ רוֹצֶה לָתֵת לָנוּ אֶת הַתּוֹרָה בְּאַהֲבָה.

  • מָה אֲנִי מַרְגִּישׁ כְּשֶׁאֲנִי חוֹשֵׁב שֶׁמַּתַּן תּוֹרָה הוּא יוֹם שֶׁל שִׂמְחָה כְּמוֹ חֲתֻנָּה?
  • אֵיזֶה מִנְהָג בַּחֲתֻנָּה מֵזְכִּיר לִי אֶת הַר סִינַי, וְלָמָּה?
  • הַיּוֹם אֲכִין קִשּׁוּט אֶחָד לְחַג הַשָּׁבוּעוֹת: פָּסוּק אֶחָד שֶׁאֲנִי אוֹהֵב.

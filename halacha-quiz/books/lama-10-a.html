רָמָה א׳ – הַמִּצְווֹת בּוֹנוֹת אוֹתִי

❓ הַשְּׁאֵלָה שֶׁל הַמּוֹשֵׁל

פַּעַם שָׁאַל מוֹשֵׁל רוֹמִי, טוּרְנוּסְרוּפוּס, אֶת רַבִּי עֲקִיבָא: ׳אִם אֱלֹהֵיכֶם אוֹהֵב עֲנִיִּים, לָמָּה הוּא לֹא מְפַרְנֵס אוֹתָם בְּעַצְמוֹ?׳

זוֹ שְׁאֵלָה טוֹבָה! וְהַתְּשׁוּבָה שֶׁל רַבִּי עֲקִיבָא מְלַמֶּדֶת אוֹתָנוּ לָמָּה יֵשׁ מִצְווֹת: הַנְּתִינָה לֶעָנִי מַצִּילָה וּמְתַקֶּנֶת גַּם אוֹתָנוּ.

הַנְּתִינָה לֶעָנִי מְתַקֶּנֶת גַּם אֶת הַנּוֹתֵן.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה לַה׳ שֶׁנָּתַן לִי זְכוּת לַעֲזֹר וּלְהִשְׁתַּפֵּר 💛

📖 לָמָּה עוֹשִׂים מִצְווֹת? – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – הַמִּצְווֹת בּוֹנוֹת אוֹתִי

🤲 הַתְּשׁוּבָה: גַּם בִּשְׁבִילֵנוּ

ה׳ מוֹצֵא הַרְבֵּה דְּרָכִים לְפַרְנֵס אֶת הָעֲנִיִּים. אִם הַמַּטָּרָה הָיְתָה רַק שֶׁלָּעָנִי יִהְיֶה, הוּא כְּבָר הָיָה דּוֹאֵג לָזֶה.

הַמִּצְוָה לָתֵת נִתְּנָה גַּם בִּשְׁבִילֵנוּ, כְּדֵי שֶׁנִּהְיֶה בְּנֵי אָדָם טוֹבִים יוֹתֵר. הֶעָנִי מְקַבֵּל בֶּאֱמֶת, וְגַם אֲנַחְנוּ מִתְחַזְּקִים.

הַמִּצְוָה נִתְּנָה גַּם כְּדֵי לְתַקֵּן אוֹתִי.

🌱

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁהַלֵּב שֶׁלִּי יִהְיֶה טוֹב וְנָדִיב, כְּמוֹ שֶׁה׳ אוֹהֵב 💛

📖 לָמָּה עוֹשִׂים מִצְווֹת? – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – הַמִּצְווֹת בּוֹנוֹת אוֹתִי

🌱 הַנּוֹתֵן מִשְׁתַּנֶּה

כְּשֶׁאֲנִי נוֹתֵן, לֹא רַק הַכִּיס שֶׁלִּי מִשְׁתַּנֶּה, גַּם הַלֵּב. כָּל פַּעַם שֶׁאֲנִי נוֹתֵן בְּאַהֲבָה, אֲנִי נַעֲשֶׂה קְצָת יוֹתֵר טוֹב לֵב.

זֶה כְּמוֹ אִמּוּן: מִי שֶׁמִּתְאַמֵּן בִּנְתִינָה, הַלֵּב שֶׁלּוֹ הוֹלֵךְ וְנַעֲשֶׂה חָזָק וְטוֹב.

כָּל נְתִינָה בְּאַהֲבָה בּוֹנָה אוֹתִי.

💛

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד לֶאֱהֹב אֶת הַחֲבֵרִים שֶׁלִּי, וְאוֹמֵר תּוֹדָה לַה׳ 💛

📖 לָמָּה עוֹשִׂים מִצְווֹת? – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, וַיִּקְרָא י״ט י״ח · מִיכָה ו׳ ח׳

רָמָה א׳ – הַמִּצְווֹת בּוֹנוֹת אוֹתִי

💛 הַתּוֹרָה אָמְרָה: ׳וְאָהַבְתָּ׳

הַתּוֹרָה לֹא אָמְרָה רַק ׳תֵּן׳. הִיא אָמְרָה ׳וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ׳. עִקַּר הַמִּצְוָה הוּא לֶאֱהֹב, וְהַנְּתִינָה הִיא הַדֶּרֶךְ שֶׁמְּגַדֶּלֶת אֶת הָאַהֲבָה.

גַּם הַנָּבִיא מִיכָה אוֹמֵר מָה ה׳ דּוֹרֵשׁ מִמֶּנּוּ: ׳וְאַהֲבַת חֶסֶד׳, לֶאֱהֹב לַעֲשׂוֹת טוֹב.

לֹא רַק לָתֵת – גַּם לֶאֱהֹב.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְבַקֵּשׁ מֵה׳ שֶׁהַמִּצְווֹת יַבְרִיקוּ אֶת הַלֵּב שֶׁלִּי 💛

📖 לָמָּה עוֹשִׂים מִצְווֹת? – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּרֵאשִׁית רַבָּה מ״ד א׳

רָמָה א׳ – הַמִּצְווֹת בּוֹנוֹת אוֹתִי

✨ לְצָרֵף אֶת הַבְּרִיּוֹת

הַמִּדְרָשׁ אוֹמֵר: ׳לֹא נִתְּנוּ הַמִּצְווֹת אֶלָּא לְצָרֵף בָּהֶן אֶת הַבְּרִיּוֹת׳.

׳לְצָרֵף׳ פֵּרוּשׁוֹ לְזַקֵּק וּלְצַחְצֵחַ, כְּמוֹ זָהָב שֶׁמְּנַקִּים בָּאֵשׁ. הַמִּצְווֹת נִתְּנוּ לְטוֹבָתֵנוּ: הֵן מְזַכּוֹת אוֹתָנוּ וּמַבְרִיקוֹת אֶת הַלֵּב.

הַמִּצְווֹת מְצָרְפוֹת וּמַבְרִיקוֹת אוֹתִי.

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד מֵאַבְרָהָם לָתֵת בְּלֵב שָׂמֵחַ 💛

📖 לָמָּה עוֹשִׂים מִצְווֹת? – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּרֵאשִׁית י״ח · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – הַמִּצְווֹת בּוֹנוֹת אוֹתִי

🏕️ אַבְרָהָם וְהָאוֹרְחִים

אַבְרָהָם רָץ לְהָכִין אֹכֶל לִשְׁלוֹשָׁה אוֹרְחִים. בַּסּוֹף הִתְבָּרֵר שֶׁהֵם מַלְאָכִים שֶׁלֹּא הָיוּ צְרִיכִים אֹכֶל כְּלָל!

אָז לָמָּה פָּרָשַׁת הַחֶסֶד הִיא דַּוְקָא אֶצְלָם? הַתּוֹרָה מְלַמֶּדֶת שֶׁהַחֶסֶד הוּא קֹדֶם כֹּל שֶׁאֲנִי אֶתֵּן, וְהַנְּתִינָה בּוֹנָה אוֹתִי.

גַּם כְּשֶׁהָאוֹרֵחַ לֹא צָרִיךְ, הַנְּתִינָה בּוֹנָה אֶת הַנּוֹתֵן.

😊

💛 אַהֲבַת הַשֵּׁם: אֲנִי נוֹתֵן בְּחִיּוּךְ, כִּי אֲנִי שָׂמֵחַ לַעֲשׂוֹת רָצוֹן ה׳ 💛

📖 לָמָּה עוֹשִׂים מִצְווֹת? – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דְּבָרִים ט״ו י׳

רָמָה א׳ – הַמִּצְווֹת בּוֹנוֹת אוֹתִי

😊 לָתֵת בְּלֵב טוֹב

יֵשׁ דָּבָר חָשׁוּב: לָתֵת בְּשִׂמְחָה. הַתּוֹרָה אוֹמֶרֶת: ׳נָתוֹן תִּתֵּן לוֹ וְלֹא־יֵרַע לְבָבְךָ בְּתִתְּךָ לוֹ׳.

אִם נוֹתְנִים וְהַלֵּב כּוֹעֵס, הַלֵּב לֹא מִתְחַזֵּק כְּמוֹ שֶׁצָּרִיךְ. אִם נוֹתְנִים בְּלֵב טוֹב, הַנְּתִינָה בּוֹנָה אוֹתָנוּ.

נוֹתְנִים בְּשִׂמְחָה, כְּדֵי שֶׁהַלֵּב יִתְחַזֵּק.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה לַה׳ עַל הַזְּכוּת לִהְיוֹת שָׁלִיחַ טוֹב שֶׁלּוֹ 💛

📖 לָמָּה עוֹשִׂים מִצְווֹת? – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🗝️ זְכוּת לָתֵת

ה׳ מְפַרְנֵס אֶת כֻּלָּם, וְלִפְעָמִים הוּא נוֹתֵן לָנוּ זְכוּת לִהְיוֹת הַשְּׁלִיחִים שֶׁלּוֹ. מִי שֶׁזּוֹכֶה לַעֲזֹר, מַעֲלָתוֹ גְּדוֹלָה.

כָּל מִצְוָה בֵּין אָדָם לַחֲבֵרוֹ הִיא אִמּוּן לַלֵּב, וְהַלֵּב הוּא מַה שֶּׁה׳ אוֹהֵב לִרְאוֹת.

הַמִּצְווֹת הֵן אִמּוּן לַלֵּב וּזְכוּת לָתֵת.

מָה לוֹקְחִים מִרָמָה א׳?

ה׳ אוֹהֵב אוֹתִי וְרוֹצֶה לְהֵיטִיב לִי. הוּא נָתַן לִי מִצְווֹת, וְהֵן לֹא רַק עֲבוֹדָה: הֵן הַדֶּרֶךְ שֶׁבָּהּ הַלֵּב שֶׁלִּי נַעֲשֶׂה טוֹב וְחָזָק. כְּשֶׁאֲנִי נוֹתֵן בְּאַהֲבָה, אֲנִי מִתְקָרֵב אֵלָיו.

🤔 חוֹשְׁבִים עַל זֶה

  • אֵיזוֹ מִצְוָה שֶׁנָּתַתִּי בָּהּ הַשָּׁבוּעַ שִׁנְּתָה אוֹתִי, וּמָה הִרְגַּשְׁתִּי?
  • אֵיךְ אֶתֵּן הַיּוֹם בְּלֵב טוֹב, אֲפִלּוּ כְּשֶׁאֵין לִי חֵשֶׁק?
  • אָכִין ׳אוֹצַר מִצְווֹת׳ אִישִׁי: בְּכָל יוֹם אַכְנִיס יַהֲלוֹם – מִצְוָה שֶׁעָשִׂיתִי וּמָה הִיא עָשְׂתָה לַלֵּב שֶׁלִּי.

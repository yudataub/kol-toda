🚜 טְרַקְטוֹר בְּלִי כִּוּוּן

טְרַקְטוֹר גָּדוֹל יוֹדֵעַ לַחְפֹּר, אֲבָל אִם אֵין מִי שֶׁיֹּאמַר לוֹ מָה לִבְנוֹת, הוּא חוֹפֵר חוֹר כָּאן וְחוֹר שָׁם, וְלֹא נִבְנֶה שׁוּם בִּנְיָן.

בַּשִּׁעוּר אוֹמְרִים: הַמִּצְווֹת הֵן כְּמוֹ טְרַקְטוֹר, וְהַתּוֹרָה הִיא הַטְּרַקְטוֹר הַגָּדוֹל מִכֻּלָּם.

אִישׁ לֹא קוֹנֶה טְרַקְטוֹר כְּדֵי לִנְסֹעַ בּוֹ. צָרִיךְ לָדַעַת מָה בּוֹנִים.

הַמִּצְווֹת הֵן כְּמוֹ טְרַקְטוֹר. צָרִיךְ לָדַעַת מָה בּוֹנִים.

🏠 מָה בּוֹנִים?

הַמִּצְווֹת הֵן לֹא הַתַּכְלִית. הֵן אֶמְצָעִי לְהַגִּיעַ לְקִרְבַת ה׳ – לְאַהֲבַת ה׳.

הַבִּנְיָן שֶׁאָנוּ בּוֹנִים בְּכָל מִצְוָה הוּא הַקֵּרֻב שֶׁלָּנוּ אֶל ה׳.

כָּל יוֹם אֲנִי יָכוֹל לִבְנוֹת עוֹד קְצָת.

הַמִּצְווֹת בּוֹנוֹת בִּנְיָן: קִרְבַת ה׳.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְכַוֵּן אֶת הַמִּצְווֹת שֶׁלִּי אֵלֶיךָ, ה׳ 💛

רָמָה א׳ – הָרָצוֹן מְכַוֵּן

🧭 הָרָצוֹן מְכַוֵּן

כְּשֶׁיֵּשׁ לָאָדָם רָצוֹן עָצוּם לְהַשִּׂיג אַהֲבַת ה׳, הַטְּרַקְטוֹר יוֹדֵעַ הֵיכָן לַחְפֹּר וּלְאֵיזֶה כִּוּוּן.

כְּשֶׁאֵין שְׁאִיפָה, כָּל יוֹם חוֹפְרִים חוֹר אַחֵר: קְצָת מִתְקָרְבִים, אֲבָל עֲדַיִן רְחוֹקִים מִבִּנְיָן.

הָרָצוֹן הוּא הַמְּכַוֵּן שֶׁל כָּל הַמִּצְווֹת.

הָרָצוֹן מַגְדִּיר לְאָן הַמִּצְווֹת הוֹלְכוֹת.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁהַמִּצְווֹת יִגְּעוּ בַּלֵּב שֶׁלִּי, ה׳ 💛

📖 רָצוֹן וְחֵשֶׁק לַעֲבוֹדַת ה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🤝 הַכֹּל תָּלוּי בָּרָצוֹן

בַּשִּׁעוּר שׁוֹאֲלִים: הַאִם עַל יְדֵי חֶסֶד נַשִּׂיג אַהֲבַת חֶסֶד? תָּלוּי כַּמָּה רָצוֹן יֵשׁ לָנוּ.

הַאִם עַל יְדֵי בְּרָכוֹת נַשִּׂיג יִרְאַת שָׁמַיִם? תָּלוּי כַּמָּה אָנוּ רוֹצִים.

כָּךְ בְּכָל מִצְוָה: כְּשֶׁאֲנִי בֶּאֱמֶת רוֹצֶה, הִיא מַגִּיעָה אֶל הַלֵּב.

כַּמָּה הַמִּצְוָה תִּגַּע בַּלֵּב – תָּלוּי כַּמָּה אֲנִי רוֹצֶה.

📋 מָה בּוֹנָה כָּל מִצְוָה?

תְּפִלָּה – קוֹנִים אֱמוּנָה.

בְּרָכוֹת – קוֹנִים יִרְאַת שָׁמַיִם.

תּוֹרָה – בּוֹנִים מִשְׁכָּן לַשְּׁכִינָה.

חֶסֶד – בּוֹנִים לֵב שֶׁאוֹהֵב חֶסֶד.

מִזְמוֹרֵי תּוֹדָה – מַכִּירִים טוֹבָה לַה׳.

חֲשִׁיבָה עַל הַשֵּׁנִי – אַהֲבַת יִשְׂרָאֵל.

לְכָל מִצְוָה יֵשׁ בִּנְיָן שֶׁהִיא בּוֹנָה בְּתוֹכִי.

רָמָה א׳ – הָרָצוֹן מְכַוֵּן

👀 רוֹאִים מַה שֶׁרוֹצִים

אָדָם רוֹאֶה אֶת מַה שֶׁהוּא רוֹצֶה לִרְאוֹת.

מִי שֶׁלּוֹמֵד וְרוֹצֶה לְהַגִּיעַ לְאַהֲבַת ה׳ – מוֹצֵא בַּלִּמּוּד הַרְבֵּה אַהֲבַת ה׳.

מִי שֶׁלֹּא שׁוֹאֵף לְכָךְ, אוּלַי לֹא יִשְׂמַח לֵב אֶל מַה שֶׁנִּמְצָא שָׁם.

מִי שֶׁרוֹצֶה אַהֲבַת ה׳ מוֹצֵא אוֹתָהּ בַּלִּמּוּד.

⏳ כְּשֶׁמּוֹנְעִים – הַשְּׁאִיפָה גְּדֵלָה

לִפְעָמִים לֹא מַצְלִיחִים לַעֲשׂוֹת מַה שֶׁרוֹצִים, כִּי יֵשׁ מְנִיעָה. אֲבָל כְּשֶׁמּוֹנְעִים מֵאָדָם דָּבָר, הַשְּׁאִיפָה שֶׁלּוֹ גְּדֵלָה.

כְּגֹדֶל הַשְּׁאִיפָה כָּךְ גֹּדֶל הַהֲנָאָה כְּשֶׁמַּשִּׂיגִים.

לָכֵן לֹא נִשְׁבָּרִים: הַמְּנִיעָה יְכוֹלָה לְהַגְדִּיל אֶת הָרָצוֹן לִקְרַבַת ה׳.

מְנִיעָה מַגְדִּילָה אֶת הַשְּׁאִיפָה.

🎤 דָּבָר אֶחָד שֶׁאֲנִי רוֹצֶה בֶּאֱמֶת

מְשׂוֹחֲחִים: מָה הַדָּבָר הָאֶחָד שֶׁאֲנִי רוֹצֶה בֶּאֱמֶת הַשָּׁבוּעַ? לְמָשָׁל, לְהִתְפַּלֵּל בְּכַוָּנָה, לַעֲזֹר בַּבַּיִת, אוֹ לִלְמֹד בִּזְמַן קָבוּעַ.

בּוֹחֲרִים אֶחָד, וְעוֹשִׂים פְּעֻלָּה אַחַת קְטַנָּה בִּשְׁבִילוֹ כְּבָר הַיּוֹם.

וּבְסוֹף הַשָּׁבוּעַ מְסַפְּרִים אֵיךְ הָיָה.

בּוֹחֲרִים דָּבָר אֶחָד שֶׁרוֹצִים, וְעוֹשִׂים פְּעֻלָּה קְטַנָּה.

מָה לוֹקְחִים מִרָמָה א׳?

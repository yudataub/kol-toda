הַפָּרָשָׁה הַחֲשׁוּבָה בְּיוֹתֵר – אֵיךְ מַגִּיעִים לְאַהֲבַת ה׳

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

שֶׁבַע פְּעֻלּוֹת לַפָּרָשָׁה, ״ה׳ אֱלֹהֵינוּ״, ״בְּכָל לְבָבְךָ וְנַפְשְׁךָ וּמְאֹדֶךָ״ וְהַדֶּרֶךְ לְאַהֲבָה

📖 פְּסוּקִים ד׳–ט׳ · שִׁעוּר שֶׁל הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁאֲנִי נֶהֱנֶה מֵאֹכֶל טָעִים, אֲנִי יָכוֹל לוֹמַר: תּוֹדָה, ה׳, שֶׁאַתָּה אוֹהֵב אוֹתִי 💛

📖 פָּרָשַׁת שְׁמַע – דְּבָרִים ו׳, פֶּרֶק ו׳, פָּסוּק ה׳ (רַשִׁ״י)

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

⚔️ בִּשְׁנֵי יְצָרֶיךָ

״בְּכָל לְבָבְךָ״ – חֲזַ״ל דּוֹרְשִׁים: בִּשְׁנֵי יְצָרֶיךָ, בַּיֵּצֶר הַטּוֹב וּבַיֵּצֶר הָרַע. כְּלוֹמַר, גַּם כְּשֶׁעוֹסְקִים בַּיֵּצֶר הָרַע, זוֹכְרִים אֶת ה׳.

יֵשׁ לָזֶה שְׁנֵי פֵּרוּשִׁים:

(א) לְהִתְגַּבֵּר – לְוַתֵּר עַל הַתַּאֲוָה לִכְבוֹד ה׳.

(ב) לְהִשְׁתַּמֵּשׁ – לָקַחַת אֶת הַכֹּחַ שֶׁל הַיֵּצֶר לְטוֹבָה. לְמָשָׁל, נֶהֱנִים מֵאֹכֶל טָעִים וּמַרְגִּישִׁים מִתּוֹךְ זֶה אַהֲבַת ה׳, אוֹ עוֹשִׂים חִידוֹן יָפֶה מִתּוֹךְ שִׂמְחָה וְרָצוֹן לִכְבוֹד ה׳.

אֶפְשָׁר לְהִתְגַּבֵּר עַל הַיֵּצֶר, וְאֶפְשָׁר גַּם לַהֲפֹךְ אֶת כֹּחוֹ לְאַהֲבַת ה׳.

1

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹחֵר לַחְשֹׁב עַל דְּבָרִים טוֹבִים שֶׁה׳ עוֹשֶׂה לִי, וְהַלֵּב שֶׁלִּי מִתְמַלֵּא אַהֲבָה 💛

📖 פָּרָשַׁת שְׁמַע – דְּבָרִים ו׳, פֶּרֶק ו׳, פָּסוּק ו׳ (הָרַמְבָּ״ם)

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

🧩 ״לֵב״ בַּתּוֹרָה – מַחְשָׁבָה עֲמֻקָּה

בַּתּוֹרָה ״לֵב״ פֵּרוּשׁוֹ מַחְשָׁבָה עֲמֻקָּה, כְּמוֹ ״וְשַׂמְתֶּם עַל לְבַבְכֶם״ וְ״עַל לְבָבֶךָ״.

אִי אֶפְשָׁר לְצַוּוֹת עַל הַלֵּב ״תַּרְגִּישׁ!״. אֲבָל אֶפְשָׁר לִבְחֹר מַחְשָׁבוֹת. מִי שֶׁמְּפָרֵשׁ דָּבָר אַחֶרֶת, חוֹשֵׁב אַחֶרֶת, וּלְאַט לְאַט גַּם מַרְגִּישׁ אַחֶרֶת.

הָרַמְבָּ״ם מְלַמֵּד: מִי שֶׁאוֹהֵב, חוֹשֵׁב עַל מִי שֶׁהוּא אוֹהֵב כָּל הַיּוֹם. כָּךְ אַהֲבַת ה׳ – הַמַּחְשָׁבוֹת רָצוֹת אֵלָיו מֵעַצְמָן.

אֵין מְצַוִּים עַל רֶגֶשׁ, אֲבָל אֶפְשָׁר לִבְחֹר מַחְשָׁבָה.

2

💛 אַהֲבַת הַשֵּׁם: הָרָצוֹן הֲכִי עָמֹק שֶׁלִּי הוּא לְשַׂמֵּחַ אֶת ה׳ 💛

📖 פָּרָשַׁת שְׁמַע – דְּבָרִים ו׳, פֶּרֶק ו׳, פָּסוּק ה׳ (רַשִׁ״י)

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

🔥 ״בְּכָל נַפְשְׁךָ״ – בְּכָל הָרָצוֹן

״נֶפֶשׁ״ הִיא הָרָצוֹן, וְזוֹ דַּרְגָּה גְּבוֹהָה עוֹד יוֹתֵר מִמַּחְשָׁבוֹת.

הָרָצוֹן הֲכִי עָמֹק שֶׁל כָּל אָדָם הוּא לִחְיוֹת. וְרַשִׁ״י מְבָאֵר: ״בְּכָל נַפְשְׁךָ – אֲפִלּוּ הוּא נוֹטֵל אֶת נַפְשְׁךָ״.

זֶה לֹא סִפּוּר מַפְחִיד. הַכַּוָּנָה פְּשׁוּטָה: הָרָצוֹן לְשַׂמֵּחַ אֶת ה׳ גָּדוֹל אֲפִלּוּ מֵהָרָצוֹן לִחְיוֹת, כֵּן, אֲפִלּוּ מֵהָרָצוֹן הֲכִי חָזָק שֶׁיֵּשׁ לָנוּ.

הָרָצוֹן לְשַׂמֵּחַ אֶת ה׳ הוּא הַגָּדוֹל מִכָּל הָרְצוֹנוֹת.

3

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁיֵּשׁ לִי מִשְׂחָק אָהוּב מְאֹד, אֲנִי יוֹדֵעַ שֶׁאַהֲבַת ה׳ גְּדוֹלָה מִמֶּנּוּ 💛

📖 פָּרָשַׁת שְׁמַע – דְּבָרִים ו׳, פֶּרֶק ו׳, פָּסוּק ה׳ (רַשִׁ״י)

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

💎 ״בְּכָל מְאֹדֶךָ״ – מְאֹד מְאֹד

מָה אַתֶּם אוֹהֲבִים ״מְאֹד מְאֹד״? אוּלַי כֶּסֶף שֶׁחוֹסְכִים, אוֹ הַמִּשְׂחָק הָאָהוּב עֲלֵיכֶם.

״בְּכָל מְאֹדֶךָ״ אוֹמֵר: אַהֲבַת ה׳ גְּדוֹלָה יוֹתֵר מִכָּל מָה שֶׁאַתֶּם אוֹהֲבִים מְאֹד מְאֹד.

זֶה בִּטּוּי, לֹא פֵּרוּשׁ מִלּוּלִי: הַכַּוָּנָה שֶׁהָאַהֲבָה לַה׳ הִיא הַחֲזָקָה מִכֻּלָּן.

אַהֲבַת ה׳ גְּדוֹלָה אֲפִלּוּ מִמָּה שֶׁאֲנִי אוֹהֵב ״מְאֹד מְאֹד״.

4

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי לוֹמֵד תּוֹרָה וְחוֹשֵׁב, אֲנִי מַרְגִּישׁ קָרוֹב יוֹתֵר לְה׳ 💛

📖 פָּרָשַׁת שְׁמַע – דְּבָרִים ו׳, פֶּרֶק ו׳, פָּסוּק ו׳ (רַשִׁ״י)

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

🛤️ ״וְהָיוּ הַדְּבָרִים״ – הַדֶּרֶךְ לָאַהֲבָה

רַשִׁ״י שׁוֹאֵל: ״וְאָהַבְתָּ״ – וְאֵיךְ מַגִּיעִים לָאַהֲבָה? תְּשׁוּבָתוֹ: ״וְהָיוּ הַדְּבָרִים הָאֵלֶּה״, שֶׁמִּתּוֹךְ כָּךְ אַתָּה מַכִּיר אֶת ה׳ וּמִדְבַּק בִּדְרָכָיו.

הַדְּבָרִים – הַלִּמּוּד, הַדִּבּוּר וְהַמִּצְווֹת – הֵם גַּם תּוֹצָאָה שֶׁל הָאַהֲבָה וְגַם הַדֶּרֶךְ אֵלֶיהָ.

בְּסֵפֶר שַׁעֲרֵי אַהֲבָה מוּסְבָּר שֶׁכָּל הַתּוֹרָה כֻּלָּהּ מְבִיאָה לְאַהֲבַת ה׳. אֲבָל לֹא כָּל לִמּוּד מַגִּיעַ לָאַהֲבָה מֵעַצְמוֹ: צָרִיךְ לִלְמֹד בְּעִיּוּן וּבְמַחְשָׁבָה.

לוֹמְדִים בְּעִיּוּן – וּמַגִּיעִים לְאַהֲבָה.

5

💛 אַהֲבַת הַשֵּׁם: בַּבֹּקֶר אֲנִי מִסְתַּכֵּל בָּאוֹר וְאוֹמֵר: כַּמָּה יָפֶה הָעוֹלָם שֶׁה׳ בָּרָא 💛

📖 פָּרָשַׁת שְׁמַע – דְּבָרִים ו׳, פֶּרֶק ו׳, פָּסוּק ה׳ (בִּרְכוֹת קְרִיאַת שְׁמַע)

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

🌅 בִּרְכוֹת שְׁמַע – שְׁנֵי שְׁעָרִים לָאַהֲבָה

לִפְנֵי שֶׁקּוֹרְאִים קְרִיאַת שְׁמַע, מְבָרְכִים בִּרְכוֹת שֶׁעוֹנוֹת עַל הַשְּׁאֵלָה ״אֵיךְ מַגִּיעִים לְאַהֲבָה?״:

קֹדֶם ״יוֹצֵר אוֹר״ – מִתְבּוֹנְנִים בַּבְּרִיאָה, בָּאוֹר, בַּחֹשֶׁךְ וּבְכָל הַנִּפְלָאוֹת.

אַחַר כָּךְ ״אַהֲבָה רַבָּה״ – הַתּוֹרָה, ״וְתֵן בְּלִבֵּנוּ בִּינָה״.

הַבְּרִיאָה בָּאָה קֹדֶם, כִּי קֹדֶם מִתְרַגְּלִים לְהִתְפַּעֵל מִדְּבָרִים פְּשׁוּטִים: לָצֵאת הַחוּצָה, לִרְאוֹת צֶמַח יָפֶה, שָׁמַיִם, אֲנָשִׁים. מִשָּׁם מַמְשִׁיכִים לַתּוֹרָה.

הַבְּרִיאָה קֹדֶם, הַתּוֹרָה אַחַר כָּךְ – שְׁנֵיהֶם מְבִיאִים לְאַהֲבָה.

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

מָה לוֹקְחִים מִרָמָה ב׳?

אַהֲבַת ה׳ אֵינָהּ רֶגֶשׁ שֶׁנּוֹפֵל עָלֵינוּ מִלְּמַעְלָה. אֲנַחְנוּ בּוֹחֲרִים מַחְשָׁבוֹת, מִתְפַּעֲלִים מִן הַבְּרִיאָה וְלוֹמְדִים בְּעִיּוּן, וּלְאַט לְאַט הַלֵּב מִתְמַלֵּא. כָּל מַחְשָׁבָה טוֹבָה הִיא צַעַד קָטָן בַּדֶּרֶךְ אֶל ה׳ שֶׁאוֹהֵב אוֹתָנוּ.

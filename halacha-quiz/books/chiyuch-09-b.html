אֵיךְ חִיּוּךְ וְ׳שָׁלוֹם׳ מְקַדְּשִׁים שֵׁם שָׁמַיִם? בַּדֶּרֶךְ, בְּבֵית הַמִּדְרָשׁ וּבַכִּתָּה

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת מֻשְׁלָם בַּמִּדּוֹת, לֹא רַק בַּלִּמּוּד 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַמְבַּ״ם, יְסוֹדֵי הַתּוֹרָה ה׳ י״א

רָמָה ב׳ – קִדּוּשׁ ה׳ בַּפָּנִים

📜 הָרַמְבָּ״ם: שְׁנֵי תַּלְמִידֵי חֲכָמִים

הָרַמְבָּ״ם בְּהִלְכוֹת יְסוֹדֵי הַתּוֹרָה מְתָאֵר שְׁנֵי סוּגִים שֶׁל לוֹמְדֵי תּוֹרָה.

הָרִאשׁוֹן: מִי שֶׁדִּבּוּרוֹ עִם הַבְּרִיּוֹת אֵינוֹ בְּנַחַת, אֵינוֹ מְקַבֵּל אוֹתָם בְּסֵבֶר פָּנִים יָפוֹת וְהוּא בַּעַל כַּעַס. זֶה חִלּוּל ה׳.

הַשֵּׁנִי: מִי שֶׁמְּדַבֵּר בְּנַחַת, מְקַבֵּל בְּסֵבֶר פָּנִים יָפוֹת, וְאִם נֶעֱלָב אֵינוֹ עוֹלֵב בַּחֲזָרָה. עַד שֶׁכֻּלָּם מְשַׁבְּחִים אוֹתוֹ וְאוֹהֲבִים אוֹתוֹ. זֶה קִדּוּשׁ ה׳.

הָרַמְבָּ״ם: סֵבֶר פָּנִים יָפוֹת הוּא קִדּוּשׁ ה׳, וְכַעַס וְדִבּוּר לֹא נָעִים הֵם חִלּוּל ה׳.

1

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי נָעִים, אֲנִי גּוֹרֵם לַאֲנָשִׁים לוֹמַר ׳אַשְׁרֵי אָבִיו׳ 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, יוֹמָא פ״ו ע״א

רָמָה ב׳ – קִדּוּשׁ ה׳ בַּפָּנִים

🏛️ הַגְּמָרָא: מָה הַבְּרִיּוֹת אוֹמְרוֹת

בַּגְּמָרָא (יוֹמָא פ״ו) לוֹמְדִים עַל מִי שֶׁקּוֹרֵא וְשׁוֹנֶה וּמְשַׁמֵּשׁ תַּלְמִידֵי חֲכָמִים, וּ׳מַשָּׂאוֹ וּמַתָּנוֹ בְּנַחַת עִם הַבְּרִיּוֹת׳.

מָה הַבְּרִיּוֹת אוֹמְרוֹת עָלָיו? ׳אַשְׁרֵי אָבִיו שֶׁלִּמְּדוֹ תּוֹרָה, רְאוּ כַּמָּה נָאִים דְּרָכָיו וְכַמָּה מְתֻקָּנִים מַעֲשָׂיו׳.

וּמִי שֶׁלּוֹמֵד תּוֹרָה וְאֵינוֹ מִתְנַהֵג יָפֶה, גּוֹרֵם לַהֶפֶךְ.

הָאֲנָשִׁים מְשַׁבְּחִים אֶת הַתּוֹרָה בִּזְכוּת הַמִּדּוֹת הַיָּפוֹת שֶׁל הַלּוֹמֵד.

2

💛 אַהֲבַת הַשֵּׁם: אֲנִי יוֹדֵעַ שֶׁאֲנָשִׁים מַסְתַּכְּלִים עָלַי, וַאֲנִי רוֹצֶה שֶׁיִּרְאוּ דְּבָרִים יָפִים 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשְׁנָה יוֹמָא ח׳ ט׳

רָמָה ב׳ – קִדּוּשׁ ה׳ בַּפָּנִים

⚖️ חִלּוּל ה׳ וְקִדּוּשׁ ה׳ בַּמִּשְׁנָה

בְּסוֹף מַסֶּכֶת יוֹמָא לוֹמְדִים שֶׁגַּם עֲבֵרָה קַלָּה, אִם אֲחֵרִים לוֹמְדִים מִמֶּנָּה לְזַלְזֵל, נַעֲשֵׂית חִלּוּל ה׳ חָמוּר.

וּלְהֶפֶךְ: קִדּוּשׁ ה׳ הוֹפֵךְ מִצְוָה קְטַנָּה לְמִצְוָה גְּדוֹלָה מְאוֹד.

לָכֵן אֲמִירַת שָׁלוֹם בְּחִיּוּךְ הִיא פְּעָמִים רַבּוֹת קִדּוּשׁ שֵׁם שָׁמַיִם: ׳רְאוּ אֵלּוּ שֶׁלָּמְדוּ תּוֹרָה, כַּמָּה יָפֶה הֵם מִתְנַהֲגִים׳.

מַעֲשֶׂה קָטָן שֶׁאֲחֵרִים רוֹאִים יָכוֹל לְקַדֵּשׁ אוֹ לְחַלֵּל אֶת הַשֵּׁם.

3

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְהִתְעַנְיֵן בֶּאֱמֶת בָּאֲנָשִׁים שֶׁאֲנִי אוֹמֵר לָהֶם שָׁלוֹם 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סֵפֶר שָׁלוֹם רַב

רָמָה ב׳ – קִדּוּשׁ ה׳ בַּפָּנִים

🕊️ שָׁלוֹם כָּרָאוּי

בְּסֵפֶר שָׁלוֹם רַב מְסֻפָּר עַל הָרַב בַּמְבֶּרְגֶּר, שֶׁאָמַר שָׁלוֹם לִיהוּדִי שֶׁלֹּא הָיָה קָרוֹב לַתּוֹרָה, הִתְעַנְיֵן בּוֹ וְלֹא הִמְשִׁיךְ הָלְאָה.

מֵאוֹתוֹ יְהוּדִי יָצְאוּ אַחֲרֵי כְּמֵאָה וַחֲמִשִּׁים שָׁנָה יוֹתֵר מֵחֲמֵשֶׁת אֲלָפִים נְשָׁמוֹת שׁוֹמְרוֹת תּוֹרָה וּמִצְווֹת.

שָׁלוֹם שֶׁנֶּאֱמַר כָּרָאוּי, בְּחִיּוּךְ וּבְעִנְיָן אֲמִתִּי, פּוֹתֵחַ לֵב.

׳שָׁלוֹם׳ בְּחִיּוּךְ וּבְהִתְעַנְיְנוּת אֲמִתִּית יָכוֹל לְשַׁנּוֹת דּוֹרוֹת שְׁלֵמִים.

4

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִלְמֹד וְלִזְכֹּר, וְלָכֵן אֲנִי נָעִים לַכֹּל 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, עֵירוּבִין נ״ד ע״א · רַשִׁ״י

רָמָה ב׳ – קִדּוּשׁ ה׳ בַּפָּנִים

📿 הָעֲנָק רָפוּי וְהַלִּמּוּד מִתְקַיֵּם

בַּגְּמָרָא (עֵירוּבִין נ״ד) נֶאֱמַר: ׳וַעֲנָקִים לְגַרְגְּרוֹתֶיךָ׳. אִם אָדָם מֵשִׂים עַצְמוֹ כַּעֲנָק רָפוּי עַל הַצַּוָּאר, תַּלְמוּדוֹ מִתְקַיֵּם בְּיָדוֹ. אִם לָאו, אֵין תַּלְמוּדוֹ מִתְקַיֵּם.

רַשִׁ״י מְפָרֵשׁ שֶׁהָעֲנָק רָפוּי וְאֵינוֹ לוֹחֵץ. כָּךְ צָרִיךְ אָדָם לִהְיוֹת נָעִים וּמְרֻצֶּה לַבְּרִיּוֹת.

זוֹ עֵצָה שֶׁנּוֹתְנִים חֲזַ״ל כְּדֵי שֶׁהַלִּמּוּד יִישָּׁאֵר וְלֹא יִשָּׁכַח.

לִהְיוֹת נָעִים וּמְרֻצֶּה לַבְּרִיּוֹת הוּא סוֹד לְלִמּוּד שֶׁנִּשְׁאָר.

5

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת כְּמוֹ יוֹסֵף: שָׂם לֵב גַּם כְּשֶׁקָּשֶׁה לִי 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּרֵאשִׁית מ׳ ו׳–ז׳

רָמָה ב׳ – קִדּוּשׁ ה׳ בַּפָּנִים

🤝 יוֹסֵף וְשִׂימַת הַלֵּב

׳וַיָּבֹא אֲלֵיהֶם יוֹסֵף בַּבֹּקֶר וַיַּרְא אֹתָם וְהִנָּם זֹעֲפִים׳. יוֹסֵף עַצְמוֹ הָיָה אָסוּר וְהָיוּ לוֹ צָרוֹת, וּמִטֶּבַע הָעוֹלָם אָדָם בְּצָרָה מִתְכַּנֵּס בְּתוֹךְ עַצְמוֹ.

בְּכָל זֹאת הוּא הִבְחִין שֶׁהַיּוֹם פְּנֵיהֶם רָעִים יוֹתֵר מֵאֶתְמוֹל. הֵם לֹא אָמְרוּ דָּבָר, וְהוּא שָׂם לֵב וְשָׁאַל.

מֵהַשְּׁאֵלָה הַזּוֹ נִפְתַּח פֶּתַח לִישׁוּעָתוֹ.

חִיּוּךְ אֲמִתִּי מַתְחִיל בְּשִׂימַת לֵב לַאֲנָשִׁים שֶׁסְּבִיבֵנוּ.

מָה לוֹקְחִים מִרָמָה ב׳?

חֲזַ״ל וְהָרַמְבָּ״ם מְלַמְּדִים שֶׁפָּנִים נְעִימוֹת הֵן קִדּוּשׁ ה׳. מִי שֶׁמְּקַדֵּשׁ אֶת הַשֵּׁם בְּחִיּוּךְ מַרְגִּישׁ שֶׁה׳ אוֹהֵב אוֹתוֹ וְנִמְצָא אִתּוֹ.

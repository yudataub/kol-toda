עַד ׳עַל לְבָבֶךָ׳ – הַלֵּב צָרִיךְ לְהַסְכִּים לְכָל מִלָּה.

לָכֵן כָּתוּב ׳וְשִׁנַּנְתָּם לְבָנֶיךָ וְדִבַּרְתָּ בָּם׳: כְּמוֹ אַבָּא שֶׁמְּלַמֵּד אֶת בְּנוֹ דְּבָרִים שֶׁעֲרֵבִים עָלָיו, שֶׁהֵם רְגִילִים בְּפִיו וְהוּא מְדַבֵּר בָּהֶם תָּמִיד. אַף אֶחָד לֹא מְדַבֵּר כָּל הַזְּמַן עַל דָּבָר שֶׁנִּמְאַס לוֹ.

מְקַבְּלִים אֶת הַמִּצְווֹת לֹא כְּמִי שֶׁדּוֹאֵג וּמִצְטַעֵר, אֶלָּא כְּאוֹהֵב שֶׁשָּׂמֵחַ לַעֲשׂוֹת אֶת רְצוֹן מִי שֶׁהוּא אוֹהֵב.

מִצְוָה שֶׁעוֹשִׂים בְּשִׂמְחָה וּבְאַהֲבָה – הִיא עֲבוֹדָה שְׁלֵמָה.

💎 טוֹטָפוֹת – ׳שִׁוִּיתִי ה׳ לְנֶגְדִּי תָמִיד׳

׳וְהָיוּ לְטֹטָפֹת בֵּין עֵינֶיךָ׳: הָרַשְׁבָּ״א מְבָאֵר שֶׁטּוֹטָפוֹת הֵן תַּכְשִׁיט (כְּמוֹ בַּמִּשְׁנָה בְּמַסֶּכֶת שַׁבָּת), וְשָׂמִים אוֹתָן בֵּין הָעֵינַיִם כְּדֵי שֶׁהַדָּבָר יִהְיֶה תָּמִיד מוּל הָעֵינַיִם, וְלֹא יַעֲבֹר מֵהַזִּכָּרוֹן אֲפִלּוּ רֶגַע. לְשֵׁם כָּךְ גַּם כּוֹתְבִים אוֹתָם עַל הַשְּׁעָרִים, וְכָךְ אֲנַחְנוּ מִתְפָּאֲרִים וּמִתְעַנְּגִים בָּהֶם.

מִדְּבָרָיו לוֹמְדִים חִדּוּשׁ גָּדוֹל: ׳שִׁוִּיתִי ה׳ לְנֶגְדִּי תָמִיד׳, שֶׁהָרַמָּ״א מֵבִיא בְּרֹאשׁ הַשֻּׁלְחָן עָרוּךְ, הִיא מִצְוָה מִן הַתּוֹרָה, הַכְּתוּבָה בַּמִּלִּים ׳וְהָיוּ לְטֹטָפֹת בֵּין עֵינֶיךָ׳. וְהַגְּרָ״א אוֹמֵר שֶׁזּוֹ כָּל מַעֲלַת הַצַּדִּיקִים.

לִזְכֹּר אֶת ה׳ תָּמִיד – זֶה הַתַּכְשִׁיט שֶׁבֵּין הָעֵינַיִם.

הָרַמְבָּ״ם: בְּכָל כְּנִיסָה וִיצִיאָה פּוֹגְשִׁים בְּיִחוּד ה׳ וְזוֹכְרִים אֶת אַהֲבָתוֹ, מִתְעוֹרְרִים מִן הַשֵּׁנָה וּמִן הַטָּעֻיּוֹת שֶׁל ׳הַבְלֵי הַזְּמַן׳, וְיוֹדְעִים שֶׁאֵין דָּבָר עוֹמֵד לְעוֹלָם אֶלָּא הַיְּדִיעָה שֶׁל צוּר הָעוֹלָם – וּמִיָּד חוֹזְרִים לַדֶּרֶךְ הַיְּשָׁרָה.

שָׁלוֹשׁ מַחְשָׁבוֹת כְּשֶׁרוֹאִים מְזוּזָה: מָה כָּתוּב בָּהּ, לְשֵׁם מָה בָּאתִי לָעוֹלָם, וּמָה תַּכְלִית כָּל הַמִּצְווֹת – ׳כָּל מִצְוֹתֶיךָ אֱמוּנָה׳ (תְּהִלִּים קי״ט).

וְהַתְּפִלִּין: הֶחָפֵץ חַיִּים מְלַמֵּד שֶׁהַקּוֹרֵא שְׁמַע בְּלִי תְּפִלִּין כְּאִלּוּ מֵעִיד עֵדוּת שֶׁקֶר. גַּם אִם יַנִּיחַ אוֹתָן אַחַר כָּךְ, צָרִיךְ שֶׁיִּהְיוּ עָלָיו כְּשֶׁהוּא אוֹמֵר ׳וּקְשַׁרְתָּם׳.

מְזוּזָה וּתְפִלִּין הֵם כֵּלִים – כְּדֵי שֶׁהַדְּבָרִים יִכָּנְסוּ לַלֵּב.

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

🛡️ כֹּחַ הַהֲכָנָה

הַחַיֵּי אָדָם: ׳שׁוֹמֵר מִצְוָה לֹא יֵדַע דָּבָר רָע׳ – שׁוֹמְרִים וּמְחַכִּים לַמִּצְוָה, מִתְיַשְּׁבִים אֵיךְ לַעֲשׂוֹת אוֹתָהּ, וְלֹא עוֹשִׂים אוֹתָהּ בְּפִתְאֹם. וְהוּא בָּדַק בְּעַצְמוֹ: מִצְוָה שֶׁנַּעֲשֵׂית בְּפִתְאֹם לֹא מִתְקַיֶּמֶת כָּרָאוּי.

נֶפֶשׁ הַחַיִּים: מֵהָרֶגַע שֶׁעוֹלָה הַמַּחְשָׁבָה לְקַיֵּם מִצְוָה, קְדֻשָּׁה עֶלְיוֹנָה חוֹפֶפֶת עַל הָאָדָם וּמַקִּיפָה אוֹתוֹ. ׳כָּל הַמְקַדֵּשׁ עַצְמוֹ מִלְּמַטָּה – מְקַדְּשִׁין אוֹתוֹ מִלְמַעְלָה׳.

רַבִּי שְׁלֹמֹה קְלוּגֶר: לֹא כָּל אָדָם מַצְלִיחַ לְקַבֵּל דָּבָר קָשֶׁה בְּשִׂמְחָה. הָעֵצָה: כְּשֶׁטּוֹב, מְקַבְּלִים מֵרֹאשׁ שֶׁאִם יָבוֹא קֹשִׁי – הוּא לְטוֹבָה. וְאָז, גַּם אִם בָּרֶגַע עַצְמוֹ קָשֶׁה לִשְׂמֹחַ, נֶחְשָׁב כְּאִלּוּ קִבֵּל אוֹתוֹ בְּשִׂמְחָה.

מִי שֶׁמְּקַבֵּל מֵרֹאשׁ – כְּאִלּוּ כְּבָר קִיֵּם.

⚖️ רְשׁוּת וְחוֹבָה – ׳בְּנִי יַקִּירִי, עַכְשָׁו׳

סֵפֶר הַחִנּוּךְ: יִרְאַת ה׳ הִיא מִצְוָה תְּמִידִית. אֲבָל כְּשֶׁבָּא לְיַד אָדָם דְּבַר עֲבֵרָה, חוֹבָה עָלָיו בְּאוֹתוֹ רֶגַע לְהִזָּכֵר שֶׁה׳ מַשְׁגִּיחַ. מִי שֶׁלֹּא שָׂם לֵב אָז – בִּטֵּל מִצְוַת עֲשֵׂה.

כָּךְ גַּם הָאֱמוּנָה וְאַהֲבַת ה׳: רֹב הַיּוֹם יֵשׁ רְשׁוּת, אֲבָל בִּשְׁעַת נִסָּיוֹן – חוֹבָה לְעוֹרֵר אֶת הַלֵּב.

כְּשֶׁהַתָּכְנִיּוֹת מִשְׁתַּנּוֹת, ה׳ כְּאִלּוּ קוֹרֵא: ׳בְּנִי יַקִּירִי, רֹב הַיּוֹם אַתָּה בּוֹחֵר אִם לְהִזָּכֵר בִּי, אֲבָל עַכְשָׁו אֲנִי מְבַקֵּשׁ שֶׁתִּזְכֹּר שֶׁאֲנִי קַיָּם, וְשֶׁאֲנִי אוֹהֵב אוֹתְךָ׳.

שְׁעַת נִסָּיוֹן הִיא שְׁעַת הַחוֹבָה – וְגַם שְׁעַת הַזְּכִיָּה.

מָה לוֹקְחִים מִרָמָה ב׳?

ה׳ לֹא רוֹצֶה שֶׁנְּקַיֵּם אֶת שְׁמַע כְּמוֹ חוֹבָה כְּבֵדָה, אֶלָּא כְּמוֹ בֵּן שֶׁשָּׂמֵחַ לַעֲשׂוֹת נַחַת לְאָבִיו. הַתְּפִלִּין, הַמְּזוּזָה וְהַהֲכָנָה מֵרֹאשׁ הֵם הַדְּרָכִים שֶׁלּוֹ לִשְׁמֹר עַל הַקֶּשֶׁר אִתָּנוּ – וּבְכָל רֶגַע קָשֶׁה הוּא קוֹרֵא לָנוּ בְּאַהֲבָה: ׳בְּנִי יַקִּירִי׳.

וְהָיוּ הַדְּבָרִים הָאֵלֶּה אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם עַל לְבָבֶךָ

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • אֵיזוֹ מִלָּה מִפָּרָשַׁת שְׁמַע הָיִיתִי רוֹצֶה שֶׁתִּהְיֶה חֲקוּקָה עַל הַלֵּב שֶׁלִּי, כְּמוֹ אוֹתִיּוֹת בְּאֶבֶן?
  • עַל מָה אֲנִי הֲכִי אוֹהֵב לְדַבֵּר – וְאֵיךְ אֶפְשָׁר שֶׁגַּם דִּבְרֵי תּוֹרָה יִהְיוּ מְתוּקִים לִי כָּךְ?
  • מָה עָלוּל לְהִשְׁתַּבֵּשׁ לִי הַשָּׁבוּעַ – וּמָה אֲנִי מְקַבֵּל עָלַי כְּבָר הַלַּיְלָה, לִפְנֵי הַשֵּׁינָה?

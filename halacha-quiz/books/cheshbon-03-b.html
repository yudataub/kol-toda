מְנַהֵל הָעֵסֶק שֶׁלִּי

רָמָה ב׳ – גַּן הַמְּבוּכָה וְהַזְּמַן הַקָּבוּעַ

אֵיךְ עוֹשִׂים חֶשְׁבּוֹן נֶפֶשׁ כְּמוֹ מְנַהֵל עֵסֶק: גַּן הַמְּבוּכָה, שְׁתֵּי שִׁיטוֹת לַחֲשֹׁב עַל הַיּוֹם, וְזְמַן קָבוּעַ בְּכָל יוֹם

📖 חֶשְׁבּוֹן הַנֶּפֶשׁ · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

💛 אַהֲבַת הַשֵּׁם: אֲנִי שׁוֹמֵעַ לָעֵצָה שֶׁל הַחֲכָמִים 💛

📖 חֶשְׁבּוֹן הַנֶּפֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בַּמִּדְבָּר כ״א כז׳ · בָּבָא בַּתְרָא עַ״ח ב׳ · מְסִלַּת יְשָׁרִים פֶּרֶק ג׳

רָמָה ב׳ – גַּן הַמְּבוּכָה וְהַזְּמַן הַקָּבוּעַ

📜 ׳בֹּאוּ חֶשְׁבּוֹן׳

בַּתּוֹרָה, בְּסִפּוּר מַסְעֵי בְּנֵי יִשְׂרָאֵל, כָּתוּב: ׳עַל כֵּן יֹאמְרוּ הַמֹּשְׁלִים בֹּאוּ חֶשְׁבּוֹן׳.

חֲזַ״ל דּוֹרְשִׁים: ׳הַמּוֹשְׁלִים בְּיִצְרָם, בֹּאוּ וְנַחֲשֹׁב חֶשְׁבּוֹנוֹ שֶׁל עוֹלָם׳ – וְהַחֶשְׁבּוֹן הוּא: הֶפְסֵד מִצְוָה כְּנֶגֶד שְׂכָרָהּ, וּשְׂכַר עֲבֵרָה כְּנֶגֶד הֶפְסֵדָהּ.

הָרַמְחַ״ל מְלַמֵּד בְּ׳מְסִלַּת יְשָׁרִים׳ שֶׁזּוֹ הָעֵצָה הָאֲמִתִּית שֶׁל מִי שֶׁכְּבָר יָצָא מִתַּחַת יַד הַיֵּצֶר.

הָעֵצָה שֶׁל הַחֲכָמִים: בֹּאוּ חֶשְׁבּוֹן.

💛 אַהֲבַת הַשֵּׁם: אֲנִי סוֹמֵךְ עַל הַחֲכָמִים כְּשֶׁאֲנִי לֹא רוֹאֶה אֶת הַדֶּרֶךְ 💛

📖 חֶשְׁבּוֹן הַנֶּפֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְסִלַּת יְשָׁרִים פֶּרֶק ג׳

רָמָה ב׳ – גַּן הַמְּבוּכָה וְהַזְּמַן הַקָּבוּעַ

🌳 מַה לּוֹמֵד גַּן הַמְּבוּכָה

הָרַמְחַ״ל מְסַפֵּר עַל גַּן שֶׁבּוֹ שְׁבִילִים רַבִּים שֶׁנִּרְאִים זֵהִים, וְרַק חֲלָקָם מַגִּיעִים לַחֲצֵר שֶׁבָּאֶמְצַע.

הַהוֹלֵךְ בֵּין הַשְּׁבִילִים אֵינוֹ יָכוֹל לָדַעַת מָה הַנָּכוֹן, אֲבָל מִי שֶׁעוֹמֵד בַּחֲצֵר רוֹאֶה אֶת הַכֹּל, וְיָכוֹל לוֹמַר: ׳לְכוּ בַּדֶּרֶךְ הַזּוֹ׳.

וּמִי שֶׁמַּאֲמִין לוֹ מַגִּיעַ. וּמִי שֶׁרוֹצֶה לָלֶכֶת אַחֲרֵי עֵינָיו, עָלוּל לִהְיוֹת אָבוּד.

מַאֲמִינִים לַחֲכָמִים שֶׁכְּבָר הִגִּיעוּ.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְפַקֵּחַ עַל עַצְמִי, וְה׳ עוֹזֵר לִי 💛

📖 חֶשְׁבּוֹן הַנֶּפֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְסִלַּת יְשָׁרִים פֶּרֶק ג׳ · מְסִלַּת יְשָׁרִים פֶּרֶק ב׳

רָמָה ב׳ – גַּן הַמְּבוּכָה וְהַזְּמַן הַקָּבוּעַ

⚖️ כְּמוֹ מְנַהֵל עֵסֶק

הָרַמְחַ״ל מְלַמֵּד שֶׁצָּרִיךְ אָדָם לְהִתְבּוֹנֵן ׳בְּזְמַן קָבוּעַ לוֹ בְּהִתְבּוֹדְדוֹ׳: מָה הַדֶּרֶךְ הָאֲמִתִּית לְפִי הַתּוֹרָה, וְאַחַר כָּךְ לְבְדֹּק אֶת הַמַּעֲשִׂים אִם הֵם עַל הַדֶּרֶךְ.

לָכֵן יוֹשְׁבִים לְבַד, בְּלִי מַפְרִיעִים, וְחוֹשְׁבִים: אֵיךְ אֶפְשָׁר לְשַׁכְלֵל אֶת עֲבוֹדַת הַבּוֹרֵא שֶׁלִּי.

וּבְסוֹף הַפֶּרֶק הַקּוֹדֵם כָּתוּב: אִם הָאָדָם מְפַקֵּחַ עַל עַצְמוֹ, הַקָּדוֹשׁ בָּרוּךְ הוּא עוֹזֵר לוֹ.

מִי שֶׁמְּפַקֵּחַ עַל עַצְמוֹ, ה׳ עוֹזֵר לוֹ.

🔀

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹחֵר שִׁיטָה וְעוֹשֶׂה אוֹתָהּ 💛

📖 חֶשְׁבּוֹן הַנֶּפֶשׁ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב · פִּרְקֵי אָבוֹת ב׳ טז׳

רָמָה ב׳ – גַּן הַמְּבוּכָה וְהַזְּמַן הַקָּבוּעַ

🔀 שְׁתֵּי שִׁיטוֹת

שִׁיטָה א׳: חֶשְׁבּוֹן עַל הַיּוֹם הַקּוֹדֵם. בּוֹדְקִים מָה לֹא הָיָה טוֹב, וְחוֹשְׁבִים עֵצוֹת לְתַקֵּן. לִפְעָמִים מַסְפִּיק לַעֲבֹר עַל קְצָת מְאֹד מֵהַיּוֹם.

שִׁיטָה ב׳: דִּמְיוֹן מוּדְרָךְ – מְתַכְנְנִים וּמְדַמְיְנִים אֵיךְ רוֹצִים שֶׁיִּרְאֶה הַיּוֹם הַבָּא עָלֵינוּ לְטוֹבָה.

וְאִם לֹא מַסְפִּיק זְמַן: ׳לֹא עָלֶיךָ הַמְּלָאכָה לִגְמֹר, וְלֹא אַתָּה בֶן חוֹרִין לְהִבָּטֵל מִמֶּנָּה׳.

שְׁתֵּי שִׁיטוֹת: לִבְדֹּק אֶתְמוֹל, אוֹ לְדַמְיֵן מָחָר.

⏰ קֶבַע שֶׁלֹּא שׁוֹכְחִים

הָרַמְחַ״ל מְלַמֵּד: ׳וְיִקְבַּע עִתִּים וְשָׁעוֹת לָזֶה, שֶׁלֹּא יִהְיֶה מִשְׁקָלוֹ עֲרַאי, אֶלָּא בִּקְבִיעוּת גְּדוֹלָה׳.

אֲנָשִׁים רַבִּים מַסְכִּימִים שֶׁהָעֵצָה טוֹבָה, וְאַחֲרֵי כַּמָּה יָמִים שׁוֹכְחִים אוֹתָהּ, כִּי הַיֵּצֶר הָרָע דּוֹאֵג שֶׁנִּשְׁכַּח.

לָכֵן קוֹבְעִים זְמַן קָבוּעַ. וּמִי שֶׁעוֹשֶׂה כָּךְ מְסַפֵּר עַל רוֹגַע נַפְשִׁי וְעַל הִתְעַלּוּת.

זְמַן קָבוּעַ שׁוֹמֵר שֶׁלֹּא נִשְׁכַּח.

מָה לוֹקְחִים מִרָמָה ב׳?

מִי שֶׁמְּפַקֵּחַ עַל עַצְמוֹ, ה׳ עוֹזֵר לוֹ. כָּל יוֹם שֶׁבּוֹ אֲנִי יוֹשֵׁב וּבוֹדֵק אֶת הַדֶּרֶךְ שֶׁלִּי, אֲנִי מַרְגִּישׁ שֶׁה׳ אִתִּי וּמַדְרִיךְ אוֹתִי.

🤔 חוֹשְׁבִים עַל זֶה

  • אִם הָיִיתִי מְנַהֵל עֵסֶק, מָה הָיִיתִי כּוֹתֵב הַיּוֹם בְּעַמּוּד ׳רֶוַח׳ וּמָה בְּעַמּוּד ׳הֶפְסֵד׳?
  • בְּאֵיזוֹ שָׁעָה הֲכִי נוֹחַ לִי לָשֶׁבֶת רֶגַע שָׁקֵט לְבַד וְלַעֲשׂוֹת חֶשְׁבּוֹן קָצָר עַל הַיּוֹם?
  • אֲצַיֵּר הַיּוֹם ׳חֲנוּת׳ שֶׁלִּי, וּבָהּ מָה הִרְוַחְתִּי הַיּוֹם: דְּבָרִים טוֹבִים שֶׁעָשִׂיתִי וְדָבָר אֶחָד שֶׁאֲנִי רוֹצֶה לְשַׁפֵּר מָחָר.

אֵיךְ הַתּוֹרָה מְשַׁבַּחַת אֶת הָאָרֶץ, לָמָּה כָּלֵב אָמַר ׳מְאֹד מְאֹד׳, וְאֵיךְ אֲנִי כּוֹתֵב חוֹבֶרֶת תַּיָּרוּת

רָמָה ב׳ – שֶׁבַח שֶׁמּוֹלִיךְ לְאַהֲבָה

🌄 תֵּאוּר שֶׁמּוֹלִיךְ לְהוֹדָיָה

הַפְּסוּקִים בְּפָרָשַׁת עֵקֶב בְּנוּיִים בְּסֵדֶר: מַיִם, תְּבוּאָה וּפֵרוֹת, ׳לֹא תֶחְסַר כֹּל בָּהּ׳, אֲבָנִים וְהָרִים עֲשִׁירִים – וְרַק בַּסּוֹף: ׳וְאָכַלְתָּ וְשָׂבָעְתָּ וּבֵרַכְתָּ אֶת ה׳׳.

הַמַּשְׁמָעוּת הַפְּשׁוּטָה שֶׁל הַפְּסוּקִים: הַתּוֹרָה מְחַנֶּכֶת אוֹתָנוּ לֶאֱהֹב אֶת אֶרֶץ יִשְׂרָאֵל, וּמִתּוֹךְ כָּךְ לְהוֹדוֹת לַה׳ עָלֶיהָ.

הַתֵּאוּר לֹא נִגְמָר בַּיֹּפִי; הוּא מוֹבִיל לִבְרָכָה.

הַתֵּאוּר שֶׁל הָאָרֶץ מוֹבִיל לְאַהֲבָה וּלְהוֹדָיָה.

📖 אַהֲבַת אֶרֶץ יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אַהֲבַת אֶרֶץ חֶמְדָּה (מָשָׁל)

רָמָה ב׳ – שֶׁבַח שֶׁמּוֹלִיךְ לְאַהֲבָה

🧺 לֹא כּוֹפִים – מְתָאֲרִים

בַּמָּשָׁל, אִם הַיֶּלֶד יְנַסֶּה לִכְפּוֹת עַל חֲבֵרוֹ לֶאֱהֹב אֶת הַמָּקוֹם, הַחֲבֵרוּת לֹא תַּצְלִיחַ. אִי אֶפְשָׁר לִכְפּוֹת אַהֲבָה.

הַדֶּרֶךְ הִיא לְתָאֵר, לְשַׁבֵּחַ, וְלִתֵּן לֶחָבֵר לַחְשֹׁב וּלְהִתְבּוֹנֵן עַד שֶׁגַּם הוּא עַצְמוֹ יִרְצֶה וְיֶאֱהַב.

כָּךְ גַּם הַתּוֹרָה: אֵין חוֹבֶרֶת תַּיָּרוּת בָּעוֹלָם שֶׁמְּשַׁבַּחַת וּמְהַלֶּלֶת אֶת הָאָרֶץ כְּמוֹ הַתּוֹרָה.

אַהֲבָה לֹא כּוֹפִים; מְתָאֲרִים וּמְשַׁבְּחִים עַד שֶׁרוֹצִים בְּעַצְמֵנוּ.

רָמָה ב׳ – שֶׁבַח שֶׁמּוֹלִיךְ לְאַהֲבָה

✨ טוֹב מְאֹד מְאֹד

כָּלֵב בֶּן יְפֻנֶּה אָמַר: ׳טוֹבָה הָאָרֶץ מְאֹד מְאֹד׳. הַשֶּׁבַח הַכָּפוּל הַזֶּה, ׳מְאֹד מְאֹד׳, אֵינוֹ מוֹפִיעַ בַּתּוֹרָה אֵצֶל שׁוּם עִנְיָן אַחֵר.

אֲפִלּוּ עַל כְּלַל הָעוֹלָם כָּתוּב רַק ׳וְהִנֵּה טוֹב מְאֹד׳, וְלֹא ׳טוֹב מְאֹד מְאֹד׳.

מִכָּאן לוֹמְדִים אֶת יַחַס הַתּוֹרָה לְאֶרֶץ יִשְׂרָאֵל: דָּבָר שֶׁצָּרִיךְ לְחַבֵּב עַד מְאֹד.

יַחַס הַתּוֹרָה לָאָרֶץ: חִבָּה מְיֻחֶדֶת וּכְפוּלָה.

🍎

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי מְבָרֵךְ עַל פְּרִי, אֲנִי נִזְכָּר בְּשִׁבְחָהּ שֶׁל הָאָרֶץ 💛

📖 אַהֲבַת אֶרֶץ יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשְׁנָה בְּרָכוֹת · אַהֲבַת אֶרֶץ חֶמְדָּה

רָמָה ב׳ – שֶׁבַח שֶׁמּוֹלִיךְ לְאַהֲבָה

🍎 שֶׁהִשְׁתַּבְּחָה בָּהֶם הָאָרֶץ

אֵין זֶה רַק בַּפָּסוּק. גַּם בַּהֲלָכָה רוֹאִים שֶׁשִּׁבְעַת הַמִּינִים חֲשׁוּבִים: לְפִי הַמִּשְׁנָה, אָדָם שֶׁיֵּשׁ לְפָנָיו פֵּרוֹת הַרְבֵּה מַקְדִּים וּמְבָרֵךְ עַל הַפְּרִי מִשִּׁבְעַת הַמִּינִים.

הַטַּעַם: אֵלּוּ הַמִּינִים שֶׁהִשְׁתַּבְּחָה בָּהֶם אֶרֶץ יִשְׂרָאֵל.

כָּךְ הַשֶּׁבַח שֶׁל הַתּוֹרָה נוֹגֵעַ גַּם בַּשֻּׁלְחָן שֶׁלָּנוּ.

פְּרִי מִשִּׁבְעַת הַמִּינִים זוֹכֶה לְהַקְדָּמָה בַּבְּרָכָה.

🔍

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹהֵב אֶת הָאָרֶץ כִּי הִיא מַתָּנָה מֵה׳ 💛

📖 אַהֲבַת אֶרֶץ יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אַהֲבַת אֶרֶץ חֶמְדָּה

רָמָה ב׳ – שֶׁבַח שֶׁמּוֹלִיךְ לְאַהֲבָה

🔍 לֹא בִּגְלַל שֶׁהִיא הַכִּי

הַמִּצְוָה לֶאֱהֹב אֶת הָאָרֶץ אֵינָהּ תְּלוּיָה בְּכָךְ שֶׁהִיא הַכִּי גְּדוֹלָה אוֹ הַכִּי יָפָה בָּעוֹלָם.

אֲנַחְנוּ אוֹהֲבִים אוֹתָהּ כִּי הִיא שֶׁל ה׳, וְהוּא נָתַן אוֹתָהּ לָנוּ, לְעַם הַסְּגֻלָּה שֶׁלּוֹ.

הַיֹּפִי שֶׁלָּהּ הוּא יֹפִי עָמֹק, וְהַחִבּוּר שֶׁלָּנוּ אֵלֶיהָ הוּא חִבּוּר שֶׁצָּרִיךְ לִלְמֹד וּלְהַעֲמִיק בּוֹ.

אוֹהֲבִים אֶת הָאָרֶץ כִּי הִיא מַתָּנָה מֵה׳, וְלֹא רַק כִּי הִיא יָפָה.

רָמָה ב׳ – שֶׁבַח שֶׁמּוֹלִיךְ לְאַהֲבָה

מָה לוֹקְחִים מִרָמָה ב׳?

אֶת הָאַהֲבָה לֹא כּוֹפִים; מְתָאֲרִים אֶת הַטּוֹב וְנוֹתְנִים לַלֵּב לְהִתְבּוֹנֵן. אֲנִי אוֹהֵב אֶת הָאָרֶץ כִּי הִיא מַתָּנָה מֵה׳, וְכָךְ הָאַהֲבָה לָאָרֶץ הִיא גַּם אַהֲבַת ה׳.

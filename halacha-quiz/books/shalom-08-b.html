🧩 ׳וַיִּקַּח׳ וְלֹא ׳וַיִּקְחוּ׳

בַּפָּרָשָׁה מְנוּיִים קֹרַח, דָּתָן, אֲבִירָם וְאוֹן – וְעִם זֶה כָּתוּב ׳וַיִּקַּח׳ בִּלְשׁוֹן יָחִיד. כָּל אֶחָד לָקַח לְעַצְמוֹ אִינְטֶרֶס אַחֵר: קֹרַח רָצָה כְּהֻנָּה, דָּתָן וַאֲבִירָם מִשֵּׁבֶט רְאוּבֵן הִתְנַגְּחוּ בְּמֹשֶׁה, וּמָאתַיִם וַחֲמִשִּׁים הַנְּשִׂיאִים רָצוּ כָּל אֶחָד דָּבָר מִשֶּׁלּוֹ.

הַמִּשְׁנָה בְּאָבוֹת קוֹרֵאת לָזֶה ׳מַחֲלֹקֶת קֹרַח וְכָל עֲדָתוֹ׳, כִּי הַשֻּׁתָּפִים עַצְמָם הָיוּ חֲלוּקִים. כְּשֶׁקְּבוּצוֹת שׁוֹנוֹת מִתְקַבְּצוֹת וְהַמְּשֻׁתָּף לָהֶן הוּא רַק הָרָצוֹן לִפְגֹּעַ בַּצַּד הַשֵּׁנִי – זֶה מַאֲבָק עַל כּוֹחַ וְלֹא מַחֲלֹקֶת עַל הָאֱמֶת.

אֲנָשִׁים שֶׁמְּאֻחָדִים רַק ׳נֶגֶד׳ עוֹסְקִים בְּמַאֲבָק עַל כּוֹחַ, לֹא בֶּאֱמֶת.

📜 אַרְבָּעָה פֵּרוּשִׁים לַפָּסוּק

הַפָּסוּק ׳וְלֹא יִהְיֶה כְקֹרַח וְכַעֲדָתוֹ׳ מִתְפָּרֵשׁ בְּכַמָּה דְּרָכִים. רַשִׁ״י: צִפּוּי הַמִּזְבֵּחַ הוּא זִכָּרוֹן, שֶׁלֹּא יִהְיֶה עוֹד אָדָם שֶׁיִּנְהַג כְּקֹרַח. הָרַמְבַּ״ם: מִי שֶׁיַּחֲלֹק עַל הַכְּהֻנָּה לֹא יֵעָנֵשׁ כְּקֹרַח אֶלָּא בְּצָרַעַת. הָרַמְבַּ״ן: זֶה אִסּוּר – שֶׁלֹּא יֹאמַר אָדָם ׳לָמָּה יַעֲבֹד הַכֹּהֵן וַאֲנִי לֹא?׳.

וְהַמִּשְׁנָה בְּרוּרָה, לְפִי הַגְּמָרָא, הַסְּמַ״ג וְרַבֵּנוּ יוֹנָה: זֶה אִסּוּר כְּלָלִי לֹא לְהַחֲזִיק בְּמַחֲלֹקֶת.

נֶחְלְקוּ הָרִאשׁוֹנִים אִם זֶה אִסּוּר מִן הַתּוֹרָה, אֲבָל לְכָל הַדֵּעוֹת יֵשׁ בָּזֶה לְפָחוֹת אִסּוּר מִדִּבְרֵי חֲכָמִים.

לְכָל הַדֵּעוֹת אָסוּר לְהַחֲזִיק בְּמַחֲלֹקֶת.

🌗 סוֹפָהּ לְהִתְקַיֵּם

הַמִּשְׁנָה בְּאָבוֹת: ׳כָּל מַחֲלֹקֶת שֶׁהִיא לְשֵׁם שָׁמַיִם סוֹפָהּ לְהִתְקַיֵּם, וְשֶׁאֵינָהּ לְשֵׁם שָׁמַיִם אֵין סוֹפָהּ לְהִתְקַיֵּם. אֵיזוֹ הִיא מַחֲלֹקֶת שֶׁהִיא לְשֵׁם שָׁמַיִם? זוֹ מַחֲלֹקֶת הִלֵּל וְשַׁמַּאי. וְשֶׁאֵינָהּ לְשֵׁם שָׁמַיִם? זוֹ מַחֲלֹקֶת קֹרַח וְכָל עֲדָתוֹ׳.

הָרַע״ב מְבָאֵר: מַטָּרָתָהּ שֶׁל מַחֲלֹקֶת לְשֵׁם שָׁמַיִם הִיא בֵּרוּר הָאֱמֶת, וְהַמַּטָּרָה הַזֹּאת מִתְקַיֶּמֶת כְּשֶׁנִּפְסְקָה הַהֲלָכָה. אֲבָל קֹרַח חִפֵּשׂ שְׁרָרָה – וְאַף אֶחָד לֹא קִבֵּל אוֹתָהּ.

חוֹלְקִים כְּדֵי לְבָרֵר אֱמֶת, לֹא כְּדֵי לְנַצֵּחַ.

🎶 בְּנֵי קֹרַח וְהַשּׁוֹפָר

לִפְנֵי תְּקִיעַת שׁוֹפָר בְּרֹאשׁ הַשָּׁנָה אוֹמְרִים אֶת הַמִּזְמוֹר ׳לַמְנַצֵּחַ לִבְנֵי קֹרַח מִזְמוֹר׳. בְּנֵי קֹרַח הִרְהֲרוּ בִּתְשׁוּבָה בָּרֶגַע הָאַחֲרוֹן – וְנִשְׁאֲרוּ. מֵהֶם יָצְאוּ מְשׁוֹרְרֵי תְּהִלִּים, וּמִצֶּאֱצָאֵיהֶם שְׁמוּאֵל הַנָּבִיא.

אוֹמְרִים אוֹתוֹ רֶגַע לִפְנֵי הַתְּקִיעוֹת כְּדֵי לְלַמֵּד: גַּם מִי שֶׁעוֹמֵד רֶגַע לִפְנֵי הַסּוֹף, קְצָת תְּשׁוּבָה וְהוּא מִתְקַיֵּם.

אַף פַּעַם לֹא מְאֻחָר מִדַּי – קְצָת תְּשׁוּבָה מַסְפִּיקָה לְהַתְחָלָה.

🕊️ אֲנִי ה׳

הַפָּסוּק ׳וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ׳ נִגְמָר בַּמִּלִּים ׳אֲנִי ה׳׳. אֶפְשָׁר לְהָבִין: הָאַהֲבָה לֶחָבֵר קְשׁוּרָה לִקְרִיבוּתוֹ לַה׳.

קֹרַח לֹא הֵבִין אֶת זֶה. הוּא רָאָה בְּמֹשֶׁה וּבְאַהֲרֹן מִתְחָרִים, וְלֹא אֶת מִי שֶׁה׳ קֵרֵב.

מִי שֶׁמֵּבִין ׳אֲנִי ה׳׳ רוֹאֶה בְּכָל יְהוּדִי אֶת מָה שֶׁה׳ רוֹאֶה בּוֹ.

אוֹהֲבִים אֶת הֶחָבֵר בִּגְלַל מָה שֶׁה׳ שָׂם בּוֹ.

👨‍👧 אַבָּא שֶׁמַּזְהִיר

מִי שֶׁמַּחֲזִיק בְּמַחֲלֹקֶת עוֹבֵר עַל אִסּוּר, וְזֶה לֹא ׳עִנְיָן שֶׁל אֹפִי׳. ה׳ רָצָה כָּל כָּךְ שֶׁנִּחְיֶה בְּשָׁלוֹם, שֶׁעָשָׂה מִן הַמַּחֲלֹקֶת אִסּוּר.

אָב שֶׁמַּזְהִיר אֶת בָּנָיו מִלָּרִיב – מַזְהִיר מֵאַהֲבָה.

הֶחָפֵץ חַיִּים אָמַר לַבַּחוּרִים שֶׁבַּמַּחֲלֹקֶת יֵשׁ סַכָּנָה גְּדוֹלָה, כִּי מְדַבְּרִים בָּהּ הַרְבֵּה לָשׁוֹן הָרָע.

הָאִסּוּר שֶׁל מַחֲלֹקֶת בָּא מֵאַהֲבָה, כְּמוֹ אַזְהָרָה שֶׁל אַבָּא.

מָה לוֹקְחִים מִרָמָה ב׳?

הָאִסּוּר לְהַחֲזִיק בְּמַחֲלֹקֶת בָּא מֵאַהֲבָה: ה׳ רוֹצֶה שֶׁנִּהְיֶה קְרוֹבִים זֶה לָזֶה. מִי שֶׁמְּחַפֵּשׂ אֱמֶת וְשָׂמֵחַ בְּחֶלְקוֹ מִתְקָרֵב אֶל ה׳ וְאוֹהֵב אוֹתוֹ.

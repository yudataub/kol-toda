לֹא מַסְפִּיק לוֹמַר אֶת הַמִּלִּים בַּפֶּה. הָרַשְׁבָּ״א מְבָאֵר: הַדְּבָרִים צְרִיכִים לִהְיוֹת חֲקוּקִים עַל הַלֵּב, כְּמוֹ אוֹתִיּוֹת שֶׁחוֹרְטִים בְּאֶבֶן, שֶׁאֵינָן נִמְחָקוֹת.

כְּשֶׁאֲנִי אוֹמֵר ׳שְׁמַע׳, אֲנִי מַסְכִּים בְּכָל הַלֵּב לְכָל מִלָּה.

קְרִיאַת שְׁמַע לֹא נִשְׁאֶרֶת בַּפֶּה – הִיא נֶחְקֶקֶת בַּלֵּב.

🗣️ וְדִבַּרְתָּ בָּם – מְדַבְּרִים עַל מָה שֶׁאוֹהֲבִים

עַל מָה יֶלֶד אוֹהֵב לְדַבֵּר? עַל הַמִּשְׂחָק שֶׁהוּא הֲכִי אוֹהֵב, עַל הַטִּיּוּל שֶׁהָיָה לוֹ. מְדַבְּרִים הַרְבֵּה עַל מָה שֶׁעָרֵב לָנוּ, וְלֹא עַל דָּבָר שֶׁנִּמְאַס לָנוּ.

לָכֵן הַתּוֹרָה אוֹמֶרֶת ׳וְשִׁנַּנְתָּם לְבָנֶיךָ וְדִבַּרְתָּ בָּם׳: דִּבְרֵי ה׳ צְרִיכִים לִהְיוֹת מְתוּקִים לָנוּ, כָּל כָּךְ שֶׁנְּדַבֵּר עֲלֵיהֶם תָּמִיד וּנְלַמֵּד אוֹתָם בְּשִׂמְחָה.

מִי שֶׁאוֹהֵב אֶת הַתּוֹרָה – מְדַבֵּר עָלֶיהָ בְּשִׂמְחָה.

💎 לְטֹטָפֹת – תַּכְשִׁיט בֵּין הָעֵינַיִם

׳וְהָיוּ לְטֹטָפֹת בֵּין עֵינֶיךָ׳. הָרַשְׁבָּ״א מְבָאֵר: טוֹטָפוֹת הֵן תַּכְשִׁיט. אֶת הַתְּפִלִּין שֶׁל רֹאשׁ מַנִּיחִים בֵּין הָעֵינַיִם, בְּמָקוֹם שֶׁכֻּלָּם רוֹאִים.

כָּךְ גַּם הָאֱמוּנָה: הִיא צְרִיכָה לִהְיוֹת תָּמִיד מוּל הָעֵינַיִם שֶׁלָּנוּ, וְלֹא לָצֵאת מֵהַזִּכָּרוֹן אֲפִלּוּ רֶגַע. וַאֲנַחְנוּ מִתְגָּאִים בָּהּ, כְּמוֹ בְּתַכְשִׁיט יָקָר.

הָאֱמוּנָה הִיא הַתַּכְשִׁיט שֶׁלִּי – תָּמִיד מוּל הָעֵינַיִם.

📦 וּקְשַׁרְתָּם – תְּפִלִּין מוּל הַלֵּב וּמוּל הַמֹּחַ

׳וּקְשַׁרְתָּם לְאוֹת עַל יָדֶךָ׳ – אֵלּוּ הַתְּפִלִּין. הַתְּפִלִּין שֶׁל יָד קְשׁוּרוֹת עַל הַזְּרוֹעַ מוּל הַלֵּב, וְהַתְּפִלִּין שֶׁל רֹאשׁ מוּנָחוֹת עַל הָרֹאשׁ, מוּל הַמֹּחַ. כָּךְ הַלֵּב וְהַמַּחְשָׁבָה שֶׁלָּנוּ קְשׁוּרִים לַה׳.

הֶחָפֵץ חַיִּים מְלַמֵּד: מִי שֶׁקּוֹרֵא קְרִיאַת שְׁמַע בְּלִי תְּפִלִּין, כְּאִלּוּ מֵעִיד עֵדוּת שֶׁקֶר – הוּא אוֹמֵר ׳וּקְשַׁרְתָּם׳, וְהַתְּפִלִּין לֹא עָלָיו! לָכֵן הַגְּדוֹלִים מַנִּיחִים תְּפִלִּין לִפְנֵי קְרִיאַת שְׁמַע שֶׁל שַׁחֲרִית.

תְּפִלִּין מוּל הַלֵּב וּמוּל הַמֹּחַ – הַכֹּל קָשׁוּר לַה׳.

🚪 וּכְתַבְתָּם עַל מְזוּזֹת – בְּכָל כְּנִיסָה וִיצִיאָה

בַּמְּזוּזָה כְּתוּבוֹת שְׁתֵּי פָּרָשִׁיּוֹת: ׳שְׁמַע׳ וְ׳וְהָיָה אִם שָׁמֹעַ׳.

הָרַמְבָּ״ם מְלַמֵּד: בְּכָל פַּעַם שֶׁנִּכְנָסִים וְיוֹצְאִים, פּוֹגְשִׁים אֶת יִחוּד ה׳ וְזוֹכְרִים אֶת אַהֲבָתוֹ. הַמְּזוּזָה מְעִירָה אוֹתָנוּ, כְּמוֹ שָׁעוֹן מְעוֹרֵר, מִדְּבָרִים שֶׁל מָה בְּכָךְ, וּמַזְכִּירָה שֶׁרַק הַקֶּשֶׁר עִם ה׳ נִשְׁאָר לְעוֹלָם.

הַמִּצְוָה הַזֹּאת שַׁיֶּכֶת לְבָנִים וְגַם לְבָנוֹת.

הַמְּזוּזָה לֹא נוֹעֲדָה לְקִשּׁוּט – הִיא בָּאָה לְהַזְכִּיר לַלֵּב.

🌅 וּבְשָׁכְבְּךָ וּבְקוּמֶךָ – פַּעֲמַיִם בְּכָל יוֹם

׳בְּשִׁבְתְּךָ בְּבֵיתֶךָ וּבְלֶכְתְּךָ בַדֶּרֶךְ וּבְשָׁכְבְּךָ וּבְקוּמֶךָ׳. רַשִׁ״י מְבָאֵר: זְמַן שְׁכִיבָה וּזְמַן קִימָה.

לָכֵן קוֹרְאִים קְרִיאַת שְׁמַע פַּעֲמַיִם בְּכָל יוֹם: בַּבֹּקֶר, בִּזְמַן שֶׁאֲנָשִׁים קָמִים, וּבָעֶרֶב, בִּזְמַן שֶׁהוֹלְכִים לִישֹׁן.

כָּךְ הַיּוֹם שֶׁלָּנוּ מַתְחִיל וְנִגְמָר בִּ׳שְׁמַע יִשְׂרָאֵל׳, וּבֵינְתַיִם – בַּבַּיִת, בַּדֶּרֶךְ וּבְבֵית הַסֵּפֶר – הַדְּבָרִים מְלַוִּים אוֹתָנוּ.

בַּבֹּקֶר וּבָעֶרֶב – הַיּוֹם שֶׁלִּי מַתְחִיל וְנִגְמָר בִּ׳שְׁמַע׳.

חֵלֶק ב׳ – אֵיךְ מְקַיְּמִים: עַל הַלֵּב, מְדַבְּרִים בָּם, תְּפִלִּין, מְזוּזָה, בַּבֹּקֶר וּבָעֶרֶב.

קֹדֶם מַאֲמִינִים וְאוֹהֲבִים – וְאַחַר כָּךְ מְקַיְּמִים בְּכָל יוֹם.

💛 אַהֲבַת הַשֵּׁם: לִפְנֵי שֶׁאֲנִי נִרְדָּם אֲנִי אוֹמֵר לַה׳: מָה שֶׁתַּעֲשֶׂה – אֲנִי יוֹדֵעַ שֶׁזֶּה מֵאַהֲבָה 💛

📖 פָּרָשַׁת שְׁמַע – דְּבָרִים ו׳, פֶּרֶק ו׳, פָּסוּק ז׳ (נֶפֶשׁ הַחַיִּים)

רָמָה א׳ – הַמִּלִּים עַל הַלֵּב

🌙 לִפְנֵי הַשֵּׁינָה – מְקַבְּלִים מֵרֹאשׁ

לִפְנֵי הַשֵּׁינָה קוֹרְאִים קְרִיאַת שְׁמַע שֶׁעַל הַמִּטָּה. זֶה הַזְּמַן לְקַבֵּל מֵרֹאשׁ: ׳אִם אָחִי הַקָּטָן יִבְכֶּה בַּלַּיְלָה וְיָעִיר אוֹתִי – אֲנִי זוֹכֵר שֶׁזֶּה מֵה׳, וְשֶׁה׳ אוֹהֵב אוֹתִי׳.

נֶפֶשׁ הַחַיִּים מְלַמֵּד: מִי שֶׁמִּתְכּוֹנֵן לְמִצְוָה, קְדֻשָּׁה עוֹטֶפֶת אוֹתוֹ כְּבָר מֵהָרֶגַע שֶׁחָשַׁב עָלֶיהָ.

מְקַבְּלִים בָּעֶרֶב – וְקַל יוֹתֵר לְקַיֵּם בַּלַּיְלָה.

💡

רָמָה א׳ – הַמִּלִּים עַל הַלֵּב

מָה לוֹקְחִים מִרָמָה א׳?

ה׳ נָתַן לָנוּ לֹא רַק מִלִּים יָפוֹת, אֶלָּא דֶּרֶךְ לִחְיוֹת אוֹתָן: בַּלֵּב, בַּפֶּה, עַל הַיָּד, בֵּין הָעֵינַיִם וְעַל הַדֶּלֶת. כָּל אֵלֶּה הֵם כְּמוֹ סִימָנִים קְטַנִּים שֶׁל אַבָּא אוֹהֵב, שֶׁפְּזוּרִים בְּכָל מָקוֹם כְּדֵי שֶׁלֹּא נִשְׁכַּח אוֹתוֹ אֲפִלּוּ רֶגַע. וְכָל פַּעַם שֶׁאֲנַחְנוּ נִזְכָּרִים – אֲנַחְנוּ מְקַיְּמִים אֶת שְׁמַע בֶּאֱמֶת.

וְהָיוּ הַדְּבָרִים הָאֵלֶּה אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם עַל לְבָבֶךָ

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • אֵיזוֹ מִלָּה מִפָּרָשַׁת שְׁמַע הָיִיתִי רוֹצֶה שֶׁתִּהְיֶה חֲקוּקָה עַל הַלֵּב שֶׁלִּי, כְּמוֹ אוֹתִיּוֹת בְּאֶבֶן?
  • עַל מָה אֲנִי הֲכִי אוֹהֵב לְדַבֵּר – וְאֵיךְ אֶפְשָׁר שֶׁגַּם דִּבְרֵי תּוֹרָה יִהְיוּ מְתוּקִים לִי כָּךְ?
  • מָה עָלוּל לְהִשְׁתַּבֵּשׁ לִי הַשָּׁבוּעַ – וּמָה אֲנִי מְקַבֵּל עָלַי כְּבָר הַלַּיְלָה, לִפְנֵי הַשֵּׁינָה?

רָמָה א׳ – שִׁירָה בַּלַּיְלָה

😊 מִי שָׁר? מִי שֶׁשָּׂמֵחַ

יוֹסִי קִבֵּל צִיּוּן מְצֻיָּן, וּבַדֶּרֶךְ הַבַּיְתָה הוּא הִתְחִיל לְזַמְזֵם. אִישׁ לֹא בִּקֵּשׁ מִמֶּנּוּ.

כָּךְ זֶה אֶצְלֵנוּ: לֵב שֶׁמָּלֵא שִׂמְחָה פּוֹרֵץ בְּשִׁיר. חֲזַ״ל לִמְּדוּ שֶׁשִּׁירָה הִיא עֲבוֹדָה שֶׁבְּשִׂמְחָה וּבְטוּב לֵבָב, וְלָכֵן אֵין אָדָם שָׁר אֶלָּא מִתּוֹךְ שִׂמְחָה.

הַיּוֹם מַתְחִילִים סִדְרָה עַל עֶשֶׂר הַשִּׁירוֹת שֶׁל עַם יִשְׂרָאֵל.

שִׁיר אֲמִתִּי יוֹצֵא מִשִּׂמְחָה.

רָמָה א׳ – שִׁירָה בַּלַּיְלָה

🗺️ הַשִּׁירָה ה-1 מִתּוֹךְ עֶשֶׂר

הַמְּכִילְתָּא מְלַמֶּדֶת: ׳עֶשֶׂר שִׁירוֹת הֵן׳.

זוֹ הַשִּׁירָה ה-1 מִתּוֹךְ עֶשֶׂר, וְהִיא נֶאֶמְרָה בְּמִצְרַיִם. אַחֲרֶיהָ יָבוֹאוּ שִׁירַת הַיָּם, הַבְּאֵר, מֹשֶׁה, יְהוֹשֻׁעַ, דְּבוֹרָה וּבָרָק, דָּוִד, שְׁלֹמֹה וִיהוֹשָׁפָט, וְהָעֲשִׂירִית תֵּאָמֵר לֶעָתִיד לָבוֹא.

הַשִּׁירָה שֶׁלָּנוּ הַיּוֹם פּוֹתַחַת אֶת הָרְשִׁימָה, וְהִיא הַתַּחֲנָה הָרִאשׁוֹנָה בַּמַּפָּה.

הַשִּׁירָה הָרִאשׁוֹנָה מִתּוֹךְ עֶשֶׂר נֶאֶמְרָה בְּמִצְרַיִם.

🔥

📖 עֶשֶׂר הַשִּׁירוֹת, שְׁמוֹת רַבָּה כ״ג ד׳

רָמָה א׳ – שִׁירָה בַּלַּיְלָה

🔥 הַנִּסִּים שֶׁעוֹד לֹא הָיָה בָּהֶם שִׁיר

אַבְרָהָם הֻצַּל מִכִּבְשַׁן הָאֵשׁ, וְיִצְחָק הֻצַּל מִן הַמַּאֲכֶלֶת. נִסִּים גְּדוֹלִים!

הַמִּדְרָשׁ אוֹמֵר: מִיּוֹם שֶׁנִּבְרָא הָעוֹלָם עַד שֶׁעָמְדוּ יִשְׂרָאֵל עַל הַיָּם לֹא מָצִינוּ אָדָם שֶׁאָמַר שִׁירָה לַה׳, אֶלָּא יִשְׂרָאֵל.

הַמְּפָרְשִׁים מַסְבִּירִים: הָיוּ שִׁירוֹת גַּם קֹדֶם, אֲבָל שִׁירָה שְׁלֵמָה שֶׁנִּכְתְּבָה בַּתּוֹרָה אָמְרוּ רַק בְּנֵי יִשְׂרָאֵל.

שִׁירָה שְׁלֵמָה שֶׁנִּכְתְּבָה בַּתּוֹרָה אָמְרוּ בְּנֵי יִשְׂרָאֵל.

🌙

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת כְּמוֹ בְּנֵי יִשְׂרָאֵל וְלָשִׁיר לַה׳ גַּם כְּשֶׁעֲדַיִן לֹא הַכֹּל גָּמוּר 💛

📖 עֶשֶׂר הַשִּׁירוֹת, רַשִׁ״י יְשַׁעְיָהוּ ל׳ כ״ט

רָמָה א׳ – שִׁירָה בַּלַּיְלָה

🌙 שִׁיר לִפְנֵי הַיְּצִיאָה

בְּמִצְרַיִם, בְּלֵיל הַפֶּסַח, בְּנֵי יִשְׂרָאֵל עֲדַיִן הָיוּ בַּבָּתִּים, וְעוֹד לֹא יָצְאוּ לַחֵרוּת.

הַנָּבִיא יְשַׁעְיָהוּ אוֹמֵר: ׳הַשִּׁיר יִהְיֶה לָכֶם כְּלֵיל הִתְקַדֶּשׁ חָג׳. רַשִׁ״י מַסְבִּיר: כְּמוֹ שֶׁאֲמַרְתֶּם שִׁיר שֶׁל הַפְּסָחִים בְּמִצְרַיִם, אָמְרוּ הַלֵּל בְּלֵיל אֲכִילַת הַפֶּסַח.

כְּלוֹמַר, הֵם הִתְחִילוּ לָשִׁיר עוֹד לִפְנֵי שֶׁיָּצְאוּ.

הַשִּׁירָה הָרִאשׁוֹנָה נֶאֶמְרָה בַּלַּיְלָה, עוֹד לִפְנֵי הַיְּצִיאָה.

🌃

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁקָּשֶׁה לִי, אֲנִי יָכוֹל לְחַפֵּשׂ דָּבָר אֶחָד לָשִׁיר עָלָיו, וְה׳ קָרוֹב אֵלַי 💛

📖 עֶשֶׂר הַשִּׁירוֹת, חֲגִיגָה י״ב ע״ב, תְּהִלִּים מ״ב ט׳

רָמָה א׳ – שִׁירָה בַּלַּיְלָה

🌃 לָמָּה חַג הַחֵרוּת בַּלַּיְלָה?

אֶת חַג הַחֵרוּת חוֹגְגִים דַּוְקָא בַּלַּיְלָה, אַף שֶׁבְּנֵי יִשְׂרָאֵל יָצְאוּ בְּפֹעַל לַחֵרוּת בַּיּוֹם.

בַּפָּסוּק כָּתוּב: ׳יוֹמָם יְצַוֶּה ה׳ חַסְדּוֹ וּבַלַּיְלָה שִׁירֹה עִמִּי׳. חֲזַ״ל לָמְדוּ מִן הַפָּסוּק: מִי שֶׁעוֹסֵק בַּתּוֹרָה בַּלַּיְלָה – ה׳ מוֹשֵׁךְ עָלָיו חוּט שֶׁל חֶסֶד בַּיּוֹם.

וּבַשִּׁעוּר מוֹסִיפִים: הַלַּיְלָה רוֹמֵז גַּם לִזְמַן שֶׁל חֹשֶׁךְ, וּמִי שֶׁשָּׁר לַה׳ גַּם בַּחֹשֶׁךְ זוֹכֶה לְחֶסֶד וּלְגְאֻלָּה.

שִׁירָה בַּחֹשֶׁךְ מְבִיאָה חֶסֶד וּגְאֻלָּה.

📖 עֶשֶׂר הַשִּׁירוֹת, תְּהִלִּים קי״ח ט״ו, הַגְּרָ״א עַל הַהַגָּדָה

רָמָה א׳ – שִׁירָה בַּלַּיְלָה

🎵 קוֹל רִנָּה – וְאַחַר כָּךְ יְשׁוּעָה

דָּנִי מְחַכֶּה לִתְשׁוּבָה עַל בַּקָּשָׁה שֶׁבִּקֵּשׁ. הוּא יָכוֹל לָשֶׁבֶת וּלְהִתְלוֹנֵן, וְהוּא יָכוֹל לוֹמַר: ׳אֲנִי בָּטוּחַ בַּה׳׳.

בַּפָּסוּק כָּתוּב: ׳קוֹל רִנָּה וִישׁוּעָה בְּאָהֳלֵי צַדִּיקִים׳. הַגְּרָ״א מְבָאֵר: הָרִנָּה, שֶׁהִיא שֶׁבַח וְהוֹדָיָה לַה׳, בָּאָה קֹדֶם הַיְשׁוּעָה, כִּי לִבִּי בָּטוּחַ בַּה׳.

הָרִנָּה בָּאָה קֹדֶם הַיְשׁוּעָה.

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי שָׁר אוֹ אוֹמֵר תּוֹדָה, אֲנִי עוֹשֶׂה בְּדִיּוּק מַה שֶּׁה׳ רוֹצֶה מֵהָעוֹלָם 💛

📖 עֶשֶׂר הַשִּׁירוֹת, מִדְרַשׁ אוֹתִיּוֹת דְּרַבִּי עֲקִיבָא

רָמָה א׳ – שִׁירָה בַּלַּיְלָה

🌍 בִּשְׁבִיל שִׁירָה נִבְרָא הָעוֹלָם

בְּמִדְרָשׁ אוֹתִיּוֹת דְּרַבִּי עֲקִיבָא כָּתוּב שֶׁה׳ אוֹמֵר: ׳שֶׁאִלְמָלֵא שִׁירָה וְזִמְרָה שֶׁהֵם אוֹמְרִים לְפָנַי בְּכָל יוֹם – לֹא בָּרָאתִי אֶת עוֹלָמִי׳.

אֶפְשָׁר לְהָבִין מִכָּאן שֶׁאַחַת הַמַּטָּרוֹת שֶׁל הַבְּרִיאָה הִיא שֶׁיִּהְיוּ בָּעוֹלָם מִי שֶׁשָּׁרִים וּמוֹדִים לַה׳.

לָכֵן כָּל שִׁירָה שֶׁלָּנוּ חֲשׁוּבָה מְאֹד.

אֶפְשָׁר לְהָבִין: הָעוֹלָם נִבְרָא כְּדֵי שֶׁיָּשִׁירוּ לַה׳.

📖 עֶשֶׂר הַשִּׁירוֹת, מִשִּׁעוּרֵי עֶשֶׂר הַשִּׁירוֹת

רָמָה א׳ – שִׁירָה בַּלַּיְלָה

🌟 הַשִּׁירָה שֶׁנּוֹלְדָה בַּחֹשֶׁךְ

מָה לָמַדְנוּ עַל הַשִּׁירָה הָרִאשׁוֹנָה? שֶׁשָּׁרִים מִתּוֹךְ שִׂמְחָה, וְשֶׁשָּׁרִים גַּם בַּלַּיְלָה, כְּשֶׁעוֹד מְחַכִּים לַיְשׁוּעָה.

לֵב שֶׁבָּטוּחַ בַּה׳ אוֹמֵר רִנָּה מֵרֹאשׁ. וְהַתַּחֲנָה הַבָּאָה שֶׁלָּנוּ בַּמַּפָּה תִּהְיֶה שִׁירַת הַיָּם.

שִׁירָה מִתּוֹךְ שִׂמְחָה, גַּם בַּחֹשֶׁךְ.

רָמָה א׳ – שִׁירָה בַּלַּיְלָה

מָה לוֹקְחִים מִרָמָה א׳?

ה׳ הוֹצִיא אֶת בְּנֵי יִשְׂרָאֵל לַחֵרוּת, וְהֵם אָמְרוּ לוֹ שִׁירָה עוֹד בַּלַּיְלָה, כִּי הֵם בָּטְחוּ בּוֹ. כְּשֶׁאֲנִי שָׁר לַה׳ מִתּוֹךְ שִׂמְחָה, אֲנִי מַרְגִּישׁ כַּמָּה הוּא אוֹהֵב אוֹתִי, וְהַלֵּב שֶׁלִּי נִפְתָּח לְאַהֲבָה.

  • מָתַי הָיִיתִי שָׂמֵחַ כָּל כָּךְ שֶׁהָיָה לִי חֵשֶׁק לָשִׁיר אוֹ לְזַמְזֵם? עַל מָה הָיְתָה הַשִּׂמְחָה?
  • מָה הַדָּבָר שֶׁאֲנִי מְחַכֶּה לוֹ עַכְשָׁיו, וְעַל מָה כֵּן אֶפְשָׁר לְהוֹדוֹת לַה׳ כְּבָר הַיּוֹם?
  • מְשִׂימָה: ׳תּוֹדָה מֵרֹאשׁ׳ – אֶבְחַר הַיּוֹם דָּבָר אֶחָד שֶׁאֲנִי עוֹד מְחַכֶּה לוֹ, אוֹמַר לַה׳ תּוֹדָה בְּבִטָּחוֹן, וּכְשֶׁהוּא יָבוֹא – אוֹדֶה שׁוּב.

שֵׂכֶל וְלֵב, לָמָּה הַלֵּב מַאֲמִין רַק לְמַה שֶׁרוֹאֶה, וְאֵיךְ מַחֲזִירִים אֶת הַיְדִיעָה אֶל הַלֵּב

רָמָה ב׳ – שִׁכְנוּעַ הַלֵּב

🧩 הַתְּשׁוּבָה הַשְּׁנִיָּה

בַּשִּׁעוּר הַקּוֹדֵם שָׁאַלְנוּ: לָמָּה לְצַוּוֹת עַל אֱמוּנָה? תֵּרוּץ א׳ הָיָה – הַשְׁגָּחָה עַל כָּל פְּרָט. תֵּרוּץ ב׳: הַמִּצְוָה הִיא לְהַאֲמִין בַּלֵּב.

הַלֵּב מִטַּבְעוֹ אֵינוֹ מַאֲמִין לְמַה שֶׁלֹּא רוֹאֶה, וּמֵעוֹלָם לֹא רָאָה שֶׁיֵּשׁ בּוֹרֵא. לָכֵן הַמִּצְוָה הִיא לִגְרוֹם לַלֵּב לְהַאֲמִין, וְזֶה נִקְרָא אֱמוּנָה: אֶצֶל הַלֵּב, מַה שֶׁלֹּא רוֹאִים כְּאִלּוּ אֵינוֹ קַיָּם.

הַמִּצְוָה: לִגְרוֹם לַלֵּב לְהַאֲמִין, אַף שֶׁהוּא לֹא רוֹאֶה.

רָמָה ב׳ – שִׁכְנוּעַ הַלֵּב

🔍 לֹא בִּגְלַל שֶׁלֹּא רָאִינוּ אֶת הַשָּׂכָר

הַרְבֵּה אוֹמְרִים: הַלֵּב לֹא מַאֲמִין לַבּוֹרֵא כִּי מֵעוֹלָם לֹא רָאִינוּ אֶת הַשָּׂכָר שֶׁהוּא מַבְטִיחַ. זוֹ תְּשׁוּבָה לֹא נְכוֹנָה.

אִם יִתְקַשְּׁרוּ אֶל אָדָם וְיוֹדִיעוּ לוֹ שֶׁזָּכָה בִּפְרָס, הוּא יִשְׂמַח מְאֹד, גַּם שֶׁעוֹד לֹא רָאָה אֶת הַכֶּסֶף. הוּא מַאֲמִין שֶׁמִּי שֶׁאָמַר – יְקַיֵּם.

אֲבָל כְּשֶׁהַבּוֹרֵא אוֹמֵר שֶׁשָּׁעָה אַחַת שֶׁל נַחַת בָּעוֹלָם הַבָּא יָקָרָה מִכָּל חַיֵּי הָעוֹלָם הַזֶּה – כָּאן הַלֵּב פָּשׁוּט לֹא מַאֲמִין לַבּוֹרֵא.

הַבְּעָיָה הִיא לֹא שֶׁלֹּא רָאִינוּ שָׂכָר, אֶלָּא שֶׁהַלֵּב לֹא מַאֲמִין לַבּוֹרֵא.

רָמָה ב׳ – שִׁכְנוּעַ הַלֵּב

📜 לַעֲבֹד אֶת ה׳ בַּלֵּב

בִּקְרִיאַת שְׁמַע אָנוּ קוֹרְאִים כָּל יוֹם פַּעֲמַיִם: ׳וּלְעָבְדוֹ בְּכָל לְבַבְכֶם׳. כְּלוֹמַר, חוֹבָה לַעֲבֹד אֶת הַבּוֹרֵא לֹא רַק בַּשֵּׂכֶל אֶלָּא גַּם בַּלֵּב.

וְאֵיךְ עוֹשִׂים אֶת זֶה? הַתּוֹרָה גִּלְּתָה: ׳וְיָדַעְתָּ הַיּוֹם וַהֲשֵׁבֹתָ אֶל לְבָבֶךָ׳ – קֹדֶם תֵּדַע בַּשֵּׂכֶל, וְאַחַר כָּךְ תָּשִׁיב אֶל הַלֵּב.

קֹדֶם יוֹדְעִים בַּשֵּׂכֶל, וְאַחַר כָּךְ מַשִּׁיבִים אֶת הַיְדִיעָה אֶל הַלֵּב.

📖 אֱמוּנָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַמָ״א אוֹרַח חַיִּים א׳ א׳ · יִרְמְיָהוּ כ״ג כ״ד

רָמָה ב׳ – שִׁכְנוּעַ הַלֵּב

👀 הַמֶּלֶךְ הַגָּדוֹל רוֹאֶה

הָרַמָ״א פּוֹתֵחַ אֶת הַשֻּׁלְחָן עָרוּךְ בִּדְבָרִים עַל אֱמוּנָה בַּלֵּב: כְּשֶׁאָדָם מֵשִׂים אֶל לִבּוֹ שֶׁהַמֶּלֶךְ הַגָּדוֹל, הַקָּדוֹשׁ בָּרוּךְ הוּא, מָלֵא כָּל הָאָרֶץ וְעוֹמֵד עָלָיו וְרוֹאֶה אֶת מַעֲשָׂיו – מִיָּד מַגִּיעָה אֵלָיו יִרְאַת שָׁמַיִם.

כָּתוּב: ׳אִם יִסָּתֵר אִישׁ בַּמִּסְתָּרִים וַאֲנִי לֹא אֶרְאֶנּוּ נְאֻם ה׳׳.

זֹאת הָאֱמוּנָה שֶׁחַיָּה בַּלֵּב: ה׳ רוֹאֶה אוֹתִי בְּכָל רֶגַע.

לֵב שֶׁמַּאֲמִין מַרְגִּישׁ שֶׁה׳ רוֹאֶה אוֹתוֹ בְּכָל רֶגַע.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְשַׁכְנֵעַ אֶת הַלֵּב שֶׁלִּי שָׁלָב אַחַר שָׁלָב 💛

📖 אֱמוּנָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׁעוּר ׳אֱמוּנָה בַּלֵּב׳

רָמָה ב׳ – שִׁכְנוּעַ הַלֵּב

🪜 שְׁלוֹשָׁה שְׁלַבִּים

הַשִּׁעוּר מְלַמֵּד לְשַׁכְנֵעַ אֶת הַלֵּב בִּשְׁלוֹשָׁה שְׁלַבִּים.

א׳: מַתְחִילִים מֵהַתְחָלָה – מִנַּיִן שֶׁיֵּשׁ בּוֹרֵא? אוּלַי הָעוֹלָם הָיָה מֵאָז וּלְעוֹלָם? וּמוֹכִיחִים לְעַצְמֵנוּ, כְּאִלּוּ אָנוּ מַסְבִּירִים לְאָדָם שֶׁיּוֹשֵׁב מוּלֵנוּ, עַד שֶׁהוּא מִשְׁתַּכְנֵעַ.

ב׳: מְשַׁכְנְעִים אֶת עַצְמֵנוּ שֶׁאוֹתוֹ בּוֹרֵא נָתַן תּוֹרָה לְעַם יִשְׂרָאֵל.

ג׳: מְשַׁכְנְעִים אֶת עַצְמֵנוּ שֶׁהַטּוֹב הָאֲמִתִּי הוּא רַק לִשְׁמֹעַ לְתוֹרָתוֹ.

א׳ יֵשׁ בּוֹרֵא, ב׳ נָתַן תּוֹרָה, ג׳ הַטּוֹב הוּא לִשְׁמֹעַ לוֹ.

רָמָה ב׳ – שִׁכְנוּעַ הַלֵּב

🔁 כַּמָּה פְּעָמִים בַּיּוֹם?

לֹא מַסְפִּיק לַעֲשׂוֹת אֶת זֶה פַּעַם אַחַת, וְגַם לֹא פַּעַם אַחַת בַּיּוֹם. צָרִיךְ כַּמָּה פְּעָמִים בַּיּוֹם: לִפְנֵי כָּל סֵדֶר לִמּוּד וְאַחֲרָיו, לִפְנֵי הַתְּפִלָּה וְאַחֲרֶיהָ.

לֹא בְּיוֹם אֶחָד מַתְחִילִים לְהַרְגִּישׁ, אֲבָל אַחֲרֵי כִּשְׁלוֹשָׁה שָׁבוּעוֹת בְּעֵרֶךְ הַלֵּב מַתְחִיל לְהַאֲמִין, וְהָאֱמוּנָה חֲזָקָה הַרְבֵּה יוֹתֵר.

מִי שֶׁהָיָה יוֹדֵעַ כַּמָּה רוֹגַע וְשִׂמְחָה הוּא קוֹנֶה, וְכַמָּה קַל לְהִלָּחֵם בַּיֵּצֶר – כֻּלָּם הָיוּ עוֹשִׂים אֶת זֶה.

שׁוּב וָשׁוּב כָּל יוֹם, וְאַחֲרֵי כִּשְׁלוֹשָׁה שָׁבוּעוֹת מַרְגִּישִׁים.

מָה לוֹקְחִים מִרָמָה ב׳?

הַתּוֹרָה לִמְּדָה אוֹתָנוּ לֹא רַק לָדַעַת אֶלָּא גַּם לְהָשִׁיב אֶל הַלֵּב. מִי שֶׁעוֹשֶׂה זֹאת, מַרְוִיחַ רוֹגַע וְשִׂמְחָה, וְאֶת הַקִּרְבָה לַה׳ שֶׁמְּעוֹרֶרֶת אַהֲבָה.

רָמָה א׳ – חֲצִי שָׁעָה לְאִמָּא

🕰️ אֵין לִי זְמַן?

אַבָּא אוֹמֵר: ׳בּוֹא נָשִׂיחַ קְצָת׳, וְהַיֶּלֶד עוֹנֶה: ׳אֵין לִי זְמַן, יֵשׁ לִי הַרְבֵּה לַעֲשׂוֹת׳.

מִי שֶׁבֶּאֱמֶת הָיָה לוֹ הַרְבֵּה לַעֲשׂוֹת הָיָה הַחֲזוֹן אִישׁ, גְּדוֹל הַדּוֹר: לִמּוּד תּוֹרָה, כְּתִיבַת סְפָרִים, וַאֲנָשִׁים שֶׁבָּאוּ לִשְׁאֹל אוֹתוֹ.

וּבְכָל זֹאת, כָּל יוֹם מָצָא חֲצִי שָׁעָה לְאִמּוֹ. הַשִּׁעוּר הַזֶּה מְסַפֵּר אֵיךְ.

גַּם מִי שֶׁעָסוּק מְאֹד מוֹצֵא זְמַן לַהוֹרִים.

רָמָה א׳ – חֲצִי שָׁעָה לְאִמָּא

🏡 כָּל יוֹם, עַל כָּל דָּבָר

הַחֲזוֹן אִישׁ הָיָה מְבַקֵּר אֶת אִמּוֹ חֲצִי שָׁעָה כָּל יוֹם.

הוּא יָשַׁב עִמָּהּ, דִּבֵּר עִמָּהּ עַל כָּל עִנְיְנֵי הָעוֹלָם, וְהָיָה גַּם מְטַיֵּל עִמָּהּ הַרְבֵּה.

לֹא רַק מַגִּיעִים לְבִקּוּר קָצָר – מְדַבְּרִים בֶּאֱמֶת, וְגַם הוֹלְכִים יַחַד.

חֲצִי שָׁעָה שֶׁל שִׂיחָה אֲמִתִּית, כָּל יוֹם.

🏃 הוּא רָץ!

פַּעַם אַחַת שָׁמַע הַחֲזוֹן אִישׁ שֶׁאִמּוֹ זְקוּקָה לוֹ. הוּא לֹא חִכָּה: הוּא יָצָא בְּרִיצָה לְבֵיתָהּ בְּרַגְלָיו.

וְזֶה קָרָה כְּשֶׁכְּבָר הָיָה זָקֵן, וְהוּא לֹא גָּר קָרוֹב אֵלֶיהָ.

כְּשֶׁהוֹרֶה צָרִיךְ אוֹתָנוּ – לֹא מְחַכִּים.

כְּשֶׁהַהוֹרֶה צָרִיךְ אוֹתִי, אֲנִי בָּא מִיָּד.

🎁

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְחַפֵּשׂ בְּעַצְמִי דְּרָכִים לְשַׂמֵּחַ אֶת הוֹרַי 💛

📖 כִּבּוּד אָב וָאֵם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרַב חַיִּים קַנְיֶבְסְקִי עַל הַחֲזוֹן אִישׁ

רָמָה א׳ – חֲצִי שָׁעָה לְאִמָּא

🎁 בְּלִי שֶׁבִּקְּשׁוּ

שָׁאֲלוּ אֶת הָרַב חַיִּים קַנְיֶבְסְקִי: אֵיךְ אִמּוֹ שֶׁל הַחֲזוֹן אִישׁ נָתְנָה לוֹ לְהַקְדִּישׁ חֲצִי שָׁעָה כָּל יוֹם?

הוּא עָנָה: הִיא לֹא בִּקְּשָׁה, הוּא בָּא מֵעַצְמוֹ.

הוּא חִפֵּשׂ בְּעַצְמוֹ אֶת הַמִּצְוָה. מִי שֶׁאוֹהֵב מִצְוָה לֹא מְחַכֶּה לְבַקָּשָׁה.

הַחֲזוֹן אִישׁ בָּא מֵעַצְמוֹ, בְּלִי בַּקָּשָׁה.

🤫

💛 אַהֲבַת הַשֵּׁם: אֲנִי שׁוֹמֵר עַל הוֹרַי מִצַּעַר, וַה׳ שׁוֹמֵר עָלַי 💛

📖 כִּבּוּד אָב וָאֵם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרַב בֶּן צִיּוֹן פַּלְמָן עַל הַחֲזוֹן אִישׁ

רָמָה א׳ – חֲצִי שָׁעָה לְאִמָּא

🤫 כְּשֶׁכּוֹאֵב – בְּשֶׁקֶט

כְּשֶׁהַחֲזוֹן אִישׁ הָיָה יֶלֶד, הָיוּ לוֹ לִפְעָמִים כְּאֵבֵי בֶּטֶן חֲזָקִים. כְּשֶׁהִרְגִּישׁ שֶׁהַכְּאֵב מִתְקָרֵב, בָּרַח לְבֵית הַכְּנֶסֶת וְנִשְׁכַּב עַל סַפְסָל.

לָמָּה? כְּדֵי שֶׁהוֹרָיו לֹא יִשְׁמְעוּ אֶת הַגְּנִיחוֹת וְלֹא יִצְטַעֲרוּ.

כִּבּוּד הוֹרִים הוּא גַּם לֹא לְצַעֵר אוֹתָם. (וּכְשֶׁיֵּשׁ כְּאֵב חָזָק אוֹ בְּעָיָה, אוֹמְרִים לַהוֹרִים בְּוַדַּאי!)

יֶלֶד שֶׁלֹּא רוֹצֶה שֶׁהַהוֹרִים יִצְטַעֲרוּ.

📖 כִּבּוּד אָב וָאֵם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מַעֲשֶׂה אִישׁ, חֵלֶק ד׳

🏆 בִּזְכוּת מָה?

בִּבְחִינָה שֶׁל תַּלְמִידֵי תַּלְמוּד תּוֹרָה שָׁאַל הַמְּלַמֵּד אֶת הַחֲזוֹן אִישׁ: ׳בַּמֶּה זָכִיתָ לְהַגִּיעַ לְמָה שֶׁהִגַּעְתָּ?׳ עֵינֵי כָּל הַיְלָדִים הִסְתַּכְּלוּ עָלָיו.

כְּדֵי לְחַנֵּךְ אוֹתָם עָנָה: ׳בִּזְכוּת כִּבּוּד הוֹרִים וּמוֹרִים!׳

הַסּוֹד שֶׁל הַחֲזוֹן אִישׁ: כִּבּוּד הוֹרִים וּמוֹרִים.

🔄 תּוֹר שֶׁלִּי לָתֵת

כְּשֶׁהָיִינוּ תִּינוֹקוֹת, הַהוֹרִים נָתְנוּ לָנוּ הַכֹּל: אֹכֶל, בַּיִת, אַהֲבָה וְהַרְבֵּה זְמַן.

הַחֲזוֹן אִישׁ הֵבִין שֶׁאָדָם צָרִיךְ לִהְיוֹת נוֹתֵן וְלֹא רַק מְקַבֵּל, כִּי מִי שֶׁרַק מְקַבֵּל רָחוֹק יוֹתֵר מֵה׳, הַנּוֹתֵן.

וְכָךְ הוּא הֶחְזִיר לְאִמּוֹ: בְּבִקּוּר, בְּשִׂיחָה וּבִזְמַן.

הַחֲזוֹן אִישׁ הֶחְזִיר לְאִמּוֹ בְּבִקּוּר וּבְשִׂיחָה.

🔟 עֲשָׂרָה דַּקּוֹת

אֵין צֹרֶךְ בַּחֲצִי שָׁעָה כְּדֵי לְהַתְחִיל. הַשָּׁבוּעַ אֲנִי בּוֹחֵר עֲשָׂרָה דַּקּוֹת בְּיוֹם לְשִׂיחָה עִם הוֹרֶה, סַבָּא אוֹ סַבְתָּא.

אֲנִי מְסַפֵּר מָה עָבַר עָלַי, שׁוֹאֵל שְׁאֵלָה, וּמַקְשִׁיב.

וּבְסוֹף הַשָּׁבוּעַ אֶרְשֹׁם מָה הִרְגַּשְׁתִּי, וּמָה הִרְגִּישׁוּ הֵם.

עֲשָׂרָה דַּקּוֹת שֶׁל שִׂיחָה בְּכָל יוֹם.

מָה לוֹקְחִים מִרָמָה א׳?

הַחֲזוֹן אִישׁ הָיָה גָּדוֹל, וְהוּא לָמַד שֶׁכִּבּוּד הוֹרִים אֵינוֹ רַק לַעֲשׂוֹת מַה שֶּׁמְּבַקְּשִׁים, אֶלָּא לָבוֹא מֵעַצְמוֹ וְלָתֵת. ה׳ נָתַן לָנוּ הוֹרִים כְּמַתָּנָה, וְכָל חֲצִי שָׁעָה אִתָּם מְקָרֶבֶת אוֹתָנוּ אֶל ה׳, שֶׁאוֹהֵב אוֹתָנוּ וְנוֹתֵן לָנוּ הַכֹּל.

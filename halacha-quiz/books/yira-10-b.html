רָמָה ב׳ – דְּרָכִים לִקְנוֹת יִרְאָה

הָעֵצָה שֶׁל רַבֵּנוּ יוֹנָה וּתְפִלָּתוֹ שֶׁל רַבִּי נַחְמָן: חֲבֵרִים שֶׁמְּחַזְּקִים זֶה אֶת זֶה בְּיִרְאַת שָׁמַיִם

📖 יִרְאַת שָׁמַיִם · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

💛 אַהֲבַת הַשֵּׁם: אֲנִי יָכוֹל לִהְיוֹת הֶחָבֵר שֶׁמְּחַזֵּק 💛

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַבֵּנוּ יוֹנָה, אִגֶּרֶת הַתְּשׁוּבָה · פְּנִינִים בְּמִצְוַת יִרְאַת הַשֵּׁם

רָמָה ב׳ – דְּרָכִים לִקְנוֹת יִרְאָה

📜 אִגֶּרֶת הַתְּשׁוּבָה

רַבֵּנוּ יוֹנָה כּוֹתֵב עֵצָה טוֹבָה לִקְנוֹת יִרְאַת שָׁמַיִם: שֶׁיִּהְיֶה לוֹ חָבֵר שֶׁיְּדַבֵּר עִמּוֹ בְּיִרְאַת שָׁמַיִם.

זֹאת עֵצָה לַחַיִּים, לֹא רַק לַמְּבֻגָּרִים: כְּשֶׁחָבֵר מְדַבֵּר עַל ה׳, הַלֵּב נִזְכָּר וְנִפְתָּח.

וְכָךְ גַּם אֲנִי יָכוֹל לִהְיוֹת הֶחָבֵר שֶׁמְּחַזֵּק אֶת חֲבֵרִי.

יִרְאַת שָׁמַיִם קוֹנִים בְּעֶזְרַת חָבֵר שֶׁמְּדַבֵּר בָּהּ.

1
🧭

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִתְרַגֵּל בִּשְׁתֵּי הַדְּרָכִים: לִזְכֹּר וְלִקְרֹא 💛

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְסִלַּת יְשָׁרִים · דְּבָרִים י״ז י״ט

רָמָה ב׳ – דְּרָכִים לִקְנוֹת יִרְאָה

🧭 שְׁתֵּי דְּרָכִים בִּמְסִלַּת יְשָׁרִים

בִּמְסִלַּת יְשָׁרִים כָּתוּב שֶׁקּוֹנִים יִרְאַת שָׁמַיִם בִּשְׁתֵּי דְּרָכִים.

הָרִאשׁוֹנָה: לְהִתְרַגֵּל לִזְכֹּר שֶׁה׳ רוֹאֶה כָּל מַעֲשֶׂה שֶׁאָדָם עוֹשֶׂה וְהוּא מְנַהֵל אֶת הַכֹּל. בְּקִצּוּר, לִחְיוֹת אֱמוּנָה שְׁלֵמָה.

הַשְּׁנִיָּה: לִלְמֹד תּוֹרָה. הַפָּסוּק: ׳וְהָיְתָה עִמּוֹ וְקָרָא בוֹ כָּל יְמֵי חַיָּיו לְמַעַן יִלְמַד לְיִרְאָה׳, וּמִכָּאן שֶׁקְרִיאָה בַּתּוֹרָה גּוֹרֶמֶת יִרְאַת שָׁמַיִם.

שְׁתֵּי דְּרָכִים: לִזְכֹּר שֶׁה׳ רוֹאֶה, וְלִקְרֹא בַּתּוֹרָה.

2
🔎

💛 אַהֲבַת הַשֵּׁם: אֲנִי קוֹרֵא פְּסוּקִים וּמַכִּיר אֶת ה׳ יוֹתֵר 💛

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דְּבָרִים ו׳ ו׳ · רַשִׁ״י · פְּנִינִים בְּמִצְוַת יִרְאַת הַשֵּׁם

רָמָה ב׳ – דְּרָכִים לִקְנוֹת יִרְאָה

🔎 פְּסוּקִים שֶׁמְּלַמְּדִים

אָדָם שֶׁמִּתְבּוֹנֵן בַּפְּסוּקִים שֶׁבַּתּוֹרָה בְּלִי מְפָרְשִׁים, אֶפְשָׁר לָשִׂים לֵב: כָּל הַפְּסוּקִים מְלַמְּדִים שֶׁה׳ שׁוֹלֵט בַּכֹּל.

כָּךְ הַקְּרִיאָה בַּתּוֹרָה מְחַבֶּרֶת בֵּין שְׁתֵּי הַדְּרָכִים: הִיא גַּם לִמּוּד וְגַם תַּרְגִּיל בִּזְכִירָה שֶׁה׳ מְנַהֵל.

בְּרַשִׁ״י עַל הַפָּסוּק ׳וְהָיוּ הַדְּבָרִים הָאֵלֶּה׳: מִתּוֹךְ כָּךְ אַתָּה מַכִּיר אֶת הַקָּדוֹשׁ בָּרוּךְ הוּא.

בַּפְּסוּקִים רוֹאִים שֶׁה׳ שׁוֹלֵט בַּכֹּל, וְכָךְ מַכִּירִים אוֹתוֹ.

💠

💛 אַהֲבַת הַשֵּׁם: יִרְאַת ה׳ הִיא הָאוֹצָר שֶׁלִּי וְהַפֶּתַח לְאַהֲבָה 💛

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שַׁבָּת ל״א א׳ · פְּנִינִים בְּמִצְוַת יִרְאַת הַשֵּׁם

רָמָה ב׳ – דְּרָכִים לִקְנוֹת יִרְאָה

💠 הָאוֹצָר הַחֲשׁוּב

חֲכָמִים אָמְרוּ שֶׁגַּם לְמִי שֶׁיֵּשׁ בּוֹ הַרְבֵּה דְּבָרִים טוֹבִים, הַדָּבָר הֶחָשׁוּב הוּא שֶׁיִּרְאַת ה׳ תִּהְיֶה הָאוֹצָר שֶׁלּוֹ.

וְעוֹד כָּתוּב: יִרְאַת ה׳ הִיא פֶּתַח לְהִכָּנֵס מִמֶּנּוּ לְאַהֲבַת ה׳. הַיִּרְאָה אֵינָהּ סוֹף הַדֶּרֶךְ, הִיא הַהַתְחָלָה שֶׁלָּהּ.

יִרְאַת ה׳ הִיא הָאוֹצָר וְהַפֶּתַח לְאַהֲבַת ה׳.

4
🌙

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְדַבֵּר עִם חֲבֵרִים וּמְאִיר אוֹתָם בְּמָה שֶׁיֵּשׁ בִּי 💛

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִתְּפִלּוֹתָיו שֶׁל רַבִּי נַחְמָן

רָמָה ב׳ – דְּרָכִים לִקְנוֹת יִרְאָה

🌙 הַתְּפִלָּה שֶׁל רַבִּי נַחְמָן

רַבִּי נַחְמָן כָּתַב תְּפִלָּה עַל יִרְאַת שָׁמַיִם: ׳שֶׁיִּהְיֶה לָנוּ יִרְאַת שָׁמַיִם בֶּאֱמֶת, וְתִהְיֶה יִרְאָתְךָ עַל פָּנֵינוּ׳. הוּא מְבַקֵּשׁ גַּם ׳יִרְאָה תַתָּאָה וְיִרְאָה עִלָּאָה׳, כְּלוֹמַר יִרְאַת הָעֹנֶשׁ וְיִרְאַת הָרוֹמְמוּת.

וְאָז: ׳שֶׁכָּל אֶחָד וְאֶחָד יַעֲסֹק וִידַבֵּר עִם חֲבֵרוֹ בְּיִרְאַת שָׁמַיִם בְּכָל יוֹם וָיוֹם׳.

בִּתְפִלָּה כָּתוּב שֶׁלְּכָל אֶחָד וְאֶחָד מִיִּשְׂרָאֵל יֵשׁ דַּעַת וְדִבּוּרִים קְדוֹשִׁים שֶׁאֵין לַחֲבֵרוֹ, וְצָרִיךְ לְהָאִיר אוֹתָם בַּחֲבֵרִים.

רַבִּי נַחְמָן מְבַקֵּשׁ יִרְאָה, וְשֶׁחֲבֵרִים יְחַזְּקוּ זֶה אֶת זֶה בְּכָל יוֹם.

5
💡

רָמָה ב׳ – דְּרָכִים לִקְנוֹת יִרְאָה

מָה לוֹקְחִים מִרָמָה ב׳?

יִרְאַת ה׳ הִיא הָאוֹצָר וְהַפֶּתַח לְאַהֲבָתוֹ. כְּשֶׁחֲבֵרִים מְחַזְּקִים זֶה אֶת זֶה, כֻּלָּם מִתְקָרְבִים אֶל ה׳.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • מִי הֶחָבֵר שֶׁיָּכוֹל לְחַזֵּק אוֹתִי בְּיִרְאַת שָׁמַיִם, וְאֶת מִי אֲנִי יָכוֹל לְחַזֵּק?
  • מָה מִשְׁפָּט אֶחָד שֶׁחִזֵּק אוֹתִי וְשֶׁאֲנִי רוֹצֶה לוֹמַר לְחָבֵר?
  • אֶבְחַר חָבֵר, וּפַעַם בַּשָּׁבוּעַ נַזְכִּיר זֶה לָזֶה מִשְׁפָּט אֶחָד שֶׁמְּחַזֵּק.

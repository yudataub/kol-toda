זְרִיזִין מַקְדִּימִין לַמִּצְווֹת

🌱 בְּעַצְמוֹ · רָמָה ב׳

בְּעַצְמִי, בִּזְרִיזוּת – הַכְּרָכָּרָה, הַמִּכְתָּב וְעֶרֶב פֶּסַח

רָמָה ב׳ – חֶסֶד וְעִקָּר

הָרַב חַיִּים זוֹנֶנְפֶלְד לֹא שׁוֹלֵחַ אֲחֵרִים: הוּא עוֹלֶה בַּגֶּשֶׁם, רָץ לְבֵית הַחוֹלִים וּמְחַלֵּק כֶּסֶף לְפֶסַח. מָה אוֹמֵר זָרִיז עַל מִצְוָה שֶׁעוֹשִׂים בְּעַצְמוֹ?

📖 זְרִיזוּת · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
🤲

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹשֶׂה חֶסֶד בְּגוּפִי, כִּי כָּךְ אֲנִי דּוֹמֶה לַה׳ 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מָרָא דְאַרְעָא יִשְׂרָאֵל

רָמָה ב׳ – חֶסֶד וְעִקָּר

🤲 גְּמִילוּת חֲסָדִים בַּגּוּף

מִפִּי עֵד רְאִיָּה מְסַפְּרִים עַל הָרַב חַיִּים: זוֹ הָיְתָה גְּמִילוּת חֶסֶד שֶׁבְּגוּפוֹ. הָרַב לֹא רַק אָמַר דְּבָרִים טוֹבִים, אֶלָּא עָשָׂה בְּגוּפוֹ הַזָּקֵן וְהֶעָיֵף.

הַחֶסֶד שֶׁבַּגּוּף קָשֶׁה יוֹתֵר מֵהַחֶסֶד שֶׁבַּדִּבּוּר, כִּי הוּא דּוֹרֵשׁ מַאֲמָץ, הַקְרָבָה וּזְרִיזוּת.

גַּם חֶסֶד שֶׁבַּגּוּף צָרִיךְ לַעֲשׂוֹת מִיָּד וְלֹא לְהַמְתִּין.

חֶסֶד שֶׁבַּגּוּף = לַעֲשׂוֹת, לֹא רַק לוֹמַר.

1
🧑

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְבַקֵּשׁ מֵה׳ לֵב חָפֵץ, וְהַגּוּף כְּבָר יִצְטָרֵף 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מַעֲלַת הַזְּרִיזוּת · מָרָא דְאַרְעָא יִשְׂרָאֵל

רָמָה ב׳ – חֶסֶד וְעִקָּר

🧑 תְּנוּעָה שֶׁל צָעִיר

הָרַב הָיָה קָרוֹב לְגִיל שְׁמוֹנִים, וְמְתָאֲרִים אוֹתוֹ כְּמִי שֶׁטִּפֵּס בִּתְנוּעָה זְרִיזָה שֶׁל צָעִיר אֶל הַמּוֹשָׁב הֶחָשׂוּף בַּגֶּשֶׁם.

הַזְּרִיזוּת אֵינָהּ תְּלוּיָה בְּגִיל: מִי שֶׁלִּבּוֹ חָפֵץ בְּמִצְוָה, גּוּפוֹ מִצְטָרֵף. כְּמוֹ שֶׁנֶּאֱמַר ׳יִתְגַּבֵּר כָּאֲרִי לַעֲמֹד בַּבֹּקֶר׳.

וְכֵן צָרִיךְ לִזְכֹּר: כְּשֶׁהָאִשָּׁה סֵרְבָה, הוּא נִשְׁאַר עַל הַמּוֹשָׁב, כִּי הַחֶסֶד לֹא צָרִיךְ לְהִתְבַּטֵּל.

הַלֵּב הַחָפֵץ מַזְרִיז אֶת הַגּוּף.

2
✉️

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹזֵר לְכָל יְהוּדִי, כִּי כֻּלָּנוּ יְלָדִים שֶׁל ה׳ 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, זָרִיז וְנִשְׂכָּר, פֶּרֶק ה׳ (הָאִישׁ עַל הַחוֹמָה)

רָמָה ב׳ – חֶסֶד וְעִקָּר

✉️ לֹא סוֹמְכִים עַל מִכְתָּב

בִּקְשׁוּ מֵהָרַב מִכְתָּב הַמְלָצָה כְּדֵי שֶׁבֵּית חוֹלִים יְקַבֵּל חוֹלֶה מְאֹד. הַחוֹלֶה הָיָה אִישׁ שֶׁחָשַׁב אַחֶרֶת מֵהָרַב עַל כַּמָּה עִנְיָנִים.

הָרַב לֹא שָׁאַל מִי הוּא. הוּא לֹא סָמַךְ עַל הַמִּכְתָּב וְרָץ בְּעַצְמוֹ בְּגֶשֶׁם שׁוֹטֵף, סִדֵּר שֶׁיְּקַבְּלוּ אוֹתוֹ, וְדָאַג לוֹ עַד שֶׁהִבְרִיא.

חֶסֶד אֱמֶת אֵינוֹ בּוֹחֵר מִי רָאוּי לוֹ.

חֶסֶד שֶׁל זָרִיז לֹא בּוֹחֵר, וְלֹא מַמְתִּין לְמִכְתָּב.

3
🕯️

💛 אַהֲבַת הַשֵּׁם: אֲנִי יוֹדֵעַ מָה הָעִקָּר וּמָה הַתּוֹסֶפֶת, וְה׳ מַדְרִיךְ אוֹתִי 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מָרָא דְאַרְעָא יִשְׂרָאֵל

רָמָה ב׳ – חֶסֶד וְעִקָּר

🕯️ סֵדֶר מְהֻדָּר אוֹ סֵדֶר לַאֲחֵרִים?

בְּעֶרֶב פֶּסַח הָרַב הַפְסִיק אֶת הַהַכָּנוֹת שֶׁלּוֹ כְּדֵי לְחַלֵּק כֶּסֶף לַאֲחֵרִים: ׳לָנוּ לֹא אִכְפַּת אִם אֵין סֵדֶר מְהֻדָּר, הָעִקָּר מַצָּה, מָרוֹר וּבִעוּר חָמֵץ כַּדִּין. לַאֲחֵרִים אִכְפַּת, כִּי הֶחָג יֵהָפֵךְ לָהֶם לַאֲבֵלוּת וְתָקוּם מְרִיבָה בְּכָל בַּיִת׳.

הַחֲשִׁיבוּת הִיא לִקְבֹּעַ סֵדֶר: קוֹדֵם מַצָּה וּבִעוּר חָמֵץ, וְאַחַר כָּךְ חֶסֶד, וְאַחַר כָּךְ הִדּוּר.

מַקְדִּימִים מִצְוָה שֶׁלֹּא יְכוֹלָה לְחַכּוֹת, וּמְוַתְּרִים עַל הִדּוּר.

4
❤️

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַרְגִּישׁ אֶת הַמִּצְוָה כְּשֶׁאֲנִי עוֹשֶׂה אוֹתָהּ מִיָּד 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מַעֲלַת הַזְּרִיזוּת

רָמָה ב׳ – חֶסֶד וְעִקָּר

❤️ זְרִיזוּת מְבִיאָה אֶת הַלֵּב

אֶפְשָׁר לְקַיֵּם מִצְוָה בַּשֵּׂכֶל, כְּשֶׁחוֹשְׁבִים וּמְחַשְּׁבִים. אֲבָל זְרִיזוּת מְבִיאָה אֶת הַלֵּב: מִי שֶׁעוֹשֶׂה מִיָּד, מַרְגִּישׁ אֶת הַמִּצְוָה.

דֻּגְמָה: אִם אִמָּא מְבַקֶּשֶׁת וַאֲנִי הוֹלֵךְ מִיָּד, וְלֹא אַחֲרֵי חָמֵשׁ דַּקּוֹת – יֵשׁ הֶבְדֵּל גָּדוֹל. הַזְּרִיזוּת מְחַבֶּרֶת אוֹתִי לַמִּצְוָה.

הִיא מְחַבֶּרֶת אוֹתִי לַמִּצְוָה דֶּרֶךְ הַלֵּב.

5
💭

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹחֵר דָּבָר טוֹב אֶחָד וְעוֹשֶׂה אוֹתוֹ בְּעַצְמִי 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – חֶסֶד וְעִקָּר

💭 מָה אֲנִי עוֹשֶׂה בְּעַצְמִי?

יֵשׁ דְּבָרִים שֶׁאֶפְשָׁר לָתֵת לְמִישֶׁהוּ אַחֵר לַעֲשׂוֹת: לַעֲזֹר לָאָח, לְסַדֵּר, לְהַכִּין, לְהוֹצִיא זְבָלִים. זָרִיז עוֹשֶׂה בְּעַצְמוֹ, וְלֹא מַעֲבִיר הָלְאָה.

כָּל יוֹם אֶפְשָׁר לִבְחֹר דָּבָר אֶחָד קָטָן וְלַעֲשׂוֹת אוֹתוֹ בְּעַצְמִי.

דָּבָר אֶחָד קָטָן, בְּעַצְמִי, כָּל יוֹם.

6
💡

רָמָה ב׳ – חֶסֶד וְעִקָּר

מָה לוֹקְחִים מִרָמָה ב׳?

זְרִיזוּת בְּחֶסֶד מְחַבֶּרֶת אֶת הַלֵּב לַמִּצְוָה. מִי שֶׁעוֹשֶׂה חֶסֶד בְּעַצְמוֹ וּמִיָּד, מַרְגִּישׁ אֶת הָאַהֲבָה שֶׁל ה׳ אֶל בְּרִיּוֹתָיו וְאֶת הָאַהֲבָה שֶׁלּוֹ אֵלָיו.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָה אֲנִי יָכוֹל לַעֲשׂוֹת הַיּוֹם בְּעַצְמִי, בִּמְקוֹם לְבַקֵּשׁ מִמִּישֶׁהוּ אַחֵר?
  • מָתַי הַשָּׁבוּעַ הֶעֱבַרְתִּי לְמִישֶׁהוּ אַחֵר מִצְוָה שֶׁיָּכֹלְתִּי לַעֲשׂוֹת בְּעַצְמִי?
  • הַיּוֹם אֶבְחַר דָּבָר טוֹב אֶחָד וְאֶעֱשֶׂה אוֹתוֹ בְּעַצְמִי, בְּשִׂמְחָה וּמִיָּד.

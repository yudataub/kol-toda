רוֹעִים בָּאוּ וְגֵרְשׁוּ אֶת בְּנוֹת יִתְרוֹ מֵהַבְּאֵר, וּמֹשֶׁה קָם וְעָזַר לָהֶן לְהַשְׁקוֹת אֶת הַצֹּאן.

מֹשֶׁה עָזַר לַאֲחֵרִים, וּבָאָה אֵלָיו טוֹבָה גְּדוֹלָה בַּחֲזָרָה.

לָמָּה? כִּי הוּא גָּדַל בְּמִדְיָן וְקִבֵּל שָׁם טוֹבָה. הוּא אָמַר: אֵינִי יָכוֹל לְהָצֵר לְמִי שֶׁהֵטִיב לִי.

חֲכָמִים אָמְרוּ בְּמָשָׁל: בּוֹר שֶׁשָּׁתִיתָ מִמֶּנּוּ – אַל תִּזְרֹק בּוֹ אֶבֶן.

לֹא פּוֹגְעִים בְּמִי שֶׁהֵטִיב לָנוּ, גַּם כְּשֶׁהַמַּצָּב מִשְׁתַּנֶּה.

בַּיִת שֶׁנִּבְנָה עַל הַכָּרַת הַטּוֹב הוּא בַּיִת יַצִּיב וְחָזָק.

מִי שֶׁדָּאַג לָנוּ בִּזְמַן קָשֶׁה – זוֹכְרִים אוֹתוֹ תָּמִיד.

הַרְבֵּה שָׁנִים אַחֲרֵי מֹשֶׁה, בִּימֵי הַמֶּלֶךְ שָׁאוּל, חָיוּ בְּנֵי בְּנֵי יִתְרוֹ בַּמִּדְבָּר, לְיַד מָקוֹם מְסֻכָּן. שָׁאוּל הִזְהִיר אוֹתָם לְהִתְרַחֵק, וְאָמַר: ׳וְאַתָּה עָשִׂיתָה חֶסֶד עִם כָּל בְּנֵי יִשְׂרָאֵל׳.

הַחֶסֶד שֶׁל יִתְרוֹ לֹא נִשְׁכַּח. אֲפִלּוּ נֶכֶד שֶׁל נֶכֶד קִבֵּל הַכָּרַת טוֹבָה.

הַכָּרַת הַטּוֹב נִמְשֶׁכֶת גַּם לַצֶּאֱצָאִים שֶׁל הַמֵּטִיב.

💛 מִי פָּתַח לִי אֶת הַדֶּלֶת?

גַּם לְךָ יֵשׁ מִישֶׁהוּ שֶׁעָזַר בִּזְמַן קָשֶׁה: חָבֵר שֶׁהִשְׁאִיל, שָׁכֵן שֶׁפָּתַח אֶת הַדֶּלֶת, מוֹרֶה שֶׁהִקְשִׁיב.

מֹשֶׁה מְלַמֵּד שְׁלוֹשָׁה דְּבָרִים: זוֹכְרִים מִי עָזַר, אוֹמְרִים תּוֹדָה וּמְבַקְשִׁים רְשׁוּת כְּשֶׁצָּרִיךְ, וּמַחְזִירִים טוֹבָה כְּשֶׁאֶפְשָׁר.

מִי שֶׁלּוֹמֵד לְהַכִּיר טוֹבָה לְאָדָם, לוֹמֵד גַּם לְהַכִּיר טוֹבָה לַה׳.

זוֹכְרִים, אוֹמְרִים תּוֹדָה, וּמַחְזִירִים טוֹבָה.

מָה לוֹקְחִים מִרָמָה א׳?

ה׳ שָׁלַח לְמֹשֶׁה בַּיִת חַם בְּמִדְיָן בִּשְׁעַת צֹרֶךְ, וּמֹשֶׁה לֹא שָׁכַח. כְּשֶׁאֲנִי זוֹכֵר מִי עָזַר לִי, אֲנִי מַרְגִּישׁ שֶׁה׳ הוּא שֶׁשָּׁלַח לִי אֶת הָעֶזְרָה, וְכָךְ אַהֲבָתִי אֵלָיו גְּדֵלָה.

  • מִי עָזַר לִי פַּעַם כְּשֶׁהָיִיתִי בְּצָרָה אוֹ בִּמְבוּכָה, וְהַאִם הוֹדֵיתִי לוֹ כָּרָאוּי?
  • אֵיךְ אֲנִי יָכוֹל לְהַחְזִיר טוֹבָה לְמִי שֶׁעָזַר לִי – בְּמִלִּים, בְּמַעֲשֶׂה אוֹ בִּבְקָשַׁת רְשׁוּת?
  • הַשָּׁבוּעַ אֶבְחַר אָדָם אֶחָד שֶׁעָזַר לִי בִּזְמַן קָשֶׁה, וְאֶכְתֹּב לוֹ תּוֹדָה אוֹ אֶעֱשֶׂה לוֹ טוֹבָה קְטַנָּה בְּחִנָּם – וְאֶרְשֹׁם מָה הִרְגַּשְׁתִּי.

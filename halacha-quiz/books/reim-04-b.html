נְשָׁמָה אַחַת

רָמָה ב׳ – שֹׁרֶשׁ וּמָחוֹל

נְשָׁמָה אֱלוֹקִית בְּכָל יְהוּדִי, שֹׁרֶשׁ אֶחָד, נְפָשׁוֹת קְשׁוּרוֹת וּמָחוֹל שֶׁל צַדִּיקִים

📖 אַהֲבַת יִשְׂרָאֵל · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
💫

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִסְתַּכֵּל פְּנִימָה וְרוֹאֶה נְשָׁמָה יָקָרָה 💛

📖 אַהֲבַת יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, כֵּיצַד מַגִּיעִים לְאַהֲבַת יִשְׂרָאֵל

רָמָה ב׳ – שֹׁרֶשׁ וּמָחוֹל

💫 הַנְּשָׁמָה הָאֱלוֹקִית

בַּעַל הַתַּנְיָא כָּתַב עֵצָה לְאַהֲבַת יִשְׂרָאֵל: לְהִתְבּוֹנֵן שֶׁבַּיְּהוּדִי שֶׁלְּפָנַי יֵשׁ נְשָׁמָה אֱלוֹקִית, וְלֶאֱהֹב אֶת הַנְּשָׁמָה שֶׁבּוֹ.

כָּךְ אֲנִי לֹא מִסְתַּכֵּל רַק עַל מַה שֶּׁרוֹאִים בַּחוּץ, כְּמוֹ מַרְגִּיז, שׁוֹנֶה אוֹ חָלָשׁ, אֶלָּא עַל מַה שֶּׁבִּפְנִים. הַנְּשָׁמָה שֶׁבִּפְנִים תָּמִיד יָקָרָה.

אוֹהֲבִים אֶת הַנְּשָׁמָה שֶׁבַּפְּנִים.

1
🌳

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׁלֵם יוֹתֵר כְּשֶׁאֲנִי אוֹהֵב אֶת כֻּלָּם 💛

📖 אַהֲבַת יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דֶּרֶךְ מִצְוֹתֶיךָ · שַׁעֲרֵי יוֹשֶׁר

רָמָה ב׳ – שֹׁרֶשׁ וּמָחוֹל

🌳 שֹׁרֶשׁ אֶחָד – חֵלֶק מִנִּשְׁמָתִי

בַּ׳דֶּרֶךְ מִצְוֹתֶיךָ׳ וּבְ׳שַׁעֲרֵי יוֹשֶׁר׳ כָּתוּב לְהִתְבּוֹנֵן בְּכָךְ שֶׁשֹּׁרֶשׁ כָּל נִשְׁמוֹת יִשְׂרָאֵל אֶחָד.

לָכֵן, אִם אָדָם אֵינוֹ אוֹהֵב יְהוּדִי אֶחָד, חָסֵר לוֹ חֵלֶק מִנִּשְׁמָתוֹ. וְכַאֲשֶׁר אוֹהֵב אוֹתוֹ, הוּא אוֹהֵב אֶת עַצְמוֹ.

הַנְּשָׁמוֹת בְּאוֹתוֹ שֹׁרֶשׁ, וְאַהֲבָה לִיהוּדִי הִיא אַהֲבָה לְעַצְמִי.

2
🔗

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִקְשֹׁר אֶת לִבִּי לִיהוּדִים אֲחֵרִים 💛

📖 אַהֲבַת יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּרֵאשִׁית מ״ד ל׳ · אוּנְקְלוֹס

רָמָה ב׳ – שֹׁרֶשׁ וּמָחוֹל

🔗 נְפָשׁוֹת מִתְקַשְּׁרוֹת

הַפָּסוּק עַל יַעֲקֹב אָבִינוּ וּבִנְיָמִין אוֹמֵר: ׳וְנַפְשׁוֹ קְשׁוּרָה בְנַפְשׁוֹ׳. אוּנְקְלוֹס מְתַרְגֵּם: ׳וְנַפְשֵׁיהּ חֲבִיבָא לֵיהּ כְּנַפְשֵׁיהּ׳, כְּלוֹמַר נַפְשׁוֹ אֲהוּבָה עָלָיו כְּנַפְשׁוֹ.

מִכָּאן לָמַדְנוּ מָה קוֹרֶה כְּשֶׁאָדָם אוֹהֵב אֶת הַשֵּׁנִי: הַנְּפָשׁוֹת מִתְקַשְּׁרוֹת.

אַהֲבָה קוֹשֶׁרֶת נְפָשׁוֹת.

3
💃

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁהַכִּוּוּן שֶׁלִּי יִהְיֶה אֶל ה׳, יַחַד עִם כֻּלָּם 💛

📖 אַהֲבַת יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תַּעֲנִית ל״א א׳

רָמָה ב׳ – שֹׁרֶשׁ וּמָחוֹל

💃 הַמָּחוֹל וְהַחִצִּים

הַגְּמָרָא אוֹמֶרֶת שֶׁלֶּעָתִיד לָבוֹא הַקָּדוֹשׁ בָּרוּךְ הוּא עוֹשֶׂה מָחוֹל לַצַּדִּיקִים וְהוּא יוֹשֵׁב בָּאֶמְצַע, וְכֻלָּם מַצְבִּיעִים: ׳הִנֵּה אֱלֹקֵינוּ זֶה׳.

בְּעִגּוּל כָּל הַחִצִּים בְּאוֹתוֹ מֶרְחָק מֵהַמֶּרְכָּז, וְכָל עוֹד הֵם יְשָׁרִים אֶל הָאֶמְצַע, אֵין חֲשִׁיבוּת מֵאֵיזֶה כִּוּוּן בָּא כָּל אֶחָד. הַמַּטָּרָה שֶׁל כֻּלָּם אֶחָד.

כֻּלָּם מְכֻוָּנִים אֶל הַמֶּרְכָּז, מִכֹּל כִּוּוּן.

4
🌟

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד מֵה׳ לֶאֱהֹב כֻּלָּם 💛

📖 אַהֲבַת יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, פְּנִינִים בְּאַהֲבַת יִשְׂרָאֵל

רָמָה ב׳ – שֹׁרֶשׁ וּמָחוֹל

🌟 הַבַּעַל שֵׁם טוֹב וְהָאַהֲבָה

הַבַּעַל שֵׁם טוֹב הָיָה אוֹמֵר: ׳הַלְוַאי שֶׁאֶאֱהַב אֶת הַצַּדִּיק הַגָּדוֹל בְּיוֹתֵר כְּמוֹ שֶׁה׳ אוֹהֵב אֶת הָרָשָׁע׳.

הַמִּשְׁפָּט הַזֶּה מְלַמֵּד כַּמָּה גָּדוֹלָה אַהֲבַת ה׳ לְכָל יְהוּדִי, גַּם לְמִי שֶׁעוֹד לֹא תִּקֵּן אֶת דְּרָכָיו. אָנוּ מְבַקְּשִׁים לֶאֱהֹב קְצָת כְּמוֹ שֶׁה׳ אוֹהֵב.

ה׳ אוֹהֵב כָּל יְהוּדִי, אֲפִלּוּ מִי שֶׁעוֹד לֹא תִּקֵּן.

5
💛

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה אַהֲבָה שְׁלֵמָה, לַחֲבֵרִים וְלַה׳ 💛

📖 אַהֲבַת יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרַב דִּיסְקִין · כֵּיצַד נִקְנֵית מִדַּת אַהֲבַת יִשְׂרָאֵל

רָמָה ב׳ – שֹׁרֶשׁ וּמָחוֹל

💛 אֵין אַהֲבָה לַחֲצָאִין

רַבִּי עֲקִיבָא אָמַר שֶׁ׳וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ׳ זֶה כְּלָל גָּדוֹל. הָרַב דִּיסְקִין הִסְבִּיר: כְּשֶׁאָדָם אוֹהֵב אֶת חֲבֵרוֹ, זֶה גּוֹרֵם לוֹ לֶאֱהֹב אֶת ה׳, מִשּׁוּם שֶׁאֵין אַהֲבָה לַחֲצָאִין.

אַהֲבָה הִיא מִדָּה בַּנֶּפֶשׁ: לֹא שַׁיָּךְ לֶאֱהֹב אָדָם אֶחָד וְאֶת הַשֵּׁנִי לֹא. מִי שֶׁקּוֹנֶה מִדָּה זוֹ בִּכְלָלוּת, אוֹהֵב גַּם בְּבֵיתוֹ וְגַם אֶת ה׳.

אַהֲבַת חָבֵר מוֹבִילָה לְאַהֲבַת ה׳, כִּי הָאַהֲבָה שְׁלֵמָה.

6
💡

רָמָה ב׳ – שֹׁרֶשׁ וּמָחוֹל

מָה לוֹקְחִים מִרָמָה ב׳?

אַהֲבָה קוֹשֶׁרֶת נְפָשׁוֹת, וְאֵין אַהֲבָה לַחֲצָאִין. כְּשֶׁאֲנִי אוֹהֵב אֶת עַם ה׳, לִבִּי נִפְתָּח לֶאֱהֹב גַּם אֶת ה׳.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • אֵיזֶה חָבֵר קָשֶׁה לִי לֶאֱהֹב, וּמָה יֵשׁ בּוֹ שֶׁאֶפְשָׁר לִרְאוֹת כִּנְשָׁמָה יָקָרָה?
  • מָתַי הִרְגַּשְׁתִּי שֶׁהַלֵּב שֶׁלִּי מִתְקַשֵּׁר לְחָבֵר?
  • אֲצַיֵּר עִגּוּל שֶׁבְּמֶרְכָּזוֹ ה׳, וַאֲצַיֵּר חֵץ שֶׁלִּי, וְאֶכְתֹּב מֵאֵיזֶה כִּוּוּן אֲנִי בָּא.

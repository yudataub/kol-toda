רַבִּי קָלְמָן גּוֹלְדְשְׁמִידְט הָיָה תַּלְמִיד שֶׁל דַּרְכּוֹ שֶׁל רַבִּי נַחְמָן מִבְּרֶסְלֶב, שֶׁלִּמֵּד שֶׁשִּׂמְחָה נוֹתֶנֶת כֹּחַ. כְּשֶׁהַמַּצָּב הָיָה קָשֶׁה מְאוֹד, הוּא קִבֵּל הַחְלָטָה.

הוּא אָמַר לְעַצְמוֹ: ׳קָלְמָן, אִם אֲנִי אֶהְיֶה עָצוּב עַכְשָׁיו, אֲנִי רַק מַכְבִּיד עַל עַצְמִי. הַדֶּרֶךְ הִיא לַעֲשׂוֹת דַּוְקָא – לִצְחֹק.׳

הוּא לֹא צָחַק כִּי הָיָה לוֹ מַצְחִיק. הוּא צָחַק כִּי הוּא רָצָה לִחְיוֹת.

הַשִּׂמְחָה שֶׁל רַבִּי קָלְמָן לֹא בָּאָה מֵהַמַּצָּב אֶלָּא מֵהַחְלָטָה.

הַצְּחוֹק שֶׁל רַבִּי קָלְמָן לֹא הָיָה הַדְחָקָה. הוּא יָדַע הֵיטֵב מָה מַצָּבוֹ, וְלֹא עָשָׂה כְּאִלּוּ הַכֹּל קַל.

הוּא הֵבִין שֶׁהַפַּחַד הוּא אוֹיֵב, וּבָחַר לֹא לָתֵת לוֹ לִשְׁלֹט.

בְּרֵאיוֹן מְצֻלָּם אָמַר שֶׁהוּא לֹא מִתְפַּחֵד, כִּי הוּא סוֹמֵךְ עַל ה׳ שֶׁמַּעֲבִיר אוֹתָנוּ בְּשָׁלוֹם מֵעוֹלָם לְעוֹלָם.

שִׂמְחָה שֶׁל בְּחִירָה אֵינָהּ בְּרִיחָה מִן הַקֹּשִׁי, אֶלָּא עֲמִידָה מוּלוֹ.

רָמָה ב׳ – הַכֹּחַ שֶׁל רַבִּי קָלְמָן

📖 הֶסְכֵּם הַדַּף הַיּוֹמִי

אֵיךְ יוֹצְרִים בִּטָּחוֹן כְּשֶׁלֹּא יוֹדְעִים מָה יִהְיֶה? רַבִּי קָלְמָן סִפֵּר: ׳עָשִׂיתִי הֶסְכֵּם עִם ה׳. הִתְחַיַּבְתִּי לִלְמֹד אֶת הַדַּף הַיּוֹמִי, וְהַמַּחֲזוֹר נִמְשָׁךְ שֶׁבַע וָחֵצִי שָׁנִים. אָמַרְתִּי: אַבָּא, אַתָּה צָרִיךְ לָתֵת לִי כֹּחַ וְחַיִּים כְּדֵי לְסַיֵּם.׳

וּבְחִיּוּךְ שׁוֹבָב הוֹסִיף: ׳אָמַרְתִּי לַחֲבֵרִים – אַל תִּלְמְדוּ מַהֵר, שֶׁיִּקַּח לָנוּ אַרְבַּע עֶשְׂרֵה שָׁנָה!׳

אֶפְשָׁר לָשִׂים לְעַצְמִי מַטָּרָה קְטַנָּה וּלְבַקֵּשׁ מֵה׳ כֹּחַ לְהַשְׁלִים אוֹתָהּ.

🏃 רִיצָה פְּנִימִית

רַבִּי קָלְמָן קָרָא לַצְּחוֹק הָעָמֹק ׳רִיצָה פְּנִימִית׳: ׳אֲנִי לֹא יָכוֹל לָרוּץ בַּחוּץ, אָז אֲנִי מֵרִיץ אֶת הָאֵיבָרִים בִּפְנִים.׳

בְּצְחוֹק עָמֹק הַסַּרְעֹפֶת – הַשְּׁרִיר שֶׁמִּתַּחַת לָרֵאוֹת – עוֹלָה וְיוֹרֶדֶת, וְהָרֵאוֹת מִתְמַלְּאוֹת אֲוִיר.

וְהַמֹּחַ מְשַׁחְרֵר אֶנְדּוֹרְפִינִים, חוֹמְרִים שֶׁמְּקַלִּים עַל הַכְּאֵב. עַל זֶה כְּבָר לָמַדְנוּ בַּשִּׁעוּר הַשִּׁשִּׁי.

צְחוֹק עָמֹק מֵרִיץ אֶת הַגּוּף מִבִּפְנִים וּמְקַל עַל הַכְּאֵב.

🎭 מִטְרָגֶדְיָה לְקוֹמֶדְיָה

רַבִּי קָלְמָן לִמֵּד גַּם אֲנָשִׁים אֲחֵרִים לְהַסְתַּכֵּל עַל הַצָּרוֹת מִלְּמַעְלָה. הוּא הָיָה מַצְבִּיעַ עַל הָרֹאשׁ הַקֵּרֵחַ שֶׁלּוֹ וְאוֹמֵר: ׳תִּרְאוּ אֵיזֶה יֹפִי – חָסַכְתִּי כֶּסֶף עַל שַׁמְפּוֹ! אֲנִי קָם בַּבֹּקֶר וְהַשֵּׂעָר כְּבָר מְסֻדָּר!׳

כְּשֶׁאֲנִי צוֹחֵק עַל הַמַּצָּב, אֲנִי מִשְׁתַּחְרֵר מִמֶּנּוּ: אֲנִי כְּבָר לֹא ׳קָרְבָּן׳ שֶׁלּוֹ, אֶלָּא ׳צוֹפֶה׳ שֶׁמְּשֻׁעֲשָׁע מִמֶּנּוּ.

מִי שֶׁצּוֹחֵק עַל הַמַּצָּב, כְּבָר לֹא חַלָּשׁ מִפָּנָיו.

📜 אַל תְּחַכּוּ!

רַבִּי קָלְמָן הִשְׁאִיר מֶסֶר: ׳אֲנָשִׁים מְחַכִּים שֶׁיִּקְרֶה מַשֶּׁהוּ נוֹרָא כְּדֵי לְהִתְעוֹרֵר עַל הַחַיִּים שֶׁלָּהֶם. לָמָּה לְחַכּוֹת כְּדֵי לְהַתְחִיל לִשְׂמֹחַ?׳

הוּא אָמַר לָנוּ, שֶׁיֵּשׁ לָנוּ רַגְלַיִם שֶׁהוֹלְכוֹת וַאֲוִיר בָּרֵאוֹת, וְלָכֵן אֵין סִבָּה לִבְכּוֹת עַל פְּקָק אוֹ לִכְעֹס כְּשֶׁהַקָּפֶה נִשְׁפָּךְ.

הַחִיּוּךְ הוּא דֶּרֶךְ לוֹמַר לַבּוֹרֵא: ׳תּוֹדָה שֶׁנָּתַתָּ לִי עוֹד יוֹם׳.

תּוֹדָה עַל מַה שֶּׁיֵּשׁ הִיא סוֹד הַשִּׂמְחָה.

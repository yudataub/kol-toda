מָה לָמְדוּ חֲכָמִים מִן הַמִּלָּה ׳וַיִּקַּח׳, אֵיךְ הִתְאַמְּצוּת בְּמִצְוָה מְבִיאָה בְּרָכָה, וְלָמָּה זְרִיזוּת הִיא מִדָּה שֶׁל לֵב

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַשִׁ״י בְּרֵאשִׁית ט׳ כ״ג

רָמָה ב׳ – זְרִיזוּת שֶׁל גּוּף וְנֶפֶשׁ

📜 רַשִׁ״י: מִי הִתְאַמֵּץ יוֹתֵר?

בִּבְרֵאשִׁית ט׳ כ״ג כָּתוּב: ׳וַיִּקַּח שֵׁם וָיֶפֶת אֶת הַשִּׂמְלָה׳. רַשִׁ״י שָׂם לֵב שֶׁלֹּא נִכְתַּב ׳וַיִּקְחוּ׳ בִּלְשׁוֹן רַבִּים, אֶלָּא ׳וַיִּקַּח׳ בִּלְשׁוֹן יָחִיד.

מִכָּאן לָמַד: שֵׁם הִתְאַמֵּץ בַּמִּצְוָה יוֹתֵר מִיֶּפֶת. שְׁנֵיהֶם עָשׂוּ, אֲבָל לֹא בְּאוֹתָהּ מִדָּה שֶׁל מֶרֶץ.

לָכֵן זָכוּ בָּנָיו שֶׁל שֵׁם לְמִצְוַת צִיצִית, וְהַשְּׁכִינָה שָׁרְתָה בְּאָהֳלֵי שֵׁם.

מִי שֶׁמִּתְאַמֵּץ יוֹתֵר בְּמִצְוָה, זוֹכֶה גַּם לִבְרָכָה לְדוֹרוֹת.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁהַשְּׁכִינָה תִּהְיֶה קְרוֹבָה לְמַעֲשִׂים שֶׁלִּי 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, עֲלֵי שׁוּר, הָרַב וָולְבֶּה · שַׁבָּת כ״א

רָמָה ב׳ – זְרִיזוּת שֶׁל גּוּף וְנֶפֶשׁ

🔥 זְרִיזוּת – מִדַּת הַשְׁרָאַת הַשְּׁכִינָה

הָרַב וָולְבֶּה מְלַמֵּד: זְרִיזוּת הִיא הַמִּדָּה שֶׁל הַשְׁרָאַת הַשְּׁכִינָה בָּאָדָם. הַגְּמָרָא אוֹמֶרֶת עַל הַכֹּהֲנִים בְּבֵית הַמִּקְדָּשׁ: ׳זְרִיזִים הֵם׳.

לְפִי הָרַב יְרוֹחָם מִמִּיר, מִצְוָה שֶׁעוֹשִׂים בִּזְרִיזוּת הִיא מַעֲשֶׂה שֶׁהַשְּׁכִינָה שׁוֹרָה עָלָיו.

וְכָל עִכּוּב בַּזְּמַן מַרְחִיק אוֹתָהּ מֵהַמַּעֲשֶׂה.

זְרִיזוּת מְקָרֶבֶת אֶת הַשְּׁכִינָה, וְעִכּוּב מַרְחִיק אוֹתָהּ.

⚡

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת מִן הַמְמַהֲרִים לְמִצְוֹת 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁמוֹת י״ב י״א · מְכִילְתָּא · עֲלֵי שׁוּר

רָמָה ב׳ – זְרִיזוּת שֶׁל גּוּף וְנֶפֶשׁ

⚡ חִפָּזוֹן שֶׁל שְׁכִינָה

בְּלֵיל יְצִיאַת מִצְרַיִם צִוָּה ה׳ לֶאֱכֹל אֶת קָרְבַּן הַפֶּסַח ׳בְּחִפָּזוֹן׳. בַּמְּכִילְתָּא נֶאֱמַר שֶׁזֶּה ׳חִפָּזוֹן שְׁכִינָה׳: הַשְּׁכִינָה מְמַהֶרֶת בְּלִי שׁוּם עִכּוּב.

הִשְׁתַּהוּת וְעִכּוּבִים הֵם מִן הָרַע שֶׁבָּא לָעוֹלָם אַחֲרֵי הַחֵטְא.

מִי שֶׁמְּמַהֵר לַעֲשׂוֹת מִצְוָה הוֹלֵךְ בְּדַרְכָּהּ שֶׁל הַשְּׁכִינָה, וְזֶה מַה שֶּׁרָאִינוּ אֵצֶל שֵׁם.

חִפָּזוֹן שֶׁל שְׁכִינָה: הַשְּׁכִינָה מְמַהֶרֶת בְּלִי עִכּוּב.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִזְדָּרֵז וְגַם חוֹשֵׁב מָה הַכִּי טוֹב לַעֲשׂוֹת 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שַׁבָּת כ״א · עֲלֵי שׁוּר

רָמָה ב׳ – זְרִיזוּת שֶׁל גּוּף וְנֶפֶשׁ

🧠 זְרִיזוּת בָּרֹאשׁ

זְרִיזוּת אֵינָהּ רַק בָּרַגְלַיִם. הַכֹּהֲנִים הָיוּ זְרִיזִים, וּבְכָל זֹאת סִדְּרוּ אֶת הָעֲבוֹדָה כְּדֵי שֶׁלֹּא יִכָּשְׁלוּ.

זְרִיזוּת הִיא גַּם בָּרֹאשׁ: מִתְמַצְּאִים מַהֵר מָה לַעֲשׂוֹת, וְחוֹשְׁבִים תּוֹךְ כְּדֵי הַמַּעֲשֶׂה.

זֶה מַה שֶּׁמַּבְדִּיל זְרִיזוּת מִפְּזִיזוּת: מַאֲמָץ יַחַד עִם מַחֲשָׁבָה.

זְרִיזוּת הִיא מְהִירוּת עִם מַחֲשָׁבָה.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לַעֲשׂוֹת מִצְוָה בְּכָל הַגּוּף וּבְכָל הַנְּשָׁמָה 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, גּוּר אַרְיֵה בְּרֵאשִׁית ט׳

רָמָה ב׳ – זְרִיזוּת שֶׁל גּוּף וְנֶפֶשׁ

🪞 גּוּף וְנֶפֶשׁ יַחַד

הַמַּהֲרָ״ל בְּגוּר אַרְיֵה מְבָאֵר שֶׁשֵּׁם הִתְאַמֵּץ, וְלָכֵן גַּם גּוּפוֹ וְגַם נִשְׁמָתוֹ זָכוּ לְכָבוֹד. יֶפֶת לֹא הִתְאַמֵּץ וְלֹא כִּוֵּן בְּנַפְשׁוֹ, וְלָכֵן זָכָה רַק גּוּפוֹ.

כְּשֶׁעוֹשִׂים בִּזְרִיזוּת, הַנְּשָׁמָה עוֹשָׂה. כְּשֶׁלֹּא מִתְאַמְּצִים, הָעוֹשֶׂה הוּא רַק הַגּוּף.

מִצְוָה בְּמַאֲמָץ הִיא מִצְוָה שֶׁל כָּל הָאָדָם.

בְּמַאֲמָץ, הַגּוּף וְהַנְּשָׁמָה עוֹשִׂים יַחַד.

🏡

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִתְגַּבֵּר עַל הָעַצְלוּת, כִּי אֲנִי אוֹהֵב אֶת ה׳ 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אָבוֹת ב׳ ט״ו · דֶּרֶךְ חַיִּים

רָמָה ב׳ – זְרִיזוּת שֶׁל גּוּף וְנֶפֶשׁ

🏡 בַּעַל הַבַּיִת דּוֹחֵק

בְּאָבוֹת (ב׳ ט״ו): ׳הַיּוֹם קָצָר, וְהַמְּלָאכָה מְרֻבָּה, וְהַפּוֹעֲלִים עֲצֵלִים, וּבַעַל הַבַּיִת דּוֹחֵק׳. בַּעַל הַבַּיִת הוּא ה׳.

הַדֶּרֶךְ חַיִּים מְבָאֵר: ה׳ אֵינוֹ כְּמוֹ גּוּף, וְאֵין אֶצְלוֹ עִיֵּפוּת וְעִכּוּב. הָאָדָם הוּא בַּעַל גּוּף, וְיֵשׁ לוֹ נְטִיָּה לְעַצְלוּת.

לָכֵן צָרִיךְ לְהִתְאַמֵּץ יוֹתֵר, וְלֹא לְהוֹסִיף עַל הָעַצְלוּת.

ה׳ אֵינוֹ מִתְעַיֵּף, וְאָנוּ מִתְגַּבְּרִים עַל הָעַצְלוּת.

מָה לוֹקְחִים מִרָמָה ב׳?

שֵׁם הִתְאַמֵּץ, וְהַבְּרָכָה הִגִּיעָה עַד לַשְּׁכִינָה בְּאָהֳלָיו. כְּשֶׁאֲנִי מִתְאַמֵּץ בְּמִצְוָה בְּגוּף וּבְנֶפֶשׁ, אֲנִי אוֹמֵר לְה׳ שֶׁאֲנִי אוֹהֵב אוֹתוֹ וְרוֹצֶה לִהְיוֹת קָרוֹב אֵלָיו.

🤔 חוֹשְׁבִים עַל זֶה

  • אֵיזוֹ מִצְוָה אֲנִי יָכוֹל לַעֲשׂוֹת הַיּוֹם מִיָּד, בְּלִי לִדְחוֹת לְ׳אַחַר כָּךְ׳?
  • אֵיפֹה הִתְאַמַּצְתִּי הַשָּׁבוּעַ בֶּאֱמֶת, וְאֵיפֹה עָשִׂיתִי רַק כְּדֵי לָצֵאת יְדֵי חוֹבָה?
  • בַּשָּׁבוּעַ הַקָּרוֹב אֶבְחַר מִצְוָה אַחַת וְאֶעֱשֶׂה אוֹתָהּ ׳כְּמוֹ שֵׁם׳: מִיָּד, בְּמֶרֶץ וְעוֹד קְצָת יוֹתֵר – וְאֶרְשֹׁם מָה הִרְגַּשְׁתִּי.

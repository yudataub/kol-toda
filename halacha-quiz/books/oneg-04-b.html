זָכוֹר אֶת יוֹם הַשַּׁבָּת לְקַדְּשׁוֹ

🌱 שַׁמַּאי וְהַזִּכָּרוֹן · רָמָה ב׳

שַׁמַּאי – זוֹכְרִים אֶת הַשַּׁבָּת כָּל הַשָּׁבוּעַ

רָמָה ב׳ – זָכוֹר וּלְקַדְּשׁוֹ

זָכוֹר אֶת יוֹם הַשַּׁבָּת: לָמָּה קוֹרְאִים רִאשׁוֹן בְּשַׁבָּת, מָה עָשָׂה שַׁמַּאי הַזָּקֵן, וְאֵיךְ מַכְנִיסִים אֶת הַשַּׁבָּת לַלֵּב בְּכָל יוֹם

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַמְבַּ״ן עַל שְׁמוֹת כ׳ ח׳

רָמָה ב׳ – זָכוֹר וּלְקַדְּשׁוֹ

📖 חֲצִי פָּסוּק – שְׁתֵּי מִצְווֹת

חֲכָמִים לָמְדוּ שֶׁאֶפְשָׁר לְחַלֵּק אֶת הַפָּסוּק לִשְׁנַיִם: ׳זָכוֹר אֶת יוֹם הַשַּׁבָּת׳ – מִצְוָה לִזְכֹּר אוֹתוֹ מִתְּחִלַּת הַשָּׁבוּעַ, אֲפִלּוּ בְּדִבְרֵי חֹל. ׳לְקַדְּשׁוֹ׳ – מִצְוָה בַּשַּׁבָּת עַצְמָהּ, לְקַדֵּשׁ אֶת הַיּוֹם בִּדְבָרִים.

הַקִּדּוּשׁ בְּלֵיל שַׁבָּת הוּא הַקִּיּוּם שֶׁל ׳לְקַדְּשׁוֹ׳.

׳זָכוֹר׳ – כָּל הַשָּׁבוּעַ. ׳לְקַדְּשׁוֹ׳ – בַּשַּׁבָּת עַצְמָהּ.

❓ הַשְּׁאֵלָה עַל שַׁמַּאי

אוּלַי תִּשְׁאֲלוּ: שַׁמַּאי זָכַר אֶת הַשַּׁבָּת בְּמַאֲכָלָיו, וְהָאֲכִילָה הִיא דָּבָר שֶׁל חֹל. אֵיפֹה הַקְּדֻשָּׁה?

הַתְּשׁוּבָה: הַחֵלֶק הָרִאשׁוֹן שֶׁל הַפָּסוּק מְדַבֵּר עַל זִכָּרוֹן גַּם בְּדִבְרֵי חֹל. הַקְּדֻשָּׁה נַעֲשֵׂית בַּשַּׁבָּת עַצְמָהּ, וְהַחֹל מַכִּין אֶת הַלֵּב.

זִכָּרוֹן בַּחֹל, קְדֻשָּׁה בַּשַּׁבָּת.

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד מִכָּל הַחֲכָמִים 💛

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סֵפֶר הַחִנּוּךְ · רַשִׁ״י וְרַמְבַּ״ן

⚖️ דֵּעָה אַחֶרֶת

יֵשׁ דֵּעָה אַחֶרֶת: סֵפֶר הַחִנּוּךְ לוֹמֵד שֶׁמִּצְוַת ׳זָכוֹר׳ הִיא רַק בַּשַּׁבָּת עַצְמָהּ.

לְעֻמָּתוֹ, רַשִׁ״י עַל הַתּוֹרָה סָבוּר כְּמוֹ הָרַמְבַּ״ן: ׳תְּנוּ לֵב לִזְכֹּר תָּמִיד אֶת יוֹם הַשַּׁבָּת׳. כְּשֶׁיֵּשׁ מַחֲלֹקֶת בֵּין גְּדוֹלִים, לוֹמְדִים מִשְּׁנֵיהֶם: אֲנַחְנוּ בְּוַדַּאי זוֹכְרִים בַּשַּׁבָּת, וְזוֹכְרִים גַּם בַּחֹל.

יֵשׁ דֵּעוֹת: רַשִׁ״י וְרַמְבַּ״ן – כָּל הַשָּׁבוּעַ; הַחִנּוּךְ – בַּשַּׁבָּת.

📚

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹסִיף עוֹד קְצָת כָּבוֹד לַשַּׁבָּת בְּכָל שָׁבוּעַ 💛

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְסִלַּת יְשָׁרִים

רָמָה ב׳ – זָכוֹר וּלְקַדְּשׁוֹ

📚 כָּל מַה שֶׁמּוֹסִיף כָּבוֹד

מְסִלַּת יְשָׁרִים מְלַמֵּד: אַחֲרֵי שֶׁהַתּוֹרָה לִמְּדָה שֶׁרְצוֹנָהּ שֶׁנְּכַבֵּד אֶת הַשַּׁבָּת, כָּל מַה שֶׁאָדָם עוֹשֶׂה כְּדֵי לְהוֹסִיף כָּבוֹד – נִכְלָל בַּמִּצְוָה.

וְלָכֵן גַּם אֲכִילַת הַשְּׁאֵרִיּוֹת וְהַקְּנִיּוֹת בְּאֶמְצַע הַשָּׁבוּעַ הֵן חֵלֶק מִכְּבוֹד הַשַּׁבָּת.

כָּל תּוֹסֶפֶת כָּבוֹד נִכְלֶלֶת בַּמִּצְוָה.

🧀

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד לְחַבֵּר אֶת כָּל הַשָּׁבוּעַ אֶל הַשַּׁבָּת 💛

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בֵּיצָה ט״ז א׳ · חִדּוּשׁ בִּכְבוֹד שַׁבָּת

רָמָה ב׳ – זָכוֹר וּלְקַדְּשׁוֹ

🧀 שֶׁלֹּא יֵלֵךְ לָאִבּוּד

הַגְּמָרָא אוֹמֶרֶת: כְּדֵי לִקְנוֹת חָדָשׁ, צָרִיךְ לֶאֱכֹל קֹדֶם אֶת הַיָּשָׁן, כְּדֵי שֶׁלֹּא יִהְיֶה בִּזְבּוּז. לָכֵן אֲכִילַת הַשְּׁאֵרִיּוֹת הִיא כְּבָר הֲכָנָה לַשַּׁבָּת.

זֶה חִדּוּשׁ יָפֶה: אֲפִלּוּ אֲרוּחָה שֶׁל יוֹם חֹל נֶחְשֶׁבֶת לִכְבוֹד שַׁבָּת, כְּשֶׁחוֹשְׁבִים עַל כָּךְ.

אֲכִילָה בַּחֹל לְצֹרֶךְ שַׁבָּת – גַּם הִיא לִכְבוֹד שַׁבָּת.

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, זָכוֹר אֶת יוֹם הַשַּׁבָּת לְקַדְּשׁוֹ

✨ מָה זֶה קָדוֹשׁ?

׳קָדוֹשׁ׳ פֵּרוּשׁוֹ מֻפְרָשׁ וּמֻבְדָּל. הַשַּׁבָּת מֻפְרֶשֶׁת מִשְּׁאָר הַיָּמִים כִּי אֵין עוֹשִׂים בָּהּ מְלָאכָה.

לָכֵן ׳לְקַדְּשׁוֹ׳ פֵּרוּשׁוֹ לוֹמַר דְּבָרִים שֶׁמַּזְכִּירִים לָנוּ שֶׁיּוֹם זֶה שׁוֹנֶה וִיקָר מִשְּׁאָר הַיָּמִים.

וְהַחִנּוּךְ מוֹסִיף: הַיּוֹם הַזֶּה גָּדוֹל מִשְּׁאָר הַיָּמִים.

׳קָדוֹשׁ׳ = מֻפְרָשׁ וּמֻבְדָּל מִשְּׁאָר הַיָּמִים.

מָה לוֹקְחִים מִרָמָה ב׳?

הַזִּכָּרוֹן בַּחֹל וְהַקְּדֻשָּׁה בַּשַּׁבָּת מִתְחַבְּרִים לְשָׁבוּעַ אֶחָד שֶׁל קֶשֶׁר עִם ה׳. מִי שֶׁמּוֹסִיף כָּבוֹד בְּכָל יוֹם, אוֹהֵב יוֹתֵר אֶת הַשַּׁבָּת וְאֶת מִי שֶׁנָּתַן אוֹתָהּ.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה אֲנִי יָכוֹל לַעֲשׂוֹת בְּיוֹם שֵׁנִי כְּדֵי לִזְכֹּר אֶת הַשַּׁבָּת?
  • אֵיזֶה דָּבָר יָפֶה אֶשְׁמֹר לַשַּׁבָּת בִּלְבַד, וְאֵיךְ אַרְגִּישׁ כְּשֶׁאוֹצִיא אוֹתוֹ?
  • אֶבְחַר דָּבָר יָפֶה אֶחָד לְ׳קוּפְסַת שַׁבָּת׳, אֶשְׁמֹר אוֹתוֹ עַד הַשַּׁבָּת, וְאֶרְשֹׁם מָה בָּחַרְתִּי וְאֵיךְ הִרְגַּשְׁתִּי.

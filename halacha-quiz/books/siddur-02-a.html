🌅 בֹּקֶר חָדָשׁ – בִּרְכוֹת הַשַּׁחַר

כְּשֶׁקָּמִים בַּבֹּקֶר, מְבָרְכִים אֶת בִּרְכוֹת הַשַּׁחַר. הַסֵּפֶר ׳סֵדֶר הַיּוֹם׳ מְלַמֵּד: מְשַׁבְּחִים אֶת ה׳ שֶׁהֶחֱזִיר לָנוּ אֶת הַנְּשָׁמָה בְּרַחֲמִים וְנָתַן לָנוּ שֵׂכֶל לְהָבִין.

וְהַדָּבָר הֶחָשׁוּב בְּיוֹתֵר: עִקַּר הַבְּרָכָה הוּא כַּוָּנַת הַלֵּב, לֹא רַק הַמִּלִּים שֶׁהַפֶּה אוֹמֵר.

לָכֵן אוֹמְרִים כָּל בְּרָכָה לְאַט, מִלָּה אַחַר מִלָּה, וְחוֹשְׁבִים עַל מָה מוֹדִים.

עִקַּר הַבְּרָכָה הוּא כַּוָּנַת הַלֵּב – לֹא רַק הַמִּלִּים.

הַבְּרָכָה הָרִאשׁוֹנָה: ׳אֲשֶׁר נָתַן לַשֶּׂכְוִי בִינָה לְהַבְחִין בֵּין יוֹם וּבֵין לָיְלָה׳. הַשֶּׂכְוִי הוּא הַתַּרְנְגוֹל שֶׁקּוֹרֵא כְּשֶׁהַבֹּקֶר מַגִּיעַ.

ה׳ נָתַן גַּם לָנוּ שֵׂכֶל – לְהַבְחִין בֵּין טוֹב לְרַע. לָכֵן מְבַקְּשִׁים: כְּמוֹ שֶׁנָּתַן לַשֶּׂכְוִי בִּינָה, יָכִין אֶת הַשֵּׂכֶל שֶׁלָּנוּ לִבְחֹר בְּכָל דָּבָר טוֹב, וְנִזְכֶּה לְקַיֵּם ׳וּבָחַרְתָּ בַּחַיִּים׳.

הַשֵּׂכֶל שֶׁנָּתַן לָנוּ ה׳ עוֹזֵר לָנוּ לִבְחֹר בַּטּוֹב.

👀 פּוֹקֵחַ עִוְרִים

׳פּוֹקֵחַ עִוְרִים׳ – ה׳ פּוֹקֵחַ לָנוּ אֶת הָעֵינַיִם בַּבֹּקֶר, וַאֲנַחְנוּ רוֹאִים אֶת הַשֶּׁמֶשׁ, אֶת הַבַּיִת וְאֶת הַפָּנִים שֶׁל מִי שֶׁאוֹהֲבִים.

אֲבָל יֵשׁ שְׁנֵי סוּגִים שֶׁל עֵינַיִם: עֵינֵי הָרְאִיָּה וְעֵינֵי הַשֵּׂכֶל. מְבַקְּשִׁים שֶׁה׳ יִפְקַח לָנוּ אֶת שְׁתֵּיהֶן, כְּדֵי שֶׁנִּרְאֶה אֶת כָּל מָה שֶׁטּוֹב וְנָכוֹן לָנוּ, וְנַעֲבִיר אֶת הָעַיִן מִדְּבָרִים רָעִים.

ה׳ פּוֹקֵחַ גַּם אֶת עֵינֵי הָרְאִיָּה וְגַם אֶת עֵינֵי הַשֵּׂכֶל.

ה׳ מַתִּיר אוֹתָנוּ מִן הַמִּטָּה, וְגַם מִן הַיֵּצֶר הָרָע.

׳מַלְבִּישׁ עֲרֻמִּים׳ – ה׳ נוֹתֵן לָנוּ בְּגָדִים. אֲבָל לְפִי ׳סֵדֶר הַיּוֹם׳ אֲנַחְנוּ מְבַקְּשִׁים גַּם בִּגְדֵי אֱמוּנָה: שֶׁלֹּא נַעֲזֹב אֶת הָאֱמוּנָה בַּתּוֹרָה כָּל יְמֵי חַיֵּינוּ.

וּמְבַקְּשִׁים שֶׁנִּהְיֶה נְקִיִּים מִמַּחֲשָׁבוֹת רָעוֹת, כְּדֵי שֶׁנֵּלֵךְ לִפְנֵי ה׳ בִּלְבוּשִׁים נָאִים וּמְתֻקָּנִים.

מַלְבִּישׁ עֲרֻמִּים – בְּגָדִים לַגּוּף, וֶאֱמוּנָה לַנְּשָׁמָה.

מִי שֶׁלּוֹמֵד תּוֹרָה וּמְקַיֵּם מִצְווֹת – עוֹמֵד זָקוּף וְגָאֶה.

רָמָה א׳ – אוֹמְרִים תּוֹדָה בַּבֹּקֶר

מָה לוֹקְחִים מִרָמָה א׳?

כָּל בֹּקֶר ה׳ מַחֲזִיר לָנוּ אֶת הַנְּשָׁמָה, פּוֹקֵחַ אֶת הָעֵינַיִם, מַלְבִּישׁ וּמְחַזֵּק, וְנוֹתֵן לָנוּ יוֹם חָדָשׁ. מִי שֶׁמְּבָרֵךְ אֶת בִּרְכוֹת הַשַּׁחַר בְּלֵב שָׁלֵם מַתְחִיל אֶת הַיּוֹם בְּאַהֲבָה לְאַבָּא שֶׁבַּשָּׁמַיִם, שֶׁדָּאַג לְכָל צָרְכֵּנוּ.

חֲדָשִׁים לַבְּקָרִים רַבָּה אֱמוּנָתֶךָ

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • אֵיזוֹ מִבִּרְכוֹת הַשַּׁחַר נוֹגַעַת לִי הַכִּי הַרְבֵּה הַיּוֹם – עֵינַיִם, בְּגָדִים, צְעָדִים אוֹ כֹּחַ – וְעַל מָה אֲנִי רוֹצֶה לוֹמַר תּוֹדָה?
  • מָה הַהֶבְדֵּל בֵּין לוֹמַר בְּרָכָה בַּפֶּה לְבֵין לְבָרֵךְ מִן הַלֵּב, וְאֵיךְ אֲכַוֵּן בְּרָכָה אַחַת בַּבֹּקֶר?
  • מָה הָיָה קָשֶׁה לִי הַשָּׁבוּעַ בַּבֹּקֶר, וְאֵיךְ אֶזְכֹּר שֶׁה׳ מְחַדֵּשׁ לִי אֶת הַכֹּחַ בְּכָל יוֹם?

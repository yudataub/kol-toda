מִשְׁפָּחוֹת רַבּוֹת מִוַּרְשָׁה קָנוּ אֲדָמָה בַּכְּפָר – כִּשְׁלֹשׁ מֵאוֹת מִשְׁפָּחוֹת. אֲבָל רֹבָּן הַגָּדוֹל נִשְׁאֲרוּ בְּוַרְשָׁה.

הַחֲלוּצִים הַבּוֹדְדִים שֶׁהִגִּיעוּ מָצְאוּ חֻרְבָּה וּשְׁמָמָה, בִּצּוֹת וְשׁוּם תַּשְׁתִּית. הֵם הָיוּ מְעַטִּים, רְחוֹקִים זֶה מִזֶּה – אֲבָל הֵם נֶאֶחֲזוּ בָּאֲדָמָה וְלֹא הִרְפּוּ.

מִשְּׁלֹשׁ מֵאוֹת מִשְׁפָּחוֹת רַק מְעַטִּים בָּאוּ – וְהֵם נִשְׁאֲרוּ.

רָמָה א׳ – מִי הָיוּ הַחֲלוּצִים

🏠 אַרְבַּע הַמִּשְׁפָּחוֹת הָרִאשׁוֹנוֹת

בִּשְׁנַת תרפ״ה עָלוּ עַל הָאֲדָמָה אַרְבַּע מִשְׁפָּחוֹת: טַבְצֶ׳נִיק, פִּישֶׁר, אַבְּרַמְסְקִי וְלַנְדָּאוּ.

טַבְצֶ׳נִיק, שֶׁבָּא מִלִּיטָא, הֵקִים בֵּית כְּנֶסֶת בְּנֹסַח אַשְׁכְּנַז בִּרְחוֹב שָׁלוֹם עֲלֵיכֶם. פִּישֶׁר הָיָה חָסִיד רַדְזִין וְהָלַךְ עִם צִיצִית תְּכֵלֶת; לֹא הָיוּ לוֹ יְלָדִים, וְהוּא תָּרַם אֶת בֵּיתוֹ לְמוֹסְדוֹת לִמּוּד תּוֹרָה. אַבְּרַמְסְקִי הָיָה חָסִיד סָדִיגוֹרָה.

אַרְבַּע מִשְׁפָּחוֹת – טַבְצֶ׳נִיק, פִּישֶׁר, אַבְּרַמְסְקִי וְלַנְדָּאוּ – הָיוּ הַתּוֹשָׁבִים הָרִאשׁוֹנִים.

🦟 בִּצָּה וְקַדַּחַת

הַדֶּרֶךְ לְחֵיפָה הָיְתָה חֲסוּמָה בְּבִצָּה גְּדוֹלָה. הָיוּ קַדַּחַת וּמָלַרְיָה שֶׁהִפִּילוּ חֲלָלִים, וְהַפַּרְנָסָה הָיְתָה בְּדֹחַק. עַד שְׁנַת תרצ״ד בְּעֵרֶךְ לֹא הָיָה חַשְׁמַל בַּכְּפָר.

הַמִּתְיַשְּׁבִים קָבְרוּ אֶת יְקִירֵיהֶם בָּאֲדָמָה, וְאָמְרוּ: לֹא נִכָּנַע. נַתְחִיל שׁוּב, בְּאוֹתוֹ מָקוֹם. הַדַּף קוֹרֵא לָזֶה ׳רָצוֹן בַּרְזֶל׳.

בִּצָּה, קַדַּחַת וְדֹחַק – וְהֵם לֹא וִתְּרוּ.

💧 מַיִם בְּמִשּׁוּרָה

בַּהַתְחָלָה לֹא הָיָה צִנּוֹר וְלֹא בָּרֶז. הָעַרְבִים הֵבִיאוּ מַיִם בִּכְלֵי עַל חֲמוֹרִים וּמָכְרוּ לַמִּתְיַשְּׁבִים בְּמִשּׁוּרָה.

וְאֶת הַלֶּחֶם אָפוּ פַּעַם בַּשָּׁבוּעַ, בְּיוֹם חֲמִישִׁי, בְּתַנּוּרֵי אֶבֶן עֲרָבִיִּים. הַיּוֹם אֲנַחְנוּ פּוֹתְחִים בָּרֶז – וְהַכֹּל מוּכָן.

מַיִם נִמְכְּרוּ בְּמִשּׁוּרָה, וְלֶחֶם אָפוּ פַּעַם בְּשָׁבוּעַ.

🛳️ שְׁמוֹנָה עָשָׂר יוֹם בַּיָּם

ר׳ מֹשֶׁה מָרְדְּכַי בּוֹיִם סִפֵּר: עֲשָׂרָה אַבְרֵכִים צְעִירִים הִגִּיעוּ בְּסוֹף הַקַּיִץ שֶׁל תרפ״ה, אַחֲרֵי שְׁמוֹנָה עָשָׂר יוֹם בְּאָנִיָּה רוֹמָנִית יְשָׁנָה, בֵּין סְעָרוֹת.

הֵם הִגִּיעוּ לִכְפָר נִדָּח, וְהַתְּנָאִים הָיוּ קָשִׁים אֲפִלּוּ לְפִי הַמּוּשָׂגִים שֶׁל מִזְרַח אֵירוֹפָּה.

אַחֲרֵי שְׁמוֹנָה עָשָׂר יוֹם בַּיָּם הִגִּיעוּ אֶל כְּפָר נִדָּח.

🐄 הָאִכָּר מִסְּטָאשׁוֹב

מֹשֶׁה מָרְדְּכַי בּוֹיִם בָּא מֵהָעֲיָרָה סְטָאשׁוֹב. הָרַב שֶׁלָּהּ דִּבֵּר בִּלְהָבָה עַל בִּנְיַן הָאָרֶץ, וַאֲנָשִׁים רַבִּים עָלוּ. בּוֹיִם הָיָה שׁוֹנֶה: הוּא נַעֲשָׂה אִכָּר.

בִּשְׁנַת תרפ״ז הֵקִים רֶפֶת, וְנָסַע לַכְּפָר הַבֵּדוּאִי סַאסָא לִקְנוֹת פָּרוֹת. פָּרָה נָתְנָה אָז 600 עַד 1,000 לִיטְרִים חָלָב בַּשָּׁנָה; הַיּוֹם פָּרָה בְּאֶרֶץ יִשְׂרָאֵל נוֹתֶנֶת 10,600.

בּוֹיִם בָּחַר לִהְיוֹת אִכָּר, וּמִשְׁפַּחְתּוֹ עוֹסֶקֶת בְּחַקְלָאוּת עַד הַיּוֹם.

⛏️ אֲנִי אֶחְפֹּר בְּאֵר מִשֶּׁלִּי

בּוֹיִם בִּקֵּשׁ מִשָּׁכֵן לִשְׁאֹב מַיִם מֵהַבְּאֵר שֶׁלּוֹ, וְהוּא סֵרֵב. אָמַר בּוֹיִם: אֲנִי אֶחְפֹּר בְּאֵר מִשֶּׁלִּי.

הוּא הִזְמִין אַנְגְּלִי שֶׁהִכָּה בְּמַקְלוֹ עַל הַקַּרְקַע וְסִמֵּן אֶת הַמָּקוֹם. אַחֲרֵי חֲמִשָּׁה מֶטְרִים הִגִּיעוּ לְסֶלַע, וְהִמְשִׁיכוּ עִם אִזְמֵל שֶׁשָּׁקַל חֲצִי טוֹן. מַיִם נִמְצְאוּ בְּעֹמֶק 12–13 מֶטֶר – וְהֵם חָפְרוּ עַד 28 מֶטֶר.

מִי שֶׁלֹּא נָתְנוּ לוֹ מַיִם – חָפַר בְּאֵר מִשֶּׁלּוֹ, עַד 28 מֶטֶר.

מָה לוֹקְחִים מִרָמָה א׳?

הַכְּפָר לֹא קָם מֵעַצְמוֹ. מִשְּׁלֹשׁ מֵאוֹת מִשְׁפָּחוֹת נִשְׁאֲרוּ מְעַטִּים, וּבִזְכוּתָם יֵשׁ לָנוּ הַיּוֹם עִיר. זֶה ׳הֶסְתֵּר י-הּ׳ – הַיָּד שֶׁל ה׳ שֶׁמּוֹבִילָה אוֹתָנוּ גַּם כְּשֶׁלֹּא רוֹאִים. וּכְשֶׁמַּבִּינִים כַּמָּה אָהַב אוֹתָנוּ ה׳ וְדָאַג לָנוּ דֹּר אַחַר דּוֹר, הַלֵּב מִתְמַלֵּא בְּאַהֲבָה וּבְתוֹדָה.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָה אֲנִי מְקַבֵּל הַיּוֹם בְּלִי מַאֲמָץ – מַיִם, אוֹר, כְּבִישׁ, בֵּית כְּנֶסֶת – וּמִי הִתְאַמֵּץ כְּדֵי שֶׁיִּהְיֶה כָּאן?
  • מָה הָיָה ׳מָרוֹר׳ שֶׁעָבַרְתִּי לָאַחֲרוֹנָה, וְאֵיךְ אֲנִי יָכוֹל לְהוֹדוֹת עָלָיו כְּמוֹ שֶׁמּוֹדִים עַל הַמָּרוֹר בְּלֵיל הַסֵּדֶר?
  • הַשָּׁבוּעַ אֲנִי שׁוֹאֵל סַבָּא, סָבְתָּא אוֹ שָׁכֵן וָתִיק: מָה הָיָה פֹּה כְּשֶׁהָיִיתָ יֶלֶד? וְאוֹמֵר לוֹ תּוֹדָה.

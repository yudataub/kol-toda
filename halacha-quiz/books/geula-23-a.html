שׁוֹמֵר הַשַּׁלְשֶׁלֶת: בֵּין שְׁנֵי הָרִים, בָּנַי רוּצוּ, וְאֶבֶן שֶׁאֵין לָהּ הוֹפְכִין

📖 הַגְּאֻלָּה · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – שׁוֹמֵר הַשַּׁלְשֶׁלֶת

🏃 מוֹסְרִים אֶת הַמַּקֵּל

בְּמֵרוֹץ שְׁלִיחִים, כָּל רָץ מוֹסֵר אֶת הַמַּקֵּל לָרָץ הַבָּא.

אִם רָץ אֶחָד עוֹצֵר וְלֹא מוֹסֵר – הַשַּׁלְשֶׁלֶת נִקְטֶעֶת.

גַּם בַּתּוֹרָה יֵשׁ מַקֵּל שֶׁעוֹבֵר מִיָּד לְיָד, וְיֵשׁ מִי שֶׁשָּׁמַר שֶׁהוּא לֹא יִפֹּל.

מוֹסְרִים הָלְאָה, וְלֹא מַפִּילִים אֶת הַמַּקֵּל.

1

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ לִהְיוֹת חֵלֶק מִן הַשַּׁלְשֶׁלֶת שֶׁל תּוֹרַת ה׳ 💛

📖 הַגְּאֻלָּה, מִשִּׁעוּרֵי הַגְּאֻלָּה, בְּמִדְבַּר כ״ז כ״ג

רָמָה א׳ – שׁוֹמֵר הַשַּׁלְשֶׁלֶת

✋ מָה זֶה סְמִיכָה

סְמִיכָה – הָרַב מַסְמִיךְ אֶת תַּלְמִידוֹ לִהְיוֹת דַּיָּן, אִישׁ מִפִּי אִישׁ עַד מֹשֶׁה.

כְּבָר מֹשֶׁה סָמַךְ אֶת יְהוֹשֻׁעַ: ׳וַיִּסְמֹךְ אֶת יָדָיו עָלָיו וַיְצַוֵּהוּ׳.

כָּךְ עָבְרָה הַסְּמִיכָה מִדּוֹר לְדוֹר, וְכָל דַּיָּן הִגִּיעַ אֶל הַשַּׁלְשֶׁלֶת הַזּוֹ.

הַסְּמִיכָה עוֹבֶרֶת מֵאִישׁ לְאִישׁ, מִמֹּשֶׁה וְעַד הַדּוֹר שֶׁלָּנוּ.

2

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ שֶׁהָיָה מִי שֶׁלֹּא וִתֵּר עַל הַתּוֹרָה 💛

📖 הַגְּאֻלָּה, סַנְהֶדְרִין י״ד ע״א

רָמָה א׳ – שׁוֹמֵר הַשַּׁלְשֶׁלֶת

📜 הַגְּזֵרָה

בַּגְּמָרָא כָּתוּב שֶׁמַּלְכוּת רוֹמִי גָּזְרָה גְּזֵרָה קָשָׁה: כָּל מִי שֶׁסּוֹמֵךְ אוֹ נִסְמָךְ יִהְיֶה בְּסַכָּנָה, וְעִיר שֶׁסּוֹמְכִים בָּהּ תֵּחָרֵב.

מִי שֶׁרָצָה לִסְמֹךְ סִכֵּן אֶת כָּל הַסּוֹבְבִים אוֹתוֹ.

הָרוֹמָאִים רָצוּ שֶׁהַשַּׁלְשֶׁלֶת תִּפָּסֵק.

הַגְּזֵרָה הִפְחִידָה אֶת כֻּלָּם, כִּי הִיא פָּגְעָה בָּעִיר כֻּלָּהּ.

3

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ שֶׁיֵּשׁ אֲנָשִׁים שֶׁדּוֹאֲגִים לַאֲחֵרִים 💛

📖 הַגְּאֻלָּה, סַנְהֶדְרִין י״ד ע״א

רָמָה א׳ – שׁוֹמֵר הַשַּׁלְשֶׁלֶת

⛰️ בֵּין שְׁנֵי הָרִים

רַבִּי יְהוּדָה בֶּן בָּבָא מָצָא פִּתְרוֹן. הוּא הָלַךְ וְיָשַׁב בֵּין שְׁנֵי הָרִים גְּדוֹלִים, בֵּין אוּשָׁא לִשְׁפַרְעָם.

זֶה לֹא בְּתוֹךְ עִיר, וְלֹא בְּתוֹךְ הַתְּחוּם שֶׁלָּהּ.

כָּךְ הַגְּזֵרָה לֹא תִּפְגַּע בְּאִישׁ מִלְּבַדּוֹ.

יָשַׁב בֵּין הֶהָרִים, כְּדֵי שֶׁלֹּא לְסַכֵּן עִיר.

4

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ שֶׁהַתּוֹרָה יָכְלָה לְהַמְשִׁיךְ לְכָל הַדּוֹרוֹת 💛

📖 הַגְּאֻלָּה, סַנְהֶדְרִין י״ד ע״א

רָמָה א׳ – שׁוֹמֵר הַשַּׁלְשֶׁלֶת

👥 חֲמִשָּׁה זְקֵנִים

שָׁם סָמַךְ חֲמִשָּׁה זְקֵנִים: רַבִּי מֵאִיר, רַבִּי יְהוּדָה, רַבִּי שִׁמְעוֹן, רַבִּי יוֹסֵי וְרַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ.

מֵהֶם נִמְשְׁכָה הַתּוֹרָה לְכָל הַדּוֹרוֹת.

הַשַּׁלְשֶׁלֶת כְּבָר לֹא נִקְטְעָה.

חֲמִשָּׁה זְקֵנִים נִסְמְכוּ, וְהַשַּׁלְשֶׁלֶת נִמְשְׁכָה.

5

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ שֶׁיֵּשׁ מִי שֶׁדּוֹאֵג לְכָךְ שֶׁהַתּוֹרָה תִּמָּשֵׁךְ 💛

📖 הַגְּאֻלָּה, סַנְהֶדְרִין י״ד ע״א, עֲבוֹדָה זָרָה ח׳ ע״ב

🏃 בָּנַי, רוּצוּ!

כְּשֶׁהָרוֹמָאִים הִכִּירוּ בָּהֶם, רַבִּי יְהוּדָה בֶּן בָּבָא אָמַר לַתַּלְמִידִים: ׳בָּנַי, רוּצוּ!׳.

הֵם שָׁאֲלוּ: ׳רַבִּי, מָה יִהְיֶה עָלֶיךָ?׳

הוּא עָנָה: ׳הֲרֵינִי מֻטָּל לִפְנֵיהֶם כְּאֶבֶן שֶׁאֵין לָהּ הוֹפְכִין׳.

זָקֵן אֶחָד נִשְׁאָר כָּאֶבֶן, וְהַתַּלְמִידִים רָצוּ.

6

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת כְּמוֹ מִי שֶׁשּׁוֹמֵר עַל מָה שֶׁחָשׁוּב, וְאוֹהֵב אֶת ה׳ 💛

📖 הַגְּאֻלָּה, סַנְהֶדְרִין י״ג ע״ב

🌟 זָכוּר לְטוֹב

הָרוֹמָאִים לֹא עָזְבוּ אוֹתוֹ, וְהוּא מָסַר אֶת נַפְשׁוֹ עַל הַסְּמִיכָה.

בִּגְלַל זֶה אוֹמֵר רַב יְהוּדָה בְּשֵׁם רַב: ׳בְּרַם זָכוּר אוֹתוֹ הָאִישׁ לַטּוֹב, וְרַבִּי יְהוּדָה בֶּן בָּבָא שְׁמוֹ׳.

בִּזְכוּתוֹ לֹא נִשְׁתַּכְּחוּ דִּינֵי קְנָסוֹת מִיִּשְׂרָאֵל.

זָכוּר לְטוֹב: הוּא שָׁמַר עַל הַשַּׁלְשֶׁלֶת.

🗺️ הַדֶּרֶךְ הַבַּיְתָה

זוֹ הַתַּחֲנָה ה-23 בַּדֶּרֶךְ.

בַּתַּחֲנָה הַקּוֹדֶמֶת רָאִינוּ אֵיךְ שָׁאֲלוּ שְׁאֵלָה מִבֵּית דִּין לְבֵית דִּין, עַד בֵּית הַדִּין הַגָּדוֹל.

עַכְשָׁיו רוֹאִים מִי שָׁמַר עַל הַשַּׁלְשֶׁלֶת שֶׁלּוֹ.

בַּתַּחֲנָה הַזּוֹ: מִי שָׁמַר עַל הַשַּׁלְשֶׁלֶת.

מָה לוֹקְחִים מִרָמָה א׳?

ה׳ נָתַן לָנוּ תּוֹרָה שֶׁעוֹבֶרֶת מִיָּד לְיָד, וְשָׁמַר עָלֶיהָ בִּזְכוּת אֲנָשִׁים אַמִּיצִים. כְּשֶׁאֲנִי מוֹסֵר דָּבָר טוֹב הָלְאָה, אֲנִי מַרְגִּישׁ אֶת אַהֲבַת ה׳.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה לִמֵּד אוֹתִי מִישֶׁהוּ, וּמִי יָכוֹל לִלְמֹד מִמֶּנִּי?
  • מָתַי הָיָה לִי אֹמֶץ לַעֲמֹד עַל מָה שֶׁחָשׁוּב, גַּם כְּשֶׁזֶּה הָיָה קָשֶׁה?
  • הַמְּשִׂימָה: לִבְחֹר דָּבָר טוֹב אֶחָד, לְלַמֵּד אוֹתוֹ לְמִישֶׁהוּ, וְלוֹמַר תּוֹדָה.

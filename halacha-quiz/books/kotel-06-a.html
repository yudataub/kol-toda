😔 שָׁנִים בְּלִי הַכֹּתֶל

הַרְבֵּה שָׁנִים הָיָה הַכֹּתֶל הַמַּעֲרָבִי רָחוֹק מִמֶּנּוּ. תֵּשַׁע עֶשְׂרֵה שָׁנִים יְהוּדִים לֹא יָכְלוּ לָבוֹא לְשָׁם וּלְהִתְפַּלֵּל.

הָיוּ רַבִּים שֶׁהִתְגַּעְגְּעוּ וְהִסְתַּכְּלוּ מֵרָחוֹק.

תֵּשַׁע עֶשְׂרֵה שָׁנִים לֹא יָכְלוּ יְהוּדִים לָבוֹא לַכֹּתֶל.

🪑 יוֹשְׁבִים וּבוֹכִים

הָאַדְמוֹ״ר הַנְּתִיבוֹת שָׁלוֹם סִפֵּר: כַּאֲשֶׁר הָעִיר הָעַתִּיקָה נִסְגְּרָה, הִתְאַסְּפוּ כְּאַלְפַּיִם אֲנָשִׁים בְּהֵיכַל יְשִׁיבַת מֵאָה שְׁעָרִים.

הֵם יָשְׁבוּ עַל הָאָרֶץ וּבָכוּ.

הוּא חָשַׁב אָז עַל הַפָּסוּק: ׳וַאֲנִי אָמַרְתִּי נִגְרַשְׁתִּי מִנֶּגֶד עֵינֶיךָ, אַךְ אוֹסִיף לְהַבִּיט אֶל הֵיכַל קָדְשֶׁךָ׳.

אֲלָפִים יָשְׁבוּ עַל הָאָרֶץ וּבָכוּ עַל הַכֹּתֶל.

רָמָה א׳ – הַכֹּתֶל חָזַר

🎉 הַכֹּתֶל חָזַר!

בְּמִלְחֶמֶת שֵׁשֶׁת הַיָּמִים, בְּיוֹם כ״ח בְּאִיָּר תשכ״ז, שֻׁחְרַר הַכֹּתֶל, וְיְהוּדִים יָכְלוּ לָבוֹא אֵלָיו שׁוּב.

הַשִּׂמְחָה הָיְתָה גְּדוֹלָה בְּכָל הָעָם.

אֲנִי שָׂמֵחַ לְסַפֵּר עַל הַשִּׂמְחָה, לֹא עַל הַמִּלְחָמָה.

בְּכ״ח בְּאִיָּר שֻׁחְרַר הַכֹּתֶל וְהָיְתָה שִׂמְחָה גְּדוֹלָה.

💃

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ לְהַרְאוֹת שִׂמְחָה אֲמִתִּית 💛

📖 כֹּתֶל הַהוֹדָיוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אָז נִדְבְּרוּ · יְשַׁעְיָהוּ נ״ו ז׳

רָמָה א׳ – הַכֹּתֶל חָזַר

💃 בּוֹכֶה וְרוֹקֵד

הָאַדְמוֹ״ר מִוִּיז׳נִיץ שָׁמַע שֶׁהַכֹּתֶל שֻׁחְרַר, וְהִתְרַגֵּשׁ מְאֹד. הוּא בָּכָה וְאַחַר כָּךְ רָקַד.

לְכָל מִי שֶׁנִּכְנַס לַחֶדֶר הוּא אָמַר: ׳הַכֹּתֶל בְּיָדֵינוּ!׳

הוּא שָׁר אֶת הַפָּסוּק ׳וַהֲבִיאוֹתִים אֶל הַר קָדְשִׁי׳.

הָאַדְמוֹ״ר מִוִּיז׳נִיץ בָּכָה, רָקַד וְשָׁר.

רָמָה א׳ – הַכֹּתֶל חָזַר

🙏 שֶׁלֹּא יֹאמְרוּ ׳כֹּחִי׳

הָאַדְמוֹ״ר חָזַר וְשִׁנֵּן: שֶׁלֹּא יֹאמְרוּ ׳כֹּחִי וְעֹצֶם יָדִי׳.

הַכֹּל בָּא מֵה׳, וְהוּא שֶׁנָּתַן לָנוּ אֶת הַנֵּס.

לָכֵן הַשִּׂמְחָה הִיא שִׂמְחָה שֶׁל תּוֹדָה.

הַשִּׂמְחָה שֶׁל הַכֹּתֶל הִיא תּוֹדָה לַה׳ עַל הַנֵּס.

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹהֵב לְסַפֵּר עַל הַנִּסִּים שֶׁל ה׳ 💛

📖 כֹּתֶל הַהוֹדָיוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אִבְּן עֶזְרָא בְּמִדְבַּר י׳ י׳

רָמָה א׳ – הַכֹּתֶל חָזַר

🎊 יוֹם שִׂמְחָה

הָאִבְּן עֶזְרָא לָמַד שֶׁיּוֹם שֶׁבּוֹ ה׳ מַצִּיל אוֹתָנוּ מֵאוֹיֵב הוּא יוֹם שִׂמְחָה.

בַּיּוֹם הַזֶּה גַּם צָרִיךְ לְסַפֵּר עַל הַנִּסִּים.

כָּךְ אֲנַחְנוּ שְׂמֵחִים וּמוֹדִים.

יוֹם שֶׁה׳ מַצִּיל אוֹתָנוּ הוּא יוֹם שִׂמְחָה וְיוֹם שֶׁל סִפּוּר נִסִּים.

🌾

📖 כֹּתֶל הַהוֹדָיוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, כּוֹתֶל הַהוֹדָיוֹת

רָמָה א׳ – הַכֹּתֶל חָזַר

🌾 בְּחַג הַשָּׁבוּעוֹת

בְּחַג הַשָּׁבוּעוֹת תשכ״ז הַרְבֵּה יְהוּדִים בָּאוּ לַכֹּתֶל.

שָׁבוּעוֹת הוּא חַג הַבִּכּוּרִים, חַג שֶׁל תּוֹדָה עַל הַתּוֹרָה וְעַל הָאָרֶץ.

יֵשׁ שֶׁמְּבָאֲרִים: לֹא בְּמִקְרֶה חָזַר הַכֹּתֶל דַּוְקָא בְּחַג הַהוֹדָיָה.

חָזְרוּ לַכֹּתֶל בְּחַג הַשָּׁבוּעוֹת, חַג הַבִּכּוּרִים וְהַהוֹדָיָה.

📖 כֹּתֶל הַהוֹדָיוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אָמוֹרוֹת טְהוֹרוֹת

רָמָה א׳ – הַכֹּתֶל חָזַר

📖 תְּהִלִּים כָּל יוֹם חֲמִישִׁי

אַבָּא שֶׁל הָאַדְמוֹ״ר מֵרַאחְמִיסְטְרִיבְקָה כָּתַב לוֹ: ׳הִתְחַלְתִּי לָלֶכֶת אֶל הַכֹּתֶל בְּכָל יוֹם חֲמִישִׁי לוֹמַר שָׁם אֶת כָּל סֵפֶר תְּהִלִּים.׳

כְּשֶׁחָמִיו שָׁמַע אֶת הַמִּכְתָּב הוּא אָמַר בְּהִתְרַגְּשׁוּת: ׳כָּזֶה עוֹלָם הַבָּא! לָמָּה זָכָה אַבָּא שֶׁלְּךָ!׳

אַחֲרֵי הַשִּׁחְרוּר הָלַךְ אָב כָּל יוֹם חֲמִישִׁי לוֹמַר תְּהִלִּים בַּכֹּתֶל.

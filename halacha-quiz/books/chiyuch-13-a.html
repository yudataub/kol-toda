אֲבָל דַּוְקָא פְּעֻלָּה ׳קְטַנָּה׳, שֶׁרַק מְזִיזָה קְצָת אֶת שְׁרִירֵי הַפָּנִים, מַכִּילָה כֹּחַ עֲצוּם.

חִיּוּךְ אֶחָד הוּא כְּמוֹ טַבְלִיָּה עִם הַרְבֵּה וִיטָמִינִים: הַרְבֵּה מִצְווֹת בְּבַת אַחַת.

בְּסֵפֶר הַחִנּוּךְ מְבֹאָר שֶׁשֹּׁרֶשׁ הַמִּצְוָה ׳וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ׳ הוּא שֶׁהָעוֹלָם נִבְנֶה עַל אַהֲבָה וְשָׁלוֹם בֵּין הַבְּרִיּוֹת.

הַחִיּוּךְ אוֹמֵר בְּלִי מִלִּים: ׳אֲנִי אוֹהֵב אוֹתְךָ, הַטּוֹב שֶׁלְּךָ חָשׁוּב לִי׳.

בָּחוּר יְשִׁיבָה שַׁקְדָן הִרְגִּישׁ בּוֹדֵד וְ׳שָׁקוּף׳ בֵּין מֵאוֹת הַבַּחוּרִים, וְהֶחְלִיט לַעֲזֹב. הוּא אָרַז בְּשֶׁקֶט אֶת הַמִּזְוָדוֹת וְהִתְחִיל לִגְרֹר אוֹתָן לַיְּצִיאָה.

בַּמִּסְדְּרוֹן עָבַר אַבְרֵךְ מְבֻגָּר שֶׁלֹּא הִכִּיר אוֹתוֹ, וּפָנָיו אוֹרוּ. הוּא עָצַר, חִיֵּךְ חִיּוּךְ רָחָב וְחַם וְאָמַר: ׳שָׁלוֹם עֲלֵיכֶם! אֵיזֶה יֹפִי לִרְאוֹת אוֹתְךָ! הַפָּנִים שֶׁלְּךָ מְאִירוֹת אֶת בֵּית הַמִּדְרָשׁ שֶׁלָּנוּ׳.

הַבָּחוּר קָפָא, מִלְמֵל ׳תּוֹדָה׳, חָזַר לַחֶדֶר וּפֵרֵק אֶת הַמִּזְוָדוֹת.

חֶסֶד בַּגּוּף גָּדוֹל מֵחֶסֶד בְּמָמוֹן: הַכֶּסֶף הוּא חִיצוֹנִי לִי, וְהַחִיּוּךְ הוּא אֲנִי.

וְאֶפְשָׁר לַעֲשׂוֹת חֶסֶד גַּם עִם עָשִׁיר, כִּי גַּם הָאָדָם הֶעָשִׁיר זָקוּק לְחִיּוּךְ, וְאֶת זֶה הוּא לֹא יָכוֹל לִקְנוֹת בְּכָל כַּסְפּוֹ.

חִיּוּךְ הוּא חֶסֶד שֶׁנּוֹתְנִים מִתּוֹךְ עַצְמֵנוּ.

׳מַה הוּא רַחוּם – אַף אַתָּה רַחוּם׳. אֲבָל יֵשׁ תֹּאַר אֶחָד שֶׁל ה׳ שֶׁשּׁוֹכְחִים לְחַקּוֹת: מֵאִיר פָּנִים.

בְּבִרְכַּת כֹּהֲנִים אוֹמְרִים ׳יָאֵר ה׳ פָּנָיו אֵלֶיךָ וִיחֻנֶּךָּ׳, וְדָוִד הַמֶּלֶךְ מְבַקֵּשׁ: ׳אֱלֹהִים יְחָנֵּנוּ וִיבָרְכֵנוּ יָאֵר פָּנָיו אִתָּנוּ סֶלָה׳.

אַחַת הַדְּרָכִים שֶׁבָּהֶן הַתּוֹרָה נִקְנֵית הִיא ׳נוֹשֵׂא בְּעֹל עִם חֲבֵרוֹ׳. הָעֹל הַכָּבֵד בְּיוֹתֵר הוּא הָעֹל שֶׁבַּלֵּב.

חִיּוּךְ שֶׁל הִשְׁתַּתְּפוּת אוֹמֵר: ׳אֲנִי רוֹאֶה אוֹתְךָ, אַתָּה לֹא לְבַד׳.

כָּךְ הַגְּרַ״שׁ אוֹיֶרְבַּךְ עָצַר בָּרְחוֹב בָּחוּר שֶׁהָלַךְ בְּרֹאשׁ מֻשְׁפָּל, הִנִּיחַ יָד עַל כְּתֵפוֹ וְאָמַר: ׳אֲנִי לֹא יוֹדֵעַ מָה עוֹבֵר עָלֶיךָ, אֲבָל הַקָּדוֹשׁ בָּרוּךְ הוּא אוֹהֵב אוֹתְךָ מְאוֹד, וְגַם אֲנִי שָׂמֵחַ לִפְגֹּשׁ אוֹתְךָ הַבֹּקֶר׳.

חִיּוּךְ שֶׁל הִשְׁתַּתְּפוּת אוֹמֵר: אַתָּה לֹא לְבַד.

8

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְקַיֵּם חָמֵשׁ מִצְווֹת בְּחִיּוּךְ אֶחָד 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סֵפֶר הַחִיּוּךְ, פֶּרֶק ד׳ (חֵלֶק א׳)

רָמָה א׳ – חָמֵשׁ הַפֵּאוֹת הָרִאשׁוֹנוֹת

💎 חָמֵשׁ פֵּאוֹת בֵּין אָדָם לַחֲבֵרוֹ

בַּיַּהֲלוֹם בַּעַל עֶשֶׂר הַפֵּאוֹת, חָמֵשׁ הַפֵּאוֹת הָרִאשׁוֹנוֹת הֵן בֵּין אָדָם לַחֲבֵרוֹ:

❤️ וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ

🤲 גְּמִילוּת חֲסָדִים בַּגּוּף

👑 וְהָלַכְתָּ בִּדְרָכָיו

🚫 לֹא לְצַעֵר אֶת הַזּוּלָת

🎒 נוֹשֵׂא בְּעֹל עִם חֲבֵרוֹ

בְּחִיּוּךְ אֶחָד מְקַיְּמִים אֶת כֻּלָּן.

חָמֵשׁ פֵּאוֹת: אַהֲבָה, חֶסֶד, דְּרָכָיו, לֹא לְצַעֵר, נוֹשֵׂא בְּעֹל.

9
💡

רָמָה א׳ – חָמֵשׁ הַפֵּאוֹת הָרִאשׁוֹנוֹת

מָה לוֹקְחִים מִרָמָה א׳?

חִיּוּךְ אֶחָד שֶׁלִּי הוּא מַתָּנָה לַחֲבֵרִים וּלְה׳. כְּשֶׁאֲנִי מַאִיר פָּנִים, אֲנִי הוֹלֵךְ בִּדְרָכָיו וּמַרְגִּישׁ אֶת אַהֲבָתוֹ.

10
❤️
אֱלֹהִים יְחָנֵּנוּ וִיבָרְכֵנוּ יָאֵר פָּנָיו אִתָּנוּ סֶלָה

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • אֵילוּ מֵעֶשֶׂר הַפֵּאוֹת הִרְגַּשְׁתִּי הַיּוֹם כְּשֶׁחִיַּכְתִּי אוֹ שֶׁחִיְּכוּ אֵלַי?
  • מִי הַ׳גֵּר׳ שֶׁלִּי: מִי בַּכִּתָּה, בְּבֵית הַכְּנֶסֶת אוֹ בַּשְּׁכוּנָה מַרְגִּישׁ חָדָשׁ אוֹ לְבַד?
  • הַשָּׁבוּעַ אֲנִי כּוֹתֵב אֶת עֶשֶׂר הַפֵּאוֹת עַל כַּרְטִיס קָטָן, מְסַמֵּן כָּל עֶרֶב אֵילוּ קִיַּמְתִּי, וּמְחַיֵּךְ בְּכַוָּנָה אֶל ׳גֵּר׳ אֶחָד.

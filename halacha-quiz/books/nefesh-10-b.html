רָמָה ב׳ – אֶרֶךְ אַפַּיִם

🧭 מִדָּה הֲפוּכָה לַכַּעַס

בְּסֵפֶר ׳אֹרֶךְ אַפַּיִם׳ יֵשׁ מִדָּה שֶׁהִיא הֵפֶךְ הַכַּעַס: לִהְיוֹת אָדָם נוֹחַ – סַבְלָנוּת.

הָרַמְבַּ״ן כָּתַב אוֹתָהּ בִּתְחִלַּת הָאִגֶּרֶת: ׳תִּתְנַהֵג תָּמִיד לְדַבֵּר כָּל דְּבָרֶיךָ בְּנַחַת, לְכָל אָדָם וּבְכָל עֵת׳.

מִי שֶׁמִּתְנַהֵג בְּסַבְלָנוּת לֹא רַק נִמְנָע מִכַּעַס, אֶלָּא גַּם מְקַיֵּם מִצְוָה.

סַבְלָנוּת הִיא הֵפֶךְ הַכַּעַס, וְאֶפְשָׁר לִקְנוֹת אוֹתָהּ.

רָמָה ב׳ – אֶרֶךְ אַפַּיִם

🌟 הוֹלְכִים בְּדַרְכֵי ה׳

בַּתּוֹרָה כָּתוּב עַל ה׳: ׳אֵל רַחוּם וְחַנּוּן אֶרֶךְ אַפַּיִם וְרַב חֶסֶד וֶאֱמֶת׳.

הַתֹּמֶר דְּבוֹרָה חִדֵּשׁ שֶׁגַּם זֶה חֵלֶק מִמִּדּוֹתָיו שֶׁל ה׳ שֶׁהָאָדָם מְחֻיָּב בָּהֶם. לָכֵן כָּל זְמַן שֶׁאָדָם מִתְנַהֵג בְּסַבְלָנוּת – הוּא הוֹלֵךְ בְּדַרְכָּיו שֶׁל ה׳.

וְכֵן בַּמַּעֲשִׂים: עַל הָאָדָם לִדְאֹג שֶׁבְּכָל מַצָּב הוּא יִהְיֶה נוֹחַ.

ה׳ אֶרֶךְ אַפַּיִם – וּמִי שֶׁסַּבְלָן הוֹלֵךְ בְּדַרְכָּיו.

🛠️

💛 אַהֲבַת הַשֵּׁם: אֲנִי קוֹנֶה סַבְלָנוּת בְּכָל פַּעַם שֶׁאֲנִי מַמְתִּין בְּנַחַת 💛

📖 עֲנָוָה וְנַחַת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּעִנְיַן מִדַּת הַכַּעַס · הַתּוֹרָה שֶׁלָּמַדְתִּי בַּכְּפָר (רַבִּי יִצְחָק מִוּוּרְקִי)

רָמָה ב׳ – אֶרֶךְ אַפַּיִם

🛠️ מִדָּה אֶפְשָׁר לִקְנוֹת

סַבְלָנוּת הִיא מִדָּה, וּמִדָּה אֶפְשָׁר לִקְנוֹת בַּעֲבוֹדָה. בַּשִּׁעוּר נֶאֱמַר שֶׁאֶפְשָׁר לִקְנוֹת בְּנַפְשׁוֹ אֶת ׳הַמִּדָּה הַנִּפְלָאָה שֶׁל סַבְלָנוּת׳.

יִצְחָק לֹא נוֹלַד סַבְלָן. הוּא הִתְאַמֵּן עֶשְׂרִים פְּעָמִים עִם יֶלֶד אֶחָד, וְכָךְ קָנָה אֶת הַמִּדָּה.

כָּל נִסָּיוֹן קָטָן שֶׁל סַבְלָנוּת מַגְדִּיל אֶת הַמִּדָּה.

סַבְלָנוּת נִקְנֵית בְּאִמּוּן, וַאֲפִלּוּ בְּיֶלֶד אֶחָד וְאוֹת אַחַת.

🍬

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי יוֹתֵר כְּשֶׁאֲנִי יוֹדֵעַ לְהַמְתִּין 💛

רָמָה ב׳ – אֶרֶךְ אַפַּיִם

🍬 נִסּוּי הַמַּרְשְׁמֶלוֹ

בַּדַּף ׳סוֹד הַסַּבְלָנוּת׳ מְסֻפָּר עַל נִסּוּי מְפֻרְסָם: שָׂמוּ יֶלֶד לְבַד בְּחֶדֶר עִם מַרְשְׁמֶלוֹ אֶחָד. אָמְרוּ לוֹ: אַתָּה יָכוֹל לֶאֱכֹל אוֹתוֹ עַכְשָׁו, אוֹ לְהַמְתִּין 15 דַּקּוֹת וּלְקַבֵּל שְׁנַיִם.

חֵלֶק מֵהַיְלָדִים אָכְלוּ מִיָּד, וְחֵלֶק הִמְתִּינוּ. הַחוֹקְרִים עָקְבוּ אַחֲרֵיהֶם עֲשָׂרוֹת שָׁנִים, וְגִלּוּ שֶׁהַיְלָדִים שֶׁהִמְתִּינוּ הָיוּ יוֹתֵר מַצְלִיחִים וּמְאֻשָּׁרִים וּבַעֲלֵי קְשָׁרִים חֶבְרָתִיִּים טוֹבִים יוֹתֵר כְּשֶׁגָּדְלוּ.

הַסּוֹד: לִפְעָמִים כְּדַאי לְהַמְתִּין, כִּי אָז מְקַבְּלִים הַרְבֵּה יוֹתֵר.

מִי שֶׁיּוֹדֵעַ לְהַמְתִּין, מְקַבֵּל הַרְבֵּה יוֹתֵר.

רָמָה ב׳ – אֶרֶךְ אַפַּיִם

🥘 עֵשָׂו שֶׁלֹּא יָכֹל לְהַמְתִּין

בַּתּוֹרָה: עֵשָׂו חָזַר מֵהַצַּיִד רָעֵב וְעָיֵף, וְיַעֲקֹב בִּשֵּׁל נְזִיד עֲדָשִׁים. עֵשָׂו אָמַר: ׳הַלְעִיטֵנִי נָא מִן הָאָדֹם הָאָדֹם הַזֶּה כִּי עָיֵף אָנֹכִי׳.

מוּסְבָּר בַּדַּף: ׳נָא׳ מַשְׁמָעוּתָהּ גַּם ׳עַכְשָׁו! מִיָּד!׳, וְ׳הָאָדֹם הָאָדֹם׳ – הָעֲדָשִׁים עוֹד לֹא הָיוּ מְבֻשָּׁלוֹת לְגַמְרֵי. עֵשָׂו לֹא יָכֹל לְהַמְתִּין וְרָצָה הַכֹּל מִיָּד.

וּבַסּוֹף הוּא מָכַר אֶת הַבְּכוֹרָה – הַדָּבָר הֲכִי חָשׁוּב. לְעֻמָּתוֹ יַעֲקֹב יָדַע לְהַמְתִּין וְלַעֲבֹד בְּסַבְלָנוּת.

מִי שֶׁלֹּא יָכוֹל לְהַמְתִּין – עָלוּל לְוַתֵּר עַל הַדָּבָר הַחָשׁוּב.

רָמָה ב׳ – אֶרֶךְ אַפַּיִם

🗝️ אֹרֶךְ רוּחַ וְאַהֲבָה

יִצְחָק הֵבִין: גַּם אָדָם חָכָם וּמוּכְשָׁר מְאֹד, בְּלִי סַבְלָנוּת, בְּלִי אֹרֶךְ רוּחַ וּבְלִי אַהֲבָה לַזּוּלַת – אֵינוֹ יָכוֹל לִפְתֹּחַ אֶת הַלְּבָבוֹת וּלְהַנְחִיל אֶת הַתּוֹרָה.

לָכֵן רַבִּי יִצְחָק מִוּוּרְקִי אָמַר כָּל חַיָּיו שֶׁהַתּוֹרָה שֶׁלָּמַד בַּכְּפָר הִיא הָאֲמִתִּית.

סַבְלָנוּת, אֹרֶךְ רוּחַ וְאַהֲבָה הֵם הַיְסוֹד לְכָל לִמּוּד, לְכָל עֲשִׂיָּה וּלְכָל קֶשֶׁר עִם בְּנֵי אָדָם.

סַבְלָנוּת, אֹרֶךְ רוּחַ וְאַהֲבָה פּוֹתְחִים לְבָבוֹת.

מָה לוֹקְחִים מִרָמָה ב׳?

ה׳ אֶרֶךְ אַפַּיִם וְרַב חֶסֶד, וְכָל פַּעַם שֶׁאֲנִי סַבְלָן, אֲנִי הוֹלֵךְ בְּדַרְכָּיו. הַסַּבְלָנוּת הִיא דֶּרֶךְ לְהַרְגִּישׁ אֶת אַהֲבַת ה׳ אֵלַי וְאֶת אַהֲבָתִי אֵלָיו.

  • אֶגְזֹר מַפְתֵּחַ מִקַּרְטוֹן, אֶכְתֹּב בְּתוֹכוֹ מִקְרֶה אֶחָד שֶׁבּוֹ סַבְלָנוּת עָזְרָה לִי, וְאֶתְלֶה אוֹתוֹ בַּחֶדֶר.

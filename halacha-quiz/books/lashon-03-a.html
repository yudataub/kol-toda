רָמָה א׳ – עוֹצְרִים וְחוֹשְׁבִים

לָמָּה חוֹשְׁבִים לִפְנֵי שֶׁמְּדַבְּרִים, וְאֵיךְ בּוֹחֲרִים אֶת הַמִּלָּה הַנְּכוֹנָה

🪙 מִלָּה בְּסֶלַע

בַּגְּמָרָא אוֹמְרִים: ׳מִלָּה בְּסֶלַע, מִשְׁתּוֹקָא בִּתְרֵין׳. סֶלַע הוּא מַטְבֵּעַ שֶׁל כֶּסֶף.

כְּלוֹמַר: אִם הַמִּלָּה שָׁוָה סֶלַע, הַשְּׁתִיקָה שָׁוָה שְׁנַיִם! הַגְּמָרָא מְלַמֶּדֶת אוֹתָנוּ שֶׁשְּׁתִיקָה חֲכָמָה הִיא יְקָרָה מְאֹד.

שְׁתִיקָה חֲכָמָה שָׁוָה יוֹתֵר מִמִּלָּה.

❓ אֲבָל לָמָּה?

יֵשׁ שֶׁשָּׁאֲלוּ: בְּדִבּוּר יֵשׁ הַרְבֵּה מִלִּים, וּבִשְׁתִיקָה שׁוֹתְקִים רַק פַּעַם אַחַת – אָז אוּלַי עָדִיף לְדַבֵּר?

הַתְּשׁוּבָה: מְדֻבָּר עַל הַשְּׁתִיקָה שֶׁבֵּין הַמִּלִּים. לִפְנֵי שֶׁמּוֹצִיאִים מִלָּה מֵהַפֶּה, עוֹצְרִים רֶגַע וְחוֹשְׁבִים.

הַשְּׁתִיקָה הַיְקָרָה הִיא שֶׁבֵּין הַמִּלִּים.

🚦 עוֹצְרִים וְחוֹשְׁבִים

דָּנִי רוֹצֶה לוֹמַר לְחָבֵר: ׳אַתָּה לֹא יוֹדֵעַ לְשַׂחֵק!׳ אֲבָל הוּא עוֹצֵר רֶגַע וְחוֹשֵׁב: ׳מָה הֶחָבֵר יַרְגִּישׁ?׳

אָז הוּא אוֹמֵר: ׳בּוֹא נְנַסֶּה שׁוּב יַחַד׳. הַשְּׁתִיקָה הַקְּטַנָּה הִצִּילָה אֶת הַמִּשְׂחָק, וְהֶחָבֵר חִיֵּךְ.

עוֹצְרִים, חוֹשְׁבִים, וְאָז אוֹמְרִים.

3
📏

💛 אַהֲבַת הַשֵּׁם: אֲנִי חוֹשֵׁב לִפְנֵי שֶׁאֲנִי מְדַבֵּר, כִּי הַפֶּה הוּא מַתָּנָה מֵה׳ 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אִגֶּרֶת הָרַמְבַּ״ן

רָמָה א׳ – עוֹצְרִים וְחוֹשְׁבִים

📏 וְחַשֹּׁב קֹדֶם

הָרַמְבַּ״ן כָּתַב בְּאִגַּרְתּוֹ: ׳וְחַשֹּׁב הַדִּבּוּר קֹדֶם שֶׁתּוֹצִיאֶנּוּ מִפִּיךָ׳.

זֶה בְּדִיּוּק מָה שֶׁמְּלַמֶּדֶת הַשְּׁתִיקָה שֶׁבֵּין הַמִּלִּים: מְחַשְּׁבִים לִפְנֵי שֶׁאוֹמְרִים, לֹא אַחֲרֵי. מִי שֶׁחוֹשֵׁב קֹדֶם, בּוֹחֵר מִלִּים טוֹבוֹת יוֹתֵר.

קוֹדֶם חוֹשְׁבִים, אַחַר כָּךְ מְדַבְּרִים.

💛 אַהֲבַת הַשֵּׁם: אֲנִי שׁוֹמֵר עַל הַמִּלִּים שֶׁלִּי, כִּי הֵן אוֹצָר שֶׁנָּתַן לִי ה׳ 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְאִירִי עַל אָבוֹת א׳ י״ז

רָמָה א׳ – עוֹצְרִים וְחוֹשְׁבִים

💰 כְּמוֹ כֶּסֶף

חַכְמֵי הַמּוּסָר אָמְרוּ: ׳אֱצֹר לְשׁוֹנְךָ כַּאֲשֶׁר תֶּאֱצֹר מָמוֹנְךָ׳. מִי שֶׁיֵּשׁ לוֹ כֶּסֶף לֹא מוֹצִיא אוֹתוֹ בְּלִי לַחְשֹׁב.

כָּךְ גַּם מִלִּים: כָּל מִלָּה יְקָרָה, וְכָדַאי לְהוֹצִיא רַק אֶת מָה שֶׁבֶּאֱמֶת צָרִיךְ.

שׁוֹמְרִים עַל הַלָּשׁוֹן כְּמוֹ שֶׁשּׁוֹמְרִים עַל כֶּסֶף.

🔤

💛 אַהֲבַת הַשֵּׁם: אֲנִי זוֹכֵר: שְׁתִיקָה בִּימִין, שִׂיחָה בִּשְׂמֹאל, וְלִפְנֵי כֹּל דִּבּוּר מְבַקֵּשׁ מֵה׳ עֶזְרָה 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – עוֹצְרִים וְחוֹשְׁבִים

🔤 שִׁי״ן יְמָנִית וּשְׂמָאלִית

בַּמִּלָּה ׳שְׁתִיקָה׳ יֵשׁ שִׁי״ן יְמָנִית (שׁ), וּבַמִּלָּה ׳שִׂיחָה׳ יֵשׁ שִׂי״ן שְׂמָאלִית (שׂ).

הַשְּׁתִיקָה הִיא הַשִּׁי״ן הַיְמָנִית, כִּי בָּהּ צְרִיכִים לְהַחֲזִיק רֹב הַזְּמַן – וְהַשִּׂיחָה בַּצַּד הַשְּׂמָאלִי. זֶה סִימָן קָטָן שֶׁעוֹזֵר לִזְכֹּר.

רֹב הַזְּמַן מַחְזִיקִים בַּשְּׁתִיקָה, וּבַשְּׂמֹאל מְדַבְּרִים.

🤝

💛 אַהֲבַת הַשֵּׁם: אֲנִי שׁוֹמֵר אֶת הַבְּרִית עִם הַלָּשׁוֹן שֶׁלִּי, מִתּוֹךְ אַהֲבָה לַה׳ 💛

רָמָה א׳ – עוֹצְרִים וְחוֹשְׁבִים

🤝 בְּרִית הַלָּשׁוֹן

יֵשׁ בְּרִית מִילָה, וְיֵשׁ גַּם בְּרִית הַלָּשׁוֹן. כְּמוֹ שֶׁנִּכְרַת בְּרִית בַּגּוּף, כָּךְ הָאָדָם עוֹשֶׂה בְּרִית עִם הַלָּשׁוֹן שֶׁלּוֹ: הוּא לֹא מוֹצִיא כָּל מָה שֶׁעוֹלֶה עַל דַּעְתּוֹ.

שְׁנֵי הַבְּרִיתוֹת הֵן מֵאֵת ה׳, וְשְׁתֵּיהֶן שׁוֹמְרוֹת עָלֵינוּ.

בְּרִית הַלָּשׁוֹן: לֹא מוֹצִיאִים כָּל מָה שֶׁעוֹלֶה עַל הַדַּעַת.

👋

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְבָרֵךְ אֶת חֲבֵרִי ׳לֵךְ לְשָׁלוֹם׳ וּמְבַקֵּשׁ מֵה׳ שֶׁיַּצְלִיחַ אֶת דַּרְכָּם 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּרָכוֹת ס״ד א׳ · שְׁמוֹת ד׳ י״ח · שְׁמוּאֵל ב׳ ט״ו ט׳

רָמָה א׳ – עוֹצְרִים וְחוֹשְׁבִים

👋 לֵךְ לְשָׁלוֹם

כְּשֶׁנִּפְרָדִים מֵחָבֵר, אוֹמְרִים לוֹ ׳לֵךְ לְשָׁלוֹם׳ וְלֹא ׳לֵךְ בְּשָׁלוֹם׳. יִתְרוֹ אָמַר לְמֹשֶׁה ׳לֵךְ לְשָׁלוֹם׳ – וְהוּא הִצְלִיחַ בְּדַרְכּוֹ.

דָּוִד אָמַר לְאַבְשָׁלוֹם ׳לֵךְ בְּשָׁלוֹם׳ – וְאַבְשָׁלוֹם לֹא הִצְלִיחַ. גַּם מִלָּה קְטַנָּה אַחַת בַּפְּרֵדָה בַּעֲלַת כֹּחַ.

בְּפֵרוּד אוֹמְרִים ׳לֵךְ לְשָׁלוֹם׳.

מָה לוֹקְחִים מִרָמָה א׳?

מִי שֶׁחוֹשֵׁב לִפְנֵי שֶׁהוּא מְדַבֵּר, בּוֹחֵר מִלִּים יָפוֹת וּטְהוֹרוֹת. כָּךְ הַפֶּה שֶׁלִּי הוֹפֵךְ לִכְלִי שֶׁל תּוֹדָה וְאַהֲבָה, וַאֲנִי מַרְגִּישׁ אֶת אַהֲבַת ה׳ אֵלַי.

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי הַשָּׁבוּעַ אָמַרְתִּי מִלָּה בְּלִי לַחְשֹׁב, וּמָה הָיִיתִי אוֹמֵר אִם הָיִיתִי עוֹצֵר רֶגַע?
  • אֵיזוֹ בְּרָכָה אוֹמְרִים כְּשֶׁנִּפְרָדִים מֵחָבֵר, וּלְמִי אֹמַר אוֹתָהּ הַיּוֹם?
  • שָׁלוֹשׁ שְׁנִיּוֹת לִפְנֵי תְּשׁוּבָה: הַיּוֹם בַּכִּתָּה וּבַבַּיִת, לִפְנֵי שֶׁאֲנִי עוֹנֶה, אֲסַפֵּר בַּלֵּב עַד שָׁלוֹשׁ וְרַק אָז אֲדַבֵּר, וְאֶרְשֹׁם כַּמָּה פְּעָמִים הִצְלַחְתִּי.

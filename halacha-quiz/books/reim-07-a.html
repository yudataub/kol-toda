בַּשִּׁעוּר ׳מֵאַהֲבַת חִנָּם לְאַחְדוּת׳ לוֹמְדִים אֵיךְ לֹא לְהַדְלִיק אֶת הָאֵשׁ, וְאֵיךְ לְכַבּוֹתָהּ לִפְנֵי שֶׁהִיא גְּדֵלָה.

רִיב מַתְחִיל לִפְעָמִים בְּמִלָּה קְצָרָה.

הַסִּפּוּר הַזֶּה מַדְלִיק אֶת הָאֵשׁ יוֹתֵר מִמַּה שֶׁקָּרָה בֶּאֱמֶת.

הַסִּפּוּר שֶׁאֲנִי מוֹסִיף מַדְלִיק אֶת הָאֵשׁ.

אֶפְשָׁר לַעֲצֹר וְלִשְׁאֹל: מָה אֲנִי יוֹדֵעַ בְּוַדַּאי, וּמָה אֲנִי מַשְׁלִים בְּעַצְמִי?

וַדַּאי: ׳הוּא עָבַר וְלֹא אָמַר שָׁלוֹם׳. מַשְׁלִים: ׳הוּא מְזַלְזֵל בִּי׳.

הַשְּׁאֵלָה לֹא מְבַטֶּלֶת פְּגִיעָה אֲמִתִּית, אֲבָל נוֹתֶנֶת סִכּוּי לְהָגִיב בְּרוֹגַע וְלֹא מִתּוֹךְ לֶהָבָה.

עוֹצְרִים: מָה אֲנִי יוֹדֵעַ, וּמָה אֲנִי מַשְׁלִים?

כְּשֶׁחוֹשְׁבִים ׳אוּלַי׳, לְפָחוֹת לֹא הוֹפְכִים אֶת הֶחָבֵר לְאוֹיֵב.

׳אוּלַי׳ מוֹנֵעַ שִׂנְאָה לִפְנֵי שֶׁהִיא מַתְחִילָה.

👂 מַקְשִׁיבִים לִפְנֵי שֶׁשּׁוֹפְטִים

חָבֵר הוּא לֹא רַק הַמַּעֲשֶׂה שֶׁהִכְעִיס אוֹתִי. יֵשׁ לוֹ פְּחָדִים, מַאֲמָצִים וְתִקְווֹת שֶׁאֲנִי לֹא רוֹאֶה.

כְּשֶׁשּׁוֹאֲלִים וּמַקְשִׁיבִים לַתְּשׁוּבָה, מְקַבְּלִים תְּמוּנָה רְחָבָה יוֹתֵר. לִפְעָמִים אֲנִי צוֹדֵק, וְלִפְעָמִים הֶחְמַצְתִּי כִּמְעַט הַכֹּל.

שׁוֹאֲלִים וּמַקְשִׁיבִים לִפְנֵי שֶׁשּׁוֹפְטִים.

🌈 הֶבְדֵּל אֵינוֹ סַכָּנָה

בְּכִתָּה טוֹבָה יֵשׁ יְלָדִים שׁוֹנִים, עִם דֵּעוֹת וְאֹפִי שׁוֹנִים. אֶפְשָׁר לְהִתְוַכֵּחַ עַל רַעְיוֹן בְּלִי לִפְגֹּעַ בָּאָדָם עַצְמוֹ.

אַהֲבַת יִשְׂרָאֵל אֵינָהּ אוֹמֶרֶת שֶׁצָּרִיךְ לְהַסְכִּים עִם כֻּלָּם. הִיא אוֹמֶרֶת שֶׁגַּם מִי שֶׁחוֹשֵׁב אַחֶרֶת הוּא אָדָם בַּעַל כָּבוֹד וּנְשָׁמָה.

אֶפְשָׁר לֶאֱהֹב וּלְכַבֵּד גַּם מִי שֶׁחוֹשֵׁב אַחֶרֶת.

🌉 מַחֲוָה שֶׁפּוֹתַחַת גֶּשֶׁר

לֹא תָּמִיד אֶפְשָׁר לִפְתֹּר רִיב בְּשִׂיחָה אַחַת. אֲבָל כִּמְעַט תָּמִיד אֶפְשָׁר לִבְחֹר מַחֲוָה קְטַנָּה שֶׁלֹּא מוֹסִיפָה אֵשׁ: לְבָרֵךְ לְשָׁלוֹם, לְהַחְזִיר חֵפֶץ, לְהוֹדוֹת בַּחֵלֶק שֶׁלִּי, אוֹ לְהִמָּנַע מִמִּשְׁפָּט מִיּוּתָּר.

מַחֲוָה קְטַנָּה שֶׁלֹּא מוֹסִיפָה אֵשׁ פּוֹתַחַת גֶּשֶׁר.

אַהֲבַת חִנָּם נִבְנֵית מֵהַרְבֵּה רְגָעִים קְטַנִּים: רוֹאִים אֶת הַטּוֹב, מְבַקְּשִׁים הֶסְבֵּר לִפְנֵי שֶׁכּוֹעֲסִים, וְזוֹכְרִים שֶׁכָּל אָדָם נִבְרָא בְּצֶלֶם אֱלֹהִים.

מִי שֶׁמִּתְאַמֵּן כָּךְ נַעֲשֶׂה בְּעַצְמוֹ מָקוֹר שֶׁל רוֹגַע, וְהָאֲוִיר סְבִיבוֹ נַעֲשֶׂה טוֹב יוֹתֵר.

אַהֲבַת חִנָּם נִבְנֵית מֵרְגָּעִים קְטַנִּים.

מָה לוֹקְחִים מִרָמָה א׳?

ה׳ רוֹאֶה אֶת הַלֵּב שֶׁלָּנוּ, וְהוּא אוֹהֵב כָּל אֶחָד מִמֶּנּוּ. כְּשֶׁאֲנִי עוֹצֵר, שׁוֹאֵל וּמַקְשִׁיב, אֲנִי לוֹמֵד לֶאֱהֹב אֶת חֲבֵרִי כְּמוֹ שֶׁה׳ אוֹהֵב אוֹתִי.

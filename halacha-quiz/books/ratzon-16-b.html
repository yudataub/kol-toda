לָמָּה קוֹרְאִים קְרִיאַת שְׁמַע פַּעֲמַיִם?

לָמָּה קוֹרְאִים קְרִיאַת שְׁמַע בַּבֹּקֶר וּבָעֶרֶב, וְאֵיךְ הִיא מְעוֹרֶרֶת בַּלֵּב שְׁאִיפָה לְהִתְקָרֵב לַה׳

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִתְעוֹרֵר כָּל יוֹם מֵחָדָשׁ לִשְׁאֹף אֵלֶיךָ, ה׳ 💛

📖 רָצוֹן וְחֵשֶׁק לַעֲבוֹדַת ה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דְּבָרִים ו׳ ד׳–ז׳

רָמָה ב׳ – חֵפֶץ הַלֵּב

🌗 שְׁתֵּי סִבּוֹת

הַשִּׁעוּר מְסַבֵּר: הַתּוֹרָה צִוְּתָה לִקְרֹא קְרִיאַת שְׁמַע פַּעֲמַיִם בְּיוֹם, כִּי הִיא מְלַמֶּדֶת מָה תַּכְלִית חַיֵּינוּ, וְכָל יוֹם צָרִיךְ לִזְכֹּר אוֹתָהּ.

וְהַסִּבָּה הַשְּׁנִיָּה: כַּמָּה נַצְלִיחַ לְהַשִּׂיג אֶת הַתַּכְלִית בְּמֶשֶׁךְ הַיּוֹם בְּעֶזְרַת הַמִּצְווֹת, תָּלוּי בַּשְּׁאִיפָה שֶׁלָּנוּ. אִם יֵשׁ שְׁאִיפָה, נַשִּׂיג, וְכָל שֶׁהִיא גְּדוֹלָה יוֹתֵר, הַמִּצְווֹת עוֹשׂוֹת בָּנוּ יוֹתֵר אֱמוּנָה וְאַהֲבַת ה׳.

לָכֵן קוֹרְאִים קְרִיאַת שְׁמַע – כְּדֵי לְעוֹרֵר אֶת הַשְּׁאִיפָה.

שְׁתֵּי סִבּוֹת: לִזְכֹּר אֶת הַתַּכְלִית, וּלְעוֹרֵר אֶת הַשְּׁאִיפָה.

1
🏹

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹדֵף אַחֲרֶיךָ, ה׳, בְּחֶמְדָּה גְּדוֹלָה 💛

📖 רָצוֹן וְחֵשֶׁק לַעֲבוֹדַת ה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְסִלַּת יְשָׁרִים י״ט · יִרְמְיָהוּ ל״א כ׳

רָמָה ב׳ – חֵפֶץ הַלֵּב

🏹 שֶׁיְּהֵא חוֹשֵׁק וּמִתְאַוֶּה

מְסִלַּת יְשָׁרִים בְּפֶרֶק י״ט: עִנְיַן הָאַהֲבָה הוּא שֶׁיִּהְיֶה הָאָדָם חוֹשֵׁק וּמִתְאַוֶּה מַמָּשׁ אֶל קִרְבָתוֹ יִתְבָּרַךְ וְרוֹדֵף אַחַר קְדֻשָּׁתוֹ, כְּמוֹ שֶׁאָדָם רוֹדֵף אַחַר דָּבָר נֶחְמָד לוֹ בְּחֶמְדָּה עַזָּה.

עַד שֶׁהַזְכָּרַת שְׁמוֹ, הַדִּבּוּר בִּתְהִלּוֹתָיו וְהָעֵסֶק בְּדִבְרֵי תוֹרָתוֹ הֵם לוֹ שַׁעֲשׁוּעַ וְעֹנֶג מַמָּשׁ, כְּמוֹ אָב שֶׁאוֹהֵב אֶת בְּנוֹ הַיָּחִיד אַהֲבָה חֲזָקָה, שֶׁאֲפִלּוּ הַדִּבּוּר בּוֹ נָעִים לוֹ.

וְכָךְ כָּתוּב: ׳כִּי מִדֵּי דַבְּרִי בּוֹ זָכֹר אֶזְכְּרֶנּוּ עוֹד׳.

אַהֲבָה: חוֹשְׁקִים, מִתְאַוִּים וְרוֹדְפִים אַחַר קְרִבַת ה׳.

🌊

💛 אַהֲבַת הַשֵּׁם: נַפְשִׁי צָמְאָה לְךָ, ה׳ 💛

📖 רָצוֹן וְחֵשֶׁק לַעֲבוֹדַת ה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְסִלַּת יְשָׁרִים ז׳ · תְּהִלִּים מ״ב, פ״ד, ס״ג

רָמָה ב׳ – חֵפֶץ הַלֵּב

🌊 חֵפֶץ הַלֵּב וּתְשׁוּקַת הַנְּשָׁמָה

מְסִלַּת יְשָׁרִים (פֶּרֶק ז׳): ׳הַנִּרְצֶה יוֹתֵר בַּעֲבוֹדַת הַבּוֹרֵא יִתְבָּרַךְ שְׁמוֹ הוּא חֵפֶץ הַלֵּב וּתְשׁוּקַת הַנְּשָׁמָה׳.

כָּךְ דָּוִד הַמֶּלֶךְ מִתְהַלֵּל בְּחֶלְקוֹ הַטּוֹב: ׳כְּאַיָּל תַּעֲרֹג עַל אֲפִיקֵי מָיִם כֵּן נַפְשִׁי תַעֲרֹג אֵלֶיךָ אֱלֹהִים׳; ׳נִכְסְפָה וְגַם כָּלְתָה נַפְשִׁי לְחַצְרוֹת יְהוָה׳; ׳צָמְאָה לְךָ נַפְשִׁי כָּמַהּ לְךָ בְשָׂרִי׳.

הַלֵּב שֶׁחוֹשֵׁק הוּא הַמַּתָּנָה שֶׁה׳ רוֹצֶה מִמֶּנּוּ.

הַנִּרְצֶה יוֹתֵר: חֵפֶץ הַלֵּב וּתְשׁוּקַת הַנְּשָׁמָה.

🧲

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹשֶׂה מִצְווֹת כְּדֵי לְהִדָּבֵק בְּךָ, ה׳ 💛

📖 רָצוֹן וְחֵשֶׁק לַעֲבוֹדַת ה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִכְתָּב מֵאֵלִיָּהוּ, חֵלֶק א׳

רָמָה ב׳ – חֵפֶץ הַלֵּב

🧲 לְשֵׁם דְּבֵקוּת בַּבּוֹרֵא

הַמִּכְתָּב מֵאֵלִיָּהוּ מְלַמֵּד: יֵשׁ שְׁנֵי אוֹפָנִים לְקַיֵּם מִצְווֹת. לְשֵׁם ה׳ – כְּדֵי לְהִדָּבֵק בּוֹ וּבְדְרָכָיו הַקְּדוֹשׁוֹת. שֶׁלֹּא לְשֵׁם ה׳ – אֲפִלּוּ עוֹשִׂים מַעֲשִׂים טוֹבִים כִּי הֵם נוֹשְׂאִים חֵן וּמַתְאִימִים לַטֶּבַע וְלַיֹּשֶׁר שֶׁלָּנוּ, אֲבָל לֹא כְּדֵי לְהִדָּבֵק בַּבּוֹרֵא.

הַשָּׂכָר הַגָּדוֹל הוּא הַדְּבֵקוּת עַצְמָהּ: ׳צַדִּיקִים יוֹשְׁבִים וְנֶהֱנִים מִזִּיו הַשְּׁכִינָה׳. מִצְוָה שֶׁהִיא בִּגְדַר דְּבֵקוּת בַּבּוֹרֵא – הַקְּדֻשָּׁה שֶׁבָּאָה לַנֶּפֶשׁ נִתֶּנֶת בִּכְפָלִים.

מִצְוָה לְשֵׁם דְּבֵקוּת בַּבּוֹרֵא – שְׂכָרָהּ כָּפוּל.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְאַמֵּן אֶת הַשְּׁאִיפָה שֶׁלִּי לַקְּדֻשָּׁה, ה׳ 💛

📖 רָצוֹן וְחֵשֶׁק לַעֲבוֹדַת ה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִכְתָּב מֵאֵלִיָּהוּ, חֵלֶק א׳

רָמָה ב׳ – חֵפֶץ הַלֵּב

🧭 הַשְּׁאִיפָה הוֹלֶכֶת אִתָּנוּ

עוֹד מְלַמֵּד הַמִּכְתָּב מֵאֵלִיָּהוּ: ׳כְּפִי אֲשֶׁר תָּבוֹא נַפְשׁוֹ מִכָּאן, כֵּן תִּהְיֶה שָׁם׳.

מִי שֶׁהָרְצוֹנוֹת שֶׁלּוֹ בַּחַיִּים הָיוּ גַּשְׁמִיִּים, לֹא יִהְיֶה לוֹ חֵשֶׁק לְרוּחָנִיּוּת גַּם שָׁם. וּמִי שֶׁהַשְּׁאִיפָה וְהָרְצוֹנוֹת שֶׁלּוֹ הָיוּ רוּחָנִיִּים, לְהִתְקָרֵב אֵלָיו יִתְבָּרַךְ, יִשְׁאַף לְרוּחָנִיּוּת וְלִקְרָבַת ה׳ גַּם שָׁם.

לָכֵן שִׁנּוּן הַשְּׁאִיפָה כָּל יוֹם הוּא דָּבָר חָשׁוּב.

מִי שֶׁשּׁוֹאֵף לְרוּחָנִיּוּת, יִשְׁאַף לָהּ תָּמִיד.

רָמָה ב׳ – חֵפֶץ הַלֵּב

מָה לוֹקְחִים מִרָמָה ב׳?

חֵפֶץ הַלֵּב וּתְשׁוּקַת הַנְּשָׁמָה הֵם הַנִּרְצֶה יוֹתֵר לִפְנֵי ה׳. מִי שֶׁשּׁוֹאֵף לִדְבֵקוּת בּוֹ, מַרְגִּישׁ שֶׁה׳ אוֹהֵב אוֹתוֹ וְעוֹזֵר לוֹ לְהִתְקָרֵב.

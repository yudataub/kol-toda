🍞 מוֹדִים עַל הַלֶּחֶם

סוֹף הָאֲרוּחָה. אֲנִי כְּבָר רוֹצֶה לָרוּץ לְשַׂחֵק, וּבִרְכַּת הַמָּזוֹן עוֹבֶרֶת מַהֵר מַהֵר.

בְּאֶמְצַע הַבְּרָכָה אוֹמְרִים ׳נוֹדֶה לְךָ׳, וְזֶה אוֹמֵר: אֲנַחְנוּ מוֹדִים לְךָ. בִּרְכַּת הַמָּזוֹן אֵינָהּ רַק מִלִּים לְסַיֵּם אֶת הַסְּעוּדָה, אֶלָּא הוֹדָיָה לַה׳.

מְסַפְּרִים עַל רַבִּי שְׁלֹמֹה זַלְמָן אוֹיֶרְבַּךְ שֶׁאָמַר אֶת הַבְּרָכָה הַזֹּאת בְּכַוָּנָה גְּדוֹלָה. נִלְמַד מִמֶּנּוּ אֵיךְ.

׳נוֹדֶה לְךָ׳ פֵּרוּשׁוֹ: אֲנַחְנוּ מוֹדִים לְךָ.

🍷 מוֹצֵא דֶּרֶךְ לְהוֹדוֹת

כְּשֶׁנּוֹלְדָה נֶכְדָּה לְרַבִּי שְׁלֹמֹה זַלְמָן, הוּא רָצָה לְהוֹדוֹת לַה׳, אַף שֶׁכְּבָר הָיוּ לוֹ נֶכְדִּים. אֲבָל לֹא נִתְקְנָה בְּרָכָה מְיֻחֶדֶת לְסָבָא שֶׁנּוֹלַד לוֹ נֶכֶד.

אָז הוּא מָצָא עֵצָה: הִשִּׂיג שְׁנֵי סוּגֵי יַיִן. עַל הַיַּיִן הַפָּחוֹת טוֹב בֵּרַךְ ׳בּוֹרֵא פְּרִי הַגֶּפֶן׳. וּכְשֶׁהֵבִיאוּ אֶת הַיַּיִן הַמְשֻׁבָּח יוֹתֵר, בֵּרַךְ ׳הַטּוֹב וְהַמֵּטִיב׳, כַּמּוּבָא בַּהֲלָכָה.

הוּא חִפֵּשׂ דְּרָכִים לְהוֹדוֹת לַה׳ עַל כָּל חֶסֶד.

מִי שֶׁרוֹצֶה לְהוֹדוֹת – מוֹצֵא דֶּרֶךְ.

🍽️ סְעוּדַת הוֹדָאָה

פַּעַם יַלְדָּה אַחַת הָיְתָה חוֹלָה, וּבְחַסְדֵי שָׁמַיִם הִבְרִיאָה. רַבִּי שְׁלֹמֹה זַלְמָן יָעַץ לְמִשְׁפַּחְתָּהּ לַעֲרֹךְ סְעוּדַת הוֹדָאָה עַל הַנֵּס.

סְעוּדָה כָּזֹאת הִיא דֶּרֶךְ לְהוֹדוֹת לַה׳ בִּסְעוּדָה, יַחַד עִם הַמִּשְׁפָּחָה, בְּשִׂמְחָה.

עַל נֵס מוֹדִים גַּם בִּסְעוּדַת הוֹדָאָה.

🚪 הוֹדָאָה מִקָּרוֹב

בְּאֶמְצַע הַסְּעוּדָה נִפְתְּחָה הַדֶּלֶת – וְרַבִּי שְׁלֹמֹה זַלְמָן נִכְנַס! הַמִּשְׁפָּחָה לֹא יָדְעָה מֵרֹב שִׂמְחָה: גְּדוֹל הַדּוֹר הִטְרִיחַ אֶת עַצְמוֹ בִּשְׁבִילָם, וְהוּא בִּקֵּשׁ רְשׁוּת לֶאֱכֹל אִתָּם.

שָׁאֲלוּ אוֹתוֹ: ׳לָמָּה הִטְרַחְתָּ אֶת עַצְמְךָ? אֶפְשָׁר לְהוֹדוֹת גַּם מֵהַבַּיִת׳.

הוּא עָנָה בְּחִיּוּךְ: ׳אֵינָהּ דּוֹמָה הוֹדָאָה הָאֲמוּרָה מִקָּרוֹב לְהוֹדָאָה הָאֲמוּרָה מֵרָחוֹק׳.

הוֹדָאָה מִקָּרוֹב, יַחַד עִם אֲחֵרִים, שׁוֹנָה מֵהוֹדָאָה מֵרָחוֹק.

💎 כְּמוֹנֶה מַרְגָּלִיּוֹת

בְּנוֹ שֶׁל רַבִּי שְׁלֹמֹה זַלְמָן סִפֵּר: לִפְעָמִים רָצִיתִי לְהַכְנִיס אֵלָיו אֲנָשִׁים בְּשָׁעָה שֶׁבֵּרַךְ, כְּדֵי שֶׁיִּרְאוּ אֵיךְ מְבָרְכִים בִּרְכַּת הַמָּזוֹן.

הוּא הָיָה אוֹמֵר ׳נוֹדֶה לְךָ ה׳ אֱלֹהֵינוּ עַל שֶׁהִנְחַלְתָּ לַאֲבוֹתֵינוּ אֶרֶץ חֶמְדָּה טוֹבָה וּרְחָבָה׳ בְּצוּרָה כָּל כָּךְ מוּחָשִׁית, שֶׁנִּדְמָה לַבֵּן שֶׁזֶּה נֹסַח שֶׁלֹּא שָׁמַע מֵעוֹלָם.

גַּם כְּשֶׁמִּהֵר בַּסְּעוּדָה, בֵּרַךְ בִּמְתִינוּת וּבְאֹרֶךְ רוּחַ, כְּמוֹנֶה מַרְגָּלִיּוֹת – כְּמוֹ מִי שֶׁסּוֹפֵר אֲבָנִים יְקָרוֹת.

מְבָרְכִים לְאַט וּבְאֹרֶךְ רוּחַ, כְּמוֹ מִי שֶׁסּוֹפֵר אֲבָנִים יְקָרוֹת.

׳כְּשֶׁבֵּרַכְתִּי אֶת נוֹדֶה לְךָ הָרִאשׁוֹן, הָאֲמִירָה הָיְתָה חֲסֵרַת כַּוָּנָה. וּמִכֵּיוָן שֶׁעִנְיַן נוֹדֶה לְךָ הוּא הוֹדָאָה, בְּלִי כַּוָּנַת הַלֵּב לֹא אָמַרְתִּי הוֹדָאָה כְּלָל׳.

📜 עַל מָה מוֹדִים בְּ׳נוֹדֶה לְךָ׳?

בְּ׳נוֹדֶה לְךָ׳ אָנוּ מוֹדִים לַה׳ עַל הַרְבֵּה דְּבָרִים: עַל אֶרֶץ חֶמְדָּה טוֹבָה וּרְחָבָה שֶׁנָּתַן לַאֲבוֹתֵינוּ, וְעַל שֶׁהוֹצִיאָנוּ מִמִּצְרַיִם וּפָדָה אוֹתָנוּ מִבֵּית עֲבָדִים.

וְגַם עַל הַבְּרִית, הַתּוֹרָה, הַחֻקִּים, הַחַיִּים, וְעַל הָאֹכֶל שֶׁה׳ זָן אוֹתָנוּ ׳בְּכָל יוֹם וּבְכָל עֵת וּבְכָל שָׁעָה׳.

כְּשֶׁאוֹמְרִים אֶת הַמִּלִּים בְּכַוָּנָה, מַרְגִּישִׁים כַּמָּה הַרְבֵּה טוֹבוֹת יֵשׁ.

בְּ׳נוֹדֶה לְךָ׳ מוֹדִים עַל אֶרֶץ, גְּאֻלָּה, תּוֹרָה, חַיִּים וְאֹכֶל.

מָה לוֹקְחִים מִרָמָה א׳?

בִּרְכַּת הַמָּזוֹן הִיא הוֹדָיָה לַה׳ עַל הָאֹכֶל וְעַל כָּל הַטּוֹבוֹת. כְּשֶׁאֲנִי אוֹמֵר אוֹתָהּ בְּלֵב, אֲנִי מַרְגִּישׁ כַּמָּה ה׳ אוֹהֵב אוֹתִי וְנוֹתֵן לִי, וְאַהֲבָתִי אֵלָיו גְּדֵלָה.

9
❤️
וְאָכַלְתָּ וְשָׂבָעְתָּ וּבֵרַכְתָּ אֶת ה׳ אֱלֹהֶיךָ

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי בַּפַּעַם הָאַחֲרוֹנָה אָמַרְתִּי ׳נוֹדֶה לְךָ׳ וְחָשַׁבְתִּי עַל הַמִּלִּים?
  • עַל אֵיזוֹ שִׂמְחָה שֶׁלִּי אֲנִי יָכוֹל לְהוֹדוֹת לַה׳ בְּמִלִּים שֶׁלִּי?
  • הַשָּׁבוּעַ בְּכָל אֲרוּחָה אֲבָרֵךְ בִּרְכַּת הַמָּזוֹן לְאַט, וּבְ׳נוֹדֶה לְךָ׳ אֶחְשֹׁב עַל דָּבָר אֶחָד שֶׁאֲנִי מוֹדֶה עָלָיו, וְאֶרְשֹׁם מָה הִרְגַּשְׁתִּי.

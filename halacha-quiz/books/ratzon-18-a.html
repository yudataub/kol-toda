בּוֹנִים מִגְדָּל מִקֻּבִּיּוֹת. אִם הַקֻּבִּיּוֹת שֶׁבַּתַּחְתִּית עֲקֻמּוֹת, הַמִּגְדָּל נוֹפֵל, גַּם אִם הַקֻּבִּיּוֹת שֶׁלְמַעְלָה יָפוֹת מְאֹד.

כָּךְ גַּם בַּעֲבוֹדַת ה׳: יֵשׁ יְסוֹד שֶׁכָּל הַבִּנְיָן עוֹמֵד עָלָיו. הַיְּסוֹד הוּא רָצוֹן אֲמִתִּי לִהְיוֹת קָרוֹב לַה׳.

אִם הַיְּסוֹד חָזָק, הַבִּנְיָן חָזָק. וְאִם הוּא חַלָּשׁ, כָּל הַבִּנְיָן בְּסַכָּנָה.

בַּשִּׁעוּר הַזֶּה לוֹמְדִים שְׁלֹשָׁה שְׁלָבִים לִבְנִיַּת הַיְּסוֹד.

הַיְּסוֹד שֶׁל כָּל הַבִּנְיָן הוּא רָצוֹן אֲמִתִּי לְהִתְקָרֵב לַה׳.

כָּל מִצְוָה, כָּל תְּפִלָּה וְכָל מַעֲשֶׂה טוֹב מְקָרְבִים אוֹתָנוּ אֵלָיו.

כָּךְ הַשָּׁלָב הַשֵּׁנִי: הַיְּדִיעָה שֶׁקִּרְבַת ה׳ הִיא הַמַּטָּרָה מְמַלְּאָה אֶת הַלֵּב וְנַעֲשֵׂית הַתְּשׁוּקָה שֶׁל הַחַיִּים.

וְלֹא אוֹמְרִים אֶת זֶה רַק ׳מִן הַשָּׂפָה וְלַחוּץ׳. רוֹצִים בֶּאֱמֶת.

שָׁלָב ב׳: הָרָצוֹן לְהִתְקָרֵב לַה׳ הוֹפֵךְ לִתְשׁוּקָה שֶׁל כָּל הַחַיִּים.

הַשָּׁלָב הַשְּׁלִישִׁי: יוֹדְעִים שֶׁעֲבֵרָה מַרְחִיקָה אוֹתָנוּ מִקִּרְבַת ה׳, וְלָכֵן נִזְהָרִים מִמֶּנָּה מְאֹד.

מִי שֶׁרוֹצֶה קִרְבָה בֶּאֱמֶת, שׁוֹמֵר עַל עַצְמוֹ מִדְּבָרִים שֶׁמַּרְחִיקִים.

אֲבָל כְּשֶׁאֵין רָצוֹן, זֶה לַחַץ, כִּי עוֹשִׂים דָּבָר נֶגֶד מַה שֶׁרוֹצִים. כְּשֶׁיֵּשׁ רָצוֹן, אֵין לַחַץ כְּלָל, כִּי הָאָדָם עוֹשֶׂה מַה שֶׁהוּא רוֹצֶה.

יֶלֶד שֶׁאוֹהֵב לְשַׂחֵק בַּכַּדּוּר לֹא מַרְגִּישׁ שֶׁהוּא ׳חַיָּב׳.

כְּשֶׁיֵּשׁ רָצוֹן אֲמִתִּי, אֵין לַחַץ.

לִפְעָמִים הָרָצוֹן קַיָּם אֲבָל עוֹד לֹא חָזָק דַּיּוֹ. זֶה לֹא אוֹמֵר שֶׁלֹּא רוֹצִים בִּכְלָל, אֶלָּא שֶׁהָרָצוֹן עוֹד חַלָּשׁ.

מִי שֶׁרוֹצֶה לְהִתְקַדֵּם צָרִיךְ לְעוֹרֵר אֶת הָרָצוֹן יוֹתֵר וְיוֹתֵר.

דֶּרֶךְ פְּשׁוּטָה: לָשֶׁבֶת בְּמָקוֹם שָׁקֵט וּלְדַבֵּר עִם ה׳ בְּמִלִּים שֶׁלָּנוּ. לְמָשָׁל: ׳אַתָּה יוֹדֵעַ שֶׁאֲנִי רוֹצֶה לִהְיוֹת קָרוֹב אֵלֶיךָ, אֲבָל הָרָצוֹן שֶׁלִּי חַלָּשׁ. עֲזֹר לִי לִרְצוֹת בֶּאֱמֶת׳.

מְחַזְּקִים אֶת הָרָצוֹן כְּשֶׁמְּדַבְּרִים עִם ה׳ וּמְבַקְּשִׁים מִמֶּנּוּ.

כְּמוֹ יֶלֶד שֶׁמִּתְאַמֵּן הַרְבֵּה וְאַחַר כָּךְ נֶהֱנֶה מֵהַמִּשְׂחָק.

אֵין דָּבָר הָעוֹמֵד כְּנֶגֶד הָרָצוֹן

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָה הַדָּבָר שֶׁאֲנִי רוֹצֶה בּוֹ הֲכִי הַרְבֵּה, וּמָה הָיָה קוֹרֶה אִם הָיִיתִי רוֹצֶה כָּךְ גַּם לִהְיוֹת קָרוֹב לַה׳?
  • בְּאֵיזוֹ מִצְוָה הָרָצוֹן שֶׁלִּי חָזָק, וּבְאֵיזוֹ הוּא עוֹד חַלָּשׁ?
  • אֲצַיֵּר מַפַּת קִרְבָה: שְׁבִיל עִם שָׁלוֹשׁ תַּחֲנוֹת, וּבְכָל תַּחֲנָה אֶכְתֹּב מַה אֲנִי עוֹשֶׂה הַשָּׁבוּעַ.

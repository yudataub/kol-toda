רָמָה ב׳ – תּוֹדָה לְכָל בְּרִיָּה

הַיְאוֹר וְהֶעָפָר שֶׁהֵגֵנּוּ, הַכְּלָבִים שֶׁקִּבְּלוּ שָׂכָר, וְהַכָּרַת הַטּוֹב שֶׁבּוֹנָה אֶת הַלֵּב

📖 הַכָּרַת הַטּוֹב · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
📖

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא פּוֹגֵעַ בְּמִי שֶׁהֵיטִיב לִי 💛

📖 הַכָּרַת הַטּוֹב לַחֲבֵרִים וְלַהוֹרִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַשִׁ״י שְׁמוֹת ז׳ י״ט

רָמָה ב׳ – תּוֹדָה לְכָל בְּרִיָּה

📖 מַה אָמַר רַשִׁ״י?

בַּפָּסוּק ׳אֱמֹר אֶל אַהֲרֹן קַח מַטְּךָ וּנְטֵה יָדְךָ עַל מֵימֵי מִצְרַיִם׳, רַשִׁ״י מְבָאֵר: הַיְאוֹר הֵגֵן עַל מֹשֶׁה כְּשֶׁהֻשְׁלַךְ לְתוֹכוֹ כְּתִינוֹק, וְלָכֵן לֹא לָקָה עַל יָדוֹ – לֹא בְּדָם וְלֹא בִּצְפַרְדְּעִים.

זֶה מַשְׁמָעוּת גְּדוֹלָה: מֹשֶׁה, שֶׁהָיָה הַמַּנְהִיג, לֹא הָיָה מוּכָן לִפְגֹּעַ בְּמִי שֶׁהֵיטִיב לוֹ.

הַיְאוֹר לֹא יוֹדֵעַ וְלֹא מַרְגִּישׁ, וְעִם זֹאת הַתּוֹרָה מַקְפִּידָה.

מֹשֶׁה לֹא פָּגַע בְּמִי שֶׁהֵיטִיב לוֹ, גַּם כְּשֶׁהוּא דּוֹמֵם.

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹקֵחַ מוּסָר גַּם מִדָּבָר קָטָן 💛

📖 הַכָּרַת הַטּוֹב לַחֲבֵרִים וְלַהוֹרִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַשִׁ״י שְׁמוֹת ח׳ י״ב · שְׁמוֹת רַבָּה י׳

רָמָה ב׳ – תּוֹדָה לְכָל בְּרִיָּה

🏖️ הֶעָפָר בְּמַכַּת כִּנִּים

בְּמַכַּת כִּנִּים כָּתוּב: ׳אֱמֹר אֶל אַהֲרֹן נְטֵה אֶת מַטְּךָ וְהַךְ אֶת עֲפַר הָאָרֶץ׳. רַשִׁ״י: הֶעָפָר הֵגֵן עַל מֹשֶׁה כְּשֶׁטָּמַן בּוֹ אֶת הַמִּצְרִי, וְלָכֵן לֹא הָיָה רָאוּי שֶׁיִּלְקֶה עַל יָדוֹ.

הַמִּדְרָשׁ אוֹמֵר: ׳עָפָר שֶׁהֵגֵן עָלֶיךָ אֵינוֹ דִּין שֶׁיִּלְקֶה עַל יָדְךָ׳.

הַתּוֹרָה הִקְפִּידָה אֲפִילּוּ עַל דָּבָר שֶׁנִּרְאֶה קָטָן כָּל כָּךְ.

הַתּוֹרָה מַקְפִּידָה עַל הַכָּרַת הַטּוֹב גַּם בְּדָבָר קָטָן.

🧱

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְפַתֵּחַ אֶת הַמִּדָּה בְּכָל מָקוֹם, כְּדֵי שֶׁאַכִּיר טוֹבָה גַּם לַה׳ 💛

📖 הַכָּרַת הַטּוֹב לַחֲבֵרִים וְלַהוֹרִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רְשִׁימַת מְקוֹרוֹת ׳הַכָּרַת הַטּוֹב׳

רָמָה ב׳ – תּוֹדָה לְכָל בְּרִיָּה

🧱 גַּם לַדּוֹמֵם

הַמִּדְרָשׁ מְלַמֵּד: גַּם לַדּוֹמֵם צִוְּתָה הַתּוֹרָה לְהַכִּיר טוֹבָה, כְּדֵי לְפַתֵּחַ בָּנוּ אֶת הַמִּדָּה.

הַלֵּב אֵינוֹ מְחַלֵּק בֵּין אָדָם לְיְאוֹר: מִי שֶׁלֹּא מַכִּיר טוֹבָה לְאֶחָד מֵהֶם, מְקַלְקֵל אֶת הַמִּדָּה כֻּלָּהּ.

לָכֵן מִי שֶׁמַּכִּיר טוֹבָה בֶּאֱמֶת מֵהַלֵּב, יַכִּיר טוֹבָה גַּם לַה׳.

מִדַּת הַכָּרַת הַטּוֹב אַחַת לָאָדָם וְלַדּוֹמֵם.

3
🐕

💛 אַהֲבַת הַשֵּׁם: אֲנִי זוֹכֵר שֶׁה׳ רוֹאֶה כָּל טוֹבָה, גַּם הַקְּטַנָּה 💛

📖 הַכָּרַת הַטּוֹב לַחֲבֵרִים וְלַהוֹרִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַשִׁ״י שְׁמוֹת כ״ב ל׳ · מְכִילְתָּא

רָמָה ב׳ – תּוֹדָה לְכָל בְּרִיָּה

🐕 ׳תְּנוּ לוֹ שְׂכָרוֹ׳

רַשִׁ״י עַל הַפָּסוּק ׳לַכֶּלֶב תַּשְׁלִכוּן אֹתוֹ׳: ׳לָמָּה כָּתוּב כֶּלֶב? לְלַמֶּדְךָ שֶׁאֵין הַקָּדוֹשׁ בָּרוּךְ הוּא מְקַפֵּחַ שְׂכַר שׁוּם בְּרִיָּה׳.

בְּלֵיל מַכַּת בְּכוֹרוֹת הַכְּלָבִים לֹא נָבְחוּ, וְה׳ אָמַר: תְּנוּ לוֹ שְׂכָרוֹ.

אִם לַכֶּלֶב יֵשׁ שָׂכָר, קַל וָחֹמֶר לָאָדָם.

ה׳ מְשַׁלֵּם שָׂכָר לְכָל בְּרִיָּה עַל טוֹבָה.

4
🐦

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה גַּם עַל טוֹבָה אַחַת, וְזוֹכֵר אוֹתָהּ זְמַן רַב 💛

📖 הַכָּרַת הַטּוֹב לַחֲבֵרִים וְלַהוֹרִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, חִדּוּשִׁים לְפָרָשַׁת וָאֵרָא · הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – תּוֹדָה לְכָל בְּרִיָּה

🐦 צִפֳּרֵי שַׁבַּת שִׁירָה

הַמִּנְהָג לְפַזֵּר אֹכֶל לַצִּפֳּרִים בְּשַׁבַּת שִׁירָה בָּא מִכָּךְ שֶׁהַצִּפֳּרִים הִקְדִּימוּ לֶאֱכֹל אֶת הַמָּן שֶׁדָּתָן וַאֲבִירָם פִּזְּרוּ בְּשַׁבָּת, וּבָכָךְ הֶעִידוּ עַל אֱמֶת דִּבְרֵי מֹשֶׁה.

עַל מַעֲשֶׂה אֶחָד שֶׁל צִפֳּרִים נוֹהֲגִים לְדוֹרוֹת לְהַכִּיר טוֹבָה.

זֶה מַרְאֶה כַּמָּה מְדַקְדְּקִים בְּהַכָּרַת הַטּוֹב.

מַכִּירִים טוֹבָה לְצִפֳּרִים לְדוֹרוֹת.

🔥

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַכִּיר טוֹבָה לְהוֹרַי וְלַמּוֹרִים שֶׁלִּי, קַל וָחֹמֶר 💛

📖 הַכָּרַת הַטּוֹב לַחֲבֵרִים וְלַהוֹרִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, חִדּוּשִׁים לְפָרָשַׁת וָאֵרָא · הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – תּוֹדָה לְכָל בְּרִיָּה

🔥 קַל וָחֹמֶר

הַלֶּקַח בַּדַּף: אִם מֹשֶׁה הִכִּיר טוֹבָה לֶעָפָר, כַּמָּה צָרִיךְ אָדָם לְהַכִּיר טוֹבָה לְהוֹרָיו, לְבֵן זוּגוֹ, לְמוֹרֵהוּ וּלְשָׁכֵנוֹ. וְאִם לַכֶּלֶב יֵשׁ שָׂכָר – לָאָדָם לֹא כָּל שֶׁכֵּן.

הַיְאוֹר, הֶעָפָר וְהַכְּלָבִים לֹא מַרְגִּישִׁים, אֲבָל הַהוֹרִים וְהַמּוֹרֶה כֵּן.

וּמִי שֶׁמַּרְגִּיל אֶת הַלֵּב שֶׁלּוֹ, הוּא גַּם מִי שֶׁמַּכִּיר טוֹבָה לַה׳.

אִם לַדּוֹמֵם מַכִּירִים טוֹבָה, קַל וָחֹמֶר לָאָדָם.

רָמָה ב׳ – תּוֹדָה לְכָל בְּרִיָּה

מָה לוֹקְחִים מִרָמָה ב׳?

הַכָּרַת הַטּוֹב לַדּוֹמֵם וְלַחַי הִיא תַּרְגִּיל לַלֵּב. מִי שֶׁמְּתַרְגֵּל, רוֹאֶה אֶת טוֹבַת הַבּוֹרֵא בְּכָל מָקוֹם וְאוֹהֵב אוֹתוֹ.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • מִי מֵהַדְּבָרִים הַפְּשׁוּטִים סְבִיבִי עָזַר לִי הַיּוֹם, וְאֵיךְ אוֹדֶה לוֹ?
  • עַל אֵיזֶה דָּבָר פָּשׁוּט אֲנִי לֹא אוֹמֵר תּוֹדָה אַף פַּעַם, וְלָמָּה כְּדַאי לְהַתְחִיל?
  • אֹמַר תּוֹדָה לִשְׁלוֹשָׁה דְּבָרִים פְּשׁוּטִים (שֻׁלְחָן, כִּסֵּא, בֵּית הַסֵּפֶר), אַסְבִּיר לָמָּה, וְאֶרְשֹׁם מָה הִרְגַּשְׁתִּי.

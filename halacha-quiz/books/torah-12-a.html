כְּשֶׁהַזְּמַן נֶעֱצָר

רָמָה א׳ – שׁוֹקְעִים בַּתּוֹרָה

הָאוֹר שָׂמֵחַ עַל הַסֻּלָּם, הָרַב צְבִי יְהוּדָה וְהָרַב מָרְדֳּכַי אֱלִיָּהוּ: מָה קּוֹרֶה כְּשֶׁשְּׁקוּעִים בַּתּוֹרָה בְּאַהֲבָה

📖 אַהֲבַת הַתּוֹרָה · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
🪜

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹהֵב לְהַגִּיעַ לְסֵפֶר חָדָשׁ, וְה׳ נָתַן לִי תּוֹרָה לְגַלּוֹת 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אֹסֶף סִפּוּרֵי אַהֲבַת תּוֹרָה

רָמָה א׳ – שׁוֹקְעִים בַּתּוֹרָה

🪜 הַיֶּלֶד עַל הַסֻּלָּם

הָאוֹר שָׂמֵחַ הָיָה פַּעַם יֶלֶד. יוֹם אֶחָד הוּא נִכְנַס לְבֵית הַכְּנֶסֶת, כְּשֶׁתְּפִלַּת עַרְבִית כִּמְעַט נִגְמְרָה.

הוּא רָאָה סֵפֶר עַל מַדָּף גָּבוֹהַּ, וְעָלָה עַל סֻלָּם כְּדֵי לְהוֹרִיד אֶת הַסֵּפֶר.

מָה קָרָה כְּשֶׁהַסֵּפֶר הָיָה כְּבָר בַּיָּד?

יֶלֶד עָלָה עַל סֻלָּם לְהוֹרִיד סֵפֶר מִמַּדָּף גָּבוֹהַּ.

1
📖

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי אוֹהֵב מַשֶּׁהוּ, אֲנִי שׁוֹכֵחַ אֶת הַזְּמַן 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אֹסֶף סִפּוּרֵי אַהֲבַת תּוֹרָה

רָמָה א׳ – שׁוֹקְעִים בַּתּוֹרָה

📖 פָּתַח וְשָׁכַח

מִיָּד כְּשֶׁהַסֵּפֶר הִגִּיעַ לְיָדוֹ הוּא פָּתַח אוֹתוֹ וְהִתְחִיל לְעַיֵּן וְלִקְרֹא.

הוּא כָּל כָּךְ שָׁקַע בַּלִּמּוּד שֶׁשָּׁכַח מֵעַצְמוֹ. הוּא לָמַד שָׁעָה אַחַר שָׁעָה, כָּל הַלַּיְלָה.

כָּךְ זֶה כְּשֶׁאוֹהֲבִים: הַזְּמַן עוֹבֵר מַהֵר.

הוּא פָּתַח סֵפֶר, שָׁקַע בּוֹ וְשָׁכַח מֵעַצְמוֹ.

2
🌅

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִשְׁתַּדֵּל לִהְיוֹת שָׁקוּעַ בְּמָה שֶׁאֲנִי לוֹמֵד 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אֹסֶף סִפּוּרֵי אַהֲבַת תּוֹרָה

רָמָה א׳ – שׁוֹקְעִים בַּתּוֹרָה

🌅 הַשַּׁמָּשׁ בַּבֹּקֶר

בַּבֹּקֶר בָּא הַשַּׁמָּשׁ, פָּתַח אֶת בֵּית הַכְּנֶסֶת, וּמָצָא יֶלֶד עוֹמֵד וְלוֹמֵד עַל הַסֻּלָּם!

הַשַּׁמָּשׁ שָׁאַל: ׳מָה אַתָּה עוֹשֶׂה כָּאן?׳

וְהָאוֹר שָׂמֵחַ הַקָּטָן שָׁאַל אוֹתוֹ בִּתְמִימוּת: ׳כְּבָר אָמְרוּ עָלֵינוּ?׳ – הוּא חָשַׁב שֶׁהַתְּפִלָּה עוֹד לֹא נִגְמְרָה.

אֲנַחְנוּ לֹא צְרִיכִים לְהִשָּׁאֵר עֵרִים כָּל הַלַּיְלָה; הַסִּפּוּר מְלַמֵּד עַל הַשְּׁקִיעָה, לֹא עַל חוֹבָה.

הוּא הָיָה שָׁקוּעַ כָּל כָּךְ שֶׁלֹּא יָדַע שֶׁעָבַר לַיְלָה שָׁלֵם.

3
🕯️

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת מִי שֶׁמְּחַכֶּה וְדוֹאֵג לַאֲחֵרִים 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אֹסֶף סִפּוּרֵי אַהֲבַת תּוֹרָה

רָמָה א׳ – שׁוֹקְעִים בַּתּוֹרָה

🕯️ הָרַב צְבִי יְהוּדָה מְחַכֶּה

הָרַב צְבִי יְהוּדָה הַכֹּהֵן קוּק הָיָה גָּדוֹל בַּתּוֹרָה. פַּעַם הָיָה תַּלְמִיד שֶׁגָּר בְּבֵיתוֹ וְהָיָה צָרִיךְ לַחְזֹר מְאֻחָר בַּלַּיְלָה.

הָרַב לֹא הָלַךְ לִישֹׁן. הוּא נִשְׁאַר לִלְמֹד עַד שֶׁהַתַּלְמִיד יַחְזֹר.

כָּךְ הוּא הִשְׁגִּיחַ עַל הַתַּלְמִיד וְגַם לָמַד.

הָרַב נִשְׁאַר לִלְמֹד עַד שֶׁהַתַּלְמִיד יַחְזֹר.

4
🌄

💛 אַהֲבַת הַשֵּׁם: הַלִּמּוּד שֶׁאֲנִי אוֹהֵב גּוֹרֵם לִי לְהַרְגִּישׁ שֶׁהַזְּמַן עוֹמֵד 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אֹסֶף סִפּוּרֵי אַהֲבַת תּוֹרָה

רָמָה א׳ – שׁוֹקְעִים בַּתּוֹרָה

🌄 מָה הַשָּׁעָה?

הַתַּלְמִיד חָזַר... רַק בַּבֹּקֶר!

הָרַב צְבִי יְהוּדָה הָיָה שָׁקוּעַ כָּל כָּךְ בַּלִּמּוּד שֶׁלֹּא שָׂם לֵב שֶׁעָבַר זְמַן רַב. הוּא חָשַׁב שֶׁעֲדַיִן לַיְלָה.

הוּא שָׁאַל אֶת הַתַּלְמִיד: ׳מָה הַשָּׁעָה?׳ וְזֵרֵז אוֹתוֹ לָלֶכֶת לִישֹׁן.

הָרַב חָשַׁב שֶׁעֲדַיִן לַיְלָה, וְהוּא כְּבָר הָיָה בֹּקֶר.

5
📚

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִלְמֹד בְּרִכּוּז כָּזֶה, וְאֶשְׁמֹר שֶׁקֶט כְּשֶׁאֲחֵרִים לוֹמְדִים 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אֲבִיהֶם שֶׁל יִשְׂרָאֵל לִילָדִים

רָמָה א׳ – שׁוֹקְעִים בַּתּוֹרָה

📚 אַרְבַּע שָׁעוֹת שֶׁל סִדּוּר

הָרַב מָרְדֳּכַי אֱלִיָּהוּ הָיָה לוֹמֵד שָׁעוֹת רַבּוֹת, וְהָיָה שָׁקוּעַ וּמְרֻכָּז מְאֹד.

פַּעַם נִכְנַס הָעוֹזֵר שֶׁלּוֹ לְסַדֵּר אֶת הַסְּפָרִים בַּחֶדֶר. הוּא רָאָה שֶׁהָרַב לוֹמֵד, וְלֹא רָצָה לְהַפְרִיעַ וְלוֹמַר שָׁלוֹם. הוּא הִתְחִיל לְסַדֵּר בְּשֶׁקֶט.

אַחֲרֵי אַרְבַּע שָׁעוֹת שֶׁבָּהֶן הִזִּיז עֲרֵמוֹת סְפָרִים, הַרַב שָׂם לֵב וְאָמַר לוֹ: ׳בָּרוּךְ הַבָּא, אָסָף׳.

הָעוֹזֵר סִדֵּר אַרְבַּע שָׁעוֹת, וְהָרַב שָׂם לֵב רַק אַחַר כָּךְ.

6
❤️

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לֶאֱהֹב אֶת הַתּוֹרָה כָּל כָּךְ שֶׁאֶשְׁכַּח אֶת הַשָּׁעוֹן 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אֹסֶף סִפּוּרֵי אַהֲבַת תּוֹרָה

רָמָה א׳ – שׁוֹקְעִים בַּתּוֹרָה

❤️ מַה הַסּוֹד?

שְׁלוֹשָׁה סִפּוּרִים, וְאוֹתוֹ סוֹד: כְּשֶׁאוֹהֲבִים, שׁוֹקְעִים. כְּשֶׁיֶּלֶד שָׁקוּעַ בְּמִשְׂחָק אָהוּב, הוּא שׁוֹכֵחַ אֶת הַשָּׁעוֹן; כָּךְ גַּם אֵלּוּ שֶׁאוֹהֲבִים תּוֹרָה.

לֹא הָיָה לָהֶם צֹרֶךְ בְּתִזְכֹּרֶת. הָאַהֲבָה הִיא שֶׁהוֹלִיכָה אוֹתָם.

וְהָאַהֲבָה הַזֹּאת שֶׁל תּוֹרָה הִיא חֵלֶק מֵאַהֲבָתָם לַה׳ שֶׁנָּתַן אוֹתָהּ.

אַהֲבָה אֲמִתִּית גּוֹרֶמֶת לִשְׁקֹעַ וְלִשְׁכֹּחַ אֶת הַזְּמַן.

7
😴

💛 אַהֲבַת הַשֵּׁם: אֲנִי שׁוֹמֵר עַל הַגּוּף שֶׁנָּתַן לִי ה׳ כְּדֵי לִלְמֹד 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אֹסֶף סִפּוּרֵי אַהֲבַת תּוֹרָה

רָמָה א׳ – שׁוֹקְעִים בַּתּוֹרָה

😴 גַּם לָדַעַת לִישֹׁן

שִׁימוּ לֵב: הָרַב צְבִי יְהוּדָה, שֶׁנִּשְׁאַר עֵר כָּל הַלַּיְלָה, זֵרֵז אֶת הַתַּלְמִיד לָלֶכֶת לִישֹׁן.

הוּא לֹא דָּרַשׁ מֵהַתַּלְמִיד לְהִשָּׁאֵר עֵר גַּם הוּא. הוּא דָּאַג לַשֵּׁנָה וְלַבְּרִיאוּת שֶׁלּוֹ.

שְׁקִיעָה בְּתוֹרָה אֵינָהּ אוֹמֶרֶת לְהַזִּיק לַגּוּף; אֲנִי לוֹמֵד, וְאַחַר כָּךְ יָשֵׁן לְמַעַן לִלְמֹד מָחָר.

אוֹהֲבִים תּוֹרָה, וְגַם דּוֹאֲגִים לַשֵּׁנָה וְלַבְּרִיאוּת.

8
💡

רָמָה א׳ – שׁוֹקְעִים בַּתּוֹרָה

מָה לוֹקְחִים מִרָמָה א׳?

אַהֲבָה לַתּוֹרָה גּוֹרֶמֶת לִשְׁקֹעַ וְלִשְׁכֹּחַ אֶת הַזְּמַן. הָאַהֲבָה הַזֹּאת הִיא חֵלֶק מֵהָאַהֲבָה לַה׳ שֶׁנָּתַן לָנוּ אֶת הַתּוֹרָה, וְכָל פַּעַם שֶׁאֲנִי לוֹמֵד, אֲנִי מִתְקָרֵב אֵלָיו.

9
❤️
שָׂשׂ אָנֹכִי עַל אִמְרָתֶךָ כְּמוֹצֵא שָׁלָל רָב

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי הָיִיתִי שָׁקוּעַ כָּל כָּךְ בְּמַשֶּׁהוּ שֶׁאָהַבְתִּי שֶׁשָּׁכַחְתִּי כַּמָּה זְמַן עָבַר?
  • מִי סְבִיבִי שָׁקוּעַ בְּלִמּוּד אוֹ בְּעֵסֶק חָשׁוּב, וְאֵיךְ אֲכַבֵּד אוֹתוֹ וְלֹא אַפְרִיעַ לוֹ?
  • אֶבְחַר סֵפֶר אוֹ שִׁעוּר שֶׁאֲנִי אוֹהֵב, אֶקְרָא חֲמֵשׁ־עֶשְׂרֵה דַּקּוֹת בְּשֶׁקֶט, אֲנַחֵשׁ כַּמָּה זְמַן עָבַר – וְאָז אֶבְדֹּק בַּשָּׁעוֹן.

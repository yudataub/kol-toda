🥁 תֹּף מִרְיָם

בַּגַּן יֵשׁ תֹּף קָטָן שֶׁהַיְּלָדִים שָׁרִים בּוֹ, וְקוֹרְאִים לוֹ ׳תֹּף מִרְיָם׳. זֶה שֵׁם שֶׁקַּיָּם רַק בְּיִשְׂרָאֵל – מִי שֶׁיִּכָּנֵס לַחֲנוּת צִיּוּד לְגַנִּים בְּחוּץ לָאָרֶץ וִיבַקֵּשׁ ׳תֹּף מִרְיָם׳, לֹא יָבִינוּ אוֹתוֹ.

לָמָּה? כִּי מִרְיָם הַנְּבִיאָה שָׁרָה בְּתֻפִּים. הִיא הָאֵם שֶׁל הַשִּׁירָה בְּיִשְׂרָאֵל.

הַתֹּף הַקָּטָן נִקְרָא עַל שֵׁם מִרְיָם, אֵם הַשִּׁירָה.

🌊

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׁר לַה׳ עַל הַנִּסִּים, כְּמוֹ מִרְיָם 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁמוֹת ט״ו כ׳־כ״א

רָמָה א׳ – אֵם הַשִּׁירָה

🌊 אֵם הַשִּׁירָה

אַחֲרֵי קְרִיעַת יַם סוּף כָּתוּב: ׳וַתִּקַּח מִרְיָם הַנְּבִיאָה… אֶת הַתֹּף בְּיָדָהּ, וַתֵּצֶאןָ כָל הַנָּשִׁים אַחֲרֶיהָ בְּתֻפִּים וּבִמְחֹלֹת׳.

מִרְיָם עָנְתָה לָהֶן: ׳שִׁירוּ לַה׳ כִּי גָאֹה גָּאָה׳. כָּל הַנְּשָׁמָה שֶׁלָּהּ הָיְתָה שִׁירָה, שִׂמְחָה וְהוֹדָיָה, וְהִיא הוֹבִילָה אֶת הַנָּשִׁים בְּשִׁירָה מִיָּד אַחֲרֵי הַנֵּס.

מִרְיָם הוֹבִילָה בְּשִׁירָה אַחֲרֵי קְרִיעַת יַם סוּף.

🧺

💛 אַהֲבַת הַשֵּׁם: אֲנִי דּוֹאֵג לַאֲחִים וְלַחֲבֵרִים שֶׁלִּי כִּי אֲנִי אוֹהֵב אוֹתָם 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁמוֹת ב׳ ד׳ · חֲזַ״ל

רָמָה א׳ – אֵם הַשִּׁירָה

🧺 הָאָחוֹת שֶׁאָהֲבָה

כְּשֶׁמֹּשֶׁה הָיָה תִּינוֹק בַּתֵּבָה עַל הַנִּיל, כָּתוּב: ׳וַתֵּתַצַּב אֲחֹתוֹ מֵרָחֹק לְדֵעָה מַה יֵּעָשֶׂה לוֹ׳.

חֲזַ״ל מְסַפְּרִים שֶׁמִּרְיָם גַּם שִׁכְנְעָה אֶת אָבִיהָ, עַמְרָם, לַחֲזֹר אֶל יוֹכֶבֶד – וְכָךְ נוֹלַד מֹשֶׁה. כָּל חַיֶּיהָ הִיא דָּאֲגָה לוֹ וְאָהֲבָה אוֹתוֹ.

מִרְיָם אָהֲבָה אֶת מֹשֶׁה וְדָאֲגָה לוֹ מִיּוֹם לֵדָתוֹ.

⚖️

💛 אַהֲבַת הַשֵּׁם: אֲנִי זוֹכֵר שֶׁלַּפֶּה יֵשׁ כֹּחַ לִשְׁנֵי כִּוּוּנִים 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דְּבָרִים כ״ד ט׳ · שִׁעוּרֵי הָרַב

רָמָה א׳ – אֵם הַשִּׁירָה

⚖️ פַּעֲמַיִם בַּתּוֹרָה

כְּשֶׁמְּחַפְּשִׂים אֶת מִרְיָם בַּתּוֹרָה, בְּמַעֲשֶׂה שֶׁהִיא עָשְׂתָה בְּעַצְמָהּ, מוֹצְאִים שְׁנֵי מְקוֹמוֹת בִּלְבַד: שִׁירַת הַיָּם, וְהַדִּבּוּר עַל מֹשֶׁה.

עַל הַשֵּׁנִי כָּתוּב: ׳זָכוֹר אֵת אֲשֶׁר עָשָׂה ה׳ אֱלֹהֶיךָ לְמִרְיָם בַּדֶּרֶךְ׳. וְהַשְּׁאֵלָה שֶׁל הַשִּׁעוּר: לָמָּה דַּוְקָא אֵם הַשִּׁירָה הִיא גַּם הַדֻּגְמָה שֶׁהַתּוֹרָה בָּחֲרָה כְּדֵי לְלַמֵּד עַל הַפֶּה?

מִרְיָם נִזְכֶּרֶת בְּמַעֲשֶׂה שֶׁלָּהּ פַּעֲמַיִם: שִׁירָה וְדִבּוּר.

⏳

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְדַבֵּר בִּזְהִירוּת, כִּי הַדִּבּוּר הוּא דָּבָר גָּדוֹל 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בַּמִּדְבָּר י״ב א׳ וט״ו

רָמָה א׳ – אֵם הַשִּׁירָה

⏳ הַמַּחֲנֶה מְחַכֶּה

מִרְיָם וְאַהֲרֹן דִּבְּרוּ עַל מֹשֶׁה, וּמִרְיָם נִסְגְּרָה מִחוּץ לַמַּחֲנֶה שִׁבְעָה יָמִים. כָּתוּב: ׳וְהָעָם לֹא נָסַע עַד הֵאָסֵף מִרְיָם׳.

כָּל הַמַּחֲנֶה חִכָּה, וְכֻלָּם יָדְעוּ לָמָּה. כְּמוֹ שֶׁכֻּלָּנוּ שׁוֹאֲלִים כְּשֶׁנִּתְקָעִים בְּפְקָק, כָּאן שִׁבְעָה יָמִים הָעָם עָמַד. הַתּוֹרָה רוֹצָה שֶׁנָּבִין כַּמָּה חָמוּר לְפַגֵּעַ בְּדִבּוּר.

הָעָם חִכָּה שִׁבְעָה יָמִים, כְּדֵי שֶׁנָּבִין כַּמָּה חָמוּר הַדִּבּוּר.

🥁

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִשְׁתַּמֵּשׁ בַּפֶּה שֶׁלִּי לְעוֹדֵד וְלָשִׁיר 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׂפַת אֱמֶת · שִׁעוּרֵי הָרַב

רָמָה א׳ – אֵם הַשִּׁירָה

🥁 לָמָּה דַּוְקָא מִרְיָם?

מִי שֶׁעוֹמֵד בְּגֹבַהּ גָּדוֹל, נִמְדָּד בִּדְקָדּוּק גָּדוֹל. כָּל מַהוּתָהּ שֶׁל מִרְיָם הָיְתָה שִׁירָה וְהוֹדָיָה, וְתָמִיד עוֹדְדָה וְשִׂמְּחָה.

לָכֵן אֶצְלָהּ כָּל סְטִיָּה קְטַנָּה הִיא כְּמוֹ רְעִידַת אֲדָמָה: הִיא פּוֹגַעַת בַּשֹּׁרֶשׁ. וְיֵשׁ מֵכָּאן לֶקַח לְכֻלָּנוּ: אוֹתוֹ פֶּה שֶׁיָּכוֹל לָשִׁיר, לְבָרֵךְ וּלְעוֹדֵד – יָכוֹל גַּם לִפְגֹּעַ.

לַפֶּה כֹּחַ לִשְׁנֵי הַכִּוּוּנִים: לָשִׁיר אוֹ לִפְגֹּעַ.

💛 אַהֲבַת הַשֵּׁם: אֲנִי דָּן לְכַף זְכוּת וְלֹא מַחְלִיט מִיָּד שֶׁחָבֵר עָשָׂה רַע 💛

רָמָה א׳ – אֵם הַשִּׁירָה

🌧️ הָרֶכֶב בַּגֶּשֶׁם

אָדָם קָבַע עִם חָבֵר שֶׁיָּבוֹא לָקַחַת אוֹתוֹ בִּמְכוֹנִית. יָרַד גֶּשֶׁם חָזָק, וּמְכוֹנִית עָצְרָה לְיַד הַבַּיִת. הוּא דָּפַק עַל הַחַלּוֹן וְדָפַק, וְהַנֶּהָג לֹא פָּתַח. הוּא עָמַד בַּגֶּשֶׁם, נִרְטַב וְלֹא הֵבִין: ׳לָמָּה הוּא לֹא פּוֹתֵחַ לִי?׳

לְמָחֳרָת הִתְבָּרֵר שֶׁהֶחָבֵר בִּכְלָל לֹא הִגִּיעַ! הַנֶּהָג הָיָה אָדָם אַחֵר לְגַמְרֵי. טוֹב שֶׁלֹּא צָעַק עָלָיו.

גַּם כְּשֶׁנִּרְאֶה דָּבָר רַע – בּוֹדְקִים, כִּי לִפְעָמִים אֵין שָׁם כְּלוּם.

רָמָה א׳ – אֵם הַשִּׁירָה

🎤 שׁוֹאֲלִים לִפְנֵי שֶׁמַּחְלִיטִים

מִשִּׁעוּר עַל מִרְיָם לוֹמְדִים: כְּשֶׁנִּרְאֶה לְךָ שֶׁחָבֵר עָשָׂה מַשֶּׁהוּ לֹא בְּסֵדֶר – לֹא מַחְלִיטִים לְבַד, הוֹלְכִים וְשׁוֹאֲלִים אוֹתוֹ. כִּמְעַט תָּמִיד מִתְבָּרֵר שֶׁהוּא לֹא הִתְכַּוֵּן, אוֹ שֶׁלֹּא הֵבַנּוּ.

וְכָל יוֹם אֲנִי בּוֹחֵר אֵיזֶה פֶּה אֲנִי מֵכִין לְעַצְמִי: פֶּה שֶׁל שִׁירָה, אוֹ פֶּה שֶׁל תְּלוּנָה.

שׁוֹאֲלִים לִפְנֵי שֶׁמַּחְלִיטִים, וּבוֹחֲרִים פֶּה שֶׁל שִׁירָה.

מָה לוֹקְחִים מִרָמָה א׳?

מִרְיָם הָיְתָה אֵם הַשִּׁירָה, וְלִמְּדָה אוֹתָנוּ שֶׁלַּפֶּה יֵשׁ כֹּחַ גָּדוֹל. כְּשֶׁאֲנִי בּוֹחֵר פֶּה שֶׁל שִׁירָה, אֲנִי מוֹדֶה לַה׳, וְאַהֲבָתִי אֵלָיו גְּדֵלָה.

🤔 חוֹשְׁבִים עַל זֶה

  • אֵיזֶה מִשְׁפָּט שֶׁל שִׁירָה אַגִּיד הַיּוֹם בַּהַפְסָקָה?
  • מָתַי הַשָּׁבוּעַ הֶחְלַטְתִּי עַל חָבֵר לִפְנֵי שֶׁשָּׁאַלְתִּי אוֹתוֹ, וְאֵיךְ אֶשְׁאַל בַּפַּעַם הַבָּאָה?
  • אֶכְתֹּב חָרוּז קָצָר עַל ׳פֶּה שֶׁל שִׁירָה׳, אֹמַר אוֹתוֹ בְּקוֹל, וְאֶשְׁאַל חָבֵר אֶחָד לִפְנֵי שֶׁאֲנִי מַחְלִיט עָלָיו.

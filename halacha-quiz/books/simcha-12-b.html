הַפַּעַם אוֹדֶה אֶת ה׳

הַמַּתָּנָה שֶׁל יוֹם הַהוּלֶדֶת

רָמָה ב׳ – מַתְּנַת הַחַיִּים

הַמַּתָּנָה הַגְּדוֹלָה בְּיוֹתֵר שֶׁקִּבַּלְנוּ הִיא עֶצֶם הַחַיִּים. לָמָּה יוֹם הַהוּלֶדֶת הוּא יוֹם שֶׁל שִׂמְחָה, וּמָה הָעֲנִי שֶׁהִתְלוֹנֵן בַּמָּטוֹס מְלַמֵּד אוֹתָנוּ

📖 שִׂמְחָה · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה לַה׳ וְשָׂמֵחַ בְּיוֹם הַהוּלֶדֶת שֶׁלִּי 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשְׁנָה בְּרוּרָה רי״ח ס״ק ל״ב · בְּרֵאשִׁית מ׳ כ׳

רָמָה ב׳ – מַתְּנַת הַחַיִּים

📅 מִצְוָה שֶׁכֻּלָּם מַסְכִּימִים עָלֶיהָ

מִי שֶׁנַּעֲשָׂה לוֹ נֵס – הַמִּשְׁנָה בְּרוּרָה מְלַמֶּדֶת – רָאוּי שֶׁבְּכָל שָׁנָה בְּאוֹתוֹ יוֹם יִתְבּוֹדֵד לְהוֹדוֹת לַה׳, לִשְׂמֹחַ וּלְסַפֵּר חַסְדּוֹ. וְטוֹב לְהַפְרִישׁ צְדָקָה כְּפִי יְכָלְתּוֹ.

בְּיוֹם הַהוּלֶדֶת נַעֲשָׂה לָאָדָם נֵס, שֶׁנּוֹשַׁע וְנוֹלַד בְּשָׁלוֹם. וְיֵשׁ עוֹד סִבָּה לְהוֹדוֹת: שֶׁנּוֹלַד כִּיהוּדִי.

יֵשׁ שֶׁאוֹמְרִים שֶׁחֲגִיגָה גְּדוֹלָה אֵינָהּ מִדַּרְכֵּנוּ, כִּי מָצָאנוּ ׳יוֹם הֻלֶּדֶת אֶת פַּרְעֹה׳. אֲבָל עַל הַהוֹדָיָה וְהַשִּׂמְחָה שֶׁנּוֹלַדְנוּ כֻּלָּם מַסְכִּימִים.

בְּיוֹם הַהוּלֶדֶת: הוֹדָיָה וְשִׂמְחָה, וְלֹא רַק מְסִבָּה.

1

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא דּוֹרֵשׁ עוֹד, אֲנִי מוֹדֶה עַל מָה שֶׁקִּבַּלְתִּי 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מָשָׁל · שִׂמְחַת יוֹם הַהוּלֶדֶת

רָמָה ב׳ – מַתְּנַת הַחַיִּים

🏠 הָעָנִי שֶׁהִתְלוֹנֵן

בַּמָּשָׁל, הַנָּדִיב נָתַן לֶעָנִי בַּיִת מְפֹאָר וְטִיסָה. בַּמָּטוֹס הֶעָנִי הִתְלוֹנֵן שֶׁהַכִּסֵּא וְהָאֹכֶל אֵינָם מֻשְׁלָמִים, וְאָמַר שֶׁהַנָּדִיב הָיָה צָרִיךְ לְסַדֵּר הַכֹּל טוֹב יוֹתֵר.

זֶה מְכֹעָר: לִדְרשׁ בְּעַזּוּת תּוֹסֶפֶת נוֹחִיּוּת, כְּאִלּוּ הַכֹּל מַגִּיעַ לוֹ.

וְכָךְ גַּם חַיֵּינוּ הֵם ״נְסִיעָה״ אֶל יַעַד נִפְלָא, הָעוֹלָם הַבָּא. מִי שֶׁמַּרְגִּישׁ תָּמִיד שֶׁ׳מַגִּיעַ לוֹ יוֹתֵר׳ דּוֹמֶה לְאוֹתוֹ עָנִי.

תְּלוּנוֹת עַל פְּרָטִים בְּמַתָּנָה גְּדוֹלָה הֵן חֻצְפָּה.

2

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְפָרֵק אֶת הַמַּחְסוֹם בְּכָךְ שֶׁאֲנִי זוֹכֵר שֶׁהַחַיִּים הֵם מַתָּנָה 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – מַתְּנַת הַחַיִּים

🚧 הַמַּחְסוֹם הָעִקָּרִי

הַמַּחְסוֹם שֶׁחוֹסֵם אֶת הַהִתְפַּעֲלוּת וְאֶת הַהַכָּרָה בְּטוֹב ה׳ הוּא הַמַּחֲשָׁבָה ׳הַכֹּל מַגִּיעַ לִי׳. כְּשֶׁאָדָם רָגִיל לְדָבָר, נִדְמֶה לוֹ שֶׁזּוֹ זְכוּתוֹ לְקַבֵּל אוֹתוֹ, וְאֵין לוֹ סִבָּה לְהוֹדוֹת.

הַתְּשׁוּבָה: הַמַּתָּנָה הַגְּדוֹלָה בְּיוֹתֵר הִיא עֶצֶם הַחַיִּים, וְהִיא לֹא ״הִגִּיעָה לָנוּ״. מִי שֶׁמֵּבִין זֹאת מוֹדֶה גַּם כְּשֶׁלֹּא הַכֹּל מְסֻדָּר כְּרָצוֹן, כְּמוֹ מִי שֶׁעָמַד בַּמָּטוֹס וְהִגִּיעַ לַיַּעַד.

הַתְּשׁוּבָה לְ׳מַגִּיעַ לִי׳ הִיא הַמַּתָּנָה שֶׁל עֶצֶם הַחַיִּים.

3

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ עַל הַחַיִּים וְאוֹהֵב אֶת ה׳ שֶׁנָּתַן אוֹתָם 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הֶחָפֵץ חַיִּים, שֵׁם עוֹלָם א׳ כ׳ · דְּבָרִים ל׳ ו׳

רָמָה ב׳ – מַתְּנַת הַחַיִּים

❤️ לִשְׂמֹחַ בִּימֵי הַחַיִּים

הֶחָפֵץ חַיִּים כּוֹתֵב שֶׁמַּה מְּאֹד צָרִיךְ הָאָדָם לִשְׂמֹחַ בִּימֵי הַחַיִּים שֶׁנָּתַן לוֹ ה׳, כִּי עַל יָדָם יָכוֹל לְהַשִּׂיג עֹנֶג נִצְחִי. וְכַאֲשֶׁר יִתְבּוֹנֵן בָּזֶה, יָבוֹא לֶאֱהֹב אֶת ה׳ אַהֲבָה עַזָּה עַל הַחַיִּים.

וְכָךְ מִתְבָּאָר הַפָּסוּק ׳וּמָל ה׳ אֱלֹהֶיךָ אֶת לְבָבְךָ … לְאַהֲבָה אֶת ה׳ אֱלֹהֶיךָ בְּכָל לְבָבְךָ וּבְכָל נַפְשְׁךָ לְמַעַן חַיֶּיךָ׳.

שִׂמְחָה עַל הַחַיִּים מְבִיאָה אַהֲבַת ה׳.

4

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְהַכִּיר בְּעֵרֶךְ הַתּוֹרָה וְהַמִּצְווֹת וְלִשְׂמֹחַ בָּהֶן 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הֶחָפֵץ חַיִּים, שֵׁם עוֹלָם א׳ כ׳

רָמָה ב׳ – מַתְּנַת הַחַיִּים

💰 הַיּוֹם בָּאוֹצָר

בַּמָּשָׁל שֶׁל הֶחָפֵץ חַיִּים הָאִישׁ אָמַר שֶׁהַיּוֹם הֶחָבִיב עָלָיו הוּא הַיּוֹם בָּאוֹצָר: רָעֵב וְלָבוּשׁ פָּשׁוּט, אֲבָל בְּכָל רֶגַע מָצָא כְּלִי יָקָר וְיָדַע שֶׁהוּא ׳עוֹלֶה לְמַעְלָה׳.

כָּךְ גַּם הָאוֹצָר שֶׁלָּנוּ הוּא הַתּוֹרָה וְהַמִּצְווֹת, שֶׁכָּל פְּרָט קָטָן מֵהֶן שָׁוֶה יוֹתֵר מִכָּל אַבְנֵי הַחֵן בָּעוֹלָם. מִי שֶׁיָּכוֹל לְלַקֵּט אוֹתָם, שָׂמֵחַ גַּם בְּיָמִים שֶׁאֵינָם קַלִּים.

הֶחָפֵץ חַיִּים מוֹסִיף: לִפְעָמִים אֵין אָנוּ מַכִּירִים בְּעֶרֶךְ הַתּוֹרָה וְהַמִּצְווֹת, וְלָכֵן קָשֶׁה לָנוּ לִשְׂמֹחַ.

מִי שֶׁמַּכִּיר בְּעֵרֶךְ הָאוֹצָר, שָׂמֵחַ בְּכָל רֶגַע.

5

💛 אַהֲבַת הַשֵּׁם: אֲנִי יוֹדֵעַ שֶׁהַהוֹרִים שֶׁלִּי שְׂמֵחִים שֶׁנּוֹלַדְתִּי 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּרֵאשִׁית כ״ט ל״ה

רָמָה ב׳ – מַתְּנַת הַחַיִּים

👪 לֵאָה: הַפַּעַם אוֹדֶה

כְּשֶׁנּוֹלַד לְלֵאָה אִמֵּנוּ בְּנָהּ הָרְבִיעִי אָמְרָה: ׳הַפַּעַם אוֹדֶה אֶת ה׳׳ וְקָרְאָה אֶת שְׁמוֹ יְהוּדָה. הִיא הוֹדְתָה עַל כָּל יֶלֶד שֶׁזָּכְתָה בּוֹ, וְעָשְׂתָה מִזֶּה הוֹדָיָה תְּמִידִית.

גַּם הַהוֹרִים רְאוּיִים לְהוֹדוֹת לַה׳ בְּיוֹם הַהוּלֶדֶת עַל הַמַּתָּנָה הַנִּפְלָאָה שֶׁל יֶלֶד יְהוּדִי.

וְגַם אֲנִי יָכוֹל לִשְׂמֹחַ וּלְהוֹדוֹת עַל עַצְמִי.

גַּם הַהוֹרִים מוֹדִים עַל כָּל יֶלֶד.

מָה לוֹקְחִים מִרָמָה ב׳?

מִי שֶׁמַּכִּיר בְּכָךְ שֶׁהַחַיִּים הֵם מַתָּנָה, מַרְגִּישׁ כַּמָּה ה׳ אוֹהֵב אוֹתוֹ. לָכֵן יוֹם הַהוּלֶדֶת הוּא לֹא רַק יוֹם לַעֲשׂוֹת מְסִבָּה, אֶלָּא יוֹם לֶאֱהֹב אֶת ה׳ שֶׁנָּתַן לָנוּ אֶת הַכֹּל.

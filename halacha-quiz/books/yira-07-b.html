רָמָה ב׳ – מִיִּרְאָה לְאַהֲבָה

📚 הִלְכוֹת תְּשׁוּבָה י׳ א׳

הָרַמְבַּ״ם כּוֹתֵב: ׳אַל יֹאמַר אָדָם: הֲרֵינִי עוֹשֶׂה מִצְווֹת הַתּוֹרָה וְעוֹסֵק בְּחָכְמָתָהּ כְּדֵי שֶׁאֲקַבֵּל כָּל הַבְּרָכוֹת, אוֹ כְּדֵי שֶׁאֶזְכֶּה לְחַיֵּי הָעוֹלָם הַבָּא, וְאֶפְרשׁ מִן הָעֲבֵרוֹת כְּדֵי שֶׁאֶנָּצֵל מִן הַקְּלָלוֹת׳.

אֵין רָאוּי לַעֲבֹד אֶת ה׳ עַל הַדֶּרֶךְ הַזֶּה. הָעוֹבֵד כָּךְ עוֹבֵד מִיִּרְאָה, וְזֹאת לֹא מַעֲלַת הַנְּבִיאִים וְלֹא מַעֲלַת הַחֲכָמִים.

אֲבָל מְחַנְּכִים אֶת הַקְּטַנִּים לַעֲבֹד מִיִּרְאָה עַד שֶׁתִּרְבֶּה דַּעְתָּם וְיַעַבְדוּ מֵאַהֲבָה.

עֲבוֹדָה רַק מִפַּחַד אוֹ בִּשְׁבִיל פְּרָס הִיא לֹא הַמַּעֲלָה הַגְּבוֹהָה.

רָמָה ב׳ – מִיִּרְאָה לְאַהֲבָה

💞 הִלְכוֹת תְּשׁוּבָה י׳ ב׳

הָעוֹבֵד מֵאַהֲבָה עוֹסֵק בַּתּוֹרָה וְהוֹלֵךְ בִּנְתִיבוֹת הַחָכְמָה לֹא מִפְּנֵי דָּבָר בָּעוֹלָם: לֹא מִפְּנֵי יִרְאַת הָרָעָה וְלֹא כְּדֵי לָרֶשֶׁת הַטּוֹבָה. הוּא עוֹשֶׂה אֶת הָאֱמֶת מִפְּנֵי שֶׁהִיא אֱמֶת, וְסוֹף הַטּוֹבָה לָבוֹא בִּגְלָלָהּ.

מַעֲלָה זוֹ גְּדוֹלָה מְאֹד, וְאֵין כָּל חָכָם זוֹכֶה לָהּ. הִיא מַעֲלַת אַבְרָהָם אָבִינוּ, שֶׁה׳ קְרָאוֹ אוֹהֲבוֹ, וְהִיא הַמַּעֲלָה שֶׁצִּוָּנוּ ה׳ עַל יְדֵי מֹשֶׁה: ׳וְאָהַבְתָּ אֵת ה׳ אֱלֹהֶיךָ׳.

אַהֲבָה גְּמוּרָה: עוֹשִׂים אֶת הָאֱמֶת מִפְּנֵי שֶׁהִיא אֱמֶת.

🤔

💛 אַהֲבַת הַשֵּׁם: אֲנִי יוֹדֵעַ שֶׁאֲנִי בְּאֶמְצַע הַדֶּרֶךְ, וְאֲנִי מַמְשִׁיךְ לַעֲלוֹת 💛

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרַב יְהוּדָה טָאוּב · רַמְבַּ״ם הִלְכוֹת תְּשׁוּבָה י׳

רָמָה ב׳ – מִיִּרְאָה לְאַהֲבָה

🤔 סְתִירָה אוֹ מַדְרֵגוֹת?

הַלּוֹמֵד אֶת הָרַמְבַּ״ם מַרְגִּישׁ סְתִירָה: בַּהֲלָכָה א׳ הַפַּחַד מֵעֹנֶשׁ הוּא דַּרְגָּה נְמוּכָה, וּבַהֲלָכָה ב׳ הָאַהֲבָה מַעֲלָה גְּדוֹלָה. לָכְאוֹרָה הֵן חוֹלְקוֹת.

וְצָרִיךְ לוֹמַר: יֵשׁ הַרְבֵּה דַּרְגּוֹת בָּאֶמְצַע. בַּהֲלָכָה א׳ מְדֻבָּר בְּמִי שֶׁעוֹבֵד אֶת ה׳ רַק מֵחֲשַׁשׁ עֹנֶשׁ, דַּרְגָּה נְמוּכָה. בַּהֲלָכָה ב׳ מְדֻבָּר בְּמִי שֶׁעוֹסֵק בַּתּוֹרָה מֵאַהֲבָה גְּמוּרָה, דַּרְגָּה גְּבוֹהָה מְאֹד.

וּבֵינֵיהֶן עוֹמְדוֹת מַדְרֵגוֹת, וְיִרְאַת הָרוֹמְמוּת הִיא אַחַת מֵהֶן.

שְׁתֵּי הַהֲלָכוֹת מְדַבְּרוֹת עַל שְׁתֵּי דַּרְגּוֹת שׁוֹנוֹת בְּאוֹתָהּ סֻלָּם.

💛 אַהֲבַת הַשֵּׁם: אֲנִי כְּבָר קָרוֹב לַה׳, וְרוֹצֶה לִשְׁמֹר עַל הַקִּרְבָה הַזֹּאת 💛

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרַב פִּינְקֶל · תְּהִלִּים נ״א י״ג

רָמָה ב׳ – מִיִּרְאָה לְאַהֲבָה

🔑 בֶּאוֹר שֶׁל הָרַב פִּינְקֶל

שָׁמַעְנוּ בְּשֵׁם הָרַב פִּינְקֶל בֵּאוּר נִפְלָא: בְּדֶרֶךְ כְּלָל אָדָם חוֹשֵׁב ׳אֵיךְ אַגִּיעַ לְיִרְאַת הָרוֹמְמוּת?׳ אֲבָל הוּא כְּבָר שָׁם.

אָדָם שֶׁיֵּשׁ לְפָנָיו חֵטְא, הַאִם הוּא נִמְנָע רַק מֵחֲמַת הַפַּחַד? בְּדֶרֶךְ כְּלָל לֹא. הוּא לֹא חוֹטֵא כִּי הוּא רוֹצֶה לְהִתְקָרֵב לַה׳ וְלֹא יַעֲשֶׂה דָּבָר הָפוּךְ. נִמְצָא שֶׁהַסִּבָּה שֶׁאֵינוֹ עוֹבֵר עֲבֵרָה הִיא הַפַּחַד לְהַפְסִיד אֶת קִרְבַת ה׳, וְזוֹ יִרְאַת הָרוֹמְמוּת – יִרְאָה לְהַפְסִיד אֶת הָרוֹמְמוּת.

וְזֶה מַה שֶּׁאוֹמְרִים ׳וְרוּחַ קָדְשְׁךָ אַל תִּקַּח מִמֶּנִּי׳: שֶׁלֹּא אַפְסִיד אֶת קִרְבַת ה׳ שֶׁיֵּשׁ לִי.

יִרְאַת הָרוֹמְמוּת הִיא לְפַחֵד לְהַפְסִיד אֶת קִרְבַת ה׳.

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הֶחָפֵץ חַיִּים

רָמָה ב׳ – מִיִּרְאָה לְאַהֲבָה

💬 הַגְדָּרָה שֶׁל הֶחָפֵץ חַיִּים

אוּלַי יֵשׁ לָזֶה סְמוּכִין מִדִּבְרֵי הֶחָפֵץ חַיִּים (קוֹבֶץ הֶעָרוֹת, חִדּוּשֵׁי אַגָּדוֹת סִימָן ג׳), שֶׁהִגְדִּיר מָהִי תְּשׁוּבָה מִיִּרְאַת הָעֹנֶשׁ: אָדָם שֶׁאֵינוֹ מִתְחָרֵט עַל הַמַּעֲשֶׂה עַצְמוֹ אֶלָּא מִתְיָרֵא מֵהָעֹנֶשׁ, וְאִלּוּ הָעֹנֶשׁ הָיָה נִמְחַל לוֹ, לֹא הָיָה מִתְחָרֵט כְּלָל.

נִמְצָא שֶׁכָּל אָדָם שֶׁיּוֹדֵעַ שֶׁגַּם בְּלִי הָעֹנֶשׁ לֹא הָיָה חוֹטֵא, מֵחֲמַת שֶׁאֵינוֹ רוֹצֶה לְהַפְסִיד אֶת הַקֶּשֶׁר עִם בּוֹרֵא עוֹלָם אוֹ לְצַעֵר אוֹתוֹ, תְּשׁוּבָתוֹ הִיא לְפָחוֹת מִיִּרְאַת הָרוֹמְמוּת.

תְּשׁוּבָה מִיִּרְאַת הָרוֹמְמוּת הִיא כְּשֶׁגַּם בְּלִי עֹנֶשׁ הָיָה מִתְחָרֵט.

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, פְּנִינִים בְּמִצְוַת יִרְאַת הַשֵּׁם · רַמְבַּ״ם הִלְכוֹת תְּשׁוּבָה י׳ ב׳

רָמָה ב׳ – מִיִּרְאָה לְאַהֲבָה

🚪 פֶּתַח לָאַהֲבָה

בַּשִּׁעוּר נֶאֱמַר גַּם: יִרְאַת ה׳ הִיא פֶּתַח לְהִכָּנֵס מִמֶּנּוּ לְאַהֲבַת ה׳. וְהָרַמְבַּ״ם מוֹסִיף: ׳בִּזְמַן שֶׁיֶּאֱהַב אָדָם אֶת ה׳ אַהֲבָה הָרְאוּיָה, מִיָּד יַעֲשֶׂה כָּל הַמִּצְווֹת מֵאַהֲבָה׳.

כְּלוֹמַר, הַיִּרְאָה וְהָאַהֲבָה אֵינָן שְׁתֵּי דְּרָכִים שֶׁחוֹלְקוֹת, אֶלָּא שְׁתֵּי מַדְרֵגוֹת. מִי שֶׁמַּתְחִיל מִיִּרְאָה, בִּדְרָכוֹ הוֹלֵךְ אֶל הָאַהֲבָה.

וְהַיִּרְאָה שֶׁל מִי שֶׁקָּרוֹב לַה׳ הִיא כְּבָר מְלֵאָה אַהֲבָה: הוּא מִתְיָרֵא לְהַפְסִיד אֶת מָה שֶׁהוּא אוֹהֵב.

הַיִּרְאָה הִיא פֶּתַח, וּמִי שֶׁקָּרוֹב לַה׳ מִתְיָרֵא מִתּוֹךְ אַהֲבָה.

מָה לוֹקְחִים מִרָמָה ב׳?

הָרַמְבַּ״ם מְלַמֵּד שֶׁעוֹלִים מִיִּרְאָה לְאַהֲבָה. הַיִּרְאָה הִיא פֶּתַח, וּמִי שֶׁקָּרוֹב לַה׳ מִתְיָרֵא לְהַפְסִיד אֶת מָה שֶׁהוּא אוֹהֵב.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה אֲנִי הֲכִי לֹא רוֹצֶה לְהַפְסִיד בַּקֶּשֶׁר שֶׁלִּי עִם ה׳?
  • מָתַי הִתְאַפַּקְתִּי לֹא כִּי פָּחַדְתִּי מֵעֹנֶשׁ, אֶלָּא כִּי רָצִיתִי לִהְיוֹת קָרוֹב לַה׳?
  • הַשָּׁבוּעַ אֶכְתֹּב שְׁלוֹשָׁה דְּבָרִים שֶׁמְּקָרְבִים אוֹתִי לַה׳ וַאֲנִי לֹא רוֹצֶה לְהַפְסִיד – וְאֶזְכֹּר אוֹתָם בִּשְׁעַת קֹשִׁי.

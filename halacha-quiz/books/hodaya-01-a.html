🍎 בְּרָכָה בְּלִי לֵב?

שְׁנֵי יְלָדִים מְבָרְכִים עַל תַּפּוּחַ. הָרִאשׁוֹן אוֹמֵר אֶת הַמִּלִּים מַהֵר, וְהָעֵינַיִם שֶׁלּוֹ כְּבָר אֵצֶל הַמִּשְׂחָק. הַשֵּׁנִי עוֹצֵר, מִסְתַּכֵּל עַל הַתַּפּוּחַ הָאָדֹם וּמְחַיֵּךְ.

הַמִּלִּים אוֹתָן מִלִּים – אֲבָל הַלֵּב שׁוֹנֶה. בַּשִּׁעוּר לוֹמְדִים שֶׁלִּפְעָמִים אֲנַחְנוּ מְבָרְכִים מִתּוֹךְ הֶרְגֵּל, וְה׳ רוֹצֶה הוֹדָיָה שֶׁיּוֹצֵאת מִן הַלֵּב.

הוֹדָיָה אֲמִתִּית יוֹצֵאת מִן הַלֵּב, לֹא רַק מִן הַפֶּה.

😊 שְׁנֵי דְּבָרִים בְּהוֹדָיָה

כְּשֶׁאָנוּ מוֹדִים לַה׳ צְרִיכִים שְׁנֵי דְּבָרִים. רִאשׁוֹן: שִׂמְחָה – לִשְׂמֹחַ בַּטּוֹב שֶׁקִּבַּלְנוּ.

שֵׁנִי: הַכָּרָה בִּגְדֻלַּת ה׳ – לְהָבִין שֶׁהוּא זֶה שֶׁנָּתַן לָנוּ, וְלֶאֱהֹב אוֹתוֹ עַל כָּךְ.

כְּשֶׁיֵּשׁ שְׁנֵיהֶם יַחַד, הַתּוֹדָה הִיא תּוֹדָה אֲמִתִּית.

הוֹדָיָה = שִׂמְחָה + הַכָּרָה בִּגְדֻלַּת ה׳.

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי שׁוֹמֵעַ עַל חֶסֶד שֶׁעָשָׂה ה׳, אֲנִי שָׂמֵחַ בַּלֵּב 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁמוֹת י״ח ט׳ · רַשִׁ״י

רָמָה א׳ – הוֹדָיָה מִן הַלֵּב

🏜️ יִתְרוֹ שָׁמַע וְשָׂמַח

יִתְרוֹ, חֹתֵן מֹשֶׁה, שָׁמַע עַל הַנִּסִּים שֶׁעָשָׂה ה׳ לְעַם יִשְׂרָאֵל וּבָא אֲלֵיהֶם לַמִּדְבָּר.

הַתּוֹרָה כּוֹתֶבֶת עָלָיו: ׳וַיִּחַדְּ יִתְרוֹ עַל כָּל הַטּוֹבָה׳. רַשִׁ״י מֵסְבִּיר: יִתְרוֹ שָׂמַח.

הוּא לֹא רַק שָׁמַע – הוּא הִרְגִּישׁ שִׂמְחָה גְּדוֹלָה בַּלֵּב.

יִתְרוֹ שָׁמַע עַל הַטּוֹב – וְשָׂמַח.

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁמוֹת י״ח י׳–י״א

רָמָה א׳ – הוֹדָיָה מִן הַלֵּב

💬 עַתָּה יָדַעְתִּי

יִתְרוֹ לֹא נִשְׁאַר בְּשִׂמְחָה בִּלְבָד. הוּא אָמַר: ׳בָּרוּךְ ה׳ אֲשֶׁר הִצִּיל אֶתְכֶם׳, וְאָמַר גַּם: ׳עַתָּה יָדַעְתִּי כִּי גָּדוֹל ה׳׳.

הַהוֹדָיָה שֶׁלּוֹ בָּאָה מִשִּׂמְחָה, וְהִבִּיאָה אוֹתוֹ לְהַכִּיר אֶת גְּדֻלַּת ה׳ יוֹתֵר מִתָּמִיד.

זוֹ הוֹדָיָה שֶׁמְּקָרֶבֶת אֶל ה׳.

הוֹדָיָה מִן הַלֵּב מְקָרֶבֶת אֶל ה׳.

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – הוֹדָיָה מִן הַלֵּב

📜 לֹא רַק נֹסַח מוּכָן

יִתְרוֹ לֹא אָמַר נֹסַח בְּרָכָה מְסֻדָּר. הוּא אָמַר מָה שֶׁהוּא הִרְגִּישׁ בַּלֵּב.

גַּם אֲנַחְנוּ אוֹמְרִים אֶת הַבְּרָכוֹת שֶׁתִּקְּנוּ חֲכָמִים, וְאַחֲרֵיהֶן אֶפְשָׁר לְהוֹסִיף מִשְׁפָּט מִשֶּׁלָּנוּ.

כָּךְ הַהוֹדָיָה נִהְיֵית אִישִׁית וְחַמָּה.

אַחֲרֵי הַבְּרָכָה אֶפְשָׁר לְהוֹסִיף מִשְׁפָּט שֶׁלִּי.

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – הוֹדָיָה מִן הַלֵּב

🥤 תּוֹדָה עַל הַמַּיִם וְהַמִּיץ

אֲנִי שׁוֹתֶה כּוֹס מִיץ. אֲנִי מְבָרֵךְ, וְאַחַר כָּךְ אוֹמֵר: ׳תּוֹדָה רַבָּה לְךָ, ה׳, עַל הַמַּיִם, עַל הַמִּיץ וְעַל הַכּוֹס׳.

הַמִּשְׁפָּט הַקָּטָן הַזֶּה מוֹסִיף שִׂמְחָה וְהַכָּרָה.

אֲפִלּוּ דָּבָר פָּשׁוּט נִהְיֶה מַתָּנָה כְּשֶׁמּוֹדִים עָלָיו.

מִשְׁפָּט תּוֹדָה אֶחָד אַחֲרֵי הַבְּרָכָה פּוֹתֵחַ אֶת הַלֵּב.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְשַׂמֵּחַ אֶת ה׳ בְּתוֹדָה שֶׁלִּי 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – הוֹדָיָה מִן הַלֵּב

👑 מַשְׂמִיחִים אֶת ה׳

כְּשֶׁיְּהוּדִי מַכְרִיז בְּקוֹל וּבְשִׂמְחָה שֶׁה׳ הוּא הַמֶּלֶךְ וּמְקוֹר כָּל הַבְּרָכוֹת – זֶה מַשְׂמִיחַ אֶת בּוֹרֵא עוֹלָם.

ה׳ יוֹדֵעַ הַכֹּל, וְהוּא אוֹהֵב לִשְׁמֹעַ אוֹתָנוּ מַכִּירִים בּוֹ בְּשִׂמְחָה.

כְּמוֹ אַבָּא שֶׁשָּׂמֵחַ כְּשֶׁהַיֶּלֶד אוֹמֵר לוֹ ׳תּוֹדָה׳ מִן הַלֵּב.

הוֹדָיָה בְּשִׂמְחָה מַשְׂמִיחָה אֶת בּוֹרֵא עוֹלָם.

🔑 לָמָּה בְּרָכוֹת?

מַטָּרַת הַבְּרָכָה אֵינָהּ לְהוֹסִיף עוֹד בְּרָכָה וְעוֹד בְּרָכָה, אֶלָּא לִבְנוֹת בִּפְנִים רֶגֶשׁ שֶׁל הַכָּרַת הַטּוֹב.

כָּל בְּרָכָה הִיא אִמּוּן קָטָן: אֲנִי עוֹצֵר, רוֹאֶה מָה קִבַּלְתִּי, וּמַרְגִּישׁ תּוֹדָה.

כָּךְ גַּם הַמִּצְווֹת: הֵן עוֹזְרוֹת לָנוּ לְהַרְגִּישׁ אֶת הַטּוֹב שֶׁה׳ נוֹתֵן.

הַבְּרָכוֹת בּוֹנוֹת בַּלֵּב הַכָּרַת הַטּוֹב.

מָה לוֹקְחִים מִרָמָה א׳?

הוֹדָיָה אֲמִתִּית יֵשׁ בָּהּ שִׂמְחָה וְהַכָּרָה בִּגְדֻלַּת ה׳, כְּמוֹ אֵצֶל יִתְרוֹ. כְּשֶׁאֲנִי מוֹסִיף אַחֲרֵי הַבְּרָכָה מִשְׁפָּט מִן הַלֵּב, אֲנִי מַרְגִּישׁ אֶת ה׳ קָרוֹב אֵלַי, וְאַהֲבָתִי אֵלָיו גְּדֵלָה.

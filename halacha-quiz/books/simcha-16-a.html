שַׂבְּעֵנוּ בַבֹּקֶר חַסְדֶּךָ וּנְרַנְּנָה וְנִשְׂמְחָה בְּכָל יָמֵינוּ

רָמָה א׳ – שִׂמְחָה שֶׁלּוֹמְדִים

שִׂמְחָה אֲמִתִּית צְרִיכָה עָמָל, לִימּוּד וְהוֹדָיָה: נְשִׁימָה, רְאִיָּה וּמַתָּנוֹת הָעוֹלָם

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שִׂמְחָה שֶׁנִּשְׁאֶרֶת בַּלֵּב 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרַב אֲבִיגְדוֹר מִילֶר · חוֹבָתוֹ בְּעוֹלָמוֹ – שִׂמְחָתוֹ בְּעוֹלָמוֹ

רָמָה א׳ – שִׂמְחָה שֶׁלּוֹמְדִים

🎢 הַלּוּנָה פַּארְק הַמְאַכְזֵב

יֶלֶד הוֹלֵךְ לְלוּנָה פַּארְק, עוֹלֶה עַל מִתְקָן גָּבוֹהַּ וְצוֹעֵק מֵהַתְרַגְּשׁוּת. וּבַסּוֹף הוּא יוֹרֵד, חוֹזֵר הַבַּיְתָה – וְהַכֹּל אוֹתוֹ דָּבָר.

הָרַב אֲבִיגְדוֹר מִילֶר לִמֵּד: מִי שֶׁמְּחַפֵּשׂ שִׂמְחָה בְּדֶרֶךְ קַלָּה מְקַבֵּל חֲוָיָה אוֹ שְׁתַּיִם, אֲבָל לֹא שִׂמְחָה.

וְהָעוֹלָם נִרְאֶה לוֹ מָקוֹם מְאַכְזֵב.

חֲוָיָה חוֹלֶפֶת אֵינָהּ שִׂמְחָה אֲמִתִּית.

1

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד לִשְׂמֹחַ, כְּמוֹ שֶׁאֲנִי לוֹמֵד תּוֹרָה 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אִיּוֹב ה׳ ז׳ · הָרַב אֲבִיגְדוֹר מִילֶר · חוֹבָתוֹ בְּעוֹלָמוֹ – שִׂמְחָתוֹ בְּעוֹלָמוֹ

רָמָה א׳ – שִׂמְחָה שֶׁלּוֹמְדִים

💪 גַּם עַל שִׂמְחָה עוֹמְלִים

הָאָדָם נוֹלַד לַעֲמֹל: לִלְמֹד, לְהִתְאַמֵּץ וְלַעֲלוֹת מַדְרֵגָה אַחַר מַדְרֵגָה.

וְיֵשׁ עָמָל שֶׁלֹּא כֻּלָּם יוֹדְעִים עָלָיו: לִלְמֹד אֵיךְ לְהִנּוֹת מִן הָעוֹלָם.

הַשִּׂמְחָה אֵינָהּ ׳קוֹרֵית׳ מֵעַצְמָהּ. הִיא חֵלֶק מֵהַתּוֹרָה, וּכְמוֹ כָּל חֵלֶק בַּתּוֹרָה, צָרִיךְ לִלְמֹד אוֹתוֹ וּלְהִתְאַמֵּץ.

שִׂמְחָה צְרִיכָה לְלַמֵּד וּלְהִתְאַמֵּץ, כְּמוֹ כָּל חֵלֶק בַּתּוֹרָה.

2

💛 אַהֲבַת הַשֵּׁם: אֲנִי נוֹשֵׁם עָמֹק וּמוֹדֶה לַה׳ עַל הָאֲוִיר 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרַב אֲבִיגְדוֹר מִילֶר · חוֹבָתוֹ בְּעוֹלָמוֹ – שִׂמְחָתוֹ בְּעוֹלָמוֹ

רָמָה א׳ – שִׂמְחָה שֶׁלּוֹמְדִים

🌬️ נְשִׁימָה אַחַת טוֹבָה

בּוֹאוּ נִנְשֹׁם עָמֹק. הָאֲוִיר נִכְנָס, מְחַמְצֵן אֶת הַדָּם, וְכָל הַגּוּף נֶהֱנֶה מֵהַחַמְצָן.

כַּמָּה שִׂמְחָה יֵשׁ בְּכָל נְשִׁימָה! אֲבָל אִם לֹא חוֹשְׁבִים עַל זֶה אַף פַּעַם – לֹא שָׂמֵחִים בָּהּ.

הָרַב מִילֶר לִמֵּד: צָרִיךְ לְהִתְחִיל לַחְשֹׁב עַל כָּךְ.

בִּנְשִׁימָה יֵשׁ שִׂמְחָה, כְּשֶׁחוֹשְׁבִים עָלֶיהָ.

3

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ שֶׁיֵּשׁ לִי עֵינַיִם שֶׁרוֹאוֹת 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרַב אֲבִיגְדוֹר מִילֶר · חוֹבָתוֹ בְּעוֹלָמוֹ – שִׂמְחָתוֹ בְּעוֹלָמוֹ

רָמָה א׳ – שִׂמְחָה שֶׁלּוֹמְדִים

👀 שְׁתֵּי מַצְלֵמוֹת מְשֻׁכְלָלוֹת

יֵשׁ לְכָל אֶחָד מֵאִתָּנוּ שְׁתֵּי מַצְלֵמוֹת מְשֻׁכְלָלוֹת בְּיוֹתֵר: הָעֵינַיִם. בְּהֶן רוֹאִים צְבָעִים, תְּנוּעָה וַאֲנָשִׁים.

וְכִי אֵין זוֹ שִׂמְחָה גְּדוֹלָה? הִיא נִשְׁאֶרֶת מְסֻתֶּרֶת רַק כִּי לֹא עָצַרְנוּ לַחְשֹׁב עָלֶיהָ.

׳אֵין דָּבָר שֶׁיָּפֶה יוֹתֵר מִן הָרְאִיָּה׳.

הָעֵינַיִם הֵן שְׁתֵּי מַצְלֵמוֹת, וּבָהֶן שִׂמְחָה גְּדוֹלָה.

4

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי אוֹמֵר ׳פּוֹקֵחַ עִוְרִים׳, אֲנִי שָׂמֵחַ שֶׁאֲנִי רוֹאֶה 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בִּרְכוֹת הַשַּׁחַר · הָרַב אֲבִיגְדוֹר מִילֶר · חוֹבָתוֹ בְּעוֹלָמוֹ – שִׂמְחָתוֹ בְּעוֹלָמוֹ

רָמָה א׳ – שִׂמְחָה שֶׁלּוֹמְדִים

🌅 בְּרָכָה עַל הָעֵינַיִם

יֵשׁ אֲנָשִׁים שֶׁקָּשֶׁה לָהֶם לִרְאוֹת, וְהֵם מְגַשְּׁשִׁים אֶת דַּרְכָּם בְּעֶזְרַת מַקֵּל.

לָנוּ יֵשׁ אֶת הָעֹנֶג הַזֶּה, וְלָכֵן מְבָרְכִים בְּשִׂמְחָה: ׳פּוֹקֵחַ עִוְרִים׳.

מִי שֶׁעָמַל בְּתוֹרַת הַשִּׂמְחָה – כְּבָר לֹא יוּכַל שֶׁלֹּא לִשְׂמֹחַ.

אוֹמְרִים ׳פּוֹקֵחַ עִוְרִים׳ בְּשִׂמְחָה, כִּי רוֹאִים.

5

💛 אַהֲבַת הַשֵּׁם: אֲנִי חוֹשֵׁב עַל ה׳, מוֹדֶה לוֹ וּמַעֲרִיךְ אֶת מַתְּנוֹתָיו 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְסִלַּת יְשָׁרִים פֶּרֶק א׳ · הָרַב אֲבִיגְדוֹר מִילֶר · חוֹבָתוֹ בְּעוֹלָמוֹ – שִׂמְחָתוֹ בְּעוֹלָמוֹ

רָמָה א׳ – שִׂמְחָה שֶׁלּוֹמְדִים

🌟 לַחְשֹׁב, לְהוֹדוֹת, לְהַעֲרִיךְ

הָרַב מִילֶר מְצַטֵּט מִמְּסִלַּת יְשָׁרִים: ׳הָאָדָם נִבְרָא לְהִתְעַנֵּג עַל ה׳׳.

וְאֵיךְ מִתְעַנְּגִים? לַחְשֹׁב עָלָיו, לְהוֹדוֹת לוֹ, וּלְהַעֲרִיךְ אֶת מַה שֶׁהוּא נוֹתֵן.

זוֹ הַדֶּרֶךְ לְשִׂמְחָה שֶׁלֹּא נֶעֱלֶמֶת.

חוֹשְׁבִים, מוֹדִים וּמַעֲרִיכִים – וְכָךְ מִתְעַנְּגִים עַל ה׳.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְקַבֵּל אֶת הָעוֹלָם כְּמַתָּנָה מֵה׳ 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרַב אֲבִיגְדוֹר מִילֶר · חוֹבָתוֹ בְּעוֹלָמוֹ – שִׂמְחָתוֹ בְּעוֹלָמוֹ

רָמָה א׳ – שִׂמְחָה שֶׁלּוֹמְדִים

🎁 הַנָּאוֹת הָעוֹלָם – מַתָּנָה

הַהֲנָאוֹת שֶׁל הָעוֹלָם אֵינָן ׳לְקִיחָה׳ שֶׁל הָעוֹלָם. הֵן קַבָּלָה שֶׁל מַה שֶׁה׳ נוֹתֵן לָנוּ בְּחַסְדּוֹ הַגָּדוֹל.

מָה צָרִיךְ? רַק לְהָבִין נָכוֹן אֶת הָעוֹלָם, וְנָכוֹן לַעֲמֹל.

וּכְשֶׁעוֹמְלִים, הַשִּׂמְחָה פּוֹרֶצֶת מִתּוֹךְ הַלֵּב.

הַנָּאוֹת הָעוֹלָם = מַתָּנָה מֵה׳, וְצָרִיךְ לִלְמֹד לְקַבֵּל.

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אָבוֹת ד׳ א׳ · הָרַב אֲבִיגְדוֹר מִילֶר · חוֹבָתוֹ בְּעוֹלָמוֹ – שִׂמְחָתוֹ בְּעוֹלָמוֹ

רָמָה א׳ – שִׂמְחָה שֶׁלּוֹמְדִים

📷 שָׁלוֹשׁ דַּקּוֹת ׳מַצְלֵמוֹת׳

הַיּוֹם נִתְנַסֶּה: עוֹצְרִים שָׁלוֹשׁ דַּקּוֹת, מִסְתַּכְּלִים בַּחֶדֶר וּמוֹצְאִים חֲמִשָּׁה דְּבָרִים שֶׁקָּשֶׁה לִרְאוֹת בְּלִי עֵינַיִם. אַחַר כָּךְ נוֹשְׁמִים עָמֹק וְאוֹמְרִים תּוֹדָה.

וְכָךְ מִתְאַמְּנִים בְּתוֹרַת הַשִּׂמְחָה.

וְחֲכָמִים אָמְרוּ: ׳אֵיזֶהוּ עָשִׁיר? הַשָּׂמֵחַ בְּחֶלְקוֹ׳.

שָׁלוֹשׁ דַּקּוֹת, חֲמִשָּׁה דְּבָרִים, נְשִׁימָה אַחַת – וְתוֹדָה.

מָה לוֹקְחִים מִרָמָה א׳?

מִי שֶׁלּוֹמֵד לִשְׂמֹחַ בִּנְשִׁימָה, בִּרְאִיָּה וּבְכָל מַה שֶׁיֵּשׁ לוֹ, מַרְגִּישׁ אֶת אַהֲבַת ה׳. הוּא יוֹדֵעַ שֶׁכָּל זֶה מַתָּנָה, וְאוֹהֵב אֶת הַנּוֹתֵן.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה הַדָּבָר שֶׁאֲנִי נֶהֱנֶה מִמֶּנּוּ בְּכָל יוֹם בְּלִי לָשִׂים לֵב, וְאֵיךְ אֶשִּׂים לֵב הַיּוֹם?
  • מָתַי חִפַּשְׂתִּי שִׂמְחָה בְּדֶרֶךְ קַלָּה וְהִתְאַכְזַבְתִּי, וּמָה הָיָה עוֹזֵר לִי לִשְׂמֹחַ בֶּאֱמֶת?
  • אֲנִי אֶעֱשֶׂה הַיּוֹם שָׁלוֹשׁ דַּקּוֹת ׳מַצְלֵמוֹת׳: אֶמְצָא בַּחֶדֶר חֲמִשָּׁה דְּבָרִים שֶׁקָּשֶׁה לִרְאוֹת בְּלִי עֵינַיִם, אֶנְשֹׁם עָמֹק וְאֹמַר תּוֹדָה.

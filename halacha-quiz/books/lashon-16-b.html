רָמָה ב׳ – מַתָּנָה מִמֶּלֶךְ

👑 מַתָּנָה מִמֶּלֶךְ

בְּמִדְרַשׁ תַּנְחוּמָא יֵשׁ מָשָׁל: מֶלֶךְ אָהַב אֶת אֶחָד מֵאֲנָשָׁיו, וּבָחַר לוֹ מָנָה יָפָה לָתֵת לוֹ. בָּא אֶחָד וְאָמַר לָאוֹהֵב שֶׁהַמָּנָה רָעָה.

הַמֶּלֶךְ יִשְׁמַע וְיֹאמַר: ׳אֵינְךָ אוֹמֵר שֶׁלֹּא כִּבַּדְתִּי אֶת אוֹהֲבִי כָּרָאוּי?׳

כָּךְ הַמְּרַגְּלִים: ׳וַתֵּרָגְנוּ בְּאָהֳלֵיכֶם׳ – תָּרוּ גְּנוּת עַל אֶרֶץ יִשְׂרָאֵל, שֶׁה׳ בְּעַצְמוֹ קָרָא לָהּ ׳אֶרֶץ טוֹבָה׳.

מִי שֶׁאוֹמֵר שֶׁהַמַּתָּנָה רָעָה, מְזַלְזֵל בַּנּוֹתֵן.

1
🏠

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִתְנַהֵג בְּכָבוֹד בְּבֵית ה׳, גַּם בְּמִלִּים שֶׁלִּי 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, וַיִּקְרָא כ״ה כ״ג · מַעְיַן בֵּית הַשּׁוֹאֵבָה · זֹהַר ח״ג קס״א א׳

רָמָה ב׳ – מַתָּנָה מִמֶּלֶךְ

🏠 הָאָרֶץ שֶׁלּוֹ

הָרַב שְׁוָאב מַסְבִּיר: בַּתּוֹרָה כָּתוּב ׳כִּי לִי הָאָרֶץ׳. כָּל הָעוֹלָם שֶׁל ה׳, אֲבָל אֶרֶץ יִשְׂרָאֵל שַׁיֶּכֶת לוֹ בְּאֹפֶן מְיֻחָד, כְּמוֹ בַּיִת שֶׁלּוֹ.

לָכֵן לָשׁוֹן הָרָע עַל אֶרֶץ יִשְׂרָאֵל נֶחְשֶׁבֶת כִּבְיָכוֹל לָשׁוֹן הָרָע עַל ה׳.

וְכָךְ גַּם בַּזֹּהַר: מִי שֶׁמּוֹצִיא דִּבָּה עַל הָאָרֶץ הַקְּדוֹשָׁה – כְּאִלּוּ מוֹצִיא עָלָיו.

אֶרֶץ יִשְׂרָאֵל הִיא בַּיִת שֶׁל ה׳, וְדִבּוּר עָלֶיהָ כְּדִבּוּר עָלָיו.

2
📖

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַאֲמִין לְמָה שֶׁה׳ אָמַר, וְשָׂמֵחַ בְּאֶרֶץ הַחֶמְדָּה 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים ק״ו כ״ד · מַלְבִּ״ים

רָמָה ב׳ – מַתָּנָה מִמֶּלֶךְ

📖 וַיִּמְאֲסוּ בְּאֶרֶץ חֶמְדָּה

בִּתְהִלִּים כָּתוּב עַל חֵטְא הַמְּרַגְּלִים: ׳וַיִּמְאֲסוּ בְּאֶרֶץ חֶמְדָּה לֹא הֶאֱמִינוּ לִדְבָרוֹ׳.

הַמַּלְבִּ״ים מְפָרֵשׁ: ׳וַיִּמְאֲסוּ בְּאֶרֶץ חֶמְדָּה׳ – שֶׁאָמְרוּ שֶׁהָאָרֶץ רָעָה.

וְ׳לֹא הֶאֱמִינוּ לִדְבָרוֹ׳ – שֶׁה׳ אָמַר שֶׁהִיא טוֹבָה, וְהֵם לֹא הֶאֱמִינוּ.

לְהוֹצִיא דִּבָּה עַל הָאָרֶץ זֶה גַּם לֹא לְהַאֲמִין לְמָה שֶׁה׳ אָמַר.

3
🔎

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִסְתַּכֵּל לְאַט וּבְעַיִן טוֹבָה, לִפְנֵי שֶׁאֲנִי מְדַבֵּר 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הִלְכוֹת דִּבַּת הָאָרֶץ

רָמָה ב׳ – מַתָּנָה מִמֶּלֶךְ

🔎 שָׁבוּעַ אֶחָד וְכָל הַחֶסְרוֹנוֹת

הָאַדְמוֹ״ר מִקְּלוּזִינְבוּרְג הָיָה פַּעַם בִּדְרוֹם אֲמֶרִיקָה. פָּגַשׁ אוֹתוֹ יְהוּדִי וְהִתְלוֹנֵן: ׳בִּקַּרְתִּי שָׁבוּעַ בְּאֶרֶץ יִשְׂרָאֵל וְרָאִיתִי שָׁם חֶסְרוֹנוֹת׳.

הָאַדְמוֹ״ר עָנָה: ׳פִּלְאִי! אֲנִי גָּר בְּאֶרֶץ יִשְׂרָאֵל שָׁנִים, וַאֲנִי עֲדַיִן לֹא רָאִיתִי אֶת כָּל הַמְּקוֹמוֹת וְהַחֶסְרוֹנוֹת שֶׁאַתָּה מְדַבֵּר עֲלֵיהֶם. וְאַתָּה, בְּשָׁבוּעַ אֶחָד, רָאִיתָ הַכֹּל!׳

מִי שֶׁהָיָה רַק שָׁבוּעַ לֹא רָאָה הַכֹּל, וְלֹא יָדוּעַ לוֹ הַטּוֹב.

🚪

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹאֶה טוֹב בְּכָל מָקוֹם בָּאָרֶץ וּבְכָל יְהוּדִי 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּקֻדָּשָׁתוֹ שֶׁל אַהֲרֹן

רָמָה ב׳ – מַתָּנָה מִמֶּלֶךְ

🚪 הַלְוַאי – וּמָה שֶׁנִּשְׁמַע בּוֹ

כְּשֶׁאוֹמְרִים עַל מָקוֹם אֶחָד ׳הַלְוַאי שֶׁכָּךְ בְּכָל הָאָרֶץ׳, נִשְׁמָע שֶׁבִּשְׁאָר הַמְּקוֹמוֹת הַמַּצָּב לֹא טוֹב.

זֶה מָה שֶׁהָרַבִּי מִבֶּעלְז שָׁמַע, וְנִצְטַעֵר: אֲבַק לְשׁוֹן הָרָע עַל יוֹשְׁבֵי הָאָרֶץ.

כְּדֵי לְשַׁבֵּחַ מָקוֹם, אֵין צֹרֶךְ לְהַקְטִין אֶת הַמְּקוֹמוֹת הָאֲחֵרִים.

מְשַׁבְּחִים מָקוֹם אֶחָד בְּלִי לוֹמַר שֶׁהָאֲחֵרִים לֹא טוֹבִים.

🌈

💛 אַהֲבַת הַשֵּׁם: אֲנִי יָכוֹל לְהַתְחִיל מֵחָדָשׁ, עִם דִּבּוּרִים טוֹבִים וְאַהֲבַת ה׳ 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּקֻדָּשָׁתוֹ שֶׁל אַהֲרֹן

רָמָה ב׳ – מַתָּנָה מִמֶּלֶךְ

🌈 דִּבּוּר שֶׁמְּשַׂמֵּחַ

הַסִּפּוּר מְלַמֵּד שֶׁהָרַבִּי לֹא כָּעַס עַל הַחֲסִיד עַד הַסּוֹף: כְּשֶׁהַחֲסִיד בָּא וְסִפֵּר דִּבּוּרִים שֶׁמְּשַׁבְּחִים אֶת הַיְּהוּדִים בָּאָרֶץ, פְּנֵי הָרַבִּי צָהֲלוּ.

כֵּן אֶפְשָׁר לָבוֹא אֶל אָדָם אַחֲרֵי שֶׁטָּעִינוּ, עִם דִּבּוּרִים טוֹבִים.

וּבִפְרָט בְּעֶרֶב יוֹם כִּפּוּר, הַיּוֹם שֶׁבּוֹ מְבַקְּשִׁים סְלִיחָה וּמִתְקָרְבִים אֶל ה׳.

אַחֲרֵי טָעוּת, דִּבּוּרִים טוֹבִים מְשַׂמְּחִים אֶת הַלֵּב.

רָמָה ב׳ – מַתָּנָה מִמֶּלֶךְ

מָה לוֹקְחִים מִרָמָה ב׳?

אֶרֶץ יִשְׂרָאֵל הִיא מַתָּנָה שֶׁל מֶלֶךְ הַמְּלָכִים, וְהַמִּלִּים שֶׁלָּנוּ עָלֶיהָ יְכוֹלוֹת לְהוֹדוֹת לַנּוֹתֵן. מִי שֶׁמַּאֲמִין לְדִבְרֵי ה׳ שֶׁהִיא טוֹבָה, אוֹהֵב אוֹתוֹ יוֹתֵר.

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי שָׁמַעְתִּי דִּבּוּר שֶׁמְּשַׁבֵּחַ מָקוֹם אוֹ אָדָם, אֲבָל גַּם הִקְטִין אֲחֵרִים, וְאֵיךְ אֶפְשָׁר לְנַסֵּחַ אוֹתוֹ אַחֶרֶת?
  • אֵיזֶה סִפּוּר טוֹב אֲנִי יָכוֹל לְהָבִיא הַיּוֹם לַבַּיִת, כְּמוֹ שֶׁהַנֶּהָגִים הֵבִיאוּ לָרַבִּי?
  • אֶקַּח מִשְׁפָּט אֶחָד שֶׁאָמַרְתִּי אוֹ שָׁמַעְתִּי, אֶכְתֹּב אוֹתוֹ בְּדַף, וְאֶכְתֹּב לְיָדוֹ נִסּוּחַ שֶׁמְּשַׁבֵּחַ בְּלִי לִפְגֹּעַ, וְאֹמַר אוֹתוֹ לְמִישֶׁהוּ בַּבַּיִת.

רָמָה ב׳ – שְׁעָרִים וּכְלִי

📜 מַה הָיָה בִּימֵי דָּוִד?

בַּגְּמָרָא וּבַלְּבוּשׁ מְסֻפָּר שֶׁבִּימֵי דָּוִד הַמֶּלֶךְ הָיְתָה צָרָה בָּעָם, וְלֹא יָדְעוּ מַדּוּעַ. דָּוִד חָקַר וְהֵבִין בְּרוּחַ הַקֹּדֶשׁ שֶׁהָעָם לֹא שִׁבֵּחַ וְלֹא בֵּרַךְ כָּרָאוּי אֶת ה׳ עַל כָּל הַטּוֹבוֹת.

לָכֵן עָמַד וְתִקֵּן מֵאָה בְּרָכוֹת בְּכָל יוֹם, וּמֵאָז פָּסְקָה הַצָּרָה.

הַלֶּקַח: כְּשֶׁלֹּא מוֹדִים עַל הַטּוֹב, אֵין עוֹמְדִים עָלָיו – וְכָךְ מִתְרַחֵק הַלֵּב.

מֵאָה בְּרָכוֹת הֵן תִּקּוּן עַל כָּךְ שֶׁלֹּא שָׂמִים לֵב לַטּוֹבוֹת.

1

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְקַיֵּם בְּשִׂמְחָה אֶת מַה שֶּׁה׳ מְבַקֵּשׁ 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְנָחוֹת מ״ג ב׳ · דְּבָרִים י׳ י״ב · סֵדֶר הַיּוֹם

רָמָה ב׳ – שְׁעָרִים וּכְלִי

❓ ׳מַה׳ אוֹ ׳מֵאָה׳?

בַּתּוֹרָה כָּתוּב: ׳וְעַתָּה יִשְׂרָאֵל, מָה ה׳ אֱלֹהֶיךָ שֹׁאֵל מֵעִמָּךְ׳. חֲכָמִים דָּרְשׁוּ: אַל תִּקְרָא ׳מָה׳ אֶלָּא ׳מֵאָה׳ – כִּי הַמִּלִּים כִּמְעַט אוֹתָהּ מִלָּה.

בַּסְּפָרִים מוֹסִיפִים: אִם זֶה כָּל מַה שֶּׁה׳ שׁוֹאֵל מִמֶּנּוּ, זֶה דָּבָר גָּדוֹל מְאֹד.

וּבַשִּׂמְחָה, מֵאָה בְּרָכוֹת הֵן לֹא עֹל – הֵן מַתָּנָה, כִּי הֵן מְקָרְבוֹת אוֹתָנוּ אֶל ה׳.

׳מָה׳ שֶׁה׳ שׁוֹאֵל מִמֶּנּוּ = ׳מֵאָה׳ בְּרָכוֹת.

2

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה גַּם כְּשֶׁקָּשֶׁה, וְה׳ הוֹפֵךְ לְטוֹבָה 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דְּבָרִים כ״ג ו׳ · בַּעַל הַטּוּרִים · הָאֲרִיזַ״ל

רָמָה ב׳ – שְׁעָרִים וּכְלִי

🔁 מֵהַקְּלָלָה לַהִלּוּל

כָּתוּב בַּתּוֹרָה: ׳וַיַּהֲפֹךְ ה׳ אֱלֹהֶיךָ לְךָ אֶת הַקְּלָלָה לִבְרָכָה׳. הָאֲרִיזַ״ל הֶעֱמִיד אֶת הָרֶמֶז: אִם נַהֲפֹךְ אֶת אוֹתִיּוֹת קְלָלָה נְקַבֵּל הַלֵּל־ק׳.

כְּלוֹמַר: מִי שֶׁמּוֹדֶה וּמְהַלֵּל אֶת ה׳ מֵאָה פְּעָמִים בַּיּוֹם, זוֹכֶה שֶׁדְּבָרִים קָשִׁים נֶהֱפָכִים לִבְרָכָה.

בַּעַל הַטּוּרִים מוֹסִיף: מֵאָה בְּרָכוֹת מְגִנּוֹת עָלֵינוּ.

הוֹדָיָה מְשַׁנָּה אֶת הַמַּבָּט: מִקְּלָלָה לִבְרָכָה.

3

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹמֵר ׳מוֹדִים׳ מֵהַלֵּב, לֹא רַק מִן הַפֶּה 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דַּעַת זְקֵנִים (דְּבָרִים י׳ י״ב) · תְּהִלִּים ק׳

רָמָה ב׳ – שְׁעָרִים וּכְלִי

🔢 מוֹדִים כְּמוֹ מֵאָה

בַּעֲלֵי הַתּוֹסָפוֹת כָּתְבוּ שֶׁ׳מוֹדִים׳ שָׁוֶה מֵאָה בְּגִימַטְרִיָּה, וְמִי שֶׁאוֹמֵר ׳מוֹדִים׳ כַּהֹגֶן, כְּאִלּוּ קִיֵּם מֵאָה בְּרָכוֹת.

וְגַם ׳מִזְמוֹר לְתוֹדָה׳ הוּא מִזְמוֹר מֵאָה בִּתְהִלִּים, רֶמֶז שֶׁאוֹמְרִים תּוֹדָה מֵאָה פְּעָמִים בַּיּוֹם.

לֹא סְתָם אוֹמְרִים אֶת הַמִּלָּה – שָׂמִים לֵב לְמַה שֶּׁאוֹמְרִים.

הַכַּוָּנָה בְּ׳מוֹדִים׳ שׁוָה לְמֵאָה בְּרָכוֹת.

4

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת כְּלִי פָּתוּחַ לְטוֹבוֹת שֶׁה׳ נוֹתֵן 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרִיקַנְטִי · הַשְּׁלָ״ה · שְׂפַת אֱמֶת (תּוֹלְדוֹת)

רָמָה ב׳ – שְׁעָרִים וּכְלִי

🚪 אֲנִי כְּלִי לַבְּרָכָה

הָרִיקַנְטִי כָּתַב: בַּמָּרוֹם יֵשׁ מֵאָה שְׁעָרִים, וְכָל בְּרָכָה פּוֹתַחַת שַׁעַר אֶחָד. הַשָּׁ״לָ״ה הַקָּדוֹשׁ מוֹסִיף שֶׁהַמֵּאָה נִרְמְזָה כְּבָר אֵצֶל אַבְרָהָם, יִצְחָק וְיַעֲקֹב.

הַשְּׂפַת אֱמֶת מְלַמֵּד: מִן הַשָּׁמַיִם רוֹצִים לְהַשְׁפִּיעַ בְּרָכוֹת תָּמִיד, רַק שֶׁהָאָדָם צָרִיךְ לִהְיוֹת כְּלִי לְקַבֵּל – וְזֶה בְּכֹחַ מֵאָה הַבְּרָכוֹת.

כְּמוֹ כּוֹס: כְּשֶׁהִיא פְּתוּחָה, אֶפְשָׁר לִמְלֹא אוֹתָהּ.

מֵאָה בְּרָכוֹת הֵן כְּלִי שֶׁמְּקַבֵּל שֶׁפַע.

5

💛 אַהֲבַת הַשֵּׁם: אֲנִי כּוֹתֵב רְשִׁימָה שֶׁל הוֹדָיוֹת וְאוֹמֵר אוֹתָהּ בְּשִׂמְחָה 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשְׁנָה בְּרוּרָה תרכ״ב · בֶּן אִישׁ חַי (וַיֵּשֶׁב) · מֵאָה הוֹדָיוֹת

רָמָה ב׳ – שְׁעָרִים וּכְלִי

🙏 שְׁבָחִים וְהוֹדָיוֹת

בַּסְּפָרִים מְסֻפָּר שֶׁמִּי שֶׁלֹּא יָכוֹל לְהַשְׁלִים מֵאָה בְּרָכוֹת, בִּשְׁעַת הַדֹּחַק יוֹצֵא בִּשְׁבָחִים וְהוֹדָיוֹת. לָכֵן כְּדַאי שֶׁתִּהְיֶה לִי רְשִׁימָה שֶׁל מֵאָה הוֹדָיוֹת שֶׁאֲנִי אוֹמֵר בְּשִׂמְחָה.

הַבֶּן אִישׁ חַי כָּתַב שֶׁכְּדַאי לְהַרְגִּיל אֶת עַצְמֵנוּ לוֹמַר אַחֲרֵי הַהוֹדָיָה אֶת הַפְּסוּקִים שֶׁדָּוִד אָמַר: ׳וַיְבָרֶךְ דָּוִיד אֶת ה׳ לְעֵינֵי כָּל הַקָּהָל׳, וְזֶה נֶחְשָׁב כִּבְרָכַת הוֹדָיָה.

הָעִקָּר הוּא לֹא רַק לִסְפֹּר, אֶלָּא לְהוֹדוֹת.

רְשִׁימַת מֵאָה הוֹדָיוֹת עוֹזֶרֶת לְהַשְׁלִים וּלְהַרְגִּישׁ.

מָה לוֹקְחִים מִרָמָה ב׳?

מֵאָה בְּרָכוֹת הֵן לֹא סְפִירָה יְבֵשָׁה אֶלָּא מֵאָה פְּעָמִים שֶׁאֲנִי מַרְגִּישׁ שֶׁה׳ נוֹתֵן לִי. כָּל תּוֹדָה פּוֹתַחַת לִי עוֹד שַׁעַר וּמְקָרֶבֶת אוֹתִי לַמֵּטִיב, וְכָךְ גּוֹדֶלֶת אַהֲבַת ה׳.

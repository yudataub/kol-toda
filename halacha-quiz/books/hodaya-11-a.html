מָה עָשָׂה הֶחָפֵץ חַיִּים בַּחֲצוֹת הַלַּיְלָה, לָמָּה פֵּרֵט כָּל טוֹבָה, וְאֵיךְ גַּם אֲנִי מוֹדֶה לִפְנֵי הַשֵּׁנָה

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן

💛 אַהֲבַת הַשֵּׁם: גַּם אֲנִי יָכוֹל לְדַבֵּר עִם ה׳ בַּשֶּׁקֶט שֶׁל הַלַּיְלָה 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סִפּוּר עַל הֶחָפֵץ חַיִּים (מִפִּי הָרַב שְׁמוּאֵל גְּרֵינְמַן)

רָמָה א׳ – הוֹדָיָה בַּלַּיְלָה

🌙 מֵאֲחוֹרֵי הַדֶּלֶת

הַלַּיְלָה עָמֹק. כָּל בְּנֵי הַבַּיִת יְשֵׁנִים, וּבַחוּץ שׁוֹרֶרֶת דְּמָמָה.

אָז נִכְנַס מָרָן הֶחָפֵץ חַיִּים לַחֶדֶר שֶׁלּוֹ, סָגַר אֶת הַדֶּלֶת בְּמַפְתֵּחַ וּבִבְרִיחַ, וְלֹא הִדְלִיק אוֹר.

לִפְעָמִים עָמְדוּ הַקְּרוֹבִים לוֹ מֵאֲחוֹרֵי הַדֶּלֶת, וּבְרֶעֶד הִקְשִׁיבוּ לְמָה שֶׁהוּא אוֹמֵר לַה׳.

הֶחָפֵץ חַיִּים דִּבֵּר עִם ה׳ בְּשֶׁקֶט שֶׁל לַיְלָה.

1
🙏

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַתְחִיל בְּתוֹדָה, וְרַק אַחַר כָּךְ מְסַפֵּר מָה אִתִּי 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סִפּוּר עַל הֶחָפֵץ חַיִּים (מִפִּי הָרַב שְׁמוּאֵל גְּרֵינְמַן)

רָמָה א׳ – הוֹדָיָה בַּלַּיְלָה

🙏 קֹדֶם שֶׁבַח וְהוֹדָיָה

הוּא שָׁפַךְ אֶת לִבּוֹ לִפְנֵי ה׳. וּבִתְחִלַּת דְּבָרָיו הוּא נָתַן שֶׁבַח וְהוֹדָיָה לַבּוֹרֵא עַל כָּל הַחֲסָדִים שֶׁעָשָׂה לוֹ.

לֹא רַק ׳תּוֹדָה עַל הַכֹּל׳, אֶלָּא בִּפְרָטִים.

לִפְנֵי שֶׁמְּדַבְּרִים הַרְבֵּה, מַתְחִילִים בְּשֶׁבַח וְהוֹדָיָה.

2

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹאֶה בְּכָל פְּרָט טוֹב זְכוּת גְּדוֹלָה 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סִפּוּר עַל הֶחָפֵץ חַיִּים (מִפִּי הָרַב שְׁמוּאֵל גְּרֵינְמַן)

רָמָה א׳ – הוֹדָיָה בַּלַּיְלָה

📋 כָּל פְּרָט – זְכוּת גְּדוֹלָה

הֶחָפֵץ חַיִּים הָיָה מְפָרֵט כָּל מְאֹרָע בְּחַיָּיו. עַל כָּל פְּרָט וּפְרָט הִסְתַּכֵּל כְּעַל זְכוּת גְּדוֹלָה, וְנָתַן שֶׁבַח וְהוֹדָיָה עַל הַחֶסֶד הַגָּדוֹל שֶׁה׳ עָשָׂה עִמּוֹ.

כְּשֶׁמְּפָרְטִים, רוֹאִים כַּמָּה טוֹבוֹת יֵשׁ.

כְּשֶׁמְּפָרְטִים אֶת הַטּוֹבוֹת, רוֹאִים כַּמָּה הֵן.

3
🕯️

💛 אַהֲבַת הַשֵּׁם: אֲנִי זוֹכֵר וּמוֹדֶה גַּם עַל עֶזְרָה מֵהֶעָבָר 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סִפּוּר עַל הֶחָפֵץ חַיִּים (מִפִּי הָרַב שְׁמוּאֵל גְּרֵינְמַן)

רָמָה א׳ – הוֹדָיָה בַּלַּיְלָה

🕯️ תּוֹדָה עַל הָעֶזְרָה בַּיַּלְדוּת

מִן הַפְּרָטִים שֶׁהֶחָפֵץ חַיִּים הוֹדָה עֲלֵיהֶם: שֶׁה׳ עָזַר לוֹ בִּיתְמוּתוֹ.

גַּם כְּשֶׁהַיַּלְדוּת הָיְתָה קָשָׁה, הוּא זָכַר אֶת הָעֶזְרָה וְהוֹדָה עָלֶיהָ.

זוֹכְרִים אֶת הָעֶזְרָה גַּם מִזְּמַן קָשֶׁה.

4

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה לַה׳ עַל מָה שֶׁלָּמַדְתִּי הַיּוֹם 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סִפּוּר עַל הֶחָפֵץ חַיִּים (מִפִּי הָרַב שְׁמוּאֵל גְּרֵינְמַן)

רָמָה א׳ – הוֹדָיָה בַּלַּיְלָה

📚 תּוֹרָה וּסְפָרִים

הוּא הוֹדָה גַּם עַל שֶׁה׳ סִיֵּעַ בְּיָדוֹ לִלְמֹד תּוֹרָה וּלְחַבֵּר סְפָרִים.

כָּל שׁוּרָה שֶׁלָּמַד וְכָל סֵפֶר שֶׁכָּתַב הָיוּ בְּעֵינָיו מַתָּנָה.

כְּשֶׁאֲנִי מַצְלִיחַ בְּלִמּוּד, אֲנִי יָכוֹל לוֹמַר: תּוֹדָה, ה׳, שֶׁעָזַרְתָּ לִי.

הַצְלָחָה בַּלִּמּוּד הִיא מַתָּנָה שֶׁמּוֹדִים עָלֶיהָ.

5
🏡

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה לַה׳ עַל אֲנָשִׁים טוֹבִים בַּחַיִּים שֶׁלִּי 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סִפּוּר עַל הֶחָפֵץ חַיִּים (מִפִּי הָרַב שְׁמוּאֵל גְּרֵינְמַן)

רָמָה א׳ – הוֹדָיָה בַּלַּיְלָה

🏡 מִשְׁפָּחָה טוֹבָה

הֶחָפֵץ חַיִּים הוֹדָה גַּם עַל כָּךְ שֶׁה׳ נָתַן לוֹ חֲתָנִים טוֹבִים.

אֲנָשִׁים טוֹבִים סְבִיבֵנוּ הֵם חֶסֶד גָּדוֹל.

גַּם אֲנִי יָכוֹל לְהוֹדוֹת עַל הַמִּשְׁפָּחָה שֶׁלִּי וְעַל הַחֲבֵרִים הַטּוֹבִים.

אֲנָשִׁים טוֹבִים סְבִיבֵנוּ הֵם חֶסֶד מֵה׳.

6

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה בְּהִתְלַהֲבוּת, כִּי אֲנִי שָׂמֵחַ 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סִפּוּר עַל הֶחָפֵץ חַיִּים (מְאוֹרוֹת הַגְּדוֹלִים)

רָמָה א׳ – הוֹדָיָה בַּלַּיְלָה

🏨 נִשְׁמַת בַּמָּלוֹן

רַבִּי יְחֶזְקֵאל אַבְרָמְסְקִי סִפֵּר: בִּימֵי נְעוּרָיו לָן בְּמָלוֹן בְּעִיר אַחַת. בַּבֹּקֶר הַשָּׁכֵם שָׁמַע מֵהַחֶדֶר הַסָּמוּךְ יְהוּדִי אוֹמֵר תְּפִלַּת נִשְׁמַת בִּמְתִיקוּת, בְּהִתְלַהֲבוּת וּבִבְכִיּוֹת, עַד שֶׁהִזְדַּעְזַע.

לְמָחֳרָת נוֹדַע לוֹ: זֶה הָיָה הֶחָפֵץ חַיִּים!

הֶחָפֵץ חַיִּים הוֹדָה בְּהִתְלַהֲבוּת גַּם מִחוּץ לְבֵיתוֹ.

😴

💛 אַהֲבַת הַשֵּׁם: בְּכָל לַיְלָה אֲנִי מוֹדֶה עַל שְׁלוֹשָׁה פְּרָטִים 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – הוֹדָיָה בַּלַּיְלָה

😴 הוֹדָיָה לִפְנֵי הַשֵּׁנָה

גַּם אֲנִי יָכוֹל לַעֲשׂוֹת בַּלַּיְלָה קְצָת כְּמוֹ הֶחָפֵץ חַיִּים: לִפְנֵי שֶׁאֲנִי נִרְדָּם, אֲנִי מוֹדֶה עַל שְׁלוֹשָׁה פְּרָטִים מֵהַיּוֹם.

לֹא רַק ׳תּוֹדָה עַל הַכֹּל׳, אֶלָּא פְּרָטִים: הַמָּנָה הַטְּעִימָה, הֶחָבֵר שֶׁעָזַר, הַמִּשְׂחָק הַטּוֹב.

וְכָךְ הַיּוֹם נִגְמָר בְּלֵב שָׂמֵחַ.

שְׁלוֹשָׁה פְּרָטִים לִפְנֵי הַשֵּׁנָה = לֵב שָׂמֵחַ.

רָמָה א׳ – הוֹדָיָה בַּלַּיְלָה

מָה לוֹקְחִים מִרָמָה א׳?

הֶחָפֵץ חַיִּים הָיָה גָּדוֹל בַּתּוֹרָה, וּבַלַּיְלָה, כְּשֶׁאִישׁ לֹא רָאָה, הוּא הוֹדָה לַה׳ עַל כָּל פְּרָט. מִי שֶׁמּוֹדֶה בִּפְרָטִים מַרְגִּישׁ כַּמָּה ה׳ אוֹהֵב אוֹתוֹ, וְכָךְ גַּם אַהֲבָתוֹ אֵלָיו גְּדֵלָה.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • אֵילוּ שְׁלוֹשָׁה פְּרָטִים מֵהַיּוֹם אֲנִי יָכוֹל לְהוֹדוֹת עֲלֵיהֶם לִפְנֵי שֶׁאֲנִי נִרְדָּם?
  • אֵיזֶה דָּבָר בְּחַיַּי נִרְאֶה לִי רָגִיל, וְאֵיךְ אֶפְשָׁר לִרְאוֹת בּוֹ זְכוּת גְּדוֹלָה?
  • הַשָּׁבוּעַ, לִפְנֵי הַשֵּׁנָה, אֲנִי מוֹדֶה לַה׳ עַל שְׁלוֹשָׁה פְּרָטִים, וְרוֹשֵׁם בַּבֹּקֶר מָה הִרְגַּשְׁתִּי.

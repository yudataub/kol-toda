רָמָה ב׳ – מִדּוֹתָיו שֶׁל ה׳

🧭 אֵיךְ הוֹלְכִים אַחַר הַשְּׁכִינָה?

רַבִּי חָמָא בְּרַבִּי חֲנִינָא שָׁאַל עַל הַפָּסוּק ׳אַחֲרֵי ה׳ אֱלֹהֵיכֶם תֵּלֵכוּ׳: וְכִי אֶפְשָׁר לָאָדָם לָלֶכֶת אַחַר הַשְּׁכִינָה? הֲרֵי ׳ה׳ אֱלֹהֶיךָ אֵשׁ אוֹכְלָה הוּא׳!

וְהוּא עוֹנֶה: לָלֶכֶת אַחַר מִדּוֹתָיו. מָה הוּא מַלְבִּישׁ עֲרוּמִּים – אַף אַתָּה הַלְבֵּשׁ עֲרוּמִּים. מָה הוּא מְבַקֵּר חוֹלִים – אַף אַתָּה בַּקֵּר חוֹלִים. וְכָךְ בְּנִחוּם אֲבֵלִים וּבִקְבוּרַת מֵתִים.

הַהוֹרָאָה אַחַת: מִתְנַהֲגִים כְּמוֹ ה׳.

הוֹלְכִים אַחֲרֵי ה׳ כְּשֶׁמִּתְנַהֲגִים כְּמוֹ שֶׁהוּא מִתְנַהֵג.

רָמָה ב׳ – מִדּוֹתָיו שֶׁל ה׳

📜 אַרְבָּעָה פְּסוּקִים

לְכָל מִדָּה יֵשׁ פָּסוּק: הַלְבָּשָׁה – ׳וַיַּעַשׂ ה׳ אֱלֹהִים… כׇּתְנוֹת עוֹר וַיַּלְבִּשֵׁם׳. בִּקּוּר חוֹלִים – ׳וַיֵּרָא אֵלָיו ה׳ בְּאֵלֹנֵי מַמְרֵא׳. נִחוּם אֲבֵלִים – ׳וַיְבָרֶךְ אֱלֹהִים אֶת יִצְחָק בְּנוֹ׳ אַחֲרֵי מוֹת אַבְרָהָם. קְבוּרָה – ׳וַיִּקְבֹּר אֹתוֹ בַּגַּי׳.

שְׁלוֹשָׁה מֵהַפְּסוּקִים בְּסֵפֶר בְּרֵאשִׁית, וְהָרְבִיעִי בְּסֵפֶר דְּבָרִים, בְּסוֹף הַתּוֹרָה.

כָּךְ רוֹאִים שֶׁהַחֶסֶד נִמְצָא מֵהַתְחָלַת הַתּוֹרָה וְעַד סוֹפָהּ.

אַרְבַּע מִדּוֹת, אַרְבָּעָה פְּסוּקִים: שְׁלוֹשָׁה בִּבְרֵאשִׁית וְאֶחָד בִּדְבָרִים.

רָמָה ב׳ – מִדּוֹתָיו שֶׁל ה׳

🔁 הַחֶסֶד שֶׁלֹּא הִגִּיעַ לוֹ

הַגְּמָרָא מְצַטֶּטֶת אֶת רַבִּי שִׂמְלַאי: הַתּוֹרָה תְּחִלָּתָהּ חֶסֶד וְסוֹפָהּ חֶסֶד. אֲבָל כָּל הַבְּרִיאָה הִיא חֶסֶד, אָז לָמָּה דַּוְקָא הַלְבָּשָׁה?

כִּי חֶסֶד אֱמִתִּי הוּא חֶסֶד עִם מִי שֶׁלֹּא מַגִּיעַ לוֹ. אָדָם הָרִאשׁוֹן בְּדִיּוּק חָטָא, וּבְכָל זֹאת ה׳ הִלְבִּישׁ אוֹתוֹ.

הַמַּהֲרַ״ל כָּתַב שֶׁהַהַתְחָלָה וְהַסּוֹף הֵם עִקַּר הַדָּבָר, וּבֵינֵיהֶם הַכֹּל מֵאִיר.

הַחֶסֶד הָרִאשׁוֹן נִתַּן גַּם לְמִי שֶׁטָּעָה.

רָמָה ב׳ – מִדּוֹתָיו שֶׁל ה׳

⏰ לְבַקֵּר בְּחֶסֶד

מִי שֶׁעוֹשֶׂה חֶסֶד חוֹשֵׁב מָה יִהְיֶה הֲכִי טוֹב לַמְקַבֵּל. לָכֵן הוּא יָבוֹא לְבַקֵּר אֶת הַחוֹלֶה בַּשָּׁעָה שֶׁנּוֹחַ לַחוֹלֶה, וְלֹא בַּשָּׁעָה שֶׁנּוֹחַ לוֹ.

הַחֶסֶד הוּא לֹא רַק מַעֲשֶׂה, אֶלָּא גַּם מַחֲשָׁבָה: שׁוֹאֲלִים ׳מָה הוּא צָרִיךְ?׳ וְלֹא רַק ׳מָה נוֹחַ לִי?׳.

כָּךְ גַּם בְּנִחוּם אֲבֵלִים: שׁוֹמְעִים, וְלֹא מְדַבְּרִים בִּמְקוֹמוֹ.

חֶסֶד זֶה לַחֲשֹׁב מָה טוֹב לַמְקַבֵּל.

📖 חֶסֶד – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – מִדּוֹתָיו שֶׁל ה׳

💭 מַעֲשִׂים קְטַנִּים שֶׁנַּעֲשִׂים חֶסֶד

אֶפְשָׁר לַהֲפֹךְ גַּם מַעֲשִׂים קְטַנִּים לְחֶסֶד. אִמָּא שֶׁמַּלְבִּישָׁה אֶת יְלָדֶיהָ יְכוֹלָה לַחֲשֹׁב: אֲנִי עוֹזֶרֶת לְבֶן שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא.

כָּךְ גַּם אֲמִירַת שָׁלוֹם, הַקְשָׁבָה לְחָבֵר אוֹ תְּפִלָּה עַל חָבֵר.

מַחֲשָׁבָה אַחַת – אֲנִי עוֹשֶׂה רְצוֹן ה׳ וּמֵיטִיב לִילָדָיו – מַעֲלָה אֶת הַמַּעֲשֶׂה לְמִצְוַת חֶסֶד.

מַחֲשָׁבָה אַחַת טוֹבָה מַעֲלָה מַעֲשֶׂה קָטָן לְמִצְוַת חֶסֶד.

📖 חֶסֶד – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אָבוֹת א׳ ב׳ · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – מִדּוֹתָיו שֶׁל ה׳

📚 תּוֹרָה בִּשְׁבִיל חֶסֶד

בְּאָבוֹת א׳ ב׳ כָּתוּב שֶׁהָעוֹלָם עוֹמֵד עַל שְׁלוֹשָׁה דְּבָרִים: עַל הַתּוֹרָה, עַל הָעֲבוֹדָה וְעַל גְּמִילוּת חֲסָדִים.

אַבְרָהָם, אִישׁ הַחֶסֶד, הוּא הָרִאשׁוֹן שֶׁבָּאָבוֹת – וּבְכָל זֹאת הַתּוֹרָה כְּתוּבָה רִאשׁוֹנָה.

הָרַב לָמַד מִכָּאן: מַקְדִּימִים תּוֹרָה כְּדֵי לָבוֹא לְחֶסֶד אֱמִתִּי.

לוֹמְדִים תּוֹרָה כְּדֵי לָדַעַת אֵיךְ לַעֲשׂוֹת חֶסֶד נָכוֹן.

מָה לוֹקְחִים מִרָמָה ב׳?

הַגְּמָרָא מְלַמֶּדֶת אוֹתָנוּ לָלֶכֶת אַחַר מִדּוֹתָיו שֶׁל ה׳. כְּשֶׁאֲנִי מַלְבִּישׁ, מְבַקֵּר וּמְנַחֵם, אֲנִי מִתְקָרֵב אֵלָיו וְאוֹהֵב אוֹתוֹ יוֹתֵר.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה הַחֶסֶד שֶׁעָשִׂיתִי הַשָּׁבוּעַ בְּדִיּוּק כְּמוֹ ה׳ – לְהַלְבִּישׁ, לְבַקֵּר אוֹ לְנַחֵם?
  • מִי הוּא חָבֵר שֶׁאֲנִי יָכוֹל לַעֲזֹר לוֹ גַּם אִם לֹא הִתְנַהֵג יָפֶה אֵלַי?
  • אֶבְחַר מַדְרֵגָה אַחַת שֶׁל חֶסֶד, אֶעֱשֶׂה אוֹתָהּ בַּשָּׁעָה שֶׁנּוֹחַ לַמְקַבֵּל, וְאֶרְשֹׁם מִי הָיָה וּמָה הִרְגַּשְׁתִּי.

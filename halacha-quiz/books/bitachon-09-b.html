מִבְחֲנֵי בִּטָּחוֹן קְטַנִּים

רָמָה ב׳ – מִתְחַזְּקִים כָּל יוֹם

מָה עוֹשִׂים כְּשֶׁהָאוֹטוֹבּוּס נָסַע, הַמִּשְׁקָפַיִם נִשְׁבְּרוּ אוֹ חָבֵר לֹא רוֹצֶה – וְאֵיךְ מִתְאַמְּנִים בְּרוֹגַע כָּל יוֹם

📖 בִּטָּחוֹן בַּה׳ · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
📋

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹדֵק אֶת הַתְּגוּבָה שֶׁלִּי וּמְחַזֵּק אֶת הַבִּטָּחוֹן 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִבְחֲנֵי אֱמוּנָה עַצְמִיִּים

רָמָה ב׳ – מִתְחַזְּקִים כָּל יוֹם

📋 מִבְחֲנֵי אֱמוּנָה עַצְמִיִּים

בְּדַף הַשִּׁעוּר יֵשׁ רְשִׁימָה שֶׁל מִבְחָנִים שֶׁשּׁוֹאֲלִים בָּהֶם: מָה תְּגוּבָתִי כְּשֶׁאֲנִי לֹא מְרֻצֶּה?

לְמָשָׁל: הָאוֹטוֹבּוּס נָסַע, הָאֲרוּחָה לֹא הִגִּיעָה, אֵין בֶּגֶד מְגֻהָץ, מִשְׁקָפַיִם אוֹ קְעָרָה נִשְׁבְּרוּ, חָבֵר טָעָה אוֹ מְסָרֵב לַעֲשׂוֹת אֶת רְצוֹנִי, נֶאֱבַד לִי חֵפֶץ בְּרֶגַע לָחוּץ, אוֹ שֶׁנִּפְּלָה טָעוּת בִּמְלָאכָה.

הַמַּטָּרָה: לִרְאוֹת בְּאֵיזֶה מַצָּבִים הַבִּטָּחוֹן שֶׁלִּי חָזָק, וּבְאֵילוּ צָרִיךְ לְחַזֵּק.

הַשְּׁאֵלָה בַּמִּבְחָן הִיא לֹא ׳מָה קָרָה׳ אֶלָּא ׳מָה תְּגוּבָתִי׳.

1
🤲

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁהַבִּטָּחוֹן יִהְיֶה מִדָּה קְבוּעָה שֶׁלִּי 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – מִתְחַזְּקִים כָּל יוֹם

🤲 הַכֹּל מֵאֵת אַבָּא טוֹב

לְפִי הַשִּׁעוּר, מִידַּת הַבִּטָּחוֹן פֵּרוּשָׁהּ שֶׁאָדָם חַי בַּהַכָּרָה שֶׁאֵין מִקְרֶה, הַכֹּל מֵאֵת ה׳ שֶׁהוּא רַק טוֹב וּמֵיטִיב.

לָכֵן הוּא לֹא עָצוּב עַל מַה שֶּׁקּוֹרֶה, וְלֹא דּוֹאֵג מָה יִקְרֶה, וְלֹא פּוֹחֵד מִמָּה שֶׁיַּעֲשׂוּ לוֹ.

מִי שֶׁפַּעַם דּוֹאֵג וּפַעַם בּוֹטֵחַ מִתְנַהֵג בְּבִטָּחוֹן, אֲבָל עֲדַיִן לֹא קָנָה אֶת הַמִּדָּה.

מִדָּה הִיא דֶּרֶךְ קְבוּעָה שֶׁנִּקְנֵית בַּלֵּב, לֹא פַּעַם אַחַת.

2
🧗

💛 אַהֲבַת הַשֵּׁם: גַּם אִם אֲנִי יוֹדֵעַ, אֲנִי מִתְחַזֵּק שׁוּב כָּל יוֹם 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַבִּי יְחֶזְקֵאל לֵוִינְשְׁטֵיין

רָמָה ב׳ – מִתְחַזְּקִים כָּל יוֹם

🧗 גַּם הַגָּדוֹל צָרִיךְ לְהִתְחַזֵּק

עַל הַמַּשְׁגִּיחַ רַבִּי יְחֶזְקֵאל לֵוִינְשְׁטֵיין מְסֻפָּר שֶׁעָמַד פַּעַם בִּישִׁיבַת פּוֹנֶבֶז׳ וְאָמַר: ׳רַבּוֹתַי, נַעַר הָיִיתִי וְגַם זָקַנְתִּי, וַאֲנִי מְדַבֵּר עֲשָׂרוֹת שָׁנִים עַל אֱמוּנָה. אֲנִי מַרְגִּישׁ שֶׁאִם אֲנִי מַסִּיחַ אֶת עַצְמִי כַּמָּה דַּקּוֹת מֵהָאֱמוּנָה, אֲנִי נוֹפֵל מִמַּדְרֵגָתִי׳.

הוּא הָיָה אָז קָרוֹב לְגִיל תִּשְׁעִים.

אָנוּ אֵינֵנוּ בְּמַדְרֵגָתוֹ, אֲבָל נָבִין שֶׁצָּרִיךְ לְהִתְחַזֵּק כָּל יוֹם.

הַטָּעוּת הַנְּפוֹצָה: ׳פַּעַם לָמַדְתִּי אֱמוּנָה וְדַי׳. צָרִיךְ כָּל יוֹם.

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד מְעַט בִּטָּחוֹן כָּל בֹּקֶר 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – מִתְחַזְּקִים כָּל יוֹם

📚 אֵיךְ קוֹנִים בִּטָּחוֹן

לְפִי הַשִּׁעוּר, קוֹנִים בִּטָּחוֹן בְּלִמּוּד סְפָרִים עַל כָּךְ, כְּמוֹ ׳נֵר לְרַגְלִי׳ ו׳גַּן הָאֱמוּנָה׳.

יָפֶה אִם הַלִּמּוּד הָרִאשׁוֹן בַּיּוֹם, לִפְנֵי כָּל לִמּוּד אַחֵר, הוּא לִמּוּד בְּעִנְיָנֵי אֱמוּנָה וּבִטָּחוֹן, כִּי זוֹ תַּכְלִית הַחַיִּים.

יֶלֶד יָכוֹל לְהַתְחִיל בְּכַמָּה דַּקּוֹת קְטַנּוֹת בְּכָל בֹּקֶר.

קוֹנִים בִּטָּחוֹן בְּלִמּוּד קָטָן, כָּל יוֹם.

🌟

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד תּוֹרָה כְּדֵי לִבְטֹחַ בְּה׳ 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הַגְּרָ״א, אֶבֶן שְׁלֵמָה ג׳

רָמָה ב׳ – מִתְחַזְּקִים כָּל יוֹם

🌟 עִקַּר הַכֹּל

הַגְּרָ״א (הַגָּאוֹן מִוִּילְנָא) כָּתַב: ׳וְעִקַּר הַכֹּל הוּא הַבִּטָּחוֹן, וּמִי שֶׁהוּא מְחֻסַּר בִּטָּחוֹן אֵין תּוֹרָתוֹ מִתְקַיֶּמֶת בְּיָדוֹ׳.

עוֹד כָּתַב הַגְּרָ״א שֶׁעִקַּר מַתַּן הַתּוֹרָה לְיִשְׂרָאֵל הוּא כְּדֵי שֶׁיָּשִׂימוּ בִּטְחוֹנָם בַּה׳.

כְּלוֹמַר, הַלִּמּוּד וְהַמִּצְווֹת בָּאִים לְהַשְׁפִּיעַ עָלֵינוּ וּלְהַגְדִּיל אֶת הַבִּטָּחוֹן.

הַבִּטָּחוֹן הוּא מַטָּרָה שֶׁל הַתּוֹרָה, וְהוּא מְקַיֵּם אוֹתָהּ בְּיָדֵינוּ.

מָה לוֹקְחִים מִרָמָה ב׳?

בִּטָּחוֹן הוּא קִרְבָה לה׳. כְּשֶׁאֲנִי מְחַזֵּק אוֹתוֹ כָּל יוֹם, אֲנִי מַרְגִּישׁ שֶׁאַבָּא טוֹב אוֹהֵב אוֹתִי וְשׁוֹמֵר עָלַי.

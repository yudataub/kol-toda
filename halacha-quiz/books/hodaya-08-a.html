וְעַתָּה נַפְשֵׁנוּ יְבֵשָׁה אֵין כֹּל בִּלְתִּי אֶל הַמָּן עֵינֵינוּ

🌱 בִּלְתִּי אֶל הַמָּן · רָמָה א׳

רְאוּ בָּנַי עַל מָה מִתְלוֹנְנִים

רָמָה א׳ – אֵיפֹה הַחֲבִיתָה?

הַמָּן בַּמִּדְבָּר, הַשֻּׁלְחָן הַמָּלֵא וְהַשְּׁאֵלָה ׳אֵיפֹה הַחֲבִיתָה?׳ – וְאֵיךְ הוֹפְכִים תְּלוּנָה לְתוֹדָה

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִסְתַּכֵּל עַל הַשֻּׁלְחָן וְרוֹאֶה כַּמָּה טוֹב ה׳ נָתַן לִי 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – אֵיפֹה הַחֲבִיתָה?

🍽️ אֵיפֹה הַחֲבִיתָה?

אַבָּא חוֹזֵר הַבַּיְתָה בְּרָעָב. עַל הַשֻּׁלְחָן מְחַכִּים סָלָט, לֶחֶם וְחֻמּוּס. אֲבָל הוּא מִסְתַּכֵּל וְשׁוֹאֵל: ׳אֵיפֹה הַחֲבִיתָה? אֵיפֹה הַחֲבִיתָה?׳ וְנִשְׁאָר מִתְעַצְבֵּן.

זֶה מַצְחִיק, נָכוֹן? הַשֻּׁלְחָן מָלֵא, וְהוּא רוֹאֶה רַק מָה שֶׁחָסֵר.

בַּשִּׁעוּר עַל הַמָּן לוֹמְדִים שֶׁגַּם אֲנַחְנוּ עוֹשִׂים לִפְעָמִים כָּךְ.

אֶפְשָׁר לִרְאוֹת שֻׁלְחָן מָלֵא וְלִשְׁאֹל רַק עַל מָה שֶׁחָסֵר.

1

💛 אַהֲבַת הַשֵּׁם: תּוֹדָה, ה׳, שֶׁאַתָּה נוֹתֵן לִי אֹכֶל כָּל יוֹם 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – אֵיפֹה הַחֲבִיתָה?

🍞 הַמָּן: אֹכֶל מְיֻחָד

בַּמִּדְבָּר לֹא הָיְתָה חֲנוּת וְלֹא שָׂדֶה, וַה׳ הוֹרִיד לְעַם יִשְׂרָאֵל אֹכֶל בְּכָל יוֹם: הַמָּן.

הַמָּן הָיָה אֹכֶל נִפְלָא, וְהָיָה בּוֹ כָּל טַעַם שֶׁרָצוּ לִטְעֹם.

הַכֹּל מוּכָן, וַאֲפִלּוּ הַטַּעַם לְפִי הָרָצוֹן!

הַמָּן הָיָה אֹכֶל נִפְלָא, עִם כָּל טַעַם שֶׁרָצוּ.

2

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא אוֹמֵר ׳אֵין לִי כְּלוּם׳, כִּי ה׳ נוֹתֵן לִי הַרְבֵּה 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בַּמִּדְבָּר י״א ד׳–ו׳

רָמָה א׳ – אֵיפֹה הַחֲבִיתָה?

😠 בִּלְתִּי אֶל הַמָּן עֵינֵינוּ

וְעִם כָּל זֶה, כָּתוּב בַּתּוֹרָה שֶׁהָעָם הִתְאַוָּה תַּאֲוָה לְאֹכֶל אַחֵר, וְהִתְחִילוּ לִבְכּוֹת: ׳מִי יַאֲכִילֵנוּ בָּשָׂר?׳

הֵם זָכְרוּ אֶת הַדָּגִים שֶׁאָכְלוּ בְּמִצְרַיִם, וְאָמְרוּ: ׳אֵין כֹּל, בִּלְתִּי אֶל הַמָּן עֵינֵינוּ׳.

שְׁמַעְתֶּם? הֵם אָמְרוּ ׳אֵין כֹּל׳ עַל אֹכֶל שֶׁיָּרַד מֵה׳ בְּכָל יוֹם!

אָמְרוּ ׳אֵין כֹּל׳, וְהָיָה לָהֶם מָן בְּכָל יוֹם.

3

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁה׳ יִשְׂמַח כְּשֶׁהוּא רוֹאֶה אוֹתִי מוֹדֶה 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַשִׁ״י, בַּמִּדְבָּר י״א ז׳

רָמָה א׳ – אֵיפֹה הַחֲבִיתָה?

👀 רְאוּ בָּנַי עַל מָה מִתְלוֹנְנִים

הַתּוֹרָה מְתָאֶרֶת אֶת הַמָּן: ׳וְהַמָּן כְּזֶרַע גַּד הוּא׳. רַשִׁ״י מְסַבֵּר: ה׳ כָּתַב אֶת זֶה בַּתּוֹרָה כְּדֵי לוֹמַר לְכָל הָעוֹלָם: ׳רְאוּ בָּנַי עַל מָה מִתְלוֹנְנִים!׳

כְּלוֹמַר: תִּרְאוּ כַּמָּה חָשׁוּב הָיָה הַמָּן, וְהֵם מִתְלוֹנְנִים עָלָיו.

ה׳ אוֹמֵר: ׳רְאוּ עַל מָה בָּנַי מִתְלוֹנְנִים׳.

4

💛 אַהֲבַת הַשֵּׁם: אֲנִי סוֹפֵר אֶת הַטּוֹבוֹת שֶׁיֵּשׁ לִי, וְנֶהֱנֶה מֵהֶן 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – אֵיפֹה הַחֲבִיתָה?

🧠 מִי שֶׁרוֹאֶה רַק מָה שֶׁאֵין

מִי שֶׁמִּתְאַוֶּה לְמָה שֶׁאֵין לוֹ, לֹא מַחֲשִׁיב אֶת מָה שֶׁיֵּשׁ לוֹ.

וּמִי שֶׁלֹּא מַחֲשִׁיב – לֹא נֶהֱנֶה. אֲפִלּוּ שֶׁהַשֻּׁלְחָן מָלֵא!

הָעֵצָה: לְהַתְחִיל לִסְפֹּר אֶת מָה שֶׁכְּבָר יֵשׁ.

מִי שֶׁרוֹאֶה רַק מָה שֶׁאֵין, לֹא נֶהֱנֶה מִמָּה שֶׁיֵּשׁ.

5

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁקָּשֶׁה לִי, אֲנִי זוֹכֵר כַּמָּה טוֹב יֵשׁ לִי 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – אֵיפֹה הַחֲבִיתָה?

😮‍💨 תְּלוּנָה גְּדוֹלָה עַל כְּאֵב קָטָן

לִפְעָמִים יֶלֶד עָיֵף, וְקָצָת כּוֹאֵב לוֹ הָרֹאשׁ, וְהוּא מִתְלוֹנֵן וּמִתְלוֹנֵן כְּאִלּוּ אֵין לוֹ כְּלוּם.

אֲבָל יֵשׁ לוֹ בַּיִת, מִשְׁפָּחָה, אֹכֶל, בְּגָדִים וּבְרִיאוּת.

ה׳ מִסְתַּכֵּל וְאוֹמֵר: ׳רְאוּ בָּנַי עַל מָה מִתְלוֹנְנִים׳. כְּשֶׁכּוֹאֵב מַשֶּׁהוּ קָטָן מוּתָּר לוֹמַר, אֲבָל לֹא לִשְׁכֹּחַ אֶת כָּל הַטּוֹב.

אֲפִלּוּ כְּשֶׁכּוֹאֵב מַשֶּׁהוּ, אֲנִי לֹא שׁוֹכֵחַ אֶת כָּל הַטּוֹב.

6

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה לְךָ, ה׳, עַל כָּל נְשִׁימָה 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים צ״ב ב׳ · תְּהִלִּים ק״נ ו׳

רָמָה א׳ – אֵיפֹה הַחֲבִיתָה?

🌟 הָעֵצָה: טוֹב לְהוֹדוֹת

מָה עוֹשִׂים? בַּתְּהִלִּים כָּתוּב: ׳טוֹב לְהֹדוֹת לַה׳׳. כְּשֶׁמַּתְחִילִים לְהוֹדוֹת, מַתְחִילִים לְהִסְתַּכֵּל עַל מָה שֶׁיֵּשׁ וְלֹא עַל מָה שֶׁאֵין.

וּבְסוֹף סֵפֶר תְּהִלִּים כָּתוּב: ׳כֹּל הַנְּשָׁמָה תְּהַלֵּל יָהּ׳ – כָּל נְשִׁימָה שֶׁלָּנוּ הִיא סִבָּה לְהוֹדוֹת.

כְּשֶׁמּוֹדִים, מַתְחִילִים לִרְאוֹת מָה שֶׁיֵּשׁ.

7

💛 אַהֲבַת הַשֵּׁם: אֲנִי הוֹפֵךְ תְּלוּנָה לְתוֹדָה, וְהַלֵּב שֶׁלִּי נִפְתָּח 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – אֵיפֹה הַחֲבִיתָה?

✏️ תְּלוּנָה שֶׁהָפְכָה לְתוֹדָה

אֶפְשָׁר לְשַׁנּוֹת מִשְׁפָּט אֶחָד, וְהַלֵּב מִשְׁתַּנֶּה. ׳אֵין לִי חֲבִיתָה!׳ ← ׳תּוֹדָה עַל הַלֶּחֶם וְהַסָּלָט!׳

׳הַמִּבְחָן קָשֶׁה׳ ← ׳תּוֹדָה שֶׁאֲנִי יָכוֹל לִלְמֹד׳.

תְּלוּנָה אַחַת שֶׁהָפַכְתִּי לְתוֹדָה – זוֹ הַמְּשִׂימָה שֶׁל הַשָּׁבוּעַ.

מַחֲלִיפִים ׳אֵין לִי׳ בְּ׳תּוֹדָה עַל׳.

8
💡

רָמָה א׳ – אֵיפֹה הַחֲבִיתָה?

מָה לוֹקְחִים מִרָמָה א׳?

מִי שֶׁמְּאַמֵּן אֶת עַצְמוֹ לִרְאוֹת אֶת מָה שֶׁיֵּשׁ, נֶהֱנֶה מִכָּל פֵּרוּסַת לֶחֶם וּמִכָּל שֻׁלְחָן. הַהוֹדָיָה לַה׳ מַחְזִירָה אוֹתָנוּ לִרְאוֹת כַּמָּה הוּא אוֹהֵב אוֹתָנוּ וְנוֹתֵן לָנוּ, וְכָךְ אַהֲבָתֵנוּ אֵלָיו גְּדֵלָה.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָה הַתְּלוּנָה שֶׁאָמַרְתִּי הַיּוֹם, וְאֵיךְ אֶפְשָׁר לְהַפְכָּהּ לְתוֹדָה?
  • עַל אֵילוּ דְּבָרִים שֶׁיֵּשׁ עַל הַשֻּׁלְחָן שֶׁלִּי אֲנִי כִּמְעַט אַף פַּעַם לֹא אוֹמֵר תּוֹדָה?

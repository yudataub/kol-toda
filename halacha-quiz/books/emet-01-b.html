הָעוֹלָם עוֹמֵד עַל הָאֱמֶת

רָמָה ב׳ – אַנְשֵׁי אֱמֶת וּמַנְהִיגִים

מִי הוּא אִישׁ אֱמֶת, לָמָּה יִתְרוֹ בִּקֵּשׁ שָׂרִים שֶׁהֵם אַנְשֵׁי אֱמֶת, וְאֵיךְ הוֹלְכִים בְּדֶרֶךְ הָאֱמֶת

📖 אֱמֶת · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
⚖️

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹסִיף לָעוֹלָם אֱמֶת כְּשֶׁאֲנִי מְדַבֵּר אֱמֶת לַחֲבֵרִים 💛

📖 אֱמֶת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אָבוֹת א׳ י״ח · זְכַרְיָה ח׳ ט״ז

רָמָה ב׳ – אַנְשֵׁי אֱמֶת וּמַנְהִיגִים

⚖️ שְׁלוֹשָׁה תְּנָאִים לַחֶבְרָה

בְּפִרְקֵי אָבוֹת: ׳עַל שְׁלוֹשָׁה דְבָרִים הָעוֹלָם עוֹמֵד, עַל הַדִּין וְעַל הָאֱמֶת וְעַל הַשָּׁלוֹם׳, וְהַמִּשְׁנָה מְבִיאָה אֶת הַפָּסוּק: ׳אֱמֶת וּמִשְׁפַּט שָׁלוֹם שִׁפְטוּ בְּשַׁעֲרֵיכֶם׳.

שְׁלוֹשָׁה תְּנָאִים לְחֶבְרָה תַּקִּינָה: דִּין – מִשְׁפָּט יָשָׁר, אֱמֶת – אֶפְשָׁר לִסְמֹךְ עַל מָה שֶׁאוֹמְרִים, שָׁלוֹם – חַיִּים יַחַד בְּלִי מְרִיבוֹת.

בַּנָּבִיא זְכַרְיָה כָּתוּב גַּם: ׳דַּבְּרוּ אֱמֶת אִישׁ אֶת רֵעֵהוּ׳.

דִּין, אֱמֶת וְשָׁלוֹם: שְׁלוֹשָׁה תְּנָאִים לְחֶבְרָה תַּקִּינָה.

1
🧭

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד לְהַנְהִיג אֶת עַצְמִי בְּאֱמֶת 💛

📖 אֱמֶת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁמוֹת י״ח כ״א · רַשִׁ״י

רָמָה ב׳ – אַנְשֵׁי אֱמֶת וּמַנְהִיגִים

🧭 מַנְהִיג שֶׁאֶפְשָׁר לִסְמֹךְ עָלָיו

יִתְרוֹ יָעַץ לְמֹשֶׁה: ׳וְאַתָּה תֶחֱזֶה מִכָּל הָעָם אַנְשֵׁי חַיִל, יִרְאֵי אֱלֹהִים, אַנְשֵׁי אֱמֶת, שֹׂנְאֵי בָצַע׳ – וְאֶת הַמַּנְהִיגִים הָאֵלֶּה לִמְנוֹת לְשָׂרִים.

רַשִׁ״י: אַנְשֵׁי אֱמֶת הֵם בַּעֲלֵי הַבְטָחָה שֶׁכְּדַאי לִסְמֹךְ עַל דִּבְרֵיהֶם.

מִי שֶׁשּׁוֹפֵט אֲנָשִׁים צָרִיךְ שֶׁיֵּדְעוּ שֶׁהַמִּלִּים שֶׁלּוֹ נֶאֱמָנוֹת, אֲחֶרֶת אֵין עַל מָה לִסְמֹךְ בְּהַחְלָטוֹת שֶׁלּוֹ.

מַנְהִיג צָרִיךְ לִהְיוֹת אִישׁ אֱמֶת: בַּעַל הַבְטָחָה.

2
🏔️

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לָגוּר קָרוֹב לַה׳ בְּדֶרֶךְ הָאֱמֶת 💛

📖 אֱמֶת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים ט״ו

רָמָה ב׳ – אַנְשֵׁי אֱמֶת וּמַנְהִיגִים

🏔️ מִי יָגוּר בְּאָהֳלֶךָ?

הַמִּזְמוֹר מְתָאֵר אֶת הָאָדָם הַקָּרוֹב לַה׳: הוֹלֵךְ תָּמִים, פּוֹעֵל צֶדֶק וְדוֹבֵר אֱמֶת בִּלְבָבוֹ.

עוֹד כָּתוּב: ׳לֹא רָגַל עַל לְשֹׁנוֹ׳ (אֵינוֹ מְדַבֵּר רָעָה עַל אֲחֵרִים), ׳לֹא עָשָׂה לְרֵעֵהוּ רָעָה׳, וְ׳שֹׁחַד עַל נָקִי לֹא לָקָח׳.

וּבַסּוֹף הַשָּׂכָר: ׳עֹשֵׂה אֵלֶּה לֹא יִמּוֹט לְעוֹלָם׳ – מִי שֶׁעוֹשֶׂה כָּךְ יַעֲמֹד יַצִּיב.

דּוֹבֵר אֱמֶת בִּלְבָבוֹ – קָרוֹב לַה׳ וְלֹא יִמּוֹט לְעוֹלָם.

3
🔍

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת כֵּן עִם עַצְמִי וְעִם ה׳ 💛

📖 אֱמֶת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרַב שְׁלֹמֹה אֲבִינֵר, תּוֹרַת אִמְּךָ

רָמָה ב׳ – אַנְשֵׁי אֱמֶת וּמַנְהִיגִים

🔍 הָאֱמֶת מְאַמֶּתֶת אֶת הַכֹּל

לְפִי דִּבְרֵי הָרַב שְׁלֹמֹה אֲבִינֵר: הָאֱמֶת עוֹשָׂה שֶׁכָּל הָעִנְיָנִים הָאֲחֵרִים יִהְיוּ אֲמִתִּיִּים.

מִי שֶׁחָסֵר לוֹ מִדַּת הָאֱמֶת, כָּל עֲרָכָיו מְסֻלָּפִים: הוּא מְשַׁקֵּר לַאֲחֵרִים וּמְשַׁקֵּר גַּם לְעַצְמוֹ.

לָכֵן אֱמֶת אֵינָהּ רַק לְהַגִּיד אֱמֶת לַחֲבֵרִים, אֶלָּא גַּם לִהְיוֹת כֵּן עִם עַצְמִי.

אֱמֶת עִם אֲחֵרִים וְאֱמֶת עִם עַצְמִי.

4
🪜

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁאֲנִי נִכְשָׁל, אֲנִי מַתְחִיל שׁוּב וְה׳ אוֹהֵב אוֹתִי 💛

📖 אֱמֶת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – אַנְשֵׁי אֱמֶת וּמַנְהִיגִים

🪜 מִתְקַדְּמִים צַעַד אַחַר צַעַד

עַל הָאָדָם מֻטֶּלֶת הַחוֹבָה לְהִשְׁתַּדֵּל לְהִתְקַדֵּם תָּמִיד בְּשִׁפּוּר הַמִּדּוֹת שֶׁלּוֹ, גַּם אִם עוֹד לֹא הִגִּיעַ לַשְּׁלֵמוּת.

לָכֵן כְּשֶׁמְּגַלִּים שֶׁלֹּא קִיַּמְנוּ הַבְטָחָה, לֹא מִתְיָאֲשִׁים: מְתַקְּנִים, וּמַתְחִילִים שׁוּב.

כָּל הַבְטָחָה שֶׁמְּקַיְּמִים הִיא עוֹד צַעַד בַּסֻּלָּם.

מִתְקַדְּמִים בַּמִּדּוֹת צַעַד אַחַר צַעַד, וְלֹא מִתְיָאֲשִׁים.

5
💡

רָמָה ב׳ – אַנְשֵׁי אֱמֶת וּמַנְהִיגִים

מָה לוֹקְחִים מִרָמָה ב׳?

אִישׁ אֱמֶת הוּא עַמּוּד שֶׁל הָעוֹלָם. הַמִּזְמוֹר מְלַמֵּד שֶׁהַדּוֹבֵר אֱמֶת בִּלְבָבוֹ זוֹכֶה לִקְרָבַת ה׳ וְלֹא יִמּוֹט לְעוֹלָם. כָּל הַבְטָחָה שֶׁאֲנִי מְקַיֵּם מְקָרֶבֶת אוֹתִי אֶל ה׳.

6
❤️
עַל שְׁלוֹשָׁה דְבָרִים הָעוֹלָם עוֹמֵד, עַל הַדִּין וְעַל הָאֱמֶת וְעַל הַשָּׁלוֹם

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • מִי הוּא אִישׁ אֱמֶת שֶׁאֲנִי מַכִּיר, וּמָה גּוֹרֵם לִי לִסְמֹךְ עַל דְּבָרָיו?
  • הַאִם הִבְטַחְתִּי הַשָּׁבוּעַ מַשֶּׁהוּ וְלֹא קִיַּמְתִּי, וְאֵיךְ אֲתַקֵּן אוֹ אֶזָּהֵר בַּפַּעַם הַבָּאָה?
  • אֶכְתֹּב עַל אִישׁ אֱמֶת שֶׁאֲנִי מַכִּיר: לָמָּה בָּחַרְתִּי בּוֹ וּשְׁתֵּי דֻּגְמָאוֹת לְכָךְ – וְאֶבְחַר הַבְטָחָה אַחַת שֶׁאֲקַיֵּם הַשָּׁבוּעַ.

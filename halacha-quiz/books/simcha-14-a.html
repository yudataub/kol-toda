יֶלֶד קִבֵּל גְּלִידָה גְּדוֹלָה וְטְעִימָה, וְהוּא קוֹרֵא: ׳אֵין לִי כַּפִּית!׳ הוּא לֹא רוֹאֶה שֶׁיֵּשׁ לוֹ גְּלִידָה.

יֵשׁ אֲנָשִׁים שֶׁמַּסְתַּכְּלִים עַל הָאַיִן, וְיֵשׁ שֶׁמַּסְתַּכְּלִים עַל הַיֵּשׁ. בַּשִּׁעוּר הַזֶּה לוֹמְדִים לִרְאוֹת אֶת הַ׳יֵּשׁ׳.

מַסְתַּכְּלִים עַל הַיֵּשׁ, לֹא רַק עַל הָאַיִן.

הָמָן הָרָשָׁע הָיָה אִישׁ חָשׁוּב מְאֹד בְּמַלְכוּת פָּרַס. הוּא סִפֵּר לַאֲנָשָׁיו עַל כְּבוֹד עָשְׁרוֹ וְרֹב בָּנָיו וְעַל כָּל מַה שֶׁהַמֶּלֶךְ גִּדֵּל אוֹתוֹ וְנִשָּׂא אוֹתוֹ עַל הַשָּׂרִים.

וְגַם הַמַּלְכָּה אֶסְתֵּר הִזְמִינָה לַמִּשְׁתֶּה רַק אוֹתוֹ עִם הַמֶּלֶךְ. הָיָה לוֹ הַכֹּל!

לְהָמָן הָיָה עֹשֶׁר, בָּנִים וְכָבוֹד.

😠 ׳וְכָל זֶה אֵינֶנּוּ שֹׁוֶה לִי׳

אֲבָל מָה אָמַר הָמָן? ׳וְכָל זֶה אֵינֶנּוּ שֹׁוֶה לִי בְּכָל עֵת אֲשֶׁר אֲנִי רֹאֶה אֶת מָרְדֳּכַי הַיְּהוּדִי יוֹשֵׁב בְּשַׁעַר הַמֶּלֶךְ׳.

יְהוּדִי אֶחָד לֹא הִשְׁתַּחֲוָה לוֹ – וְהוּא שָׁכַח אֶת כָּל הַיֵּשׁ וְרָאָה רַק אֶת הָאַיִן הַקָּטָן.

הָמָן שָׁכַח אֶת הַיֵּשׁ וְרָאָה רַק אֶת הָאַיִן.

💛 אַהֲבַת הַשֵּׁם: אֲנִי לָמֵד מֵהַמּוֹרָה לְהַסְתַּכֵּל גַּם עַל מַה שֶׁיֵּשׁ 💛

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, ׳לְהַסְתַּכֵּל עַל מַה שֶׁיֵּשׁ׳

רָמָה א׳ – יֵשׁ וְאַיִן

🛋️ הַמּוֹרָה וְהַסָּלוֹן

מוֹרָה אַחַת סִפְּרָה: גַּרְתִּי בְּדִירָה קְטַנָּה בְּקוֹמָה רְבִיעִית, וְעָבַרְנוּ לְדִירָה גְּדוֹלָה עִם אַרְבָּעָה חֲדָרִים, מִטְבָּח גָּדוֹל וּמִרְפְּסוֹת. הָיִינוּ שְׂמֵחִים מְאֹד.

אֲבָל לִקְרַאת הַחַג הִגִּיעוּ אוֹרְחִים, וְהָיָה צָרִיךְ לְהַגְדִּיל אֶת הַסָּלוֹן. לְשֵׁם כָּךְ חָסְרוּ לָנוּ חֲמֵשׁ מֵאוֹת דּוֹלָר, וְלֹא הָיוּ לָנוּ.

חָסְרוּ חֲמֵשׁ מֵאוֹת דּוֹלָר לְהַגְדִּיל סָלוֹן.

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁחָסֵר לִי דָּבָר, אֲנִי לֹא שׁוֹכֵחַ אֶת כָּל הַטּוֹב שֶׁיֵּשׁ לִי 💛

😞 מָה זָכְרָה?

מֵאָז הִתְחִילָה לָלֶכֶת עֲצוּבָה. מַחֲשָׁבָה אַחַת מִלְּאָה אֶת הָרֹאשׁ: ׳אֲנִי צְרִיכָה חֲמֵשׁ מֵאוֹת דּוֹלָר, וְאֵין לִי׳.

הִיא שָׁכְחָה אֶת הַשִּׂמְחָה בַּדִּירָה הַחֲדָשָׁה, אֶת הַשְּׁכֵנִים שֶׁהֵם שׁוֹמְרֵי תּוֹרָה, וַאֲפִלּוּ אֶת שְׁנֵי הַכִּיּוֹרִים בַּמִּטְבָּח. הִיא זָכְרָה רַק מַה שֶׁאֵין.

הִיא זָכְרָה רַק אֶת הָאַיִן.

💡 הִיא מֵבִינָה

בְּעֶרֶב פּוּרִים הִגִּיעָה לְהַרְצָאָה. הַמַּרְצָה דִּבְּרָה עַל הָמָן, שֶׁ׳כָּל זֶה אֵינֶנּוּ שֹׁוֶה לוֹ׳.

פִּתְאוֹם הִיא הֵבִינָה: ׳אֲנִי עוֹשָׂה קְצָת כְּמוֹ הָמָן! חֲסֵרִים לִי חֲמֵשׁ מֵאוֹת דּוֹלָר, אֲבָל בִּגְלַל זֶה לְהִתְעַצֵּב וְלִשְׁכֹּחַ אֶת כָּל הַטּוֹב שֶׁיֵּשׁ לִי?׳

אֶפְשָׁר לֹא לִשְׁכֹּחַ אֶת הַיֵּשׁ בִּגְלַל אַיִן.

📖 שִׂמְחָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַבִּי חַיִּים מִוֹּלוֹזִ׳ין

🗣️ רַבִּי חַיִּים מִוֹּלוֹזִ׳ין

רַבִּי חַיִּים מִוֹּלוֹזִ׳ין לִמֵּד: אָסוּר לְהוֹצִיא מֵהַפֶּה מִשְׁפָּט שֶׁמַּתְחִיל בַּמִּלִּים ׳אֵין לִי׳. מִי שֶׁמַּתְחִיל מִשְׁפָּט בְּ׳אֵין׳ – כְּאִלּוּ קָבַע גּוֹרָל.

לָכֵן תָּמִיד אוֹמְרִים ׳יֵשׁ׳. וְאִם כְּבָר יָצָא ׳אֵין׳ – מְסַיְּמִים מִיָּד: ׳אֵין עוֹד מִלְּבַדּוֹ׳.

מַתְחִילִים מִשְׁפָּט בְּ׳יֵשׁ׳, לֹא בְּ׳אֵין׳.

🥛 שִׁירַת הַכּוֹס הַמְּלֵאָה

יֵשׁ שִׁיר שֶׁמַּחֲלִיף כָּל תְּלוּנָה בְּצַד הַטּוֹב שֶׁלָּהּ. ׳אֲנִי אֲסִיר תּוֹדָה לַבְּלָגָן אַחֲרֵי הַמְּסִבָּה – כִּי זֶה אוֹמֵר שֶׁהָיִיתִי מֻקָּף חֲבֵרִים. לַבְּגָדִים שֶׁקְּצָת צְפוּפִים – כִּי יֵשׁ לִי מַסְפִּיק לֶאֱכֹל. לַדֶּשֶׁא שֶׁצָּרִיךְ לְכַסֵּחַ – כִּי יֵשׁ לִי בַּיִת׳.

כָּךְ כָּל תְּלוּנָה הוֹפֶכֶת לְ׳יֵשׁ׳.

כָּל תְּלוּנָה יֵשׁ לָהּ צַד שֶׁל ׳יֵשׁ׳.

⚖️ מֵאֲזָן שֶׁל יֵשׁ

הַמּוֹרָה אָמְרָה: כָּל אֶחָד, כְּשֶׁיַּעֲרֹךְ מֵאֲזָן, יִרְאֶה שֶׁהַיֵּשׁ רַב מְאֹד, אֲפִלּוּ בִּתְנָאִים קָשִׁים. כָּךְ נִכְנָסִים לְ׳אֵיזֶהוּ עָשִׁיר – הַשָּׂמֵחַ בְּחֶלְקוֹ׳.

עוֹרְכִים מֵאֲזָן: בְּצַד אֶחָד כּוֹתְבִים מַה שֶׁיֵּשׁ, וּבְצַד שֵׁנִי מַה שֶׁחָסֵר. וְרוֹאִים אֵיזֶה צַד כָּבֵד.

עוֹרְכִים מֵאֲזָן: יֵשׁ מוּל אַיִן.

מָה לוֹקְחִים מִרָמָה א׳?

כְּשֶׁאֲנִי מִסְתַּכֵּל עַל הַ׳יֵּשׁ׳, אֲנִי רוֹאֶה כַּמָּה ה׳ נָתַן לִי. וְכָךְ הַלֵּב שָׂמֵחַ וְאוֹהֵב אוֹתוֹ – בִּמְקוֹם לִרְאוֹת רַק אֶת הָאַיִן.

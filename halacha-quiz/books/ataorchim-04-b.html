✨ הַסְּטֵייפְלֶר מִתְפַּלֵּל · רָמָה ב׳

כְּשֶׁמָּרָן הַסְּטֵייפְלֶר מִתְפַּלֵּל עָלֵינוּ

רָמָה ב׳ – הַשּׁוֹחֲטִים וְהַתְּפִלָּה

הַשּׁוֹחֲטִים שֶׁל כְּפַר אָתָא, פֶּרֶק תְּהִלִּים אֶחָד וְהַמִּשְׂרָד שֶׁלֹּא הִסְכִּים: ׳כְּשֶׁהוּא מִתְפַּלֵּל – אֲנַחְנוּ מַצְלִיחִים׳

📖 תּוֹלְדוֹת קִרְיַת אָתָא · מֵאַרְכִיּוֹן הַקּוֹל הַחֲמִישִׁי

🎯 לַחִידוֹן

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִשְׁמֹר עַל מִנְהֲגֵי הַכַּשְׁרוּת 💛

📖 תּוֹלְדוֹת קִרְיַת אָתָא – מֵאַרְכִיּוֹן הַקּוֹל הַחֲמִישִׁי, תְּפִלּוֹתָיו שֶׁל הַסְּטֵייפְלֶר – צַדִּיקִים אוֹרְחִים בְּכְפַר אָתָא

רָמָה ב׳ – הַשּׁוֹחֲטִים וְהַתְּפִלָּה

🐔 שְׁנֵי שׁוֹחֲטִים כַּמִּנְהָג

בְּכְפַר אָתָא הִקְפִּידוּ שֶׁשְׁנֵי שׁוֹחֲטִים יַעַמְדוּ אֵצֶל הַשְּׁחִיטָה, כַּנָּהוּג בַּקְּהִלּוֹת יִשְׂרָאֵל הַכְּשֵׁרוֹת מִדּוֹר לְדוֹר.

לְמִשְׂרַד הַדָּתוֹת לֹא הָיָה ׳תֶּקֶן׳ לָזֶה, וְהַכְּפָר שִׁלֵּם אֶת הַשָּׂכָר מִכִּיסוֹ.

שְׁנֵי שׁוֹחֲטִים – כִּי כָּךְ נוֹהֲגִים בַּקְּהִלּוֹת הַכְּשֵׁרוֹת.

1

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד לְפְנוֹת לְתְפִלָּה גַּם כְּשֶׁיֵּשׁ בְּעָיָה גְּדוֹלָה 💛

📖 תּוֹלְדוֹת קִרְיַת אָתָא – מֵאַרְכִיּוֹן הַקּוֹל הַחֲמִישִׁי, תְּפִלּוֹתָיו שֶׁל הַסְּטֵייפְלֶר – צַדִּיקִים אוֹרְחִים בְּכְפַר אָתָא

רָמָה ב׳ – הַשּׁוֹחֲטִים וְהַתְּפִלָּה

☎️ הַמְּצוּקָה וְהַטֶּלֶפוֹן

פָּקִיד בַּמִּשְׂרָד הִקְשָׁה עַל הַכְּפָר, וְהָיְתָה פְּגִישָׁה מַכְרִיעָה. הָרַב שְׁמוּאֵל שׁוֹלֶזִינְגֶר הִתְקַשֵּׁר לְאָחִיו: ׳לֵךְ מִיָּד לַסְּטֵייפְלֶר שֶׁיִּתְפַּלֵּל. כְּשֶׁהוּא מִתְפַּלֵּל – נַצְלִיחַ בְּעֶזְרַת ה׳׳.

׳הוּא הָאָב שֶׁלָּנוּ, וּכְשֶׁהוּא מִתְפַּלֵּל עָלֵינוּ, ה׳ שׁוֹמֵעַ לוֹ. כְּבָר הַרְבֵּה פְּעָמִים הָיָה כָּךְ׳.

הָרַב שְׁלָחוֹ בְּדֶרֶךְ שֶׁל תְּפִלָּה.

2

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד שֶׁלְּתְפִלָּה אֵין זְמַן לְחַכּוֹת 💛

📖 תּוֹלְדוֹת קִרְיַת אָתָא – מֵאַרְכִיּוֹן הַקּוֹל הַחֲמִישִׁי, תְּפִלּוֹתָיו שֶׁל הַסְּטֵייפְלֶר – צַדִּיקִים אוֹרְחִים בְּכְפַר אָתָא

רָמָה ב׳ – הַשּׁוֹחֲטִים וְהַתְּפִלָּה

🏃 מְהִירוּת וּלְבָבוֹת

הָרַב מֹשֶׁה מָרְדֳּכַי רָץ אֶל הַסְּטֵייפְלֶר, וְתוֹךְ שָׁלוֹשׁ דַּקּוֹת הָיָה בַּחֶדֶר. הַסְּטֵייפְלֶר קָרָא אֶת הַסִּפּוּר וְאָמַר: ׳רִבּוֹנוֹ שֶׁל עוֹלָם חָזָק מִכֻּלָּם. לָנוּ עֵצָה אַחַת – תְּפִלָּה׳.

הוּא לֹא חִכָּה לָעֶרֶב, אֶלָּא הִתְחִיל לוֹמַר פֶּרֶק תְּהִלִּים וְעוֹד פֶּרֶק, בְּקוֹל רָם וּבְהִתְלַהֲבוּת.

הַתְּפִלָּה הִתְחִילָה מִיָּד, בְּקוֹל רָם.

3

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְהִתְפַּלֵּל וְלִבְטֹחַ בְּה׳ 💛

📖 תּוֹלְדוֹת קִרְיַת אָתָא – מֵאַרְכִיּוֹן הַקּוֹל הַחֲמִישִׁי, פְּנִינֵי רַבֵּנוּ הַקְּהִלּוֹת יַעֲקֹב – הָרַב מ״מ שׁוֹלֶזִינְגֶר

רָמָה ב׳ – הַשּׁוֹחֲטִים וְהַתְּפִלָּה

😊 ׳הַחֲרֵדִים יַצְלִיחוּ׳

אַחֲרֵי הַתְּפִלָּה אָמַר הַסְּטֵייפְלֶר: ׳ה׳ יַעֲזֹר, יִהְיֶה טוֹב, הַחֲרֵדִים יַצְלִיחוּ׳.

אֶת הַדְּבָרִים הָאֵלֶּה כָּתַב הָרַב מֹשֶׁה מָרְדֳּכַי בְּסִפְרוֹ.

הַתְּפִלָּה הֵבִיאָה בִּטָּחוֹן שֶׁיִּהְיֶה טוֹב.

4

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִזְכֹּר מִי שֶׁהִתְפַּלֵּל עָלַי 💛

📖 תּוֹלְדוֹת קִרְיַת אָתָא – מֵאַרְכִיּוֹן הַקּוֹל הַחֲמִישִׁי, פְּנִינֵי רַבֵּנוּ הַקְּהִלּוֹת יַעֲקֹב – הָרַב מ״מ שׁוֹלֶזִינְגֶר

רָמָה ב׳ – הַשּׁוֹחֲטִים וְהַתְּפִלָּה

🕯️ זִכָּרוֹן לִשְׁנֵי אַחִים

אַחֲרֵי פְּטִירַת הַסְּטֵייפְלֶר, בִּשְׁנַת תשמ״ה, אַבְרֵכִים וּבְנֵי יְשִׁיבוֹת שֶׁל קִרְיַת אָתָא קָרְאוּ לַעֲצֶרֶת הִתְעוֹרְרוּת.

הַמּוֹדָעָה הַקְּטַנָּה גָּרְמָה לָרַב מֹשֶׁה מָרְדֳּכַי לְהִתְרַגֵּשׁ, כִּי הוּא זָכַר אֶת הַתְּפִלָּה שֶׁהִתְפַּלֵּל הַסְּטֵייפְלֶר עֲבוּר הָעִיר.

הָעִיר זָכְרָה אֶת הַצַּדִּיק שֶׁהִתְפַּלֵּל עָלֶיהָ.

מָה לוֹקְחִים מִרָמָה ב׳?

שְׁנֵי שׁוֹחֲטִים, פֶּרֶק תְּהִלִּים וְיָד ה׳. כָּךְ ה׳ מַנְהִיג – וְזוֹ אַהֲבָה.

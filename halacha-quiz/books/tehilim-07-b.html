רָמָה ב׳ – בִּטָּחוֹן עִם מְפָרְשִׁים

🗣️ אֹמַר לַה׳

בַּפָּסוּק ׳אֹמַר לַה׳ מַחְסִי׳ הַמִּלָּה ׳אֹמַר׳ חֲשׁוּבָה. הַסְּפוֹרְנוֹ מְבָאֵר: ׳בְּהִתְפַּלֵּל אֵלָיו׳ – כְּלוֹמַר, מִתְפַּלְּלִים וְאוֹמְרִים בַּפֶּה.

וְהַפָּסוּק מוֹסִיף ׳אֶבְטַח בּוֹ׳, וְהָרוֹקֵחַ מְבָאֵר: כָּל יָמַי. וְרַשִׁ״י: ׳וְהַכֹּל יִלְמְדוּ מִמֶּנִּי׳ – מִי שֶׁבּוֹטֵחַ נֶעֱשֶׂה דֻּגְמָה לַאֲחֵרִים.

בִּטָּחוֹן אוֹמְרִים בַּפֶּה, וְהוּא נֶעֱשֶׂה דֻּגְמָה לַאֲחֵרִים.

1

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְנַסֶּה לִבְטֹחַ בַּה׳ וְרוֹאֶה שֶׁזֶּה עוֹזֵר 💛

📖 מִזְמוֹרֵי תְּהִלִּים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים צ״א ג׳ · רַדַ״ק, אִבְן עֶזְרָא

רָמָה ב׳ – בִּטָּחוֹן עִם מְפָרְשִׁים

🔬 מִנִּסָּיוֹן

׳כִּי הוּא יַצִּילְךָ מִפַּח יָקוּשׁ׳. הָרַדַ״ק מְבָאֵר: ׳כְּמוֹ שֶׁהִצִּילַנִי, שֶׁבָּטַחְתִּי בּוֹ׳. וְאִבְן עֶזְרָא: ׳אֲנִי נִסִּיתִי זֶה, וַאֲנִי אֶהְיֶה לְךָ עָרֵב׳.

כְּלוֹמַר, מִי שֶׁאוֹמֵר אֶת הַדְּבָרִים לֹא מְדַבֵּר רַק בִּתְאוֹרְיָה. הוּא בָּדַק וְרָאָה שֶׁהַבִּטָּחוֹן בַּה׳ עוֹזֵר, וְלָכֵן הוּא מַבְטִיחַ גַּם לָנוּ.

הַבִּטָּחוֹן הַזֶּה לֹא תֵּאוֹרְיָה – מִי שֶׁאוֹמֵר אוֹתוֹ בָּדַק אוֹתוֹ בְּעַצְמוֹ.

2

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַרְגִּישׁ שֶׁהַהַבְטָחָה שֶׁל ה׳ שׁוֹמֶרֶת עָלַי מִכָּל צַד 💛

📖 מִזְמוֹרֵי תְּהִלִּים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים צ״א ד׳ · רַדַ״ק, הַמֵּאִירִי, רַשִׁ״י

רָמָה ב׳ – בִּטָּחוֹן עִם מְפָרְשִׁים

🛡️ מָגֵן וּמַקִּיף

׳צִנָּה וְסֹחֵרָה אֲמִתּוֹ׳. הָרַדַ״ק: הַהַבְטָחָה שֶׁל ה׳ ׳תִּהְיֶה לְךָ כְּצִנָּה וְסֹחֵרָה׳ – כְּלֵי מָגֵן. הַמֵּאִירִי מוֹסִיף: אֵלּוּ כֵּלִים שֶׁהָאָדָם מֵגֵן בָּהֶם עַל עַצְמוֹ מִן הַחֲרָבוֹת וְהַחִצִּים.

הַהֶבְדֵּל: ׳צִנָּה׳ הִיא הַמָּגֵן, וּ׳סֹחֵרָה׳ – רַשִׁ״י אוֹמֵר – מַקִּיפָה אֶת הָאָדָם סָחוֹר סָחוֹר, מִכָּל צַד.

הַהַבְטָחָה שֶׁל ה׳ מְגִנָּה עָלֵינוּ מִכָּל הַצְּדָדִים.

3

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְבַקֵּשׁ מֵה׳ שֶׁהַדְּאָגָה לֹא תְּנַהֵל אוֹתִי 💛

📖 מִזְמוֹרֵי תְּהִלִּים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים צ״א י׳ · סַנְהֶדְרִין ק״ג א׳, רַשִׁ״י כְּתֻבּוֹת י׳ ב׳

רָמָה ב׳ – בִּטָּחוֹן עִם מְפָרְשִׁים

🌥️ הַדְּאָגוֹת

׳לֹא תְאֻנֶּה אֵלֶיךָ רָעָה׳. הַגְּמָרָא בְּסַנְהֶדְרִין קַ״ג מְפָרֶשֶׁת: שֶׁלֹּא יְבַעֲתוּךָ הִרְהוּרִים רָעִים. וְרַשִׁ״י (כְּתֻבּוֹת י׳): ׳מַחֲשָׁבָה רָעָה – דְּאָגָה׳.

וְעוֹד נֶאֱמַר שָׁם עַל ׳לֹא יִקְרַב בְּאָהֳלֶךָ׳: שֶׁלֹּא יִהְיֶה לְךָ בֵּן אוֹ תַּלְמִיד שֶׁמַּקְדִּיחַ אֶת תַּבְשִׁילוֹ בָּרַבִּים, כְּלוֹמַר שֶׁלֹּא תִּתְבַּיֵּשׁ.

הַמִּזְמוֹר לֹא מַבְטִיחַ חַיִּים בְּלִי קֹשִׁי, אֶלָּא שֶׁהַדְּאָגָה לֹא תְּשַׁלֵּט עָלֵינוּ.

הַמִּזְמוֹר עוֹזֵר לֹא לְהִתְמַלֵּא דְּאָגָה וּבוּשָׁה.

4

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁקָּשֶׁה לִי, אֲנִי קוֹרֵא לַה׳ וּמַרְגִּישׁ שֶׁהוּא אִתִּי 💛

📖 מִזְמוֹרֵי תְּהִלִּים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים צ״א ט״ו · שְׁמוֹת רַבָּה ב׳ ה׳, מַלְבִּ״ם

רָמָה ב׳ – בִּטָּחוֹן עִם מְפָרְשִׁים

👬 עִמּוֹ אָנֹכִי בְצָרָה

׳יִקְרָאֵנִי וְאֶעֱנֵהוּ, עִמּוֹ אָנֹכִי בְצָרָה׳. רַבִּי יַנַּאי אָמַר בַּמִּדְרָשׁ: ׳מָה הַתְּאוֹמִים הָאֵלֶּה – אִם חָשׁ אֶחָד בְּרֹאשׁוֹ, חֲבֵרוֹ מַרְגִּישׁ; כָּךְ אָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא, כִּבְיָכוֹל, ׳עִמּוֹ אָנֹכִי בְצָרָה׳׳.

הַמַּלְבִּ״ם שׁוֹאֵל: לָמָּה ה׳ מַנִּיחַ אוֹתָנוּ לִפְעָמִים בְּמַצָּב קָשֶׁה? כְּדֵי שֶׁנִּקְרָא אֵלָיו וְנַרְגִּישׁ בְּחוּשׁ שֶׁהוּא אִתָּנוּ. אִם הַכֹּל הָיָה קַל, הָיִינוּ חוֹשְׁבִים שֶׁזֶּה סְתָם טֶבַע.

ה׳ אִתִּי בַּצָּרָה, וְהַקֹּשִׁי נוֹתֵן לִי לְהַרְגִּישׁ אוֹתוֹ.

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי נִצּוֹל, אֲנִי יוֹדֵעַ שֶׁה׳ שָׂמֵחַ אִתִּי 💛

📖 מִזְמוֹרֵי תְּהִלִּים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים צ״א ט״ז · תַּנְחוּמָא אַחֲרֵי מוֹת י״ב

🌅 יְשׁוּעָתִי

׳וְאַרְאֵהוּ בִּישׁוּעָתִי׳. רַבִּי אַבָּהוּ אָמַר בַּמִּדְרָשׁ תַּנְחוּמָא: ׳כָּל יְשׁוּעָה שֶׁבָּאָה לְיִשְׂרָאֵל – הִיא שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא׳. הַהוֹכָחָה: בַּפָּסוּק כָּתוּב ׳יְשׁוּעָתִי׳ וְלֹא ׳יְשׁוּעַתְכֶם׳, וְכֵן ׳כִּי קְרוֹבָה יְשׁוּעָתִי לָבוֹא׳ בִּישַׁעְיָהוּ.

כְּלוֹמַר, כְּשֶׁיִּשְׂרָאֵל נוֹשָׁעִים, ה׳ עַצְמוֹ שָׂמֵחַ. הַמִּזְמוֹר נִגְמָר בִּשְׂמְחָה מְשֻׁתֶּפֶת.

הַיְשׁוּעָה שֶׁלָּנוּ הִיא גַּם הַיְשׁוּעָה שֶׁל ה׳.

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְפָרְשִׁים מְלַמְּדִים שֶׁהַבִּטָּחוֹן נוֹלָד מִנִּסָּיוֹן, וְשֶׁה׳ אִתָּנוּ גַּם בַּצָּרָה. מִי שֶׁקּוֹרֵא לוֹ מַרְגִּישׁ אֶת הָאַהֲבָה שֶׁלּוֹ, וּבַיְשׁוּעָה גַּם הוּא שָׂמֵחַ.

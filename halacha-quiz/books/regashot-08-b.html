🎛️ וְלִבּוֹ מָסוּר בְּיָדוֹ

בְּסֵפֶר הַחִנּוּךְ נִשְׁאֶלֶת שְׁאֵלָה חֲשׁוּבָה: אֵיךְ הַתּוֹרָה מְצַוָּה ׳לֹא תַחְמֹד׳ עַל רֶגֶשׁ שֶׁאָדָם לֹא יָכוֹל לַעֲמֹד עָלָיו? הוּא עוֹנֶה: זֶה לֹא נָכוֹן. בְּיַד הָאָדָם לִמְנֹעַ אֶת מַחְשְׁבוֹתָיו וְתַאֲוֹתָיו, וְלֹא רַק אֶת מַעֲשָׂיו.

הוּא כּוֹתֵב: ׳וְלִבּוֹ מָסוּר בְּיָדוֹ, עַל כָּל אֲשֶׁר יַחְפֹּץ יַטֶּנּוּ׳. זוֹ בְּשׂוֹרַת חֵרוּת גְּדוֹלָה: עוֹלָמוֹ הַפְּנִימִי שֶׁל הָאָדָם הוּא בְּיָדָיו, וְאֶפְשָׁר לְהָסִיר אֶת הַחֶמְדָּה מִן הַלֵּב.

בְּדִבְּרוֹת שֶׁבְּסֵפֶר דְּבָרִים כָּתוּב גַּם ׳וְלֹא תִתְאַוֶּה׳, וְלֹא רַק ׳לֹא תַחְמֹד׳.

בְּיַד הָאָדָם לִמְנֹעַ אֶת מַחְשְׁבוֹתָיו וְתַאֲוֹתָיו; לָכֵן הַתּוֹרָה מְצַוָּה גַּם עַל הַלֵּב.

👀 עַיִן רוֹאָה וְלֵב חוֹמֵד – מָה גִּלּוּ?

בְּ׳נִסָּיוֹן הַמַּרְשְׁמַלּוֹ׳ יְלָדִים קְטַנִּים הָיוּ צְרִיכִים לִדְחוֹת אֶת הָרָצוֹן לֶאֱכֹל. כְּשֶׁרָאוּ אֶת הַפְּרָס, הָיָה לָהֶם הַרְבֵּה יוֹתֵר קָשֶׁה לְחַכּוֹת. כְּשֶׁהַפְּרָס הֻסְתַּר – חִכּוּ זְמַן אָרֹךְ כִּמְעַט פִּי עֶשֶׂר.

חוֹקְרִים מָצְאוּ שֶׁמִּי שֶׁיּוֹדֵעַ לִדְחוֹת סִפּוּק, קַל לוֹ יוֹתֵר לְהַצִּיב מַטָּרוֹת וּלְהַגִּיעַ אֲלֵיהֶן.

חֲכָמֵינוּ אָמְרוּ: ׳עַיִן רוֹאָה וְלֵב חוֹמֵד׳. וְגַם: ׳אֵין לְךָ בֶּן חוֹרִין אֶלָּא מִי שֶׁעוֹסֵק בְּתַלְמוּד תּוֹרָה׳ – בֶּן חוֹרִין הוּא מִי שֶׁשּׁוֹלֵט בִּרְצוֹנוֹתָיו.

כְּשֶׁהַפְּרָס מֻסְתָּר, אֶפְשָׁר לְחַכּוֹת כִּמְעַט פִּי עֶשֶׂר יוֹתֵר – ׳עַיִן רוֹאָה וְלֵב חוֹמֵד׳.

📖 הָרְגָשׁוֹת שֶׁלִּי בַּתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁמוֹת כ׳ י״ד · הָרַב יַעֲקֹב גָּלִינְסְקִי

רָמָה ב׳ – מָה לָמְדוּ הַחֲכָמִים וְהַחוֹקְרִים

🧳 וְכֹל אֲשֶׁר לְרֵעֶךָ

הָרַב יַעֲקֹב גָּלִינְסְקִי שָׁאַל עַל הַדִּבְּרָה הָעֲשִׂירִית: ׳לֹא תַחְמֹד בֵּית רֵעֶךָ … וְשׁוֹרוֹ וַחֲמֹרוֹ וְכֹל אֲשֶׁר לְרֵעֶךָ׳ – לָמָּה לְפָרֵט כָּל כָּךְ הַרְבֵּה, אִם בַּסּוֹף כָּתוּב ׳וְכֹל אֲשֶׁר לְרֵעֶךָ׳?

מִי שֶׁחוֹמֵד אֶת הַדָּבָר שֶׁל חֲבֵרוֹ חוֹשֵׁב: ׳זֶה מַתְאִים לִי, מַגִּיעַ לִי כְּמוֹ שֶׁיֵּשׁ לוֹ׳. אוֹמֵר לוֹ הַפָּסוּק: אַתָּה רוֹצֶה אֶת שֶׁלּוֹ? קַח אֶת הַכֹּל! לַחֲבֵרְךָ יֵשׁ גַּם דְּבָרִים שֶׁאֵינָם כָּל כָּךְ טוֹבִים וְלֹא הָיִיתָ רוֹצֶה בָּהֶם.

הִתְבּוֹנְנוּת שְׁלֵמָה – לֹא רַק בַּטּוֹב שֶׁל הַזּוּלַת, אֶלָּא בְּכָל מָה שֶׁיֵּשׁ לוֹ – מוֹנַעַת חֶמְדָּה.

מִי שֶׁמִּתְבּוֹנֵן בְּכָל מָה שֶׁיֵּשׁ לַחֲבֵרוֹ, וְלֹא רַק בַּחֵלֶק הַיָּפֶה, פָּחוֹת חוֹמֵד.

📖 הָרְגָשׁוֹת שֶׁלִּי בַּתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים ל״ד י״ג–י״ד · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🌿 הֶחָפֵץ חַיִּים

׳מִי הָאִישׁ הֶחָפֵץ חַיִּים אֹהֵב יָמִים לִרְאוֹת טוֹב? נְצֹר לְשׁוֹנְךָ מֵרָע׳. לָמָּה מִי שֶׁשּׁוֹמֵר אֶת לְשׁוֹנוֹ נִקְרָא דַּוְקָא ׳חָפֵץ חַיִּים׳?

שָׁמַעְנוּ פֵּרוּשׁ: אָדָם מְקַנֵּא בַּחֲבֵרוֹ וּמְדַבֵּר עָלָיו רַע כִּי הוּא לֹא מִתְבּוֹנֵן בַּחֲסָדִים שֶׁיֵּשׁ לוֹ עַצְמוֹ. הוּא חוֹשֵׁב שֶׁהַדְּבָרִים הַטּוֹבִים שֶׁל חֲבֵרוֹ הָיוּ צְרִיכִים לִהְיוֹת שֶׁלּוֹ.

אֲבָל מִי שֶׁהוּא ׳חָפֵץ חַיִּים׳ מִתְבּוֹנֵן בְּחַיָּיו וּבַדְּבָרִים הַטּוֹבִים שֶׁיֵּשׁ לוֹ, וְלֹא מַגִּיעַ לְקַנֵּא וְלֹא לְדַבֵּר עַל הַשֵּׁנִי.

מִי שֶׁמִּתְבּוֹנֵן בַּטּוֹב שֶׁבְּחַיָּיו – לֹא מְקַנֵּא וְלֹא מְדַבֵּר עַל חֲבֵרוֹ.

📖 הָרְגָשׁוֹת שֶׁלִּי בַּתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים ק׳ א׳ · הָרַב חַיִּים קַנְיֶבְסְקִי · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🕯️ קָרְבַּן תּוֹדָה וְהֶחָמֵץ

מַפְתִּיעַ: הַהוֹדָיָה קְשׁוּרָה גַּם לְקִנְאָה. בְּכָל הַקָּרְבָּנוֹת אֵין חָמֵץ, חוּץ מִקָּרְבַּן תּוֹדָה (וְשְׁתֵּי הַלֶּחֶם בְּשָׁבוּעוֹת). בְּסִפְרֵי חֲסִידוּת חָמֵץ מְרַמֵּז עַל כֹּחוֹת הַגּוּף וְהַתְּכוּנוֹת הָרָעוֹת, וּמַצָּה – עַל פְּשִׁיטוּת וַעֲנָוָה. כְּשֶׁאָדָם בָּא לְהוֹדוֹת, הוּא מְתַקֵּן וּמְעַדֵּן אֶת הָרְצוֹנוֹת שֶׁלּוֹ.

בַּמִּזְמוֹר ׳מִזְמוֹר לְתוֹדָה הָרִיעוּ לַה׳ כָּל הָאָרֶץ׳ שׁוֹאֲלִים: לָמָּה ׳כָּל הָאָרֶץ׳? הָרַב חַיִּים קַנְיֶבְסְקִי הִסְבִּיר: כְּשֶׁרוֹאִים שֶׁלַּזּוּלַת קָרָה דָּבָר טוֹב, מִתְבּוֹנְנִים וְרוֹאִים אֶת הַחֲסָדִים שֶׁה׳ עוֹשֶׂה גַּם אִתִּי כָּל יוֹם, וּמוֹדִים.

הוֹדָיָה מְעַדֶּנֶת אֶת הָרְצוֹנוֹת, וּכְשֶׁרוֹאִים טוֹב אֵצֶל אַחֵר – מוֹדִים עַל הַטּוֹב שֶׁלָּנוּ.

💛 וְרָאֲךָ וְשָׂמַח בְּלִבּוֹ

מֹשֶׁה לֹא רָצָה לְקַבֵּל אֶת הַשְּׁלִיחוּת וְאָמַר: ׳שְׁלַח נָא בְּיַד תִּשְׁלָח׳. הוּא דָּאַג מָה יַרְגִּישׁ אַהֲרֹן אָחִיו הַגָּדוֹל, שֶׁהִתְנַבֵּא לְיִשְׂרָאֵל בְּמִצְרַיִם, כְּשֶׁיִּרְאֶה אוֹתוֹ עוֹלֶה לִגְדֻלָּה.

ה׳ אָמַר לוֹ: ׳הִנֵּה הוּא יֹצֵא לִקְרָאתֶךָ וְרָאֲךָ וְשָׂמַח בְּלִבּוֹ׳. רַשִׁ״י מְפָרֵשׁ: לֹא כְּמוֹ שֶׁאַתָּה סָבוּר, שֶׁיַּקְפִּיד עָלֶיךָ. וּמִשָּׁם זָכָה אַהֲרֹן לַחֹשֶׁן הַנָּתוּן עַל הַלֵּב.

אַהֲרֹן מוֹכִיחַ שֶׁאֶפְשָׁר: לֵב שֶׁאֵינוֹ מְקַנֵּא אֶלָּא שָׂמֵחַ בֶּאֱמֶת.

מָה לוֹקְחִים מִרָמָה ב׳?

כְּשֶׁאֲנִי שׁוֹלֵט בַּלֵּב שֶׁלִּי וּמוֹדֶה עַל מָה שֶׁנָּתַן לִי ה׳, אֲנִי מַרְגִּישׁ שֶׁהוּא אוֹהֵב אוֹתִי וְנוֹתֵן לִי בְּדִיּוּק מָה שֶׁצָּרִיךְ. כָּךְ אַהֲבָתִי אֵלָיו גְּדֵלָה.

7
❤️
לֹא תַחְמֹד בֵּית רֵעֶךָ

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי הָיָה לַחֲבֵרִי דָּבָר חָדָשׁ וְהַלֵּב אָמַר ׳גַּם אֲנִי רוֹצֶה׳, וּמָה עָשִׂיתִי?
  • מָה אֲנִי יָכוֹל לְהַסְתִּיר מֵהָעַיִן, אוֹ לְהַסִּיט מִמֶּנּוּ אֶת הַמַּבָּט, כְּדֵי שֶׁהַלֵּב לֹא יַחְמֹד?
  • הַיּוֹם כְּשֶׁאֶרְאֶה מַשֶּׁהוּ שֶׁיֵּשׁ לַחֲבֵרִי וְאֵין לִי, אַסִּיט אֶת הָעֵינַיִם וְאוֹמַר תּוֹדָה עַל דָּבָר אֶחָד שֶׁיֵּשׁ לִי.

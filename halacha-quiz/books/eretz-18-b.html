חֲכָמִים מְלַמְּדִים שֶׁיֵּשׁ שְׁלֹשָׁה דְבָרִים שֶׁצָּרִיךְ לְחַבֵּב כְּאִלּוּ נִתְּנוּ הַיּוֹם: תּוֹרַת יִשְׂרָאֵל (׳הַיּוֹם הַזֶּה ה׳ אֱלֹהֶיךָ מְצַוְּךָ׳ – רַשִׁ״י: בְּכָל יוֹם יִהְיוּ בְּעֵינֶיךָ כַּחֲדָשִׁים), בְּרִית עַם יִשְׂרָאֵל (׳הַיּוֹם הַזֶּה נִהְיֵיתָ לְעָם׳), וְאֶרֶץ יִשְׂרָאֵל (׳וּנְתָנָהּ לָךְ׳).

כָּךְ יוֹצֵא שֶׁהָאָרֶץ חֲשׁוּבָה כְּמוֹ הַתּוֹרָה וְהָעָם: אֵין מַרְגִּישִׁים בָּהּ הֶרְגֵּל, אֶלָּא הִתְחַדְּשׁוּת.

הַתּוֹרָה, הָעָם וְהָאָרֶץ – שְׁלָשְׁתָּם ׳הַיּוֹם׳, תָּמִיד חֲדָשִׁים.

🧺 לָמָּה בִּכּוּרִים?

מִצְוַת הַבִּכּוּרִים מְיֻחֶדֶת: הַפֵּרוֹת אֵינָם רַק אֹכֶל לַכֹּהֲנִים, אֶלָּא דֶּרֶךְ לְהוֹדוֹת לַה׳ עַל מַתְּנַת הָאָרֶץ. לָכֵן מֵבִיאִים אוֹתָם בְּטֶנֶא, וְאוֹמְרִים אֶת מִקְרָא הַבִּכּוּרִים.

מְבִיאִים רַק מִי שֶׁהַפֵּרוֹת גָּדְלוּ בְּחֶלְקוֹ שֶׁלּוֹ בָּאָרֶץ, וְרַק מִשִּׁבְעַת הַמִּינִים, שֶׁהֵם תַּמְצִית שִׁבְחָהּ שֶׁל הָאָרֶץ.

כָּךְ הַמִּצְוָה מַזְכִּירָה: ׳הַיּוֹם׳ ה׳ מַמְשִׁיךְ לָתֵת לָנוּ אֶת הָאָרֶץ.

בִּכּוּרִים הֵם הוֹדָיָה לַה׳ עַל מַתְּנַת הָאָרֶץ, שֶׁנִּמְשֶׁכֶת עַד הַיּוֹם.

❓ הַשְּׁאֵלָה שֶׁל הַפּוֹנִיבֶז׳

הָרַב מִפּוֹנִיבֶז׳ שָׁאַל: ׳וְכִי רַק הָעוֹלִים הַחֲדָשִׁים נִצְטַוּוּ לְהָבִיא בִּכּוּרִים, וְכָל מִי שֶׁנּוֹלַד בָּאָרֶץ אֵינוֹ בְּמִצְוָה זוֹ?׳

וְהַתְּשׁוּבָה: הַכַּוָּנָה הִיא שֶׁכָּל אֶחָד מִוָּתִיקֵי הָאָרֶץ יֹאהַב אוֹתָהּ אַהֲבַת נְעוּרִים נִצְחִית, כְּאִלּוּ רַק הַיּוֹם בָּא אֵלֶיהָ. וְכֵן כָּתַב רַשִׁ״י: ׳תְּהֵא בְּעֵינֶיךָ כְּאִלּוּ נִתְּנָה לְךָ בּוֹ בַּיּוֹם׳.

מִי שֶׁגָּדַל בָּאָרֶץ אֵינוֹ מְקַבֵּל אֶת הָאַהֲבָה בְּחִנָּם: הוּא צָרִיךְ לְחַדֵּשׁ אוֹתָהּ.

גַּם מִי שֶׁנּוֹלַד בָּאָרֶץ צָרִיךְ לְחַדֵּשׁ אֶת הָאַהֲבָה אֵלֶיהָ.

🎉 בְּכָל יוֹם כַּחֲדָשִׁים

הָרַב ׳נְתִיבוֹת שָׁלוֹם׳ מִסְּלוֹנִים הֵסִּיק מִפָּסוּק הַבִּכּוּרִים עִקָּרוֹן לַחַיִּים: ׳הִגַּדְתִּי הַיּוֹם׳ מְלַמֵּד שֶׁמִּצְוַת יְשִׁיבַת אֶרֶץ יִשְׂרָאֵל צְרִיכָה לִהְיוֹת אֵצֶל יְהוּדִי בִּבְחִינַת ׳בְּכָל יוֹם יִהְיוּ בְּעֵינֶיךָ כַּחֲדָשִׁים׳.

לְכָךְ הוּא עָרַךְ מְסִבָּה אַחֲרֵי חֲמִישִׁים שָׁנָה בָּאָרֶץ, וּבִקֵּשׁ מִכֻּלָּם לְבָרֵךְ וּלְהוֹדוֹת לַה׳ עַל כָּל חֲסָדָיו.

הַהוֹדָיָה הִיא מַה שֶּׁמְּשַׁמֵּר אֶת הַהִתְחַדְּשׁוּת.

הוֹדָיָה לַה׳ עַל הָאָרֶץ שׁוֹמֶרֶת עַל הַהִתְחַדְּשׁוּת שֶׁל הָאַהֲבָה.

✨ בְּכָל רֶגַע אוֹתָהּ הַהֶאָרָה

אַדְמוֹ״ר אֶחָד הִתְלוֹנֵן לִפְנֵי הָאַדְמוֹ״ר מִסָּדִיגוֹרָה, בַּעַל ׳אֲבִיר יַעֲקֹב׳: זֶה הַבִּקּוּר הַשְּׁלִישִׁי שֶׁלּוֹ בָּאָרֶץ, וְהוּא כְּבָר אֵינוֹ מַרְגִּישׁ אֶת הַהֶאָרָה שֶׁהָיְתָה לוֹ בַּבִּקּוּר הָרִאשׁוֹן.

ה׳אֲבִיר יַעֲקֹב׳ הֵשִׁיב: לָנוּ יֵשׁ בְּכָל רֶגַע וָרֶגַע שֶׁל שְׁהוּתֵנוּ בָּאָרֶץ אוֹתָהּ הֶאָרָה וְאוֹתָהּ הִשְׁתּוֹקְקוּת שֶׁהָיְתָה לָנוּ בָּרֶגַע הָרִאשׁוֹן שֶׁרַגְלֵנוּ דָּרְכוּ עַל אַדְמַת הָאָרֶץ.

אֶפְשָׁר לְהַחֲזִיק אֶת הַהֶרְגֵּשׁ הָרִאשׁוֹן – אִם שָׂמִים לֵב.

מִי שֶׁשָּׂם לֵב מַרְגִּישׁ בְּכָל רֶגַע אֶת הַהֶאָרָה שֶׁל הָרֶגַע הָרִאשׁוֹן.

🍂 הַפֵּרוֹת וְהַזְּמַנִּים

בָּאָרֶץ יֵשׁ סֵדֶר קָבוּעַ: הַשְּׂעוֹרָה צוֹמַחַת בְּנִיסָן, וְלָכֵן מְבִיאִים בְּפֶסַח אֶת הָעֹמֶר מֵהַשְּׂעוֹרִים. הַחִטָּה בְּשָׁבוּעוֹת, וְאָז מְבִיאִים אֶת שְׁתֵּי הַלֶּחֶם. הָעֲנָבִים מִתַּמּוּז, הַתְּאֵנִים מֵאָב, וּבְאֱלוּל וּבְתִשְׁרֵי בָּאִים הַתְּמָרִים וְאַחֲרֵיהֶם הַזֵּיתִים.

אַחֲרֵי חֲנֻכָּה אֵין מְבִיאִים בִּכּוּרִים, כִּי הַפֵּרוֹת מִתְיַבְּשִׁים.

כָּךְ הָאָרֶץ מַחְדֶּשֶׁת אֶת עַצְמָהּ כָּל חֹדֶשׁ, וְאֶפְשָׁר לְהוֹדוֹת עַל כָּל חֹדֶשׁ מֵחָדָשׁ.

בְּכָל תְּקוּפָה בָּאָרֶץ צוֹמֵחַ פְּרִי אַחֵר, וְכָל אֶחָד הוּא סִבָּה לְהוֹדָיָה.

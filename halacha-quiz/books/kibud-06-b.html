לְכַבֵּד בְּשִׂמְחָה

רָמָה ב׳ – שְׁנֵי הַהוֹרִים וְהַשִּׂמְחָה

מָה לָמַד הַסְּטַיְיפְלֶר עַל יֶלֶד קָטָן, אֵיךְ מְכַבְּדִים אֶת שְׁנֵי הַהוֹרִים, וְלָמָּה פָּנִים שְׂמֵחוֹת הֵן חֵלֶק מֵהַמִּצְוָה

📖 כִּבּוּד אָב וָאֵם · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🧒

💛 אַהֲבַת הַשֵּׁם: אֲנִי דֻּגְמָה לַקְּטַנִּים, וְרוֹצֶה שֶׁיִּלְמְדוּ מִמֶּנִּי טוֹב 💛

📖 כִּבּוּד אָב וָאֵם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הַסְּטַיְיפְלֶר · הָרַב אָרוּש

רָמָה ב׳ – שְׁנֵי הַהוֹרִים וְהַשִּׂמְחָה

🧒 הַבַּיִת מְחַנֵּךְ

הַסְּטַיְיפְלֶר לִמֵּד שֶׁיֵּשׁ עִנְיָן גָּדוֹל לְלַמֵּד בַּבַּיִת אֶת חֲשִׁיבוּת הַמִּצְוָה וּלְשַׁנֵּן אוֹתָהּ הֵיטֵב, כְּבָר בְּגִיל קָטָן. אֵין לוֹמְדִים אוֹתָהּ רַק מִבַּקָּשׁוֹת, אֶלָּא מִשִּׂיחָה וּמִסִּפּוּרִים.

וְיֵשׁ עוֹד תְּנַאי: מִי שֶׁמְּלַמֵּד צָרִיךְ קֹדֶם לְהַפְנִים כַּמָּה גְּדוֹלָה הַמִּצְוָה. כֵּן גַּם הַגָּדוֹל מֵבִין שֶׁהוּא דֻּגְמָה לַקְּטַנִּים.

מִי שֶׁמַּפְנִים אֶת הַמִּצְוָה, מְלַמֵּד אוֹתָהּ בְּדֻגְמָה.

1
🧩

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹאֶה כָּבוֹד בַּבַּיִת וְלוֹמֵד אוֹתוֹ 💛

📖 כִּבּוּד אָב וָאֵם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרַב מִיכֵל זִילְבֶּר · הָרַב טִיקוֹצִ׳ינְסְקִי

רָמָה ב׳ – שְׁנֵי הַהוֹרִים וְהַשִּׂמְחָה

🧩 שְׁנֵי צְדָדִים

הָרַב מֹשֶׁה טִיקוֹצִ׳ינְסְקִי, מַשְׁגִּיחַ בִּישִׁיבַת סְלוֹבּוֹדְקָה, שָׁמַע עַל הוֹרִים שֶׁמִּתְבַּיְּשִׁים לִתְבֹּעַ אֶת כְּבוֹדָם. הָעֵצָה שֶׁלּוֹ: הָאָב מְחַנֵּךְ לְכַבֵּד אֶת הָאֵם, וְהָאֵם מְחַנֶּכֶת לְכַבֵּד אֶת הָאָב.

כָּךְ לֹא צָרִיךְ אָדָם לְדַבֵּר עַל עַצְמוֹ, וְהַיְלָדִים רוֹאִים שְׁנֵי הוֹרִים שֶׁמְּכַבְּדִים זֶה אֶת זֶה. וְיֶלֶד שֶׁרוֹאֶה בַּבַּיִת כָּבוֹד – לוֹמֵד אוֹתוֹ בְּקַלּוּת.

שְׁנֵי הוֹרִים שֶׁמְּכַבְּדִים זֶה אֶת זֶה, מְחַנְּכִים בְּיַחַד.

2
🪞

💛 אַהֲבַת הַשֵּׁם: אֲנִי נוֹתֵן כָּבוֹד, וְה׳ מְסַדֵּר שֶׁגַּם אֲנִי אֲקַבֵּל 💛

📖 כִּבּוּד אָב וָאֵם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סוֹטָה ח׳ ב׳

רָמָה ב׳ – שְׁנֵי הַהוֹרִים וְהַשִּׂמְחָה

🪞 מִדָּה כְּנֶגֶד מִדָּה

חֲכָמִים לִמְּדוּ: ׳בְּמִדָּה שֶׁאָדָם מוֹדֵד בָּהּ – מוֹדְדִין לוֹ׳. כְּלוֹמַר, מִי שֶׁמְּכַבֵּד אֶת הוֹרָיו – גַּם יְלָדָיו יְכַבְּדוּ אוֹתוֹ, וּמִי שֶׁחַס וְשָׁלוֹם לֹא, גַּם יְלָדָיו יִנְהֲגוּ בּוֹ כָּךְ.

זֶה לֹא עֹנֶשׁ אֶלָּא דֶּרֶךְ הָעוֹלָם: מַה שֶּׁאָדָם נוֹתֵן, הוּא מְקַבֵּל. לָכֵן כְּדַאי לְהַתְחִיל כְּבָר עַכְשָׁו.

מַה שֶּׁנּוֹתְנִים לַהוֹרִים, חוֹזֵר בַּעֲתִיד.

3
😍

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁהַבַּיִת שֶׁלָּנוּ יִהְיֶה מָלֵא אַהֲבָה 💛

📖 כִּבּוּד אָב וָאֵם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – שְׁנֵי הַהוֹרִים וְהַשִּׂמְחָה

😍 כְּשֶׁהַהוֹרִים נִמְסִים

יֵשׁ מִצְוָה שֶׁהִיא כָּל כָּךְ מְצוּיָה, וה׳ קָבַע שֶׁשְּׂכָרָהּ גָּדוֹל מִמַּה שֶּׁאֶפְשָׁר לְתָאֵר. וְאֶפְשָׁר לִרְאוֹת אוֹתוֹ כְּבָר עַכְשָׁו: כְּשֶׁהַיֶּלֶד קָם לִפְנֵי הוֹרָיו, הֵם אוֹהֲבִים אוֹתוֹ מְאֹד.

אָמְרוּ בִּלְשׁוֹן חֲרִיפָה קְצָת: ׳הַהוֹרִים נִמְסִים׳. וְנִרְאָה פַּעַם אָב שֶׁבָּא לְהוֹדוֹת לִבְנוֹ, וְהָיוּ לוֹ דְּמָעוֹת בָּעֵינַיִם.

וּמִי שֶׁגָּר בְּבַיִת שֶׁכֻּלָּם אוֹהֲבִים אוֹתוֹ – יֵשׁ לוֹ תַּעֲנוּג בָּעוֹלָם הַזֶּה.

כִּבּוּד הוֹרִים מְבִיא אַהֲבָה, וְאֹשֶׁר בַּבַּיִת.

🎊

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹבֵד אֶת ה׳ בְּשִׂמְחָה 💛

📖 כִּבּוּד אָב וָאֵם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים ק׳ ב׳

רָמָה ב׳ – שְׁנֵי הַהוֹרִים וְהַשִּׂמְחָה

🎊 בְּשִׂמְחָה – פִּי כַּמָּה

בִּתְהִלִּים כָּתוּב: ׳עִבְדוּ אֶת ה׳ בְּשִׂמְחָה, בֹּאוּ לְפָנָיו בִּרְנָנָה׳. אֶפְשָׁר לַעֲשׂוֹת אֶת אוֹתָהּ פְּעֻלָּה שֶׁל כִּבּוּד הוֹרִים וּלְקַבֵּל עָלֶיהָ פִּי כַּמָּה, תָּלוּי אִם עוֹשִׂים בְּשִׂמְחָה שֶׁזָּכִינוּ.

מִי שֶׁעוֹשֶׂה בְּשִׂמְחָה שָׁוֶה לוֹ מִילְיוֹנִים, וּמִי שֶׁעוֹשֶׂה בְּפָנִים עֲצוּבוֹת – כַּמָּה הוּא מַפְסִיד.

הַשִּׂמְחָה מַכְפִּילָה אֶת הַשָּׂכָר.

👋

💛 אַהֲבַת הַשֵּׁם: אֲנִי פּוֹתֵחַ כָּל כְּנִיסָה בְּשָׁלוֹם 💛

📖 כִּבּוּד אָב וָאֵם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּרָכוֹת ב׳ א׳

רָמָה ב׳ – שְׁנֵי הַהוֹרִים וְהַשִּׂמְחָה

👋 שָׁלוֹם – כָּבוֹד קָטָן

כְּשֶׁנִּכְנָסִים הַבַּיְתָה אוֹמְרִים שָׁלוֹם לַהוֹרִים. בַּמִּשְׁנָה בְּמַסֶּכֶת בְּרָכוֹת כָּתוּב שֶׁאֲפִלּוּ בֵּין הַפְּרָקִים שֶׁל קְרִיאַת שְׁמַע מֻתָּר לִשְׁאֹל בִּשְׁלוֹם אָדָם שֶׁצָּרִיךְ לְכַבְּדוֹ.

מִכָּאן רוֹאִים כַּמָּה הַכָּבוֹד וְהַשָּׁלוֹם חֲשׁוּבִים. וְאִם כָּךְ לְכָל אָדָם חָשׁוּב – קַל וָחֹמֶר לַהוֹרִים שֶׁנָּתְנוּ לָנוּ חַיִּים.

שָׁלוֹם חַם בַּכְּנִיסָה הוּא כָּבוֹד קָטָן וְחָשׁוּב.

רָמָה ב׳ – שְׁנֵי הַהוֹרִים וְהַשִּׂמְחָה

מָה לוֹקְחִים מִרָמָה ב׳?

כִּבּוּד הוֹרִים אֵינוֹ מַשִּׂימָה כְּבֵדָה אֶלָּא הִזְדַּמְּנוּת לְהַרְגִּישׁ אַהֲבָה. כְּשֶׁאֲנִי עוֹשֶׂה אוֹתוֹ בְּשִׂמְחָה, אֲנִי עוֹבֵד אֶת ה׳ בְּשִׂמְחָה וּמַרְגִּישׁ כַּמָּה הוּא אוֹהֵב אוֹתִי וְנָתַן לִי בַּיִת מָלֵא אַהֲבָה.

🤔 חוֹשְׁבִים עַל זֶה

  • בְּאֵיזֶה מַצָּב הַשָּׁבוּעַ עָשִׂיתִי מַה שֶּׁבִּקְשׁוּ מִמֶּנִּי בְּפָנִים חֲמוּצוֹת, וְאֵיךְ אֶעֱשֶׂה אוֹתוֹ בְּשִׂמְחָה בַּפַּעַם הַבָּאָה?
  • אֵיךְ אֲנִי יָכוֹל לְכַבֵּד אֶת אִמָּא כְּשֶׁאַבָּא כָּאן, וְאֶת אַבָּא כְּשֶׁאִמָּא כָּאן?
  • אֲחַבֵּר שִׁיר תּוֹדָה קָצָר לְאִמָּא וּלְאַבָּא, אָשִׁיר אוֹתוֹ לָהֶם בְּחִיּוּךְ, וְאֶרְשֹׁם מָה הֵם הִרְגִּישׁוּ.

📜 הַצַּוָּאָה וְהַבְּחִירָה

בְּסוֹף יָמָיו הוֹרָה רַבִּי יְהוּדָה הַנָּשִׂיא: ׳שִׁמְעוֹן בְּנִי חָכָם, גַּמְלִיאֵל בְּנִי נָשִׂיא, חֲנִינָא בַּר חָמָא יֵשֵׁב בָּרֹאשׁ׳.

לְכָל אֶחָד תַּפְקִיד: חָכָם, נָשִׂיא וְרֹאשׁ יְשִׁיבָה.

רַבִּי חֲנִינָא קִבֵּל אֶת הַמִּשְׂרָה הַחֲשׁוּבָה בְּיוֹתֵר בְּעוֹלַם הַתּוֹרָה, וְדַוְקָא אָז נִבְחַן: מָה יַעֲשֶׂה?

הַצַּוָּאָה נָתְנָה לְרַבִּי חֲנִינָא אֶת רֹאשׁ הַיְשִׁיבָה.

🔢 לֹא קִבֵּל

הַגְּמָרָא מְסַפֶּרֶת: ׳לֹא קִבֵּל רַבִּי חֲנִינָא, שֶׁהָיָה רַבִּי אַפָּס גָּדוֹל מִמֶּנּוּ שְׁתֵּי שָׁנִים וּמֶחֱצָה׳.

זֶה הֶפְרֵשׁ קָטָן מְאֹד, וּבְכָל זֹאת רַבִּי חֲנִינָא לֹא הִשְׁתַּמֵּשׁ בַּצַּוָּאָה כְּדֵי לְהִתְקַדֵּם. כְּבוֹד הַגָּדוֹל מִמֶּנּוּ הָיָה חָשׁוּב לוֹ יוֹתֵר מִן הַמִּשְׂרָה.

רַבִּי אַפָּס יָשַׁב בָּרֹאשׁ, רַבִּי חֲנִינָא יָשַׁב בַּחוּץ, וְרַבִּי לֵוִי בָּא וְיָשַׁב לְיָדוֹ.

כְּבוֹד הַגָּדוֹל חָשׁוּב מִמִּשְׂרָה, גַּם כְּשֶׁהַהֶפְרֵשׁ קָטָן.

רָמָה ב׳ – כָּבוֹד לַגָּדוֹל

⚖️ אֲבָל מָה עִם הַצַּוָּאָה?

הַגְּמָרָא שׁוֹאֶלֶת: אִם רַבִּי חֲנִינָא וִתֵּר, מָה עִם דִּבְרֵי הַנָּשִׂיא? וְהִיא עוֹנָה: דִּבְרֵי צַדִּיק מִתְקַיְּמִים, ׳וְתִגְזַר אֹמֶר וְיָקׇם לָךְ׳.

אַחֲרֵי הַזְּמַן רַבִּי חֲנִינָא יָשַׁב בָּרֹאשׁ, וְכָךְ הִתְקַיְּמוּ דְּבָרָיו שֶׁל רַבִּי.

הַוִּתּוּר לֹא בִּטֵּל אֶת רְצוֹן הַנָּשִׂיא, אֶלָּא קָבַע סֵדֶר: קֹדֶם הַגָּדוֹל, וְאַחַר כָּךְ הוּא.

וִתּוּר בִּכְבוֹד לֹא מַפְסִיד, כִּי דִּבְרֵי צַדִּיק מִתְקַיְּמִים.

רָמָה ב׳ – כָּבוֹד לַגָּדוֹל

🥿 חַמִּין וָשֶׁמֶן

הַגְּמָרָא מְסַפֶּרֶת שֶׁרַבִּי חֲנִינָא, בֶּן שְׁמוֹנִים, עָמַד עַל רֶגֶל אַחַת וְחָלַץ וְנָעַל אֶת הַמִּנְעָל. הוּא עַצְמוֹ הִסְבִּיר: ׳חַמִּין וָשֶׁמֶן שֶׁסָּכַתְנִי אִמִּי בְּיַלְדוּתִי – הֵם שֶׁעָמְדוּ לִי בְּעֵת זִקְנוּתִי׳.

הוּא הִכִּיר טוֹבָה עַל הַדְּאָגָה שֶׁל אִמּוֹ.

הַסִּפּוּר שֶׁל הָרַב מוֹסִיף סוֹד נוֹסָף: הוּא מֵעוֹלָם לֹא תָּפַס אֶת מְקוֹמוֹ שֶׁל מִי שֶׁגָּדוֹל מִמֶּנּוּ.

הַכָּרַת הַטּוֹב עַל מָה שֶׁנָּתְנוּ לִי, וְכָבוֹד לַגְּדוֹלִים.

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא מַפְסִיד כְּשֶׁאֲנִי נוֹתֵן כָּבוֹד לַחָבֵר 💛

🧒 לָמָּה קָשֶׁה לְוַתֵּר?

לִפְעָמִים קָשֶׁה לָנוּ לְוַתֵּר עַל דְּבָרִים קְטַנִּים הַרְבֵּה יוֹתֵר: מִי יַעֲמֹד רִאשׁוֹן בַּתּוֹר, מִי יֵשֵׁב בַּכִּסֵּא שֶׁלְּיַד הַחַלּוֹן, מִי יַנְהִיג אֶת הַמִּשְׂחָק בַּהַפְסָקָה.

אָנוּ פּוֹחֲדִים: ׳אִם אֲוַתֵּר, אֵצֵא פְּרַאיֶיר וְאַפְסִיד׳.

אֲבָל כָּבוֹד אֲמִתִּי אֵינוֹ נִמְדָּד בַּכִּסֵּא שֶׁאֲנִי יוֹשֵׁב עָלָיו, אֶלָּא בַּכָּבוֹד שֶׁאֲנִי נוֹתֵן לַאֲחֵרִים.

כָּבוֹד אֲמִתִּי הוּא כָּבוֹד שֶׁנּוֹתְנִים לַאֲחֵרִים.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת מִי שֶׁעוֹשֶׂה שָׁלוֹם בְּוִתּוּר, וּלְהַרְגִּישׁ שֶׁה׳ אִתִּי 💛

רָמָה ב׳ – כָּבוֹד לַגָּדוֹל

🕊️ וִתּוּר שֶׁשּׁוֹמֵר שָׁלוֹם

כְּשֶׁכֻּלָּם רוֹצִים לִהְיוֹת רִאשׁוֹנִים, קַל לְהַתְחִיל מְרִיבָה. כְּשֶׁאֶחָד מְוַתֵּר בִּכְבוֹד, הַשָּׁלוֹם נִשְׁמָר.

כָּךְ הָיָה בְּבֵית הַמִּדְרָשׁ: רַבִּי חֲנִינָא קִבֵּל אֶת הַמִּשְׂרָה בַּסּוֹף ׳מִבְּלִי לִפְגֹּעַ בְּאִישׁ׳.

וּמִי שֶׁמְּוַתֵּר עַל הַכָּבוֹד שֶׁלּוֹ לְמַעַן הַשֵּׁנִי, הַקָּדוֹשׁ בָּרוּךְ הוּא בְּעַצְמוֹ שׁוֹמֵר עָלָיו.

הַוִּתּוּר שֶׁל אֶחָד שׁוֹמֵר עַל הַשָּׁלוֹם שֶׁל כֻּלָּם.

מָה לוֹקְחִים מִרָמָה ב׳?

הַוִּתּוּר שֶׁל רַבִּי חֲנִינָא הָיָה וִתּוּר שֶׁל אָדָם שֶׁבָּטַח בַּקָּדוֹשׁ בָּרוּךְ הוּא. מִי שֶׁמְּוַתֵּר בְּלֵב שָׁלֵם, מַרְגִּישׁ שֶׁה׳ אוֹהֵב אוֹתוֹ וְנוֹתֵן לוֹ אֶת הַטּוֹב בִּזְמַנּוֹ.

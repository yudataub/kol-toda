דַּקָּה אַחַת בָּעֶרֶב – תּוֹדָה, סְלִיחָה, עֶזְרָה

אֵיךְ מְסַיְּמִים אֶת הַיּוֹם בְּשִׂיחָה קְצָרָה עִם ה׳: לְהוֹדוֹת, לְבַקֵּשׁ סְלִיחָה אוֹ לְבַקֵּשׁ עֶזְרָה לְמָחָר

📖 תְּפִלָּה מִן הַלֵּב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הוֹשֵׁעַ י״ד ג׳

רָמָה ב׳ – לַחְזֹר אֶל ה׳ בִּדְבָרִים

🗣️ קְחוּ עִמָּכֶם דְּבָרִים

הַנָּבִיא הוֹשֵׁעַ אוֹמֵר: ׳קְחוּ עִמָּכֶם דְּבָרִים וְשׁוּבוּ אֶל ה׳׳. כְּשֶׁאָדָם רוֹצֶה לְתַקֵּן, הוּא לוֹקֵחַ אִתּוֹ מִלִּים וְאוֹמֵר אוֹתָן לַה׳.

זֶה הָעִקָּר בַּדַּקָּה שֶׁל הָעֶרֶב: לֹא רַק לַחְשֹׁב בַּלֵּב, אֶלָּא לְדַבֵּר – תּוֹדָה, סְלִיחָה, עֶזְרָה.

וְאֶפְשָׁר לְהוֹסִיף: ׳אִמְרוּ אֵלָיו׳, כְּלוֹמַר לְדַבֵּר אֵלָיו בְּעַצְמוֹ.

לַחְזֹר אֶל ה׳ בְּמִלִּים, לֹא רַק בְּמַחְשָׁבָה.

1
⛵

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁהַלֵּב שֶׁלִּי יִהְיֶה סָמוּךְ אֵלֶיךָ, ה׳ 💛

📖 תְּפִלָּה מִן הַלֵּב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, קִדּוּשִׁין פ״ב א׳ · רַשִׁ״י

רָמָה ב׳ – לַחְזֹר אֶל ה׳ בִּדְבָרִים

⛵ הַסַּפָּנִים

בַּגְּמָרָא אָמְרוּ: ׳הַסַּפָּנִין – רֻבָּן חֲסִידִים׳. רַשִׁ״י מֵסְבִּיר: הֵם שָׁטִים בַּיָּם, בְּמָקוֹם סַכָּנָה, וּבְכָל רֶגַע לִבָּם לְאֲבִיהֶם שֶׁבַּשָּׁמַיִם.

לָמַדְנוּ: אָדָם שֶׁכָּל הַזְּמַן לִבּוֹ לַשָּׁמַיִם יָכוֹל לְהַגִּיעַ לְדַרְגָּה נִפְלָאָה שֶׁל חֲסִידוּת.

וְיֵשׁ בָּהֶם גַּם עֲנָוָה: הֵם מַרְגִּישִׁים שֶׁבְּלִי ה׳ אִי אֶפְשָׁר לִחְיוֹת.

מִי שֶׁלִּבּוֹ תָּמִיד לַשָּׁמַיִם יָכוֹל לְהַגִּיעַ לַחֲסִידוּת.

2
💭

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁקָּשֶׁה לִי, אֲנִי פּוֹנֶה אֵלֶיךָ, ה׳, מִכָּל הַלֵּב 💛

📖 תְּפִלָּה מִן הַלֵּב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – לַחְזֹר אֶל ה׳ בִּדְבָרִים

💭 מָה לוֹקְחִים מִזֶּה

אֶפְשָׁר לְסַכֵּם שְׁלוֹשָׁה דְּבָרִים לְמַעֲשֶׂה.

רִאשׁוֹן: קְשָׁיִים שֶׁבָּאִים עוֹזְרִים לִי לְכַוֵּן אֶת הַלֵּב אֶל אָבִינוּ שֶׁבַּשָּׁמַיִם.

שֵׁנִי: מָה שֶׁהוֹפֵךְ אָדָם לְחָסִיד הוּא תְּפִלָּה מִן הַלֵּב, וְלָכֵן כְּדַאי לְדַבֵּר עִם ה׳ בְּכָל יוֹם.

שְׁלִישִׁי: גַּם מִי שֶׁאֵינוֹ לוֹמֵד הַרְבֵּה, כְּשֶׁלִּבּוֹ לַשָּׁמַיִם הוּא יָכוֹל לַעֲלוֹת בְּמַעֲלָה גְּדוֹלָה.

תְּפִלָּה מִן הַלֵּב בְּכָל יוֹם הִיא דֶּרֶךְ לְהִתְקָרֵב.

3
🎯

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִחְיוֹת קָרוֹב אֵלֶיךָ, ה׳, כָּל הַיּוֹם 💛

📖 תְּפִלָּה מִן הַלֵּב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – לַחְזֹר אֶל ה׳ בִּדְבָרִים

🎯 הַמַּטָּרָה

הַמַּטָּרָה בַּשִּׂיחָה עִם ה׳ אֵינָהּ רַק לִפְתֹּר אֶת הַבְּעָיוֹת שֶׁלִּי. גַּם זֶה בְּסֵדֶר, אֲבָל הָעִקָּר הוּא שֶׁאֶחְיֶה קָרוֹב לַה׳.

לָכֵן בַּדַּקָּה אֲנִי מוֹדֶה, מְבַקֵּשׁ סְלִיחָה, וּמְבַקֵּשׁ עֶזְרָה לִהְיוֹת טוֹב יוֹתֵר, וְלֹא רַק שֶׁהַכֹּל יִסְתַּדֵּר.

הָעִקָּר הוּא לִחְיוֹת קָרוֹב לַה׳.

4
🌱

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַחְלִיט וּמְבַקֵּשׁ עֶזְרָה, וְה׳ עוֹזֵר 💛

📖 תְּפִלָּה מִן הַלֵּב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – לַחְזֹר אֶל ה׳ בִּדְבָרִים

🌱 מַחְלִיטִים צַעַד קָטָן

אַחֲרֵי שֶׁאוֹמְרִים סְלִיחָה, שׁוֹאֲלִים: מָה אֶעֱשֶׂה מָחָר? לֹא מַסְתַּפְּקִים בְּ׳הַלְוַאי׳. אֲנִי מְקַבֵּל עַל עַצְמִי צַעַד קָטָן מַמָּשִׁי, לְמָשָׁל לָקוּם עֶשֶׂר דַּקּוֹת קֹדֶם, וּמְבַקֵּשׁ מֵה׳ שֶׁיַּעֲזֹר לִי לְקַיֵּם אוֹתוֹ.

כָּךְ הַבַּקָּשָׁה אֵינָהּ רֵיקָה: יֵשׁ בָּהּ גַּם הַחְלָטָה שֶׁלִּי.

סְלִיחָה, הַחְלָטָה קְטַנָּה, וּבַקָּשָׁה לְעֶזְרָה.

מָה לוֹקְחִים מִרָמָה ב׳?

מִי שֶׁמְּדַבֵּר עִם ה׳ בְּסוֹף הַיּוֹם מַרְגִּישׁ שֶׁה׳ קָרוֹב אֵלָיו, כְּמוֹ אַבָּא שֶׁמַּקְשִׁיב. הַמַּטָּרָה הִיא לִחְיוֹת קָרוֹב אֵלָיו, וְזוֹ אַהֲבַת ה׳ בְּמַעֲשֶׂה.

מָה הָיָה הָאֶשֶׁל שֶׁנָּטַע אַבְרָהָם, מָה פֵּרוּשׁ רָאשֵׁי הַתֵּבוֹת, וְלָמָּה הַמִּצְוָה שְׁלֵמָה רַק עִם לְוָיָה

רָמָה ב׳ – שְׁלֵמוּת הַמִּצְוָה

📜 מַחְלֹקֶת חֲכָמִים עַל הָאֶשֶׁל

חֲכָמִים (סוֹטָה י ע״א) נֶחְלְקוּ: יֵשׁ אוֹמֵר שֶׁהָאֶשֶׁל הָיָה פַּרְדֵּס, כְּדֵי לָתֵת מִמֶּנּוּ פֵּרוֹת לָאוֹרְחִים. וְיֵשׁ אוֹמֵר שֶׁהָיָה פּוּנְדָּק, כְּדֵי לָלוּן בּוֹ.

בְּאָבוֹת דְּרַבִּי נָתָן מוּבָא שֶׁבָּנָה ׳פְּלַטְרִין׳ עַל הַדְּרָכִים וְהִנִּיחַ שָׁם מַאֲכָל וּמַשְׁקֶה לְכָל עוֹבֵר.

בְּכָל הַדֵּעוֹת, הָאֶשֶׁל הָיָה מָקוֹם שֶׁכָּל עוֹבֵר דֶּרֶךְ יָכוֹל לַעֲצֹר בּוֹ.

הָאֶשֶׁל הָיָה פַּרְדֵּס אוֹ פּוּנְדָּק – מָקוֹם לְכָל עוֹבֵר.

🔠

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹשֶׂה אֶת מַה שֶׁאֶפְשָׁר, בְּלֵב שָׂמֵחַ 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, חֶפֶץ חַיִּים, אַהֲבַת חֶסֶד ח״ג פ״ב

רָמָה ב׳ – שְׁלֵמוּת הַמִּצְוָה

🔠 רָאשֵׁי תֵּבוֹת – אֲכִילָה, שְׁתִיָּה, לְוָיָה

הֶחָפֵץ חַיִּים מֵבִיא: ׳אֶשֶׁל׳ הוּא אֲכִילָה, שְׁתִיָּה, לְוָיָה (וְיֵשׁ אוֹמְרִים: לִינָה).

וְעֵצָה: ׳מַה טּוֹב לָאָדָם אִם יָכוֹל לִלְמֹד הַמִּדָּה מֵאַבְרָהָם אָבִינוּ לִהְיוֹת לוֹ חֶדֶר מְיֻחָד בִּשְׁבִיל הָאוֹרְחִים, עַכְשָׁו לְלִינָתָם׳, וּבַבֹּקֶר יְקַיֵּם גַּם לְוָיָה.

וְאִם לֹא יָכוֹל לְקַיֵּם אֲכִילָה וּשְׁתִיָּה כָּרָאוּי – לְפָחוֹת יְקַיֵּם לִינָה וְלְוָיָה.

אִם אִי אֶפְשָׁר הַכֹּל – לְפָחוֹת לִינָה וְלְוָיָה.

🙏

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי מְאָרֵחַ, אֲנִי יָכוֹל לַעֲזֹר גַּם לְהוֹדוֹת לַה׳ 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּרֵאשִׁית כ״א ל״ג · חֶפֶץ חַיִּים

רָמָה ב׳ – שְׁלֵמוּת הַמִּצְוָה

🙏 וַיִּקְרָא בְּשֵׁם ה׳

עַל הַמִּלִּים ׳וַיִּקְרָא שָׁם בְּשֵׁם ה׳ אֵל עוֹלָם׳ אָמְרוּ חֲכָמִים: אַבְרָהָם הִקְרִיא שֵׁם שָׁמַיִם בְּפִי כָּל עוֹבֵר.

אַחֲרֵי שֶׁאָכְלוּ וְשָׁתוּ, בִּקֵּשׁ שֶׁיְבָרְכוּ אֶת מִי שֶׁנָּתַן לָהֶם.

כָּךְ הָפְכָה הָאֲרוּחָה לְפֶתַח שֶׁל אֱמוּנָה: הָאוֹרֵחַ לֹא רַק שָׂבַע, הוּא גַּם הִכִּיר טוֹבָה לַמֵּטִיב.

אַבְרָהָם עָזַר לָאוֹרְחִים לְהוֹדוֹת לַה׳ עַל הַטּוֹבָה.

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא מְפַטֵּר אֶת עַצְמִי מֵהַלְּוָיָה, כִּי הִיא שׁוֹמֶרֶת עַל שְׁנֵינוּ 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, חֶפֶץ חַיִּים, אַהֲבַת חֶסֶד ח״ג פ״ב

רָמָה ב׳ – שְׁלֵמוּת הַמִּצְוָה

🚶 כַּמָּה צָרִיךְ לְלַוֹּת?

בִּימֵי חֲכָמִים הָלְכוּ מִיל וּפַרְסָה, כִּי הַדְּרָכִים הָיוּ מְסֻכָּנוֹת. הֶחָפֵץ חַיִּים מֵבִיא שֶׁהַיּוֹם לְפָחוֹת אַרְבַּע אַמּוֹת, כְּלוֹמַר לָלֶכֶת עִם הָאוֹרֵחַ עַד הַשַּׁעַר.

וְהוּא מוֹסִיף: ׳חָלִילָה לִפְטֹר אֶת עַצְמוֹ מִמִּצְוַת לְוָיָה לְגַמְרֵי׳, כִּי הִיא סְגֻלָּה שֶׁלֹּא יִקְרֶה לוֹ אָסוֹן בַּדֶּרֶךְ.

וְאִם הָאוֹרֵחַ לֹא מַכִּיר אֶת הַדֶּרֶךְ – מִצְוָה רַבָּה לָלֶכֶת אִתּוֹ אוֹ לְהַסְבִּיר לוֹ הֵיטֵב.

מְלַוִּים לְפָחוֹת אַרְבַּע אַמּוֹת, וּבְדֶרֶךְ לֹא מוּכֶּרֶת – יוֹתֵר.

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אוֹצַר פְּנִינִים וְעוֹבְדוֹת · חֶפֶץ חַיִּים

רָמָה ב׳ – שְׁלֵמוּת הַמִּצְוָה

🔥 הָעָשִׁיר מִוִּילְנָה

בְּוִילְנָה הָיָה עָשִׁיר גָּדוֹל, מַכְנִיס אוֹרְחִים. נֶזֶק גָּדוֹל בְּאֵשׁ פָּגַע בְּבֵיתוֹ. שָׁאֲלוּ אֶת הַגְּרָ״א לָמָּה לֹא הֵגִנָּה עָלָיו הַמִּצְוָה שֶׁהָיָה זָהִיר בָּהּ.

הַגְּרָ״א הִסְבִּיר: הוּא נָתַן אֲכִילָה וּשְׁתִיָּה בִּלְבַד וְלֹא קִיֵּם לְוָיָה, וְנִשְׁאֲרוּ מִן הָאֶשֶׁל רַק הָאוֹתִיּוֹת ׳אֵשׁ׳.

הֶחָפֵץ חַיִּים מְסַיֵּם: ׳שֵׁם אֶשֶׁל אֵינוֹ מֻנָּח אֶלָּא אִם כֵּן הִשְׁלִימָהּ כָּרָאוּי׳.

מִצְוָה חֲסֵרָה אֵינָהּ מְגִנָּה כְּמוֹ מִצְוָה שְׁלֵמָה.

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַבֵּנוּ בַּחְיֵי · בְּרֵאשִׁית י״ח ט״ז

רָמָה ב׳ – שְׁלֵמוּת הַמִּצְוָה

🤝 הַלְּוָיָה מוֹכִיחָה אֶת הַלֵּב

רַבֵּנוּ בַּחְיֵי כּוֹתֵב שֶׁאַחֲרֵי ׳וְאַבְרָהָם הֹלֵךְ עִמָּם לְשַׁלְּחָם׳ כָּתוּב ׳וַה׳ אָמַר׳ – בִּזְכוּת הַלְּוָיָה בָּא אֵלָיו הַדִּבּוּר.

לָמָּה הַלְּוָיָה מְיֻחֶדֶת? כְּשֶׁהַמְּאָרֵחַ מְאָרֵחַ, הָאוֹרֵחַ אֵינוֹ יָכוֹל לָדַעַת אִם הוּא שָׂמֵחַ בֶּאֱמֶת. אֲבָל כְּשֶׁהַמְּאָרֵחַ יוֹצֵא אִתּוֹ מִן הַבַּיִת אַחֲרֵי שֶׁכְּבָר יָצָא יְדֵי חוֹבָתוֹ, זֶה סִימָן שֶׁהָאֲרוּחָה הָיָה מִן הַלֵּב.

הַלְּוָיָה מַרְאָה שֶׁהָאֲרוּחָה הָיָה מִן הַלֵּב.

רָמָה ב׳ – שְׁלֵמוּת הַמִּצְוָה

מָה לוֹקְחִים מִרָמָה ב׳?

ה׳ נוֹתֵן לָנוּ טוֹבָה שְׁלֵמָה, וְרוֹצֶה מִצְווֹת שְׁלֵמוֹת. כְּשֶׁאֲנִי מַשְׁלִים אֶת הַמִּצְוָה עַד הַלְּוָיָה, אֲנִי מַרְאֶה שֶׁאֲנִי עוֹשֶׂה אוֹתָהּ בִּשְׁבִילוֹ וּמֵאַהֲבָה, וְלֹא רַק כְּדֵי לְסַמֵּן.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה בַּבַּיִת שֶׁלִּי יָכוֹל לִהְיוֹת ׳אֶשֶׁל׳ – חֶדֶר, מָקוֹם, אוֹ שָׁעָה לְאוֹרֵחַ?
  • מָתַי הִפְסַקְתִּי בְּאֶמְצַע, בְּלִי לְסַיֵּם אֶת הַחֶסֶד, וְאֵיךְ אַשְׁלִים בַּפַּעַם הַבָּאָה?
  • הַשָּׁבוּעַ אֲכִין חֲבִילַת אֶשֶׁ״ל לְאוֹרֵחַ דִּמְיוֹנִי – אֲכִילָה, שְׁתִיָּה, לִינָה וְלְוָיָה – וַאֲסַפֵּר עָלֶיהָ בַּבַּיִת.

לָמָּה נִבְרָא הַפֶּה, אֵיךְ רְאִיַּת הַטּוֹב מְבִיאָה לְהוֹדָיָה, וְלָמָּה מִי שֶׁמְּחַפֵּשׂ רַע מוֹצֵא אוֹתוֹ אֲפִלּוּ בַּמָּן

רָמָה ב׳ – עַיִן טוֹבָה, פֶּה טוֹב

📜 חַיִּים טוֹבִים, לֹא רַק אֲרֻכִּים

׳מִי הָאִישׁ הֶחָפֵץ חַיִּים אֹהֵב יָמִים לִרְאוֹת טוֹב׳. מְבָאֲרִים: מִי שֶׁרוֹצֶה חַיִּים לֹא רַק בַּכַּמּוּת אֶלָּא גַּם בָּאֵיכוּת. ׳אֹהֵב יָמִים׳ – יָמִים מְלֵאִים בְּאַהֲבַת ה׳ וְאַהֲבַת הַזּוּלָת.

וְהַדֶּרֶךְ לְכָךְ: לִרְאוֹת טוֹב – לְהִתְרַגֵּל לִרְאוֹת אֶת הַטּוֹב בַּזּוּלָת, בַּבְּרִיאָה וּבַבּוֹרֵא. זֶה הַמַּתְכּוֹן לְאוֹשֶׁר בִּשְׁנֵי הָעוֹלָמוֹת.

אוֹשֶׁר בָּא מִ׳לִרְאוֹת טוֹב׳: בַּזּוּלָת, בַּבְּרִיאָה וּבַבּוֹרֵא.

רָמָה ב׳ – עַיִן טוֹבָה, פֶּה טוֹב

📿 הַזְּכִירוֹת וְ׳לְהוֹדוֹת לְךָ׳

בִּבְרָכַת ׳אַהֲבָה רַבָּה׳ בַּבֹּקֶר מְכַוְּנִים, לְפִי הָאֲרִ״י בַּמָּגֵן אַבְרָהָם, לְכַמָּה זְכִירוֹת: ׳וּבָנוּ בָחַרְתָּ׳ – מַתַּן תּוֹרָה, ׳וְקֵרַבְתָּנוּ׳ – מַעֲמָד הַר סִינַי, ׳לְשִׁמְךָ הַגָּדוֹל׳ – מַעֲשֵׂה עֲמָלֵק.

וּבַמִּלִּים ׳לְהוֹדוֹת לְךָ׳ מְכַוְּנִים: הַפֶּה נִבְרָא רַק לְהוֹדוֹת, וְלֹא לְדַבֵּר לָשׁוֹן הָרָע.

בְּ׳לְהוֹדוֹת לְךָ׳ מְכַוְּנִים: הַפֶּה נִבְרָא לְהוֹדוֹת.

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹחֵר לִרְאוֹת טוֹב בַּחֲבֵרִים וְלֹא לְחַפֵּשׂ רַע 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מָגֵן אַבְרָהָם ס׳ · שִׁעוּרֵי הָרַב

רָמָה ב׳ – עַיִן טוֹבָה, פֶּה טוֹב

🔍 לָמָּה דַּוְקָא לָשׁוֹן הָרָע?

הַמָּגֵן אַבְרָהָם לֹא אָמַר ׳לְהוֹדוֹת לְךָ וְלֹא לְשַׁקֵּר׳, אֶלָּא ׳לֹא לְדַבֵּר לָשׁוֹן הָרָע׳. מַטָּרַת הַפֶּה הִיא לְהוֹדוֹת, וְהוֹדָיָה בָּאָה עַל יְדֵי רְאִיַּת הַטּוֹב.

הַהֵפֶךְ הַגָּמוּר מִ׳לִרְאוֹת טוֹב׳ הוּא לָשׁוֹן הָרָע: הָאָדָם מִתְרַכֵּז בָּרַע שֶׁבַּזּוּלָת, חוֹשֵׁב רַע וּמְדַבֵּר רַע. לָכֵן הוּא מוֹצִיא אֶת הָאָדָם מִשִּׁירָה וּמִשִּׂמְחָה.

רְאִיַּת טוֹב מְבִיאָה הוֹדָיָה, וְלָשׁוֹן הָרָע הוּא רְאִיָּה שֶׁל רַע.

🍯

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְשַׁנֶּה אֶת הַהֶרְגֵּל: מְחַפֵּשׂ טוֹב וְלֹא רַע 💛

📖 שְׁמִירַת הַלָּשׁוֹן – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בַּמִּדְבָּר י״א ו׳־ז׳ · שִׁעוּרֵי הָרַב

רָמָה ב׳ – עַיִן טוֹבָה, פֶּה טוֹב

🍯 מִי שֶׁמְּחַפֵּשׂ רַע מוֹצֵא

בְּפָרָשַׁת בְּהַעֲלֹתְךָ מְתָאֶרֶת הַתּוֹרָה אֶת הַמָּן: ׳וְעֵינוֹ כְּעֵין הַבְּדֹלַח׳. מָצִינוּ שֶׁהָעָם אָמַר ׳וְעַתָּה נַפְשֵׁנוּ יְבֵשָׁה אֵין כֹּל׳.

הָרַב סִפֵּר שֶׁשָּׁנִים קָרָא אֶת הַפָּרָשָׁה וְלֹא הֵבִין: אֵיךְ מָן כָּזֶה וּבְאוֹתוֹ זְמַן ׳אֵין כֹּל׳? עַד שֶׁרָאָה אֲנָשִׁים שֶׁגַּם יְשׁוּעָה גְּדוֹלָה הִתְקַבְּלָה אֶצְלָם בִּתְלוּנוֹת. וְאָז הֵבִין: מִי שֶׁהִתְרַגֵּל לְחַפֵּשׂ רַע – יִמְצָא אוֹתוֹ גַּם בַּמָּן.

הֶרְגֵּל שֶׁל חִפּוּשׂ רַע מוֹצִיא תְּלוּנָה מִכָּל דָּבָר.

רָמָה ב׳ – עַיִן טוֹבָה, פֶּה טוֹב

🕵️ עַיִן שֶׁלֹּא מַאֲמִינָה בְּטוֹב

הָרַב סִפֵּר עַל יְהוּדִי שֶׁהֵבִיא תְּרוּפוֹת מֵהָעִיר הַגְּדוֹלָה לְכָל אַנְשֵׁי עִירוֹ, שֶׁלֹּא הָיָה בָּהּ בֵּית מִרְקַחַת – בְּלִי תַּשְׁלוּם. שָׁאֲלוּ אוֹתוֹ: ׳מָה הִרְוַחְתָּ?׳ וְהוּא עָנָה: ׳מִצְווֹת׳. הַשּׁוֹאֵל לֹא הִצְלִיחַ לְהָבִין.

יֵשׁ אֲנָשִׁים שֶׁקָּשֶׁה לָהֶם לְהַאֲמִין שֶׁמִּישֶׁהוּ עוֹשֶׂה טוֹב סְתָם כָּךְ, וְתָמִיד מְחַפְּשִׂים כַּוָּנָה רָעָה. אֲבָל הָאָדָם בְּטִבְעוֹ אוֹהֵב לַעֲשׂוֹת טוֹב.

עַיִן טוֹבָה מַאֲמִינָה שֶׁאֲנָשִׁים עוֹשִׂים טוֹב סְתָם כָּךְ.

מָה לוֹקְחִים מִרָמָה ב׳?

הוֹדָיָה בָּאָה מֵעַיִן טוֹבָה. מִי שֶׁרוֹאֶה אֶת הַטּוֹב וּמוֹדֶה עָלָיו, מִשְׁתַּמֵּשׁ בַּפֶּה כְּמוֹ שֶׁנִּבְרָא, וּמַרְגִּישׁ כַּמָּה ה׳ אוֹהֵב אוֹתוֹ.

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְסִלַּת יְשָׁרִים פֶּרֶק ח׳

רָמָה ב׳ – שְׁאִיפָה, תָּכְנִית וְהֶרְגֵּל

🙏 הַתְבּוֹנְנוּת בְּטוֹבוֹת

בַּמְּסִלַּת יְשָׁרִים: כְּשֶׁאָדָם יוֹדֵעַ כַּמָּה גָּדוֹל עֵרֶךְ הַמִּצְווֹת, לִבּוֹ מִתְעוֹרֵר לָעֲבוֹדָה. וְעוֹד יוֹתֵר מוֹעִיל לְהִסְתַּכֵּל בְּרֹב הַטּוֹבוֹת שֶׁה׳ עוֹשֶׂה עִם הָאָדָם בְּכָל עֵת וּבְכָל שָׁעָה, מִיּוֹם הַלֵּדָה.

מִי שֶׁמִּסְתַּכֵּל כָּךְ מַכִּיר חוֹבָה רַבָּה לָאֵל הַמֵּיטִיב. וְהוֹאִיל וְאִי אֶפְשָׁר לִגְמֹל לוֹ, לְפָחוֹת יוֹדֶה לִשְׁמוֹ וִיקַיֵּם מִצְווֹתָיו.

אֵין אָדָם, עָשִׁיר אוֹ עָנִי, שֶׁלֹּא רוֹאֶה נִפְלָאוֹת וְטוֹבוֹת בְּמַצָּבוֹ.

מִי שֶׁרוֹאֶה אֶת הַטּוֹבוֹת – לֹא מִתְעַצֵּל לְהוֹדוֹת וְלַעֲבֹד.

💔

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ, כִּי שִׂמְחָה מְסַלֶּקֶת עַצְלוּת 💛

רָמָה ב׳ – שְׁאִיפָה, תָּכְנִית וְהֶרְגֵּל

💔 מֵאַיִן בָּאָה הִתְרַפּוּת?

הָרַב חַיִּים וִיטָל כּוֹתֵב בְּ׳שַׁעֲרֵי קְדֻשָּׁה׳ שֶׁהָעַצְלוּת הִיא תּוֹצָאָה שֶׁל עֶצֶב, כְּלוֹמַר מִיֵּאוּשׁ וְשֶׁלֹּא מַכִּירִים אֶת הָעֵרֶךְ שֶׁל הַמַּעֲשִׂים.

כְּשֶׁאָדָם מוֹצֵא בְּעַצְמוֹ מְנִיעָה, עָלָיו לִלְמֹד מוּסָר עַל חֲשִׁיבוּת הָעִנְיָן. כְּשֶׁיַּכִּיר בְּכָךְ, יִזְדָּרֵז יוֹתֵר.

כָּל אֶחָד מִתְעַצֵּל בִּתְחוּם אַחֵר: אֶחָד זָרִיז בְּחֶסֶד וּמִתְעַצֵּל בַּעֲבוֹדָה, שֵׁנִי רָץ לַגְּמָרָא וּמִתְעַצֵּל בִּתְפִלָּה.

עַצְלוּת בָּאָה מֵעֶצֶב וּמֵאִי־הַכָּרָה בְּעֵרֶךְ הַמַּעֲשֶׂה.

🪡

💛 אַהֲבַת הַשֵּׁם: אֲנִי שׁוֹאֵף לְהִתְקָרֵב לַה׳, וְה׳ מְקָרֵב אוֹתִי 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִכְתָּב מֵאֵלִיָּהוּ ח״ג · תַּנְחוּמָא וַיֵּרָא

רָמָה ב׳ – שְׁאִיפָה, תָּכְנִית וְהֶרְגֵּל

🪡 פִּתְחוֹ שֶׁל מַחַט

בְּ׳מִכְתָּב מֵאֵלִיָּהוּ׳ מְבֹאָר: מֶרְחַק הַשֶּׁמֶשׁ מִן הָאָרֶץ גָּדוֹל מְאֹד, וְאוֹרָהּ מַגִּיעַ אֵלֵינוּ בְּרגָעִים. לְעֻמַּת זֹאת, כְּשֶׁחוֹפְרִים מִנְהָרָה בְּהַר, אֲפִלּוּ כַּמָּה מֶטְרִים לוֹקְחִים זְמַן רַב. דֶּרֶךְ הָעֲלִיָּה הָרוּחָנִית קָשָׁה.

אֲבָל כְּשֶׁהָאָדָם עוֹשֶׂה ׳פִּתְחוֹ שֶׁל מַחַט׳ – שְׁאִיפָה גְּדוֹלָה – אוֹרוֹ שֶׁל ה׳ מְמַהֵר לָבוֹא אֶל הַלֵּב כְּבָרָק.

כָּךְ אַבְרָהָם, חַלָּשׁ וְכוֹאֵב, רָץ בְּשְׁאִיפָה לַחֶסֶד, וְרִיצָה זוֹ הֵבִיאָה גִּלּוּי גָּדוֹל לְבָנָיו: ׳מִסִּינַי בָּא וְזָרַח מִשֵּׂעִיר לָמוֹ׳.

שְׁאִיפָה גְּדוֹלָה פּוֹתַחַת אֶת הַלֵּב לַעֲזָרָה מֵה׳.

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, חוֹבַת הַתַּלְמִידִים

❤️ אַהֲבַת הָעֲבוֹדָה

בַּסֵּפֶר ׳חוֹבַת הַתַּלְמִידִים׳ שׁוֹאֲלִים בָּחוּר מִתְרַפֶּה: כְּשֶׁאַתָּה צָרִיךְ לִנְסֹעַ נְסִיעָה שֶׁאַתָּה אוֹהֵב, אַתָּה מַשְׁכִּים, לוֹבֵשׁ מַהֵר, וְכָל מַעֲשֶׂיךָ בִּזְרִיזוּת. וּבְעֶרֶב פֶּסַח אַתָּה מְמַהֵר לַעֲשׂוֹת הַכֹּל. לָמָּה? כִּי הָעֲבוֹדָה אֲהוּבָה עָלֶיךָ וְנַפְשְׁךָ שְׂמֵחָה בָּהּ.

כְּשֶׁאוֹהֲבִים, הַנֶּפֶשׁ מִתְעוֹרֶרֶת בְּכֹחַ, וְכָל הַגּוּף חַי וּמִתְאַמֵּץ.

לָכֵן צָרִיךְ לִלְמֹד לֶאֱהֹב אֶת הָעֲבוֹדָה וְאֶת הָעָמָל.

כְּשֶׁאוֹהֲבִים אֶת הָעֲבוֹדָה – הַנֶּפֶשׁ מִתְעוֹרֶרֶת וּמִזְדָּרֶזֶת.

רָמָה ב׳ – שְׁאִיפָה, תָּכְנִית וְהֶרְגֵּל

📋 תָּכְנִית יוֹמִית

הָעֵצָה: לִקְצֹב לְכָל יוֹם שִׁעוּרִים, לֹא הַרְבֵּה מִדַּי בְּפַעַם אַחַת. כְּשֶׁעוֹבְרִים כַּמָּה יָמִים וּמַשְׁלִימִים אֶת הַשִּׁעוּרִים יוֹם בְּיוֹמוֹ, מוֹסִיפִים בָּהֶם מְעַט.

טוֹב לִכְתֹּב עַל נְיָר אֶת סֵדֶר הַיּוֹם, וְכָל יוֹם לְהַבִּיט בָּרְשִׁימָה לְהַשְׁלִים אֶת הַחֹק.

מִי שֶׁלֹּא עָמַד בָּרְשִׁימָה מֵרִשְׁלָנוּת, מִצְטַעֵר וּמַשְׁלִים בְּיוֹם הַשֵּׁנִי. כָּךְ הַשִּׁעוּרִים נַעֲשִׂים לְחֹק קָבוּעַ.

תָּכְנִית קְטַנָּה, כְּתִיבָה, בְּדִיקָה וְהַשְׁלָמָה.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִתְרַגֵּל לְמַעֲשִׂים טוֹבִים, וְהַלֵּב שֶׁלִּי מִתְקָרֵב לַה׳ 💛

🔁 הֶרְגֵּל וּגְדוֹלִים

בְּ׳מִגְדַּל עֹז׳: מִי שֶׁמַּרְגִּיל אֶת עַצְמוֹ בִּזְרִיזוּת בִּימֵי הַנַּעֲרוּת, יִהְיֶה זָרִיז וְהוֹלֵךְ אֲפִלּוּ בִּזְקֵנוּתוֹ, כְּמוֹ שֶׁנֶּאֱמַר: ׳חֲנֹךְ לַנַּעַר עַל פִּי דַרְכּוֹ׳. וְגַם אִם רוֹאִים שֶׁהַבֵּן זָרִיז, מוֹסִיפִים לְזָרֵז אוֹתוֹ.

סֵפֶר הַחִנּוּךְ: ׳הָאָדָם נִפְעָל כְּפִי פְּעֻלּוֹתָיו, וְלִבּוֹ וְכָל מַחְשְׁבוֹתָיו אַחַר מַעֲשָׂיו׳. אֲפִלּוּ מִי שֶׁמְּשַׁקֵּעַ בַּתּוֹרָה וּבְמִצְווֹת, נוֹטֶה אֶל הַטּוֹב.

וְהַדֶּרֶךְ הָאַחֲרוֹנָה: לִלְמֹד מִמַּעֲשֵׂי הַגְּדוֹלִים, כִּי ׳אַחֲרֵי הַפְּעֻלּוֹת נִמְשָׁכִים הַלְּבָבוֹת׳.

הֶרְגֵּל טוֹב נִשְׁאָר לְכָל הַחַיִּים, וְהַמַּעֲשִׂים מוֹשְׁכִים אֶת הַלֵּב.

מָה לוֹקְחִים מִרָמָה ב׳?

מִי שֶׁמַּכִּיר בְּחֶסֶד ה׳ וְשׁוֹאֵף לְהִתְקָרֵב אֵלָיו, מוֹצֵא אֶת הַכֹּחַ לִהְיוֹת זָרִיז. הַשְּׁאִיפָה הַגְּדוֹלָה פּוֹתַחַת אֶת הַלֵּב, וּבָהּ ה׳ עוֹזֵר לָנוּ בְּמִהֵרָה.

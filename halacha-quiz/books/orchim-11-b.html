רָמָה ב׳ – כָּבוֹד לָאוֹרֵחַ

אֵיךְ מַרְגִּישׁ אוֹרֵחַ עָיֵף שֶׁמַּגִּיעַ, אֵילוּ דְּבָרִים קְטַנִּים הוֹפְכִים אֶת הַבַּיִת לְחַם וְנָעִים, וּמָה עוֹשֶׂה אוֹרֵחַ טוֹב

📖 הַכְנָסַת אוֹרְחִים · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹחֵר דָּבָר קָטָן אֶחָד שֶׁיְּשַׂמַּח מִישֶׁהוּ הַיּוֹם 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – כָּבוֹד לָאוֹרֵחַ

🤲 לָמָּה דָּבָר קָטָן מַשְׁפִּיעַ?

מָה גּוֹרֵם לָאוֹרֵחַ לִזְכֹּר אֶת הַשַּׁבָּת שֶׁלּוֹ אֶצְלֵנוּ גַּם אַחֲרֵי שָׁנִים? לֹא תָּמִיד הַדְּבָרִים הַגְּדוֹלִים. לִפְעָמִים דַּוְקָא דָּבָר קָטָן שֶׁעוֹלֶה מְעַט.

מִשְׁפָּחָה שֶׁנָּסְעָה שָׁעוֹת אֲרֻכּוֹת וְהִגִּיעָה מְיֻגַּעַת, מַרְגִּישָׁה שֶׁהָרִעֲנוּן שֶׁמְּחַכֶּה לָהּ שָׁוֶה יוֹתֵר מִמְּחִירוֹ. כָּךְ גַּם סְלַסְלַת קַבָּלַת פָּנִים עִם כַּמָּה דְּבָרִים לֶאֱכֹל וּמַשֶּׁהוּ מְיֻחָד מִן הַמָּקוֹם.

הַהַפְתָּעוֹת הָאֵלֶּה יוֹצְרוֹת הַרְגָּשָׁה טוֹבָה וּמוֹנְעוֹת תְּלוּנוֹת.

דְּבָרִים קְטַנִּים שֶׁעוֹלִים מְעַט יוֹצְרִים זִכָּרוֹן טוֹב לְשָׁנִים.

1
🧼

💛 אַהֲבַת הַשֵּׁם: אֲנִי דּוֹאֵג לְנִקָּיוֹן וְלִיֹשֶׁר גַּם בִּדְבָרִים שֶׁאִישׁ אֵינוֹ רוֹאֶה, כִּי ה׳ רוֹאֶה אוֹתִי 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – כָּבוֹד לָאוֹרֵחַ

🧼 מַצָּעִים מְכֻבָּסִים

אוֹרֵחַ מַנִּיחַ שֶׁהַמַּצָּעִים שֶׁהוּא מְקַבֵּל מְכֻבָּסִים. מַצָּעִים מְשֻׁמָּשִׁים, אֲפִלּוּ אִם הֵם נִרְאִים נְקִיִּים, הֵם מֵעֵין גְּנֵבַת דַּעַת: מַטְעִים אֶת הָאוֹרֵחַ בְּמַה שֶׁהוּא בָּטוּחַ בּוֹ.

לָכֵן מַצָּעִים מְכֻבָּסִים לְכָל אוֹרֵחַ הֵם חוֹבָה. גַּם מְנַקִּים בִּקְבִיעוּת, מְאַוְרְרִים הַרְבֵּה וְדוֹאֲגִים לְרֵיחַ נָעִים.

נִקָּיוֹן אֵינוֹ מוֹתָרוּת בְּאֵרוּחַ. הוּא חֵלֶק מִן הַיֹּשֶׁר וּמִכְּבוֹד הָאוֹרֵחַ.

נִקָּיוֹן הוּא חֵלֶק מִן הַיֹּשֶׁר וּמִכְּבוֹד הָאוֹרֵחַ.

2
🛡️

💛 אַהֲבַת הַשֵּׁם: אֲנִי חוֹשֵׁב מֵרֹאשׁ עַל שְׁלוֹם הָאוֹרֵחַ, כִּי שְׁמִירַת חַיִּים הִיא מִצְוָה גְּדוֹלָה 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דְּבָרִים כ״ב ח׳

רָמָה ב׳ – כָּבוֹד לָאוֹרֵחַ

🛡️ מַה שֶּׁהָאוֹרֵחַ לֹא יוֹדֵעַ

בַּעַל הַבַּיִת רָגִיל לַבַּיִת שֶׁלּוֹ. הוּא יוֹדֵעַ אֵיזֶה בֶּרֶז צָרִיךְ לְסַבֵּב חָזָק וְאֵיפֹה הַמַּדְרֵגָה הַמַּחֲלִיקָה. הָאוֹרֵחַ לֹא יוֹדֵעַ.

לָכֵן כְּדַאי לְהַנִּיחַ דַּף הוֹרָאוֹת בָּרוּר: אֵיפֹה נִמְצָאִים הַדְּבָרִים הַחֲשׁוּבִים, אֵיךְ מַדְלִיקִים אֶת הַבּוֹיילֶר, מָה יֵשׁ בַּסְּבִיבָה, וְעַל מָה צָרִיךְ לְהִזָּהֵר.

בַּתּוֹרָה כָּתוּב: ׳וְלֹא תָשִׂים דָּמִים בְּבֵיתֶךָ׳, וְזֶה חָל גַּם עַל סַכָּנָה שֶׁאֲנַחְנוּ מַכִּירִים וְהָאוֹרֵחַ לֹא.

מַסְבִּירִים לָאוֹרֵחַ מָה שֶׁהוּא לֹא יָכוֹל לָדַעַת, וְכָךְ שׁוֹמְרִים עָלָיו.

👑

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת כְּמוֹ ה׳ שֶׁנּוֹתֵן טוֹב לְכָל אֶחָד, בְּלִי הֶבְדֵּל 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אָבוֹת ג׳ י״ד · בְּרֵאשִׁית ל״ג י׳

רָמָה ב׳ – כָּבוֹד לָאוֹרֵחַ

👑 כָּבוֹד שֶׁלֹּא תָּלוּי בְּמִי שֶׁבָּא

הָאוֹרֵחַ הוּא צֶלֶם אֱלֹהִים. חֲכָמִים אָמְרוּ: ׳חָבִיב אָדָם שֶׁנִּבְרָא בְּצֶלֶם׳. מִי שֶׁמְּכַבֵּד אָדָם – מְכַבֵּד אֶת הַצֶּלֶם שֶׁעָלָיו.

יַעֲקֹב אָמַר לְעֵשָׂו ׳כִּרְאֹת פְּנֵי אֱלֹהִים׳, וְכָךְ גַּם אֲנַחְנוּ דּוֹאֲגִים לְכָל אוֹרֵחַ כְּאִלּוּ הוּא אָדָם חָשׁוּב מְאֹד.

הַכָּבוֹד אֵינוֹ תָּלוּי בְּמַה שֶּׁיֵּשׁ לָאוֹרֵחַ, וְגַם לֹא בְּמַה שֶּׁהוּא נוֹתֵן לָנוּ.

הַמְּאָרֵחַ שֶׁמְּכַבֵּד אֶת כֻּלָּם בְּאוֹתָהּ מִדָּה, הוֹלֵךְ בְּדַרְכּוֹ שֶׁל ה׳, שֶׁעוֹרֵךְ שֻׁלְחָן לִפְנֵי כָּל אֶחָד.

הַכָּבוֹד לָאוֹרֵחַ לֹא תָּלוּי בְּמַה שֶּׁיֵּשׁ לוֹ.

🧳

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁאֲנִי אוֹרֵחַ, אֲנִי מוֹסִיף בְּרָכָה וְלֹא מַכְבִּיד, כְּמוֹ שֶׁה׳ רוֹצֶה 💛

📖 הַכְנָסַת אוֹרְחִים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – כָּבוֹד לָאוֹרֵחַ

🧳 הָאוֹרֵחַ הַטּוֹב

אוֹרֵחַ טוֹב עוֹשֶׂה כַּמָּה דְּבָרִים. הוּא שׁוֹאֵל מֵרֹאשׁ מָה יֵשׁ בַּבַּיִת וּמָה אֵין, וְלֹא מַנִּיחַ שֶׁדָּבָר שֶׁלֹּא שָׁאַל עָלָיו קַיָּם.

אִם שָׁבַר מַשֶּׁהוּ בְּטָעוּת, הוּא מְסַפֵּר לְבַעַל הַבַּיִת וְלֹא נוֹתֵן לוֹ לְגַלּוֹת לְבַד. רֹב הַזְּמַן הַמְּאָרֵחַ יַעֲרִיךְ אֶת זֶה וְיִמְחַל.

הוּא מַשְׁאִיר נָקִי: מְסַדֵּר, מְקַפֵּל וּמְנַקֶּה. אִם נִשְׁאַר אֹכֶל אוֹ שְׁתִיָּה, הוּא מַשְׁאִיר לְבַעַל הַבַּיִת כְּמַתָּנָה קְטַנָּה. וְהוּא לֹא מַרְעִישׁ בִּשְׁעוֹת הַמְּנוּחָה, כְּדֵי לִשְׁמֹר עַל הַשְּׁכֵנִים.

אוֹרֵחַ טוֹב שׁוֹאֵל, מְעַדְכֵּן, מַשְׁאִיר נָקִי וְשׁוֹמֵר עַל הַשְּׁכֵנִים.

רָמָה ב׳ – כָּבוֹד לָאוֹרֵחַ

מָה לוֹקְחִים מִרָמָה ב׳?

אֶת הַכָּבוֹד לָאוֹרֵחַ אֵין מוֹדְדִים בְּכֶסֶף, אֶלָּא בַּלֵּב: נִקָּיוֹן וְיֹשֶׁר, דָּבָר קָטָן שֶׁחָשַׁבְנוּ עָלָיו, וְהַכָּרַת טוֹבָה. בָּעוֹלָם הַזֶּה כֻּלָּנוּ אוֹרְחִים אֵצֶל ה׳. מִי שֶׁמַּכִּיר בְּכָל טִרְחָה מִתְאַמֵּן לִהְיוֹת אוֹרֵחַ טוֹב בָּעוֹלָם, וְלֶאֱהֹב אֶת בַּעַל הַבַּיִת הַגָּדוֹל.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה הַהַפְתָּעָה הַקְּטַנָּה שֶׁאֲנִי יָכוֹל לְהָכִין הַשָּׁבוּעַ לְאוֹרֵחַ אוֹ לִקְרוֹב מִשְׁפָּחָה?
  • מָתַי הָיִיתִי אוֹרֵחַ, וּמָה יָכֹלְתִּי לוֹמַר אוֹ לַעֲשׂוֹת כְּדֵי לְהַכִּיר טוֹבָה לַמְּאָרֵחַ?
  • אֶבְחַר הַפְתָּעָה קְטַנָּה אַחַת, אֲבַצֵּעַ אוֹתָהּ בְּאֵרוּחַ הַקָּרוֹב, וְאֶרְשֹׁם מָה הִרְגַּשְׁתִּי וּמָה הִרְגִּישׁ הָאוֹרֵחַ.

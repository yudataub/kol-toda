לִפְנֵי שָׁנִים רַבּוֹת גָּר בִּכְפָר קָטָן אִישׁ עָנִי וּשְׁמוֹ יוֹסֵף. הוּא חַי בִּצְרִיף קָטָן, וְכָל הַשָּׁבוּעַ עָבַד קָשֶׁה מְאֹד.

אֶת הַכֶּסֶף שֶׁהִרְוִיחַ הוּא חָסַךְ לְצָרְכֵי שַׁבָּת. בִּימֵי הַשָּׁבוּעַ הוּא וּמִשְׁפַּחְתּוֹ אָכְלוּ בְּצִמְצוּם, כְּדֵי שֶׁבַּשַּׁבָּת יִהְיֶה הַכֹּל מְכֻבָּד.

יוֹסֵף הָיָה עָנִי, אֲבָל חָסַךְ כְּדֵי לְכַבֵּד אֶת הַשַּׁבָּת.

בְּעֶרֶב שַׁבָּת יוֹסֵף הָיָה לוֹקֵחַ אֶת כָּל הַכֶּסֶף שֶׁחָסַךְ וְקוֹנֶה בּוֹ מִצְרָכִים לַשַּׁבָּת. הַמּוֹכְרִים בַּשּׁוּק יָדְעוּ: אֶת הַטּוֹב בְּיוֹתֵר קוֹנֶה יוֹסֵף לִכְבוֹד שַׁבָּת!

וּכְשֶׁהִגִּיעָה הַשַּׁבָּת, הָיָה שֻׁלְחָנוֹ מָלֵא מִכָּל טוּב, וְלִבּוֹ מָלֵא שִׂמְחָה.

לָכֵן קָרְאוּ לוֹ כֻּלָּם: ׳יוֹסֵף מוֹקִיר שַׁבָּת׳.

קָרְאוּ לוֹ ׳מוֹקִיר שַׁבָּת׳ כִּי הוֹקִיר אֶת הַשַּׁבָּת.

יוֹם אֶחָד חָלַם חֲלוֹם מוּזָר, שֶׁהִבְהִיל אוֹתוֹ: ׳כָּל כַּסְפְּךָ וּרְכוּשְׁךָ יַעֲבֹר לְיוֹסֵף מוֹקִיר שַׁבָּת, כִּי מְכַבֵּד הוּא אֶת הַשַּׁבָּת׳.

בַּחֲלוֹם: רְכוּשׁוֹ שֶׁל הֶעָשִׁיר יַעֲבֹר לְיוֹסֵף מוֹקִיר שַׁבָּת.

👴 עֵצַת הַחֲכָמִים

הֶעָשִׁיר נִבְהַל. הוּא קָרָא לַחֲכָמִים וְסִפֵּר לָהֶם אֶת הַחֲלוֹם.

הַחֲכָמִים אָמְרוּ לוֹ: ׳כְּדַאי לְךָ לִמְכּוֹר אֶת כָּל מַה שֶׁיֵּשׁ לְךָ וְלִקְנוֹת אֶבֶן יְקָרָה. תּוּכַל לִשְׁמֹר עַל הָאֶבֶן, וְכָךְ לֹא תֹּאבַד אֶת כַּסְפְּךָ׳.

הוּא עָשָׂה כְּדִבְרֵיהֶם: מָכַר אֶת הַכֹּל, וְקָנָה בְּכָל כַּסְפּוֹ אֶבֶן יְקָרָה מְאֹד.

הֶעָשִׁיר מָכַר אֶת כָּל רְכוּשׁוֹ וְקָנָה אֶבֶן יְקָרָה.

יוֹם אֶחָד עָבַר עַל גֶּשֶׁר מֵעַל לַנָּהָר, שָׂמֵחַ וּמְרֻצֶּה, וְאָמַר לְעַצְמוֹ: ׳אִישׁ לֹא יִקַּח מִמֶּנִּי אֶת כַּסְפִּי. כָּל רְכוּשִׁי בְּתוֹךְ כּוֹבָעִי!׳

הֶעָשִׁיר הִרְגִּישׁ בָּטוּחַ, כִּי כָּל רְכוּשׁוֹ בַּכּוֹבַע.

🐟 הַדָּג הַגָּדוֹל

הַדַּיָּגִים הֶעֱלוּ אֶת הַדָּג הַגָּדוֹל בִּרְשָׁתָם וְשָׂמְחוּ מְאֹד. הָיָה עֶרֶב שַׁבָּת, וְהַשָּׁעָה כְּבָר מְאֻחֶרֶת.

הֵם רָצוּ לַשּׁוּק, אֲבָל הַדָּג הָיָה יָקָר, וְהַקּוֹנִים לֹא רָצוּ לִקְנוֹתוֹ.

עַד שֶׁאֶחָד הַמּוֹכְרִים יָעַץ: ׳לְכוּ אֶל יוֹסֵף מוֹקִיר שַׁבָּת. הוּא אֵינוֹ חוֹסֵךְ מִכַּסְפּוֹ לִכְבוֹד הַשַּׁבָּת!׳

אַף קוֹנֶה לֹא רָצָה אֶת הַדָּג, חוּץ מִיּוֹסֵף.

  • אֶכְתֹּב שֵׁם לְפִי הַמַּעֲשִׂים שֶׁלִּי הַיּוֹם (׳דָּנִי מְסַדֵּר…׳), אֶבְחַר מַעֲשֶׂה לַשַּׁבָּת, וְאֶרְשֹׁם אֵיךְ הִרְגַּשְׁתִּי.

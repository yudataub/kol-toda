כַּעַס שֶׁמַּשְׁכִּיחַ

רָמָה ג׳ – מָה עוֹשִׂים הַיּוֹם

לָמָּה בִּשְׁעַת כַּעַס מִסְתַּלְּקוֹת הַחָכְמָה וְהָאֱמוּנָה, מָה לָמְדוּ חֲכָמִים עַל מִי שֶׁשּׁוֹבֵר בַּחֲמָתוֹ, וְאֵיךְ מִתְרַחֲקִים מִן הַכַּעַס

📖 עֲנָוָה וְנַחַת · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
🪜

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹצֵר עַל הַמַּדְרֵגָה הַקְּטַנָּה, וְה׳ עוֹזֵר לִי 💛

📖 עֲנָוָה וְנַחַת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שַׁבָּת קה ב׳

רָמָה ג׳ – מָה עוֹשִׂים הַיּוֹם

🪜 מַדְרֵגוֹת הַכַּעַס

הַיּוֹם עוֹשִׂים כְּמוֹ בַּשִּׁעוּר: מְצַיְּרִים עַל דַּף מַדְרֵגוֹת. עַל כָּל מַדְרֵגָה כּוֹתְבִים צַעַד קָטָן שֶׁל כַּעַס, מֵהַהַתְחָלָה הַקְּטַנָּה וְעַד שְׁבִירַת דְּבָרִים.

אַחַר כָּךְ מְחַפְּשִׂים: עַל אֵיזוֹ מַדְרֵגָה אֲנִי רוֹצֶה לַעֲצֹר? הַגְּמָרָא לִמְּדָה שֶׁהַיֵּצֶר מוֹבִיל צַעַד אַחַר צַעַד, וְלָכֵן עוֹצְרִים כְּשֶׁהַצַּעַד עוֹדֶנּוּ קָטָן.

וְזוֹכְרִים: בִּשְׁעַת כַּעַס הַחָכְמָה מִסְתַּלֶּקֶת.

מְצַיְּרִים מַדְרֵגוֹת וְעוֹצְרִים עַל הַמַּדְרֵגָה הַקְּטַנָּה.

1
🧭

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְנַסֶּה הַיּוֹם לִשְׁתֹּק פַּעַם אַחַת בְּמָקוֹם לְהָשִׁיב 💛

📖 עֲנָוָה וְנַחַת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַמְבַּ״ם, הִלְכוֹת דֵּעוֹת ב׳ · אֹרֶךְ אַפַּיִם

רָמָה ג׳ – מָה עוֹשִׂים הַיּוֹם

🧭 מִתְאַמְּנִים לֹא לְהָשִׁיב

הָרַמְבַּ״ם לִמֵּד: הַצַּדִּיקִים ׳שׁוֹמְעִים חֶרְפָּתָם וְאֵינָם מֵשִׁיבִין׳, וְצָרִיךְ לְלַמֵּד אֶת עַצְמוֹ שֶׁלֹּא יִכְעַס.

אֶפְשָׁר לְהִתְאַמֵּן: כְּשֶׁמִּישֶׁהוּ אוֹמֵר לִי מַשֶּׁהוּ שֶׁמַּרְגִּיז, אֲנִי מְנַסֶּה לִשְׁמֹעַ וְלֹא לְהָשִׁיב.

שְׁתִיקָה כָּזוֹ שָׁוָה לְפִי ׳אֹרֶךְ אַפַּיִם׳ יוֹתֵר מִכָּל הַתַּעֲנִיּוֹת.

מִתְאַמְּנִים לִשְׁמֹעַ וְלֹא לְהָשִׁיב, וְהַשָּׂכָר גָּדוֹל.

2
😊

💛 אַהֲבַת הַשֵּׁם: הַיּוֹם אֲנִי מוֹדֶה לַה׳ עַל דָּבָר אֶחָד וּמֵאִיר פָּנִים לְחָבֵר 💛

📖 עֲנָוָה וְנַחַת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אוֹצָר שֶׁל מִדּוֹת

רָמָה ג׳ – מָה עוֹשִׂים הַיּוֹם

😊 מוֹדִים וּמְאִירִים פָּנִים

הַסִּימָן הַיָּפֶה שֶׁל מִדּוֹת מְתֻקָּנוֹת: אָדָם שֶׁמּוֹדֶה עַל כָּל דָּבָר וּמֵאִיר פָּנִים לִסְבִיבָתוֹ. אֵצֶל אָדָם כָּזֶה הַכַּעַס רָחוֹק.

אֶפְשָׁר לְהַתְחִיל הַיּוֹם: בַּבֹּקֶר אוֹמְרִים תּוֹדָה לַה׳ עַל דָּבָר אֶחָד, וְחוֹשְׁבִים עַל מִי שֶׁאֲנִי יָכוֹל לְהָאִיר לוֹ פָּנִים.

הַשְּׂמֵחִים וְהַמּוֹדִים נִשְׁאָרִים חֲכָמִים וּמְאֻמָּנִים.

תּוֹדָה לַה׳ וּפָנִים מְאִירוֹת – וְהַכַּעַס רָחוֹק.

3
💡

רָמָה ג׳ – מָה עוֹשִׂים הַיּוֹם

מָה לוֹקְחִים מִרָמָה ג׳?

הַיּוֹם אֲנִי יָכוֹל לַעֲצֹר עַל מַדְרֵגָה קְטַנָּה, לְהוֹדוֹת לַה׳ וּלְהָאִיר פָּנִים. כָּךְ אֲנִי שׁוֹמֵר עַל הַחָכְמָה, עַל הָאֱמוּנָה וְעַל הָאַהֲבָה שֶׁלִּי לַה׳.

אַל תְּבַהֵל בְּרוּחֲךָ לִכְעוֹס כִּי כַעַס בְּחֵיק כְּסִילִים יָנוּחַ

סוֹף רָמָה ג׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי הַשָּׁבוּעַ הִרְגַּשְׁתִּי שֶׁהַכַּעַס עוֹלֶה, וּמָה קָרָה לַמַּחֲשָׁבוֹת שֶׁלִּי בְּאוֹתוֹ רֶגַע?
  • מָהוּ הַצַּעַד הַקָּטָן הָרִאשׁוֹן שֶׁל הַכַּעַס אֶצְלִי, וְאֵיךְ אֶעֱצֹר אוֹתוֹ הַפַּעַם הַבָּאָה?
  • אֲצַיֵּר מַדְרֵגוֹת שֶׁל כַּעַס, אֶכְתֹּב עַל כָּל מַדְרֵגָה צַעַד, וְאֶסַמֵּן אֶת הַמַּדְרֵגָה שֶׁבָּהּ אֲנִי רוֹצֶה לַעֲצֹר.

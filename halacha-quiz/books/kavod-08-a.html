רָמָה א׳ – הַסִּפּוּר שֶׁל רַבִּי מֹשֶׁה

🏙️ שְׁאֵלָה בְּשַׁבָּת

בְּשַׁבָּת אַחַת בָּאוּ אֲנָשִׁים אֶל רַבִּי מֹשֶׁה קְלֵירְס, רַבָּהּ שֶׁל טְבֶרְיָה, וְאָמְרוּ: ׳הָעֵרוּב שֶׁל הָעִיר לֹא כַּשּׁוּרָה!׳ עֵרוּב הוּא סִדּוּר מְיֻחָד בָּעִיר, שֶׁעוֹזֵר לְטַלְטֵל בְּשַׁבָּת.

הָרַב בֵּרֵר וְגִלָּה: הָעֵרוּב הוּתְקַן בְּעֶרֶב שַׁבָּת עַל פִּי הוֹרָאָה שֶׁל הָרַב הַסְּפָרַדִּי שֶׁל הָעִיר, וְתַחַת פִּקּוּחוֹ.

מָה עָשָׂה רַבִּי מֹשֶׁה?

רַבִּי מֹשֶׁה שׁוֹמֵעַ תְּלוּנָה וּבוֹדֵק לְפִי הָעוּבְדּוֹת.

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, קֻנְטְרֵס הַכָּבוֹד

רָמָה א׳ – הַסִּפּוּר שֶׁל רַבִּי מֹשֶׁה

✋ פְּסַק קָצָר

רַבִּי מֹשֶׁה אָמַר לָאֲנָשִׁים: הָעֵרוּב כָּשֵׁר, אֲבָל אֵין לִי עַכְשָׁו זְמַן לְהַסְבִּיר לָכֶם אֶת הַטַּעַם.

הוּא רַק אָמַר שֶׁהָעֵרוּב כָּשֵׁר, וְלֹא הִתְחִיל לְדַבֵּר עַל הָרַב הַסְּפָרַדִּי.

בְּיוֹם רִאשׁוֹן הוּא הָלַךְ לְטַפֵּל בַּדָּבָר בְּשֶׁקֶט.

רַבִּי מֹשֶׁה אֵינוֹ מְדַבֵּר עַל הָרַב, וּמְטַפֵּל בַּדָּבָר בְּשֶׁקֶט.

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא מְנַצֵּחַ בְּוִכּוּחַ, אֲנִי מַזְמִין אֶת חֲבֵרִי לִלְמֹד יַחַד 💛

רָמָה א׳ – הַסִּפּוּר שֶׁל רַבִּי מֹשֶׁה

🚪 בְּיוֹם רִאשׁוֹן

רַבִּי מֹשֶׁה הָלַךְ אֶל בֵּיתוֹ שֶׁל הָרַב הַסְּפָרַדִּי וְאָמַר: ׳יֵשׁ לִי עִנְיָן קָשֶׁה בַּגְּמָרָא, וַאֲנִי צָרִיךְ אֶת עֶזְרָתְךָ׳.

הָרַב הַסְּפָרַדִּי חִיֵּךְ וְשָׁאַל: ׳הַאִם הָרַב מְשַׁטֶּה בִּי?׳

אֲבָל רַבִּי מֹשֶׁה אָמַר שֶׁקָּשֶׁה לוֹ לְבָרֵר אֶת הַדִּין לְבַד, וּבִקֵּשׁ שֶׁיֵּלְכוּ יַחַד לְבֵית כְּנֶסֶת לְבָרֵר אֶת הַדִּין.

רַבִּי מֹשֶׁה בָּא בְּשֶׁקֶט, וּמְבַקֵּשׁ עֶזְרָה בִּשְׁאֵלָה.

📖

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹהֵב לִלְמֹד עִם חֲבֵרִים, וְלִשְׂמֹחַ יַחַד בְּתוֹרָה 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, קֻנְטְרֵס הַכָּבוֹד

רָמָה א׳ – הַסִּפּוּר שֶׁל רַבִּי מֹשֶׁה

📖 לוֹמְדִים יַחַד

בְּבֵית הַכְּנֶסֶת הוֹצִיא רַבִּי מֹשֶׁה מֵהָאָרוֹן אֶת הַגְּמָרָא עֵרוּבִין, וְהֵם לָמְדוּ יַחַד אֶת הַסּוּגְיָה.

הֵם הִתְעַמְּקוּ בַּגְּמָרָא, בַּמְפָרְשִׁים הַקַּדְמוֹנִים וּבַפּוֹסְקִים.

רַבִּי מֹשֶׁה לֹא אָמַר לוֹ ׳טָעִיתָ׳. הוּא רַק לָמַד אִתּוֹ.

לוֹמְדִים יַחַד, וְלֹא אוֹמְרִים לַחָבֵר ׳טָעִיתָ׳.

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי טוֹעֶה, אֲנִי רוֹצֶה שֶׁיְּתַקְּנוּ אוֹתִי בְּעֵדִינוּת, וְכָךְ אֲנִי נוֹהֵג בַּאֲחֵרִים 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, קֻנְטְרֵס הַכָּבוֹד

רָמָה א׳ – הַסִּפּוּר שֶׁל רַבִּי מֹשֶׁה

😮 הוּא מְגַלֶּה בְּעַצְמוֹ

פִּתְאוֹם הָרַב הַסְּפָרַדִּי הֵבִין, וְקָרָא בְּקוֹל: ׳אוֹי לִי שֶׁטָּעִיתִי בְּעֶרֶב שַׁבָּת הָאַחֲרוֹן! שֶׁלֹּא כַּדִּין עָשִׂיתִי!׳

אַף אֶחָד לֹא אָמַר לוֹ שֶׁטָּעָה. הוּא עַצְמוֹ גִּלָּה אֶת הַטָּעוּת, מִתּוֹךְ הַלִּמּוּד.

כָּךְ אֶפְשָׁר לְתַקֵּן טָעוּת בְּלִי שֶׁמִּישֶׁהוּ יִתְבַּיֵּשׁ.

מִי שֶׁמְּגַלֶּה בְּעַצְמוֹ אֶת הַטָּעוּת, נִשְׁאָר בְּכָבוֹד.

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, קֻנְטְרֵס הַכָּבוֹד

רָמָה א׳ – הַסִּפּוּר שֶׁל רַבִּי מֹשֶׁה

🕊️ לָכֵן בָּאתִי

רַבִּי מֹשֶׁה הִרְגִּיעַ אוֹתוֹ: ׳לָכֵן בָּאתִי אֵלֶיךָ עַכְשָׁו. בְּשַׁבָּת פָּסַקְתִּי שֶׁהָעֵרוּב כָּשֵׁר, וְעַכְשָׁו בָּאתִי׳.

הָרַב הַסְּפָרַדִּי הִתְפַּלֵּא, וְאָז הִסְבִּיר לוֹ רַבִּי מֹשֶׁה אֶת הַטַּעַם.

מֵהַסִּפּוּר לוֹמְדִים: צָרִיךְ לְתַקֵּן אֶת הַטָּעוּת, אֲבָל בְּלִי לְבַיֵּשׁ אֶת מִי שֶׁטָּעָה.

מְתַקְּנִים טָעוּת, וְשׁוֹמְרִים עַל הַכָּבוֹד שֶׁל מִי שֶׁטָּעָה.

רָמָה א׳ – הַסִּפּוּר שֶׁל רַבִּי מֹשֶׁה

💎 כְּבוֹד הַתּוֹרָה וּכְבוֹד הַבְּרִיּוֹת

הַטַּעַם שֶׁהִסְבִּיר רַבִּי מֹשֶׁה: ׳מִצְוַת עֵרוּב הִיא מִדִּבְרֵי רַבָּנָן בָּעִיר שֶׁלָּנוּ, וְכְבוֹד הַתּוֹרָה וּכְבוֹד הַבְּרִיּוֹת הוּא מִדְּאוֹרַיְתָא׳.

זֹאת אוֹמֶרֶת: הַתּוֹרָה עַצְמָהּ מְצַוָּה לְכַבֵּד אֶת הָאֲנָשִׁים.

וְגַם רַב גָּדוֹל זָכָה לְכָבוֹד, כִּי לֹא הִכְלִימוּ אוֹתוֹ.

הַתּוֹרָה עַצְמָהּ מְצַוָּה לְכַבֵּד אֶת הַבְּרִיּוֹת.

חָבֵר כָּתַב בַּלּוּחַ תְּשׁוּבָה לֹא נְכוֹנָה. אֶפְשָׁר לִצְעֹק ׳טָעִיתָ!׳, וְאֶפְשָׁר לְהִתְקָרֵב בְּשֶׁקֶט.

כְּמוֹ רַבִּי מֹשֶׁה: אוֹמְרִים ׳בּוֹא נִבְדֹּק יַחַד׳, וְנוֹתְנִים לוֹ לְגַלּוֹת בְּעַצְמוֹ.

כָּךְ הַתִּקּוּן קוֹרֶה, וְהֶחָבֵר נִשְׁאָר בְּכָבוֹד.

אוֹמְרִים ׳בּוֹא נִבְדֹּק יַחַד׳ – וְלֹא ׳טָעִיתָ׳.

מָה לוֹקְחִים מִרָמָה א׳?

רַבִּי מֹשֶׁה לָמַד שֶׁגַּם כְּשֶׁצָּרִיךְ לְתַקֵּן, אֶפְשָׁר לַעֲשׂוֹת זֹאת בְּאַהֲבָה. מִי שֶׁשּׁוֹמֵר עַל כְּבוֹדָם שֶׁל בְּנֵי אָדָם, שׁוֹמֵר עַל מַה שֶׁה׳ בָּרָא, וְכָךְ גְּדֵלָה אַהֲבָתוֹ אֵלָיו.

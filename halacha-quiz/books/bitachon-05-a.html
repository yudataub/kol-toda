לָמָּה הַפַּחַד מַפִּיל, מָה מֻתָּר וּמָה אָסוּר לַעֲשׂוֹת עִם פַּחַד, וְאֵיךְ בִּטָּחוֹן בַּה׳ מַרְגִּיעַ

רָמָה א׳ – קֶרֶשׁ וּפַחַד

🪵 קֶרֶשׁ עַל הָרִצְפָּה

הַמַּהֲרַ״ל מַסְבִּיר דָּבָר מַפְתִּיעַ עַל פַּחַד. דִּמְיְנוּ קֶרֶשׁ עֵץ אָרֹךְ שֶׁמֻּנָּח עַל הָרִצְפָּה בֶּחָצֵר.

כִּמְעַט וַדַּאי שֶׁנַּעֲבֹר עָלָיו מִקָּצֶה לְקָצֶה בְּלִי שׁוּם בְּעָיָה.

הָרַגְלַיִם יוֹדְעוֹת מָה לַעֲשׂוֹת, וְהַכֹּל פָּשׁוּט.

עַל קֶרֶשׁ שֶׁעַל הָרִצְפָּה עוֹבְרִים בְּקַלּוּת.

1
🏢

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְבַקֵּשׁ מֵה׳ שֶׁיְּחַזֵּק אֶת הַלֵּב שֶׁלִּי 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מַהֲרַ״ל

רָמָה א׳ – קֶרֶשׁ וּפַחַד

🏢 אוֹתוֹ קֶרֶשׁ, אֲבָל גָּבוֹהַּ

עַכְשָׁו דִּמְיְנוּ אֶת אוֹתוֹ קֶרֶשׁ בְּדִיּוּק מֻנָּח בֵּין שְׁנֵי בִּנְיָנִים גְּבוֹהִים, הַרְבֵּה מֵעַל הָאֲדָמָה.

הַמַּהֲרַ״ל אוֹמֵר: יֵשׁ סִכּוּי גָּדוֹל שֶׁמִּי שֶׁיְּנַסֶּה לַעֲבֹר יִפֹּל בָּאֶמְצַע.

הַקֶּרֶשׁ אוֹתוֹ קֶרֶשׁ וְהָרַגְלַיִם אוֹתָן רַגְלַיִם. מָה הִשְׁתַּנָּה? רַק דָּבָר אֶחָד: הַפַּחַד.

(זֶה רַק דִּמְיוֹן – לֹא מְנַסִּים אֶת זֶה בַּבַּיִת!)

הַקֶּרֶשׁ אוֹתוֹ קֶרֶשׁ – הִשְׁתַּנָּה רַק הַפַּחַד.

2
😟

💛 אַהֲבַת הַשֵּׁם: אֲנִי שׁוֹמֵר עַל הַמַּחֲשָׁבוֹת שֶׁלִּי וּמְמַלֵּא אוֹתָן בְּבִטָּחוֹן בַּה׳ 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מַהֲרַ״ל · אִיּוֹב ג׳ כ״ה

רָמָה א׳ – קֶרֶשׁ וּפַחַד

😟 הַמַּחֲשָׁבָה שֶׁמַּפִּילָה

הַמַּהֲרַ״ל מַסְבִּיר: הַמַּחֲשָׁבָה עַל הַנְּפִילָה הִיא שֶׁפּוֹעֶלֶת בָּאָדָם וּמַפִּילָה אוֹתוֹ.

וְכָךְ אָמַר אִיּוֹב: ׳כִּי פַחַד פָּחַדְתִּי וַיֶּאֱתָיֵנִי׳. מִי שֶׁמִּתְיָרֵא מִדָּבָר מַקְטִין אֶת עַצְמוֹ לְפָנָיו וְנוֹתֵן לוֹ מָקוֹם לִפְעֹל.

לָכֵן חָשׁוּב מְאֹד מָה נִכְנָס לַמַּחֲשָׁבָה שֶׁלָּנוּ.

מַחֲשָׁבָה מַפְחִידָה גְּדֵלָה וְנוֹתֶנֶת לַפַּחַד כֹּחַ.

3
🌡️

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁאֲנִי מַרְגִּישׁ פַּחַד, אֲנִי יָכוֹל לְהִתְקָרֵב אֶל ה׳ 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַמְבַּ״ם, הִלְכוֹת מְלָכִים

רָמָה א׳ – קֶרֶשׁ וּפַחַד

🌡️ פַּחַד טִבְעִי – מֻתָּר

פַּחַד בָּא מֵעַצְמוֹ, וְאִי אֶפְשָׁר לְצַוּוֹת עַל רֶגֶשׁ. הָרַמְבַּ״ם כָּתַב עַל חַיָּל בַּקְרָב, לֹא עַל מִי שֶׁמַּרְגִּישׁ פַּחַד, אֶלָּא עַל מִי שֶׁמַּתְחִיל לַחְשֹׁב וּלְהַרְהֵר וּמַבְהִיל אֶת עַצְמוֹ.

לָכֵן פַּחַד טִבְעִי אֵינוֹ אָסוּר, וַאֲפִלּוּ יֵשׁ בּוֹ צַד טוֹב: הוּא גּוֹרֵם לָנוּ לְהָבִין שֶׁיֵּשׁ סַכָּנָה וְלִהְיוֹת זְהִירִים.

מָה אָסוּר? לְהַגְדִּיל אֶת הַפַּחַד בְּמַחֲשָׁבוֹת.

פַּחַד שֶׁבָּא מֵעַצְמוֹ מֻתָּר; לְהַגְדִּיל אוֹתוֹ בְּמַחֲשָׁבוֹת – לֹא.

4
🧮

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁחוֹזֶרֶת מַחֲשָׁבָה מַפְחִידָה, אֲנִי אוֹמֵר: יֵשׁ לִי אַבָּא טוֹב 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַמְבַּ״ם · רַבֵּנוּ יוֹנָה (דְּבָרִים כ׳ א׳)

רָמָה א׳ – קֶרֶשׁ וּפַחַד

🧮 ׳מָה אִם… וּמָה אִם…׳

הָרַמְבַּ״ם נוֹתֵן דֻּגְמָה: לוֹחֵם שֶׁחוֹשֵׁב בְּאֶמְצַע הַקְּרָב מָה יִהְיֶה עִם הַמִּשְׁפָּחָה שֶׁלּוֹ – נַעֲשֶׂה חַלָּשׁ.

רַבֵּנוּ יוֹנָה נוֹתֵן דֻּגְמָה אַחֶרֶת: מִי שֶׁמַּתְחִיל לַחְשֹׁב כַּמָּה יֵשׁ לָהֶם וְכַמָּה יֵשׁ לָנוּ, מָה הַיִּתְרוֹן שֶׁלָּהֶם וּמָה שֶׁלָּנוּ – הַחֶשְׁבּוֹנוֹת מַגְדִּילִים אֶת הַפַּחַד.

גַּם אֲנִי מַכִּיר אֶת זֶה: ׳מָה יִהְיֶה אִם…׳ שֶׁחוֹזֵר וְחוֹזֵר, וּבְדִיּוּק זֶה מַבְהִיל.

מַחֲשָׁבוֹת וְחֶשְׁבּוֹנוֹת שֶׁחוֹזְרוֹת וְחוֹזְרוֹת מַגְדִּילוֹת אֶת הַפַּחַד.

🗣️

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְסַפֵּר לַחֲבֵרִים דְּבָרִים שֶׁמְּחַזְּקִים אֶת הַלֵּב 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מַהֲרַ״ל

רָמָה א׳ – קֶרֶשׁ וּפַחַד

🗣️ לֹא מַפְחִידִים חֲבֵרִים

יֶלֶד מְסַפֵּר בַּהַפְסָקָה סִפּוּר אֵימָה, וְכָל הַחֲבֵרִים מִתְכַּנְּסִים וּמִתְפַּחֲדִים.

הַמַּהֲרַ״ל מַסְבִּיר שֶׁמִּי שֶׁמַּכְנִיס בַּאֲנָשִׁים פְּחָדִים אֵינוֹ רַק גּוֹרֵם לָהֶם צַעַר – הוּא עָלוּל לָתֵת לַדְּבָרִים מָקוֹם לִקְרוֹת. זֶה עִנְיָן שֶׁצָּרִיךְ לְשַׁקֵּל בְּכֹבֶד רֹאשׁ.

לָכֵן עָדִיף לֹא לְסַפֵּר סִפּוּרֵי אֵימָה, וְאִם מִישֶׁהוּ מַתְחִיל – אֶפְשָׁר לְהַצִּיעַ סִפּוּר אַחֵר שֶׁמְּחַזֵּק.

לֹא מַפְחִידִים חֲבֵרִים – מְחַזְּקִים אוֹתָם.

6
📗

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹטֵחַ בַּה׳ וְעוֹזֵר גַּם לַחֲבֵרִים לִבְטֹחַ 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הֶחָפֵץ חַיִּים, מַחֲנֵה יִשְׂרָאֵל · מִשְׁלֵי כ״ט כ״ה

רָמָה א׳ – קֶרֶשׁ וּפַחַד

📗 לְפִי גֹּדֶל הַבִּטָּחוֹן

הֶחָפֵץ חַיִּים כָּתַב סֵפֶר לְחַיָּלִים יְהוּדִים, ׳מַחֲנֵה יִשְׂרָאֵל׳. הוּא הִסְבִּיר: לְפִי גֹּדֶל הַבִּטָּחוֹן – כָּךְ גֹּדֶל הַהַצְלָחָה.

וּכְשֵׁם שֶׁמִּצְוָה לְהִתְחַזֵּק בְּעַצְמוֹ, צָרִיךְ לְחַזֵּק גַּם אֶת הַחֲבֵרִים בְּבִטָּחוֹן.

בְּמִשְׁלֵי כָּתוּב: ׳חֶרְדַּת אָדָם יִתֵּן מוֹקֵשׁ וּבוֹטֵחַ בַּה׳ יְשֻׂגָּב׳.

מִתְחַזְּקִים בְּבִטָּחוֹן, וּמְחַזְּקִים גַּם אֶת הַחֲבֵרִים.

🌤️

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי מַרְגִּישׁ פַּחַד, אֲנִי אוֹמֵר לַה׳: אֵלֶיךָ אֶבְטָח 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים נ״ו ד׳

רָמָה א׳ – קֶרֶשׁ וּפַחַד

🌤️ יוֹם אִירָא – אֵלֶיךָ אֶבְטָח

בִּתְהִלִּים כָּתוּב: ׳יוֹם אִירָא אֲנִי אֵלֶיךָ אֶבְטָח׳. כְּשֶׁבָּא פַּחַד, בּוֹטְחִים בַּה׳.

בִּטָּחוֹן פֵּרוּשׁוֹ לִהְיוֹת רָגוּעַ וְלֹא לְפַחֵד מָה יִהְיֶה, כִּי יֵשׁ לִי אַבָּא טוֹב שֶׁאוֹהֵב אוֹתִי.

וְהַפְּעֻלָּה לַשָּׁבוּעַ: כְּשֶׁמַּתְחִילִים סִפּוּר מַפְחִיד – מַצִּיעִים סִפּוּר מְחַזֵּק, וּכְשֶׁבָּאָה מַחֲשָׁבָה מַפְחִידָה – אוֹמְרִים בַּלֵּב ׳אֵלֶיךָ אֶבְטָח׳.

כְּשֶׁבָּא פַּחַד אוֹמְרִים: ׳אֵלֶיךָ אֶבְטָח׳.

רָמָה א׳ – קֶרֶשׁ וּפַחַד

מָה לוֹקְחִים מִרָמָה א׳?

פַּחַד הוּא רֶגֶשׁ טִבְעִי, אֲבָל אֲנִי יָכוֹל לִבְחֹר מָה לַעֲשׂוֹת בּוֹ: לֹא לְהַגְדִּיל אוֹתוֹ, אֶלָּא לוֹמַר ׳אֵלֶיךָ אֶבְטָח׳. כְּשֶׁאֲנִי זוֹכֵר שֶׁיֵּשׁ לִי אַבָּא טוֹב שֶׁאוֹהֵב אוֹתִי, הַלֵּב נִרְגָּע וְאַהֲבָתִי אֵלָיו גְּדֵלָה.

  • מָה מַפְחִיד אוֹתִי הַשָּׁבוּעַ, וּמָה אֲנִי אוֹמֵר לְעַצְמִי כְּשֶׁהַמַּחֲשָׁבָה הַזֹּאת חוֹזֶרֶת?
  • אֵיזֶה סִפּוּר מְחַזֵּק אֲסַפֵּר לַחֲבֵרִים בַּפַּעַם הַבָּאָה שֶׁמִּישֶׁהוּ מַתְחִיל סִפּוּר מַפְחִיד?
  • אֶרְשֹׁם הַשָּׁבוּעַ כָּל פַּעַם שֶׁאָמַרְתִּי בַּלֵּב ׳אֵלֶיךָ אֶבְטָח׳, וְאֶרְשֹׁם מָה קָרָה לַלֵּב שֶׁלִּי אַחֲרֵי כֵן.

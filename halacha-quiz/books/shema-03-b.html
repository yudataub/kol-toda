הַמִּשְׁנָה בְּמַסֶּכֶת בְּרָכוֹת: ׳בְּכָל לְבָבְךָ – בִּשְׁנֵי יְצָרֶיךָ, בְּיֵצֶר טוֹב וּבְיֵצֶר רָע. וּבְכָל נַפְשְׁךָ – אֲפִלּוּ הוּא נוֹטֵל אֶת נַפְשְׁךָ. וּבְכָל מְאֹדֶךָ – בְּכָל מָמוֹנְךָ׳.

שְׁלוֹשָׁה ׳בְּכָל׳ – שְׁלוֹשָׁה אָבוֹת, וְכָל אֶחָד מֵהֶם מְלַמֵּד אוֹתָנוּ דֶּרֶךְ שֶׁל אַהֲבָה.

🔥 אַהֲבָה שֶׁגּוֹבֶרֶת עַל כָּל הַחֲשָׁקִים

אֵיךְ אוֹהֲבִים אֶת ה׳ בַּיֵּצֶר הָרָע? הָרַשְׁבָּ״א מְבָאֵר: הַיֵּצֶר הָרָע מוֹשֵׁךְ אוֹתָנוּ אַחֲרֵי ׳חֶמְדַּת הַלֵּב׳ – מָה שֶׁאֲנַחְנוּ חוֹשְׁקִים בּוֹ. ה׳ מְבַקֵּשׁ שֶׁהַחֵשֶׁק לַעֲבֹד אוֹתוֹ יִהְיֶה גָּדוֹל מִכָּל הַחֲשָׁקִים, כְּמוֹ שֶׁכָּתוּב: ׳אִם יִתֵּן אִישׁ אֶת כָּל הוֹן בֵּיתוֹ בָּאַהֲבָה, בּוֹז יָבוּזוּ לוֹ׳.

כָּךְ אַבְרָהָם: חִכָּה כָּל חַיָּיו לְבֵן, הָיָה אִישׁ שֶׁל חֶסֶד, וְאָהַב אֶת יִצְחָק מְאוֹד – וְאַהֲבַת ה׳ גָּבְרָה עַל הַכֹּל. הַנִּסָּיוֹן שֶׁלּוֹ הָיָה בָּרָצוֹן, לֹא רַק בַּמַּעֲשֶׂה, וְלָכֵן נִקְרָא ׳זֶרַע אַבְרָהָם אֹהֲבִי׳.

אַבְרָהָם רָצָה אֶת רְצוֹן ה׳ יוֹתֵר מִכָּל מָה שֶׁאָהַב.

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

🪨 בְּכָל נַפְשְׁךָ – מִן הַקָּטָן וְעַד הַגָּדוֹל

׳בְּכָל נַפְשְׁךָ׳ – הָרַשְׁבָּ״א מְבָאֵר שֶׁזֶּה כּוֹלֵל כָּל צַעַר וְכָל נֵזֶק שֶׁבָּאִים בִּגְלַל מִצְוָה, וְעַד מְסִירוּת נֶפֶשׁ. חֲזַ״ל מְסַפְּרִים עַל יְהוּדִים שֶׁסָּבְלוּ קָשׁוֹת מִן הַשּׂוֹנְאִים, רַק מִפְּנֵי שֶׁמָּלוּ אֶת בְּנֵיהֶם אוֹ נָטְלוּ לוּלָב.

זוֹ מִדָּתוֹ שֶׁל יִצְחָק אָבִינוּ, שֶׁהִסְכִּים לַעֲלוֹת עַל הַמִּזְבֵּחַ, כִּי הֵבִין שֶׁה׳ רוֹצֶה בְּכָךְ.

וְלָכֵן גַּם צַעַר קָטָן בִּשְׁבִיל ה׳ – לָקוּם עָיֵף כְּדֵי לַעֲזֹר, לָלֶכֶת בַּקֹּר לִתְפִלָּה – הוּא חֵלֶק מִ׳בְּכָל נַפְשְׁךָ׳.

בְּכָל נַפְשְׁךָ: כָּל צַעַר בִּשְׁבִיל ה׳, מִן הַקָּטָן וְעַד הַגָּדוֹל.

📖 פָּרָשַׁת שְׁמַע – דְּבָרִים ו׳, פֶּרֶק ו׳, פָּסוּק ה׳ (הָרַשְׁבָּ״א)

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

⛺ בְּכָל מְאֹדֶךָ – יַעֲקֹב וּמְעָרַת הַמַּכְפֵּלָה

׳בְּכָל מְאֹדֶךָ׳ – לִהְיוֹת מוּכָן לְוַתֵּר עַל כָּל הַמָּמוֹן כְּשֶׁצָּרִיךְ בִּשְׁבִיל עֲבוֹדַת ה׳. זוֹ מִדָּתוֹ שֶׁל יַעֲקֹב אָבִינוּ: הוּא עָזַב אֶת הָעֹשֶׁר שֶׁל בֵּית אָבִיו וּבָחַר לִהְיוֹת ׳יֹשֵׁב אֹהָלִים׳.

הוּא הִבְטִיחַ: ׳וְכֹל אֲשֶׁר תִּתֶּן לִי עַשֵּׂר אֲעַשְּׂרֶנּוּ לָךְ׳.

וְהַמִּדְרָשׁ מְסַפֵּר שֶׁנָּתַן אֶת כָּל הָרְכוּשׁ שֶׁהֵבִיא מִחוּץ לָאָרֶץ כְּדֵי לִקְנוֹת אֶת מְעָרַת הַמַּכְפֵּלָה. לָכֵן אָמַר ׳בְּקִבְרִי אֲשֶׁר כָּרִיתִי לִי׳ – מִלְּשׁוֹן ׳כְּרִי׳, עֲרֵמָה שֶׁל כֶּסֶף.

בְּכָל מְאֹדֶךָ: הַכֶּסֶף מְשָׁרֵת אֶת הָאַהֲבָה לְה׳, וְלֹא לְהֶפֶךְ.

📖 פָּרָשַׁת שְׁמַע – דְּבָרִים ו׳, פֶּרֶק ו׳, פָּסוּק ה׳ (רַבִּי עֲקִיבָא)

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

✍️ ״רַבֵּנוּ, עַד כָּאן?״ – הַהֲכָנָה וְהַשְּׁטָר

הַתַּלְמִידִים יָדְעוּ אֶת הַדִּין שֶׁל ׳בְּכָל נַפְשְׁךָ׳. הֵם שָׁאֲלוּ: אֵיךְ מַגִּיעִים לְמַדְרֵגָה כָּזוֹ, שֶׁגַּם בְּרֶגַע קָשֶׁה כָּל כָּךְ מְקַבְּלִים עֹל מַלְכוּת שָׁמַיִם בְּאַהֲבָה?

רַבִּי עֲקִיבָא עָנָה: ׳כָּל יָמַי הָיִיתִי מִצְטַעֵר׳ – זֶה לֹא בָּא בְּרֶגַע אֶחָד. הִתְכּוֹנַנְתִּי בְּכָל יוֹם, וְלָכֵן עַכְשָׁו זֶה לֹא קָשֶׁה לִי.

׳וַאֲקַיְּמֶנּוּ׳ – כְּמוֹ קִיּוּם שְׁטָר: כָּל יוֹם, בִּקְרִיאַת שְׁמַע, ׳חָתַם׳ רַבִּי עֲקִיבָא שֶׁהוּא מוּכָן, וְחִכָּה לָתֵת לַחֲתִימָה תֹּקֶף אֲמִתִּי.

אֶת הַכֹּחַ לַעֲמֹד בְּנִסָּיוֹן קוֹנִים בַּהֲכָנָה – יוֹם אַחֲרֵי יוֹם.

📖 פָּרָשַׁת שְׁמַע – דְּבָרִים ו׳, פֶּרֶק ו׳, פָּסוּק ה׳ (מִשְׁנָה בְּרוּרָה, חַיֵּי אָדָם)

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

🕯️ לָמָּה חִכָּה – וּמָה מְכַוְּנִים הַיּוֹם

לָמָּה רָצָה רַבִּי עֲקִיבָא בָּרֶגַע הַזֶּה, וְלֹא רַק לְהַמְשִׁיךְ לְלַמֵּד? כִּי הָאָדָם בָּא לָעוֹלָם לְהַכִּיר אֶת רַחֲמֵי ה׳. הַמַּדְרֵגָה הַגְּבוֹהָה בְּיוֹתֵר הִיא לְהַאֲמִין שֶׁה׳ טוֹב דַּוְקָא כְּשֶׁנִּרְאֶה שֶׁיֵּשׁ דִּין.

הַמִּשְׁנָה בְּרוּרָה: קוֹרְאִים שְׁמַע בְּאֵימָה וּבְיִרְאָה, וּמְכַוְּנִים לִהְיוֹת מוּכָנִים לִמְסֹר אֶת הַנֶּפֶשׁ עַל קִדּוּשׁ ה׳. הַחַיֵּי אָדָם: מְכַוְּנִים לְכָךְ בַּמִּלָּה ׳אֶחָד׳, וּמִי שֶׁלֹּא הִסְפִּיק – לְפָחוֹת בַּמִּלִּים ׳בְּכָל נַפְשְׁךָ׳.

הָרַשְׁבָּ״א: ׳כִּי עָלֶיךָ הֹרַגְנוּ כָל הַיּוֹם׳ – אִי אֶפְשָׁר לָמוּת כָּל יוֹם! אֶלָּא מִי שֶׁמַּסְכִּים בַּלֵּב, נֶחְשָׁב לוֹ כְּאִלּוּ עָשָׂה.

מִי שֶׁמַּסְכִּים בְּלֵב שָׁלֵם בִּקְרִיאַת שְׁמַע – נֶחְשָׁב לוֹ כְּאִלּוּ עָשָׂה.

מָה לוֹקְחִים מִרָמָה ב׳?

רַבִּי עֲקִיבָא לֹא נוֹלַד גִּבּוֹר – הוּא הִתְכּוֹנֵן כָּל יוֹם, בְּכָל ׳שְׁמַע יִשְׂרָאֵל׳. גַּם אֲנַחְנוּ, בְּכָל פַּעַם שֶׁאֲנַחְנוּ אוֹמְרִים ׳בְּכָל לְבָבְךָ וּבְכָל נַפְשְׁךָ וּבְכָל מְאֹדֶךָ׳, חוֹתְמִים לְה׳ עַל אַהֲבָה. וְה׳, שֶׁאוֹהֵב אוֹתָנוּ כְּמוֹ אָב, שׁוֹמֵר כָּל חֲתִימָה כָּזוֹ וְשָׂמֵחַ בָּהּ.

  • כְּשֶׁאֲנִי אוֹמֵר ׳בְּכָל נַפְשְׁךָ׳ בִּקְרִיאַת שְׁמַע, עַל מָה אֲנִי מַסְכִּים בַּלֵּב?
  • מָה תִּהְיֶה ׳הָעֲקֵדָה שֶׁלִּי׳ הַשָּׁבוּעַ – מָתַי אֶקּוּם בְּשִׂמְחָה לַעֲשׂוֹת רְצוֹן ה׳, גַּם כְּשֶׁקָּשֶׁה לִי?

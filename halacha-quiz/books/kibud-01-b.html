רָמָה ב׳ – בֵּית סֵפֶר לְהַכָּרַת הַטּוֹב

🔑 הַמִּצְוָה הַחֲשׁוּבָה

הַמִּשְׁנָה בְּקִדּוּשִׁין אוֹמֶרֶת: ׳כָּל הָעוֹשֶׂה מִצְוָה אַחַת – מֵטִיבִין לוֹ וּמַאֲרִיכִין לוֹ יָמָיו׳.

הַיְרוּשַׁלְמִי שׁוֹאֵל: אֵיזוֹ מִצְוָה? וְעוֹנֶה: כִּבּוּד אָב וָאֵם, שֶׁהִיא מִצְוָה חֲשׁוּבָה בְּיוֹתֵר.

אָדָם יָכוֹל לִלְמֹד הַרְבֵּה תּוֹרָה וְלַעֲשׂוֹת הַרְבֵּה חֶסֶד, וּבְכָל זֹאת הַתּוֹרָה תָּלְתָה אֶת הַשָּׂכָר בְּכִבּוּד הוֹרִים.

הַיְרוּשַׁלְמִי: הַמִּצְוָה שֶׁמְּאַרִיכָה יָמִים הִיא כִּבּוּד אָב וָאֵם.

🙃 הָעוֹלָם נִבְרָא הָפוּךְ

בְּדֶרֶךְ כְּלָל חוֹשְׁבִים: הַהוֹרִים נָתְנוּ לִי הַרְבֵּה, לָכֵן צָרִיךְ לְכַבֵּד אוֹתָם. הַשִּׁעוּר מְלַמֵּד גַּם צַד הָפוּךְ.

ה׳ הִסְתַּכֵּל בַּתּוֹרָה וְרָאָה אֵילוּ מִדּוֹת טוֹבוֹת הָאָדָם צָרִיךְ לִקְנוֹת. אָז בָּרָא עוֹלָם שֶׁיֵּשׁ בּוֹ הִזְדַּמְּנוּת לִקְנוֹת אוֹתָן.

אַחַת הַמִּדּוֹת הִיא הַכָּרַת הַטּוֹב, וְהַהוֹרִים הֵם הַמָּקוֹם הֶחָשׁוּב לְהִתְאַמֵּן בָּהּ. כִּבּוּד הוֹרִים הוּא לֹא רַק ׳כִּי יֵשׁ לִי הוֹרִים׳, אֶלָּא כִּי אֲנִי צָרִיךְ לִבְנוֹת בַּלֵּב אֶת הַמִּדָּה.

ה׳ נָתַן לָנוּ הוֹרִים כְּדֵי שֶׁנִּבְנֶה בַּלֵּב אֶת מִדַּת הַכָּרַת הַטּוֹב.

רָמָה ב׳ – בֵּית סֵפֶר לְהַכָּרַת הַטּוֹב

🎶 שִׁירָה שֶׁמַּעֲלָה

מְסַפְּרִים בַּסְּפָרִים שֶׁכְּשֶׁיִּתֵּן לָנוּ ה׳ שָׂכָר עַל הַמִּצְוֹת, נַכִּיר לוֹ טוֹבָה, נוֹדֶה וְנָשִׁיר, וְכָךְ נַעֲלֶה לְמַדְרֵגָה גְּבוֹהָה יוֹתֵר.

דֻּגְמָה: יְהוֹשֻׁעַ אָמַר לַשֶּׁמֶשׁ ׳שֶׁמֶשׁ בְּגִבְעוֹן דֹּם׳. רש״י מְפָרֵשׁ: ׳דֹּם׳ מִלּוֹמַר שִׁירָה, כִּי הַשֶּׁמֶשׁ הוֹלֶכֶת בִּזְכוּת הַשִּׁירָה לַה׳.

מִי שֶׁלֹּא לָמַד הַכָּרַת הַטּוֹב, קָשֶׁה לוֹ לָשִׁיר וּלְהוֹדוֹת.

הַכָּרַת הַטּוֹב מְאַפְשֶׁרֶת הוֹדָיָה וְשִׁירָה, וְהַשִּׁירָה מַעֲלָה אוֹתָנוּ.

🧩 גַּם כְּשֶׁנּוֹתְנִים מִטִּבְעָם

אוּלַי יֹאמַר יֶלֶד: ׳הַהוֹרִים עָשׂוּ בִּשְׁבִילָם, כִּי הֵם רָצוּ יְלָדִים׳. הַשִּׁעוּר עוֹנֶה: הַהוֹרִים נוֹתְנִים מִטִּבְעָם, בְּדִיּוּק כְּמוֹ שֶׁה׳ נוֹתֵן מִטּוּב לִבּוֹ.

מִי שֶׁלּוֹמֵד לְהוֹדוֹת גַּם לְמִי שֶׁנּוֹתֵן מִטִּבְעוֹ, יוּכַל לְהוֹדוֹת לַה׳. מִי שֶׁלֹּא, יִשְׁאַל: ׳לָמָּה לְהוֹדוֹת? הֲרֵי טוֹב לוֹ לָתֵת׳, וְיִתְקַשֶּׁה לְהִתְקָרֵב אֵלָיו.

מוֹדִים גַּם לְמִי שֶׁנּוֹתֵן מִטִּבְעוֹ, וְכָךְ לוֹמְדִים לְהוֹדוֹת לַה׳.

🚰 צִנּוֹרוֹת הַשֶּׁפַע

כָּל אָדָם מְקַבֵּל טוֹבָה דֶּרֶךְ צִנּוֹרוֹת, וְהַצִּנּוֹרוֹת הָרִאשׁוֹנִים הֵם הַהוֹרִים. מִי שֶׁמּוֹדֶה לָהֶם, הַצִּנּוֹר אֶצְלוֹ פָּתוּחַ וְהַטּוֹב זוֹרֵם אֵלָיו.

הָרַב שְׁטֵיינְמָן אָמַר שֶׁגְּדוֹלֵי הַדּוֹרוֹת שֶׁלְּפָנֵינוּ זָכוּ לִגְדֻלָּתָם גַּם בִּזְכוּת כִּבּוּד אָב וָאֵם.

הַהוֹרִים הֵם הַצִּנּוֹר הָרִאשׁוֹן שֶׁדַּרְכּוֹ מַגִּיעָה אֵלֵינוּ הַטּוֹבָה.

⏳ אֵין בִּזְבּוּז זְמַן

לִפְעָמִים מִתְחַשֵּׁק לִגְמֹר מַהֵר, כִּי הַזְּמַן יָקָר. אֲבָל הַתּוֹרָה כָּתְבָה ׳לְמַעַן יַאֲרִכוּן יָמֶיךָ׳.

מַה שֶׁמַּשְׁקִיעִים בְּכִבּוּד הוֹרִים חוֹזֵר בְּאֹרֶךְ יָמִים, וְלָכֵן אֵין לַחֲשֹׁשׁ שֶׁנַּפְסִיד זְמַן.

וּכְשֶׁלֹּא עוֹזְרִים, הַהוֹרִים לֹא מַפְסִידִים. מִי שֶׁמַּפְסִיד הוּא אֲנִי, כִּי פִּסְפַּסְתִּי הִזְדַּמְּנוּת לִקְנוֹת תּוֹדָה.

הַזְּמַן שֶׁמַּשְׁקִיעִים בְּכִבּוּד הוֹרִים חוֹזֵר בְּאֹרֶךְ יָמִים.

מָה לוֹקְחִים מִרָמָה ב׳?

הַהוֹרִים הֵם הַצִּנּוֹר הָרִאשׁוֹן וּבֵית הַסֵּפֶר הָרִאשׁוֹן לְהַכָּרַת הַטּוֹב. מִי שֶׁבּוֹנֶה אֶת הַמִּדָּה הַזֹּאת בַּלֵּב, יוֹדֵעַ לְהוֹדוֹת גַּם לַה׳ וְלֶאֱהֹב אוֹתוֹ.

פָּנָיו שֶׁל אָדָם הֵם רְשׁוּת הָרַבִּים, וְהַלֵּב הוּא רְשׁוּת הַיָּחִיד. אֵיךְ שׁוֹמְרִים עַל פָּנִים שֶׁלֹּא מַזִּיקוֹת לְאַף אֶחָד?

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד לְהַפְרִיד: בַּלֵּב אֲנִי חוֹשֵׁב, וְעַל הַפָּנִים אֲנִי נוֹתֵן שָׁלוֹם 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הַסַּבָּא מִסְּלַבּוֹדְקָא · מַסֶּכֶת בָּבָא קַמָּא

רָמָה ב׳ – בּוֹר, חֲשָׁד וּזְכֻיּוֹת

🏘️ הַפָּנִים וְהַלֵּב

הַסַּבָּא מִסְּלַבּוֹדְקָא לָמַד מִדִּינֵי רְשׁוּת הַיָּחִיד וּרְשׁוּת הָרַבִּים תּוֹרָה שֶׁבֵּין אָדָם לַחֲבֵרוֹ.

אֵין אָדָם עוֹשֶׂה בִּרְשׁוּת הָרַבִּים מַעֲשִׂים שֶׁיֵּשׁ בָּהֶם תַּקָּלָה לַחֲבֵרוֹ. לָכֵן כָּל אָדָם כּוֹבֵשׁ אֶת יִצְרוֹ וּמַסְבִּיר לַחֲבֵרוֹ פָּנִים יָפוֹת.

אֲבָל בַּלֵּב, בִּרְשׁוּת הַיָּחִיד, יֵשׁ לוֹ רְשׁוּת, וְשָׁם הוּא עוֹשֶׂה אֶת חֶשְׁבּוֹן הַנֶּפֶשׁ שֶׁלּוֹ.

אֵין כָּאן סְתִירָה לַכֵּנוּת: הַלֵּב נִשְׁאָר אֲמִתִּי, וְהַפָּנִים מְכַבְּדוֹת אֶת הָרַבִּים.

הַלֵּב לְחֶשְׁבּוֹן נֶפֶשׁ, וְהַפָּנִים לְשָׁלוֹם עִם הָרַבִּים.

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הַסַּבָּא מִסְּלַבּוֹדְקָא · הָרַב פִּינְקֶל

רָמָה ב׳ – בּוֹר, חֲשָׁד וּזְכֻיּוֹת

🕳️ מַה קּוֹרֶה כְּשֶׁאֵין סֵבֶר פָּנִים

בִּסְלַבּוֹדְקָא אָמְרוּ שֶׁפָּנִים זְעוּפוֹת הֵן ׳בּוֹר בִּרְשׁוּת הָרַבִּים׳: כְּמוֹ שֶׁאָסוּר לַחְפֹּר בּוֹר בָּרְחוֹב, כָּךְ אָסוּר לְהִסְתּוֹבֵב עִם פָּנִים זְעוּפוֹת, וּמִי שֶׁנִּתְקָל בָּהֶן ׳נוֹפֵל׳.

הָרַב פִּינְקֶל מְתָאֵר: חוֹסֶר סֵבֶר פָּנִים יָפוֹת גּוֹרֵם לְמַחֲלֹקֶת, לְמֶתַח וּלְכַעַס, ׳וְהָעַצְבָּנִי הַזֶּה לֹא מֵבִין מָה הוּא עָשָׂה לַשֵּׁנִי׳.

פַּעַם בָּחוּר שֶׁרָצָה לִלְמֹד פָּגַשׁ קֹר וַאֲדִישׁוּת, וְזֶה הָיָה הָרֶגַע שֶׁבִּגְלָלוֹ לֹא חָזַר לְבֵית הַמִּדְרָשׁ.

פָּנִים קָרוֹת גּוֹרְמוֹת מַחֲלֹקֶת, מֶתַח וְכַעַס.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁהַפָּנִים שֶׁלִּי לֹא יַפְחִידוּ אַף אֶחָד 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, נְחֶמְיָה ב׳ ב׳ · רַשִׁ״י

רָמָה ב׳ – בּוֹר, חֲשָׁד וּזְכֻיּוֹת

👑 נְחֶמְיָה: הַסַּכָּנָה שֶׁבַּפָּנִים

נְחֶמְיָה הִגִּישׁ כּוֹס יַיִן לַמֶּלֶךְ, וְהַמֶּלֶךְ שָׁאַל: ׳מַדּוּעַ פָּנֶיךָ רָעִים וְאַתָּה אֵינְךָ חוֹלֶה? אֵין זֶה כִּי אִם רֹעַ לֵב׳.

רַשִׁ״י מְפָרֵשׁ שֶׁהַמֶּלֶךְ חָשַׁשׁ בִּגְלַל פָּנָיו הָרָעִים שֶׁנְּחֶמְיָה רוֹצֶה לְהַזִּיק לוֹ.

נְחֶמְיָה פָּחַד מֵהַחֲשָׁד, וְהִסְבִּיר שֶׁפָּנָיו רָעִים בִּגְלַל יְרוּשָׁלַיִם הֶחָרֵבָה.

לָמַדְנוּ כַּמָּה מְסֻכָּנוֹת פָּנִים רָעוֹת: הֵן מְעוֹרְרוֹת חֲשָׁד וּפַחַד, גַּם כְּשֶׁאֵין בַּלֵּב שׁוּם רָעָה.

פָּנִים רָעוֹת מְעוֹרְרוֹת חֲשָׁד גַּם כְּשֶׁהַלֵּב טוֹב.

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הַחֲזוֹן אִישׁ

רָמָה ב׳ – בּוֹר, חֲשָׁד וּזְכֻיּוֹת

😔 פָּנִים רְצִינִיּוֹת וַעֲצוּבוֹת

מוּבָא בְּשֵׁם הַחֲזוֹן אִישׁ שֶׁאָדָם שֶׁפָּנָיו רְצִינִיּוֹת וַעֲצוּבוֹת כָּל הַזְּמַן, מַפְחִיד בְּנֵי אָדָם וּמְצַעֵר אֶת הַבְּרִיּוֹת.

מִי שֶׁעָצוּב אֵינוֹ חַיָּב לִהְיוֹת שָׂמֵחַ בַּלֵּב, אֲבָל הַפָּנִים שַׁיָּכוֹת לַאֲחֵרִים.

לָכֵן אֲפִלּוּ אָדָם רְצִינִי צָרִיךְ לְהַרְאוֹת לְמִי שֶׁפּוֹגֵשׁ אוֹתוֹ פָּנִים נְעִימוֹת.

פָּנִים עֲצוּבוֹת כָּל הַזְּמַן מְצַעֲרוֹת אֶת מִי שֶׁסְּבִיבֵנוּ.

💛 אַהֲבַת הַשֵּׁם: כָּל חִיּוּךְ שֶׁלִּי הוּא זְכוּת, וַאֲנִי אוֹסֵף אוֹתָן בְּשִׂמְחָה 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הַסַּבָּא מִסְּלַבּוֹדְקָא

🧳 זְכֻיּוֹת בְּחִיּוּךְ

נֶכֶד שֶׁל הַסַּבָּא מִסְּלַבּוֹדְקָא בָּא לְבַקֵּשׁ בְּרָכָה לִפְנֵי דֶּרֶךְ מְסֻכֶּנֶת. הַסַּבָּא שָׁאַל עִם מִי הוּא הוֹלֵךְ, וְכַאֲשֶׁר שָׁמַע, אָמַר: ׳אַל תֵּלֵךְ עִמּוֹ. פָּנָיו זְעוּפוֹת, וַאֲנִי מִתְיָרֵא שֶׁחֲסֵרוֹת לוֹ זְכֻיּוֹת׳.

מִכָּאן לוֹמְדִים כַּמָּה זְכֻיּוֹת אֶפְשָׁר לִקְנוֹת בְּחִיּוּךְ נֶחְמָד לֶחָבֵר וּבַהַרְגָּשָׁה טוֹבָה שֶׁנּוֹתְנִים לַשֵּׁנִי, וְכַמָּה זְכֻיּוֹת מַפְסִידִים בְּפָנִים זְעוּפוֹת.

חִיּוּךְ קוֹנֶה זְכֻיּוֹת, וּפָנִים זְעוּפוֹת מַפְסִידוֹת אוֹתָן.

מָה לוֹקְחִים מִרָמָה ב׳?

מִי שֶׁשּׁוֹמֵר עַל פָּנִים נְעִימוֹת לָרַבִּים, וּבַלֵּב עוֹשֶׂה חֶשְׁבּוֹן נֶפֶשׁ, חַי בֶּאֱמֶת וּבְטוֹבָה. כָּךְ הוּא מִתְקָרֵב אֶל ה׳, שֶׁאוֹהֵב אֶת כָּל בְּרִיָּה.

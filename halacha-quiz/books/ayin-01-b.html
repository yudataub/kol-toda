הַשְּׁאֵלָה הָיְתָה עַל הַמִּדָּה הֲכִי חֲשׁוּבָה, זוֹ שֶׁאִם הָאָדָם נִדְבָּק בָּהּ, הוּא זוֹכֶה לְכָל הַמִּדּוֹת הַטּוֹבוֹת.

כְּלוֹמַר, עַיִן טוֹבָה הִיא הָרֹאשׁ שֶׁל הַדֶּרֶךְ: מִי שֶׁרוֹאֶה טוֹב, מִתְחַבֵּר אֶל הַטּוֹב וְאֶל הַמִּדּוֹת הַטּוֹבוֹת.

עַיִן טוֹבָה הִיא הַמִּדָּה שֶׁמּוֹבִילָה לְכָל הַמִּדּוֹת הַטּוֹבוֹת.

📖 לָמָּה סִפֵּר הֶחָסִיד עַל הַשִּׁנַּיִם?

בְּ׳חוֹבוֹת הַלְּבָבוֹת׳ כָּתוּב שֶׁהֶחָסִיד רָצָה לְהוֹכִיחַ אֶת תַּלְמִידָיו עַל כָּךְ שֶׁסִּפְּרוּ בִּגְנוּת הַנִּבְלָה.

הוּא הִסְבִּיר: אִם כְּבָר גְּנַאי לְסַפֵּר בִּגְנוּת כֶּלֶב מֵת, כָּל שֶׁכֵּן אָסוּר לְדַבֵּר רָע עַל אָדָם חַי.

כַּוָּנָתוֹ הָיְתָה שֶׁלֹּא יַרְגִּילוּ אֶת הַלָּשׁוֹן לְדַבֵּר רָע, כִּי הַלָּשׁוֹן נַעֲשֵׂית טֶבַע. וּכְשֶׁמַּרְגִּילִים אוֹתָהּ לְדַבֵּר טוֹב – גַּם זֶה נַעֲשֶׂה טֶבַע קָבוּעַ.

מַרְגִּילִים אֶת הַלָּשׁוֹן, וְהַהֶרְגֵּל נַעֲשֶׂה טֶבַע.

🪰 הַזְּבוּב וְהָעֵצָה שֶׁל הַחֲכָמִים

רַבֵּנוּ יוֹנָה מְלַמֵּד: מִי שֶׁמְּחַפֵּשׂ מוּמֵי בְּנֵי אָדָם, וְלֹא מְדַבֵּר לְעוֹלָם בַּשֶּׁבַח וּבַדָּבָר הַטּוֹב שֶׁנִּמְצָא בָּהֶם – דּוֹמֶה לַזְּבוּבִים, שֶׁנּוֹחֲתִים עַל הַמְּקוֹמוֹת הַמְּלֻכְלָכִים.

לְעֻמָּתוֹ, דֶּרֶךְ הַיְשָׁרִים הִיא לְשַׁבֵּחַ אֶת הָאָדָם, כִּי נִמְצָא בּוֹ דָּבָר טוֹב.

וְהַכְּלִי יָקָר מוֹסִיף: אֲפִלּוּ אָדָם חָכָם וּמְכֻבָּד – הַזְּבוּבִים יְחַפְּשׂוּ בּוֹ דַּוְקָא אֶת מַה שֶּׁלֹּא כֻּלָּם מֻשְׁלָמִים.

דֶּרֶךְ הַיְשָׁרִים: לְשַׁבֵּחַ אֶת הָאָדָם עַל הַדָּבָר הַטּוֹב שֶׁבּוֹ.

🖼️ הַתְּמוּנָה שֶׁבַּפְּנִים

יֵשׁ אֲנָשִׁים שֶׁרוֹאִים רַק אֶת הַיָּפֶה וְהַמּוּצְלָח, וְיֵשׁ שֶׁרוֹאִים רַק אֶת הַשְּׁחוֹרוֹת. כָּל אֶחָד בָּטוּחַ שֶׁהוּא רוֹאֶה אֶת הַמְּצִיאוּת כְּמוֹ שֶׁהִיא.

אֲבָל בֶּאֱמֶת שְׁנֵיהֶם מְצַיְּרִים אֶת הַתְּמוּנָה שֶׁבַּפְּנִים שֶׁלָּהֶם.

בָּעוֹלָם יֵשׁ הַרְבֵּה יוֹתֵר דְּבָרִים יָפִים מִדְּבָרִים מְכֹעָרִים, וְהָאָדָם בּוֹחֵר אִם יִתְמַקֵּד בַּחִיּוּבִי אוֹ בַּשְּׁלִילִי.

הָאָדָם בּוֹחֵר אִם לְהִתְמַקֵּד בַּחִיּוּבִי אוֹ בַּשְּׁלִילִי.

🌟 אוֹפְּטִימִיּוּת: אֱמוּנָה שֶׁיִּהְיֶה טוֹב

אוֹפְּטִימִיּוּת הִיא אֱמוּנָה שֶׁיִּהְיֶה טוֹב: שֶׁבַּסּוֹף הָעִנְיָנִים יִסְתַּדְּרוּ לְטוֹבָה, וְשֶׁאֲנָשִׁים טוֹבִים בְּעֶצֶם.

הַהֵפֶךְ שֶׁלָּהּ הוּא פֶּסִימִיּוּת. הַפֶּסִימִי רוֹאֶה קֹשִׁי בְּכָל הִזְדַּמְּנוּת, וְהָאוֹפְּטִימִי רוֹאֶה הִזְדַּמְּנוּת בְּכָל קֹשִׁי.

בְּתוֹרָה אָנוּ לוֹמְדִים עוֹד: מִי שֶׁבּוֹטֵחַ בַּה׳, הוּא רוֹאֶה גַּם בַּמַּצָּב הַקָּשֶׁה אֶת הַחֶסֶד שֶׁלּוֹ.

הַפֶּסִימִי רוֹאֶה קֹשִׁי בְּכָל הִזְדַּמְּנוּת, וְהָאוֹפְּטִימִי – הִזְדַּמְּנוּת בְּכָל קֹשִׁי.

😊 מָה הָרֶוַח שֶׁל עַיִן טוֹבָה?

הָרַב אֲבִיגְדוֹר מִילֵר מְלַמֵּד: הַהַתְחָלָה שֶׁל הַשִּׂמְחָה הִיא לִקְנוֹת אֶת הַמִּדָּה שֶׁנִּקְרֵאת ׳טוֹב עַיִן׳, כְּלוֹמַר לָשִׂים לֵב לַצַּד הַטּוֹב שֶׁבְּכָל דָּבָר.

מִי שֶׁמִּסְתַּכֵּל כָּךְ עַל הַכֹּל מַרְגִּישׁ סִפּוּק וְשִׂבְעַת רָצוֹן, וְהַנָּאָה תְּמִידִית.

וְיַחַד אִתָּם יֵשׁ לוֹ בִּטָּחוֹן וְתִקְוָה.

עַיִן טוֹבָה מְבִיאָה שִׂמְחָה, סִפּוּק, בִּטָּחוֹן וְתִקְוָה.

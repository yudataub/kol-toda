וְשָׂמַחְתָּ בְכָל הַטּוֹב אֲשֶׁר נָתַן לְךָ ה׳ אֱלֹהֶיךָ וּלְבֵיתֶךָ

בִּכּוּרִים: לְסַפֵּר מֵהַהַתְחָלָה

רָמָה ב׳ – הַדִּבּוּר מְעוֹרֵר

לָמָּה הַחַקְלָאִי אוֹמֵר פָּרָשָׁה שֶׁל הוֹדָיָה, מָה לָמַד רַבִּי יְרוּחָם, וְאֵיךְ מוֹדִים בִּפְרָטִים

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַכִּיר טוֹבָה עַל כָּל הַדֶּרֶךְ שֶׁעָבַר עַמִּי 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דְּבָרִים כ״ו ה׳–י׳

רָמָה ב׳ – הַדִּבּוּר מְעוֹרֵר

📖 מִקְרָא בִּכּוּרִים

בַּפָּרָשָׁה כָּתוּב: הַחַקְלָאִי הַמֵּבִיא בִּכּוּרִים אוֹמֵר לִפְנֵי הַכֹּהֵן: ׳אֲרַמִּי אֹבֵד אָבִי, וַיֵּרֶד מִצְרַיְמָה וַיָּגָר שָׁם בִּמְתֵי מְעָט׳.

הוּא מְסַפֵּר עַל הַהַתְחָלָה: יַעֲקֹב בֵּית לָבָן, יְרִידָה לְמִצְרַיִם, שִׁעְבּוּד, וְאַחַר כָּךְ: ׳וַיִּשְׁמַע ה׳ אֶת קֹלֵנוּ… וַיּוֹצִאֵנוּ ה׳ מִמִּצְרַיִם׳.

וְאָז הוּא אוֹמֵר: ׳וַיְבִאֵנוּ אֶל הַמָּקוֹם הַזֶּה… וְעַתָּה הִנֵּה הֵבֵאתִי אֶת רֵאשִׁית פְּרִי הָאֲדָמָה׳.

הַמֵּבִיא מְסַפֵּר אֶת כָּל הַדֶּרֶךְ מִיַּעֲקֹב עַד הַפֵּרוֹת.

1

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְעוֹרֵר אֶת הַלֵּב שֶׁלִּי בְּמִלִּים שֶׁל תּוֹדָה 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סֵפֶר הַחִנּוּךְ, מִצְוָה תרו

רָמָה ב׳ – הַדִּבּוּר מְעוֹרֵר

🔥 הַדִּבּוּר מְעוֹרֵר

סֵפֶר הַחִנּוּךְ מִסְבִּיר: ׳לְפִי שֶׁהָאָדָם מְעוֹרֵר מַחְשְׁבוֹתָיו וּמְצַיֵּר בִּלְבָבוֹ הָאֱמֶת בְּכֹחַ דִּבְרֵי פִּיו׳. כְּשֶׁה׳ הֵיטִיב לוֹ וּבֵרַךְ אֶת אַדְמָתוֹ – רָאוּי שֶׁיְּעוֹרֵר אֶת לִבּוֹ בְּדִבּוּר.

וּבְעִקָּר, הוּא צָרִיךְ לָזכֹּר שֶׁהַכֹּל הִגִּיעַ אֵלָיו מֵאֲדוֹן הָעוֹלָם, וְלָכֵן מְסַפֵּר.

מִתּוֹךְ הִתְעוֹרְרוּת הַנֶּפֶשׁ בְּשֶׁבַח ה', זוֹכֶה וּמִתְבָּרֶכֶת אַרְצוֹ.

דִּבּוּר שֶׁל הוֹדָיָה מְעוֹרֵר אֶת הַלֵּב וּמְבִיא בְּרָכָה.

2

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא מִתְבַּיֵּשׁ לְהַאֲרִיךְ בְּתוֹדָה 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַבִּי יְרוּחָם מִמִּיר (צַדִּיקִים מוֹדִים לַה׳)

רָמָה ב׳ – הַדִּבּוּר מְעוֹרֵר

🗣️ צְרִיכִים לְדַבֵּר הַרְבֵּה

רַבִּי יְרוּחָם מִמִּיר שָׁאַל: וְכִי לֹא מַסְפִּיק שֶׁהַחַקְלָאִי הֵבִיא אֶת הַפֵּרוֹת? לָמָּה צָרִיךְ עוֹד לְסַפֵּר בַּפֶּה כָּל כָּךְ הַרְבֵּה?

וְעָנָה: כְּשֶׁמַּגִּיעַ הַדָּבָר לְהַכָּרַת הַטּוֹב – צְרִיכִים לְדַבֵּר וּלְהַאֲרִיךְ. אֲנִי רוֹאֶה לִפְעָמִים תַּלְמִיד שֶׁאֲסִיר תּוֹדָה, וּפָנָיו מְעִידוֹת, אֲבָל הוּא חוֹשֵׁב שֶׁלֹּא מַתְאִים לוֹמַר. וַאֲנִי אוֹמֵר: כֵּן! כֵּן צָרִיךְ לוֹמַר.

שֶׁלֹּא נִהְיֶה כְּמוֹ הַסּוֹחֵר שֶׁעוֹבֵר מַהֵר.

אֵין מַסְפִּיק לְהַרְגִּישׁ אֶת הַתּוֹדָה – צְרִיכִים לוֹמַר אוֹתָהּ בְּמִלִּים רַבּוֹת.

3

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה כְּדֵי לִהְיוֹת שָׂמֵחַ בְּכָל הַטּוֹב 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דְּבָרִים כ״ו י״א · אוֹר הַחַיִּים · הוֹדָיָה פּוֹתַחַת שְׁעָרִים

רָמָה ב׳ – הַדִּבּוּר מְעוֹרֵר

😊 וְשָׂמַחְתָּ בְכָל הַטּוֹב

הַפָּסוּק בְּסוֹף הַפָּרָשָׁה: ׳וְשָׂמַחְתָּ בְכָל הַטּוֹב אֲשֶׁר נָתַן לְךָ ה׳ אֱלֹהֶיךָ׳. פֹּה יֵשׁ קֻשְׁיָה: הַחַקְלָאִי לֹא אוֹכֵל אֶת הַבִּכּוּרִים, אָז בְּמָה הוּא שָׂמֵחַ?

בַּשִּׁעוּר מִתְבָּאֵר: זוֹ בְּרָכָה לְמִי שֶׁמּוֹדֶה – מִי שֶׁמּוֹדֶה עַל כָּל מָה שֶׁעָבַר עָלָיו, זוֹכֶה לִשְׂמֹחַ בְּכָל הַטּוֹב. ׳הַטּוֹב׳ הוּא גַּם ה׳ עַצְמוֹ, מְקוֹר כָּל טוֹב.

שִׂמְחָה בְּכָל הַטּוֹב הִיא בְּרָכָה לְמִי שֶׁמּוֹדֶה.

4

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה בִּפְרָטִים וּפוֹתֵחַ שְׁעָרִים 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סֵפֶר הַחִנּוּךְ, מִצְוָה תרו · תְּהִלִּים ק׳ ד׳

רָמָה ב׳ – הַדִּבּוּר מְעוֹרֵר

🔑 הוֹדָיָה פּוֹתַחַת שְׁעָרִים

הַשִּׁעוּר מְסַכֵּם כַּמָּה יְסוֹדוֹת מִדִּבְרֵי הַחִנּוּךְ: מְחַפְּשִׂים לְהוֹדוֹת מֵהַהַתְחָלָה וּמְסַפְּרִים הַכֹּל; אֵין מַסְפִּיק ׳תּוֹדָה׳ בִּלְבַד אֶלָּא גַּם לְעוֹרֵר אֶת הַנֶּפֶשׁ בְּשֶׁבַח ה'; אַחֲרֵי הַשֶּׁבַח מֻתָּר לְבַקֵּשׁ.

ה׳ רוֹצֶה מְאֹד לְהֵיטִיב לָנוּ, וְהוֹדָיָה נוֹתֶנֶת לוֹ אֶת הָאֶפְשָׁרוּת. עַל כֵּן נֶאֱמַר: ׳בֹּאוּ שְׁעָרָיו בְּתוֹדָה׳.

הוֹדָיָה מְפַרֶטֶת וּמְסַפֶּרֶת פּוֹתַחַת שְׁעָרִים שֶׁל שֶׁפַע.

5
💡

רָמָה ב׳ – הַדִּבּוּר מְעוֹרֵר

מָה לוֹקְחִים מִרָמָה ב׳?

מִי שֶׁמְּסַפֵּר בַּפֶּה אֶת כָּל הַחֲסָדִים שֶׁה׳ עָשָׂה אִתּוֹ, מְעוֹרֵר אֶת לִבּוֹ וּמַרְגִּישׁ אֶת אַהֲבַת ה׳. הוֹדָיָה בִּפְרָטִים הִיא הַדֶּרֶךְ לְאַהֲבָה גְּדוֹלָה יוֹתֵר.

6
❤️
וְשָׂמַחְתָּ בְכָל הַטּוֹב אֲשֶׁר נָתַן לְךָ ה׳ אֱלֹהֶיךָ וּלְבֵיתֶךָ

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • עַל מִי שֶׁעָזַר לִי הַשָּׁבוּעַ אָמַרְתִּי רַק ׳תּוֹדָה׳, וּמָה אֶפְשָׁר לְהוֹסִיף כְּדֵי שֶׁיַּרְגִּישׁ?
  • מֵאַיִן בָּאָה הַמִּשְׁפָּחָה שֶׁלִּי, וְעַל אֵלּוּ שְׁלַבִּים בַּדֶּרֶךְ אֲנִי מוֹדֶה לַה׳?
  • אֲסַפֵּר בַּבַּיִת אֶת ׳הַסִּפּוּר שֶׁל הַמִּשְׁפָּחָה שֶׁלִּי׳ מֵהַהַתְחָלָה, וְאֶרְשֹׁם עַל מָה הוֹדֵיתִי.

אַשְׁרֵי הָאִישׁ – הָעֵץ וְהַמֹּץ

רָמָה ב׳ – הֶעָלֶה וְהַפְּרִי

שְׁלוֹשָׁה צְעָדִים שֶׁמַּרְחִיקִים מֵהַטּוֹב, עֵץ עַל מַיִם, וּמֹץ שֶׁהָרוּחַ מְעִיפָה

📖 מִזְמוֹרֵי תְּהִלִּים · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹצֵר כְּבָר בַּצַּעַד הָרִאשׁוֹן 💛

📖 מִזְמוֹרֵי תְּהִלִּים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – הֶעָלֶה וְהַפְּרִי

🚶 שָׁלוֹשׁ מַדְרֵגוֹת

הַמִּזְמוֹר מַזְהִיר מִשְּׁלוֹשָׁה דְּבָרִים: ׳עֲצַת רְשָׁעִים׳, ׳דֶּרֶךְ חַטָּאִים׳ וּ׳מוֹשַׁב לֵצִים׳. הַפְּעָלִים הֵם: הָלַךְ, עָמַד, יָשַׁב.

הַפֵּרוּשׁ הַפָּשׁוּט: ׳הָלַךְ׳ – זְמַן קָצָר בַּדֶּרֶךְ, ׳עָמַד׳ – מִתְעַכֵּב יוֹתֵר, ׳יָשַׁב׳ – יְשִׁיבַת קֶבַע. אָדָם לֹא נוֹפֵל בְּבַת אַחַת; הוּא יוֹרֵד מַדְרֵגָה אַחַר מַדְרֵגָה.

מִי שֶׁלֹּא הוֹלֵךְ בַּהַתְחָלָה – לֹא יִגְמַר בְּיֶשִׁיבָה.

1

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי יוֹשֵׁב עִם חֲבֵרִים אֲנִי מַעֲלֶה נוֹשְׂאִים טוֹבִים 💛

📖 מִזְמוֹרֵי תְּהִלִּים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, פִּרְקֵי אָבוֹת ג׳ ב׳

רָמָה ב׳ – הֶעָלֶה וְהַפְּרִי

🪑 מוֹשַׁב – גַּם בְּלִי לֵצִים

הַמִּלָּה ׳מוֹשַׁב׳ מְפַתַּחַת שְׁאֵלָה: לָמָּה לֹא כָּתוּב ׳חֶבְרַת לֵצִים׳? תְּשׁוּבָה: מוֹשַׁב הוּא הַמָּקוֹם עַצְמוֹ, גַּם כְּשֶׁאֵין שָׁם לֵצִים כְּרֶגַע – כִּי הוּא מָקוֹם שֶׁלָּהֶם וַאֲנַחְנוּ לֹא רוֹצִים לֵשֵׁב בּוֹ.

הַמִּשְׁנָה בְּאָבוֹת לוֹמֶדֶת מֵהַפָּסוּק: שְׁנַיִם שֶׁיּוֹשְׁבִים וְאֵין בֵּינֵיהֶם דִּבְרֵי תּוֹרָה – זֶה מוֹשַׁב לֵצִים. מִי שֶׁמְּחַפֵּשׂ יֶשִׁיבָה מְלֵאָה, יְכוֹלָה לִהְיוֹת רַק עִם דְּבָרִים טוֹבִים.

מָקוֹם שֶׁאֵין בּוֹ דִּבְרֵי תּוֹרָה הוּא מוֹשַׁב לֵצִים.

2

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹחֵר בַּטּוֹב כִּי יֵשׁ לִי חֵפֶץ בָּהּ 💛

📖 מִזְמוֹרֵי תְּהִלִּים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִדְרַשׁ תְּהִלִּים א׳

רָמָה ב׳ – הֶעָלֶה וְהַפְּרִי

💛 סוּר מֵרָע וַעֲשֵׂה טוֹב

הַפָּסוּק הַבָּא פּוֹתֵחַ ב׳כִּי אִם׳: ׳כִּי אִם בְּתוֹרַת ה׳ חֶפְצוֹ׳. כְּלוֹמַר, אֵין מַסְפִּיק לָסוּר מֵרָע; צָרִיךְ גַּם לַעֲשׂוֹת טוֹב.

הַמִּדְרָשׁ בִּתְהִלִּים א׳ לוֹמֵד זֹאת מִפָּסוּק אַחֵר: ׳סוּר מֵרָע וַעֲשֵׂה טוֹב׳ – סוּר מֵרָע עַל מְנָת לַעֲשׂוֹת טוֹב. הַחֵפֶץ שֶׁל הָאִישׁ הוּא בַּתּוֹרָה, כִּי זוֹ תּוֹרָתוֹ שֶׁל ה׳.

אֹשֶׁר אֲמִתִּי = לָסוּר מֵרָע וְגַם לַעֲשׂוֹת טוֹב.

3

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד בְּאַהֲבָה, וְהַתּוֹרָה נַעֲשֵׂית שֶׁלִּי 💛

📖 מִזְמוֹרֵי תְּהִלִּים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, עֲבוֹדָה זָרָה י״ט עַמּוּד א׳

רָמָה ב׳ – הֶעָלֶה וְהַפְּרִי

📖 תּוֹרַת ה׳ וְאַחַר כָּךְ תּוֹרָתוֹ

׳בְּתוֹרַת ה׳ חֶפְצוֹ וּבְתוֹרָתוֹ יֶהְגֶּה׳. רָבָא אָמַר: בַּהַתְחָלָה הַתּוֹרָה נִקְרֵאת עַל שְׁמוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא, וּבַסּוֹף נִקְרֵאת עַל שֵׁם הַלּוֹמֵד – ׳תּוֹרָתוֹ׳.

מִי שֶׁעָמֵל בַּתּוֹרָה, הִיא נַעֲשֵׂית שֶׁלּוֹ. וְעוֹד אָמְרוּ: אָדָם לוֹמֵד תּוֹרָה בְּאֹפֶן הַטּוֹב בְּיוֹתֵר מִמָּקוֹם שֶׁלִּבּוֹ חָפֵץ.

הַתּוֹרָה הַזֹּאת שֶׁל ה׳ נַעֲשֵׂית שֶׁלִּי כְּשֶׁאֲנִי עָמֵל.

4

💛 אַהֲבַת הַשֵּׁם: אֲנִי יוֹנֵק מִשְּׁנֵי נְהָרוֹת: לִמּוּד וּמַעֲשִׂים טוֹבִים 💛

📖 מִזְמוֹרֵי תְּהִלִּים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – הֶעָלֶה וְהַפְּרִי

🌳 עֵץ בֵּין שְׁנֵי נְהָרוֹת

׳כְּעֵץ שָׁתוּל עַל פַּלְגֵי מָיִם׳. ׳פֶּלֶג׳ בִּלְשׁוֹן חִלּוּק: מָקוֹם שֶׁבּוֹ שְׁנֵי נְהָרוֹת מִתְפַּצְּלִים, וְהָעֵץ יוֹנֵק מִשְּׁנֵיהֶם. גַּם אִם אֶחָד יִתְיַבֵּשׁ, הוּא יוֹנֵק מֵהַשֵּׁנִי.

׳אֲשֶׁר פִּרְיוֹ יִתֵּן בְּעִתּוֹ וְעָלֵהוּ לֹא יִבּוֹל׳: הַפְּרִי הוּא הַמִּצְווֹת בִּזְמַנָּן; הֶעָלֶה הוּא דְּבָרִים שֶׁאֵינָם הַפְּרִי עַצְמוֹ, אֲבָל צוֹרֶךְ לַלִּמּוּד – כְּמוֹ אֲכִילָה וּמְנוּחָה.

גַּם הָעָלִים, כְּלוֹמַר דְּבָרִים שֶׁמְּסַיְּעִים לִלְמֹד, חֲשׁוּבִים.

5

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹחֵר לִהְיוֹת עֵץ עִם פְּרִי 💛

📖 מִזְמוֹרֵי תְּהִלִּים – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – הֶעָלֶה וְהַפְּרִי

🌾 הָעֵץ מוּל הַמֹּץ

׳לֹא כֵן הָרְשָׁעִים כִּי אִם כַּמֹּץ אֲשֶׁר תִּדְּפֶנּוּ רוּחַ׳. הַמֹּץ הוּא הַקְּלִפָּה הַדַּקָּה שֶׁעוֹטֶפֶת אֶת הַתְּבוּאָה; בַּדִּישׁ מְפַנִּים אוֹתָהּ, וְהָרוּחַ נוֹשֵׂאת אוֹתָהּ.

הַמִּזְמוֹר מַצִּיג מָשָׁל הָפוּךְ: הָעֵץ מְקֻיָּם וּמוֹצִיא פְּרִי, וְהַמֹּץ קַל וְאֵין בּוֹ תּוֹעֶלֶת. בַּסּוֹף: ׳כִּי יוֹדֵעַ ה׳ דֶּרֶךְ צַדִּיקִים׳ – ה׳ מַשְׁגִּיחַ עַל הַדֶּרֶךְ הַטּוֹבָה וְאוֹהֵב אוֹתָהּ.

עֵץ עִם שֹׁרָשִׁים אוֹ מֹץ שֶׁהָרוּחַ נוֹשֵׂאת – הַבְּחִירָה שֶׁלָּנוּ.

6
💡

רָמָה ב׳ – הֶעָלֶה וְהַפְּרִי

מָה לוֹקְחִים מִרָמָה ב׳?

ה׳ מַשְׁגִּיחַ עַל הַדֶּרֶךְ הַטּוֹבָה וְאוֹהֵב אוֹתָהּ. כְּשֶׁאֲנִי עוֹמֵד בְּנִסְיוֹן קָטָן, אֲנִי מַרְגִּישׁ אֶת הַקֶּשֶׁר אֵלָיו.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה הַצַּעַד הַקָּטָן הָרִאשׁוֹן שֶׁיָּכוֹל לִגְרֹר אוֹתִי לְמָקוֹם לֹא טוֹב, וְאֵיךְ אֶעֱצֹר אוֹתוֹ?
  • מָה ׳הַמַּיִם׳ שֶׁלִּי – מָה גּוֹרֵם לִי לְגַדֵּל פְּרִי?

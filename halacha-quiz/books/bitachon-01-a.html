🕊️ מַהוּ בִּטָּחוֹן?

בְּכָל יוֹם אֲנַחְנוּ אוֹמְרִים ׳שְׁמַע יִשְׂרָאֵל׳. בָּזֶה אֲנַחְנוּ מְעִידִים שֶׁה׳ בָּרָא אֶת הָעוֹלָם וּמַשְׁגִּיחַ עַל כָּל מַה שֶּׁקּוֹרֶה בּוֹ, וְשֶׁאֵין לְאַף אֶחָד כֹּחַ לַעֲשׂוֹת דָּבָר בִּלְעָדָיו.

בִּטָּחוֹן פֵּרוּשׁוֹ: לִהְיוֹת רָגוּעַ וְשָׁלֵו, וְלֹא לִפְחֹד מַה יִּהְיֶה.

כִּי יֵשׁ לִי אַבָּא טוֹב שֶׁאוֹהֵב אוֹתִי יוֹתֵר מִמַּה שֶּׁאֲנִי אוֹהֵב אֶת עַצְמִי, וְכָל מָה שֶׁהוּא עוֹשֶׂה – רַק לְטוֹבָתִי.

אַבָּא טוֹב שֶׁאוֹהֵב אוֹתִי יוֹתֵר מִכְּפִי שֶׁאֲנִי אוֹהֵב אֶת עַצְמִי.

חוֹבוֹת הַלְּבָבוֹת מְלַמֵּד שֶׁכְּדֵי לִבְטֹחַ צָרִיךְ לָדַעַת שִׁבְעָה דְּבָרִים עַל ה׳. הִנֵּה שְׁנַיִם מֵהֶם.

הָרִאשׁוֹן: ה׳ מַשְׁגִּיחַ עָלַי מִתְּחִלַּת חַיַּי וְעַד סוֹפָם.

הַשֵּׁנִי: ה׳ אוֹהֵב אוֹתִי, מְרַחֵם עָלַי וְחָס עָלַי. וַאֲנִי כְּבָר רָאִיתִי אֵיךְ הוּא עָשָׂה לִי טוֹבוֹת גְּדוֹלוֹת.

ה׳ מַשְׁגִּיחַ עָלַי מֵהַהַתְחָלָה וְעַד הַסּוֹף, וְאוֹהֵב אוֹתִי.

הַשְּׁלִישִׁי: מֵרוֹב אַהֲבָה, ה׳ לֹא מַסִּיחַ דַּעְתּוֹ מִמֶּנִּי וְלֹא מִתְעַצֵּל לְמַלֵּא אֶת מִשְׁאֲלוֹתַי.

הָרְבִיעִי: ה׳ יוֹדֵעַ בְּדִיּוּק מַה טּוֹב לִי – לֹא רַק לְפִי מַה שֶּׁנִּרְאֶה לָעַיִן, אֶלָּא מַה טּוֹב לִי בֶּאֱמֶת.

יֶלֶד רוֹצֶה מַמְתָּק לִפְנֵי הָאֲרוּחָה, וְאַבָּא אוֹמֵר ׳לֹא עַכְשָׁיו׳. הוּא לֹא מִתְעַלֵּם מִמֶּנּוּ – הוּא יוֹדֵעַ מַה טּוֹב לוֹ. כָּךְ גַּם ה׳ יוֹדֵעַ אֶת הַטּוֹב הָאֲמִתִּי שֶׁלִּי.

ה׳ לֹא מִתְעַצֵּל, וְיוֹדֵעַ מַה טּוֹב לִי בֶּאֱמֶת.

ה׳ חָזָק, וּבְלִי רְצוֹנוֹ אַף אֶחָד לֹא מַזִּיק וְלֹא מוֹעִיל.

הַשְּׁבִיעִי: ה׳ נְדִיב לֵב וּבַעַל חֶסֶד בְּתַכְלִית, גַּם לְמִי שֶׁרָאוּי וְגַם לְמִי שֶׁפָּחוֹת רָאוּי, וְחַסְדּוֹ בְּלִי הֶפְסֵק.

הַנְּתִיבוֹת שָׁלוֹם מוֹסִיף: בִּטָּחוֹן כּוֹלֵל לְהַאֲמִין שֶׁה׳ אוֹהֵב אוֹתִי כְּמוֹ שֶׁאַבָּא אוֹהֵב אֶת בְּנוֹ, גַּם כְּשֶׁאֲנִי לֹא מֻשְׁלָם.

ה׳ נָדִיב תָּמִיד, וְאוֹהֵב אוֹתִי כְּמוֹ אַבָּא אוֹהֵב אֶת בְּנוֹ.

🏅 מִדָּה – לֹא רַק פַּעַם אַחַת

אוֹמְרִים ׳מִדַּת הַבִּטָּחוֹן׳, כְּמוֹ מִדַּת הָעֲנָוָה וּמִדַּת הָרַחֲמִים. מִדָּה פֵּרוּשָׁהּ שֶׁהַדָּבָר קָנוּי בַּלֵּב וְקָבוּעַ.

מִי שֶׁרַק פַּעַם אַחַת מְרַחֵם, עֲדַיִן לֹא קָנָה אֶת מִדַּת הָרַחֲמִים; הוּא רַק הִתְגַּבֵּר עַכְשָׁו. כָּךְ גַּם בִּטָּחוֹן: מִי שֶׁפַּעַם דּוֹאֵג וּפַעַם בּוֹטֵחַ, עוֹד לֹא קָנָה אֶת הַמִּדָּה.

אֲבָל אֶפְשָׁר לְהִתְאַמֵּן, צַעַד אַחַר צַעַד.

📝 מָה עוֹשִׂים כְּשֶׁיֵּשׁ דְּאָגָה?

כְּשֶׁיֵּשׁ בְּעָיָה, כְּדַאי לְהַעֲבִיר בַּמֹּחַ מַחֲשָׁבוֹת שֶׁל בִּטָּחוֹן: ה׳ מַשְׁגִּיחַ, אוֹהֵב, חָזָק, יוֹדֵעַ מַה טּוֹב וְנָדִיב.

זוֹ הַמְּשִׂימָה שֶׁל הַשָּׁבוּעַ: מְדַרְגִּים שְׁלוֹשָׁה דְּבָרִים שֶׁמַּדְאִיגִים אוֹתִי, וְלִצְדָּם כּוֹתְבִים מַה אֲנִי יוֹדֵעַ עַל ה׳ שֶׁיַּרְגִּיעַ אוֹתִי.

כָּךְ הַלֵּב לוֹמֵד לִהְיוֹת רָגוּעַ.

מָה לוֹקְחִים מִרָמָה א׳?

  • אֵיזֶה מִשִּׁבְעַת הַדְּבָרִים עַל ה׳ הוּא הֲכִי קָרוֹב לִלְבִּי, וְלָמָּה?
  • אֶרְשֹׁם שְׁלוֹשָׁה דְּבָרִים שֶׁמַּדְאִיגִים אוֹתִי, וְלִצְדָּם אֶכְתֹּב מָה אֲנִי יוֹדֵעַ עַל ה׳ שֶׁמַּרְגִּיעַ – וְאֶקְרָא אֶת זֶה בַּלַּיְלָה לִפְנֵי הַשֵּׁנָה.

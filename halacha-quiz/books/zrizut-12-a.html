זְרִיזִין מַקְדִּימִין לַמִּצְווֹת

🌱 בְּעַצְמוֹ · רָמָה א׳

בְּעַצְמִי, בִּזְרִיזוּת – הַכְּרָכָּרָה, הַמִּכְתָּב וְעֶרֶב פֶּסַח

רָמָה א׳ – חֶסֶד בַּגֶּשֶׁם

הָרַב חַיִּים זוֹנֶנְפֶלְד לֹא שׁוֹלֵחַ אֲחֵרִים: הוּא עוֹלֶה בַּגֶּשֶׁם, רָץ לְבֵית הַחוֹלִים וּמְחַלֵּק כֶּסֶף לְפֶסַח. מָה אוֹמֵר זָרִיז עַל מִצְוָה שֶׁעוֹשִׂים בְּעַצְמוֹ?

📖 זְרִיזוּת · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
🌧️

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹאֶה חֶסֶד קָטָן, וּמוֹדֶה לַה׳ עַל הַטּוֹבִים שֶׁבָּעוֹלָם 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, זָרִיז וְנִשְׂכָּר, פֶּרֶק ה׳ (הָאִישׁ עַל הַחוֹמָה)

רָמָה א׳ – חֶסֶד בַּגֶּשֶׁם

🌧️ גֶּשֶׁם בְּשׁוּק מַחֲנֵה יְהוּדָה

בְּיוֹם גֶּשֶׁם שׁוֹטֵף עָמְדוּ אֲנָשִׁים מִתַּחַת לַגַּגּוֹן שֶׁבַּשּׁוּק, כְּדֵי לֹא לְהִתְרַטֵּב. לְיָדָם עָמְדָה אִשָּׁה עִם תִּינוֹק עַל הַיָּדַיִם.

פִּתְאוֹם עָבְרָה כְּרָכָּרָה שְׁחוֹרָה, וְיָשַׁב בְּתוֹכָהּ הָרַב חַיִּים זוֹנֶנְפֶלְד, רַבָּהּ שֶׁל יְרוּשָׁלַיִם. הוּא חָזַר מִבְּרִית מִילָה בְּבֵית הַחוֹלִים ׳שַׁעֲרֵי צֶדֶק׳.

אִישׁ מִזִּקְנֵי יְרוּשָׁלַיִם, שֶׁרָאָה אֶת הַכֹּל, סִפֵּר אַחַר כָּךְ מָה קָרָה.

סִפּוּר עַל גֶּשֶׁם וְעַל חֶסֶד, שֶׁרָאוּ בְּעֵינַיִם.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוּכָן לְוַתֵּר עַל הַנוֹחוּת שֶׁלִּי בִּשְׁבִיל מִישֶׁהוּ אַחֵר, וְה׳ אוֹהֵב אוֹתִי 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מָרָא דְאַרְעָא יִשְׂרָאֵל

רָמָה א׳ – חֶסֶד בַּגֶּשֶׁם

🐴 הַכְּרָכָּרָה נֶעֱצֶרֶת

לְפֶתַע נֶעֶצְרָה הַכְּרָכָּרָה. הָרַב, שֶׁהָיָה קָרוֹב לְגִיל שְׁמוֹנִים, קָפַץ וְהִתְיַשֵּׁב בִּתְנוּעָה זְרִיזָה שֶׁל צָעִיר לְיַד הָעַגְלוֹן, עַל הַמּוֹשָׁב הַחָשׂוּף בַּגֶּשֶׁם.

וּמִשָּׁם הוּא הִזְמִין אֶת הָאִשָּׁה עִם הַתִּינוֹק לַעֲלוֹת לִפְנִים הַכְּרָכָּרָה, לַמָּקוֹם הַיָּבֵשׁ, וּלְמַהֵר.

הוּא נִרְטַב בַּגֶּשֶׁם, כְּדֵי שֶׁמִּישֶׁהוּ אַחֵר יִהְיֶה יָבֵשׁ.

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מָרָא דְאַרְעָא יִשְׂרָאֵל

רָמָה א׳ – חֶסֶד בַּגֶּשֶׁם

🙏 ״אֲנִי לֹא חוֹזֵר לַמָּקוֹם״

הָאִשָּׁה לֹא הִרְגִּישָׁה בְּנוֹחַ, וְנִסְּתָה לְסָרֵב בִּנְעִימוּת. אֲבָל הָרַב הִצְהִיר: גַּם אִם הִיא לֹא תִּכָּנֵס, הוּא לֹא יַחֲזֹר לִמְקוֹמוֹ – הַמָּקוֹם שֶׁמָּסַר לָהּ לְמַעֲשֵׂה חֶסֶד.

בַּסּוֹף הִיא נִכְנְעָה וְעָלְתָה עִם הַתִּינוֹק לַמָּקוֹם הַיָּבֵשׁ.

כָּךְ מַרְאֶה זָרִיז שֶׁהַחֶסֶד אֵינוֹ רַק מִלִּים: הוּא נַעֲשֶׂה בַּגּוּף.

חֶסֶד אֶפְשָׁר לַעֲשׂוֹת גַּם בַּגּוּף, לֹא רַק בְּמִלִּים.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְהִתְקָרֵב לְכָל יְהוּדִי, גַּם לְמִי שֶׁשּׁוֹנֶה מִמֶּנִּי 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, זָרִיז וְנִשְׂכָּר, פֶּרֶק ה׳ (הָאִישׁ עַל הַחוֹמָה)

רָמָה א׳ – חֶסֶד בַּגֶּשֶׁם

✉️ בַּקָּשָׁה לְמִכְתָּב

אֲנָשִׁים בָּאוּ אֶל הָרַב חַיִּים וּבִקְשׁוּ מִמֶּנּוּ מִכְתָּב הַמְלָצָה לְבֵית הַחוֹלִים ׳שַׁעֲרֵי צֶדֶק׳. הָיָה שָׁם אִישׁ חוֹלֶה מְאֹד, שֶׁהָיָה רָחוֹק מִדֶּרֶךְ הָרַב.

מִכְתָּב הוּא דָּבָר טוֹב. אֲבָל הָרַב הִרְגִּישׁ: הַחוֹלֶה צָרִיךְ אֶת הָעֶזְרָה עַכְשָׁו, וְלֹא סִפּוּר שֶׁעוֹבֵר מִיָּד לְיָד.

מִכְתָּב עוֹזֵר, אֲבָל אֲנִי עוֹזֵר יוֹתֵר כְּשֶׁאֲנִי בָּא בְּעַצְמִי.

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, זָרִיז וְנִשְׂכָּר, פֶּרֶק ה׳ (הָאִישׁ עַל הַחוֹמָה)

רָמָה א׳ – חֶסֶד בַּגֶּשֶׁם

🏥 בְּעַצְמוֹ לְבֵית הַחוֹלִים

הָרַב לֹא הִסְתַּפֵּק בַּמִּכְתָּב. בְּגֶשֶׁם שׁוֹטֵף הוּא מִהֵר בְּעַצְמוֹ לְבֵית הַחוֹלִים, בְּלִי לְעַכֵּב.

שָׁם סִדֵּר שֶׁיְּקַבְּלוּ אֶת הַחוֹלֶה וִיטַפְּלוּ בּוֹ, עַד שֶׁהִבְרִיא.

הָרַב הֶאֱמִין שֶׁהַדֶּרֶךְ הַטּוֹבָה בְּיוֹתֵר לְהַצִּיל אָדָם הִיא לָבוֹא בְּעַצְמִי, וְלֹא לִשְׁלֹחַ נְיָר.

בְּגֶשֶׁם שׁוֹטֵף הוּא הָלַךְ בְּעַצְמוֹ, וְהַחוֹלֶה הֶחְלִים.

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מָרָא דְאַרְעָא יִשְׂרָאֵל

רָמָה א׳ – חֶסֶד בַּגֶּשֶׁם

🕯️ עֶרֶב פֶּסַח

פַּעַם בָּאוּ כְּסָפִים לַיִּשּׁוּב בְּעֶרֶב פֶּסַח. הָרַב אָמַר לָרַבָּנִית: ׳הַפְסִיקִי אֶת הַהַכָּנוֹת, וּלְכִי לַחֲלֹק אֶת הַכֶּסֶף בְּבָתֵּי יִשְׂרָאֵל, כִּי כֻּלָּם עֲסוּקִים בְּהַכָּנוֹת הֶחָג.׳

הִיא עָנְתָה: ׳גַּם אֲנִי עֲסוּקָה מְאֹד!׳ הָרַב הִסְבִּיר: לָנוּ לֹא אִכְפַּת אִם הַסֵּדֶר לֹא יִהְיֶה מְהֻדָּר; הָעִקָּר מַצָּה, מָרוֹר וְבִעוּר חָמֵץ. לַאֲחֵרִים אִכְפַּת, כִּי אִם לֹא יְקַבְּלוּ אֶת מָה שֶׁצָּרִיךְ, הֶחָג יִהְיֶה עָצוּב.

הָרַב דָּחָה אֶת הַהֲדָר שֶׁלּוֹ כְּדֵי שֶׁלַּאֲחֵרִים יִהְיֶה חַג.

💛 אַהֲבַת הַשֵּׁם: אֲנִי יוֹדֵעַ: כְּשֶׁאֲנִי מְשַׂמֵּחַ אֲחֵרִים, הַשִּׂמְחָה חוֹזֶרֶת אֵלַי 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מָרָא דְאַרְעָא יִשְׂרָאֵל

רָמָה א׳ – חֶסֶד בַּגֶּשֶׁם

🎉 שִׂמְחָה בְּכָל הַבָּתִּים

הָרַבָּנִית לָקְחָה אֶת הַכֶּסֶף, הָלְכָה בְּכָל בָּתֵּי יִשְׂרָאֵל וְחִלְּקָה אֶת הַמָּעוֹת לְפֶסַח. כָּל הַיּוֹם יָצָא לָהּ בְּכָךְ, וּבְבֵיתָהּ לֹא הֵכִינָה כְּלוּם.

אַחַר כָּךְ הָיְתָה שִׂמְחָה בְּכָל בָּתֵּי יִשְׂרָאֵל, וְעוֹד יוֹתֵר בְּבֵיתוֹ שֶׁל הָרַב.

הָרַב שָׁלַח אֶת הַמְּלָאכָה בְּעַצְמוֹ וּבְמִשְׁפַּחְתּוֹ, וְלֹא חִכָּה שֶׁאֲחֵרִים יַעֲשׂוּ.

מִי שֶׁמְּוַתֵּר בִּשְׁבִיל אֲחֵרִים, מְקַבֵּל שִׂמְחָה גְּדוֹלָה יוֹתֵר.

🔍

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹשֶׂה בְּעַצְמִי, וְשָׂמֵחַ לִהְיוֹת שְׁלִיחַ שֶׁל ה׳ 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – חֶסֶד בַּגֶּשֶׁם

🔍 בְּעַצְמִי אוֹ בִּשְׁלִיחוּת?

בִּשְׁלוֹשָׁה הַסִּפּוּרִים הָרַב בָּא בְּעַצְמוֹ: הוּא עָלָה עַל הַכְּרָכָּרָה, הָלַךְ לְבֵית הַחוֹלִים, וְשָׁלַח אֶת הָרַבָּנִית לַחֲלֹק כֶּסֶף וְהִסְכִּים לְוַתֵּר.

לִפְעָמִים אִי אֶפְשָׁר לָבוֹא בְּעַצְמִי וְשׁוֹלְחִים שָׁלִיחַ, וְזֶה בְּסֵדֶר. אֲבָל כְּשֶׁאֶפְשָׁר – הַמִּצְוָה הַכִּי יָפָה הִיא כְּשֶׁאֲנִי עוֹשֶׂה אוֹתָהּ בְּעַצְמִי.

כְּשֶׁאֶפְשָׁר, אֲנִי עוֹשֶׂה בְּעַצְמִי.

🌟

💛 אַהֲבַת הַשֵּׁם: מִי שֶׁזָּרִיז בְּחֶסֶד, דּוֹמֶה לַה׳ שֶׁעוֹשֶׂה חֶסֶד עִם כֻּלָּם 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – חֶסֶד בַּגֶּשֶׁם

🌟 מָה לוֹמְדִים מֵהַגֶּשֶׁם?

הָרַב חַיִּים עָשָׂה שְׁלוֹשָׁה דְּבָרִים: וִתֵּר עַל נוֹחוּת, הִתְחִיל בְּעַצְמוֹ, וְלֹא הִתְעַכֵּב בְּגַלֵּל הַגֶּשֶׁם.

בַּחַיִּים שֶׁלָּנוּ אֶפְשָׁר לַעֲשׂוֹת כָּמוֹהוּ: לַעֲזֹר לְחָבֵר, לָקוּם בִּשְׁבִיל אַחֵר, לַעֲשׂוֹת חֶסֶד שֶׁאֲחֵרִים מַמְתִּינִים לוֹ.

זָרִיז = עוֹשֶׂה בְּעַצְמוֹ, מִיָּד, אֲפִלּוּ כְּשֶׁזֶּה לֹא נוֹחַ.

9
💡

רָמָה א׳ – חֶסֶד בַּגֶּשֶׁם

מָה לוֹקְחִים מִרָמָה א׳?

הָרַב חַיִּים עָשָׂה חֶסֶד בְּגוּפוֹ, כִּי אָהַב אֶת ה׳ וְאֶת בְּרִיּוֹתָיו. כָּל חֶסֶד שֶׁאֲנִי עוֹשֶׂה בְּעַצְמִי הוּא דֶּרֶךְ לְהִתְקָרֵב לַה׳.

זְרִיזִין מַקְדִּימִין לַמִּצְווֹת

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָה אֲנִי יָכוֹל לַעֲשׂוֹת הַיּוֹם בְּעַצְמִי, בִּמְקוֹם לְבַקֵּשׁ מִמִּישֶׁהוּ אַחֵר?
  • מָתַי הַשָּׁבוּעַ הֶעֱבַרְתִּי לְמִישֶׁהוּ אַחֵר מִצְוָה שֶׁיָּכֹלְתִּי לַעֲשׂוֹת בְּעַצְמִי?
  • הַיּוֹם אֶבְחַר דָּבָר טוֹב אֶחָד וְאֶעֱשֶׂה אוֹתוֹ בְּעַצְמִי, בְּשִׂמְחָה וּמִיָּד.

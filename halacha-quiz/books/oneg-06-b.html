טוֹב לְהֹדוֹת לַה׳ וּלְזַמֵּר לְשִׁמְךָ עֶלְיוֹן

🌱 שִׁיר הַשַּׁבָּת · רָמָה ב׳

שִׁירוֹ שֶׁל אָדָם הָרִאשׁוֹן

רָמָה ב׳ – תְּשׁוּבָה, הוֹדָיָה וְשִׁיר

אֵיךְ נוֹלַד שִׁיר הַשַּׁבָּת מִתְּשׁוּבָה, מָה קָשׁוּר בֵּין הוֹדָיָה לְטָעוּת, וְלָמָּה שָׁרִים ׳טוֹב לְהוֹדוֹת׳ בְּשַׁבָּת

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
📜

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ שֶׁתְּשׁוּבָה אֶפְשָׁרִית תָּמִיד 💛

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּרֵאשִׁית רַבָּה כ״ב י״ג

רָמָה ב׳ – תְּשׁוּבָה, הוֹדָיָה וְשִׁיר

📜 הַמִּדְרָשׁ בִּשְׁלֵמוּתוֹ

רַבִּי חָמָא בְּשֵׁם רַבִּי חֲנִינָא: קַיִן יָצָא שָׂמֵחַ, כִּי תְּשׁוּבָתוֹ נִתְקַבְּלָה. פָּגַע בּוֹ אָדָם הָרִאשׁוֹן: ׳מָה נַעֲשָׂה בְּדִינְךָ?׳ – ׳עָשִׂיתִי תְּשׁוּבָה וְנִתְפַּשַּׁרְתִּי׳.

הִתְחִיל אָדָם מְטַפֵּחַ עַל פָּנָיו: ׳כָּךְ הִיא כֹּחָהּ שֶׁל תְּשׁוּבָה, וַאֲנִי לֹא הָיִיתִי יוֹדֵעַ?!׳ מִיָּד עָמַד וְאָמַר: ׳מִזְמוֹר שִׁיר לְיוֹם הַשַּׁבָּת׳.

רַבִּי לֵוִי: אָדָם אֲמָרוֹ, וְנִשְׁכַּח מִדּוֹרוֹ, וּבָא מֹשֶׁה וְחִדְּשׁוֹ עַל שְׁמוֹ.

הַמִּזְמוֹר נוֹלַד מִתְּשׁוּבָה, וּמֹשֶׁה חִדֵּשׁ אוֹתוֹ.

1
❓

💛 אַהֲבַת הַשֵּׁם: בְּשַׁבָּת אֲנִי עוֹצֵר וּמַרְגִּישׁ אֶת הַטּוֹב שֶׁבָּעוֹלָם 💛

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בִּיאוּר מִזְמוֹר שִׁיר לְיוֹם הַשַּׁבָּת

רָמָה ב׳ – תְּשׁוּבָה, הוֹדָיָה וְשִׁיר

❓ מָה הַקֶּשֶׁר?

מְפָרְשִׁים שׁוֹאֲלִים שְׁתֵּי שְׁאֵלוֹת: מָה הַקֶּשֶׁר בֵּין תְּשׁוּבָתוֹ שֶׁל אָדָם לַמִּזְמוֹר הַזֶּה? וְאֵיךְ הוּא קָשׁוּר לְשַׁבָּת, אִם לֹא כָּתוּב בּוֹ דָּבָר עַל הַשַּׁבָּת?

הַתְּשׁוּבָה: הַמִּזְמוֹר הוּא הוֹדָיָה עַל הַבְּרִיאָה הַנִּפְלָאָה שֶׁה׳ יָצַר לְטוֹבָתֵנוּ, וְזוֹ בְּדִיּוּק מַטָּרַת הַשַּׁבָּת.

בַּשַּׁבָּת עוֹצְרִים וּמַרְגִּישִׁים אֶת הַטּוֹב וְהַחֶסֶד שֶׁבָּעוֹלָם.

הַשִּׁיר הוּא הוֹדָיָה עַל הַבְּרִיאָה, וְזוֹ מַטָּרַת הַשַּׁבָּת.

2
🔑

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְהַכִּיר בַּטּוֹב כְּדֵי שֶׁלֹּא אֶטְעֶה 💛

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בִּיאוּר מִזְמוֹר שִׁיר לְיוֹם הַשַּׁבָּת

רָמָה ב׳ – תְּשׁוּבָה, הוֹדָיָה וְשִׁיר

🔑 שֹׁרֶשׁ הַטָּעוּת

אָדָם שָׁב בִּתְשׁוּבָה עַל שֹׁרֶשׁ הַטָּעוּת שֶׁלּוֹ: חוֹסֶר הַכָּרָה בְּטוּב ה׳. אִלּוּ הִכִּיר בְּכָל הַטּוֹב שֶׁעָשָׂה עִמּוֹ ה׳, לֹא הָיָה טוֹעֶה.

עַל זֶה שָׁר: ׳טוֹב לְהֹדוֹת לַה׳… כִּי שִׂמַּחְתַּנִי ה׳ בְּפָעֳלֶךָ׳.

וּבַמִּדְרָשׁ מוּבָא שֶׁהַשֹּׁרֶשׁ הַזֶּה הוּא שֹׁרֶשׁ כָּל טָעוּת.

חוֹסֶר הַכָּרָה בַּטּוֹב הוּא שֹׁרֶשׁ כָּל טָעוּת.

3
🌍

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׁר, מוֹדֶה וּמְתַקֵּן 💛

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִדְרַשׁ שׁוֹחֵר טוֹב, תְּהִלִּים צ״ב

רָמָה ב׳ – תְּשׁוּבָה, הוֹדָיָה וְשִׁיר

🌍 בִּי יִלְמְדוּ כָּל הַדּוֹרוֹת

אָמַר אָדָם: ׳בִּי יִלָּמְדוּ כָּל הַדּוֹרוֹת שֶׁכָּל מִי שֶׁמְּשׁוֹרֵר וּמְזַמֵּר לְשֵׁם עֶלְיוֹן וּמוֹדֶה עַל פְּשָׁעָיו וְעוֹזֵב אוֹתָם – נִצּוֹל מִן הַדִּין הַקָּשֶׁה׳.

מִכָּאן שֶׁשִּׁיר שֶׁל הוֹדָיָה וּתְשׁוּבָה הוֹלְכִים יַחַד.

מִי שֶׁשָּׁר לַה׳ וּמְתַקֵּן אֶת דְּרָכָיו, הוּא בְּדֶרֶךְ טוֹבָה.

שִׁיר, הוֹדָיָה וּתְשׁוּבָה הוֹלְכִים יַחַד.

4
🩺

💛 אַהֲבַת הַשֵּׁם: אֲנִי סוֹמֵךְ עַל ה׳ שֶׁהַכֹּל לְטוֹבָתִי 💛

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִדְרַשׁ שׁוֹחֵר טוֹב, תְּהִלִּים צ״ב

רָמָה ב׳ – תְּשׁוּבָה, הוֹדָיָה וְשִׁיר

🩺 הָרוֹפֵא וְהַחוֹלֶה

הַמִּדְרָשׁ מְסַפֵּר שֶׁאָדָם הֵבִין שֶׁהַהוֹרָאוֹת שֶׁל ה׳ הֵן לְטוֹבָתֵנוּ, כְּמוֹ עֵצָה שֶׁל רוֹפֵא לְחוֹלֶה: ׳דָּבָר פְּלוֹנִי אַל תֹּאכַל, הוּא מַזִּיק לְךָ׳.

מִי שֶׁלֹּא שָׁמַע לָרוֹפֵא, לֹא יָכוֹל לְהַאֲשִׁים אֶת הָרוֹפֵא. אָדָם אָמַר: ׳אֲנִי גָּרַמְתִּי לְעַצְמִי׳.

לָכֵן כָּתוּב בַּמִּזְמוֹר: ׳לְהַגִּיד כִּי יָשָׁר ה׳׳ – כָּל הוֹרָאוֹתָיו טוֹבוֹת לָנוּ.

הוֹרָאוֹת ה׳ הֵן עֵצָה טוֹבָה, כְּמוֹ שֶׁל רוֹפֵא.

5
🍽️

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי שָׁר בַּשַּׁבָּת, אֲנִי אוֹהֵב אֶת ה׳ יוֹתֵר 💛

📖 כְּבוֹד שַׁבָּת וְעֹנֶג שַׁבָּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שַׁבָּת וְשִׁירָה

רָמָה ב׳ – תְּשׁוּבָה, הוֹדָיָה וְשִׁיר

🍽️ שִׁיר בַּסְּעוּדָה

הַמִּזְמוֹר הוּא ׳שִׁיר לְיוֹם הַשַּׁבָּת׳. בְּבֵית הַמִּקְדָּשׁ הַלְוִיִּם שָׁרוּ אוֹתוֹ בְּכָל שַׁבָּת, וּבְיִשְׂרָאֵל נָהֲגוּ לָשִׁיר בִּסְעוּדוֹת הַשַּׁבָּת בְּכָל הַדּוֹרוֹת.

הַהוֹדָיָה מִן הַלֵּב בַּשַּׁבָּת הִיא מַטָּרַת הַיּוֹם.

כְּשֶׁשָּׁרִים בְּשִׂמְחָה, מַרְגִּישִׁים אֶת הַטּוֹב וְאוֹהֲבִים אֶת מִי שֶׁנָּתַן אוֹתוֹ.

שִׁיר שֶׁל הוֹדָיָה הוּא לֵב הַשַּׁבָּת.

מָה לוֹקְחִים מִרָמָה ב׳?

שֹׁרֶשׁ כָּל טָעוּת הוּא חוֹסֶר הַכָּרָה בְּטוּב ה׳, וְהַתִּקּוּן הוּא הוֹדָיָה. מִי שֶׁשָּׁר לַה׳ מִתּוֹךְ הַכָּרַת הַטּוֹב, מַרְגִּישׁ אֶת אַהֲבָתוֹ וְאוֹהֵב אוֹתוֹ.

🤔 חוֹשְׁבִים עַל זֶה

  • עַל אֵיזוֹ טוֹבָה שֶׁקִּבַּלְתִּי הַשָּׁבוּעַ אֲנִי רוֹצֶה לְהוֹדוֹת לַה׳?
  • מָתַי הִרְגַּשְׁתִּי הֲקָלָה אַחֲרֵי שֶׁתִּקַּנְתִּי טָעוּת, וְאֵיךְ זֶה קָשׁוּר לְשִׁיר שֶׁל הוֹדָיָה?
  • בְּשַׁבָּת הַקְּרוֹבָה אֲנַסֶּה לָשִׁיר יַחַד עִם הַמִּשְׁפָּחָה ׳טוֹב לְהוֹדוֹת׳, וְאֶרְשֹׁם מָה הִרְגַּשְׁתִּי.

סֻלָּם הַיִּרְאָה

רָמָה א׳ – הַסֻּלָּם

שָׁלוֹשׁ מַדְרֵגוֹת: יִרְאַת הָעֹנֶשׁ, יִרְאַת הָרוֹמְמוּת וְיִרְאַת הַחֶסֶד, וְאִי אֶפְשָׁר לְדַלֵּג עַל הָרִאשׁוֹנָה

📖 יִרְאַת שָׁמַיִם · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🪜

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹלֶה בְּסֻלַּם הַיִּרְאָה צַעַד אַחַר צַעַד 💛

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׁעוּר ׳סֻלָּם יִרְאַת שָׁמַיִם׳

רָמָה א׳ – הַסֻּלָּם

🪜 מָשָׁל הַסֻּלָּם

יֶלֶד רוֹצֶה לְהַגִּיעַ לַמַּדָּף הַגָּבוֹהַּ שֶׁבַּסַּלּוֹן. הוּא לֹא יָכוֹל לִקְפֹּץ אֵלָיו, וְגַם לֹא לְדַלֵּג עַל הַמַּדְרֵגוֹת הַנְּמוּכוֹת. הוּא צָרִיךְ לַעֲלוֹת מַדְרֵגָה אַחַר מַדְרֵגָה.

כָּךְ גַּם יִרְאַת שָׁמַיִם: הִיא כְּמוֹ סֻלָּם. אִי אֶפְשָׁר לְהַגִּיעַ לַשָּׁלָב הָעֶלְיוֹן בְּלִי לַעֲבֹר קֹדֶם בַּשָּׁלָב הַתַּחְתּוֹן.

אִי אֶפְשָׁר לְהַגִּיעַ לַשָּׁלָב הָעֶלְיוֹן בְּלִי הַתַּחְתּוֹן.

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׁעוּר ׳סֻלָּם יִרְאַת שָׁמַיִם׳

3️⃣ שָׁלוֹשׁ מַדְרֵגוֹת

בְּסֻלָּם הַיִּרְאָה יֵשׁ שָׁלוֹשׁ מַדְרֵגוֹת.

הַתַּחְתּוֹנָה: יִרְאַת הָעֹנֶשׁ.

הָאֶמְצָעִית: יִרְאַת הָרוֹמְמוּת.

וְהַשְּׁלִישִׁית: יִרְאָה שֶׁבָּאָה מֵהִתְבּוֹנְנוּת בְּחַסְדֵי ה׳.

בַּכַּרְטִיסִים הַבָּאִים נַכִּיר כׇּל מַדְרֵגָה.

שָׁלוֹשׁ מַדְרֵגוֹת: עֹנֶשׁ, רוֹמְמוּת וָחֶסֶד.

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַתְחִיל מֵהַמַּדְרֵגָה הָרִאשׁוֹנָה 💛

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׁעוּר ׳סֻלָּם יִרְאַת שָׁמַיִם׳

רָמָה א׳ – הַסֻּלָּם

⚠️ הַמַּדְרֵגָה הַתַּחְתּוֹנָה

הַמַּדְרֵגָה הָרִאשׁוֹנָה וְהַיְּסוֹדִית בְּיוֹתֵר הִיא יִרְאַת הָעֹנֶשׁ: אֲנִי נִזְהָר שֶׁלֹּא לַעֲבֹר עַל מִצְווֹת ה׳, כִּי אֲנִי יוֹדֵעַ שֶׁזֶּה חָשׁוּב.

זֹאת הַנְּקֻדָּה שֶׁמִּמֶּנָּה מַתְחִיל הַכֹּל. בִּלְעָדֶיהָ אִי אֶפְשָׁר לְהַגִּיעַ לְשׁוּם מַדְרֵגָה גְּבוֹהָה יוֹתֵר.

יִרְאַת הָעֹנֶשׁ: הַנְּקֻדָּה שֶׁמִּמֶּנָּה מַתְחִילִים.

📖

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד הֲלָכוֹת כְּדֵי לְקַיֵּם אֶת רְצוֹן ה׳ 💛

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׁעוּר ׳סֻלָּם יִרְאַת שָׁמַיִם׳

רָמָה א׳ – הַסֻּלָּם

📖 צָרִיךְ לָדַעַת

כְּדֵי שֶׁיִּרְאַת הָעֹנֶשׁ תִּהְיֶה אֲמִתִּית, צָרִיךְ קֹדֶם לָדַעַת אֶת הַהֲלָכוֹת: מָה כָּתוּב בַּתּוֹרָה, מָה אָסוּר וּמָה מֻתָּר.

מִי שֶׁלֹּא יוֹדֵעַ מָה ה׳ מְבַקֵּשׁ מִמֶּנּוּ, לֹא יָכוֹל לְהִזָּהֵר, כִּי הוּא לֹא יוֹדֵעַ אֵיפֹה הַגְּבוּל.

זֶה הָאָלֶף-בֵּית: לִלְמֹד אֶת הַהֲלָכָה, וְרַק אַחַר כָּךְ לְהִתְקַדֵּם.

קֹדֶם לוֹמְדִים מָה ה׳ מְבַקֵּשׁ.

🌌

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִסְתַּכֵּל עַל הָעוֹלָם וּמַרְגִּישׁ אֶת גְּדֻלַּת ה׳ 💛

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׁעוּר ׳סֻלָּם יִרְאַת שָׁמַיִם׳

רָמָה א׳ – הַסֻּלָּם

🌌 יִרְאַת הָרוֹמְמוּת

מֵעַל יִרְאַת הָעֹנֶשׁ נִמְצֵאת מַדְרֵגָה גְּבוֹהָה יוֹתֵר: יִרְאַת הָרוֹמְמוּת.

אָדָם מִתְבּוֹנֵן אֵיךְ הַקָּדוֹשׁ בָּרוּךְ הוּא בָּרָא אֶת כׇּל הָעוֹלָם בְּמַאֲמַר פִּיו, וּמְחַדֵּשׁ אוֹתוֹ בְּכׇל רֶגַע וָרֶגַע.

כְּשֶׁהוּא מִסְתַּכֵּל עַל הַגְּדֻלָּה הַזֹּאת, הוּא מַרְגִּישׁ כַּמָּה הוּא עַצְמוֹ קָטָן לְעֻמָּתָהּ.

יִרְאַת הָרוֹמְמוּת: מִתְבּוֹנְנִים בִּגְדֻלַּת הַבּוֹרֵא.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹאֶה אֶת הַחֲסָדִים שֶׁל ה׳ וְאוֹהֵב אוֹתוֹ 💛

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׁעוּר ׳סֻלָּם יִרְאַת שָׁמַיִם׳

רָמָה א׳ – הַסֻּלָּם

🌞 יִרְאַת הַחֶסֶד

יֵשׁ עוֹד מַדְרֵגָה, וְגַם הִיא נִקְרֵאת יִרְאַת שָׁמַיִם: יִרְאָה שֶׁבָּאָה מֵהִתְבּוֹנְנוּת בְּחַסְדֵּי ה׳.

מִי שֶׁמִּסְתַּכֵּל כַּמָּה הַקָּדוֹשׁ בָּרוּךְ הוּא רַב חֶסֶד, וְרוֹאֶה שֶׁכׇּל מַה שֶּׁהוּא עוֹשֶׂה אִתָּנוּ הוּא חֶסֶד, מַרְגִּישׁ גַּם הוּא יִרְאָה.

זֶה נִשְׁמָע כְּמוֹ אַהֲבָה, אֲבָל גַּם הַהִתְבּוֹנְנוּת בְּגֹדֶל הַחֶסֶד מוֹלִידָה הַכְנָעָה וְיִרְאָה.

יִרְאַת הַחֶסֶד: מִתְבּוֹנְנִים בְּחַסְדֵּי ה׳.

🏢

💛 אַהֲבַת הַשֵּׁם: אֲפִלּוּ הַגְּדוֹלִים זְקוּקִים לַיְּסוֹד, וְגַם אֲנִי 💛

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׁעוּר ׳כַּנְפֵי הַתּוֹרָה׳

רָמָה א׳ – הַסֻּלָּם

🏢 שְׁתֵּי קוֹמוֹת

הַיִּרְאָה כְּמוֹ בִּנְיָן שֶׁל שְׁתֵּי קוֹמוֹת. אַהֲרֹן הַכֹּהֵן הָיָה צַדִּיק הַדּוֹר, וְעִם זֶה ה׳ אָמַר לוֹ שֶׁלֹּא יִכָּנֵס אֶל הַקֹּדֶשׁ בְּכׇל עֵת.

מִכָּאן לוֹמְדִים: כׇּל אָדָם צָרִיךְ יִרְאַת שָׁמַיִם, לֹא מְשַׁנֶּה בְּאֵיזוֹ מַדְרֵגָה הוּא נִמְצָא.

אֵין אָדָם שֶׁפָּטוּר מֵהַקּוֹמָה הָרִאשׁוֹנָה, גַּם לֹא הַגָּדוֹל שֶׁבַּדּוֹר.

גַּם הַגָּדוֹל שֶׁבַּדּוֹר צָרִיךְ אֶת הַקּוֹמָה הָרִאשׁוֹנָה.

💛 אַהֲבַת הַשֵּׁם: אֲנִי יוֹדֵעַ אֵיפֹה אֲנִי עַל הַסֻּלָּם וּמַמְשִׁיךְ לַעֲלוֹת 💛

📖 יִרְאַת שָׁמַיִם – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

📍 אֵיפֹה אֲנִי?

עַכְשָׁיו אֶפְשָׁר לִשְׁאֹל: אֵיפֹה אֲנִי עַל הַסֻּלָּם?

אִם אֲנִי עוֹד בַּמַּדְרֵגָה הַתַּחְתּוֹנָה, זֶה בְּסֵדֶר גָּמוּר. כָּל אֶחָד מַתְחִיל מִשָּׁם.

וְאִם אֲנִי מַרְגִּישׁ בְּתוֹכִי עַל הָאֶמְצָעִית אוֹ הָעֶלְיוֹנָה, אֲנִי עֲדַיִן זָקוּק גַּם לַתַּחְתּוֹנָה.

כָּל אֶחָד מַתְחִיל מֵהַמַּדְרֵגָה הָרִאשׁוֹנָה.

מָה לוֹקְחִים מִרָמָה א׳?

סֻלַּם הַיִּרְאָה מוֹלִיךְ אֶל ה׳ צַעַד אַחַר צַעַד. כׇּל מַדְרֵגָה מְקָרֶבֶת אוֹתִי, וּבַסּוֹף הָאַהֲבָה וְהַיִּרְאָה נִפְגָּשׁוֹת בַּלֵּב.

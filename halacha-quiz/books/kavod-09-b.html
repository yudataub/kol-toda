כְּבוֹד שָׁמַיִם וְכְבוֹד הַמִּצְווֹת

רָמָה ב׳ – דְּרָכִים לְכַבֵּד

מָה זֶה כָּבוֹד לַקָּדוֹשׁ בָּרוּךְ הוּא, וְאֵיךְ אֶפְשָׁר לְכַבֵּד מִצְווֹת בְּשִׂמְחָה, בַּזְּמַן וּבְהִדּוּר

📖 כָּבוֹד הַבְּרִיּוֹת · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
👑

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִתְפַּלֵּל כְּדֵי לְהַכִּיר בְּגֹדֶל ה׳ 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – דְּרָכִים לְכַבֵּד

👑 כָּבוֹד הוּא הַכָּרָה בְּמַעֲלָה

מַהוּ ׳כְּבוֹד שָׁמַיִם׳? יֵשׁ לָדַעַת יְסוֹד: כְּשֶׁאָנוּ אוֹמְרִים שֶׁה׳ בָּרָא אֶת הַכֹּל לִכְבוֹדוֹ, אֵין הַכַּוָּנָה שֶׁהוּא צָרִיךְ כָּבוֹד, חַס וְשָׁלוֹם.

כָּבוֹד פֵּרוּשׁוֹ הַכָּרָה בְּמַעֲלָה. מַטָּרַת הָעוֹלָם שֶׁבְּנֵי אָדָם יַכִּירוּ בְּה׳, וְזֶה נִקְרָא ׳לִכְבוֹדוֹ׳.

לָכֵן כָּל תְּפִלַּת שַׁחֲרִית סוֹבֶבֶת סְבִיב הַכָּבוֹד: ׳יְהִי כְבוֹד׳, ׳הוֹד וְהָדָר׳, ׳כְּבוֹד מַלְכוּתְךָ יֹאמֵרוּ׳.

כְּבוֹד שָׁמַיִם = שֶׁאֲנָשִׁים מַכִּירִים בְּמַעֲלַת ה׳.

1
🗣️

💛 אַהֲבַת הַשֵּׁם: אֲנִי שׁוֹמֵר עַל הַדִּבּוּר, כְּדֵי לְקַדֵּשׁ אֶת ה׳ 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַבֵּנוּ יוֹנָה

רָמָה ב׳ – דְּרָכִים לְכַבֵּד

🗣️ לְקַדֵּשׁ אֶת ה׳ בַּדִּבּוּר

רַבֵּנוּ יוֹנָה כּוֹתֵב שֶׁכְּשֶׁאָדָם מִתְיַצֵּב בְּתוֹךְ הָעָם וּמְדַבֵּר עִם חֲבֵרָיו, עָלָיו לְהִתְבּוֹנֵן וְלִשְׁמֹר עַל מוֹצָא שְׂפָתָיו, לְקַדֵּשׁ אֶת ה׳ בִּדְבָרָיו וְלָדַבֵּר בְּשֶׁבַח עֲבוֹדָתוֹ וְיִרְאָיו.

בְּאֹפֶן כָּזֶה הוּא זוֹכֶה לִזְכוּת גְּדוֹלָה, כִּי זֶה מֵעִקְרֵי יְצִירַת הָאָדָם.

מֵעַל לְכָל שִׂיחָה, אֶפְשָׁר לִשְׁאֹל: מָה אֲנִי בּוֹחֵר לְהַגִּיד, וְעַל מָה אֲנִי מְשַׁבֵּחַ?

הַדִּבּוּר שֶׁלָּנוּ יָכוֹל לְהַרְבּוֹת כְּבוֹד שָׁמַיִם.

2
😊

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ וּמוֹדֶה, וְכָךְ אֲנִי מְכַבֵּד אֶת ה׳ 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – דְּרָכִים לְכַבֵּד

😊 שִׂמְחָה הִיא כָּבוֹד

בַּכֹּל בּוֹ כָּתוּב שֶׁכְּבוֹד ה׳ הוּא בְּשִׂמְחַת הַנֶּפֶשׁ. כְּשֶׁאָדָם שָׂמֵחַ שֶׁזָּכָה לְקַיֵּם מִצְוָה, הוּא מַרְאֶה כַּמָּה הִיא חֲשׁוּבָה.

וְגַם הוֹדָיָה לַה׳ עַל הַטּוֹב שֶׁנָּתַן לָנוּ נוֹתֶנֶת כָּבוֹד, כִּי כָּךְ אָנוּ מַכִּירִים שֶׁה׳ טוֹב.

לָכֵן חֲשׁוּבָה לָנוּ הַשִּׂמְחָה: הִיא מַרְאָה שֶׁהַמִּצְוָה יְקָרָה לָנוּ.

שִׂמְחָה וְהוֹדָיָה הֵן דֶּרֶךְ לִתֵּן כָּבוֹד לַה׳.

🎁

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְכַבֵּד מִצְווֹת בְּמַעֲשִׂים, לֹא רַק בְּמִלִּים 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – דְּרָכִים לְכַבֵּד

🎁 אֵיךְ מְכַבְּדִים מִצְווֹת?

יֵשׁ דְּרָכִים מַעֲשִׂיּוֹת לְכַבֵּד מִצְווֹת. רִאשׁוֹנָה: לְהַגִּיעַ לַשִּׁעוּר בַּזְּמַן שֶׁהוּא מַתְחִיל, כִּי זֶה מַרְאֶה חֲבִיבוּת.

שְׁנִיָּה: לַעֲשׂוֹת אֶת הַמִּצְוָה בְּהִדּוּר, עִם כֵּלִים יָפִים.

שְׁלִישִׁית: לְסַכֵּם אֶת הַתּוֹרָה, לַחֲזֹר עָלֶיהָ, וּלְהִתְבּוֹנֵן בְּסוֹף הַסֵּדֶר מָה יָפֶה בְּמָה שֶׁלָּמַדְנוּ.

כָּל אֵלֶּה אוֹמְרִים: ׳הַמִּצְוָה הַזּוֹ יְקָרָה לִי׳.

בַּזְּמַן, בְּהִדּוּר, וּבְסִכּוּם – כָּךְ מְכַבְּדִים.

4
🌟

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְשַׁבֵּחַ אֲחֵרִים, וְכָךְ מַרְבֶּה כְּבוֹד שָׁמַיִם 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – דְּרָכִים לְכַבֵּד

🌟 לְשַׁבֵּחַ אֲחֵרִים

מְלַמְּדִים שֶׁלֹּא כְּדַאי לְהַפְחִיד אֲנָשִׁים מִמִּצְווֹת, אֶלָּא לוֹמַר לָהֶם כַּמָּה הַמִּצְווֹת שֶׁהֵם עוֹשִׂים חֲשׁוּבוֹת. כְּשֶׁאָדָם מַרְגִּישׁ מְאֻשָּׁר שֶׁהוּא זוֹכֶה לַעֲשׂוֹת עוֹד מִצְווֹת, הוּא מִתְקַדֵּם.

כְּשֶׁגּוֹרְמִים לְאָדָם לְהַעֲרִיךְ מִצְוָה, זֶה עַצְמוֹ כְּבוֹד שָׁמַיִם.

כָּךְ מְחַזְּקִים חֲבֵרִים וּמַרְבִּים כְּבוֹד ה׳.

שִׁבּוּחַ אֲמִתִּי מוֹסִיף חֲשִׁיבוּת לַמִּצְוָה.

💛 אַהֲבַת הַשֵּׁם: אֲנִי נוֹתֵן דֻּגְמָה טוֹבָה לַאֲחֵרִים 💛

📖 כְּבוֹד הַבְּרִיּוֹת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, יוֹמָא (סוֹף הַמַּסֶּכֶת)

רָמָה ב׳ – דְּרָכִים לְכַבֵּד

⚠️ לֹא מְזַלְזְלִים

הַגְּמָרָא בְּסוֹף מַסֶּכֶת יוֹמָא נוֹתֶנֶת דֻּגְמָה: אִם אָדָם גָּדוֹל, כְּמוֹ רַבִּי יוֹחָנָן, הָיָה הוֹלֵךְ אַרְבַּע אַמּוֹת בְּלִי תְּפִלִּין, אֲנָשִׁים הָיוּ לוֹמְדִים לְזַלְזֵל בַּמִּצְוָה.

הַלֶּקַח: מַה שֶּׁאָנוּ עוֹשִׂים נִרְאֶה לַאֲחֵרִים, וְהוּא יָכוֹל לְהַעֲלוֹת אֶת הַמִּצְוָה בְּעֵינֵיהֶם אוֹ לְהוֹרִיד אוֹתָהּ.

לָכֵן מִי שֶׁמְּכַבֵּד מִצְווֹת, דּוֹאֵג גַּם לְהוֹרוֹת לַאֲחֵרִים כַּמָּה הֵן יָקָרוֹת.

הַמַּעֲשִׂים שֶׁלָּנוּ מְלַמְּדִים אֲחֵרִים כַּמָּה הַמִּצְוָה חֲשׁוּבָה.

מָה לוֹקְחִים מִרָמָה ב׳?

הַכָּבוֹד לַה׳ אֵינוֹ רַק בְּמִלִּים: הוּא נִרְאֶה בְּשִׂמְחָה, בַּהַגָּעָה בַּזְּמַן, בְּהִדּוּר וּבְשִׁבּוּחַ לַאֲחֵרִים. כָּל אֵלֶּה מַרְאִים שֶׁאֲנִי אוֹהֵב אֶת ה׳ וְאֶת מִצְווֹתָיו.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה אֲנִי עוֹשֶׂה הַשָּׁבוּעַ שֶׁמַּרְאֶה שֶׁמִּצְוָה אַחַת חֲשׁוּבָה לִי?
  • מִצְוָה אַחַת שֶׁאֲנִי יָכוֹל לַעֲשׂוֹת בְּהִדּוּר: אֵיזוֹ וְאֵיךְ?
  • אֶבְחַר חָבֵר שֶׁעָשָׂה מִצְוָה, אוֹמַר לוֹ בְּהִתְלַהֲבוּת כַּמָּה זֶה חָשׁוּב, וְאֶרְשֹׁם מָה הוּא הִרְגִּישׁ.

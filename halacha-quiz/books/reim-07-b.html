רָמָה ב׳ – לִרְאוֹת אֶת הָאָדָם

⚖️ לָמָּה שִׂנְאַת חִנָּם חֲמוּרָה?

שִׂנְאַת חִנָּם אֵינָהּ רַק רֶגֶשׁ לֹא נָעִים. זֶה מַצָּב שֶׁבּוֹ אָדָם מַחְלִיט שֶׁהַשֵּׁנִי אֵינוֹ רָאוּי לְמָקוֹם, לְהַקְשָׁבָה אוֹ לְטוֹבָה, בְּלִי סִבָּה שֶׁמַּצְדִּיקָה זֹאת. כְּשֶׁזֶּה נַעֲשֶׂה הֶרְגֵּל, כָּל מַעֲשֶׂה שֶׁל הַשֵּׁנִי נִקְרָא לְרַע, וְכָל טָעוּת הוֹפֶכֶת לְהוֹכָחָה שֶׁהוּא פָּסוּל.

לָכֵן חֲכָמִים רָאוּ בְּפֵרוּד סַכָּנָה גְּדוֹלָה. בַּיִת יָכוֹל לִהְיוֹת מָלֵא תּוֹרָה וּמַעֲשִׂים, אֲבָל אִם בְּנֵי הַבַּיִת מְבַזִּים זֶה אֶת זֶה, מַשֶּׁהוּ בַּיְּסוֹד נִשְׁבָּר.

שִׂנְאַת חִנָּם הִיא לְהַחְלִיט שֶׁהַשֵּׁנִי לֹא רָאוּי לְטוֹבָה, בְּלִי סִבָּה.

רָמָה ב׳ – לִרְאוֹת אֶת הָאָדָם

🔥 הָאֵשׁ שֶׁנִּדְלֶקֶת מִמִּלָּה

רִיב מַתְחִיל לִפְעָמִים בְּמִלָּה קְצָרָה, אֲבָל מִיָּד אָנוּ מְסַפְּרִים לְעַצְמֵנוּ סִפּוּר שָׁלֵם: ׳הוּא מְזַלְזֵל בִּי, הִיא תָּמִיד עוֹשָׂה זֹאת, אֵין עִם מִי לְדַבֵּר׳. הַסִּפּוּר מַדְלִיק אֶת הָאֵשׁ יוֹתֵר מִן הָאֵרוּעַ הַמְקוֹרִי.

אֶפְשָׁר לַעֲצֹר וְלִשְׁאֹל: מָה אֲנִי יוֹדֵעַ בְּוַדַּאי, וּמָה אֲנִי מַשְׁלִים בְּעַצְמִי? אוּלַי הָאָדָם הָיָה עָיֵף, אוּלַי נִסַּח אֶת עַצְמוֹ גָּרוּעַ, אוּלַי יֵשׁ פֶּצַע יָשָׁן שֶׁצּוֹבֵעַ אֶת הַהוֹוֶה. הַשְּׁאֵלָה לֹא מְבַטֶּלֶת פְּגִיעָה אֲמִתִּית, אֲבָל נוֹתֶנֶת סִכּוּי לְהָגִיב בְּמִדָּה וְלֹא מִתּוֹךְ לֶהָבָה.

מַחְלֹקֶת מִתְעַצֶּמֶת כְּשֶׁמְּגִיבִים לְכַוָּנָה מְדֻמְיֶנֶת, וְלֹא לְעוּבְדוֹת.

רָמָה ב׳ – לִרְאוֹת אֶת הָאָדָם

👁️ לִרְאוֹת אֶת הָאָדָם

אָדָם אֵינוֹ רַק הַמַּעֲשֶׂה שֶׁהִרְגִּיז אוֹתָנוּ. יֵשׁ לוֹ פְּחָדִים, מַאֲמָצִים, עָבָר וְתִקְווֹת שֶׁאֲנַחְנוּ לֹא רוֹאִים. כְּשֶׁאָנוּ שׁוֹאֲלִים וּמַקְשִׁיבִים לַתְּשׁוּבָה, אָנוּ מְקַבְּלִים תְּמוּנָה רְחָבָה יוֹתֵר. לִפְעָמִים נִגְלֶה שֶׁהֵבַנּוּ נָכוֹן, וְלִפְעָמִים שֶׁהֶחְמַצְנוּ כִּמְעַט הַכֹּל.

הַקְשָׁבָה אֵינָהּ וִתּוּר עַל גְּבוּלוֹת: אֶפְשָׁר לוֹמַר שֶׁמַּעֲשֶׂה מְסֻיָּם פָּגַע בִּי, וּבְכָל זֹאת לְהַכִּיר בִּכְבוֹדוֹ שֶׁל מִי שֶׁעָשָׂה אוֹתוֹ. כָּךְ מְתַקְּנִים בְּלִי לַהֲפֹךְ אֶת הַשֵּׁנִי לְאוֹיֵב קָבוּעַ.

הַקְשָׁבָה אֵינָהּ וִתּוּר עַל גְּבוּלוֹת, אֶלָּא הַכָּרָה בִּכְבוֹד הָאָדָם.

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אֶת כֻּלָּנוּ יַחַד, גַּם כְּשֶׁאָנוּ שׁוֹנִים 💛

רָמָה ב׳ – לִרְאוֹת אֶת הָאָדָם

🌈 אַחְדוּת אֵינָהּ מְחִיקָה

יֵשׁ פִּתּוּי לַחְשֹׁב שֶׁאִם כֻּלָּנוּ נִהְיֶה זֵהִים, לֹא יִהְיוּ מְרִיבוֹת. אֲבָל חֶבְרָה בְּרִיאָה כּוֹלֶלֶת אֲנָשִׁים שׁוֹנִים, בַּעֲלֵי נִסָּיוֹן, אֹפִי וְדֵעוֹת מְגֻוָּנוֹת. הַשְּׁאֵלָה אֵינָהּ אֵיךְ לְהַעֲלִים אֶת הַהֶבְדֵּל, אֶלָּא אֵיךְ מַחְזִיקִים אוֹתוֹ בְּלִי לְבַזּוֹת.

אֶפְשָׁר לְהִתְוַכֵּחַ עַל רַעְיוֹן וְלִזָּהֵר שֶׁלֹּא לַהֲפֹךְ אֶת הַוִּכּוּחַ לְמִשְׁפָּט עַל הָאִישִׁיּוּת. אֶפְשָׁר לִלְמֹד מִן הַשֵּׁנִי גַּם כְּשֶׁלֹּא מִשְׁתַּכְנְעִים. אַחְדוּת הִיא יְכֹלֶת לְהִשָּׁאֵר בְּקֶשֶׁר גַּם כְּשֶׁאֵין הַסְכָּמָה.

אַחְדוּת אֵינָהּ מְחִיקָה שֶׁל דֵּעוֹת, אֶלָּא קֶשֶׁר גַּם בְּלִי הַסְכָּמָה.

רָמָה ב׳ – לִרְאוֹת אֶת הָאָדָם

🌉 מַחֲוָה שֶׁפּוֹתַחַת גֶּשֶׁר

לֹא תָּמִיד אֶפְשָׁר לִפְתֹּר סִכְסוּךְ בְּשִׂיחָה אַחַת, אֲבָל כִּמְעַט תָּמִיד אֶפְשָׁר לִבְחֹר מַחֲוָה שֶׁאֵינָהּ מוֹסִיפָה אֵשׁ: לְבָרֵךְ לְשָׁלוֹם, לְהַחְזִיר חֵפֶץ, לְהוֹדוֹת בַּחֵלֶק שֶׁלָּנוּ, אוֹ לְהִמָּנַע מִמִּשְׁפָּט מִיּוּתָּר. הַמַּחֲוָה אֵינָהּ אוֹמֶרֶת שֶׁהַכֹּל נִפְתַּר; הִיא אוֹמֶרֶת שֶׁאֵין אָנוּ מוּכָנִים לָתֵת לַשִּׂנְאָה לְנַהֵל אֶת הֶעָתִיד.

אַהֲבַת חִנָּם נִבְנֵית מֵהַרְבֵּה רְגָעִים כָּאֵלֶּה: לִרְאוֹת אֶת הַטּוֹב, לְבַקֵּשׁ הֶבְהָרָה לִפְנֵי שֶׁכּוֹעֲסִים, וְלִזְכֹּר שֶׁהַזּוּלַת נִבְרָא בְּצֶלֶם. צַעַד קָטָן שֶׁל כָּבוֹד יָכוֹל לְהַפְסִיק מַעְגָּל שֶׁל עוֹיְנוּת.

מַחֲוָה קְטַנָּה אֵינָהּ פּוֹתֶרֶת הַכֹּל, אֲבָל לֹא מַשְׁאִירָה לַשִּׂנְאָה לְנַהֵל.

מָה לוֹקְחִים מִרָמָה ב׳?

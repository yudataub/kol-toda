רִקּוּד עַל הַשְּׂרֵפָה

רָמָה ב׳ – הוֹדָיָה גַּם בְּיוֹם קָשֶׁה

רַבִּי שְׁמוּאֵל מוּנְקִיס, הַדָּבָר הַגָּדוֹל שֶׁאִי אֶפְשָׁר לִשְׂרֹף, וּבִרְכוֹת הַשַּׁחַר

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
🔥

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְהָבִין אֵיךְ אֶפְשָׁר לְהוֹדוֹת גַּם כְּשֶׁקָּשֶׁה 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, צַדִּיקִים מוֹדִים לַה׳ (רַבִּי שְׁמוּאֵל מוּנְקִיס)

רָמָה ב׳ – הוֹדָיָה גַּם בְּיוֹם קָשֶׁה

🔥 הַסִּפּוּר וְהַשְּׁאֵלָה

רַבִּי שְׁמוּאֵל מוּנְקִיס, מֵחֲסִידָיו שֶׁל בַּעַל הַתַּנְיָא, אִבֵּד בִּשְׂרֵפָה כִּמְעַט אֶת כָּל רְכוּשׁוֹ. הוּא הִצִּיל רַק בַּקְבּוּק אֶחָד, רָץ לְבֵית הַמִּדְרָשׁ לִשְׁתּוֹת ׳לֶחָיִים׳ וְרָקַד.

חֲבֵרָיו חָשְׁשׁוּ שֶׁהִשְׁתַּבְּשָׁה דַּעְתּוֹ מֵרֹב צַעַר.

הַשְּׁאֵלָה: אֵיךְ אָדָם יָכוֹל לָשִׁיר וְלִרְקֹד אַחֲרֵי הֶפְסֵד גָּדוֹל? הַתְּשׁוּבָה שֶׁלּוֹ מְלַמֶּדֶת אוֹתָנוּ אֵיפֹה לְחַפֵּשׂ סִבָּה לְהוֹדוֹת.

הַשְּׁאֵלָה: אֵיךְ רוֹקְדִים אַחֲרֵי הֶפְסֵד? הַתְּשׁוּבָה שֶׁלּוֹ מְלַמֶּדֶת אוֹתָנוּ.

1
💬

💛 אַהֲבַת הַשֵּׁם: הַקֶּשֶׁר שֶׁלִּי אֶל ה׳ לֹא יָכוֹל לְהִשָּׂרֵף 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, צַדִּיקִים מוֹדִים לַה׳ (רַבִּי שְׁמוּאֵל מוּנְקִיס)

רָמָה ב׳ – הוֹדָיָה גַּם בְּיוֹם קָשֶׁה

💬 הַתְּשׁוּבָה שֶׁלּוֹ

הוּא עָנָה: אִלּוּ הָיָה מִמִּי שֶׁמִּשְׁתַּחֲוִים לַהֶבֶל וָרִיק, לָאֲלִילִים, גַּם אֱלֹהָיו הָיָה נִשְׂרַף בַּשְּׂרֵפָה. אֲבָל הוּא יְהוּדִי, וֵאלֹהָיו הַגָּדוֹל חַי וְקַיָּם.

עַל זֶה בִּלְבַד רָאוּי לְהוֹדוֹת בְּכָל לֵב, וּלְכָךְ הוּא הִתְכַּוֵּן כְּשֶׁשָּׁר ׳שֶׁלֹּא עָשַׂנִי גּוֹי׳.

כָּל מָה שֶׁאָבַד בַּשְּׂרֵפָה הוּא חֵלֶק מִן הָרְכוּשׁ, וְהַקָּשֶׁר אֶל ה׳ נִשְׁאַר.

ה׳ חַי וְקַיָּם – וְעַל זֶה מוֹדִים בְּכָל לֵב.

2
🌅

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂם לֵב לַחֲסָדִים הַקְּטַנִּים שֶׁבַּבֹּקֶר 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הַבַּ״ח · בְּהוֹדָיָה צָרִיךְ לְהַאֲרִיךְ

רָמָה ב׳ – הוֹדָיָה גַּם בְּיוֹם קָשֶׁה

🌅 בְּרָכָה עַל כָּל חֶסֶד

הַבַּ״ח שָׁאַל: לָמָּה לֹא תִּקְּנוּ בְּבִרְכוֹת הַשַּׁחַר בְּרָכָה אַחַת כְּלָלִית, ׳שֶׁעָשַׂנִי יִשְׂרָאֵל׳? וְעָנָה: כִּי רוֹצִים לְהַאֲרִיךְ בְּהוֹדָאוֹת וּלְבָרֵךְ עַל כָּל חֶסֶד בִּפְנֵי עַצְמוֹ.

לָכֵן גַּם עַל קִימָה מִן הַמִּטָּה יֵשׁ בְּרָכוֹת: ׳מַתִּיר אֲסוּרִים׳ כְּשֶׁמַּנִּיעִים עֲצָמוֹת שֶׁהָיוּ כָּל הַלַּיְלָה כְּאִלּוּ קְשׁוּרוֹת, וְ׳זוֹקֵף כְּפוּפִים׳ כְּשֶׁעוֹמְדִים.

מַאֲרִיכִים בְּהוֹדָאוֹת: בְּרָכָה נִפְרֶדֶת לְכָל חֶסֶד.

3
🎯

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְפָרֵט בַּלֵּב אֶת הַחֲסָדִים שֶׁקִּבַּלְתִּי 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּהוֹדָיָה צָרִיךְ לְהַאֲרִיךְ

רָמָה ב׳ – הוֹדָיָה גַּם בְּיוֹם קָשֶׁה

🎯 לְהַרְגִּישׁ אֶת הַפְּרָטִים

מִי שֶׁמְּדַקְדֵּק בְּכַוָּנַת הַלֵּב לְפִי הַמִּלִּים, שׁוֹמֵעַ יוֹתֵר וְיוֹתֵר חֶסֶד: הַפֵּרוּט וְהַהַדְרָגָה שֶׁבִּבְרָכוֹת מְלַמְּדִים כַּמָּה הַחֶסֶד גָּדוֹל.

אֵין מִסְתַּפְּקִים בְּ׳תּוֹדָה׳ כְּלָלִית. הוֹדָאָה אֲמִתִּית נוֹגַעַת בְּכָל פְּרָט וּפְרָט.

הוֹדָאָה אֲמִתִּית נוֹגַעַת בְּכָל פְּרָט.

4
📜

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה עַל כָּל טוֹבָה, גַּם אַחַת בִּלְבַד 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּהוֹדָיָה צָרִיךְ לְהַאֲרִיךְ (מַלְבִּ״ים עַל הַהַגָּדָה)

רָמָה ב׳ – הוֹדָיָה גַּם בְּיוֹם קָשֶׁה

📜 דַּיֵּנוּ

גַּם בַּהַגָּדָה, לִפְנֵי הַהַלֵּל, מַזְכִּירִים אֶת כָּל הַטּוֹבוֹת: ׳אִלּוּ הוֹצִיאָנוּ מִמִּצְרַיִם וְלֹא עָשָׂה בָּהֶם שְׁפָטִים – דַּיֵּנוּ׳. הַמַּלְבִּ״ים מְפָרֵשׁ: כְּדֵי לִתֵּן בִּינָה בַּלֵּב עַד כַּמָּה חַיָּבִים לְהַלֵּל.

גַּם טוֹבָה אַחַת דַּי כְּדֵי לְהוֹדוֹת, וְעַל אַחַת כַּמָּה וְכַמָּה בְּקִבּוּץ כָּל הַטּוֹבוֹת יַחַד.

גַּם טוֹבָה אַחַת דַּי כְּדֵי לְהוֹדוֹת – וְעוֹד יוֹתֵר כֻּלָּן יַחַד.

5
💛

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה גַּם עַל הַגָּדוֹל וְגַם עַל הַקָּטָן 💛

📖 הַכָּרַת הַטּוֹב לַה׳ (הוֹדָיָה) – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, צַדִּיקִים מוֹדִים לַה׳ (רַבִּי שְׁמוּאֵל מוּנְקִיס)

רָמָה ב׳ – הוֹדָיָה גַּם בְּיוֹם קָשֶׁה

💛 מַה לּוֹקְחִים?

מֵרַבִּי שְׁמוּאֵל לוֹקְחִים שְׁלוֹשָׁה דְּבָרִים: לְחַפֵּשׂ דָּבָר גָּדוֹל לְהוֹדוֹת עָלָיו גַּם בְּיוֹם קָשֶׁה; לְהוֹדוֹת בְּשִׂמְחָה, אֲפִילּוּ בְּשִׁיר וּבְרִקּוּד; וּלְהַרְגִּישׁ אֶת הַחֲסָדִים הַקְּטַנִּים שֶׁבְּכָל בֹּקֶר, כְּמוֹ בְּבִרְכוֹת הַשַּׁחַר.

דָּבָר גָּדוֹל, שִׂמְחָה, וַחֲסָדִים קְטַנִּים.

💡

רָמָה ב׳ – הוֹדָיָה גַּם בְּיוֹם קָשֶׁה

מָה לוֹקְחִים מִרָמָה ב׳?

הוֹדָיָה אֲמִתִּית מוֹצֵאת דָּבָר גָּדוֹל לְהוֹדוֹת עָלָיו, גַּם בְּיוֹם קָשֶׁה, וּמְפָרֶטֶת אֶת הַחֲסָדִים הַקְּטַנִּים. כָּךְ הַלֵּב מִתְמַלֵּא שִׂמְחָה וְאַהֲבַת ה׳.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה הַדָּבָר הַגָּדוֹל בְּיוֹתֵר שֶׁיֵּשׁ לִי, שֶׁאִי אֶפְשָׁר לִשְׂרֹף אוֹ לָקַחַת?
  • מָתַי בַּפַּעַם הָאַחֲרוֹנָה הוֹדֵיתִי עַל חֶסֶד קָטָן בַּבֹּקֶר?
  • מָחָר בַּבֹּקֶר אֹמַר בִּרְכַּת שַׁחַר אַחַת לְאַט, אַרְגִּישׁ אֶת הַחֶסֶד, וְאֶרְשֹׁם מָה הִרְגַּשְׁתִּי.

🏠 מִי הוֹלֵךְ רִאשׁוֹן?

שְׁנֵי חֲבֵרִים רָבוּ. כָּל אֶחָד יוֹשֵׁב בַּצַּד וְחוֹשֵׁב: ׳שֶׁהוּא יָבוֹא אֵלַי רִאשׁוֹן׳. עוֹבֵר יוֹם, עוֹבֵר עוֹד יוֹם – וְאַף אֶחָד לֹא זָז.

בַּפָּרָשָׁה שֶׁלָּנוּ מֹשֶׁה רַבֵּנוּ עָשָׂה אַחֶרֶת: הוּא קָם וְהָלַךְ בְּעַצְמוֹ. בַּשִּׁעוּר לוֹמְדִים מָה אֶפְשָׁר לִלְמֹד מִזֶּה.

לֹא מְחַכִּים שֶׁהַשֵּׁנִי יָבוֹא – קָמִים וְהוֹלְכִים.

🚶 וַיָּקָם מֹשֶׁה וַיֵּלֶךְ

כָּתוּב: ׳וַיָּקָם מֹשֶׁה וַיֵּלֶךְ אֶל דָּתָן וַאֲבִירָם׳. דָּתָן וַאֲבִירָם הִתְנַגְּדוּ לְמֹשֶׁה וּפָגְעוּ בּוֹ.

מֹשֶׁה, הַגָּדוֹל שֶׁבַּדּוֹר, לֹא חִכָּה שֶׁיָּבוֹאוּ אֵלָיו. הוּא הָלַךְ בְּעַצְמוֹ אֲלֵיהֶם כְּדֵי לְנַסּוֹת לְפַיֵּס.

דַּוְקָא מִי שֶׁפָּגְעוּ בּוֹ – הוּא זֶה שֶׁהוֹלֵךְ.

מֹשֶׁה הָלַךְ בְּעַצְמוֹ אֶל מִי שֶׁפָּגַע בּוֹ.

📜 אֵין מַחֲזִיקִין בְּמַחֲלֹקֶת

בַּגְּמָרָא אָמַר רֵישׁ לָקִישׁ: ׳וַיָּקָם מֹשֶׁה וַיֵּלֶךְ׳ – מִכָּאן שֶׁאֵין מַחֲזִיקִין בְּמַחֲלֹקֶת.

לֹא מַסְתַּפְּקִים בְּזֶה שֶׁלֹּא מַתְחִילִים רִיב. גַּם כְּשֶׁאֲנִי לֹא הִתְחַלְתִּי אוֹתוֹ, אֲנִי מְנַסֶּה לְהַפְסִיק אוֹתוֹ וּלְפַיֵּס.

לֹא מַחֲזִיקִים בְּמַחֲלֹקֶת – מְנַסִּים לְפַיֵּס.

בְּסֵפֶר שְׁאִילְתּוֹת כָּתוּב: לֹא רַק הַקָּטָן צָרִיךְ לְהִתְכּוֹפֵף לַגָּדוֹל, אֶלָּא גַּם הַגָּדוֹל צָרִיךְ לְהִתְכּוֹפֵף לַקָּטָן, כְּדֵי שֶׁלֹּא תִּהְיֶה מַחֲלֹקֶת.

גַּם אִם אֲנִי הַגָּדוֹל בַּכִּתָּה אוֹ הֶחָכָם יוֹתֵר – אֲנִי הוֹלֵךְ לְפַיֵּס, כְּמוֹ מֹשֶׁה.

הַגָּדוֹל הוֹלֵךְ לְפַיֵּס אֶת הַקָּטָן.

🏃 שָׁלוֹם – רוֹדְפִים אַחֲרָיו

בַּתְּהִלִּים כָּתוּב: ׳בַּקֵּשׁ שָׁלוֹם וְרָדְפֵהוּ׳. חֲכָמִים אָמְרוּ: גָּדוֹל הַשָּׁלוֹם. בְּכָל הַמִּצְוֹת כָּתוּב ׳כִּי תִרְאֶה׳, ׳כִּי תִפְגַּע׳ – אִם הַמִּצְוָה בָּאָה לְיָדְךָ, אַתָּה חַיָּב לְקַיֵּם אוֹתָהּ.

אֲבָל בְּשָׁלוֹם: ׳בַּקְשֵׁהוּ בִּמְקוֹמְךָ וּרְדְפֵהוּ לְמָקוֹם אַחֵר׳.

שָׁלוֹם הוּא הַמִּצְוָה שֶׁרוֹדְפִים אַחֲרֶיהָ.

אֶת הַשָּׁלוֹם לֹא מְחַכִּים – רוֹדְפִים אַחֲרָיו.

🤫 וְאַהֲרֹן שׁוֹתֵק

כְּשֶׁקֹּרַח קָם, חָשְׁדוּ בְּאַהֲרֹן שֶׁהוּא רוֹדֵף כָּבוֹד. אַהֲרֹן, שֶׁאוֹהֵב שָׁלוֹם, לֹא הִתְגוֹנֵן, לֹא הִתְוַכֵּחַ וְלֹא הֵגִיב – הוּא שָׁתַק.

הָרַמְבַּ״ן מְבָאֵר: אַהֲרֹן לֹא עָנָה דָּבָר בְּכָל הַמַּחֲלֹקֶת, וְעָשָׂה כִּדְבַר מֹשֶׁה.

לִפְעָמִים הַשְּׁתִיקָה הִיא הַגְּבוּרָה הַגְּדוֹלָה, וְהִיא מְכַבָּה אֶת הָאֵשׁ.

לִפְעָמִים הַשְּׁתִיקָה הִיא גְּבוּרָה.

🌿 הַמַּטֶּה שֶׁפָּרַח

מִי שֶׁשּׁוֹתֵק בִּשְׁעַת מַחֲלֹקֶת מוֹסֵר אֶת הַכָּבוֹד שֶׁלּוֹ לַה׳. וְה׳ רָאָה אֶת שְׁתִיקָתוֹ שֶׁל אַהֲרֹן, וְהֵעִיד עָלָיו.

מַטֵּה אַהֲרֹן פָּרַח, הוֹצִיא פְּרָחִים וְגַם שְׁקֵדִים, וְכֻלָּם רָאוּ שֶׁאַהֲרֹן נִבְחַר.

הַשְּׁתִיקָה לֹא הִפְסִידָה לוֹ כְּלוּם.

ה׳ מְכַבֵּד אֶת מִי שֶׁשּׁוֹתֵק בִּשְׁעַת מַחֲלֹקֶת.

💛 גַּם אֶל הָרְחוֹקִים

דָּתָן וַאֲבִירָם הָיוּ רְחוֹקִים מְאֹד, וְהַרְבֵּה אֲנָשִׁים הָלְכוּ אַחֲרֵיהֶם. מֹשֶׁה הָלַךְ אֲלֵיהֶם בְּכָל זֹאת.

שָׁלוֹם אֵינוֹ רַק עִם מִי שֶׁמַּסְכִּים אִתִּי. לִפְעָמִים צָרִיךְ לָלֶכֶת דַּוְקָא אֶל מִי שֶׁהֲכִי רָחוֹק.

וְכִבְיָכוֹל ה׳ עַצְמוֹ ׳רוֹדֵף׳ אַחֲרֵינוּ כְּדֵי לַעֲשׂוֹת אִתָּנוּ שָׁלוֹם – וּמִי שֶׁהוֹלֵךְ לְפַיֵּס הוֹלֵךְ בְּדַרְכָּיו.

מָה לוֹקְחִים מִרָמָה א׳?

שֻׁתָּפִים בַּתּוֹרָה

מִי שֶׁעוֹזֵר לְאַחֵר לִלְמֹד, וּמִי שֶׁמְּלַמֵּד, נַעֲשֶׂה שֻׁתָּף בַּתּוֹרָה

📖 אַהֲבַת הַתּוֹרָה · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🌟

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁיִּהְיֶה לִי חֵלֶק בַּתּוֹרָה שֶׁל כֻּלָּם 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׁעוּר ׳אוֹר הַתּוֹרָה שֶׁמְּחַיֶּה׳ · תַּנְיָא

רָמָה ב׳ – שֻׁתָּפוּת, נְתִינָה וְהוֹרָאָה

🌟 שֻׁתָּף אֲמִתִּי, לֹא רַק נוֹתֵן

בַּשִּׁעוּר ׳אוֹר הַתּוֹרָה שֶׁמְּחַיֶּה׳ נֶאֱמַר: מִי שֶׁגּוֹרֵם לַאֲחֵרִים לִלְמֹד וּמַחְזִיק אֶת לוֹמְדֵי הַתּוֹרָה, מְסַיֵּעַ לָהֶם וּמְשַׂמֵּחַ אוֹתָם – אֵינוֹ רַק נוֹתֵן צְדָקָה. הוּא נַעֲשֶׂה שֻׁתָּף אֲמִתִּי בַּתּוֹרָה, וְיֵשׁ לוֹ חֵלֶק בָּאוֹר שֶׁמְּחַיֶּה.

וּבְאוֹתוֹ שִׁעוּר הוּבָא מָשָׁל מִסֵּפֶר הַתַּנְיָא: הַמִּצְווֹת הֵן כְּמוֹ לְבוּשִׁים וְהַתּוֹרָה כְּמוֹ מָזוֹן. מִי שֶׁיֵּשׁ לוֹ הַרְבֵּה בְּגָדִים וְאֵין לוֹ מָה לֶאֱכֹל – עַל מָה יַלְבִּישׁ אוֹתָם?

וְלָכֵן מִי שֶׁעוֹזֵר שֶׁתִּהְיֶה תּוֹרָה, עוֹזֵר שֶׁיִּהְיוּ חַיִּים לְכָל הַמִּצְווֹת.

מִי שֶׁמַּחְזִיק וּמְשַׂמֵּחַ לוֹמְדִים הוּא שֻׁתָּף אֲמִתִּי, וְלוֹ חֵלֶק בַּתּוֹרָה שֶׁלָּהֶם.

1
🕯️

💛 אַהֲבַת הַשֵּׁם: אֲנִי נוֹתֵן מִזְּמַנִּי וּמִנּוֹחוּתִי לְמַעַן הַתּוֹרָה 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הֶחָפֵץ חַיִּים (מִתּוֹךְ שִׁעוּר ׳אוֹר הַתּוֹרָה שֶׁמְּחַיֶּה׳)

רָמָה ב׳ – שֻׁתָּפוּת, נְתִינָה וְהוֹרָאָה

🕯️ הַוִּתּוּר הוּא הַשֻּׁתָּפוּת

הֶחָפֵץ חַיִּים הִזְכִּיר אֶת הַנָּשִׁים שֶׁנּוֹתְנוֹת לִבְעָלֵיהֶן לָלֶכֶת לִלְמֹד. הֵן מְוַתְּרוֹת עַל עֶזְרָתָם וְעַל נוֹכְחוּתָם בַּבַּיִת בְּאוֹתָן שָׁעוֹת, וּמְחַכּוֹת עַד שֶׁהֵם שָׁבִים.

הַוִּתּוּר הַזֶּה אֵינוֹ דָּבָר צְדָדִי: הוּא עַצְמוֹ מַה שֶׁהוֹפֵךְ אוֹתָן לְשֻׁתָּפוֹת מְלֵאוֹת בַּתּוֹרָה שֶׁנִּלְמֶדֶת, וּמַעֲנִיק לָהֶן אֶת הַטַּל שֶׁמְּחַיֶּה.

מִכָּאן שֶׁגַּם מִי שֶׁלֹּא יוֹשֵׁב וְלוֹמֵד בְּאוֹתָהּ שָׁעָה, יָכוֹל לִהְיוֹת שֻׁתָּף – כְּשֶׁהוּא נוֹתֵן מִזְּמַנּוֹ וּמִנּוֹחוּתוֹ.

הַוִּתּוּר וְהַהַמְתָּנָה לְמַעַן הַלִּמּוּד – הֵם עַצְמָם הַשֻּׁתָּפוּת בַּתּוֹרָה.

2
🎁

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד כְּדֵי לָתֵת, לֹא רַק כְּדֵי לָקַחַת 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, לַחְשֹׁב לִפְנֵי הַלִּמּוּד (בַּ״ח)

רָמָה ב׳ – שֻׁתָּפוּת, נְתִינָה וְהוֹרָאָה

🎁 לִלְמֹד כִּנְתִינָה

אוֹמְרִים שֶׁחָשׁוּב לְהִתְרַגֵּל לִלְמֹד כִּנְתִינָה וְלֹא כִּנְטִילָה. כְּשֶׁאֲנִי לוֹמֵד, אֲנִי לֹא רַק לוֹקֵחַ.

מָה אֲנִי נוֹתֵן? הַלִּמּוּד שֶׁלִּי מוֹרִיד קְדֻשָּׁה לָאָרֶץ, וְהָאָרֶץ מֵאִירָה מִכְּבוֹדוֹ. וְה׳ נֶהֱנֶה שֶׁבְּנוֹ חֲבִיבוֹ נַעֲשֶׂה יוֹתֵר קָדוֹשׁ וְטָהוֹר.

כָּךְ אֲנִי מַתְחִיל כָּל לִמּוּד בְּכַוָּנָה: ׳אֶלְמַד כְּדֵי לְשַׂמֵּחַ אֶת אָבִי שֶׁבַּשָּׁמַיִם׳.

לוֹמְדִים כִּנְתִינָה: הַלִּמּוּד מוֹרִיד קְדֻשָּׁה וּמְשַׂמֵּחַ אֶת הָאָב שֶׁבַּשָּׁמַיִם.

3
📚

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְבַקֵּשׁ מֵה׳ לִזְכֹּר אֶת מָה שֶׁלָּמַדְתִּי, כְּדֵי לְלַמֵּד אֲחֵרִים 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, לַחְשֹׁב לִפְנֵי הַלִּמּוּד

רָמָה ב׳ – שֻׁתָּפוּת, נְתִינָה וְהוֹרָאָה

📚 לִלְמֹד כְּדֵי לְלַמֵּד

כַּוָּנָה נוֹסֶפֶת לִפְנֵי הַלִּמּוּד: ׳כְּדֵי שֶׁאוּכַל לְלַמֵּד אֲחֵרִים׳. בְּכָךְ הַלִּמּוּד נֶחְשָׁב תּוֹרַת חֶסֶד.

וְגַם: ׳עַל יְדֵי שֶׁאֶהְיֶה יוֹתֵר יְרֵא שָׁמַיִם, אוּכַל יוֹתֵר לְהַשְׁפִּיעַ עַל אֲחֵרִים׳. הַמִּדּוֹת הַטּוֹבוֹת שֶׁלִּי הֵן חֵלֶק מֵהַהוֹרָאָה.

וְיֵשׁ בַּקָּשָׁה: ׳אָנָּא ה׳, עֲזֹר לִי לִזְכֹּר אֶת הַלִּמּוּד, כְּדֵי לְלַמֵּד אֶת עַם יִשְׂרָאֵל׳. וְכָל הָעוֹסְקִים עִם הַצִּבּוּר, שֶׁיִּהְיוּ עוֹסְקִים עִמָּהֶם לְשֵׁם שָׁמַיִם.

לוֹמְדִים כְּדֵי לְלַמֵּד – תּוֹרַת חֶסֶד, וּמְבַקְשִׁים עֵזֶר לִזְכֹּר.

4
📜

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׁלִיחַ קָטָן שֶׁמַּעֲבִיר הָלְאָה אֶת הַמַּתָּנָה שֶׁל ה׳ 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, חִדּוּשִׁים לְפָרָשַׁת הַאֲזִינוּ (דִּבְרֵי הָרַב אֱלִיָּהוּ דִּיסְקִין)

רָמָה ב׳ – שֻׁתָּפוּת, נְתִינָה וְהוֹרָאָה

📜 הַמַּתָּנָה עוֹבֶרֶת הָלְאָה

ה׳ מָסַר לָנוּ אֶת הַתּוֹרָה כְּדֵי שֶׁנַּעֲבִיר אוֹתָהּ הָלְאָה. מִי שֶׁמְּלַמֵּד, מַמְשִׁיךְ אֶת מַעֲשֵׂה הָאַהֲבָה שֶׁל נְתִינַת הַתּוֹרָה.

דִּבְרֵי הָרַב אֱלִיָּהוּ דִּיסְקִין מִבִּיאִים מְקוֹרוֹת לָזֶה: הָרַמְבַּ״ם בְּסֵפֶר הַמִּצְווֹת, הַסִּפְרֵי בְּפָרָשַׁת וָאֶתְחַנַּן, וְהַמַּהַרַ״ל בְּ׳דֶּרֶךְ חַיִּים׳.

וְכָל יוֹם בְּבִרְכַּת הַתּוֹרָה מִתְפַּלְּלִים בְּכַוָּנָה עַל הַיְלָדִים וְעַל הַתַּלְמִידִים שֶׁלָּנוּ בִּשְׁמוֹתֵיהֶם.

הַמְלַמֵּד מַמְשִׁיךְ אֶת מַעֲשֵׂה הָאַהֲבָה שֶׁל נְתִינַת הַתּוֹרָה, וּמִתְפַּלֵּל עַל תַּלְמִידָיו.

🤝

💛 אַהֲבַת הַשֵּׁם: אֲנִי שֻׁתָּף בַּתּוֹרָה בְּכָל דֶּרֶךְ שֶׁאֲנִי יָכוֹל 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – שֻׁתָּפוּת, נְתִינָה וְהוֹרָאָה

🤝 בָּנִים וּבָנוֹת, כֻּלָּנוּ שֻׁתָּפִים

מִי שֶׁמְּסַיֵּעַ לְחָבֵר לִלְמֹד – בְּחַבְרוּתָא, בִּנְתִינַת מָקוֹם שָׁקֵט, בְּעֶזְרָה בְּשִׁעוּרֵי בַּיִת – שֻׁתָּף בַּתּוֹרָה.

גַּם יְלָדִים וְגַם יְלָדוֹת יְכוֹלִים: מִי שֶׁלּוֹמֵד, מִי שֶׁעוֹזֵר, מִי שֶׁמְּוַתֵּר, וּמִי שֶׁמְּלַמֵּד.

וּמִי שֶׁעוֹזְרִים לוֹ לִלְמֹד, חָשׁוּב שֶׁיֹּאמַר תּוֹדָה, כִּי הַשֻּׁתָּף נוֹתֵן וְהוּא רוֹאֶה אֶת הַטּוֹב.

כֻּלָּנוּ שֻׁתָּפִים: הַלּוֹמֵד, הָעוֹזֵר, הַמְוַתֵּר וְהַמְלַמֵּד.

💡

רָמָה ב׳ – שֻׁתָּפוּת, נְתִינָה וְהוֹרָאָה

מָה לוֹקְחִים מִרָמָה ב׳?

הַתּוֹרָה הִיא מַתָּנָה שֶׁנּוֹתְנִים הָלְאָה. כְּשֶׁאֲנִי לוֹמֵד כִּנְתִינָה, עוֹזֵר לַאֲחֵרִים וּמוֹדֶה, אֲנִי חֵלֶק מִשַּׁרְשֶׁרֶת הָאַהֲבָה שֶׁל ה׳ וְעַמּוֹ.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • מִי עוֹזֵר לִי לִלְמֹד – בַּבַּיִת, בַּכִּתָּה אוֹ בֵּין הַחֲבֵרִים – וְאֵיךְ אֲנִי יוֹדֵעַ שֶׁהוּא עוֹזֵר?
  • אֶת מִי אֲנִי יָכוֹל לַעֲזֹר הַשָּׁבוּעַ לִלְמֹד, וּמָה בְּדִיּוּק אֶעֱשֶׂה?
  • אֶכְתֹּב מִכְתָּב תּוֹדָה קָצָר לְמִישֶׁהוּ שֶׁעוֹזֵר לִי לִלְמֹד וְאֶתֵּן לוֹ אוֹתוֹ, וְאֶלְמַד דָּבָר אֶחָד לְחָבֵר אוֹ לְאָח אוֹ לְאָחוֹת קְטַנִּים.

👤 מִי הָיָה הָרַב גֵּפֶן?

הָרַב יְהוּדָה אַרְיֵה לֵיב גֵּפֶן נוֹלַד בִּשְׁנַת תר״ע (1910). בְּדַף הַתּוֹלְדוֹת הוּא נִקְרָא ׳הָאֲרִי מִכְּפַר אָתָא׳.

הוּא הָיָה הָרַב שֶׁבָּא אַחֲרֵי הָרַב מֶנְדֵּלְסוֹן, וְהַסִּפּוּר שֶׁלּוֹ מַתְחִיל רָחוֹק מְאֹד מִכָּאן – בְּבָתֵּי הַמִּדְרָשׁ שֶׁל גָּלִיצְיָה וּפּוֹלִין.

אַרְיֵה בַּשֵּׁם, וְאֵשׁ בַּלֵּב.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִלְמֹד תּוֹרָה בְּחֶדְוָה וּבְעֹמֶק 💛

📖 תּוֹלְדוֹת קִרְיַת אָתָא – מֵאַרְכִיּוֹן הַקּוֹל הַחֲמִישִׁי, הָרַב גֵּפֶן – הַדְּרָשָׁה הָאַחֲרוֹנָה

רָמָה א׳ – מִי הָיָה הָרַב גֵּפֶן

📚 תַּלְמִיד חָכָם שֶׁשְּׁמוֹ הָלַךְ לְפָנָיו

בָּעִיר טַרְנוֹב הוּא הָיָה תַּלְמִיד חָכָם צָעִיר, וְלָמַד אֵצֶל הַגָּאוֹן רַבִּי מֵאִיר אָרָק. מִשָּׁם עָבַר לִישִׁיבַת ׳חַכְמֵי לוּבְּלִין׳, אֵצֶל רַבּוֹ הַגָּדוֹל, הַמַּהֲרַ״ם שַׁפִּירָא.

כְּשֶׁהָיָה קָם בַּשִּׁעוּר, הָיָה שֶׁקֶט בְּבֵית הַמִּדְרָשׁ – כִּי כֻּלָּם יָדְעוּ שֶׁתֵּכֶף תָּבוֹא שְׁאֵלָה חַדָּה שֶׁתַּעֲצֹר אֲפִלּוּ גְּדוֹלִים.

שְׁאֵלוֹת חַדּוֹת, וְלֵב שֶׁרוֹצֶה לְהָבִין עַד הַסּוֹף.

📖 תּוֹלְדוֹת קִרְיַת אָתָא – מֵאַרְכִיּוֹן הַקּוֹל הַחֲמִישִׁי, הָאֲרִי מִכְּפַר אָתָא

רָמָה א׳ – מִי הָיָה הָרַב גֵּפֶן

🕯️ רַב בְּעִיֶּרֶת לִיפְּסְק

בִּשְׁנַת תרצ״ו (1936) הוּא נַעֲשָׂה רַב בָּעֲיָרָה לִיפְּסְק. הוּא הָיָה רַב צָעִיר, אֲבָל כְּבָר אָז הָלַךְ לְפָנָיו שֵׁם שֶׁל תַּלְמִיד חָכָם וְאִישׁ צִבּוּר.

אַחַר כָּךְ בָּאוּ שָׁנִים קָשׁוֹת מְאֹד לַיַּהֲדוּת פּוֹלִין.

רַב צָעִיר שֶׁמִּתְכּוֹנֵן לִתְפְּקִיד גָּדוֹל.

3

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן כֹּחַ לְמִי שֶׁשּׁוֹמֵר עַל תּוֹרָתוֹ 💛

📖 תּוֹלְדוֹת קִרְיַת אָתָא – מֵאַרְכִיּוֹן הַקּוֹל הַחֲמִישִׁי, הָאֲרִי מִכְּפַר אָתָא

רָמָה א׳ – מִי הָיָה הָרַב גֵּפֶן

❄️ מְסִירוּת נֶפֶשׁ עַל הַתּוֹרָה

בְּשָׁנִים הַקָּשׁוֹת שֶׁל הַמִּלְחָמָה הוּא הוּגְלָה לְסִיבִּיר הַקָּרָה, וְשָׁם מָסַר אֶת נַפְשׁוֹ עַל הַתּוֹרָה וְהַמִּצְווֹת.

אַחֲרֵי הַמִּלְחָמָה הוּא עָבַר לְמַחֲנוֹת הַפְּלִיטִים בְּגֶרְמַנְיָה וְשָׁם עִדֵּד אֶת שְׁאֵרִית הַפְּלֵיטָה, וְהִתְחִיל לַחְלֹם עַל עֲלִיָּה לְאֶרֶץ יִשְׂרָאֵל.

גַּם בַּקֹּר הַגָּדוֹל, הוּא לֹא וִתֵּר עַל הַתּוֹרָה.

💛 אַהֲבַת הַשֵּׁם: ה׳ מֵבִיא אוֹתָנוּ הַבַּיְתָה, גַּם אַחֲרֵי דֶּרֶךְ אֲרֻכָּה 💛

📖 תּוֹלְדוֹת קִרְיַת אָתָא – מֵאַרְכִיּוֹן הַקּוֹל הַחֲמִישִׁי, הָאֲרִי מִכְּפַר אָתָא

רָמָה א׳ – מִי הָיָה הָרַב גֵּפֶן

🚢 הַחֲלוֹם מִתְקַיֵּם

הוּא עָלָה לְאֶרֶץ יִשְׂרָאֵל וְהִתְיַשֵּׁב בְּחֵיפָה. יַחַד עִם הָאַדְמוֹ״ר מִסַּעֲרֵט-וִיזְ׳נִיץ הוּא יִסֵּד אֶת הַיְשִׁיבָה ׳יִחַל יִשְׂרָאֵל׳.

הוּא הָיָה גַּם דְּמוּת מֶרְכָּזִית בְּסָנִיף חֵיפָה שֶׁל פּוֹעֲלֵי אַגּוּדַּת יִשְׂרָאֵל.

הַחֲלוֹם מֵהַמַּחֲנוֹת נַעֲשָׂה מַעֲשֶׂה בְּחֵיפָה.

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ לְכַבֵּד אֶת הָרַב שֶׁל הַקְּהִלָּה שֶׁלִּי 💛

📖 תּוֹלְדוֹת קִרְיַת אָתָא – מֵאַרְכִיּוֹן הַקּוֹל הַחֲמִישִׁי, הָרַב גֵּפֶן – הַדְּרָשָׁה הָאַחֲרוֹנָה

רָמָה א׳ – מִי הָיָה הָרַב גֵּפֶן

👑 יוֹם הַהַכְתָּרָה

בְּרֹאשׁ חֹדֶשׁ אֲדָר תשי״ג הוּכְתַּר לְרַבָּהּ שֶׁל כְּפַר אָתָא, וְהָיְתָה זוֹ חֲגִיגָה שֶׁל עַם שָׁלֵם: חֲזִיתוֹת הַבָּתִּים קֻשְּׁטוּ בִּשְׁטִיחִים, בַּכְּנִיסָה לַכְּפָר עָמַד שַׁעַר כָּבוֹד מְקֻשָּׁט בְּיָרֹק וּבְדְגָלִים, וַאֲלָפִים קִבְּלוּ אֶת פָּנָיו בְּשִׁירָה.

בְּרֹאשׁ הַמַּכְתִּירִים עָמַד הָרַב הָרָאשִׁי, הָרַב הֶרְצוֹג.

הַכְּפָר כֻּלּוֹ יָצָא לְקַבֵּל אֶת רַבּוֹ בְּשִׂמְחָה.

💛 אַהֲבַת הַשֵּׁם: גַּם אֲנִי רוֹצֶה לָדַעַת לְמַעַן מִי אֲנִי עוֹבֵד 💛

📖 תּוֹלְדוֹת קִרְיַת אָתָא – מֵאַרְכִיּוֹן הַקּוֹל הַחֲמִישִׁי, הָאֲרִי מִכְּפַר אָתָא

רָמָה א׳ – מִי הָיָה הָרַב גֵּפֶן

🛖 תַּלְמוּד תּוֹרָה בִּצְרִיפִים

עַד בּוֹאוֹ שָׁכַן תַּלְמוּד הַתּוֹרָה בִּצְרִיפִים רְעוּעִים. הָרַב גֵּפֶן הִתְמַסֵּר לְהָקִים בִּנְיָן לְתִפְאֶרֶת, כְּדֵי שֶׁיְּלָדִים יִלְמְדוּ תּוֹרָה בְּמָקוֹם רָאוּי.

בִּשְׁבִיל הַיְלָדִים הוּא הָיָה מוּכָן לְהִתְאַמֵּץ וְלִפְעֹל בְּכָל כֹּחוֹ.

הוּא רָצָה בַּיִת יָפֶה לַתּוֹרָה שֶׁל הַיְלָדִים.

📖 תּוֹלְדוֹת קִרְיַת אָתָא – מֵאַרְכִיּוֹן הַקּוֹל הַחֲמִישִׁי, הָרַב גֵּפֶן – הַדְּרָשָׁה הָאַחֲרוֹנָה

רָמָה א׳ – מִי הָיָה הָרַב גֵּפֶן

🎤 הַדְּרָשָׁה הָאַחֲרוֹנָה

בְּכ״ו בִּשְׁבָט תשי״ז עָלָה הָרַב גֵּפֶן לַבָּמָה בְּמָלוֹן ׳יִשְׂרָאֵל׳ בְּתֵל אָבִיב, בְּכִנּוּס שֶׁל פּוֹעֲלֵי אַגּוּדַּת יִשְׂרָאֵל עַל קְלִיטַת עוֹלִים מֵהוּנְגַּרְיָה.

חָמֵשׁ דַּקּוֹת בִּלְבַד דִּבֵּר בְּהִתְרַגְּשׁוּת, וְקוֹלוֹ הָיָה זְעָקָה: ׳לְכָל יְהוּדִי שׁוֹמֵר תּוֹרָה לַעֲלוֹת לָאָרֶץ!׳

חָמֵשׁ דַּקּוֹת שֶׁל לֶהָבָה לְמַעַן עֲלִיָּה.

📖 תּוֹלְדוֹת קִרְיַת אָתָא – מֵאַרְכִיּוֹן הַקּוֹל הַחֲמִישִׁי, הָרַב גֵּפֶן – הַדְּרָשָׁה הָאַחֲרוֹנָה

רָמָה א׳ – מִי הָיָה הָרַב גֵּפֶן

🕊️ הַפְּטִירָה וְהַמוֹרָשָׁה

בְּשִׂיא הַדְּבָרִים נִפְטַר הָרַב גֵּפֶן. הוּא הָיָה בֶּן אַרְבָּעִים וָשֶׁבַע, כְּמִנְיַן שְׁנוֹת רַבּוֹ, הַמַּהֲרַ״ם שַׁפִּירָא.

הוּא לֹא זָכָה לִרְאוֹת אֶת בִּנְיַן הַתַּלְמוּד תּוֹרָה שֶׁהִקִּים, וְהִשְׁאִיר כְּתַב יָד עַל רַבּוֹ. הַהַלְוָיָה יָצְאָה מִבֵּית הַכְּנֶסֶת ׳הֵיכַל שְׁלֹמֹה׳, וּבַיּוֹם הַהוּא לֹא הָיוּ לִמּוּדִים בַּגַּן שֶׁבְּבֵית הַכְּנֶסֶת.

הַדְּרָשָׁה הָאַחֲרוֹנָה שֶׁלּוֹ נִשְׁאֲרָה צַוָּאָה.

מָה לוֹקְחִים מִרָמָה א׳?

הָרַב גֵּפֶן לֹא שָׁמַר עַל נַפְשׁוֹ אֶלָּא עַל תּוֹרָתוֹ וְעַל עַם יִשְׂרָאֵל. כָּךְ ה׳ מַעֲמִיד לְעַמּוֹ אֲנָשִׁים שֶׁלִּבָּם בּוֹעֵר לַאֲחֵרִים – וְזֶה סִימָן שֶׁה׳ אוֹהֵב אוֹתָנוּ.

  • מִי הָיָה הָרַב אוֹ הַמּוֹרֶה שֶׁלִּי שֶׁנָּתַן לִי מִכֹּחוֹ, וְאֵיךְ אֲנִי יָכוֹל לְהוֹדוֹת לוֹ?

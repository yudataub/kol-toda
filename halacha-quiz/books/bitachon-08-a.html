כִּי בוֹ יִשְׂמַח לִבֵּנוּ

רָמָה א׳ – שִׂמְחָה שֶׁל בּוֹטֵחַ

אֵיךְ הַבִּטָּחוֹן מֵבִיא שִׂמְחָה, אֵיךְ בּוֹדְקִים אֶת ׳מַד הַבִּטָּחוֹן׳, וְלָמָּה לֵב שָׂמֵחַ מוּכָן לְקַבֵּל מֵה׳

📖 בִּטָּחוֹן בַּה׳ · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
🎒

💛 אַהֲבַת הַשֵּׁם: אֲנִי זוֹכֵר שֶׁה׳ אוֹהֵב אוֹתִי, וְהַלֵּב שֶׁלִּי נִרְגָּע 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – שִׂמְחָה שֶׁל בּוֹטֵחַ

🎒 בֹּקֶר שֶׁל דְּאָגָה

לִפְנֵי מִבְחָן הַבֶּטֶן שֶׁל דָּנִי כְּבָר מִתְכַּוֶּצֶת: ׳מָה יִהְיֶה אִם אֶכְשַׁל?׳

הוּא עוֹצֵר רֶגַע וְנוֹשֵׁם עָמֹק. הוּא נִזְכָּר: ׳יֵשׁ לִי אַבָּא טוֹב שֶׁאוֹהֵב אוֹתִי, וְהוּא יוֹדֵעַ מָה טוֹב לִי׳. הַבֶּטֶן מִתְרַפָּה, וְהוּא מַתְחִיל לְחַיֵּךְ.

בַּשִּׁעוּר לוֹמְדִים: בִּטָּחוֹן מֵבִיא שִׂמְחָה וְרוֹגַע.

בִּטָּחוֹן אוֹמֵר: יֵשׁ לִי אַבָּא טוֹב, וְלָכֵן אֲנִי יָכוֹל לִהְיוֹת שָׂמֵחַ וְרָגוּעַ.

1
🎵

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ כִּי אֲנִי בּוֹטֵחַ בַּה׳ 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים ל״ג כ״א

רָמָה א׳ – שִׂמְחָה שֶׁל בּוֹטֵחַ

🎵 הַפָּסוּק שֶׁל הַשִּׂמְחָה

בְּתְהִלִּים כָּתוּב: ׳כִּי בוֹ יִשְׂמַח לִבֵּנוּ כִּי בְשֵׁם קָדְשׁוֹ בָטָחְנוּ׳.

מִי שֶׁיּוֹדֵעַ שֶׁכָּל מָה שֶׁקָּרָה לוֹ וְשֶׁיִּקְרֶה לוֹ הוּא רַק טוֹב, וְשֶׁיֵּשׁ לוֹ אַבָּא טוֹב וּמֵיטִיב – לָמָּה שֶׁלֹּא יִהְיֶה שָׂמֵחַ?

הַשִּׂמְחָה הִיא הַתּוֹצָאָה הַטִּבְעִית שֶׁל הַבִּטָּחוֹן.

הַלֵּב שָׂמֵחַ כִּי הוּא בּוֹטֵחַ בַּה׳.

2
🧭

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹדֵק אֶת הַמַּד שֶׁלִּי, וּמְחַזֵּק אֶת הַבִּטָּחוֹן 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – שִׂמְחָה שֶׁל בּוֹטֵחַ

🧭 מַד הַבִּטָּחוֹן

בָּרֶכֶב יֵשׁ מַד שֶׁמַּרְאֶה כַּמָּה דֶּלֶק נִשְׁאַר. גַּם בַּלֵּב אֶפְשָׁר לִדְמוֹת מַד כָּזֶה.

אִם אֲנִי מוּדָאָג, בְּמַצַּב רוּחַ לֹא טוֹב, אוֹ לֹא שָׂמֵחַ וְרָגוּעַ – זֶה סִימָן שֶׁחָסֵר לִי עוֹד בִּטָּחוֹן.

זֶה לֹא עֹנֶשׁ וְלֹא סִימָן שֶׁאֲנִי יֶלֶד רַע. זֶה רַק אוֹת שֶׁמַּזְכִּיר לִי לַחְזֹר אֶל הָאַבָּא הַטּוֹב.

דְּאָגָה וּמַצַּב רוּחַ רַע הֵם סִימָן שֶׁצָּרִיךְ לְחַזֵּק אֶת הַבִּטָּחוֹן.

3
🪟

💛 אַהֲבַת הַשֵּׁם: אֲנִי פּוֹתֵחַ אֶת הַלֵּב בְּשִׂמְחָה 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – שִׂמְחָה שֶׁל בּוֹטֵחַ

🪟 דְּאָגָה סוֹגֶרֶת, שִׂמְחָה פּוֹתַחַת

כְּשֶׁהַתְּרִיס יָרוּד וְהַחַלּוֹן סָגוּר, הַשֶּׁמֶשׁ לֹא נִכְנֶסֶת לַחֶדֶר. כָּךְ גַּם בַּלֵּב.

בַּמְּקוֹרוֹת נֶאֱמַר: ׳הַיָּגוֹן סוֹתֵם אֶת מְקוֹרֵי הַנֶּפֶשׁ׳, וְהַשִּׂמְחָה פּוֹתַחַת אוֹתָם.

ה׳ רוֹצֶה לְהַשְׁפִּיעַ עָלֵינוּ טוֹבָה, וּכְשֶׁהַלֵּב שָׂמֵחַ הוּא פָּתוּחַ וּמוּכָן לְקַבֵּל.

שִׂמְחָה פּוֹתַחַת אֶת הַלֵּב לְקַבֵּל טוֹבָה.

4
🤗

💛 אַהֲבַת הַשֵּׁם: אֲנִי הַבֵּן שֶׁל ה׳, וְשִׂמְחָתִי מְשַׂמַּחַת אוֹתוֹ 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – שִׂמְחָה שֶׁל בּוֹטֵחַ

🤗 אַבָּא שָׂמֵחַ בְּשִׂמְחָתִי

כְּשֶׁיֶּלֶד שָׂמֵחַ, אַבָּא שֶׁלּוֹ שָׂמֵחַ גַּם הוּא.

בַּשִּׁעוּר אָמְרוּ: אַתָּה בְּנוֹ שֶׁל ה׳, וְאִם אַתָּה שָׂמֵחַ – אֵיךְ הוּא לֹא יִשְׂמַח?

כְּשֶׁאֲנִי נִשְׁאָר שָׂמֵחַ גַּם כְּשֶׁיֵּשׁ לִי בְּעָיָה, אֲנִי מְשַׂמֵּחַ אוֹתוֹ.

כְּשֶׁאֲנִי שָׂמֵחַ בִּבְטָחוֹן, אֲנִי מְשַׂמֵּחַ אֶת ה׳.

5
💪

💛 אַהֲבַת הַשֵּׁם: אֲנִי בִּידַיִם טוֹבוֹת, וְגַם כְּשֶׁקָּשֶׁה אֲנִי נִרְגָּע 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – שִׂמְחָה שֶׁל בּוֹטֵחַ

💪 שָׂמֵחַ גַּם כְּשֶׁקָּשֶׁה

נֹעַם הִפְסִיד בַּמִּשְׂחָק וְהָיָה מְאֻכְזָב. הוּא לָקַח נְשִׁימָה: ׳אֲנִי בִּידַיִם טוֹבוֹת׳.

מִי שֶׁחוֹשֵׁב רַק ׳יִהְיֶה רַע׳ – נַעֲשֶׂה עָצוּב. מִי שֶׁזּוֹכֵר שֶׁהוּא בִּידַיִם טוֹבוֹת – נִרְגָּע.

שִׂמְחָה כָּזוֹ בִּזְמַן קֹשִׁי חֲשׁוּבָה מְאֹד.

זְכִירָה שֶׁאֲנִי בִּידַיִם טוֹבוֹת מְרַגַּעַת וּמְשַׂמֶּחֶת גַּם בִּזְמַן קֹשִׁי.

6
🌅

💛 אַהֲבַת הַשֵּׁם: אֲנִי מֵכִין אֶת הַלֵּב לְקַבֵּל אֶת הָאוֹר שֶׁל ה׳ 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּמִדְבַּר ו׳ כ״ה

רָמָה א׳ – שִׂמְחָה שֶׁל בּוֹטֵחַ

🌅 שִׂמְחָה שֶׁמְּקַבֶּלֶת אוֹר

בְּבִרְכַּת כֹּהֲנִים כָּתוּב: ׳יָאֵר ה׳ פָּנָיו אֵלֶיךָ׳. ה׳ רוֹצֶה לְהַשְׁפִּיעַ לָנוּ אוֹר וְטוֹבָה.

וְהַלֵּב הַשָּׂמֵחַ מוּכָן לְקַבֵּל אֶת הָאוֹר הַזֶּה.

כָּךְ הַבִּטָּחוֹן, שֶׁמֵּבִיא שִׂמְחָה, מְכִינָה אוֹתָנוּ לְקַבֵּל מֵה׳.

הַלֵּב הַשָּׂמֵחַ מוּכָן לְקַבֵּל אֶת הַטּוֹב שֶׁה׳ רוֹצֶה לָתֵת.

7
🎶

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַתְחִיל אֶת הַיּוֹם בְּשִׂמְחָה וּבְבִטָּחוֹן 💛

📖 בִּטָּחוֹן בַּה׳ – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – שִׂמְחָה שֶׁל בּוֹטֵחַ

🎶 הַתְחָלָה שֶׁל יוֹם

אֶפְשָׁר לְהַתְחִיל כָּל בֹּקֶר בְּפָסוּק אֶחָד: ׳כִּי בוֹ יִשְׂמַח לִבֵּנוּ׳.

בַּבֹּקֶר שׁוֹאֲלִים: ׳מָה הַמַּד שֶׁלִּי הַיּוֹם?׳ וּבָעֶרֶב בּוֹדְקִים שׁוּב.

כָּךְ מִתְאַמְּנִים יוֹם אַחַר יוֹם, וְהַלֵּב לוֹמֵד לִהְיוֹת שָׂמֵחַ וְרָגוּעַ.

אִמּוּן יוֹמִי קָטָן מְחַזֵּק אֶת הַבִּטָּחוֹן וְאֶת הַשִּׂמְחָה.

8
💡

רָמָה א׳ – שִׂמְחָה שֶׁל בּוֹטֵחַ

מָה לוֹקְחִים מִרָמָה א׳?

הַבִּטָּחוֹן הוּא הַדֶּרֶךְ לְשִׂמְחָה אֲמִתִּית. כְּשֶׁאֲנִי זוֹכֵר שֶׁה׳ אוֹהֵב אוֹתִי כְּמוֹ אַבָּא טוֹב, הַלֵּב שֶׁלִּי שָׂמֵחַ, וְכָךְ גַּם אַהֲבָתִי אֵלָיו גְּדֵלָה.

כִּי בוֹ יִשְׂמַח לִבֵּנוּ כִּי בְשֵׁם קָדְשׁוֹ בָטָחְנוּ

  • מָתַי הַשָּׁבוּעַ הָיִיתִי דּוֹאֵג, וּמָה אֶזְכֹּר בַּפַּעַם הַבָּאָה?

לָמָּה גַּאֲוָה הִיא שִׁכְחָה, וְאֵיךְ הַתּוֹדָה שֶׁל בִּרְכַּת הַמָּזוֹן מַצִּילָה אוֹתָנוּ

💛 אַהֲבַת הַשֵּׁם: אֲנִי שׁוֹמֵר עַל הַלֵּב שֶׁלִּי שֶׁלֹּא יִשְׁכַּח אֶת ה׳ 💛

📖 עֲנָוָה וְנַחַת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דְּבָרִים ח׳ י״ב–י״ח

רָמָה ב׳ – תּוֹדָה נֶגֶד גַּאֲוָה

🏡 מֵאַיִן מַתְחֶלֶת הַשִּׁכְחָה?

בַּפְּסוּקִים מְתֻאָר סֵדֶר: ׳פֶּן תֹּאכַל וְשָׂבָעְתָּ, וּבָתִּים טוֹבִים תִּבְנֶה וְיָשָׁבְתָּ, וּבְקָרְךָ וְצֹאנְךָ יִרְבְּיֻן, וְכֶסֶף וְזָהָב יִרְבֶּה לָּךְ׳.

אַחֲרֵי כָּל הַטּוֹב הַזֶּה בָּאָה הַסַּכָּנָה: ׳וְרָם לְבָבֶךָ וְשָׁכַחְתָּ אֶת ה׳׳. הַטּוֹבוֹת עַצְמָן אֵינָן רָעוֹת – הַבְּעָיָה הִיא שֶׁהַלֵּב יָכוֹל לְהִתְנַשֵּׂא.

וּבַפָּסוּק הַבָּא: ׳וְאָמַרְתָּ בִּלְבָבֶךָ: כֹּחִי וְעֹצֶם יָדִי׳ – הַשִּׁכְחָה הִיא מַחֲשָׁבָה שֶׁבַּלֵּב. וְהַתְּרוּפָה: ׳וְזָכַרְתָּ אֶת ה׳ אֱלֹהֶיךָ׳.

הַשִּׁכְחָה מַתְחִילָה בַּלֵּב, וְהַתְּרוּפָה הִיא ׳וְזָכַרְתָּ׳.

💛 אַהֲבַת הַשֵּׁם: אֲנִי עָנָו וּמַכִּיר טוֹבָה לַנּוֹתֵן 💛

📖 עֲנָוָה וְנַחַת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סְמַ״ג לָאוִין ס״ד · סוֹטָה ה׳ א׳

רָמָה ב׳ – תּוֹדָה נֶגֶד גַּאֲוָה

📜 הַסְּמַ״ג: אַזְהָרָה לֹא לְהִתְגָּאוֹת

הַסְּמַ״ג מְבָאֵר: ׳הִשָּׁמֶר לְךָ פֶּן תִּשְׁכַּח׳ הִיא אַזְהָרָה שֶׁלֹּא יִתְגָּאוּ יִשְׂרָאֵל כְּשֶׁה׳ מַשְׁפִּיעַ עֲלֵיהֶם טוֹבָה וְיֹאמְרוּ ׳כֹּחִי וְעֹצֶם יָדִי׳.

מִכָּאן שֶׁלֹּא יִתְגָּאֶה אָדָם בְּמָה שֶׁחָנַן אוֹתוֹ ה׳ – בְּמָמוֹן, בְּיֹפִי וּבְחָכְמָה – אֶלָּא יִהְיֶה עָנָו וְיַכִּיר טוֹבָה לַנּוֹתֵן.

הַסְּמַ״ג מְסַפֵּר שֶׁבַּהַתְחָלָה לֹא מָנָה זֹאת כְּמִצְוָה, כִּי גַּם הָרַמְבַּ״ם לֹא מָנָה אוֹתָהּ, עַד שֶׁרָאָה בַּחֲלוֹם: ׳שָׁכַחְתָּ אֶת הָעִקָּר׳. הוּא הֵבִיא רְאָיָה מֵהַגְּמָרָא: ׳אַזְהָרָה לְגַסֵּי הָרוּחַ מִנַּיִן? שֶׁנֶּאֱמַר: וְרָם לְבָבֶךָ וְשָׁכַחְתָּ׳.

אָסוּר לְהִתְגָּאוֹת בְּמָמוֹן, בְּיֹפִי וּבְחָכְמָה – הֵם מַתָּנָה.

🍽️

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי מוֹדֶה לַה׳, הַלֵּב שֶׁלִּי נִהְיֶה צָנוּעַ וְקָרוֹב אֵלָיו 💛

📖 עֲנָוָה וְנַחַת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דְּבָרִים ח׳ י׳–י״ז · חִדּוּשִׁים לְפָרָשַׁת עֵקֶב

רָמָה ב׳ – תּוֹדָה נֶגֶד גַּאֲוָה

🍽️ בִּרְכַּת הַמָּזוֹן מוּל ׳כֹּחִי׳

כְּשֶׁמִּסְתַּכְּלִים בַּפְּסוּקִים רוֹאִים שֶׁהַתּוֹרָה מַצִּיעָה אֶת הַהוֹדָאָה לַה׳ כְּהַצָּלָה מִגַּאֲוָה. לָכֵן בִּרְכַּת הַמָּזוֹן מוֹפִיעָה בַּפָּרָשָׁה בְּסָמוּךְ לִפְנֵי אִסּוּר הַגַּאֲוָה.

מִי שֶׁמּוֹדֶה לַה׳ עַל הָאֹכֶל לֹא יֹאמַר: ׳כֹּחִי וְעֹצֶם יָדִי עָשָׂה לִי אֶת הַחַיִל הַזֶּה׳.

וְכָךְ נוֹהֵג מִי שֶׁעָנָו: ׳עָנָו מְאֹד וְשָׁפָל בֶּרֶךְ לִפְנֵי ה׳ וְלִפְנֵי אֲנָשִׁים׳, וּמוֹדֶה לְבוֹרְאוֹ עַל הַמַּעֲלָה שֶׁחָנַן אוֹתוֹ.

הַהוֹדָאָה לַה׳ מַצִּילָה מִן הַמַּחֲשָׁבָה ׳כֹּחִי וְעֹצֶם יָדִי׳.

💧

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִתְמַעֵט בְּעֵינַי כְּדֵי שֶׁתּוֹרַת ה׳ תִּשָּׁאֵר אֶצְלִי 💛

📖 עֲנָוָה וְנַחַת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מַהֲרַ״ל, דֶּרֶךְ חַיִּים · תַּעֲנִית ז׳ א׳

רָמָה ב׳ – תּוֹדָה נֶגֶד גַּאֲוָה

💧 מַיִם הוֹלְכִים לַנָּמוּךְ

הַמַּהַרַ״ל מְבָאֵר: הַתּוֹרָה הִיא שֵׂכֶל אֱלֹהִי, וְהָאָדָם בַּעַל גּוּף. כְּדֵי לְקַבֵּל אוֹתָהּ צָרִיךְ לְהִתְרַחֵק מִמִּדּוֹת גַּשְׁמִיּוֹת. גַּסּוּת הָרוּחַ הִיא מִדָּה גַּשְׁמִית, וְהָעֲנָוָה הִיא הַמִּדָּה הַנִּבְדֶּלֶת מִן הַחֹמֶר.

לָכֵן ׳אֵין הַתּוֹרָה מִתְקַיֶּמֶת בְּגַסֵּי רוּחַ׳, וְדִבְרֵי תּוֹרָה נִמְשְׁלוּ לְמַיִם שֶׁהוֹלְכִים לַמָּקוֹם הַנָּמוּךְ.

וּמֹשֶׁה רַבֵּנוּ זָכָה לַתּוֹרָה בִּגְלַל עַנְוָתָנוּתוֹ: ׳הוֹאִיל וּמִעַטְתָּ אֶת עַצְמְךָ, תִּקָּרֵא עַל שִׁמְךָ׳.

הָעֲנָוָה הִיא הַמִּדָּה הַחֲשׁוּבָה לִקְנִיַּת הַתּוֹרָה.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹאֶה בְּכָל מַתָּנָה אַהֲבָה מֵה׳ 💛

📖 עֲנָוָה וְנַחַת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, חִדּוּשִׁים לְפָרָשַׁת עֵקֶב

רָמָה ב׳ – תּוֹדָה נֶגֶד גַּאֲוָה

🌟 הָ׳אֲנִי׳ שֶׁמַּסְתִּיר, וְהַתּוֹדָה שֶׁמְּגַלָּה

הַגַּאֲוָה מַסְתִּירָה אֶת ה׳ מֵאֲחוֹרֵי הַ׳אֲנִי׳: כְּשֶׁאֲנִי חוֹשֵׁב שֶׁהַכֹּל בִּזְכוּתִי, אֵין לִי סִבָּה לִזְכֹּר אֶת הַנּוֹתֵן.

הָעֲנָוָה מְגַלָּה שֶׁכָּל מַה שֶּׁיֵּשׁ לָנוּ הוּא מַתָּנָה. וּמִי שֶׁרוֹאֶה מַתָּנוֹת, אוֹהֵב אֶת הַנּוֹתֵן.

וְעוֹד לָמַדְנוּ: כָּל שֶׁה׳ נוֹתֵן לָנוּ יוֹתֵר, כָּךְ עָלֵינוּ לִזְכֹּר יוֹתֵר מֵאַיִן זֶה בָּא.

הַגַּאֲוָה הִיא שִׁכְחָה, וְהָעֲנָוָה הִיא זִכָּרוֹן.

רָמָה ב׳ – תּוֹדָה נֶגֶד גַּאֲוָה

מָה לוֹקְחִים מִרָמָה ב׳?

כָּל טוֹבָה שֶׁאֲנִי מְקַבֵּל מַזְמִינָה אוֹתִי לִזְכֹּר אֶת הַנּוֹתֵן. מִי שֶׁמּוֹדֶה רוֹאֶה בְּכָל מַתָּנָה אַהֲבָה, וְלִבּוֹ נִפְתָּח לֶאֱהֹב אֶת ה׳.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • עַל אֵיזוֹ הַצְלָחָה שֶׁלִּי אֲנִי יָכוֹל לוֹמַר הַיּוֹם: ׳תּוֹדָה, ה׳, שֶׁנָּתַתָּ לִי כֹּחַ׳?
  • מָתַי הַשָּׁבוּעַ הָיָה לִי קַל לוֹמַר ׳הַכֹּל בִּזְכוּתִי׳, וְאֵיךְ אֶזְכֹּר אֶת הַנּוֹתֵן בַּפַּעַם הַבָּאָה?
  • אֶשְׁפֹּךְ מַיִם מִכּוֹס וְאֶרְאֶה לְאָן הֵם הוֹלְכִים, וְאֶכְתֹּב שָׁלוֹשׁ טוֹבוֹת שֶׁקִּבַּלְתִּי מֵה׳ וְאֵינִי רוֹצֶה לִשְׁכֹּחַ.

הַשֵּׂכֶל יוֹדֵעַ שֶׁה׳ מֵזִיז אֶת יָדִי, אֲבָל בַּלֵּב נִדְמֶה לִי שֶׁאֲנִי עוֹשֶׂה זֹאת לְבַד. לָכֵן מְבַקְּשִׁים בַּתְּפִלָּה: שֶׁמָּה שֶׁהַשֵּׂכֶל יוֹדֵעַ יַשְׁפִּיעַ גַּם עַל הַלֵּב. אֲנִי יָכוֹל לַעֲשׂוֹת רַק אֶת הַהַתְחָלָה, וְה׳ עוֹזֵר לִי לִקְנוֹת אֶת זֶה בַּלֵּב.

וּבְסוֹף כָּל בְּרָכָה כְּדַאי לְהַרְגִּישׁ שִׂמְחָה: ׳יֵשׁ לִי אַבָּא שֶׁיֵּשׁ לוֹ אֶת כָּל הַיְּכוֹלוֹת הַנִּפְלָאוֹת׳.

אֲנִי נִבְרֵאתִי לְהוֹדוֹת, וַאֲנִי מְבַקֵּשׁ שֶׁהַלֵּב יַרְגִּישׁ מָה שֶׁהַשֵּׂכֶל יוֹדֵעַ.

🕎 מוֹדִים דְּרַבָּנָן וְעַל הַנִּסִּים

׳מוֹדִים דְּרַבָּנָן׳ מוֹדֶה לַה׳ שֶׁהוּא ׳אֱלֹהֵי כָל בָּשָׂר, יוֹצְרֵנוּ יוֹצֵר בְּרֵאשִׁית׳, וּמְבַקֵּשׁ: ׳כֵּן תְּחַיֵּנוּ וּתְקַיְּמֵנוּ׳, כְּמוֹ שֶׁהֶחֱיִיתָנוּ וְקִיַּמְתָּנוּ. הוּא נִגְמָר: ׳עַל שֶׁאֲנַחְנוּ מוֹדִים לָךְ. בָּרוּךְ אֵל הַהוֹדָאוֹת׳.

בְּ׳עַל הַנִּסִּים׳ שֶׁל חֲנֻכָּה: מַלְכוּת יָוָן רָצְתָה ׳לְהַשְׁכִּיחָם תּוֹרָתֶךָ וּלְהַעֲבִירָם מֵחֻקֵּי רְצוֹנֶךָ׳, וְה׳ עָמַד לָהֶם ׳בְּעֵת צָרָתָם׳. הַנֵּס: ׳גִּבּוֹרִים בְּיַד חַלָּשִׁים, וְרַבִּים בְּיַד מְעַטִּים, וּטְמֵאִים בְּיַד טְהוֹרִים, וּרְשָׁעִים בְּיַד צַדִּיקִים, וְזֵדִים בְּיַד עוֹסְקֵי תוֹרָתֶךָ׳.

אַחֲרֵי הַנִּצָּחוֹן: ׳פִּנּוּ אֶת הֵיכָלֶךָ, וְטִהֲרוּ אֶת מִקְדָּשֶׁךָ, וְהִדְלִיקוּ נֵרוֹת בְּחַצְרוֹת קָדְשֶׁךָ׳, וְקָבְעוּ שְׁמוֹנַת יְמֵי חֲנֻכָּה ׳לְהוֹדוֹת וּלְהַלֵּל לְשִׁמְךָ הַגָּדוֹל׳.

הַנֵּס הוּא שֶׁהַחַלָּשִׁים וְהַמְעַטִּים נִצְּחוּ – וְעַל זֶה מוֹדִים וּמְהַלְּלִים.

🏺 שִׂים שָׁלוֹם – לָמָּה דַּוְקָא ׳שִׂים׳?

בִּבְרָכַת כֹּהֲנִים רֹב הַבְּרָכוֹת הֵן עַל מָה שֶׁה׳ עוֹשֶׂה אִתָּנוּ: ׳יְבָרֶכְךָ, יִשְׁמְרֶךָ, יָאֵר פָּנָיו, וִיחֻנֶּךָּ׳. רַק הַשָּׁלוֹם הוּא בֵּין הָאֲנָשִׁים. לָכֵן אֶפְשָׁר לְהָבִין שֶׁנֶּאֱמַר ׳וְיָשֵׂם לְךָ שָׁלוֹם׳ – לָשִׂים וּלְהַנִּיחַ אֶת הַשָּׁלוֹם בְּתוֹכֵנוּ.

וּבְ׳שִׂים שָׁלוֹם׳: ׳כִּי בְאוֹר פָּנֶיךָ נָתַתָּ לָּנוּ... תּוֹרַת חַיִּים וְאַהֲבַת חֶסֶד, וּצְדָקָה וּבְרָכָה וְרַחֲמִים וְחַיִּים וְשָׁלוֹם׳. חֶסֶד וְרַחֲמִים הֵן שְׁתֵּי מִדּוֹת שֶׁל ה׳ שֶׁהוֹלְכוֹת הַרְבֵּה פְּעָמִים יַחַד; וּכְשֶׁאָדָם נוֹהֵג בְּחֶסֶד, גַּם ה׳ נוֹהֵג עִמּוֹ בְּחֶסֶד.

וְהַמָּשָׁל: הַשָּׁלוֹם הוּא הַכְּלִי. אַחֲרֵי הַתְּפִלָּה קִבַּלְנוּ שֶׁפַע, וּבְלִי שָׁלוֹם הוּא נִשְׁפָּךְ מֵהַחוֹר שֶׁבַּכְּלִי. הַבְּרָכָה נִגְמֶרֶת: ׳הַמְבָרֵךְ אֶת עַמּוֹ יִשְׂרָאֵל בַּשָּׁלוֹם׳.

הַשָּׁלוֹם הוּא הַכְּלִי שֶׁמַּחֲזִיק אֶת כָּל הַשֶּׁפַע – בְּלִי שָׁלוֹם הַכֹּל נִשְׁפָּךְ.

6
💡

רָמָה ב׳ – מִן הַשֵּׂכֶל אֶל הַלֵּב

מָה לוֹקְחִים מִרָמָה ב׳?

ה׳ מְקַבֵּל אֶת הַתְּפִלָּה בְּאַהֲבָה, זוֹכֵר אוֹתָנוּ לְטוֹבָה, וְכָל הַחַיִּים וְהַטּוֹבוֹת שֶׁלָּנוּ הֵם מַתָּנָה מִמֶּנּוּ. כְּשֶׁאֲנִי אוֹמֵר ׳מוֹדִים׳ וּמַרְגִּישׁ אֶת הַתּוֹדָה בַּלֵּב, וְעוֹשֶׂה שָׁלוֹם עִם הַחֲבֵרִים, אֲנִי מַרְאֶה לְאַבָּא שֶׁבַּשָּׁמַיִם כַּמָּה אֲנִי אוֹהֵב אוֹתוֹ.

הַטּוֹב שִׁמְךָ וּלְךָ נָאֶה לְהוֹדוֹת

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • עַל אֵילוּ שָׁלוֹשׁ טוֹבוֹת שֶׁקִּבַּלְתִּי הַיּוֹם אֲנִי רוֹצֶה לוֹמַר לַה׳ ׳מוֹדִים אֲנַחְנוּ לָךְ׳?
  • כְּשֶׁאֲנִי אוֹמֵר ׳רְצֵה׳, אֵיךְ אֲנִי מַרְגִּישׁ שֶׁה׳ מְקַבֵּל אֶת הַתְּפִלָּה שֶׁלִּי בְּאַהֲבָה?
  • עִם מִי רִבִּיתִי הַשָּׁבוּעַ, וּמָה אֲנִי יָכוֹל לַעֲשׂוֹת כְּדֵי לַעֲשׂוֹת שָׁלוֹם לִפְנֵי שֶׁאֲנִי אוֹמֵר ׳שִׂים שָׁלוֹם׳?

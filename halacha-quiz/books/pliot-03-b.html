👑 לָמָּה אוֹמְרִים ׳גָּלוּי וְיָדוּעַ׳?

הַנּוּסָח מוּזָר: ׳גָּלוּי וְיָדוּעַ לִפְנֵי כִסֵּא כְבוֹדֶךָ׳. הֲרֵי אֲנַחְנוּ לֹא מְסַפְּרִים לַמֶּלֶךְ דָּבָר חָדָשׁ! הָרַב גּוֹלְדְוַסֶּר וְהָרַב פִּינְקוּס מְבָאֲרִים: זוֹ בַּקָּשָׁה סְמוּיָה. כְּמוֹ מִי שֶׁאוֹמֵר לַמֶּלֶךְ ׳אַתָּה יוֹדֵעַ שֶׁבִּלְעָדֵי זֶה אֲנִי לֹא אוּכַל לִחְיוֹת׳ – וְהַמֶּלֶךְ מֵבִין.

הָרַב פִּינְקוּס מוֹסִיף: אֲנַחְנוּ לֹא מְסַפְּרִים, אֶלָּא מְתָאֲרִים חָכְמָה נִפְלָאָה שֶׁרַק ה׳ יָכוֹל לְהָבִין עַד הַסּוֹף.

וְהַגְּר״א מְבָאֵר שֶׁ׳כִּסֵּא כְבוֹדֶךָ׳ בָּא לְהַגִּיד: ה׳ מַשְׁגִּיחַ גַּם עַל דְּבָרִים שֶׁנִּרְאִים לָנוּ פְּשׁוּטִים וּקְטַנִּים.

גָּלוּי וְיָדוּעַ = בַּקָּשָׁה סְמוּיָה, וְתֵאוּר שֶׁל חָכְמָה שֶׁרַק ה׳ מֵבִין.

🎚️ יִפָּתַח וְיִסָּתֵם

הַלְּבוּשׁ מְבָאֵר שֶׁ׳יִפָּתַח׳ וְ׳יִסָּתֵם׳ אֵינָם אוֹמְרִים סְתִימָה אוֹ פְּתִיחָה גְּמוּרָה, אֶלָּא יוֹתֵר מִדַּי אוֹ פָּחוֹת מִדַּי מִמָּה שֶׁצָּרִיךְ. הָעָרוּךְ הַשֻּׁלְחָן מוֹסִיף: ה׳ בָּרָא אֶת כָּל אֵלֶּה ׳בְּמִדָּה וּבְמִשְׁקָל כְּפִי צָרְכּוֹ׳.

הָאַשְׁכְּנַזִּים אוֹמְרִים ׳שֶׁאִם יִפָּתֵחַ אוֹ יִסָּתֵם׳, וְהַסְּפָרַדִּים אוֹמְרִים ׳שֶׁאִם יִסָּתֵם אוֹ יִפָּתֵחַ׳. כָּל קְהִלָּה נוֹהֶגֶת כְּמִנְהָגָהּ.

וְהַמִּלִּים ׳וְלַעֲמֹד לְפָנֶיךָ׳ מְלַמְּדוֹת: בְּלִי גּוּף בָּרִיא קָשֶׁה גַּם לְהִתְפַּלֵּל וּלְבָרֵךְ.

יִפָּתַח וְיִסָּתֵם = יוֹתֵר מִדַּי אוֹ פָּחוֹת מִדַּי מִן הַמִּדָּה הַמְדֻיֶּקֶת.

בַּגְּמָרָא אָמַר רַב שֶׁחוֹתְמִים ׳רוֹפֵא חוֹלִים׳, וְרַב פָּפָּא סִיֵּם: אוֹמְרִים גַּם ׳רוֹפֵא כָל בָּשָׂר׳ וְגַם ׳מַפְלִיא לַעֲשׂוֹת׳.

הַשֻּׁלְחָן עָרוּךְ מְבָאֵר: ׳רוֹפֵא כָל בָּשָׂר׳ עַל שֵׁם הַנְּקָבִים שֶׁדַּרְכָּם הַגּוּף מוֹצִיא מָה שֶׁלֹּא טוֹב לוֹ, וְזוֹ רְפוּאָה. וְהַדְּרִישָׁה מוֹסִיף: יֵשׁ רְפוּאָה גַּם בְּלִי חוֹלִי. מָה שֶׁה׳ מְקַיֵּם אֶת הָאִזּוּן בַּגּוּף, שֶׁדָּבָר אֶחָד לֹא יִתְגַּבֵּר עַל חֲבֵרוֹ, זֹאת הָרְפוּאָה.

לָכֵן מוֹדִים גַּם כְּשֶׁהַכֹּל תַּקִּין.

רְפוּאָה אֵינָהּ רַק כְּשֶׁחוֹלִים: קִיּוּם הָאִזּוּן בַּגּוּף הוּא רְפוּאָה.

הָרַב פִּינְקוּס מְסַבֵּר: יֵשׁ דְּבָרִים שֶׁאִי אֶפְשָׁר לְחַבֵּר, כְּמוֹ לְהַכְנִיס פָּסוּק לְתוֹךְ סִיר. וְאַף עַל פִּי כֵן ה׳ חִבֵּר נֶפֶשׁ רוּחָנִית לְגוּף גַּשְׁמִי. זֶה פֶּלֶא עָצוּם.

מַפְלִיא לַעֲשׂוֹת: הָרוּחַ נִשְׁמֶרֶת, הַגּוּף בּוֹרֵר, וְהַנְּשָׁמָה מְחֻבֶּרֶת לַגּוּף.

😊 שִׂמְחָה מְחַזֶּקֶת

הָרֶמָ״א אוֹמֵר שֶׁהַנְּשָׁמָה נִשְׁמֶרֶת בַּגּוּף כְּשֶׁהַגּוּף בָּרִיא. וְיֵשׁ גַּם הֶפֶךְ: בְּרִיאוּת הַנֶּפֶשׁ מְחַזֶּקֶת אֶת הַגּוּף.

הַגְּר״א מְפָרֵשׁ אֶת הַפָּסוּק ׳רוּחַ־אִישׁ יְכַלְכֵּל מַחֲלֵהוּ וְרוּחַ נְכֵאָה מִי יִשָּׂאֶנָּה׳: מִי שֶׁשָּׂמֵחַ תָּמִיד, הַשִּׂמְחָה עוֹזֶרֶת לוֹ לְהִתְחַזֵּק גַּם כְּשֶׁקָּשֶׁה. אֲבָל רוּחַ עֲצוּבָה כְּבֵדָה מְאֹד לָשֵׂאת.

לָכֵן הַהוֹדָיָה עַל הַגּוּף אֵינָהּ רַק מִלִּים: כְּשֶׁמּוֹדִים, הַנֶּפֶשׁ שְׂמֵחָה, וְהַשִּׂמְחָה טוֹבָה גַּם לַגּוּף.

  • הַיּוֹם אֲבָרֵךְ ׳אֲשֶׁר יָצַר׳ אַחַת בְּכַוָּנָה, לְאַט וּמִתּוֹךְ הַסִּדּוּר, וְאֶחְשֹׁב עַל פֶּלֶא אֶחָד שֶׁלָּמַדְנוּ.

עַל מָה אֶפְשָׁר לְוַתֵּר, עַל מָה לֹא, וְאֵיךְ חוֹלְקִים בְּשָׁלוֹם בְּאַרְבָּעָה סִימָנִים

🧭

💛 אַהֲבַת הַשֵּׁם: אֲנִי יוֹדֵעַ לְוַתֵּר וְיוֹדֵעַ לַעֲמֹד, וְשְׁנֵיהֶם מְקָרְבִים לַה׳ 💛

📖 שָׁלוֹם וְוִתּוּר – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בָּבָא מְצִיעָא ל׳ ב׳ · מִשְׁנָה בְּרוּרָה

רָמָה ב׳ – אַרְבָּעָה סִימָנִים

🧭 שְׁלוֹשָׁה מַצָּבִים

אִם יֵשׁ סַכָּנָה רוּחָנִית אֲמִתִּית – מָתַי מְוַתְּרִים וּמָתַי עוֹמְדִים? יֵשׁ לְחַלֵּק לִשְׁלוֹשָׁה מַצָּבִים. כְּשֶׁמְּדֻבָּר בְּמָמוֹן וְכַדּוֹמֶה – מִצְוָה לְוַתֵּר כְּדֵי שֶׁלֹּא תִּהְיֶה מַחֲלֹקֶת.

כְּשֶׁמְּדֻבָּר בְּהִדּוּר בִּלְבַד, כְּלוֹמַר אֹפֶן יָפֶה יוֹתֵר לְקַיֵּם מִצְוָה – לִפְעָמִים כְּדַאי לְוַתֵּר כְּדֵי לְהִמָּנַע מִמַּחֲלֹקֶת.

וּכְשֶׁמְּדֻבָּר בְּמִצְוָה גְּמוּרָה אוֹ בְּאִסּוּר, אֲפִלּוּ מִדִּבְרֵי חֲכָמִים – אֵין שׁוּם הֶתֵּר לְוַתֵּר, כְּמוֹ שֶׁכָּתַב הַמִּשְׁנָה בְּרוּרָה.

בְּמָמוֹן מְוַתְּרִים, בְּהִדּוּר לִפְעָמִים, וּבְמִצְוָה גְּמוּרָה אוֹ אִסּוּר – לֹא.

1
🏛️

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְוַתֵּר עַל שֶׁלִּי, כְּדֵי שֶׁיִּהְיֶה שָׁלוֹם בֵּין בְּנֵי ה׳ 💛

📖 שָׁלוֹם וְוִתּוּר – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בָּבָא מְצִיעָא ל׳ ב׳

רָמָה ב׳ – אַרְבָּעָה סִימָנִים

🏛️ לִפְנִים מִשּׁוּרַת הַדִּין

הַגְּמָרָא אוֹמֶרֶת: ׳לֹא חָרְבָה יְרוּשָׁלַיִם אֶלָּא עַל שֶׁהֶעֱמִידוּ דִּבְרֵיהֶם עַל דִּין תּוֹרָה, וְלֹא עָשׂוּ לִפְנִים מִשּׁוּרַת הַדִּין׳.

כְּלוֹמַר, כָּל אֶחָד עָמַד עַל מָה שֶׁמַּגִּיעַ לוֹ, וְלֹא וִתֵּר. הַוִּתּוּר הוּא לִפְעָמִים בְּדִיּוּק מַה שֶּׁשּׁוֹמֵר עַל הַשָּׁלוֹם.

אֲבָל הַוִּתּוּר הוּא עַל מָה שֶׁשַּׁיָּךְ לִי – כֶּסֶף וְכָבוֹד – וְלֹא עַל דְּבַר ה׳.

וִתּוּר עַל שֶׁלִּי שׁוֹמֵר עַל שָׁלוֹם, אֲבָל לֹא עַל דְּבַר ה׳.

2
🔎

💛 אַהֲבַת הַשֵּׁם: מַחֲלֹקֶת נְקִיָּה אֶפְשָׁר לְהָבִיא לִפְנֵי ה׳ בְּלִי בּוּשָׁה 💛

📖 שָׁלוֹם וְוִתּוּר – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, רַמְבַּ״ם הִלְכוֹת תְּשׁוּבָה ג׳ י״ד · יְבָמוֹת י״ג ע״ב

רָמָה ב׳ – אַרְבָּעָה סִימָנִים

🔎 אַרְבָּעָה סִימָנִים

אֵיךְ יוֹדְעִים אִם מַחֲלֹקֶת הִיא לְשֵׁם שָׁמַיִם? אַרְבָּעָה סִימָנִים: (א) הַצָּגָה הוֹגֶנֶת שֶׁל הַצַּד הַשֵּׁנִי, וְלֹא לְצַיֵּר אוֹתוֹ כְּמַגְחִיךְ כְּדֵי לְנַצֵּחַ. (ב) בְּלִי כִּנּוּיִים – הַהֲלָכָה אוֹסֶרֶת לְכַנּוֹת שֵׁם לֶחָבֵר, וְגַם אָסוּר לִנְקֹם בְּמִי שֶׁחוֹלֵק עָלַי.

(ג) שְׁמִירָה עַל אֵמוּן: הַמִּשְׁנָה בִּיבָמוֹת מְסַפֶּרֶת שֶׁבֵּית שַׁמַּאי וּבֵית הִלֵּל לֹא נִמְנְעוּ לְהִתְחַתֵּן אֵלּוּ עִם אֵלּוּ. (ד) עִנְיָן וְלֹא אָדָם: מַחֲלֹקֶת אֲסוּרָה הִיא אִישִׁית, וּמֻתֶּרֶת הִיא עַל נוֹשֵׂא מְסֻיָּם.

הוֹגְנוּת, בְּלִי כִּנּוּיִים, אֵמוּן, וְעִנְיָן וְלֹא אָדָם.

3
🔭

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְחַפֵּשׂ אֶת הַטּוֹב שֶׁכֻּלָּנוּ רוֹצִים, וְכָךְ מִתְקָרְבִים לַה׳ 💛

📖 שָׁלוֹם וְוִתּוּר – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – אַרְבָּעָה סִימָנִים

🔭 עִקָּר גָּבוֹהַּ יוֹתֵר

וִתּוּר פּוֹתֵר הַרְבֵּה, אֲבָל רַק בִּדְבָרִים שֶׁאֵינָם עִקָּרִיִּים מִדַּי. בְּדָבָר עִקָּרִי הַוִּתּוּר לֹא הוֹלֵךְ, כִּי כָּל צַד מַרְגִּישׁ שֶׁהוּא מְוַתֵּר עַל מַשֶּׁהוּ שֶׁאָסוּר לוֹ לְוַתֵּר עָלָיו.

אָז נִכְנָס מוֹדֶל שֵׁנִי: לֹא וִתּוּר עַל הָעִקָּר, אֶלָּא מְצִיאַת עִקָּר גָּבוֹהַּ יוֹתֵר שֶׁשְּׁנֵי הַצְּדָדִים שׁוּתָפִים לוֹ.

שׁוֹאֲלִים: מָה הַצַּד הַשֵּׁנִי רוֹצֶה בֶּאֱמֶת, וּמָה טוֹב בֶּאֱמֶת לְכֻלָּנוּ? בְּדֶרֶךְ כְּלָל שְׁנֵי הַצְּדָדִים רוֹצִים אֶת הַטּוֹב, הֵם רַק חֲלוּקִים מַה טּוֹב.

כְּשֶׁהַוִּתּוּר לֹא מַסְפִּיק, מְחַפְּשִׂים עִקָּר גָּבוֹהַּ מְשֻׁתָּף.

4
🪞

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹאֶה אֶת הַלֵּב שֶׁלִּי, וַאֲנִי רוֹצֶה שֶׁיִּהְיֶה נָקִי 💛

📖 שָׁלוֹם וְוִתּוּר – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשְׁלֵי כ״א ב׳

רָמָה ב׳ – אַרְבָּעָה סִימָנִים

🪞 מַרְאָה כֵּנָה לַלֵּב

בַּפַּעַם הַבָּאָה שֶׁאֲנִי מִתְוַכֵּחַ, אֶבְדֹּק אֶת עַצְמִי בְּאַרְבַּעַת הַסִּימָנִים – הֵם מַרְאָה כֵּנָה לַלֵּב.

ה׳ הוּא ׳תֹּכֵן לִבּוֹת׳, הוּא רוֹאֶה מָה בֶּאֱמֶת מְנִיעַ אוֹתִי. מַחֲלֹקֶת נְקִיָּה הִיא מַחֲלֹקֶת שֶׁאֶפְשָׁר לְהָבִיא לְפָנָיו בְּלִי בּוּשָׁה.

בּוֹדְקִים: הוֹגֵן? בְּלִי כִּנּוּי? אֵמוּן? עִנְיָן?

5
💛

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹהֵב אֶת חֲבֵרִי וְאֶת ה׳, וְלָכֵן אֲנִי יוֹדֵעַ מָתַי לְוַתֵּר וּמָתַי לַעֲמֹד 💛

📖 שָׁלוֹם וְוִתּוּר – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשְׁנָה בְּרוּרָה · בָּבָא מְצִיעָא ל׳ ב׳

רָמָה ב׳ – אַרְבָּעָה סִימָנִים

💛 מֵאֵיפֹה בָּאָה הָאַהֲבָה?

הַוִּתּוּר עַל שֶׁלִּי וְהָעֲמִידָה עַל שֶׁל ה׳ נוֹבְעִים מֵאוֹתָהּ אַהֲבָה. אֲנִי מְוַתֵּר כִּי אֲנִי אוֹהֵב אֶת חֲבֵרִי, וַאֲנִי עוֹמֵד כִּי אֲנִי אוֹהֵב אֶת ה׳.

עַל הַכָּבוֹד וְהַמָּמוֹן שֶׁלִּי – לְוַתֵּר. עַל דְּבַר ה׳ – לַעֲמֹד. וְגַם כְּשֶׁעוֹמְדִים – בְּדֶרֶךְ שֶׁל שָׁלוֹם.

וִתּוּר וַעֲמִידָה נוֹבְעִים מֵאוֹתָהּ אַהֲבָה.

מָה לוֹקְחִים מִרָמָה ב׳?

מִי שֶׁיּוֹדֵעַ מָתַי לְוַתֵּר וְאֵיךְ לַעֲמֹד בְּשָׁלוֹם, מְקַיֵּם אֶת רָצוֹן ה׳ וְאוֹהֵב אֶת בָּנָיו. ה׳ תֹּכֵן לִבּוֹת, וְהוּא אוֹהֵב לֵב נָקִי.

🤔 חוֹשְׁבִים עַל זֶה

  • עַל מָה קָשֶׁה לִי לְוַתֵּר, וְעַל מָה לֹא כְּדַאי לִי לְוַתֵּר?
  • מָתַי הִצַּגְתִּי אֶת דַּעַת הַשֵּׁנִי בְּהֲגִינוּת, וּמָתַי הִצַּרְתִּי אוֹתָהּ בְּצוּרָה מְגֻחֶכֶת?
  • אֲקַבֵּל מִקְרִים בִּפְתָקִים וְאַחְלִיט: מְוַתֵּר אוֹ עוֹמֵד – וְאֶסְבִּיר לָמָּה.

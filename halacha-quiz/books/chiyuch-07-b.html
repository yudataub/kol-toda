ה׳ שֹׁמְרֶךָ ה׳ צִלְּךָ עַל יַד יְמִינֶךָ

😊 ה׳ צִלְּךָ · רָמָה ב׳

ה׳ צִלְּךָ – כְּמַיִם הַפָּנִים לַפָּנִים

רָמָה ב׳ – הַמִּדְרָשׁ, הַזֹּהַר וְאוֹר הַפָּנִים

אֵיךְ הַחִיּוּךְ שֶׁלָּנוּ מְעוֹרֵר חִיּוּךְ גַּם מִלְּמַעְלָה: הַצֵּל, הַמַּרְאָה וְאוֹר הַפָּנִים

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְהִתְנַהֵג עִם ה׳ בְּשִׂמְחָה וּבְאַהֲבָה 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִדְרָשׁ, נֶפֶשׁ הַחַיִּים א׳ ז׳

רָמָה ב׳ – הַמִּדְרָשׁ, הַזֹּהַר וְאוֹר הַפָּנִים

📜 הַמִּדְרָשׁ: ה׳ צִלְּךָ

אָמַר ה׳ לְמֹשֶׁה: ׳לֵךְ אֱמֹר לָהֶם לְיִשְׂרָאֵל: שְׁמִי אֶהְיֶה אֲשֶׁר אֶהְיֶה׳. מָה פֵּרוּשׁ ׳אֶהְיֶה אֲשֶׁר אֶהְיֶה׳? כְּשֵׁם שֶׁאַתָּה הֹוֶה עִמִּי, כָּךְ אֲנִי הֹוֶה עִמְּךָ.

וְכֵן אָמַר דָּוִד: ׳ה׳ צִלְּךָ עַל יַד יְמִינֶךָ׳. מָה פֵּרוּשׁ ׳צִלְּךָ׳? כְּמוֹ הַצֵּל שֶׁלְּךָ: אִם אַתָּה מְשַׂחֵק לוֹ – הוּא מְשַׂחֵק לְךָ, וְאִם אַתָּה מַרְאֶה לוֹ פָּנִים זְעוּמוֹת – גַּם הוּא נוֹתֵן לְךָ כָּךְ.

כְּלוֹמַר: ה׳ מִתְנַהֵג אִתָּנוּ כְּמוֹ שֶׁאָנוּ מִתְנַהֲגִים אִתּוֹ.

הַצֵּל מַחְזִיר לָאָדָם אֶת מָה שֶׁהוּא עוֹשֶׂה לוֹ, וְכָךְ גַּם ה׳.

1

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹחֵר לַעֲמֹד בְּשִׂמְחָה, כְּדֵי שֶׁיָּאִירוּ לִי מִלְּמַעְלָה 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, זֹהַר, תְּצַוֶּה קפ״ד ב׳

רָמָה ב׳ – הַמִּדְרָשׁ, הַזֹּהַר וְאוֹר הַפָּנִים

🌗 הַזֹּהַר: עוֹלָם תַּחְתּוֹן וְעוֹלָם עֶלְיוֹן

הַזֹּהַר מְלַמֵּד: הָעוֹלָם הָעֶלְיוֹן נוֹתֵן לָעוֹלָם הַתַּחְתּוֹן לְפִי הַמַּצָּב שֶׁבּוֹ הוּא עוֹמֵד.

אִם הָאָדָם עוֹמֵד לְמַטָּה בַּהֲאָרַת פָּנִים – כָּךְ מְאִירִים לוֹ מִלְּמַעְלָה. וְאִם הוּא עוֹמֵד בְּעַצְבוּת – מְקַבֵּל מִלְּמַעְלָה לְפִי מַצָּבוֹ.

לָכֵן נֶאֱמַר ׳עִבְדוּ אֶת ה׳ בְּשִׂמְחָה׳: שִׂמְחַת הָאָדָם מוֹשֶׁכֶת אֵלָיו שִׂמְחָה עֶלְיוֹנָה.

שִׂמְחָה שֶׁל אָדָם מוֹשֶׁכֶת אֵלָיו שִׂמְחָה מִלְּמַעְלָה.

2

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי מִתְפַּלֵּל, אֲנִי מַרְגִּישׁ שֶׁה׳ מֵאִיר לִי פָּנִים 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁמוֹנֶה עֶשְׂרֵה, שִׂים שָׁלוֹם; מִדְרַשׁ תְּהִלִּים ד׳

רָמָה ב׳ – הַמִּדְרָשׁ, הַזֹּהַר וְאוֹר הַפָּנִים

🌟 כִּי בְאוֹר פָּנֶיךָ

בְּבִרְכַּת ׳שִׂים שָׁלוֹם׳ אוֹמְרִים: ׳בָּרְכֵנוּ אָבִינוּ כֻּלָּנוּ כְּאֶחָד בְּאוֹר פָּנֶיךָ, כִּי בְאוֹר פָּנֶיךָ נָתַתָּ לָּנוּ… תּוֹרַת חַיִּים וְאַהֲבַת חֶסֶד, וּצְדָקָה וּבְרָכָה וְרַחֲמִים וְחַיִּים וְשָׁלוֹם׳.

כָּל הַשֶּׁפַע בָּא מֵאוֹר פָּנָיו שֶׁל ה׳. וְכָךְ אָמְרוּ יִשְׂרָאֵל בְּמִדְרַשׁ תְּהִלִּים: ׳אֵין לָנוּ אֶלָּא הֵאָרַת פָּנֶיךָ׳.

וּבְבִרְכַּת כֹּהֲנִים, ׳יָאֵר ה׳ פָּנָיו אֵלֶיךָ׳, מְפָרֵשׁ רַשִׁ״י: יַרְאֶה לְךָ פָּנִים שׂוֹחֲקוֹת.

כָּל הַשֶּׁפַע מַגִּיעַ מֵאוֹר פָּנָיו שֶׁל ה׳.

3

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִזְכּוֹת לְאוֹר פָּנִים, וְלָכֵן אֲנִי מֵאִיר פָּנִים 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׂים שָׁלוֹם; אֶהְיֶה אֲשֶׁר אֶהְיֶה

רָמָה ב׳ – הַמִּדְרָשׁ, הַזֹּהַר וְאוֹר הַפָּנִים

🔆 אֵיךְ זוֹכִים לְאוֹר פָּנִים?

וְאֵיךְ זוֹכִים לְ׳אוֹר פָּנֶיךָ׳? עַל יְדֵי שֶׁאֲנַחְנוּ מְאִירִים פָּנִים – ׳אֶהְיֶה אֲשֶׁר אֶהְיֶה׳.

לָכֵן אֶפְשָׁר לְבָאֵר שֶׁזֹּאת הַבַּקָּשָׁה עַצְמָהּ: ׳בָּרְכֵנוּ… בְּאוֹר פָּנֶיךָ׳ – שֶׁהַפָּנִים שֶׁלָּנוּ יִהְיוּ מְאִירוֹת, כִּי שֶׁפַע הַשִּׂמְחָה מַגִּיעַ מֵאוֹר פָּנָיו שֶׁל ה׳.

וּכְשֶׁאָדָם מֵאִיר פָּנִים לַזּוּלַת, הוּא זוֹכֶה שֶׁגַּם ה׳ יָאִיר לוֹ פָּנִים – וְאֵין בְּרָכָה גְּדוֹלָה מִזּוֹ.

מִי שֶׁמֵּאִיר פָּנִים לַזּוּלַת, זוֹכֶה שֶׁה׳ יָאִיר לוֹ פָּנִים.

4

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁחֲבֵרַי יִגְדְּלוּ לְיַד הָאוֹר שֶׁלִּי 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, דְּבָרִים כ׳ י״ט

רָמָה ב׳ – הַמִּדְרָשׁ, הַזֹּהַר וְאוֹר הַפָּנִים

🌳 כְּמוֹ צֶמַח מוּל הַשֶּׁמֶשׁ

׳כִּי הָאָדָם עֵץ הַשָּׂדֶה׳. הָעֵץ זָקוּק לְחֹם הַשֶּׁמֶשׁ כְּדֵי לִגְדֹּל, וְעֵץ בְּלִי שֶׁמֶשׁ נוֹבֵל וּמִתְיַבֵּשׁ.

כָּךְ הָאָדָם, וּבְמֻיְחָד הַיֶּלֶד, זָקוּק לְחֹם וּלְהֵאָרַת פָּנִים כְּדֵי שֶׁנַּפְשׁוֹ תִּתְפַּתֵּחַ.

הַכֹּל מַגִּיעַ מֵהֵאָרַת פָּנִים: אֲנַחְנוּ מֵאוֹר פָּנָיו שֶׁל ה׳, וְהַזּוּלַת מֵאוֹר פָּנֵינוּ. ׳בּוֹאוּ וְנִהְיֶה הַשֶּׁמֶשׁ שֶׁל הָעוֹלָם׳.

הַזּוּלַת זָקוּק לְאוֹר הַפָּנִים שֶׁלָּנוּ כְּמוֹ עֵץ לַשֶּׁמֶשׁ.

5

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ שֶׁהַסְּגֻלָּה הַכִּי טוֹבָה הִיא חִיּוּךְ 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִדְרָשׁ וְזֹהַר

רָמָה ב׳ – הַמִּדְרָשׁ, הַזֹּהַר וְאוֹר הַפָּנִים

🎁 סְגֻלָּה לְשֶׁפַע

עַם יִשְׂרָאֵל הוּא ׳עַם סְגֻלָּה׳, וַאֲנָשִׁים מְחַפְּשִׂים סְגֻלּוֹת: לְפַרְנָסָה, לְשֶׁפַע וְלַנַּחַת מֵהַיְלָדִים.

וְהִנֵּה מְפֹרָשׁ בַּמִּדְרָשׁ וּבַזֹּהַר: הַשִּׂמְחָה וְהֵאָרַת הַפָּנִים מְזַכּוֹת אֶת הָאָדָם בְּכָל הַשֶּׁפַע הַזֶּה.

זוֹ סְגֻלָּה שֶׁכָּל יֶלֶד יָכוֹל לְקַיֵּם בְּכָל יוֹם, בְּלִי הוֹצָאָה וּבְלִי צֹרֶךְ בְּשׁוּם דָּבָר.

שִׂמְחָה וְהֵאָרַת פָּנִים הֵן סְגֻלָּה לְשֶׁפַע.

6

💛 אַהֲבַת הַשֵּׁם: אַחֲרֵי הַתְּפִלָּה אֲנִי מְחַיֵּךְ לָרִאשׁוֹן שֶׁאֲנִי פּוֹגֵשׁ 💛

📖 וּלְבֶן שִׁנַּיִם מֵחָלָב – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׂים שָׁלוֹם

רָמָה ב׳ – הַמִּדְרָשׁ, הַזֹּהַר וְאוֹר הַפָּנִים

🙏 כַּוָּנָה בִּתְפִלָּה

כְּשֶׁאוֹמְרִים בִּתְפִלָּה ׳בָּרְכֵנוּ אָבִינוּ כֻּלָּנוּ כְּאֶחָד בְּאוֹר פָּנֶיךָ׳, אֶפְשָׁר לְכַוֵּן: ׳רִבּוֹנוֹ שֶׁל עוֹלָם, תֵּן לִי פָּנִים מְאִירוֹת, שֶׁאוּכַל לְשַׂמֵּחַ אֶת בָּנֶיךָ׳.

וְאַחֲרֵי הַתְּפִלָּה? מְאִירִים פָּנִים לָרִאשׁוֹן שֶׁפּוֹגְשִׁים.

כָּךְ הַתְּפִלָּה הוֹפֶכֶת לְמַעֲשֶׂה.

מְבַקְשִׁים פָּנִים מְאִירוֹת, וְאַחַר כָּךְ נוֹתְנִים אוֹתָן.

💡

רָמָה ב׳ – הַמִּדְרָשׁ, הַזֹּהַר וְאוֹר הַפָּנִים

מָה לוֹקְחִים מִרָמָה ב׳?

הַפָּסוּק ׳ה׳ צִלְּךָ׳ מְלַמֵּד שֶׁה׳ קָרוֹב אֵלֵינוּ כְּמוֹ הַצֵּל. כְּשֶׁאֲנַחְנוּ מְאִירִים פָּנִים לַזּוּלַת, אֲנַחְנוּ מִתְקָרְבִים לַקָּדוֹשׁ בָּרוּךְ הוּא, וְהוּא מֵאִיר לָנוּ פָּנִים בַּחֲזָרָה. זוֹ אַהֲבָה בֵּין אַבָּא לִילָדָיו.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָה הָיָה הַהֶבְדֵּל כְּשֶׁחִיַּכְתִּי לַחֲבֵרִים הַיּוֹם, לְעֻמַּת כְּשֶׁהָיִיתִי כּוֹעֵס?
  • אֵיךְ הַצֵּל וְהַמַּרְאָה מְלַמְּדִים אוֹתִי אֵיךְ ה׳ נוֹהֵג עִמִּי?
  • הַשָּׁבוּעַ אֲנִי מְכַוֵּן בַּתְּפִלָּה ׳בְּאוֹר פָּנֶיךָ׳, וְאַחֲרֵי הַתְּפִלָּה מֵאִיר פָּנִים לָרִאשׁוֹן שֶׁאֲנִי פּוֹגֵשׁ.

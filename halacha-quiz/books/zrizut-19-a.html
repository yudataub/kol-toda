הָעֵצָה הָרִאשׁוֹנָה בַּמְּסִלַּת יְשָׁרִים: לְהִסְתַּכֵּל בְּרֹב הַטּוֹבוֹת שֶׁה׳ עוֹשֶׂה עִם הָאָדָם בְּכָל עֵת, מִיּוֹם שֶׁנּוֹלַד.

אֵין אָדָם שֶׁלֹּא רוֹאֶה בְּמַצָּבוֹ נִפְלָאוֹת: הֶעָשִׁיר חַיָּב לַה׳ עַל עָשְׁרוֹ, וְהֶעָנִי חַיָּב לוֹ שֶׁגַּם בְּעָנְיוֹ ה׳ מַמְצִיא לוֹ פַּרְנָסָה.

כְּשֶׁרוֹאִים אֶת הַטּוֹבוֹת – רוֹצִים לַעֲבֹד וְלֹא לְהִתְעַצֵּל.

הַדֶּרֶךְ הַשְּׁנִיָּה: לְהַכִּיר אֶת הָעֵרֶךְ שֶׁל הַמַּעֲשֶׂה. חֲכָמִים אָמְרוּ: אִלּוּ יָדַע רְאוּבֵן שֶׁהַתּוֹרָה תִּכְתֹּב עָלָיו, הָיָה מַצִּיל אֶת יוֹסֵף וּמוֹלִיךְ אוֹתוֹ עַל כְּתֵפוֹ אֶל אָבִיו. וְאִלּוּ יָדַע בּוֹעַז, הָיָה מַאֲכִיל אֶת רוּת עֲגָלִים פְּטוּמִים.

זֶה לֹא בִּשְׁבִיל לְהִתְפָּאֵר, אֶלָּא כִּי כְּשֶׁיּוֹדְעִים כַּמָּה הַמַּעֲשֶׂה חָשׁוּב, מִתְלַהֲבִים יוֹתֵר וּמְשַׁקְּעִים בּוֹ יוֹתֵר.

כְּשֶׁיּוֹדְעִים כַּמָּה מַעֲשֶׂה חָשׁוּב – מִתְלַהֲבִים.

🔍 אֵיפֹה אֲנִי מִתְעַצֵּל?

לֹא כֻּלָּם מִתְעַצְּלִים בְּאוֹתוֹ דָּבָר. אֶחָד זָרִיז מְאֹד בְּחֶסֶד, אֲבָל לַעֲבוֹדָה שֶׁלּוֹ הוּא מִתְעַצֵּל. שֵׁנִי אוֹהֵב לִלְמֹד גְּמָרָא וְרָץ אֶל הַשִּׁעוּר, אֲבָל לִתְפִלָּה הוּא מִתְעַצֵּל. שְׁלִישִׁי דַּוְקָא אוֹהֵב תְּפִלָּה, וְקָשֶׁה לוֹ לִלְמֹד.

לָכֵן כָּל אֶחָד צָרִיךְ לִבְדֹּק אֵיפֹה הוּא שֶׁלּוֹ מִתְעַצֵּל, וְלִלְמֹד עַד כַּמָּה הַדָּבָר חָשׁוּב.

כָּל אֶחָד מִתְעַצֵּל בְּמָקוֹם אַחֵר – וְצָרִיךְ לִמְצֹא אֵיפֹה.

🏃 שְׁאִיפָה גְּדוֹלָה

הַדֶּרֶךְ הָרְבִיעִית: לִשְׁאֹף הַרְבֵּה. אַבְרָהָם אָבִינוּ רָץ שָׁלוֹשׁ פְּעָמִים בְּפָרָשַׁת וַיֵּרָא: ׳וַיָּרָץ לִקְרָאתָם׳, ׳וְאֶל הַבָּקָר רָץ׳, ׳וַיְמַהֵר לַעֲשׂוֹת׳.

הוּא עָשָׂה זֹאת מִתּוֹךְ שְׁאִיפָה גְּדוֹלָה לַחֶסֶד, אַף שֶׁהָיָה חַלָּשׁ וְכוֹאֵב.

וְהַמִּדְרָשׁ אוֹמֵר: בִּשְׂכַר שָׁלֹשׁ הָרִיצוֹת, ה׳ יָרוּץ לִפְנֵי בָּנָיו שָׁלֹשׁ רִיצוֹת בִּמְעַמַּד הַר סִינַי.

שְׁאִיפָה גְּדוֹלָה גּוֹרֶמֶת לִרְצוֹן וְלִזְרִיזוּת.

הָעֵצָה הַמַּעֲשִׂית בַּסֵּפֶר ׳חוֹבַת הַתַּלְמִידִים׳: קֹדֶם כָּל עֲבוֹדָה, מִתְבּוֹנְנִים כַּמָּה זְמַן הִיא צְרִיכָה, וּמְצַוִּים עַל עַצְמֵנוּ לְסַיֵּם בַּזְּמַן הַזֶּה.

אַחֲרֵי שֶׁגּוֹמְרִים, בּוֹדְקִים: אִם לֹא אִחַרְנוּ – שְׂמֵחִים. אִם אִחַרְנוּ בְּלִי סִבָּה – מִצְטַעֲרִים.

כְּשֶׁמִּתְרַגְּלִים כָּךְ, מַרְגִּישִׁים כְּאִלּוּ מִישֶׁהוּ עוֹמֵד לְיָדֵנוּ וְאוֹמֵר: ׳אַל תִּתְרַשֵּׁל׳.

קוֹבְעִים זְמַן לְכָל עֲבוֹדָה, וְאַחַר כָּךְ בּוֹדְקִים.

📋 רְשִׁימַת הַיּוֹם

עוֹד עֵצָה: לִכְתֹּב עַל דַּף אֶת סֵדֶר הַיּוֹם: בַּבֹּקֶר בַּשָּׁעָה הַזֹּאת אֲנִי קָם, לוֹבֵשׁ, נוֹטֵל יָדַיִם, וְאַחַר כָּךְ מַתְחִיל לִלְמֹד. כָּל יוֹם מִסְתַּכְּלִים בָּרְשִׁימָה וּבוֹדְקִים אִם הִשְׁלַמְנוּ.

לֹא מַעֲמִיסִים הַרְבֵּה מִדַּי בְּבַת אַחַת: מוֹסִיפִים מְעַט מְעַט.

וְאִם פִּסְפַּסְנוּ בִּגְלַל רִשְׁלָנוּת, מִצְטַעֲרִים וּמַשְׁלִימִים בַּיּוֹם הַבָּא.

כּוֹתְבִים סֵדֶר יוֹם, בּוֹדְקִים, וּמַשְׁלִימִים מַה שֶׁחָסֵר.

הֶרְגֵּל וּתְנוּעָה מַהִירָה גּוֹרְמִים לַחֵשֶׁק לָבוֹא.

👣 הוֹלְכִים בְּעִקְבוֹת גְּדוֹלִים

הַדֶּרֶךְ הָאַחֲרוֹנָה: לִלְמֹד מִמַּעֲשֵׂי הַגְּדוֹלִים. חֲכָמִים אָמְרוּ: ׳חַיָּב אָדָם לוֹמַר: מָתַי יַגִּיעוּ מַעֲשַׂי לְמַעֲשֵׂי אֲבוֹתַי אַבְרָהָם יִצְחָק וְיַעֲקֹב׳.

וְכָתוּב עוֹד: כְּדַאי שֶׁכָּל יוֹם נַחֲשֹׁב שָׁעָה פְּנוּיָה: מָה עָשָׂה מֹשֶׁה רַבֵּנוּ, וּמָה עָשָׂה דָּוִד?

בִּשְׁבוּעוֹת הַקּוֹדְמִים לָמַדְנוּ סִפּוּרִים עַל גְּדוֹלִים זְרִיזִים – וְכָךְ גַּם אֲנִי לוֹמֵד מֵהֶם.

לוֹמְדִים מִמַּעֲשֵׂי הַגְּדוֹלִים וְרוֹצִים לִהְיוֹת כְּמוֹתָם.

רָמָה א׳ – דְּרָכִים לִזְרִיזוּת

מָה לוֹקְחִים מִרָמָה א׳?

זְרִיזוּת אֶפְשָׁר לִקְנוֹת: מִסְתַּכְּלִים בַּטּוֹבוֹת שֶׁה׳ עוֹשֶׂה, שׁוֹאֲפִים הַרְבֵּה, קוֹבְעִים זְמַן וּמִתְרַגְּלִים. הַהַתְבּוֹנְנוּת בְּחֶסֶד ה׳ מְעוֹרֶרֶת אֶת הָאַהֲבָה, וְהָאַהֲבָה מְעוֹרֶרֶת זְרִיזוּת.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • בְּאֵיזֶה תְּחוּם אֲנִי זָרִיז, וּבְאֵיזֶה תְּחוּם אֲנִי מִתְעַצֵּל?
  • מָה הַטּוֹבָה שֶׁה׳ עוֹשֶׂה לִי שֶׁעוֹזֶרֶת לִי לְהִתְעוֹרֵר לַעֲבוֹדָה?
  • אֶכְתֹּב טַבְלַת זְרִיזוּת: בְּכָל יוֹם אֲסַמֵּן אִם הָיִיתִי זָרִיז בִּשְׁלוֹשָׁה דְּבָרִים.

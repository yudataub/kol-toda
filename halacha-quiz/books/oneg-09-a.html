🎉 לְאָן לוֹקְחִים אֶת הַיְּלָדִים?

אַבָּא הוֹלֵךְ לָעֲבוֹדָה – וְלֹא לוֹקֵחַ אִתּוֹ אֶת הַיְּלָדִים. אֲבָל כְּשֶׁיֵּשׁ מִסִּבָּה, שִׂמְחָה וְשִׁירִים – לוֹקְחִים אֶת כֻּלָּם, גְּדוֹלִים וּקְטַנִּים!

עַל זֶה הָיָה הַוִּכּוּחַ בֵּין מֹשֶׁה לְפַרְעֹה. מָה אָמַר פַּרְעֹה, וּמָה עָנָה מֹשֶׁה?

לַעֲבוֹדָה הוֹלְכִים הַגְּבָרִים; לְחַג וְלִמְסִבָּה – כֻּלָּם.

רָמָה א׳ – חַג לְכֻלָּם

⚖️ פַּרְעֹה אוֹמֵר, מֹשֶׁה עוֹנֶה

פַּרְעֹה כְּבָר הִרְגִּישׁ שֶׁקָּשֶׁה לוֹ, וְאָמַר: ׳לְכוּ נָא הַגְּבָרִים וְעִבְדוּ אֶת ה׳, כִּי אֹתָהּ אַתֶּם מְבַקְשִׁים׳. עֲבוֹדָה – שֶׁיֵּלְכוּ הַגְּבָרִים בִּלְבַד.

מֹשֶׁה לֹא הִסְכִּים: ׳בִּנְעָרֵינוּ וּבִזְקֵנֵינוּ נֵלֵךְ, בְּבָנֵינוּ וּבִבְנוֹתֵנוּ… כִּי חַג ה׳ לָנוּ׳.

פַּרְעֹה: רַק הַגְּבָרִים. מֹשֶׁה: כֻּלָּם!

🎊 חַג, לֹא עֲבוֹדָה קָשָׁה

מֹשֶׁה מַסְבִּיר: אֶצְלֵנוּ עֲבוֹדַת ה׳ אֵינָהּ עֲבוֹדָה קָשָׁה. הִיא חַג וּמִסִּבָּה.

וּלְמִסִּבָּה הוֹלְכִים כֻּלָּם: גְּדוֹלִים וּקְטַנִּים, זְקֵנִים וּנְעָרִים.

זֶה הַסּוֹד שֶׁל ׳בִּנְעָרֵינוּ וּבִזְקֵנֵינוּ נֵלֵךְ׳: כְּשֶׁעֲבוֹדַת ה׳ הִיא חַג, אֶפְשָׁר לָקַחַת אֵלֶיהָ אֶת כֻּלָּם.

אֶצְלֵנוּ עֲבוֹדַת ה׳ הִיא חַג, וְלֶחָג לוֹקְחִים אֶת כֻּלָּם.

🏡 בַּיִת שֶׁכָּל מִצְוָה בּוֹ חֲגִיגָה

אֵיךְ נִרְאוֹת הַמִּצְווֹת אֶצְלֵנוּ בַּבַּיִת? כְּמוֹ ׳עֲבוֹדָה׳ שֶׁל הַמְּבֻגָּרִים, דָּבָר כָּבֵד שֶׁצָּרִיךְ לַעֲשׂוֹת? אוֹ כְּמוֹ חֲגִיגָה שֶׁכֻּלָּם שׁוּתָּפִים לָהּ?

כְּשֶׁהַמִּצְווֹת נַעֲשׂוֹת בְּשִׂמְחָה וּבְשִׁירָה, יֶלֶד לֹא צָרִיךְ שֶׁיְּשַׁכְנְעוּ אוֹתוֹ לְהִצְטָרֵף. אַף אֶחָד לֹא צָרִיךְ לְהַכְרִיחַ יֶלֶד לָלֶכֶת לְשִׂמְחַת תּוֹרָה – הוּא רָץ לְשָׁם מֵעַצְמוֹ.

בְּבַיִת שֶׁבּוֹ מִצְווֹת הֵן חֲגִיגָה, הַיְּלָדִים רוֹצִים לִהְיוֹת שָׁם.

🎤 מָה זוֹכְרִים בְּסוֹף הַשַּׁבָּת?

בְּשַׁבָּת אַחַת כֻּלָּם דִּבְּרוּ וְדִבְּרוּ, וּבְמוֹצָאֵי שַׁבָּת קָשֶׁה לִזְכֹּר מָה אָמְרוּ. אֲבָל דָּבָר אֶחָד כֻּלָּם זָכְרוּ: דְּבַר תּוֹרָה שֶׁאָמַר הָרַב בְּשִׁירָה, בַּנִּגּוּן ׳בִּנְעָרֵינוּ וּבִזְקֵנֵינוּ נֵלֵךְ׳.

הוּא חָזַר וְעָלָה מִסְּעוּדָה לִסְעוּדָה, עַד סְעוּדָה שְׁלִישִׁית. מִכָּל מָה שֶׁנֶּאֱמַר בַּשַּׁבָּת, נִשְׁאַר בַּזִּכָּרוֹן דַּוְקָא מָה שֶׁנֶּאֱמַר בְּשִׁירָה.

מָה שֶׁנֶּאֱמַר בְּשִׁירָה נִשְׁאַר בַּזִּכָּרוֹן.

🔁 לָמָּה שִׁיר לֹא נִמְאָס?

נְסוּ לַחֲזֹר עַל אוֹתוֹ מִשְׁפָּט פַּעַם, פַּעֲמַיִם, שָׁלוֹשׁ. אַחֲרֵי חָמֵשׁ פְּעָמִים זֶה כְּבָר נִמְאָס. אֲבָל שִׁיר אֶפְשָׁר לָשִׁיר עוֹד פַּעַם וְעוֹד פַּעַם!

לָמָּה? כִּי שִׁיר אֵינוֹ נוֹגֵעַ בַּשֵּׂכֶל. הוּא נוֹגֵעַ בָּרְגָשׁוֹת וּבַנְּשָׁמָה. וְשִׁיר שֶׁמַּכִּירִים – אוֹהֲבִים יוֹתֵר מִשִּׁיר חָדָשׁ.

הַשֵּׂכֶל מִתְעַיֵּף מֵחֲזָרָה, וְהַנְּשָׁמָה אוֹהֶבֶת לַחֲזֹר עַל שִׁיר.

💡 מִשְׁנָה בְּנִגּוּן

מָה שֶׁאָדָם לוֹמֵד בַּשֵּׂכֶל נִכְנָס לַמֹּחַ, וְלִפְעָמִים גַּם יוֹצֵא מִמֶּנּוּ. הַשִּׁירָה נוֹגַעַת בָּרֶגֶשׁ, וְאֵין גְּבוּל כַּמָּה עָמֹק הִיא יְכוֹלָה לְהִשָּׁאֵר.

לָכֵן בְּבָתֵּי סֵפֶר רַבִּים לוֹמְדִים מִשְׁנָיוֹת בְּנִגּוּן – אֲפִלּוּ אֶת כָּל פִּרְקֵי אָבוֹת – וְהַיְּלָדִים נֶהֱנִים וְשָׁרִים.

מִשְׁנָה שֶׁנִּלְמֶדֶת בְּנִגּוּן נִכְנֶסֶת אֶל הַלֵּב.

🌊 שִׁירַת הַיָּם בְּכָל יוֹם

שִׁירַת הַיָּם לֹא נֶאֶמְרָה פַּעַם אַחַת עַל שְׂפַת הַיָּם וְנִשְׁכְּחָה. הִיא נַעֲשֵׂית חֵלֶק מֵהַתְּפִלָּה: בְּכָל יוֹם, גַּם בְּיוֹם חֹל רָגִיל, אוֹמְרִים בְּשַׁחֲרִית ׳אָז יָשִׁיר מֹשֶׁה וּבְנֵי יִשְׂרָאֵל אֶת הַשִּׁירָה הַזֹּאת׳.

כִּי שִׁירָה בָּאָה מִן הָרֶגֶשׁ, וְהִיא מְעוֹרֶרֶת מֵחָדָשׁ אֶת הַהוֹדָיָה.

שִׁירָה שֶׁבָּאָה מִן הָרֶגֶשׁ אֵינָהּ מִתְיַשֶּׁנֶת.

מָה לוֹקְחִים מִרָמָה א׳?

כְּשֶׁעֲבוֹדַת ה׳ הִיא חַג, כֻּלָּם רוֹצִים לְהִצְטָרֵף – וַאֲנִי גַּם. כָּל שִׁיר וְכָל מִצְוָה בְּשִׂמְחָה מְקָרְבִים אוֹתִי אֶל ה׳ וּמְמַלְּאִים אֶת הַלֵּב בְּאַהֲבָה אֵלָיו.

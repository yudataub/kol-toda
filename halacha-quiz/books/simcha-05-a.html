חֲזַ״ל תִּאֲרוּ אָדָם שֶׁעוֹשֶׂה דָּבָר בְּלִי חַיּוּת כְּמוֹ תִּינוֹק הַבּוֹרֵחַ מִבֵּית הַסֵּפֶר. הוּא נִמְצָא בַּמָּקוֹם הַנָּכוֹן, אֲבָל כָּל גּוּפוֹ מְחַכֶּה לָצֵאת.

יֶלֶד שֶׁבּוֹרֵחַ: הַגּוּף כָּאן וְהַלֵּב בַּחוּץ.

אָדָם יָכוֹל לְהִתְפַּלֵּל כִּי הִגִּיעַ הַזְּמַן: לִקְרֹא אֶת הַמִּלִּים מַהֵר וְלָצֵאת. הוּא יָכוֹל לַעֲשׂוֹת מִצְוָה וּבַלֵּב לַחְשֹׁב: הַלְוַאי שֶׁזֶּה כְּבָר יִגָּמֵר.

זֶה לֹא אוֹמֵר שֶׁכָּל רֶגַע צָרִיךְ לִהְיוֹת מְרוֹמָם. יֵשׁ יָמִים שֶׁל עֲיֵפוּת וְטִרְדּוֹת, וְזֶה בְּסֵדֶר.

הַשְּׁאֵלָה הַחֲשׁוּבָה הִיא לְאָן הַלֵּב פּוֹנֶה בְּדֶרֶךְ כְּלָל.

לֹא כָּל רֶגַע מְרוֹמָם, אֲבָל חָשׁוּב לְאָן הַלֵּב פּוֹנֶה.

🛠️ הַנַּגָּר וְהָאָרוֹן

נַגָּר בָּנָה אָרוֹן. אִם הוּא עָבַד בְּשִׂמְחָה אוֹ בְּפָנִים כּוֹעֲסוֹת, הַלָּקוֹחַ יְשַׁלֵּם אֶת אוֹתוֹ מְחִיר. מַה שֶּׁחָשׁוּב הוּא שֶׁהָאָרוֹן יֵצֵא טוֹב.

בְּמִצְוָה זֶה שׁוֹנֶה. מִצְוָה אֵינָהּ שֵׁרוּת שֶׁמּוֹסְרִים לְלָקוֹחַ. הִיא מַעֲשֶׂה שֶׁאָדָם עוֹשֶׂה מוּל מִי שֶׁנָּתַן לוֹ אֶת הַחַיִּים.

לָכֵן גַּם הַלֵּב הוּא חֵלֶק מִן הַמַּעֲשֶׂה.

בְּמִצְוָה לֹא מוֹסְרִים מוּצָר, בּוֹנִים קֶשֶׁר.

🎁 מַתָּנָה אַחַת, שְׁנֵי יְלָדִים

שְׁנֵי יְלָדִים קִבְּלוּ אוֹתוֹ חֵפֶץ בְּדִיּוּק. הָרִאשׁוֹן לָקַח אוֹתוֹ בְּלִי לְהַסְתַּכֵּל עַל הַנּוֹתֵן וּבְלִי לוֹמַר תּוֹדָה. הַשֵּׁנִי רָאָה אֶת הַמַּתָּנָה, הִבְחִין בַּמַּחֲשָׁבָה שֶׁהוּשְׁקְעָה בָּהּ וְהִרְגִּישׁ אַהֲבָה.

הַחֵפֶץ זֵהֶה, אֲבָל הַקֶּשֶׁר שֶׁנּוֹצַר שׁוֹנֶה לְגַמְרֵי.

מִצְוָה הִיא מַתָּנָה כְּפוּלָה: הִיא נוֹתֶנֶת מַעֲשֶׂה טוֹב, וְנוֹתֶנֶת דֶּרֶךְ לִפְגֹּשׁ אֶת הַבּוֹרֵא. הַשִּׂמְחָה הִיא הַסִּימָן שֶׁהַמַּתָּנָה הִתְקַבְּלָה בַּלֵּב.

אוֹתָהּ מַתָּנָה, אֲבָל קֶשֶׁר אַחֵר, כִּי הַלֵּב רָאָה אֶת הַנּוֹתֵן.

🌿 הַנְּשָׁמָה נֶהֱנֵית

בַּשִּׁעוּר דִּבְּרוּ עַל שְׁלוֹשָׁה חֲלָקִים בָּאָדָם. הַגּוּף נֶהֱנֶה מִמְּנוּחָה וְאֹכֶל, הַשֵּׂכֶל נֶהֱנֶה מֵהֲבָנָה וּפִתְרוֹן, וְהַנְּשָׁמָה נֶהֱנֵית מִקֵּרוּב לַבּוֹרֵא וּמִמַּעֲשֶׂה טוֹב.

לִפְעָמִים הַמְּהִירוּת וְהַהֶרְגֵּל מְכַסִּים אֶת הַתַּעֲנוּג, אֲבָל הוּא עֲדַיִן שָׁם.

כְּשֶׁעוֹשִׂים מִצְוָה בְּיִשּׁוּב הַדַּעַת, אֶפְשָׁר לְגַלּוֹת בָּהּ טַעַם נָעִים.

הַגּוּף נֶהֱנֶה מִמְּנוּחָה, הַשֵּׂכֶל מֵהֲבָנָה, וְהַנְּשָׁמָה מִקֵּרוּב לַבּוֹרֵא.

🕯️ שִׂמְחָה שְׁקֵטָה

יֵשׁ יְלָדִים שֶׁחוֹשְׁבִים: ׳אֲנִי לֹא רוֹקֵד בַּתְּפִלָּה, אָז אֲנִי לֹא שָׂמֵחַ׳. אֲבָל שִׂמְחָה אֵינָהּ נִמְדֶּדֶת בְּרַעַשׁ.

אֶפְשָׁר לָשֶׁבֶת בְּשֶׁקֶט אַחֲרֵי בְּרָכָה וּלְהַרְגִּישׁ תּוֹדָה, אוֹ לִפְתֹּחַ סֵפֶר וְלִשְׂמֹחַ שֶׁאֶפְשָׁר לְהָבִין עוֹד פָּסוּק.

הַשִּׂמְחָה דּוֹמָה לְאוֹר קָטָן בַּחֶדֶר: הוּא לֹא חַיָּב לְסַנְוֵר כְּדֵי לְשַׁנּוֹת אֶת הָאֲוִירָה. מַסְפִּיק לָדַעַת שֶׁעָשִׂיתִי דָּבָר טוֹב.

שִׂמְחָה יְכוֹלָה לִהְיוֹת שְׁקֵטָה: לָדַעַת שֶׁעָשִׂיתִי דָּבָר טוֹב.

🐢 קָטָן וְלֹא מְמַהֵר

אֶפְשָׁר לִפְגֹּעַ בַּהֲנָאָה שֶׁבְּמִצְוָה בְּלִי לָשִׂים לֵב: לְמַהֵר כָּל כָּךְ שֶׁלֹּא מֵבִינִים אֶת הַמִּלִּים, אוֹ לְהַעֲמִיס עַל עַצְמִי דְּרִישׁוֹת גְּדוֹלוֹת מִדַּי.

הַפִּתְרוֹן הוּא לֹא לְוַתֵּר, אֶלָּא לְהַתְחִיל בְּצַעַד קָטָן: תְּפִלָּה אַחַת שֶׁקּוֹרְאִים בָּהּ לְאַט יוֹתֵר, אוֹ קֶטַע קָצָר שֶׁמְּבִינִים בֶּאֱמֶת.

כָּךְ הַמִּצְוָה חוֹזֶרֶת לִהְיוֹת מַתָּנָה וְלֹא מַשָּׂא.

צַעַד קָטָן וְלֹא מָהִיר: תְּפִלָּה אַחַת שֶׁקּוֹרְאִים לְאַט.

🧭 לְאָן הַלֵּב פּוֹנֶה?

הַתִּינוֹק הַבּוֹרֵחַ מְלַמֵּד אוֹתָנוּ שְׁאֵלָה אַחַת: לְאָן הַלֵּב פּוֹנֶה? לֹא צָרִיךְ לְהַעֲמִיד פָּנִים, וְלֹא צָרִיךְ לְהִתְרַגֵּשׁ בְּכָל רֶגַע.

צָרִיךְ לְחַפֵּשׂ בְּכָל מִצְוָה אֶת הַטּוֹב שֶׁבָּהּ, וּלְהַחֲזִיר לָהּ אֶת הַקֶּשֶׁר.

מִי שֶׁמְּחַפֵּשׂ, מוֹצֵא לְאַט לְאַט שִׂמְחָה אֲמִתִּית.

לֹא מַעֲמִידִים פָּנִים, מְחַפְּשִׂים אֶת הַטּוֹב בַּמִּצְוָה.

מָה לוֹקְחִים מִרָמָה א׳?

מִצְוָה אֵינָהּ מְטָלָה שֶׁרוֹצִים לְסַיֵּם. הִיא מַתָּנָה מֵה׳ וְדֶרֶךְ לִפְגֹּשׁ אוֹתוֹ. כְּשֶׁהַלֵּב פּוֹנֶה אֵלָיו, גַּם מִצְוָה קְטַנָּה מוֹסִיפָה שִׂמְחָה וְאַהֲבָה.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי הִרְגַּשְׁתִּי הַשָּׁבוּעַ שֶׁאֲנִי ׳בּוֹרֵחַ׳ מִמִּצְוָה וְרוֹצֶה שֶׁהִיא כְּבָר תִּסְתַּיֵּם?
  • אֵיזוֹ מִצְוָה שֶׁאֲנִי עוֹשֶׂה כָּל יוֹם נִדְמֵית לִי ׳סְתָם׳, וּמָה הַטּוֹב שֶׁבָּהּ?
  • הַיּוֹם אֹמַר בְּרָכָה אַחַת לְאַט, אֶזְכֹּר מִי נָתַן לִי אֶת הַטּוֹב, וְאֶרְשֹׁם בַּסּוֹף מָה הִרְגַּשְׁתִּי.

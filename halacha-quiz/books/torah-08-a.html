אֲנִי לְדוֹדִי וְדוֹדִי לִי

🌱 אֱלוּל בִּירוּשָׁלַיִם · רָמָה א׳

הַתְּרִיסִים נִסְגָּרִים – אֱלוּל בִּירוּשָׁלַיִם

רָמָה א׳ – חֲנוּת אַחַר חֲנוּת

לָמָּה חֲנֻיּוֹת נִסְגָּרוֹת בַּצָּהֳרַיִם בְּאֱלוּל, מָה מְיֻחָד בַּחֹדֶשׁ הַזֶּה, וְאֵיךְ קוֹבְעִים עִתִּים לַתּוֹרָה

📖 אַהֲבַת הַתּוֹרָה · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
🏪

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְהוֹסִיף בְּאֱלוּל בִּלְמוּד תּוֹרָה, כִּי כָּךְ אֲנִי מִתְקָרֵב לַה׳ 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מָסֹרֶת מִירוּשָׁלַיִם הָעַתִּיקָה

רָמָה א׳ – חֲנוּת אַחַר חֲנוּת

🏪 חֲנוּת אַחַר חֲנוּת

בִּירוּשָׁלַיִם הָעַתִּיקָה, בִּשְׁכוּנַת מֵאָה שְׁעָרִים, הָיָה מִנְהָג מְיֻחָד בְּחֹדֶשׁ אֱלוּל: כֻּלָּם רָצוּ לְהוֹסִיף בַּקֹּדֶשׁ.

כְּבָר בַּצָּהֳרַיִם חֲנוּת אַחַר חֲנוּת הָיְתָה נִסְגֶּרֶת, וּבַעֲלֵי הַחֲנֻיּוֹת הָלְכוּ לְבֵית הַמִּדְרָשׁ לִלְמֹד תּוֹרָה.

אֶפְשָׁר הָיָה לִשְׁמֹעַ אֶת הַתְּרִיסִים נִסְגָּרִים בָּזֶה אַחַר זֶה, וְלִרְאוֹת אֶת הָאֲנָשִׁים נִכְנָסִים לְבֵית הַמִּדְרָשׁ. זֶה הָיָה מַחֲזֶה מְיֻחָד מְאֹד.

בְּאֱלוּל סוֹגְרִים אֶת הַחֲנוּת וְהוֹלְכִים לִלְמֹד תּוֹרָה.

1
🌙

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַרְגִּישׁ שֶׁה׳ קָרוֹב אֵלַי בְּאֱלוּל 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, טוּר, אוֹרַח חַיִּים תקפ״א

רָמָה א׳ – חֲנוּת אַחַר חֲנוּת

🌙 חֹדֶשׁ שֶׁל קֵרוּב

לָמָּה דַּוְקָא בְּאֱלוּל? כִּי אֱלוּל הוּא חֹדֶשׁ שֶׁל קִרְבַּת ה׳.

בְּרֹאשׁ חֹדֶשׁ אֱלוּל עָלָה מֹשֶׁה רַבֵּנוּ לָהָר לְקַבֵּל אֶת הַלּוּחוֹת הָאַחֲרוֹנִים, וְהֶעֱבִירוּ שׁוֹפָר בַּמַּחֲנֶה. לָכֵן תּוֹקְעִים בַּשּׁוֹפָר בְּרֹאשׁ חֹדֶשׁ אֱלוּל.

כָּל חַג חוֹזֵר עַל מָה שֶׁהָיָה פַּעַם, וְהָאוֹר שֶׁהָיָה אָז חוֹזֵר וּמֵאִיר בְּכָל שָׁנָה.

אֱלוּל הוּא זְמַן שֶׁל קִרְבָה.

💝

💛 אַהֲבַת הַשֵּׁם: אֲנִי עוֹשֶׂה צַעַד אֶחָד אֶל ה׳, וְהוּא עוֹזֵר לִי 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׁיר הַשִּׁירִים ו׳ ג׳ · חִידָ״א

רָמָה א׳ – חֲנוּת אַחַר חֲנוּת

💝 אֲנִי לְדוֹדִי

בְּחֹדֶשׁ אֱלוּל יֵשׁ פָּסוּק מְיֻחָד: ׳אֲנִי לְדוֹדִי וְדוֹדִי לִי׳. רָאשֵׁי הַתֵּבוֹת שֶׁלּוֹ הֵם אֱלוּל.

׳אֲנִי לְדוֹדִי׳ – אֲנִי מִתְעוֹרֵר קֹדֶם וּמִתְקָרֵב אֶל ה׳. ׳וְדוֹדִי לִי׳ – וְאָז ה׳ עוֹזֵר לִי וּמְקָרֵב אוֹתִי.

חֲכָמִים אָמְרוּ: ׳הַבָּא לִטַּהֵר – מְסַיְּעִין אוֹתוֹ׳.

אֲנִי מַתְחִיל לְהִתְקָרֵב, וְה׳ עוֹזֵר.

3
🏷️

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְנַצֵּל אֶת הַמִּבְצָע שֶׁל אֱלוּל וּמִתְקָרֵב לַה׳ 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – חֲנוּת אַחַר חֲנוּת

🏷️ מִבְצָע בַּחֲנוּת

כָּל בַּעַל חֲנוּת יוֹדֵעַ מַה זֶּה מִבְצָע: קוֹנִים אוֹתוֹ דָּבָר בַּחֲצִי מְחִיר.

כָּךְ מְסַפְּרִים עַל אֱלוּל: ה׳ עוֹשֶׂה ׳מִבְצָע׳ עַל דְּבָרִים רוּחָנִיִּים. מַה שֶּׁבְּחֳדָשִׁים אֲחֵרִים קָשֶׁה לְהַשִּׂיג, בְּאֱלוּל אֶפְשָׁר לְהַשִּׂיג בַּחֲצִי הַזְּמַן.

כְּשֶׁיֵּשׁ מִבְצָע, חֲכָמִים קוֹנִים!

בְּאֱלוּל קַל יוֹתֵר לְהִתְקָרֵב וְלִלְמֹד.

🦁

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא מַחְמִיץ אֶת הַהִזְדַּמְּנוּת לְהִתְקָרֵב לַה׳ 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, עָמוֹס ג׳ ח׳ · שִׁעוּר ׳שְׁלֹשָׁה רְמָזִים לֶאֱלוּל׳

רָמָה א׳ – חֲנוּת אַחַר חֲנוּת

🦁 אַרְיֵה שָׁאָג

בַּפָּסוּק ׳אַרְיֵה שָׁאָג מִי לֹא יִירָא׳ יֵשׁ רֶמֶז: אַרְיֵ״ה – אֱלוּל, רֹאשׁ הַשָּׁנָה, יוֹם כִּפּוּר, הוֹשַׁעְנָא רַבָּה.

שָׁאֲלוּ: אֱלוּל הוּא חֹדֶשׁ שֶׁל רַחֲמִים וּסְלִיחוֹת, אָז עַל מָה הַשְּׁאָגָה?

הַתְּשׁוּבָה מִסִּפְרֵי הָרִאשׁוֹנִים: הַשְּׁאָגָה הִיא עַל מִי שֶׁמַּחְמִיץ אֶת אֱלוּל. זוֹ הִזְדַּמְּנוּת פַּעַם בַּשָּׁנָה לְהִתְקָרֵב אֶל ה׳ וְלִקְשֹׁר אֶת עַצְמֵנוּ אֵלָיו.

אֱלוּל הוּא הִזְדַּמְּנוּת – אֵין לְהַחְמִיץ אוֹתָהּ.

😢

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁאֱלוּל יִהְיֶה לִי חֹדֶשׁ מָתוֹק 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׁעוּר ׳שְׁלֹשָׁה רְמָזִים לֶאֱלוּל׳

רָמָה א׳ – חֲנוּת אַחַר חֲנוּת

😢 הַבֶּכִי עַל אֱלוּל

מְסַפְּרִים עַל אֶחָד מִגְּדוֹלֵי יְרוּשָׁלַיִם שֶׁבְּעֶרֶב רֹאשׁ הַשָּׁנָה הָלַךְ לְבֵית הַכְּנֶסֶת וְבָכָה.

שָׁאֲלוּ אוֹתוֹ: ׳לָמָּה אַתָּה בּוֹכֶה? עוֹד לֹא נִכְנַסְנוּ לְרֹאשׁ הַשָּׁנָה!׳

עָנָה: ׳אֲנִי לֹא בּוֹכֶה עַל רֹאשׁ הַשָּׁנָה. אֲנִי בּוֹכֶה עַל אֱלוּל. נִשְׁאֲרוּ רַק כַּמָּה דַּקּוֹת מֵהַחֹדֶשׁ הַמָּתוֹק הַזֶּה, וְהַיָּמִים שֶׁל קִרְבַת ה׳ הוֹלְכִים וְנִגְמָרִים.׳

אֱלוּל כָּל כָּךְ יָקָר, שֶׁבּוֹכִים כְּשֶׁהוּא נִגְמָר.

🌅

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִרְאוֹת בְּאֱלוּל הִזְדַּמְּנוּת, לֹא עֹל 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שִׁעוּר ׳שְׁלֹשָׁה רְמָזִים לֶאֱלוּל׳

🌅 סְלִיחוֹת בַּבֹּקֶר

זָקֵן צַדִּיק סִפֵּר: הוּא קָם בִּשְׁלוֹשׁ וָחֵצִי לִפְנוֹת בֹּקֶר וְהָלַךְ לְבֵית הַכְּנֶסֶת, שֶׁהָיָה מָלֵא מִפֹּה לְפֹה. אָמְרוּ סְלִיחוֹת בְּהִתְעוֹרְרוּת גְּדוֹלָה.

׳אֵיזֶה יֹפִי, אֵיזֶה מָתוֹק!׳ אָמַר. ׳אֲנָשִׁים הִרְגִּישׁוּ אֶת ה׳ מַמָּשׁ.׳

וְהוּא הוֹסִיף: בְּאֶמְצַע הַשָּׁנָה, גַּם אִם תִּתֵּן הַרְבֵּה כֶּסֶף, לֹא תַּצְלִיחַ לְהָקִים אֶת כֻּלָּם. רַק בְּאֱלוּל זֶה קוֹרֶה. הֵם לֹא רָאוּ בָּזֶה עֹל, אֶלָּא הִזְדַּמְּנוּת.

בְּאֱלוּל קָמִים לַסְּלִיחוֹת בְּשִׂמְחָה.

⏰ קוֹבְעִים עִתִּים

חֲכָמִים אָמְרוּ שֶׁכְּשֶׁאָדָם בָּא לַדִּין שׁוֹאֲלִים אוֹתוֹ: ׳קָבַעְתָּ עִתִּים לַתּוֹרָה?׳

שַׁמַּאי אָמַר: ׳עֲשֵׂה תוֹרָתְךָ קֶבַע׳ – שֶׁהַלִּמּוּד יִהְיֶה דָּבָר קָבוּעַ בַּיּוֹם, וְלֹא רַק לִפְעָמִים.

אֱלוּל הוּא זְמַן נֶהְדָּר לְהַתְחִיל: לִקְבֹּעַ שָׁעָה קְבוּעָה בַּיּוֹם לַתּוֹרָה.

תּוֹרָה בִּזְמַן קָבוּעַ בְּכָל יוֹם.

מָה לוֹקְחִים מִרָמָה א׳?

בְּאֱלוּל ה׳ קָרוֹב, וְהַדֶּלֶת פְּתוּחָה. מִי שֶׁעוֹשֶׂה צַעַד אֶחָד אֵלָיו מַרְגִּישׁ אֶת הַקִּרְבָה, וְאַהֲבָתוֹ לַה׳ גְּדֵלָה.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָה אֲנִי יָכוֹל לְהוֹסִיף הַשָּׁבוּעַ בְּלִמּוּד תּוֹרָה, כְּמוֹ בַּעֲלֵי הַחֲנֻיּוֹת שֶׁסָּגְרוּ אֶת הַחֲנוּת וְהָלְכוּ לִלְמֹד?
  • מָתַי בַּיּוֹם יֵשׁ לִי זְמַן שֶׁיָּכוֹל לִהְיוֹת קָבוּעַ לַתּוֹרָה, וְאֵיךְ אֶשְׁמֹר עָלָיו?
  • אֶקְבַּע שָׁעָה קְבוּעָה לְלִמּוּד תּוֹרָה, אֶכְתֹּב אוֹתָהּ עַל דַּף וְאֶתְלֶה אוֹתוֹ בַּחֶדֶר, וּבְסוֹף הַשָּׁבוּעַ אֶרְשֹׁם כַּמָּה פְּעָמִים הִצְלַחְתִּי.

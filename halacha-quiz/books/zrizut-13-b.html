בַּקֵּשׁ שָׁלוֹם וְרָדְפֵהוּ

🌱 לְפַיֵּס מִיָּד · רָמָה ב׳

הַמָּשִׁיחַ מַתִּיר אֶת הַתַּחְבּוֹשֶׁת – זְרִיזוּת לְפַיֵּס

רָמָה ב׳ – שָׁלוֹם בְּלִי דַּקּוֹת מְבֻזְבָּזוֹת

הָרַב חַיִּים זוֹנֶנְפֶלְד הוֹלֵךְ בַּלַּיְלָה לְפַיֵּס, וּמֵבִיא סִפּוּר מֵהַגְּמָרָא עַל הַמָּשִׁיחַ: כָּל דַּקָּה חֲשׁוּבָה, גַּם כְּשֶׁמְּפַיְּסִים.

📖 זְרִיזוּת · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
🕊️

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹדֵף שָׁלוֹם בְּאַהֲבָה, כִּי ה׳ אוֹהֵב שָׁלוֹם 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, תְּהִלִּים ל״ד ט״ו · זָרִיז וְנִשְׂכָּר, פֶּרֶק ה׳ (הָאִישׁ עַל הַחוֹמָה)

רָמָה ב׳ – שָׁלוֹם בְּלִי דַּקּוֹת מְבֻזְבָּזוֹת

🕊️ מִצְוַת הַשָּׁלוֹם

עַל הָרַב חַיִּים מְסַפְּרִים: ׳כְּשֶׁהָיָה שׁוֹמֵעַ עַל סִכְסוּךְ… הָיָה אוֹזֵר כְּגֶבֶר חֲלָצָיו, לֹא נָח וְלֹא שָׁקַט עַד שֶׁהֵבִיא אֶת שְׁנֵי הַצְּדָדִים לְפִיּוּס וְהִתְרַצּוּת׳.

הַכָּתוּב ׳בַּקֵּשׁ שָׁלוֹם וְרָדְפֵהוּ׳ מְלַמֵּד שֶׁשָּׁלוֹם אֵינוֹ רַק דָּבָר שֶׁמְּחַכִּים לוֹ: רוֹדְפִים אַחֲרָיו.

וְכָךְ גַּם זָרִיז: ׳יִהְיֶה זָרִיז לְבַקֵּשׁ שָׁלוֹם וְלִרְדֹּף אַחֲרָיו׳.

רוֹדְפִים שָׁלוֹם, לֹא מְחַכִּים לוֹ.

1
🌙

💛 אַהֲבַת הַשֵּׁם: אֲנִי דּוֹחֶה אֶת הַנּוֹחוּת שֶׁלִּי כְּדֵי לַעֲשׂוֹת שָׁלוֹם, וְה׳ רוֹאֶה וְשָׂמֵחַ 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, זָרִיז וְנִשְׂכָּר, פֶּרֶק ה׳ (הָאִישׁ עַל הַחוֹמָה)

רָמָה ב׳ – שָׁלוֹם בְּלִי דַּקּוֹת מְבֻזְבָּזוֹת

🌙 בֶּאֶמְצַע הַלַּיְלָה

לִפְעָמִים הָרַב הִתְעַטֵּף בַּגְּלִימָה אַחֲרֵי סְעוּדַת הָעֶרֶב וְהָלַךְ אֶל הַמְּסֻכְסָכִים. וְאִם הָיְתָה הַשָּׁעָה מְאֻחֶרֶת, לֹא הִסֵּס לְהָעִיר אֶת הָאִישׁ מִשְּׁנָתוֹ.

הָאִישׁ הִתְבַּלְבֵּל וְנִבְהַל, וְרָעַד מִקֹּר הַלַּיְלָה. כְּשֶׁרָאָה הָרַב שֶׁהָעַקְשָׁנוּת הִתְרַכְּכָה, הוּא פָּתַח בְּהַסְבָּרָה.

שִׂימוּ לֵב: הָרַב הָלַךְ בְּעַצְמוֹ, בִּשְׁעָה קָשָׁה, כִּי הַשָּׁלוֹם הָיָה חָשׁוּב לוֹ יוֹתֵר מִשֵּׁנָה.

הַשָּׁלוֹם חָשׁוּב מִנּוֹחוּת, וּמִשֵּׁנָה.

2
🏠

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְוַתֵּר קְצָת, כִּי רוֹצֶה שֶׁה׳ יִהְיֶה קָרוֹב אֵלֵינוּ 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, זָרִיז וְנִשְׂכָּר, פֶּרֶק ה׳ (הָאִישׁ עַל הַחוֹמָה)

רָמָה ב׳ – שָׁלוֹם בְּלִי דַּקּוֹת מְבֻזְבָּזוֹת

🏠 הַשְּׁכִינָה מִסְתַּלֶּקֶת

הָרַב הִסְבִּיר לָאִישׁ: ׳הַשְּׁכִינָה בַּגָּלוּת סוֹבֶלֶת יִסּוּרִים מִבָּנִים שֶׁגָּלוּ מֵעַל שֻׁלְחַן אֲבִיהֶם. וְהַמַּחֲזִיקִים בְּמַחֲלֹקֶת גּוֹרְמִים לַשְּׁכִינָה לְהִסְתַּלֵּק מִמְּעוֹנָם.׳

בְּלָשׁוֹן פְּשׁוּטָה: כְּשֶׁיֵּשׁ מְרִיבָה בַּבַּיִת, ה׳ כְּאִלּוּ פָּחוֹת קָרוֹב. כְּשֶׁמְּפַיְּסִים, הוּא חוֹזֵר.

וְהַשְּׁאֵלָה שֶׁל הָרַב: הַאִם אֵין אַתָּה רוֹאֶה חוֹבָה לְעַצְמְךָ לְוַתֵּר מַשֶּׁהוּ, כְּדֵי לְהַחֲזִיר לַבַּיִת אֶת הַשְּׁכִינָה?

שָׁלוֹם בַּבַּיִת מַחְזִיר אֶת קִרְבַת ה׳.

3
🩹

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוּכָן וְלֹא מְאַחֵר, כִּי אֲנִי מְצַפֶּה לַגְּאֻלָּה 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, סַנְהֶדְרִין צ״ח א׳

רָמָה ב׳ – שָׁלוֹם בְּלִי דַּקּוֹת מְבֻזְבָּזוֹת

🩹 מָשִׁיחַ וְתַחְבּוֹשֶׁת

הָרַב לָמַד מֵהַגְּמָרָא: ׳מָשִׁיחַ יוֹשֵׁב בְּפִתְחָא דְרוֹמִי, וּמַאי סִימָנֵיהּ? יָתֵיב בֵּין עֲנִיֵּי סוֹבְלֵי חֳלָאִים׳. כֻּלָּם שְׁרוּ וְאָסְרוּ בְּבַת אַחַת; אֲבָל הוּא שָׁרֵי חַד וְאָסַר חַד.

וְלָמָּה? ׳דִּלְמָא מִבָּעֵי לִי וְלֹא אֵיעַכֵּב׳ – אוּלַי יִצְטָרְכוּ אוֹתִי, וְלֹא אֲאַחֵר.

אִם הַמָּשִׁיחַ, שֶׁמְּחַכִּים לוֹ הַרְבֵּה שָׁנִים, אֵינוֹ רוֹצֶה לְאַחֵר אֲפִלּוּ רֶגַע – כַּמָּה יוֹתֵר אֲנַחְנוּ!

הַמָּשִׁיחַ מוּכָן לָבוֹא בְּכָל רֶגַע וְלֹא מִתְעַכֵּב.

4
⏱️

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא מְבַזְבֵּז זְמַן כְּשֶׁאֶפְשָׁר לְהַחְזִיר שָׁלוֹם 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, זָרִיז וְנִשְׂכָּר, פֶּרֶק ה׳ (הָאִישׁ עַל הַחוֹמָה)

רָמָה ב׳ – שָׁלוֹם בְּלִי דַּקּוֹת מְבֻזְבָּזוֹת

⏱️ גַּם דַּקָּה אַחַת

הַשְּׁאֵלָה הִיא: מַה הַהֶבְדֵּל בֵּין חַיִּים רְגִילִים לְבִין הַתָּרַת תַּחְבּוֹשֶׁת? דַּקָּה אוֹ שְׁתַּיִם. וְאָז אֶפְשָׁר לִטְעֹן: ׳כִּמְעַט אַלְפַּיִם שָׁנָה אָנוּ בַּגָּלוּת, וּמָה זֶה עוֹד כַּמָּה דַּקּוֹת׳?

הַהֶסְבֵּר שֶׁל הָרַב: כְּשֶׁמַּגִּיעַ הַזְּמַן, גַּם דַּקָּה אַחַת שֶׁל שְׁהִיָּה הִיא הַרְבֵּה.

וּבְעִנְיַן הַמַּחֲלֹקֶת: כָּל שֶׁתַּקְדִּימוּ לְהַשְׁלִים, כָּךְ תְּמַהֵר הַשְּׁכִינָה לַחֲזֹר לְבֵיתְכֶם.

כָּל שֶׁמְּקַדְּמִים אֶת הַפִּיּוּס, הַשָּׁלוֹם חוֹזֵר מַהֵר.

🤝

💛 אַהֲבַת הַשֵּׁם: אֲנִי הוֹלֵךְ רִאשׁוֹן לְפַיֵּס, וְה׳ מְבָרֵךְ אוֹתִי 💛

📖 זְרִיזוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – שָׁלוֹם בְּלִי דַּקּוֹת מְבֻזְבָּזוֹת

🤝 מִי הוֹלֵךְ רִאשׁוֹן?

בְּרִיב בֵּין חֲבֵרִים, כֻּלָּם יוֹדְעִים מִי צָרִיךְ לִבְקֹשׁ סְלִיחָה. לִפְעָמִים כָּל אֶחָד מְחַכֶּה שֶׁהַשֵּׁנִי יַתְחִיל.

הָרַב חַיִּים לֹא חִכָּה: הוּא הָיָה הַהוֹלֵךְ הָרִאשׁוֹן, גַּם כְּשֶׁלֹּא הוּא שֶׁרָב. זָרִיז בְּשָׁלוֹם מִתְחִיל, וְלֹא מוֹדֵד מִי הִתְחִיל בָּרִיב.

הַזָּרִיז בְּשָׁלוֹם הוּא הַהוֹלֵךְ הָרִאשׁוֹן.

רָמָה ב׳ – שָׁלוֹם בְּלִי דַּקּוֹת מְבֻזְבָּזוֹת

מָה לוֹקְחִים מִרָמָה ב׳?

הַשָּׁלוֹם הוּא הַמַּתָּנָה הַגְּדוֹלָה שֶׁה׳ אוֹהֵב. מִי שֶׁזָּרִיז לְפַיֵּס, מַחְזִיר אֶת הַשְּׁכִינָה לַבַּיִת וּמְקָרֵב אֶת לִבּוֹ לַה׳.

  • עִם מִי אֲנִי רָב עַכְשָׁו, וְאֵיךְ אֶפְנֶה אֵלָיו הַיּוֹם?
  • מָתַי בָּעָבָר חִכִּיתִי שֶׁמִּישֶׁהוּ אַחֵר יַתְחִיל לְפַיֵּס, וּמָה קָרָה?
  • הַיּוֹם אֲפַיֵּס מִיָּד חָבֵר אוֹ אָח, וְאַגִּיד לוֹ: ׳סְלַח לִי, בּוֹא נַעֲשֶׂה שָׁלוֹם׳.

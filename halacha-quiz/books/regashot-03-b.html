וְנָתַתָּ לְעַבְדְּךָ לֵב שֹׁמֵעַ

💗 לֵב שׁוֹמֵעַ · רָמָה ב׳

הַלֵּב רוֹאֶה, שׁוֹמֵעַ וּמְדַבֵּר

רָמָה ב׳ – מָה אוֹמְרִים חֲזַ״ל וְהַמְּקוֹרוֹת

אֵיךְ מְתָאֶרֶת הַתּוֹרָה אֶת הַלֵּב כְּמוֹ אָדָם: רוֹאֶה, שׁוֹמֵעַ, שָׂמֵחַ, חָרֵד וְחַם

📖 הָרְגָשׁוֹת שֶׁלִּי בַּתּוֹרָה · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד לְהַכִּיר אֶת הַלֵּב שֶׁלִּי 💛

📖 הָרְגָשׁוֹת שֶׁלִּי בַּתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, קֹהֶלֶת רַבָּה א׳ ט״ז

רָמָה ב׳ – מָה אוֹמְרִים חֲזַ״ל וְהַמְּקוֹרוֹת

📜 דִּבַּרְתִּי אֲנִי עִם לִבִּי

חֲזַ״ל בַּמִּדְרָשׁ מְנִים כַּמָּה פְּעֻלּוֹת שֶׁל הַלֵּב, וּלְכָל אַחַת פָּסוּק. הַלֵּב רוֹאֶה (׳וְלִבִּי רָאָה׳), שׁוֹמֵעַ (׳לֵב שֹׁמֵעַ׳), מְדַבֵּר (׳דִּבַּרְתִּי עִם לִבִּי׳), שָׂמֵחַ, צוֹעֵק (׳צָעַק לִבָּם אֶל ה׳׳), מִתְנַחֵם (׳דַּבְּרוּ עַל לֵב יְרוּשָׁלַיִם׳).

הַתּוֹרָה מְתָאֶרֶת אֶת הַלֵּב כְּמוֹ אָדָם חַי, עִם הַרְבֵּה רְגָשׁוֹת.

הַתּוֹרָה מְתָאֶרֶת אֶת הַלֵּב כְּמוֹ אָדָם: רוֹאֶה, שׁוֹמֵעַ, מְדַבֵּר, שָׂמֵחַ וְצוֹעֵק.

1

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְחַזֵּק אֶת לִבִּי כְּדֵי לִרְאוֹת אֶת הַטּוֹב 💛

📖 הָרְגָשׁוֹת שֶׁלִּי בַּתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, קֹהֶלֶת רַבָּה א׳ ט״ז

רָמָה ב׳ – מָה אוֹמְרִים חֲזַ״ל וְהַמְּקוֹרוֹת

🔎 הַלֵּב רוֹאֶה מָה שֶׁהוּא רוֹצֶה

לְפִי הֶסְבֵּר שֶׁל הַמִּדְרָשׁ, אָדָם רוֹאֶה אֶת מָה שֶׁלִּבּוֹ חָפֵץ. לְפִי הָרֶגֶשׁ שֶׁמִּתְעוֹרֵר בַּלֵּב כָּךְ נִרְאֶה לָנוּ הָעוֹלָם.

אִם אֲנִי מַרְגִּישׁ פַּחַד, אֲפָרֵשׁ אֶת הַמְּצִיאוּת כְּאִלּוּ יֵשׁ מִמָּה לִפְחֹד. אִם אֲנִי מַרְגִּישׁ אֹמֶץ, אֲפָרֵשׁ שֶׁאֵין מִמָּה לַחְשֹׁשׁ, וְאַרְגִּישׁ אֶת נוֹכְחוּת ה׳ יוֹתֵר.

לָכֵן כְּדַאי לְחַזֵּק אֶת הַלֵּב.

הָרֶגֶשׁ שֶׁבַּלֵּב קוֹבֵעַ אֵיךְ אֲנִי רוֹאֶה אֶת הַמְּצִיאוּת.

2

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַקְשִׁיב לַחֲבֵרִים בְּלֵב שׁוֹמֵעַ 💛

📖 הָרְגָשׁוֹת שֶׁלִּי בַּתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מְלָכִים א׳ ג׳ ט׳

רָמָה ב׳ – מָה אוֹמְרִים חֲזַ״ל וְהַמְּקוֹרוֹת

👂 שְׁלוֹשָׁה סוּגֵי הַקְשָׁבָה

״לֵב שׁוֹמֵעַ״ הוּא הַקְשָׁבָה פְּעִילָה. יֵשׁ שָׁלוֹשׁ רָמוֹת.

רָמָה רִאשׁוֹנָה: שׁוֹמְעִים, אֲבָל חוֹשְׁבִים עַל עַצְמִי – מָה זֶה אוֹמֵר לִי. רָמָה שְׁנִיָּה: הַקְשָׁבָה מְלֵאָה לָאָדָם שֶׁמּוּלִי, בְּלִי שִׁפּוּט, וּמְנַסִּים לִרְאוֹת בְּעֵינָיו. רָמָה שְׁלִישִׁית: מַקְשִׁיבִים לַכֹּל: לַמֶּסֶר, לִשְׂפַת הַגּוּף, לְמַצַּב הָרוּחַ, וְלַמָּה שֶׁלֹּא נֶאֱמַר.

לֵב שׁוֹמֵעַ מַקְשִׁיב בֶּאֱמֶת: לַפְּרָטִים, לָרֶגֶשׁ, וְגַם לְמָה שֶׁלֹּא נֶאֱמַר.

3

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא מִתְבַּיֵּשׁ בָּרְגָשׁוֹת שֶׁלִּי 💛

📖 הָרְגָשׁוֹת שֶׁלִּי בַּתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁמוּאֵל א׳ ד׳ י״ג · שְׁמוּאֵל א׳ י״ז ל״ב

רָמָה ב׳ – מָה אוֹמְרִים חֲזַ״ל וְהַמְּקוֹרוֹת

🌟 גַּם הַגְּדוֹלִים הִרְגִּישׁוּ

הַתָּנָ״ךְ לֹא מְסַתֵּר אֶת הָרְגָשׁוֹת שֶׁל גְּדוֹלִים. עֵלִי הַכֹּהֵן הָיָה כֹּהֵן גָּדוֹל, וְ׳לִבּוֹ הָיָה חָרֵד עַל אֲרוֹן הָאֱלֹהִים׳.

וְדָוִד, כְּשֶׁהָיָה נַעַר, נָתַן כֹּחַ לַאֲחֵרִים: ׳אַל יִפֹּל לֵב אָדָם עָלָיו׳. הַלֵּב יָכוֹל ׳לִפֹּל׳ בְּיֵאוּשׁ, וְיָכוֹל ׳לַעֲמֹד׳ בְּהִתְחַזְּקוּת וְתִקְוָה (׳הֲיַעֲמֹד לִבֵּךְ׳).

גַּם הַגְּדוֹלִים הִרְגִּישׁוּ חֲרָדָה, וְגַם הֵם חִזְּקוּ אֶת הַלֵּב.

4

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לֵב חַם שֶׁעוֹזֵר 💛

📖 הָרְגָשׁוֹת שֶׁלִּי בַּתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, יְהוֹשֻׁעַ ז׳ ה׳ · תְּהִלִּים ל״ט ד׳

רָמָה ב׳ – מָה אוֹמְרִים חֲזַ״ל וְהַמְּקוֹרוֹת

🔥 חַם וְנָמֵס

בַּתָּנָ״ךְ ׳וַיִּמַּס לְבַב הָעָם׳ הוּא פַּחַד, אֲבָל הַיּוֹם ׳הַלֵּב נָמֵס׳ מְבַטֵּא אַהֲבָה, חֶמְלָה וְרַחֲמִים עַזִּים, לְמָשָׁל כְּשֶׁשּׁוֹמְעִים סִפּוּר עַל יֶלֶד קָטָן שֶׁצָּרִיךְ עֶזְרָה.

׳חַם לִבִּי בְּקִרְבִּי׳ הוּא תְּשׁוּקָה עֲצוּמָה לְדָבָר טוֹב. ׳לֵב חַם׳ בִּלְשׁוֹנֵנוּ הוּא אָדָם שֶׁמַּרְגִּישׁ אֶמְפַּתְיָה לַאֲחֵרִים וְרוֹצֶה לַעֲזֹר לָהֶם.

הַלֵּב יָכוֹל לָחֹם וְלִהְיוֹת נָמֵס מֵאַהֲבָה וְרַחֲמִים.

מָה לוֹקְחִים מִרָמָה ב׳?

גַּם הַגְּדוֹלִים הִרְגִּישׁוּ, וְה׳ אֵינוֹ מַסְתִּיר אֶת זֶה בַּתּוֹרָה. כְּשֶׁאֲנִי מַרְגִּישׁ שֶׁה׳ מְכַבֵּד אֶת הָרְגָשׁוֹת שֶׁלִּי, אַהֲבָתִי אֵלָיו גְּדֵלָה.

😴 כְּשֶׁאֵין חֵשֶׁק

יֶלֶד יוֹדֵעַ שֶׁכְּדַאי לְהִתְפַּלֵּל בְּכַוָּנָה, אֲבָל אֵין לוֹ חֵשֶׁק. הוּא לֹא מַרְגִּישׁ שֶׁהוּא רוֹצֶה.

מָה עוֹשִׂים כְּשֶׁאֵין רָצוֹן? חֲזַ״ל מְלַמְּדִים: מְבַקְּשִׁים מֵה׳ שֶׁיִּתֵּן לָנוּ רָצוֹן.

גַּם עַל הָרָצוֹן עַצְמוֹ מוּתָּר לְהִתְפַּלֵּל.

בִּתְהִלִּים כָּתוּב: ׳רְצוֹן יְרֵאָיו יַעֲשֶׂה וְאֶת שַׁוְעָתָם יִשְׁמַע וְיוֹשִׁיעֵם׳.

ה׳ עוֹשֶׂה רָצוֹן לְמִי שֶׁיָּרֵא מִמֶּנּוּ, וְהוּא עוֹשֶׂה זֹאת עַל יְדֵי תְּפִלָּה: ׳וְאֶת שַׁוְעָתָם יִשְׁמַע׳.

וְזֶה מַתְאִים גַּם לָרָצוֹן עַצְמוֹ: מְבַקְּשִׁים, וְה׳ שׁוֹמֵעַ.

ה׳ עוֹשֶׂה רָצוֹן לְמִי שֶׁמִּתְפַּלֵּל.

🪜 שָׁלוֹשׁ מַדְרֵגוֹת

הָעֵצָה פְּשׁוּטָה וּמַעֲשִׂית, בִּשְׁלוֹשָׁה שְׁלָבִים.

א. בּוֹדְקִים: הַאִם אֲנִי בֶּאֱמֶת רוֹצֶה? ב. מְבַקְּשִׁים: ׳ה׳, תֵּן לִי רָצוֹן׳. ג. עוֹשִׂים וּמִתְפַּלְּלִים שֶׁנַּצְלִיחַ.

כָּךְ הַרְבֵּה יוֹתֵר קַל לְהַתְחִיל.

בּוֹדְקִים אִם רוֹצִים, מְבַקְּשִׁים רָצוֹן, וְעוֹשִׂים.

🤫 מָקוֹם שָׁקֵט

הָרַב מַמְלִיץ: בּוֹחֲרִים מָקוֹם שָׁקֵט שֶׁאֵין בּוֹ אֲנָשִׁים, וּמְדַבְּרִים עִם ה׳.

בַּשֶּׁקֶט קַל יוֹתֵר לִשְׁמֹעַ אֶת הַלֵּב.

הָעִקָּר הוּא שֶׁזֶּה בֶּאֱמֶת מֵהַלֵּב.

מְדַבְּרִים עִם ה׳ בְּמָקוֹם שָׁקֵט.

🗣️ מָה אוֹמְרִים?

אוֹמְרִים בַּלָּשׁוֹן שֶׁלָּנוּ, לְמָשָׁל: ׳רִבּוֹנוֹ שֶׁל עוֹלָם, אַתָּה יוֹדֵעַ שֶׁאֲנִי רוֹצֶה לִהְיוֹת קָרוֹב אֵלֶיךָ. אֲבָל הָרָצוֹן שֶׁלִּי חָלָשׁ. אָנָּא, עֲזֹר לִי שֶׁאֶרְצֶה בֶּאֱמֶת׳.

אֵין צֹרֶךְ בְּמִלִּים מְיֻחָדוֹת. כָּל אֶחָד מְדַבֵּר כְּמוֹ שֶׁהוּא.

אוֹמְרִים לַה׳ ׳הָרָצוֹן שֶׁלִּי חָלָשׁ, עֲזֹר לִי׳.

🔁 יוֹם אַחַר יוֹם

אֵין צֹרֶךְ לְבַקֵּשׁ רַק פַּעַם אַחַת. מְבַקְּשִׁים יוֹם אַחַר יוֹם, כָּל אֶחָד לְפִי כֹּחוֹ.

יֵשׁ בָּזֶה שְׁתֵּי מַעֲלוֹת: הַדִּבּוּר בְּעַצְמוֹ מְעוֹרֵר אֶת הָרָצוֹן, וְה׳ שׁוֹמֵעַ אֶת הַתְּפִלָּה וְעוֹזֵר.

מִי שֶׁמַּתְמִיד, בְּוַדַּאי ה׳ יַעֲזֹר לוֹ לִרְצוֹת בֶּאֱמֶת.

מְבַקְּשִׁים יוֹם אַחַר יוֹם, וְה׳ עוֹזֵר.

💬 מִשְׁפָּט שֶׁנָּגַע בִּי

לִקְרֹא כַּמָּה דָּבָר חָשׁוּב עֲדַיִן לֹא מַסְפִּיק כְּדֵי שֶׁיִּהְיֶה לָנוּ רָצוֹן.

לוֹקְחִים מִשְׁפָּט אֶחָד שֶׁנָּגַע בַּלֵּב, עוֹמְדִים בְּמָקוֹם שָׁקֵט, וְחוֹזְרִים עָלָיו בְּמִלִּים שֶׁלָּנוּ עוֹד וָעוֹד.

כַּמָּה יָמִים כָּךְ, וּמְבַקְּשִׁים מֵה׳ שֶׁיִּתֵּן לָנוּ אֶת הַדָּבָר הַזֶּה.

לוֹקְחִים מִשְׁפָּט שֶׁנָּגַע בַּלֵּב וְחוֹזְרִים עָלָיו.

📖 שׁוֹמְעִים צַדִּיק

עוֹד דֶּרֶךְ: לִשְׁמֹעַ בְּהַתְמָדָה סִפּוּרִים וּדְבָרִים שֶׁל צַדִּיק, שֶׁמְּדַבֵּר מִתּוֹךְ הַנְּשָׁמָה.

חָשׁוּב שֶׁבַּסּוֹף הַהִתְעוֹרְרוּת תָּבוֹא גַּם מִבִּפְנִים, כִּי הִתְעוֹרְרוּת שֶׁבָּאָה רַק מִבַּחוּץ אֵין לָהּ קִיּוּם.

לָכֵן כָּל אֶחָד בּוֹחֵר אֶת הַדֶּרֶךְ שֶׁמַּתְאִימָה לוֹ.

שׁוֹמְעִים צַדִּיק, וְהָרָצוֹן צָרִיךְ לְהִתְעוֹרֵר גַּם מִבִּפְנִים.

מָה לוֹקְחִים מִרָמָה א׳?

גַּם כְּשֶׁאֵין לִי חֵשֶׁק, אֲנִי יָכוֹל לִפְנוֹת אֶל ה׳ וּלְבַקֵּשׁ. הוּא שׁוֹמֵעַ אֶת הַשַּׁוְעָה, וְכָךְ אֲנִי מַרְגִּישׁ כַּמָּה הוּא קָרוֹב וְאוֹהֵב אוֹתִי.

יֵשׁ הַרְבֵּה דְּרָכִים

רָמָה ב׳ – הַצָּבָא וְהַמַּבָּט

לָמָּה כָּל אֶחָד בָּטוּחַ שֶׁדַּרְכּוֹ נְכוֹנָה, מָה לָמַדְנוּ מֵהַחֵילוֹת בַּצָּבָא, וְאֵיךְ מִסְתַּכְּלִים בְּעַיִן טוֹבָה עַל חָבֵר שֶׁעוֹשֶׂה אַחֶרֶת

📖 דָּן לְכַף זְכוּת · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן
📖

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְנַסֶּה לְהָבִין אֶת הַמַּבָּט שֶׁל חֲבֵרִי, וְה׳ בּוֹחֵן אֶת הַלֵּב שֶׁל כֻּלָּנוּ 💛

📖 דָּן לְכַף זְכוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשְׁלֵי כ״א ב׳

רָמָה ב׳ – הַצָּבָא וְהַמַּבָּט

📖 כׇּל דֶּרֶךְ אִישׁ יָשָׁר בְּעֵינָיו

בְּמִשְׁלֵי כ״א ב׳: כׇּל דֶּרֶךְ אִישׁ יָשָׁר בְּעֵינָיו וְתֹכֵן לִבּוֹת ה׳.

הַכַּוָּנָה: אָדָם תָּמִיד חוֹשֵׁב שֶׁדַּרְכּוֹ צוֹדֶקֶת, בֶּאֱמֶת וּמִבִּפְנִים. וְה׳ הוּא שֶׁבּוֹחֵן אֶת הַלְּבָבוֹת.

לָכֵן מְלַמְּדִים בַּשִּׁעוּר: נִשְׁתַּדֵּל לִרְאוֹת אֶת הַדְּבָרִים לְפִי מַבָּטוֹ שֶׁל הָאַחֵר. מִי שֶׁמַּבִּיט כָּךְ מֵבִין אֶת הָאַחֵר לִפְנֵי שֶׁהוּא שׁוֹפֵט אוֹתוֹ.

אָדָם בָּטוּחַ שֶׁדַּרְכּוֹ יְשָׁרָה, לָכֵן מְנַסִּים לִרְאוֹת לְפִי מַבָּטוֹ.

1
🪖

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד לְהַעֲרִיךְ גַּם דְּרָכִים שֶׁשּׁוֹנוֹת מִשֶּׁלִּי 💛

📖 דָּן לְכַף זְכוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – הַצָּבָא וְהַמַּבָּט

🪖 הַחֵילוֹת בַּצָּבָא

הַמָּשָׁל בַּשִּׁעוּר: יֵשׁ הַרְבֵּה דְּרָכִים לַעֲבֹד אֶת ה׳, כְּמוֹ שֶׁיֵּשׁ בַּצָּבָא הַרְבֵּה חֵילוֹת.

הַטַּיָּס, הַמַּלָּח וְהַחַיָּל בַּשֶּׁטַח לֹא עוֹשִׂים אוֹתוֹ דָּבָר, וְאַף אֶחָד מֵהֶם אֵינוֹ יָכוֹל לַעֲשׂוֹת אֶת הָעֲבוֹדָה שֶׁל כֻּלָּם. הַצָּבָא שׁוֹמֵר עַל הַמְּדִינָה בִּזְכוּת שֶׁיֵּשׁ בּוֹ הַכֹּל.

לָכֵן אֶפְשָׁר לַחְשֹׁב שֶׁהַדֶּרֶךְ שֶׁלִּי נְכוֹנָה, וְלִזְכֹּר שֶׁיֵּשׁ אֲחֵרִים שֶׁחוֹשְׁבִים אַחֶרֶת.

הַרְבֵּה חֵילוֹת בַּצָּבָא, וְהַרְבֵּה דְּרָכִים לַעֲבֹד אֶת ה׳.

2
🧠

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹמֵר לְחָבֵר: ׳בָּזֶה אַתָּה טוֹב מִמֶּנִּי׳, וְשָׂמֵחַ בִּשְׁבִילוֹ 💛

📖 דָּן לְכַף זְכוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, הָרַב דֶּסְלֶר

רָמָה ב׳ – הַצָּבָא וְהַמַּבָּט

🧠 כָּל אֶחָד מְצֻיָּן בְּמַשֶּׁהוּ

הָרַב דֶּסְלֶר מוֹסִיף: כָּל אָדָם חוֹשֵׁב שֶׁהוּא מְצֻיָּן מִמְּךָ בְּאֵיזֶה דָּבָר, וּמִן הַסְּתָם הוּא צוֹדֵק.

הָעֵצָה: לְהַסְכִּים אִתּוֹ. כֵּן, בַּדָּבָר הַזֶּה הוּא טוֹב מִמֶּנִּי. כָּךְ הוּא מַרְגִּישׁ שֶׁרוֹאִים אוֹתוֹ, וְאוֹהֵב אוֹתִי.

מִי שֶׁמַּסְכִּים שֶׁלַּחָבֵר יֵשׁ כֹּחַ מְיֻחָד, לֹא צָרִיךְ לְהִתְחָרוֹת בּוֹ.

מַסְכִּימִים שֶׁהֶחָבֵר מְצֻיָּן בְּמַשֶּׁהוּ, וְכָךְ אוֹהֲבִים.

🧭

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא פּוֹסֵק מַהֵר, וְאִם אֲנִי לֹא בָּטוּחַ אֲנִי שׁוֹאֵל 💛

📖 דָּן לְכַף זְכוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – הַצָּבָא וְהַמַּבָּט

🧭 מָה פֵּרוּשׁ ׳יִתָּכֵן׳?

בַּשִּׁעוּר: ׳יֵשׁ כָּאֵלֶּה שֶׁחוֹשְׁבִים אַחֶרֶת, וְיִתָּכֵן שֶׁגַּם דַּרְכָּם נְכוֹנָה׳. שִׂימוּ לֵב: לֹא ׳בְּוַדַּאי׳, אֶלָּא ׳יִתָּכֵן׳.

זֶה אוֹמֵר שֶׁלֹּא מְמַהֲרִים לִפְסֹק. כְּשֶׁחָבֵר עוֹשֶׂה אַחֶרֶת, חוֹשְׁבִים ׳אוּלַי׳ וְשׁוֹאֲלִים אוֹתוֹ בְּנִימוּס.

כְּשֶׁהַשְּׁאֵלָה הִיא ׳אֵיךְ צָרִיךְ לְקַיֵּם מִצְוָה׳, שׁוֹאֲלִים מוֹרֶה אוֹ רַב, וְלֹא מַחְלִיטִים לְבַד.

׳יִתָּכֵן׳ אוֹמֵר: לֹא מְמַהֲרִים לִפְסֹק, וּבְסָפֵק שׁוֹאֲלִים רַב.

4
⚖️

💛 אַהֲבַת הַשֵּׁם: אֲנִי נֶאֱמָן לְדַרְכִּי וְשָׂמֵחַ בְּדֶרֶךְ שֶׁל חֲבֵרִי 💛

📖 דָּן לְכַף זְכוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – הַצָּבָא וְהַמַּבָּט

⚖️ גַּם שֶׁלִּי וְגַם שֶׁלּוֹ

בַּשִּׁעוּר אוֹמְרִים לַתַּלְמִיד: נָכוֹן שֶׁאַתָּה חוֹשֵׁב שֶׁכָּךְ הַדֶּרֶךְ לַעֲבֹד אֶת ה׳. אֵין צֹרֶךְ לְוַתֵּר עָלֶיהָ.

וְאָז מוֹסִיפִים: יֵשׁ כָּאֵלֶּה שֶׁחוֹשְׁבִים אַחֶרֶת. אֵין סְתִירָה בֵּין הַשְּׁנַיִם, כְּמוֹ בַּצָּבָא: כָּל חַיָּל נֶאֱמָן לְחֵיל שֶׁלּוֹ, וְיוֹדֵעַ שֶׁהַחֵילוֹת הָאֲחֵרִים נְחוּצִים.

מִי שֶׁחוֹשֵׁב כָּךְ דָּן אֶת חֲבֵרוֹ לְכַף זְכוּת.

אֶפְשָׁר לְהַאֲמִין בַּדֶּרֶךְ שֶׁלִּי וְלִכְבֹּד אֶת הַדֶּרֶךְ שֶׁל חֲבֵרִי.

🏫

💛 אַהֲבַת הַשֵּׁם: אֲנִי מַחֲלִיף ׳הוּא מוּזָר׳ בְּ׳אוּלַי זוֹ הַדֶּרֶךְ שֶׁלּוֹ׳ 💛

📖 דָּן לְכַף זְכוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה ב׳ – הַצָּבָא וְהַמַּבָּט

🏫 שְׁלוֹשָׁה מַצָּבִים בַּכִּתָּה

חָבֵר לוֹמֵד בְּקוֹל רָם, חָבֵר אַחֵר מְסַכֵּם בִּצְבָעִים, וַאֲחֵרוֹן מִתְפַּלֵּל לְאַט וּבְנִגּוּן.

לִפְנֵי שֶׁאוֹמְרִים ׳הוּא מוּזָר׳, חוֹשְׁבִים: אוּלַי זוֹ הַדֶּרֶךְ שֶׁלּוֹ.

אִם הָעִנְיָן מַפְרִיעַ לִי מַמָּשׁ, אֶפְשָׁר לְבַקֵּשׁ בְּנִימוּס. וְאִם הוּא נוֹגֵעַ לַהֲלָכָה, שׁוֹאֲלִים מוֹרֶה.

לִפְנֵי ׳הוּא מוּזָר׳ חוֹשְׁבִים: אוּלַי זוֹ הַדֶּרֶךְ שֶׁלּוֹ.

מָה לוֹקְחִים מִרָמָה ב׳?

הַצָּבָא שֶׁל עַם יִשְׂרָאֵל גָּדוֹל, וְיֵשׁ בּוֹ הַרְבֵּה חֵילוֹת. מִי שֶׁשָּׂמֵחַ בַּדֶּרֶךְ שֶׁל חֲבֵרוֹ וְרוֹאֶה בּוֹ כֹּחַ מְיֻחָד, מַכִּיר בְּכָךְ שֶׁה׳ בָּרָא אֶת כֻּלָּנוּ שׁוֹנִים, וְאוֹהֵב אוֹתָנוּ כֻּלָּנוּ.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה אֲנִי עוֹשֶׂה בְּדֶרֶךְ שֶׁלִּי, וּמָה חָבֵר שֶׁלִּי עוֹשֶׂה אַחֶרֶת, וּמָה אוּלַי טוֹב בְּדַרְכּוֹ?
  • מָתַי אָמַרְתִּי עַל חָבֵר ׳הוּא טוֹעֶה׳ וְלֹא שָׁאַלְתִּי אוֹתוֹ לָמָּה?
  • אֶכְתֹּב מָה הַכֹּחַ הַמְיֻחָד שֶׁלִּי וְשֶׁל שְׁלוֹשָׁה חֲבֵרִים בַּכִּתָּה, וְאוֹמֵר לָהֶם – וְאֶרְשֹׁם מָה הֵם אָמְרוּ וּמָה הִרְגַּשְׁתִּי.

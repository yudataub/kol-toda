רָמָה א׳ – רֵעֲךָ

לָמָּה כָּתוּב ׳רֵעֲךָ׳ וְלֹא ׳אָחִיךָ׳, אֵיךְ מַעֲמִיקִים אֶת הָאַהֲבָה לַחָבֵר הַקָּרוֹב, וְאֵיךְ בּוֹנִים לֵב שֶׁיּוֹדֵעַ לֶאֱהֹב

📖 אַהֲבַת יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, וַיִּקְרָא י״ט, י״ז–י״ח

רָמָה א׳ – רֵעֲךָ

🔎 שָׁלוֹשׁ מִלִּים בַּפָּסוּק

בַּתּוֹרָה, בְּפָרָשַׁת קְדוֹשִׁים, כָּתוּב: ׳לֹא תִשְׂנָא אֶת אָחִיךָ בִּלְבָבֶךָ׳, ׳לֹא תִקֹּם וְלֹא תִטֹּר אֶת בְּנֵי עַמֶּךָ׳, וְאַחֲרֵיהֶם: ׳וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ׳.

שָׁלוֹשׁ מִלִּים שׁוֹנוֹת: אָחִיךָ, עַמֶּךָ, רֵעֲךָ. לָמָּה בַּפָּסוּק שֶׁל הָאַהֲבָה כָּתוּב דַּוְקָא ׳רֵעֲךָ׳, חָבֵר קָרוֹב, וְלֹא ׳אָחִיךָ׳?

זוֹ שְׁאֵלָה שֶׁשָּׁוֶה לַחְשֹׁב עָלֶיהָ.

בַּפָּסוּק שֶׁל הָאַהֲבָה כָּתוּב ׳רֵעֲךָ׳ – חָבֵר שֶׁכְּבָר קָרוֹב אֵלַי.

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא שׂוֹנֵא אַף אֶחָד, וְלַחֲבֵרִים הַקְּרוֹבִים אֲנִי נוֹתֵן אַהֲבָה מִיֻּחֶדֶת 💛

📖 אַהֲבַת יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, וַיִּקְרָא י״ט, י״ז–י״ח · בְּאַהֲבָה וּבְאַחְדוּת

רָמָה א׳ – רֵעֲךָ

🧱 מָה מְבַקְּשִׁים מִכָּל אֶחָד?

מִכָּל יְהוּדִי, גַּם מִי שֶׁרָחוֹק מִמֶּנִּי, הַתּוֹרָה מְבַקֶּשֶׁת שֶׁלֹּא אֶשְׂנָא, לֹא אֶנְקֹם וְלֹא אֶטֹּר טִינָה. זֶה הַמִּינִימוּם.

אֲבָל לְמִי שֶׁכְּבָר ׳רֵעַ׳ – יְדִיד קָרוֹב – הַתּוֹרָה אוֹמֶרֶת: תּוֹסִיף אַהֲבָה עַד שֶׁתַּגִּיעַ לְ׳כָּמוֹךָ׳.

כְּלוֹמַר, הֶחָבֵר הַקָּרוֹב הוּא הַמָּקוֹם לְהַעֲמִיק בּוֹ אֶת הָאַהֲבָה.

לְכָל יְהוּדִי – לֹא שׂוֹנְאִים. לַחָבֵר הַקָּרוֹב – מוֹסִיפִים אַהֲבָה.

🌳

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לֶאֱהֹב עוֹד יוֹתֵר אֶת מִי שֶׁכְּבָר אָהוּב עָלַי 💛

📖 אַהֲבַת יִשְׂרָאֵל – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, בְּאַהֲבָה וּבְאַחְדוּת · מַעְגְּלֵי הָאַהֲבָה

רָמָה א׳ – רֵעֲךָ

🌳 לְהַעֲמִיק, לֹא רַק לְהַרְחִיב

הַרְבֵּה יְלָדִים חוֹשְׁבִים: ׳אֲנִי צָרִיךְ לֶאֱהֹב עוֹד וְעוֹד יְלָדִים׳, כְּאִלּוּ מְחַפְּשִׂים ׳לְקוֹחוֹת חֲדָשִׁים׳ לָאַהֲבָה.

אֲבָל הַתּוֹרָה מְלַמֶּדֶת: הָעֲבוֹדָה הַגְּדוֹלָה הִיא לְהַעֲמִיק אֶת מָה שֶׁכְּבָר קַיָּם – לֶאֱהֹב עוֹד יוֹתֵר אֶת הַחֲבֵרִים וְהַקְּרוֹבִים שֶׁאֲנִי כְּבָר אוֹהֵב, עַד שֶׁהָאַהֲבָה הִיא ׳כָּמוֹךָ׳ מַמָּשׁ.

כְּמוֹ אִילָן: כְּשֶׁהַשָּׁרָשִׁים עֲמֻקִּים, הוּא יָכוֹל לִגְדֹּל גָּבוֹהַּ.

אֲנִי לֹא מְחַפֵּשׂ עוֹד חֲבֵרִים לֶאֱהֹב – אֲנִי מַעֲמִיק אֶת הָאַהֲבָה שֶׁיֵּשׁ.

🤝 חָבֵר טוֹב

בְּפִרְקֵי אָבוֹת, רַבָּן יוֹחָנָן בֶּן זַכַּאי שָׁאַל אֶת תַּלְמִידָיו: ׳צְאוּ וּרְאוּ אֵיזוֹהִי דֶּרֶךְ יְשָׁרָה שֶׁיִּדְבַּק בָּהּ הָאָדָם׳.

רַבִּי אֱלִיעֶזֶר אָמַר: ׳עַיִן טוֹבָה׳. רַבִּי יְהוֹשֻׁעַ אָמַר: ׳חָבֵר טוֹב׳. וְרַבִּי אֶלְעָזָר אָמַר: ׳לֵב טוֹב׳.

חָבֵר טוֹב הוּא מִי שֶׁאֲנִי קָרוֹב אֵלָיו וְאוֹהֵב אוֹתוֹ. כָּךְ לוֹמְדִים לֶאֱהֹב בֶּאֱמֶת.

רַבִּי יְהוֹשֻׁעַ בָּחַר: ׳חָבֵר טוֹב׳.

❤️ לֵב שֶׁיּוֹדֵעַ לֶאֱהֹב

מָה הַמַּטָּרָה שֶׁל מִצְוַת ׳וְאָהַבְתָּ לְרֵעֲךָ׳? לֹא רַק שֶׁיִּהְיוּ הַרְבֵּה אֲנָשִׁים שֶׁאוֹהֲבִים זֶה אֶת זֶה, אֶלָּא גַּם: שֶׁיִּהְיֶה לִי לֵב שֶׁיּוֹדֵעַ לֶאֱהֹב.

זֶה מַה שֶּׁרַבִּי אֶלְעָזָר אָמַר בְּפִרְקֵי אָבוֹת: ׳לֵב טוֹב׳. וְרַבָּן יוֹחָנָן אָמַר שֶׁבִּדְבָרָיו כְּלוּלִים כָּל הַדְּבָרִים.

כְּמוֹ שֶׁמְּאַמְּנִים שְׁרִירִים, כָּךְ אֶפְשָׁר לְאַמֵּן לֵב לֶאֱהֹב.

הַמַּטָּרָה: לִבְנוֹת לֵב שֶׁיּוֹדֵעַ לֶאֱהֹב.

רָמָה א׳ – רֵעֲךָ

🎁 שְׁתֵּי מָנוֹת לְאִישׁ אֶחָד

בְּפוּרִים יֵשׁ שְׁתֵּי מִצְווֹת שֶׁל נְתִינָה. בְּמַתָּנוֹת לָאֶבְיוֹנִים נוֹתְנִים לְכָל עָנִי מַתָּנָה אַחַת, וְנוֹתְנִים לִשְׁנֵי עֲנִיִּים. וּבְמִשְׁלוֹחַ מָנוֹת שׁוֹלְחִים שְׁתֵּי מָנוֹת לְאָדָם אֶחָד.

לָמָּה כָּךְ? מִדַּת הַחֶסֶד גְּדֵלָה כְּשֶׁנּוֹתְנִים עוֹד וְעוֹד פְּעָמִים. אֲבָל אַהֲבָה נִקְנֵית כְּשֶׁמַּשְׁקִיעִים הַרְבֵּה בְּאָדָם אֶחָד.

לָכֵן בְּמִשְׁלוֹחַ מָנוֹת מַשְׁקִיעִים בְּחָבֵר אֶחָד.

אַהֲבָה גְּדֵלָה כְּשֶׁמַּשְׁקִיעִים הַרְבֵּה בְּאָדָם אֶחָד.

🌊 מִשָּׁם הָאַהֲבָה מִתְרַחֶבֶת

שָׁלוֹם וְאַהֲבָה בַּמַּעְגָּל הַפְּנִימִי – בַּחֲבֵרִים וּבַקְּרוֹבִים – מוֹלִידִים שָׁלוֹם גַּם כְּלַפֵּי חוּץ.

מִי שֶׁלֹּא בָּנָה אַהֲבָה יַצִּיבָה בַּמַּעְגָּל הַקָּרוֹב אֵלָיו, קָשֶׁה לוֹ מְאוֹד לְהַאֲמִין שֶׁיִּבְנֶה אוֹתָהּ בַּמַּעְגָּלִים הָרְחוֹקִים.

כְּמוֹ אֶבֶן שֶׁנּוֹפֶלֶת לַמַּיִם: הַגַּלִּים מַתְחִילִים קָרוֹב, וְאַחַר כָּךְ מִתְרַחֲבִים.

אַהֲבָה בַּמַּעְגָּל הַקָּרוֹב מִתְרַחֶבֶת לְכָל הַמַּעְגָּלִים.

🔑 מֵאַהֲבַת יִשְׂרָאֵל לְאַהֲבַת ה׳

כַּמָּה שִׂמְחָה יֵשׁ לְאַבָּא שֶׁרוֹאֶה שֶׁאוֹהֲבִים אֶת הַיְלָדִים שֶׁלּוֹ! כָּךְ גַּם ה׳ שָׂמֵחַ כְּשֶׁאָנוּ אוֹהֲבִים זֶה אֶת זֶה, וְהוּא אוֹהֵב אֶת כָּל בָּנָיו.

הִלֵּל לִמֵּד שֶׁאַהֲבַת יִשְׂרָאֵל מְבִיאָה עַד אַהֲבַת ה׳. הַמִּצְווֹת שֶׁבֵּין אָדָם לַחֲבֵרוֹ נוֹעֲדוּ לִבְנוֹת לֵב אוֹהֵב, וְלֵב כָּזֶה יָכוֹל לֶאֱהֹב גַּם אֶת ה׳.

אַהֲבָה לַחֲבֵרִים בּוֹנָה לֵב שֶׁיָּכוֹל לֶאֱהֹב גַּם אֶת ה׳.

מָה לוֹקְחִים מִרָמָה א׳?

  • בְּאֵיזֶה מַצָּב הַשָּׁבוּעַ הָיָה אֶפְשָׁר לְהוֹסִיף לְחָבֵר קָרוֹב עוֹד קְצָת אַהֲבָה – מִלָּה חַמָּה, זְמַן אוֹ תְּשׂוּמֶת לֵב?
  • הַיּוֹם אֶכְתֹּב דָּבָר אֶחָד שֶׁאֶעֱשֶׂה לְחָבֵר קָרוֹב כְּדֵי שֶׁהָאַהֲבָה תִּהְיֶה ׳כָּמוֹךָ׳ – וְאֶעֱשֶׂה אוֹתוֹ, וְאֶרְשֹׁם מָה הִרְגַּשְׁתִּי.

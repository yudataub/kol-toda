רָמָה א׳ – שָׂדֶה בִּשְׁבִיל תּוֹרָה

🛤️ שְׁנֵי חֲכָמִים בַּדֶּרֶךְ

דִּמְיְנוּ יֶלֶד שֶׁחָסַךְ כֶּסֶף הַרְבֵּה זְמַן בִּשְׁבִיל מַשֶּׁהוּ שֶׁהוּא רוֹצֶה מְאֹד. מָה הוּא קוֹנֶה בּוֹ בֶּאֱמֶת?

בַּסִּפּוּר שֶׁלָּנוּ, רַבִּי יוֹחָנָן הָלַךְ בַּדֶּרֶךְ מִטְּבֶרְיָה לְצִפּוֹרִי. לְיָדוֹ הָלַךְ חָכָם אַחֵר, רַבִּי חִיָּיא בַּר אַבָּא, שֶׁנִּשְׁעַן עָלָיו.

בַּדֶּרֶךְ הֵם עָבְרוּ לְיַד שָׂדוֹת, וְרַבִּי יוֹחָנָן הִתְחִיל לְסַפֵּר.

הַסִּפּוּר מַתְחִיל בִּשְׁנֵי חֲכָמִים שֶׁהוֹלְכִים בַּדֶּרֶךְ.

💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה יְקָרָה לִי יוֹתֵר מִשָּׂדוֹת 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, וַיִּקְרָא רַבָּה ל׳

רָמָה א׳ – שָׂדֶה בִּשְׁבִיל תּוֹרָה

🌾 הַשָּׂדֶה, הַכֶּרֶם וְהַזַּיִת

הֵם הִגִּיעוּ לְשָׂדֶה. אָמַר רַבִּי יוֹחָנָן: ׳הַשָּׂדֶה הַזֹּאת הָיְתָה שֶׁלִּי, וּמָכַרְתִּי אוֹתָהּ כְּדֵי לִזְכּוֹת בַּתּוֹרָה׳.

אַחַר כָּךְ הִגִּיעוּ לְכֶרֶם, וְהוּא אָמַר אוֹתָהּ מִלָּה: ׳הַכֶּרֶם הָיָה שֶׁלִּי, וּמָכַרְתִּי אוֹתוֹ כְּדֵי לִזְכּוֹת בַּתּוֹרָה׳.

וְכָךְ גַּם בְּשָׂדֵה הַזֵּיתִים.

רַבִּי יוֹחָנָן מָכַר שָׂדֶה, כֶּרֶם וְזֵיתִים כְּדֵי לִזְכּוֹת בַּתּוֹרָה.

😢

💛 אַהֲבַת הַשֵּׁם: אֲנִי דּוֹאֵג לַאֲחֵרִים כְּמוֹ רַבִּי חִיָּיא 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, וַיִּקְרָא רַבָּה ל׳

רָמָה א׳ – שָׂדֶה בִּשְׁבִיל תּוֹרָה

😢 לָמָּה בָּכָה רַבִּי חִיָּיא?

רַבִּי חִיָּיא הִתְחִיל לִבְכּוֹת. רַבִּי יוֹחָנָן שָׁאַל: ׳מַה לְךָ בּוֹכֶה?׳

רַבִּי חִיָּיא עָנָה: ׳בּוֹכֶה אֲנִי עַל שֶׁלֹּא הִשְׁאַרְתָּ לְעַצְמְךָ כְּלוּם לַזִּקְנָה׳.

הוּא דָּאַג לְרַבִּי יוֹחָנָן, כִּי מָכַר אֶת כָּל מַה שֶׁהָיָה לוֹ.

רַבִּי חִיָּיא בָּכָה מִדְּאָגָה, כִּי לֹא נִשְׁאַר כְּלוּם לַזִּקְנָה.

3
⚖️

💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה הִיא מַתָּנָה גְּדוֹלָה מֵה׳ 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, וַיִּקְרָא רַבָּה ל׳

רָמָה א׳ – שָׂדֶה בִּשְׁבִיל תּוֹרָה

⚖️ קַל בְּעֵינֶיךָ?

רַבִּי יוֹחָנָן הֵשִׁיב: ׳קַלָּה הִיא בְּעֵינֶיךָ מָה שֶׁעָשִׂיתִי? מָכַרְתִּי דָּבָר שֶׁנִּבְרָא לְשִׁשָּׁה יָמִים, וְקָנִיתִי דָּבָר שֶׁנִּתַּן בְּאַרְבָּעִים יוֹם׳.

הַשָּׂדוֹת הֵם חֵלֶק מִן הָעוֹלָם, שֶׁנִּבְרָא בְּשִׁשָּׁה יָמִים. וְהַתּוֹרָה נִתְּנָה בְּאַרְבָּעִים יוֹם.

הַתּוֹרָה נִתְּנָה בְּאַרְבָּעִים יוֹם, וְהִיא יְקָרָה יוֹתֵר מִשָּׂדוֹת.

4
⛰️

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ שֶׁקִּבַּלְנוּ תּוֹרָה 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁמוֹת ל״ד כ״ח · דְּבָרִים ט׳ ט׳

רָמָה א׳ – שָׂדֶה בִּשְׁבִיל תּוֹרָה

⛰️ אַרְבָּעִים יוֹם עַל הָהָר

מֵאַיִן לָמַד רַבִּי יוֹחָנָן עַל אַרְבָּעִים יוֹם? מִן הַתּוֹרָה: מֹשֶׁה עָלָה אֶל הָהָר וְהָיָה שָׁם אַרְבָּעִים יוֹם וְאַרְבָּעִים לַיְלָה, וְקִבֵּל אֶת הַלּוּחוֹת.

בְּכָל אוֹתָם יָמִים מֹשֶׁה לָמַד אֶת הַתּוֹרָה.

זֶה הַזְּמַן שֶׁבּוֹ הִתְקַבְּלָה הַתּוֹרָה לְעַם יִשְׂרָאֵל.

מֹשֶׁה הָיָה עַל הָהָר אַרְבָּעִים יוֹם וְאַרְבָּעִים לַיְלָה.

5
📖

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹהֵב לִהְיוֹת עָסוּק בַּתּוֹרָה 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, יְרוּשַׁלְמִי בְּרָכוֹת ה׳ א׳

רָמָה א׳ – שָׂדֶה בִּשְׁבִיל תּוֹרָה

📖 רַבִּי יוֹחָנָן בְּצִפּוֹרִי

סִפּוּר נוֹסָף עַל רַבִּי יוֹחָנָן. הוּא יָשַׁב בְּצִפּוֹרִי וְקָרָא וְלָמַד לִפְנֵי בֵּית הַכְּנֶסֶת.

עָבַר שָׁם שַׁלִּיט, וְרַבִּי יוֹחָנָן לֹא קָם מִמְּקוֹמוֹ, כִּי הָיָה עָסוּק בַּתּוֹרָה.

בַּסִּפּוּר נֶאֱמַר: ׳הַנִּיחוּ לוֹ, הוּא עָסוּק בַּתּוֹרָה׳.

רַבִּי יוֹחָנָן הָיָה כֹּה עָסוּק בְּלִמּוּד, שֶׁאֲפִלּוּ שַׁלִּיט לֹא הִפְרִיעַ לוֹ.

🎒

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוּכָן לָתֵת קְצָת מִזְּמַנִּי בִּשְׁבִיל הַתּוֹרָה 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – שָׂדֶה בִּשְׁבִיל תּוֹרָה

🎒 מָה אֲנִי נוֹתֵן בִּשְׁבִיל תּוֹרָה?

לֹא כָּל אֶחָד מוֹכֵר שָׂדֶה. אֲבָל כָּל אֶחָד יָכוֹל לָתֵת מַשֶּׁהוּ שֶׁיָּקָר לוֹ: קְצָת זְמַן מִשְׂחָק, כְּדֵי לִלְמֹד.

מִי שֶׁאוֹהֵב תּוֹרָה, מֵבִין שֶׁהִיא שָׁוָה יוֹתֵר מִכְּסָפִים וּמִשָּׂדוֹת.

וְהַשִּׂמְחָה שֶׁבַּלִּמּוּד מְחַזֶּרֶת לָנוּ הַרְבֵּה.

נוֹתְנִים מַשֶּׁהוּ קָטָן כְּדֵי לִלְמֹד, וְשָׂמֵחִים.

💛

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד תּוֹרָה כְּדֵי לְהִתְקָרֵב לַה׳ 💛

📖 אַהֲבַת הַתּוֹרָה – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

רָמָה א׳ – שָׂדֶה בִּשְׁבִיל תּוֹרָה

💛 הַתּוֹרָה – מַתָּנָה מֵה׳

רַבִּי יוֹחָנָן הֵבִין שֶׁהַתּוֹרָה אֵינָהּ סְתָם סֵפֶר. הִיא דִּבְרֵי ה׳ אֶל עַם יִשְׂרָאֵל.

מִי שֶׁלּוֹמֵד תּוֹרָה, מַרְגִּישׁ שֶׁה׳ מְדַבֵּר אֵלָיו.

לָכֵן הָיָה מוּכָן לְוַתֵּר עַל שָׂדוֹת, כִּי הוּא אָהַב אֶת הַתּוֹרָה וְאֶת נוֹתֵן הַתּוֹרָה.

הַתּוֹרָה הִיא מַתָּנָה מֵה׳, וּמִי שֶׁאוֹהֵב אוֹתָהּ אוֹהֵב אֶת ה׳.

רָמָה א׳ – שָׂדֶה בִּשְׁבִיל תּוֹרָה

מָה לוֹקְחִים מִרָמָה א׳?

רַבִּי יוֹחָנָן הֵבִין שֶׁהַתּוֹרָה הִיא מַתָּנָה גְּדוֹלָה שֶׁנָּתַן לָנוּ ה׳, וְשָׁוָה יוֹתֵר מִכָּל שָׂדֶה. כָּל פַּעַם שֶׁאֲנִי לוֹמֵד, אֲנִי מִתְקָרֵב לַה׳ וְאוֹהֵב אוֹתוֹ יוֹתֵר.

🤔 חוֹשְׁבִים עַל זֶה

  • מָה הַדָּבָר הֶחָשׁוּב לִי בֶּאֱמֶת, וְעַל מָה אֲנִי מוּכָן לְוַתֵּר כְּדֵי לִלְמֹד תּוֹרָה?
  • מָתַי הַשָּׁבוּעַ הִרְגַּשְׁתִּי שֶׁלִּלְמֹד זֶה שָׁוֶה יוֹתֵר מִמִּשְׂחָק, וְאֵיךְ הָיָה לִי אַחַר כָּךְ?
  • אֶכְתֹּב עַל פֶּתֶק דָּבָר קָטָן אֶחָד שֶׁאֲנִי שָׂם בַּצַּד כְּדֵי לִלְמֹד תּוֹרָה הַשָּׁבוּעַ, וְאֶעֱשֶׂה אוֹתוֹ בְּשִׂמְחָה.

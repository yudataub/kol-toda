רָמָה ב׳ – הֶסְבֵּרִים טוֹבִים

הַסִּפּוּר הַמָּלֵא מִן הַגְּמָרָא, חֲמִשָּׁה הֶסְבֵּרִים טוֹבִים, וְאֵיךְ דָּנִים חָבֵר לִזְכוּת

📖 עַיִן טוֹבָה וְאוֹפְּטִימִיּוּת · מִשִּׁעוּרֵי הָרַב יְהוּדָה טָאוּב

🎯 לַחִידוֹן

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְנַסּוֹת לְהָבִין אֶת הַחָבֵר שֶׁלִּי לִפְנֵי שֶׁאֲנִי שׁוֹפֵט 💛

📖 עַיִן טוֹבָה וְאוֹפְּטִימִיּוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, אָבוֹת ב׳ ד׳

רָמָה ב׳ – הֶסְבֵּרִים טוֹבִים

🏛️ ׳עַד שֶׁתַּגִּיעַ לִמְקוֹמוֹ׳

הִלֵּל הַזָּקֵן לִמֵּד בְּפִרְקֵי אָבוֹת: ׳אַל תָּדִין אֶת חֲבֵרְךָ עַד שֶׁתַּגִּיעַ לִמְקוֹמוֹ׳. כְּלוֹמַר, עַד שֶׁתָּבִין מָה עוֹבֵר עָלָיו וּבְאֵיזֶה מַצָּב הוּא נִמְצָא.

הַשָּׂכִיר לֹא הִגִּיעַ לִמְקוֹמוֹ שֶׁל בַּעַל הַבַּיִת וְלֹא יָדַע עַל הַנֶּדֶר, וְלָכֵן בָּחַר לֹא לָדוּן אוֹתוֹ לְרַע. כָּךְ גַּם חָבֵר שֶׁלֹּא עָנָה לִי שָׁלוֹם: אוּלַי הוּא דּוֹאֵג מִמַּשֶּׁהוּ, וַאֲנִי לֹא יוֹדֵעַ.

לִפְנֵי שֶׁדָּנִים חָבֵר, מְנַסִּים לְהָבִין אֶת הַמַּצָּב שֶׁלּוֹ.

1

💛 אַהֲבַת הַשֵּׁם: אֲנִי יָכוֹל לְהַתְחִיל לְהִתְאַמֵּן בְּעַיִן טוֹבָה עוֹד הַיּוֹם 💛

📖 עַיִן טוֹבָה וְאוֹפְּטִימִיּוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שְׁאִילְתּוֹת, פָּרָשַׁת שְׁמוֹת · הַחָתָם סוֹפֵר

רָמָה ב׳ – הֶסְבֵּרִים טוֹבִים

📖 מִי הָיָה הַשָּׂכִיר?

בַּשְּׁאִילְתּוֹת (פָּרָשַׁת שְׁמוֹת) מוֹפִיעַ אוֹתוֹ סִפּוּר עִם שֵׁמוֹת: הַשָּׂכִיר הוּא רַבִּי עֲקִיבָא, שֶׁעֲדַיִן לֹא לָמַד תּוֹרָה, וּבַעַל הַבַּיִת הוּא רַבִּי אֱלִיעֶזֶר בֶּן הוּרְקָנוֹס.

הַחָתָם סוֹפֵר מֵבִיא שֶׁגַּם כְּשֶׁרַבִּי עֲקִיבָא עוֹד לֹא לָמַד, הוּא כְּבָר הָיָה בַּעַל מִדּוֹת טוֹבוֹת: הוּא דָּן אֶת בַּעַל הַבַּיִת לִזְכוּת, אֲפִלּוּ בְּיוֹם שֶׁאֵין לוֹ שָׂכָר.

עַיִן טוֹבָה הִיא מִדָּה שֶׁאֶפְשָׁר לְהַתְחִיל לְהִתְאַמֵּן בָּהּ בְּכָל גִּיל.

לְפִי הַשְּׁאִילְתּוֹת, הַשָּׂכִיר הָיָה רַבִּי עֲקִיבָא, כְּבָר אָז בַּעַל עַיִן טוֹבָה.

2

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹחֵר הֶסְבֵּר טוֹב, כְּמוֹ הַשָּׂכִיר 💛

📖 עַיִן טוֹבָה וְאוֹפְּטִימִיּוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שַׁבָּת קכ״ז ב׳

רָמָה ב׳ – הֶסְבֵּרִים טוֹבִים

🔍 חֲמִשָּׁה הֶסְבֵּרִים טוֹבִים

לְכָל ׳אֵין לִי׳ הָיָה לַשָּׂכִיר הֶסְבֵּר טוֹב, וְכָל הֶסְבֵּר נִשְׁמַע הַגּוּן:

אֵין מָעוֹת – אוּלַי נִקְנְתָה סְחוֹרָה בְּזוֹל. אֵין בְּהֵמָה – אוּלַי הִיא מֻשְׂכֶּרֶת לַאֲחֵרִים. אֵין קַרְקַע – אוּלַי הִיא מֻחְכֶּרֶת. אֵין פֵּרוֹת – אוּלַי לֹא הִפְרִישׁוּ מֵהֶם מַעֲשֵׂר. אֵין כָּרִים וּכְסָתוֹת – אוּלַי הוּקְדְּשׁוּ לַשָּׁמַיִם.

הַשָּׂכִיר לֹא הִתְעַלֵּם מֵהָעֻבְדָּה שֶׁלֹּא קִבֵּל שָׂכָר. הוּא רַק בָּחַר אֵיזֶה הֶסְבֵּר לָתֵת לָהּ.

לְכָל מַצָּב לֹא נָעִים יֵשׁ כַּמָּה הֶסְבֵּרִים, וְאֶפְשָׁר לִבְחֹר אֶת הַטּוֹב.

3

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי דָּן לִזְכוּת, ה׳ דָּן גַּם אוֹתִי לִזְכוּת 💛

📖 עַיִן טוֹבָה וְאוֹפְּטִימִיּוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, שַׁבָּת קכ״ז ב׳

רָמָה ב׳ – הֶסְבֵּרִים טוֹבִים

🔄 הַתּוֹעֶלֶת הִיא גַּם לִי

חֲכָמִים לִמְּדוּ שֶׁהַתּוֹעֶלֶת שֶׁבְּלִמּוּד זְכוּת אֵינָהּ רַק לַחָבֵר: מִי שֶׁדָּן אֶת הַזּוּלַת לִזְכוּת, גַּם הוּא נִדּוֹן לִזְכוּת בְּבֵית דִּין שֶׁל מַעְלָה, וְכַף הַמֹּאזְנַיִם שֶׁל מַעֲשָׂיו נוֹטָה אֶל הַזְּכֻיּוֹת.

זֶה מַה שֶּׁאָמַר בַּעַל הַבַּיִת בַּסּוֹף: ׳כְּשֵׁם שֶׁדַּנְתַּנִי לִזְכוּת – הַמָּקוֹם יָדִין אוֹתְךָ לִזְכוּת׳.

מִי שֶׁדָּן לִזְכוּת, גַּם הוּא נִדּוֹן לִזְכוּת.

4

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִתְאַמֵּן לָעֲצֹר לִפְנֵי שֶׁאֲנִי כּוֹעֵס 💛

📖 עַיִן טוֹבָה וְאוֹפְּטִימִיּוּת – מִשִּׁעוּרָיו שֶׁל הָרַב יְהוּדָה טָאוּב, מִשִּׁעוּרֵי עַיִן טוֹבָה

רָמָה ב׳ – הֶסְבֵּרִים טוֹבִים

🚦 שְׁלוֹשָׁה צְעָדִים

חָבֵר הִבְטִיחַ לָבוֹא לַמִּשְׂחָק וְלֹא בָּא. הַמְּלַמֵּד לֹא עָנָה לִי שָׁלוֹם. מַה עוֹשִׂים?

צַעַד רִאשׁוֹן: עוֹצְרִים לִפְנֵי שֶׁכּוֹעֲסִים. צַעַד שֵׁנִי: חוֹשְׁבִים לְפָחוֹת שְׁנֵי הֶסְבֵּרִים טוֹבִים: אוּלַי לֹא שָׁמַע, אוּלַי הָיָה טָרוּד. צַעַד שְׁלִישִׁי: אִם אֶפְשָׁר, שׁוֹאֲלִים בְּנַחַת, וּמַקְשִׁיבִים לַהֶסְבֵּר.

חֲכָמִים פָּתְחוּ וְסִיְּמוּ אֶת הַסִּפּוּר בְּאוֹתוֹ רַעְיוֹן, כִּי הוּא הָעִקָּר.

עוֹצְרִים, חוֹשְׁבִים הֶסְבֵּר טוֹב, וְשׁוֹאֲלִים בְּנַחַת.

5
💡

רָמָה ב׳ – הֶסְבֵּרִים טוֹבִים

מָה לוֹקְחִים מִרָמָה ב׳?

בַּעַל הַבַּיִת אָמַר: ׳כְּשֵׁם שֶׁדַּנְתַּנִי לִזְכוּת – הַמָּקוֹם יָדִין אוֹתְךָ לִזְכוּת׳. זֶה מְגַלֶּה שֶׁה׳ אוֹהֵב אוֹתָנוּ כְּמוֹ אַבָּא: כְּשֶׁאֲנַחְנוּ טוֹבִים לַאֲחֵרִים, הוּא מְחַפֵּשׂ אֶת הַטּוֹב בָּנוּ, וְאוֹהֵב אוֹתָנוּ גַּם כְּשֶׁאֲנַחְנוּ טוֹעִים.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָתַי הִרְגַּשְׁתִּי שֶׁחָבֵר פָּגַע בִּי, וּמָה הֶסְבֵּר טוֹב אֶפְשָׁר לַחְשֹׁב שֶׁיָּכוֹל הָיָה לִהְיוֹת הַסִּבָּה?
  • מָתַי מִישֶׁהוּ לֹא הֵבִין אוֹתִי נָכוֹן, וְאֵיךְ הִרְגַּשְׁתִּי כְּשֶׁהִסְבַּרְתִּי לוֹ?
  • הַיּוֹם, כְּשֶׁמַּשֶּׁהוּ יְאַכְזֵב אוֹתִי, אֲנִי עוֹצֵר וְחוֹשֵׁב שְׁנֵי הֶסְבֵּרִים טוֹבִים לָמָּה זֶה קָרָה – וְרַק אָז מֵגִיב.

🌍 חִידוֹנֵי חֻמַּשׁ בְּרֵאשִׁית

47 חִידוֹנִים לְפִי תָּכְנִית הַלִּמּוּדִים שֶׁל מִשְׂרַד הַחִנּוּךְ (תַּנַ״ךְ חֶמֶ״ד), לְכִתּוֹת א׳–ב׳. בְּכָל חִידוֹן כַּרְטִיסֵי לִמּוּד עִם אַהֲבַת ה׳ 💛 וְשָׁלוֹשׁ רָמוֹת: תְּמוּנוֹת, שְׁאֵלוֹת לַחֲכָמִים, וּבָרָמָה הַשְּׁלִישִׁית מִשְׂחָקִים – בְּכָל פַּעַם מִשְׂחָק אַחֵר

🌍 כִּתָּה א׳ – פָּרָשַׁת בְּרֵאשִׁית

1. בָּרָא אֱלֹהִים: שְׁלוֹשָׁה יָמִים רִאשׁוֹנִיםפֶּרֶק א׳, פְּסוּקִים א׳–י״ג
2. מְאוֹרוֹת, דָּגִים וְעוֹפוֹתפֶּרֶק א׳, פְּסוּקִים י״ד–כ״ג
3. חַיּוֹת הָאָרֶץ וְהָאָדָםפֶּרֶק א׳, פְּסוּקִים כ״ד–ל״א
4. שַׁבָּת קֹדֶשׁפֶּרֶק ב׳, פְּסוּקִים א׳–ג׳
5. גַּן עֵדֶןפֶּרֶק ב׳, פְּסוּקִים ד׳–י״ז
6. לֹא טוֹב הֱיוֹת הָאָדָם לְבַדּוֹפֶּרֶק ב׳, פְּסוּקִים י״ח–כ״ה
7. הַנָּחָשׁ וְעֵץ הַדַּעַתפֶּרֶק ג׳, פְּסוּקִים א׳–ז׳
8. אַיֶּכָּה?פֶּרֶק ג׳, פְּסוּקִים ח׳–כ״ד
9. קַיִן וְהֶבֶל: הַקָּרְבָּנוֹתפֶּרֶק ד׳, פְּסוּקִים א׳–ז׳
10. אֵי הֶבֶל אָחִיךָ?פֶּרֶק ד׳, פְּסוּקִים ח׳–ט״ז
11. מִשֵּׁת עַד נֹחַפֶּרֶק ד׳, פְּסוּקִים כ״ה–כ״ו; פֶּרֶק ה׳
12. דּוֹר הַמַּבּוּלפֶּרֶק ו׳, פְּסוּקִים א׳, ה׳–ח׳

🌈 כִּתָּה א׳ – פָּרָשַׁת נֹחַ

13. נֹחַ אִישׁ צַדִּיק וְהַתֵּבָהפֶּרֶק ו׳, פְּסוּקִים ט׳–כ״ב
14. הַמַּבּוּלפֶּרֶק ז׳
15. עוֹרֵב, יוֹנָה וְקָרְבָּןפֶּרֶק ח׳
16. הַקֶּשֶׁת בֶּעָנָןפֶּרֶק ט׳, פְּסוּקִים א׳–י״ז
17. נֹחַ וּבָנָיופֶּרֶק ט׳, פְּסוּקִים י״ח–כ״ט
18. שִׁבְעִים אֻמּוֹתפֶּרֶק י׳
19. מִגְדַּל בָּבֶלפֶּרֶק י״א, פְּסוּקִים א׳–ט׳
20. מִנֹּחַ עַד אַבְרָםפֶּרֶק י״א, פְּסוּקִים י׳–ל״ב

22. אַבְרָם בְּמִצְרַיִםפֶּרֶק י״ב, פְּסוּקִים י׳–כ׳
23. אַבְרָם וְלוֹטפֶּרֶק י״ג
24. אַרְבָּעָה מְלָכִיםפֶּרֶק י״ד, פְּסוּקִים א׳–י״ב
25. אַבְרָם מַצִּיל אֶת לוֹטפֶּרֶק י״ד, פְּסוּקִים י״ג–כ״ד
26. בְּרִית בֵּין הַבְּתָרִיםפֶּרֶק ט״ו
27. הָגָר וְיִשְׁמָעֵאלפֶּרֶק ט״ז
28. בְּרִית מִילָהפֶּרֶק י״ז, פְּסוּקִים א׳–י״ד
29. שָׂרָה וְיִצְחָקפֶּרֶק י״ז, פְּסוּקִים ט״ו–כ״ז

⛺ כִּתָּה ב׳ – וַיֵּרָא

1. הַכְנָסַת אוֹרְחִים אֵצֶל אַבְרָהָםפֶּרֶק י״ח, פְּסוּקִים א׳–ט״ז
2. תְּפִלַּת אַבְרָהָם עַל סְדֹםפֶּרֶק י״ח, פְּסוּקִים י״ז–ל״ג
3. לוֹט וְצוֹעַרפֶּרֶק י״ט, פְּסוּקִים א׳–כ״ט
4. אַבְרָהָם וַאֲבִימֶלֶךְ בִּגְרָרפֶּרֶק כ׳

6. הָעֲקֵדָה – ׳הִנֵּנִי׳פֶּרֶק כ״ב, פְּסוּקִים א׳–י״ט

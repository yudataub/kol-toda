🍎 ׳תַּפּוּחֵי זָהָב בְּמַשְׂכִּיּוֹת כָּסֶף׳

׳תַּפּוּחֵי זָהָב בְּמַשְׂכִּיּוֹת כָּסֶף דָּבָר דָּבֻר עַל אָפְנָיו׳. תַּפּוּחֵי זָהָב בְּתוֹךְ כְּלִי כֶּסֶף יָפִים מְאוֹד, וְכָךְ גַּם דִּבּוּר שֶׁנֶּאֱמַר בְּדִיּוּק בַּזְּמַן הַנָּכוֹן. אוֹתָהּ מִלָּה, בְּרֶגַע אֶחָד הִיא פּוֹגַעַת וּבְרֶגַע אַחֵר הִיא מְחַיָּה. לְפָעָמִים הַשְׁתִּיקָה חֲשׁוּבָה וְלִפְעָמִים הַמִּלָּה הַטּוֹבָה חַיֶּבֶת לָצֵאת – וְצָרִיךְ חָכְמָה לְהַבְחִין.

דִּבּוּר שֶׁנֶּאֱמַר בְּדִיּוּק בַּזְּמַן הַנָּכוֹן יָפֶה כְּמוֹ תַּפּוּחֵי זָהָב.

💍 ׳נֶזֶם זָהָב וַחֲלִי כָתֶם׳

׳נֶזֶם זָהָב וַחֲלִי כָתֶם מוֹכִיחַ חָכָם עַל אֹזֶן שֹׁמָעַת׳. תּוֹכָחָה שֶׁל חָכָם, אִם הָאָזְנַיִם פְּתוּחוֹת, הִיא כְּמוֹ תַּכְשִׁיט יָקָר. מִי שֶׁמַּקְשִׁיב לְמִי שֶׁאוֹהֵב אוֹתוֹ וּמוֹכִיחַ אוֹתוֹ, מְקַבֵּל מַתָּנָה. מִי שֶׁלֹּא מַקְשִׁיב – הַתַּכְשִׁיט לֹא נִרְאֶה.

הַתּוֹכָחָה שֶׁל הֶחָכָם הִיא תַּכְשִׁיט – כְּשֶׁיֵּשׁ אֹזֶן שׁוֹמַעַת.

💛 אַהֲבַת הַשֵּׁם: ה׳ נֶאֱמָן לְכָל הַבְטָחָה, וַאֲנִי רוֹצֶה לִהְיוֹת נֶאֱמָן כָּמוֹהוּ 💛

רָמָה א׳ – מִלָּה בִּזְמַנָּהּ

❄️ ׳כְּצִנַּת שֶׁלֶג בְּיוֹם קָצִיר׳

׳כְּצִנַּת שֶׁלֶג בְּיוֹם קָצִיר צִיר נֶאֱמָן לְשֹׁלְחָיו וְנֶפֶשׁ אֲדֹנָיו יָשִׁיב׳. בְּיוֹם חַם שֶׁל קָצִיר, קֹר נָעִים הוּא מַתָּנָה. כָּךְ גַּם שָׁלִיחַ נֶאֱמָן: מִי שֶׁשּׁוֹלֵחַ אוֹתוֹ יָכוֹל לִסְמֹךְ עָלָיו וְהוּא מַרְגִּישׁ רֶוַח. חָבֵר שֶׁעוֹשֶׂה אֶת מָה שֶׁהוּא הִבְטִיחַ הוּא כְּמוֹ הַקֹּר הַזֶּה.

שָׁלִיחַ נֶאֱמָן, שֶׁעוֹשֶׂה מַה שֶּׁהִבְטִיחַ, מֵנִיחַ אֶת דַּעְתּוֹ שֶׁל מִי שֶׁשְּׁלָחוֹ.

🕊️

💛 אַהֲבַת הַשֵּׁם: ה׳ מְדַבֵּר אֵלַי בְּרַכּוּת, וַאֲנִי לוֹמֵד לְדַבֵּר כָּךְ 💛

📖 מִשְׁלֵי, פֶּרֶק כ״ה, פָּסוּק ט״ו

רָמָה א׳ – מִלָּה בִּזְמַנָּהּ

🕊️ ׳וְלָשׁוֹן רַכָּה תִּשְׁבָּר גֶּרֶם׳

׳בְּאֹרֶךְ אַפַּיִם יְפֻתֶּה קָצִין וְלָשׁוֹן רַכָּה תִּשְׁבָּר גָּרֶם׳. קָצִין הוּא אָדָם חָזָק וְקָשֶׁה. הָאֲרֻכַּת אַף וְדִבּוּר רַךְ מַצְלִיחִים לְהַרְגִּיעַ אוֹתוֹ, וְהַלָּשׁוֹן הָרַכָּה אֲפִלּוּ ׳תִּשְׁבֹּר גֶּרֶם׳, כְּלוֹמַר תְּשַׁנֶּה לֵב קָשֶׁה כְּמוֹ עֶצֶם. קוֹל נָעִים חָזָק יוֹתֵר מִצְּעָקָה.

דִּבּוּר רַךְ וְסַבְלָנוּת יְכוֹלִים לִשְׁבֹּר אֶת הַלֵּב הַקָּשֶׁה בְּיוֹתֵר.

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לַכֹּל, אֲפִלּוּ לְמִי שֶׁלֹּא מוֹדֶה, וַאֲנִי לוֹמֵד מִמֶּנּוּ 💛

📖 מִשְׁלֵי, פֶּרֶק כ״ה, פְּסוּקִים כ״א–כ״ב

רָמָה א׳ – מִלָּה בִּזְמַנָּהּ

🍞 ׳אִם רָעֵב שֹׂנַאֲךָ הַאֲכִלֵהוּ לָחֶם׳

׳אִם רָעֵב שֹׂנַאֲךָ הַאֲכִלֵהוּ לָחֶם וְאִם צָמֵא הַשְׁקֵהוּ מָיִם׳. מִי שֶׁשּׂוֹנֵא אוֹתִי, אִם הוּא רָעֵב – אֲנִי נוֹתֵן לוֹ לֶאֱכֹל. זֶה לֹא קַל, וְזֶה בְּדִיּוּק מָה שֶׁמְּשַׁנֶּה אֶת הַלֵּב שֶׁלּוֹ וְשֶׁלִּי. ׳וַה׳ יְשַׁלֶּם לָךְ׳ – ה׳ יַחֲזִיר אֶת הַטּוֹב שֶׁעָשִׂיתָ.

גַּם לְמִי שֶׁשּׂוֹנֵא אוֹתִי אֲנִי נוֹתֵן לֶאֱכֹל וְלִשְׁתּוֹת, וְה׳ יְשַׁלֵּם לִי.

🏰 ׳עִיר פְּרוּצָה אֵין חוֹמָה׳

׳עִיר פְּרוּצָה אֵין חוֹמָה אִישׁ אֲשֶׁר אֵין מַעְצָר לְרוּחוֹ׳. עִיר בְּלִי חוֹמָה פְּתוּחָה לְכָל אוֹיֵב. כָּךְ גַּם אָדָם שֶׁלֹּא יָכוֹל לַעֲצֹר אֶת הָרוּחַ שֶׁלּוֹ: אֶת הַכַּעַס, הָרָצוֹן וְהַדִּבּוּר שֶׁיּוֹצֵא בְּלִי מַחְשָׁבָה. ׳מַעְצָר לְרוּחוֹ׳ הוּא כְּמוֹ חוֹמָה פְּנִימִית שֶׁשּׁוֹמֶרֶת עָלֵינוּ.

שְׁלִיטָה עַצְמִית הִיא חוֹמָה שֶׁשּׁוֹמֶרֶת עַל הָאָדָם.

🔥 ׳בְּאֶפֶס עֵצִים תִּכְבֶּה אֵשׁ׳

׳בְּאֶפֶס עֵצִים תִּכְבֶּה אֵשׁ וּבְאֵין נִרְגָּן יִשְׁתֹּק מָדוֹן׳. אֵשׁ בְּלִי עֵצִים כָּבְתָה, וְרִיב בְּלִי מִי שֶׁמְּלַהֵט אוֹתוֹ נִרְגָּע. ׳נִרְגָּן׳ הוּא מִי שֶׁמְּסַפֵּר רְכִילוּת וּמַבְעִיר אֶת הַמַּחֲלֹקֶת. אִם אֲנִי לֹא מַשְׁלִיךְ עֵצִים לָאֵשׁ – אִם אֲנִי לֹא מְסַפֵּר, לֹא מַסִּית וְלֹא מַגְדִּיל – הָרִיב כּוֹבֶה מֵעַצְמוֹ.

בְּלִי עֵצִים הָאֵשׁ כּוֹבָה, וּבְלִי נִרְגָּן הָרִיב נִשְׁקָט.

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹמֵר עָלַי מִבּוֹרוֹת, וַאֲנִי שׁוֹמֵר שֶׁלֹּא לָחְפֹּר בּוֹרוֹת לַאֲחֵרִים 💛

🕳️ ׳כֹּרֶה שַׁחַת בָּהּ יִפֹּל׳

׳כֹּרֶה שַׁחַת בָּהּ יִפֹּל וְגֹלֵל אֶבֶן אֵלָיו תָּשׁוּב׳. מִי שֶׁחוֹפֵר בּוֹר לַחֲבֵרוֹ, סוֹפוֹ שֶׁהוּא עַצְמוֹ נוֹפֵל בּוֹ. וּמִי שֶׁמְּגַלְגֵּל אֶבֶן עַל אֲחֵרִים, הָאֶבֶן חוֹזֶרֶת אֵלָיו. כָּל רָעָה שֶׁאֲנִי מְתַכְנֵן לָזוּלַת, חוֹזֶרֶת אֵלַי. לָכֵן כְּדַאי לָבְחֹר טוֹב, כִּי הַטּוֹב חוֹזֵר גַּם הוּא.

הָרָעָה שֶׁאֲנִי מְתַכְנֵן לַאֲחֵרִים חוֹזֶרֶת אֵלַי – וְכֵן גַּם הַטּוֹב.

מָה לוֹקְחִים מִרָמָה א׳?

אַנְשֵׁי חִזְקִיָּהוּ הֶעְתִּיקוּ עוֹד מִשְׁלֵי שְׁלֹמֹה, וּבַפְּרָקִים הָאֵלֶּה יֵשׁ לָנוּ אוֹצָר שֶׁל עֵצוֹת לַפֶּה וְלַלֵּב: לְדַבֵּר בַּזְּמַן הַנָּכוֹן, לְהִתְנַהֵג בִּנְדִיבוּת אֶל הַשּׂוֹנֵא, לֹא לְהַרְבּוֹת רְכִילוּת. ה׳ אוֹהֵב אוֹתָנוּ, וְלָכֵן הוּא נוֹתֵן לָנוּ דֶּרֶךְ לְהִיטִיב גַּם עִם מִי שֶׁקָּשֶׁה לָנוּ.

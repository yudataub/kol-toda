לִפְנֵי שֶׁבֶר גָּאוֹן

רָמָה ג׳ – מַעֲמִיקִים

שְׁלֹמֹה הַמֶּלֶךְ מְלַמֵּד: מִי שֶׁמַּרְגִּישׁ גָּבוֹהַּ נוֹפֵל, וּמִי שֶׁמַּנִּיחַ אֶת הַדְּאָגוֹת עַל ה׳ – מַרְגִּישׁ אוֹתוֹ לְצִדּוֹ

📖 פֶּרֶק ט״ז

🎯 לַחִידוֹן
⚠️

📜 הַפָּסוּקתּוֹעֲבַת ה׳ כָּל גְּבַהּ לֵב יָד לְיָד לֹא יִנָּקֶה

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁאֶזְכֹּר אוֹתוֹ תָּמִיד, וְלָכֵן הוּא שׁוֹמֵר אוֹתִי מִגַּאֲוָה 💛

📖 מִשְׁלֵי, פֶּרֶק ט״ז, פָּסוּק ה׳ (סוֹטָה ד׳ ע״ב)

רָמָה ג׳ – מַעֲמִיקִים

⚠️ הַגְּמָרָא: גַּסּוּת הָרוּחַ

בְּמַסֶּכֶת סוֹטָה (ד׳ ע״ב) אוֹמֵר רַבִּי יוֹחָנָן מִשּׁוּם רַבִּי שִׁמְעוֹן בֶּן יוֹחַאי: ׳כָּל אָדָם שֶׁיֵּשׁ בּוֹ גַּסּוּת הָרוּחַ – כְּאִלּוּ עוֹבֵד עֲבוֹדָה זָרָה׳, שֶׁנֶּאֱמַר ׳תּוֹעֲבַת ה׳ כָּל גְּבַהּ לֵב׳. וְרַבִּי יוֹחָנָן בְּעַצְמוֹ אָמַר: ׳כְּאִלּוּ כָּפַר בָּעִקָּר׳, שֶׁנֶּאֱמַר ׳וְרָם לְבָבֶךָ וְשָׁכַחְתָּ אֶת ה׳ אֱלֹהֶיךָ׳. כְּלוֹמַר: מִי שֶׁמִּתְנַשֵּׂא מִדַּי שׁוֹכֵחַ שֶׁהַכֹּל מֵה׳.

הַגְּמָרָא: גַּאֲוָה פּוֹתַחַת דֶּלֶת לִשְׁכֹּחַ אֶת ה׳.

1
💔

📜 הַפָּסוּקלִפְנֵי שֶׁבֶר גָּאוֹן וְלִפְנֵי כִשָּׁלוֹן גֹּבַהּ רוּחַ

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי עָנָו, אֲנִי רוֹאֶה יוֹתֵר טוֹב כַּמָּה ה׳ אוֹהֵב אוֹתִי 💛

📖 מִשְׁלֵי, פֶּרֶק ט״ז, פָּסוּק י״ח

רָמָה ג׳ – מַעֲמִיקִים

💔 הַגַּאֲוָה מַסְתִּירָה אֶת אַהֲבַת ה׳

הָרַב לִמֵּד בְּשִׁעוּר שֶׁהַגְּמָרָא אוֹמֶרֶת עַל הַגֵּאֶה: ׳אֵין אֲנִי וְהוּא יְכוֹלִין לָדוּר בָּעוֹלָם׳. הַהֶסְבֵּר הָרָגִיל הוּא שֶׁהַגָּאֶה לוֹקֵחַ מִן הַכָּבוֹד שֶׁל ה׳. הָרַב הוֹסִיף: הַגָּאֶה לוֹקֵחַ מְעַצְמוֹ – הַגַּאֲוָה מוֹנַעַת מֵהָאָדָם לְהַרְגִּישׁ כַּמָּה ה׳ אוֹהֵב אוֹתוֹ. וְעוֹד לִמֵּד: מִי שֶׁגֵּאֶה כְּלַפֵּי אֲנָשִׁים גֵּאֶה גַּם כְּלַפֵּי ה׳, וְלָכֵן אֵינוֹ מַכִּיר טוֹבָה עַל הַחֲסָדִים שֶׁה׳ נוֹתֵן לוֹ.

עֲנָוָה פּוֹתַחַת אֶת הָעֵינַיִם לִרְאוֹת אֶת חַסְדֵי ה׳ וּלְהַגִּיד תּוֹדָה.

2
👃

📜 הַפָּסוּקטוֹב אֶרֶךְ אַפַּיִם מִגִּבּוֹר וּמֹשֵׁל בְּרוּחוֹ מִלֹּכֵד עִיר

💛 אַהֲבַת הַשֵּׁם: ה׳ מַאֲרִיךְ אַפּוֹ אִתִּי, וְאֲנִי לוֹמֵד מִמֶּנּוּ לְהַאֲרִיךְ אַף עִם אֲחֵרִים 💛

📖 מִשְׁלֵי, פֶּרֶק ט״ז, פָּסוּק ל״ב

רָמָה ג׳ – מַעֲמִיקִים

👃 הַשִּׁעוּר עַל ׳אֶרֶךְ אַפַּיִם׳

הָרַב סִפֵּר עַל שִׁעוּר לִפְנֵי תַּלְמִידוֹת: מָה פֵּרוּשׁ ׳אֶרֶךְ אַפַּיִם׳? אַחַת עָנְתָה מִיָּד: סַבְלָנוּת. הָרַב שָׁאַל: אֲבָל לָמָּה סַבְלָנוּת נִקְרֵאת אֹרֶךְ שֶׁל אַף? הוּא בִּקֵּשׁ מִמִּי שֶׁאוֹהֶבֶת לְצַיֵּר: אֵיךְ מְצַיְּרִים מִישֶׁהוּ כּוֹעֵס? פָּנָיו אֲדֻמּוֹת וְהַכַּעַס יוֹשֵׁב בְּקָצֶה הָאַף. אָדָם רָגוּעַ הוּא הַהֶפֶךְ, כְּאִלּוּ יֵשׁ לוֹ אַף אָרֹךְ: הַכַּעַס רָחוֹק, וְעַד שֶׁהוּא מַגִּיעַ הָאָדָם כְּבָר הִסְפִּיק לְהִרָּגַע. כָּל סִפּוּר שֶׁבּוֹ הִתְאַפַּקְנוּ וְלֹא צָעַקְנוּ הוּא סִפּוּר שֶׁל גְּבוּרָה.

אֶרֶךְ אַפַּיִם הוּא מִי שֶׁהַכַּעַס שֶׁלּוֹ רָחוֹק וְהוּא יוֹדֵעַ לְהִרָּגַע.

3
🌟

📜 הַפָּסוּקטוֹב שְׁפַל רוּחַ אֶת עֲנָוִים מֵחַלֵּק שָׁלָל אֶת גֵּאִים

💛 אַהֲבַת הַשֵּׁם: ה׳ מְקַבֵּל אוֹתִי כְּמוֹ שֶׁאֲנִי וְעוֹזֵר לִי לִגְדֹּל 💛

📖 מִשְׁלֵי, פֶּרֶק ט״ז, פָּסוּק י״ט

רָמָה ג׳ – מַעֲמִיקִים

🌟 מָה מִשְׁלֵי מְלַמֵּד אוֹתִי הַיּוֹם

שְׁלֹמֹה לֹא אוֹמֵר לָנוּ לִהְיוֹת קְטַנִּים, אֶלָּא לִהְיוֹת נְכוֹנִים: לְבַקֵּשׁ מֵה׳ אֶת הַמִּלִּים, לְגַלְגֵּל אֵלָיו אֶת הַדְּאָגוֹת, לִבְחֹר חֲבֵרִים עֲנָוִים, וּלְהַאֲרִיךְ אַף. הַיּוֹם אֶבְחַר מַצָּב אֶחָד שֶׁבּוֹ אֲנִי נוֹטֶה לְהִתְפָּרֵץ אוֹ לְהִתְנַשֵּׂא, וַאֲנַסֶּה לְהַשְׁתִּיק אֶת הַקּוֹל הַפְּנִימִי עֲשָׂרָה שְׁנִיּוֹת לִפְנֵי שֶׁאֲדַבֵּר.

עֲנָוָה אֲמִתִּית: לְבַקֵּשׁ עֶזְרָה מֵה׳, לְהַמְתִּין עֲשָׂרָה שְׁנִיּוֹת וְלִשְׁמֹעַ אֶת הַזּוּלַת.

4
💡

רָמָה ג׳ – מַעֲמִיקִים

מָה לוֹקְחִים מִרָמָה ג׳?

חֲכָמֵינוּ הִזְהִירוּ מִגַּאֲוָה, וְהָרַב הִסְבִּיר אֶת הַסּוֹד: הַגַּאֲוָה לֹא רַק פּוֹגַעַת בַּאֲחֵרִים – הִיא מוֹנַעַת מִמֶּנּוּ לְהַרְגִּישׁ כַּמָּה ה׳ אוֹהֵב אוֹתָנוּ. וְהַגְּבוּרָה הַנְּכוֹנָה הִיא לְהַאֲרִיךְ אַף.

סוֹף רָמָה ג׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳גֹּל אֶל ה׳ מַעֲשֶׂיךָ׳ – עַל אֵיזוֹ דְּאָגָה אֲנִי רוֹצֶה לְגַלְגֵּל הַיּוֹם אֶל ה׳, וְלֹא לִסְחֹב אוֹתָהּ לְבַד?
  • מָתַי בַּפַּעַם הָאַחֲרוֹנָה הִרְגַּשְׁתִּי ׳אֲנִי הֲכִי טוֹב׳, וּמָה קָרָה אַחַר כָּךְ? אֵיךְ אֲנִי יָכוֹל לַעֲבֹר מִגַּאֲוָה לַעֲנָוָה בְּלִי לְהַרְגִּישׁ קָטָן?
  • הַפָּסוּק אוֹמֵר ׳טוֹב אֶרֶךְ אַפַּיִם מִגִּבּוֹר׳. מָתַי הָיָה לִי קָשֶׁה לְהִרָּגַע, וּמָה עָזַר לִי לְהַאֲרִיךְ אֶת הָאַף?

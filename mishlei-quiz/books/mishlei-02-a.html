הַפֶּרֶק פּוֹתֵחַ בִּשְׁלוֹשָׁה ׳אִם׳: ׳בְּנִי אִם תִּקַּח אֲמָרַי וּמִצְוֺתַי תִּצְפֹּן אִתָּךְ׳. לוֹקְחִים אֶת הַדְּבָרִים, וְשׁוֹמְרִים אוֹתָם כְּמוֹ דָּבָר יָקָר. ׳לְהַקְשִׁיב לַחָכְמָה אָזְנֶךָ, תַּטֶּה לִבְּךָ לַתְּבוּנָה׳ – מַקְשִׁיבִים בָּאֹזֶן, וְגַם מַטִּים אֶת הַלֵּב.

לוֹמְדִים לֹא רַק בָּאֹזֶן – גַּם בַּלֵּב.

׳כִּי אִם לַבִּינָה תִקְרָא, לַתְּבוּנָה תִּתֵּן קוֹלֶךָ׳. מִי שֶׁרוֹצֶה לְהָבִין צָרִיךְ גַּם לִקְרֹא לָהּ, כְּלוֹמַר לְבַקֵּשׁ, לִשְׁאֹל וְלֹא לְהִתְבַּיֵּשׁ. תַּלְמִיד שֶׁשּׁוֹאֵל שְׁאֵלוֹת מְקָרֵב אֵלָיו אֶת הַהֲבָנָה.

רוֹצִים לְהָבִין – שׁוֹאֲלִים וְקוֹרְאִים לַהֲבָנָה.

💰 ׳אִם תְּבַקְשֶׁנָּה כַכָּסֶף׳

הַפָּסוּק הַמֶּרְכָּזִי: ׳אִם תְּבַקְשֶׁנָּה כַכָּסֶף וְכַמַּטְמוֹנִים תַּחְפְּשֶׂנָּה׳. מִי שֶׁמְּחַפֵּשׂ אוֹצָר לֹא מַפְסִיק אַחֲרֵי חֲמִשָּׁה צְעָדִים. הוּא חוֹפֵר, בּוֹדֵק, חוֹזֵר. כָּךְ גַּם בַּחָכְמָה: מְחַפְּשִׂים בְּסַבְלָנוּת וּבְהַתְמָדָה.

אֶת הַחָכְמָה מְחַפְּשִׂים כְּמוֹ אוֹצָר – בְּסַבְלָנוּת וּבְהַתְמָדָה.

🌟 ׳אָז תָּבִין יִרְאַת ה׳׳

וּמָה מוֹצְאִים? ׳אָז תָּבִין יִרְאַת ה׳ וְדַעַת אֱלֹהִים תִּמְצָא׳. הַמִּלָּה ׳אָז׳ חֲשׁוּבָה: קֹדֶם מְחַפְּשִׂים, וְאַחַר כָּךְ מַבִּינִים. לֹא תָּמִיד מַבִּינִים מִיָּד, אֲבָל מִי שֶׁלֹּא מַפְסִיק לְחַפֵּשׂ – מוֹצֵא.

אַחֲרֵי הַחִפּוּשׂ בָּאָה הַהֲבָנָה: ׳אָז תָּבִין׳.

הַסּוֹד הַיָּפֶה: ׳כִּי ה׳ יִתֵּן חָכְמָה, מִפִּיו דַּעַת וּתְבוּנָה׳. אֲנַחְנוּ מִתְאַמְּצִים, אֲבָל הַחָכְמָה הִיא מַתָּנָה מִה׳. לָכֵן לִפְנֵי הַלִּמּוּד מְבַקְשִׁים מִמֶּנּוּ שֶׁיַּעֲזֹר לָנוּ לְהָבִין.

אֲנִי מִתְאַמֵּץ, וְה׳ הוּא הַנּוֹתֵן אֶת הַחָכְמָה.

🛡️ ׳מָגֵן לְהֹלְכֵי תֹם׳

ה׳ שׁוֹמֵר אֶת מִי שֶׁהוֹלֵךְ בִּתְמִימוּת: ׳יִצְפֹּן לַיְשָׁרִים תּוּשִׁיָּה, מָגֵן לְהֹלְכֵי תֹם׳. תּוּשִׁיָּה הִיא עֵצָה טוֹבָה. מִי שֶׁפּוֹעֵל בִּיוֹשֶׁר, ה׳ מֵכִין לוֹ עֵצוֹת טוֹבוֹת וּמָגֵן שֶׁמֵּגֵן עָלָיו.

מִי שֶׁהוֹלֵךְ בִּתְמִימוּת וּבְיוֹשֶׁר – ה׳ הוּא הַמָּגֵן שֶׁלּוֹ.

🏠

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי הֲבָנָה שֶׁשּׁוֹמֶרֶת עָלַי כָּל הַיּוֹם 💛

📖 מִשְׁלֵי, פֶּרֶק ב׳, פְּסוּקִים י׳–י״א

🏠 ׳כִּי תָבוֹא חָכְמָה בְלִבֶּךָ׳

כְּשֶׁהַחָכְמָה נִכְנֶסֶת אֶל הַלֵּב, ׳וְדַעַת לְנַפְשְׁךָ יִנְעָם׳ – הַלִּמּוּד נַעֲשֶׂה נָעִים. וְהִיא גַּם שׁוֹמֶרֶת: ׳מְזִמָּה תִשְׁמֹר עָלֶיךָ, תְּבוּנָה תִנְצְרֶכָּה׳. מִי שֶׁמֵּבִין הֵיטֵב, יוֹדֵעַ לְהִשָּׁמֵר מִטָּעֻיּוֹת.

חָכְמָה שֶׁנִּכְנְסָה לַלֵּב הִיא נָעִימָה, וְהִיא שׁוֹמֶרֶת עָלֵינוּ.

🌳

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי מָקוֹם בָּעוֹלָם, וְהוּא מְחַזֵּק אֶת הַשָּׁרָשִׁים שֶׁלִּי 💛

📖 מִשְׁלֵי, פֶּרֶק ב׳, פְּסוּקִים כ׳–כ״ב

🌳 ׳יְשָׁרִים יִשְׁכְּנוּ אָרֶץ׳

הַפֶּרֶק מִסְתַּיֵּם בַּהַבְטָחָה: ׳לְמַעַן תֵּלֵךְ בְּדֶרֶךְ טוֹבִים וְאָרְחוֹת צַדִּיקִים תִּשְׁמֹר׳. מִי שֶׁהוֹלֵךְ בְּדֶרֶךְ הַצַּדִּיקִים ׳יִשְׁכְּנוּ אָרֶץ׳ – יִשָּׁאֲרוּ שָׁם, בְּיֵשׁוּב וּבְבִטָּחוֹן, כְּמוֹ עֵץ עִם שָׁרָשִׁים.

מִי שֶׁהוֹלֵךְ בְּדֶרֶךְ הַטּוֹבִים – נִשְׁאָר יַצִּיב וּבָטוּחַ.

מָה לוֹקְחִים מִרָמָה א׳?

מִי שֶׁמֹּצֵא אוֹצָר מִתְמַלֵּא שִׂמְחָה. שְׁלֹמֹה אוֹמֵר: הַחָכְמָה וְיִרְאַת ה׳ הֵן אוֹצָר אֲמִתִּי, וְאֶפְשָׁר לִמְצֹא אוֹתוֹ אִם מְחַפְּשִׂים. וְהַבְּשׂוֹרָה הַיָּפָה: ה׳ בְּעַצְמוֹ נוֹתֵן אֶת הַחָכְמָה, כְּמוֹ אַבָּא שֶׁרוֹצֶה שֶׁבְּנוֹ יִמְצָא.

🤔 חוֹשְׁבִים עַל זֶה

  • אִם הָיִיתִי יוֹדֵעַ שֶׁיֵּשׁ אוֹצָר חָבוּי בַּגִּנָּה, כַּמָּה זְמַן הָיִיתִי מְחַפֵּשׂ? מָה הָאוֹצָר שֶׁאֲנִי רוֹצֶה לִמְצֹא בַּלִּמּוּד הַשָּׁנָה?
  • ׳כִּי ה׳ יִתֵּן חָכְמָה׳ – מָתַי בִּקַּשְׁתִּי מִה׳ שֶׁיַּעֲזֹר לִי לְהָבִין? אֵיךְ אֲנִי יָכוֹל לְבַקֵּשׁ לִפְנֵי שֶׁאֲנִי לוֹמֵד?
  • הַפָּסוּק אוֹמֵר ׳אָז תָּבִין׳ – אַחֲרֵי הַחִפּוּשׂ. עַל אֵיזֶה דָּבָר קָשֶׁה אֲנִי כְּבָר מְחַפֵּשׂ הֲבָנָה, וּמָה יָכוֹל לַעֲזֹר לִי לֹא לְוַתֵּר?

שְׁלֹמֹה מְלַמֵּד שֶׁהַחֶסֶד וְהָאֱמֶת שׁוֹמְרִים עַל הַמֶּלֶךְ וְעַל כָּל אָדָם, וְשֶׁהַנְּשָׁמָה הִיא נֵר שֶׁה׳ מַדְלִיק בְּתוֹכֵנוּ

🕊️ ׳תִּפְאַרְתּוֹ עֲבֹר עַל פָּשַׁע׳

׳שֵׂכֶל אָדָם הֶאֱרִיךְ אַפּוֹ וְתִפְאַרְתּוֹ עֲבֹר עַל פָּשַׁע׳. אָדָם חָכָם מַאֲרִיךְ אֶת הָאַף: הוּא לֹא רָץ מִיָּד לְהִתְקַטֵּט. וּמַה יוֹתֵר יָפֶה? ׳עֲבֹר עַל פָּשַׁע׳ – לִמְחֹל. כְּשֶׁחָבֵר פָּגַע בִּי וַאֲנִי מוֹחֵל לוֹ, זוֹ לֹא חוּלְשָׁה אֶלָּא הַהַצְלָחָה הַיָּפָה בְּיוֹתֵר שֶׁלִּי.

הַתִּפְאֶרֶת שֶׁל הָאָדָם הִיא לַעֲבֹר עַל פְּגִיעָה וְלִמְחֹל.

🪙 ׳מַלְוֵה ה׳ חוֹנֵן דָּל׳

׳מַלְוֵה ה׳ חוֹנֵן דָּל וּגְמֻלוֹ יְשַׁלֶּם לוֹ׳. כְּשֶׁאָדָם נוֹתֵן צְדָקָה לְעָנִי, הוּא כְּאִלּוּ מַלְוֶה לַה׳. וְה׳ אוֹמֵר שֶׁהוּא יַחֲזִיר: אֵין הַצְּדָקָה הוֹלֶכֶת לְאִבּוּד. זֶה לֹא עֲסָקִים – זֶה אֹמֶץ לֵב שֶׁל אָדָם שֶׁלֹּא פּוֹחֵד לָתֵת, כִּי הוּא יוֹדֵעַ שֶׁה׳ נוֹתֵן יוֹתֵר.

הַנּוֹתֵן לְעָנִי מַלְוֶה לַה׳, וְה׳ מַחֲזִיר לוֹ בְּשֶׁפַע.

💭 ׳רַבּוֹת מַחֲשָׁבוֹת בְּלֶב אִישׁ׳

אֲנַחְנוּ חוֹשְׁבִים הַרְבֵּה: מָה נַעֲשֶׂה, לְאָן נֵלֵךְ, מָה נִהְיֶה. ׳רַבּוֹת מַחֲשָׁבוֹת בְּלֶב אִישׁ וַעֲצַת ה׳ הִיא תָקוּם׳. לִפְעָמִים הַתּוֹכְנִיּוֹת הַרַבּוֹת לֹא מִתְקַיְּמוֹת, וְדַוְקָא מַה שֶׁה׳ מְנַהֵל, זֶה שֶׁיִּקְרֶה. זֶה לֹא אוֹמֵר לֹא לְתַכְנֵן – זֶה אוֹמֵר לִהְיוֹת רְגוּעִים: ה׳ יוֹדֵעַ מָה טוֹב לָנוּ.

יֵשׁ לִי הַרְבֵּה תֻּכְנִיּוֹת, אֲבָל מַה שֶׁה׳ רוֹצֶה – הוּא שֶׁיִּתְקַיֵּם, וְהוּא לְטוֹבָתִי.

💛 אַהֲבַת הַשֵּׁם: ה׳ שָׂם בְּלֵב כָּל חָכָם מַעְיָן, וְאֲנִי זוֹכֶה לִדְלוֹת מִמֶּנּוּ 💛

🌊 ׳מַיִם עֲמֻקִּים עֵצָה בְלֶב אִישׁ׳

׳מַיִם עֲמֻקִּים עֵצָה בְלֶב אִישׁ וְאִישׁ תְּבוּנָה יִדְלֶנָּה׳. אֲנָשִׁים חֲכָמִים אֵינָם מְדַבְּרִים הַכֹּל בַּחוּץ: הַחָכְמָה שֶׁלָּהֶם עֲמֻקָּה כְּמוֹ מַיִם בִּבְאֵר, וְצָרִיךְ אָדָם נָבוֹן שֶׁיֵּדַע לִדְלוֹת אוֹתָה, כְּלוֹמַר לִשְׁאֹל, לְהַקְשִׁיב וְלָשִׁית לֵב. לָכֵן מַתְחִילִים לִשְׁאֹל שְׁאֵלוֹת טוֹבוֹת.

חָכְמָה עֲמֻקָּה כְּמוֹ בְּאֵר, וְשׁוֹאֲלִים שְׁאֵלוֹת טוֹבוֹת כְּדֵי לִדְלוֹת אוֹתָהּ.

🤲 ׳רָב אָדָם יִקְרָא אִישׁ חַסְדּוֹ׳

הַרְבֵּה אֲנָשִׁים אוֹמְרִים עַל עַצְמָם: ׳אֲנִי אִישׁ חֶסֶד׳. אֲבָל שְׁלֹמֹה שׁוֹאֵל: ׳וְאִישׁ אֱמוּנִים מִי יִמְצָא׳? מִי הוּא מִי שֶׁבֶּאֱמֶת נוֹתֵן וְעוֹשֶׂה אֶת מָה שֶׁהוּא אוֹמֵר? חֶסֶד אֲמִתִּי הוּא לֹא רַק בְּמִלִּים, אֶלָּא בְּמַעֲשֶׂה, וְעַל זֶה אֶפְשָׁר לִסְמֹךְ.

לֹא מַסְפִּיק לוֹמַר ׳אֲנִי אִישׁ חֶסֶד׳ – צָרִיךְ לַעֲשׂוֹת בְּפֹעַל, וְכָךְ נִהְיִים אִישׁ אֱמוּנִים.

⏳ ׳אַל תֹּאמַר אֲשַׁלְּמָה רָע׳

כְּשֶׁמִּישֶׁהוּ פָּגַע בִּי, יֵשׁ רָצוֹן לְהָגִיב: ׳אֲנִי אֶחֱזִיר לוֹ!׳. הַפָּסוּק אוֹמֵר: ׳אַל תֹּאמַר אֲשַׁלְּמָה רָע קַוֵּה לַה׳ וְיֹשַׁע לָךְ׳. ׳קַוֵּה לַה׳׳ הוּא בִּטָּחוֹן: אֲנִי מַמְתִּין לַה׳, מִתְפַּלֵּל, וְה׳ מוֹשִׁיעַ. מִי שֶׁלֹּא נוֹקֵם נִשְׁאָר עִם לֵב רָגוּעַ.

לֹא מַחֲזִירִים רָע – מְקַוִּים לַה׳, וְהוּא מוֹשִׁיעַ.

🕯️ ׳נֵר ה׳ נִשְׁמַת אָדָם׳

׳נֵר ה׳ נִשְׁמַת אָדָם חֹפֵשׂ כָּל חַדְרֵי בָטֶן׳. הַנְּשָׁמָה שֶׁבְּתוֹכֵנוּ הִיא נֵר שֶׁה׳ הִדְלִיק, וְהָאוֹר שֶׁלָּהּ מֵאִיר לָנוּ אֶת הַפְּנִים: הוּא מְסַיֵּעַ לָנוּ לְהָבִין מָה נָכוֹן וּמָה לֹא, וְלִרְאוֹת אֶת הַמַּחְשָׁבוֹת הַנִּסְתָּרוֹת שֶׁלָּנוּ. כָּל אָדָם נוֹשֵׂא בְּתוֹכוֹ נֵר.

הַנְּשָׁמָה שֶׁלִּי הִיא נֵר שֶׁה׳ הִדְלִיק, וְהִיא מְאִירָה לִי אֶת הַדֶּרֶךְ.

👑 ׳חֶסֶד וֶאֱמֶת יִצְּרוּ מֶלֶךְ׳

׳חֶסֶד וֶאֱמֶת יִצְּרוּ מֶלֶךְ וְסָעַד בַּחֶסֶד כִּסְאוֹ׳. מֶלֶךְ שֶׁפּוֹעֵל בְּחֶסֶד וּבֶאֱמֶת – הַחֶסֶד וְהָאֱמֶת שׁוֹמְרִים עָלָיו, וְהַכִּסֵּא שֶׁלּוֹ עוֹמֵד בְּחֶסֶד. כָּל אֶחָד מֵאִתָּנוּ הוּא מֶלֶךְ בַּחַיִּים הַקְּטַנִּים שֶׁלּוֹ. מִי שֶׁנּוֹהֵג בְּחֶסֶד וּבֶאֱמֶת מוֹצֵא אוֹהֲבִים רַבִּים וְשׁוֹמְרִים אוֹתוֹ.

חֶסֶד וֶאֱמֶת שׁוֹמְרִים עַל מִי שֶׁנּוֹהֵג בָּהֶם.

מָה לוֹקְחִים מִרָמָה א׳?

שְׁנֵי הַפְּרָקִים הָאֵלֶּה מְלַמְּדִים מָה הַכֹּחַ שֶׁשּׁוֹמֵר עַל אָדָם: חֶסֶד וֶאֱמֶת. מִי שֶׁעוֹבֵר עַל פְּגִיעָה, נוֹתֵן לְעָנִי, וְשׁוֹמֵעַ אֶת הַנֵּר שֶׁבִּפְנִים – ה׳ שׁוֹמֵר עָלָיו, וּמְחַזֵּק אֶת הַלֵּב שֶׁלּוֹ בְּאַהֲבָה.

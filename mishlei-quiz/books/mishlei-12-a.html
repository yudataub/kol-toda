רָמָה א׳ – חֲבֵרוּת וְדִבּוּר

🥖 ׳טוֹב פַּת חֲרֵבָה וְשַׁלְוָה בָהּ׳

׳טוֹב פַּת חֲרֵבָה וְשַׁלְוָה בָהּ מִבַּיִת מָלֵא זִבְחֵי רִיב׳. פַּת חֲרֵבָה הִיא פְּרוּסָה יְבֵשָׁה שֶׁל לֶחֶם, וַאֲפִילּוּ כָּךְ יוֹתֵר טוֹב לֶאֱכֹל אוֹתָהּ בְּשָׁלוֹם מִלֶּאֱכֹל סְעוּדָה גְּדוֹלָה בְּבַיִת שֶׁכֻּלּוֹ צְעָקוֹת וּמְרִיבוֹת. הַשָּׁלוֹם בַּבַּיִת וּבַכִּתָּה שָׁווֶה יוֹתֵר מִכָּל מַעֲדָנִים.

שָׁלוֹם וְשַׁלְוָה שָׁוִים יוֹתֵר מִסְּעוּדָה גְּדוֹלָה עִם רִיב.

💛 אַהֲבַת הַשֵּׁם: ה׳ מוֹחֵל וְלֹא מַזְכִּיר, וַאֲנִי לוֹמֵד מִמֶּנּוּ לְכַסּוֹת עַל חֲבֵרִים 💛

📖 מִשְׁלֵי, פֶּרֶק י״ז, פָּסוּק ט׳

רָמָה א׳ – חֲבֵרוּת וְדִבּוּר

🤲 ׳מְכַסֶּה פֶּשַׁע מְבַקֵּשׁ אַהֲבָה׳

כְּשֶׁחָבֵר פּוֹגֵעַ בִּי, יֵשׁ לִי בְּרֵרָה: לְסַפֵּר לְכֻלָּם, אוֹ לְכַסּוֹת. ׳מְכַסֶּה פֶּשַׁע מְבַקֵּשׁ אַהֲבָה וְשֹׁנֶה בְדָבָר מַפְרִיד אַלּוּף׳. מִי שֶׁמְּכַסֶּה עַל הַטָּעוּת שֶׁל חֲבֵרוֹ וְלֹא מַזְכִּיר אוֹתָהּ שׁוּב וָשׁוּב – בּוֹנֶה אַהֲבָה. וּמִי שֶׁחוֹזֵר וּמַזְכִּיר אֶת הָעִנְיָן – מַפְרִיד בֵּין חֲבֵרִים טוֹבִים.

לְכַסּוֹת עַל טָעוּת בּוֹנֶה אַהֲבָה, וְלַחֲזֹר וּלְהַזְכִּיר אוֹתָהּ מַפְרִיד חֲבֵרִים.

2

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי כֹּחַ לַעֲצֹר רִיב לִפְנֵי שֶׁהוּא גָּדֵל 💛

📖 מִשְׁלֵי, פֶּרֶק י״ז, פָּסוּק י״ד

רָמָה א׳ – חֲבֵרוּת וְדִבּוּר

💧 ׳פּוֹטֵר מַיִם רֵאשִׁית מָדוֹן׳

אִם יֵשׁ חֹר קָטָן בִּסְכָר, הַמַּיִם מִתְחִילִים לָצֵאת, וְהַחֹר גָּדֵל. כָּךְ גַּם רִיב: ׳פּוֹטֵר מַיִם רֵאשִׁית מָדוֹן וְלִפְנֵי הִתְגַּלַּע הָרִיב נְטוֹשׁ׳. הַהַתְחָלָה קְטַנָּה – מִלָּה, מַבָּט, צְחוֹק – וְאִם לֹא עוֹצְרִים אוֹתָהּ בַּהַתְחָלָה, הִיא גְּדֵלָה. הַפָּסוּק מְיַעֵץ: עִזְבוּ אֶת הָרִיב לִפְנֵי שֶׁהוּא מִתְפָּרֵץ.

עוֹצְרִים רִיב בַּהַתְחָלָה, כִּי כְּמוֹ חֹר בְּסָכָר הוּא רַק גָּדֵל.

3

💛 אַהֲבַת הַשֵּׁם: ה׳ הוּא הַחָבֵר הֲכִי קָרוֹב, בְּכָל עֵת וּבְכָל רֶגַע 💛

📖 מִשְׁלֵי, פֶּרֶק י״ז, פָּסוּק י״ז; פֶּרֶק י״ח, פָּסוּק כ״ד

👫 ׳בְּכָל עֵת אֹהֵב הָרֵעַ׳

׳בְּכָל עֵת אֹהֵב הָרֵעַ וְאָח לְצָרָה יִוָּלֵד׳. חָבֵר אֲמִתִּי אוֹהֵב בְּכָל עֵת, לֹא רַק כְּשֶׁטּוֹב. וְהַפֶּרֶק הַבָּא מוֹסִיף: ׳וְיֵשׁ אֹהֵב דָּבֵק מֵאָח׳ – יֵשׁ חָבֵר שֶׁמְּחֻבָּר אֵלֶיךָ יוֹתֵר מֵאָח. מִי שֶׁעוֹזֵר בִּשְׁעַת צָרָה הוּא כְּמוֹ אָח שֶׁנּוֹלַד בִּשְׁבִיל הָרֶגַע הַזֶּה.

חָבֵר אֲמִתִּי נִמְצָא בְּכָל עֵת, וּבִפְרָט בְּעֵת צָרָה.

4

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי סִבּוֹת לִשְׂמֹחַ בְּכָל יוֹם 💛

📖 מִשְׁלֵי, פֶּרֶק י״ז, פָּסוּק כ״ב

רָמָה א׳ – חֲבֵרוּת וְדִבּוּר

😄 ׳לֵב שָׂמֵחַ יֵיטִב גֵּהָה׳

׳לֵב שָׂמֵחַ יֵיטִב גֵּהָה וְרוּחַ נְכֵאָה תְּיַבֶּשׁ גָּרֶם׳. שִׂמְחָה הִיא כְּמוֹ תְּרוּפָה לַגּוּף: כְּשֶׁהַלֵּב שָׂמֵחַ גַּם הַגּוּף מַרְגִּישׁ טוֹב יוֹתֵר, וְכַאֲשֶׁר הָרוּחַ עֲצוּבָה הַגּוּף כְּאִלּוּ מִתְיַבֵּשׁ. לָכֵן כְּדַאי לִשְׂמֹחַ בְּמַה שֶׁיֵּשׁ, לָשִׁיר, לְחַיֵּךְ וּלְהַצְחִיק חָבֵר עָצוּב.

שִׂמְחָה מְרַפֵּאת, וְעֶצֶב מְיַבֵּשׁ – אֲנִי בּוֹחֵר בְּשִׂמְחָה וּמְשַׂמֵּחַ אֲחֵרִים.

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי רַבָּנִים וּמוֹרִים שֶׁמְּלַמְּדִים אוֹתִי מַעְיָן שֶׁל חָכְמָה 💛

🌊 ׳מַיִם עֲמֻקִּים דִּבְרֵי פִי אִישׁ׳

׳מַיִם עֲמֻקִּים דִּבְרֵי פִי אִישׁ נַחַל נֹבֵעַ מְקוֹר חָכְמָה׳. דִּבְרֵי חָכָם הֵם כְּמוֹ מַיִם עֲמֻקִּים – מִי שֶׁמַּקְשִׁיב לָהֶם צָרִיךְ לָרֶדֶת וְלִשְׁאֹב מֵהֶם, וְאָז הוּא מוֹצֵא מַעְיָן שֶׁלֹּא נִגְמָר. וְיֵשׁ מֵאֲחוֹרֵי הַמִּלִּים שֶׁלּוֹ הַרְבֵּה מַחְשָׁבָה, וְלָכֵן כְּדַאי לְהַקְשִׁיב.

דִּבְרֵי חָכָם עֲמֻקִּים כְּמוֹ בְּאֵר, וְצָרִיךְ לִשְׁאֹב מֵהֶם בְּסַבְלָנוּת.

💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לִי פֶּה שֶׁיָּכוֹל לְבָרֵךְ, לְהוֹדוֹת וּלְשַׂמֵּחַ 💛

💬 ׳מָוֶת וְחַיִּים בְּיַד לָשׁוֹן׳

׳מָוֶת וְחַיִּים בְּיַד לָשׁוֹן וְאֹהֲבֶיהָ יֹאכַל פִּרְיָהּ׳. הַלָּשׁוֹן קְטַנָּה, אֲבָל יֵשׁ לָהּ כֹּחַ גָּדוֹל: מִלָּה טוֹבָה יְכוֹלָה לְהַחֲיוֹת לֵב שָׁבוּר, וּמִלָּה רָעָה יְכוֹלָה לִפְגֹּעַ. מִי שֶׁאוֹהֵב לְדַבֵּר, יֹאכַל אֶת הַפְּרִי שֶׁל הַדִּבּוּר שֶׁלּוֹ, טוֹב אוֹ רַע. לָכֵן כְּדַאי לִבְחֹר מִלִּים שֶׁנּוֹתְנוֹת חַיִּים.

הַמִּלִּים שֶׁלִּי יְכוֹלוֹת לָתֵת חַיִּים אוֹ לִפְגֹּעַ – אֲנִי בּוֹחֵר מִלִּים שֶׁנּוֹתְנוֹת חַיִּים.

💛 אַהֲבַת הַשֵּׁם: ה׳ הוּא הַמִּגְדָּל שֶׁלִּי, וְהַדֶּלֶת תָּמִיד פְּתוּחָה 💛

📖 מִשְׁלֵי, פֶּרֶק י״ח, פָּסוּק י׳

🗼 ׳מִגְדַּל עֹז שֵׁם ה׳׳

כְּשֶׁיֵּשׁ צָרָה, לְאָן רָצִים? ׳מִגְדַּל עֹז שֵׁם ה׳ בּוֹ יָרוּץ צַדִּיק וְנִשְׂגָּב׳. שֵׁם ה' הוּא כְּמוֹ מִגְדָּל חָזָק: אֲנִי קוֹרֵא לוֹ, מִתְפַּלֵּל אֵלָיו, וּמַרְגִּישׁ שֶׁאֲנִי לֹא לְבַד. ׳יָרוּץ׳ – הַצַּדִּיק לֹא הוֹלֵךְ לְאַט, הוּא רָץ מִיָּד לַמָּקוֹם הַבָּטוּחַ.

כְּשֶׁקָּשֶׁה, רָצִים אֶל ה׳ בִּתְפִלָּה, וּמַרְגִּישִׁים מְחֻזָּקִים.

מָה לוֹקְחִים מִרָמָה א׳?

בִּשְׁנֵי הַפְּרָקִים הָאֵלֶּה שְׁלֹמֹה לוֹמֵד אִתָּנוּ אֵיךְ לִהְיוֹת חָבֵר טוֹב וְאֵיךְ לְהִשְׁתַּמֵּשׁ בַּלָּשׁוֹן: לְכַסּוֹת פְּגִיעָה, לִמְנֹעַ רִיב, לִהְיוֹת שָׂמֵחַ, וּלְדַבֵּר בְּעֹמֶק. וּכְשֶׁקָּשֶׁה, יֵשׁ מָקוֹם בְּטוּחַ: שֵׁם ה׳. ה׳ הוּא הָאוֹהֵב הַדָּבֵק מֵאָח.

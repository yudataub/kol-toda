רַשִׁ״י: ׳אֵשֶׁת חַיִל׳ הִיא הַתּוֹרָה – אַשְׁרֵי הַזּוֹכֶה לִמְצֹא אוֹתָהּ, וְהִיא יְקָרָה מִמַּרְגָּלִיּוֹת.

✨ מְצוּדַת דָּוִד: מָשָׁל עַל הַנֶּפֶשׁ

מְצוּדַת דָּוִד כּוֹתֵב בְּסוֹף הַפֶּרֶק: ׳וְכָל הָעִנְיָן אָמוּר גַּם לְמָשָׁל עַל הַנֶּפֶשׁ הַמַּשְׂכֶּלֶת, שֶׁהִיא עוֹזֶרֶת לְבַעֲלָהּ כְּאִשָּׁה לְבַעֲלָהּ׳. וְעַל ׳שֶׁקֶר הַחֵן וְהֶבֶל הַיֹּפִי׳ הוּא מְבָאֵר: ׳הָעִיּוּנִים הַדַּקִּים בִּשְׁאָר הַחָכְמוֹת אֵין תּוֹעַלְתָּן רַבָּה מוּל חָכְמַת הַתּוֹרָה׳. רַשִׁ״י מוֹסִיף: ׳שֶׁקֶר הַחֵן וְהֶבֶל הַיֹּפִי – שֶׁב עַל כָּל הָעֲבוֹדוֹת זָרוֹת, וְהֶבֶל גְּדֻלָּתָן׳, וְ׳אִשָּׁה יִרְאַת ה׳ הִיא לְבַד תִּתְהַלָּל׳.

מְצוּדַת דָּוִד: הַפֶּרֶק הוּא גַּם מָשָׁל עַל הַנֶּפֶשׁ הַחֲכָמָה – וְחָכְמַת הַתּוֹרָה עוֹלָה עַל כֻּלָּן.

רָמָה ב׳ – הַמָּשָׁל עַל הַתּוֹרָה

מָה לוֹקְחִים מִרָמָה ב׳?

רַשִׁ״י אוֹמֵר בְּפֵרוּשׁ: ׳לְפִי הַמָּשָׁל עַל הַתּוֹרָה וְלוֹמְדֶיהָ הַפָּרָשָׁה מִתְבָּאֶרֶת׳, וְהוּא מְפָרֵשׁ פָּסוּק אַחַר פָּסוּק מָה הַתּוֹרָה עוֹשָׂה לְלוֹמְדֶיהָ. וּמְצוּדַת דָּוִד מַסְבִּיר שֶׁהַכֹּל מָשָׁל גַּם עַל הַנֶּפֶשׁ הַמַּשְׂכֶּלֶת. כָּךְ הַשִּׁיר מוֹדֶה לַתּוֹרָה בְּאַהֲבָה.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • מָה אֲנִי יָכוֹל לַעֲשׂוֹת בְּבַיִת שֶׁלִּי כְּדֵי לִהְיוֹת ׳בַּיִת שֶׁל חֶסֶד וַעֲבוֹדָה׳, גַּם כְּשֶׁאֲנִי לֹא הַמְּבֻגָּר?
  • מִי הוּא הָאָדָם שֶׁ׳פִּיו פָּתַח בְּחָכְמָה וְתוֹרַת חֶסֶד עַל לְשׁוֹנוֹ׳, וּמָה אֲנִי לוֹמֵד מִמֶּנּוּ?
  • אִם הַתּוֹרָה הִיא ׳אֵשֶׁת חַיִל׳ שֶׁשָּׂמָה לִי אוֹצָר, מָה הוּא הַחֵלֶק הַיּוֹמִי שֶׁאֲנִי מוֹסִיף לָהּ?

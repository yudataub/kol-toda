רָמָה א׳ – נוֹפְלִים וְקָמִים

🌟 ׳בְּנִי אִם חָכַם לִבֶּךָ׳

׳בְּנִי אִם חָכַם לִבֶּךָ יִשְׂמַח לִבִּי גַם אָנִי׳. הַמְּחַנֵּךְ, הֶהוֹרֶה וְהַמּוֹרֶה, שְׂמֵחִים מְאוֹד כְּשֶׁהַתַּלְמִיד מִתְחַכֵּם וּמְדַבֵּר ׳מֵישָׁרִים׳. הַשִּׂמְחָה שֶׁלָּהֶם גְּדוֹלָה אֲפִלּוּ יוֹתֵר מִשִּׂמְחַת הַתַּלְמִיד עַצְמוֹ. כְּשֶׁאֲנִי לוֹמֵד וּמִשְׁתַּפֵּר, אֲנִי מְשַׂמֵּחַ אֶת כָּל מִי שֶׁאוֹהֵב אוֹתִי.

כְּשֶׁאֲנִי לוֹמֵד וּמִשְׁתַּפֵּר, אֲנִי מְשַׂמֵּחַ אֶת הוֹרַי וְאֶת מוֹרַי.

1

💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לִי הוֹרִים שֶׁאוֹהֲבִים אוֹתִי, וְהוּא אוֹהֵב אוֹתִי יוֹתֵר 💛

📖 מִשְׁלֵי, פֶּרֶק כ״ג, פְּסוּקִים כ״ב, כ״ה

רָמָה א׳ – נוֹפְלִים וְקָמִים

👨‍👧 ׳שְׁמַע לְאָבִיךָ זֶה יְלָדֶךָ׳

׳שְׁמַע לְאָבִיךָ זֶה יְלָדֶךָ וְאַל תָּבוּז כִּי זָקְנָה אִמֶּךָ׳. הֲרֵי זֶה אָבִיךָ שֶׁהוֹלִיד אוֹתְךָ – מִן הַסְּתָם הוּא רוֹצֶה בְּטוֹבָתְךָ. וְגַם כְּשֶׁהוֹרִים מִזְדַּקְּנִים, אֵין לְזַלְזֵל בִּדְבָרֵיהֶם: הֵם עָבְרוּ הַרְבֵּה וְרוֹאִים רָחוֹק. ׳יִשְׂמַח אָבִיךָ וְאִמֶּךָ׳ – זֶה הַפְּרִי שֶׁל הַקְשָׁבָה.

מַקְשִׁיבִים לְהוֹרִים, כִּי הֵם רוֹצִים בְּטוֹבָתֵנוּ, וְזֶה מְשַׂמֵּחַ אוֹתָם.

2

💛 אַהֲבַת הַשֵּׁם: ה׳ הוּא אֱמֶת, וְכָל פַּעַם שֶׁאֲנִי אוֹמֵר אֱמֶת אֲנִי קָרוֹב אֵלָיו 💛

📖 מִשְׁלֵי, פֶּרֶק כ״ג, פָּסוּק כ״ג

רָמָה א׳ – נוֹפְלִים וְקָמִים

💎 ׳אֱמֶת קְנֵה וְאַל תִּמְכֹּר׳

׳אֱמֶת קְנֵה וְאַל תִּמְכֹּר חָכְמָה וּמוּסָר וּבִינָה׳. אֱמֶת הִיא דָּבָר שֶׁקּוֹנִים בְּמַאֲמָץ, וְאֵין לְוַתֵּר עָלָיו אֲפִלּוּ בִּשְׁבִיל רֶוַח. אִם קַשֶּׁה לֶאֱמֹר אֱמֶת וּלְהוֹדוֹת בְּטָעוּת, כְּדַאי לְהִתְאַמֵּץ וְלֹא לִמְכֹּר אֶת הָאֱמֶת תְּמוּרַת שֶׁלֹּא יִצְחֲקוּ עָלֵינוּ.

קוֹנִים אֱמֶת, חָכְמָה וּבִינָה, וְלֹא מוֹכְרִים אוֹתָם בִּשְׁבִיל שׁוּם דָּבָר.

3

💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לִי בַּיִת שֶׁל אַהֲבָה, וְאֲנִי עוֹזֵר לְבַנּוֹת אוֹתוֹ 💛

📖 מִשְׁלֵי, פֶּרֶק כ״ד, פְּסוּקִים ג׳–ד׳

רָמָה א׳ – נוֹפְלִים וְקָמִים

🏠 ׳בְּחָכְמָה יִבָּנֶה בָּיִת׳

׳בְּחָכְמָה יִבָּנֶה בָּיִת וּבִתְבוּנָה יִתְכּוֹנָן. וּבְדַעַת חֲדָרִים יִמָּלְאוּ כָּל הוֹן יָקָר וְנָעִים׳. בַּיִת טוֹב נִבְנֶה מֵחָכְמָה: אֲנָשִׁים שֶׁמַּקְשִׁיבִים זֶה לָזֶה, מְדַבְּרִים בְּנֹעַם וּמְבִינִים זֶה אֶת זֶה. וְהַחֲדָרִים מִתְמַלְּאִים לֹא רַק בְּרָהִיטִים אֶלָּא בְּדַעַת, בְּאַהֲבָה וּבְדִבְרֵי טַעַם.

בַּיִת בָּנוּי עַל חָכְמָה וּתְבוּנָה, וְהַחֲדָרִים שֶׁלּוֹ מְלֵאִים בְּדַעַת.

4

💛 אַהֲבַת הַשֵּׁם: ה׳ שָׂם בְּדַרְכִּי אֲנָשִׁים טוֹבִים שֶׁיְּכוֹלִים לְיַעֵץ לִי 💛

📖 מִשְׁלֵי, פֶּרֶק כ״ד, פְּסוּקִים ה׳–ו׳

רָמָה א׳ – נוֹפְלִים וְקָמִים

🧠 ׳בְּרֹב יוֹעֵץ תְּשׁוּעָה׳

׳גֶּבֶר חָכָם בַּעוֹז וְאִישׁ דַּעַת מְאַמֶּץ כֹּחַ׳; ׳כִּי בְתַחְבֻּלוֹת תַּעֲשֶׂה לְּךָ מִלְחָמָה וּתְשׁוּעָה בְּרֹב יוֹעֵץ׳. חָכְמָה נוֹתֶנֶת כֹּחַ. כְּשֶׁיֵּשׁ מִלְחָמָה אוֹ אֶתְגָּר גָּדוֹל, לֹא מִסְתַּפְּקִים בְּדֵעָה אַחַת – מִתְיָעֲצִים עִם הַרְבֵּה אֲנָשִׁים חֲכָמִים. מִי שֶׁשּׁוֹאֵל, מַצְלִיחַ.

לֹא מַחְלִיטִים לְבַד – מִתְיָעֲצִים עִם אֲנָשִׁים חֲכָמִים.

5

💛 אַהֲבַת הַשֵּׁם: ה׳ מְלַמֵּד אוֹתִי לַעֲזֹר, כִּי הוּא עוֹזֵר לִי בְּכָל רֶגַע 💛

📖 מִשְׁלֵי, פֶּרֶק כ״ד, פְּסוּקִים י׳–י״א

רָמָה א׳ – נוֹפְלִים וְקָמִים

🤝 ׳הִתְרַפִּיתָ בְּיוֹם צָרָה׳

׳הִתְרַפִּיתָ בְּיוֹם צָרָה צַר כֹּחֶכָה׳. אִם חָבֵר בְּצָרָה וְאֲנִי עוֹמֵד מִן הַצַּד, הַכֹּחַ שֶׁלִּי עַצְמִי מִצְטַמְצֵם: כְּשֶׁאֶזְדַּקֵּק לְעֶזְרָה, אַף אֶחָד לֹא יָבוֹא. וְהַפָּסוּק מוֹסִיף: ׳הַצֵּל לְקֻחִים לַמָּוֶת׳ – כְּשֶׁאָדָם בְּסַכָּנָה, אֵין לִמְנֹעַ עֶזְרָה.

בְּיוֹם צָרָה לֹא מִתְרַפִּים – הָעֶזְרָה לַזּוּלָת מַגְדִּילָה אֶת הַכֹּחַ שֶׁלִּי.

6

💛 אַהֲבַת הַשֵּׁם: ה׳ מֵרִים אוֹתִי בְּכָל פַּעַם שֶׁאֲנִי נוֹפֵל, וְאוֹהֵב אוֹתִי כְּשֶׁאֲנִי קָם 💛

רָמָה א׳ – נוֹפְלִים וְקָמִים

🔄 ׳כִּי שֶׁבַע יִפּוֹל צַדִּיק וָקָם׳

הַפָּסוּק הַמְפֻרְסָם שֶׁל הַפֶּרֶק: ׳כִּי שֶׁבַע יִפּוֹל צַדִּיק וָקָם וּרְשָׁעִים יִכָּשְׁלוּ בְרָעָה׳. גַּם הַצַּדִּיק נוֹפֵל – אֲפִלּוּ שֶׁבַע פְּעָמִים – אֲבָל מַה שֶּׁמַּבְדִּיל אוֹתוֹ הוּא שֶׁהוּא קָם. מִי שֶׁטּוֹעֶה וְלוֹקֵחַ מִכָּךְ לֶקַח, וְקָם לְנַסּוֹת שׁוּב, אֶת הַנְּפִילוֹת שֶׁלּוֹ הוֹפֵךְ לְמַדְרֵגוֹת.

הַצַּדִּיק נוֹפֵל, אֲבָל קָם בְּכָל פַּעַם – וְכָךְ הוּא עוֹלֶה.

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב גַּם אֶת מִי שֶׁפָּגַע בִּי, וְרוֹצֶה שֶׁגַּם אֲנִי אֶרְחַם עָלָיו 💛

📖 מִשְׁלֵי, פֶּרֶק כ״ד, פְּסוּקִים י״ז–י״ח

רָמָה א׳ – נוֹפְלִים וְקָמִים

🙊 ׳בִּנְפֹל אוֹיִבְךָ אַל תִּשְׂמָח׳

׳בִּנְפֹל אוֹיִבְךָ אַל תִּשְׂמָח וּבִכָּשְׁלוֹ אַל יָגֵל לִבֶּךָ׳. גַּם כְּשֶׁמִּישֶׁהוּ שֶׁפָּגַע בִּי נוֹפֵל, אֵין לִשְׂמֹחַ. הַנִּמּוּק: ׳פֶּן יִרְאֶה ה׳ וְרַע בְּעֵינָיו׳ – ה׳ לֹא אוֹהֵב לִרְאוֹת אוֹתָנוּ שְׂמֵחִים בְּצַעַר הַזּוּלַת. לָכֵן כְּשֶׁהַתַּלְמִיד שֶׁצָּחַק עָלַי מְקַבֵּל צִיּוּן נָמוּךְ, אֲנִי לֹא חוֹגֵג.

לֹא שְׂמֵחִים בִּנְפִילָה שֶׁל אוֹיֵב – ה׳ אוֹהֵב אֶת כֻּלָּם.

מָה לוֹקְחִים מִרָמָה א׳?

שִׁבְעָה פְּעָמִים נוֹפֵל הַצַּדִּיק, וְקָם. זֶה הַסּוֹד: לֹא מְחַפְּשִׂים צַדִּיק שֶׁלֹּא נוֹפֵל, אֶלָּא צַדִּיק שֶׁקָּם. ה׳ אוֹהֵב אוֹתָנוּ, מֵרִים אוֹתָנוּ כָּל פַּעַם, וּמְלַמֵּד אוֹתָנוּ לְהָרִים גַּם אֲחֵרִים.

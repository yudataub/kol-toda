ראש השנה - חג מיוחד ויחיד! ילדים יקרים! איזה חג הוא היחיד בשנה שחוגגים אותו שני ימים? נכון - ראש השנה! אבל למה דווקא שני ימים? כי לראש השנה יש שתי בחינות מיוחדות! | שתי הבחינות של ראש השנה יום הדין "בראש השנה יכתבון" שלושה ספרים נפתחים זמן של בדיקה ומשפט יום המלכות מכתירים את ה' כמלך אומרים מלכויות בתפילה זמן של שמחה וכתרת המלך השאלה הגדולה: איזה מהם העיקר? יום הדין או יום המלכות? התשובה מפתיעה - זה תלוי איפה אנחנו נמצאים! | השופר חושף את הסוד! גם בתקיעת השופר יש שתי הבחינות: הכתרת המלך: תוקעים שופר כמו שתוקעים חצוצרות כשמלך עולה לכסא קריאה לתשובה: "שפרו מעשיכם!" - השופר מזהיר ומעורר אז איך נדע מתי השופר אומר מה? שופר של מלכות תקיעות חזקות ושמחות "המלך בא!" שופר של תשובה תרועות ושברים "התעוררו לתשובה!" | הגילוי המפתיע על התפילות! חכמים גילו דבר מדהים: בדקו את כל התפילות שחוברו לראש השנה במקומות שונים... תפילות בארץ ישראל נכתבו בזמן חז"ל כולן עוסקות במלכות! "מלכנו ואלוקינו" פיוטים בחוץ לארץ נכתבו בזמן הגלות כולם עוסקים בדין! "יום דין, יום תוכחה" המסקנה המדהימה: בארץ ישראל - ראש השנה הוא בעיקר יום מלכות! בחוץ לארץ - ראש השנה הוא בעיקר יום דין! | עזרא חושף את המפתח! כשעזרא הסופר עלה לארץ ישראל עם העם, מה הוא אמר להם? "אל תעצבו כי חדות ה' היא מעוזכם!" נחמיה ח עזרא לימד אותנו: כשעם ישראל חוזר לארץ ישראל, השמחה חוזרת! כי בארץ ישראל ה' הוא המלך, ועם המלך תמיד שמחים! | איפה יש מלך אמיתי? הרמח"ל (רבי משה חיים לוצאטו) מסביר את הסוד: מלכות ה' יכולה להתגלות רק במקום מסוים! בחוץ לארץ "כמי שאין לו אלוקה" קשה להרגיש שיש מלך בעיקר זמן של דין בארץ ישראל כאן ה' מלך באמת! המלכות מורגשת זמן של שמחה ומלכות למה? כי צריכים שלושה דברים למלכות: מלך - הקב"ה עם - עם ישראל ארץ - ארץ ישראל רק כשהכל ביחד - יש מלכות אמיתית! | מאמר החכמה והנה לשתתחזק הקדושה כראוי צריך שיהיו ישראל בארץ ישראל, כי הם עם הנבחר והיא הארץ הנבחרת ובישראל ימלוך מלך מבית דוד שהוא המלך הנבחר ובו נקשרים כל ישראל ליקשר בקדושה ואם אחד מהתנאים האלה חסר, אין הקדושה מתחזקת כראוי, ואמנם על פי הסדר הזה מתפללים, והיינו אחר שהתפללנו על עיטור כללם של ישראל מתפללים על הארץ והיינו שמחה לארצך וכו' ואחר כך על מלכות בית דוד והיינו צמיחות וכו' | איפה יש את כל המצוות הגדולות? בואו נבדוק: איפה יכולים לקיים את המצוות של ניהול עם? רק בארץ ישראל! מלך סנהדרין (בית דין הגדול) שופטים שוטרים כל המצוות של ניהול עם - רק בארץ ישראל! למה? כי רק בארץ ישראל יש מלכות אמיתית! בחוץ לארץ אי אפשר להיות עם שלם עם מלך שלם! | מי רוצה להצטרף לעם המלך? עוד הוכחה מדהימה לכוח של ארץ ישראל: בחוץ לארץ כמעט אף אחד לא התגייר מי רוצה להצטרף לעם נרדף? קשה לראות את הגדולה בארץ ישראל הרבה גרים התגיירו! רואים את מלכות ה' רוצים להיות חלק מהעם המלכותי למה? כי בארץ ישראל רואים את המלכות האמיתית! כשרואים עם עם מלך אמיתי - רוצים להצטרף! | כשעם ישראל חזר הביתה בבית שני, כשעם ישראל עלה בחזרה לארץ ישראל, הם חשבו שזה ימשיך כמו בגלות... אבל עזרא אמר להם: "אל תעצבו!" "חדות ה' היא מעוזכם!" עכשיו אתם בארץ - עכשיו יש שמחה של מלכות! זה לימד אותנו: החזרה לארץ ישראל משנה את כל אופי ראש השנה! מיום דין עצוב - ליום מלכות שמח! | מה אנחנו מבקשים בתפילה? כשאנחנו מתפללים בראש השנה, מה אנחנו אומרים? "תן כבוד לעמך, שמחה לארצך" כבוד לעמך שעם ישראל יהיה מכובד שיכירו במלכות שלנו שמחה לארצך שארץ ישראל תהיה שמחה שתהיה מקום של מלכות אנחנו מבקשים שהעם והארץ יהיו ביחד - כי אז יש מלכות אמיתית! | התורה משפיעה על המציאות! דבר מדהים: התורה שאנחנו לומדים יכולה לשנות את העולם! דוגמה: תורת הגר"א (הגאון מוילנה) 1 הגר"א למד תורה 2 התלמידים עלו לארץ 3 התחיל הישוב! כך גם הרב אלקלעי לימד: ללמוד תורה על ארץ ישראל - זה מביא לרצון לעלות לארץ ולהקים כאן מלכות! | איך זה קורה בפועל היום? בואו נראה איך מלכות ה' מתגלה בפועל בארץ ישראל היום! העולם כולו רואה דברים מדהימים שקורים כאן: רפואה עם אהבה רופא יהודי מטפל בילד ערבי מתוך אהבת ה' ואהבת האדם העולם רואה את האור של התורה! שמחה והצלחה למרות כל האויבים מסביב - עם ישראל מצליח ושמח! התורה נותנת כוח פנימי מדהים! ניצחונות מפתיעים צבא קטן מנצח צבאות ענקיים כמו דוד וגוליית - שוב ושוב! העולם רואה יד ה' עוזרת! חקלאות מופלאה מדבר הפך לגן עדן! פירות ומים במקום שהיה חרב תוך כדי לימוד תורה! | העולם כולו תמה מהצלחת ישראל העולם כולו תמה: היי-טק מתקדם + לימוד תורה כלכלה משגשגת + שמירת מצוות חינוך מעולה + ערכי תורה רפואה מובילה + רחמים וחסד איך זה אפשרי? כי כשעם ישראל חי בארץ ישראל על פי התורה - מתגלה בעולם הכוח האמיתי של מלכות ה'! זה לא מקרה - זו התגלות אלוהית שכל העולם רואה! | מצווה מיוחדת של מלכות כשאנחנו בארץ ישראל, יש לנו מצווה מיוחדת: לקרב יהודים בתשובה! זה חלק ממצוות אהבת ה' - להראות לכל יהודי כמה יפה ונפלא להיות חלק ממלכות ה'! למה דווקא בארץ ישראל? כי כאן הכי קל לראות את המלכות! כשיהודי רואה יהודים אחרים חיים בקדושה בארץ הקודש - הוא מתחבר בחזרה למלכות שלו! | אפשר מלכות גם בלי משיח? שאלה מעניינת: האם צריך לחכות למשיח בשביל מלכות? התשובה מפתיעה - לא! סדר ברכות שמונה עשרה מלמד אותנו: ברכה 14 "ולירושלים עירך ברחמים תשוב" חזרה לירושלים ברכה 15 "את צמח דוד עבדך מהרה תצמיח" בוא המשיח רואים? קודם מבקשים חזרה לירושלים, ואחר כך משיח! זה אומר שיש מלכות כבר עכשיו - גם לפני המשיח! | מה הקשר לצום גדליה? שאלה חכמה: מה הקשר בין צום גדליה לעשרת ימי תשובה? למה בדיוק אחרי ראש השנה - יום שצריך להיות שמח - יש צום? התשובה: אין תשובה שלימה של עם ישראל אלא רק בארץ ישראל! צום גדליה מזכיר לנו: כשהרגו את גדליה - נגמר הישוב בארץ העם הלך לגלות רק כשנחזור לארץ - תהיה תשובה שלימה לכן הצום בא אחרי ראש השנה - להזכיר שהמלכות השלימה תהיה רק כשכל העם יחזור לארץ! | הפסוק שמסביר הכל "אחר ישובו בני ישראל ובקשו את ה' אלוקיהם ואת דוד מלכם ופחדו אל ה' ואל טובו באחרית הימים" הושע ג, ה חז"ל מסבירים את הפסוק: "ובקשו את ה' אלוקיהם" - זו מלכות שמים "ואת דוד מלכם" - מלכות בית דוד "ופחדו אל ה' ואל טובו" - זה בית המקדש והרב יהודה אלקלעי מוסיף הסבר חשוב: "שיבקשו מלכות שמים - היינו לשוב לארץ ישראל, כדי שייקרא הקב"ה מלך!" | מה זה אומר לחיים שלנו? עכשיו אנחנו מבינים למה אנחנו כל כך בני מזל! אנחנו חיים בארץ ישראל! המשמעות לנו: ראש השנה שלנו יותר מלכות ופחות דין השופר שלנו יותר הכתרת מלך השמחה שלנו גדולה יותר! התפקיד שלנו להראות מלכות לעולם | מה התחייבות שלנו? אם אנחנו זוכים לחיות בארץ ישראל ולהרגיש את המלכות - מה האחריות שלנו? להיות נציגי המלך להתנהג בקדושה לכבד זה את זה להראות לעולם איך נראית מלכות לאהוב את הארץ לשמור על ארץ ישראל לבנות אותה לעזור לכל יהודי לחזור המטרה שלנו בראש השנה: להכתיר את ה' כמלך - לא רק על העולם, אלא במיוחד על ארץ ישראל! | סיכום: הסוד הגדול של ראש השנה הגילוי המדהים 1 בחוץ לארץ ראש השנה = יום דין 2 חזרה לארץ משנה הכל! 3 בארץ ישראל ראש השנה = יום מלכות למה? כי מלכות אמיתית צריכה שלושה דברים: מלך הקב"ה 1 עם עם ישראל 2 ארץ ארץ ישראל 3 רק ביחד - יש מלכות! | ברכה מיוחדת לשנה החדשה לשנה טובה ומתוקה! שנזכה להכתיר את ה' כמלך על ארץ ישראל ולהיות נציגים ראויים של המלכות הקדושה! שנזכה השנה: להרגיש את מלכות ה' בכל יום בארץ ישראל לשמוע בשופר קול של שמחה ומלכות לקרב כל יהודי למלכות ה' לבנות את ארץ ישראל ולהראות לעולם את האור לאהוב את הארץ ואת העם ואת המלך תזכרו תמיד: אנחנו בני המלך, בארץ המלך! | הנחיות למורה להמשך רעיונות להעשרה: דיון: איך אנחנו מרגישים שאנחנו בארץ המלך? יצירה: עיצוב כתר למלכות ה' בארץ ישראל סיפורים: דוגמאות ממוחשות מהחיים של מלכות בארץ שירה: חיבור שיר על השמחה בארץ ישראל שאלות לדיון מתקדם: איך אנחנו יכולים להיות נציגים טובים של המלך? מה ההבדל בין גלות למלכות?

📢 ׳צַו אֶת אַהֲרֹן׳

ה׳ אוֹמֵר לְמֹשֶׁה: ׳צַו אֶת אַהֲרֹן וְאֶת בָּנָיו׳. רַשִׁ״י מְסַבֵּר: ׳צַו׳ הִיא מִלָּה שֶׁל זֵרוּז – לַעֲשׂוֹת מַהֵר, עַכְשָׁו וְגַם לְכָל הַדּוֹרוֹת. הָעוֹלָה נִשְׁאֶרֶת עַל הַמִּזְבֵּחַ ׳כָּל הַלַּיְלָה עַד הַבֹּקֶר׳, וְהָאֵשׁ דּוֹלֶקֶת כָּל הַזְּמַן.

אֶת מִצְווֹת ה׳ עוֹשִׂים בִּזְרִיזוּת.

רָמָה א׳ – כָּל הַפָּרָשָׁה

🧹 תְּרוּמַת הַדֶּשֶׁן

כָּל בֹּקֶר, לִפְנֵי הַכֹּל, הַכֹּהֵן לוֹבֵשׁ בִּגְדֵי בַּד לְבָנִים – ׳וְלָבַשׁ הַכֹּהֵן מִדּוֹ בַד׳. הוּא לוֹקֵחַ מַחְתָּה מְלֵאָה מֵהָאֵפֶר שֶׁעַל הַמִּזְבֵּחַ – הַדֶּשֶׁן – ׳וְשָׂמוֹ אֵצֶל הַמִּזְבֵּחַ׳. רַשִׁ״י מְסַבֵּר: ׳מִדּוֹ׳ – כֻּתֹּנֶת שֶׁמַּתְאִימָה בְּדִיּוּק לַמִּדָּה שֶׁלּוֹ.

הַיּוֹם בַּמִּקְדָּשׁ מַתְחִיל בְּנִקָּיוֹן.

👕

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְכַבֵּד אֶת ה׳ גַּם בַּבְּגָדִים שֶׁלִּי 💛

📖 וַיִּקְרָא, פֶּרֶק ו׳, פָּסוּק ד׳

רָמָה א׳ – כָּל הַפָּרָשָׁה

👕 בְּגָדִים אֲחֵרִים

כְּשֶׁיֵּשׁ הַרְבֵּה אֵפֶר, הַכֹּהֵן מוֹצִיא אוֹתוֹ אֶל מִחוּץ לַמַּחֲנֶה. ׳וּפָשַׁט אֶת בְּגָדָיו וְלָבַשׁ בְּגָדִים אֲחֵרִים׳. רַשִׁ״י מְסַבֵּר: זֶה ׳דֶּרֶךְ אֶרֶץ׳ – בְּגָדִים שֶׁמִּתְלַכְלְכִים בַּעֲבוֹדָה מְלַכְלֶכֶת, לֹא לוֹבְשִׁים כְּשֶׁמְּשָׁרְתִים אֶת הַמֶּלֶךְ.

לְכָל עֲבוֹדָה – הַבְּגָדִים שֶׁמַּתְאִימִים לָהּ.

🔥

💛 אַהֲבַת הַשֵּׁם: הָאַהֲבָה שֶׁלִּי לַה׳ דּוֹלֶקֶת תָּמִיד 💛

📖 וַיִּקְרָא, פֶּרֶק ו׳, פְּסוּקִים ה׳–ו׳

רָמָה א׳ – כָּל הַפָּרָשָׁה

🔥 ׳אֵשׁ תָּמִיד׳

׳וּבִעֵר עָלֶיהָ הַכֹּהֵן עֵצִים בַּבֹּקֶר בַּבֹּקֶר׳ – כָּל בֹּקֶר מוֹסִיפִים עֵצִים. ׳אֵשׁ תָּמִיד תּוּקַד עַל הַמִּזְבֵּחַ לֹא תִכְבֶה׳ – הָאֵשׁ לֹא כָּבְתָה אַף פַּעַם, לֹא בַּיּוֹם וְלֹא בַּלַּיְלָה. רַשִׁ״י מְסַבֵּר: מֵהָאֵשׁ הַזֹּאת הָיוּ מַדְלִיקִים גַּם אֶת נֵרוֹת הַמְּנוֹרָה.

אֵשׁ שֶׁלֹּא כָּבָה – וְכָל יוֹם מוֹסִיפִים לָהּ עֵצִים.

🫓

💛 אַהֲבַת הַשֵּׁם: ה׳ מַזְמִין אֶת הַמְּשָׁרְתִים שֶׁלּוֹ לְשֻׁלְחָנוֹ 💛

📖 וַיִּקְרָא, פֶּרֶק ו׳, פְּסוּקִים ז׳–י״א

רָמָה א׳ – כָּל הַפָּרָשָׁה

🫓 תּוֹרַת הַמִּנְחָה

׳וְזֹאת תּוֹרַת הַמִּנְחָה׳: הַכֹּהֵן לוֹקֵחַ מְלֹא כַּף יָד מֵהַסֹּלֶת וְהַשֶּׁמֶן, עִם כָּל הַלְּבוֹנָה, וּמַקְטִיר – ׳רֵיחַ נִיחֹחַ׳. אֶת הַשְּׁאָר אוֹכְלִים הַכֹּהֲנִים ׳מַצּוֹת... בַּחֲצַר אֹהֶל מוֹעֵד׳. ׳לֹא תֵאָפֶה חָמֵץ׳ – רַק מַצָּה.

הַכֹּהֲנִים אוֹכְלִים בְּמָקוֹם קָדוֹשׁ, כְּמוֹ אוֹרְחִים אֵצֶל ה׳.

🌅

💛 אַהֲבַת הַשֵּׁם: בַּבֹּקֶר וּבָעֶרֶב אֲנִי זוֹכֵר אֶת ה׳ 💛

📖 וַיִּקְרָא, פֶּרֶק ו׳, פְּסוּקִים י״ב–ט״ז

רָמָה א׳ – כָּל הַפָּרָשָׁה

🌅 מִנְחָה בַּבֹּקֶר וּבָעֶרֶב

הַכֹּהֵן הַגָּדוֹל מֵבִיא מִנְחָה מִשֶּׁלּוֹ ׳מִנְחָה תָּמִיד מַחֲצִיתָהּ בַּבֹּקֶר וּמַחֲצִיתָהּ בָּעָרֶב׳. אוֹפִים אוֹתָהּ בְּשֶׁמֶן עַל מַחֲבַת, וְהִיא ׳כָּלִיל תָּקְטָר׳ – כֻּלָּהּ עוֹלָה לַמִּזְבֵּחַ. רַשִׁ״י מְסַבֵּר: כֹּהֵן רָגִיל מֵבִיא אוֹתָהּ פַּעַם אַחַת, בַּיּוֹם הָרִאשׁוֹן שֶׁלּוֹ – וְהַכֹּהֵן הַגָּדוֹל בְּכָל יוֹם.

פּוֹתְחִים וְסוֹגְרִים אֶת הַיּוֹם עִם ה׳.

🙇

💛 אַהֲבַת הַשֵּׁם: ה׳ שָׂמַח שֶׁיֵּשׁ מִי שֶׁעוֹזֵר לִי לָשׁוּב 💛

📖 וַיִּקְרָא, פֶּרֶק ו׳, פָּסוּק י״ח – פֶּרֶק ז׳, פָּסוּק י׳

רָמָה א׳ – כָּל הַפָּרָשָׁה

🙇 הַחַטָּאת וְהָאָשָׁם

גַּם לְחַטָּאת וּלְאָשָׁם – הַקָּרְבָּנוֹת שֶׁל מִי שֶׁטָּעָה – יֵשׁ ׳תּוֹרָה׳. הֵם ׳קֹדֶשׁ קָדָשִׁים׳, וְהַכֹּהֲנִים אוֹכְלִים מֵהֶם ׳בְּמָקוֹם קָדוֹשׁ׳. הַכֹּהֵן שֶׁעָזַר לָאִישׁ לְכַפֵּר – הוּא מְקַבֵּל: ׳הַכֹּהֵן אֲשֶׁר יְכַפֶּר בּוֹ לוֹ יִהְיֶה׳.

הַכֹּהֵן עוֹזֵר לָאָדָם לְתַקֵּן – וְזוֹכֶה בְּחֵלֶק.

🎉

💛 אַהֲבַת הַשֵּׁם: ה׳ הִצִּיל אוֹתִי, וַאֲנִי אוֹמֵר לוֹ תּוֹדָה 💛

📖 וַיִּקְרָא, פֶּרֶק ז׳, פְּסוּקִים י״א–י״ד

רָמָה א׳ – כָּל הַפָּרָשָׁה

🎉 קָרְבַּן תּוֹדָה

׳אִם עַל תּוֹדָה יַקְרִיבֶנּוּ׳ – מִי שֶׁקָּרָה לוֹ נֵס מֵבִיא קָרְבַּן תּוֹדָה. רַשִׁ״י מְסַבֵּר: אַרְבָּעָה צְרִיכִים לְהוֹדוֹת – ׳יוֹרְדֵי הַיָּם וְהוֹלְכֵי מִדְבָּרוֹת וַחֲבוּשֵׁי בֵּית הָאֲסוּרִים וְחוֹלֶה שֶׁנִּתְרַפֵּא׳. עִם הַקָּרְבָּן מְבִיאִים אַרְבָּעָה סוּגֵי לֶחֶם – מַצּוֹת וְגַם ׳לֶחֶם חָמֵץ׳.

עַל נֵס – לֹא שׁוֹתְקִים, מוֹדִים לַה׳.

📖 וַיִּקְרָא, פֶּרֶק ז׳, פְּסוּקִים ט״ו–ט״ז

רָמָה א׳ – כָּל הַפָּרָשָׁה

🌙 ׳בְּיוֹם קָרְבָּנוֹ יֵאָכֵל׳

אֶת קָרְבַּן הַתּוֹדָה אוֹכְלִים ׳בְּיוֹם קָרְבָּנוֹ... לֹא יַנִּיחַ מִמֶּנּוּ עַד בֹּקֶר׳ – רַק יוֹם וְלַיְלָה אֶחָד. וְיֵשׁ הַרְבֵּה לֶחֶם! לָכֵן מִי שֶׁמֵּבִיא תּוֹדָה מַזְמִין מִשְׁפָּחָה וַחֲבֵרִים, וְכֻלָּם שׁוֹמְעִים עַל הַנֵּס.

תּוֹדָה גְּדוֹלָה הִיא תּוֹדָה שֶׁכֻּלָּם שׁוֹמְעִים.

🎁

🎁 חֵלֶק לַכֹּהֲנִים

מִכָּל קָרְבַּן שְׁלָמִים, חֵלֶק עוֹלֶה לַמִּזְבֵּחַ וְחֵלֶק נִתָּן לַכֹּהֲנִים: ׳וָאֶתֵּן אֹתָם לְאַהֲרֹן הַכֹּהֵן וּלְבָנָיו לְחָק עוֹלָם׳. הַכֹּהֲנִים עוֹבְדִים בַּמִּקְדָּשׁ בִּשְׁבִיל כֻּלָּם, וְכָל עַם יִשְׂרָאֵל דּוֹאֵג לָהֶם.

מִי שֶׁעוֹבֵד בִּשְׁבִיל כֻּלָּם – כֻּלָּם דּוֹאֲגִים לוֹ.

📜

💛 אַהֲבַת הַשֵּׁם: ה׳ שָׂמֵחַ כְּשֶׁאֲנִי לוֹמֵד אֶת הַתּוֹרָה שֶׁלּוֹ 💛

📖 וַיִּקְרָא, פֶּרֶק ז׳, פְּסוּקִים ל״ז–ל״ח

📜 ׳זֹאת הַתּוֹרָה׳

הַתּוֹרָה מְסַכֶּמֶת: ׳זֹאת הַתּוֹרָה לָעֹלָה לַמִּנְחָה וְלַחַטָּאת וְלָאָשָׁם וְלַמִּלּוּאִים וּלְזֶבַח הַשְּׁלָמִים׳ – כָּל זֶה ה׳ צִוָּה לְמֹשֶׁה ׳בְּהַר סִינָי׳. רַבֵּנוּ בַּחְיֵי מֵבִיא מֵחֲזַ״ל: מִי שֶׁלּוֹמֵד אֶת פָּרָשַׁת הָעוֹלָה – כְּאִלּוּ הִקְרִיב עוֹלָה.

גַּם הַיּוֹם – הַלִּמּוּד עַל הַקָּרְבָּנוֹת הוּא כְּמוֹ קָרְבָּן.

👥 כָּל הָעֵדָה בְּפֶתַח אֹהֶל מוֹעֵד

עַכְשָׁו מַתְחִילִים ׳יְמֵי הַמִּלּוּאִים׳ – הַיָּמִים שֶׁבָּהֶם אַהֲרֹן וּבָנָיו נַעֲשִׂים כֹּהֲנִים. ה׳ אוֹמֵר: ׳וְאֵת כָּל הָעֵדָה הַקְהֵל אֶל פֶּתַח אֹהֶל מוֹעֵד׳. רַשִׁ״י מְסַבֵּר: זֶה אֶחָד הַמְּקוֹמוֹת ׳שֶׁהֶחֱזִיק מֻעָט אֶת הַמְרֻבֶּה׳ – מָקוֹם קָטָן, וְכֻלָּם נִכְנְסוּ!

כָּל הָעָם רָאָה אֶת אַהֲרֹן מִתְמַנֶּה לְכֹהֵן.

👘 מֹשֶׁה מַלְבִּישׁ אֶת אַהֲרֹן

׳וַיַּקְרֵב מֹשֶׁה אֶת אַהֲרֹן וְאֶת בָּנָיו וַיִּרְחַץ אֹתָם בַּמָּיִם׳. אַחַר כָּךְ מֹשֶׁה מַלְבִּישׁ אֶת אַהֲרֹן: כֻּתֹּנֶת, אַבְנֵט, מְעִיל, אֵפוֹד, חֹשֶׁן, מִצְנֶפֶת – וְצִיץ הַזָּהָב עַל הַמֵּצַח, ׳נֵזֶר הַקֹּדֶשׁ׳.

מֹשֶׁה שָׂמֵחַ לְהַלְבִּישׁ אֶת אָחִיו בִּגְדֵי כְּבוֹד.

🫒 שֶׁמֶן הַמִּשְׁחָה

׳וַיִּקַּח מֹשֶׁה אֶת שֶׁמֶן הַמִּשְׁחָה וַיִּמְשַׁח אֶת הַמִּשְׁכָּן וְאֶת כָּל אֲשֶׁר בּוֹ וַיְקַדֵּשׁ אֹתָם׳. אַחַר כָּךְ ׳וַיִּצֹק מִשֶּׁמֶן הַמִּשְׁחָה עַל רֹאשׁ אַהֲרֹן׳. הַשֶּׁמֶן מְסַמֵּן: מֵעַכְשָׁו זֶה קָדוֹשׁ לַה׳.

הַמִּשְׁחָה הוֹפֶכֶת דָּבָר רָגִיל לְקָדוֹשׁ.

⛺ שִׁבְעָה יָמִים בְּפֶתַח אֹהֶל מוֹעֵד

שִׁבְעָה יָמִים מֹשֶׁה מְלַמֵּד אֶת אַהֲרֹן וּבָנָיו אֶת הָעֲבוֹדָה, וְהֵם אוֹכְלִים אֶת לֶחֶם הַמִּלּוּאִים. ׳וּפֶתַח אֹהֶל מוֹעֵד תֵּשְׁבוּ יוֹמָם וָלַיְלָה שִׁבְעַת יָמִים וּשְׁמַרְתֶּם אֶת מִשְׁמֶרֶת ה׳׳ – הֵם לֹא זָזִים מִשָּׁם.

לִפְנֵי תַּפְקִיד גָּדוֹל – מִתְכּוֹנְנִים בְּסַבְלָנוּת.

✅ ׳וַיַּעַשׂ אַהֲרֹן וּבָנָיו׳

הַפָּרָשָׁה נִגְמֶרֶת: ׳וַיַּעַשׂ אַהֲרֹן וּבָנָיו אֵת כָּל הַדְּבָרִים אֲשֶׁר צִוָּה ה׳ בְּיַד מֹשֶׁה׳. רַשִׁ״י מְסַבֵּר: זֶה שֶׁבַח גָּדוֹל – ׳שֶׁלֹּא הִטּוּ יָמִין וּשְׂמֹאל׳. הֵם עָשׂוּ בְּדִיּוּק כְּמוֹ שֶׁה׳ אָמַר, בְּלִי לְהוֹסִיף וּבְלִי לְהַחְסִיר.

לַעֲשׂוֹת בְּדִיּוּק מָה שֶׁה׳ אָמַר – זֶה שֶׁבַח.

מָה לוֹקְחִים מִרָמָה א׳?

בְּפָרָשַׁת צַו הַכֹּהֲנִים עוֹבְדִים בַּמִּקְדָּשׁ בְּכָל בֹּקֶר וּבְכָל עֶרֶב, וְהָאֵשׁ עַל הַמִּזְבֵּחַ לֹא כָּבָה אַף פַּעַם. מִי שֶׁה׳ עָשָׂה לוֹ נֵס – מֵבִיא תּוֹדָה. וְאַהֲרֹן וּבָנָיו מִתְכּוֹנְנִים שִׁבְעָה יָמִים, וְעוֹשִׂים הַכֹּל בְּדִיּוּק כְּמוֹ שֶׁה׳ צִוָּה.

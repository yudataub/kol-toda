אָדָם כִּי יַקְרִיב מִכֶּם

רָמָה א׳ – כָּל הַפָּרָשָׁה

כָּל פָּרָשַׁת וַיִּקְרָא: ה׳ קוֹרֵא לְמֹשֶׁה בְּאַהֲבָה, וּמְלַמֵּד אֵיךְ מִתְקָרְבִים – עוֹלָה, מִנְחָה, שְׁלָמִים, וּמָה עוֹשִׂים כְּשֶׁטּוֹעִים

📖 פְּרָקִים א׳–ה׳

🎯 לַחִידוֹן
📣

💛 אַהֲבַת הַשֵּׁם: ה׳ קוֹרֵא גַּם לִי בְּאַהֲבָה, לְהִתְקָרֵב אֵלָיו 💛

📖 וַיִּקְרָא, פֶּרֶק א׳, פָּסוּק א׳

רָמָה א׳ – כָּל הַפָּרָשָׁה

📣 ׳וַיִּקְרָא אֶל מֹשֶׁה׳

הַמִּשְׁכָּן עוֹמֵד, וְהָעָנָן שׁוֹרֶה עָלָיו. מִתּוֹךְ אֹהֶל מוֹעֵד ה׳ קוֹרֵא לְמֹשֶׁה: ׳וַיִּקְרָא אֶל מֹשֶׁה׳. רַשִׁ״י מְסַבֵּר: קְרִיאָה הִיא ׳לְשׁוֹן חִבָּה׳ – כְּמוֹ שֶׁקּוֹרְאִים לְמִישֶׁהוּ אָהוּב בִּשְׁמוֹ. וְרַק מֹשֶׁה שׁוֹמֵעַ אֶת הַקּוֹל.

סֵפֶר וַיִּקְרָא נִפְתָּח בְּמִלָּה שֶׁל אַהֲבָה.

1
🤝

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁאֶהְיֶה קָרוֹב אֵלָיו 💛

📖 וַיִּקְרָא, פֶּרֶק א׳, פְּסוּקִים א׳–ב׳

רָמָה א׳ – כָּל הַפָּרָשָׁה

🤝 ׳אָדָם כִּי יַקְרִיב מִכֶּם׳

ה׳ מְלַמֵּד: ׳אָדָם כִּי יַקְרִיב מִכֶּם קָרְבָּן לַה׳׳. הַמִּלָּה ׳קָרְבָּן׳ בָּאָה מִ׳קָרוֹב׳ – הַקָּרְבָּן מְקָרֵב אֶת הָאָדָם אֶל ה׳. רַשִׁ״י מְסַבֵּר: לָמָּה כָּתוּב ׳אָדָם׳? כְּמוֹ אָדָם הָרִאשׁוֹן, שֶׁהַכֹּל הָיָה שֶׁלּוֹ – גַּם אַתֶּם לֹא תָּבִיאוּ קָרְבָּן מִגָּזֵל.

קָרְבָּן = לְהִתְקָרֵב אֶל ה׳, רַק מִמָּה שֶׁשַּׁיָּךְ לִי בְּיֹשֶׁר.

2
🔥

💛 אַהֲבַת הַשֵּׁם: אֲנִי יָכוֹל לַעֲשׂוֹת נַחַת רוּחַ לַה׳ 💛

📖 וַיִּקְרָא, פֶּרֶק א׳, פְּסוּקִים ג׳–ט׳

רָמָה א׳ – כָּל הַפָּרָשָׁה

🔥 הָעוֹלָה – הַכֹּל עוֹלֶה לַה׳

יֵשׁ קָרְבָּן שֶׁנִּקְרָא ׳עוֹלָה׳ – כִּי כֻּלּוֹ עוֹלֶה עַל הַמִּזְבֵּחַ. הַכֹּהֲנִים מְסַדְּרִים עֵצִים עַל הָאֵשׁ, וְהָעָשָׁן עוֹלֶה לַשָּׁמַיִם: ׳עֹלָה אִשֵּׁה רֵיחַ נִיחוֹחַ לַה׳׳. רַשִׁ״י מְסַבֵּר: ׳נַחַת רוּחַ לְפָנַי, שֶׁאָמַרְתִּי וְנַעֲשָׂה רְצוֹנִי׳.

כְּשֶׁעוֹשִׂים מָה שֶׁה׳ בִּקֵּשׁ – זֶה ׳רֵיחַ נִיחוֹחַ׳.

3
🕊️

💛 אַהֲבַת הַשֵּׁם: ה׳ שָׂמֵחַ גַּם בַּמַּתָּנָה הַקְּטַנָּה שֶׁלִּי 💛

📖 וַיִּקְרָא, פֶּרֶק א׳, פְּסוּקִים י״ד–י״ז

רָמָה א׳ – כָּל הַפָּרָשָׁה

🕊️ הָעוֹלָה שֶׁל הֶעָנִי

מִי שֶׁאֵין לוֹ כֶּסֶף לִבְהֵמָה גְּדוֹלָה, מֵבִיא ׳מִן הַתֹּרִים אוֹ מִן בְּנֵי הַיּוֹנָה׳ – צִפּוֹר קְטַנָּה. וְגַם עָלֶיהָ כָּתוּב ׳רֵיחַ נִיחֹחַ לַה׳׳! רַשִׁ״י מְסַבֵּר: הַכֹּהֵן מַקְטִיר אוֹתָהּ עִם הַנּוֹצוֹת, ׳כְּדֵי שֶׁיְּהֵא הַמִּזְבֵּחַ שָׂבֵעַ וּמְהֻדָּר בְּקָרְבָּנוֹ שֶׁל עָנִי׳.

הַקָּרְבָּן הַקָּטָן שֶׁל הֶעָנִי חָשׁוּב לִפְנֵי ה׳ כְּמוֹ הַגָּדוֹל.

4
🌾

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹאֶה כַּמָּה אֲנִי מִשְׁתַּדֵּל, לֹא רַק כַּמָּה אֲנִי נוֹתֵן 💛

📖 וַיִּקְרָא, פֶּרֶק ב׳, פְּסוּקִים א׳–ג׳

רָמָה א׳ – כָּל הַפָּרָשָׁה

🌾 הַמִּנְחָה – סֹלֶת, שֶׁמֶן וּלְבוֹנָה

׳וְנֶפֶשׁ כִּי תַקְרִיב קָרְבַּן מִנְחָה לַה׳ סֹלֶת יִהְיֶה קָרְבָּנוֹ׳. מִנְחָה הִיא קֶמַח דַּק – סֹלֶת, עִם שֶׁמֶן זַיִת, וּלְבוֹנָה שֶׁמְּרִיחָה טוֹב. רַשִׁ״י מְסַבֵּר: מִי מֵבִיא מִנְחָה? הֶעָנִי. אָמַר ה׳: ׳מַעֲלֶה אֲנִי עָלָיו כְּאִלּוּ הִקְרִיב נַפְשׁוֹ׳.

הֶעָנִי נוֹתֵן מְעַט – אֲבָל נוֹתֵן אֶת כָּל הַלֵּב.

5
🍳

💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לִי לֶחֶם, וַאֲנִי מוֹדֶה לוֹ עָלָיו 💛

📖 וַיִּקְרָא, פֶּרֶק ב׳, פְּסוּקִים ד׳–י׳

רָמָה א׳ – כָּל הַפָּרָשָׁה

🍳 בַּתַּנּוּר, בַּמַּחֲבַת וּבַמַּרְחֶשֶׁת

יֵשׁ מְנָחוֹת שֶׁאוֹפִים: ׳מַאֲפֵה תַנּוּר׳, ׳עַל הַמַּחֲבַת׳ וְ׳מִנְחַת מַרְחֶשֶׁת׳. רַשִׁ״י מְסַבֵּר: מַרְחֶשֶׁת הִיא כְּלִי עָמֹק, מָלֵא שֶׁמֶן, וְהַבָּצֵק בְּתוֹכוֹ ׳רוֹחֵשׁ׳ – זָז וּמְבַעְבֵּעַ בַּשֶּׁמֶן. הַכֹּהֵן לוֹקֵחַ מְעַט וּמַקְטִיר, וְאֶת הַשְּׁאָר אוֹכְלִים הַכֹּהֲנִים.

גַּם לֶחֶם וּמַאֲפֶה יְכוֹלִים לִהְיוֹת מַתָּנָה לַה׳.

6
🧂

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹמֵר אֶת הַהַבְטָחוֹת שֶׁלּוֹ – לְכָל דָּבָר בָּעוֹלָם 💛

📖 וַיִּקְרָא, פֶּרֶק ב׳, פְּסוּקִים י״א–י״ג

רָמָה א׳ – כָּל הַפָּרָשָׁה

🧂 ׳מֶלַח בְּרִית׳

בְּמִנְחָה לֹא שָׂמִים שְׂאֹר וּדְבַשׁ, אֲבָל ׳עַל כָּל קָרְבָּנְךָ תַּקְרִיב מֶלַח׳. רַשִׁ״י מְסַבֵּר: מִשֵּׁשֶׁת יְמֵי בְּרֵאשִׁית ה׳ הִבְטִיחַ לַמַּיִם הַתַּחְתּוֹנִים, שֶׁהֵם יִקְרְבוּ עַל הַמִּזְבֵּחַ – בַּמֶּלַח שֶׁבָּא מֵהַיָּם, וּבְנִסּוּךְ הַמַּיִם בְּחַג הַסֻּכּוֹת.

גַּם הַמֶּלַח הַפָּשׁוּט מִן הַיָּם זוֹכֶה לַעֲלוֹת לַמִּזְבֵּחַ.

7
🌱

💛 אַהֲבַת הַשֵּׁם: ה׳ מְגַדֵּל לִי אֶת הַתְּבוּאָה, וְהָרִאשׁוֹן שַׁיָּךְ לוֹ 💛

📖 וַיִּקְרָא, פֶּרֶק ב׳, פְּסוּקִים י״ד–ט״ז

רָמָה א׳ – כָּל הַפָּרָשָׁה

🌱 מִנְחַת הַבִּכּוּרִים

׳וְאִם תַּקְרִיב מִנְחַת בִּכּוּרִים לַה׳ אָבִיב קָלוּי בָּאֵשׁ׳. רַשִׁ״י מְסַבֵּר: זוֹ מִנְחַת הָעֹמֶר, מִשְּׂעוֹרִים חֲדָשׁוֹת, בָּאָבִיב – כְּשֶׁהַתְּבוּאָה מַבְשִׁילָה. קוֹלִים אוֹתָהּ עַל הָאֵשׁ, כִּי הִיא עֲדַיִן לַחָה, וְאַחַר כָּךְ טוֹחֲנִים.

אֶת הָרִאשׁוֹן – קֹדֶם מְבִיאִים לַה׳.

8
🕊

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב שָׁלוֹם, וְשָׂמֵחַ כְּשֶׁאֲנִי מְחַלֵּק עִם אֲחֵרִים 💛

📖 וַיִּקְרָא, פֶּרֶק ג׳, פְּסוּקִים א׳–ה׳

רָמָה א׳ – כָּל הַפָּרָשָׁה

🕊 שְׁלָמִים – קָרְבַּן שָׁלוֹם

׳וְאִם זֶבַח שְׁלָמִים קָרְבָּנוֹ׳. אֶת הַשְּׁלָמִים מְבִיאִים בְּשִׂמְחָה, וְאוֹכְלִים מֵהֶם בִּסְעוּדָה. רַשִׁ״י מְסַבֵּר: הֵם נִקְרָאִים שְׁלָמִים ׳שֶׁמְּטִילִים שָׁלוֹם בָּעוֹלָם׳, וְגַם כִּי יֵשׁ בָּהֶם חֵלֶק לְכֻלָּם – לַמִּזְבֵּחַ, לַכֹּהֲנִים וְלַבְּעָלִים.

כְּשֶׁכֻּלָּם מְקַבְּלִים חֵלֶק – יֵשׁ שָׁלוֹם.

9
🙇

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי דֶּרֶךְ לְתַקֵּן גַּם כְּשֶׁאֲנִי טוֹעֶה 💛

📖 וַיִּקְרָא, פֶּרֶק ד׳, פְּסוּקִים א׳–כ״א

רָמָה א׳ – כָּל הַפָּרָשָׁה

🙇 מִי שֶׁטָּעָה – בִּשְׁגָגָה

׳נֶפֶשׁ כִּי תֶחֱטָא בִשְׁגָגָה׳ – בְּלִי כַּוָּנָה, בְּטָעוּת. גַּם הַכֹּהֵן הַגָּדוֹל, גַּם כָּל הָעֵדָה, גַּם הַנָּשִׂיא וְגַם אִישׁ פָּשׁוּט יְכוֹלִים לִטְעוֹת. מִי שֶׁטָּעָה מֵבִיא קָרְבַּן חַטָּאת, וְהַכֹּהֵן מְכַפֵּר עָלָיו.

כֻּלָּם יְכוֹלִים לִטְעוֹת – וְכֻלָּם יְכוֹלִים לְתַקֵּן.

10
👑

💛 אַהֲבַת הַשֵּׁם: ה׳ מוֹחֵל לְמִי שֶׁמּוֹדֶה בְּטָעוּתוֹ 💛

📖 וַיִּקְרָא, פֶּרֶק ד׳, פְּסוּקִים כ״ב–כ״ו

רָמָה א׳ – כָּל הַפָּרָשָׁה

👑 ׳אֲשֶׁר נָשִׂיא יֶחֱטָא׳

גַּם הַנָּשִׂיא – הַמַּנְהִיג שֶׁל הַשֵּׁבֶט – מֵבִיא קָרְבָּן כְּשֶׁהוּא טוֹעֶה. וְהַתּוֹרָה אוֹמֶרֶת: ׳וְכִפֶּר עָלָיו הַכֹּהֵן מֵחַטָּאתוֹ וְנִסְלַח לוֹ׳. רַשִׁ״י מְסַבֵּר: ׳אֲשֶׁר׳ – לְשׁוֹן ׳אַשְׁרֵי׳: אַשְׁרֵי הַדּוֹר שֶׁהַנָּשִׂיא שֶׁלּוֹ לֹא מִתְבַּיֵּשׁ לְהוֹדוֹת שֶׁטָּעָה!

מַנְהִיג גָּדוֹל אֵינוֹ מִתְבַּיֵּשׁ לוֹמַר ׳טָעִיתִי׳.

יֵשׁ עוֹד דְּבָרִים שֶׁמְּבִיאִים עֲלֵיהֶם קָרְבָּן – לְמָשָׁל עֵד שֶׁיּוֹדֵעַ מָה קָרָה וְלֹא סִפֵּר. לִפְנֵי הַקָּרְבָּן צָרִיךְ לְהַגִּיד בַּפֶּה מָה עָשִׂיתִי: ׳וְהִתְוַדָּה אֲשֶׁר חָטָא עָלֶיהָ׳. זֶה הַוִּדּוּי – לוֹמַר בְּכֵנוּת: טָעִיתִי, וַאֲנִי מִצְטַעֵר.

תְּשׁוּבָה מַתְחִילָה בְּמִלָּה כֵּנָה: ׳טָעִיתִי׳.

💛 אַהֲבַת הַשֵּׁם: ה׳ פּוֹתֵחַ דֶּלֶת לְכֻלָּם – גַּם לֶעָנִי 💛

📖 וַיִּקְרָא, פֶּרֶק ה׳, פְּסוּקִים ז׳–י״ג

רָמָה א׳ – כָּל הַפָּרָשָׁה

⚖️ קָרְבָּן לְפִי הַכִּיס

מִי שֶׁאֵין לוֹ כֶּבֶשׂ – מֵבִיא שְׁתֵּי יוֹנִים: ׳וְאִם לֹא תַגִּיעַ יָדוֹ דֵּי שֶׂה׳. וּמִי שֶׁאֵין לוֹ גַּם יוֹנִים – מֵבִיא קְצָת סֹלֶת: ׳עֲשִׂירִת הָאֵפָה סֹלֶת׳. וְגַם עָלָיו כָּתוּב ׳וְנִסְלַח לוֹ׳. אֵין אָדָם שֶׁאֵין לוֹ דֶּרֶךְ לְהִתְקָרֵב.

כָּל אֶחָד מֵבִיא לְפִי מָה שֶׁיֵּשׁ לוֹ.

13
🏛️

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְכַבֵּד אֶת מָה שֶׁשַּׁיָּךְ לַה׳ 💛

📖 וַיִּקְרָא, פֶּרֶק ה׳, פְּסוּקִים י״ד–י״ט

רָמָה א׳ – כָּל הַפָּרָשָׁה

🏛️ מִי שֶׁנֶּהֱנָה מֵהֶקְדֵּשׁ

׳נֶפֶשׁ כִּי תִמְעֹל מַעַל וְחָטְאָה בִּשְׁגָגָה מִקָּדְשֵׁי ה׳׳. רַשִׁ״י מְסַבֵּר: זֶה מִי שֶׁנֶּהֱנָה בְּטָעוּת מִדָּבָר שֶׁהֻקְדַּשׁ לַמִּקְדָּשׁ. הוּא מַחֲזִיר אֶת מָה שֶׁלָּקַח, מוֹסִיף עוֹד חֲמִישִׁית, וּמֵבִיא קָרְבַּן אָשָׁם.

מָה שֶׁשַּׁיָּךְ לַה׳ – שׁוֹמְרִים עָלָיו בִּזְהִירוּת.

14
💰

💛 אַהֲבַת הַשֵּׁם: ה׳ מְחַכֶּה לִי, וְסוֹלֵחַ כְּשֶׁאֲנִי מְתַקֵּן 💛

📖 וַיִּקְרָא, פֶּרֶק ה׳, פְּסוּקִים כ׳–כ״ו

רָמָה א׳ – כָּל הַפָּרָשָׁה

💰 מַחֲזִירִים – וּמוֹסִיפִים

מִי שֶׁלָּקַח מֵחֲבֵרוֹ בְּלִי רְשׁוּת, אוֹ שֶׁלֹּא הֶחֱזִיר פִּקָּדוֹן – צָרִיךְ לְתַקֵּן: ׳וְהֵשִׁיב אֶת הַגְּזֵלָה אֲשֶׁר גָּזָל׳, וְעוֹד מוֹסִיף חֲמִישִׁית מִשֶּׁלּוֹ. רַק אַחַר כָּךְ מֵבִיא קָרְבָּן, ׳וְנִסְלַח לוֹ׳. רַשִׁ״י מְסַבֵּר: זֶה כְּשֶׁהוּא מַחְלִיט לַחְזֹר בִּתְשׁוּבָה.

קֹדֶם מְתַקְּנִים אֶת מָה שֶׁפָּגַעְנוּ בַּחֲבֵרִים – וְאָז מִתְקָרְבִים לַה׳.

15
💡

רָמָה א׳ – כָּל הַפָּרָשָׁה

מָה לוֹקְחִים מִרָמָה א׳?

בְּפָרָשַׁת וַיִּקְרָא ה׳ קוֹרֵא לָנוּ בְּאַהֲבָה וּמְלַמֵּד אֵיךְ מִתְקָרְבִים אֵלָיו: הֶעָשִׁיר וְהֶעָנִי, הַנָּשִׂיא וְהָאִישׁ הַפָּשׁוּט – לְכָל אֶחָד יֵשׁ דֶּרֶךְ. וְגַם מִי שֶׁטָּעָה יָכוֹל לְתַקֵּן, ׳וְנִסְלַח לוֹ׳.

16
❤️

🤔 חוֹשְׁבִים עַל זֶה

  • ה׳ קָרָא לְמֹשֶׁה ׳בְּחִבָּה׳. מָתַי אֲנִי מַרְגִּישׁ שֶׁה׳ קָרוֹב אֵלַי?
  • הֶעָנִי הֵבִיא קְצָת סֹלֶת – וְה׳ שָׂמַח כְּאִלּוּ הִקְרִיב נַפְשׁוֹ. מָה אֲנִי יָכוֹל לָתֵת הַיּוֹם מִכָּל הַלֵּב?
  • מִי שֶׁלָּקַח – מַחֲזִיר וּמוֹסִיף. יֵשׁ מַשֶּׁהוּ שֶׁאֲנִי רוֹצֶה לְתַקֵּן עִם חָבֵר?

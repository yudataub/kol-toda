וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ אֲנִי ה׳

📖 חֻמַּשׁ וַיִּקְרָא · פָּרָשַׁת קְדֹשִׁים · רָמָה ב׳

קְדֹשִׁים תִּהְיוּ

רָמָה ב׳ – הַמְּפָרְשִׁים מְסַבְּרִים

כָּל פָּרָשַׁת קְדֹשִׁים: כִּבּוּד הוֹרִים וְשַׁבָּת, פֵּאָה וְלֶקֶט, שְׁמִירַת הַלָּשׁוֹן, ׳וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ׳, ׳מִפְּנֵי שֵׂיבָה תָּקוּם׳ וּמֹאזְנֵי צֶדֶק

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁאֶהְיֶה קָדוֹשׁ בְּכָל רֶגַע 💛

רָמָה ב׳ – הַמְּפָרְשִׁים מְסַבְּרִים

🍽️ קָדוֹשׁ גַּם בַּמֻּתָּר

הָרַמְבַּ״ן שׁוֹאֵל: מָה זֶה ׳קְדֹשִׁים תִּהְיוּ׳? אֶפְשָׁר לִשְׁמֹר אֶת כָּל הָאִסּוּרִים – וְעַדַיִן לֶאֱכֹל בְּלִי גְּבוּל וּלְדַבֵּר מִלִּים מְכֹעָרוֹת, וְלִהְיוֹת ׳נָבָל בִּרְשׁוּת הַתּוֹרָה׳. לָכֵן הַתּוֹרָה מְבַקֶּשֶׁת: לִהְיוֹת קָדוֹשׁ גַּם בַּדְּבָרִים הַמֻּתָּרִים, וְ׳יִשְׁמֹר פִּיו וּלְשׁוֹנוֹ׳. וּמָה אֲנִי לוֹמֵד? קְדֻשָּׁה הִיא גַּם אֵיךְ אֲנִי אוֹכֵל וְאֵיךְ אֲנִי מְדַבֵּר.

קְדֻשָּׁה הִיא גַּם בַּשֻּׁלְחָן וּבַדִּבּוּר.

👶 לָמָּה אִמָּא קוֹדֶמֶת?

בַּפָּסוּק כָּתוּב ׳אִמּוֹ וְאָבִיו׳ – אִמָּא לִפְנֵי אַבָּא. הָאִבְּן עֶזְרָא מְבָאֵר: ׳כִּי הַקָּטָן אֵינֶנּוּ מַכִּיר בַּתְּחִלָּה כִּי אִם אִמּוֹ, וְאַחַר כֵּן אָבִיו, וְאַחַר כֵּן שַׁבָּת׳ – וְאַחַר כָּךְ מַכִּיר אֶת ה׳ שֶׁשָּׁבַת בַּיּוֹם הַשְּׁבִיעִי. וּמָה אֲנִי לוֹמֵד? הַהוֹרִים הֵם הָרִאשׁוֹנִים שֶׁמְּלַמְּדִים אוֹתִי לְהַכִּיר אֶת ה׳.

דֶּרֶךְ הַהוֹרִים אֲנִי מַגִּיעַ לְהַכִּיר אֶת ה׳.

רָמָה ב׳ – הַמְּפָרְשִׁים מְסַבְּרִים

👁️ ה׳ מַכִּיר מַחְשְׁבוֹתֶיךָ

לָמָּה כָּתוּב ׳וְיָרֵאתָ מֵאֱלֹהֶיךָ׳ דַּוְקָא לְיַד ׳לִפְנֵי עִוֵּר׳? רַשִׁ״י מְבָאֵר: מִי שֶׁנָּתַן עֵצָה רָעָה יָכוֹל לוֹמַר ׳לְטוֹבָה נִתְכַּוַּנְתִּי׳, וְאַף אֶחָד לֹא יֵדַע. אֲבָל ה׳ הוּא ׳הַמַּכִּיר מַחְשְׁבוֹתֶיךָ׳. וּמָה אֲנִי לוֹמֵד? גַּם כְּשֶׁאַף אֶחָד לֹא רוֹאֶה – ה׳ רוֹאֶה, וַאֲנִי רוֹצֶה שֶׁהַלֵּב שֶׁלִּי יִהְיֶה נָקִי.

מָה שֶׁבַּלֵּב – ה׳ יוֹדֵעַ.

📜 הַפָּסוּקלֹא תֵלֵךְ רָכִיל

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁכָּל הַגּוּף שֶׁלִּי יְכַבֵּד אֶת הַחֲבֵרִים 💛

רָמָה ב׳ – הַמְּפָרְשִׁים מְסַבְּרִים

😉 לֹא קוֹרְצִים בָּעֵינַיִם

תַּרְגּוּם אֻנְקְלוֹס מְתַרְגֵּם ׳לֹא תֵלֵךְ רָכִיל׳: ׳לָא תֵיכוּל קוּרְצִין׳. וְרַשִׁ״י מְבָאֵר אֶת הַמִּלָּה: מְסַפְּרֵי רְכִילוּת דַּרְכָּם ׳לִקְרֹץ בְּעֵינֵיהֶם׳ – רוֹמְזִים בָּעֵינַיִם, כְּדֵי שֶׁאֲחֵרִים לֹא יָבִינוּ. וּמָה אֲנִי לוֹמֵד? לָשׁוֹן הָרַע אֶפְשָׁר לְסַפֵּר גַּם בְּלִי מִלִּים – בִּקְרִיצָה וּבִרְמִיזָה – וְגַם מִזֶּה נִזְהָרִים.

שְׁמִירַת הַלָּשׁוֹן הִיא גַּם שְׁמִירַת הָעֵינַיִם וְהַיָּדַיִם.

🗣️ מַדּוּעַ כָּכָה עָשִׂיתָ?

הָרַמְבַּ״ן מְבָאֵר אֶת ׳הוֹכֵחַ תּוֹכִיחַ׳: כְּשֶׁחָבֵר עָשָׂה לְךָ דָּבָר שֶׁלֹּא כִּרְצוֹנְךָ – אַל תִּשְׂנָא אוֹתוֹ בַּלֵּב, ׳אֲבָל תּוֹכִיחֶנּוּ: מַדּוּעַ כָּכָה עָשִׂיתָ עִמָּדִי׳. כָּךְ, אוֹמֵר הָרַמְבַּ״ן, הוּא יִתְנַצֵּל, אוֹ יוֹדֶה – וְתִסְלַח לוֹ. וּמָה אֲנִי לוֹמֵד? בִּמְקוֹם לִכְעֹס בְּשֶׁקֶט – שׁוֹאֲלִים וּמְדַבְּרִים.

שִׂיחָה פְּתוּחָה מְרַפֵּאת כַּעַס.

👑 יְהוֹנָתָן וְדָוִד

הָרַמְבַּ״ן מְבָאֵר אֶת ׳וְאָהַבְתָּ לְרֵעֲךָ כָּמוֹךָ׳: לִרְצוֹת שֶׁיִּהְיֶה לֶחָבֵר כָּל טוֹב – עֹשֶׁר, כָּבוֹד וְחָכְמָה – בְּלִי קִנְאָה, וַאֲפִלּוּ יוֹתֵר מִמֶּנִּי. כָּךְ יְהוֹנָתָן אָהַב אֶת דָּוִד ׳כִּי אַהֲבַת נַפְשׁוֹ אֲהֵבוֹ׳, וְאָמַר לוֹ: ׳וְאַתָּה תִּמְלֹךְ עַל יִשְׂרָאֵל׳. וּמָה אֲנִי לוֹמֵד? אַהֲבָה אֲמִתִּית שְׂמֵחָה בַּהַצְלָחָה שֶׁל הֶחָבֵר.

מִי שֶׁאוֹהֵב בֶּאֱמֶת – שָׂמֵחַ כְּשֶׁהֶחָבֵר מַצְלִיחַ.

כֻּלָּנוּ יְלָדִים שֶׁל אוֹתוֹ בּוֹרֵא.

📖 מִי שֶׁחָכָם בַּתּוֹרָה

תַּרְגּוּם אֻנְקְלוֹס מְתַרְגֵּם ׳מִפְּנֵי שֵׂיבָה תָּקוּם׳: ׳מִן קֳדָם דְּסַבַּר בְּאוֹרַיְתָא תְּקוּם׳ – קוּם לִפְנֵי מִי שֶׁמֵּבִין בַּתּוֹרָה. כְּמוֹ רַשִׁ״י: ׳אֵין זָקֵן אֶלָּא שֶׁקָּנָה חָכְמָה׳. וּמָה אֲנִי לוֹמֵד? הַכָּבוֹד הַגָּדוֹל הוּא לַתּוֹרָה, וְכָל מִי שֶׁלּוֹמֵד אוֹתָהּ – רָאוּי לְכָבוֹד.

קָמִים מִפְּנֵי הַתּוֹרָה שֶׁבְּתוֹךְ הָאָדָם.

רָמָה ב׳ – הַמְּפָרְשִׁים מְסַבְּרִים

⏳ שָׁוֶה לְחַכּוֹת

שָׁלֹשׁ שָׁנִים לֹא אוֹכְלִים מֵהָעֵץ, וְשָׁנָה רְבִיעִית מַעֲלִים לִירוּשָׁלַיִם. רַשִׁ״י מֵבִיא אֶת רַבִּי עֲקִיבָא: ׳דִּבְּרָה תוֹרָה כְּנֶגֶד יֵצֶר הָרָע׳ – שֶׁלֹּא יֹאמַר אָדָם ׳אַרְבַּע שָׁנִים אֲנִי מִצְטַעֵר חִנָּם׳. לָכֵן כָּתוּב ׳לְהוֹסִיף לָכֶם תְּבוּאָתוֹ׳ – בִּזְכוּת הַמִּצְוָה ה׳ מְבָרֵךְ אֶת הַפֵּרוֹת. וּמָה אֲנִי לוֹמֵד? מִצְוָה אַף פַּעַם לֹא גּוֹרֶמֶת הֶפְסֵד.

מִי שֶׁמְּחַכֶּה בִּשְׁבִיל מִצְוָה – מַרְוִיחַ יוֹתֵר.

🧂 מִשְׁקָל בְּתוֹךְ הַמֶּלַח

לָמָּה כָּתוּב לְיַד ׳מֹאזְנֵי צֶדֶק׳ – ׳אֲשֶׁר הוֹצֵאתִי אֶתְכֶם מֵאֶרֶץ מִצְרָיִם׳? רַשִׁ״י מְבָאֵר: בְּמִצְרַיִם ה׳ הִבְחִין מִי בְּכוֹר וּמִי לֹא – דָּבָר שֶׁאַף אָדָם לֹא יָכוֹל לָדַעַת. ׳וַאֲנִי הַנֶּאֱמָן לִפָּרַע מִמִּי שֶׁטּוֹמֵן מִשְׁקְלוֹתָיו בַּמֶּלַח׳ – מִי שֶׁמַּחְבִּיא רַמָּאוּת. וּמָה אֲנִי לוֹמֵד? אֶת ה׳ אִי אֶפְשָׁר לְרַמּוֹת, לָכֵן כְּדַאי לִהְיוֹת יָשָׁר תָּמִיד.

ה׳ רוֹאֶה גַּם אֶת מָה שֶׁמֻּסְתָּר.

מָה לוֹקְחִים מִרָמָה ב׳?

הָרַמְבַּ״ן, הָאִבְּן עֶזְרָא, רַשִׁ״י וְאֻנְקְלוֹס מְלַמְּדִים: קְדֻשָּׁה הִיא גַּם בַּמֻּתָּר, אַהֲבָה הִיא לִשְׂמֹחַ בַּטּוֹב שֶׁל הֶחָבֵר, וְה׳ – שֶׁבָּרָא אֶת כֻּלָּנוּ – רוֹאֶה גַּם אֶת מָה שֶׁבַּלֵּב.

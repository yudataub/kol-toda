רָמָה ג׳ – מָה זֶה אוֹמֵר לִי

🎁 שְׁתֵּי מַתָּנוֹת בַּיּוֹם הַחֲמִשִּׁים

רַב בְּאֶרֶץ יִשְׂרָאֵל לִמֵּד תְּשׁוּבָה לַשְּׁאֵלָה ׳מָה עִנְיַן שְׁמִטָּה אֵצֶל הַר סִינַי?׳: בְּהַר סִינַי, בַּיּוֹם הַחֲמִשִּׁים, ה׳ נָתַן לָנוּ שְׁתֵּי מַתָּנוֹת – תּוֹרָה וְאֶרֶץ יִשְׂרָאֵל. וְגַם הַיּוֹבֵל בָּא בִּשְׁנַת הַחֲמִשִּׁים! לָכֵן פָּרָשׁוֹת מִצְווֹת הָאָרֶץ פּוֹתְחוֹת וְנִגְמָרוֹת בְּ׳הַר סִינַי׳. הַיּוֹם, בְּבִרְכַּת הַמָּזוֹן, אֲנִי מְכַוֵּן בַּמִּלִּים ׳עַל הָאָרֶץ וְעַל הַמָּזוֹן׳.

הַתּוֹרָה וְהָאָרֶץ – שְׁתֵּי מַתָּנוֹת שֶׁל אוֹתוֹ יוֹם.

מִי שֶׁיּוֹדֵעַ לְהוֹדוֹת – מְקַבֵּל עוֹד.

חֲזַ״ל אָמְרוּ שֶׁה׳ אָמַר לְיִשְׂרָאֵל: ׳זִרְעוּ שֵׁשׁ וְהַשְׁמִיטוּ שֶׁבַע, כְּדֵי שֶׁתֵּדְעוּ שֶׁהָאָרֶץ שֶׁלִּי הִיא׳. רַב בְּאֶרֶץ יִשְׂרָאֵל הִסְבִּיר: אֶרֶץ יִשְׂרָאֵל הִיא הָאַרְמוֹן שֶׁל מֶלֶךְ הַמְּלָכִים, וַאֲנַחְנוּ אוֹרְחִים בּוֹ. בַּשְּׁמִטָּה הָאֲדָמָה נָחָה, וַאֲנַחְנוּ נִזְכָּרִים שֶׁל מִי הִיא. הַיּוֹם, כְּשֶׁאֲנִי רוֹאֶה עֵץ אוֹ שָׂדֶה, אֲנִי אוֹמֵר בַּלֵּב: ׳תּוֹדָה, ה׳, שֶׁאַתָּה מְאָרֵחַ אוֹתִי׳.

אֲנַחְנוּ אוֹרְחִים בְּאַרְמוֹן הַמֶּלֶךְ.

🔑 מָה חוֹזֵר בַּיּוֹבֵל?

רַב בְּאֶרֶץ יִשְׂרָאֵל שָׁאַל: בַּיּוֹבֵל חוֹזְרִים שְׁנֵי דְּבָרִים – הָעֲבָדִים לְבָתֵּיהֶם וְהַשָּׂדוֹת לַבְּעָלִים. מָה הַקֶּשֶׁר? שְׁנֵיהֶם לֹא הָיוּ בֶּאֱמֶת שֶׁל מִי שֶׁהֶחְזִיק בָּהֶם: הָעֶבֶד שַׁיָּךְ לַה׳, וְגַם הַקַּרְקַע שַׁיֶּכֶת לַה׳. הַיּוֹבֵל מַזְכִּיר שֶׁהַכֹּל פִּקָּדוֹן. הַיּוֹם אֲנִי מַחֲזִיר דָּבָר שֶׁשָּׁאַלְתִּי – בִּזְמַן וּבְמַצָּב טוֹב.

מָה שֶׁיֵּשׁ לִי הוּא פִּקָּדוֹן מֵה׳.

רָמָה ג׳ – מָה זֶה אוֹמֵר לִי

🍊 לֶחֶם שֶׁל אֶרֶץ יִשְׂרָאֵל

רַב בְּאֶרֶץ יִשְׂרָאֵל סִפֵּר עַל צַדִּיק שֶׁגָּר בִּצְפַת, בַּעַל הַסֵּפֶר ׳בַּת עַיִן׳. הֵבִיאוּ לְפָנָיו לֶחֶם מִחוּץ לָאָרֶץ, וְהוּא אָמַר: אֲנִי רוֹצֶה רַק לֶחֶם שֶׁל אֶרֶץ יִשְׂרָאֵל. לָמָּה? הַתּוֹרָה שׁוֹאֶלֶת ׳מַה נֹּאכַל בַּשָּׁנָה הַשְּׁבִיעִת׳ – וְלֹא עוֹנָה ׳תָּבִיאוּ מִבַּחוּץ׳. סִימָן שֶׁהַקְּדֻשָּׁה בְּפֵרוֹת הָאָרֶץ. הַיּוֹם אֲנִי בּוֹחֵר פְּרִי שֶׁגָּדַל בָּאָרֶץ, וּמְבָרֵךְ עָלָיו בְּשִׂמְחָה.

פֵּרוֹת הָאָרֶץ – קְדוֹשִׁים וַחֲבִיבִים.

✏️ הַלְוָאָה לְחָבֵר

רַב בְּאֶרֶץ יִשְׂרָאֵל לִמֵּד: כְּשֶׁחָבֵר מְבַקֵּשׁ הַלְוָאָה, יֵשׁ לְפָנַי לֹא רַק אִסּוּר – לֹא לָקַחַת רִבִּית – אֶלָּא גַּם מִצְוָה: לַעֲזֹר לוֹ בְּחִנָּם, כְּדֵי שֶׁיּוּכַל לַעֲמֹד עַל רַגְלָיו. הַשְּׁאֵלָה הִיא לֹא רַק ׳מָה מֻתָּר׳, אֶלָּא ׳מָה הַתּוֹרָה רוֹצָה מִמֶּנִּי עַכְשָׁו׳. הַיּוֹם, אִם חָבֵר צָרִיךְ עִפָּרוֹן אוֹ מַחְבֶּרֶת – אֲנִי נוֹתֵן מִיָּד, בְּלֵב שָׂמֵחַ וּבְלִי לְבַקֵּשׁ כְּלוּם.

הַלְוָאָה הִיא חֶסֶד, לֹא עֵסֶק.

💡 מָה פָּרָשַׁת בְּהַר מְלַמֶּדֶת אוֹתִי הַיּוֹם

הָאָרֶץ שֶׁלָּנוּ הִיא מַתָּנָה מֵה׳, וַאֲנַחְנוּ אוֹרְחִים בְּבֵיתוֹ. לָכֵן אֲנִי מוֹדֶה עָלֶיהָ, סוֹמֵךְ עַל ה׳ כְּמוֹ בַּשְּׁמִטָּה, שׁוֹמֵר עַל הַמִּלִּים שֶׁלִּי שֶׁלֹּא יִפְגְּעוּ, עוֹזֵר לְחָבֵר לִפְנֵי שֶׁהוּא נוֹפֵל, וּמְכַבֵּד כָּל יְהוּדִי. ׳כִּי לִי הָאָרֶץ׳.

הַכֹּל שֶׁל ה׳ – וַאֲנִי שָׂמֵחַ לְהִשְׁתַּמֵּשׁ בַּמַּתָּנוֹת שֶׁלּוֹ כְּמוֹ שֶׁהוּא רוֹצֶה.

מָה לוֹקְחִים מִרָמָה ג׳?

אֲנִי לוֹמֵד לִרְאוֹת בָּאָרֶץ מַתָּנָה מֵה׳, לְהוֹדוֹת עָלֶיהָ, לְהַלְווֹת לְחָבֵר בְּלֵב טוֹב וּלְכַבֵּד כָּל יְהוּדִי. כָּךְ אֲנִי גָּר בָּאָרֶץ – כְּמוֹ אוֹרֵחַ אָהוּב בְּבֵית הַמֶּלֶךְ.

  • בַּשְּׁמִטָּה כֻּלָּם שָׁוִים לְיַד הָעֵצִים. מָה אֲנִי יָכוֹל לַחְלֹק הַיּוֹם עִם אֲחֵרִים?
  • מָתַי אָמַרְתִּי מִלָּה שֶׁפָּגְעָה בְּחָבֵר? אֵיךְ אֲתַקֵּן אוֹתָהּ?
  • מָה ה׳ נָתַן לִי בְּמַתָּנָה, וְעוֹד לֹא אָמַרְתִּי עָלָיו תּוֹדָה?

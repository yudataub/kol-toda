בְּסוֹף הָעֲבוֹדָה אַהֲרֹן פּוֹשֵׁט אֶת בִּגְדֵי הַלָּבָן וּמַנִּיחַ אוֹתָם בַּמִּשְׁכָּן. הוּא רוֹחֵץ שׁוּב, ׳וְלָבַשׁ אֶת בְּגָדָיו וְיָצָא׳ – בִּגְדֵי הַזָּהָב – וּמַקְטִיר עוֹלָה עַל הַמִּזְבֵּחַ, בִּשְׁבִילוֹ וּבִשְׁבִיל כָּל הָעָם.

אַחֲרֵי הַכַּפָּרָה יוֹצְאִים מְחֻדָּשִׁים, מוּכָנִים לְהַמְשִׁיךְ.

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי בְּכָל שָׁנָה הִזְדַּמְּנוּת לִהְיוֹת טָהוֹר 💛

📖 וַיִּקְרָא, פֶּרֶק ט״ז, פְּסוּקִים כ״ט–ל״ד

רָמָה א׳ – כָּל הַפָּרָשָׁה

📅 לִפְנֵי ה׳ תִּטְהָרוּ

הַתּוֹרָה קוֹבַעַת: בַּחֹדֶשׁ הַשְּׁבִיעִי, בָּעֲשִׂירִי בַּחֹדֶשׁ – הוּא יוֹם הַכִּפּוּרִים – ׳תְּעַנּוּ אֶת נַפְשֹׁתֵיכֶם׳: צָמִים וְלֹא עוֹשִׂים מְלָאכָה. ׳כִּי בַיּוֹם הַזֶּה יְכַפֵּר עֲלֵיכֶם לְטַהֵר אֶתְכֶם... לִפְנֵי ה׳ תִּטְהָרוּ׳. כָּךְ בְּכָל שָׁנָה, ׳אַחַת בַּשָּׁנָה׳.

יוֹם הַכִּפּוּרִים הוּא מַתָּנָה: יוֹם שֶׁל טָהֳרָה לְכָל אֶחָד.

🧭

💛 אַהֲבַת הַשֵּׁם: אֲנִי גֵּאֶה לִהְיוֹת שַׁיָּךְ לַה׳ 💛

📖 וַיִּקְרָא, פֶּרֶק י״ח, פְּסוּקִים א׳–ד׳

רָמָה א׳ – כָּל הַפָּרָשָׁה

🧭 לֹא כְּמוֹ מִצְרַיִם

אַחַר כָּךְ ה׳ אוֹמֵר: ׳אֲנִי ה׳ אֱלֹהֵיכֶם. כְּמַעֲשֵׂה אֶרֶץ מִצְרַיִם... לֹא תַעֲשׂוּ׳. אֲנַחְנוּ לֹא הוֹלְכִים אַחֲרֵי מִנְהָגִים רָעִים שֶׁל עַמִּים אֲחֵרִים. רַשִׁ״י מְסַבֵּר: יֵשׁ ׳מִשְׁפָּטִים׳ – מִצְווֹת שֶׁהַשֵּׂכֶל מֵבִין, וְיֵשׁ ׳חֻקִּים׳ – ׳גְּזֵרַת הַמֶּלֶךְ׳, כְּמוֹ לֹא לֶאֱכֹל חֲזִיר. אֶת שְׁנֵיהֶם שׁוֹמְרִים.

יְהוּדִי הוֹלֵךְ בַּדֶּרֶךְ שֶׁל ה׳, גַּם כְּשֶׁאֲחֵרִים עוֹשִׂים אַחֶרֶת.

🌱

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי מִצְווֹת כְּדֵי שֶׁאֶחְיֶה חַיִּים טוֹבִים 💛

📖 וַיִּקְרָא, פֶּרֶק י״ח, פָּסוּק ה׳

רָמָה א׳ – כָּל הַפָּרָשָׁה

🌱 וָחַי בָּהֶם

ה׳ מַבְטִיחַ: ׳וּשְׁמַרְתֶּם אֶת חֻקֹּתַי וְאֶת מִשְׁפָּטַי אֲשֶׁר יַעֲשֶׂה אֹתָם הָאָדָם וָחַי בָּהֶם׳. רַשִׁ״י מְסַבֵּר: ׳וָחַי בָּהֶם׳ – חַיִּים לָעוֹלָם הַבָּא. וְ׳אֲנִי ה׳׳ – ׳נֶאֱמָן לְשַׁלֵּם שָׂכָר׳. הַמִּצְווֹת הֵן הַחַיִּים שֶׁלָּנוּ.

הַמִּצְווֹת נוֹתְנוֹת לִי חַיִּים – כָּאן וְלָעוֹלָם הַבָּא.

רָמָה א׳ – כָּל הַפָּרָשָׁה

מָה לוֹקְחִים מִרָמָה א׳?

פַּעַם בַּשָּׁנָה הַכֹּהֵן הַגָּדוֹל נִכְנָס בְּבִגְדֵי לָבָן אֶל הַמָּקוֹם הֲכִי קָדוֹשׁ, וּמְבַקֵּשׁ סְלִיחָה עַל כָּל עַם יִשְׂרָאֵל. וְה׳ מַבְטִיחַ: ׳לִפְנֵי ה׳ תִּטְהָרוּ׳ – בְּכָל שָׁנָה מֵחָדָשׁ.

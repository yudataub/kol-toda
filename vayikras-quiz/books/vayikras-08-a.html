׳אַךְ בֶּעָשׂוֹר לַחֹדֶשׁ הַשְּׁבִיעִי הַזֶּה יוֹם הַכִּפֻּרִים הוּא׳ – ׳וְעִנִּיתֶם אֶת נַפְשֹׁתֵיכֶם׳, ׳כִּי יוֹם כִּפֻּרִים הוּא לְכַפֵּר עֲלֵיכֶם לִפְנֵי ה׳ אֱלֹהֵיכֶם׳. וְהוּא נִמְשָׁךְ ׳מֵעֶרֶב עַד עֶרֶב׳ – מֵעֶרֶב תִּשְׁעָה בְּתִשְׁרֵי עַד הָעֶרֶב שֶׁאַחֲרָיו.

יוֹם שָׁלֵם שֶׁל כַּפָּרָה וְשֶׁל קִרְבָה לַה׳.

🍂

💛 אַהֲבַת הַשֵּׁם: ה׳ מִלֵּא לִי אֶת הַבַּיִת בְּטוֹב, וַאֲנִי שָׂמֵחַ בַּחַג שֶׁלּוֹ 💛

📖 וַיִּקְרָא, פֶּרֶק כ״ג, פְּסוּקִים ל״ג–ל״ט

🍂 חַג הָאָסִיף

׳בַּחֲמִשָּׁה עָשָׂר יוֹם לַחֹדֶשׁ הַשְּׁבִיעִי הַזֶּה חַג הַסֻּכּוֹת שִׁבְעַת יָמִים לַה׳׳. זֶה הַזְּמַן ׳בְּאָסְפְּכֶם אֶת תְּבוּאַת הָאָרֶץ׳ – כְּשֶׁאוֹסְפִים אֶת הַיְּבוּל הַבַּיְתָה. וְאַחֲרָיו – ׳בַּיּוֹם הַשְּׁמִינִי... עֲצֶרֶת הִוא׳, שְׁמִינִי עֲצֶרֶת.

כְּשֶׁהַמַּחְסָנִים מְלֵאִים – שְׂמֵחִים לִפְנֵי ה׳.

13
🍋

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ לִפְנֵי ה׳ עִם הַלּוּלָב וְהָאֶתְרוֹג שֶׁלִּי 💛

📖 וַיִּקְרָא, פֶּרֶק כ״ג, פְּסוּקִים מ׳–מ״א

רָמָה א׳ – כָּל הַפָּרָשָׁה

🍋 אַרְבָּעַת הַמִּינִים

׳וּלְקַחְתֶּם לָכֶם בַּיּוֹם הָרִאשׁוֹן פְּרִי עֵץ הָדָר כַּפֹּת תְּמָרִים וַעֲנַף עֵץ עָבֹת וְעַרְבֵי נָחַל׳. רַשִׁ״י מְסַבֵּר: ׳הָדָר׳ – זֶה אֶתְרוֹג; ׳כַּפֹּת תְּמָרִים׳ – לוּלָב אֶחָד; ׳עֵץ עָבֹת׳ – הֲדַס, שֶׁעָלָיו קְלוּעִים כְּמוֹ חֶבֶל. ׳וּשְׂמַחְתֶּם לִפְנֵי ה׳ אֱלֹהֵיכֶם שִׁבְעַת יָמִים׳.

בְּסֻכּוֹת לוֹקְחִים אֶת הַמִּינִים וּשְׂמֵחִים לִפְנֵי ה׳.

14
🛖

💛 אַהֲבַת הַשֵּׁם: ה׳ עוֹטֵף אוֹתִי בַּאֲהָבָה כְּמוֹ עֲנַן כָּבוֹד 💛

📖 וַיִּקְרָא, פֶּרֶק כ״ג, פְּסוּקִים מ״ב–מ״ד

רָמָה א׳ – כָּל הַפָּרָשָׁה

🛖 בַּסֻּכֹּת תֵּשְׁבוּ

׳בַּסֻּכֹּת תֵּשְׁבוּ שִׁבְעַת יָמִים... לְמַעַן יֵדְעוּ דֹרֹתֵיכֶם כִּי בַסֻּכּוֹת הוֹשַׁבְתִּי אֶת בְּנֵי יִשְׂרָאֵל׳. רַשִׁ״י מְסַבֵּר: אֵלּוּ ׳עַנְנֵי כָבוֹד׳ – הָעֲנָנִים שֶׁה׳ הִקִּיף בָּהֶם אֶת עַם יִשְׂרָאֵל בַּמִּדְבָּר. וּבְסוֹף: ׳וַיְדַבֵּר מֹשֶׁה אֶת מֹעֲדֵי ה׳ אֶל בְּנֵי יִשְׂרָאֵל׳.

הַסֻּכָּה מַזְכִּירָה שֶׁה׳ שָׁמַר עָלֵינוּ בַּמִּדְבָּר.

🕎

💛 אַהֲבַת הַשֵּׁם: ה׳ מֵאִיר לִי אֶת הַדֶּרֶךְ כְּמוֹ נֵר 💛

📖 וַיִּקְרָא, פֶּרֶק כ״ד, פְּסוּקִים א׳–ד׳

רָמָה א׳ – כָּל הַפָּרָשָׁה

🕎 נֵר תָּמִיד

׳צַו אֶת בְּנֵי יִשְׂרָאֵל וְיִקְחוּ אֵלֶיךָ שֶׁמֶן זַיִת זָךְ כָּתִית לַמָּאוֹר לְהַעֲלֹת נֵר תָּמִיד׳. אַהֲרֹן מַדְלִיק אֶת הַמְּנוֹרָה ׳מֵעֶרֶב עַד בֹּקֶר׳. רַשִׁ״י מְסַבֵּר: ׳זָךְ׳ – הַשֶּׁמֶן הָרִאשׁוֹן וְהַטָּהוֹר שֶׁיּוֹצֵא מִן הַזַּיִת. וְלִפְנֵי הַהַדְלָקָה מְנַקִּים אֶת הַמְּנוֹרָה מִן הָאֵפֶר.

לַמְּנוֹרָה שֶׁל ה׳ נוֹתְנִים אֶת הַשֶּׁמֶן הֲכִי טָהוֹר.

16
🥖

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי לֶחֶם בְּכָל שָׁבוּעַ, וַאֲנִי מְבָרֵךְ עָלָיו 💛

📖 וַיִּקְרָא, פֶּרֶק כ״ד, פְּסוּקִים ה׳–ט׳

רָמָה א׳ – כָּל הַפָּרָשָׁה

🥖 לֶחֶם הַפָּנִים

עַל הַשֻּׁלְחָן שֶׁבַּמִּשְׁכָּן שָׂמִים שְׁתֵּים עֶשְׂרֵה חַלּוֹת, בִּשְׁתֵּי שׁוּרוֹת שֶׁל שֵׁשׁ, וּלְבוֹנָה זַכָּה – בֹּשֶׂם – עֲלֵיהֶן. ׳בְּיוֹם הַשַּׁבָּת בְּיוֹם הַשַּׁבָּת יַעַרְכֶנּוּ לִפְנֵי ה׳ תָּמִיד׳ – בְּכָל שַׁבָּת מַחְלִיפִים, וְהַכֹּהֲנִים אוֹכְלִים אֶת הַלֶּחֶם בְּמָקוֹם קָדוֹשׁ.

גַּם הַלֶּחֶם שֶׁלָּנוּ יָכוֹל לִהְיוֹת ׳לִפְנֵי ה׳׳.

⚖️

💛 אַהֲבַת הַשֵּׁם: ה׳ הוּא אֱלֹהֵי כֻּלָּנוּ, וְשׁוֹמֵר עַל כָּל אֶחָד 💛

📖 וַיִּקְרָא, פֶּרֶק כ״ד, פָּסוּק כ״ב

⚖️ מִשְׁפָּט אֶחָד

בְּסוֹף הַפָּרָשָׁה הַתּוֹרָה מְלַמֶּדֶת כְּלָל: ׳מִשְׁפַּט אֶחָד יִהְיֶה לָכֶם כַּגֵּר כָּאֶזְרָח יִהְיֶה כִּי אֲנִי ה׳ אֱלֹהֵיכֶם׳. אוֹתוֹ דִּין לְכֻלָּם – לְמִי שֶׁנּוֹלַד בְּיִשְׂרָאֵל וּלְמִי שֶׁהִצְטָרֵף אֵלֵינוּ.

הַתּוֹרָה שׁוֹפֶטֶת אֶת כֻּלָּם בְּאוֹתוֹ צֶדֶק.

מָה לוֹקְחִים מִרָמָה א׳?

פָּרָשַׁת אֱמֹר הִיא לוּחַ הַשָּׁנָה שֶׁל עַם יִשְׂרָאֵל: שַׁבָּת בְּכָל שָׁבוּעַ, וּמוֹעֲדִים מִנִּיסָן עַד תִּשְׁרֵי. בְּכָל אֶחָד מֵהֶם – ׳מִקְרָא קֹדֶשׁ׳, פְּגִישָׁה שְׂמֵחָה עִם ה׳.

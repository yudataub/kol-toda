🐄 סִימְנֵי הַבְּהֵמָה הַטְּהוֹרָה

׳זֹאת הַחַיָּה אֲשֶׁר תֹּאכְלוּ׳ – בְּהֵמָה כְּשֵׁרָה צְרִיכָה שְׁנֵי סִימָנִים: ׳מַפְרֶסֶת פַּרְסָה וְשֹׁסַעַת שֶׁסַע׳ – פַּרְסָה שֶׁמְּחֻלֶּקֶת לִשְׁנַיִם, וּ׳מַעֲלַת גֵּרָה׳ – לוֹעֶסֶת שׁוּב אֶת הָאֹכֶל. רַשִׁ״י מְסַבֵּר: מֹשֶׁה הָיָה אוֹחֵז כָּל חַיָּה וּמַרְאֶה לָעָם – ׳זֹאת תֹּאכְלוּ וְזֹאת לֹא תֹאכְלוּ׳.

פָּרָה, כֶּבֶשׂ וְעֵז – יֵשׁ לָהֶן שְׁנֵי הַסִּימָנִים.

13
🐫

💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לִי סִימָנִים בְּרוּרִים, כְּדֵי שֶׁלֹּא אֶטְעֶה 💛

📖 וַיִּקְרָא, פֶּרֶק י״א, פְּסוּקִים ד׳–ח׳

רָמָה א׳ – כָּל הַפָּרָשָׁה

🐫 רַק סִימָן אֶחָד – לֹא מַסְפִּיק

הַגָּמָל, הַשָּׁפָן וְהָאַרְנֶבֶת מַעֲלִים גֵּרָה, אֲבָל הַפַּרְסָה שֶׁלָּהֶם לֹא מְחֻלֶּקֶת. הַחֲזִיר מַפְרִיס פַּרְסָה, אֲבָל ׳גֵּרָה לֹא יִגָּר׳. לָכֵן כֻּלָּם ׳טְמֵאִים הֵם לָכֶם׳ – צְרִיכִים אֶת שְׁנֵי הַסִּימָנִים יַחַד.

בְּהֵמָה כְּשֵׁרָה – רַק עִם שְׁנֵי הַסִּימָנִים.

14
🐟

💛 אַהֲבַת הַשֵּׁם: ה׳ בָּרָא אֶת הַדָּגִים, וְלִמֵּד אוֹתִי אֵיךְ לְהַבְחִין 💛

📖 וַיִּקְרָא, פֶּרֶק י״א, פְּסוּקִים ט׳–י״ב

רָמָה א׳ – כָּל הַפָּרָשָׁה

🐟 סְנַפִּיר וְקַשְׂקֶשֶׂת

׳אֶת זֶה תֹּאכְלוּ מִכֹּל אֲשֶׁר בַּמָּיִם כֹּל אֲשֶׁר לוֹ סְנַפִּיר וְקַשְׂקֶשֶׂת׳. דָּג כָּשֵׁר צָרִיךְ סְנַפִּירִים – כְּנָפַיִם קְטַנּוֹת שֶׁבָּהֶן הוּא שׂוֹחֶה, וְקַשְׂקַשִּׂים – הַ׳שִּׁרְיוֹן׳ הַקָּטָן עַל הָעוֹר. דָּג בְּלִי קַשְׂקַשִּׂים – לֹא אוֹכְלִים.

גַּם בַּיָּם יֵשׁ סִימָנִים שֶׁל כַּשְׁרוּת.

15
🦅

💛 אַהֲבַת הַשֵּׁם: ה׳ בָּרָא עוֹלָם מָלֵא חָכְמָה 💛

📖 וַיִּקְרָא, פֶּרֶק י״א, פְּסוּקִים י״ג–כ״ג

רָמָה א׳ – כָּל הַפָּרָשָׁה

🦅 הָעוֹפוֹת הָאֲסוּרִים

בָּעוֹפוֹת הַתּוֹרָה לֹא נוֹתֶנֶת סִימָנִים – הִיא מוֹנָה אֶת הָעוֹפוֹת הָאֲסוּרִים, כְּמוֹ ׳הַנֶּשֶׁר׳, ׳הָעֹרֵב׳ וְ׳הַחֲסִידָה׳. רַשִׁ״י מְסַבֵּר: לָמָּה קוֹרְאִים לָהּ חֲסִידָה? ׳שֶׁעוֹשָׂה חֲסִידוּת עִם חַבְרוֹתֶיהָ בִּמְזוֹנוֹת׳ – הִיא נוֹתֶנֶת אֹכֶל לַחֲבֵרוֹת שֶׁלָּהּ.

הַתּוֹרָה מְלַמֶּדֶת מָה לֹא לֶאֱכֹל – וְגַם מָה לִלְמֹד מֵהַחַיּוֹת.

16
👑

💛 אַהֲבַת הַשֵּׁם: ה׳ הֶעֱלָה אוֹתִי מִמִּצְרַיִם כְּדֵי שֶׁאֶהְיֶה קָדוֹשׁ 💛

📖 וַיִּקְרָא, פֶּרֶק י״א, פְּסוּקִים מ״ד–מ״ז

רָמָה א׳ – כָּל הַפָּרָשָׁה

👑 ׳וִהְיִיתֶם קְדֹשִׁים׳

לָמָּה כָּל זֶה? ה׳ אוֹמֵר: ׳וְהִתְקַדִּשְׁתֶּם וִהְיִיתֶם קְדֹשִׁים כִּי קָדוֹשׁ אָנִי׳, ׳כִּי אֲנִי ה׳ הַמַּעֲלֶה אֶתְכֶם מֵאֶרֶץ מִצְרַיִם׳. הַפָּרָשָׁה נִגְמֶרֶת: ׳לְהַבְדִּיל בֵּין הַטָּמֵא וּבֵין הַטָּהֹר׳. גַּם בַּמִּטְבָּח וּבַשֻּׁלְחָן – יְהוּדִי קָדוֹשׁ.

הַכַּשְׁרוּת הוֹפֶכֶת אֶת הַשֻּׁלְחָן שֶׁלִּי לִמְקוֹם קְדֻשָּׁה.

17
💡

רָמָה א׳ – כָּל הַפָּרָשָׁה

מָה לוֹקְחִים מִרָמָה א׳?

בַּיּוֹם הַשְּׁמִינִי אַהֲרֹן עוֹבֵד בַּמִּשְׁכָּן, מְבָרֵךְ אֶת הָעָם, וְאֵשׁ יוֹרֶדֶת מִן הַשָּׁמַיִם – ה׳ שׁוֹרֶה בֵּינֵינוּ. וְאַחַר כָּךְ ה׳ מְלַמֵּד אוֹתָנוּ ׳לְהַבְדִּיל׳: מָה אוֹכְלִים וּמָה לֹא, כְּדֵי שֶׁנִּהְיֶה קְדוֹשִׁים כְּמוֹתוֹ.

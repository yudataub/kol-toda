׳מִזְמוֹר שִׁיר לְיוֹם הַשַּׁבָּת׳ – רַשִׁ״י מְבָאֵר: ׳מִזְמוֹר שִׁיר – שֶׁל לְוִיִּם. לְיוֹם הַשַּׁבָּת – שֶׁאוֹמְרִים אוֹתוֹ בְּשַׁבָּתוֹת, וְהוּא מְדַבֵּר בְּעִנְיַן הָעוֹלָם הַבָּא שֶׁכֻּלּוֹ שַׁבָּת׳. מְצוּדַת דָּוִד: ׳הַמִּזְמוֹר הַזֶּה הוּקְבַּע לְזַמֵּר עַל הַדּוּכָן בְּשַׁבָּת׳. כְּלוֹמַר: הַשַּׁבָּת שֶׁלָּנוּ הִיא דֻּגְמָה קְטַנָּה לָעוֹלָם הַבָּא.

הַשַּׁבָּת הִיא טַעַם קָטָן שֶׁל הָעוֹלָם הַבָּא שֶׁכֻּלּוֹ שַׁבָּת.

📜 הַפָּסוּקלְהַגִּיד בַּבֹּקֶר חַסְדֶּךָ וֶאֱמוּנָתְךָ בַּלֵּילוֹת

רָמָה ב׳ – עִיּוּן בְּשִׁיר הַשַּׁבָּת

🌅 רַשִׁ״י: בֹּקֶר וְלֵילוֹת

׳לְהַגִּיד בַּבֹּקֶר חַסְדֶּךָ וֶאֱמוּנָתְךָ בַּלֵּילוֹת׳ – רַשִׁ״י מְבָאֵר: ׳בְּעֵת הַגְּאֻלָּה׳ – זֶה הַבֹּקֶר, ׳וּבְעוֹד צָרַת הַגָּלוּת לְהַאֲמִין בְּךָ שֶׁתִּשְׁמֹר הַבְטָחָתְךָ׳ – אֵלֶּה הַלֵּילוֹת. ׳כָּל זֶה נָאֶה וְטוֹב׳. מְצוּדַת דָּוִד: ׳קִיּוּם הַבְטָחָתְךָ נַגִּיד בְּכָל הַלֵּילוֹת׳. אָדָם יָכוֹל לְהוֹדוֹת בַּבֹּקֶר כְּשֶׁטּוֹב, וְגַם לְהַאֲמִין בַּלַּיְלָה כְּשֶׁקָּשֶׁה.

מוֹדִים עַל הַחֶסֶד כְּשֶׁטּוֹב, וּמַאֲמִינִים כְּשֶׁקָּשֶׁה – שְׁנֵיהֶם נָאִים וְטוֹבִים.

📜 הַפָּסוּקמַה גָּדְלוּ מַעֲשֶׂיךָ ה׳ מְאֹד עָמְקוּ מַחְשְׁבֹתֶיךָ

💛 אַהֲבַת הַשֵּׁם: ה׳ גָּדוֹל מִמֶּנִּי, וַאֲנִי שָׂמֵחַ שֶׁאֲנִי לֹא מֵבִין הַכֹּל 💛

📖 תְּהִלִּים, פֶּרֶק צ״ב, פָּסוּק ו׳ (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בְּשִׁיר הַשַּׁבָּת

🧠 מְצוּדַת דָּוִד: ׳עָמְקוּ מַחְשְׁבֹתֶיךָ׳

׳מַה גָּדְלוּ מַעֲשֶׂיךָ ה׳ מְאֹד עָמְקוּ מַחְשְׁבֹתֶיךָ׳ – מְצוּדַת דָּוִד מְבָאֵר: ׳עָמְקוּ – הֵם דְּבָרִים עֲמֻקִּים וְאֵין לְהָבִינָם׳. הַמְשׁוֹרֵר אוֹמֵר שֶׁאֲפִלּוּ אַחֲרֵי שֶׁהוּא רוֹאֶה הַרְבֵּה, הוּא יוֹדֵעַ שֶׁיֵּשׁ עוֹד הַרְבֵּה שֶׁהוּא לֹא מֵבִין. זוֹ עֲנָוָה: לְהוֹדוֹת וְלֹא לְהֵבִין הַכֹּל.

יֵשׁ דְּבָרִים עֲמֻקִּים שֶׁאֵין לָנוּ יְכוֹלֶת לְהָבִין – וְזֶה בְּסֵדֶר.

🌴

📜 הַפָּסוּקצַדִּיק כַּתָּמָר יִפְרָח כְּאֶרֶז בַּלְּבָנוֹן יִשְׂגֶּה

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי כֹּחַ לִתֵּן פְּרִי וּלְהִתְחַדֵּשׁ 💛

📖 תְּהִלִּים, פֶּרֶק צ״ב, פָּסוּק י״ג (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בְּשִׁיר הַשַּׁבָּת

🌴 רַשִׁ״י: תָּמָר וְאֶרֶז

׳צַדִּיק כַּתָּמָר יִפְרָח כְּאֶרֶז בַּלְּבָנוֹן יִשְׂגֶּה׳ – רַשִׁ״י מְבָאֵר: ׳כַּתָּמָר – לַעֲשׂוֹת פֵּרוֹת, וְכָאֶרֶז בַּלְּבָנוֹן – לְהַחֲלִיף גִּזְעוֹ׳. מְצוּדַת דָּוִד: ׳יַעֲשֶׂה פְּרִי בְּבָנִים וּבְבָנוֹת כַּתָּמָר הַזֶּה, וְ״יִשְׂגֶּה״ – יִתְגַּדֵּל וְיַעֲלֶה מַעְלָה מַעְלָה כְּקוֹמַת הָאֶרֶז׳. שְׁנֵי דִּמְיוֹנוֹת: תָּמָר נוֹתֵן פְּרִי, אֶרֶז גָּדֵל לְמַעְלָה וּמַחֲלִיף אֶת גִּזְעוֹ – הַצַּדִּיק נוֹתֵן וְגַם מִתְחַדֵּשׁ.

הַצַּדִּיק נוֹתֵן פְּרִי כְּמוֹ תָּמָר וּמִתְחַדֵּשׁ כְּמוֹ אֶרֶז.

🏫 מְצוּדַת דָּוִד: ׳שְׁתוּלִים בְּבֵית ה׳׳

׳שְׁתוּלִים בְּבֵית ה׳ בְּחַצְרוֹת אֱלֹהֵינוּ יַפְרִיחוּ׳ – מְצוּדַת דָּוִד מְבָאֵר: ׳יִהְיוּ נְטוּעִים וּקְבוּעִים בְּבֵית ה׳, לוֹמְדֵי הַתּוֹרָה וְהַחָכְמָה׳. הָעֵץ שֶׁשָּׁתוּל בִּמְקוֹם קָבוּעַ גָּדֵל. כָּךְ גַּם אָדָם שֶׁקָּבוּעַ בְּבֵית הַמִּדְרָשׁ וּבְבֵית הַכְּנֶסֶת, שֶׁשָּׁם הוּא שׁוֹרֵשׁ וְגָדֵל. רַשִׁ״י: ׳יִהְיוּ הַצַּדִּיקִים בְּבֵית ה׳׳.

מִי שֶׁקָּבוּעַ בְּבֵית הַמִּדְרָשׁ גָּדֵל כְּמוֹ עֵץ שָׁתוּל.

📜 הַפָּסוּקעוֹד יְנוּבוּן בְּשֵׂיבָה דְּשֵׁנִים וְרַעֲנַנִּים יִהְיוּ

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי כֹּחַ לִפְרֹחַ בְּכָל גִּיל 💛

📖 תְּהִלִּים, פֶּרֶק צ״ב, פְּסוּקִים ט״ו–ט״ז (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בְּשִׁיר הַשַּׁבָּת

🍂 מְצוּדַת דָּוִד: ׳עוֹד יְנוּבוּן בְּשֵׂיבָה׳

׳עוֹד יְנוּבוּן בְּשֵׂיבָה דְּשֵׁנִים וְרַעֲנַנִּים יִהְיוּ׳ – מְצוּדַת דָּוִד: ׳בִּימֵי זִקְנָתָם יַרְבּוּ עוֹד תְּנוּבָה וְהַצְמָחָה וְיִהְיוּ דְּשֵׁנִים וּרְטוּבִים׳. רַשִׁ״י: ׳שְׁמֵנִים וְרַעֲנַנִּים יִהְיוּ, וְאָז יַגִּידוּ כִּי יָשָׁר ה׳׳. וְהַסִּבָּה לְפִי מְצוּדַת דָּוִד: ׳לְמַעַן יִוָּדַע לַכֹּל כִּי יָשָׁר ה׳ וְאֵין בּוֹ עַוְלָה׳. הַזִּקְנָה הַפּוֹרַחַת הִיא הוֹכָחָה שֶׁה׳ יָשָׁר.

הַצַּדִּיק פּוֹרֵחַ גַּם בַּזִּקְנָה, וְזֶה מַרְאֶה שֶׁה׳ יָשָׁר.

📜 הַפָּסוּקעֵדֹתֶיךָ נֶאֶמְנוּ מְאֹד לְבֵיתְךָ נַאֲוָה קֹדֶשׁ ה׳ לְאֹרֶךְ יָמִים

💛 אַהֲבַת הַשֵּׁם: ה׳ מְקַיֵּם אֶת הַהַבְטָחוֹת שֶׁלּוֹ, וַאֲנִי בָּטוּחַ בְּכָךְ 💛

׳עֵדֹתֶיךָ נֶאֶמְנוּ מְאֹד לְבֵיתְךָ נַאֲוָה קֹדֶשׁ׳ – רַשִׁ״י מְבָאֵר: ׳שֶׁהֵעִידוּ וְהִבְטִיחוּ נְבִיאֶיךָ לְבֵיתְךָ שֶׁהוּא נָאוֶה קֹדֶשׁ... וְאַף עַל פִּי שֶׁאָרְכוּ הַיָּמִים, נֶאֱמָנִים הֵם לַה׳ מְאֹד׳. מְצוּדַת דָּוִד: ׳דִּבְרֵי הַנְּבִיאִים הַמְעִידִים עַל הַתְּשׁוּעָה הֵם נֶאֱמָנִים מְאֹד, וְיִהְיֶה לְבֵיתְךָ נוֹי וְיוֹפִי קְדֻשָּׁה וְתַתְמִיד לְאֹרֶךְ יָמִים וְלֹא תֶחֱרַב עוֹד׳. ה׳ לֹא שׁוֹכֵחַ אֶת הַהַבְטָחוֹת.

הַהַבְטָחוֹת שֶׁל הַנְּבִיאִים נֶאֱמָנוֹת, גַּם כְּשֶׁהַזְּמַן אָרֹךְ.

מָה לוֹקְחִים מִרָמָה ב׳?

רַשִׁ״י וּמְצוּדַת דָּוִד מַסְבִּירִים: הַמִּזְמוֹר מְדַבֵּר עַל הָעוֹלָם הַבָּא, עַל בֹּקֶר שֶׁל גְּאֻלָּה וְעַל לֵילוֹת שֶׁל אֱמוּנָה, וְעַל צַדִּיקִים שֶׁגְּדֵלִים כְּעֵצִים. הַכֹּל מִתּוֹךְ אַהֲבָה שֶׁל ה׳ לְעַמּוֹ.

🤔 חוֹשְׁבִים עַל זֶה

  • ׳טוֹב לְהוֹדוֹת לַה׳׳ – עַל מָה אֲנִי רוֹצֶה לְהוֹדוֹת לַה׳ הַשַּׁבָּת, וְאֵיךְ אֶעֱשֶׂה אֶת זֶה בֶּאֱמֶת מִן הַלֵּב?
  • ׳צַדִּיק כַּתָּמָר יִפְרָח׳ – מָה הַדָּבָר הַטּוֹב שֶׁאֲנִי רוֹצֶה שֶׁיִּגְדַּל בִּי כְּמוֹ הַתָּמָר וְהָאֶרֶז?
  • ׳לְהַגִּיד בַּבֹּקֶר חַסְדֶּךָ וֶאֱמוּנָתְךָ בַּלֵּילוֹת׳ – אֵיךְ אֶפְשָׁר לְהַאֲמִין גַּם כְּשֶׁקָּשֶׁה? מָה עוֹזֵר לִי לֵאמֹר ׳ה׳ טוֹב׳ גַּם בַּלַּיְלָה?

שִׁיר הַבְּרִיאָה שֶׁאוֹמְרִים בְּרֹאשׁ חֹדֶשׁ – אוֹר, הָרִים, מַעְיָנוֹת, לֶחֶם וְיַיִן וְשָׁמֶן, וְכָל בְּרִיָּה שֶׁמְּחַכָּה לְאָכְלָהּ מִיַּד ה׳

רָמָה א׳ – הַבְּרִיאָה שָׁרָה

☀️ ׳עֹטֶה אוֹר כַּשַּׂלְמָה׳

הַמִּזְמוֹר מַתְחִיל בַּבְּרָכָה לְה׳: ׳בָּרֲכִי נַפְשִׁי אֶת ה׳ ה׳ אֱלֹהַי גָּדַלְתָּ מְּאֹד הוֹד וְהָדָר לָבָשְׁתָּ׳. ה׳ ׳עֹטֶה אוֹר כַּשַּׂלְמָה נוֹטֶה שָׁמַיִם כַּיְרִיעָה׳ – הוּא לָבוּשׁ אוֹר כְּמוֹ שִׂמְלָה, וְהַשָּׁמַיִם פְּרוּסִים כְּמוֹ יְרִיעָה שֶׁל אֹהֶל. הַבְּרִיאָה הִיא בַּיִת גָּדוֹל, וְהָאוֹר הוּא הַבֶּגֶד שֶׁלּוֹ.

ה׳ עוֹטֶה אוֹר כְּמוֹ שִׂמְלָה, וְהַשָּׁמַיִם יְרִיעָה שֶׁל אֹהֶל.

1

💛 אַהֲבַת הַשֵּׁם: ה׳ בָּנָה בִּשְׁבִילִי אֶרֶץ יַצִּיבָה לָגוּר בָּהּ 💛

📖 תְּהִלִּים, פֶּרֶק ק״ד, פְּסוּקִים ה׳–ט׳

רָמָה א׳ – הַבְּרִיאָה שָׁרָה

🏔️ ׳יָסַד אֶרֶץ עַל מְכוֹנֶיהָ׳

ה׳ יָסַד אֶת הָאָרֶץ חֲזָקָה: ׳יָסַד אֶרֶץ עַל מְכוֹנֶיהָ בַּל תִּמּוֹט עוֹלָם וָעֶד׳. הַמַּיִם כִּסּוּ אֶת הֶהָרִים, אֲבָל ה׳ קָבַע לָהֶם גְּבוּל: ׳גְּבוּל שַׂמְתָּ בַּל יַעֲבֹרוּן בַּל יְשׁוּבוּן לְכַסּוֹת הָאָרֶץ׳. הַיָּם לֹא יָשׁוּב וִיכַסֶּה אֶת הָאָרֶץ – כָּךְ יֵשׁ לָנוּ מָקוֹם לָגוּר.

ה׳ קָבַע לַיָּם גְּבוּל, וְכָךְ יֵשׁ לָנוּ אֶרֶץ יַבָּשָׁה.

2

💛 אַהֲבַת הַשֵּׁם: ה׳ דּוֹאֵג לְמַיִם וּלְאֹכֶל לְכָל בְּרִיָּה, גַּם לִי 💛

📖 תְּהִלִּים, פֶּרֶק ק״ד, פְּסוּקִים י׳–י״ג

רָמָה א׳ – הַבְּרִיאָה שָׁרָה

💧 ׳הַמְשַׁלֵּחַ מַעְיָנִים׳

הַמַּיִם נוֹבְעִים מִתּוֹךְ הֶהָרִים: ׳הַמְשַׁלֵּחַ מַעְיָנִים בַּנְּחָלִים בֵּין הָרִים יְהַלֵּכוּן׳. כָּל חַיָּה בַּשָּׂדֶה שׁוֹתָה: ׳יַשְׁקוּ כָּל חַיְתוֹ שָׂדָי יִשְׁבְּרוּ פְרָאִים צְמָאָם׳, וְהָעוֹף יוֹשֵׁב עַל הָעֲנָפִים וְשָׁר. ׳מַשְׁקֶה הָרִים מֵעֲלִיּוֹתָיו מִפְּרִי מַעֲשֶׂיךָ תִּשְׂבַּע הָאָרֶץ׳ – ה׳ מַשְׁקֶה אֶת הֶהָרִים וְהָאָרֶץ שׂוֹבַעַת מִפְּרִי מַעֲשָׂיו.

ה׳ שׁוֹלֵחַ מַיִם לְכָל בְּרִיָּה, וְהָאָרֶץ שׂוֹבַעַת.

3

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי לֶחֶם, שִׂמְחָה וְאוֹר 💛

📖 תְּהִלִּים, פֶּרֶק ק״ד, פְּסוּקִים י״ד–ט״ו

רָמָה א׳ – הַבְּרִיאָה שָׁרָה

🍞 ׳לֶחֶם וְיַיִן וְשָׁמֶן׳

ה׳ דּוֹאֵג גַּם לָאָדָם: ׳מַצְמִיחַ חָצִיר לַבְּהֵמָה וְעֵשֶׂב לַעֲבֹדַת הָאָדָם לְהוֹצִיא לֶחֶם מִן הָאָרֶץ׳. ׳וְיַיִן יְשַׂמַּח לְבַב אֱנוֹשׁ לְהַצְהִיל פָּנִים מִשָּׁמֶן וְלֶחֶם לְבַב אֱנוֹשׁ יִסְעָד׳. מִן הָאָרֶץ יוֹצְאִים הַלֶּחֶם, הַיַּיִן וְהַשֶּׁמֶן – לֶחֶם מְחַזֵּק, יַיִן מְשַׂמֵּחַ, וְשֶׁמֶן מַאִיר אֶת הַפָּנִים.

הָאָרֶץ נוֹתֶנֶת לֶחֶם, יַיִן וְשֶׁמֶן – וְכָל אֶחָד מֵהֶם מַעֲנִיק מַשֶּׁהוּ אַחֵר לָאָדָם.

4

💛 אַהֲבַת הַשֵּׁם: לְכָל אֶחָד ה׳ הֵכִין בַּיִת, וְגַם לִי 💛

📖 תְּהִלִּים, פֶּרֶק ק״ד, פְּסוּקִים ט״ז–י״ח

רָמָה א׳ – הַבְּרִיאָה שָׁרָה

🦌 ׳הָרִים הַגְּבֹהִים לַיְּעֵלִים׳

לְכָל בַּעַל חַיִּים יֵשׁ בַּיִת: ׳יִשְׂבְּעוּ עֲצֵי ה׳ אַרְזֵי לְבָנוֹן אֲשֶׁר נָטָע׳. ׳אֲשֶׁר שָׁם צִפֳּרִים יְקַנֵּנוּ חֲסִידָה בְּרוֹשִׁים בֵּיתָהּ׳ – הַצִּפֳּרִים בּוֹנוֹת קֵן עַל הָאֲרָזִים, וְהַחֲסִידָה עַל הַבְּרוֹשִׁים. ׳הָרִים הַגְּבֹהִים לַיְּעֵלִים סְלָעִים מַחְסֶה לַשְׁפַנִּים׳. ה׳ דָּאַג לְכָל אֶחָד מָקוֹם מַתְאִים.

לְכָל בְּרִיָּה ה׳ הֵכִין בַּיִת: צִפּוֹר בָּאֶרֶז, יָעֵל בֶּהָרִים, שָׁפָן בַּסְּלָעִים.

5

💛 אַהֲבַת הַשֵּׁם: ה׳ סִדֵּר לָנוּ יוֹם וְלַיְלָה כְּדֵי שֶׁנָּנוּחַ וְנַעֲבֹד 💛

📖 תְּהִלִּים, פֶּרֶק ק״ד, פְּסוּקִים י״ט–כ״ג

רָמָה א׳ – הַבְּרִיאָה שָׁרָה

🌙 ׳עָשָׂה יָרֵחַ לְמוֹעֲדִים׳

ה׳ בָּרָא יָרֵחַ לִסְפֹּר אֶת הַזְּמַנִּים: ׳עָשָׂה יָרֵחַ לְמוֹעֲדִים שֶׁמֶשׁ יָדַע מְבוֹאוֹ׳. בַּלַּיְלָה יוֹצְאוֹת חַיּוֹת הַיַּעַר, ׳תִּזְרַח הַשֶּׁמֶשׁ יֵאָסֵפוּן וְאֶל מְעוֹנֹתָם יִרְבָּצוּן׳ – וְכַשֶּׁהַשֶּׁמֶשׁ זוֹרַחַת הֵן חוֹזְרוֹת לִמְעוֹנוֹתֵיהֶן. ׳יֵצֵא אָדָם לְפָעֳלוֹ וְלַעֲבֹדָתוֹ עֲדֵי עָרֶב׳. הַיּוֹם שַׁיָּךְ לָאָדָם וְהַלַּיְלָה לַחַיּוֹת.

הַיָּרֵחַ סוֹפֵר אֶת הַמּוֹעֲדִים, וְהַשֶּׁמֶשׁ יוֹצֵאת וְשׁוֹקַעַת בַּזְּמַן.

6

💛 אַהֲבַת הַשֵּׁם: ה׳ פּוֹתֵחַ אֶת יָדוֹ וְנוֹתֵן לִי הַכֹּל 💛

📖 תְּהִלִּים, פֶּרֶק ק״ד, פְּסוּקִים כ״ד–כ״ח

רָמָה א׳ – הַבְּרִיאָה שָׁרָה

🌊 ׳מָה רַבּוּ מַעֲשֶׂיךָ׳

הַפָּסוּק הַמֻּכָּר בְּיוֹתֵר: ׳מָה רַבּוּ מַעֲשֶׂיךָ ה׳ כֻּלָּם בְּחָכְמָה עָשִׂיתָ מָלְאָה הָאָרֶץ קִנְיָנֶךָ׳. הַמְשׁוֹרֵר מִסְתַּכֵּל עַל הַיָּם הַגָּדוֹל וְהָרָחָב, וְעַל מִסְפָּר הַחַיּוֹת הַגְּדוֹלוֹת וְהַקְּטַנּוֹת. ׳כֻּלָּם אֵלֶיךָ יְשַׂבֵּרוּן לָתֵת אָכְלָם בְּעִתּוֹ׳. ׳תִּתֵּן לָהֶם יִלְקֹטוּן תִּפְתַּח יָדְךָ יִשְׂבְּעוּן טוֹב׳ – ה׳ פּוֹתֵחַ אֶת יָדוֹ, וְכֻלָּם שׂוֹבְעִים טוֹב.

כָּל הַבְּרִיּוֹת מְחַכּוֹת שֶׁה׳ יִפְתַּח אֶת יָדוֹ, וְהוּא נוֹתֵן לְכֻלָּם.

7

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לָשִׁיר לַה׳ כָּל יְמֵי חַיַּי 💛

📖 תְּהִלִּים, פֶּרֶק ק״ד, פְּסוּקִים ל״א–ל״ד

רָמָה א׳ – הַבְּרִיאָה שָׁרָה

🎤 ׳אָשִׁירָה לַה׳ בְּחַיָּי׳

הַמִּזְמוֹר מִסְתַּיֵּם בְּהַבְטָחָה אִישִׁית: ׳יְהִי כְבוֹד ה׳ לְעוֹלָם יִשְׂמַח ה׳ בְּמַעֲשָׂיו׳. ׳אָשִׁירָה לַה׳ בְּחַיָּי אֲזַמְּרָה לֵאלֹהַי בְּעוֹדִי׳. ׳יֶעֱרַב עָלָיו שִׂיחִי אָנֹכִי אֶשְׂמַח בַּה׳׳. הַמְשׁוֹרֵר מַחְלִיט לָשִׁיר לַה׳ כָּל חַיָּיו. וּבַסּוֹף: ׳בָּרֲכִי נַפְשִׁי אֶת ה׳ הַלְלוּ יָהּ׳.

אֲנִי שָׁר לַה׳ כָּל חַיַּי, וְהַשִּׁיר שֶׁלִּי נָעִים לוֹ.

8
💡

מָה לוֹקְחִים מִרָמָה א׳?

מִזְמוֹר ק״ד הוּא מַסָּע בְּתוֹךְ הַבְּרִיאָה: אוֹר, שָׁמַיִם, הָרִים, מַעְיָנוֹת, בַּעֲלֵי חַיִּים, אֲנָשִׁים, יָם. הַמְשׁוֹרֵר רוֹאֶה בְּכָל דָּבָר אֶת הַיָּד שֶׁל ה׳, וְשָׁר לוֹ בְּכָל הַלֵּב. ה׳ אוֹהֵב אוֹתָנוּ וְדוֹאֵג לְכָל בְּרִיָּה.

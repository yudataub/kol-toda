רָמָה א׳ – תּוֹדָה מִכָּל הַלֵּב

שְׁנֵי מִזְמוֹרִים שֶׁל הוֹדָיָה: ׳הָרִיעוּ לַה׳ כָּל הָאָרֶץ׳ וְ׳בָּרֲכִי נַפְשִׁי׳ – ה׳ סוֹלֵחַ, רוֹפֵא, וְרוֹחֵם עָלֵינוּ כְּמוֹ אָב

📖 תְּהִלִּים, פֶּרֶק ק׳, פְּסוּקִים א׳–ב׳

רָמָה א׳ – תּוֹדָה מִכָּל הַלֵּב

🎉 ׳הָרִיעוּ לַה׳ כָּל הָאָרֶץ׳

הַמִּזְמוֹר נִקְרָא ׳מִזְמוֹר לְתוֹדָה׳. הוּא פּוֹתֵחַ בִּקְרִיאָה לְכָל הָעוֹלָם: ׳הָרִיעוּ לַה׳ כָּל הָאָרֶץ׳. ׳עִבְדוּ אֶת ה׳ בְּשִׂמְחָה בֹּאוּ לְפָנָיו בִּרְנָנָה׳. שִׂמְחָה הִיא הַדֶּרֶךְ הַטּוֹבָה לַעֲבֹד אֶת ה׳.

עוֹבְדִים אֶת ה׳ בְּשִׂמְחָה, וּבָאִים לְפָנָיו בִּרְנָנָה.

1
🌱

💛 אַהֲבַת הַשֵּׁם: אֲנִי שֶׁל ה׳, וְה׳ שֶׁלִּי 💛

📖 תְּהִלִּים, פֶּרֶק ק׳, פָּסוּק ג׳

רָמָה א׳ – תּוֹדָה מִכָּל הַלֵּב

🌱 ׳הוּא עָשָׂנוּ וְלוֹ אֲנַחְנוּ׳

הַלֵּב שֶׁל הַמִּזְמוֹר: ׳דְּעוּ כִּי ה׳ הוּא אֱלֹהִים הוּא עָשָׂנוּ וְלוֹ אֲנַחְנוּ עַמּוֹ וְצֹאן מַרְעִיתוֹ׳. ה׳ הוּא שֶׁעָשָׂה אוֹתָנוּ, וְאֲנַחְנוּ שֶׁלּוֹ – עַמּוֹ וְצֹאן מַרְעִיתוֹ. אֵין מָה לְפַחֵד, כִּי הָרוֹעֶה שֶׁלָּנוּ הוּא הַבּוֹרֵא.

ה׳ בָּרָא אוֹתָנוּ, וְאֲנַחְנוּ עַמּוֹ וְצֹאן מַרְעִיתוֹ.

2
🚪

💛 אַהֲבַת הַשֵּׁם: ה׳ טוֹב לְעוֹלָם, וַאֲנִי נִכְנָס אֵלָיו בְּתוֹדָה 💛

📖 תְּהִלִּים, פֶּרֶק ק׳, פְּסוּקִים ד׳–ה׳

רָמָה א׳ – תּוֹדָה מִכָּל הַלֵּב

🚪 ׳בֹּאוּ שְׁעָרָיו בְּתוֹדָה׳

אֵיךְ נִכְנָסִים לְבֵית ה׳? ׳בֹּאוּ שְׁעָרָיו בְּתוֹדָה חֲצֵרֹתָיו בִּתְהִלָּה הוֹדוּ לוֹ בָּרֲכוּ שְׁמוֹ׳. נִכְנָסִים בְּתוֹדָה, מִתּוֹךְ הִתְפַּעֲלוּת וְשִׁירָה. וְהַסִּבָּה בַּפָּסוּק הָאַחֲרוֹן: ׳כִּי טוֹב ה׳ לְעוֹלָם חַסְדּוֹ וְעַד דֹּר וָדֹר אֱמוּנָתוֹ׳ – ה׳ טוֹב תָּמִיד, וְהַחֶסֶד וְהָאֱמוּנָה שֶׁלּוֹ לָעַד.

נִכְנָסִים לִפְנֵי ה׳ בְּתוֹדָה, כִּי הוּא טוֹב לְעוֹלָם.

3
🙏

💛 אַהֲבַת הַשֵּׁם: אֲנִי מוֹדֶה לְה׳ מִכָּל הַלֵּב עַל כָּל הַטּוֹבוֹת 💛

📖 תְּהִלִּים, פֶּרֶק ק״ג, פְּסוּקִים א׳–ב׳

רָמָה א׳ – תּוֹדָה מִכָּל הַלֵּב

🙏 ׳בָּרֲכִי נַפְשִׁי׳

דָּוִד הַמֶּלֶךְ מְדַבֵּר אֶל נַפְשׁוֹ: ׳לְדָוִד בָּרֲכִי נַפְשִׁי אֶת ה׳ וְכָל קְרָבַי אֶת שֵׁם קָדְשׁוֹ׳. ׳בָּרֲכִי נַפְשִׁי אֶת ה׳ וְאַל תִּשְׁכְּחִי כָּל גְּמוּלָיו׳. הוּא מְבַקֵּשׁ מִכָּל חֶלְקֵי הַפְּנִים שֶׁלּוֹ לְהוֹדוֹת. וְאֶת הַטּוֹבוֹת צָרִיךְ לִזְכֹּר: לֹא לִשְׁכֹּחַ אֶפְלוּ אַחַת.

מוֹדִים מִכָּל הַלֵּב, וְלֹא שׁוֹכְחִים שׁוּם טוֹבָה.

4
💊

💛 אַהֲבַת הַשֵּׁם: ה׳ סוֹלֵחַ לִי, מְרַפֵּא אוֹתִי וּמְחַדֵּשׁ אוֹתִי 💛

📖 תְּהִלִּים, פֶּרֶק ק״ג, פְּסוּקִים ג׳–ה׳

רָמָה א׳ – תּוֹדָה מִכָּל הַלֵּב

💊 ׳הַסֹּלֵחַ... הָרֹפֵא... הַגּוֹאֵל׳

דָּוִד מוֹנֶה אֶת הַטּוֹבוֹת: ׳הַסֹּלֵחַ לְכָל עֲוֺנֵכִי הָרֹפֵא לְכָל תַּחֲלֻאָיְכִי׳. ׳הַגּוֹאֵל מִשַּׁחַת חַיָּיְכִי הַמְעַטְּרֵכִי חֶסֶד וְרַחֲמִים׳. ׳הַמַּשְׂבִּיַע בַּטּוֹב עֶדְיֵךְ תִּתְחַדֵּשׁ כַּנֶּשֶׁר נְעוּרָיְכִי׳. ה׳ סוֹלֵחַ, מְרַפֵּא, גּוֹאֵל וּמַשְׂבִּיעַ. וְהוּא מְחַדֵּשׁ אֶת הַכֹּחַ ׳כַּנֶּשֶׁר׳, כְּאִלּוּ חָזַרְנוּ לִהְיוֹת צְעִירִים.

ה׳ סוֹלֵחַ, מְרַפֵּא, גּוֹאֵל, וּמְחַדֵּשׁ אֶת הַכֹּחַ.

5
🕊️

💛 אַהֲבַת הַשֵּׁם: ה׳ סוֹלֵחַ לִי בְּרַחֲמִים כְּשֶׁאֲנִי חוֹזֵר אֵלָיו 💛

📖 תְּהִלִּים, פֶּרֶק ק״ג, פְּסוּקִים ח׳–י׳

רָמָה א׳ – תּוֹדָה מִכָּל הַלֵּב

🕊️ ׳רַחוּם וְחַנּוּן ה׳׳

הַפָּסוּק הַמְפֻרְסָם: ׳רַחוּם וְחַנּוּן ה׳ אֶרֶךְ אַפַּיִם וְרַב חָסֶד׳. ׳לֹא לָנֶצַח יָרִיב וְלֹא לְעוֹלָם יִטּוֹר׳. ה׳ לֹא כּוֹעֵס לָנֶצַח. ׳לֹא כַחֲטָאֵינוּ עָשָׂה לָנוּ וְלֹא כַעֲוֺנֹתֵינוּ גָּמַל עָלֵינוּ׳ – הוּא לֹא מֵשִׁיב לָנוּ כְּפִי הַחֵטְא, אֶלָּא בְּרַחֲמִים.

ה׳ רַחוּם וְחַנּוּן, וְאֵינוֹ כּוֹעֵס לָנֶצַח.

6
👨‍👦

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אוֹתִי כְּמוֹ שֶׁאַבָּא אוֹהֵב אֶת בְּנוֹ 💛

📖 תְּהִלִּים, פֶּרֶק ק״ג, פְּסוּקִים י״א–י״ד

רָמָה א׳ – תּוֹדָה מִכָּל הַלֵּב

👨‍👦 ׳כְּרַחֵם אָב עַל בָּנִים׳

שְׁתֵּי תְּמוּנוֹת יָפוֹת: ׳כִּי כִגְבֹהַּ שָׁמַיִם עַל הָאָרֶץ גָּבַר חַסְדּוֹ עַל יְרֵאָיו׳ – חֶסֶד גָּבוֹהַּ כִּשְׁמַיִם. ׳כִּרְחֹק מִזְרָח מִמַּעֲרָב הִרְחִיק מִמֶּנּוּ אֶת פְּשָׁעֵינוּ׳ – הַחֲטָאִים רְחוֹקִים מִמֶּנּוּ כְּמוֹ מִזְרָח מִמַּעֲרָב. ׳כְּרַחֵם אָב עַל בָּנִים רִחַם ה׳ עַל יְרֵאָיו׳. ׳כִּי הוּא יָדַע יִצְרֵנוּ זָכוּר כִּי עָפָר אֲנָחְנוּ׳. ה׳ מַכִּיר אוֹתָנוּ, יוֹדֵעַ שֶׁאֲנַחְנוּ בְּשָׂר וָדָם, וְרוֹחֵם עָלֵינוּ כְּמוֹ אַבָּא.

ה׳ רוֹחֵם עָלֵינוּ כְּמוֹ אַבָּא, וְיוֹדֵעַ שֶׁאֲנַחְנוּ בְּנֵי אָדָם.

🌸

💛 אַהֲבַת הַשֵּׁם: הַחֶסֶד שֶׁל ה׳ נִשְׁאָר גַּם אַחֲרֵינוּ, לַדּוֹרוֹת הַבָּאִים 💛

📖 תְּהִלִּים, פֶּרֶק ק״ג, פְּסוּקִים ט״ו–כ״ב

רָמָה א׳ – תּוֹדָה מִכָּל הַלֵּב

🌸 ׳וְחֶסֶד ה׳ מֵעוֹלָם וְעַד עוֹלָם׳

הַחַיִּים שֶׁלָּנוּ קְצָרִים: ׳אֱנוֹשׁ כֶּחָצִיר יָמָיו כְּצִיץ הַשָּׂדֶה כֵּן יָצִיץ׳. אֲבָל ׳וְחֶסֶד ה׳ מֵעוֹלָם וְעַד עוֹלָם עַל יְרֵאָיו וְצִדְקָתוֹ לִבְנֵי בָנִים׳. הַחֶסֶד שֶׁל ה׳ נִשְׁאָר לָעַד, עַל מִי שֶׁשּׁוֹמֵר אֶת הַבְּרִית וְעַל הַדּוֹרוֹת הַבָּאִים. לָכֵן הַמִּזְמוֹר מִסְתַּיֵּם בְּקְרִיאָה לְכָל הַבְּרִיאָה: ׳בָּרֲכוּ ה׳ כָּל מַעֲשָׂיו בְּכָל מְקֹמוֹת מֶמְשַׁלְתּוֹ בָּרֲכִי נַפְשִׁי אֶת ה׳׳.

הַחַיִּים קְצָרִים אֲבָל הַחֶסֶד שֶׁל ה׳ לָעַד.

8
💡

רָמָה א׳ – תּוֹדָה מִכָּל הַלֵּב

מָה לוֹקְחִים מִרָמָה א׳?

מִזְמוֹר ק׳ הוּא ׳מִזְמוֹר לְתוֹדָה׳ – שִׁיר שֶׁאוֹמְרִים כְּשֶׁמּוֹדִים לַה׳. מִזְמוֹר קג׳ הוּא שִׁיר שֶׁל דָּוִד, שֶׁבּוֹ הוּא מְבָרֵךְ אֶת ה׳ בְּכָל הַנֶּפֶשׁ עַל כָּל הַטּוֹב. ה׳ אוֹהֵב אוֹתָנוּ כְּמוֹ אָב אוֹהֵב אֶת בָּנָיו.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳וְאַל תִּשְׁכְּחִי כָּל גְּמוּלָיו׳ – אֵילוּ שְׁלוֹשָׁה דְּבָרִים טוֹבִים ה׳ עָשָׂה לִי הַשָּׁבוּעַ, וְלֹא הוֹדֵיתִי עֲלֵיהֶם?
  • ׳עִבְדוּ אֶת ה׳ בְּשִׂמְחָה׳ – מָה הַמִּצְוָה שֶׁאֲנִי עוֹשֶׂה בְּשִׂמְחָה, וּמָה אֲנִי עוֹשֶׂה רַק מִתּוֹךְ הֶרְגֵּל?
  • ׳כְּרַחֵם אָב עַל בָּנִים׳ – אֵיךְ אֲנִי מַרְגִּישׁ אֶת הָאַהֲבָה שֶׁל ה׳ אֵלַי כְּמוֹ אָב אוֹהֵב? מָתַי הִרְגַּשְׁתִּי אוֹתָהּ?

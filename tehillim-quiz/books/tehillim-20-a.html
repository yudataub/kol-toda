לְכוּ נְרַנְּנָה לַה׳

חֲמִשָּׁה מִזְמוֹרִים שֶׁאוֹמְרִים בְּקַבָּלַת שַׁבָּת – שִׁירָה חֲדָשָׁה, שִׂמְחָה שֶׁל כָּל הָעוֹלָם, וְה׳ הָרוֹעֶה שֶׁלָּנוּ

💛 אַהֲבַת הַשֵּׁם: ה׳ הוּא הַצּוּר שֶׁלִּי, וַאֲנִי שָׁר לוֹ בְּשִׂמְחָה 💛

רָמָה א׳ – הַשִּׁירָה שֶׁל קַבָּלַת שַׁבָּת

🎵 ׳לְכוּ נְרַנְּנָה לַה׳׳

הַמִּזְמוֹר הָרִאשׁוֹן קוֹרֵא לְכֻלָּנוּ לָבוֹא וְלָשִׁיר יַחַד: ׳לְכוּ נְרַנְּנָה לַה׳ נָרִיעָה לְצוּר יִשְׁעֵנוּ׳. ׳נְקַדְּמָה פָנָיו בְּתוֹדָה בִּזְמִרוֹת נָרִיעַ לוֹ׳. אֲנַחְנוּ בָּאִים לִפְנֵי ה׳ עִם תּוֹדָה, וְשָׁרִים לוֹ בְּשִׂמְחָה.

שָׁרִים לַה׳ יַחַד, עִם תּוֹדָה בַּלֵּב.

1
🌊

💛 אַהֲבַת הַשֵּׁם: ה׳ שֶׁבָּרָא אֶת הַכֹּל הוּא גַּם הָאַבָּא הָאוֹהֵב שֶׁלִּי 💛

📖 תְּהִלִּים, פֶּרֶק צ״ה, פְּסוּקִים ג׳–ה׳

רָמָה א׳ – הַשִּׁירָה שֶׁל קַבָּלַת שַׁבָּת

🌊 ׳אֲשֶׁר בְּיָדוֹ מֶחְקְרֵי אָרֶץ׳

הַמְשׁוֹרֵר מִסְתַּכֵּל עַל הָעוֹלָם: ׳כִּי אֵל גָּדוֹל ה׳ וּמֶלֶךְ גָּדוֹל עַל כָּל אֱלֹהִים׳. בְּיָדוֹ הֶעָמְקִים וְהֶהָרִים, ׳אֲשֶׁר לוֹ הַיָּם וְהוּא עָשָׂהוּ וְיַבֶּשֶׁת יָדָיו יָצָרוּ׳. הַיָּם וְהַיַּבָּשָׁה – הַכֹּל מֵאִתּוֹ. מִי שֶׁעָשָׂה אֶת הַכֹּל, רוֹצֶה שֶׁנַּכִּיר בּוֹ.

ה׳ בָּרָא אֶת הַיָּם וְאֶת הַיַּבָּשָׁה, וְהוּא מֶלֶךְ גָּדוֹל.

2
🙇

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִתְפַּלֵּל לַה׳ בְּכָל הַגּוּף וּבְכָל הַלֵּב 💛

📖 תְּהִלִּים, פֶּרֶק צ״ה, פָּסוּק ו׳

רָמָה א׳ – הַשִּׁירָה שֶׁל קַבָּלַת שַׁבָּת

🙇 ׳בֹּאוּ נִשְׁתַּחֲוֶה וְנִכְרָעָה׳

מִתּוֹךְ הַשִּׂמְחָה אֲנַחְנוּ גַּם נִכְנָעִים לִפְנֵי הַבּוֹרֵא: ׳בֹּאוּ נִשְׁתַּחֲוֶה וְנִכְרָעָה נִבְרְכָה לִפְנֵי ה׳ עֹשֵׂנוּ׳. נִשְׁתַּחֲוֶה, נִכְרָעָה, נִבְרְכָה – שָׁלוֹשׁ דַּרְכֵי הַתְּפִלָּה בַּגּוּף, כְּדֵי לְהַרְגִּישׁ שֶׁאֲנַחְנוּ לִפְנֵי ה׳.

גַּם הַגּוּף מִתְפַּלֵּל: מְשַׁתְּחֲוִים, כּוֹרְעִים וּמְבָרְכִים לִפְנֵי ה׳.

3
🐑

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹעֶה אוֹתִי כִּכְבְשַׂת הָרוֹעֶה הָאוֹהֵב 💛

📖 תְּהִלִּים, פֶּרֶק צ״ה, פְּסוּקִים ז׳–ח׳

רָמָה א׳ – הַשִּׁירָה שֶׁל קַבָּלַת שַׁבָּת

🐑 ׳עַם מַרְעִיתוֹ וְצֹאן יָדוֹ׳

ה׳ הוּא הָרוֹעֶה וַאֲנַחְנוּ הַצֹּאן: ׳כִּי הוּא אֱלֹהֵינוּ וַאֲנַחְנוּ עַם מַרְעִיתוֹ וְצֹאן יָדוֹ׳. הָרוֹעֶה מַכִּיר כָּל כִּבְשָׂה וְשׁוֹמֵר עָלָיו. וְאָז בָּאָה בַּקָּשָׁה קְצָרָה: ׳הַיּוֹם אִם בְּקֹלוֹ תִשְׁמָעוּ׳ – הַיּוֹם! אָסוּר לְהַשְׁאִיר לְמָחָר, ׳אַל תַּקְשׁוּ לְבַבְכֶם כִּמְרִיבָה׳.

ה׳ הוּא הָרוֹעֶה שֶׁלָּנוּ, וְהוּא מְחַכֶּה שֶׁנִּשְׁמַע בְּקוֹלוֹ הַיּוֹם.

4
🎶

💛 אַהֲבַת הַשֵּׁם: בְּכָל יוֹם ה׳ נוֹתֵן לִי סִבָּה חֲדָשָׁה לְהוֹדוֹת 💛

📖 תְּהִלִּים, פֶּרֶק צ״ו, פְּסוּקִים א׳–ג׳

רָמָה א׳ – הַשִּׁירָה שֶׁל קַבָּלַת שַׁבָּת

🎶 ׳שִׁירוּ לַה׳ שִׁיר חָדָשׁ׳

הַמִּזְמוֹר הַבָּא קוֹרֵא לְכָל הָאָרֶץ: ׳שִׁירוּ לַה׳ שִׁיר חָדָשׁ שִׁירוּ לַה׳ כָּל הָאָרֶץ׳. ׳שִׁירוּ לַה׳ בָּרֲכוּ שְׁמוֹ בַּשְּׂרוּ מִיּוֹם לְיוֹם יְשׁוּעָתוֹ׳. אוֹמְרִים אֶת הַבְּשׂוֹרָה מִיּוֹם לְיוֹם. בְּכָל יוֹם יֵשׁ סִבָּה חֲדָשָׁה לָשִׁיר, וַאֲפִלּוּ לָעַמִּים: ׳סַפְּרוּ בַגּוֹיִם כְּבוֹדוֹ בְּכָל הָעַמִּים נִפְלְאוֹתָיו׳.

בְּכָל יוֹם יֵשׁ שִׁיר חָדָשׁ לַה׳ – גַּם בְּיוֹם רָגִיל.

5
🌳

💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לָעוֹלָם כֻּלּוֹ סִבָּה לְשִׂמְחָה, גַּם לִי 💛

📖 תְּהִלִּים, פֶּרֶק צ״ו, פְּסוּקִים י״א–י״ב

רָמָה א׳ – הַשִּׁירָה שֶׁל קַבָּלַת שַׁבָּת

🌳 ׳יִשְׂמְחוּ הַשָּׁמַיִם וְתָגֵל הָאָרֶץ׳

הַמִּזְמוֹר מְתָאֵר עוֹלָם שֶׁכֻּלּוֹ שָׁר: ׳יִשְׂמְחוּ הַשָּׁמַיִם וְתָגֵל הָאָרֶץ יִרְעַם הַיָּם וּמְלֹאוֹ׳, ׳יַעֲלֹז שָׂדַי וְכָל אֲשֶׁר בּוֹ אָז יְרַנְּנוּ כָּל עֲצֵי יָעַר׳. הַיָּם, הַשָּׂדוֹת, וַאֲפִלּוּ הָעֵצִים – כֻּלָּם שָׁרִים לִפְנֵי ה׳. כָּל הַבְּרִיאָה שְׂמֵחָה כִּי ה׳ בָּא.

כָּל הַבְּרִיאָה שָׁרָה לַה׳, וְאֲנִי חֵלֶק מֵהַמַּקְהֵלָה.

6
🌤️

💛 אַהֲבַת הַשֵּׁם: אֲנִי אוֹהֵב אֶת ה׳, וְהוּא שׁוֹמֵר עָלַי 💛

📖 תְּהִלִּים, פֶּרֶק צ״ז, פְּסוּקִים י׳–י״א

רָמָה א׳ – הַשִּׁירָה שֶׁל קַבָּלַת שַׁבָּת

🌤️ ׳אוֹר זָרֻעַ לַצַּדִּיק׳

הַמִּזְמוֹר הַשְּׁלִישִׁי אוֹמֵר: ׳אֹהֲבֵי ה׳ שִׂנְאוּ רָע שֹׁמֵר נַפְשׁוֹת חֲסִידָיו מִיַּד רְשָׁעִים יַצִּילֵם׳. מִי שֶׁאוֹהֵב אֶת ה׳ אוֹהֵב גַּם אֶת הַטּוֹב וְשׂוֹנֵא אֶת הָרַע, וְה׳ שׁוֹמֵר עָלָיו. ׳אוֹר זָרֻעַ לַצַּדִּיק וּלְיִשְׁרֵי לֵב שִׂמְחָה׳ – כְּמוֹ זֶרַע שֶׁצּוֹמֵחַ, כָּךְ הָאוֹר וְהַשִּׂמְחָה מְחַכִּים לַצַּדִּיק.

אוֹהֲבֵי ה׳ שׂוֹנְאִים רַע, וְה׳ שׁוֹמֵר עֲלֵיהֶם; הָאוֹר כְּבָר זָרוּעַ בִּשְׁבִילָם.

🎺

💛 אַהֲבַת הַשֵּׁם: ה׳ הוּא הַמֶּלֶךְ שֶׁלִּי, וַאֲנִי שָׁר לְפָנָיו 💛

📖 תְּהִלִּים, פֶּרֶק צ״ח, פְּסוּקִים ה׳–ו׳

רָמָה א׳ – הַשִּׁירָה שֶׁל קַבָּלַת שַׁבָּת

🎺 ׳בַּחֲצֹצְרוֹת וְקוֹל שׁוֹפָר׳

הַמִּזְמוֹר הָרְבִיעִי פּוֹתֵחַ שׁוּב בְּ׳שִׁירוּ לַה׳ שִׁיר חָדָשׁ׳, כִּי ׳נִפְלָאוֹת עָשָׂה׳. ׳זַמְּרוּ לַה׳ בְּכִנּוֹר בְּכִנּוֹר וְקוֹל זִמְרָה׳, ׳בַּחֲצֹצְרוֹת וְקוֹל שׁוֹפָר הָרִיעוּ לִפְנֵי הַמֶּלֶךְ ה׳׳. נְגִינָה, כִּנּוֹר, חֲצוֹצְרוֹת וְשׁוֹפָר – הַכֹּל מִתְחַבֵּר לְשִׂמְחָה אַחַת, וְאֲפִלּוּ הַנְּהָרוֹת מוֹחֲאִים כַּף.

שׁוֹמְעִים כִּנּוֹר, חֲצוֹצְרוֹת וְשׁוֹפָר – כָּל הָעוֹלָם שָׁר לַמֶּלֶךְ.

רָמָה א׳ – הַשִּׁירָה שֶׁל קַבָּלַת שַׁבָּת

מָה לוֹקְחִים מִרָמָה א׳?

בְּכָל לֵיל שַׁבָּת אָנוּ עוֹמְדִים וְשָׁרִים חֲמִשָּׁה מִזְמוֹרִים שֶׁל שִׂמְחָה: ׳לְכוּ נְרַנְּנָה׳, ׳שִׁירוּ לַה׳ שִׁיר חָדָשׁ׳, ׳ה׳ מָלָךְ תָּגֵל הָאָרֶץ׳. ה׳ הוּא הָרוֹעֶה שֶׁלָּנוּ, וְהוּא רוֹצֶה שֶׁנַּכְנִיס אֶת הַשַּׁבָּת בְּשִׂמְחָה.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳נְקַדְּמָה פָנָיו בְּתוֹדָה׳ – מָה הַדָּבָר הַטּוֹב שֶׁקָּרָה לִי הַשָּׁבוּעַ וְעַל מָה אֲנִי רוֹצֶה לְהוֹדוֹת לַה׳ בְּלֵיל שַׁבָּת?
  • ׳הַיּוֹם אִם בְּקֹלוֹ תִשְׁמָעוּ׳ – מָה ה׳ מְבַקֵּשׁ מִמֶּנִּי הַיּוֹם, וְאֵיךְ אֶשְׁמַע אֶת הַקּוֹל הַזֶּה?
  • ׳שִׁירוּ לַה׳ שִׁיר חָדָשׁ׳ – מָה הַשִּׁיר הַחָדָשׁ שֶׁלִּי? אֵיזֶה דָּבָר חָדָשׁ אֲנִי יָכוֹל לְהוֹדוֹת עָלָיו הַשָּׁבוּעַ?

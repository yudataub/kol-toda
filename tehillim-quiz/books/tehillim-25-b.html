נֵר לְרַגְלִי דְבָרֶךָ וְאוֹר לִנְתִיבָתִי

🎵 סֵפֶר תְּהִלִּים · מָה אָהַבְתִּי תּוֹרָתֶךָ · רָמָה ב׳

מָה אָהַבְתִּי תּוֹרָתֶךָ

רָמָה ב׳ – עִיּוּן בְּאַהֲבַת הַתּוֹרָה

הַמִּזְמוֹר הָאָרֹךְ בְּיוֹתֵר בַּתְּהִלִּים – שִׁיר אַהֲבָה לַתּוֹרָה, בְּסֵדֶר הָאָלֶף־בֵּית: נֵר לְרַגְלִי, מָתוֹק מִדְּבַשׁ, וּמְחַפֵּשׂ שֶׂה אוֹבֵד

📖 מִזְמוֹר קי״ט

🎯 לַחִידוֹן

📜 הַפָּסוּקבַּמֶּה יְזַכֶּה נַּעַר אֶת אָרְחוֹ לִשְׁמֹר כִּדְבָרֶךָ

💛 אַהֲבַת הַשֵּׁם: ה׳ עוֹזֵר לִי לְהַתְחִיל בַּדֶּרֶךְ הַטּוֹבָה כְּבָר עַכְשָׁיו 💛

📖 תְּהִלִּים, פֶּרֶק קי״ט, פָּסוּק ט׳ (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בְּאַהֲבַת הַתּוֹרָה

🚶 מְצוּדַת דָּוִד: מִנְּעוּרִים

׳בַּמֶּה יְזַכֶּה נַּעַר אֶת אָרְחוֹ לִשְׁמֹר כִּדְבָרֶךָ׳ – מְצוּדַת דָּוִד מְבָאֵר: ׳בְּאֵיזֶה עִנְיָן יַעֲשֶׂה אֶת אָרְחוֹ זַכַּאי בְּעֵת יַרְגִּיל עַצְמוֹ מִנְּעוּרָיו לָלֶכֶת בְּאֹרַח מִישׁוֹר׳. וְ׳לִשְׁמֹר כִּדְבָרֶךָ – לֹא יֹאמַר אִי אִפְשִׁי בַּעֲבֵרָה אֶלָּא יֹאמַר אִפְשִׁי בָּהּ אֲבָל אֶשְׁמֹר בִּדְבַר ה׳׳. כְּלוֹמַר: הַתְחָלָה בִּגְיל צָעִיר, וְלֹא לוֹמַר ״אֵינִי רוֹצֶה״ אֶלָּא לוֹמַר ״אֲנִי רוֹצֶה, וְאָשִׁיב לַה׳״.

מִתְרַגְּלִים מִנְּעוּרִים לָלֶכֶת בָּדֶרֶךְ הַיְשָׁרָה, וּמְנַצְּחִים אֶת הַיֵּצֶר כְּשֶׁרוֹצִים אוֹתוֹ.

1

📜 הַפָּסוּקבְּלִבִּי צָפַנְתִּי אִמְרָתֶךָ לְמַעַן לֹא אֶחֱטָא לָךְ

💛 אַהֲבַת הַשֵּׁם: ה׳ עוֹזֵר לִי לִצְפֹּן בַּלֵּב דְּבָרִים טוֹבִים 💛

📖 תְּהִלִּים, פֶּרֶק קי״ט, פָּסוּק י״א (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בְּאַהֲבַת הַתּוֹרָה

🔐 רַשִׁ״י: ׳בְּלִבִּי צָפַנְתִּי׳

׳בְּלִבִּי צָפַנְתִּי אִמְרָתֶךָ לְמַעַן לֹא אֶחֱטָא לָךְ׳ – רַשִׁ״י: ׳לֹא נְתַתִּיהָ לְהִשְׁתַּכֵּחַ מִמֶּנִּי׳. מְצוּדַת דָּוִד: ׳שְׁמוּרִים הֵם בְּלִבִּי, לְמַעַן לֹא אֶחֱטָא לְךָ, כִּי אִם אֶשְׁכַּח מָה מֵהֶם אֶהְיֶה נִלְכָּד בְּחֵטְא בִּבְלִי דַעַת׳. הַלֵּב הוּא כְּמוֹ כַּסֶּפֶת: מַה שֶּׁשָּׁמוּר בָּהּ לֹא יֵאָבֵד. מִי שֶׁזּוֹכֵר דִּבְרֵי תּוֹרָה לֹא יֶחֱטָא מִתּוֹךְ חוֹסֶר יְדִיעָה.

זוֹכְרִים אֶת הַתּוֹרָה בַּלֵּב כְּדֵי לֹא לְהִכָּשֵׁל בְּחֵטְא בִּבְלִי דַעַת.

2

📜 הַפָּסוּקגַּל עֵינַי וְאַבִּיטָה נִפְלָאוֹת מִתּוֹרָתֶךָ

💛 אַהֲבַת הַשֵּׁם: ה׳ מְגַלֶּה לִי אֶת הָאוֹצָרוֹת שֶׁבַּתּוֹרָה 💛

📖 תְּהִלִּים, פֶּרֶק קי״ט, פָּסוּק י״ח (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בְּאַהֲבַת הַתּוֹרָה

🔍 רַשִׁ״י: ׳נִפְלָאוֹת מִתּוֹרָתֶךָ׳

׳גַּל עֵינַי וְאַבִּיטָה נִפְלָאוֹת מִתּוֹרָתֶךָ׳ – רַשִׁ״י מְבָאֵר: ׳דְּבָרִים הַמְכֻסִּים בָּהּ שֶׁאֵינָם מְפֹרָשִׁים בָּהּ׳. מְצוּדַת דָּוִד: ׳הָאֵר עֵינַי לְהָבִין מִתּוֹרָתְךָ דְּבָרִים נִפְלָאִים וּמְכֻסִּים׳. בַּתּוֹרָה יֵשׁ דְּבָרִים גְּלוּיִים, וְיֵשׁ דְּבָרִים כְּמוֹ אוֹצָר חָבוּי. מִי שֶׁלּוֹמֵד בְּעֵינַיִם פְּתוּחוֹת מוֹצֵא אוֹתָם.

בַּתּוֹרָה יֵשׁ דְּבָרִים גְּלוּיִים וּדְבָרִים מְכֻסִּים, וְצָרִיךְ לְבַקֵּשׁ שֶׁה׳ יִפְתַּח אֶת הָעֵינַיִם.

3

📜 הַפָּסוּקמַה נִּמְלְצוּ לְחִכִּי אִמְרָתֶךָ מִדְּבַשׁ לְפִי

💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה מְתוּקָה לִי, וַאֲנִי אוֹהֵב לַחֲזֹר אֵלֶיהָ 💛

📖 תְּהִלִּים, פֶּרֶק קי״ט, פְּסוּקִים צ״ז–ק״ג (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בְּאַהֲבַת הַתּוֹרָה

🍬 רַשִׁ״י: ׳נִמְלְצוּ׳ – נִמְתְּקוּ

׳מַה נִּמְלְצוּ לְחִכִּי אִמְרָתֶךָ מִדְּבַשׁ לְפִי׳ – רַשִׁ״י: ׳נִמְלְצוּ – נִמְתְּקוּ׳. מְצוּדַת דָּוִד: ׳מַה מְּאֹד נֶחְשְׁבוּ אִמְרָתְךָ לְחִכִּי לְדִבְרֵי מְלִיצָה, וְהֵם עֲרֵבִים לִי יוֹתֵר מֵעֲרֵבוּת הַדְּבַשׁ לַפֶּה׳. ׳מָה אָהַבְתִּי׳ – ׳מָה מְאֹד אָהַבְתִּי הַתּוֹרָה עַד שֶׁכָּל הַיּוֹם הִיא מַאֲמַר פִּי׳. הַתּוֹרָה לֹא חוֹבָה כְּבֵדָה – הִיא מַתָּק, וְהַחִיךְ טוֹעֵם אוֹתָהּ.

הַתּוֹרָה עֲרֵבָה לַפֶּה יוֹתֵר מִדְּבַשׁ.

4

📜 הַפָּסוּקנֵר לְרַגְלִי דְבָרֶךָ וְאוֹר לִנְתִיבָתִי

💛 אַהֲבַת הַשֵּׁם: ה׳ מֵאִיר לִי אֶת הַצַּעַד הַבָּא 💛

📖 תְּהִלִּים, פֶּרֶק קי״ט, פָּסוּק ק״ה (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בְּאַהֲבַת הַתּוֹרָה

🔦 רַשִׁ״י וּמְצוּדַת דָּוִד: ׳נֵר לְרַגְלִי׳

׳נֵר לְרַגְלִי דְבָרֶךָ׳ – רַשִׁ״י: ׳כְּשֶׁאֲנִי בָּא לְהוֹרוֹת הוֹרָאָה, אֲנִי רוֹאֶה בַּתּוֹרָה וְהִיא מְפָרֶשֶׁת אוֹתִי מִן הָאָסוּר, כַּנֵּר הַמַּצִּיל אֶת הָאָדָם מִן הַפְּחָתִים׳. מְצוּדַת דָּוִד: ׳כְּמוֹ הַנֵּר מַצִּיל הוּא בַּחֲשֵׁכַת הַלַּיְלָה מִכָּל מִכְשׁוֹל לְבַל תָּגֹף הָרֶגֶל, כֵּן דְּבָרְךָ מַצִּיל אוֹתִי מִכָּל מִכְשׁוֹל עֲבֵרָה׳. הַנֵּר הוּא לֹא לְכָל הַדֶּרֶךְ – הוּא לָרֶגֶל, צַעַד אַחַר צַעַד.

הַתּוֹרָה מְאִירָה אֶת הַצַּעַד הַבָּא, וּמַצִּילָה מִמִּכְשׁוֹל.

5

📜 הַפָּסוּקפֵּתַח דְּבָרֶיךָ יָאִיר מֵבִין פְּתָיִים

💛 אַהֲבַת הַשֵּׁם: ה׳ פּוֹתֵחַ לִי אֶת הַלֵּב כְּשֶׁאֲנִי מַתְחִיל לִלְמֹד 💛

📖 תְּהִלִּים, פֶּרֶק קי״ט, פָּסוּק ק״ל (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בְּאַהֲבַת הַתּוֹרָה

🌅 רַשִׁ״י: ׳פֵּתַח דְּבָרֶיךָ יָאִיר׳

׳פֵּתַח דְּבָרֶיךָ יָאִיר מֵבִין פְּתָיִים׳ – רַשִׁ״י מְבָאֵר: ׳תְּחִלַּת דְּבָרֶיךָ הֵאִירוּ אֶת לֵב יִשְׂרָאֵל, שֶׁאַתָּה הַמֵּבִין פְּתָאִים בְּאָמְרְךָ ״אָנֹכִי אֲשֶׁר הוֹצֵאתִיךָ״, הוֹדַעְתָּם בְּטוֹבָה שֶׁעָשִׂיתָ לָהֶם׳; וְ׳מֵבִין פְּתָיִים – מִשָּׁם יָבִינוּ הַכֹּל וְיִפְתְּחוּ בְּדִבְרֵי תּוֹרָה׳. מְצוּדַת דָּוִד: ׳כַּאֲשֶׁר יַתְחִיל הָאָדָם לְהִתְבּוֹנֵן בַּתּוֹרָה יָאִירוּ עֵינָיו וְיוֹסִיף תְּבוּנָה בְּכָל עֵת׳. הַתּוֹרָה הִתְחִילָה בְּ״אָנֹכִי״ – טוֹבָה אֲמִתִּית.

כְּשֶׁמַּתְחִילִים לִלְמֹד, הָאוֹר נִכְנָס, וְהַתּוֹרָה מַתְחִילָה בְּ׳אָנֹכִי׳ – ה׳ שֶׁהוֹצִיא אוֹתָנוּ.

6

📜 הַפָּסוּקתָּעִיתִי כְּשֶׂה אֹבֵד בַּקֵּשׁ עַבְדֶּךָ כִּי מִצְוֺתֶיךָ לֹא

💛 אַהֲבַת הַשֵּׁם: ה׳ הָרוֹעֶה שֶׁלִּי מְחַפֵּשׂ אוֹתִי כְּשֶׁאֲנִי תּוֹעֶה 💛

📖 תְּהִלִּים, פֶּרֶק קי״ט, פָּסוּק קע״ו (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בְּאַהֲבַת הַתּוֹרָה

🐑 מְצוּדַת דָּוִד: הַשֶּׂה וְהָרוֹעֶה

׳תָּעִיתִי כְּשֶׂה אֹבֵד בַּקֵּשׁ עַבְדֶּךָ׳ – מְצוּדַת דָּוִד מְבָאֵר: ׳כְּמוֹ הַשֶּׂה הַזֶּה כְּשֶׁהוּא אָבוּד מִן הָעֵדֶר אֵינוֹ יוֹדֵעַ לַחֲזֹר עַד שֶׁיְּבַקְּשׁוּהוּ הָרוֹעֶה, כֵּן תוֹעֶה אֲנִי בְּדַעְתִּי וְלֹא אֵדַע מֵעַצְמִי׳. כְּלוֹמַר: הַשֶּׂה לֹא יָכוֹל לַחֲזֹר לְבַד – הוּא צָרִיךְ אֶת הָרוֹעֶה. וְדָוִד אוֹמֵר: אֲנִי צָרִיךְ אֶת ה׳ שֶׁיְּחַפֵּשׂ אוֹתִי.

כְּשֶׁאָדָם תּוֹעֶה, הוּא צָרִיךְ אֶת הָרוֹעֶה שֶׁיְּחַפֵּשׂ אוֹתוֹ.

7
💡

רָמָה ב׳ – עִיּוּן בְּאַהֲבַת הַתּוֹרָה

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מַסְבִּירִים: מָה זֶה לִצְפֹּן, מָה הַנִּפְלָאוֹת שֶׁבַּתּוֹרָה, אֵיךְ נֵר מַצִּיל מִמִּכְשׁוֹל, וּמָה הַקֶּשֶׁר בֵּין הַתּוֹרָה לַרוֹעֶה שֶׁמְּחַפֵּשׂ אֶת הַשֶּׂה. הַתּוֹרָה וְהָאַהֲבָה לָהּ מֵאִירוֹת אֶת הַדֶּרֶךְ.

נֵר לְרַגְלִי דְבָרֶךָ וְאוֹר לִנְתִיבָתִי

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳בְּלִבִּי צָפַנְתִּי אִמְרָתֶךָ׳ – מָה הַפָּסוּק אוֹ הַדָּבָר מִן הַתּוֹרָה שֶׁאֲנִי רוֹצֶה ׳לִצְפֹּן׳ בַּלֵּב הַשָּׁבוּעַ?
  • ׳נֵר לְרַגְלִי דְבָרֶךָ׳ – מָתַי הַתּוֹרָה הִדְרִיכָה אוֹתִי בְּהַחְלָטָה קָשָׁה? אֵיפֹה אֲנִי צָרִיךְ עוֹד אוֹר?
  • ׳תָּעִיתִי כְּשֶׂה אֹבֵד בַּקֵּשׁ עַבְדֶּךָ׳ – מָתַי הִרְגַּשְׁתִּי אָבוּד, וּמָה עָזַר לִי לַחֲזֹר?

רָמָה ג׳ – מַעֲמִיקִים

בְּנֵי קֹרַח שׁוֹמְרִים עַל הַכְּמִיהָה לַה׳ גַּם בַּגָּלוּת: נֶפֶשׁ צְמֵאָה כְּמוֹ אַיָּל שֶׁצָּמֵא לְמַיִם, וְה׳ מַרְוֶה אוֹתָהּ

📜 הַפָּסוּקכְּאַיָּל תַּעֲרֹג עַל אֲפִיקֵי מָיִם כֵּן נַפְשִׁי תַעֲרֹג אֵלֶיךָ אֱלֹהִים צָמְאָה נַפְשִׁי לֵאלֹהִים לְאֵל

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי מַרְגִּישׁ רָצוֹן לְהִתְקָרֵב לַה׳, אֲנִי אוֹמֵר: ׳אֵלָיו, עַכְשָׁיו׳ 💛

📖 תְּהִלִּים, פֶּרֶק מ״ב, פָּסוּק ב׳

רָמָה ג׳ – מַעֲמִיקִים

🕊️ הַכְּמִיהָה – קוֹל הַנְּשָׁמָה

אֵין יְהוּדִי שֶׁלֹּא מַרְגִּישׁ מִדֵּי פַּעַם הִתְעוֹרְרוּת – רָצוֹן לְהִתְקָרֵב לַה׳. מֵאַיִן זֶה בָּא? מִן הַנְּשָׁמָה, שֶׁמִּשְׁתּוֹקֶקֶת לְהִתְחַבֵּר לַקָּדוֹשׁ בָּרוּךְ הוּא. גַּם אֲנָשִׁים רְחוֹקִים מְאֹד מַרְגִּישִׁים לִפְעָמִים כְּמִיהָה רוּחָנִית, וְהַרְבֵּה חוֹזְרִים בִּתְשׁוּבָה מִשָּׁם. ׳כְּאַיָּל תַּעֲרֹג׳ אֵינוֹ סְתָם דִּמּוּי: הָאַיָּל צָמֵא לְמַיִם וּבִלְעָדֵיהֶם יָמוּת – כָּךְ הַנֶּפֶשׁ צְמֵאָה לְמָזוֹן שֶׁל תּוֹרָה וּמִצְוֹת. לָכֵן צָרִיךְ לִתְפֹּס אֶת הָרְגָעִים הָאֵלֶּה וְלֹא לָתֵת לָהֶם לַעֲבֹר.

רְגָעֵי הַכְּמִיהָה לַה׳ הֵם הַנְּשָׁמָה שֶׁצְּמֵאָה – צָרִיךְ לָתֹפְסָם.

1
🙏

📜 הַפָּסוּקכְּאַיָּל תַּעֲרֹג עַל אֲפִיקֵי מָיִם כֵּן נַפְשִׁי תַעֲרֹג אֵלֶיךָ אֱלֹהִים

💛 אַהֲבַת הַשֵּׁם: אֲנִי קוֹרֵא לַמָּקוֹר, וְה׳ הוּא הַמָּקוֹר 💛

📖 תְּהִלִּים, פֶּרֶק מ״ב, פָּסוּק ב׳ (רַשִׁ״י)

רָמָה ג׳ – מַעֲמִיקִים

🙏 מִי שֶׁצּוֹעֵק – ה׳ מְרַחֵם

רַשִׁ״י מֵבִיא אֶת דִּבְרֵי רַבּוֹתֵינוּ עַל הָאַיֶּלֶת הַחֲסִידָה: כְּשֶׁהַחַיּוֹת צְמֵאוֹת, הֵן מִתְכַּנְּסוֹת אֵלֶיהָ, הִיא תּוֹלָה עֵינֶיהָ לַמָּרוֹם, חוֹפֶרֶת גּוּמָּא, וְגוֹעָה, וְה׳ מְרַחֵם וְהַתְּהוֹם מַעֲלָה מַיִם. מִי שֶׁמִּתְפַּלֵּל בִּשְׁבִיל אֲחֵרִים וְצוֹעֵק אֶל ה׳ – תְּפִלָּתוֹ מַעֲלָה גַּם לַאֲחֵרִים מַיִם. וְיֵשׁ בָּזֶה גַּם בִּטָּחוֹן: כְּשֶׁאֲנִי צָמֵא, אֲנִי לֹא מְחַפֵּשׂ מַיִם לְבַדִּי – אֲנִי קוֹרֵא לַמָּקוֹר.

תְּפִלָּה שֶׁל אֶחָד מַעֲלָה מַיִם לְכֻלָּם.

2
📖

📜 הַפָּסוּקמַה תִּשְׁתּוֹחֲחִי נַפְשִׁי וּמַה תֶּהֱמִי עָלָי הוֹחִילִי לֵאלֹהִים כִּי עוֹד אוֹדֶנּוּ יְשׁוּעֹת פָּנַי וֵאלֹהָי

💛 אַהֲבַת הַשֵּׁם: הַגַּעְגּוּעִים שֶׁלִּי הֵם הַדֶּרֶךְ שֶׁבָּהּ ה׳ מוֹשֵׁךְ אוֹתִי אֵלָיו 💛

📖 תְּהִלִּים, פֶּרֶק מ״ב, פְּסוּקִים ו׳, י״ב

רָמָה ג׳ – מַעֲמִיקִים

📖 מָה הַמִּזְמוֹר מְלַמֵּד אוֹתִי הַיּוֹם

הַמִּזְמוֹר מְלַמֵּד שֶׁהַגַּעְגּוּעִים לַה׳ הֵם סִימָן טוֹב, שֶׁאֶפְשָׁר לְדַבֵּר אֶל הַנֶּפֶשׁ וּלְחַזֵּק אוֹתָהּ ׳הוֹחִילִי לֵאלֹהִים׳, וְשֶׁזִּכָּרוֹן שֶׁל שִׂמְחָה עִם ה׳ נוֹתֵן כֹּחַ בַּיָּמִים הַקָּשִׁים.

הַגַּעְגּוּעִים לַה׳ הֵם מַתָּנָה, וְהַזִּכָּרוֹן הַטּוֹב מְחַזֵּק.

3
💡

רָמָה ג׳ – מַעֲמִיקִים

מָה לוֹקְחִים מִרָמָה ג׳?

הַכְּמִיהָה הַפִּתְאוֹמִית לְהִתְקָרֵב לַה׳ הִיא קוֹל הַנְּשָׁמָה, וְחֲזַ״ל בְּרַשִׁ״י לִמְּדוּ מֵהָאַיֶּלֶת הַחֲסִידָה אֵיךְ ה׳ מְרַחֵם עַל מִי שֶׁצּוֹעֵק אֵלָיו.

סוֹף רָמָה ג׳

🤔 חוֹשְׁבִים עַל זֶה

  • אַיָּל צָמֵא לְמַיִם כִּי בִּלְעָדֵיהֶם לֹא יִחְיֶה. מָה הַנֶּפֶשׁ שֶׁלִּי צְמֵאָה לוֹ בֶּאֱמֶת?
  • הַמִּזְמוֹר אוֹמֵר לַנֶּפֶשׁ: ׳מַה תִּשְׁתּוֹחֲחִי... הוֹחִילִי לֵאלֹהִים׳. מָתַי דִּבַּרְתִּי אֶל עַצְמִי בְּעִידוֹדִים כָּאֵלֶּה?
  • הַמִּזְמוֹר זוֹכֵר עֲלִיָּה לָרֶגֶל עִם הָמוֹן חוֹגֵג. מָה הַזִּכָּרוֹן הַטּוֹב שֶׁלִּי שֶׁמְּחַזֵּק אוֹתִי כְּשֶׁקָּשֶׁה?

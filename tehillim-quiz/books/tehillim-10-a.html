📖 תְּהִלִּים, פֶּרֶק ל׳, פָּסוּק א׳

רָמָה א׳ – מִבֶּכִי לְרִנָּה

🏠 ׳שִׁיר חֲנֻכַּת הַבַּיִת׳

הַמִּזְמוֹר נִקְרָא ׳מִזְמוֹר שִׁיר חֲנֻכַּת הַבַּיִת לְדָוִד׳. הוּא שִׁיר שֶׁל חֲנֻכָּה – שֶׁל הַתְחָלָה חֲדָשָׁה. דָּוִד שָׁר עַל הַבַּיִת שֶׁל ה׳ וְגַם עַל הַחַיִּים שֶׁלּוֹ.

הַמִּזְמוֹר הוּא שִׁיר שֶׁל הַתְחָלָה חֲדָשָׁה.

1
⬆️

💛 אַהֲבַת הַשֵּׁם: ה׳ מֵרִים אוֹתִי כְּשֶׁאֲנִי נוֹפֵל 💛

📖 תְּהִלִּים, פֶּרֶק ל׳, פָּסוּק ב׳

רָמָה א׳ – מִבֶּכִי לְרִנָּה

⬆️ ׳אֲרוֹמִמְךָ ה׳ כִּי דִלִּיתָנִי׳

דָּוִד פּוֹתֵחַ: ׳אֲרוֹמִמְךָ ה׳ כִּי דִלִּיתָנִי וְלֹא שִׂמַּחְתָּ אֹיְבַי לִי׳. ׳דִּלִּיתָנִי׳ – הֶעֱלִיתָנִי, כְּמוֹ דְּלִי שֶׁמַּעֲלִים מִן הַבְּאֵר. ה׳ הֶעֱלָה אוֹתוֹ וְלֹא נָתַן לָאוֹיְבִים לִשְׂמֹחַ.

ה׳ הֶעֱלָה אוֹתִי מִן הַבּוֹר וְלֹא נָתַן לָאוֹיְבִים לִשְׂמֹחַ.

2
🩹

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי קוֹרֵא לַה׳, הוּא רוֹפֵא אוֹתִי 💛

📖 תְּהִלִּים, פֶּרֶק ל׳, פְּסוּקִים ג׳–ד׳

רָמָה א׳ – מִבֶּכִי לְרִנָּה

🩹 ׳שִׁוַּעְתִּי אֵלֶיךָ וַתִּרְפָּאֵנִי׳

׳ה׳ אֱלֹהָי שִׁוַּעְתִּי אֵלֶיךָ וַתִּרְפָּאֵנִי ה׳ הֶעֱלִיתָ מִן שְׁאוֹל נַפְשִׁי חִיִּיתַנִי׳. דָּוִד קָרָא לַה׳ בְּקוֹל, וְה׳ רִפֵּא אוֹתוֹ וְהֶחֱיָה אוֹתוֹ מֵחָדָשׁ.

הַקְּרִיאָה לַה׳ מְבִיאָה רְפוּאָה וְחַיִּים.

3
🎶

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְשַׁתֵּף אֲחֵרִים בְּשִׂמְחָה שֶׁל תּוֹדָה 💛

📖 תְּהִלִּים, פֶּרֶק ל׳, פָּסוּק ה׳

רָמָה א׳ – מִבֶּכִי לְרִנָּה

🎶 ׳זַמְּרוּ לַה׳ חֲסִידָיו׳

דָּוִד מַזְמִין אֲחֵרִים: ׳זַמְּרוּ לַה׳ חֲסִידָיו וְהוֹדוּ לְזֵכֶר קָדְשׁוֹ׳. כְּשֶׁטּוֹב לוֹ, הוּא לֹא שָׁר לְבַד – הוּא מַזְמִין אֶת כֻּלָּם לָשִׁיר יַחַד.

מַזְמִינִים אֶת כֻּלָּם לְהוֹדוֹת יַחַד.

4
⏱️

💛 אַהֲבַת הַשֵּׁם: הָאַהֲבָה שֶׁל ה׳ אֵלַי גְּדוֹלָה יוֹתֵר מִכָּל כַּעַס 💛

📖 תְּהִלִּים, פֶּרֶק ל׳, פָּסוּק ו׳

רָמָה א׳ – מִבֶּכִי לְרִנָּה

⏱️ ׳כִּי רֶגַע בְּאַפּוֹ׳

׳כִּי רֶגַע בְּאַפּוֹ חַיִּים בִּרְצוֹנוֹ׳. הַכַּעַס שֶׁל ה׳ הוּא רַק רֶגַע קָצָר, וְהָרָצוֹן שֶׁלּוֹ – חַיִּים אֲרֻכִּים. הַטּוֹב שֶׁלּוֹ הוּא הַרְבֵּה יוֹתֵר מִן הַכַּעַס.

הַכַּעַס רֶגַע, וְהָרָצוֹן שֶׁל ה׳ חַיִּים.

5
🌅

💛 אַהֲבַת הַשֵּׁם: גַּם אַחֲרֵי יוֹם עָצוּב ה׳ מֵבִיא לִי בֹּקֶר חָדָשׁ 💛

📖 תְּהִלִּים, פֶּרֶק ל׳, פָּסוּק ו׳

רָמָה א׳ – מִבֶּכִי לְרִנָּה

🌅 ׳בָּעֶרֶב יָלִין בֶּכִי׳

הַמִּשְׁפָּט הַמֻּכָּר: ׳בָּעֶרֶב יָלִין בֶּכִי וְלַבֹּקֶר רִנָּה׳. הַלַּיְלָה יָכוֹל לִהְיוֹת קָשֶׁה וְעָצוּב, אֲבָל הַבֹּקֶר בָּא עִם שִׂמְחָה. הַצַּעַר לֹא נִשְׁאָר לָנוּ לְתָמִיד.

אַחֲרֵי לַיְלָה שֶׁל בֶּכִי בָּא בֹּקֶר שֶׁל רִנָּה.

6
😨

💛 אַהֲבַת הַשֵּׁם: כָּל טוֹב שֶׁיֵּשׁ לִי הוּא מַתָּנָה מֵה׳ 💛

📖 תְּהִלִּים, פֶּרֶק ל׳, פְּסוּקִים ז׳–ח׳

רָמָה א׳ – מִבֶּכִי לְרִנָּה

😨 ׳הִסְתַּרְתָּ פָנֶיךָ הָיִיתִי נִבְהָל׳

דָּוִד מְסַפֵּר: ׳וַאֲנִי אָמַרְתִּי בְשַׁלְוִי בַּל אֶמּוֹט לְעוֹלָם... הִסְתַּרְתָּ פָנֶיךָ הָיִיתִי נִבְהָל׳. כְּשֶׁהָיָה לוֹ טוֹב, חָשַׁב שֶׁלֹּא יִפֹּל לְעוֹלָם, וּכְשֶׁה׳ הִסְתִּיר פָּנָיו, נִבְהַל. הוּא לָמַד: הַכֹּל מֵה׳.

כְּשֶׁטּוֹב לָנוּ אָסוּר לִשְׁכֹּחַ שֶׁהַכֹּל מֵה׳.

7
💃

💛 אַהֲבַת הַשֵּׁם: ה׳ מְשַׂמֵּחַ אוֹתִי, וַאֲנִי שָׁר לוֹ 💛

📖 תְּהִלִּים, פֶּרֶק ל׳, פְּסוּקִים י״ב–י״ג

רָמָה א׳ – מִבֶּכִי לְרִנָּה

💃 ׳הָפַכְתָּ מִסְפְּדִי לְמָחוֹל׳

הַסּוֹף הַמְּשַׂמֵּחַ: ׳הָפַכְתָּ מִסְפְּדִי לְמָחוֹל לִי פִּתַּחְתָּ שַׂקִּי וַתְּאַזְּרֵנִי שִׂמְחָה לְמַעַן יְזַמֶּרְךָ כָבוֹד וְלֹא יִדֹּם׳. ה׳ הִתִּיר אֶת הַשַּׂק, הִלְבִּישׁ שִׂמְחָה, וְדָוִד מַבְטִיחַ: לְעוֹלָם לֹא אֶשְׁתֹּק מֵהַתּוֹדָה.

ה׳ הוֹפֵךְ אֶת הַצַּעַר לְרִקּוּד, וַאֲנִי לֹא שׁוֹתֵק מִתּוֹדָה.

8
💡

מָה לוֹקְחִים מִרָמָה א׳?

הַמִּזְמוֹר הַזֶּה הוּא שִׁיר תּוֹדָה: דָּוִד נָפַל, ה׳ הֶעֱלָה אוֹתוֹ, וְעַכְשָׁיו הוּא שָׁר. הוּא מְלַמֵּד שֶׁגַּם אַחֲרֵי לַיְלָה שֶׁל בֶּכִי בָּא בֹּקֶר שֶׁל רִנָּה, וְשֶׁה׳ אוֹהֵב לִשְׁמֹעַ אוֹתָנוּ שָׁרִים לוֹ תּוֹדָה.

🤔 חוֹשְׁבִים עַל זֶה

  • דָּוִד אוֹמֵר: ׳בָּעֶרֶב יָלִין בֶּכִי וְלַבֹּקֶר רִנָּה׳. מָתַי הָיָה לִי יוֹם עָצוּב שֶׁנִּגְמַר בְּשִׂמְחָה, וְעַל מָה אֲנִי מוֹדֶה לַה׳ עַכְשָׁיו?
  • דָּוִד אוֹמֵר שֶׁבִּזְמַן שְׁלֵוָה חָשַׁב שֶׁלֹּא יִפֹּל לְעוֹלָם, וְאָז ה׳ הִסְתִּיר פָּנָיו וְהוּא נִבְהַל. מָה זֶה מְלַמֵּד אוֹתִי עַל הַבִּטָּחוֹן שֶׁלִּי?
  • ׳לְמַעַן יְזַמֶּרְךָ כָבוֹד וְלֹא יִדֹּם׳. אֵיךְ אֲנִי יָכוֹל לְהַשְׁתַּמֵּשׁ בַּחַיִּים שֶׁנָּתַן לִי ה׳ כְּדֵי לְהוֹדוֹת לוֹ?

לַה׳ הָאָרֶץ וּמְלוֹאָהּ

רָמָה ב׳ – מַה מְּבָאֲרִים הַמְּפָרְשִׁים

דָּוִד שׁוֹאֵל מִי יָכוֹל לַעֲלוֹת לְהַר ה׳, וְעוֹנֶה: מִי שֶׁיָּדָיו נְקִיּוֹת וְלִבּוֹ טָהוֹר – וְאָז הַשְּׁעָרִים נִפְתָּחִים לְמֶלֶךְ הַכָּבוֹד

📖 מִזְמוֹר כ״ד

🎯 לַחִידוֹן
🗺️

📜 הַפָּסוּקלְדָוִד מִזְמוֹר לַיהוָה הָאָרֶץ וּמְלוֹאָהּ תֵּבֵל וְיֹשְׁבֵי בָהּ

💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לָנוּ אֶרֶץ מְיֻחֶדֶת, וְהוּא אוֹהֵב אוֹתָהּ 💛

📖 תְּהִלִּים, פֶּרֶק כ״ד, פָּסוּק א׳ (רַשִׁ״י)

רָמָה ב׳ – מַה מְּבָאֲרִים הַמְּפָרְשִׁים

🗺️ רַשִׁ״י: ׳הָאָרֶץ׳ וְ׳תֵּבֵל׳

׳לַה׳ הָאָרֶץ וּמְלוֹאָהּ תֵּבֵל וְיֹשְׁבֵי בָהּ׳ – רַשִׁ״י מְבָאֵר: ׳לַה׳ הָאָרֶץ – אֶרֶץ יִשְׂרָאֵל. תֵּבֵל – שְׁאָר אֲרָצוֹת׳. כְּלוֹמַר, הַמִּזְמוֹר מְדַבֵּר גַּם עַל אֶרֶץ יִשְׂרָאֵל וְגַם עַל כָּל הָעוֹלָם: שְׁנֵיהֶם שֶׁל ה׳, אֲבָל לְאֶרֶץ יִשְׂרָאֵל יֵשׁ מַעֲלָה.

רַשִׁ״י: ׳הָאָרֶץ׳ הִיא אֶרֶץ יִשְׂרָאֵל, ׳תֵּבֵל׳ הֵן שְׁאָר הָאֲרָצוֹת.

1
🌊

📜 הַפָּסוּקכִּי הוּא עַל יַמִּים יְסָדָהּ וְעַל נְהָרוֹת יְכוֹנְנֶהָ

💛 אַהֲבַת הַשֵּׁם: ה׳ הֶעֱמִיד אֶת אֶרֶץ יִשְׂרָאֵל בִּגְבוּרָה 💛

📖 תְּהִלִּים, פֶּרֶק כ״ד, פָּסוּק ב׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – מַה מְּבָאֲרִים הַמְּפָרְשִׁים

🌊 רַשִׁ״י: שִׁבְעָה יַמִּים וְאַרְבָּעָה נְהָרוֹת

׳עַל יַמִּים יְסָדָהּ וְעַל נְהָרוֹת יְכוֹנְנֶהָ׳ – רַשִׁ״י מְבָאֵר: ׳שִׁבְעָה יַמִּים מַקִּיפִין אֶרֶץ יִשְׂרָאֵל, וְאַרְבַּע נְהָרוֹת: יַרְדֵּן, וְיַרְמוּךְ, קַרְמִיּוֹן וּפוּגָה׳. מְצוּדַת דָּוִד: ׳בְּדֶרֶךְ הַטֶּבַע הָיְתָה הָאָרֶץ יוֹרֶדֶת תַּחַת הַמַּיִם׳ – אֲבָל ה׳ הֶעֱמִיד אוֹתָהּ מִלְמַעְלָה.

רַשִׁ״י וּמְצוּדַת דָּוִד: אֶרֶץ יִשְׂרָאֵל מֻקֶּפֶת יַמִּים וּנְהָרוֹת, וְה׳ הֶעֱמִיד אוֹתָהּ מֵעַל הַמַּיִם.

🎯

📜 הַפָּסוּקמִי יַעֲלֶה בְהַר ה׳ וּמִי יָקוּם בִּמְקוֹם קָדְשׁוֹ

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְהִתְכּוֹנֵן כְּדֵי לְהִתְקָרֵב לַה׳ 💛

📖 תְּהִלִּים, פֶּרֶק כ״ד, פָּסוּק ג׳ (רַשִׁ״י)

רָמָה ב׳ – מַה מְּבָאֲרִים הַמְּפָרְשִׁים

🎯 רַשִׁ״י: אֵין כֻּלָּם כְּדַאי

׳מִי יַעֲלֶה בְהַר ה׳׳ – רַשִׁ״י מְבָאֵר: ׳אַף עַל פִּי שֶׁכָּל יוֹשְׁבֵי תֵבֵל שֶׁלּוֹ, אֵין הַכֹּל כְּדַאי לְהִקָּרֵב אֵלָיו אֶלָּא נְקִי כַפַּיִם וְגוֹ׳׳. מְצוּדַת דָּוִד: ׳עִם שֶׁכֻּלָּם הֵמָּה לַה׳, מִמָּקוֹם אֵין כֻּלָּם רְאוּיִים לָבוֹא בְּהַר ה׳׳. הַשְּׁאֵלָה לֹא ׳מִי מוּתָּר׳ אֶלָּא ׳מִי מוּכָן׳.

רַשִׁ״י: הַכֹּל שֶׁל ה׳, אֲבָל לְהִקָּרֵב אֵלָיו צָרִיךְ הֲכָנָה.

3
🙅

📜 הַפָּסוּקנְקִי כַפַּיִם וּבַר לֵבָב אֲשֶׁר לֹא נָשָׂא לַשָּׁוְא נַפְשִׁי וְלֹא נִשְׁבַּע לְמִרְמָה

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אֶת הַיָּדַיִם שֶׁלִּי כְּשֶׁהֵן נְקִיּוֹת 💛

📖 תְּהִלִּים, פֶּרֶק כ״ד, פָּסוּק ד׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – מַה מְּבָאֲרִים הַמְּפָרְשִׁים

🙅 רַשִׁ״י: ׳לֹא נָשָׂא לַשָּׁוְא נַפְשִׁי׳

׳אֲשֶׁר לֹא נָשָׂא לַשָּׁוְא נַפְשִׁי׳ – רַשִׁ״י מְבָאֵר: ׳לֹא נִשְׁבַּע בִּשְׁמִי וּבְנַפְשִׁי לַשָּׁוְא; מָצִינוּ לָשׁוֹן שְׁבוּעָה נוֹפֵל עַל נֶפֶשׁ, שֶׁנֶּאֱמַר ״נִשְׁבַּע ה׳ צְבָאוֹת בְּנַפְשׁוֹ״׳. מְצוּדַת דָּוִד: ׳נְקִי כַפַּיִם – מִמָּמוֹן שֶׁאֵינוֹ שֶׁל יֹשֶׁר, וְלִבּוֹ בָּרוּר בְּיִרְאַת ה׳ וְלֹא מִפַּחַד אָדָם׳.

מְצוּדַת דָּוִד: יָדַיִם נְקִיּוֹת מִכֶּסֶף לֹא כָּשֵׁר, וְלֵב בָּרוּר בְּיִרְאַת ה׳.

👥

📜 הַפָּסוּקזֶה דּוֹר דֹּרְשָׁיו מְבַקְשֵׁי פָנֶיךָ יַעֲקֹב סֶלָה

💛 אַהֲבַת הַשֵּׁם: ה׳ מְחַכֶּה לְמִי שֶׁדּוֹרֵשׁ אוֹתוֹ וּמְקַבֵּל אוֹתוֹ בְּאַהֲבָה 💛

📖 תְּהִלִּים, פֶּרֶק כ״ד, פָּסוּק ו׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – מַה מְּבָאֲרִים הַמְּפָרְשִׁים

👥 רַשִׁ״י: ׳זֶה דּוֹר דֹּרְשָׁיו׳

׳זֶה דּוֹר דֹּרְשָׁיו׳ – רַשִׁ״י מְבָאֵר: ׳זֶה – שֶׁכָּךְ מַעֲשָׂיו, הוּא דּוֹר דֹּרְשָׁיו׳. מְצוּדַת דָּוִד: ׳זֶה הַדּוֹר שֶׁהֵמָּה דּוֹרְשֵׁי ה׳, וְהֵמָּה קְרוּאִים זֶרַע יַעֲקֹב הָרְאוּיִים לְבַקֵּשׁ פָּנֶיךָ לְעוֹלָם וְלָבֹא לְבֵית הַמִּקְדָּשׁ׳. הַמְּבַקֵּשׁ הוּא מִי שֶׁעוֹשֶׂה אֶת מַה שֶּׁנֶּאֱמַר לְמַעְלָה.

רַשִׁ״י: ׳דּוֹר דֹּרְשָׁיו׳ הוּא מִי שֶׁמַּעֲשָׂיו כְּמוֹ שֶׁתֻּאַר בַּפְּסוּקִים.

5
🚪

📜 הַפָּסוּקשְׂאוּ שְׁעָרִים רָאשֵׁיכֶם וְהִנָּשְׂאוּ פִּתְחֵי עוֹלָם וְיָבוֹא מֶלֶךְ הַכָּבוֹד

💛 אַהֲבַת הַשֵּׁם: ה׳ זוֹכֵר אֶת הַחֶסֶד שֶׁל מִי שֶׁאָהַב אוֹתוֹ 💛

📖 תְּהִלִּים, פֶּרֶק כ״ד, פָּסוּק ז׳ (רַשִׁ״י)

רָמָה ב׳ – מַה מְּבָאֲרִים הַמְּפָרְשִׁים

🚪 רַשִׁ״י: הַשְּׁעָרִים בִּימֵי שְׁלֹמֹה

׳שְׂאוּ שְׁעָרִים רָאשֵׁיכֶם׳ – רַשִׁ״י מְבָאֵר: ׳בִּימֵי שְׁלֹמֹה בְנוֹ, כְּשֶׁבָּא לְהַכְנִיס אָרוֹן לְבֵית קָדְשֵׁי הַקֳּדָשִׁים וְדָבְקוּ שְׁעָרִים זוֹ בָּזוֹ, אָמַר כ״ד רְנָנוֹת וְלֹא נֶעֱנָה, עַד שֶׁאָמַר ״אַל תָּשֵׁב פְּנֵי מְשִׁיחֶךָ זָכְרָה לְחַסְדֵי דָוִד עַבְדֶּךָ״׳. הַשְּׁעָרִים נִפְתְּחוּ בִּזְכוּת דָּוִד – כִּי ה׳ זוֹכֵר אֶת הַחֶסֶד.

רַשִׁ״י: הַשְּׁעָרִים נִפְתְּחוּ לִשְׁלֹמֹה כְּשֶׁהִזְכִּיר אֶת חַסְדֵי דָוִד.

🏛️

📜 הַפָּסוּקשְׂאוּ שְׁעָרִים רָאשֵׁיכֶם וּשְׂאוּ פִּתְחֵי עוֹלָם וְיָבֹא מֶלֶךְ הַכָּבוֹד

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְחַכֶּה לַיּוֹם שֶׁבּוֹ כָּל הַשְּׁעָרִים יִפָּתְחוּ 💛

📖 תְּהִלִּים, פֶּרֶק כ״ד, פָּסוּק ט׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – מַה מְּבָאֲרִים הַמְּפָרְשִׁים

🏛️ מְצוּדַת דָּוִד: בַּיִת רִאשׁוֹן וְעָתִיד

׳שְׂאוּ שְׁעָרִים רָאשֵׁיכֶם וּשְׂאוּ פִּתְחֵי עוֹלָם׳ – רַשִׁ״י: ׳פְּתָחִים שֶׁקְּדֻשָּׁתָן עוֹלָמִית׳. מְצוּדַת דָּוִד: ׳כֵּן יִקָּרֵא פִּתְחֵי בֵּית הַמִּקְדָּשׁ, כִּי הַמִּשְׁכָּן לֹא הָיָה קָבוּעַ בְּמָקוֹם אֶחָד, אֲבָל בֵּית הַמִּקְדָּשׁ קָבוּעַ בִּמְקוֹמוֹ עַד עוֹלָם׳. וְעוֹד: ׳וְיִתָּכֵן שֶׁהָאֶחָד אָמוּר עַל בַּיִת רִאשׁוֹן וְהָאֶחָד עַל בַּיִת הֶעָתִיד׳. הַפָּסוּק חוֹזֵר פַּעֲמַיִם, וְיֵשׁ בּוֹ רֶמֶז גַּם לִגְאֻלָּה.

מְצוּדַת דָּוִד: ׳פִּתְחֵי עוֹלָם׳ – הַפָּסוּק חוֹזֵר, וְאוּלַי מִתְכַּוֵּן גַּם לַבַּיִת הֶעָתִיד.

מָה לוֹקְחִים מִרָמָה ב׳?

רַשִׁ״י וּמְצוּדַת דָּוִד מַסְבִּירִים מִי נִקְרָא ׳הָאָרֶץ׳ וּמִי ׳תֵבֵל׳, מָה פֵּרוּשׁ ׳נָשָׂא לַשָּׁוְא׳, וּמָה קָרָה בַּשְּׁעָרִים בִּימֵי שְׁלֹמֹה. בְּכָל אֶחָד מֵהֶם רוֹאִים אֶת הָרָצוֹן שֶׁל ה׳ שֶׁנִּתְקָרֵב.

🤔 חוֹשְׁבִים עַל זֶה

  • הָעוֹלָם כֻּלּוֹ שֶׁל ה׳. מָה אֲנִי יָכוֹל לַעֲשׂוֹת כְּדֵי לְהַרְגִּישׁ שֶׁהָעוֹלָם הוּא שֶׁל ה׳ וְאֲנִי אוֹרֵחַ בּוֹ?
  • ׳נְקִי כַפַּיִם וּבַר לֵבָב׳. מָה הַיָּדַיִם שֶׁלִּי עוֹשׂוֹת הַיּוֹם, וּמָה הַלֵּב שֶׁלִּי חוֹשֵׁב – וְאֵיךְ אֶפְשָׁר לְנַקּוֹת?

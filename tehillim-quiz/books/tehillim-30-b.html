הַמִּזְמוֹר הוֹלֵךְ מִכְּלָל אֶל פְּרָט: קוֹל הַהִלּוּל מַתְחִיל בַּשָּׁמַיִם וְיוֹרֵד אֶל כָּל בְּרִיָּה.

מִכּוֹכְבֵי הַלַּיְלָה וְעַד הַשָּׁמַיִם הָעֶלְיוֹנִים – כֻּלָּם מְהַלְּלִים.

׳וַיַּעֲמִידֵם לָעַד לְעוֹלָם חָק נָתַן וְלֹא יַעֲבוֹר׳ – רַשִׁ״י: ׳חֹק נָתַן בָּהֶם: זֶה יְשַׁמֵּשׁ בַּיּוֹם וְזֶה יְשַׁמֵּשׁ בַּלַּיְלָה, וְלֹא יַעֲבוֹר אוֹתוֹ חֹק׳. מְצוּדַת דָּוִד: ׳לְכָל אֶחָד נָתַן חֹק מָתַי יְשַׁמֵּשׁ זֶה וּמָתַי יְשַׁמֵּשׁ זֶה, וְלֹא יַעֲבוֹר הַחֹק הַהוּא׳. וְעַל ׳כִּי הוּא צִוָּה וְנִבְרָאוּ׳: ׳הוּא צִוָּה שֶׁיִּהְיוּ נִבְרָאִים, וְהִנֵּה נִבְרָאוּ׳. הַסֵּדֶר שֶׁבַּטֶּבַע הוּא שִׁיר, כִּי כָּל אֶחָד מְמַלֵּא אֶת הַתַּפְקִיד שֶׁלּוֹ.

הַסֵּדֶר בַּטֶּבַע – כָּל אֶחָד בְּזְמַנּוֹ – הוּא הַהִלּוּל שֶׁלּוֹ.

ה׳ נִשְׂגָּב מִכֹּל, וְיִשְׂרָאֵל הוּא ׳עַם קְרֹבוֹ׳.

׳כִּי רוֹצֶה ה׳ בְּעַמּוֹ יְפָאֵר עֲנָוִים בִּישׁוּעָה׳ – מְצוּדַת דָּוִד: ׳עַתָּה נִרְאֶה לַכֹּל כִּי רוֹצֶה ה׳ בְּעַמּוֹ, וְאֶת הָעֲנָוִים יְפָאֵר בִּישׁוּעָה׳. וְעַל ׳יַעְלְזוּ חֲסִידִים בְּכָבוֹד יְרַנְּנוּ עַל מִשְׁכְּבוֹתָם׳: ׳בַּעֲבוּר כָּבוֹד הַבָּא לַה׳׳, ו׳עַל מִשְׁכְּבוֹתָם – עַל מְנוּחָתָם, כִּי עַל הַמִּשְׁכָּב יָנוּחַ הָאָדָם, וְרוֹצֶה לוֹמַר יְרַנְּנוּ לַה׳ כְּשֶׁיִּהְיוּ בְּאַרְצָם בִּמְנוּחָה׳. הַחֲסִידִים שָׂרִים כְּשֶׁהֵם בְּאַרְצָם וּבִמְנוּחָה.

🔔 רַשִׁ״י וּמְצוּדַת דָּוִד: ׳צִלְצְלֵי שָׁמַע׳ וְ׳כֹּל הַנְּשָׁמָה׳

׳הַלְלוּהוּ בְצִלְצְלֵי שָׁמַע הַלְלוּהוּ בְּצִלְצְלֵי תְרוּעָה׳ – רַשִׁ״י: ׳מְצִלְתַּיִם הַמַּשְׁמִיעִים קוֹל׳. מְצוּדַת דָּוִד: ׳בִּכְלֵי נִגּוּן הַנִּקְרָא מְצִלְתַּיִם הַמַּשְׁמִיעַ קוֹל לְמֵרָחוֹק׳. וְהַסִּיכּוּם: ׳כֹּל הַנְּשָׁמָה תְּהַלֵּל יָהּ׳ – מְצוּדַת דָּוִד: ׳כָּל מִי שֶׁיֵּשׁ בּוֹ נְשָׁמָה וְנֶפֶשׁ מַשְׂכֶּלֶת תְּהַלֵּל אֶת יָהּ׳. כָּל כְּלֵי הַשִּׁיר הַחִיצוֹנִיִּים טוֹבִים, אֲבָל הַנְּשָׁמָה שֶׁבִּפְנִים הִיא הַמְּהַלֶּלֶת בֶּאֱמֶת.

אֶת סֵפֶר תְּהִלִּים מְסַיְּמִים ב׳כֹּל הַנְּשָׁמָה׳ – כָּל מִי שֶׁיֵּשׁ בּוֹ נְשָׁמָה.

רָמָה ב׳ – עִיּוּן בְּסוֹף הַסֵּפֶר

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מַסְבִּירִים אֶת מִי הֵם ׳כּוֹכְבֵי אוֹר׳, מָה זֶה ׳חָק נָתַן׳, מָה פֵּרוּשׁ ׳עַל מִשְׁכְּבוֹתָם׳ וּמָה הִיא ׳צִלְצְלֵי שָׁמַע׳. הַכֹּל מְחַבֵּר אֶל הַסְּפִירָה הָאַחֲרוֹנָה – הַנְּשָׁמָה.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳כֹּל הַנְּשָׁמָה תְּהַלֵּל יָהּ׳ – מָתַי הִרְגַּשְׁתִּי שֶׁהַנְּשִׁימָה עַצְמָהּ הִיא מַתָּנָה? אֵיךְ אֶפְשָׁר לְהוֹדוֹת עָלֶיהָ?
  • ׳הַלְלוּהוּ כָּל כּוֹכְבֵי אוֹר׳ – אִם גַּם הַשֶּׁמֶשׁ, הַיָּרֵחַ וְהָעֵצִים מְהַלְּלִים, אֵיךְ אֲנִי מְהַלֵּל אֶת ה׳ בְּדֶרֶךְ שֶׁלִּי?
  • ׳כִּי רוֹצֶה ה׳ בְּעַמּוֹ׳ – מָה אֲנִי מַרְגִּישׁ כְּשֶׁאֲנִי חוֹשֵׁב שֶׁה׳ רוֹצֶה בִּי?

׳וְחַטָּאתִי נֶגְדִּי תָמִיד׳ – רַשִׁ״י מְבָאֵר: ׳מִתּוֹךְ שֶׁאֲנִי מִתְחָרֵט וְדוֹאֵג עָלָיו דּוֹמֶה לִי כְּאִלּוּ הוּא לִפְנֵי תָּמִיד כָּל שָׁעָה׳. מְצוּדַת דָּוִד: ׳מַכִּיר אֲנִי שֶׁפָּשַׁעְתִּי וְתוֹהֶה אֲנִי עָלָיו... לִזְכֹּר בּוֹ וְלִדְאֹג עָלָיו׳. הַחֵרָטָה הִיא לֹא עֹנֶשׁ – הִיא סִימָן שֶׁהַלֵּב חַי וְרוֹצֶה לִתְקֹן.

רַשִׁ״י: מִי שֶׁמִּתְחָרֵט – הַחֵטְא נִדְמֶה לוֹ כְּאִלּוּ הוּא תָּמִיד לְפָנָיו.

📜 הַפָּסוּק… תִּצְדַּק בְּדָבְרֶךָ תִּזְכֶּה בְשָׁפְטֶךָ

רָמָה ב׳ – מַה מְּבָאֲרִים הַמְּפָרְשִׁים

⚖️ רַשִׁ״י: ׳לְמַעַן תִּצְדַּק בְּדָבְרֶךָ׳

רַשִׁ״י: דָּוִד מוֹדֶה שֶׁה׳ צוֹדֵק – וְזֶה הַצַּעַד הָרִאשׁוֹן שֶׁל תְּשׁוּבָה אֲמִתִּית.

📜 הַפָּסוּקתְּחַטְּאֵנִי בְאֵזוֹב וְאֶטְהָר תְּכַבְּסֵנִי וּמִשֶּׁלֶג אַלְבִּין

💛 אַהֲבַת הַשֵּׁם: סְלִיחַת ה׳ שְׁלֵמָה – בְּלִי שׁוּם כֶּתֶם 💛

📖 תְּהִלִּים, פֶּרֶק נ״א, פָּסוּק ט׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – מַה מְּבָאֲרִים הַמְּפָרְשִׁים

׳תְּחַטְּאֵנִי בְאֵזוֹב וְאֶטְהָר׳ – רַשִׁ״י מְבָאֵר: ׳כְּמוֹ שֶׁמְּטַהֵר הַמְצֹרָע וּטְמֵא מֵת׳. מְצוּדַת דָּוִד: ׳תָּסִיר חַטָּאִי כְּמוֹ שֶׁמְּסִירִין הַטֻּמְאָה עִם הַזָּאַת הָאֵזוֹב, וְאֶהְיֶה טָהוֹר מִטֻּמְאַת הַחֵטְא׳, ו׳תְּכַבְּסֵנִי – מְחֹל לִי לְבַל יִשָּׁאֵר רֹשֶׁם עָוֹן וְאֶהְיֶה לָבָן יוֹתֵר מִן הַשֶּׁלֶג מִבְּלִי רֹשֶׁם עָוֹן כְּלָל׳. דָּוִד מְבַקֵּשׁ סְלִיחָה כָּל כָּךְ שְׁלֵמָה, שֶׁלֹּא יִשָּׁאֵר שׁוּם רֹשֶׁם.

מְצוּדַת דָּוִד: דָּוִד מְבַקֵּשׁ סְלִיחָה שְׁלֵמָה – שֶׁלֹּא יִשָּׁאֵר רֹשֶׁם כְּלָל.

📜 הַפָּסוּקתַּשְׁמִיעֵנִי שָׂשׂוֹן וְשִׂמְחָה תָּגֵלְנָה עֲצָמוֹת דִּכִּיתָ

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹמֵר לִי: ׳סָלַחְתִּי׳ – וְהַלֵּב שֶׁלִּי מִתְמַלֵּא שִׂמְחָה 💛

📖 תְּהִלִּים, פֶּרֶק נ״א, פָּסוּק י׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – מַה מְּבָאֲרִים הַמְּפָרְשִׁים

🎉 רַשִׁ״י: ׳תַּשְׁמִיעֵנִי שָׂשׂוֹן׳

׳תַּשְׁמִיעֵנִי שָׂשׂוֹן וְשִׂמְחָה תָּגֵלְנָה עֲצָמוֹת דִּכִּיתָ׳ – רַשִׁ״י מְבָאֵר: ׳תַּשְׁמִיעֵנִי שָׂשׂוֹן וְשִׂמְחָה – סְלִיחַת הַחֵטְא. עֲצָמוֹת דִּכִּיתָ – כְּשֶׁכָּעַסְתָּ עָלַי׳. מְצוּדַת דָּוִד: ׳הַשְׁמִיעֵנִי סָלַחְתִּי, וְיִהְיֶה לִי שִׂמְחָה וְשָׂשׂוֹן׳. הַסְּלִיחָה אֵינָהּ רַק שֶׁלֹּא יַעַנִישׁוּ – הִיא שִׂמְחָה שֶׁשּׁוֹמְעִים: ׳סָלַחְתִּי׳.

רַשִׁ״י: הַסְּלִיחָה הִיא ׳שָׂשׂוֹן וְשִׂמְחָה׳ שֶׁשּׁוֹמְעִים מֵה׳.

📜 הַפָּסוּקאֲלַמְּדָה פֹשְׁעִים דְּרָכֶיךָ וְחַטָּאִים אֵלֶיךָ יָשׁוּבוּ

💛 אַהֲבַת הַשֵּׁם: הַסְּלִיחָה שֶׁקִּבַּלְתִּי מַעֲנִיקָה לִי כֹּחַ לַעֲזֹר לַאֲחֵרִים 💛

📖 תְּהִלִּים, פֶּרֶק נ״א, פָּסוּק ט״ו (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – מַה מְּבָאֲרִים הַמְּפָרְשִׁים

📚 רַשִׁ״י וּמְצוּדַת דָּוִד: ׳אֲלַמְּדָה פֹשְׁעִים׳

׳אֲלַמְּדָה פֹשְׁעִים דְּרָכֶיךָ׳ – רַשִׁ״י מְבָאֵר: ׳וּמִמֶּנִּי יִלְמְדוּ וְיָשׁוּבוּ אִם יִרְאוּ שֶׁתִּסְלַח לִי׳. מְצוּדַת דָּוִד: ׳אָז אוּכַל לְלַמֵּד אֶת דְּרָכֶיךָ אֶת הַפּוֹשְׁעִים וּלְהַזְהִירָם עַל הַתְּשׁוּבָה, וְלוֹמַר לָהֶם: ראוּ מָה עָלְתָה בִּי עַל יְדֵי הַתְּשׁוּבָה׳. הַסִּפּוּר שֶׁל דָּוִד הוּא מוֹפֵת: אִם הַמֶּלֶךְ שָׁב וְנִסְלַח, כָּל אֶחָד יָכוֹל.

רַשִׁ״י: הַתְּשׁוּבָה שֶׁל דָּוִד מַרְאָה לַאֲחֵרִים שֶׁאֶפְשָׁר לָשׁוּב.

📜 הַפָּסוּקאֲדֹנָי שְׂפָתַי תִּפְתָּח וּפִי יַגִּיד תְּהִלָּתֶךָ

💛 אַהֲבַת הַשֵּׁם: אַחֲרֵי שֶׁה׳ מוֹחֵל לִי, הַפֶּה שֶׁלִּי נִפְתָּח לָשִׁיר 💛

📖 תְּהִלִּים, פֶּרֶק נ״א, פָּסוּק י״ז (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – מַה מְּבָאֲרִים הַמְּפָרְשִׁים

׳אֲדֹנָי שְׂפָתַי תִּפְתָּח וּפִי יַגִּיד תְּהִלָּתֶךָ׳ – רַשִׁ״י מְבָאֵר: ׳מְחֹל לִי וִיהִי לִי פִּתְחוֹן פֶּה לְהַגִּיד תְּהִלָּתֶךָ׳. מְצוּדַת דָּוִד: ׳בְּמָה שֶׁתִּמְחֹל לִי תִּפְתַּח פִּי לְהַגִּיד תְּהִלָּתְךָ לָרַבִּים וּלְהָעִיר לִבָּם עַל הַתְּשׁוּבָה׳. מִי שֶׁלִּבּוֹ כָּבֵד מֵחֵטְא – הַפֶּה סָגוּר. אַחֲרֵי הַסְּלִיחָה – הַפֶּה נִפְתָּח לְהוֹדוֹת.

רַשִׁ״י: אַחֲרֵי הַסְּלִיחָה מִתְפַּתֵּחַ הַפֶּה לְהַגִּיד תְּהִלָּה.

📜 הַפָּסוּקזִבְחֵי אֱלֹהִים רוּחַ נִשְׁבָּרָה לֵב נִשְׁבָּר וְנִדְכֶּה אֱלֹהִים לֹא תִבְזֶה

🤔 חוֹשְׁבִים עַל זֶה

  • דָּוִד לֹא מָצָא תֵּרוּצִים – הוּא אָמַר: ׳חָטָאתִי׳. מָתַי הָיָה לִי קָשֶׁה לוֹמַר ׳טָעִיתִי׳, וְאֵיךְ אֶפְשָׁר לַעֲשׂוֹת אֶת זֶה בְּקַלּוּת?
  • ׳לֵב טָהוֹר בְּרָא לִי אֱלֹהִים׳. מָה פֵּרוּשׁ לֵב טָהוֹר בַּשְׁבִילִי, וּמָה אֲנִי יָכוֹל לַעֲשׂוֹת כְּדֵי שֶׁהַלֵּב שֶׁלִּי יִהְיֶה נָקִי יוֹתֵר?
  • ׳זִבְחֵי אֱלֹהִים רוּחַ נִשְׁבָּרָה׳. אֵיךְ אֶפְשָׁר לְהַרְגִּישׁ שֶׁה׳ קָרוֹב אֵלַי דַּוְקָא כְּשֶׁאֲנִי מַרְגִּישׁ קָטָן וְשָׁבוּר?

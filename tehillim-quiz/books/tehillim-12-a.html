🦌 ׳כְּאַיָּל תַּעֲרֹג׳

׳כְּאַיָּל תַּעֲרֹג עַל אֲפִיקֵי מָיִם כֵּן נַפְשִׁי תַעֲרֹג אֵלֶיךָ אֱלֹהִים׳. אַיָּל שֶׁצָּמֵא רוֹצֶה מַיִם, וְכָךְ הַנֶּפֶשׁ רוֹצָה אֶת ה׳. הַכְּמִיהָה הִיא חֹסֶר שֶׁרוֹצֶה לְהִתְמַלֵּא.

הַנֶּפֶשׁ צְמֵאָה לַה׳ כְּמוֹ אַיָּל לְמַיִם.

1

💛 אַהֲבַת הַשֵּׁם: אֲנִי מִתְגַּעְגֵּעַ לִפְגֹּשׁ אֶת ה׳ בִּתְפִלָּה 💛

📖 תְּהִלִּים, פֶּרֶק מ״ב, פָּסוּק ג׳

רָמָה א׳ – נֶפֶשׁ צְמֵאָה

💧 ׳צָמְאָה נַפְשִׁי׳

׳צָמְאָה נַפְשִׁי לֵאלֹהִים לְאֵל חָי מָתַי אָבוֹא וְאֵרָאֶה פְּנֵי אֱלֹהִים׳. הַמְּשׁוֹרֵר שׁוֹאֵל: מָתַי אַגִּיעַ שׁוּב לְבֵית הַמִּקְדָּשׁ? הוּא לֹא רוֹצֶה רַק מָקוֹם – הוּא רוֹצֶה לִפְגֹּשׁ אֶת ה׳.

מִי שֶׁקָּרוֹב לַה׳ מִתְגַּעְגֵּעַ לִפְגֹּשׁ אוֹתוֹ.

2

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹאֶה כָּל דִּמְעָה שֶׁלִּי 💛

📖 תְּהִלִּים, פֶּרֶק מ״ב, פָּסוּק ד׳

רָמָה א׳ – נֶפֶשׁ צְמֵאָה

😢 ׳הָיְתָה לִּי דִמְעָתִי לֶחֶם׳

הַגּוֹלֶה בּוֹכֶה: ׳הָיְתָה לִּי דִמְעָתִי לֶחֶם יוֹמָם וָלָיְלָה בֶּאֱמֹר אֵלַי כָּל הַיּוֹם אַיֵּה אֱלֹהֶיךָ׳. הַדְּמָעוֹת הֵן כְּמוֹ אֹכֶל, וַאֲחֵרִים לוֹעֲגִים: ׳אַיֵּה אֱלֹהֶיךָ׳? אֲבָל הוּא לֹא מַפְסִיק לִבְטֹחַ.

גַּם כְּשֶׁלּוֹעֲגִים, הַנֶּפֶשׁ מַמְשִׁיכָה לְהִתְגַּעְגֵּעַ לַה׳.

3

💛 אַהֲבַת הַשֵּׁם: זִכְרוֹנוֹת טוֹבִים עִם ה׳ נוֹתְנִים לִי כֹּחַ 💛

📖 תְּהִלִּים, פֶּרֶק מ״ב, פָּסוּק ה׳

רָמָה א׳ – נֶפֶשׁ צְמֵאָה

🎉 ׳הָמוֹן חוֹגֵג׳

הוּא מִתְגַּעְגֵּעַ וְזוֹכֵר: ׳אֶעֱבֹר בַּסָּךְ אֶדַּדֵּם עַד בֵּית אֱלֹהִים בְּקוֹל רִנָּה וְתוֹדָה הָמוֹן חוֹגֵג׳. הוּא זוֹכֵר אֶת הָעֲלִיָּה לָרֶגֶל לִירוּשָׁלַיִם עִם הָמוֹן אֲנָשִׁים שֶׁשָּׁרִים וּמוֹדִים. הַזִּכָּרוֹן הַטּוֹב נוֹתֵן כֹּחַ.

הַזִּכָּרוֹן שֶׁל שִׂמְחָה עָבָרָה מְחַזֵּק בַּזְּמַן הַקָּשֶׁה.

4

💛 אַהֲבַת הַשֵּׁם: אֲנִי מְדַבֵּר אֶל עַצְמִי בְּאַהֲבָה וּבְתִקְוָה 💛

📖 תְּהִלִּים, פֶּרֶק מ״ב, פָּסוּק ו׳

רָמָה א׳ – נֶפֶשׁ צְמֵאָה

💬 ׳מַה תִּשְׁתּוֹחֲחִי נַפְשִׁי׳

הוּא מְדַבֵּר אֶל הַנֶּפֶשׁ: ׳מַה תִּשְׁתּוֹחֲחִי נַפְשִׁי וַתֶּהֱמִי עָלָי הוֹחִילִי לֵאלֹהִים כִּי עוֹד אוֹדֶנּוּ יְשׁוּעוֹת פָּנָיו׳. ׳הוֹחִילִי׳ – חַכִּי. הוּא מַרְגִּיעַ אֶת עַצְמוֹ וְאוֹמֵר: עוֹד אוֹדֶה לוֹ עַל הַיְשׁוּעָה.

מְדַבְּרִים אֶל הַנֶּפֶשׁ וְאוֹמְרִים לָהּ: חַכִּי, עוֹד נוֹדֶה.

5

💛 אַהֲבַת הַשֵּׁם: אֲנִי זוֹכֵר אֶת ה׳ בְּכָל מָקוֹם שֶׁאֲנִי נִמְצָא 💛

📖 תְּהִלִּים, פֶּרֶק מ״ב, פָּסוּק ז׳

רָמָה א׳ – נֶפֶשׁ צְמֵאָה

🏔️ ׳מֵאֶרֶץ יַרְדֵּן וְחֶרְמוֹנִים׳

׳עַל כֵּן אֶזְכָּרְךָ מֵאֶרֶץ יַרְדֵּן וְחֶרְמוֹנִים מֵהַר מִצְעָר׳. אֲפִלּוּ רָחוֹק, בְּקָצֶה הָאָרֶץ, הוּא זוֹכֵר אֶת ה׳. מִכָּל מָקוֹם אֶפְשָׁר לִזְכֹּר אוֹתוֹ.

מִכָּל מָקוֹם, גַּם רָחוֹק, אֶפְשָׁר לִזְכֹּר אֶת ה׳.

6

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁהַגַּלִּים גְּדוֹלִים, ה׳ מַחֲזִיק אוֹתִי 💛

📖 תְּהִלִּים, פֶּרֶק מ״ב, פָּסוּק ח׳

רָמָה א׳ – נֶפֶשׁ צְמֵאָה

🌊 ׳תְּהוֹם אֶל תְּהוֹם קוֹרֵא׳

בַּגָּלוּת הַצָּרוֹת רַבּוֹת: ׳תְּהוֹם אֶל תְּהוֹם קוֹרֵא לְקוֹל צִנּוֹרֶיךָ כָּל מִשְׁבָּרֶיךָ וְגַלֶּיךָ עָלַי עָבָרוּ׳. כְּמוֹ גַּלִּים שֶׁעוֹבְרִים אֶחָד אַחַר הַשֵּׁנִי. אֲבָל גַּם בְּתוֹךְ הַגַּלִּים הוּא לֹא עוֹזֵב אֶת ה׳.

כְּשֶׁהַצָּרוֹת מִתְרַבּוֹת, ה׳ עֲדַיִן אִתִּי.

7

💛 אַהֲבַת הַשֵּׁם: גַּם בַּלַּיְלָה אֲנִי יָכוֹל לָשִׁיר לַה׳ 💛

📖 תְּהִלִּים, פֶּרֶק מ״ב, פָּסוּק ט׳

רָמָה א׳ – נֶפֶשׁ צְמֵאָה

🌙 ׳וּבַלַּיְלָה שִׁירוֹ עִמִּי׳

הַנֶּפֶשׁ מִתְחַזֶּקֶת: ׳יוֹמָם יְצַוֶּה ה׳ חַסְדּוֹ וּבַלַּיְלָה שִׁירוֹ עִמִּי תְּפִלָּה לְאֵל חַיָּי׳. בַּיּוֹם ה׳ שׁוֹלֵחַ חֶסֶד, וּבַלַּיְלָה יֵשׁ לוֹ שִׁיר וּתְפִלָּה. אֲנִי לֹא לְבַד.

בַּיּוֹם חֶסֶד, וּבַלַּיְלָה שִׁיר וּתְפִלָּה.

8

רָמָה א׳ – נֶפֶשׁ צְמֵאָה

מָה לוֹקְחִים מִרָמָה א׳?

בְּנֵי קֹרַח שָׁרִים מִתּוֹךְ הַגָּלוּת וְהַגַּעְגּוּעִים: הַנֶּפֶשׁ צְמֵאָה לַה׳ כְּמוֹ אַיָּל לְמַיִם. הֵם זוֹכְרִים אֶת הָעֲלִיָּה לָרֶגֶל וְאוֹמְרִים לַנֶּפֶשׁ: אַל תִּשְׁתּוֹחֲחִי, עוֹד נוֹדֶה לוֹ. וְה׳ שׁוֹמֵעַ אֶת הַגַּעְגּוּעִים וְאוֹהֵב אוֹתָם.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • אַיָּל צָמֵא לְמַיִם כִּי בִּלְעָדֵיהֶם לֹא יִחְיֶה. מָה הַנֶּפֶשׁ שֶׁלִּי צְמֵאָה לוֹ בֶּאֱמֶת?
  • הַמִּזְמוֹר אוֹמֵר לַנֶּפֶשׁ: ׳מַה תִּשְׁתּוֹחֲחִי... הוֹחִילִי לֵאלֹהִים׳. מָתַי דִּבַּרְתִּי אֶל עַצְמִי בְּעִידוֹדִים כָּאֵלֶּה?
  • הַמִּזְמוֹר זוֹכֵר עֲלִיָּה לָרֶגֶל עִם הָמוֹן חוֹגֵג. מָה הַזִּכָּרוֹן הַטּוֹב שֶׁלִּי שֶׁמְּחַזֵּק אוֹתִי כְּשֶׁקָּשֶׁה?

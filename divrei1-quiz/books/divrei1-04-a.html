דָּוִיד לֹא מַחְלִיט לְבַד. ׳וַיִּוָּעַץ דָּוִיד עִם שָׂרֵי הָאֲלָפִים וְהַמֵּאוֹת לְכָל נָגִיד׳, וְאוֹמֵר לְכָל הַקָּהָל: ׳וְנָסֵבָּה אֶת אֲרוֹן אֱלֹהֵינוּ אֵלֵינוּ כִּי לֹא דְרַשְׁנֻהוּ בִּימֵי שָׁאוּל׳. הוּא רוֹצֶה לְהָבִיא אֶת הָאָרוֹן לִירוּשָׁלַיִם, כִּי יָמִים רַבִּים לֹא דָּרְשׁוּ אוֹתוֹ.

דָּוִיד מִתְיַעֵץ עִם שָׂרֵי הָעָם וּמַחְלִיט לְהָבִיא אֶת הָאָרוֹן לִירוּשָׁלַיִם.

🗺️ ׳מִן שִׁיחוֹר מִצְרַיִם וְעַד לְבוֹא חֲמָת׳

דָּוִיד מַקְהִיל אֶת כָּל יִשְׂרָאֵל ׳מִן שִׁיחוֹר מִצְרַיִם וְעַד לְבוֹא חֲמָת׳ – מִגְּבוּל מִצְרַיִם עַד הַצָּפוֹן. כֻּלָּם הוֹלְכִים לְקִרְיַת יְעָרִים, הַמָּקוֹם שֶׁבּוֹ הָיָה הָאָרוֹן, לְהָבִיא אוֹתוֹ מִשָּׁם.

כָּל יִשְׂרָאֵל מִכָּל גְּבוּלוֹת הָאָרֶץ הוֹלְכִים לְהָבִיא אֶת הָאָרוֹן מִקִּרְיַת יְעָרִים.

רָמָה א׳ – הַעֲלָיַת הָאָרוֹן

🎶 ׳וְדָוִיד וְכָל יִשְׂרָאֵל מְשַׂחֲקִים׳

הָאָרוֹן עָלָה עַל עֲגָלָה חֲדָשָׁה. ׳וְדָוִיד וְכָל יִשְׂרָאֵל מְשַׂחֲקִים לִפְנֵי הָאֱלֹהִים בְּכָל עֹז וּבְשִׁירִים וּבְכִנֹּרוֹת וּבִנְבָלִים וּבְתֻפִּים וּבִמְצִלְתַּיִם וּבַחֲצֹצְרוֹת׳. הַדֶּרֶךְ הָיְתָה מְלֵאָה בְּשִׂמְחָה וּמוּזִיקָה.

כָּל הָעָם שָׂמַח וְשָׁר לִפְנֵי הָאָרוֹן בִּכְלֵי נְגִינָה רַבִּים.

⏸️

💛 אַהֲבַת הַשֵּׁם: ה׳ מְלַמֵּד אוֹתָנוּ לַעֲשׂוֹת בִּזְהִירוּת אֶת מָה שֶׁחָשׁוּב לוֹ 💛

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק י״ג, פְּסוּקִים ט׳–י״א

רָמָה א׳ – הַעֲלָיַת הָאָרוֹן

⏸️ ׳פֶּרֶץ עֻזָּא׳

בַּדֶּרֶךְ ׳שָׁמְטוּ הַבָּקָר׳, וְעֻזָּא שָׁלַח אֶת יָדוֹ לֶאֱחֹז בָּאָרוֹן. הָיָה כָּאן אֵרוּעַ קָשֶׁה וּמָקוֹם שֶׁנִּקְרָא מִכָּאן ׳פֶּרֶץ עֻזָּא׳. הַהַעֲלָיָה נֶעֶצְרָה, וְדָוִיד הִבִּין שֶׁמַּשֶּׁהוּ בַּדֶּרֶךְ לֹא הָיָה כְּפִי שֶׁצָּרִיךְ.

בַּדֶּרֶךְ קָרָה אֵרוּעַ קָשֶׁה, הַהַעֲלָיָה נֶעֶצְרָה, וְדָוִיד הִבִּין שֶׁצָּרִיךְ לַעֲשׂוֹת אֶת הַדָּבָר בְּדֶרֶךְ אַחֶרֶת.

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק י״ג, פְּסוּקִים י״ב–י״ג

😟 ׳הֵיךְ אָבִיא אֵלַי אֶת אֲרוֹן הָאֱלֹהִים׳

דָּוִיד נִבְהַל: ׳וַיִּירָא דָוִיד אֶת הָאֱלֹהִים בַּיּוֹם הַהוּא לֵאמֹר הֵיךְ אָבִיא אֵלַי אֵת אֲרוֹן הָאֱלֹהִים׳. הוּא לֹא הוֹלִיךְ אֶת הָאָרוֹן אֶל עִיר דָּוִיד, אֶלָּא הִנִּיחַ אוֹתוֹ זְמַן מָה בְּמָקוֹם קָרוֹב. הוּא רוֹצֶה לַעֲשׂוֹת הַכֹּל נָכוֹן.

דָּוִיד נִבְהַל וְרוֹצֶה לָדַעַת אֵיךְ לְהָבִיא אֶת הָאָרוֹן כַּמִּשְׁפָּט.

🏠

💛 אַהֲבַת הַשֵּׁם: ה׳ מְבָרֵךְ אֶת מִי שֶׁמְּכַבֵּד אֶת הַקֹּדֶשׁ בְּבֵיתוֹ 💛

רָמָה א׳ – הַעֲלָיַת הָאָרוֹן

🏠 ׳וַיְבָרֶךְ ה׳ אֶת בֵּית עֹבֵד אֱדֹם׳

הָאָרוֹן נִשְׁאַר שְׁלוֹשָׁה חֳדָשִׁים בְּבֵית עֹבֵד אֱדֹם הַגִּתִּי: ׳וַיֵּשֶׁב אֲרוֹן הָאֱלֹהִים עִם בֵּית עֹבֵד אֱדֹם בְּבֵיתוֹ שְׁלֹשָׁה חֳדָשִׁים וַיְבָרֶךְ ה׳ אֶת בֵּית עֹבֵד אֱדֹם וְאֶת כָּל אֲשֶׁר לוֹ׳. מִי שֶׁהִכְנִיס אֶת הָאָרוֹן לְבֵיתוֹ קִבֵּל בְּרָכָה.

הָאָרוֹן הָיָה שְׁלוֹשָׁה חֳדָשִׁים בְּבֵית עֹבֵד אֱדֹם, וְה׳ בֵּרֵךְ אֶת בֵּיתוֹ.

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק ט״ו, פְּסוּקִים א׳–י״ד

רָמָה א׳ – הַעֲלָיַת הָאָרוֹן

🧭 ׳לֹא לָשֵׂאת אֶת אֲרוֹן הָאֱלֹהִים כִּי אִם הַלְוִיִּם׳

דָּוִיד הֵכִין בִּירוּשָׁלַיִם מָקוֹם וְאֹהֶל לָאָרוֹן, וְאָמַר: ׳לֹא לָשֵׂאת אֶת אֲרוֹן הָאֱלֹהִים כִּי אִם הַלְוִיִּם כִּי בָם בָּחַר ה׳ לָשֵׂאת אֶת אֲרוֹן ה׳׳. הוּא קָרָא לַכֹּהֲנִים וְלַלְוִיִּם ׳הִתְקַדְּשׁוּ׳, כִּי הַפַּעַם יַעֲשׂוּ הַכֹּל כְּפִי שֶׁהַתּוֹרָה אָמְרָה.

דָּוִיד הֵכִין אֹהֶל, וְקָבַע שֶׁרַק הַלְוִיִּם יִשְׂאוּ אֶת הָאָרוֹן.

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק ט״ו, פְּסוּקִים ט״ו, כ״ה–כ״ח

רָמָה א׳ – הַעֲלָיַת הָאָרוֹן

🎺 ׳וַיִּשְׂאוּ בְנֵי הַלְוִיִּם בִּכְתֵפָם׳

הַפַּעַם ׳וַיִּשְׂאוּ בְנֵי הַלְוִיִּם אֵת אֲרוֹן הָאֱלֹהִים כַּאֲשֶׁר צִוָּה מֹשֶׁה כִּדְבַר ה׳ בִּכְתֵפָם בַּמֹּטוֹת עֲלֵיהֶם׳. וְכָל יִשְׂרָאֵל מַעֲלִים אֶת הָאָרוֹן ׳בִּתְרוּעָה וּבְקוֹל שׁוֹפָר וּבַחֲצֹצְרוֹת וּבִמְצִלְתַּיִם׳. שִׁירָה וְשִׂמְחָה – וְהַכֹּל כְּמוֹ בַּתּוֹרָה.

הַלְוִיִּם נָשְׂאוּ אֶת הָאָרוֹן בְּכָתֵף כְּמוֹ שֶׁמֹּשֶׁה צִוָּה, בִּתְרוּעָה וּבְשִׁיר.

מָה לוֹקְחִים מִרָמָה א׳?

דָּוִיד רוֹצֶה שֶׁהָאָרוֹן, שֶׁבּוֹ הַלּוּחוֹת, יִהְיֶה בְּתוֹךְ יְרוּשָׁלַיִם. הוּא מַקְהִיל אֶת כָּל יִשְׂרָאֵל, וּבַפַּעַם הָרִאשׁוֹנָה לֹא הַכֹּל מִתְנַהֵל כַּנָּכוֹן. דָּוִיד לוֹמֵד, חוֹזֵר, וְהַפַּעַם נוֹשְׂאִים אֶת הָאָרוֹן כְּמוֹ בַּתּוֹרָה, בְּשִׂמְחָה וּבְשִׁיר. ה׳ אוֹהֵב אוֹתָנוּ וּמְלַמֵּד אוֹתָנוּ אֶת הַדֶּרֶךְ.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • דָּוִיד רָצָה לְכַבֵּד אֶת הָאָרוֹן, אֲבָל הִתְחִיל בְּדֶרֶךְ שֶׁלֹּא הָיְתָה הַדֶּרֶךְ שֶׁהַתּוֹרָה אָמְרָה. מָתַי רָצִיתִי לַעֲשׂוֹת מִצְוָה בְּכַוָּנָה טוֹבָה, וְהִבַּנְתִּי שֶׁצָּרִיךְ לַעֲשׂוֹת אוֹתָהּ ׳כַּמִּשְׁפָּט׳?
  • דָּוִיד לֹא הִרְפָּה וְחָזַר אַחֲרֵי שְׁלוֹשָׁה חֳדָשִׁים עִם הֲכָנוֹת. אֵיךְ אֲנִי מִתְמוֹדֵד עִם טָעוּת – נֶעֱצָב וּמוֹתֵר, אוֹ לוֹמֵד וּמִתְקַדֵּם?
  • הַלְוִיִּם שָׁרוּ בִּכְלֵי שִׁיר כְּשֶׁנָּשְׂאוּ אֶת הָאָרוֹן. אֵיפֹה אֲנִי יָכוֹל לְהוֹסִיף שִׁירָה וְשִׂמְחָה לְמִצְוָה שֶׁאֲנִי עוֹשֶׂה?

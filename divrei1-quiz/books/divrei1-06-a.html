דָּוִיד רוֹצֶה לִבְנוֹת בַּיִת לַה׳ – נָתָן אוֹמֵר לוֹ ׳לֹא אַתָּה תִבְנֶה׳, וְה׳ מַבְטִיחַ לוֹ בַּיִת אַחֵר, וְדָוִיד עוֹמֵד לְהוֹדוֹת בְּעַנְוָה

רָמָה א׳ – הַבַּיִת וְהַבְּרָכָה

🏠 ׳אָנֹכִי יוֹשֵׁב בְּבֵית הָאֲרָזִים׳

דָּוִיד יוֹשֵׁב בְּבֵיתוֹ וְאוֹמֵר לְנָתָן הַנָּבִיא: ׳הִנֵּה אָנֹכִי יוֹשֵׁב בְּבֵית הָאֲרָזִים וַאֲרוֹן בְּרִית ה׳ תַּחַת יְרִיעוֹת׳. הוּא מַרְגִּישׁ שֶׁלֹּא נָכוֹן שֶׁהוּא גָּר בְּבַיִת יָפֶה וְהָאָרוֹן עוֹמֵד בְּאֹהֶל.

דָּוִיד רוֹצֶה שֶׁלָּאָרוֹן יִהְיֶה בַּיִת, כִּי הוּא עַצְמוֹ גָּר בְּבֵית אֲרָזִים.

💬 ׳כֹּל אֲשֶׁר בִּלְבָבְךָ עֲשֵׂה׳

נָתָן עוֹנֶה מִיָּד: ׳כֹּל אֲשֶׁר בִּלְבָבְךָ עֲשֵׂה כִּי הָאֱלֹהִים עִמָּךְ׳. הוּא מִתְרַשֵּׁם מֵהָרָצוֹן הַטּוֹב שֶׁל הַמֶּלֶךְ, וּמַסְכִּים בְּלִי לִשְׁאֹל אֶת ה׳.

נָתָן הַנָּבִיא מַסְכִּים לְדָוִיד: ׳כֹּל אֲשֶׁר בִּלְבָבְךָ עֲשֵׂה׳.

🌙

💛 אַהֲבַת הַשֵּׁם: ה׳ מְדַבֵּר אֵלֵינוּ בְּאֱמֶת, גַּם כְּשֶׁהַתְּשׁוּבָה אֵינָהּ ׳כֵּן׳ 💛

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק י״ז, פְּסוּקִים ג׳–ד׳

רָמָה א׳ – הַבַּיִת וְהַבְּרָכָה

🌙 ׳לֹא אַתָּה תִבְנֶה־לִּי הַבַּיִת׳

בַּלַּיְלָה הַהוּא ה׳ שׁוֹלֵחַ אֶת נָתָן: ׳לֵךְ וְאָמַרְתָּ אֶל דָּוִיד עַבְדִּי כֹּה אָמַר ה׳ לֹא אַתָּה תִבְנֶה לִּי הַבַּיִת לָשָׁבֶת׳. נָתָן חוֹזֵר וְאוֹמֵר לְדָוִיד אֶת מָה שֶׁה׳ אָמַר, גַּם כְּשֶׁזֶּה שׁוֹנֶה מִמָּה שֶׁאָמַר קֹדֶם.

ה׳ אוֹמֵר לְדָוִיד: אַתָּה לֹא תִבְנֶה אֶת הַבַּיִת.

⛺

💛 אַהֲבַת הַשֵּׁם: ה׳ הוֹלֵךְ אִתָּנוּ בְּכָל מָקוֹם, גַּם בְּאֹהֶל פָּשׁוּט 💛

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק י״ז, פְּסוּקִים ה׳–ו׳

רָמָה א׳ – הַבַּיִת וְהַבְּרָכָה

⛺ ׳מֵאֹהֶל אֶל אֹהֶל׳

ה׳ מַסְבִּיר: ׳כִּי לֹא יָשַׁבְתִּי בְּבַיִת מִן הַיּוֹם אֲשֶׁר הֶעֱלֵיתִי אֶת יִשְׂרָאֵל עַד הַיּוֹם הַזֶּה וָאֶהְיֶה מֵאֹהֶל אֶל אֹהֶל וּמִמִּשְׁכָּן׳. מֵאָז יְצִיאַת מִצְרַיִם ה׳ הָיָה עִם עַמּוֹ בְּאֹהֶל, וְלֹא בִּקֵּשׁ בַּיִת.

ה׳ הָיָה עִם הָעָם בְּאֹהֶל וּבְמִשְׁכָּן מֵאָז יְצִיאַת מִצְרַיִם, וְלֹא בִּקֵּשׁ בַּיִת.

🐑

💛 אַהֲבַת הַשֵּׁם: ה׳ לוֹקֵחַ אוֹתִי מִמָּקוֹם קָטָן וְנוֹתֵן לִי הִזְדַּמְּנוּת גְּדוֹלָה 💛

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק י״ז, פְּסוּקִים ז׳–ח׳

רָמָה א׳ – הַבַּיִת וְהַבְּרָכָה

🐑 ׳אֲנִי לְקַחְתִּיךָ מִן הַנָּוֶה׳

ה׳ מְזַכֵּר לְדָוִיד: ׳אֲנִי לְקַחְתִּיךָ מִן הַנָּוֶה מִן אַחֲרֵי הַצֹּאן לִהְיוֹת נָגִיד עַל עַמִּי יִשְׂרָאֵל׳. ׳וָאֶהְיֶה עִמְּךָ בְּכֹל אֲשֶׁר הָלַכְתָּ... וְעָשִׂיתִי לְךָ שֵׁם כְּשֵׁם הַגְּדוֹלִים אֲשֶׁר בָּאָרֶץ׳. מֵרוֹעֶה צֹאן לְמֶלֶךְ – וְה׳ הָיָה אִתּוֹ.

ה׳ לָקַח אֶת דָּוִיד מִן הַצֹּאן לִהְיוֹת מֶלֶךְ, וְהָיָה אִתּוֹ בְּכָל דֶּרֶךְ.

💛 אַהֲבַת הַשֵּׁם: ה׳ מְשַׁלֵּם לְמִי שֶׁרוֹצֶה לַעֲשׂוֹת טוֹב, גַּם אִם אַחֵר מַשְׁלִים 💛

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק י״ז, פְּסוּקִים י׳–י״ב

רָמָה א׳ – הַבַּיִת וְהַבְּרָכָה

🏡 ׳וּבַיִת יִבְנֶה לְּךָ ה׳׳

ה׳ מַבְטִיחַ: ׳וָאַגִּד לָךְ וּבַיִת יִבְנֶה לְּךָ ה׳׳. וְאָז: ׳וַהֲקִימוֹתִי אֶת זַרְעֲךָ אַחֲרֶיךָ אֲשֶׁר יִהְיֶה מִבָּנֶיךָ וַהֲכִינוֹתִי אֶת מַלְכוּתוֹ הוּא יִבְנֶה לִּי בָּיִת׳. דָּוִיד לֹא יִבְנֶה, אֲבָל בְּנוֹ יִבְנֶה, וְהַמַּלְכוּת תִּמָּשֵׁךְ.

ה׳ מַבְטִיחַ לְדָוִיד בַּיִת שֶׁל מַלְכוּת, וְשֶׁבְּנוֹ יִבְנֶה אֶת בֵּית ה׳.

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק י״ז, פְּסוּקִים י״ג–י״ד

רָמָה א׳ – הַבַּיִת וְהַבְּרָכָה

👨‍👦 ׳אֲנִי אֶהְיֶה לּוֹ לְאָב׳

ה׳ אוֹמֵר עַל הַבֵּן שֶׁיִּבְנֶה אֶת הַבַּיִת: ׳אֲנִי אֶהְיֶה לּוֹ לְאָב וְהוּא יִהְיֶה לִּי לְבֵן וְחַסְדִּי לֹא אָסִיר מֵעִמּוֹ... וְכִסְאוֹ יִהְיֶה נָכוֹן עַד עוֹלָם׳. ה׳ אוֹהֵב אוֹתוֹ כְּמוֹ אַבָּא אֶת בְּנוֹ.

ה׳ מַבְטִיחַ לִהְיוֹת כְּמוֹ אָב לַבֵּן הַמֶּלֶךְ, וְחַסְדּוֹ לֹא יָסוּר.

🪑

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב כְּשֶׁאֲנִי מוֹדֶה בְּעַנְוָה עַל הַטּוֹבוֹת שֶׁקִּבַּלְתִּי 💛

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק י״ז, פָּסוּק ט״ז

רָמָה א׳ – הַבַּיִת וְהַבְּרָכָה

🪑 ׳מִי אֲנִי ה׳ אֱלֹהִים׳

דָּוִיד שׁוֹמֵעַ וְאוֹמֵר תּוֹדָה. ׳וַיָּבֹא הַמֶּלֶךְ דָּוִיד וַיֵּשֶׁב לִפְנֵי ה׳ וַיֹּאמֶר מִי אֲנִי ה׳ אֱלֹהִים וּמִי בֵיתִי כִּי הֲבִיאֹתַנִי עַד הֲלֹם׳. הוּא לֹא אוֹמֵר ׳מַגִּיעַ לִי׳, אֶלָּא ׳מִי אֲנִי׳. זוֹ עַנְוָה.

דָּוִיד יוֹשֵׁב לִפְנֵי ה׳ וְאוֹמֵר: ׳מִי אֲנִי ה׳ אֱלֹהִים וּמִי בֵיתִי׳.

רָמָה א׳ – הַבַּיִת וְהַבְּרָכָה

מָה לוֹקְחִים מִרָמָה א׳?

דָּוִיד רוֹאֶה שֶׁהוּא גָּר בְּבֵית אֲרָזִים וְהָאָרוֹן עוֹמֵד בְּאֹהֶל, וְרוֹצֶה לִבְנוֹת בַּיִת לַה׳. ה׳ עוֹנֶה: אֶת הַבַּיִת יִבְנֶה בִּנְךָ, וְאֲנִי אֶבְנֶה לְךָ בַּיִת שֶׁל מַלְכוּת. ה׳ אוֹהֵב אֶת הָרָצוֹן הַטּוֹב, וְדָוִיד מוֹדֶה לוֹ מִתּוֹךְ עַנְוָה.

  • דָּוִיד חִשֵּׁב ׳אֲנִי גָּר בְּבֵית אֲרָזִים וְהָאָרוֹן תַּחַת יְרִיעוֹת׳. מָתַי הִרְגַּשְׁתִּי שֶׁיֵּשׁ לִי יוֹתֵר מִדַּי וְרָצִיתִי לָתֵת?
  • ה׳ אָמַר לוֹ ׳לֹא אַתָּה תִבְנֶה׳, וְדָוִיד לֹא כָּעַס אֶלָּא הוֹדָה. אֵיךְ אֲנִי מִתְנַהֵג כְּשֶׁאוֹמְרִים לִי ׳לֹא׳ עַל מַשֶּׁהוּ שֶׁרָצִיתִי מְאֹד?
  • דָּוִיד אָמַר ׳מִי אֲנִי׳. אֵיפֹה אֲנִי יָכוֹל לְהַגִּיד ׳תּוֹדָה, זֶה לֹא בִּזְכוּתִי׳?

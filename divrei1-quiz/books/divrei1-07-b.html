🪙 רַשִׁ״י: חֲמִשִּׁים מִכָּל שֵׁבֶט

בְּדִבְרֵי הַיָּמִים כָּתוּב ׳שִׁקְלֵי זָהָב מִשְׁקָל שֵׁשׁ מֵאוֹת׳, וּבִשְׁמוּאֵל כָּתוּב חֲמִשִּׁים שִׁקְלֵי כֶּסֶף. רַשִׁ״י מְיַשֵּׁב: ׳הָא כֵּיצַד? נָטַל מִכָּל שֵׁבֶט וְשֵׁבֶט חֲמִשִּׁים שְׁקָלִים, הֲרֵי שֵׁשׁ מֵאוֹת שְׁקָלִים, כְּדֵי שֶׁיִּהְיֶה חֵלֶק לְכָל יִשְׂרָאֵל בַּמִּזְבֵּחַ׳.

רַשִׁ״י: דָּוִיד לָקַח חֲמִשִּׁים שְׁקָלִים מִכָּל שֵׁבֶט, כְּדֵי שֶׁלְּכָל יִשְׂרָאֵל יִהְיֶה חֵלֶק בַּמִּזְבֵּחַ.

רָמָה ב׳ – עִיּוּן

💎 מְצוּדַת דָּוִד: ׳כֶּסֶף מָלֵא׳ וְ׳חִנָּם׳

מְצוּדַת דָּוִד מְבָאֵר: ׳בְּכֶסֶף מָלֵא – בִּמְחִיר כֶּסֶף שָׁלֵם כָּל מִשְׁקָלוֹ׳. וְעַל ׳לֹא אֶשָּׂא אֲשֶׁר לְךָ לַה׳׳ הוּא כּוֹתֵב: ׳לֹא אֶקַּח אֲשֶׁר לְךָ לְתִתָּם לַה׳... וּלְהַעֲלוֹת עוֹלָה מֵהַדָּבָר הַנִּתָּן לִי בְּחִנָּם׳. מִצְוָה צְרִיכָה לָבוֹא מִמָּמוֹן שֶׁלִּי, וְלֹא מִמָּמוֹן שֶׁל מִישֶׁהוּ אַחֵר.

מְצוּדַת דָּוִד: דָּוִיד לֹא רָצָה לְהַעֲלוֹת עוֹלָה מֵהַדָּבָר שֶׁנִּתַּן לוֹ בְּחִנָּם – הַמִּצְוָה צְרִיכָה לְהִיוֹת מִשֶּׁלּוֹ.

📖 רַשִׁ״י: ׳מִשּׁוּם כְּבוֹדוֹ שֶׁל דָּוִד׳

הַפָּסוּק: ׳בָּעֵת הַהִיא בִּרְאוֹת דָּוִיד כִּי עָנָהוּ ה׳ בְּגֹרֶן אָרְנָן הַיְבוּסִי וַיִּזְבַּח שָׁם׳. רַשִׁ״י מְבָאֵר: ׳פָּרָשָׁה זוֹ לֹא נִכְתְּבָה בִּשְׁמוּאֵל וּמִשּׁוּם כְּבוֹדוֹ שֶׁל דָּוִד נִכְתְּבָה כָּאן שֶׁעָשָׂה מִזְבֵּחַ׳. הַסֵּפֶר שֶׁלָּנוּ בּוֹחֵר לְהַדְגִּישׁ אֶת הַטּוֹב שֶׁעָשָׂה דָּוִיד.

רַשִׁ״י: דִּבְרֵי הַיָּמִים מַדְגִּישׁ אֶת הַטּוֹב שֶׁעָשָׂה דָּוִיד, מִשּׁוּם כְּבוֹדוֹ.

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק כ״ב, פָּסוּק א׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן

🏛️ רַשִׁ״י: ׳עַל אוֹתוֹ מִזְבֵּחַ׳

דָּוִיד אוֹמֵר ׳זֶה הוּא בֵּית ה׳ הָאֱלֹהִים וְזֶה מִזְבֵּחַ לְעֹלָה לְיִשְׂרָאֵל׳. רַשִׁ״י מְבָאֵר: ׳מִכָּאן וָאֵילָךְ שֶׁכָּל שֶׁמְּבַקֵּשׁ ה׳ לְהַקְרִיב לוֹ קָרְבָּן יֵצֵא וְיַקְרִיב עַל אוֹתוֹ מִזְבֵּחַ׳. וְעַד אָז הַמִּזְבֵּחַ שֶׁל מֹשֶׁה ׳עֲדַיִן בְּעֵת הַהִיא בַּבָּמָה אֲשֶׁר בְּגִבְעוֹן׳. מִכָּאן נִקְבַּע הַמָּקוֹם לָעוֹלָם.

רַשִׁ״י: מִכָּאן וָאֵילָךְ מַקְרִיבִים עַל הַמִּזְבֵּחַ שֶׁבְּגֹרֶן אָרְנָן, וְלֹא עוֹד בְּגִבְעוֹן.

הַכָּתוּב: ׳וַיֹּאמֶר דָּוִיד לִכְנוֹס אֶת הַגֵּרִים אֲשֶׁר בְּאֶרֶץ יִשְׂרָאֵל וַיַּעֲמֵד חֹצְבִים׳. רַשִׁ״י מְבָאֵר: ׳גֵּרִים שֶׁנִּתְגַּיְּרוּ, שֶׁלֹּא רָצָה לְהַעֲבִיד בְּיִשְׂרָאֵל עֲבוֹדַת פֶּרֶךְ׳. כְּלוֹמַר: בֵּית ה׳ נִבְנֶה בִּידֵי הַגֵּרִים שֶׁבָּאָרֶץ, וְלֹא בִּידֵי הָעָם בְּכֹפִיָּה.

רַשִׁ״י: דָּוִיד כָּנַס אֶת הַגֵּרִים לַעֲבוֹדַת הַבִּנְיָן, כְּדֵי שֶׁלֹּא לְהַעֲבִיד בְּיִשְׂרָאֵל בְּפֶרֶךְ.

רָמָה ב׳ – עִיּוּן

☮️ מְצוּדַת דָּוִד: ׳אִישׁ מְנוּחָה׳

ה׳ אָמַר לְדָוִיד: ׳הִנֵּה בֵן נוֹלַד לָךְ הוּא יִהְיֶה אִישׁ מְנוּחָה וַהֲנִחוֹתִי לוֹ מִכָּל אוֹיְבָיו מִסָּבִיב׳. מְצוּדַת דָּוִד מְבָאֵר: ׳אִישׁ מְנוּחָה – יִהְיֶה חֶפְצוֹ בִּמְנוּחָה וְלֹא יִתְגָּרֶה בְּמִי׳, ׳וְשָׁלוֹם וְכוּ׳ – זִכְרוֹן שְׁמוֹ הוּא לְאוֹת עַל הַשָּׁלוֹם׳. דָּוִיד נִלְחַם מִלְחָמוֹת, וְשְׁלֹמֹה יִבְנֶה בְּשָׁלוֹם.

מְצוּדַת דָּוִד: שְׁלֹמֹה ׳אִישׁ מְנוּחָה׳ וְשִׁמּוֹ ׳שָׁלוֹם׳ – הוּא יִבְנֶה בְּשָׁלוֹם.

🪙 מְצוּדַת דָּוִד: ׳בְּעָנְיִי׳

דָּוִיד אוֹמֵר לִשְׁלֹמֹה: ׳וְהִנֵּה בְעָנְיִי הֲכִינוֹתִי לְבֵית ה׳ זָהָב כִּכָּרִים מֵאָה אֶלֶף וְכֶסֶף אֶלֶף אֲלָפִים כִּכָּרִים׳. מְצוּדַת דָּוִד מְבָאֵר: ׳בְּעָנְיִי – עִם שֶׁאֲנִי נֶחְשָׁב לְעָנִי מוּל הַצֹּרֶךְ הָרָאוּי לְבִנְיַן בֵּית ה׳, הִנֵּה הֵכַנְתִּי זָהָב וְכוּ׳׳. כָּל כָּךְ הַרְבֵּה – וְעֲדַיִן הוּא אוֹמֵר ׳בְּעָנְיִי׳.

מְצוּדַת דָּוִד: דָּוִיד הֵכִין כִּכָּרֵי זָהָב וָכֶסֶף רַבִּים, וְעִם זֹאת קָרָא לָזֶה ׳בְּעָנְיִי׳ מוּל גֹּדֶל הַבַּיִת.

🤔 חוֹשְׁבִים עַל זֶה

  • אָרְנָן הִצִּיעַ לָתֵת לְדָוִיד הַכֹּל בְּחִנָּם, אֲבָל דָּוִיד אָמַר ׳לֹא כִּי קָנֹה אֶקְנֶה בְּכֶסֶף מָלֵא׳. מָתַי בָּחַרְתִּי לְשַׁלֵּם מִכִּיסִי עַל דָּבָר חָשׁוּב, וְלֹא לְקַבֵּל חִנָּם?
  • דָּוִיד לֹא זָכָה לִבְנוֹת, אֲבָל הֵכִין הַכֹּל. עַל אֵיזֶה דָּבָר אֲנִי יָכוֹל לְהָכִין עֲבוּר מִישֶׁהוּ אַחֵר שֶׁיַּשְׁלִים?
  • שְׁלֹמֹה נִקְרָא ׳אִישׁ מְנוּחָה׳ וְ׳שָׁלוֹם׳. אֵיךְ אֲנִי יָכוֹל לְהָבִיא מְנוּחָה וְשָׁלוֹם לַמָּקוֹם שֶׁבּוֹ אֲנִי נִמְצָא?

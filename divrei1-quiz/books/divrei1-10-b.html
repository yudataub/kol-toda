הָעָם מִתְנַדֵּב לַמִּקְדָּשׁ בְּלֵב שָׁלֵם, דָּוִיד מְבָרֵךְ אֶת ה׳ ׳לְךָ ה׳ הַגְּדֻלָּה׳ וְאוֹמֵר ׳כִּי מִמְּךָ הַכֹּל וּמִיָּדְךָ נָתַנּוּ לָךְ׳, וּשְׁלֹמֹה מוּמְלָךְ

🏗️

📜 הַפָּסוּקשְׁלֹמֹה בְנִי אֶחָד בָּחַר בּוֹ אֱלֹהִים נַעַר וָרָךְ וְהַמְּלָאכָה גְדוֹלָה כִּי לֹא לְאָדָם הַבִּירָה כִּי לַה׳ אֱלֹהִים

💛 אַהֲבַת הַשֵּׁם: ה׳ אֵינוֹ מְבַקֵּשׁ מִמֶּנִּי יוֹתֵר מִמָּה שֶׁאֲנִי יָכוֹל, וְנוֹתֵן לִי עֶזְרָה 💛

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק כ״ט, פָּסוּק א׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּבְּרָכָה

🏗️ רַשִׁ״י: ׳לֹא לְאָדָם הַבִּירָה׳

רַשִׁ״י מְבָאֵר: ׳שְׁלֹמֹה בְנִי אֶחָד – וַאֲפִלּוּ אִם הָיוּ מֵאָה לֹא דַּי לְבָנוֹתוֹ, וְעוֹד שֶׁהוּא נַעַר וָרָךְ׳. וְעַל ׳וְהַמְּלָאכָה גְדוֹלָה׳: ׳אִם תֹּאמַר תַּקְטִין הַבַּיִת – זֶה אִי אֶפְשָׁר, כִּי לֹא לְאָדָם הַבִּירָה כִּי אִם לַה׳ אֱלֹהִים הַגָּדוֹל וְהַנּוֹרָא... וְאַף עַל פִּי כֵן אַל תִּבָּהֵל לִבְנוֹת בַּיִת גָּדוֹל כָּזֶה, כִּי אֲנִי בְּכָל כֹּחִי הֲכִינוֹתִי׳.

רַשִׁ״י: הַבַּיִת גָּדוֹל כִּי הוּא לַה׳, וְדָוִיד הֵכִין כָּל כָּךְ הַרְבֵּה כְּדֵי שֶׁלְּשְׁלֹמֹה יִהְיֶה קַל.

📜 הַפָּסוּקוַיִּשְׂמְחוּ הָעָם עַל הִתְנַדְּבָם כִּי בְּלֵב שָׁלֵם הִתְנַדְּבוּ לַה׳ וְגַם דָּוִיד הַמֶּלֶךְ שָׂמַח שִׂמְחָה גְדוֹלָה

💛 אַהֲבַת הַשֵּׁם: ה׳ שָׂמֵחַ בְּמַתָּנָה שֶׁבָּאָה מֵרָצוֹן 💛

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק כ״ט, פָּסוּק ט׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּבְּרָכָה

😊 רַשִׁ״י: לֵב אֶחָד, לֹא שְׁנֵי לְבָבוֹת

הַפָּסוּק: ׳וַיִּשְׂמְחוּ הָעָם עַל הִתְנַדְּבָם כִּי בְּלֵב שָׁלֵם הִתְנַדְּבוּ׳. רַשִׁ״י מְבָאֵר: ׳בְּלֵב שָׁלֵם – בְּלֵב אֶחָד נָתְנוּ בִּרְצוֹן נַפְשָׁם וְלֹא בִּשְׁנֵי לְבָבוֹת, שֶׁיֵּשׁ אָדָם שֶׁנּוֹתֵן בְּעַל כָּרְחוֹ וְלֹא בְּנֶפֶשׁ חֲפֵצָה, אוֹ שֶׁמִּתְבַּיֵּשׁ מֵאֲחֵרִים׳. וְעַל דָּוִיד: ׳שָׂמַח שִׂמְחָה גְּדוֹלָה – יוֹתֵר מִכֻּלָּם, כִּי הוּא בַּעַל הַמַּעֲשֶׂה׳.

רַשִׁ״י: ׳בְּלֵב שָׁלֵם׳ – נָתְנוּ מֵרָצוֹן, לֹא בְּעַל כָּרְחָם וְלֹא מִבּוּשָׁה; וְדָוִיד שָׂמַח יוֹתֵר מִכֻּלָּם.

👴

📜 הַפָּסוּקוַיְבָרֶךְ דָּוִיד אֶת ה׳ לְעֵינֵי כָּל הַקָּהָל וַיֹּאמֶר דָּוִיד בָּרוּךְ אַתָּה ה׳ אֱלֹהֵי יִשְׂרָאֵל אָבִינוּ

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹמֵר אֶת הַנְּדָרִים הַטּוֹבִים שֶׁל הָאָבוֹת, וְאֶת שֶׁלָּנוּ 💛

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק כ״ט, פָּסוּק י׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּבְּרָכָה

👴 רַשִׁ״י: ׳יִשְׂרָאֵל אָבִינוּ׳

דָּוִיד אָמַר ׳בָּרוּךְ אַתָּה ה׳ אֱלֹהֵי יִשְׂרָאֵל אָבִינוּ׳. רַשִׁ״י מְבָאֵר: ׳לְפִיכָךְ הִזְכִּיר יִשְׂרָאֵל יוֹתֵר מֵאַבְרָהָם וְיִצְחָק, לְפִי שֶׁיַּעֲקֹב גַּם הוּא נָדַר, דִּכְתִיב ״וַיִּדַּר יַעֲקֹב נֶדֶר״׳. ״יִשְׂרָאֵל אָבִינוּ״ הוּא ״יִשְׂרָאֵל סָבָא״, וְהַנּוֹדֵר צָרִיךְ לְהַזְכִּיר אֶת יִשְׂרָאֵל בְּנִדְרוֹ. דָּוִיד וְהָעָם נוֹדְרִים לַמִּקְדָּשׁ כְּמוֹ יַעֲקֹב.

רַשִׁ״י: דָּוִיד הִזְכִּיר ״אֱלֹהֵי יִשְׂרָאֵל אָבִינוּ״, כִּי יַעֲקֹב נָדַר, וְהַנְּדָבָה כְּמוֹ נֶדֶר.

📜 הַפָּסוּקלְךָ ה׳ הַגְּדֻלָּה וְהַגְּבוּרָה וְהַתִּפְאֶרֶת וְהַנֵּצַח וְהַהוֹד כִּי כֹל בַּשָּׁמַיִם וּבָאָרֶץ

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק כ״ט, פָּסוּק י״א (רַשִׁ״י)

🖐️ רַשִׁ״י: חֲמִשָּׁה שְׁבָחִים

רַשִׁ״י מְפָרֵשׁ אֶת ׳לְךָ ה׳ הַגְּדֻלָּה וְהַגְּבוּרָה וְהַתִּפְאֶרֶת וְהַנֵּצַח וְהַהוֹד׳ עַל מַה שֶּׁדָּוִיד קִבֵּל: ׳הַגְּדֻלָּה – שֶׁנָּתַן לוֹ כֹּחַ לְקַבֵּץ מָמוֹן כָּזֶה לְבִנְיַן בֵּית הַמִּקְדָּשׁ... וְהַגְּבוּרָה – שֶׁנָּתַתָּ לִי גְּבוּרָה לְהִלָּחֵם עִם שׂוֹנְאֵי יִשְׂרָאֵל... וְהַנֵּצַח – שֶׁנָּתַן לוֹ נִצָּחוֹן לִשְׁלֹל שְׁלַל הָעַמִּים לְצֹרֶךְ הַבִּנְיָן׳.

רַשִׁ״י: חֲמֵשֶׁת הַשְּׁבָחִים שֶׁל דָּוִיד הֵם הוֹדָיָה עַל הַכֹּחַ, הַנִּצָּחוֹן וְהַמָּמוֹן שֶׁקִּבֵּל לְבִנְיַן הַמִּקְדָּשׁ.

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק כ״ט, פָּסוּק י״ד (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּבְּרָכָה

🌾 רַשִׁ״י: ׳וּמִיָּדְךָ נָתַנּוּ לָךְ׳

רַשִׁ״י מְבָאֵר אֶת ׳וְכִי מִי אֲנִי וּמִי עַמִּי כִּי נַעְצֹר כֹּחַ לְהִתְנַדֵּב׳: ׳שֶׁדָּחַקְנוּ עַצְמֵנוּ לְהִתְנַדֵּב כָּל כָּךְ בַּנְּדָבָה הַזֹּאת, וְלֹא מִשֶּׁלָּנוּ אֶלָּא מִמְּךָ הַכֹּל, וְחָלַקְתָּ לָנוּ מְעַט. ״וּמִידְךָ נָתַנּוּ לָךְ״ – וּמַה שֶּׁנָּתַתָּ לָנוּ מִיָּדְךָ נָתַנּוּ לְךָ מִקְצָת׳.

רַשִׁ״י: אֵין לָנוּ גַּאֲוָה בַּנְּדָבָה – הַכֹּל מֵה׳, וְאָנוּ מַחֲזִירִים מִקְצָת.

הַפָּסוּק: ׳כִּי גֵרִים אֲנַחְנוּ לְפָנֶיךָ וְתוֹשָׁבִים כְּכָל אֲבֹתֵינוּ כַּצֵּל יָמֵינוּ עַל הָאָרֶץ וְאֵין מִקְוֶה׳. רַשִׁ״י מְבָאֵר: ׳כְּצֵל יָמֵינוּ – מְפֹרָשׁ בְּמִדְרַשׁ קֹהֶלֶת: וְלֹא כְּצֵל הָאִילָן... אֶלָּא כְּצֵל כַּנְפֵי הַדְּבוֹרָה, שֶׁיֵּשׁ לָהּ כְּנָפַיִם וְאֵין לָהּ צֵל׳. מְצוּדַת דָּוִד: ״אֵין מִקְוֶה – אֵין תִּקְוָה לָאָדָם לְהַאֲרִיךְ יָמָיו יוֹתֵר מִן הַקָּצוּב״. כִּי הַחַיִּים קְצָרִים, מוּטָב לָתֵת וְלִבְנוֹת.

רַשִׁ״י: חַיֵּי אָדָם כְּצֵל שֶׁל כְּנַף דְּבוֹרָה – קְצָרִים מְאֹד, וְלָכֵן חֲשׁוּב לִתֵּן וְלִבְנוֹת דְּבָרִים שֶׁנִּשְׁאָרִים.

רָמָה ב׳ – עִיּוּן בַּבְּרָכָה

🔍 רַשִׁ״י: ׳שָׁמְרָה זֹאת לְעוֹלָם׳

דָּוִיד מִתְפַּלֵּל: ׳שָׁמְרָה זֹאת לְעוֹלָם לְיֵצֶר מַחְשְׁבוֹת לְבַב עַמֶּךָ וְהָכֵן לְבָבָם אֵלֶיךָ׳. רַשִׁ״י מְבָאֵר: ׳שָׁמְרָה זֹאת – זְכֹר לָנוּ זֹאת הַנְּדָבָה לְמִצְוָה רַבָּה וְלִצְדָקָה׳, ׳וְהָכֵן לְבָבָם אֵלֶיךָ – שְׁנֵי לִבָּם שָׁלֵם לִהְיוֹת לֵב אֶחָד אֵלֶיךָ׳. הַתְּפִלָּה הִיא שֶׁהַלֵּב הַטּוֹב שֶׁנִּרְאָה בַּנְּדָבָה יִשָּׁאֵר לְתָמִיד.

רַשִׁ״י: דָּוִיד מִתְפַּלֵּל שֶׁה׳ יִזְכֹּר אֶת הַנְּדָבָה וְיָכִין אֶת לִבָּם שֶׁל הָעָם לִהְיוֹת לֵב אֶחָד אֵלָיו.

רָמָה ב׳ – עִיּוּן בַּבְּרָכָה

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מְגַלִּים אֶת הַמַּעֲמַקִּים: מָה הוּא הַלֵּב הַשָּׁלֵם, לָמָּה בְּרָכָה עַל ״יִשְׂרָאֵל אָבִינוּ״, חֲמִשָּׁה שְׁבָחִים עַל חֲמִשָּׁה דְּבָרִים שֶׁקִּבֵּל דָּוִיד, וְלָמָּה ׳כִּי מִמְּךָ הַכֹּל׳.

כִּי מִמְּךָ הַכֹּל וּמִיָּדְךָ נָתַנּוּ לָךְ

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • דָּוִיד אָמַר ׳כִּי מִמְּךָ הַכֹּל וּמִיָּדְךָ נָתַנּוּ לָךְ׳. מַה יֵשׁ לִי שֶׁקִּבַּלְתִּי מִה׳, וְאֵיךְ אֲנִי יָכוֹל ״לְהַחְזִיר״ חֵלֶק מִמֶּנּוּ?
  • הָעָם הִתְנַדֵּב ׳בְּלֵב שָׁלֵם׳ וְשָׂמַח. מָתַי נָתַתִּי צְדָקָה בְּשִׂמְחָה, וּמָתַי נָתַתִּי מִתּוֹךְ חוֹבָה?
  • דָּוִיד אָמַר ׳גֵּרִים אֲנַחְנוּ לְפָנֶיךָ וְתוֹשָׁבִים... כַּצֵּל יָמֵינוּ׳. מָה אֲנִי רוֹצֶה שֶׁיִּשָּׁאֵר מִימֵי הַקְּצָרִים?

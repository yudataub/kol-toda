בְּעָנְיִי הֲכִינוֹתִי

רָמָה א׳ – הַגֹּרֶן וְהַהֲכָנוֹת

דָּוִיד קוֹנֶה אֶת הַגֹּרֶן שֶׁבּוֹ יִבָּנֶה בֵּית הַמִּקְדָּשׁ ׳בְּכֶסֶף מָלֵא׳, וּמַכִּין לִשְׁלֹמֹה בְּנוֹ אֶת כָּל מַה שֶּׁצָּרִיךְ לַבִּנְיָן

רָמָה א׳ – הַגֹּרֶן וְהַהֲכָנוֹת

📍 ׳לְהָקִים מִזְבֵּחַ בְּגֹרֶן אָרְנָן׳

בַּזְּמַן שֶׁמַּגֵּפָה פָּגְעָה בָּעָם, ה׳ אָמַר לְדָוִיד דֶּרֶךְ גָּד הַחֹזֶה: ׳כִּי יַעֲלֶה דָוִיד לְהָקִים מִזְבֵּחַ לַה׳ בְּגֹרֶן אָרְנָן הַיְבֻסִי׳. דָּוִיד עוֹלֶה לַגֹּרֶן בִּדְבַר ה׳.

ה׳ אָמַר לְדָוִיד לְהָקִים מִזְבֵּחַ בְּגֹרֶן אָרְנָן, וְדָוִיד הָלַךְ מִיָּד.

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב כְּשֶׁאֲנַחְנוּ מְכַבְּדִים זֶה אֶת זֶה 💛

רָמָה א׳ – הַגֹּרֶן וְהַהֲכָנוֹת

🌾 ׳וְאָרְנָן דָּשׁ חִטִּים׳

אָרְנָן הָיָה בַּגֹּרֶן וְדָשׁ חִטִּים. כְּשֶׁרָאָה אֶת דָּוִיד ׳וַיֵּצֵא מִן הַגֹּרֶן וַיִּשְׁתַּחוּ לְדָוִיד אַפַּיִם אָרְצָה׳. הוּא כִּבֵּד אֶת הַמֶּלֶךְ.

אָרְנָן מְקַבֵּל אֶת הַמֶּלֶךְ בְּכָבוֹד וּמִשְׁתַּחֲוֶה לוֹ.

💰

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב לֵב נוֹתֵן, גַּם שֶׁל אָרְנָן וְגַם שֶׁל דָּוִיד 💛

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק כ״א, פְּסוּקִים כ״ב–כ״ג

💰 ׳בְּכֶסֶף מָלֵא תְּנֵהוּ לִי׳

דָּוִיד אוֹמֵר לְאָרְנָן: ׳תְּנָה לִּי מְקוֹם הַגֹּרֶן וְאֶבְנֶה בּוֹ מִזְבֵּחַ לַה׳ בְּכֶסֶף מָלֵא תְּנֵהוּ לִי׳. אָרְנָן עוֹנֶה: ׳קַח לָךְ... רְאֵה נָתַתִּי הַבָּקָר לָעֹלוֹת וְהַמּוֹרִגִּים לָעֵצִים וְהַחִטִּים לַמִּנְחָה׳ – הוּא מוּכָן לָתֵת הַכֹּל בְּחִנָּם.

דָּוִיד רוֹצֶה לִקְנוֹת בְּכֶסֶף מָלֵא, וְאָרְנָן מַצִּיעַ לָתֵת הַכֹּל בְּחִנָּם.

⚖️

💛 אַהֲבַת הַשֵּׁם: מַה שֶּׁאֲנִי נוֹתֵן לַה׳ צָרִיךְ לָבוֹא מִמֶּנִּי, וְזֶה מַה שֶּׁעוֹשֶׂה אוֹתוֹ יָקָר 💛

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק כ״א, פָּסוּק כ״ד

רָמָה א׳ – הַגֹּרֶן וְהַהֲכָנוֹת

⚖️ ׳לֹא אֶשָּׂא אֲשֶׁר לְךָ לַה׳׳

דָּוִיד מְסָרֵב: ׳לֹא כִּי קָנֹה אֶקְנֶה בְּכֶסֶף מָלֵא כִּי לֹא אֶשָּׂא אֲשֶׁר לְךָ לַה׳ וְהַעֲלוֹת עוֹלָה חִנָּם׳. הוּא לֹא רוֹצֶה לְהַקְרִיב לַה׳ מַשֶּׁהוּ שֶׁלֹּא עָלָה לוֹ כְּלוּם. קָרְבָּן צָרִיךְ לִהְיוֹת מִשֶּׁלִּי.

דָּוִיד לֹא מְקַבֵּל חִנָּם: קָרְבָּן לַה׳ צָרִיךְ לִהְיוֹת מִשֶּׁלּוֹ.

🔥

💛 אַהֲבַת הַשֵּׁם: ה׳ עוֹנֶה לִתְפִלָּה שֶׁל מִי שֶׁנּוֹתֵן מִלִּבּוֹ 💛

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק כ״א, פְּסוּקִים כ״ה–כ״ז

רָמָה א׳ – הַגֹּרֶן וְהַהֲכָנוֹת

🔥 ׳וַיַּעֲנֵהוּ בָאֵשׁ׳

דָּוִיד נוֹתֵן לְאָרְנָן ׳שִׁקְלֵי זָהָב מִשְׁקָל שֵׁשׁ מֵאוֹת׳ וּבוֹנֶה שָׁם מִזְבֵּחַ. ׳וַיַּעַל עֹלוֹת וּשְׁלָמִים וַיִּקְרָא אֶל ה׳ וַיַּעֲנֵהוּ בָאֵשׁ מִן הַשָּׁמַיִם עַל מִזְבַּח הָעֹלָה׳. ה׳ עָנָה לוֹ, וְהַמַּגֵּפָה נֶעֶצְרָה.

דָּוִיד שִׁלֵּם שֵׁשׁ מֵאוֹת שִׁקְלֵי זָהָב, בָּנָה מִזְבֵּחַ, וְה׳ עָנָה לוֹ בְּאֵשׁ.

5
🏛️

💛 אַהֲבַת הַשֵּׁם: ה׳ בָּחַר מָקוֹם בִּשְׁבִילֵנוּ לָבוֹא אֵלָיו 💛

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק כ״ב, פָּסוּק א׳

רָמָה א׳ – הַגֹּרֶן וְהַהֲכָנוֹת

🏛️ ׳זֶה הוּא בֵּית ה׳ הָאֱלֹהִים׳

דָּוִיד מֵבִין מַה קָּרָה: ׳וַיֹּאמֶר דָּוִיד זֶה הוּא בֵּית ה׳ הָאֱלֹהִים וְזֶה מִזְבֵּחַ לְעֹלָה לְיִשְׂרָאֵל׳. הַמָּקוֹם הַזֶּה, גֹּרֶן אָרְנָן, הוּא הַמָּקוֹם שֶׁבּוֹ יִבָּנֶה בֵּית הַמִּקְדָּשׁ.

דָּוִיד מַכְרִיז: גֹּרֶן אָרְנָן הוּא הַמָּקוֹם שֶׁבּוֹ יִבָּנֶה בֵּית ה׳.

6
🧱

💛 אַהֲבַת הַשֵּׁם: ה׳ שָׂמֵחַ כְּשֶׁאֲנִי מַכִּין מֵרֹאשׁ לַדָּבָר הַטּוֹב 💛

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק כ״ב, פְּסוּקִים ב׳–ד׳

רָמָה א׳ – הַגֹּרֶן וְהַהֲכָנוֹת

🧱 ׳אַבְנֵי גָזִית... וּבַרְזֶל לָרֹב׳

דָּוִיד מַתְחִיל לְהָכִין: ׳וַיַּעֲמֵד חֹצְבִים לַחְצוֹב אַבְנֵי גָזִית לִבְנוֹת בֵּית הָאֱלֹהִים׳, ׳וּבַרְזֶל לָרֹב לַמִּסְמְרִים לְדַלְתוֹת הַשְּׁעָרִים וְלַמְחַבְּרוֹת... וְעֵצֵי אֲרָזִים לְאֵין מִסְפָּר׳, אֲשֶׁר הֵבִיאוּ הַצִּידֹנִים וְהַצֹּרִים. כָּל הַחוֹמְרִים מוּכָנִים.

דָּוִיד הֵכִין אֲבָנִים, בַּרְזֶל, נְחֹשֶׁת וַאֲרָזִים לָרֹב לַבִּנְיָן.

👦

💛 אַהֲבַת הַשֵּׁם: ה׳ דּוֹאֵג לַדּוֹר הַבָּא, וְהוֹרֵי שֶׁלִּי דּוֹאֲגִים לִי כְּמוֹ שֶׁדָּוִיד דָּאַג לִשְׁלֹמֹה 💛

📖 דִּבְרֵי הַיָּמִים א, פֶּרֶק כ״ב, פְּסוּקִים ה׳, ט׳

רָמָה א׳ – הַגֹּרֶן וְהַהֲכָנוֹת

👦 ׳שְׁלֹמֹה בְנִי נַעַר וָרָךְ׳

דָּוִיד אוֹמֵר: ׳שְׁלֹמֹה בְנִי נַעַר וָרָךְ וְהַבַּיִת לִבְנוֹת לַה׳ לְהַגְדִּיל לְמַעְלָה... אָכִינָה נָּא לוֹ׳. הוּא יוֹדֵעַ שֶׁהַבֵּן עֲדַיִן צָעִיר, וְלָכֵן הוּא מַכִּין לוֹ אֶת הַכֹּל. וְה׳ אָמַר עַל שְׁלֹמֹה: ׳הוּא יִהְיֶה אִישׁ מְנוּחָה... שָׁלוֹם וָשֶׁקֶט אֶתֵּן עַל יִשְׂרָאֵל׳.

דָּוִיד מַכִּין לִשְׁלֹמֹה בְּנוֹ הַצָּעִיר, שֶׁיִּהְיֶה ׳אִישׁ מְנוּחָה׳ וְשָׁלוֹם.

מָה לוֹקְחִים מִרָמָה א׳?

דָּוִיד מוֹצֵא אֶת הַמָּקוֹם שֶׁבּוֹ יִבָּנֶה הַמִּקְדָּשׁ, וְקוֹנֶה אוֹתוֹ בְּכֶסֶף מָלֵא. אַחַר כָּךְ הוּא מַכִּין זָהָב, אֲבָנִים וַאֲרָזִים לִשְׁלֹמֹה, כִּי הוּא יֵדַע שֶׁהוּא עַצְמוֹ לֹא יִבְנֶה. ה׳ אוֹהֵב מִי שֶׁנּוֹתֵן מִלִּבּוֹ וְעוֹשֶׂה לְמַעַן הַדּוֹר הַבָּא.

🤔 חוֹשְׁבִים עַל זֶה

  • אָרְנָן הִצִּיעַ לָתֵת לְדָוִיד הַכֹּל בְּחִנָּם, אֲבָל דָּוִיד אָמַר ׳לֹא כִּי קָנֹה אֶקְנֶה בְּכֶסֶף מָלֵא׳. מָתַי בָּחַרְתִּי לְשַׁלֵּם מִכִּיסִי עַל דָּבָר חָשׁוּב, וְלֹא לְקַבֵּל חִנָּם?
  • דָּוִיד לֹא זָכָה לִבְנוֹת, אֲבָל הֵכִין הַכֹּל. עַל אֵיזֶה דָּבָר אֲנִי יָכוֹל לְהָכִין עֲבוּר מִישֶׁהוּ אַחֵר שֶׁיַּשְׁלִים?
  • שְׁלֹמֹה נִקְרָא ׳אִישׁ מְנוּחָה׳ וְ׳שָׁלוֹם׳. אֵיךְ אֲנִי יָכוֹל לְהָבִיא מְנוּחָה וְשָׁלוֹם לַמָּקוֹם שֶׁבּוֹ אֲנִי נִמְצָא?

פְּתִיחָה: מָעוֹז צוּר יְשׁוּעָתִי

בְּלֵב הַכְּאֵב הַגָּדוֹל שֶׁל מִלְחֶמֶת הַתְּקוּמָה – הָאֲבֵדוֹת הַקָּשׁוֹת, הַפְּצוּעִים, הַחֲטוּפִים וְהַשְּׁכוֹל – צָמְחוּ עָצְמוֹת אֲדִירוֹת שֶׁל גְּבוּרָה, אַחְדוּת וֶאֱמוּנָה. חוֹבֶרֶת זוֹ נוֹעֲדָה לְהָאִיר אֶת הַצַּד הַשֵּׁנִי שֶׁל הַמַּטְבֵּעַ: הַנִּסִּים הַגְּדוֹלִים שֶׁרָאִינוּ. כְּבָר לִמְּדוּנוּ נְבִיאִים וַחֲכָמִים (יִרְמְיָהוּ ט"ז, יד-טו; בְּרָכוֹת יב.), שֶׁנִּסֵּי גְּאֻלָּתֵנוּ הָאַחֲרוֹנָה יִהְיוּ גְּדוֹלִים כָּל כָּךְ עַד שֶׁיַּשְׁכִּיחוּ אֶת נִסֵּי יְצִיאַת מִצְרַיִם. יִתָּכֵן שֶׁמִּלְחָמָה זוֹ הִיא תְּחִלַּת הִתְגַּשְּׁמוּתוֹ שֶׁל חָזוֹן זֶה. רַק מִתּוֹךְ רְאִיַּת הָעָצְמָה וְהִתְבּוֹנְנוּת בְּיַד ה' הַמְכַוֶּנֶת, נוּכַל לִשְׁאֹב כֹּחוֹת לְהִתְמוֹדֵד עִם הַקְּשָׁיִים וּלְהִתְרוֹמֵם לַשָּׁלָב הַבָּא בִּגְאֻלָּתֵנוּ.

הַנִּסִּים בַּמִּלְחָמָה הַזּוֹ אֵינָם רַק סִדְרַת אֵרוּעִים מִקְרִיִּים, אֶלָּא "פַּאזֶל" אֱ-לֹהִי מֻרְכָּב, שֶׁבּוֹ כָּל נֵס הָיָה הֲכָנָה לַנֵּס הַבָּא, וְכֻלָּם יַחַד הוֹבִילוּ לַנִּצָּחוֹן הַגָּדוֹל עַל אוֹיְבֵינוּ. פִּרְסוּם הַנֵּס וְהַנִּצָּחוֹן הוּא מִצְוָה שֶׁל מַמָּשׁ. עָלֵינוּ לְסַפֵּר אֶת נִפְלְאוֹת ה' וְאֶת מְסִירוּת הַנֶּפֶשׁ שֶׁל חַיָּלֵינוּ, שֶׁבְּמִלְחַמְתָּם וּבְמוֹתָם הִצִּילוּ אֶת מְדִינַת יִשְׂרָאֵל. שִׁדּוּר נִצָּחוֹן מְחַזֵּק אֶת בִּטְחוֹנָהּ שֶׁל הַמְּדִינָה, בְּעוֹד שֶׁשִּׁדּוּר הֶפְסֵד מֵבִיא אֶת הַמִּלְחָמָה הַבָּאָה.

חוֹבֶרֶת זוֹ מְחֻלֶּקֶת לִשְׁנֵי חֲלָקִים. בַּחֵלֶק הָרִאשׁוֹן נִקְרָא בְּכָל לַיְלָה מִלֵּילוֹת חֲנֻכָּה קֶטַע קָצָר, הַמְּתָאֵר נֵס אַחֵר וְחֵלֶק נוֹסָף מֵהַפַּאזֶל הָאֱ-לֹהִי. בַּחֵלֶק הַשֵּׁנִי נַרְחִיב וְנַעֲמִיק בַּמַּבָּט הָאֱמוּנִי עַל כָּל הַמַּהֲלָךְ.

לַיּוֹם הָרִאשׁוֹן שֶׁל חֲנֻכָּה

קֹדֶם לַמִּלְחָמָה הָיִינוּ שְׁרוּיִים בְּפִלּוּג וּבְקִטּוּב פְּנִימִי עָמֹק, שֶׁהִגִּיעַ לְשִׂיאוֹ בְּאֵרוּעֵי יוֹם הַכִּפּוּרִים בְּתֵל אָבִיב. אוֹיְבֵינוּ, שֶׁצָּפוּ בָּנוּ מֵהַצַּד, רָאוּ עַם מְפֻצָּל וְחָלָשׁ, וְחָשְׁבוּ שֶׁזּוֹהִי הִזְדַּמְּנוּת חֲלוֹמִית לְהַכְרִיעַ אוֹתָנוּ. בְּשִׂמְחַת תּוֹרָה הִכָּה בָּנוּ פּוֹגְרוֹם נוֹרָא: 1,200 יְהוּדִים נִרְצְחוּ בְּאַכְזָרִיּוּת, וְ-251 נוֹסָפִים נֶחְטְפוּ לְשִׁבְיָה. הַכְּאֵב הָיָה עָצוּם, אַךְ דַּוְקָא מִתּוֹךְ הַשֶּׁבֶר הַגָּדוֹל הַזֶּה צָמַח הַנֵּס הָרִאשׁוֹן וְהַמְּכוֹנֵן – "נֵס הָאַחְדוּת".

נֵס הָאַחְדוּת – נְקֻדַּת הַמִּפְנֶה

מִתּוֹךְ הַכְּאֵב, עַם יִשְׂרָאֵל בָּחַר לְהִתְאַחֵד מֵחָדָשׁ. הֵבַנּוּ שֶׁכֻּלָּנוּ יַחַד בַּסִּירָה אַחַת, וְשֶׁאוֹיְבֵינוּ מְבַקְּשִׁים לְהַשְׁמִיד אֶת כֻּלָּנוּ. הָאַחְדוּת הָיְתָה נִפְלָאָה וּבָאָה לִידֵי בִּטּוּי בְּהִתְגַּיְּסוּת מוּפְלָאָה לְמִלּוּאִים, שֶׁכָּלְלָה גַּם אֶת אֵלּוּ שֶׁהִצְהִירוּ שֶׁלֹּא יָבוֹאוּ. עַשְׂרוֹת אַלְפֵי חַיָּלִים שָׁבוּ מִחוּ"ל, וְכ-45,000 חַיָּלִים שֶׁכְּבָר הָיָה לָהֶם פְּטוֹר מִמִּלּוּאִים, הִתְעַקְּשׁוּ לְהִתְגַּיֵּס. הָאַחְדוּת הַזּוֹ הָיְתָה גַּם נַחֲלָתָהּ שֶׁל יַהֲדוּת הַתְּפוּצוֹת שֶׁהִתְגַּיְּסָה בְּכָל כֹּחָהּ.

בָּרֶגַע שֶׁעַם יִשְׂרָאֵל הִתְאַחֵד, הֵחֵל הַמִּפְנֶה בַּמִּלְחָמָה. הַכְּאֵב הָיָה עֲדַיִן גָּדוֹל, אַךְ מֵעַתָּה וְאֵילָךְ, הֵחֵלּוּ לְהִתְרַחֵשׁ נִסִּים וְנִפְלָאוֹת שֶׁלֹּא נִרְאוּ כְּמוֹתָם בְּמֶשֶׁךְ אַלְפֵי שָׁנִים.

לַיּוֹם הַשֵּׁנִי שֶׁל חֲנֻכָּה

הַפֵּרוּד הֵבִיא עָלֵינוּ אֶת הַמִּלְחָמָה, וְהָאַחְדוּת הֵחֵלָּה אֶת הַתְּקוּמָה. עוֹד בְּאוֹתוֹ יוֹם שֶׁל שִׂמְחַת תּוֹרָה, בּוֹ הִתְאַחַדְנוּ, זָכִינוּ לִרְאוֹת נִסִּים עֲצוּמִים שֶׁהָיוּ תְּחִלָּתוֹ שֶׁל הַמַּהְפָּךְ.

  1. הֲשָׁבַת מִלְחָמָה לָאוֹיֵב: בְּמֶשֶׁךְ כָּל הַדּוֹרוֹת, סָבַל עַם יִשְׂרָאֵל מִפּוֹגְרוֹמִים, וּמֵעוֹלָם לֹא הָיְתָה לָנוּ הַיְּכֹלֶת לְהָגֵן עַל עַצְמֵנוּ. זוֹ הַפַּעַם הָרִאשׁוֹנָה בַּהִיסְטוֹרְיָה שֶׁבְּתוֹךְ הַפּוֹגְרוֹם עַצְמוֹ, קָם הָעָם עַל רַגְלָיו וְנִלְחַם בִּגְבוּרָה. אֶזְרָחִים וְחַיָּלִים חֵרְפוּ נַפְשָׁם וְהֵשִׁיבוּ מִלְחָמָה שַׁעְרָה.
  2. נֵס הַהִתְגַּיְּסוּת: כֹּחֵנוּ הָאֲמִתִּי הִתְגַּלָּה בְּרֶגַע הָאֱמֶת. רָאִינוּ הִתְגַּיְּסוּת שֶׁל מֵאָה וּשְׁלוֹשִׁים אֲחוּזִים. אֲנָשִׁים עָזְבוּ הַכֹּל, בָּאָרֶץ וּבְחוּ"ל, וּבָאוּ לְהָגֵן עַל הַבַּיִת. גִּלִּינוּ שׁוּב אֶת הָאַחְוָה, שֶׁ"כָּל יִשְׂרָאֵל חֲבֵרִים".
  3. נֵס הַצָּפוֹן וְכִשָּׁלוֹן תָּכְנִית הָאוֹיֵב: לְחִזְבַּאלְלָה הָיָה צָבָא עָצוּם שֶׁהָיָה מוּכָן לְהִצְטָרֵף לַמַּתְקָפָה מֵהַצָּפוֹן וּלְהַגְדִּיל אֶת הָאָסוֹן בְּעַשְׂרוֹת מוֹנִים. אָנוּ יוֹדְעִים כַּיּוֹם שֶׁהָיְתָה תָּכְנִית מִתְקָפָה רַב-זִירָתִית לְהַשְׁמָדַת יִשְׂרָאֵל, וּבְחַסְדֵי ה' חִזְבַּאלְלָה לֹא הִצְטָרֵף, לְלֹא שׁוּם סִבָּה הֶגְיוֹנִית. נוֹסָף עַל כָּךְ, הָיָה נֵס גָּדוֹל בְּתוֹךְ הַחֹשֶׁךְ, שֶׁגַּם הֶחָמָאס לֹא מִמֵּשׁ אֶת תָּכְנִיתוֹ הַמְּלֵאָה לְהַגִּיעַ לְעָרִים כְּמוֹ קִרְיַת גַּת וְתֵל אָבִיב. מְחַבְּלָיו, שְׁטוּפֵי זִמָּה, הִתְעַכְּבוּ בְּבִזּוּעַ מַעֲשֵׂי זוועה בִּמְסִבַּת הַנּוֹבָה, וּבְכָךְ נִמְנַע אָסוֹן גָּדוֹל פִּי כַּמָּה.

בְּתוֹךְ הַחֹשֶׁךְ הַגָּדוֹל שֶׁל הַפּוֹגְרוֹם, זָרַח אוֹר. הַנֵּס הַבָּא, שֶׁעָתִיד הָיָה לְהִתְרַחֵשׁ, יְשַׁנֶּה לְגַמְרֵי אֶת פְּנֵי הַמַּעֲרָכָה בַּצָּפוֹן.

לַיּוֹם הַשְּׁלִישִׁי שֶׁל חֲנֻכָּה

בְּמֶשֶׁךְ שָׁנִים אֲרֻכּוֹת, בָּנָה חִזְבַּאלְלָה צָבָא עָצוּם שֶׁהִוָּה אִיּוּם קִיּוּמִי עַל מְדִינַת יִשְׂרָאֵל. תָּכְנִיתָם הָיְתָה מַפְחִידָה: שִׁגּוּר שְׁלוֹשֶׁת אֲלָפִים טִילִים בְּיוֹם וּפְלִישָׁה לְיִשּׁוּבִים בַּצָּפוֹן בְּמַטָּרָה לַהֲרֹג עַשְׂרוֹת אַלְפֵי יְהוּדִים. הֵם הָיוּ בְּטוּחִים בְּנִצְחוֹנָם. וְאָז הִגִּיעָה נְקֻדַּת הַמִּפְנֶה הַמַּכְרַעַת – "נֵס הַבִּיפֶּרִים".

בְּמִבְצָע גְּאוֹנִי שֶׁאֵין לוֹ תַּקְדִּים בַּהִיסְטוֹרְיָה, הָפְכוּ הַבִּיפֶּרִים וּמַכְשִׁירֵי הַקֶּשֶׁר שֶׁל אַלְפֵי מְחַבְּלֵי חִזְבַּאלְלָה לִכְלֵי נֶשֶׁק נֶגְדָּם. אַלְפֵי מְחַבְּלִים, בְּתוֹכָם מְפַקְּדִים בְּכִירִים, נִפְגְּעוּ וְנֻטְרְלוּ בְּרֶגַע אֶחָד. הַנֵּס הַגָּדוֹל לֹא רַק פָּגַע בַּמְּחַבְּלִים, אֶלָּא יָצַר פָּאנִיקָה וּמְהוּמָה בִּשְׁבִילֵיהֶם. הֵם אִבְּדוּ אֶת יְכֹלֶת הַתִּקְשֹׁרֶת הַצְּבָאִית וְהָפְכוּ לִמְשֻׁתָּקִים.

חִסּוּל "רֹאשׁ הַנָּחָשׁ" וְהַמַּהְפָּךְ הָאַסְטְרָטֵגִי

נַסְרַאלְלָה, שֶׁעַד אָז הִתְרַבְרֵב וְהִתְפָּאֵר, עָשָׂה טָעוּת גּוֹרָלִית. הוּא כִּנֵּס אֶת צַמֶּרֶת הָאִרְגּוּן בְּבוּנְקֶר, וּבְמִבְצָע מְדֻיָּק וּמַדְהִים, צַהַ"ל הִגִּיעַ אֲלֵיהֶם וְחִסֵּל כִּמְעַט אֶת כָּל הַנְהָגַת חִזְבַּאלְלָה. תוֹךְ שָׁבוּעוֹת סְפוּרִים, פֵּרַקְנוּ אֶת מַה שֶּׁהֵם בָּנוּ בְּמֶשֶׁךְ עֲשׂוֹרִים. אוֹיֵב שֶׁאִיֵּם עַל עֶצֶם קִיּוּמֵנוּ, הָפַךְ לְאִרְגּוּן פָּגִיעַ וּמְשֻׁתָּק, שֶׁאֵינוֹ מְסֻגָּל עוֹד לְמַמֵּשׁ אֶת זְמָמוֹ. נֵס עָצוּם שֶׁל "וְנַהֲפוֹךְ הוּא".

אִישׁ לֹא צִפָּה שֶׁמִּיָּד לְמָחֳרָת יִתְרַחֵשׁ נֵס נוֹסָף, גָּדוֹל אַף יוֹתֵר, שֶׁהִתְאַפְּשֵׁר בִּזְכוּת הַמַּהֲלָךְ הַזֶּה.

לַיּוֹם הָרְבִיעִי שֶׁל חֲנֻכָּה

סוּרְיָה. אֶחָד מֵאוֹיְבֵינוּ הַמְּסֻכָּנִים וְהַוָּתִיקִים בְּיוֹתֵר. מִיָּד עִם סִיּוּם הַמַּכָּה לְחִזְבַּאלְלָה, הִתְרַחֵשׁ נֵס נוֹסָף: נְפִילַת מִשְׁטַר אַסַאד. בְּמַהֲלָךְ מַפְתִּיעַ, הִצְלִיחוּ הַמּוֹרְדִים בְּסוּרְיָה, תּוֹךְ 12 יָמִים בִּלְבַד, לְהַפִּיל אֶת הַמִּשְׁטָר. כָּאן, עוֹד חֵלֶק מֵהַפַּאזֶל הָאֱ-לֹהִי נִכְנַס לִמְקוֹמוֹ בְּדִיּוּק עוֹצֵר נְשִׁימָה, מַהֲלָךְ עַל לוּחַ הַשַּׁחְמָט הַגֵּאוֹפּוֹלִיטִי שֶׁשּׁוּם אַסְטְרָטֵג אֱנוֹשִׁי לֹא יָכוֹל הָיָה לַחֲזוֹת.

לְלֹא "נֵס הַבִּיפֶּרִים", לֹא הָיְתָה יִשְׂרָאֵל מְחַסֶּלֶת אֶת חִזְבַּאלְלָה. לְלֹא חִסּוּל חִזְבַּאלְלָה, לֹא הָיוּ מַצְלִיחִים הַמּוֹרְדִים לְחַסֵּל אֶת מִשְׁטַר אַסַאד.

בְּהַחְלָטָה נוֹעֶזֶת, יָצָא צַהַ"ל לְמִבְצָע "חֵץ הַבָּשָׁן" – מַתְקָפָה יְזוּמָה לְהַשְׁמָדַת הַצָּבָא הַסּוּרִי. מַה שֶּׁקָּרָה בַּיּוֹמַיִם הַבָּאִים הָיָה חֲסַר תַּקְדִּים. בְּשִׁלּוּב כֹּחוֹת שֶׁל חֵיל הָאֲוִיר, הַיָּם וְהַיַּבָּשָׁה, הִשְׁמִיד צַהַ"ל אֶת הַצָּבָא הַסּוּרִי כֻּלּוֹ. הַנָּתוּן הַמַּדְהִים בְּיוֹתֵר, שֶׁאֵין לוֹ אָח וָרֵעַ בַּהִיסְטוֹרְיָה הַצְּבָאִית, הוּא שֶׁהַהֶשֵּׂג הָאַדִּיר הַזֶּה הֻשַּׂג לְלֹא אַף נִפְגָּע אֶחָד לְכֹחוֹתֵינוּ. הָאִיּוּם הַסּוּרִי הֻסַּר. אֲבָל עוֹד נוֹתַר אִיּוּם גָּדוֹל יוֹתֵר – אִירָאן.

לַיּוֹם הַחֲמִישִׁי שֶׁל חֲנֻכָּה

בְּעוֹד שֶׁמַּעֲצָמוֹת הָעוֹלָם, כְּמוֹ אַרְצוֹת הַבְּרִית, צָרְפַת וְאַנְגְּלִיָּה, חָשְׁשׁוּ לְהִתְעַמֵּת עִם אִירָאן, דַּוְקָא יִשְׂרָאֵל הַקְּטַנָּה הֶחְלִיטָה לֹא לִפְחֹד. יָצָאנוּ לְמִלְחָמָה מוּל אִימְפֶּרְיָה שֶׁל 90 מִילְיוֹן אִישׁ, בְּמֶרְחָק 2,000 קִילוֹמֶטְרִים מֵהַבַּיִת. תוֹךְ 12 יָמִים, שֻׁנָּה הָעוֹלָם. צַהַ"ל הִשִּׂיג שְׁלִיטָה מֻחְלֶטֶת בִּשְׁמֵי אִירָאן, הִשְׁמִיד אֶת פְּרוֹיֶקְט הַגַּרְעִין, חִסֵּל אֶת רָאשֵׁי הַצָּבָא וּמַדְּעָנֵי הַגַּרְעִין, וּפָגַע אֲנוּשׁוֹת בְּכַלְכָּלָתָם. כְּדֵי לְהָבִין אֶת גֹּדֶל הַנֵּס, נַשְׁוֶה אוֹתוֹ לְמִלְחֶמֶת הַמִּפְרָץ.

הַשְׁוָאָה לְמִלְחֶמֶת הַמִּפְרָץ שֶׁל אַרְצוֹת הַבְּרִית בְּעִירָאק:

  • קוֹאָלִיצְיָה: 33 מְדִינוֹת.
  • כֹּחַ אָדָם: כְּמִילְיוֹן חַיָּלִים.
  • מֶשֶׁךְ: לְמַעְלָה מֵחֲצִי שָׁנָה.
  • אֲבֵדוֹת: מֵאוֹת הֲרוּגִים וּמְטוֹסִים רַבִּים שֶׁהֻפְּלוּ. טִיל סְקַאד עִירָאקִי אֶחָד הָרַג 28 חַיָּלִים אָמֶרִיקָאִים בְּמַכָּה אַחַת.

לְעֻמַּת זֹאת, מִלְחֶמֶת יִשְׂרָאֵל בְּאִירָאן:

  • קוֹאָלִיצְיָה: כִּמְעַט לְבַד (עִם סִיּוּעַ אָמֶרִיקָאִי בַּסּוֹף).
  • כֹּחַ אָדָם: כֹּחַ קָטָן וּמְדֻיָּק.
  • מֶשֶׁךְ: 12 יָמִים.
  • אֲבֵדוֹת: אַף חַיָּל לֹא נֶהֱרַג וְלֹא נִפְגַּע. אַף מָטוֹס לֹא הֻפַּל.

נוֹסָף עַל כָּךְ, הַתַּחֲזִיּוֹת דִּבְּרוּ עַל עַשְׂרוֹת אַלְפֵי הֲרוּגִים בָּעֹרֶף הַיִּשְׂרְאֵלִי. בַּפֹּעַל, לַמְרוֹת הַכְּאֵב עַל כָּל פְּגִיעָה, זָכִינוּ לְהַשְׁגָּחָה פְּרָטִית וְהַפְּגִיעָה הָיְתָה מִזְעָרִית לְעֻמַּת הַצָּפוּי. כֵּיצַד אֵרַע נֵס כֹּה מַשְׁמָעוּתִי, שֶׁאֵין לוֹ תַּקְדִּים הִיסְטוֹרִי?

לַיּוֹם הַשִּׁשִּׁי שֶׁל חֲנֻכָּה

נֵס פּוּרִים לִמֵּד אוֹתָנוּ שֶׁכְּדֵי לִרְאוֹת אֶת יַד ה', צָרִיךְ לְהִתְבּוֹנֵן בְּכָל הַפְּרָטִים הַקְּטַנִּים – בַּ"פַּאזֶל" הָאֱ-לֹהִי הַשָּׁלֵם. רַק כַּאֲשֶׁר מְחַבְּרִים אֶת מִשְׁתֵּה אֲחַשְׁוֵרוֹשׁ, סֵרוּבָהּ שֶׁל וַשְׁתִּי, מַעֲשֵׂה בִּגְתָן וָתֶרֶשׁ, וּמָרְדְּכַי עַל הַסּוּס, מְבִינִים אֶת הַהַשְׁגָּחָה הָעֶלְיוֹנָה. כָּךְ גַּם בְּמִלְחַמְתֵּנוּ. עַל מִי סָמְכָה אִירָאן כָּל הַשָּׁנִים? עַל בַּעֲלֵי בְּרִיתָהּ: רוּסְיָה, סוּרְיָה, חִזְבַּאלְלָה וְחַמַאס. וְכָאן הִתְגַּלָּה הַפַּאזֶל בִּמְלוֹא תִּפְאַרְתּוֹ. מִי יָכוֹל הָיָה לְתַזְמֵר סִדְרָה כָּזוֹ? אֵיזוֹ יָד כִּוְּנָה אֶת הָאֵרוּעִים הַשּׁוֹנִים הָאֵלֶּה – מִלְחָמָה בְּאֵירוֹפָּה, יָהֳרָתוֹ שֶׁל אִרְגּוּן טֶרוֹר, מַהֲלָךְ טֶכְנוֹלוֹגִי גְּאוֹנִי – לְסִימְפוֹנְיָה אַחַת שֶׁל יְשׁוּעָה?

  1. מַדּוּעַ רוּסְיָה לֹא סִיְּעָה? מִכֵּיוָן שֶׁהָיְתָה שְׁקוּעָה בַּמִּלְחָמָה עִם אוּקְרָאִינָה – מִלְחָמָה שֶׁפָּרְצָה דַּוְקָא כַּאֲשֶׁר טְרַאמְפּ לֹא הָיָה בַּשִּׁלְטוֹן.
  2. מַדּוּעַ חִזְבַּאלְלָה לֹא סִיֵּעַ? מִכֵּיוָן שֶׁבְּחַסְדֵי ה' חֻסַּל בְּעִקְבוֹת "נֵס הַבִּיפֶּרִים".
  3. מַדּוּעַ סוּרְיָה לֹא סִיְּעָה? מִכֵּיוָן שֶׁצְּבָאָהּ הֻשְׁמַד בְּ"מִבְצָע חֵץ הַבָּשָׁן", שֶׁהִתְאַפְּשֵׁר רַק לְאַחַר נְפִילַת מִשְׁטַר אַסַאד, שֶׁנָּפַל בִּזְכוּת חִסּוּל חִזְבַּאלְלָה.

וְכָאן הִגִּיעַ הַ"וְנַהֲפוֹךְ הוּא" הַגָּדוֹל. סוּרְיָה לֹא רַק שֶׁלֹּא הִפְרִיעָה, אֶלָּא הָפְכָה לְגוֹרֵם מְסַיֵּעַ. הִיא הָפְכָה לִתְחֲנַת מַעֲבָר וְתִדְלוּק עֲבוּר מְטוֹסֵי צַהַ"ל בְּדַרְכָּם לְתַקֹּף בְּאִירָאן. אוֹיֵב מַר הָפַךְ לִכְלִי עֵזֶר. וְהַפַּאזֶל עוֹד לֹא הֻשְׁלַם.

לַיּוֹם הַשְּׁבִיעִי שֶׁל חֲנֻכָּה

הַפַּאזֶל הָאֱ-לֹהִי הָלַךְ וְהִתְרַחֵב, וְכָלַל גּוֹרְמִים בִּלְתִּי צְפוּיִים לְגַמְרֵי. הַחוּתִ'ים בְּתֵימָן, שֶׁבְּמֶשֶׁךְ חֳדָשִׁים הִצִּיקוּ לָנוּ, הִתְבָּרְרוּ כְּ"מְאַמְּנִים" הֲכִי טוֹבִים שֶׁל חֵיל הָאֲוִיר. בִּזְכוּתָם, תִּרְגְּלוּ הַטַּיָּסִים תְּקִיפוֹת אֲרֻכּוֹת-טְוָח "עַל רָטֹב" לְמֶרְחָק שֶׁל 2,000 קִילוֹמֶטְרִים, וְהִגִּיעוּ מוּכָנִים וּמְאֻמָּנִים לַמִּלְחָמָה הַגְּדוֹלָה בְּאִירָאן. לַצַּד זֶה, "נֵס טְרַאמְפּ" הָיָה חֵלֶק מַכְרִיעַ. הָעֻבְדָּה שֶׁלֹּא נִבְחַר לְקָדֶנְצְיָה רְצוּפָה אִפְשְׁרָה אֶת מִלְחֶמֶת רוּסְיָה-אוּקְרָאִינָה (שֶׁהִשְׁאִירָה אֶת רוּסְיָה מְנֻטְרֶלֶת מִלְּסַיֵּעַ לְאִירָאן), וַחֲזָרָתוֹ לַשִּׁלְטוֹן סִפְּקָה סִיּוּעַ קְרִיטִי לְיִשְׂרָאֵל. עַל כָּךְ יֵשׁ לְהוֹסִיף אֶת הַנֵּס הַכַּלְכָּלִי (הַבּוּרְסָה, הַשֶּׁקֶל וְהַהַייטֶק שֶׁפָּרְחוּ בְּעֵת מִלְחָמָה) וְאֶת נֵס הַהֲגָנָה מִפְּנֵי הַטִּילִים: כ-27,000 טִילִים שֻׁגְּרוּ לְעֵבֶר יִשְׂרָאֵל, וְזָכִינוּ לְהֲגָנָה שֶׁמַּדְּעָנִים מַגְדִּירִים כְּנֵס גָּלוּי. הֵם טָעֲנוּ שֶׁ-85% יֵרוּט יִהְיֶה נֵס, וּבְחַסְדֵי ה' הִגַּעְנוּ לְ-99% – נָתוּן שֶׁאֵין לוֹ הֶסְבֵּר מַדָּעִי.

סִכּוּם הַפַּאזֶל הָאֱ-לֹהִי: סִבָּה וְתוֹצָאָה

  • הַפְסָדוֹ שֶׁל טְרַאמְפּ בַּבְּחִירוֹת הוֹבִיל לְמִלְחֶמֶת רוּסְיָה-אוּקְרָאִינָה ← רוּסְיָה לֹא יְכוֹלָה לַעֲזֹר לְאוֹיְבֵינוּ.
  • "נֵס הַבִּיפֶּרִים" הוֹבִיל לְחִסּוּל חִזְבַּאלְלָה ← חִזְבַּאלְלָה לֹא יָכֹל לַעֲזֹר לְסוּרְיָה וּלְאִירָאן.
  • חִסּוּל חִזְבַּאלְלָה הוֹבִיל לִנְפִילַת מִשְׁטַר אַסַאד ← צַהַ"ל הִשְׁמִיד אֶת הַצָּבָא הַסּוּרִי.
  • הַשְּׁמָדַת הַצָּבָא הַסּוּרִי הוֹבִילָה לְכָךְ שֶׁסּוּרְיָה הָפְכָה לְ"תַחֲנַת תִּדְלוּק" בַּדֶּרֶךְ לְאִירָאן.
  • הַתְּקִיפוֹת שֶׁל הַחוּתִ'ים הוֹבִילוּ לְכָךְ שֶׁטַּיָּסֵינוּ הָיוּ מְאֻמָּנִים לִתְקִיפָה אֲרֻכַּת טְוָח.

לַיּוֹם הַשְּׁמִינִי שֶׁל חֲנֻכָּה

הַנִּצָּחוֹן הַמֻּחְלָט שֶׁל יִשְׂרָאֵל וְהַפְּגָנַת הָעָצְמָה שֶׁלָּהּ בְּכָל מָקוֹם – מִתֵּימָן וְעַד קָטָאר – הִטִּילוּ פַּחַד גָּדוֹל עַל הָעוֹלָם הָעַרְבִי. הֵם הֵבִינוּ שֶׁעַם יִשְׂרָאֵל לֹא יִשְׁקֹט עַד שֶׁיָּשׁוּבוּ הַחֲטוּפִים, וְשֶׁצַּהַ"ל יַמְשִׁיךְ לְהַשְׁמִיד אֶת עַזָּה עַד הַסּוֹף. הָעָצְמָה הַזּוֹ, בְּשִׁלּוּב סִיּוּעוֹ שֶׁל טְרַאמְפּ, הוֹבִילָה לְנֵס "וְשָׁבוּ בָנִים": כָּל הַחֲטוּפִים חָזְרוּ! גַּם בַּתַּרְחִישִׁים הָאוֹפְּטִימִיִּים בְּיוֹתֵר, אִישׁ לֹא הֶאֱמִין שֶׁנִּרְאֶה אֶת כֻּלָּם בַּבַּיִת. הָעִסְקָה לֹא נַעֲשְׂתָה מִתּוֹךְ חֻלְשָׁה, אֶלָּא מִתּוֹךְ עָצְמָה אֲדִירָה שֶׁל מְדִינָה שֶׁנִּצְּחָה אֶת אוֹיְבֶיהָ. נוֹסָף עַל כָּךְ, זָכִינוּ לְהָבִיא לְקֶבֶר יִשְׂרָאֵל כִּמְעַט אֶת כָּל הַחֲלָלִים שֶׁנֶּחְטְפוּ, וְאַף אֶת חַלְלֵי הֶעָבָר, הֲדָר גּוֹלְדִין וְאוֹרוֹן שָׁאוּל.

מִמַּשְׁבֵּר לְנִצָּחוֹן

  • אַחְדוּת: מִתּוֹךְ הַפֵּרוּד וְהַכְּאֵב, צָמְחָה אַחְדוּת מֻפְלָאָה וֶאֱמוּנָה גְּדוֹלָה שֶׁהוֹבִילוּ לְנִסִּים עֲצוּמִים.
  • הֲסָרַת הָאִיּוּם הַקִּיּוּמִי: אַרְבַּעַת "רָאשֵׁי הַנָּחָשׁ" – אִירָאן, סוּרְיָה, חִזְבַּאלְלָה וְחַמַאס – שֶׁאִיְּמוּ לְהַשְׁמִידֵנוּ, כְּבָר לֹא מְהַוִּים אִיּוּם קִיּוּמִי. בִּטְחוֹנָהּ שֶׁל מְדִינַת יִשְׂרָאֵל טוֹב יוֹתֵר מִכָּל זְמַן אַחֵר.
  • הַתְּחִיָּה הָרוּחָנִית: גִּלִּינוּ מֵחָדָשׁ אֶת נִשְׁמַת הָעָם. חַיָּלִים מְבַקְּשִׁים צִיצִית וּתְפִלִּין, וְנֹעַר וּמְבֻגָּרִים מִתְקָרְבִים לַמָּסֹרֶת. לִמּוּד הַתּוֹרָה בִּשְׂדֵה הַקְּרָב הִגִּיעַ לְשִׂיא חָדָשׁ.
  • הַצְלָחָה כַּלְכָּלִית: בְּנִגּוּד לְכָל הַתַּחֲזִיּוֹת, וּבְנִגּוּד לְכָל מִלְחָמָה אַחֶרֶת בַּהִיסְטוֹרְיָה, כַּלְכָּלַת יִשְׂרָאֵל חֲזָקָה מִתָּמִיד.
  • דּוֹר הַנִּצָּחוֹן: חָשַׁבְנוּ שֶׁיֵּשׁ לָנוּ "דּוֹר שֶׁל מָסַכִּים", וְגִלִּינוּ דּוֹר שֶׁל אֲרָיוֹת. דּוֹר שֶׁל שְׁלִיחוּת. דּוֹר הַמַּכַּבִּים.

עָבַרְנוּ שְׁנָתַיִם שֶׁל בֶּכִי וּקְשָׁיִים, אַךְ בְּסוֹפוֹ שֶׁל דָּבָר, יַד ה' סִיְּעָה בַּעֲדֵנוּ, וְיָצָאנוּ זְקוּפִים, מַאֲמִינִים וַחֲזָקִים. אָנוּ מַמְשִׁיכִים לִצְעֹד לְקוֹמָה נוֹסֶפֶת בְּתַהֲלִיךְ הַגְּאֻלָּה שֶׁל עַם יִשְׂרָאֵל.

הַרְחָבָה: מַבָּט אֱמוּנִי עַל נִסֵּי מִלְחֶמֶת הַתְּקוּמָה

מֵחֲנֻכָּה לְמִלְחֶמֶת הַתְּקוּמָה

מִלְחֶמֶת הַחַשְׁמוֹנָאִים לֹא הָיְתָה רַק מַאֲבָק עַל חֹפֶשׁ דָּתִי, אֶלָּא מַעֲבָר מִ"מָּוֶת פָּסִיבִי" עַל קִדּוּשׁ ה' לְמִלְחָמָה אַקְטִיבִית עַל קִדּוּשׁ ה', עַל קִיּוּם הַתּוֹרָה וְעַל עַצְמָאוּת מְדִינִית. יְהוּדָה הַמַּכַּבִּי וְלוֹחֲמָיו נִלְחֲמוּ כִּמְעַטִּים מוּל רַבִּים, כְּשֶׁהֵם מְשַׁלְּבִים אֱמוּנָה גְּדוֹלָה בְּצַד תַּכְסִיסֵי מִלְחָמָה מִקְצוֹעִיִּים. בַּקְּרָב הָרִאשׁוֹן בְּמַעֲלֵה לְבוֹנָה, חִזֵּק יְהוּדָה אֶת לוֹחֲמָיו בְּאָמְרוֹ: "הֲיַד ה' תִּקְצַר לָתֵת רַבִּים בְּיַד מְעַטִּים". הַמַּטָּרָה הָיְתָה לֹא רַק לְטַהֵר אֶת הַמִּקְדָּשׁ, אֶלָּא לְהַשִּׂיג רִבּוֹנוּת. הָרַמְבַּ"ם (הִלְכוֹת חֲנֻכָּה פ"ג ה"א) מְסַכֵּם אֶת נֵס חֲנֻכָּה לֹא רַק בְּנֵס פַּךְ הַשֶּׁמֶן, אֶלָּא גַּם בָּעֻבְדָּה שֶׁ"חָזְרָה מַלְכוּת לְיִשְׂרָאֵל יֶתֶר עַל מָאתַיִם שָׁנָה". גַּם בְּמִלְחַמְתֵּנוּ, זָכִינוּ לְהַמְשִׁיךְ אֶת דֶּרֶךְ הַמַּכַּבִּים: לְהִלָּחֵם לֹא רַק עַל הֲגָנָה, אֶלָּא עַל קִיּוּם מְדִינָתֵנוּ, עַל שִׁלְטוֹנֵנוּ וְעַל מַלְכוּתֵנוּ הָרִבּוֹנִית.

פּוּרִים וּמִלְחֶמֶת הַתְּקוּמָה

הַהַקְבָּלָה בֵּין פָּרַס הָעַתִּיקָה (הָמָן) לְפָרַס הַמּוֹדֶרְנִית (אִירָאן) הִיא מַדְהִימָה. אֶלָּא שֶׁכַּיּוֹם, בְּנֵס הִיסְטוֹרִי, אָנוּ זוֹכִים לְהָגֵן עַל עַצְמֵנוּ. עִיּוּן מַעֲמִיק בִּמְגִלַּת אֶסְתֵּר מְגַלֶּה שֶׁהַפָּסוּק "לַיְּהוּדִים הָיְתָה אוֹרָה וְשִׂמְחָה וְשָׂשׂן וִיקָר" (אֶסְתֵּר ח', טז) נֶאֱמַר עוֹד לִפְנֵי הַנִּצָּחוֹן בַּקְּרָבוֹת, מִיָּד כַּאֲשֶׁר נִתְּנָה לָהֶם הָרְשׁוּת לְהִקָּהֵל וְלַעֲמֹד עַל נַפְשָׁם. עֶצֶם הַיְּכֹלֶת לְהָגֵן עַל עַצְמֵנוּ הִיא יְכֹלֶת שֶׁל גְּאֻלָּה, וְהִיא עַצְמָהּ סִבָּה לְשִׂמְחָה גְּדוֹלָה. כְּמוֹ בִּימֵי פּוּרִים, מִי שֶׁחָפֵץ הָיָה לִפְגֹּעַ בָּנוּ – נִפְגַּע קָשׁוֹת.

"אִם מִזֶּרַע הַיְּהוּדִים מָרְדֳּכַי אֲשֶׁר הַחִלּוֹתָ לִנְפֹּל לְפָנָיו לֹא תוּכַל לוֹ כִּי נָפוֹל תִּפּוֹל לְפָנָיו" (אֶסְתֵּר ו', יג).

הֶבְדֵּלִים בֵּין פּוּרִים וּבֵין מִלְחֶמֶת הַתְּקוּמָה

הַגְּמָרָא (מְגִלָּה יד.) מְבָרֶרֶת מַדּוּעַ לֹא אוֹמְרִים הַלֵּל בְּפוּרִים וּמְבִיאָה שָׁלוֹשׁ תְּשׁוּבֹות. כְּשֶׁבּוֹחֲנִים אוֹתָן, מְגַלִּים שֶׁהַנֵּס שֶׁלָּנוּ גָּדוֹל אַף יוֹתֵר מִנֵּס פּוּרִים, וְלָכֵן מְחַיֵּב הַלֵּל וְהוֹדָאָה גְּדוֹלִים יוֹתֵר.

  • "אֵין אוֹמְרִים שִׁירָה עַל נֵס שֶׁבְּחוּצָה לָאָרֶץ" ← בָּרוּךְ ה', הַנֵּס שֶׁלָּנוּ הוּא בְּאֶרֶץ יִשְׂרָאֵל!
  • "עֲדַיִן עַבְדֵי אֲחַשְׁוֵרוֹשׁ אֲנַן" ← בָּרוּךְ ה', יֵשׁ לָנוּ מְדִינָה רִבּוֹנִית מִשֶּׁלָּנוּ. אֵין אָנוּ עַבְדֵי אֲחַשְׁוֵרוֹשׁ!
  • "קְרִיאָתָהּ זוֹ הִלּוּלָהּ" ← קְרִיאַת הַמְּגִלָּה בָּאָה כְּתַחְלִיף לַהַלֵּל, כֵּיוָן שֶׁנֵּס פּוּרִים הוּא נֵס נִסְתָּר שֶׁדּוֹרֵשׁ הִתְבּוֹנְנוּת בַּ"פַּאזֶל". גַּם בַּמִּלְחָמָה שֶׁלָּנוּ נִדְרֶשֶׁת הִתְבּוֹנְנוּת, אַךְ הַפַּאזֶל גָּדוֹל וְעָצוּם הַרְבֵּה יוֹתֵר, וּמְגַלֶּה אֶת יַד ה' בְּאֹפֶן חָזָק יוֹתֵר.

פִּרְקֵי "מְגִלַּת הַתְּקוּמָה"

הַנִּסִּים הַגְּדוֹלִים שֶׁל מִלְחַמְתֵּנוּ נִכְתָּבִים בַּשָּׁמַיִם. כְּשֶׁאָנוּ מְחַבְּרִים אֶת כָּל חֶלְקֵי הַפַּאזֶל לְסִפּוּר אֶחָד, אָנוּ יְכוֹלִים לְדַמְיֵן אֶת פִּרְקֵי אוֹתָהּ מְגִלָּה שְׁמֵימִית, הַנִּפְרֶשֶׂת לְעֵינֵינוּ.

פֶּרֶק א' – הַהֲכָנוֹת הָרְחוֹקוֹת

בְּאֶרֶץ רְחוֹקָה מִצִּיּוֹן, הִתְעוֹרֵר סִכְסוּךְ – חוּט בְּשָׁטִיחַ שֶׁעוֹד לֹא נִרְאָה. לִפְנֵי כְּשָׁלוֹשׁ שָׁנִים, פָּתְחָה רוּסְיָה בְּמִלְחָמָה עִם אוּקְרָאִינָה. הָעִתּוּי לֹא הָיָה מִקְרִי; הוּא הִתְאַפְּשֵׁר בְּהֶעְדֵּר מַנְהִיג חָזָק בְּאַרְצוֹת הַבְּרִית. אוֹתָהּ מִלְחָמָה, שֶׁנִּרְאֲתָה לָנוּ זָרָה וְלֹא נוֹגַעַת, הָיְתָה לְמַעֲשֶׂה הַנָּחַת אֶבֶן הַפִּנָּה בְּבִנְיַן הַפֶּלֶא שֶׁיָּגֵן עַל יִשְׂרָאֵל.

פֶּרֶק ב' – הֶסְתֵּר פָּנִים בְּאֶרֶץ הַקֹּדֶשׁ

בְּעוֹד הָעוֹלָם הָיָה טָרוּד בְּמַאֲבָקָיו, בְּתוֹךְ אֶרֶץ הַקֹּדֶשׁ פָּשָׂה פֵּרוּד נוֹרָא, וְאָח קָם עַל אָחִיו. הַקִּטּוּב הִגִּיעַ לְשִׂיאוֹ בַּשֶּׁבֶר שֶׁל תְּפִלַּת יוֹם הַכִּפּוּרִים בְּתֵל אָבִיב, וְהַשְּׁכִינָה כְּאִלּוּ הִסְתַּלְּקָה. כְּשֶׁיֵּשׁ מְרִיבָה בֵּין הָאַחִים, נִכְנָס הַקָּדוֹשׁ בָּרוּךְ הוּא לְ"הֶסְתֵּר פָּנִים".

פֶּרֶק ג' – מֵחֹשֶׁךְ לְאוֹר גָּדוֹל

בְּשִׂמְחַת תּוֹרָה, נִצֵּל הָאוֹיֵב אֶת אוֹתוֹ "רֶגַע קָטֹן" שֶׁל הֶסְתֵּר פָּנִים, וּפוֹגְרוֹם נוֹרָא פָּקַד אוֹתָנוּ. אַךְ דַּוְקָא בְּאוֹתוֹ יוֹם, מִתּוֹךְ הַשֶּׁבֶר, צָמַח הַנֵּס הַגָּדוֹל בְּיוֹתֵר: עַם יִשְׂרָאֵל הִתְאַחֵד! הָאַחְדוּת הַזּוֹ הֵשִׁיבָה אֶת פְּנֵי ה', כְּמוֹ שֶׁנֶּאֱמַר: "בְּרֶגַע קָטֹן עֲזַבְתִּיךְ וּבְרַחֲמִים גְּדֹלִים אֲקַבְּצֵךְ" (יְשַׁעְיָהוּ נ"ד, ז). וּמִיָּד עִם שִׁיבַת הָאַחְדוּת, הֵחֵלּוּ הַנִּסִּים לְהִתְגַּלּוֹת: הַיְּכֹלֶת לְהָשִׁיב מִלְחָמָה בְּאוֹתוֹ הַיּוֹם, הַהִתְגַּיְּסוּת הַמֻּפְלָאָה, וְהַשְּׁתִיקָה הָרוֹעֶמֶת מִצַּד צְבָא הַטֶּרוֹר בַּצָּפוֹן.

פֶּרֶק ד' – נִטְרוּל רֹאשׁ הַנָּחָשׁ הָרִאשׁוֹן

הָאִיּוּם הַקִּיּוּמִי הַגָּדוֹל בְּיוֹתֵר, חִזְבַּאלְלָה, נֻטְרַל. "נֵס הַבִּיפֶּרִים", בְּרַק שְׁמֵימִי שֶׁל גְּאוֹנוּת טֶכְנוֹלוֹגִית וְהַשְׁגָּחָה עֶלְיוֹנָה, הֵבִיא לִפְגִיעָה בְּאַלְפֵי מְפַקְּדִים, לְשִׁתּוּק מַעֲרֶכֶת הַתִּקְשֹׁרֶת, וּלְחִסּוּל צַמֶּרֶת הָאִרְגּוּן. אוֹיֵב שֶׁאִיֵּם עַל הַשְׁמָדָתֵנוּ, כְּבָר לֹא הִוָּה אִיּוּם קִיּוּמִי.

פֶּרֶק ה' – קְרִיסַת הַדּוֹמִינוֹ

בְּלַיְלָה אֶחָד, מִיָּד לְאַחַר הַמַּכָּה לְחִזְבַּאלְלָה, נָפַל מִשְׁטַר אַסַאד בְּסוּרְיָה. הַקֶּשֶׁר הַסִּבָּתִי הָיָה בָּרוּר: הַמּוֹרְדִים הִצְלִיחוּ רַק מִשּׁוּם שֶׁתּוֹמְכוֹ הָעִקָּרִי שֶׁל אַסַאד, חִזְבַּאלְלָה, חֻסַּל. צַהַ"ל נִצֵּל אֶת הַהִזְדַּמְּנוּת, וּבְ"מִבְצָע חֵץ הַבָּשָׁן" הִשְׁמִיד אֶת הַצָּבָא הַסּוּרִי תּוֹךְ יוֹמַיִם, לְלֹא נִפְגָּעִים.

פֶּרֶק ו' – הַהִתְמוֹדְדוּת עִם לֵב הָרִשְׁעָה

וְאָז יָצְאָה יִשְׂרָאֵל, לְבַדָּהּ, לְמִלְחָמָה נֶגֶד אִימְפֶּרְיַת הָרֶשַׁע הָאִירָאנִית. הַנִּצָּחוֹן, תּוֹךְ 12 יָמִים וּבְאֶפֶס אֲבֵדוֹת, הִתְאַפְּשֵׁר בִּזְכוּת כָּל חֶלְקֵי הַפַּאזֶל שֶׁהֻנְּחוּ בִּמְקוֹמָם בְּדִיּוּק מֻפְלָא: רוּסְיָה עֲסוּקָה, חִזְבַּאלְלָה חֻסַּל, חַמַאס פָּצוּעַ, וְסוּרְיָה – בְּ"וְנַהֲפוֹךְ הוּא" – הָפְכָה מֵאוֹיֵב לִתְחֲנַת מַעֲבָר הַמְּסַיַּעַת לְיִשְׂרָאֵל.

פֶּרֶק ז' – חֲתִימַת הַפַּאזֶל וְשִׁיבַת בָּנִים

הַפַּאזֶל הֻשְׁלַם עִם נִסִּים נוֹסָפִים: הַחוּתִ'ים שֶׁ"אִמְּנוּ" אֶת הַטַּיָּסִים, טְרַאמְפּ שֶׁחָזַר לַשִּׁלְטוֹן, הַכַּלְכָּלָה שֶׁפָּרְחָה, וְהַהֲגָנָה הַשְּׁמֵימִית מִפְּנֵי טִילִים. וְאָז, כְּשֶׁהָעוֹלָם כֻּלּוֹ רָאָה אֶת עָצְמָתָהּ שֶׁל יִשְׂרָאֵל, הִגִּיעַ הַנֵּס הַבִּלְתִּי יֵאָמֵן: "וְשָׁבוּ בָנִים לִגְבוּלָם" – כָּל הַחֲטוּפִים שָׁבוּ הַבַּיְתָה, וּמִלְחֶמֶת הַתְּקוּמָה נֶחְתְּמָה בְּנִצָּחוֹן שֶׁלֹּא נִרְאָה כְּמוֹתוֹ אַלְפֵי שָׁנִים.

מִבְחַן חִדּוּד

1. מַהוּ הַקֶּשֶׁר הַסִּבָּתִי הַמֶּרְכָּזִי בֵּין "נֵס הַבִּיפֶּרִים" לְהַצְלָחַת "מִבְצָע חֵץ הַבָּשָׁן" נֶגֶד הַצָּבָא הַסּוּרִי?

תְּשׁוּבָה נְכוֹנָה! נִמּוּק: הַתְּשׁוּבָה הַנְּכוֹנָה הִיא ב'. חִזְבַּאלְלָה הָיָה הַתּוֹמֵךְ הַצְּבָאִי הָעִקָּרִי שֶׁל מִשְׁטַר אַסַאד. רַק לְאַחַר שֶׁחִזְבַּאלְלָה קָרַס בְּעִקְבוֹת נֵס הַבִּיפֶּרִים, נוֹצַר "חַלּוֹן הִזְדַּמְּנוּיוֹת" שֶׁאִפְשֵׁר לַמּוֹרְדִים לְהַפִּיל אֶת הַמִּשְׁטָר. נְפִילַת הַמִּשְׁטָר הִיא שֶׁיָּצְרָה אֶת הַהִזְדַּמְּנוּת לְצַהַ"ל לִתְקֹף וּלְהַשְׁמִיד אֶת הַצָּבָא הַסּוּרִי בְּ"חֵץ הַבָּשָׁן".

2. כֵּיצַד תָּרְמָה מִלְחֶמֶת רוּסְיָה-אוּקְרָאִינָה, שֶׁנִּרְאֲתָה לֹא קְשׁוּרָה, לְנִצְחוֹנָהּ שֶׁל יִשְׂרָאֵל עַל אִירָאן?

מְצֻיָּן! נִמּוּק: הַתְּשׁוּבָה הַנְּכוֹנָה הִיא ג'. אַחַת מֵהַהַנָּחוֹת הָאַסְטְרָטֵגִיּוֹת שֶׁל סוּרְיָה וְאִירָאן הָיְתָה שֶׁבְּעֵת מִלְחָמָה, רוּסְיָה תַּעֲמֹד לְצִדָּן. הָעֻבְדָּה שֶׁרוּסְיָה הָיְתָה שְׁקוּעָה עַד צַוָּאר בַּמִּלְחָמָה הַמַּתִּישָׁה עִם אוּקְרָאִינָה נִטְרְלָה אוֹתָהּ כְּגוֹרֵם שֶׁיָּכוֹל לְהַפְרִיעַ לְיִשְׂרָאֵל אוֹ לְסַיֵּעַ לְאוֹיְבֶיהָ.

3. עַל פִּי הַנִּתּוּחַ, מַה הָיָה הַתַּפְקִיד הַבִּלְתִּי צָפוּי שֶׁל הַחוּתִ'ים בַּ"פַּאזֶל הָאֱ-לֹהִי"?

נָכוֹן מְאוֹד! נִמּוּק: הַתְּשׁוּבָה הַנְּכוֹנָה הִיא ב'. כְּדֵי לְהִתְכּוֹנֵן לִתְקִיפָה מֻרְכֶּבֶת בְּמֶרְחָק שֶׁל 2,000 קִילוֹמֶטְרִים, הָיָה צֹרֶךְ בְּאִמּוּן "עַל רָטֹב", כְּלוֹמַר בִּתְנָאֵי אֱמֶת. הַתְּקִיפוֹת הַמַּתְמִידוֹת נֶגֶד הַחוּתִ'ים בְּתֵימָן, שֶׁנִּמְצָאִים בְּמֶרְחָק דּוֹמֶה, שִׁמְּשׁוּ כְּאִמּוּן מֻשְׁלָם וּבִלְתִּי מְתֻכְנָן לַטַּיָּסִים לִקְרַאת הַמִּבְצָע הָעִקָּרִי בְּאִירָאן.

4. הַמַּאֲמָר טוֹעֵן שֶׁהַנֵּס שֶׁל מִלְחֶמֶת הַתְּקוּמָה גָּדוֹל מִנֵּס פּוּרִים. מַדּוּעַ, עַל פִּי הַהֶסְבֵּר שֶׁנִּתַּן?

תְּשׁוּבָה מְדֻיֶּקֶת! נִמּוּק: הַתְּשׁוּבָה הַנְּכוֹנָה הִיא ג'. הַמַּאֲמָר מַסְבִּיר שֶׁשְּׁתַּיִם מֵהַסִּבּוֹת הַמֶּרְכָּזִיּוֹת שֶׁבִּגְלָלָן לֹא אוֹמְרִים הַלֵּל בְּפוּרִים הֵן שֶׁהַנֵּס אֵרַע בְּחוּ"ל ("נֵס שֶׁבְּחוּצָה לָאָרֶץ") וְשֶׁהַיְּהוּדִים נִשְׁאֲרוּ מְשֻׁעְבָּדִים לַמַּלְכוּת ("עַבְדֵי אֲחַשְׁוֵרוֹשׁ"). בְּמִלְחֶמֶת הַתְּקוּמָה, הַנֵּס הָיָה בְּאֶרֶץ יִשְׂרָאֵל וְתַחַת שִׁלְטוֹן יִשְׂרְאֵלִי עַצְמָאִי, וְלָכֵן מַדְרֵגָתוֹ גְּבוֹהָה יוֹתֵר.

5. מַהִי נְקֻדַּת הַמִּפְנֶה הָרוּחָנִית שֶׁעַל פִּי הַטֶּקְסְט, הֵחֵלָּה אֶת סִדְרַת הַנִּסִּים בַּמִּלְחָמָה?

בּוֹל בַּמַּטָּרָה! נִמּוּק: הַתְּשׁוּבָה הַנְּכוֹנָה הִיא ג'. הַמַּאֲמָר מַדְגִּישׁ שׁוּב וָשׁוּב שֶׁהַפֵּרוּד שֶׁקָּדַם לַמִּלְחָמָה הָיָה סִבָּה לְ"הֶסְתֵּר פָּנִים". "נֵס הָאַחְדוּת" – הַהִתְגַּיְּסוּת הַמְּיָדִית וְהַתְּחוּשָׁה שֶׁכֻּלָּנוּ עַם אֶחָד – הָיָה הַמַּהְפָּךְ הָרוּחָנִי שֶׁהֵשִׁיב אֶת הַסִּיַּעְתָּא דִּשְׁמַיָּא וְ"פָּתַח אֶת הַדֶּלֶת" לְכָל שְׁאַר הַנִּסִּים שֶׁבָּאוּ אַחֲרָיו.

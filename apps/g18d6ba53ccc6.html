בַּיִת 1
שִׁבְעִים נֶפֶשׁ יָרְדוּ לְמִצְרָיִם
יַעֲקֹב וּבָנָיו בְּאַהֲבָה תְּמִימָה
הַשֵּׁם סוֹפֵר אוֹתָנוּ בְּכָל שָׁעָה
כְּמוֹ אֲבָנִים טוֹבוֹת וּמַרְגָּלִיּוֹת יְקָרָה
טְרוּם פִּזְמוֹן
נֶפֶשׁ אַחַת הָיִינוּ, אַחְדוּת וְשִׂמְחָה
עוֹבְדִים אֶת הַשֵּׁם בְּלֵב וּבְנִשְׁמָה
וְהַשֵּׁם הִבְטִיחַ לְיַעֲקֹב לִפְנֵי הַיְּרִידָה
לְגוֹי גָּדוֹל אֲשִׂימְךָ שָׁם - זוֹ הַהַבְטָחָה
פִּזְמוֹן ⭐
פָּרוּ וַיִּשְׁרְצוּ וַיִּרְבּוּ וַיַּעַצְמוּ בִּמְאֹד מְאֹד
שֵׁשׁ מִלִּים שֶׁל נֵס גָּדוֹל, כֹּחַ מֵהַשֵּׁם הַגָּדוֹל
וְכַאֲשֶׁר יְעַנּוּ אוֹתוֹ כֵּן יִרְבֶּה וְכֵן יִפְרֹץ
הָאֱמוּנָה נוֹתֶנֶת כֹּחַ, הָעָם לֹא נִשְׁבָּר וְלֹא נֶעֱצָר בַּמָּצוֹר
בַּיִת 2
וַיָּקָם מֶלֶךְ חָדָשׁ עַל מִצְרַיִם
אֲשֶׁר לֹא יָדַע אֶת יוֹסֵף הַצַּדִּיק וְהַתָּמִים
כְּפוּי טוֹבָה הָיָה, רָשָׁע וְאַכְזָר
הָבָה נִתְחַכְּמָה לוֹ - גָּזַר בְּחָכְמָה רַע וּמָר
טְרוּם פִּזְמוֹן
עֲבוֹדַת פֶּרֶךְ וְלַחַץ, לִבְנוֹת עָרִים וּמַחְסָנִים
טִיט וּלְבֵנִים, מַשָּׂאוֹת כְּבֵדִים רַבִּים
אֲבָל נָשֵׁינוּ הַגִּבּוֹרוֹת לֹא נִכְנָעוּ
אָמְרוּ: סוֹף הַקָּדוֹשׁ בָּרוּךְ הוּא גוֹאֵל אוֹתָנוּ
פִּזְמוֹן ⭐
פָּרוּ וַיִּשְׁרְצוּ וַיִּרְבּוּ וַיַּעַצְמוּ בִּמְאֹד מְאֹד
שֵׁשׁ מִלִּים שֶׁל נֵס גָּדוֹל, כֹּחַ מֵהַשֵּׁם הַגָּדוֹל
וְכַאֲשֶׁר יְעַנּוּ אוֹתוֹ כֵּן יִרְבֶּה וְכֵן יִפְרֹץ
הָאֱמוּנָה נוֹתֶנֶת כֹּחַ, הָעָם לֹא נִשְׁבָּר וְלֹא נֶעֱצָר בַּמָּצוֹר

[הַפְסָקָה אִינְסְטְרוּמֶנְטָלִית - סוֹלוֹ קְלַרְנִית שִׂמְחָתִי]

גֶּשֶׁר - אֶנֶרְגֶטִי יוֹתֵר
שִׁפְרָה וּפוּעָה, יוֹכֶבֶד וּמִרְיָם
יָרְאוּ אֶת הָאֱלֹהִים, לֹא שָׁמְעוּ לְפַרְעֹה הָרָשָׁע תָּמִיד
וַתְּחַיֶּיןָ אֶת הַיְלָדִים, סִפְּקוּ מַיִם וּמָזוֹן
יִרְאַת שָׁמַיִם נָתְנָה כֹּחַ לְמַעַן הַחֶסֶד וְהָרָצוֹן
דֶּקְלוּם - מְדֻבָּר בְּעוֹצְמָה
הַשֵּׁם הִבְטִיחַ לְיַעֲקֹב: אָנֹכִי אֶעֱלְךָ גַם עָלֹה
וַיַּעַשׂ לָהֶם בָּתִּים - בָּתֵּי כְהֻנָּה וְלֵוִיָּה וּמַלְכוּת
לֹא לְעוֹלָם מִשְׁתַּעְבְּדִין בָּנוּ, הָאֱמוּנָה שֶׁלָּנוּ חֲזָקָה

[מוֹדוּלַצְיָה - מַעֲבָר לְסוּלָם גָּבוֹהַּ יוֹתֵר]

פִּזְמוֹן - גָּבוֹהַּ וְעָצְמָתִי יוֹתֵר ⭐⭐
פָּרוּ וַיִּשְׁרְצוּ וַיִּרְבּוּ וַיַּעַצְמוּ בִּמְאֹד מְאֹד
שֵׁשׁ מִלִּים שֶׁל נֵס גָּדוֹל, כֹּחַ מֵהַשֵּׁם הַגָּדוֹל
וְכַאֲשֶׁר יְעַנּוּ אוֹתוֹ כֵּן יִרְבֶּה וְכֵן יִפְרֹץ
הָאֱמוּנָה נוֹתֶנֶת כֹּחַ, הָעָם לֹא נִשְׁבָּר וְלֹא נֶעֱצָר בַּמָּצוֹר
בַּיִת 3 - מְנַצֵּחַ
כָּל כֹּחַ הַמִּצְרִים לֹא עָזַר כְּלָל
הָעָם הִתְחַזֵּק וְהִתְעַצֵּם בְּכָל
כִּי בְּכָל מָה שֶׁהֵם נוֹתְנִים לֵב לְעַנּוֹת
כֵּן לֵב הַקָּדוֹשׁ בָּרוּךְ הוּא לְהַרְבּוֹת וּלְהַפְרִיץ אֶת הַמִּנְיָן
פִּזְמוֹן סִיּוּם עִם הַרְמוֹנִיּוֹת ⭐⭐⭐
פָּרוּ וַיִּשְׁרְצוּ וַיִּרְבּוּ וַיַּעַצְמוּ בִּמְאֹד מְאֹד
שֵׁשׁ מִלִּים שֶׁל נֵס גָּדוֹל, כֹּחַ מֵהַשֵּׁם הַגָּדוֹל
וְכַאֲשֶׁר יְעַנּוּ אוֹתוֹ כֵּן יִרְבֶּה וְכֵן יִפְרֹץ

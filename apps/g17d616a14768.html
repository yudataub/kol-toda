הַפּרוֹפֵסוֹר שֶׁשִּׁחְרֵר אֶת חֶבְרוֹן: סִפּוּרוֹ שֶׁל בֶּן צִיּוֹן טַבְגֶּר

הַמַּדְעָן שֶׁעָזַב אֶת הָאוּנִיבֶרְסִיטָה כְּדֵי לְהִלָּחֵם עַל נִשְׁמָתָהּ שֶׁל עִיר הָאָבוֹת

עִבּוּד עַל פִּי רֵאָיוֹן עִם יַעֲקֹב טַבְגֶּר, בְּנוֹ שֶׁל הַפּרוֹפֵסוֹר

תֹּכֶן הָעִנְיָנִים

דַּמְיְנוּ אֶת הַמַּעֲבָר הֶחָרִיף: פּרוֹפֵסוֹר לְפִיזִיקָה, אִישׁ שֶׁשְּׁמוֹ הוֹלֵךְ לְפָנָיו בְּמַסְדְּרוֹנוֹת הָאָקָדֶמְיָה הַסּוֹבְיֶטִית הַקְּשׁוּחָה, מֵסִיר אֶת חֲלִיפָתוֹ הַמְּהֻדֶּרֶת, עוֹזֵב אֶת הַכָּתֵדְרָה הַנִּכְבָּדָה בְּאוּנִיבֶרְסִיטַת תֵּל אָבִיב, וְיוֹרֵד, בְּמוֹ יָדָיו, לַחְפֹּר בְּתוֹךְ דִּיר עִזִּים וּמַזְבֵּלָה שֶׁהָיוּ פַּעַם בֵּית כְּנֶסֶת מְפֹאָר. זֶהוּ, עַל קְצֵה הַמַּזְלֵג, סִפּוּרוֹ שֶׁל פּרוֹפֵסוֹר בֶּן צִיּוֹן טַבְגֶּר, דְּמוּת שֶׁשִּׁלְּבָה בְּאֹפֶן נָדִיר אִינְטֶלֶקְט מַבְרִיק עִם עֲשִׂיָּה חֲלוּצִית, שֶׁיָּדָיו הָיוּ מְלֻכְלָכוֹת בְּעַפְרָהּ הַמְּחֻלָּל שֶׁל חֶבְרוֹן. בְּעוֹד שֶׁרַבִּים מַכִּירִים אֶת הַמַּאֲבָקִים הַגְּדוֹלִים עַל עִיר הָאָבוֹת, פָּעֳלוֹ שֶׁל טַבְגֶּר, הָאִישׁ שֶׁגָּאַל בִּנְחִישׁוּת שְׁקֵטָה אֶת בֵּית כְּנֶסֶת "אַבְרָהָם אָבִינוּ" וְאֶת בֵּית הָעַלְמִין הֶעָתִיק, לְעִתִּים נִשְׁכַּח. הוּא הָיָה אִישׁ שֶׁהִקְדִּים אֶת זְמַנּוֹ, שֶׁהֵבִין כִּי הַמַּאֲבָק עַל אֶרֶץ יִשְׂרָאֵל אֵינוֹ רַק עִנְיָן שֶׁל פּוֹלִיטִיקָה גְּבוֹהָה, אֶלָּא שֶׁל מַעֲשִׂים פְּשׁוּטִים, עִקְבִיִּים וַאֲמִיתִּיִּים בַּשֶּׁטַח.

1. מִסִּיבִּיר לְתֵל אָבִיב - הַמַּסָּע שֶׁעִצֵּב אֶת הַלּוֹחֵם

כְּדֵי לְהָבִין אֶת הַנְּחִישׁוּת הַבִּלְתִּי מִתְפַּשֶּׁרֶת שֶׁל פּרוֹפֵסוֹר טַבְגֶּר בְּסִמְטָאוֹת חֶבְרוֹן, חַיָּבִים לַחֲזֹר לַשָּׁנִים שֶׁבָּהֶן חֻשַּׁל בְּכוּר הַהִתּוּךְ שֶׁל בְּרִית הַמּוֹעֲצוֹת. הַהִתְמוֹדְדוּת עִם הַ-ק.ג.ב., הַהִתְעַקְּשׁוּת הַמַּדָּעִית מוּל מִמְסָד סְקֶפְּטִי, וְהַיְּכֹלֶת לַחְשֹׁב מִחוּץ לַקֻּפְסָה בִּשְׁעַת מְצוּקָה – כָּל אֵלֶּה הָיוּ הַהַכְשָׁרָה הַמֻּשְׁלֶמֶת לַמַּאֲבָקִים שֶׁצָּפוּ לוֹ בְּעִיר הָאָבוֹת.

  • הַצְטַיְּנוּת מְדֻכֵּאת: כְּבָר מִגִּיל צָעִיר הִתְבַּלֵּט טַבְגֶּר בְּכִשְׁרוֹנוֹתָיו. הוּא הִתְקַבֵּל לַפָקוּלְטָה לְפִיזִיקָה וְהִצְטַיֵּן בְּלִמּוּדָיו. אַךְ בַּתְּקוּפָה הַסְּטָלִינִיסְטִית, הַמִּשְׁטָר לֹא רָאָה בְּעַיִן יָפָה אֶת הַצְלָחָתוֹ שֶׁל יְהוּדִי וְנִסָּה לַעֲצֹר אֶת הִתְקַדְּמוּתוֹ. טַבְגֶּר, בְּחָכְמָה וּבִתְעוּזָה, פָּנָה לְסוֹפֵר יְהוּדִי שֶׁהָיָה חָבֵר אִישִׁי שֶׁל סְטָאלִין, וּבְעֶזְרָתוֹ הֻחְזַר לָאָקָדֶמְיָה וְהִמְשִׁיךְ בְּדַרְכּוֹ הַמַּדָּעִית.
  • תַּגְלִית מַהְפְּכָנִית: פָּעֳלוֹ הַמַּדָּעִי הִגִּיעַ לְשִׂיא עִם גִּלּוּי הַ"סִימֶטְרִיָּה הַמַּגְנֵטִית". זוֹ לֹא הָיְתָה עוֹד תַּגְלִית, אֶלָּא "כְּלַל-עַל" בְּפִיזִיקָה, עִקָּרוֹן יְסוֹדִי שֶׁשִּׁנָּה אֶת הַהֲבָנָה הַקַּיֶּמֶת. כְּשֶׁמַּדְּעָנִים אֲחֵרִים טָעֲנוּ שֶׁהַדָּבָר "לֹא יִתָּכֵן" וְנוֹגֵד אֶת חֻקֵּי הַפִיזִיקָה הַיְּדוּעִים, טַבְגֶּר לֹא וִתֵּר. הוּא הִתְעַקֵּשׁ, נִפְגַּשׁ עִם גְּדוֹלֵי הַמַּדָּע בְּרוּסְיָה וְשִׁכְנֵעַ אוֹתָם בְּצִדְקָתוֹ, עַד שֶׁתַּגְלִיתוֹ נִכְלְלָה בְּסִפְרֵי הַיְּסוֹד שֶׁל הַתְּחוּם.
  • הַבְּרִיחָה מֵהַ-ק.ג.ב.: בְּמַקְבִּיל לַעֲבוֹדָתוֹ הַמַּדָּעִית, הֵקִים טַבְגֶּר קְבוּצָה צִיּוֹנִית שֶׁלָּמְדָה יַהֲדוּת וְצִיּוֹנוּת. כַּאֲשֶׁר הַ-ק.ג.ב. עָלָה עַל הַקְּבוּצָה וְעָצַר אֶת חֲבֵרָיו, טַבְגֶּר הִצְלִיחַ לְהִתְחַמֵּק בְּדֶרֶךְ יְצִירָתִית וּמַדְהִימָה: הוּא אִשְׁפֵּז אֶת עַצְמוֹ בְּבֵית מְשֻׁגָּעִים כְּדֵי לְהִתְחַבֵּא מֵהָרָשֻׁיּוֹת, וּלְאַחַר מִכֵּן הִסְתַּתֵּר בְּיַעֲרוֹת הַוּוֹלְגָה. שׁוּב, חֲשִׁיבָה לֹא שִׁגְרָתִית הִצִּילָה אֶת חַיָּיו.
  • עֲלִיָּה וְקָרְיֶרָה בָּאָרֶץ: בִּשְׁנַת 1972, הִצְלִיחַ טַבְגֶּר לַעֲלוֹת אַרְצָה. שִׁמְעוֹ הַמַּדָּעִי הִקְדִּים אוֹתוֹ, וּפְרוֹפֵסוֹר יוּבַל נֶאֱמָן, נְשִׂיא אוּנִיבֶרְסִיטַת תֵּל אָבִיב דַּאֲזַי, הִזְמִין אוֹתוֹ לְהַרְצוֹת בָּאוּנִיבֶרְסִיטָה. אַךְ לְהַפְתָּעַת הַכֹּל, לְאַחַר שְׁנָתַיִם בִּלְבַד, הוּא הֶחְלִיט לַעֲזֹב אֶת הַמִּשְׂרָה הַנִּכְסֶפֶת וְלַעֲבֹר לְחֶבְרוֹן.

הַנִּסָּיוֹן שֶׁצָּבַר בְּרוּסְיָה לִמֵּד אוֹתוֹ שֶׁהַמִּמְסָד יָכוֹל לִהְיוֹת לֹא רַק מֻשְׁחָת, אֶלָּא גַּם אַבְּסוּרְדִּי. כְּשֶׁהִגִּיעַ לְחֶבְרוֹן, הוּא גִּלָּה לְתַדְהֵמָתוֹ מְצִיאוּת דּוֹמָה – רַק שֶׁהַפַּעַם, תַּחַת שִׁלְטוֹן יְהוּדִי.

2. חֶבְרוֹן, שְׁנוֹת הַ-70: מַרְאֶה שֶׁל חֻרְבָּן וְהֶפְקֵרוּת

כְּדֵי לְהָבִין אֶת גֹּדֶל פָּעֳלוֹ שֶׁל פּרוֹפֵסוֹר טַבְגֶּר, חַיָּבִים לְצַיֵּר אֶת תְּמוּנַת הַמַּצָּב הַקּוֹדֶרֶת שֶׁל חֶבְרוֹן בְּאֶמְצַע שְׁנוֹת הַשִּׁבְעִים. לֹא הָיָה זֶה מַאֲבָק רַק נֶגֶד אוֹיֵב עֲרָבִי, אֶלָּא מַאֲבָק בָּאֲדִישׁוּת, בְּהֶפְקֵרוּת וּבִמְדִינִיּוּת מְסֻלֶּפֶת שֶׁל הַמִּמְשָׁל הַצְּבָאִי הַיִּשְׂרְאֵלִי. זֶה מַה שֶּׁרָאָה טַבְגֶּר כְּשֶׁהִגִּיעַ לָעִיר:

  1. יִשּׁוּב יְהוּדִי אֵין: בְּתוֹךְ חֶבְרוֹן עַצְמָהּ לֹא הָיָה יִשּׁוּב יְהוּדִי כְּלָל. קִרְיַת אַרְבַּע הָיְתָה גִּבְעָה מְרֻחֶקֶת, מְנֻתֶּקֶת מֵהַמֶּרְכָּז הַהִיסְטוֹרִי שֶׁל עִיר הָאָבוֹת.
  2. מְעָרַת הַמַּכְפֵּלָה: יְהוּדִים לֹא יָכְלוּ לְהִתְפַּלֵּל בַּמְּעָרָה בְּאֹפֶן חָפְשִׁי. כְּנִיסָתָם הֻתְּרָה רַק בִּשְׁעוֹת תַּיָּרוּת מֻגְבָּלוֹת, כַּמָּה שָׁעוֹת בַּבֹּקֶר וְכַמָּה שָׁעוֹת בַּצָּהֳרַיִם, בְּמַצָּב הַמַּזְכִּיר אֶת הַמַּצָּב בְּהַר הַבַּיִת כַּיּוֹם.
  3. בֵּית הָעַלְמִין הֶעָתִיק: הַמָּקוֹם הָיָה מְחֻלָּל בְּאֹפֶן מְזַעֲזֵעַ. מַצֵּבוֹת נֶעֶקְרוּ, וְהַשֶּׁטַח הָפַךְ לְגַן יָרָק שֶׁעֻבַּד עַל יְדֵי הָעֲרָבִים. חֲלָקִים אֲחֵרִים שֶׁל בֵּית הָעַלְמִין שִׁמְּשׁוּ כְּמַזְבֵּלָה עִירוֹנִית.
  4. בֵּית הַכְּנֶסֶת עַ"שׁ אַבְרָהָם אָבִינוּ: כָּאן הִגִּיעַ הַבִּזָּיוֹn לְשִׂיא. הַמָּקוֹם הַקָּדוֹשׁ, שֶׁעָמַד בְּלֵב הָרֹבַע הַיְּהוּדִי, הָפַךְ לְשִׁלּוּב בִּלְתִּי נִתְפָּס שֶׁל שְׁלוֹשָׁה שִׁמּוּשִׁים.
    "הִצְלִיחוּ לְהָחִיל שָׁם שְׁלוֹשָׁה דְּבָרִים: דִּיר עִזִּים, מַזְבֵּלָה עִירוֹנִית, וּסְלִיחָה מִכְּבוֹדְכֶם, בֵּית הַשִּׁמּוּשׁ הַצִּבּוּרִי הַיָּחִיד בְּכָל חֶבְרוֹן."

אַךְ הַגִּלּוּי הַמַּזְעֲזֵעַ בְּיוֹתֵר, זֶה שֶׁדָּחַף אֶת טַבְגֶּר לִפְעֻלָּה, הָיָה שֶׁהַחֻרְבָּן לֹא רַק שֶׁלֹּא נֶעֱצַר תַּחַת שִׁלְטוֹן יִשְׂרְאֵלִי, אֶלָּא אַף קִבֵּל חוֹתֶמֶת רִשְׁמִית. הוּא גִּלָּה לְתַדְהֵמָתוֹ כִּי הַמִּמְשָׁל הַצְּבָאִי הֶחְכִּיר בְּאֹפֶן רִשְׁמִי, שָׁנָה אַחַר שָׁנָה, אֶת שֶׁטַח בֵּית הַכְּנֶסֶת לְבַעַל הָעִזִּים הָעֲרָבִי. הֶעָלְמַת הָעַיִן הָפְכָה לְחִלּוּל פָּעִיל בְּחָסוּת מְדִינַת יִשְׂרָאֵל.

מוּל הַמְּצִיאוּת הַבִּלְתִּי נִסְבֶּלֶת הַזּוֹ, טַבְגֶּר לֹא חִכָּה לְאִשּׁוּרִים – הוּא הֵחֵל לִפְעֹל, תְּחִלָּה בַּנְּקֻדָּה הָרְגִישָׁה בְּיוֹתֵר: מְעָרַת הַמַּכְפֵּלָה.

3. פְּעֻלּוֹת רִאשׁוֹנוֹת: הַשַּׁרְשֶׁרֶת בַּמְּעָרָה וְהַפַּאזֶל בְּבֵית הָעַלְמִין

הַטַּקְטִיקָה שֶׁל פּרוֹפֵסוֹר טַבְגֶּר בִּפְעֻלּוֹתָיו הָרִאשׁוֹנוֹת הָיְתָה מֻבְרֶקֶת. הוּא נִמְנַע מֵעִמּוּת חֲזִיתִי, וּבִמְקוֹמוֹ יָצַר עוּבְדּוֹת בַּשֶּׁטַח בְּאֶמְצָעוּת שִׁלּוּב שֶׁל חָכְמָה, נְחִישׁוּת וְהִתְנַהֲגוּת שֶׁהָיְתָה פָּסִיבִית וְאַקְטִיבִית בְּעֵת וּבְעוֹנָה אַחַת. הוּא לֹא הִתְוַכֵּחַ, הוּא פָּשׁוּט עָשָׂה, וְהֶעֱמִיד אֶת הָרָשֻׁיּוֹת מוּל דִּילֶמָּה.

הַמַּאֲבָק עַל הַתְּפִלָּה בַּמְּעָרָה: כְּדֵי לְעַרְעֵר עַל הַהַגְבָּלָה הַמַּשְׁפִּילָה עַל תְּפִלַּת יְהוּדִים, הִגִּיעַ טַבְגֶּר לִמְעָרַת הַמַּכְפֵּלָה עִם בְּנוֹ וְקָשַׁר אֶת עַצְמָם בְּשַׁרְשְׁרָאוֹת לַצִּיּוּן שֶׁל שָׂרָה אִמֵּנוּ. כַּאֲשֶׁר הַחַיָּלִים דָּרְשׁוּ מֵהֶם לָצֵאת, הֵם נִתְקְלוּ בְּסִדְרַת מַהֲלָכִים מְתֻחְכָּמִים:

  • הַפְּעֻלָּה הַשְּׁקֵטָה: טַבְגֶּר הִתְעַלֵּם מֵהַחַיָּלִים וְהִמְשִׁיךְ לִקְרֹא תְּהִלִּים, כְּאִלּוּ דָּבָר אֵינוֹ מַטְרִיד אוֹתוֹ.
  • הַתְּשׁוּבָה הַלּוֹגִית: רַק כְּשֶׁהִפְצִירוּ בּוֹ, עָנָה בְּהִגָּיוֹן קַר: "אֲנִי קָשׁוּר, אֲנִי פָּסִיבִי. אֲנִי לֹא עוֹשֶׂה כְּלוּם". הוּא הֶעֱבִיר אֶת נֵטֶל הַהוֹכָחָה וְהַפְּעֻלָּה אֲלֵיהֶם.
  • הַתַּחְבּוּלָה עִם הַמַּפְתֵּחַ: הַחַיָּלִים רָאוּ מַפְתֵּחַ מֻנָּח לְיָדוֹ, אַךְ כַּמּוּבָן שֶׁטַבְגֶּר דָּאַג שֶׁזֶּה יִהְיֶה מַפְתֵּחַ לֹא מַתְאִים, וּבְכָךְ קָנָה זְמַן יָקָר.
  • הַמֶּסֶר: הַפְּעֻלָּה כֻּלָּהּ שִׁדְּרָה מֶסֶר בָּרוּר, לְלֹא מִלִּים: "אֲנִי פֹּה כְּדֵי לְהִשָּׁאֵר. הַשָּׁעוֹת שֶׁקְּבַעְתֶּם אֵינָן חֻקִּיּוֹת". הַמַּאֲבָק הַזֶּה פָּתַח אֶת הַדֶּרֶךְ לִתְפִלַּת יְהוּדִים קְבוּעָה בַּמְּעָרָה.

תְּחִלַּת הָעֲבוֹדָה בְּבֵית הָעַלְמִין: הַהִזְדַּמְּנוּת לְשַׁקֵּם אֶת בֵּית הָעַלְמִין הִגִּיעָה כַּאֲשֶׁר טַבְגֶּר הִתְקַבֵּל לַעֲבוֹדָה כְּשׁוֹמֵר עַל קִבְרוֹ הַטָּרִי שֶׁל הַתִּינוֹק אַבְרָהָם יְדִידְיָה נַחְשׁוֹן ז"ל, שֶׁנִּקְבַּר שָׁם בִּמְסִירוּת נֶפֶשׁ. הַשְּׁמִירָה עַל קֶבֶר יָחִיד הָפְכָה מַהֵר מְאֹד לְמִבְצָע שִׁקּוּם שֶׁל בֵּית הָעַלְמִין כֻּלּוֹ.

  • סִלּוּק הַחַקְלָאוּת: הַשָּׁלָב הָרִאשׁוֹן הָיָה לְסַלֵּק אֶת הַחִלּוּל הַחַקְלָאִי. טַבְגֶּר עָקַר אֶת שְׂדֵה הָעַגְבָנִיּוֹת וְאֶת הַגְּפָנִים, וּלְהַפְתָּעָתוֹ, הַשּׁוֹמֵר הָעֲרָבִי שֶׁעָבַד בַּמָּקוֹם, חוסיין, שִׁתֵּף עִמּוֹ פְּעֻלָּה וְאַף עָזַר לוֹ.
  • נִקּוּי הַמַּזְבֵּלָה: כְּדֵי לְנַקּוֹת אֶת הַפֶּסֹלֶת שֶׁנִּזְרְקָה מֵהַבָּתִּים הַסְּמוּכִים, נָקַט טַבְגֶּר בְּשִׁיטָה יְצִירָתִית. הוּא פָּשׁוּט הֵחֵל לִזְרֹק אֶת הָאַשְׁפָּה בַּחֲזָרָה לְכִוּוּן חַצְרוֹת הַשְּׁכֵנִים. הַלָּלוּ, שֶׁלֹּא רָצוּ לִחְיוֹת בְּתוֹךְ הַזֶּבֶל, שָׁלְחוּ אֶת יַלְדֵיהֶם לְנַקּוֹת אֶת כָּל הַשֶּׁטַח.
  • חֲשִׂיפַת הַקְּבָרִים: לְאַחַר הַנִּקָּיוֹן, הֵחֵל הַשָּׁלָב הַמְרַגֵּשׁ. הוּא מָצָא שִׁבְרֵי מַצֵּבוֹת שֶׁשֻּׁלְּבוּ בִּבְנִיַּת טֵרָסוֹת, וְהֵחֵל לְהַרְכִּיב אוֹתָן כְּמוֹ פַּאזֶל עַד שֶׁגִּלָּה אֶת שְׁמוֹת נִרְצְחֵי תרפ"ט. כְּמוֹ כֵן, חָשַׂף אֶת קִבְרוֹ שֶׁל הַמְקֻבָּל הָרַב אֵלִיָּהוּ דִּי וִידַאשׂ, בַּעַל הַ"רֵאשִׁית חָכְמָה", וְאֶת קִבְרָהּ שֶׁל הָרַבָּנִית מְנוּחָה רָחֵל.

הַצְלָחוֹתָיו הָרִאשׁוֹנוֹת, שֶׁהֻשְּׂגוּ בְּשֵׂכֶל וּבִנְחִישׁוּת, חִזְּקוּ אֶת רוּחוֹ לִגֶּשֶׁת לָאֶתְגָּר הַגָּדוֹל, הַמּוּרְכָּב וְהַמְזֹהָם מִכֹּל: בֵּית כְּנֶסֶת אַבְרָהָם אָבִינוּ.

4. קוּם וּבְנֵה: גְּאֻלַּת בֵּית הַכְּנֶסֶת אַבְרָהָם אָבִינוּ

הַמַּאֲבָק עַל בֵּית הַכְּנֶסֶת אַבְרָהָם אָבִינוּ לֹא הָיָה רַק מִבְצָע הַנְדָּסִי שֶׁל חֲפִירָה וְנִקּוּי. זוֹ הָיְתָה הִתְנַגְּשׁוּת חֲזִיתִית בֵּין הַהִגָּיוֹן הַצִּיּוֹנִי הַפָּשׁוּט וְהַיָּשָׁר שֶׁל טַבְגֶּר לְבֵין הַבִּירוֹקְרַטְיָה, הַפּוֹלִיטִיקָה וְהָאַבְּסוּרְד שֶׁל הַמִּמְשָׁל הַצְּבָאִי. בְּמַאֲבָק זֶה, נֶאֱלַץ טַבְגֶּר לִלְחֹם בִּשְׁתֵּי חֲזִיתוֹת בּוֹ-זְמַנִּית.

הָאֶתְגָּר הָעַרְבִי וְהָאֶתְגָּר הַצְּבָאִי

בְּאֹפֶן מַפְתִּיעַ, הָאֶתְגָּר מוּל הָעֲרָבִים הַמְּקוֹמִיִּים הָיָה הַפָּשׁוּט יוֹתֵר. כַּאֲשֶׁר יְהוּדִי בָּא עִם הָאֱמֶת שֶׁלּוֹ, בְּאֹפֶן נָחוּשׁ וּמוּחְלָט, הֵם הֵבִינוּ וְכִבְּדוּ זֹאת. עִם הַזְּמַן, הֵם אֲפִלּוּ הֵחֵלּוּ לַעֲזֹר לוֹ: הִזְהִירוּ אוֹתוֹ מִפְּנֵי בּוֹרוֹת מַיִם נִסְתָּרִים, הֵבִיאוּ לוֹ מַיִם לִשְׁתִיָּה. הַמַּאֲבָק הָאֲמִיתִּי, הַמֻּרְכָּב וְהַמַּרְתִּיחַ, הָיָה מוּל הַצָּבָא וְהַמִּשְׁטָרָה. כְּפִי שֶׁטַבְגֶּר עַצְמוֹ נִתֵּחַ: הַבְּעָיָה מַתְחִילָה "כַּאֲשֶׁר הַפּוֹלִיטִיקָה נִכְנֶסֶת לַצָּבָא". הַחַיָּלִים נִשְׁלְחוּ לֹא כְּדֵי לְהָגֵן עַל יְהוּדִים, אֶלָּא כְּדֵי לִמְנֹעַ מִיְּהוּדִי לְנַקּוֹת בֵּית כְּנֶסֶת חָרֵב.

מִלְחֶמֶת הַתָּשָׁה מִשְׁפָּטִית וּפִיזִית

הָרָשֻׁיּוֹת הִפְעִילוּ נֶגְדּוֹ מַעֲצָרִים חוֹזְרִים וְנִשְׁנִים, אַךְ טַבְגֶּר הָפַךְ אֶת הַמַּעֲצָרִים לְכֵלִים בְּמִלְחֶמֶת הַהַתָּשָׁה שֶׁלּוֹ, בְּשִׁיטוֹת שֶׁזִּקֵּק עוֹד בְּהִתְמוֹדְדוּתוֹ עִם הַבִּירוֹקְרַטְיָה הָאַבְּסוּרְדִּית בִּבְרִית הַמּוֹעֲצוֹת. הוּא נָהַג לְהַתִּישׁ אֶת הַמַּעֲרֶכֶת בְּאֶמְצָעוּת הוּמוֹר וְתַחְכּוּם. הוּא הֶחְלִיף אֶת הַשְּׁלָטִים בֵּין מִשְׂרַד מְפַקֵּד הַתַּחֲנָה לְבֵין הַשֵּׁרוּתִים, וְכַאֲשֶׁר רָצוּ לְשַׁחְרֵר אוֹתוֹ, סֵרֵב בִּטְעָנָה שֶׁהוּא עָיֵף וְרוֹצֶה לָנוּחַ. בְּאַחַד הַמִּקְרִים, כְּשֶׁקָּצִין נִסָּה לְהִתְעַלֵּם מִמֶּנּוּ, טַבְגֶּר "דָּחַף אוֹתוֹ בְּכַוָּנָה" כְּדֵי לִכְפּוֹת אֶת הַמַּעֲצָר וּלְהַגְחִיךְ אֶת הַמַּצָּב.

הַשִּׂיא הִגִּיעַ בְּבֵית הַמִּשְׁפָּט הַצְּבָאִי. טַבְגֶּר, שֶׁהָיָה בַּעַל מֹחַ לוֹגִי חַד, נֶהֱנָה לַחְשֹׂף אֶת הָאַבְּסוּרְד שֶׁבַּהַאֲשָׁמוֹת. הַדִּיאָלוֹג הַבָּא עִם הַשּׁוֹפֵט מַדְגִּים זֹאת הֵיטֵב:

הַשּׁוֹפֵט: "אַתָּה נֶאֱשָׁם שֶׁנִּכְנַסְתָּ לְשֶׁטַח צְבָאִי סָגוּר."
טַבְגֶּר: "אֲנִי לֹא מֵבִין."
הַשּׁוֹפֵט: "מָה אַתָּה לֹא מֵבִין? שֶׁטַח צְבָאִי סָגוּר, נִכְנַסְתָּ!"
טַבְגֶּר: "אֲבָל אִם הוּא שֶׁטַח צְבָאִי סָגוּר, אִם הוּא סָגוּר, אֵיךְ נִכְנַסְתִּי?"

נִצְחוֹנוֹתָיו שֶׁל טַבְגֶּר, שֶׁהֻשְּׂגוּ לֹא רַק בְּאֵת וּבְמַעְדֵּר אֶלָּא גַּם בְּשֵׂכֶל חָרִיף וּבְהוּמוֹר, לֹא הָיוּ מִקְרִיִּים; הֵם נָבְעוּ מִתְּפִיסַת עוֹלָם מְגֻבֶּשֶׁת שֶׁרָאֲתָה בָּאַבְּסוּרְד אֶתְגָּר וְלֹא מִכְשׁוֹל.

5. הַפִילוֹסוֹפְיָה שֶׁל טַבְגֶּר: אַחֲרָיוּת אִישִׁית מוּל "חֹק מוֹק"

מַעֲשָׂיו שֶׁל פּרוֹפֵסוֹר טַבְגֶּר לֹא הָיוּ אִימְפּוּלְסִיבִיִּים. הֵם נָבְעוּ מִתְּפִיסַת עוֹלָם עֲמֻקָּה וּמְגֻבֶּשֶׁת, שֶׁהִבְחִינָה בְּחַדּוּת בֵּין חֹק אֲמִיתִּי, שֶׁנּוֹעַד לְטוֹבַת הָאָדָם, לְבֵין פּוֹlִיטִיקָה הַמִּתְחַפֶּשֶׂת לְחֹק. פֶּרֶק זֶה מְזַקֵּק אֶת עֶקְרוֹנוֹת הַיְּסוֹד שֶׁהִנְחוּ אוֹתוֹ:

  1. לְקִיחַת אַחֲרָיוּת אִישִׁית: זֶהוּ הָעִקָּרוֹן הַמֶּרְכָּזִי בְּמִשְׁנָתוֹ. "אֲנִי נִסִּיתִי לַעֲשׂוֹת מַה שֶּׁאֲנִי יָכוֹל", אָמַר. הוּא הֶאֱמִין שֶׁכְּדֵי לַעֲשׂוֹת אֶת הַדָּבָר הַנָּכוֹן, לֹא רַק שֶׁאֵין צֹרֶךְ לְבַקֵּשׁ רְשׁוּת, אֶלָּא שֶׁעֶצֶם בַּקָּשַׁת הָרְשׁוּת הִיא מַעֲשֶׂה לֹא מוּסָרִי. מַדּוּעַ? כִּי בַּקָּשַׁת אִשּׁוּר מַעֲבִירָה אֶת הָאַחֲרָיוּת מֵהָאָדָם הַפּוֹעֵל אֶל פָּקִיד אוֹ קָצִין בַּעַל אִינְטֶרֶסִים מְסֻבָּכִים. זוֹהִי הִתְחַמְּקוּת מֵהַחוֹבָה הָאֱנוֹשִׁית הַבְּסִיסִית לִפְעֹל וּלְתַקֵּן.
  2. הַבְחָנָה בֵּין חֹק לְפוֹלִיטִיקָה: אֶת הַשִּׁעוּר הֶחָשׁוּב בְּיוֹתֵר לָמַד טַבְגֶּר מִמּוֹשֵׁל חֶבְרוֹן הַצְּבָאִי. הַמּוֹשֵׁל הִסְבִּיר לוֹ אֶת עִקְּרוֹן הַ"חֹק מוֹק". כְּלוֹמַר, יֵשׁ אֶת הַחֹק הַכָּתוּב, וְיֵשׁ אֶת הַ"מוֹק" (מַה שֶּׁמִּתְחַשֵּׁק לַשִּׁלְטוֹן) – וְהַשִּׁלְטוֹן פּוֹעֵל לְפִי הַ"מוֹק". הֲבָנָה זוֹ שִׁחְרְרָה אוֹתוֹ מֵהַפַּחַד מִפְּנֵי הַמַּעֲרֶכֶת הַמִּשְׁפָּטִית. הוּא רָאָה בַּמִּשְׁפָּטִים לֹא אִיּוּם, אֶלָּא הִזְדַּמְּנוּת "לְהוֹכִיחַ אֶת קְלוֹנָם בָּרַבִּים", בִּידִיעָה שֶׁהוּא לֹא עוֹבֵר עַל שׁוּם חֹק אֲמִיתִּי.
  3. עֲשִׂיָּה עַל פְּנֵי דִּבּוּרִים: חָנָן פּוֹרָת, מִמַּנְהִיגֵי גּוּשׁ אֱמוּנִים, זִהָה אֶת הַחִדּוּשׁ שֶׁהֵבִיא טַבְגֶּר: "בִּמְקוֹם לְדַבֵּר וְלַעֲשׂוֹת הַפְגָּנוֹת, בּוֹר פֹּה, בּוֹר שָׁם". טַבְגֶּר הֶאֱמִין בְּכֹחָהּ שֶׁל הָעֲשִׂיָּה הַפְּשׁוּטָה, הַמְּמֻקֶּדֶת. הוּא הוֹכִיחַ שֶׁחֲפִירָה שְׁקֵטָה בְּבֵית כְּנֶסֶת וְקִיּוּם תְּפִלָּה עִקֶּשֶׁת הֵם כֵּלִים יְעִילִים הַרְבֵּה יוֹתֵר מִכָּל מַאֲבָק פּוֹלִיטִי רַעֲשָׁנִי.

מוֹרַשְׁתּוֹ שֶׁל טַבְגֶּר אֵינָהּ רַק הַבִּנְיָנִים שֶׁשִּׁקֵּם, אֶלָּא תְּפִיסַת הָעוֹלָם הַחֲלוּצִית וְהָאַמִּיצָה שֶׁהִנְחִיל, תְּפִיסָה שֶׁרֶלֵוַנְטִית כַּיּוֹם יוֹתֵר מִתָּמִיד.

6. סִיּוּם: מוֹרֶשֶׁת חַיָּה וְנוֹשֶׁמֶת

פָּעֳלוֹ שֶׁל פּרוֹפֵסוֹר בֶּן צִיּוֹן טַבְגֶּר הוּא עֵדוּת לְכֹחוֹ שֶׁל אָדָם אֶחָד, חָמוּשׁ בֶּאֱמוּנָה, הִגָּיוֹן וּנְחִישׁוּת, לְשַׁנּוֹת מְצִיאוּת. דְּבָרִים שֶׁנִּרְאִים הַיּוֹם מוּבָנִים מֵאֲלֵיהֶם בְּחֶבְרוֹן – בֵּית כְּנֶסֶת פָּעִיל הַמָּלֵא בְּמִתְפַּלְּלִים, בֵּית עַלְמִין מְשֻׁקָּם שֶׁפּוֹקְדִים אֶת קִבְרוֹתָיו – הֵם תּוֹצָאָה יְשִׁירָה שֶׁל מַאֲבָקוֹ הָעִקֵּשׁ, שֶׁנִּהֵל לְבַדּוֹ בְּרֹב הַמְּקָרִים.

אַרְבָּעִים שָׁנָה לְאַחַר פְּטִירָתוֹ, סִפְרוֹ "חֶבְרוֹן שֶׁלִּי" מַמְשִׁיךְ לִהְיוֹת רֶלֵוַנְטִי וּמְעוֹרֵר הַשְׁרָאָה. כְּפִי שֶׁמְּעִידִים פְּעִילִים, הוּא מְשַׁמֵּשׁ עֲבוּרָם כְּמִין "תַּנַ"ךְ פּוֹלִיטִי", מַדְרִיךְ לַעֲשִׂיָּה נְכוֹנָה וְאַמִּיצָה. בְּיוֹם הַשָּׁנָה הָ-40 לִפְטִירָתוֹ, שֶׁצֻּיַּן בְּאֵרוּעַ מְיֻחָד, הֻדְגַּשׁ שׁוּב כַּמָּה דַּרְכּוֹ הַמְּיֻחֶדֶת – דֶּרֶךְ שֶׁל אַחֲרָיוּת אִישִׁית, שֵׂכֶל יָשָׁר וַעֲשִׂיָּה – מַמְשִׁיכָה לְהָאִיר אֶת הַדֶּרֶךְ לַדּוֹרוֹת הַבָּאִים.

🎯 מִבְחַן חִדּוּד

1. מָה הָיְתָה הַסִּבָּה הָעִקָּרִית שֶׁפְּעֻלָּתוֹ הָרִאשׁוֹנָה שֶׁל פּרוֹפ' טַבְגֶּר הָיְתָה קְשִׁירַת עַצְמוֹ בְּשַׁרְשֶׁרֶת בִּמְעָרַת הַמַּכְפֵּלָה?

  • א. כְּדֵי לִגְרֹם לִפְרוֹבוֹקַצְיָה אֲלִימָה וּלְעוֹרֵר אֶת דַּעַת הַקָּהָל.
  • ב. כְּדֵי לְהַפְגִּין פָּסִיבִיוּת שֶׁמַּעֲמִידָה אֶת הָרָשֻׁיּוֹת בִּפְנֵי דִּילֶמָּה, וּלְקַבֵּעַ עוּבְדָּה שֶׁ"אֲנִי פֹּה כְּדֵי לְהִשָּׁאֵר".
  • ג. כִּי הוּא יָדַע שֶׁהַחַיָּלִים לֹא יָעֵזּוּ לִפְגֹּעַ בּוֹ מִפְּנֵי מַעֲמָדוֹ כִּפְרוֹפֵסוֹר.
  • ד. כְּדֵי לְקַבֵּל כּוֹתָרוֹת בַּעִתּוֹנוּת וּלְגַיֵּס תְּמִיכָה צִבּוּרִית לַמַּאֲבָק.
תְּשׁוּבָה נְכוֹנָה: ב'. הַטַּקְטִיקָה שֶׁל טַבְגֶּר לֹא חִפְּשָׂה עִמּוּת, אֶלָּא יְצִירַת עוּבְדָּה בַּשֶּׁטַח. הַקְּשִׁירָה אִפְשְׁרָה לוֹ לִטְעֹן "אֲנִי פָּסִיבִי", וּבְכָךְ לְאַתְגֵּר אֶת הַחַיָּלִים לְפַנּוֹת אוֹתוֹ בְּכֹחַ. הַמַּטָּרָה הָיְתָה לְעַרְעֵר עַל חֻקִּיּוּת הַהַגְבָּלָה וְלִקְבֹּעַ נוֹכָחוּת יְהוּדִית קְבוּעָה.

2. עַל פִּי הַמַּאֲמָר, מַדּוּעַ הָיָה הָאֶתְגָּר מוּל הָעֲרָבִים בְּשִׁקּוּם בֵּית הַכְּנֶסֶת "פָּשׁוּט יוֹתֵר" מֵהָאֶתְגָּר מוּל הַצָּבָא וְהַמִּשְׁטָרָה?

  • א. כִּי הָעֲרָבִים פָּחֲדוּ מִפְּנֵי הַצָּבָא וְלָכֵן לֹא הִתְעַמְּתוּ אִתּוֹ.
  • ב. כִּי טַבְגֶּר שִׁלֵּם לָעֲרָבִים כֶּסֶף כְּדֵי שֶׁיַּעַזְרוּ לוֹ בַּעֲבוֹדָה.
  • ג. כִּי הָעֲרָבִים כִּבְּדוּ אֶת נְחִישׁוּתוֹ וְהֵבִינוּ אֶת הָאֱמֶת הַפְּשׁוּטָה שֶׁבְּמַעֲשָׂיו, בְּעוֹד שֶׁהַצָּבָא פָּעַל מִתּוֹךְ שִׁקּוּלִים פּוֹלִיטִיִּים.
  • ד. כִּי הַמִּשְׁטָרָה אָסְרָה עַל הָעֲרָבִים לְהִתְקָרֵב לָאֵזוֹר הַחֲפִירוֹת.
תְּשׁוּבָה נְכוֹנָה: ג'. הַמַּאֲמָר מַדְגִּישׁ שֶׁכַּאֲשֶׁר יְהוּדִי בָּא עִם "הָאֱמֶת הַמֻּחְלֶטֶת" שֶׁלּוֹ, הָעֲרָבִים הֵבִינוּ וְכִבְּדוּ זֹאת. הַבְּעָיָה הָאֲמִיתִּית הָיְתָה עִם הַצָּבָא, שֶׁפָּעַל לֹא עַל פִּי הִגָּיוֹן אוֹ צֶדֶק, אֶלָּא עַל פִּי "פּוֹלִיטִיקָה שֶׁנִּכְנְסָה לַצָּבָא".

3. מָהוּ עִקְרוֹן הַ-"חֹק מוֹק" שֶׁלָּמַד טַבְגֶּר מִמּוֹשֵׁל חֶבְרוֹן, וְכֵיצַד הוּא הִשְׁפִּיעַ עַל פְּעֻלּוֹתָיו?

  • א. זֶהוּ חֹק צְבָאִי קָשׁוּחַ, שֶׁגָּרַם לְטַבְגֶּר לַחְשֹׁשׁ מֵהַמִּשְׁפָּטִים וּלְהִסְתַּתֵּר.
  • ב. זֶהוּ שֵׁם קוֹד לַחֹק הַיַּרְדֵּנִי, וְטַבְגֶּר נִסָּה לְהוֹכִיחַ שֶׁהוּא לֹא תָּקֵף.
  • ג. הָעִקָּרוֹן שֶׁהַשִּׁלְטוֹן פּוֹעֵל לְפִי מַה שֶּׁ"מִּתְחַשֵּׁק לוֹ" (מוֹק) וְלֹא לְפִי הַחֹק. הֲבָנָה זוֹ שִׁחְרְרָה אוֹתוֹ מֵהַפַּחַד וְהָפְכָה אֶת הַמִּשְׁפָּטִים לְהִזְדַּמְּנוּת לַחְשֹׂף אֶת צְבִיעוּת הַשִּׁלְטוֹן.
  • ד. זֶהוּ חֹק עַתִּיק שֶׁאוֹסֵר חֲפִירָה בַּאֲתָרִים קְדוֹשִׁים, וְטַבְגֶּר כִּבֵּד אוֹתוֹ.
תְּשׁוּבָה נְכוֹנָה: ג'. הַהֲבָנָה שֶׁהָרָשֻׁיּוֹת לֹא פּוֹעֲלוֹת עַל פִּי חֹק וָצֶדֶק, אֶלָּא עַל פִּי אִינְטֶרֶסִים פּוֹלִיטִיִּים ("מוֹק"), נָתְנָה לְטַבְגֶּר אֶת הַכֹּחַ לֹא לַחְשֹׁשׁ מֵהַמַּעֲרֶכֶת הַמִּשְׁפָּטִית וְאַף לְהִשְׁתַּמֵּשׁ בָּהּ כְּבָמָה לַחְשֹׂף אֶת קְלוֹנָהּ.

4. מַדּוּעַ, לְפִי הַפִילוֹסוֹפְיָה שֶׁל טַבְגֶּר, זֶה "לֹא מוּסָרִי לְבַקֵּשׁ אִשּׁוּר" לְמַעֲשֶׂה טוֹב?

  • א. כִּי בַּקָּשַׁת אִשּׁוּר גּוֹרֶמֶת לְעִכּוּבִים בִּירוֹקְרָטִיִּים מְיֻתָּרִים.
  • ב. כִּי הָרָשֻׁיּוֹת תָּמִיד יְסָרְבוּ, וְלָכֵן חֲבָל עַל הַזְּמַן.
  • ג. כִּי הַבַּקָּשָׁה מַעֲבִירָה אֶת הָאַחֲרָיוּת מֵהָאָדָם הַפּוֹעֵל אֶל הַפָּקִיד, וּמְשַׁחְרֶרֶת אֶת הָאָדָם מֵחובָתוֹ לַעֲשׂוֹת אֶת הַטּוֹב.
  • ד. כִּי זֶה מַרְאֶה עַל חֹסֶר אֱמוּנָה בְּצִדְקַת הַדֶּרֶךְ.
תְּשׁוּבָה נְכוֹנָה: ג'. עִקְּרוֹן הַיְּסוֹד שֶׁל טַבְגֶּר הָיָה לְקִיחַת אַחֲרָיוּת אִישִׁית. הוּא הֶאֱמִין שֶׁכַּאֲשֶׁר אָדָם מְבַקֵּשׁ אִשּׁוּר, הוּא בְּעֶצֶם אוֹמֵר: "זוֹ לֹא הָאַחֲרָיוּת שֶׁלִּי, אֶלָּא שֶׁלְּךָ". הוּא רָאָה בְּכָךְ הִתְחַמְּקוּת מֵהַחוֹבָה הָאֱנוֹשִׁית הַבְּסִיסִית לִפְעֹל וּלְתַקֵּן.

5. מָה הָיָה הַ"חִדּוּשׁ" שֶׁהֵבִיא טַבְגֶּר לְמַאֲבַק הַהִתְיַשְּׁבוּת, לְפִי דְּבָרָיו שֶׁל חָנָן פּוֹרָת?

  • א. הַיְּכֹלֶת לְגַיֵּס כְּסָפִים מִתּוֹרְמִים בְּחוּ"ל.
  • ב. הַמַּעֲבָר מִדִּבּוּרִים וְהַפְגָּנוֹת לַעֲשִׂיָּה פְּרַקְטִית וְיַצִּיבָה בַּשֶּׁטַח ("בּוֹר פֹּה, בּוֹר שָׁם").
  • ג. הַשִּׁמּוּשׁ בְּעוֹרְכֵי דִּין מְמֻלָּחִים כְּדֵי לְנַצֵּחַ בְּבָתֵּי הַמִּשְׁפָּט.
  • ד. הַקֶּשֶׁר הַטּוֹב שֶׁיָּצַר עִם הַתִּקְשֹׁרֶת הַזָּרָה.
תְּשׁוּבָה נְכוֹנָה: ב'. חָנָן פּוֹרָת זִהָה שֶׁטַבְגֶּר שִׁנָּה אֶת כְּלָלֵי הַמִּשְׂחָק. בִּמְקוֹם לִצְעֹק סִיסְמָאוֹת וּלְאַרְגֵּן הַפְגָּנוֹT, הוּא פָּשׁוּט הָלַךְ וְעָשָׂה. גִּישָׁה זוֹ שֶׁל יְצִירַת עוּבְדּוֹת בַּשֶּׁטַח הָיְתָה יְעִילָה וּמַהְפְּכָנִית לִתְקוּפָתָהּ.

📚 מְקוֹרוֹת

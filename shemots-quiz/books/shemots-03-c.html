וּלְכָל בְּנֵי יִשְׂרָאֵל הָיָה אוֹר

רָמָה ג׳ – מָה זֶה אוֹמֵר לִי

כָּל סִפּוּר פָּרָשַׁת בֹּא: אַרְבֶּה וְחֹשֶׁךְ, הַחֹדֶשׁ הַזֶּה לָכֶם, קָרְבַּן פֶּסַח, הַיְּצִיאָה מִמִּצְרַיִם וּמִצְווֹת הַזִּכָּרוֹן

🎉

💛 אַהֲבַת הַשֵּׁם: לַעֲבֹד אֶת ה׳ – זוֹ הַשִּׂמְחָה הֲכִי גְּדוֹלָה שֶׁלִּי 💛

📖 שְׁמוֹת, פֶּרֶק י׳, פָּסוּק ט׳

🔗 לְהַרְחָבָה בָּאֲתָר📝 שִׁעוּר: הִתְבּוֹנְנוּת בַּחֲסָדִים – שֹׁרֶשׁ הָעֲבוֹדָה וְהַתְּשׁוּבָה

רָמָה ג׳ – מָה זֶה אוֹמֵר לִי

🎉 ׳כִּי חַג ה׳ לָנוּ׳

רַב בְּאֶרֶץ יִשְׂרָאֵל לִמֵּד: פַּרְעֹה חָשַׁב שֶׁעֲבוֹדַת ה׳ הִיא ׳עֲבוֹדָה קָשָׁה׳ – לָכֵן רַק הַגְּבָרִים יֵלְכוּ. מֹשֶׁה עָנָה: אֵצֶלֵנוּ זוֹ חֲגִיגָה, ׳כִּי חַג ה׳ לָנוּ׳! וְהָרַב סִפֵּר עַל שְׁתֵּי מִשְׁפָּחוֹת שֶׁשָּׁמְרוּ שַׁבָּת: הַיְּלָדִים שֶׁל הָאַבָּא שֶׁשָּׂמַח בַּשַּׁבָּת – הָלְכוּ בְּדַרְכּוֹ. הַשַּׁבָּת הַזֹּאת אֲנִי שָׁר בְּשִׂמְחָה זֶמֶר אֶחָד לִכְבוֹד הַשַּׁבָּת.

מִצְווֹת הֵן חֲגִיגָה, לֹא מַשָּׂא.

📜 הַפָּסוּקוַיֹּאכַל אֶת כָּל עֵשֶׂב הָאָרֶץ וְאֵת כָּל פְּרִי הָעֵץ אֲשֶׁר הוֹתִיר הַבָּרָד וְלֹא נוֹתַר כָּל יֶרֶק

💛 אַהֲבַת הַשֵּׁם: ה׳ מַשְׁאִיר לִי תָּמִיד הַרְבֵּה טוֹב 💛

📖 שְׁמוֹת, פֶּרֶק י׳, פָּסוּק ט״ו

🔗 לְהַרְחָבָה בָּאֲתָר📝 שִׁעוּר: הִתְבּוֹנְנוּת בַּחֲסָדִים – שֹׁרֶשׁ הָעֲבוֹדָה וְהַתְּשׁוּבָה

רָמָה ג׳ – מָה זֶה אוֹמֵר לִי

🌾 לִרְאוֹת מָה נִשְׁאַר

שָׁלֹשׁ פְּעָמִים כָּתוּב שֶׁהָאַרְבֶּה אָכַל אֶת מָה שֶׁנִּשְׁאַר אַחֲרֵי הַבָּרָד. רַב בְּאֶרֶץ יִשְׂרָאֵל לִמֵּד: גַּם כְּשֶׁמַּשֶּׁהוּ נִלְקָח, תָּמִיד נִשְׁאָר הַרְבֵּה טוֹב – אֲנִי נוֹשֵׁם, רוֹאֶה, הוֹלֵךְ. מִי שֶׁשָּׂם לֵב לַטּוֹב שֶׁנִּשְׁאַר וּמוֹדֶה עָלָיו – זוֹ תְּשׁוּבָה אֲמִתִּית. הַיּוֹם, כְּשֶׁמַּשֶּׁהוּ לֹא מִסְתַּדֵּר, אֲנִי מוֹנֶה שְׁלֹשָׁה דְּבָרִים טוֹבִים שֶׁיֵּשׁ לִי.

מִסְתַּכְּלִים עַל מָה שֶׁנִּשְׁאַר – וּמוֹדִים עָלָיו.

💛 אַהֲבַת הַשֵּׁם: ה׳ עוֹשֶׂה הַכֹּל לְטוֹבָתֵנוּ, גַּם כְּשֶׁאֲנַחְנוּ לֹא מְבִינִים 💛

רָמָה ג׳ – מָה זֶה אוֹמֵר לִי

🖼️ הָרוּבָּלִים בַּמִּסְגֶּרֶת

רַב בְּאֶרֶץ יִשְׂרָאֵל סִפֵּר: יְהוּדִי אֶחָד הִפְסִיד אֶת כָּל כַּסְפּוֹ כְּשֶׁהַמַּטְבֵּעַ – הָרוּבָּל – אִבֵּד אֶת עֶרְכּוֹ. בִּגְלַל זֶה עָלָה לְאֶרֶץ יִשְׂרָאֵל, וְכָךְ נִצַּל מֵהַצָּרוֹת הַגְּדוֹלוֹת שֶׁבָּאוּ אַחַר כָּךְ עַל הַמָּקוֹם שֶׁעָזַב. הוּא שָׂם כַּמָּה רוּבָּלִים בְּמִסְגֶּרֶת בַּבַּיִת – ׳לְמַעַן תְּסַפֵּר בְּאָזְנֵי בִנְךָ׳. הַיּוֹם אֲנִי שׁוֹאֵל אֶת סַבָּא אוֹ אֶת אַבָּא עַל נֵס שֶׁקָּרָה בַּמִּשְׁפָּחָה.

מָה שֶׁנִּרְאֶה רַע – יָכוֹל לִהְיוֹת הַהַצָּלָה.

👂

📜 הַפָּסוּקוְהִגַּדְתָּ לְבִנְךָ בַּיּוֹם הַהוּא לֵאמֹר בַּעֲבוּר זֶה עָשָׂה ה׳ לִי בְּצֵאתִי מִמִּצְרָיִם

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹמֵעַ וְשָׂמֵחַ כְּשֶׁאֲנִי מְסַפֵּר עָלָיו 💛

רָמָה ג׳ – מָה זֶה אוֹמֵר לִי

👂 ה׳ מַקְשִׁיב לַסִּפּוּר

רַב בְּאֶרֶץ יִשְׂרָאֵל לִמֵּד מִן הַזֹּהַר הַקָּדוֹשׁ: כְּשֶׁיְּהוּדִים מְסַפְּרִים עַל יְצִיאַת מִצְרַיִם וּשְׂמֵחִים, ה׳ כִּבְיָכוֹל קוֹרֵא לְכָל הַפָּמַלְיָא שֶׁל מַעְלָה: ׳בּוֹאוּ שִׁמְעוּ אֶת סִפּוּר הַשֶּׁבַח שֶׁמְּסַפְּרִים בָּנַי׳. כָּל סִפּוּר כָּזֶה עוֹלֶה לְמַעְלָה. בְּלֵיל הַסֵּדֶר, וְגַם הַשָּׁבוּעַ בַּשֻּׁלְחָן, אֲנִי מְסַפֵּר בְּשִׂמְחָה פְּרָט אֶחָד מִיְּצִיאַת מִצְרַיִם.

כְּשֶׁאֲנִי מְסַפֵּר עַל הַנִּסִּים בְּשִׂמְחָה – ה׳ שָׂמֵחַ.

🌛

📜 הַפָּסוּקהַחֹדֶשׁ הַזֶּה לָכֶם רֹאשׁ חֳדָשִׁים רִאשׁוֹן הוּא לָכֶם לְחָדְשֵׁי הַשָּׁנָה

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי כָּל פַּעַם הִזְדַּמְּנוּת חֲדָשָׁה 💛

📖 שְׁמוֹת, פֶּרֶק י״ב, פָּסוּק ב׳

🔗 לְהַרְחָבָה בָּאֲתָר📝 שִׁעוּר: הַחֹדֶשׁ הַזֶּה לָכֶם

רָמָה ג׳ – מָה זֶה אוֹמֵר לִי

🌛 חֹדֶשׁ – מִלְּשׁוֹן חִדּוּשׁ

רַב בְּאֶרֶץ יִשְׂרָאֵל לִמֵּד בַּכִּתָּה: הַשֶּׁמֶשׁ תָּמִיד אוֹתָהּ שֶׁמֶשׁ, אֲבָל הַיָּרֵחַ נֶעֱלָם וְחוֹזֵר וּמִתְחַדֵּשׁ. ׳חֹדֶשׁ׳ – מִלְּשׁוֹן חִדּוּשׁ. גַּם אֲנַחְנוּ תָּמִיד יְכוֹלִים לְהִתְחַדֵּשׁ: אַחֲרֵי רִיב אוֹ טָעוּת – פּוֹתְחִים דַּף חָדָשׁ, וּמַתְחִילִים בְּצַעַד קָטָן, כְּמוֹ הַיָּרֵחַ הַדַּק. הַיּוֹם אֲנִי מַשְׁלִים עִם מִי שֶׁרַבְתִּי אִתּוֹ.

כְּמוֹ הַיָּרֵחַ – תָּמִיד אֶפְשָׁר לְהַתְחִיל מֵחָדָשׁ.

📜 הַפָּסוּקוּלְכָל בְּנֵי יִשְׂרָאֵל הָיָה אוֹר בְּמוֹשְׁבֹתָם

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹמֵר עַל הַבַּיִת שֶׁלִּי וּמֵאִיר אוֹתוֹ 💛

רָמָה ג׳ – מָה זֶה אוֹמֵר לִי

🏘️ אוֹתָהּ אֶרֶץ – וְאוֹר בַּבָּתִּים

רַב בְּאֶרֶץ יִשְׂרָאֵל לִמֵּד בַּכִּתָּה: בְּאוֹתָהּ אֶרֶץ וּבְאוֹתוֹ זְמַן – הַמִּצְרִים יוֹשְׁבִים בְּחֹשֶׁךְ כָּבֵד וְלֹא יְכוֹלִים לָזוּז, וּבְבָתֵּי יִשְׂרָאֵל אוֹר. כְּמוֹ בְּכָל הַמַּכּוֹת, ה׳ מַבְדִּיל וּמַרְאֶה לְכֻלָּם עַל מִי הוּא שׁוֹמֵר. הָעֶרֶב, כְּשֶׁאֲנִי מַדְלִיק אוֹר בַּבַּיִת, אֲנִי אוֹמֵר בַּלֵּב: ׳תּוֹדָה, ה׳, עַל הָאוֹר שֶׁבַּבַּיִת שֶׁלָּנוּ׳.

ה׳ מֵאִיר אֶת הַבָּתִּים שֶׁל יִשְׂרָאֵל.

📦 תְּפִלִּין וּמְזוּזָה – זֵכֶר בְּכָל יוֹם

הַתְּפִלִּין וְהַמְּזוּזָה מַזְכִּירוֹת לָנוּ אֶת ה׳ בְּכָל יוֹם.

💡 מָה פָּרָשַׁת בֹּא מְלַמֶּדֶת אוֹתִי הַיּוֹם

עֲבוֹדַת ה׳ הִיא חֲגִיגָה – ׳כִּי חַג ה׳ לָנוּ׳, וְכָל הַמִּשְׁפָּחָה שׁוֹתֶפָה בָּהּ. אֲנִי רוֹצֶה לְחַבֵּב אֶת הַמִּצְווֹת כְּמוֹ יִשְׂרָאֵל שֶׁנָּשְׂאוּ אֶת הַבָּצֵק עַל הַכָּתֵף, לְהַתְחִיל מֵחָדָשׁ כְּמוֹ הַיָּרֵחַ, וּלְסַפֵּר לְכֻלָּם עַל הַנִּסִּים שֶׁל ה׳ – ׳וְהִגַּדְתָּ לְבִנְךָ׳. הַיּוֹם אֲנִי עוֹשֶׂה מִצְוָה אַחַת בְּשִׂמְחָה, כְּמוֹ בְּחַג.

מִצְוָה בְּשִׂמְחָה, הַתְחָלָה חֲדָשָׁה, וְסִפּוּר עַל נִסֵּי ה׳.

מָה לוֹקְחִים מִרָמָה ג׳?

פָּרָשַׁת בֹּא מְלַמֶּדֶת שֶׁהַמִּצְווֹת הֵן חַג. כְּשֶׁאֲנִי שָׂמֵחַ בַּמִּצְוָה, רוֹאֶה אֶת הַטּוֹב שֶׁנִּשְׁאַר, מִתְחַדֵּשׁ וּמְסַפֵּר עַל נִסֵּי ה׳ – יֵשׁ אוֹר גַּם בַּבַּיִת שֶׁלִּי.

  • יִשְׂרָאֵל נָשְׂאוּ אֶת הַמִּצְוָה עַל הַכָּתֵף כִּי אָהֲבוּ אוֹתָהּ. אֵיזוֹ מִצְוָה אֲנִי הֲכִי אוֹהֵב?
  • הַיָּרֵחַ מִתְחַדֵּשׁ. בְּאֵיזֶה דָּבָר אֲנִי רוֹצֶה לְהַתְחִיל דַּף חָדָשׁ?

לָמָּה מִזְבַּח הַקְּטֹרֶת כָּתוּב רַק בְּסוֹף הַפָּרָשָׁה? רַב בְּאֶרֶץ יִשְׂרָאֵל לִמֵּד בְּשֵׁם אֶחָד מִגְּדוֹלֵי יִשְׂרָאֵל: הָיוּ שְׁנֵי מִזְבְּחוֹת – הַחִיצוֹן שֶׁל נְחֹשֶׁת, וְהַפְּנִימִי שֶׁל זָהָב, שֶׁעָלָיו רֵיחַ טוֹב שֶׁל קְטֹרֶת. יֵשׁ יְהוּדִי שֶׁהַמִּצְווֹת כְּבֵדוֹת עָלָיו, וְיֵשׁ יְהוּדִי שֶׁהַתְּפִלָּה מְתוּקָה לוֹ וְהַשַּׁבָּת נְעִימָה לוֹ – כִּי הוּא קָשׁוּר לַה׳ מִבִּפְנִים. ׳קְטֹרֶת׳ – לְשׁוֹן קֶשֶׁר. הַיּוֹם אֲנִי חוֹשֵׁב עַל מִלָּה אַחַת בַּתְּפִלָּה וְאוֹמֵר אוֹתָהּ מִכָּל הַלֵּב.

מִי שֶׁקָּשׁוּר לַה׳ מִבִּפְנִים – הַמִּצְווֹת מְתוּקוֹת לוֹ.

🚂 מִמִּצְוָה לֹא מַפְסִידִים

׳נֵר תָּמִיד׳ – וְהַתּוֹרָה וְהַמִּצְווֹת הֵן אוֹר. סִפּוּר יָדוּעַ: הָרַב מִפּוֹנֶבִיז׳ קָבַע פְּגִישָׁה עִם נָדִיב בְּתַחֲנַת הָרַכֶּבֶת בִּשְׁמוֹנֶה בַּבֹּקֶר, כְּדֵי לְבַקֵּשׁ תְּרוּמָה לַיְּשִׁיבָה – וְהִתְעוֹרֵר מְאֻחָר. הוּא הֶחְלִיט: קֹדֶם אֶתְפַּלֵּל בְּמִנְיָן. כְּשֶׁהִגִּיעַ, הָרַכֶּבֶת כְּבָר יָצְאָה – וּפִתְאוֹם הַנָּדִיב נִכְנַס, גַּם הוּא אֵחַר, וְנָתַן תְּרוּמָה גְּדוֹלָה. הַיּוֹם אֲנִי זוֹכֵר: מִמִּצְוָה אַף פַּעַם לֹא מַפְסִידִים.

מִי שֶׁשּׁוֹמֵר מִצְוָה – לֹא מַפְסִיד.

💡 מָה פָּרָשַׁת תְּצַוֶּה מְלַמֶּדֶת אוֹתִי הַיּוֹם

אַהֲרֹן נָשָׂא אֶת כָּל עַם יִשְׂרָאֵל עַל הַלֵּב, וּמֹשֶׁה שָׂמַח בִּשְׁבִיל אָחִיו וְוִתֵּר עַל הַכָּבוֹד שֶׁלּוֹ. אֲנִי רוֹצֶה לָתֵת לַה׳ אֶת הַטּוֹב בְּיוֹתֵר, לְהִתְפַּלֵּל עַל אֲחֵרִים, לִשְׁמֹר עַל הַפֶּה כְּמוֹ רִמּוֹן, וְלִשְׂמֹחַ בְּהַצְלָחָה שֶׁל חֲבֵרַי. כָּךְ גַּם אֲנִי מֵאִיר, כְּמוֹ נֵר תָּמִיד. הַיּוֹם אֲנִי בּוֹחֵר דָּבָר אֶחָד מֵאֵלֶּה – וְעוֹשֶׂה אוֹתוֹ.

כְּמוֹ אַהֲרֹן – גַּם אֲנִי יָכוֹל לָשֵׂאת אֶת כֻּלָּם עַל הַלֵּב.

מָה לוֹקְחִים מִרָמָה ג׳?

בִּגְדֵי הַכֹּהֵן מְלַמְּדִים אוֹתִי אֵיךְ לִחְיוֹת: לָשֵׂאת אֶת כֻּלָּם עַל הַלֵּב, לִשְׂמֹחַ בִּשְׁבִיל אֲחֵרִים, לִשְׁמֹר עַל הַפֶּה, וְלִהְיוֹת קָשׁוּר לַה׳ מִבִּפְנִים.

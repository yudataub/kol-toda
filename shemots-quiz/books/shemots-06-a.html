נַעֲשֶׂה וְנִשְׁמָע

פָּרָשַׁת מִשְׁפָּטִים: מִצְווֹת שֶׁל יֹשֶׁר, חֶסֶד וְרַחֲמִים, שְׁלֹשֶׁת הָרְגָלִים, וְהַבְּרִית בְּהַר סִינַי

🍽️ ׳וְאֵלֶּה הַמִּשְׁפָּטִים׳

אַחֲרֵי עֲשֶׂרֶת הַדִּבְּרוֹת ה׳ אוֹמֵר לְמֹשֶׁה: ׳וְאֵלֶּה הַמִּשְׁפָּטִים אֲשֶׁר תָּשִׂים לִפְנֵיהֶם׳ – דִּינִים שֶׁל צֶדֶק בֵּין אָדָם לַחֲבֵרוֹ. רַשִׁ״י מְסַבֵּר: ׳וְאֵלֶּה׳ – כְּמוֹ שֶׁהַדִּבְּרוֹת מִסִּינַי, גַּם הַמִּשְׁפָּטִים מִסִּינַי. וְ׳תָּשִׂים לִפְנֵיהֶם׳ – ׳כְּשֻׁלְחָן הֶעָרוּךְ וּמוּכָן לֶאֱכֹל׳: מֹשֶׁה צָרִיךְ לְהַסְבִּיר אוֹתָם הֵיטֵב.

גַּם הַדִּינִים בֵּין אָדָם לַחֲבֵרוֹ – מִסִּינַי.

🩺

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁכָּל אֶחָד יִהְיֶה בָּרִיא וְשָׁלֵם 💛

📖 שְׁמוֹת, פֶּרֶק כ״א, פְּסוּקִים י״ח–י״ט

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🩺 ׳וְרַפֹּא יְרַפֵּא׳

אִם אָדָם פָּגַע בַּחֲבֵרוֹ, וְהֶחָבֵר שָׁכַב חוֹלֶה – הַתּוֹרָה אוֹמֶרֶת: ׳רַק שִׁבְתּוֹ יִתֵּן וְרַפֹּא יְרַפֵּא׳. רַשִׁ״י מְסַבֵּר: הוּא צָרִיךְ לְשַׁלֵּם עַל הַיָּמִים שֶׁהֶחָבֵר לֹא יָכֹל לַעֲבֹד, ׳יְשַׁלֵּם שְׂכַר הָרוֹפֵא׳.

מִי שֶׁגָּרַם צַעַר – דּוֹאֵג לְתַקֵּן.

2
🕳️

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁאֶשְׁמֹר עַל כָּל מִי שֶׁעוֹבֵר בַּדֶּרֶךְ 💛

📖 שְׁמוֹת, פֶּרֶק כ״א, פְּסוּקִים ל״ג–ל״ד

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🕳️ ׳וְלֹא יְכַסֶּנּוּ׳

׳וְכִי יִפְתַּח אִישׁ בּוֹר... וְלֹא יְכַסֶּנּוּ, וְנָפַל שָׁמָּה שׁוֹר אוֹ חֲמוֹר׳ – בַּעַל הַבּוֹר יְשַׁלֵּם. רַשִׁ״י מְסַבֵּר: מְדֻבָּר בְּבוֹר שֶׁחָפְרוּ בִּרְשׁוּת הָרַבִּים – בַּמָּקוֹם שֶׁכֻּלָּם עוֹבְרִים. מִי שֶׁפּוֹתֵחַ בּוֹר – צָרִיךְ לְכַסּוֹת אוֹתוֹ.

אַחְרָאִי: לֹא מַשְׁאִירִים מִכְשׁוֹל בַּדֶּרֶךְ.

3
🔥

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב כְּשֶׁאֲנִי זָהִיר וְשׁוֹמֵר עַל אֲחֵרִים 💛

📖 שְׁמוֹת, פֶּרֶק כ״ב, פְּסוּקִים ד׳–ה׳

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🔥 ׳כִּי תֵצֵא אֵשׁ׳

׳כִּי תֵצֵא אֵשׁ וּמָצְאָה קֹצִים׳ – אִם מִישֶׁהוּ הִדְלִיק אֵשׁ, וְהִיא עָבְרָה לַשָּׂדֶה שֶׁל הַשָּׁכֵן וְשָׂרְפָה אֶת הַתְּבוּאָה – ׳שַׁלֵּם יְשַׁלֵּם הַמַּבְעִר אֶת הַבְּעֵרָה׳. גַּם בְּהֵמָה שֶׁאָכְלָה בַּשָּׂדֶה שֶׁל אַחֵר – הַבְּעָלִים מְשַׁלֵּם.

שׁוֹמְרִים שֶׁהַדְּבָרִים שֶׁלִּי לֹא יַזִּיקוּ לַאֲחֵרִים.

4
🔐

💛 אַהֲבַת הַשֵּׁם: ה׳ סוֹמֵךְ עָלַי שֶׁאֶהְיֶה נֶאֱמָן 💛

📖 שְׁמוֹת, פֶּרֶק כ״ב, פְּסוּקִים ו׳–י״ד

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🔐 שׁוֹמֵר עַל חֵפֶץ שֶׁל חָבֵר

׳כִּי יִתֵּן אִישׁ אֶל רֵעֵהוּ כֶּסֶף אוֹ כֵלִים לִשְׁמֹר׳ – הַתּוֹרָה מְלַמֶּדֶת אֶת דִּינֵי הַשּׁוֹמֵר. מִי שֶׁקִּבֵּל דָּבָר לִשְׁמִירָה – צָרִיךְ לִשְׁמֹר עָלָיו הֵיטֵב, כְּמוֹ עַל שֶׁלּוֹ.

מָה שֶּׁנָּתְנוּ לִי לִשְׁמֹר – אֲנִי שׁוֹמֵר בְּנֶאֱמָנוּת.

5
🧳

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אֶת הַגֵּר, וְגַם אֲנִי 💛

📖 שְׁמוֹת, פֶּרֶק כ״ב, פָּסוּק כ׳

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🧳 ׳וְגֵר לֹא תוֹנֶה׳

׳וְגֵר לֹא תוֹנֶה וְלֹא תִלְחָצֶנּוּ כִּי גֵרִים הֱיִיתֶם בְּאֶרֶץ מִצְרָיִם׳. רַשִׁ״י מְסַבֵּר: ׳לֹא תוֹנֶה׳ – לֹא מְצַעֲרִים בְּמִלִּים, וְ׳לֹא תִלְחָצֶנּוּ׳ – לֹא לוֹקְחִים מִמֶּנּוּ כֶּסֶף. ׳גֵּר׳ הוּא מִי שֶׁבָּא מִמָּקוֹם אַחֵר. אַתֶּם הֱיִיתֶם גֵּרִים בְּמִצְרַיִם – וְאַתֶּם יוֹדְעִים כַּמָּה זֶה קָשֶׁה.

מִי שֶׁהָיָה זָר – יוֹדֵעַ לְקָרֵב אֶת הַזָּר.

👂

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹמֵעַ כָּל לֵב שָׁבוּר 💛

📖 שְׁמוֹת, פֶּרֶק כ״ב, פְּסוּקִים כ״א–כ״ב

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

👂 ׳שָׁמֹעַ אֶשְׁמַע צַעֲקָתוֹ׳

׳כָּל אַלְמָנָה וְיָתוֹם לֹא תְעַנּוּן׳ – לֹא מְצַעֲרִים יָתוֹם, יֶלֶד שֶׁאֵין לוֹ אַבָּא אוֹ אִמָּא. ה׳ אוֹמֵר: ׳כִּי אִם צָעֹק יִצְעַק אֵלַי – שָׁמֹעַ אֶשְׁמַע צַעֲקָתוֹ׳. רַשִׁ״י מְסַבֵּר: הַדִּין הַזֶּה ׳לְכָל אָדָם׳ – אֲבָל הַתּוֹרָה מַזְכִּירָה אוֹתָם, כִּי הֵם חַלָּשִׁים.

לֹא מְצַעֲרִים אַף אֶחָד – וּבְמִיּוּחָד אֶת הֶחָלָשׁ.

🪙

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי, כְּדֵי שֶׁגַּם אֲנִי אֶתֵּן 💛

📖 שְׁמוֹת, פֶּרֶק כ״ב, פָּסוּק כ״ד

רָמָה א׳ – כָּל סִפּוּר הַפָּרָשָׁה

🪙 ׳אִם כֶּסֶף תַּלְוֶה׳

׳אִם כֶּסֶף תַּלְוֶה אֶת עַמִּי אֶת הֶעָנִי עִמָּךְ׳ – מַלְוִים כֶּסֶף לְמִי שֶׁצָּרִיךְ. רַשִׁ״י מְסַבֵּר: ׳הֱוֵי מִסְתַּכֵּל בְּעַצְמְךָ כְּאִלּוּ אַתָּה עָנִי׳. וְ׳לֹא תִהְיֶה לוֹ כְּנֹשֶׁה׳ – לֹא לוֹחֲצִים עָלָיו וְלֹא מְבַיְּשִׁים אוֹתוֹ, ׳לֹא תַכְלִימֵהוּ׳. וְלֹא לוֹקְחִים רִבִּית.

עוֹזְרִים לֶעָנִי – וְשׁוֹמְרִים עַל הַכָּבוֹד שֶׁלּוֹ.

8
🧥

💛 אַהֲבַת הַשֵּׁם: ה׳ חַנּוּן – וְהוּא דּוֹאֵג גַּם לַשְּׂמִיכָה שֶׁל הֶעָנִי 💛

📖 שְׁמוֹת, פֶּרֶק כ״ב, פְּסוּקִים כ״ה–כ״ו

🧥 הַשְּׂמִיכָה שֶׁל הֶעָנִי

אִם לָקַחְתָּ מִן הֶעָנִי מַשְׁכּוֹן – דָּבָר שֶׁלּוֹ עַד שֶׁיַּחֲזִיר אֶת הַכֶּסֶף – ׳עַד בֹּא הַשֶּׁמֶשׁ תְּשִׁיבֶנּוּ לוֹ׳. ׳כִּי הִוא כְסוּתוֹ לְבַדָּהּ... בַּמֶּה יִשְׁכָּב?׳ – הֲרֵי זֶה הַבֶּגֶד שֶׁהוּא מִתְכַּסֶּה בּוֹ בַּלַּיְלָה! ה׳ אוֹמֵר: ׳וְשָׁמַעְתִּי כִּי חַנּוּן אָנִי׳.

חוֹשְׁבִים: מָה הַשֵּׁנִי צָרִיךְ עַכְשָׁו?

מְבִיאִים לַה׳ בִּזְמַן אֶת הַתְּרוּמוֹת וְאֶת הַבְּכוֹרוֹת, ׳וְאַנְשֵׁי קֹדֶשׁ תִּהְיוּן לִי׳ – לָכֵן לֹא אוֹכְלִים בְּשַׂר טְרֵפָה, ׳לַכֶּלֶב תַּשְׁלִכוּן אֹתוֹ׳. רַשִׁ״י מְסַבֵּר: לָמָּה דַּוְקָא לַכֶּלֶב? כִּי הַכְּלָבִים לֹא נָבְחוּ בְּיִשְׂרָאֵל בְּלֵיל יְצִיאַת מִצְרַיִם, ׳וְאֵין הַקָּדוֹשׁ בָּרוּךְ הוּא מְקַפֵּחַ שְׂכַר כָּל בְּרִיָּה׳.

ה׳ מְשַׁלֵּם שָׂכָר אֲפִלּוּ לַכֶּלֶב.

׳לֹא תִשָּׂא שֵׁמַע שָׁוְא׳ – רַשִׁ״י מְסַבֵּר: לֹא מְקַבְּלִים דִּבּוּר שֶׁל שֶׁקֶר – זוֹ ׳אַזְהָרָה לִמְקַבֵּל לָשׁוֹן הָרָע׳. וְ׳לֹא תִהְיֶה אַחֲרֵי רַבִּים לְרָעֹת׳ – רַשִׁ״י מְסַבֵּר: אִם רַבִּים עוֹשִׂים דָּבָר לֹא יָשָׁר, לֹא אוֹמְרִים ׳כֻּלָּם עוֹשִׂים – גַּם אֲנִי׳.

לֹא שׁוֹמְעִים לָשׁוֹן הָרָע, וְלֹא הוֹלְכִים אַחֲרֵי הָרֹב לְרָעָה.

סוֹף רָמָה א׳

🤔 חוֹשְׁבִים עַל זֶה

  • הַתּוֹרָה אוֹמֶרֶת לַעֲזֹר גַּם לְמִי שֶׁלֹּא אוֹהֵב אוֹתִי. לְמִי אֲנִי יָכוֹל לַעֲזֹר הַיּוֹם?
  • ׳וְגֵר לֹא תוֹנֶה׳. אֵיךְ אֲנִי מְקָרֵב יֶלֶד חָדָשׁ אוֹ בּוֹדֵד?
  • יִשְׂרָאֵל אָמְרוּ ׳נַעֲשֶׂה׳ לִפְנֵי ׳נִשְׁמָע׳. אֵיזוֹ מִצְוָה אֲנִי רוֹצֶה לַעֲשׂוֹת בְּלֵב שָׁלֵם?

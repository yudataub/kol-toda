🐪 יִתְרוֹ בָּא אֶל הַמִּדְבָּר

יִתְרוֹ בָּא אֶל מֹשֶׁה, אֶל הַמִּדְבָּר ׳אֲשֶׁר הוּא חֹנֶה שָׁם הַר הָאֱלֹהִים׳. אִתּוֹ בָּאוּ צִפּוֹרָה אֵשֶׁת מֹשֶׁה וּשְׁנֵי בָּנָיו: ׳גֵּרְשֹׁם – כִּי אָמַר גֵּר הָיִיתִי בְּאֶרֶץ נָכְרִיָּה׳, וֶ׳אֱלִיעֶזֶר – כִּי אֱלֹהֵי אָבִי בְּעֶזְרִי׳.

גַּם הַשֵּׁמוֹת שֶׁל בְּנֵי מֹשֶׁה הֵם תּוֹדָה לַה׳.

🤗 ׳וַיֵּצֵא מֹשֶׁה לִקְרַאת חֹתְנוֹ׳

מֹשֶׁה יוֹצֵא לִקְרַאת יִתְרוֹ, מִשְׁתַּחֲוֶה לוֹ וּמְנַשֵּׁק אוֹתוֹ, ׳וַיִּשְׁאֲלוּ אִישׁ לְרֵעֵהוּ לְשָׁלוֹם׳. רַשִׁ״י מְסַבֵּר: ׳כָּבוֹד גָּדוֹל נִתְכַּבֵּד יִתְרוֹ בְּאוֹתָהּ שָׁעָה׳ – כְּשֶׁמֹּשֶׁה יָצָא, יָצְאוּ גַּם אַהֲרֹן, נָדָב וַאֲבִיהוּא, וְכֻלָּם יָצְאוּ אַחֲרֵיהֶם.

מְקַבְּלִים אוֹרֵחַ בְּכָבוֹד וּבְשִׂמְחָה.

🗣️ מֹשֶׁה מְסַפֵּר

בָּאֹהֶל, ׳וַיְסַפֵּר מֹשֶׁה לְחֹתְנוֹ אֵת כָּל אֲשֶׁר עָשָׂה ה׳ לְפַרְעֹה וּלְמִצְרַיִם׳. הוּא סִפֵּר גַּם עַל ׳כָּל הַתְּלָאָה׳ – כָּל הַקְּשָׁיִים שֶׁהָיוּ בַּדֶּרֶךְ – ׳וַיַּצִּלֵם ה׳׳.

מְסַפְּרִים לַאֲחֵרִים עַל הַנִּסִּים שֶׁל ה׳.

׳וַיִּחַדְּ יִתְרוֹ עַל כָּל הַטּוֹבָה׳ – רַשִׁ״י מְסַבֵּר: ׳וַיִּשְׂמַח יִתְרוֹ׳. וְיִתְרוֹ אוֹמֵר: ׳בָּרוּךְ ה׳ אֲשֶׁר הִצִּיל אֶתְכֶם מִיַּד מִצְרַיִם׳, וְ׳עַתָּה יָדַעְתִּי כִּי גָדוֹל ה׳ מִכָּל הָאֱלֹהִים׳.

יִתְרוֹ שָׂמֵחַ בַּטּוֹב שֶׁל אֲחֵרִים וּמְבָרֵךְ אֶת ה׳.

יִתְרוֹ מֵבִיא קָרְבָּנוֹת לַה׳, ׳וַיָּבֹא אַהֲרֹן וְכֹל זִקְנֵי יִשְׂרָאֵל לֶאֱכָל לֶחֶם עִם חֹתֵן מֹשֶׁה לִפְנֵי הָאֱלֹהִים׳. כֻּלָּם שְׂמֵחִים יַחַד בַּסְּעוּדָה.

לְמָחֳרָת מֹשֶׁה יוֹשֵׁב לִשְׁפֹּט אֶת הָעָם, ׳וַיַּעֲמֹד הָעָם עַל מֹשֶׁה מִן הַבֹּקֶר עַד הָעָרֶב׳. כָּל מִי שֶׁיֵּשׁ לוֹ שְׁאֵלָה אוֹ רִיב – בָּא אֶל מֹשֶׁה. וּמֹשֶׁה מְלַמֵּד אוֹתָם ׳אֶת חֻקֵּי הָאֱלֹהִים וְאֶת תּוֹרֹתָיו׳.

מֹשֶׁה עוֹבֵד קָשֶׁה כָּל הַיּוֹם בִּשְׁבִיל הָעָם.

גַּם אָדָם גָּדוֹל צָרִיךְ עֶזְרָה.

יִתְרוֹ מְיַעֵץ: בְּחַר שׁוֹפְטִים – ׳אַנְשֵׁי חַיִל יִרְאֵי אֱלֹהִים אַנְשֵׁי אֱמֶת שֹׂנְאֵי בָצַע׳. רַשִׁ״י מְסַבֵּר: ׳אַנְשֵׁי אֱמֶת׳ – אֲנָשִׁים שֶׁאֶפְשָׁר לִסְמֹךְ עַל דִּבְרֵיהֶם. יִהְיוּ שָׂרֵי אֲלָפִים, שָׂרֵי מֵאוֹת, שָׂרֵי חֲמִשִּׁים וְשָׂרֵי עֲשָׂרוֹת. הֵם יִשְׁפְּטוּ אֶת הַדְּבָרִים הַקְּטַנִּים, וְהַדָּבָר הַגָּדוֹל יָבִיאוּ אֶל מֹשֶׁה.

מֹשֶׁה קוֹרֵא לְזִקְנֵי הָעָם וּמְסַפֵּר לָהֶם אֶת דִּבְרֵי ה׳. ׳וַיַּעֲנוּ כָל הָעָם יַחְדָּו וַיֹּאמְרוּ: כֹּל אֲשֶׁר דִּבֶּר ה׳ נַעֲשֶׂה׳. וּמֹשֶׁה מֵשִׁיב אֶת דִּבְרֵי הָעָם אֶל ה׳.

📯 קוֹלוֹת וְקוֹל שׁוֹפָר

בַּבֹּקֶר שֶׁל הַיּוֹם הַשְּׁלִישִׁי: ׳קֹלֹת וּבְרָקִים וְעָנָן כָּבֵד עַל הָהָר וְקֹל שֹׁפָר חָזָק מְאֹד׳. מֹשֶׁה מוֹצִיא אֶת הָעָם ׳לִקְרַאת הָאֱלֹהִים׳ – רַשִׁ״י מְסַבֵּר: ה׳ יָצָא לִקְרָאתָם ׳כְּחָתָן הַיּוֹצֵא לִקְרַאת כַּלָּה׳. הַר סִינַי עָשֵׁן כֻּלּוֹ, ׳וַיְהִי קוֹל הַשּׁוֹפָר הוֹלֵךְ וְחָזֵק מְאֹד׳.

מַתַּן תּוֹרָה – יוֹם הַשִּׂמְחָה הַגָּדוֹל שֶׁל עַם יִשְׂרָאֵל.

1️⃣ ׳אָנֹכִי ה׳ אֱלֹהֶיךָ׳

׳וַיְדַבֵּר אֱלֹהִים אֵת כָּל הַדְּבָרִים הָאֵלֶּה׳ – עֲשֶׂרֶת הַדִּבְּרוֹת. הַדִּבֵּר הָרִאשׁוֹן: ׳אָנֹכִי ה׳ אֱלֹהֶיךָ אֲשֶׁר הוֹצֵאתִיךָ מֵאֶרֶץ מִצְרַיִם׳. הַשֵּׁנִי: ׳לֹא יִהְיֶה לְךָ אֱלֹהִים אֲחֵרִים׳ – לֹא עוֹבְדִים שׁוּם דָּבָר אַחֵר, רַק אֶת ה׳.

ה׳ שֶׁהוֹצִיא אוֹתָנוּ מִמִּצְרַיִם – הוּא הָאֱלֹקִים שֶׁלָּנוּ.

הַדִּבֵּר הַשְּׁלִישִׁי: ׳לֹא תִשָּׂא אֶת שֵׁם ה׳ אֱלֹהֶיךָ לַשָּׁוְא׳ – לֹא מַזְכִּירִים אֶת שֵׁם ה׳ סְתָם. הָרְבִיעִי: ׳זָכוֹר אֶת יוֹם הַשַּׁבָּת לְקַדְּשׁוֹ׳ – שִׁשָּׁה יָמִים עוֹבְדִים, וּבַיּוֹם הַשְּׁבִיעִי נָחִים, כְּמוֹ שֶׁה׳ נָח אַחֲרֵי שֶׁבָּרָא אֶת הָעוֹלָם. רַשִׁ״י מְסַבֵּר: אִם מָצָאתָ דָּבָר יָפֶה – שְׁמֹר אוֹתוֹ לְשַׁבָּת.

הַדִּבֵּר הַחֲמִישִׁי: ׳כַּבֵּד אֶת אָבִיךָ וְאֶת אִמֶּךָ לְמַעַן יַאֲרִכוּן יָמֶיךָ עַל הָאֲדָמָה אֲשֶׁר ה׳ אֱלֹהֶיךָ נֹתֵן לָךְ׳. זוֹ מִצְוָה שֶׁיֵּשׁ עָלֶיהָ הַבְטָחָה: חַיִּים אֲרֻכִּים בְּאֶרֶץ יִשְׂרָאֵל.

😮 ׳וַיַּעַמְדוּ מֵרָחֹק׳

׳וְכָל הָעָם רֹאִים אֶת הַקּוֹלֹת וְאֶת הַלַּפִּידִם׳ – הָעָם פּוֹחֵד וְעוֹמֵד מֵרָחוֹק. הֵם אוֹמְרִים לְמֹשֶׁה: ׳דַּבֵּר אַתָּה עִמָּנוּ וְנִשְׁמָעָה׳. מֹשֶׁה עוֹנֶה: ׳אַל תִּירָאוּ׳ – ה׳ בָּא כְּדֵי שֶׁתִּהְיֶה בָּכֶם יִרְאַת שָׁמַיִם, ׳לְבִלְתִּי תֶחֱטָאוּ׳.

יִרְאַת שָׁמַיִם שׁוֹמֶרֶת עַל הָאָדָם מֵחֵטְא.

🪨 מִזְבַּח אֲדָמָה

ה׳ אוֹמֵר: ׳מִזְבַּח אֲדָמָה תַּעֲשֶׂה לִּי׳, וּמַבְטִיחַ: ׳בְּכָל הַמָּקוֹם אֲשֶׁר אַזְכִּיר אֶת שְׁמִי אָבוֹא אֵלֶיךָ וּבֵרַכְתִּיךָ׳. וְאִם בּוֹנִים מִזְבַּח אֲבָנִים – לֹא חוֹתְכִים אוֹתָן בְּבַרְזֶל. רַשִׁ״י מְסַבֵּר: הַמִּזְבֵּחַ מַאֲרִיךְ חַיִּים, וְהַבַּרְזֶל – כְּלֵי מִלְחָמָה – מְקַצֵּר חַיִּים.

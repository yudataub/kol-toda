רָמָה ג׳ – מַעֲמִיקִים

📜 ׳מִי בָכֶם מִכָּל עַמּוֹ׳ – כּוֹרֶשׁ מַכְרִיז

הָרַב לִמֵּד: סֵפֶר עֶזְרָא מַתְחִיל בְּדִבְרֵי כּוֹרֶשׁ: ׳כֹּל מַמְלְכוֹת הָאָרֶץ נָתַן לִי ה׳ אֱלֹהֵי הַשָּׁמַיִם וְהוּא פָקַד עָלַי לִבְנוֹת לוֹ בַיִת בִּירוּשָׁלַיִם׳. הַמְפָרְשִׁים מְבָאֲרִים שֶׁזֶּה מִתּוֹךְ נְבוּאַת יְשַׁעְיָהוּ – כּוֹרֶשׁ הֵבִין שֶׁהַנְּבוּאָה מְדַבֶּרֶת עָלָיו. הוּא לֹא חִיֵּב אִישׁ: ׳מִי בָכֶם מִכָּל עַמּוֹ יְהִי אֱלֹהָיו עִמּוֹ וְיַעַל׳. אֲבָל עָלוּ רַק כְּאַרְבָּעִים וּשְׁנַיִם אֶלֶף. לָמָּה? הָיָה לָהֶם טוֹב בַּגּוֹלָה, וְהַמָּקוֹם הָיָה חָרֵב; וְגַם צִפּוּ לִגְאֻלָּה עִם נִסִּים גְּלוּיִים, וְלֹא לְמֶלֶךְ נָכְרִי שֶׁאוֹמֵר ׳מִי שֶׁרוֹצֶה – יַעֲלֶה׳.

הַגְּאֻלָּה שֶׁל כּוֹרֶשׁ לֹא הִיא הַגְּאֻלָּה שֶׁצִּפּוּ לָהּ, וְלָכֵן עָלוּ מְעַטִּים; וְדַוְקָא דֶּרֶךְ הַמְעַטִּים נִבְנָה בַּיִת שֵׁנִי.

✉️ ׳כּוֹרֶשׁ הַשֵּׁנִי׳ – הַצְהָרַת בַּלְפוּר

אַלְפַּיִם שָׁנָה אַחֲרֵי כּוֹרֶשׁ, בְּמִלְחֶמֶת הָעוֹלָם הָרִאשׁוֹנָה, נִכְנְסָה טוּרְקִיָּה לַמִּלְחָמָה וּבְרִיטַנְיָה כָּבְשָׁה אֶת אֶרֶץ יִשְׂרָאֵל. בַּשֵּׁנִי בְּנוֹבֶמְבֶּר 1917 נִשְׁלַח מִכְתָּב: מֶמְשֶׁלֶת בְּרִיטַנְיָה ׳רוֹאָה בְּעַיִן יָפָה הֲקָמַת בַּיִת לְאֻמִּי לָעָם הַיְּהוּדִי בְּאֶרֶץ יִשְׂרָאֵל׳. הָרַב אָמַר שֶׁבְּרִיטַנְיָה וְהַהַצְהָרָה שֶׁלָּהּ הָיוּ כִּבְיָכוֹל ׳כּוֹרֶשׁ הַשֵּׁנִי׳. וְשׁוּב, כְּמוֹ אָז, עָלוּ מְעַטִּים: רַבִּים לֹא רָאוּ בַּדָּבָר קְרִיאָה אֱלֹהִית. וְהוּא הוֹסִיף: כְּפִי שֶׁכּוֹרֶשׁ קָרוּי ׳מְשִׁיחוֹ׳ – שָׁלִיחַ שֶׁל ה׳ – כָּךְ גַּם בַּדּוֹרוֹת הָאַחֲרוֹנִים שָׁלַח ה׳ שְׁלִיחִים מֵאֻמּוֹת הָעוֹלָם.

כְּמוֹ כּוֹרֶשׁ, גַּם בַּדּוֹר שֶׁלָּנוּ ה׳ שָׁלַח שְׁלִיחִים מֵאֻמּוֹת הָעוֹלָם – וְשׁוּב צָרִיךְ לָקוּם וְלַעֲלוֹת.

🤝 שְׁלוֹשָׁה אֲנָשִׁים בַּמָּקוֹם הַנָּכוֹן

הָרַב סִפֵּר: בְּדִיּוּק בַּתְּקוּפָה שֶׁבְּרִיטַנְיָה כָּבְשָׁה אֶת הָאָרֶץ עָלוּ לַשִּׁלְטוֹן שְׁלוֹשָׁה אֲנָשִׁים בְּדֶרֶךְ מַפְתִּיעָה. הַמֶּלֶךְ ג׳וֹרְג׳ הַחֲמִישִׁי לֹא הָיָה אָמוּר לִמְלֹךְ – אָחִיו הַבְּכוֹר נִפְטַר. רֹאשׁ הַמֶּמְשָׁלָה הִתְחַלֵּף כַּמָּה שָׁבוּעוֹת לִפְנֵי הַהַצְהָרָה, וְהַחָדָשׁ אָהַד אֶת הַיְּהוּדִים. וְשַׂר הַחוּץ, בַּלְפוּר, לָמַד תַּנַ״ךְ. שְׁלָשְׁתָּם הָיוּ בַּעֲלֵי אֱמוּנָה חֲזָקָה בַּתַּנַ״ךְ. ׳אֵין זֶה מַעֲשֵׂה צֵרוּף מִקְרִים׳, אָמַר הָרַב, ׳זוֹ יַד ה׳ שֶׁמְּסַדֶּרֶת אֶת הָאֲנָשִׁים הַנְּכוֹנִים בַּמְּקוֹמוֹת הַנְּכוֹנִים׳.

ה׳ מְסַדֵּר אֶת הָאֲנָשִׁים הַנְּכוֹנִים בַּמְּקוֹמוֹת הַנְּכוֹנִים – כְּמוֹ ׳הָאֹמֵר לְכוֹרֶשׁ רֹעִי׳.

רָמָה ג׳ – מַעֲמִיקִים

🪨 ׳גַּל עֵד׳ – מוֹצָאֵי שַׁבָּת וַיֵּצֵא

הָרַב הֵבִיא רֶמֶז כְּדֵי לִזְכֹּר מִסְפָּרִים: הַהַצְבָּעָה בְּאוּ״ם עַל הֲקָמַת מְדִינָה יְהוּדִית הָיְתָה בְּמוֹצָאֵי שַׁבָּת פָּרָשַׁת וַיֵּצֵא תש״ח. בְּסוֹף הַפָּרָשָׁה יַעֲקֹב קָרָא לַגַּל ׳גַּלְעֵד׳, וְלָבָן קָרָא לוֹ ׳יְגַר שָׂהֲדוּתָא׳. ׳גַּל׳ בְּגִימַטְרִיָּא שְׁלוֹשִׁים וְשָׁלוֹשׁ – שְׁלוֹשִׁים וְשָׁלוֹשׁ מְדִינוֹת הִצְבִּיעוּ בְּעַד, וְהָיוּ ׳עֵדִים׳ טוֹבִים לְיַעֲקֹב; ׳יְגַר׳ מַתְחִיל בָּאוֹתִיּוֹת יוֹ״ד גִּימֶ״ל, שְׁלוֹשׁ עֶשְׂרֵה – שְׁלוֹשׁ עֶשְׂרֵה מְדִינוֹת הִצְבִּיעוּ נֶגֶד. הָרַב הִדְגִּישׁ: גַּם לוּחַ הַזְּמַנִּים שֶׁל הַהִיסְטוֹרְיָה אֵינוֹ מִקְרִי.

הַהִיסְטוֹרְיָה לֹא מִקְרִית: בְּשַׁבָּת ׳וַיֵּצֵא׳ הִצְבִּיעוּ שְׁלוֹשִׁים וְשָׁלוֹשׁ בְּעַד וּשְׁלוֹשׁ עֶשְׂרֵה נֶגֶד – ׳גַּל עֵד׳.

🪡 פֶּתַח כְּחוּדוֹ שֶׁל מַחַט

הָרַב הֵבִיא אֶת דִּבְרֵי הָרַב יְהוּדָה אַלְקָלַי, שֶׁפִּרְסֵם אֶת סְפָרָיו עֲשָׂרוֹת שָׁנִים לִפְנֵי הַצְהָרַת בַּלְפוּר: גְּאֻלַּת יִשְׂרָאֵל תִּהְיֶה עַל יְדֵי מַלְכֵי הָאָרֶץ, כְּמוֹ שֶׁגְּאֻלַּת בָּבֶל הָיְתָה עַל יְדֵי כּוֹרֶשׁ. ׳יִשְׂרָאֵל יִפְתְּחוּ פֶּתַח כְּחוּדוֹ שֶׁל מַחַט – יַעֲשׂוּ הִשְׁתַּדְּלוּת – וְהַקָּדוֹשׁ בָּרוּךְ הוּא יִפְתַּח לָהֶם פֶּתַח כְּפִתְחוֹ שֶׁל אוּלָם׳. הָרַב הוֹסִיף: אֵין לְחַכּוֹת שֶׁהַכֹּל יֵעָשֶׂה מִלְּמַעְלָה בְּלִי מַעֲשֶׂה, וְאֵין לַחֲשֹׁב שֶׁהַכֹּל תָּלוּי בְּכֹחֲנוּ: ׳הַהִשְׁתַּדְּלוּת שֶׁל יִשְׂרָאֵל קְטַנָּה, אֲבָל הִיא מְעוֹרֶרֶת אֶת הַסִּיּוּעַ הַגָּדוֹל מִלְּמַעְלָה׳.

אָנוּ פּוֹתְחִים פֶּתַח כְּחוּדוֹ שֶׁל מַחַט, וְה׳ פּוֹתֵחַ פֶּתַח כְּפִתְחוֹ שֶׁל אוּלָם.

רָמָה ג׳ – מַעֲמִיקִים

🌅 ׳יוֹצֵר אוֹר וּבוֹרֵא חֹשֶׁךְ׳ – בִּבְרָכוֹת קְרִיאַת שְׁמַע

בְּכָל בֹּקֶר, בִּבְרָכוֹת קְרִיאַת שְׁמַע, אָנוּ אוֹמְרִים: ׳יוֹצֵר אוֹר וּבוֹרֵא חֹשֶׁךְ, עֹשֶׂה שָׁלוֹם וּבוֹרֵא אֶת הַכֹּל׳ – מִלִּים מִן הַפָּסוּק שֶׁלָּנוּ (מ״ה ז׳), בְּשִׁנּוּי אֶחָד: בִּמְקוֹם ׳וּבוֹרֵא רָע׳ אוֹמְרִים ׳וּבוֹרֵא אֶת הַכֹּל׳. רַשִׁ״י מְבָאֵר בַּפָּסוּק: ׳יוֹצֵר אוֹר – לַצַּדִּיקִים, וּבוֹרֵא חֹשֶׁךְ – לְבָבֶל׳. הָרַב שָׁאַל: אִם ה׳ ׳יוֹצֵר אוֹר׳, מַשְׁמָע שֶׁלִּפְנֵי כֵן הָיָה חֹשֶׁךְ; הָאוֹר בָּא לְסַלֵּק אֶת הַחֹשֶׁךְ – וְכָךְ אָנוּ מוֹדִים בַּבֹּקֶר עַל כָּל מַה שֶּׁמַּאִיר לָנוּ אֶת הַחַיִּים.

בְּכָל בֹּקֶר אוֹמְרִים ׳יוֹצֵר אוֹר... עֹשֶׂה שָׁלוֹם וּבוֹרֵא אֶת הַכֹּל׳ – ה׳ הוּא מְקוֹר הָאוֹר וְהַשָּׁלוֹם.

רָמָה ג׳ – מַעֲמִיקִים

💞 תְּשׁוּבָה מִתּוֹךְ הַכָּרַת הַטּוֹב

בְּסֵפֶר ׳אוֹרְחוֹת צַדִּיקִים׳, בְּשַׁעַר הַתְּשׁוּבָה, נֶאֱמַר: מִי שֶׁחוֹשֵׁב עַל כָּל הַטּוֹבוֹת שֶׁה׳ עָשָׂה לוֹ, שָׁב מִרְצוֹנוֹ אֶל ה׳ וּמְבַקֵּשׁ מְחִילָה. וְרַבֵּנוּ יוֹנָה בְּ׳שַׁעֲרֵי תְּשׁוּבָה׳ מְבָאֵר שֶׁהַתְּשׁוּבָה צְרִיכָה לָבוֹא מִתּוֹךְ הַכָּרַת הַטּוֹב: הָאָדָם אוֹמֵר לְעַצְמוֹ ׳אֵיךְ מָרִיתִי אֶת ה׳ יִתְבָּרַךְ, וְלֹא זָכַרְתִּי אֶת מִי שֶׁעָשָׂה אוֹתִי יֵשׁ מֵאַיִן וְעָשָׂה עִמִּי חֶסֶד כָּל כָּךְ׳. זוֹ תְּשׁוּבָה מֵאַהֲבָה, וּבָהּ – כְּפִי שֶׁאָמְרוּ חֲזַ״ל – הָעֲוֹנוֹת נֶהְפָּכִים לִזְכֻיּוֹת. וְזֶה בְּדִיּוּק ׳שׁוּבָה אֵלַי כִּי גְאַלְתִּיךָ׳: קֹדֶם הַטּוֹבָה, אַחַר כָּךְ הַחֲזָרָה.

תְּשׁוּבָה מֵאַהֲבָה: חוֹשְׁבִים עַל כָּל הַטּוֹבוֹת שֶׁה׳ עָשָׂה לָנוּ – וְשָׁבִים אֵלָיו.

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לַחֲזֹר אֶל ה׳ מֵאַהֲבָה, כִּי הוּא כְּבָר גָּאַל אוֹתִי 💛

רָמָה ג׳ – מַעֲמִיקִים

📖 מָה יְשַׁעְיָהוּ מְלַמֵּד אוֹתִי הַיּוֹם

יְשַׁעְיָהוּ מְלַמֵּד אוֹתִי: ה׳ כְּבָר מָחַק וּכְבָר גָּאַל – וְאֲנִי רַק צָרִיךְ לָשׁוּב; לְזַהוֹת אֶת הַהַזְדַּמְּנוּיוֹת שֶׁה׳ שׁוֹלֵחַ גַּם כְּשֶׁהֵן לֹא נִרְאוֹת כְּמוֹ שֶׁדִּמְיַנְתִּי; לַעֲשׂוֹת צַעַד קָטָן, וְה׳ יִפְתַּח אֶת הַפֶּתַח הַגָּדוֹל; וּלְהוֹדוֹת כָּל בֹּקֶר עַל הָאוֹר. כָּךְ אֲנִי עוֹנֶה לַקְּרִיאָה ׳שׁוּבָה אֵלַי׳.

שָׁבִים לַה׳ מִתּוֹךְ הַכָּרַת הַטּוֹב, פּוֹתְחִים פֶּתַח קָטָן וְלֹא מַחְמִיצִים אֶת הַהִזְדַּמְּנוּת.

רָמָה ג׳ – מַעֲמִיקִים

מָה לוֹקְחִים מִרָמָה ג׳?

פֶּרֶק מ״ד–מ״ה מִתְקַיֵּם לְנֶגֶד עֵינֵינוּ: אַחֲרֵי אַלְפַּיִם שָׁנָה, שׁוּב מֶלֶךְ נָכְרִי נוֹתֵן לַיְּהוּדִים רְשׁוּת לַחֲזֹר, וְעוֹד הַרְבֵּה מִסִּימָנֵי הַשְּׁגָחָה. וְכָל בֹּקֶר אָנוּ אוֹמְרִים אֶת הַפָּסוּק שֶׁל ׳יוֹצֵר אוֹר׳.

סוֹף רָמָה ג׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳מָחִיתִי כָעָב פְּשָׁעֶיךָ׳ – ה׳ קוֹדֵם מוֹחֵק, וְרַק אַחַר כָּךְ אוֹמֵר ׳שׁוּבָה אֵלַי׳. מָה זֶה אוֹמֵר עַל הַיַּחַס שֶׁלּוֹ אֵלֵינוּ כְּשֶׁאָנוּ רוֹצִים לָשׁוּב?
  • כּוֹרֶשׁ, מֶלֶךְ נָכְרִי, הָיָה שָׁלִיחַ שֶׁל ה׳. מָתַי הִרְגַּשְׁתִּי שֶׁה׳ שָׁלַח לִי עֶזְרָה מִמָּקוֹם לֹא צָפוּי?
  • הַבָּאִים לַגְּאֻלָּה שֶׁל כּוֹרֶשׁ הָיוּ מְעַטִּים, כִּי הַדֶּרֶךְ לֹא נִרְאֲתָה כְּמוֹ שֶׁצִּפּוּ. מָה אֲנִי יָכוֹל לַעֲשׂוֹת כְּדֵי לְזַהוֹת הִזְדַּמְּנוּת טוֹבָה גַּם כְּשֶׁהִיא לֹא נִרְאֵית כְּמוֹ שֶׁדִּמְיַנְתִּי?

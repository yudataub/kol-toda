הַבַּעַל פּוֹנֶה אֶל הַשּׁוֹמְעִים: ׳וְעַתָּה יוֹשֵׁב יְרוּשָׁלִַם וְאִישׁ יְהוּדָה שִׁפְטוּ נָא בֵּינִי וּבֵין כַּרְמִי׳. וְהוּא שׁוֹאֵל שְׁאֵלָה שֶׁאֵין עָלֶיהָ תְּשׁוּבָה: ׳מַה לַּעֲשׂוֹת עוֹד לְכַרְמִי וְלֹא עָשִׂיתִי בּוֹ מַדּוּעַ קִוֵּיתִי לַעֲשׂוֹת עֲנָבִים וַיַּעַשׂ בְּאֻשִׁים׳. הוּא עָשָׂה הַכֹּל. הַבְּעָיָה אֵינָהּ אֶצְלוֹ.

׳מַה לַּעֲשׂוֹת עוֹד לְכַרְמִי וְלֹא עָשִׂיתִי בּוֹ׳ – ה׳ עָשָׂה לָנוּ הַכֹּל.

🌵 ׳הָסֵר מְשׂוּכָּתוֹ׳

אִם הַכֶּרֶם לֹא נוֹתֵן פֵּרוֹת, הַבַּעַל יוֹדִיעַ מָה יַעֲשֶׂה: ׳הָסֵר מְשׂוּכָּתוֹ וְהָיָה לְבָעֵר פָּרֹץ גְּדֵרוֹ וְהָיָה לְמִרְמָס׳. הוּא יֶפֶסִיק לִשְׁמֹר עָלָיו, וְיַעֲלוּ בּוֹ ׳שָׁמִיר וָשָׁיִת׳, קֹצִים וַעֲשָׂבִים שׁוֹטִים. אָדָם שֶׁלֹּא מְטַפֵּחַ אֶת הַגִּנָּה שֶׁלּוֹ רוֹאֶה אוֹתָהּ מִתְפָּרֶצֶת בְּקוֹצִים.

כֶּרֶם שֶׁלֹּא מְטַפְּחִים – גָּדֵר נִפְרֶצֶת וְקוֹצִים צוֹמְחִים.

🏛️ ׳כִּי כֶרֶם ה׳ צְבָאוֹת בֵּית יִשְׂרָאֵל׳

וְאָז בָּא הַנִּמְשָׁל: ׳כִּי כֶרֶם ה׳ צְבָאוֹת בֵּית יִשְׂרָאֵל וְאִישׁ יְהוּדָה נְטַע שַׁעֲשׁוּעָיו׳ – אֲנַחְנוּ הַכֶּרֶם, וְאֲנַחְנוּ ׳נְטַע שַׁעֲשׁוּעָיו׳, הַגִּנָּה שֶׁה׳ הֲכִי אוֹהֵב. הוּא קִוָּה לְ׳מִשְׁפָּט׳ – לְצֶדֶק – וְקִבֵּל ׳מִשְׂפָּח׳, חֵטְא. הוּא קִוָּה לִ׳צְדָקָה׳, וְשָׁמַע ׳צְעָקָה׳ שֶׁל אֲנָשִׁים שֶׁנִּפְגְּעוּ.

הַנִּמְשָׁל: בֵּית יִשְׂרָאֵל הוּא ׳כֶרֶם ה׳ צְבָאוֹת׳ וּ׳נְטַע שַׁעֲשׁוּעָיו׳; הַפֵּרוֹת שֶׁה׳ רוֹצֶה: מִשְׁפָּט וּצְדָקָה.

🏘️ ׳הוֹי מַגִּיעֵי בַיִת בְּבַיִת׳

עַכְשָׁו הַנָּבִיא מַסְבִּיר אֵיזֶה פֵּרוֹת רָעִים גָּדְלוּ. הַפַּעַם הוּא קוֹרֵא ׳הוֹי׳ – קְרִיאַת צַעַר: ׳הוֹי מַגִּיעֵי בַיִת בְּבַיִת שָׂדֶה בְשָׂדֶה יַקְרִיבוּ עַד אֶפֶס מָקוֹם׳. בַּעֲלֵי כֹּחַ קוֹנִים עוֹד וָעוֹד בָּתִּים וּשְׂדוֹת עַד שֶׁלֹּא נִשְׁאָר מָקוֹם לַחַלָּשׁ. ׳וְהוּשַׁבְתֶּם לְבַדְּכֶם בְּקֶרֶב הָאָרֶץ׳ – הַאִם בֶּאֱמֶת רוֹצִים לָגוּר לְבַד?

חֲמְדָנוּת שֶׁלֹּא מַשְׁאִירָה מָקוֹם לַחַלָּשׁ – זֶה פְּרִי רַע; ה׳ רוֹצֶה שֶׁיִּהְיֶה מָקוֹם לְכֻלָּם.

🎻 ׳וְאֵת פֹּעַל ה׳ לֹא יַבִּיטוּ׳

עוֹד ׳הוֹי׳: ׳הוֹי מַשְׁכִּימֵי בַבֹּקֶר שֵׁכָר יִרְדֹּפוּ׳. הֵם יוֹשְׁבִים בַּמִּשְׁתֶּה, יֵשׁ ׳כִּנּוֹר וָנֶבֶל תֹּף וְחָלִיל׳, אֲבָל ׳וְאֵת פֹּעַל ה׳ לֹא יַבִּיטוּ וּמַעֲשֵׂה יָדָיו לֹא רָאוּ׳. הַנָּבִיא לֹא נֶגֶד שִׂמְחָה וּמוּזִיקָה, אֶלָּא נֶגֶד שִׂמְחָה שֶׁשּׁוֹכַחַת אֶת ה׳, וְלֹא מַבִּיטָה בַּיָּפֶה שֶׁבְּרָא.

שִׂמְחָה אֲמִתִּית מַבִּיטָה בְּ׳פֹּעַל ה׳׳ – בְּכָל מָה שֶׁהוּא בָּרָא וְעוֹשֶׂה.

הַ׳הוֹי׳ הַמְפֻרְסָם בְּיוֹתֵר: ׳הוֹי הָאֹמְרִים לָרַע טוֹב וְלַטּוֹב רָע שָׂמִים חֹשֶׁךְ לְאוֹר וְאוֹר לְחֹשֶׁךְ שָׂמִים מַר לְמָתוֹק וּמָתוֹק לְמָר׳. מִי שֶׁמַּחֲלִיף בֵּין טוֹב לָרַע דּוֹמֶה לְמִי שֶׁטּוֹעֶה בֵּין אוֹר לְחֹשֶׁךְ וּבֵין מַר לְמָתוֹק. וְאָז ׳הוֹי חֲכָמִים בְּעֵינֵיהֶם וְנֶגֶד פְּנֵיהֶם נְבֹנִים׳ – מִי שֶׁחוֹשֵׁב שֶׁהוּא יוֹדֵעַ הַכֹּל, לֹא לוֹמֵד.

לֹא מַחֲלִיפִים בֵּין טוֹב לָרַע, בֵּין אוֹר לְחֹשֶׁךְ – וְלֹא חוֹשְׁבִים שֶׁאֲנַחְנוּ ׳חֲכָמִים בְּעֵינֵינוּ׳.

מָה לוֹקְחִים מִרָמָה א׳?

יְשַׁעְיָהוּ לוֹקֵחַ שִׁיר, וּבוֹ מָשָׁל שֶׁכָּל יֶלֶד מֵבִין: כֶּרֶם שֶׁבַּעַל הַבַּיִת טָרַח עָלָיו כְּכָל הַיָּכוֹל. הַשְּׁאֵלָה הַגְּדוֹלָה: ׳מַה לַּעֲשׂוֹת עוֹד לְכַרְמִי וְלֹא עָשִׂיתִי בּוֹ׳ – ה׳ נָתַן לָנוּ הַכֹּל, בְּאַהֲבָה, וְהוּא מְחַכֶּה לַפֵּרוֹת שֶׁלָּנוּ.

🤔 חוֹשְׁבִים עַל זֶה

  • בַּעַל הַכֶּרֶם עָשָׂה הַכֹּל: גָּדַר, סִקֵּל, נָטַע, בָּנָה מִגְדָּל. מָה ה׳ עָשָׂה בִּשְׁבִילִי הַשָּׁנָה, וְאֵיזֶה ׳עֲנָבִים טוֹבִים׳ אֲנִי רוֹצֶה לְהָבִיא לוֹ בַּתְּמוּרָה?
  • ׳וַיְקַו לְמִשְׁפָּט וְהִנֵּה מִשְׂפָּח׳ – הוּא קִוָּה לְצֶדֶק וְקִבֵּל חֵטְא. מָתַי קִוּוּ מִמֶּנִּי לַעֲשׂוֹת מַשֶּׁהוּ טוֹב, וְאֵיךְ אֲנִי יָכוֹל לַעֲמֹד בְּזֶה?
  • ׳הוֹי הָאֹמְרִים לָרַע טוֹב וְלַטּוֹב רָע׳. אֵיךְ אֲנִי יוֹדֵעַ לְהַבְחִין בֵּין מָה שֶׁבֶּאֱמֶת טוֹב לְבֵין מָה שֶׁרַק נִרְאֶה נָעִים?

וְה׳ עוֹנֶה בְּמִשְׁפָּט הֲכִי אוֹהֵב בַּפֶּרֶק: ׳הֲתִשְׁכַּח אִשָּׁה עוּלָהּ מֵרַחֵם בֶּן בִּטְנָהּ גַּם אֵלֶּה תִשְׁכַּחְנָה וְאָנֹכִי לֹא אֶשְׁכָּחֵךְ׳. אֲפִלּוּ אִם הוֹרֶה שׁוֹכֵחַ – וְזֶה כִּמְעַט לֹא קוֹרֶה – ה׳ לֹא שׁוֹכֵחַ לְעוֹלָם. ׳הֵן עַל כַּפַּיִם חַקֹּתִיךְ חוֹמֹתַיִךְ נֶגְדִּי תָּמִיד׳ – הַחוֹמוֹת שֶׁל יְרוּשָׁלַיִם חֲקוּקוֹת עַל הַכַּפַּיִם שֶׁלּוֹ, תָּמִיד לְנֶגֶד עֵינָיו.

  • ׳הֵן עַל כַּפַּיִם חַקֹּתִיךְ חוֹמֹתַיִךְ נֶגְדִּי תָּמִיד׳. מָה אֲנִי רוֹצֶה שֶׁיִּהְיֶה ׳חָקוּק עַל הַכַּף׳ שֶׁלִּי, כְּלוֹמַר תָּמִיד לְנֶגֶד עֵינַי?
  • יְשַׁעְיָהוּ אָמַר ׳לְרִיק יָגַעְתִּי׳ – וּבְכָל זֹאת לֹא הִפְסִיק. מָתַי הָיָה לִי נִדְמֶה שֶׁמַּאֲמָצַי לֹא עוֹזְרִים, וּמָה נָתַן לִי כֹּחַ לְהַמְשִׁיךְ?

🔑 רַשִׁ״י: ׳מַפְתֵּחַ בֵּית דָּוִד׳ – אֶלְיָקִים

רַשִׁ״י מְבָאֵר: ׳מַפְתֵּחַ בֵּית דָּוִד – תַּרְגּוּם: מַפְתֵּחַ בֵּית מַקְדְּשָׁא וְשׁוּלְטָן בֵּית דָּוִד׳, וְעַל ׳יָתֵד בְּמָקוֹם נֶאֱמָן׳: ׳לָשׁוֹן קִיּוּם – מָקוֹם שֶׁהוּא נֶאֱמָן לַנִּשְׁעָנִים עָלָיו כִּי לֹא יָמוּט׳. מְצוּדַת דָּוִד עַל ׳וְהָיָה לְאָב׳: ׳וְלֹא כְּמוֹ אַתָּה שֶׁרָצִיתָ לִמְסֹר הָעִיר בְּיַד הָאוֹיֵב׳. אֶלְיָקִים נָשָׂא בַּעֹל, וּמִשְׁפַּחְתּוֹ נִשְׁעֶנֶת עָלָיו.

רַשִׁ״י: ׳מַפְתֵּחַ בֵּית דָּוִד׳ וּ׳יָתֵד בְּמָקוֹם נֶאֱמָן׳ – מַנְהִיג שֶׁאֶפְשָׁר לִסְמֹךְ עָלָיו, ׳לְאָב׳ לַעִיר.

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מְגַלִּים: כְּשֶׁה׳ מֵבִיא גְּאֻלָּה, גַּם הָאֻמּוֹת מַכִּירוֹת אוֹתוֹ; וּמֵאֲחוֹרֵי ׳שֹׁמֵר מַה מִּלַּיְלָה׳ עוֹמֶדֶת תִּקְוָה וּתְשׁוּבָה. וּבְמַעֲשֶׂה שֶׁבְנָא וְאֶלְיָקִים – הַנֶּאֱמָן גּוֹבֵר.

וּתְקַעְתִּיו יָתֵד בְּמָקוֹם נֶאֱמָן

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳אָתָא בֹקֶר וְגַם לָיְלָה׳ – ה׳ מַבְטִיחַ בֹּקֶר, וְגַם אוֹמֵר שֶׁיָּבוֹא לַיְלָה, וְאָז ׳שֻׁבוּ אֵתָיוּ׳. מָה אֲנִי עוֹשֶׂה כְּשֶׁחוֹשֵׁךְ, וְאֵיךְ אֲנִי מְחַכֶּה לַבֹּקֶר?
  • שֶׁבְנָא רָצָה לִהְיוֹת חָשׁוּב וְנִכְשַׁל; אֶלְיָקִים הָיָה נֶאֱמָן. מָה הַהֶבְדֵּל בֵּין מִי שֶׁרוֹצֶה כָּבוֹד לְבֵין מִי שֶׁנֶּאֱמָן?
  • ׳אָכוֹל וְשָׁתוֹ כִּי מָחָר נָמוּת׳ – הָעָם בָּחַר בְּשִׂמְחָה שֶׁל הֶרֶף. מָה הַשִּׂמְחָה הָאֲמִתִּית שֶׁלִּי, זוֹ שֶׁמְּחַבֶּרֶת אוֹתִי לַה׳?

💪 הָרַב אַלְקָלַי – ׳לֹא הִתְיָאֵשׁ׳

הָרַב אַלְקָלַי כָּתַב עֲשָׂרוֹת שָׁנִים לִפְנֵי שֶׁקָּרָה, שֶׁתָּקוּם מְדִינָה לַיְּהוּדִים וְשֶׁבְּרִיטַנְיָה תִּהְיֶה הָרִאשׁוֹנָה לְסַיֵּעַ. אָז לֹא הֶאֱמִינוּ לוֹ וְלֹא הֵבִינוּ אוֹתוֹ. הָרַב לִמֵּד: אָדָם שֶׁרוֹאֶה אֶת הָאֱמֶת וְאֵין מְקַבְּלִים אֶת דְּבָרָיו עָלוּל לְהִתְיָאֵשׁ – אֲבָל הָרַב אַלְקָלַי לֹא הִתְיָאֵשׁ. הוּא הִמְשִׁיךְ לִכְתֹּב וְלִפְעֹל עַד יוֹמוֹ הָאַחֲרוֹן, אַף שֶׁלֹּא רָאָה תּוֹצָאוֹת. וְזֶה בְּדִיּוּק יְשַׁעְיָהוּ: ׳לְרִיק יָגַעְתִּי... אָכֵן מִשְׁפָּטִי אֶת ה׳ וּפְעֻלָּתִי אֶת אֱלֹהָי׳.

עוֹשִׂים אֶת הַמַּאֲמָץ וְלֹא מִתְיָאֲשִׁים, גַּם כְּשֶׁלֹּא רוֹאִים תּוֹצָאוֹת – ׳פְּעֻלָּתִי אֶת אֱלֹהָי׳.

6
📖

📜 הַפָּסוּקוַאֲנִי אָמַרְתִּי לְרִיק יָגַעְתִּי לְתֹהוּ וְהֶבֶל כֹּחִי כִלֵּיתִי אָכֵן מִשְׁפָּטִי אֶת־ה׳ וּפְעֻלָּתִי אֶת־אֱלֹהָי

💛 אַהֲבַת הַשֵּׁם: אֲנִי חָקוּק בְּלֵב ה׳, וְהוּא חָקוּק בְּלִבִּי 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק מ״ט, פְּסוּקִים ד׳, ט״ו–ט״ז

רָמָה ג׳ – מַעֲמִיקִים

📖 מָה יְשַׁעְיָהוּ מְלַמֵּד אוֹתִי הַיּוֹם

יְשַׁעְיָהוּ מְלַמֵּד אוֹתִי: כְּשֶׁנִּדְמֶה לִי שֶׁה׳ שָׁכַח – לִזְכֹּר ׳וְאָנֹכִי לֹא אֶשְׁכָּחֵךְ׳; כְּשֶׁנִּדְמֶה לִי שֶׁ׳לְרִיק יָגַעְתִּי׳ – לְהַמְשִׁיךְ, כִּי ׳פְּעֻלָּתִי אֶת אֱלֹהָי׳; וּכְשֶׁאֲנִי רוֹאֶה אֶת הַנְּבוּאָה מִתְקַיֶּמֶת – לְהוֹדוֹת. וְלִהְיוֹת אֶחָד מִן הַבָּנִים הַשָּׁבִים הַבַּיְתָה.

ה׳ לֹא שׁוֹכֵחַ, אֲנִי לֹא מִתְיָאֵשׁ, וְאֲנִי מוֹדֶה עַל הַנְּבוּאוֹת שֶׁמִּתְקַיְּמוֹת.

7
💡

רָמָה ג׳ – מַעֲמִיקִים

מָה לוֹקְחִים מִרָמָה ג׳?

הַנְּבוּאָה שֶׁל פֶּרֶק מ״ט מִתְקַיֶּמֶת לְנֶגֶד עֵינֵינוּ: הָעַמִּים מְסַיְּעִים, הָאָרֶץ נוֹתֶנֶת פֵּרוֹתֶיהָ, הַמָּקוֹם הוֹלֵךְ וּמִתְמַלֵּא. וְכָל זֶה בָּא מִפִּי ה׳, שֶׁאָמַר: ׳לֹא אֶשְׁכָּחֵךְ׳.

❤️
וְאָנֹכִי לֹא אֶשְׁכָּחֵךְ הֵן עַל כַּפַּיִם חַקֹּתִיךְ חוֹמֹתַיִךְ נֶגְדִּי תָּמִיד

סוֹף רָמָה ג׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳וְאָנֹכִי לֹא אֶשְׁכָּחֵךְ׳ – ה׳ אוֹמֵר שֶׁהוּא לֹא שׁוֹכֵחַ. מָתַי הִרְגַּשְׁתִּי שֶׁשָּׁכְחוּ אוֹתִי, וְאֵיךְ אֲנִי יָכוֹל לִזְכֹּר שֶׁה׳ זוֹכֵר?
  • ׳הֵן עַל כַּפַּיִם חַקֹּתִיךְ חוֹמֹתַיִךְ נֶגְדִּי תָּמִיד׳. מָה אֲנִי רוֹצֶה שֶׁיִּהְיֶה ׳חָקוּק עַל הַכַּף׳ שֶׁלִּי, כְּלוֹמַר תָּמִיד לְנֶגֶד עֵינַי?
  • יְשַׁעְיָהוּ אָמַר ׳לְרִיק יָגַעְתִּי׳ – וּבְכָל זֹאת לֹא הִפְסִיק. מָתַי הָיָה לִי נִדְמֶה שֶׁמַּאֲמָצַי לֹא עוֹזְרִים, וּמָה נָתַן לִי כֹּחַ לְהַמְשִׁיךְ?

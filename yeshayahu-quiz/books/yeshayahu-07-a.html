רָמָה א׳ – עוֹלָם שֶׁל שָׁלוֹם וְדֵעָה

🌱 ׳וְיָצָא חֹטֶר מִגֵּזַע יִשָׁי׳

אַחֲרֵי הַפֶּרֶק שֶׁבּוֹ ה׳ מַפִּיל אֶת אַשּׁוּר, בָּאָה נְבוּאָה חֲדָשָׁה: ׳וְיָצָא חֹטֶר מִגֵּזַע יִשָׁי וְנֵצֶר מִשָּׁרָשָׁיו יִפְרֶה׳. גֶּזַע יָשָׁן שֶׁנִּרְאֶה כְּאִלּוּ חָתוּךְ, וְיוֹצֵא מִמֶּנּוּ עָנָף חַי. זֶה הַמֶּלֶךְ מִבֵּית דָּוִד, הוּא מֶלֶךְ הַמָּשִׁיחַ.

מִגֶּזַע יִשַׁי, אֲבִי דָּוִד, יִצְמַח מֶלֶךְ חָדָשׁ – הַמָּשִׁיחַ.

1
🧠

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לַמַּנְהִיג הַטּוֹב רוּחַ חָכְמָה וְדַעַת, כְּדֵי לְהוֹבִיל אוֹתָנוּ בְּצֶדֶק 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק י״א, פְּסוּקִים ב׳–ג׳

רָמָה א׳ – עוֹלָם שֶׁל שָׁלוֹם וְדֵעָה

🧠 ׳וְנָחָה עָלָיו רוּחַ ה׳׳

עַל הַמֶּלֶךְ הַזֶּה ׳וְנָחָה עָלָיו רוּחַ ה׳ רוּחַ חָכְמָה וּבִינָה רוּחַ עֵצָה וּגְבוּרָה רוּחַ דַּעַת וְיִרְאַת ה׳׳. הוּא אֵינוֹ שׁוֹפֵט לְפִי מָה שֶׁהוּא רוֹאֶה: ׳וְלֹא לְמַרְאֵה עֵינָיו יִשְׁפּוֹט וְלֹא לְמִשְׁמַע אָזְנָיו יוֹכִיחַ׳. הוּא מַרְגִּישׁ בְּלִבּוֹ מִי טוֹב וּמִי לֹא, ׳וַהֲרִיחוֹ בְּיִרְאַת ה׳׳ – כְּמוֹ שֶׁמַּרְגִּישִׁים בְּרֵיחַ.

הַמֶּלֶךְ הַצַּדִּיק שׁוֹפֵט בְּחָכְמָה וּבְיִרְאַת ה׳, ׳וְלֹא לְמַרְאֵה עֵינָיו׳.

2
⚖️

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אֶת הַדַּלִּים וְעַנְוֵי הָאָרֶץ, וְדּוֹאֵג שֶׁיִּשְׁפְּטוּ אוֹתָם בְּצֶדֶק 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק י״א, פְּסוּקִים ד׳–ה׳

רָמָה א׳ – עוֹלָם שֶׁל שָׁלוֹם וְדֵעָה

⚖️ ׳וְשָׁפַט בְּצֶדֶק דַּלִּים׳

הַמֶּלֶךְ הַזֶּה דּוֹאֵג לַחַלָּשִׁים: ׳וְשָׁפַט בְּצֶדֶק דַּלִּים וְהוֹכִיחַ בְּמִישׁוֹר לְעַנְוֵי אָרֶץ׳. וְהַבֶּגֶד שֶׁלּוֹ הוּא כֹּשֶׁר הַלֵּב: ׳וְהָיָה צֶדֶק אֵזוֹר מָתְנָיו וְהָאֱמוּנָה אֵזוֹר חֲלָצָיו׳ – כְּמוֹ חֲגוֹרָה שֶׁקּוֹשֶׁרֶת אֶת הַבְּגָדִים, כָּךְ צֶדֶק וֶאֱמוּנָה קוֹשְׁרִים אֶת הַמֶּלֶךְ.

הַמֶּלֶךְ הַצַּדִּיק שׁוֹפֵט דַּלִּים בְּצֶדֶק; ׳צֶדֶק׳ וְ׳אֱמוּנָה׳ הֵם הַחֲגוֹרָה שֶׁלּוֹ.

3
🐺

💛 אַהֲבַת הַשֵּׁם: ה׳ מְחַכֶּה לְיוֹם שֶׁבּוֹ כֻּלָּם יַחַד בְּשָׁלוֹם, וַאֲנִי יָכוֹל לְהַתְחִיל בּוֹ כְּבָר עַכְשָׁו 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק י״א, פְּסוּקִים ו׳–ז׳

רָמָה א׳ – עוֹלָם שֶׁל שָׁלוֹם וְדֵעָה

🐺 ׳וְגָר זְאֵב עִם כֶּבֶשׂ׳

וְהַתְּמוּנָה הַמְפֻרְסֶמֶת: ׳וְגָר זְאֵב עִם כֶּבֶשׂ וְנָמֵר עִם גְּדִי יִרְבָּץ וְעֵגֶל וּכְפִיר וּמְרִיא יַחְדָּו וְנַעַר קָטֹן נֹהֵג בָּם׳. כֻּלָּם יַחַד – ׳וּפָרָה וָדֹב תִּרְעֶינָה יַחְדָּו... וְאַרְיֵה כַּבָּקָר יֹאכַל תֶּבֶן׳. הַחַיּוֹת הַחֲזָקוֹת לֹא מְפַחֲדוֹת מִן הַקְּטַנּוֹת, וְנַעַר קָטָן מַנְהִיג אוֹתָן.

בִּימֵי הַמָּשִׁיחַ יִהְיֶה שָׁלוֹם בֵּין חֲזָקִים וַחֲלָשִׁים: ׳וְגָר זְאֵב עִם כֶּבֶשׂ׳.

4
🌊

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁכֻּלָּם יַכִּירוּ אֶת ה׳, לֹא יִהְיֶה צֹרֶךְ לְהַזִּיק – כִּי מַכִּירִים אֶת הָאַהֲבָה שֶׁלּוֹ 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק י״א, פָּסוּק ט׳

רָמָה א׳ – עוֹלָם שֶׁל שָׁלוֹם וְדֵעָה

🌊 ׳כִּי מָלְאָה הָאָרֶץ דֵּעָה אֶת ה׳׳

וְהַסִּבָּה לְכָל הַשָּׁלוֹם: ׳לֹא יָרֵעוּ וְלֹא יַשְׁחִיתוּ בְּכָל הַר קָדְשִׁי כִּי מָלְאָה הָאָרֶץ דֵּעָה אֶת ה׳ כַּמַּיִם לַיָּם מְכַסִּים׳. כְּמוֹ שֶׁהַמַּיִם מְכַסִּים אֶת הַיָּם עַד שֶׁלֹּא רוֹאִים אֶת הַקַּרְקַע, כָּךְ הָעוֹלָם יִתְמַלֵּא בְּהַכָּרַת ה׳.

הַשָּׁלוֹם בָּא מִן הַדֵּעָה: ׳כִּי מָלְאָה הָאָרֶץ דֵּעָה אֶת ה׳ כַּמַּיִם לַיָּם מְכַסִּים׳.

5
🚩

💛 אַהֲבַת הַשֵּׁם: ה׳ לֹא שׁוֹכֵחַ אַף אֶחָד – הוּא מְקַבֵּץ גַּם אֶת הַנִּדָּחִים 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק י״א, פְּסוּקִים י׳–י״ב

רָמָה א׳ – עוֹלָם שֶׁל שָׁלוֹם וְדֵעָה

🚩 ׳לְנֵס עַמִּים אֵלָיו גּוֹיִם יִדְרֹשׁוּ׳

הַמֶּלֶךְ הַזֶּה יִהְיֶה כְּמוֹ דֶּגֶל בָּרוּם: ׳שֹׁרֶשׁ יִשַׁי אֲשֶׁר עֹמֵד לְנֵס עַמִּים אֵלָיו גּוֹיִם יִדְרֹשׁוּ וְהָיְתָה מְנֻחָתוֹ כָּבוֹד׳. וְאָז ה׳ יְקַבֵּץ אֶת עַמּוֹ: ׳וְהָיָה בַּיּוֹם הַהוּא יוֹסִיף ה׳ שֵׁנִית יָדוֹ לִקְנוֹת אֶת שְׁאָר עַמּוֹ... וְאָסַף נִדְחֵי יִשְׂרָאֵל וּנְפֻצוֹת יְהוּדָה יְקַבֵּץ מֵאַרְבַּע כַּנְפוֹת הָאָרֶץ׳.

ה׳ יְקַבֵּץ אֶת עַמּוֹ מֵאַרְבַּע כַּנְפוֹת הָאָרֶץ, וְהָעַמִּים יִדְרְשׁוּ אֶת הַמֶּלֶךְ.

6
🙏

💛 אַהֲבַת הַשֵּׁם: ה׳ הוּא הָעוֹז שֶׁלִּי וְהַשִּׁיר שֶׁלִּי, וּבוֹ אֲנִי בּוֹטֵחַ 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק י״ב, פְּסוּקִים א׳–ג׳

רָמָה א׳ – עוֹלָם שֶׁל שָׁלוֹם וְדֵעָה

🙏 ׳אוֹדְךָ ה׳ כִּי אָנַפְתָּ בִּי׳

וּבַפֶּרֶק הַבָּא יֵשׁ שִׁיר הוֹדָיָה: ׳וְאָמַרְתָּ בַּיּוֹם הַהוּא אוֹדְךָ ה׳ כִּי אָנַפְתָּ בִּי יָשֹׁב אַפְּךָ וּתְנַחֲמֵנִי׳. הֲרֵי הָאָדָם מוֹדֶה גַּם עַל הַקֹּשִׁי, כִּי רָאָה שֶׁאַחֲרָיו בָּא נֶחָמָה. ׳הִנֵּה אֵל יְשׁוּעָתִי אֶבְטַח וְלֹא אֶפְחָד כִּי עָזִּי וְזִמְרָת יָהּ ה׳ וַיְהִי לִי לִישׁוּעָה׳. וְ׳וּשְׁאַבְתֶּם מַיִם בְּשָׂשׂוֹן מִמַּעַיְנֵי הַיְשׁוּעָה׳.

׳הִנֵּה אֵל יְשׁוּעָתִי אֶבְטַח וְלֹא אֶפְחָד׳ – בִּטָּחוֹן וְהוֹדָיָה; וְ׳וּשְׁאַבְתֶּם מַיִם בְּשָׂשׂוֹן׳.

🎤 ׳הוֹדוּ לַה׳ קִרְאוּ בִשְׁמוֹ׳

הַשִּׁיר מִתְפַּשֵּׁט לְכָל הָעוֹלָם: ׳וַאֲמַרְתֶּם בַּיּוֹם הַהוּא הוֹדוּ לַה׳ קִרְאוּ בִשְׁמוֹ הוֹדִיעוּ בָעַמִּים עֲלִילֹתָיו הַזְכִּירוּ כִּי נִשְׂגָּב שְׁמוֹ׳. וְהוּא נִגְמָר בְּקָרִיאַת שִׂמְחָה: ׳צַהֲלִי וָרֹנִּי יוֹשֶׁבֶת צִיּוֹן כִּי גָדוֹל בְּקִרְבֵּךְ קְדוֹשׁ יִשְׂרָאֵל׳. אֶת ה׳ מְפַרְסְמִים בְּשִׁירָה וּבְשִׂמְחָה.

מוֹדִים לַה׳, קוֹרְאִים בִּשְׁמוֹ וּמוֹדִיעִים לָעַמִּים: ׳כִּי גָדוֹל בְּקִרְבֵּךְ קְדוֹשׁ יִשְׂרָאֵל׳.

רָמָה א׳ – עוֹלָם שֶׁל שָׁלוֹם וְדֵעָה

מָה לוֹקְחִים מִרָמָה א׳?

יְשַׁעְיָהוּ רוֹאֶה עֵץ חָדָשׁ שֶׁצּוֹמֵחַ מִגֶּזַע יָשָׁן, וְעוֹלָם שֶׁבּוֹ הַזְּאֵב וְהַכֶּבֶשׂ שׁוֹכְנִים יַחַד. וְאָז הוּא שָׁר שִׁיר שֶׁאֲנַחְנוּ שָׁרִים בְּכָל מוֹצָאֵי שַׁבָּת: ׳הִנֵּה אֵל יְשׁוּעָתִי אֶבְטַח וְלֹא אֶפְחָד׳. ה׳ אוֹהֵב אוֹתָנוּ וּמֵכִין לָנוּ עוֹלָם טוֹב.

  • בְּהַבְדָּלָה אֲנִי אוֹמֵר ׳הִנֵּה אֵל יְשׁוּעָתִי אֶבְטַח וְלֹא אֶפְחָד׳. מָה גָּרַם לִי הַשָּׁבוּעַ לְהַרְגִּישׁ בִּטָּחוֹן, וְעַל מָה אֲנִי רוֹצֶה לְהוֹדוֹת לְה׳?
  • ׳וּשְׁאַבְתֶּם מַיִם בְּשָׂשׂוֹן׳ – מָתַי הָיְתָה לִי שִׂמְחָה אֲמִתִּית שֶׁל לִמּוּד אוֹ שֶׁל חֶסֶד?

יְרוּשָׁלַיִם קָמָה לָאוֹר, הָעַמִּים בָּאִים אֶל אוֹרָהּ, וְהַבָּנִים חוֹזְרִים הַבַּיְתָה – וְה׳ עַצְמוֹ הוּא הָאוֹר שֶׁלָּהּ לְעוֹלָם

📜 הַפָּסוּקקוּמִי אוֹרִי כִּי בָא אוֹרֵךְ וּכְבוֹד ה׳ עָלַיִךְ זָרָח כִּי־הִנֵּה הַחֹשֶׁךְ יְכַסֶּה־אֶרֶץ וַעֲרָפֶל לְאֻמִּים וְעָלַיִךְ יִזְרַח ה׳ וּכְבוֹדוֹ עָלַיִךְ יֵרָאֶה

רָמָה ג׳ – נְבוּאָה שֶׁמִּתְקַיֶּמֶת

🕎 הַפְטָרָה, ׳לְכָה דוֹדִי׳ וּמְנוֹרָה – ׳אוֹר לַגּוֹיִם׳

הַנְּבוּאָה ׳קוּמִי אוֹרִי׳ הִיא הַפְטָרַת פָּרָשַׁת כִּי תָבוֹא, וְשָׁרִים אוֹתָהּ גַּם בְּ׳לְכָה דוֹדִי׳: ׳הִתְעוֹרְרִי הִתְעוֹרְרִי כִּי בָא אוֹרֵךְ קוּמִי אוֹרִי׳. הָרַב לָמַד עָלֶיהָ בְּשִׁעוּר ׳אוֹר לַגּוֹיִם׳: הַנָּבִיא מַבְטִיחַ שֶׁיָּבוֹא יוֹם שֶׁבּוֹ הָעַמִּים יָבוֹאוּ לִלְמֹד מֵעַם יִשְׂרָאֵל אֵיךְ לִחְיוֹת. וְגַם בִּזְמַנֵּנוּ רַבִּים בָּעוֹלָם רוֹאִים בְּעַם יִשְׂרָאֵל עַם חָכָם, וְהַפָּסוּק מְבַשֵּׂר שֶׁזֶּה רַק הַהַתְחָלָה. כְּשֶׁחִפְּשׂוּ סֵמֶל לַמְּדִינָה, בָּחֲרוּ בַּמְּנוֹרָה – כִּי עַם יִשְׂרָאֵל נוֹעַד לִהְיוֹת ׳אוֹר לַגּוֹיִם׳, וְהָאוֹר שֶׁלּוֹ, שֶׁל מְנוֹרַת בֵּית הַמִּקְדָּשׁ, הֵפִיץ אֶת הָאוֹר הַזֶּה בָּעוֹלָם.

הַפְטָרָה שֶׁשָּׁרִים בְּ׳לְכָה דוֹדִי׳: עַם יִשְׂרָאֵל נוֹעַד לִהְיוֹת אוֹר לָעוֹלָם, וּמְנוֹרָה הִיא הַסֵּמֶל.

💛 אַהֲבַת הַשֵּׁם: ה׳ מַחֲזִיר אוֹתִי הַבַּיְתָה, וְאֶת הַגַּעְגּוּעִים שֶׁלִּי הוּא כְּבָר שָׂם בְּתוֹכִי 💛

רָמָה ג׳ – נְבוּאָה שֶׁמִּתְקַיֶּמֶת

☁️ ׳כָּעָב תְּעוּפֶינָה וְכַיּוֹנִים׳ – סִפּוּר הָרַב קוּק

פַּעַם בִּקְּרָה אֵצֶל הָרַב קוּק קְבוּצַת עוֹלִים מִגֶּרְמָנְיָה, שֶׁנֶּעֶקְרוּ מִמְּקוֹמָם בְּחֶזְקַת הַיָּד וְהִשְׁאִירוּ רְכוּשׁ מֵאַחֲרֵיהֶם, וְנַפְשָׁם הָיְתָה עֲגוּמָה. הָרַב הִרְגִּישׁ אֶת הַקֹּשִׁי וְאָמַר לָהֶם: ׳יְשַׁעְיָהוּ הַנָּבִיא אָמַר: מִי אֵלֶּה כָּעָב תְּעוּפֶינָה וְכַיּוֹנִים אֶל אֲרֻבֹּתֵיהֶם. מַה בֵּין תְּעוּפַת הָעָבִים לִתְעוּפַת הַיּוֹנִים? הָעָבִים אֵינָם נִשָּׂאִים מִרְצוֹנָם – רוּחַ הַסְּעָרָה דּוֹחֶפֶת אוֹתָם. הַיּוֹנִים חוֹזְרוֹת מִכֹּחַ הַגַּעְגּוּעִים וְהַחִבָּה לְבֵיתָן׳. הַנְּבוּאָה מְנַחֶמֶת: יִהְיוּ כָּאֵלֶּה שֶׁיָּשׁוּבוּ לָאָרֶץ מֵרָצוֹן פְּנִימִי, כַּיּוֹנִים, וְיִהְיוּ כָּאֵלֶּה שֶׁרוּחוֹת סוֹעֲרוֹת תַּעֲקֹרְנָה אוֹתָם וְתִדְחֹפְנָה אוֹתָם לַעֲלוֹת, כְּעָבִים. וְהָרַב בֵּרַךְ אוֹתָם: ׳הַקָּדוֹשׁ בָּרוּךְ הוּא יַעֲזֹר לָכֶם שֶׁגַּם אִם תְּחִלַּת עֲלִיָּתְכֶם הָיְתָה בְּ״כָּעָב תְּעוּפֶינָה״, תִּגְמְרוּ אוֹתָהּ מִתּוֹךְ הַרְגָּשָׁה שֶׁל ״כַּיּוֹנִים אֶל אֲרֻבֹּתֵיהֶם״׳.

גַּם מִי שֶׁהוּבָא לָאָרֶץ ׳כָּעָב׳ בְּרוּחַ סְעָרָה, יָכוֹל לְהִתְיַשֵּׁב בָּהּ ׳כַּיּוֹנָה׳ מֵאַהֲבָה.

רָמָה ג׳ – נְבוּאָה שֶׁמִּתְקַיֶּמֶת

✨ ׳וְעַמֵּךְ כֻּלָּם צַדִּיקִים׳ – לִפְנֵי פִּרְקֵי אָבוֹת

אֶת הַפָּסוּק ׳וְעַמֵּךְ כֻּלָּם צַדִּיקִים לְעוֹלָם יִירְשׁוּ אָרֶץ נֵצֶר מַטָּעַי מַעֲשֵׂה יָדַי לְהִתְפָּאֵר׳ אֲנַחְנוּ אוֹמְרִים לִפְנֵי כָּל פֶּרֶק בְּפִרְקֵי אָבוֹת, בַּמִּשְׁנָה בְּסַנְהֶדְרִין: ׳כָּל יִשְׂרָאֵל יֵשׁ לָהֶם חֵלֶק לָעוֹלָם הַבָּא׳. הָרַב שָׁאַל: הַפָּסוּק מְדַבֵּר עַל יְרֻשַּׁת הָאָרֶץ, אֲבָל הַמִּשְׁנָה לוֹמֶדֶת מִמֶּנּוּ חֵלֶק בָּעוֹלָם הַבָּא – אֵיךְ? וְהוּא עָנָה עִם הָאִבְּן עֶזְרָא: ׳מִי שֶׁיֵּשׁ לוֹ חֵלֶק בְּאֶרֶץ יִשְׂרָאֵל, מִמֵּילָא יֵשׁ לוֹ חֵלֶק בָּעוֹלָם הַבָּא׳. הַמִּשְׁנָה מְחַבֶּרֶת בֵּין שְׁנֵי הַדְּבָרִים כִּי הֵם דָּבָר אֶחָד.

׳וְעַמֵּךְ כֻּלָּם צַדִּיקִים לְעוֹלָם יִירְשׁוּ אָרֶץ׳ – הַחֵלֶק בָּעוֹלָם הַבָּא עוֹבֵר דֶּרֶךְ יְרֻשַּׁת הָאָרֶץ.

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי הִזְדַּמְּנוּת לְזָרֵז אֶת הַגְּאֻלָּה בִּידַיִם שֶׁלִּי 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ס׳, פָּסוּק כ״ב

רָמָה ג׳ – נְבוּאָה שֶׁמִּתְקַיֶּמֶת

⏳ ׳בְּעִתָּהּ אֲחִישֶׁנָּה׳ – הַדֶּרֶךְ הַשְּׁלִישִׁית שֶׁל הַגְּרָ״א

חֲזַ״ל דּוֹרְשִׁים: ׳זָכוּ – אֲחִישֶׁנָּה, לֹא זָכוּ – בְּעִתָּהּ׳. כְּלוֹמַר: אוֹ שֶׁהַגְּאֻלָּה תָּבוֹא מְהֵרָה בִּזְכוּת מַעֲשִׂים טוֹבִים, אוֹ שֶׁתָּבוֹא בִּזְמַנָּהּ הַקָּבוּעַ גַּם בְּלִי זְכוּת. הָרַב לָמַד שֶׁלַּגְּר״א הָיָה חִדּוּשׁ: זֶה פְּשׁוּטוֹ שֶׁל הַפָּסוּק – ׳בְּעִתָּהּ אֲחִישֶׁנָּה׳ יַחַד. גַּם כְּשֶׁמַּגִּיעַ הַזְּמַן ׳בְּעִתָּהּ׳, אֶפְשָׁר שֶׁהַגְּאֻלָּה תָּבוֹא לְאַט לְאַט וְאֶפְשָׁר שֶׁתָּבוֹא מַהֵר. וְאֵיךְ זוֹכִים שֶׁגַּם ׳בְּעִתָּהּ׳ יִהְיֶה ׳אֲחִישֶׁנָּה׳? עַל יְדֵי מַעֲשִׂים שֶׁל יִשּׁוּב אֶרֶץ יִשְׂרָאֵל. לָכֵן תַּלְמִידֵי הַגְּרָ״א בָּנוּ, נָטְעוּ וְהֵקִימוּ שְׁכוּנוֹת, וְרָאוּ בַּבְּנִיָּה וּבַנְּטִיעָה מַעֲשֶׂה רוּחָנִי שֶׁמְּזָרֵז אֶת הַגְּאֻלָּה. אֲפִלּוּ הַפַּרְדֵּס הַקָּטָן לְיַד מְעָרַת שִׁמְעוֹן הַצַּדִּיק הָיָה נִסָּיוֹן לְקַיֵּם בְּפֹעַל אֶת ׳אֲחִישֶׁנָּה׳.

לְפִי הַגְּרָ״א, גַּם כְּשֶׁהַגְּאֻלָּה בָּאָה ׳בְּעִתָּהּ׳, יִשּׁוּב הָאָרֶץ מְזָרֵז אוֹתָהּ ׳אֲחִישֶׁנָּה׳.

👪 ׳הַקָּטֹן יִהְיֶה לָאֶלֶף׳ – תְּמוּנָה בַּכּוֹתֶל

הָרַב סִיֵּם שִׁעוּר ׳נְבוּאוֹת לְנֶגֶד עֵינֵינוּ׳ בִּתְמוּנָה: אִשָּׁה שֶׁעָבְרָה אֶת הַשּׁוֹאָה, בַּת מֵאָה וְאַרְבַּע, בִּקְשָׁה מִכָּל צֶאֱצָאֶיהָ לְהַגִּיעַ לַכּוֹתֶל בְּיוֹם הַשּׁוֹאָה לְצִלּוּם מְשֻׁתָּף. הַתְּמוּנָה שֶׁהִתְקַבְּלָה מַרְאָה צִבּוּר שָׁלֵם, מֵאוֹת אֲנָשִׁים, כֻּלָּם יוֹצְאֵי חֲלָצֶיהָ שֶׁל אִשָּׁה אַחַת. עַל זֶה נֶאֱמַר: ׳הַקָּטֹן יִהְיֶה לָאֶלֶף וְהַצָּעִיר לְגוֹי עָצוּם אֲנִי ה׳ בְּעִתָּהּ אֲחִישֶׁנָּה׳. הָרַב סִיֵּם: יֵשׁ עוֹד נְבוּאוֹת שֶׁלֹּא הִתְקַיְּמוּ. הַדּוֹרוֹת שֶׁלְּפָנֵינוּ קָרְאוּ אֶת הַפְּסוּקִים וְחָלְמוּ, וַאֲנַחְנוּ זָכִינוּ לִרְאוֹת חֵלֶק מֵהֶם – וְהָאַחְרָיוּת שֶׁלָּנוּ הִיא לָהוֹדוֹת, לִשְׂמֹחַ וּלְהִתְפַּלֵּל לִרְאוֹת גַּם אֶת מַה שֶּׁעוֹד לֹא הִתְקַיֵּם.

׳הַקָּטֹן יִהְיֶה לָאֶלֶף׳ – אִשָּׁה אַחַת מִן הַשּׁוֹאָה וּמֵאוֹת צֶאֱצָאִים סְבִיבָהּ בַּכּוֹתֶל.

📜 הַפָּסוּקקוּמִי אוֹרִי כִּי בָא אוֹרֵךְ וּכְבוֹד ה׳ עָלַיִךְ זָרָח

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁה׳ יִהְיֶה הָאוֹר שֶׁלִּי – וְשֶׁאֶהְיֶה אוֹר לַאֲחֵרִים 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ס׳, פְּסוּקִים א׳, ח׳, כ״א–כ״ב

רָמָה ג׳ – נְבוּאָה שֶׁמִּתְקַיֶּמֶת

📖 מָה יְשַׁעְיָהוּ מְלַמֵּד אוֹתִי הַיּוֹם

הַנְּבוּאָה קוֹרֵאת לִי: ׳קוּם, אוֹר!׳. אֲנִי רוֹצֶה לִהְיוֹת אוֹר לַסְּבִיבָה שֶׁלִּי – תּוֹרָה, מִדּוֹת טוֹבוֹת וְשִׂמְחָה. לִרְאוֹת בְּכָל יְהוּדִי ׳נֵצֶר מַטָּעַי׳, לָשׁוּב הַבַּיְתָה בְּשִׂמְחָה וְלֹא רַק מֵאֹנֶס, וְלִזְכֹּר שֶׁגַּם אֲנִי יָכוֹל לְזָרֵז אֶת הַגְּאֻלָּה. וּלְהוֹדוֹת לַה׳ עַל מָה שֶׁכְּבָר הִתְקַיֵּם בִּימֵינוּ.

אֲנִי קָם לָאוֹר: אוֹר בַּלֵּב וְאוֹר לַסְּבִיבָה, וְתוֹדָה עַל מַה שֶּׁכְּבָר מִתְקַיֵּם.

רָמָה ג׳ – נְבוּאָה שֶׁמִּתְקַיֶּמֶת

מָה לוֹקְחִים מִרָמָה ג׳?

נְבוּאָה שֶׁקָּרָאנוּ בַּהַפְטָרָה וְשָׁרִים בְּ׳לְכָה דוֹדִי׳ מִתְקַיֶּמֶת לְנֶגֶד עֵינֵינוּ: הָעַמִּים רוֹאִים אֶת הָאוֹר, הַבָּנִים חוֹזְרִים לָאָרֶץ, וְהַקָּטָן נַעֲשֶׂה אֶלֶף. נוֹדֶה לַה׳ וְנִשְׂמַח.

וְהָיָה לָךְ ה׳ לְאוֹר עוֹלָם

סוֹף רָמָה ג׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳קוּמִי אוֹרִי כִּי בָא אוֹרֵךְ׳ – מָתַי הִרְגַּשְׁתִּי שֶׁהַלֵּב שֶׁלִּי ׳קָם לָאוֹר׳, וּמָה גָּרַם לָזֶה?
  • ׳מִי אֵלֶּה כָּעָב תְּעוּפֶינָה וְכַיּוֹנִים אֶל אֲרֻבֹּתֵיהֶם׳ – אֵיךְ אֲנִי חוֹזֵר הַבַּיְתָה: כְּמוֹ עָנָן שֶׁדּוֹחֲפִים אוֹתוֹ, אוֹ כְּמוֹ יוֹנָה שֶׁרוֹצָה בְּעַצְמָהּ?
  • ׳וְעַמֵּךְ כֻּלָּם צַדִּיקִים׳ – אֵיךְ אֲנִי יָכוֹל לִרְאוֹת אֶת הַטּוֹב שֶׁיֵּשׁ בְּכָל יְהוּדִי, גַּם בְּמִי שֶׁקָּשֶׁה לִי אִתּוֹ?

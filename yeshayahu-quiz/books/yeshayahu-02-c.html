רָמָה ג׳ – מַעֲמִיקִים

🏛️ ׳עֲתִידִין בָּתֵּי מִדְרָשׁוֹת... שֶׁיִּקָּבְעוּ בְּאֶרֶץ יִשְׂרָאֵל׳

בְּשִׂמְחַת תּוֹרָה, כְּשֶׁמּוֹצִיאִים אֶת סִפְרֵי הַתּוֹרָה, כֻּלָּם שָׁרִים ׳כִּי מִצִּיּוֹן תֵּצֵא תוֹרָה׳. וְהַגְּמָרָא (מְגִלָּה כ״ט ע״א) אוֹמֶרֶת: ׳עֲתִידִין בָּתֵּי כְנֵסִיּוֹת וּבָתֵּי מִדְרָשׁוֹת שֶׁבְּבָבֶל שֶׁיִּקָּבְעוּ בְּאֶרֶץ יִשְׂרָאֵל׳. בְּמֶשֶׁךְ דּוֹרוֹת הֵבִינוּ זֹאת יְהוּדִים בִּפְשִׁיטוּת: מְסֻפָּר עַל רַב שֶׁבָּא לְבֵית כְּנֶסֶת שֶׁהָיְתָה בּוֹ רִצְפָּה שֶׁל חוֹל, וְאָמַר לַמִּתְפַּלְּלִים: כְּשֶׁיָּבוֹא הַמָּשִׁיחַ, בֵּית הַכְּנֶסֶת יָעוּף לְאֶרֶץ יִשְׂרָאֵל – אֲבָל שֶׁלָּכֶם יִשָּׁאֵר כָּאן, כִּי אֵין לוֹ רִצְפָּה! בְּאוֹתוֹ שָׁבוּעַ כֻּלָּם תָּרְמוּ וְעָשׂוּ רִצְפָּה. הַגְּמָרָא אָכֵן הִתְקַיְּמָה – רַק בְּדֶרֶךְ אַחֶרֶת: לֹא הַבִּנְיָנִים עָפוּ, אֶלָּא בָּתֵּי הַמִּדְרָשׁ עַצְמָם, עִם הָרַבָּנִים, הַתַּלְמִידִים וְהַדֶּרֶךְ, עָבְרוּ לְאֶרֶץ יִשְׂרָאֵל.

הַגְּמָרָא הִבְטִיחָה שֶׁבָּתֵּי הַמִּדְרָשׁ שֶׁל הַגּוֹלָה יִקָּבְעוּ בְּאֶרֶץ יִשְׂרָאֵל – וְזֶה קָרָה בְּדוֹרֵנוּ, בְּדֶרֶךְ שֶׁלֹּא דִּמְיְנוּ.

🏫 הַיְשִׁיבוֹת שֶׁעָלוּ לְאֶרֶץ יִשְׂרָאֵל

לִפְנֵי מֵאָה שָׁנָה עִקַּר הַתּוֹרָה הָיָה בְּלִיטָא וּבְפּוֹלִין. יְשִׁיבַת מִיר נָדְדָה בַּמִּלְחָמָה עַד שַׁנְחַאי שֶׁבְּסִין, וְהַיּוֹם הִיא בִּירוּשָׁלַיִם. רַבִּי יוֹסֵף שְׁלֹמֹה כַּהֲנֶמַן בָּנָה אֶת פּוֹנִיבֶז׳ בִּבְנֵי בְּרַק בִּשְׁנַת תש״ד, בְּאֶמְצַע הַשּׁוֹאָה, בִּנְיָן עֲנָק לְתַלְמִידִים מְעַטִּים. וְרַבִּי שְׁמוּאֵל הַלֵּוִי וָאזְנֶר, תַּלְמִיד יְשִׁיבַת חַכְמֵי לוּבְּלִין, קָפַץ מִן הָאֳנִיָּה וְשָׂחָה אֶל חוֹף הָאָרֶץ – וְחִדֵּשׁ כָּאן אֶת הַיְשִׁיבָה. הָרַב כַּהֲנֶמַן אָמַר: ׳כְּבָר אַלְפַּיִם שָׁנָה לֹא הָיָה כָּל כָּךְ הַרְבֵּה תּוֹרָה כְּמוֹ שֶׁיֵּשׁ הַיּוֹם בְּאֶרֶץ יִשְׂרָאֵל׳ – וְאָז הָיוּ כָּאן 148 יְשִׁיבוֹת. הַיּוֹם יֵשׁ יוֹתֵר מֵאֶלֶף וְשֵׁשׁ מֵאוֹת, וְרֹב סִפְרֵי הַתּוֹרָה הַחֲשׁוּבִים שֶׁל הַדּוֹרוֹת הָאַחֲרוֹנִים נִכְתְּבוּ כָּאן.

עִקַּר הַתּוֹרָה בָּעוֹלָם יוֹצֵא הַיּוֹם מֵאֶרֶץ יִשְׂרָאֵל – ׳כִּי מִצִּיּוֹן תֵּצֵא תוֹרָה׳ לְנֶגֶד עֵינֵינוּ.

🌍 ׳וְנָהֲרוּ אֵלָיו כָּל הַגּוֹיִם׳ – כְּבָר הַיּוֹם

יְשַׁעְיָהוּ לֹא רָאָה עוֹלָם שֶׁל שִׂנְאָה, אֶלָּא עוֹלָם שֶׁבָּא לִלְמֹד: ׳וְיֹרֵנוּ מִדְּרָכָיו וְנֵלְכָה בְּאֹרְחֹתָיו׳. וְזֶה מַתְחִיל לִקְרוֹת: רַב בְּאֶרֶץ יִשְׂרָאֵל סִפֵּר שֶׁבָּאָה אֵלָיו קְבוּצָה שֶׁל אֲנָשִׁים מִמִּזְרַח אַסְיָה שֶׁבִּקְּשׁוּ לִשְׁמֹעַ תּוֹרָה. וְעוֹד סִפֵּר עַל קְבוּצַת תַּיָּרִים מִדְּרוֹם קוֹרֵיאָה שֶׁרָאֲתָה בְּעֶרֶב סֻכּוֹת יְהוּדִים בּוֹנִים סֻכָּה, הִתְלַהֲבָה, וּבִקְּשָׁה לִרְאוֹת אֶת הַסֻּכָּה וְאֶת אַרְבַּעַת הַמִּינִים שֶׁעֲלֵיהֶם קָרְאוּ בַּתַּנַ״ךְ – וּבַסּוֹף הַמַּנְהִיג שֶׁלָּהֶם בִּקֵּשׁ בְּרָכָה מִן הַיְּהוּדִי. הֵם לֹא בָּאִים לִכְבֹּשׁ – הֵם בָּאִים לִשְׁמֹעַ: ׳לְכוּ וְנַעֲלֶה אֶל הַר ה׳׳.

הֶחָזוֹן שֶׁל יְשַׁעְיָהוּ הוּא עוֹלָם שֶׁבָּא לִלְמֹד, וְהוּא מַתְחִיל לְהִתְקַיֵּם כְּבָר בְּדוֹרֵנוּ.

📗 סֵפֶר שֶׁהִתְגַּלְגֵּל לְכָל הָעוֹלָם

רַב בִּירוּשָׁלַיִם כָּתַב סֵפֶר עַל שֶׁבַע מִצְווֹת בְּנֵי נֹחַ – הַמִּצְווֹת שֶׁכָּל אָדָם בָּעוֹלָם מְצֻוֶּה בָּהֶן – וְעַל הַחוֹבָה לְלַמֵּד אוֹתָן. הַסֵּפֶר שָׁכַב זְמַן רַב: מִי יִקְרָא אוֹתוֹ? עַד שֶׁהֶחְלִיט: אֲנִי עוֹשֶׂה אֶת שֶׁלִּי, וְהוֹצִיא אוֹתוֹ לָאוֹר. הַסֵּפֶר הִתְגַּלְגֵּל לִידֵי אֲנָשִׁים מֵעַמִּים רַבִּים שֶׁהִתְלַהֲבוּ, בָּאוּ אֵלָיו וּבִקְּשׁוּ לִלְמֹד. הוּא תֻּרְגַּם לְאַנְגְּלִית, צָרְפָתִית, גֶּרְמָנִית, סְפָרַדִּית, רוּסִית וְעוֹד. וְלַיְשִׁיבָה שֶׁהִקִּים קָרְאוּ ׳דְּבַר יְרוּשָׁלַיִם׳ – עַל שֵׁם הַפָּסוּק ׳וּדְבַר ה׳ מִירוּשָׁלָיִם׳.

מָה שֶׁאָדָם אֶחָד עוֹשֶׂה בִּירוּשָׁלַיִם יָכוֹל לְהָאִיר לְכָל הָעוֹלָם – ׳וּדְבַר ה׳ מִירוּשָׁלָיִם׳.

🕯️ חֲזַ״ל: ׳כִּשְׁרָגָא בְּטִיהֲרָא׳

הַגְּמָרָא (שַׁבָּת ס״ג ע״א) שׁוֹאֶלֶת אֶת רַבִּי אֱלִיעֶזֶר: לָמָּה יִתְבַּטְּלוּ כְּלֵי הַנֶּשֶׁק לִימוֹת הַמָּשִׁיחַ? ׳אָמַר לָהֶן: לְפִי שֶׁאֵינָן צְרִיכִין, שֶׁנֶּאֱמַר: לֹא יִשָּׂא גוֹי אֶל גּוֹי חֶרֶב׳. שָׁאֲלוּ: שֶׁיִּשָּׁאֲרוּ לְנוֹי, כְּקִשּׁוּט! ׳אָמַר אַבַּיֵּי: מִידֵּי דַּהֲוָה אַשְּׁרָגָא בְּטִיהֲרָא׳ – כְּמוֹ נֵר בְּצָהֳרֵי הַיּוֹם. נֵר הוּא דָּבָר יָפֶה, אֲבָל בְּאוֹר הַשֶּׁמֶשׁ הוּא לֹא מוֹסִיף כְּלוּם. הַיּוֹם מְכַבְּדִים אָדָם עִם נֶשֶׁק, כִּי הוּא נָחוּץ; כְּשֶׁאִישׁ לֹא יִצְטָרֵךְ לְהָגֵן עַל עַצְמוֹ – הַחֶרֶב תִּהְיֶה כְּמוֹ פָּנָס בְּאֶמְצַע הַיּוֹם.

בִּימוֹת הַמָּשִׁיחַ הַנֶּשֶׁק יִתְבַּטֵּל כִּי לֹא יִהְיֶה בּוֹ צֹרֶךְ – ׳כִּשְׁרָגָא בְּטִיהֲרָא׳, כְּמוֹ נֵר בַּצָּהֳרַיִם.

🌱

📜 הַפָּסוּקוְשָׁפַט בֵּין הַגּוֹיִם וְהוֹכִיחַ לְעַמִּים רַבִּים וְכִתְּתוּ חַרְבוֹתָם לְאִתִּים וַחֲנִיתוֹתֵיהֶם לְמַזְמֵרוֹת לֹא־יִשָּׂא גוֹי אֶל־גּוֹי חֶרֶב וְלֹא־יִלְמְדוּ עוֹד מִלְחָמָה

💛 אַהֲבַת הַשֵּׁם: ה׳ מוֹבִיל אֶת הָעוֹלָם בְּסַבְלָנוּת, צַעַד אַחַר צַעַד, אֶל הַטּוֹב 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ב׳, פָּסוּק ד׳ (רַמְבַּ״ם, הִלְכוֹת מְלָכִים י״ב)

רָמָה ג׳ – מַעֲמִיקִים

🌱 הָרַמְבַּ״ם: ׳עוֹלָם כְּמִנְהָגוֹ נוֹהֵג׳

אֵיךְ יִתְקַיֵּם ׳לֹא יִשָּׂא גוֹי אֶל גּוֹי חֶרֶב׳? בְּדִבְרֵי הָרַמְבַּ״ם (הִלְכוֹת מְלָכִים י״ב, א׳): ׳אַל יַעֲלֶה עַל הַלֵּב שֶׁבִּימוֹת הַמָּשִׁיחַ יִבָּטֵל דָּבָר מִמִּנְהָגוֹ שֶׁל עוֹלָם... אֶלָּא עוֹלָם כְּמִנְהָגוֹ נוֹהֵג׳. אִם כֵּן, הַשָּׁלוֹם יָכוֹל לָבוֹא בְּתַהֲלִיכִים: עַמִּים שֶׁנִּלְחֲמוּ בֶּעָבָר מַחְלִיטִים לְאַט לְאַט שֶׁעָדִיף לָהֶם שָׁלוֹם. וַאֲנַחְנוּ עוֹד בָּאֶמְצַע הַדֶּרֶךְ – עֲדַיִן צְרִיכִים לִשְׁמֹר עַל עַצְמֵנוּ, וְהַשָּׁלוֹם שֶׁחָזוּ הַנְּבִיאִים רָחוֹק. אֲבָל כְּמוֹ שֶׁחֵלֶק מִן הַנְּבוּאָה כְּבָר הִתְקַיֵּם לְנֶגֶד עֵינֵינוּ, כָּךְ יִתְקַיֵּם גַּם הַשְּׁאָר.

לְפִי הָרַמְבַּ״ם הַשָּׁלוֹם יָכוֹל לָבוֹא בְּדֶרֶךְ הַטֶּבַע, צַעַד אַחַר צַעַד – וַאֲנַחְנוּ עוֹד בַּדֶּרֶךְ.

📜 הַפָּסוּקבֵּית יַעֲקֹב לְכוּ וְנֵלְכָה בְּאוֹר ה׳

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה שֶׁמִּי שֶׁיִּסְתַּכֵּל עָלַי יִרְאֶה קְצָת מִן הָאוֹר שֶׁל ה׳ 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ב׳, פָּסוּק ה׳; פֶּרֶק ג׳, פָּסוּק י׳

רָמָה ג׳ – מַעֲמִיקִים

📖 מָה יְשַׁעְיָהוּ מְלַמֵּד אוֹתִי הַיּוֹם

אִם הָעוֹלָם בָּא לִלְמֹד מֵאִתָּנוּ – מָה הוּא יִרְאֶה? לָכֵן יְשַׁעְיָהוּ אוֹמֵר מִיָּד: ׳בֵּית יַעֲקֹב לְכוּ וְנֵלְכָה בְּאוֹר ה׳׳. וַחֲזַ״ל (קִדּוּשִׁין מ׳ ע״א) מְבָאֲרִים אֶת ׳אִמְרוּ צַדִּיק כִּי טוֹב׳: ׳טוֹב לַשָּׁמַיִם וְטוֹב לַבְּרִיּוֹת – זֶהוּ צַדִּיק טוֹב׳. הַנָּבִיא מְלַמֵּד אוֹתִי: לִלְמֹד תּוֹרָה בְּשִׂמְחָה, לִהְיוֹת טוֹב גַּם לַה׳ וְגַם לַאֲנָשִׁים, לְוַתֵּר עַל גַּאֲוָה – וּלְהוֹדוֹת לַה׳ שֶׁזָּכִיתִי לִחְיוֹת בַּדּוֹר שֶׁבּוֹ ׳מִצִּיּוֹן תֵּצֵא תוֹרָה׳.

הוֹלְכִים בְּאוֹר ה׳ כְּבָר הַיּוֹם: טוֹבִים לַשָּׁמַיִם וְלַבְּרִיּוֹת, וּמוֹדִים עַל הַנְּבוּאָה שֶׁמִּתְקַיֶּמֶת.

רָמָה ג׳ – מַעֲמִיקִים

מָה לוֹקְחִים מִרָמָה ג׳?

יְשַׁעְיָהוּ רָאָה אֶת הַתּוֹרָה יוֹצֵאת מִצִּיּוֹן וְאֶת הָעַמִּים בָּאִים לִלְמֹד – וְאֶת תְּחִלַּת הַחֲזוֹן הַזֶּה אֲנַחְנוּ זוֹכִים לִרְאוֹת בְּעֵינֵינוּ. כְּשֶׁנָּשִׁיר בַּפַּעַם הַבָּאָה ׳כִּי מִצִּיּוֹן תֵּצֵא תוֹרָה׳, נָשִׁיר בְּעֵינַיִם פְּקוּחוֹת, נוֹדֶה לַה׳ עַל הַחֶסֶד, וְנִתְפַּלֵּל לִרְאוֹת אֶת הַנְּבוּאָה בִּשְׁלֵמוּתָהּ.

סוֹף רָמָה ג׳

🤔 חוֹשְׁבִים עַל זֶה

  • יְשַׁעְיָהוּ אוֹמֵר ׳לְכוּ וְנֵלְכָה בְּאוֹר ה׳׳ – לֹא ׳לֵךְ׳ אֶלָּא ׳נֵלְכָה׳, כֻּלָּנוּ יַחַד. אֵיזֶה דָּבָר טוֹב אֲנִי יָכוֹל לְהַזְמִין חָבֵר לַעֲשׂוֹת אִתִּי הַשָּׁבוּעַ?
  • הָעַמִּים אוֹמְרִים ׳וְיֹרֵנוּ מִדְּרָכָיו׳ – הֵם רוֹצִים לִלְמֹד. מָתַי בְּפַעַם הָאַחֲרוֹנָה לָמַדְתִּי תּוֹרָה כִּי רָצִיתִי, וְלֹא כִּי הָיִיתִי צָרִיךְ?
  • ׳כִּי מִצִּיּוֹן תֵּצֵא תוֹרָה׳ מִתְקַיֵּם בְּדוֹרֵנוּ. אֵיךְ אֲנִי יָכוֹל לִהְיוֹת חֵלֶק מִזֶּה – וְעַל מָה אֲנִי רוֹצֶה לְהוֹדוֹת לַה׳ כְּשֶׁאֲנִי שָׁר אֶת הַפָּסוּק בְּבֵית הַכְּנֶסֶת?

רָמָה ג׳ – מַעֲמִיקִים

יְשַׁעְיָהוּ מְנַבֵּא עַל מַפַּלְתָּהּ שֶׁל בָּבֶל וְעַל גַּאֲוַת מַלְכָּהּ – וּמִתּוֹךְ כָּךְ מַבְטִיחַ: ה׳ יְרַחֵם עַל יַעֲקֹב וְיִבְחַר בּוֹ שׁוּב

📖 פְּרָקִים י״ג–י״ח

🎯 לַחִידוֹן
🚶

📜 הַפָּסוּקאֵיךְ נָפַלְתָּ מִשָּׁמַיִם הֵילֵל בֶּן־שָׁחַר נִגְדַּעְתָּ לָאָרֶץ חוֹלֵשׁ עַל־גּוֹיִם וְאַתָּה אָמַרְתָּ בִלְבָבְךָ הַשָּׁמַיִם אֶעֱלֶה מִמַּעַל לְכוֹכְבֵי־אֵל אָרִים כִּסְאִי וְאֵשֵׁב בְּהַר־מוֹעֵד בְּיַרְכְּתֵי צָפוֹן

💛 אַהֲבַת הַשֵּׁם: ה׳ מְשַׁלֵּם שָׂכָר גָּדוֹל עַל כָּל דָּבָר שֶׁנַּעֲשֶׂה לִכְבוֹדוֹ 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק י״ד, פְּסוּקִים י״ב–י״ד (סַנְהֶדְרִין צ״ו ע״א)

רָמָה ג׳ – מַעֲמִיקִים

🚶 שָׁלוֹשׁ הַפְּסִיעוֹת שֶׁל נְבוּכַדְנֶצַּר

הָרַב סִפֵּר מִסַּנְהֶדְרִין צ״ו ע״א: כְּשֶׁעָמְדָה הַחַמָּה לְחִזְקִיָּהוּ הַמֶּלֶךְ, שָׁלְחוּ מִבָּבֶל אִגֶּרֶת לְחִזְקִיָּהוּ, וְהִקְדִּימוּ בָּהּ אֶת כְּבוֹד חִזְקִיָּהוּ לִכְבוֹד בּוֹרֵא עוֹלָם. כְּשֶׁרָאָה זֹאת נְבוּכַדְנֶצַּר, הִקְפִּיד מְאֹד: ״כָּךְ כּוֹתְבִים? בּוֹרֵא עוֹלָם צָרִיךְ לִהְיוֹת רִאשׁוֹן!״ הוּא רָץ לְהַחְזִיר אֶת הַשָּׁלִיחַ, פָּסַע שָׁלוֹשׁ פְּסִיעוֹת וְעָצְרוּ אוֹתוֹ. וּבִשְׂכַר כָּל פְּסִיעָה זָכָה לְמַלְכוּת עַל כָּל הָעוֹלָם דּוֹר אֶחָד. הַחָפֵץ חַיִּים הוֹסִיף: כְּשֶׁאָדָם עוֹשֶׂה מִצְוָה לִכְבוֹד ה׳, אֵין קֵץ לִשְׂכָרָהּ.

אֲפִלּוּ מֶלֶךְ גֵּאֶה קִבֵּל שָׂכָר עַל שָׁלוֹשׁ פְּסִיעוֹת לִכְבוֹד שָׁמַיִם – מִצְוָה לִכְבוֹד ה׳ אֵין לָהּ קֵץ.

1
🎁

📜 הַפָּסוּקבָּעֵת הַהִיא יוּבַל־שַׁי לַה׳ צְבָאוֹת עַם מְמֻשָּׁךְ וּמוֹרָט וּמֵעַם נוֹרָא מִן־הוּא וָהָלְאָה גּוֹי קַו־קָו וּמְבוּסָה אֲשֶׁר בָּזְאוּ נְהָרִים אַרְצוֹ אֶל־מְקוֹם שֵׁם־ה׳ צְבָאוֹת הַר־צִיּוֹן

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹמֵר אֶת הַתִּקְוָה שֶׁלָּנוּ, וְעוֹד יָבִיא אוֹתָנוּ לְצִיּוֹן 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק י״ח, פָּסוּק ז׳ (מְצוּדַת דָּוִד)

רָמָה ג׳ – מַעֲמִיקִים

🎁 מְצוּדַת דָּוִד: ׳גּוֹי קַו קָו׳

בְּסוֹף הַנְּבוּאָה יֵשׁ שֶׁי: ׳בָּעֵת הַהִיא יוּבַל שַׁי לַה׳ צְבָאוֹת עַם מְמֻשָּׁךְ וּמוֹרָט... גּוֹי קַו קָו וּמְבוּסָה אֲשֶׁר בָּזְאוּ נְהָרִים אַרְצוֹ׳. מְצוּדַת דָּוִד מְבָאֵר: ׳גּוֹי קַו קָו – אֲשֶׁר בְּכָל יְמֵי גָּלוּתָם הָיָה מְקַוֶּה לַה׳ וְחוֹזֵר וּמְקַוֶּה, וְעִם כִּי בָּזְזוּ אֶת אַרְצוֹ וּרְמָסוּהוּ בְּרַגְלֵיהֶם׳. הָעָם שֶׁהִקְוָה תָּמִיד הוּא זֶה שֶׁמַּגִּיעַ אֶל הַר צִיּוֹן.

מְצוּדַת דָּוִד: ׳גּוֹי קַו קָו׳ = עַם שֶׁבְּכָל הַגָּלוּת חָזַר וְקִוָּה לַה׳ – וְלָכֵן מַגִּיעַ אֶל הַר צִיּוֹן.

📜 הַפָּסוּקכִּי יְרַחֵם ה׳ אֶת־יַעֲקֹב וּבָחַר עוֹד בְּיִשְׂרָאֵל וְהִנִּיחָם עַל־אַדְמָתָם וְנִלְוָה הַגֵּר עֲלֵיהֶם וְנִסְפְּחוּ עַל־בֵּית יַעֲקֹב

💛 אַהֲבַת הַשֵּׁם: ה׳ בּוֹחֵר בִּי שׁוּב, וְאֲנִי בּוֹחֵר בּוֹ 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק י״ד, פְּסוּקִים א׳, י״ב, כ״ז

רָמָה ג׳ – מַעֲמִיקִים

📖 מָה יְשַׁעְיָהוּ מְלַמֵּד אוֹתִי הַיּוֹם

יְשַׁעְיָהוּ מְלַמֵּד אוֹתִי: גַּאֲוָה נוֹפֶלֶת – ׳אֵיךְ נָפַלְתָּ מִשָּׁמַיִם׳; לָכֵן צְרִיכִים לִהְיוֹת עֲנָוִים. ה׳ מַבְטִיחַ ׳וּבָחַר עוֹד בְּיִשְׂרָאֵל׳, וְלָכֵן אֶפְשָׁר לִבְטֹחַ גַּם כְּשֶׁהַכֹּל קָשֶׁה, כִּי ׳ה׳ צְבָאוֹת יָעָץ וּמִי יָפֵר׳. וְכֻלָּנוּ יְכוֹלִים לִהְיוֹת ׳קַו קָו׳ – לִקְווֹת, וְלָשׁוּב וְלִקְווֹת.

עֲנָוָה, בִּטָּחוֹן בַּבְטָחוֹת ה׳, וְתִקְוָה שֶׁלֹּא נִגְמֶרֶת.

רָמָה ג׳ – מַעֲמִיקִים

מָה לוֹקְחִים מִרָמָה ג׳?

אַחֲרֵי שֶׁיְּשַׁעְיָהוּ מְתָאֵר אֶת גַּאֲוַת בָּבֶל, חֲכָמִים מְלַמְּדִים שֶׁאֲפִלּוּ מֶלֶךְ בָּבֶל זָכָה לְרֶגַע שֶׁל הַכָּרַת כְּבוֹד שָׁמַיִם. וּמְצוּדַת דָּוִד מְלַמֵּד מִי הָעָם שֶׁ׳הוּבַל שַׁי׳ לְצִיּוֹן: עַם שֶׁהִקְוָה.

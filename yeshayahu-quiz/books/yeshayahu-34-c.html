רָמָה ג׳ – מַעֲמִיקִים

🎶 ׳טֶרֶם יִקְרָאוּ׳ – שֶׁבַח לִפְנֵי תְּפִלָּה

חֲזַ״ל אוֹמְרִים (בְּרָכוֹת ל״ב ע״א): ׳לְעוֹלָם יְסַדֵּר אָדָם שִׁבְחוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא וְאַחַר כָּךְ יִתְפַּלֵּל׳. הָאַלְשִׁיךְ מְבָאֵר אֶת ׳טֶרֶם יִקְרָאוּ וַאֲנִי אֶעֱנֶה עוֹד הֵם מְדַבְּרִים׳: ׳עוֹד הֵם מְדַבְּרִים׳ – כְּשֶׁהֵם מַזְכִּירִים אֶת כֹּחִי וּמִדּוֹתַי וּמְשַׁבְּחִים אוֹתִי, עוֹד לִפְנֵי שֶׁיִּקְרְאוּ וִיבַקְּשׁוּ, אֲנִי כְּבָר עוֹנֶה. הָרַב לָמַד: עַל תְּפִלָּה יָכוֹל לָקוּם מְקַטְרֵג שֶׁהָאָדָם אֵינוֹ רָאוּי, אֲבָל עַל שֶׁבַח אֱמֶת אֵין מִי שֶׁיֹּאמַר שֶׁאֵינוֹ אֱמֶת. רַבִּי חִיָּיא וּבָנָיו: כְּשֶׁאָמַר ׳מַשִּׁיב הָרוּחַ׳ – נָשְׁבָה הָרוּחַ, וּכְשֶׁאָמַר ׳מוֹרִיד הַגֶּשֶׁם׳ – יָרַד הַגֶּשֶׁם. הֵם לֹא בִּקְשׁוּ, רַק הִזְכִּירוּ אֶת שִׁבְחוֹ שֶׁל ה׳.

הַשֶּׁבַח קוֹדֵם לַתְּפִלָּה – ׳טֶרֶם יִקְרָאוּ וַאֲנִי אֶעֱנֶה׳.

📜 הַפָּסוּקזְאֵב וְטָלֶה יִרְעוּ כְאֶחָד וְאַרְיֵה כַּבָּקָר יֹאכַל־תֶּבֶן וְנָחָשׁ עָפָר לַחְמוֹ לֹא־יָרֵעוּ וְלֹא־יַשְׁחִיתוּ בְּכָל־הַר קָדְשִׁי אָמַר ה׳

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שָׁלוֹם בֵּין כָּל בְּרִיּוֹתָיו, בֵּין אֲנָשִׁים וּבֵין עַמִּים 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ס״ה, פָּסוּק כ״ה

רָמָה ג׳ – מַעֲמִיקִים

⚖️ ׳זְאֵב וְטָלֶה יִרְעוּ כְאֶחָד׳ – רַב סַעֲדְיָה גָּאוֹן וְהָרַמְבַּ״ם

הַפָּסוּק דּוֹמֶה לִ׳וְגָר זְאֵב עִם כֶּבֶשׂ׳ (י״א ו׳), וְהָרַב לָמַד מַחֲלֹקֶת עַל פֵּרוּשׁוֹ. בְּדִבְרֵי הָרַמְבַּ״ם: ׳אַל יַעֲלֶה עַל הַלֵּב שֶׁבִּימוֹת הַמָּשִׁיחַ יְבַטֵּל דָּבָר מִמִּנְהָגוֹ שֶׁל עוֹלָם... אֶלָּא עוֹלָם כְּמִנְהָגוֹ נוֹהֵג׳, וְהַפָּסוּק ׳מָשָׁל וְחִידָה׳: הַזְּאֵב וְהַנָּמֵר הֵם הָעַמִּים, וְהַכֶּבֶשׂ וְהַגְּדִי – יִשְׂרָאֵל, וְיִהְיוּ ׳יוֹשְׁבִים לָבֶטַח׳ עִם הָעַמִּים. לְעֻמַּת זֶה רַב סַעֲדְיָה גָּאוֹן קָבַע שֶׁהַנְּבוּאָה עוֹד לֹא הִתְקַיְּמָה, כִּי ׳הֲרֵי אָנוּ רוֹאִים אוֹתָם הַיּוֹם בְּטִבְעָם׳, וְלָכֵן הַזְּאֵב וְהַכֶּבֶשׂ יִחְיוּ בֶּאֱמֶת יַחַד. וּבְתַרְגּוּם יוֹנָתָן נֶאֱמַר שֶׁגַּם הַשָּׁלוֹם יִתְפַּשֵּׁט עַד שֶׁבַּעֲלֵי הַחַיִּים יְשַׁנּוּ אֶת טִבְעָם. וְגַם אִבְּן עֶזְרָא (כְּפִי שֶׁמְּצַטֵּט הָאַבְרַבְּנֵאל): ׳בָּזֶה הִבְטִיחַ עַל הַשָּׁלוֹם שֶׁיִּהְיֶה בִּימוֹת הַמָּשִׁיחַ׳. בֵּין כָּךְ וּבֵין כָּךְ – שָׁלוֹם.

רַמְבַּ״ם: ׳מָשָׁל וְחִידָה׳ עַל שָׁלוֹם בֵּין עַמִּים; רַב סַעֲדְיָה: כְּפַשׁוּטוֹ – בַּעֲלֵי חַיִּים בֶּאֱמֶת בְּשָׁלוֹם.

📜 הַפָּסוּקוְהָיָה מִדֵּי־חֹדֶשׁ בְּחָדְשׁוֹ וּמִדֵּי שַׁבָּת בְּשַׁבַּתּוֹ יָבוֹא כָל־בָּשָׂר לְהִשְׁתַּחֲוֺת לְפָנַי אָמַר ה׳

💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לִי נְשָׁמָה נוֹסֶפֶת בְּשַׁבָּת וּבְרֹאשׁ חֹדֶשׁ, כְּדֵי שֶׁאֶהֱנֶה לְהוֹדוֹת לוֹ 💛

רָמָה ג׳ – מַעֲמִיקִים

🌙 ׳מִדֵּי חֹדֶשׁ בְּחָדְשׁוֹ וּמִדֵּי שַׁבָּת בְּשַׁבַּתּוֹ׳ – הַנְּשָׁמָה הַיְתֵרָה

סוֹף הַסֵּפֶר – ׳הַשָּׁמַיִם כִּסְאִי׳ וְעַד ׳וְהָיָה מִדֵּי חֹדֶשׁ בְּחָדְשׁוֹ וּמִדֵּי שַׁבָּת בְּשַׁבַּתּוֹ יָבוֹא כָל בָּשָׂר לְהִשְׁתַּחֲוֺת לְפָנַי אָמַר ה׳׳ – נִקְרָא כְּהַפְטָרָה בְּשַׁבָּת רֹאשׁ חֹדֶשׁ. הָרַב לָמַד, בְּשֵׁם הַזֹּהַר (רַבִּי שִׁמְעוֹן עַל הַפָּסוּק) וְהַגְּר״א, שֶׁבְּכָל יוֹם שֶׁיֵּשׁ בּוֹ מוּסָף יֵשׁ ׳נְשָׁמָה יְתֵרָה׳. הַנְּשָׁמָה הַיְתֵרָה גּוֹרֶמֶת לַיְּהוּדִי, אֲפִלּוּ בְּלִי נֵס, לְהוֹדוֹת וּלְהַלֵּל – לָכֵן אוֹמְרִים הַלֵּל בְּרֹאשׁ חֹדֶשׁ וּבַחַגִּים. בְּסוֹף הַהַלֵּל יֵשׁ שְׁמוֹנֶה לְשׁוֹנוֹת – ׳יוֹדוּ וִיבָרְכוּ וִישַׁבְּחוּ וִיפָאֲרוּ וִירוֹמְמוּ וְיַעֲרִיצוּ וְיַקְדִּישׁוּ וְיַמְלִיכוּ׳ – כְּנֶגֶד ׳שְׁמוֹנֶה׳ – נְשָׁמָה. ׳אָבִינוּ שֶׁבַּשָּׁמַיִם נָתַן לָנוּ נְשָׁמָה טְהוֹרָה, וְהוֹסִיף לָנוּ עוֹד מִמֶּנָּה בְּשַׁבָּתוֹת, בַּחַגִּים וּבְרָאשֵׁי חֳדָשִׁים׳.

בְּשַׁבָּת וּבְרֹאשׁ חֹדֶשׁ יֵשׁ לָנוּ נְשָׁמָה יְתֵרָה, וּבָהּ אֲנַחְנוּ בָּאִים ׳לְהִשְׁתַּחֲוֺת לְפָנֵי ה׳׳.

📖 מָה יְשַׁעְיָהוּ מְלַמֵּד אוֹתִי הַיּוֹם

סוֹף הַסֵּפֶר הוּא בְּשׂוֹרָה: ה׳ מוּכָן לְקַבֵּל אוֹתִי, עוֹנֶה עוֹד לִפְנֵי שֶׁאֲנִי קוֹרֵא, מַבִּיט אֶל הֶעָנִי וְהֶחָרֵד. הַלִּמּוּד: לְהַתְחִיל כָּל תְּפִלָּה בְּשֶׁבַח, לִהְיוֹת עָנָו וְחָרֵד עַל דִּבְרֵי ה׳, לַעֲשׂוֹת צַעַד לְשָׁלוֹם עִם כָּל אָדָם, וּלְנַצֵּל אֶת הַנְּשָׁמָה הַיְתֵרָה בְּשַׁבָּת וּבְרֹאשׁ חֹדֶשׁ כְּדֵי לְהוֹדוֹת וּלְהַלֵּל.

שֶׁבַח, עֲנָוָה, שָׁלוֹם וְהוֹדָיָה – וְה׳ כְּבָר עוֹנֶה.

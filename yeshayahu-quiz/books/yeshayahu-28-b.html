🌍 רַשִׁ״י וּמְצוּדַת דָּוִד: הַדְּאָגָה שֶׁל הַגֵּר

הַגֵּר חוֹשֵׁשׁ: ׳הַבְדֵּל יַבְדִּילַנִי ה׳ מֵעַל עַמּוֹ׳. רַשִׁ״י מְבָאֵר: ׳לָמָּה אֶתְגַּיֵּיר, הֲלֹא הַקָּדוֹשׁ בָּרוּךְ הוּא יַבְדִּילֵנִי מֵעַל עַמּוֹ כְּשֶׁיְּשַׁלֵּם שְׂכָרָם׳. וּמְצוּדַת דָּוִד: ׳אֶהְיֶה נִבְדָּל מִיִּשְׂרָאֵל בְּעֵת יְקַבְּלוּ הַטּוֹבָה, וְלֹא אֲקַבֵּל עִמָּהֶם׳. מְצוּדַת צִיּוֹן מְבָאֵר אֶת ׳הַנִּלְוֶה׳: ׳עִנְיַן הַחֶבְרָה וְהָרֵעוּת׳. ה׳ עוֹנֶה בְּפָסוּק ו׳: ׳כָּל שֹׁמֵר שַׁבָּת... וּמַחֲזִיקִים בִּבְרִיתִי׳ – מַשְׁמָע: כָּל הָאוֹחֵז בַּתּוֹרָה, הוּא חֵלֶק מִן הַחֶבְרָה.

הַנִּלְוֶה – ׳עִנְיַן חֶבְרָה וְרֵעוּת׳: ה׳ אוֹמֵר לַגֵּר שֶׁלֹּא יִבָּדֵל, אֶלָּא יִהְיֶה חָבֵר.

🌳 מְצוּדַת צִיּוֹן: ׳יָד׳ הוּא מָקוֹם

הַנָּבִיא מַזְכִּיר אָדָם שֶׁחוֹשֵׁב: ׳הֵן אֲנִי עֵץ יָבֵשׁ׳. רַשִׁ״י מְבָאֵר: ׳לָמָּה אֵיטִיב דְּרָכַי וּמַעֲלָלַי, הֵן אֲנִי כְּעֵץ יָבֵשׁ מֵאֵין זִכָּרוֹן׳. וְה׳ עוֹנֶה. מְצוּדַת צִיּוֹן: ׳יָד – מָקוֹם׳, כְּמוֹ ׳וְיָד תִּהְיֶה לְךָ׳. וּמְצוּדַת דָּוִד מְבָאֵר אֶת ׳בְּבֵיתִי וּבְחוֹמֹתַי׳: ׳בְּבֵית הַמִּקְדָּשׁ... שָׁם אֶתֵּן לוֹ מָקוֹם וְזִכָּרוֹן שֵׁם, וְהוּא דָּבָר הַטּוֹב יוֹתֵר מִזִּכָּרוֹן הַבָּא מִבָּנִים וּבָנוֹת׳.

׳יָד וָשֵׁם׳ – ׳מָקוֹם וְשֵׁם׳ בְּבֵית ה׳, טוֹבִים מִזִּכָּרוֹן שֶׁל בָּנִים וּבָנוֹת.

📜 הַפָּסוּקכִּי בֵיתִי בֵּית תְּפִלָּה יִקָּרֵא לְכָל הָעַמִּים

רַשִׁ״י: בֵּית הַמִּקְדָּשׁ אֵינוֹ לְיִשְׂרָאֵל בִּלְבַד, ׳כִּי אַף לַגֵּרִים׳.

רַשִׁ״י מְבָאֵר: ׳כִּבְשׁוּ מְסִלָּה סְלוּלָה, פַּנּוּ יֵצֶר הָרַע מִדַּרְכֵּיכֶם׳, וְ׳הָרִימוּ מִכְשׁוֹל – סַלְּקוּ הָאֲבָנִים שֶׁאַתֶּם נִכְשָׁלִים בָּהֶם, הֵם הִרְהוּרֵי רֶשַׁע׳. מְצוּדַת דָּוִד לוֹמֵד לְפִי הָרֶקַע הַהִסְטוֹרִי: ׳חִזְקִיָּה שֶׁחָסָה בִּי יֹאמַר סֹלּוּ סֹלּוּ׳ – כְּלוֹמַר עֲשׂוּ דֶּרֶךְ כְּבוּשָׁה לָעָם, ׳כִּי לֹא יִפְחֲדוּ עוֹד מִן הָאוֹיֵב׳. לְפִי רַשִׁ״י סוֹלְלִים דֶּרֶךְ בַּלֵּב, וּלְפִי מְצוּדַת דָּוִד – דֶּרֶךְ בָּרְחוֹב.

רָמָה ב׳ – עִיּוּן בְּבֵית הַתְּפִלָּה

🤲 רַשִׁ״י: ׳לֹא לְעוֹלָם אָרִיב׳ וּ׳לָרָחוֹק וְלַקָּרוֹב׳

רַשִׁ״י מְבָאֵר: ׳כִּי לֹא לְעוֹלָם אָרִיב – אִם אָבִיא יִסּוּרִין עַל הָאָדָם, אֵין תַּחֲרוּתִי עָלָיו לְאֹרֶךְ יָמִים וְלֹא קִצְפִּי לָנֶצַח׳. וְעַל ׳לָרָחוֹק וְלַקָּרוֹב׳ רַשִׁ״י מְבָאֵר: ׳שְׁנֵיהֶם שָׁוִין – מִי שֶׁנִּתְיַשֵּׁן וְהוּרְגַּל בְּתוֹרָתִי וַעֲבוֹדָתִי מִנְּעוּרָיו, וּמִי שֶׁנִּתְקָרֵב עַתָּה מִקָּרוֹב לָשׁוּב מִדַּרְכּוֹ׳. ה׳ אוֹמֵר לִשְׁנֵיהֶם: ׳וּרְפָאתִיו׳.

רַשִׁ״י: מִי שֶׁקָּרוֹב מִנְּעוּרָיו וּמִי שֶׁחָזַר עַכְשָׁיו – ה׳ אוֹמֵר לִשְׁנֵיהֶם שָׁלוֹם.

רָמָה ב׳ – עִיּוּן בְּבֵית הַתְּפִלָּה

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מְגַלִּים אֶת הָאַהֲבָה: ה׳ נוֹתֵן לְכָל אֶחָד ׳מָקוֹם וָשֵׁם׳ בְּבֵיתוֹ, ׳אֵין תַּחֲרוּתִי עָלָיו לְאֹרֶךְ יָמִים׳, וְהוּא מַרְכִּין אֶת שְׁכִינָתוֹ עַל הַשָּׁפָל.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳לְכָל הָעַמִּים׳ – מִי הוּא הָאָדָם שֶׁהָיָה לִי קָשֶׁה לְקַבֵּל אוֹתוֹ בַּקְּבוּצָה שֶׁלָּנוּ, וְאֵיךְ אֲנִי יָכוֹל לַעֲשׂוֹת לוֹ מָקוֹם?
  • ׳שֵׁם טוֹב מִבָּנִים וּמִבָּנוֹת׳ – מָה אֲנִי רוֹצֶה שֶׁיִּזְכְּרוּ עָלַי, וְאֵיזֶה מַעֲשֶׂה טוֹב אֲנִי יָכוֹל לַעֲשׂוֹת הַשָּׁבוּעַ?
  • ׳וְאֶת דַּכָּא וּשְׁפַל רוּחַ׳ – מָתַי הִרְגַּשְׁתִּי שֶׁהַלֵּב שֶׁלִּי שָׁבוּר, וְאֵיךְ אֶפְשָׁר לִזְכֹּר שֶׁה׳ קָרוֹב אֵלַי דַּוְקָא אָז?

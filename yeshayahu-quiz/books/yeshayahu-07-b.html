רַשִׁ״י מְבָאֵר אֶת ׳וּשְׁאַבְתֶּם׳: ׳וּתְקַבְּלוּן אוּלְפָן חַדְתָּא׳ – תְּקַבְּלוּ לִמּוּד חָדָשׁ, וְעַל ׳מִמַּעַיְנֵי הַיְשׁוּעָה׳: ׳כִּי יַרְחִיב לִבָּם עַל יְדֵי יְשׁוּעָה הַבָּאָה לָהֶם וְיִתְגַּלּוּ לָהֶם רָזֵי הַתּוֹרָה שֶׁנִּשְׁתַּכְּחוּ בְּבָבֶל עַל יְדֵי הַצָּרוֹת׳. מְצוּדַת דָּוִד: ׳כְּמוֹ הַשּׁוֹאֵב מַיִם מִן הַמַּעְיָן הַנּוֹבֵעַ, הֲרֵי שׁוֹאֵב הוּא בְּשָׂשׂוֹן כִּי לֹא יִדְאַג פֶּן יִכְלוּ הַמַּיִם׳.

רַשִׁ״י וּמְצוּדַת דָּוִד: הַיְשׁוּעָה כְּמוֹ מַעְיָן – לֹא נִגְמָר; וְעִמָּהּ בָּא לִמּוּד תּוֹרָה חָדָשׁ.

רָמָה ב׳ – עִיּוּן בַּנְּבוּאָה

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מְגַלִּים כַּמָּה אַהֲבָה בַּנְּבוּאָה: ה׳ מְנַחֵם גַּם אֶת הַגּוֹלִים מֵאַשּׁוּר, שׁוֹפֵט מִן הַלֵּב, וְהַשִּׂמְחָה בַּיְשׁוּעָה כְּמוֹ מַעְיָן שֶׁאֵינוֹ מִתְיַבֵּשׁ – ׳וּשְׁאַבְתֶּם מַיִם בְּשָׂשׂוֹן׳.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳כִּי מָלְאָה הָאָרֶץ דֵּעָה אֶת ה׳׳. מָה אֲנִי יָכוֹל לַעֲשׂוֹת הַיּוֹם כְּדֵי שֶׁיִּהְיֶה בָּעוֹלָם קְצָת יוֹתֵר דֵּעָה וְשָׁלוֹם?
  • בְּהַבְדָּלָה אֲנִי אוֹמֵר ׳הִנֵּה אֵל יְשׁוּעָתִי אֶבְטַח וְלֹא אֶפְחָד׳. מָה גָּרַם לִי הַשָּׁבוּעַ לְהַרְגִּישׁ בִּטָּחוֹן, וְעַל מָה אֲנִי רוֹצֶה לְהוֹדוֹת לְה׳?
  • ׳וּשְׁאַבְתֶּם מַיִם בְּשָׂשׂוֹן׳ – מָתַי הָיְתָה לִי שִׂמְחָה אֲמִתִּית שֶׁל לִמּוּד אוֹ שֶׁל חֶסֶד?

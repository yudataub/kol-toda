בְּאַחֲרִית הַיָּמִים

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

יְשַׁעְיָהוּ רוֹאֶה אֶת כָּל הָעַמִּים עוֹלִים לִלְמֹד תּוֹרָה בְּהַר ה׳, וְקוֹרֵא לָנוּ לָלֶכֶת כְּבָר עַכְשָׁיו בְּאוֹר ה׳

📖 פְּרָקִים ב׳–ד׳

📜 הַסֵּפֶר הַמָּלֵא

📜 הַפָּסוּקוְהָיָה בְּאַחֲרִית הַיָּמִים נָכוֹן יִהְיֶה הַר בֵּית ה׳ בְּרֹאשׁ הֶהָרִים וְנִשָּׂא מִגְּבָעוֹת

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹמֵר לָנוּ נִסִּים גְּדוֹלִים עוֹד יוֹתֵר מִמַּה שֶּׁהָיָה 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ב׳, פָּסוּק ב׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

✨ רַשִׁ״י: ׳וְנִשָּׂא מִגְּבָעוֹת׳

מָתַי זֶה יִהְיֶה? מְצוּדַת דָּוִד מְבָאֵר: ׳בְּאַחֲרִית הַיָּמִים – בִּימֵי הַמָּשִׁיחַ׳. וּמַה פֵּרוּשׁ ׳וְנִשָּׂא מִגְּבָעוֹת׳? רַשִׁ״י מְבָאֵר: ׳יִגְדַּל נֵס שֶׁנַּעֲשָׂה בּוֹ מִנִּסֵּי סִינַי וְכַרְמֶל וְתָבוֹר׳. כְּלוֹמַר: הַנֵּס שֶׁיֵּעָשֶׂה בְּהַר הַבַּיִת יִהְיֶה גָּדוֹל מִן הַנִּסִּים שֶׁנַּעֲשׂוּ בְּהָרִים אֲחֵרִים, וְגַם מִמַּתַּן תּוֹרָה בְּסִינַי.

רַשִׁ״י: הַר הַבַּיִת ׳יִנָּשֵׂא׳ בְּנֵס גָּדוֹל מִנִּסֵּי סִינַי, כַּרְמֶל וְתָבוֹר.

📜 הַפָּסוּק…וְנָהֲרוּ אֵלָיו כָּל הַגּוֹיִם

💛 אַהֲבַת הַשֵּׁם: קִרְבַת ה׳ מוֹשֶׁכֶת אֶת הַלֵּב כְּמוֹ נָהָר 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ב׳, פָּסוּק ב׳ (רַשִׁ״י וּמְצוּדַת צִיּוֹן)

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

🌊 מְצוּדַת צִיּוֹן: ׳וְנָהֲרוּ אֵלָיו׳

׳וְנָהֲרוּ אֵלָיו כָּל הַגּוֹיִם׳ – מְצוּדַת צִיּוֹן מְבָאֵר: ׳יִמָּשְׁכוּ אֵלָיו כְּנָהָר הַמּוֹשֵׁךְ׳, וְרַשִׁ״י: ׳יִתְקַבְּצוּ, יִמָּשְׁכוּ אֵלָיו כִּנְהָרוֹת׳. נָהָר רָגִיל זוֹרֵם לְמַטָּה, מִן הָהָר אֶל הָעֵמֶק. כָּאן הָעַמִּים זוֹרְמִים לְמַעְלָה, אֶל רֹאשׁ הָהָר – כִּי מַשֶּׁהוּ מוֹשֵׁךְ אוֹתָם לְשָׁם.

׳וְנָהֲרוּ׳ – הָעַמִּים יִמָּשְׁכוּ אֶל הַר ה׳ כְּמוֹ נָהָר, בְּלִי שֶׁמִּישֶׁהוּ יַכְרִיחַ אוֹתָם.

🏠

📜 הַפָּסוּקוְהָלְכוּ עַמִּים רַבִּים וְאָמְרוּ לְכוּ וְנַעֲלֶה אֶל הַר ה׳ אֶל בֵּית אֱלֹהֵי יַעֲקֹב

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁיִּהְיֶה לָנוּ אִתּוֹ בַּיִת, לֹא רַק מָקוֹם לְבַקֵּר בּוֹ 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ב׳, פָּסוּק ג׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

🏠 רַשִׁ״י: ׳אֶל בֵּית אֱלֹהֵי יַעֲקֹב׳

לָמָּה כָּתוּב ׳בֵּית אֱלֹהֵי יַעֲקֹב׳, וְלֹא אַבְרָהָם אוֹ יִצְחָק? רַשִׁ״י מְבָאֵר: ׳לְפִי שֶׁהוּא קְרָאוֹ בֵּית אֵל, לְפִיכָךְ יִקָּרֵא עַל שְׁמוֹ. אֲבָל אַבְרָהָם קְרָאוֹ הַר... יִצְחָק קְרָאוֹ שָׂדֶה׳. יַעֲקֹב רָאָה בַּמָּקוֹם הַזֶּה בַּיִת – מָקוֹם שֶׁגָּרִים בּוֹ יַחַד, כְּמוֹ מִשְׁפָּחָה.

אַבְרָהָם קָרָא לַמָּקוֹם ׳הַר׳, יִצְחָק ׳שָׂדֶה׳, וְיַעֲקֹב ׳בַּיִת׳ – וְעַל שְׁמוֹ נִקְרָא ׳בֵּית אֱלֹהֵי יַעֲקֹב׳.

⚖️

📜 הַפָּסוּקוְשָׁפַט בֵּין הַגּוֹיִם וְהוֹכִיחַ לְעַמִּים רַבִּים וְכִתְּתוּ חַרְבוֹתָם לְאִתִּים וַחֲנִיתוֹתֵיהֶם לְמַזְמֵרוֹת לֹא יִשָּׂא גוֹי אֶל גּוֹי חֶרֶב וְלֹא יִלְמְדוּ עוֹד מִלְחָמָה

💛 אַהֲבַת הַשֵּׁם: ה׳ יָבִיא לָעוֹלָם מִשְׁפָּט צוֹדֵק, וְאָז יִהְיֶה שָׁלוֹם אֲמִתִּי 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ב׳, פָּסוּק ד׳ (מְצוּדַת דָּוִד וּמְצוּדַת צִיּוֹן)

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

⚖️ מְצוּדַת דָּוִד: לָמָּה לֹא תִּהְיֶה מִלְחָמָה?

מְצוּדַת דָּוִד מְבָאֵר אֶת ׳וְשָׁפַט בֵּין הַגּוֹיִם׳: ׳הַשּׁוֹפֵט וְהוּא מֶלֶךְ הַמָּשִׁיחַ הוּא יִשְׁפֹּט בֵּין הָאֻמּוֹת בְּדִבְרֵי רִיבוֹתָם׳. כְּשֶׁיֵּשׁ שׁוֹפֵט צוֹדֵק שֶׁכֻּלָּם סוֹמְכִים עָלָיו, לֹא צָרִיךְ לָרִיב בְּחֶרֶב. וּמְצוּדַת צִיּוֹן מְבָאֵר אֶת הַמִּלִּים: ׳אִתִּים – כְּלֵי חֲפִירָה׳, ׳מַזְמֵרוֹת – הַכְּלִי שֶׁכּוֹרְתִין בּוֹ הַזְּמוֹרוֹת׳.

הַמָּשִׁיחַ יִשְׁפֹּט בְּצֶדֶק בֵּין הָעַמִּים, וְלָכֵן הַחֲרָבוֹת יֵהָפְכוּ לְאִתִּים וּלְמַזְמֵרוֹת.

📜 הַפָּסוּקבֵּית יַעֲקֹב לְכוּ וְנֵלְכָה בְּאוֹר ה׳

💛 אַהֲבַת הַשֵּׁם: גַּם כְּשֶׁה׳ מוֹכִיחַ אוֹתִי, הוּא מַרְאֶה לִי קֹדֶם לְאָן אֲנִי יָכוֹל לְהַגִּיעַ 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ב׳, פָּסוּק ה׳ (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

🌅 חָזוֹן ← תּוֹכָחָה ← נֶחָמָה

הַנְּבוּאָה בְּנוּיָה בִּשְׁלֹשָׁה חֲלָקִים: קֹדֶם חָזוֹן נִפְלָא (ב׳ ב׳–ה׳), אַחֲרָיו תּוֹכָחָה עַל הַגַּאֲוָה וְהַחֵטְא (ב׳ ו׳ – ג׳ ט״ו), וּבַסּוֹף נֶחָמָה (ד׳ ב׳–ו׳). וְהַגֶּשֶׁר הוּא פָּסוּק ה׳. מְצוּדַת דָּוִד: ׳הַנָּבִיא מְזָרְזָם לוֹמַר: הוֹאִיל וְכֵן יִהְיֶה, נֵלְכָה נָא בְּאוֹר ה׳, רְצוֹנוֹ לוֹמַר בְּאוֹר הַתּוֹרָה וְהַמִּצְוָה, כִּי יֵשׁ תִּקְוָה׳.

יְשַׁעְיָהוּ מַתְחִיל בְּחָזוֹן, מוֹכִיחַ, וּמְסַיֵּם בְּנֶחָמָה – ׳כִּי יֵשׁ תִּקְוָה׳.

🗼

📜 הַפָּסוּקעֵינֵי גַּבְהוּת אָדָם שָׁפֵל וְשַׁח רוּם אֲנָשִׁים וְנִשְׂגַּב ה׳ לְבַדּוֹ בַּיּוֹם הַהוּא

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא צָרִיךְ לְהִתְגָּאוֹת כְּדֵי לִהְיוֹת חָשׁוּב – אֲנִי חָשׁוּב לַה׳ 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ב׳, פְּסוּקִים י״א–י״ז (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

🗼 הַמִּלָּה הַמַּנְחָה: רָם וְשָׁפֵל

בְּפֶרֶק ב׳ חוֹזְרוֹת מִלִּים שֶׁל גֹּבַהּ: ׳גֵּאֶה וָרָם׳, ׳הָרָמִים וְהַנִּשָּׂאִים׳, ׳מִגְדָּל גָּבֹהַּ׳ – וּמִלִּים שֶׁל נְמִיכוּת: ׳וְשַׁח׳, ׳וְשָׁפֵל׳. רַשִׁ״י מְבָאֵר: ׳עֵינֵי גַבְהוּת אָדָם – גַּסּוּת הָרוּחַ׳. וּמְצוּדַת דָּוִד עַל ׳וְנִשְׂגַּב ה׳ לְבַדּוֹ׳: ׳אָז יִוָּדַע לַכֹּל שֶׁהַכֹּחַ הוּא לַה׳ לְבַדּוֹ׳.

כָּל מָה שֶׁ׳רָם׳ מִתּוֹךְ גַּאֲוָה יֵרֵד, וְיִוָּדַע ׳שֶׁהַכֹּחַ הוּא לַה׳ לְבַדּוֹ׳.

📜 הַפָּסוּקכִּי יִתְפֹּשׂ אִישׁ בְּאָחִיו בֵּית אָבִיו שִׂמְלָה לְכָה קָצִין תִּהְיֶה לָּנוּ וְהַמַּכְשֵׁלָה הַזֹּאת תַּחַת יָדֶךָ

💛 אַהֲבַת הַשֵּׁם: כָּל מָה שֶׁאֲנִי לוֹמֵד הַיּוֹם יָכוֹל לְהַאִיר לַאֲחֵרִים מָחָר 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ג׳, פְּסוּקִים ו׳–ז׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בֶּחָזוֹן

🧥 ׳שִׂמְלָה לְכָה – קָצִין תִּהְיֶה לָּנוּ׳

יְשַׁעְיָהוּ מְתָאֵר זְמַן שֶׁאֵין בּוֹ מַנְהִיגִים: אִישׁ תּוֹפֵס אֶת קְרוֹבוֹ וְאוֹמֵר: ׳שִׂמְלָה לְכָה קָצִין תִּהְיֶה לָּנוּ׳. וְהַלָּה עוֹנֶה: ׳לֹא אֶהְיֶה חֹבֵשׁ וּבְבֵיתִי אֵין לֶחֶם וְאֵין שִׂמְלָה׳. רַשִׁ״י מְבָאֵר בְּדֶרֶךְ הַדְּרָשׁ: ׳עָשִׁיר אַתָּה בַּתּוֹרָה וּמְחֻוֶּרֶת הִיא בְּיָדְךָ כְּשִׂמְלָה׳ – וְהוּא עוֹנֶה שֶׁאֵין בְּיָדוֹ ׳לֹא טַעַם מִשְׁנָה וְלֹא טַעַם אַגָּדָה׳.

כְּשֶׁאֵין לוֹמְדֵי תּוֹרָה, אֵין מִי שֶׁיַּנְהִיג; הַתּוֹרָה הִיא הַ׳שִּׂמְלָה׳ שֶׁל הַמַּנְהִיג.

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מְגַלִּים כַּמָּה אַהֲבָה יֵשׁ בֶּחָזוֹן: הָעַמִּים ׳נוֹהֲרִים׳ אֶל הָהָר כְּמוֹ נָהָר, ה׳ רוֹצֶה ׳בַּיִת׳ אִתָּנוּ כְּמוֹ שֶׁיַּעֲקֹב קָרָא לוֹ, וְאַחֲרֵי הַתּוֹכָחָה תָּמִיד בָּאָה תִּקְוָה – ׳נֵלְכָה נָא בְּאוֹר ה׳... כִּי יֵשׁ תִּקְוָה׳.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • יְשַׁעְיָהוּ אוֹמֵר ׳לְכוּ וְנֵלְכָה בְּאוֹר ה׳׳ – לֹא ׳לֵךְ׳ אֶלָּא ׳נֵלְכָה׳, כֻּלָּנוּ יַחַד. אֵיזֶה דָּבָר טוֹב אֲנִי יָכוֹל לְהַזְמִין חָבֵר לַעֲשׂוֹת אִתִּי הַשָּׁבוּעַ?
  • הָעַמִּים אוֹמְרִים ׳וְיֹרֵנוּ מִדְּרָכָיו׳ – הֵם רוֹצִים לִלְמֹד. מָתַי בְּפַעַם הָאַחֲרוֹנָה לָמַדְתִּי תּוֹרָה כִּי רָצִיתִי, וְלֹא כִּי הָיִיתִי צָרִיךְ?
  • ׳כִּי מִצִּיּוֹן תֵּצֵא תוֹרָה׳ מִתְקַיֵּם בְּדוֹרֵנוּ. אֵיךְ אֲנִי יָכוֹל לִהְיוֹת חֵלֶק מִזֶּה – וְעַל מָה אֲנִי רוֹצֶה לְהוֹדוֹת לַה׳ כְּשֶׁאֲנִי שָׁר אֶת הַפָּסוּק בְּבֵית הַכְּנֶסֶת?

רָמָה ג׳ – מַעֲמִיקִים

🏘️ ׳לְיַשֵּׁב אֶת הָאֲדָמָה וְלֹא אֶת הָאֲוִיר׳

הָרַב לִמֵּד: מִצְוַת יִשּׁוּב אֶרֶץ יִשְׂרָאֵל – ״וִישַׁבְתֶּם בָּהּ״ – אֵינָהּ שֶׁכֻּלָּם יִתְקַבְּצוּ בְּמָקוֹם אֶחָד וְיִבְנוּ בָּתִּים לְגוֹבַהּ שְׁלוֹשִׁים קוֹמוֹת. ׳הַמִּצְוָה הִיא לְיַשֵּׁב אֶת הָאֲדָמָה, לֹא אֶת הָאֲוִיר׳. הַמָּקוֹר: ׳הַרְחִיבִי מְקוֹם אָהֳלֵךְ... כִּי יָמִין וּשְׂמֹאול תִּפְרֹצִי׳. וְשִׂימוּ לֵב, אָמַר הָרַב: לֹא כָּתוּב ״הַרְחִיבִי גְּבוּלֵךְ״ אֶלָּא ״מְקוֹם אָהֳלֵךְ״ – מְקוֹם הַמְּגוּרִים. אִם הַמְּגוּרִים רַק בִּירוּשָׁלַיִם, מַרְחִיבִים אוֹתָם לִסְבִיבוֹת יְרוּשָׁלַיִם, וּמִשָּׁם לֶהָרִים, עַד קְצוֹת גְּבוּלוֹת יִשְׂרָאֵל.

מִצְוַת הַיִּשּׁוּב: לְהִתְפָּרֵשׂ בָּאָרֶץ – ׳הַרְחִיבִי מְקוֹם אָהֳלֵךְ׳ – וְלֹא רַק לִבְנוֹת לְגוֹבַהּ.

🌱 הַשְּׁכִינָה מִתְרַחֶבֶת עִם הַיִּשּׁוּב

לָמָּה חֲשׁוּבָה כָּל כָּךְ הַרְחָבַת הַיִּשּׁוּב? הָרַב הִסְבִּיר: אֶרֶץ יִשְׂרָאֵל הִיא מְקוֹם הַשְּׁכִינָה, וְכָל מֵרְחָב מְיֻשָּׁב מוֹסִיף מָקוֹם שֶׁבּוֹ שׁוֹרָה הַשְּׁכִינָה. מִי שֶׁמַּפְרִיחַ אֶת הָאָרֶץ, כִּבְיָכוֹל מַרְחִיב אֶת הַשְׁרָאַת הַשְּׁכִינָה בָּהּ. וְהֵבִיא אֶת יִצְחָק אָבִינוּ, שֶׁחָפַר בְּאֵר וְקָרָא לָהּ ״רְחֹבוֹת״ וְאָמַר: ״כִּי עַתָּה הִרְחִיב ה׳ לָנוּ וּפָרִינוּ בָאָרֶץ״ – הַבְּרָכָה שֶׁל ״וּפָרִינוּ״ תְּלוּיָה בְּהַרְחָבָה. כָּל בַּיִת חָדָשׁ, שָׂדֶה חָדָשׁ, וּשְׁכוּנָה חֲדָשָׁה – מוֹסִיפִים מָקוֹם שֶׁשּׁוֹרָה בּוֹ הַשְּׁכִינָה.

כָּל בַּיִת, שָׂדֶה וּשְׁכוּנָה חֲדָשִׁים בָּאָרֶץ מוֹסִיפִים מָקוֹם לַשְּׁכִינָה – עֲבוֹדַת קֹדֶשׁ.

רָמָה ג׳ – מַעֲמִיקִים

🛠️ ״אֲנַחְנוּ רַק מְקַטְרְגִים״ – בּוֹנִים

הָרַב סִפֵּר עַל אַדְמוֹ״ר שֶׁבָּא לָאָרֶץ אַחֲרֵי הַשּׁוֹאָה וְחָלַם לִבְנוֹת קִרְיָה חֲדָשָׁה לַחֲסִידָיו. כְּשֶׁהִצִּיעוּ לוֹ לִבְנוֹת בִּירוּשָׁלַיִם אוֹ בִּבְנֵי בְּרַק, דָּחָה: ׳בִּירוּשָׁלַיִם יֵשׁ קְדֻשָּׁה וְתוֹרָה בְּשֶׁפַע; אֲנִי רוֹצֶה לְהוֹסִיף בָּאָרֶץ מָקוֹם חָדָשׁ׳. הוּא גַּם אָמַר מִשְׁפָּט מְפֹרְסָם: ׳אֲנַחְנוּ הַיְרֵאִים עוֹסְקִים בְּקִטְרוּג בִּלְבָד, וְהַבּוֹנִים הֵם אֲחֵרִים. לֹא עוֹד – נִפְעַל בְּכָל כֹּחֵנוּ לְבִנְיַן אַרְצֵנוּ הַקְּדוֹשָׁה׳. הָרַב אָמַר: קַל לַעֲמֹד מִן הַצַּד וּלְבַקֵּר, וְהַרְבֵּה יוֹתֵר חָשׁוּב לְהַפְשִׁיל שַׁרְווּלִים וְלִבְנוֹת. בְּדִיּוּק ׳הַרְחִיבִי מְקוֹם אָהֳלֵךְ׳ – לֹא רַק לִדְבֹּר, לְהַרְחִיב.

לֹא רַק מְבַקְּרִים וְנוֹזְפִים, אֶלָּא בּוֹנִים וּמַרְחִיבִים – ׳הַרְחִיבִי מְקוֹם אָהֳלֵךְ׳.

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי מוֹדֶה לַה׳ עַל הַטּוֹב, אֲנִי מוֹשֵׁךְ אֵלַי עוֹד חֶסֶד שֶׁלֹּא יָזוּז 💛

רָמָה ג׳ – מַעֲמִיקִים

🌈 הַכָּרַת הַטּוֹב וּמֵי נֹחַ

הָרַב הִסְבִּיר שֶׁהַהַבְטָחָה ׳כִּי מֵי נֹחַ זֹאת לִי׳ קְשׁוּרָה לְהַכָּרַת הַטּוֹב: הַכָּרַת הַטּוֹב שֶׁל נֹחַ – שֶׁהוֹדָה לַה׳ אַחֲרֵי הַמַּבּוּל – הִיא זוֹ שֶׁהִתִּירָה לוֹ לֶאֱכֹל בָּשָׂר, וְהִיא זוֹ שֶׁהִצִּילָה אֶת כָּל הַבְּרִיאָה מִמֵּי הַמַּבּוּל, וּבִזְכוּתָהּ כָּרַת ה׳ בְּרִית עִם הַבְּרִיאָה כֻּלָּהּ לְדוֹרֵי דוֹרוֹת. וְכַתּוּב שֶׁם: ״כְּשֵׁם שֶׁחוֹסֶר הַכָּרַת הַטּוֹב מוֹנֵעַ חֶסֶד, כָּךְ הַכָּרַת הַטּוֹב מַרְבָּה אוֹתוֹ״. מִכָּאן: מִי שֶׁמּוֹדֶה לַה׳ מַרְבֶּה אֶת הַחֶסֶד שֶׁבָּא אֵלָיו.

הַכָּרַת הַטּוֹב מַרְבָּה חֶסֶד – וּבִזְכוּתָהּ יֵשׁ בְּרִית שֶׁלֹּא תִּתְבַּטֵּל, כְּמוֹ בֵּימֵי נֹחַ.

רָמָה ג׳ – מַעֲמִיקִים

📖 מָה יְשַׁעְיָהוּ מְלַמֵּד אוֹתִי הַיּוֹם

יְשַׁעְיָהוּ מְלַמֵּד אוֹתִי: הַקֹּשִׁי הוּא רֶגַע וְהָרַחֲמִים גְּדוֹלִים; חַסְדּוֹ שֶׁל ה׳ חָזָק מֵהֶהָרִים וְלֹא יָזוּז גַּם כְּשֶׁאֲנִי לֹא מַגִּיעַ; וּכְשֶׁיֵּשׁ הַרְבֵּה בָּנִים וְהַמָּקוֹם צַר – מַרְחִיבִים אֶת הָאֹהֶל וְלֹא חוֹסְכִים. אֲנִי רוֹצֶה לִהְיוֹת מִי שֶׁבּוֹנֶה וּמַרְחִיב, מִי שֶׁמּוֹדֶה וּמִי שֶׁלּוֹמֵד תּוֹרָה, כְּדֵי שֶׁיִּהְיֶה שָׁלוֹם רַב.

הַקֹּשִׁי רֶגַע וְהַחֶסֶד לְעוֹלָם; בּוֹנִים וּמַרְחִיבִים, מוֹדִים וְלוֹמְדִים.

סוֹף רָמָה ג׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳בְּרֶגַע קָטֹן עֲזַבְתִּיךְ וּבְרַחֲמִים גְּדֹלִים אֲקַבְּצֵךְ׳ – הַקֹּשִׁי הוּא רֶגַע, וְהָרַחֲמִים גְּדוֹלִים. מָתַי הִרְגַּשְׁתִּי שֶׁה׳ הִרְחִיב אֶת הַלֵּב שֶׁלִּי אַחֲרֵי זְמַן קָשֶׁה?

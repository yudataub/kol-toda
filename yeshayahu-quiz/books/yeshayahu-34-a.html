טֶרֶם יִקְרָאוּ וַאֲנִי אֶעֱנֶה

רָמָה א׳ – עוֹלָם חָדָשׁ

ה׳ פּוֹרֵס יָדַיִם כָּל הַיּוֹם, מְחַדֵּשׁ שָׁמַיִם וָאָרֶץ, עוֹנֶה עוֹד לִפְנֵי שֶׁקּוֹרְאִים, וּמַבִּיט אֶל הֶעָנִי וְהֶחָרֵד עַל דְּבָרוֹ – וַאֲנַחְנוּ נָבוֹא לְהִשְׁתַּחֲווֹת לְפָנָיו מִדֵּי חֹדֶשׁ וּמִדֵּי שַׁבָּת

📖 יְשַׁעְיָהוּ, פֶּרֶק ס״ה, פְּסוּקִים א׳–ב׳

רָמָה א׳ – עוֹלָם חָדָשׁ

🤲 ׳פֵּרַשְׂתִּי יָדַי כָּל הַיּוֹם׳

ה׳ אוֹמֵר: ׳נִדְרַשְׁתִּי לְלוֹא שָׁאָלוּ נִמְצֵאתִי לְלֹא בִקְשֻׁנִי אָמַרְתִּי הִנֵּנִי הִנֵּנִי׳. וְהוּא מוֹסִיף: ׳פֵּרַשְׂתִּי יָדַי כָּל הַיּוֹם׳. רַשִׁ״י מְבָאֵר: ׳פֵּרַשְׂתִּי יָדַי – כְּדֵי לְקַבְּלָם בִּתְשׁוּבָה׳, וְעַל ׳הִנֵּנִי הִנֵּנִי׳: ׳שׁוּבוּ אֵלַי וְהִנְנִי מוּכָן לְקַבֵּל׳. ה׳ מוּכָן וּפָתוּחַ, עוֹד לִפְנֵי שֶׁמְּבַקְּשִׁים.

ה׳ פּוֹרֵס יָדַיִם כָּל הַיּוֹם וְאוֹמֵר ׳הִנֵּנִי׳ – מוּכָן לְקַבֵּל תְּשׁוּבָה.

1

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹאֶה בִּי אֶת הַגַּרְגֵּר הַמָּתוֹק, וְלָכֵן שׁוֹמֵר עָלַי 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ס״ה, פְּסוּקִים ח׳–י׳

רָמָה א׳ – עוֹלָם חָדָשׁ

🍇 ׳אַל תַּשְׁחִיתֵהוּ כִּי בְרָכָה בּוֹ׳

ה׳ אוֹמֵר: ׳כַּאֲשֶׁר יִמָּצֵא הַתִּירוֹשׁ בָּאֶשְׁכּוֹל וְאָמַר אַל תַּשְׁחִיתֵהוּ כִּי בְרָכָה בּוֹ כֵּן אֶעֱשֶׂה לְמַעַן עֲבָדַי לְבִלְתִּי הַשְׁחִית הַכֹּל׳. מִי שֶׁמּוֹצֵא אֶשְׁכּוֹל וְרוֹאֶה בּוֹ אֲפִלּוּ גַּרְגֵּר אֶחָד מָתוֹק, אוֹמֵר: אַל תַּשְׁחִיתוּ! יֵשׁ בּוֹ בְּרָכָה. כָּךְ ה׳ שׁוֹמֵר עַל כֻּלָּם בִּזְכוּת הַצַּדִּיקִים. וְהוּא מַבְטִיחַ: ׳וְהוֹצֵאתִי מִיַּעֲקֹב זֶרַע׳ וְ׳וְהָיָה הַשָּׁרוֹן לִנְוֵה צֹאן׳.

בִּזְכוּת הַצַּדִּיקִים שֶׁבְּתוֹכָם ה׳ שׁוֹמֵר עַל כֻּלָּם, כְּמוֹ אֶשְׁכּוֹל שֶׁיֵּשׁ בּוֹ בְּרָכָה.

2

💛 אַהֲבַת הַשֵּׁם: ה׳ שָׂמֵחַ בְּעַמּוֹ, וַאֲנִי חֵלֶק מִן הַשִּׂמְחָה שֶׁלּוֹ 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ס״ה, פְּסוּקִים י״ז–י״ט

רָמָה א׳ – עוֹלָם חָדָשׁ

🌠 ׳הִנְנִי בוֹרֵא שָׁמַיִם חֲדָשִׁים׳

ה׳ מַבְטִיחַ: ׳כִּי הִנְנִי בוֹרֵא שָׁמַיִם חֲדָשִׁים וָאָרֶץ חֲדָשָׁה וְלֹא תִזָּכַרְנָה הָרִאשֹׁנוֹת וְלֹא תַעֲלֶינָה עַל לֵב׳. ׳כִּי אִם שִׂישׂוּ וְגִילוּ עֲדֵי עַד אֲשֶׁר אֲנִי בוֹרֵא כִּי הִנְנִי בוֹרֵא אֶת יְרוּשָׁלִַם גִּילָה וְעַמָּהּ מָשׂוֹשׂ׳. ׳וְגַלְתִּי בִירוּשָׁלִַם וְשַׂשְׂתִּי בְעַמִּי וְלֹא יִשָּׁמַע בָּהּ עוֹד קוֹל בְּכִי׳. ה׳ עַצְמוֹ שָׂמֵחַ בִּירוּשָׁלַיִם וּבְעַמּוֹ.

עוֹלָם חָדָשׁ שֶׁבּוֹ הַצָּרוֹת נִשְׁכָּחוֹת, וה׳ עַצְמוֹ שָׂמֵחַ בִּירוּשָׁלַיִם.

3

💛 אַהֲבַת הַשֵּׁם: ה׳ לֹא נוֹתֵן לַעֲמָלִי לֶאֱבֹד לָרִיק 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ס״ה, פְּסוּקִים כ״א–כ״ג

רָמָה א׳ – עוֹלָם חָדָשׁ

🏡 ׳וּבָנוּ בָתִּים וְיָשָׁבוּ׳

הַנְּבוּאָה מְתָאֶרֶת חַיִּים שֶׁל שְׁלֵוָה: ׳וּבָנוּ בָתִּים וְיָשָׁבוּ וְנָטְעוּ כְרָמִים וְאָכְלוּ פִּרְיָם לֹא יִבְנוּ וְאַחֵר יֵשֵׁב לֹא יִטְּעוּ וְאַחֵר יֹאכֵל׳. כָּל אֶחָד נֶהֱנֶה מִן הַפְּרִי שֶׁל עֲמָלוֹ. ׳כִּי כִימֵי הָעֵץ יְמֵי עַמִּי׳ – חַיִּים אֲרֻכִּים כְּמוֹ עֵץ. ׳לֹא יִיגְעוּ לָרִיק כִּי זֶרַע בְּרוּכֵי ה׳ הֵמָּה׳.

כָּל אֶחָד בּוֹנֶה וְנוֹטֵעַ וְנֶהֱנֶה מִפְּרִי עֲמָלוֹ – ׳לֹא יִיגְעוּ לָרִיק׳.

4

💛 אַהֲבַת הַשֵּׁם: ה׳ יוֹדֵעַ מָה אֲנִי צָרִיךְ עוֹד לִפְנֵי שֶׁאֲנִי מְבַקֵּשׁ 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ס״ה, פָּסוּק כ״ד

רָמָה א׳ – עוֹלָם חָדָשׁ

💬 ׳טֶרֶם יִקְרָאוּ וַאֲנִי אֶעֱנֶה׳

זֶה הַפָּסוּק הַיָּפֶה בְּיוֹתֵר: ׳וְהָיָה טֶרֶם יִקְרָאוּ וַאֲנִי אֶעֱנֶה עוֹד הֵם מְדַבְּרִים וַאֲנִי אֶשְׁמָע׳. מְצוּדַת דָּוִד: ׳פְּעָמִים אֶעֱנֶה לָהֶם לְמַלֵּא שְׁאֵלוֹת לִבָּם עַד לֹא יִקְרְאוּ לְהִתְפַּלֵּל בַּפֶּה, פְּעָמִים כְּשֶׁעוֹד הֵם מְדַבְּרִים בִּתְפִלָּה אֶשְׁמַע׳. ה׳ קָרוֹב כָּל כָּךְ, שֶׁהוּא עוֹנֶה עוֹד לִפְנֵי שֶׁסִּיַּמְנוּ לְבַקֵּשׁ.

ה׳ עוֹנֶה לִפְנֵי שֶׁקּוֹרְאִים, וְשׁוֹמֵעַ עוֹד כְּשֶׁמְּדַבְּרִים.

5

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה עוֹלָם שֶׁל שָׁלוֹם, וַאֲנִי יָכוֹל לְהַתְחִיל אוֹתוֹ הַיּוֹם 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ס״ה, פָּסוּק כ״ה

רָמָה א׳ – עוֹלָם חָדָשׁ

🐑 ׳זְאֵב וְטָלֶה יִרְעוּ כְאֶחָד׳

הַנָּבִיא מְצַיֵּר שָׁלוֹם: ׳זְאֵב וְטָלֶה יִרְעוּ כְאֶחָד וְאַרְיֵה כַּבָּקָר יֹאכַל תֶּבֶן וְנָחָשׁ עָפָר לַחְמוֹ לֹא יָרֵעוּ וְלֹא יַשְׁחִיתוּ בְּכָל הַר קָדְשִׁי אָמַר ה׳׳. רַשִׁ״י: ׳יֹאכַל תֶּבֶן – וְלֹא יִצְטָרֵךְ לְהַשְׁחִית בַּבְּהֵמָה׳. הַשָּׁלוֹם בָּא לְכָל הַבְּרִיאָה, וּבְכָל הַר קָדְשִׁי.

יָבוֹא שָׁלוֹם גָּדוֹל, שֶׁבּוֹ אֲפִלּוּ הַזְּאֵב וְהַטָּלֶה יַחְדָּו.

6

💛 אַהֲבַת הַשֵּׁם: ה׳ מַבִּיט אֵלַי כְּשֶׁאֲנִי עָנָו וּמַקְשִׁיב לִדְבָרָיו 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ס״ו, פְּסוּקִים א׳–ב׳

רָמָה א׳ – עוֹלָם חָדָשׁ

🌌 ׳וְאֶל זֶה אַבִּיט אֶל עָנִי׳

בְּסוֹף הַסֵּפֶר ה׳ אוֹמֵר: ׳הַשָּׁמַיִם כִּסְאִי וְהָאָרֶץ הֲדֹם רַגְלָי אֵי זֶה בַיִת אֲשֶׁר תִּבְנוּ לִי׳. ה׳ גָּדוֹל מִכָּל בַּיִת, וְעִם זֶה: ׳וְאֶל זֶה אַבִּיט אֶל עָנִי וּנְכֵה רוּחַ וְחָרֵד עַל דְּבָרִי׳. מְצוּדַת דָּוִד: ׳וְעִם שֶׁאֲנִי רָם עַל רָמִים עַכְשָׁיו אַבִּיט וְאַשְׁגִּיחַ בַּשְּׁפָלִים׳.

ה׳ מַבִּיט דַּוְקָא אֶל הֶעָנִי, שְׁבַר הָרוּחַ וְהֶחָרֵד עַל דְּבָרוֹ.

7

💛 אַהֲבַת הַשֵּׁם: ה׳ מְנַחֵם אוֹתִי כְּמוֹ שֶׁהוֹרֶה אוֹהֵב מְנַחֵם אֶת יַלְדּוֹ 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ס״ו, פְּסוּקִים י׳–י״ד

רָמָה א׳ – עוֹלָם חָדָשׁ

🕊️ ׳שִׂמְחוּ אֶת יְרוּשָׁלִַם׳

בַּסּוֹף הַנָּבִיא קוֹרֵא לְכֻלָּנוּ: ׳שִׂמְחוּ אֶת יְרוּשָׁלִַם וְגִילוּ בָהּ כָּל אֹהֲבֶיהָ שִׂישׂוּ אִתָּהּ מָשׂוֹשׂ כָּל הַמִּתְאַבְּלִים עָלֶיהָ׳. ה׳ מַבְטִיחַ: ׳הִנְנִי נֹטֶה אֵלֶיהָ כְּנָהָר שָׁלוֹם׳. וְהַנֶּחָמָה הַגְּדוֹלָה: ׳כְּאִישׁ אֲשֶׁר אִמּוֹ תְּנַחֲמֶנּוּ כֵּן אָנֹכִי אֲנַחֶמְכֶם וּבִירוּשָׁלִַם תְּנֻחָמוּ׳. ׳וּרְאִיתֶם וְשָׂשׂ לִבְּכֶם וְעַצְמוֹתֵיכֶם כַּדֶּשֶׁא תִפְרַחְנָה׳.

ה׳ מְנַחֵם אוֹתָנוּ בִּירוּשָׁלַיִם כְּמוֹ מְנַחֶמֶת אִמָּא, וְנוֹטֶה אֵלֶיהָ שָׁלוֹם כְּמוֹ נָהָר.

💡

רָמָה א׳ – עוֹלָם חָדָשׁ

מָה לוֹקְחִים מִרָמָה א׳?

סוֹף הַסֵּפֶר מְלֵא נֶחָמָה: ה׳ פָּרַשׂ יָדַיִם כָּל הַיּוֹם, הוּא מְחַדֵּשׁ שָׁמַיִם וָאָרֶץ, הַשִּׂמְחָה בִּירוּשָׁלַיִם נִמְשֶׁכֶת, וְהוּא עוֹנֶה ׳טֶרֶם יִקְרָאוּ׳. זֶה הָעוֹלָם שֶׁאֲנַחְנוּ מִתְפַּלְּלִים אֵלָיו.

  • ׳וְאֶל זֶה אַבִּיט אֶל עָנִי וּנְכֵה רוּחַ וְחָרֵד עַל דְּבָרִי׳ – מָה אֲנִי יָכוֹל לַעֲשׂוֹת כְּדֵי לִהְיוֹת כָּזֶה, שֶׁה׳ מַבִּיט אֵלָיו?
  • ׳זְאֵב וְטָלֶה יִרְעוּ כְאֶחָד׳ – עִם מִי קָשֶׁה לִי לִהְיוֹת ׳יַחַד׳, וְאֵיךְ אֲנִי יָכוֹל לַעֲשׂוֹת צַעַד קָטָן לְשָׁלוֹם?

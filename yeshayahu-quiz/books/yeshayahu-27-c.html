דִּרְשׁוּ ה׳ בְּהִמָּצְאוֹ

רָמָה ג׳ – מַעֲמִיקִים

יְשַׁעְיָהוּ מַזְמִין אֶת כֻּלָּנוּ לָבוֹא וְלִשְׁתּוֹת בְּחִנָּם מִמַּיִם חַיִּים, וּמְגַלֶּה לָנוּ שֶׁה׳ קָרוֹב וּמוּכָן לִסְלֹחַ הַרְבֵּה

📖 פֶּרֶק נ״ה

🎯 לַחִידוֹן

📜 הַפָּסוּקדִּרְשׁוּ ה׳ בְּהִמָּצְאוֹ קְרָאֻהוּ בִּהְיוֹתוֹ קָרוֹב

💛 אַהֲבַת הַשֵּׁם: כָּל שָׁנָה ה׳ נוֹתֵן לִי שְׁנֵי זְמַנִּים מְיֻחָדִים שֶׁבָּהֶם הוּא קָרוֹב אֵלַי יוֹתֵר 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק נ״ה, פָּסוּק ו׳ (חֲזַ״ל)

רָמָה ג׳ – מַעֲמִיקִים

📅 ׳דִּרְשׁוּ ה׳ בְּהִמָּצְאוֹ׳ – עֲשֶׂרֶת יְמֵי תְּשׁוּבָה וֶאֱלוּל

חֲזַ״ל דּוֹרְשִׁים: ׳דִּרְשׁוּ ה׳ בְּהִמָּצְאוֹ׳ – אֵלּוּ עֲשֶׂרֶת הַיָּמִים שֶׁבֵּין רֹאשׁ הַשָּׁנָה לְיוֹם הַכִּפּוּרִים. וְהַ׳פָּנִים יָפוֹת׳ (בַּעַל הַהַפְלָאָה) שׁוֹאֵל: אִם אֵלּוּ הַיָּמִים שֶׁבָּהֶם דּוֹרְשִׁים, מָתַי ׳קְרָאֻהוּ בִּהְיוֹתוֹ קָרוֹב׳? וְהוּא עוֹנֶה: זֶה חֹדֶשׁ אֱלוּל – בּוֹ ה׳ קָרוֹב אֵלֵינוּ. כָּךְ לָמַד הָרַב בְּשִׁיעוּר עַל חֹדֶשׁ אֱלוּל: כְּבָר בֶּאֱלוּל הַלֵּב מַרְגִּישׁ ׳וַאֲנִי קִרְבַת אֱלֹהִים לִי טוֹב׳, וְאֵין צוֹרֶךְ לְחַכּוֹת לְרֹאשׁ הַשָּׁנָה.

בֶּאֱלוּל ה׳ ׳קָרוֹב׳, וּבַעֲשֶׂרֶת יְמֵי תְּשׁוּבָה ׳דּוֹרְשִׁים׳ אוֹתוֹ – שְׁנֵי זְמַנִּים שֶׁל קִרְבָה.

1

📜 הַפָּסוּקיַעֲזֹב רָשָׁע דַּרְכּוֹ וְאִישׁ אָוֶן מַחְשְׁבֹתָיו וְיָשֹׁב אֶל־ה׳ וִירַחֲמֵהוּ וְאֶל־אֱלֹהֵינוּ כִּי־יַרְבֶּה לִסְלוֹחַ

💛 אַהֲבַת הַשֵּׁם: בְּכָל פַּעַם שֶׁיּוֹרֵד גֶּשֶׁם אֲנִי יָכוֹל לִזְכֹּר שֶׁה׳ סוֹלֵחַ לִי וְאוֹהֵב אוֹתִי 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק נ״ה, פְּסוּקִים ז׳, י׳ (תַּעֲנִית ז׳ ע״ב)

רָמָה ג׳ – מַעֲמִיקִים

🌧️ ׳אֵין הַגְּשָׁמִים יוֹרְדִים׳ – גֶּשֶׁם וּסְלִיחָה

בַּפֶּרֶק הַגֶּשֶׁם וְהַסְּלִיחָה בָּאִים יַחַד: ׳כִּי יַרְבֶּה לִסְלוֹחַ׳ וְאַחַר כָּךְ ׳כַּאֲשֶׁר יֵרֵד הַגֶּשֶׁם׳. חֲזַ״ל אוֹמְרִים (תַּעֲנִית ז׳ ע״ב): ׳אֵין הַגְּשָׁמִים יוֹרְדִים אֶלָּא אִם כֵּן נִמְחֲלוּ עֲוֹנוֹתֵיהֶן שֶׁל יִשְׂרָאֵל׳, שֶׁנֶּאֱמַר ׳רָצִיתָ ה׳ אַרְצֶךָ... נָשָׂאתָ עֲוֺן עַמֶּךָ׳. כְּשֶׁיּוֹרֵד גֶּשֶׁם, זֶה סִימָן שֶׁה׳ רוֹצֶה בָּנוּ, סוֹלֵחַ לָנוּ וּמַשְׁפִּיעַ עָלֵינוּ טוֹב.

הַגֶּשֶׁם הוּא סִימָן לִסְלִיחָה: כְּשֶׁה׳ סוֹלֵחַ, הוּא גַּם מַשְׁפִּיעַ טוֹב.

2

📜 הַפָּסוּקכִּי כַּאֲשֶׁר יֵרֵד הַגֶּשֶׁם וְהַשֶּׁלֶג מִן־הַשָּׁמַיִם וְשָׁמָּה לֹא יָשׁוּב כִּי אִם־הִרְוָה אֶת־הָאָרֶץ וְהוֹלִידָהּ וְהִצְמִיחָהּ וְנָתַן זֶרַע לַזֹּרֵעַ וְלֶחֶם לָאֹכֵל

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁה׳ נוֹתֵן לִי טוֹב, אֲנִי שָׂמֵחַ וְסוֹמֵךְ עָלָיו שֶׁיִּשְׁמֹר 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק נ״ה, פָּסוּק י׳

רָמָה ג׳ – מַעֲמִיקִים

🏖️ ׳עַל דָּבָר טוֹב לֹא מִתְפַּלְּלִים שֶׁיִּפָּסֵק׳

רַב בְּאֶרֶץ יִשְׂרָאֵל סִפֵּר: בְּכַמָּה שָׁנִים הָיָה מְעַט גֶּשֶׁם, מִפְלַס הַכִּנֶּרֶת יָרַד, וְכֻלָּם הִתְפַּלְּלוּ לְגֶשֶׁם. כְּשֶׁסּוֹף סוֹף יָרַד שֶׁפַע – בִּשְׁנַת תשנ״ב – הַיַּרְדֵּן עָלָה עַל גְּדוֹתָיו, וְהָיָה חֲשָׁשׁ שֶׁטְּבֶרְיָה תּוּצַף. בִּקְּשׁוּ מֵהָרַב מָרְדֳּכַי אֱלִיָּהוּ: ׳תִּתְפַּלֵּל שֶׁהַמַּיִם לֹא יַעֲלוּ׳. הוּא עָנָה: כָּל כָּךְ הַרְבֵּה שָׁנִים הִתְפַּלַּלְנוּ לַגֶּשֶׁם הַזֶּה, וְעַכְשָׁיו שֶׁה׳ נוֹתֵן – אֵיךְ אֶפְשָׁר לְהִתְפַּלֵּל שֶׁיֵּפָּסֵק? ׳עַל דָּבָר טוֹב לֹא מִתְפַּלְּלִים שֶׁיִּפָּסֵק, וְאַתֶּם אַל תִּדְאֲגוּ – ה׳ יִדְאַג לָכֶם׳.

בְּרָכָה מֵה׳ מְקַבְּלִים בְּשִׂמְחָה, וְלֹא מְבַקְשִׁים שֶׁתִּפָּסֵק – ׳ה׳ יִדְאַג לָכֶם׳.

3

📜 הַפָּסוּקהוֹי כָּל־צָמֵא לְכוּ לַמַּיִם וַאֲשֶׁר אֵין־לוֹ כָּסֶף לְכוּ שִׁבְרוּ וֶאֱכֹלוּ וּלְכוּ שִׁבְרוּ בְּלוֹא־כֶסֶף וּבְלוֹא מְחִיר יַיִן וְחָלָב

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לִהְיוֹת הַצָּמֵא שֶׁבָּא, וְה׳ הוּא הַמַּיִם 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק נ״ה, פְּסוּקִים א׳, ו׳–ז׳

רָמָה ג׳ – מַעֲמִיקִים

📖 מָה יְשַׁעְיָהוּ מְלַמֵּד אוֹתִי הַיּוֹם

יְשַׁעְיָהוּ לֹא רַק דּוֹרֵשׁ מִמֶּנִּי – הוּא מַזְמִין. ה׳ מַזְמִין אוֹתִי לַמַּיִם, לַלֶּחֶם, לַבְּרִית, וְלִסְלִיחָה. הַלִּמּוּד: לַעֲמֹד אֶל הַדֶּלֶת הַפְּתוּחָה בְּלִי לְהִתְבַּיֵּשׁ, לִלְמֹד תּוֹרָה כִּי הִיא מַשְׂבִּיעָה, לָשׁוּב לַ׳הַיּוֹם׳ וְלֹא לְמָחָר, וּלְהַאֲמִין שֶׁדְּבַר ה׳ ׳לֹא יָשׁוּב רֵיקָם׳ – גַּם הַמִּלָּה הַקְּטַנָּה שֶׁאֲנִי אוֹמֵר לוֹ בִּתְפִלָּה.

ה׳ מַזְמִין, קָרוֹב וְסוֹלֵחַ – אֲנִי רַק צָרִיךְ לָבוֹא הַיּוֹם.

💡

רָמָה ג׳ – מַעֲמִיקִים

מָה לוֹקְחִים מִרָמָה ג׳?

׳דִּרְשׁוּ ה׳ בְּהִמָּצְאוֹ׳ נוֹגֵעַ בָּנוּ כָּל שָׁנָה: בֶּאֱלוּל, בַּעֲשֶׂרֶת יְמֵי תְּשׁוּבָה, וּבְכָל גֶּשֶׁם שֶׁיּוֹרֵד. נִלְמַד מֵחֲזַ״ל וּמִסִּפּוּרִים אֵיךְ ה׳ קָרוֹב אֵלֵינוּ, וְנִרְאֶה אֶת אַהֲבָתוֹ.

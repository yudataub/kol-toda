🕯️ מְצוּדַת דָּוִד: ׳וְקָרָאתָ לַשַּׁבָּת עֹנֶג׳

מְצוּדַת דָּוִד מְבָאֵר: ׳אִם תָּשִׁיב מִשַּׁבָּת – אִם הֶחֱזַקְתָּ בַּדֶּרֶךְ לָצֵאת חוּץ לַתְּחוּם וְנִזְכַּרְתָּ שֶׁהוּא שַׁבָּת וְהִשְׁבַּתָּ רַגְלְךָ בִּשְׁבִיל הַשַּׁבָּת׳. ׳וְקָרָאתָ לַשַּׁבָּת עֹנֶג – תִּקְרָא אֶת הַשַּׁבָּת יוֹם עֹנֶג, יִהְיֶה מְיֻחָד לְךָ לְהִתְעַנֵּג בּוֹ׳. וְעַל ׳וְכִבַּדְתּוֹ׳: ׳לֹא בִּקְרִיאַת שֵׁם לְבַד, כִּי אִם תְּכַבְּדֵהוּ בְּעִנְיָנֶיךָ׳. הַשַּׁבָּת הִיא גַּם מְנוּחָה, גַּם עֹנֶג וְגַם כָּבוֹד.

שַׁבָּת: מְנוּחָה, עֹנֶג וְכָבוֹד – ׳לֹא בִּקְרִיאַת שֵׁם לְבַד׳.

6
🏞️

📜 הַפָּסוּקאָז תִּתְעַנַּג עַל ה׳ וְהִרְכַּבְתִּיךָ עַל בָּמֳותֵי אָרֶץ וְהַאֲכַלְתִּיךָ נַחֲלַת יַעֲקֹב אָבִיךָ

💛 אַהֲבַת הַשֵּׁם: ה׳ מְחַכֶּה שֶׁאֶתְעַנֵּג דַּוְקָא עָלָיו, כִּי הוּא הָעֹנֶג הַגָּדוֹל בְּיוֹתֵר 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק נ״ח, פָּסוּק י״ד (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּצּוֹם וּבַשַּׁבָּת

🏞️ רַשִׁ״י וּמְצוּדַת דָּוִד: ׳נַחֲלַת יַעֲקֹב אָבִיךָ׳

מְצוּדַת דָּוִד מְבָאֵר אֶת ׳אָז תִּתְעַנַּג עַל ה׳׳: ׳לֹא תִתְעַנֵּג לְעֶדְנַת הַגּוּף כִּי אִם לְהַעֲמִיד אֶת הַגּוּף בְּמַצָּב הָרָאוּי לְהַשְׂכִּיל בְּתוֹרַת ה׳׳. וְרַשִׁ״י מְבָאֵר: ׳נַחֲלַת יַעֲקֹב אָבִיךָ – נַחֲלָה בְּלֹא מְצָרִים׳ (כְּמוֹ ׳וּפָרַצְתָּ יָמָּה וָקֵדְמָה׳), וְ׳יַעֲקֹב שָׁמַר אֶת הַשַּׁבָּת׳, שֶׁנֶּאֱמַר ׳וַיִּחַן אֶת פְּנֵי הָעִיר׳ – ׳שֶׁקָּבַע תְּחוּמִין עִם דִּמְדּוּמֵי חַמָּה׳, כָּךְ פֵּרֵשׁ רַבִּי שִׁמְשׁוֹן. כְּלוֹמַר: הַשַּׁבָּת הִיא הַדֶּרֶךְ לִנְחֲלָה גְּדוֹלָה, וְהָעֹנֶג הוּא כְּדֵי לְהִתְקָרֵב לַה׳.

מְצוּדַת דָּוִד: הָעֹנֶג בַּשַּׁבָּת אֵינוֹ רַק לְהִתְפַּנֵּק – הוּא כְּדֵי לְהַשְׂכִּיל בְּתוֹרַת ה׳.

7
💡

רָמָה ב׳ – עִיּוּן בַּצּוֹם וּבַשַּׁבָּת

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מַרְאִים שֶׁצּוֹם אֲמִתִּי כּוֹלֵל נְתִינָה וּשְׁמִירָה עַל הַלֵּב: ׳תַּנְחוּמֵי דְּבָרִים טוֹבִים׳ מִלְּבַד הַלֶּחֶם, ׳לֹא תִתְעַנֵּג לְעֶדְנַת הַגּוּף׳ אֶלָּא לְהַשְׂכִּיל בְּתוֹרַת ה׳, וְה׳ עוֹנֶה ׳הִנֵּנִי׳.

8
❤️
אָז תִּקְרָא וַה׳ יַעֲנֶה תְּשַׁוַּע וְיֹאמַר הִנֵּנִי

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳פָּרֹס לָרָעֵב לַחְמֶךָ׳ – מִי נִמְצָא סְבִיבִי שֶׁצָּרִיךְ מִשֶּׁהוּ מִמֶּנִּי, וּמָה אֲנִי יָכוֹל לָתֵת לוֹ הַיּוֹם?
  • ׳אָז תִּקְרָא וַה׳ יַעֲנֶה... הִנֵּנִי׳ – מָה אֶרְגִּישׁ אִם ה׳ עוֹנֶה לִי ׳הִנֵּנִי׳ כְּשֶׁאֲנִי קוֹרֵא אֵלָיו, וְעַל מָה אֲנִי רוֹצֶה לִקְרֹא אֵלָיו הַיּוֹם?
  • ׳וְקָרָאתָ לַשַּׁבָּת עֹנֶג׳ – מָה הַדָּבָר שֶׁהוּא הָעֹנֶג הַכִּי גָּדוֹל שֶׁלִּי בַּשַּׁבָּת, וְאֵיךְ הוּא מְקָרֵב אוֹתִי לַה׳?

יְרוּשָׁלַיִם קָמָה לָאוֹר, הָעַמִּים בָּאִים אֶל אוֹרָהּ, וְהַבָּנִים חוֹזְרִים הַבַּיְתָה – וְה׳ עַצְמוֹ הוּא הָאוֹר שֶׁלָּהּ לְעוֹלָם

📜 הַפָּסוּקכִּי הִנֵּה הַחֹשֶׁךְ יְכַסֶּה אֶרֶץ וַעֲרָפֶל לְאֻמִּים וְעָלַיִךְ יִזְרַח ה׳ וּכְבוֹדוֹ עָלַיִךְ יֵרָאֶה

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹרֶה בְּתוֹכִי וּמֵאִיר אוֹתִי, גַּם כְּשֶׁסְּבִיבִי חָשׁוּךְ 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ס׳, פְּסוּקִים א׳–ב׳ (מְצוּדַת דָּוִד וּמְצוּדַת צִיּוֹן)

רָמָה ב׳ – עִיּוּן בָּאוֹר

🌅 מְצוּדַת דָּוִד וּמְצוּדַת צִיּוֹן: חֹשֶׁךְ וְאוֹר

מְצוּדַת דָּוִד מְבָאֵר אֶת ׳קוּמִי אוֹרִי׳: ׳קוּמִי וְהָאִירִי, כִּי בָּא לָךְ הַהֲאָרָה הָרְאוּיָה לָךְ, וְהוּא מָשָׁל עַל הַשִּׂמְחָה וְהַטּוֹבָה׳, וְ׳כְּבוֹד ה׳ – הַשְּׁכִינָה תִּתְגַּלֶּה עָלַיִךְ וְתִשְׁרֶה בָּךְ׳. עַל ׳כִּי הִנֵּה הַחֹשֶׁךְ׳: ׳צָרוֹת יִתְרַבּוּ בָּעוֹלָם׳. וּמְצוּדַת צִיּוֹן: ׳וַעֲרָפֶל – עַב הֶעָנָן׳. הַנָּבִיא מַנִּיחַ זֶה לְעֻמַּת זֶה: חֹשֶׁךְ וַעֲרָפֶל עַל הָעוֹלָם – וְאוֹר ה׳ עַל יְרוּשָׁלַיִם.

מְצוּדַת דָּוִד: ׳הַהֲאָרָה׳ הִיא מָשָׁל עַל הַשִּׂמְחָה, וּ׳כְבוֹד ה׳׳ – הַשְּׁכִינָה שׁוֹרָה בִּירוּשָׁלַיִם.

💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לָנוּ תּוֹרָה כְּדֵי שֶׁנָּאִיר בָּהּ אֶת הָעוֹלָם 💛

רָמָה ב׳ – עִיּוּן בָּאוֹר

🕯️ מְצוּדַת דָּוִד: ׳וְהָלְכוּ גוֹיִם לְאוֹרֵךְ׳

מְצוּדַת דָּוִד מְבָאֵר אֶת ׳לְאוֹרֵךְ׳: ׳מִמֵּךְ יִלְמְדוּ דַּרְכֵי ה׳ וְאַתְּ תָּאִירִי עֵינֵיהֶם׳. וּמְצוּדַת צִיּוֹן עַל ׳לְנֹגַהּ׳: ׳עִנְיַן הֲאָרָה׳, כְּמוֹ ׳מִנֹּגַהּ נֶגְדּוֹ׳. כְּלוֹמַר: הָאוֹר הוּא לִמּוּד. מִי שֶׁהוֹלֵךְ ׳לְאוֹרֵךְ׳ אֵינוֹ בָּא לִכְבֹּשׁ אֶלָּא לְהִתְחַכֵּם, וְיִשְׂרָאֵל הֵם הַמּוֹרִים שֶׁמְּאִירִים אֶת הַדֶּרֶךְ.

מְצוּדַת דָּוִד: הָאוֹר שֶׁל יִשְׂרָאֵל הוּא לִמּוּד דַּרְכֵי ה׳ – ׳וְאַתְּ תָּאִירִי עֵינֵיהֶם׳.

📜 הַפָּסוּקאָז תִּרְאִי וְנָהַרְתְּ וּפָחַד וְרָחַב לְבָבֵךְ

💛 אַהֲבַת הַשֵּׁם: ה׳ מְשַׂמֵּחַ אוֹתִי בְּשִׂמְחָה גְּדוֹלָה מִכְּפִי שֶׁהִתְכּוֹנַנְתִּי 💛

רָמָה ב׳ – עִיּוּן בָּאוֹר

😮 מְצוּדַת דָּוִד: ׳וּפָחַד וְרָחַב לְבָבֵךְ׳

מְצוּדַת דָּוִד מְבָאֵר: ׳אָז תִּרְאִי – אֶת בָּנַיִךְ בָּאִים מִכָּל צַד וְיָאִיר פָּנַיִךְ מֵרֹב הַשִּׂמְחָה׳. וְעַל ׳וּפָחַד׳: ׳דֶּרֶךְ הָאָדָם לִהְיוֹת נִבְהָל וְכַאֲלוּ מְפַחֵד בְּבוֹא אֵלָיו טוֹבָה פִּתְאוֹם׳, וְעַל ׳וְרָחַב׳: ׳יִתְרַחֵב לִבֵּךְ בִּשְׁבִיל רֹב הַטּוֹבָה וְהַשִּׂמְחָה׳. מְצוּדַת צִיּוֹן: ׳וְנָהַרְתְּ – עִנְיַן הֲאָרָה׳. הַשִּׂמְחָה כֹּה גְּדוֹלָה, שֶׁאֲפִלּוּ מִתְפַּחֲדִים מִמֶּנָּה.

מְצוּדַת דָּוִד: כְּשֶׁטּוֹבָה גְּדוֹלָה בָּאָה פִּתְאוֹם, הַלֵּב ׳נִבְהָל׳ וְאַחַר כָּךְ מִתְרַחֵב.

🕊️

📜 הַפָּסוּקמִי אֵלֶּה כָּעָב תְּעוּפֶינָה וְכַיּוֹנִים אֶל אֲרֻבֹּתֵיהֶם

💛 אַהֲבַת הַשֵּׁם: ה׳ מַחֲזִיר אוֹתִי לַבַּיִת שֶׁלִּי כְּמוֹ יוֹנָה לַשּׁוֹבָךְ 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ס׳, פְּסוּקִים ח׳–ט׳ (מְצוּדַת דָּוִד, מְצוּדַת צִיּוֹן וְרַשִׁ״י)

רָמָה ב׳ – עִיּוּן בָּאוֹר

🕊️ מְצוּדַת דָּוִד וּמְצוּדַת צִיּוֹן: ׳אֲרֻבֹּתֵיהֶם׳

מְצוּדַת דָּוִד מְבָאֵר: ׳כְּאִלּוּ הַבְּרִיּוֹת יִשְׁאֲלוּ עַל בְּנֵי הַגּוֹלָה: מִי אֵלֶּה? שֶׁמְּמַהֲרִים לָשׁוּב כַּעֲפִיפַת הֶעָנָן וְכַיּוֹנִים הַמְּמַהֲרִים לָעוּף אֶל אֲרֻבּוֹתֵיהֶם׳. מְצוּדַת צִיּוֹן: ׳אֲרֻבּוֹתֵיהֶם – חַלּוֹנוֹת הַשּׁוֹבָךְ, וְהוּא מְקוֹם מְדוֹרָם׳. הַמָּשָׁל מַרְאֶה בְּנֵי אָדָם שֶׁחוֹזְרִים הַבַּיְתָה כְּמוֹ יוֹנִים, בְּמֶרֶץ וּבְשִׂמְחָה. וְרַשִׁ״י עַל ׳בָּרִאשֹׁנָה׳: ׳כְּבָרִאשׁוֹנָה בִּימֵי שְׁלֹמֹה׳, כְּשֶׁאֳנִיּוֹת תַּרְשִׁישׁ הָיוּ בָּאוֹת לַמֶּלֶךְ.

מְצוּדַת צִיּוֹן: ׳אֲרֻבּוֹתֵיהֶם׳ הֵן חַלּוֹנוֹת הַשּׁוֹבָךְ – מְקוֹם הַמָּדוֹר שֶׁחוֹזְרִים אֵלָיו.

💛

📜 הַפָּסוּקכִּי בְקִצְפִּי הִכִּיתִיךְ וּבִרְצוֹנִי רִחַמְתִּיךְ

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה לְרַחֵם עָלַי – זֶה הָרָצוֹן שֶׁלּוֹ, לֹא רַק הַתּוֹצָאָה 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ס׳, פָּסוּק י׳ (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בָּאוֹר

💛 מְצוּדַת דָּוִד: ׳בְקִצְפִּי הִכִּיתִיךְ וּבִרְצוֹנִי רִחַמְתִּיךְ׳

מְצוּדַת דָּוִד מְבָאֵר: ׳כָּל הַבָּאָה עָלַיִךְ לֹא הָיְתָה בְּדֶרֶךְ הַטֶּבַע, כִּי לְשֶׁעָבַר הִכֵּיתִי אוֹתָךְ בְּקִצְפִּי וְלָזֶה הָיִית בְּתַכְלִית הַשִּׁפְלוּת, וְעַתָּה רִחַמְתִּי אוֹתָךְ בִּרְצוֹנִי הַטּוֹב׳. וְעַל ׳וּבָנוּ בְנֵי נֵכָר חֹמֹתַיִךְ׳: ׳חוֹמוֹת הֶעָרִים׳. הַקֶּצֶף הָיָה זְמַנִּי; הָרָצוֹן הוּא הַסּוֹף. ה׳ אוֹמֵר: אֲנִי לֹא רוֹצֶה לְהַכּוֹת – אֲנִי רוֹצֶה לְרַחֵם.

מְצוּדַת דָּוִד: הָרַחֲמִים בָּאִים ׳בִּרְצוֹנִי הַטּוֹב׳, לֹא בְּדֶרֶךְ הַטֶּבַע.

📜 הַפָּסוּקתַּחַת הַנְּחֹשֶׁת אָבִיא זָהָב וְתַחַת הַבַּרְזֶל אָבִיא כֶסֶף וְתַחַת הָעֵצִים נְחֹשֶׁת וְתַחַת הָאֲבָנִים בַּרְזֶל וְשַׂמְתִּי פְקֻדָּתֵךְ שָׁלוֹם וְנֹגְשַׂיִךְ צְדָקָה

💛 אַהֲבַת הַשֵּׁם: ה׳ מַהֲפֵךְ אֶת הַצָּרוֹת שֶׁלִּי לְשָׁלוֹם 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ס׳, פָּסוּק י״ז (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בָּאוֹר

⚖️ הַשְׁוָאָה: ׳פְּקֻדָּתֵךְ שָׁלוֹם׳ – רַשִׁ״י וּמְצוּדַת דָּוִד

רַשִׁ״י מְבָאֵר: ׳תַּחַת הַנְּחֹשֶׁת – שֶׁנָּטְלוּ מִמֵּךְ׳. עַל ׳וְשַׂמְתִּי פְקֻדָּתֵךְ שָׁלוֹם׳ יֵשׁ שְׁנֵי פֵּרוּשִׁים: רַשִׁ״י מְבִיא אֶת רַבּוֹתֵינוּ: ׳הַפְּקוּדוֹת שֶׁבָּאוּ עָלַיִךְ בְּגָלוּתֵךְ וְהַנּוֹגְשִׂים שֶׁדְּחָקוּךְ יִהְיוּ לָךְ לְשָׁלוֹם וּלְצְדָקָה׳. וּמְצוּדַת דָּוִד: ׳תַּחַת הַפְּקִידִים שֶׁהָיוּ נוֹתְנִים עֲלֵיכֶם לִגְבּוֹת הַמַּס, אָשִׂים עָלַיִךְ שָׁלוֹם, כִּי לֹא יָבוֹאוּ עוֹד בְּתַגָּר לִגְבּוֹת הַמַּס, כִּי אִם יָבוֹאוּ לִשְׁאֹל בִּשְׁלוֹמֵךְ׳. לְפִי רַבּוֹתֵינוּ הַצָּרוֹת עַצְמָן נֶהֱפָכוֹת לְטוֹבָה; לְפִי הַמְּצוּדָה – בְּמָקוֹם הַגּוֹבִים בָּאִים אוֹהֲבִים.

מְצוּדַת דָּוִד: הַגּוֹבִים יִהְיוּ אוֹהֲבִים – ׳יָבוֹאוּ לִשְׁאֹל בִּשְׁלוֹמֵךְ׳.

רַשִׁ״י מְבָאֵר: ׳לֹא יִהְיֶה לָךְ עוֹד – לֹא תִּצְטָרְכִי לְאוֹר הַשֶּׁמֶשׁ׳. מְצוּדַת דָּוִד: ׳אֲבָל אוֹר הַשְּׁכִינָה יִהְיֶה לָךְ לְאוֹר עוֹלָם בַּיּוֹם וּבַלַּיְלָה׳, וְ׳לְתִפְאַרְתֵּךְ – תִּתְפָּאֲרִי בִּכְבוֹד ה׳׳. וְעַל ׳וְשָׁלְמוּ יְמֵי אֶבְלֵךְ׳: ׳יִתַּמּוּ וְיִגָּמְרוּ יְמֵי אֶבְלֵךְ כִּי לֹא תִּתְאַבְּלִי עוֹד׳. אַהֲבַת ה׳ מוּבְטַחַת: הוּא הָאוֹר, וְהָאֵבֶל נִגְמָר.

מְצוּדַת דָּוִד: ה׳ הוּא הָאוֹר, וְ׳יְמֵי אֶבְלֵךְ׳ יִתַּמּוּ.

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מְגַלִּים כַּמָּה אַהֲבָה בַּנְּבוּאָה: ׳לֹא הָיָה בְּדֶרֶךְ הַטֶּבַע׳ אֶלָּא ׳בִּרְצוֹנִי הַטּוֹב׳, ה׳ מְאִיר אֶת הָעוֹלָם דֶּרֶךְ יְרוּשָׁלַיִם, וְיָמֵי הָאֵבֶל ׳יִתַּמּוּ׳.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳קוּמִי אוֹרִי כִּי בָא אוֹרֵךְ׳ – מָתַי הִרְגַּשְׁתִּי שֶׁהַלֵּב שֶׁלִּי ׳קָם לָאוֹר׳, וּמָה גָּרַם לָזֶה?
  • ׳מִי אֵלֶּה כָּעָב תְּעוּפֶינָה וְכַיּוֹנִים אֶל אֲרֻבֹּתֵיהֶם׳ – אֵיךְ אֲנִי חוֹזֵר הַבַּיְתָה: כְּמוֹ עָנָן שֶׁדּוֹחֲפִים אוֹתוֹ, אוֹ כְּמוֹ יוֹנָה שֶׁרוֹצָה בְּעַצְמָהּ?
  • ׳וְעַמֵּךְ כֻּלָּם צַדִּיקִים׳ – אֵיךְ אֲנִי יָכוֹל לִרְאוֹת אֶת הַטּוֹב שֶׁיֵּשׁ בְּכָל יְהוּדִי, גַּם בְּמִי שֶׁקָּשֶׁה לִי אִתּוֹ?

זֶה הַדֶּרֶךְ לְכוּ בוֹ

רָמָה ג׳ – מַעֲמִיקִים

יְשַׁעְיָהוּ שׁוֹאֵל עַל מִי סוֹמְכִים – עַל סוּסִים מִמִּצְרַיִם אוֹ עַל ה׳ – וְה׳ מְחַכֶּה לָנוּ וְאוֹמֵר: ׳זֶה הַדֶּרֶךְ לְכוּ בוֹ׳

📖 פְּרָקִים ל׳–ל״א

🎯 לַחִידוֹן

📜 הַפָּסוּקוְאָזְנֶיךָ תִּשְׁמַעְנָה דָבָר מֵאַחֲרֶיךָ לֵאמֹר זֶה הַדֶּרֶךְ לְכוּ בוֹ כִּי תַאֲמִינוּ וְכִי תַשְׂמְאִילוּ

💛 אַהֲבַת הַשֵּׁם: ה׳ קוֹרֵא לִי גַּם כְּשֶׁאֲנִי רָחוֹק, וְעוֹד לֹא מְאֻחָר 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ל׳, פָּסוּק כ״א

רָמָה ג׳ – מַעֲמִיקִים

🐑 לִשְׁמֹעַ אֶת קוֹל הָרוֹעֶה

הָרַב לִמֵּד: אָדָם שֶׁתָּעָה בַּדֶּרֶךְ יָכוֹל לִהְיוֹת בְּמַצָּבִים שׁוֹנִים. יֵשׁ מִי שֶׁכְּבָר אֵינוֹ שׁוֹמֵעַ מֵאַיִן קוֹרְאִים לוֹ, וְיֵשׁ מִי שֶׁעוֹד שׁוֹמֵעַ הֵד רָחוֹק. גַּם אִם אֵינוֹ יוֹדֵעַ אֶת כָּל הַדֶּרֶךְ חֲזָרָה, עֶצֶם שְׁמִיעַת הַקּוֹל נוֹתֶנֶת כִּוּוּן. הָרַב קָשַׁר לְכָךְ אֶת הַפָּסוּק: ׳תָּעִיתִי כְּשֶׂה אֹבֵד בַּקֵּשׁ עַבְדֶּךָ׳ (תְּהִלִּים קי״ט, קע״ו) – אֵין צֹרֶךְ לְהַעֲמִיד פָּנִים שֶׁלֹּא תָּעִינוּ, אֶפְשָׁר לוֹמַר אֶת הָאֱמֶת וּלְבַקֵּשׁ שֶׁלֹּא נִשָּׁאֵר אֲבוּדִים. כָּל עוֹד הַלֵּב שׁוֹמֵעַ אֶת קוֹל הָרוֹעֶה, אֶפְשָׁר לְהַתְחִיל לַחֲזֹר.

כְּשֶׁהַלֵּב עוֹד שׁוֹמֵעַ אֶת הַקּוֹל, אֶפְשָׁר לַחֲזֹר – ׳זֶה הַדֶּרֶךְ לְכוּ בוֹ׳.

1

📜 הַפָּסוּקוְנָתַן לָכֶם אֲדֹנָי לֶחֶם צָר וּמַיִם לָחַץ וְלֹא־יִכָּנֵף עוֹד מוֹרֶיךָ וְהָיוּ עֵינֶיךָ רֹאוֹת אֶת־מוֹרֶיךָ

💛 אַהֲבַת הַשֵּׁם: ה׳ נוֹתֵן לִי מַה שֶּׁצָּרִיךְ, וַאֲנִי שָׂמֵחַ בּוֹ 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ל׳, פָּסוּק כ׳ (רַשִׁ״י וּמְצוּדַת דָּוִד)

רָמָה ג׳ – מַעֲמִיקִים

🍞 ׳לֶחֶם צָר וּמַיִם לָחַץ׳ – לְהַסְתַּפֵּק

הַפָּסוּק אוֹמֵר: ׳וְנָתַן לָכֶם אֲדֹנָי לֶחֶם צָר וּמַיִם לָחַץ׳. רַשִׁ״י מְבָאֵר: ׳לֹא תִהְיוּ כְּרוּכִים אַחַר תַּעֲנוּגִים כַּאֲשֶׁר אַתֶּם עַתָּה׳. וּמְצוּדַת דָּוִד: ׳הַמָּקוֹם יִתֵּן בִּלְבַבְכֶם לְהִסְתַּפֵּק בְּלֶחֶם צַר וּמַיִם לַחַץ, וְלֹא תִהְיוּ שְׁטוּפִים אַחַר הַתַּעֲנוּגִים׳. כְּלוֹמַר, בַּגְּאֻלָּה ה׳ מְלַמֵּד אוֹתָנוּ לִהְיוֹת שְׂמֵחִים בְּמַה שֶּׁיֵּשׁ.

מִי שֶׁשָּׂמֵחַ בְּמַה שֶּׁיֵּשׁ לוֹ אֵינוֹ רָץ אַחַר תַּעֲנוּגִים.

2

📜 הַפָּסוּקכִּי כֹה־אָמַר אֲדֹנָי ה׳ קְדוֹשׁ יִשְׂרָאֵל בְּשׁוּבָה וָנַחַת תִּוָּשֵׁעוּן בְּהַשְׁקֵט וּבְבִטְחָה תִּהְיֶה גְּבוּרַתְכֶם וְלֹא אֲבִיתֶם

💛 אַהֲבַת הַשֵּׁם: אֲנִי בּוֹטֵחַ בְּךָ, ה׳, וְשׁוֹמֵעַ אֶת קוֹלְךָ 💛

📖 יְשַׁעְיָהוּ, פֶּרֶק ל׳, פְּסוּקִים ט״ו, כ״א

רָמָה ג׳ – מַעֲמִיקִים

📖 מָה יְשַׁעְיָהוּ מְלַמֵּד אוֹתִי הַיּוֹם

יְשַׁעְיָהוּ מְלַמֵּד אוֹתִי לִבְחֹר בְּבִטָּחוֹן בִּמְקוֹם בְּפַחַד: לִשְׁאֹל אֶת ה׳ לִפְנֵי שֶׁרָצִים לַאֲחֵרִים, לִשְׁמֹעַ אֶת הָאֱמֶת גַּם כְּשֶׁהִיא לֹא נְעִימָה, לָנוּחַ וְלִבְטֹחַ, לְהַסְתַּפֵּק, וּלְהַקְשִׁיב לַקּוֹל שֶׁאוֹמֵר ׳זֶה הַדֶּרֶךְ לְכוּ בוֹ׳. ה׳ מְחַכֶּה לִי, שׁוֹמֵעַ אוֹתִי מִיָּד וְשׁוֹמֵר עָלַי כְּצִפֳּרִים עָפוֹת.

שׁוֹאֲלִים אֶת ה׳, בּוֹטְחִים בְּשֶׁקֶט, וְשׁוֹמְעִים אֶת הַקּוֹל שֶׁאוֹמֵר ׳זֶה הַדֶּרֶךְ׳.

💡

רָמָה ג׳ – מַעֲמִיקִים

מָה לוֹקְחִים מִרָמָה ג׳?

הַקּוֹל שֶׁאוֹמֵר ׳זֶה הַדֶּרֶךְ לְכוּ בוֹ׳ חַי גַּם הַיּוֹם: בָּרַעַשׁ שֶׁל הַחַיִּים אֶפְשָׁר לִשְׁמֹעַ אֶת קוֹל הָרוֹעֶה, לְהַסְתַּפֵּק בְּמָה שֶׁיֵּשׁ וְלִבְטֹחַ.

🤔 חוֹשְׁבִים עַל זֶה

  • ׳בְּהַשְׁקֵט וּבְבִטְחָה תִּהְיֶה גְּבוּרַתְכֶם׳ – אֵיפֹה בַּחַיִּים שֶׁלִּי אֲנִי מְמַהֵר וּמִתְרַגֵּשׁ, וְאֵיךְ אֶפְשָׁר לִבְטֹחַ בַּה׳ וּלְהַרְגִּישׁ שָׁלֵו?
  • ׳וְלָכֵן יְחַכֶּה ה׳ לַחֲנַנְכֶם׳ – ה׳ מְחַכֶּה לָנוּ. עַל מָה אֲנִי מְחַכֶּה שֶׁיַּעֲזֹר לִי, וְעַל מָה ה׳ מְחַכֶּה שֶׁאֶעֱשֶׂה אֲנִי?
  • ׳זֶה הַדֶּרֶךְ לְכוּ בוֹ׳ – מָתַי הִרְגַּשְׁתִּי שֶׁמִּישֶׁהוּ מַכְוִין אוֹתִי לַדֶּרֶךְ הַנְּכוֹנָה? וְאֵיךְ אֲנִי יָכוֹל לִשְׁמֹעַ טוֹב יוֹתֵר?

הַפָּסוּק ׳כִּי נִחַם ה׳ צִיּוֹן... שָׂשׂוֹן וְשִׂמְחָה יִמָּצֵא בָהּ תּוֹדָה וְקוֹל זִמְרָה׳ נִקְרָא בְּהַפְטָרַת פָּרָשַׁת עֵקֶב, מִשֶּׁבַע דְּנֶחָמָתָא. הָרַב לִמֵּד: כְּשֵׁם שֶׁיֵּשׁ מִצְוָה לִטְעֹם מִתַּבְשִׁילֵי הַשַּׁבָּת בְּעֶרֶב שַׁבָּת (׳טוֹעֲמֶיהָ חַיִּים זָכוּ׳), כָּךְ ה׳ נוֹתֵן לְעַם יִשְׂרָאֵל ׳מַטְעַמִּים שֶׁל גְּאֻלָּה׳ לִפְנֵי הַגְּאֻלָּה הַשְּׁלֵמָה. וּמַה הֵם? ׳תּוֹדָה וְקוֹל זִמְרָה׳ – הַהוֹדָיָה.

הַהוֹדָיָה וְהַשִּׁיר הֵם ׳מַטְעַמִּים׳ – טַעַם רִאשׁוֹן שֶׁל הַגְּאֻלָּה.

רָמָה ג׳ – מַעֲמִיקִים

🏗️ חֲצִי רִאשׁוֹן וַחֲצִי שֵׁנִי

הָרַב חִלֵּק אֶת הַפָּסוּק לִשְׁנַיִם. חֶצְיוֹ הָרִאשׁוֹן: ׳כִּי נִחַם ה׳ צִיּוֹן נִחַם כָּל חָרְבֹתֶיהָ׳ – חָרְבָּה שֶׁנֶּהֶפְכֶת לְבִנְיָן; כָּל חָרְבָּה בָּעִיר נִבְנֵית, בּוֹנִים וּבוֹנִים. וְחֶצְיוֹ הַשֵּׁנִי: ׳שָׂשׂוֹן וְשִׂמְחָה יִמָּצֵא בָהּ תּוֹדָה וְקוֹל זִמְרָה׳. מִי שֶׁרוֹאֶה אֶת הַחֵצִי הָרִאשׁוֹן מִתְקַיֵּם לְנֶגֶד עֵינָיו, צָרִיךְ לְקַיֵּם אֶת הַחֵצִי הַשֵּׁנִי: עַל כָּל בִּנְיָן שֶׁקָּם – לוֹמַר תּוֹדָה וְלָשִׁיר.

רוֹאִים בִּנְיָן קָם? אוֹמְרִים תּוֹדָה וְשָׁרִים – זֶה הַחֵצִי הַשֵּׁנִי שֶׁל הַפָּסוּק.

רָמָה ג׳ – מַעֲמִיקִים

🎶 הַקָּרְבָּן שֶׁנִּשְׁאָר

הָרַב הֵבִיא אֶת דִּבְרֵי חֲזַ״ל: לֶעָתִיד לָבוֹא כָּל הַקָּרְבָּנוֹת בְּטֵלִים, וּמָה נִשְׁאָר? קָרְבַּן תּוֹדָה וּמִזְמוֹר לְתוֹדָה (מִדְרַשׁ וַיִּקְרָא רַבָּה). הַמְפָרְשִׁים מְבָאֲרִים שֶׁהַכַּוָּנָה לַקָּרְבָּנוֹת שֶׁבָּאִים עַל חֵטְא – לֶעָתִיד לֹא יִהְיֶה בָּהֶם צֹרֶךְ. הוּא קָשַׁר זֹאת לַפָּסוּק שֶׁלָּנוּ: ׳תּוֹדָה וְקוֹל זִמְרָה׳. וְהוּא הוֹסִיף שֶׁבַּדּוֹר שֶׁלָּנוּ ה׳ נוֹתֵן אוֹר מַדְהִים שֶׁל הוֹדָיָה: מִזְמוֹר לְתוֹדָה מִתְנַגֵּן וּמָלֵא מַנְגִּינוֹת, מַתְחִילִים לְדַבֵּר עַל הוֹדָיָה בַּחֲתֻנּוֹת, וַאֲנָשִׁים מַרְגִּישִׁים אוֹר שֶׁל גְּאֻלָּה.

לֶעָתִיד לָבוֹא תִּשָּׁאֵר רַק הַתּוֹדָה; וַאֲנַחְנוּ כְּבָר מַתְחִילִים לָשִׁיר אוֹתָהּ.

רָמָה ג׳ – מַעֲמִיקִים

💑 ׳כִּי אֶחָד קְרָאתִיו׳ – אֲחִדּוּת בַּבַּיִת

הָרַב הֵבִיא אֶת דִּבְרֵי הַמַּהַר״ל עַל ׳הַבִּיטוּ אֶל אַבְרָהָם אֲבִיכֶם וְאֶל שָׂרָה תְּחוֹלֶלְכֶם כִּי אֶחָד קְרָאתִיו׳: אַבְרָהָם וְשָׂרָה הִגִּיעוּ לְמַצָּב שֶׁבּוֹ נַעֲשׂוּ אֶחָד. וּמִכָּאן, אָמַר הָרַב, אֶפְשָׁר לִלְמֹד: הַגְּאֻלָּה תָּבוֹא כְּשֶׁתִּהְיֶה אֲחִדּוּת בֵּין אִישׁ לְאִשְׁתּוֹ. בַּבְּרָכָה הָאַחֲרוֹנָה שֶׁל שֶׁבַע בְּרָכוֹת, בְּתוֹךְ בְּרָכָה עַל חָתָן וְכַלָּה, מְדַבְּרִים פִּתְאֹם עַל ׳עָרֵי יְהוּדָה וְחוּצוֹת יְרוּשָׁלַיִם׳ – כִּי מִבַּיִת שֶׁיֵּשׁ בּוֹ אַהֲבָה, אַחֲוָה, שָׁלוֹם וְרֵעוּת צוֹמַחַת הַגְּאֻלָּה. כָּל בַּיִת כָּזֶה הוּא אֶבֶן בְּבִנְיַן הֶעָתִיד.

בַּיִת שֶׁיֵּשׁ בּוֹ אַהֲבָה, אַחֲוָה, שָׁלוֹם וְרֵעוּת הוּא כְּבָר עַכְשָׁיו אֶבֶן בְּבִנְיַן הַגְּאֻלָּה.

📖 מָה יְשַׁעְיָהוּ מְלַמֵּד אוֹתִי הַיּוֹם

יְשַׁעְיָהוּ מְלַמֵּד אוֹתִי: לַעֲנוֹת ׳הִנְנִי׳ כְּשֶׁה׳ קוֹרֵא; לִהְיוֹת מְנַחֵם בְּמִלָּה חַמָּה לְמִי שֶׁעָיֵף; לְהַבִּיט אֶל אַבְרָהָם וְשָׂרָה וְלִבְנוֹת בַּיִת שֶׁיֵּשׁ בּוֹ אֲחִדּוּת; וּלְקַיֵּם אֶת חֲצִי הַפָּסוּק הַשֵּׁנִי – עַל כָּל טוֹבָה לוֹמַר ׳תּוֹדָה וְקוֹל זִמְרָה׳.

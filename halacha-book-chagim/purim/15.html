סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת פּוּרִים
מַהֲדוּרַת יְלָדִים

📜 פּוּרִים וּמִקְרָא מְגִלָּה

פֶּרֶק ט״ו · לַמַּהֲדוּרָה לַמּוֹרֶה
לֵיל פּוּרִים. בֵּית הַכְּנֶסֶת מָלֵא: יְלָדִים בְּתַחְפֹּשׂוֹת, רַעֲשָׁנִים בַּיָּדַיִם, וְהַחַזָּן פּוֹרֵשׂ מְגִלָּה אֲרֻכָּה כְּמוֹ מִכְתָּב. ״אִישׁ יְהוּדִי הָיָה בְּשׁוּשַׁן הַבִּירָה...״ – וְכֻלָּם צוֹעֲקִים יַחַד! אֲבָל מָתַי מֻתָּר לְהַרְעִישׁ? מָה עוֹשִׂים אִם פִּסְפַסְתָּ מִלָּה? וְלָמָּה בִּירוּשָׁלַיִם חוֹגְגִים יוֹם אַחֲרֵי כֻּלָּם? בַּפֶּרֶק הַזֶּה נִלְמַד עַל נֵס פּוּרִים וְעַל הַמִּצְוָה הָרִאשׁוֹנָה שֶׁל הַחַג – מִקְרָא מְגִלָּה.

1נֵס פּוּרִים

בֵּית הַמִּקְדָּשׁ הָרִאשׁוֹן נֶחְרַב, וְיִשְׂרָאֵל גָּלוּ מֵאַרְצָם. כּוֹרֶשׁ הִרְשָׁה לַיְּהוּדִים לַחֲזֹר, אֲבָל רַק מְעַטִּים עָלוּ. רַבִּים רָצוּ לִהְיוֹת כְּמוֹ הַגּוֹיִים, וַאֲפִלּוּ נֶהֱנוּ מִסְּעוּדַת אֲחַשְׁוֵרוֹשׁ, שֶׁבָּהּ הִשְׁתַּמְּשׁוּ בִּכְלֵי בֵּית הַמִּקְדָּשׁ. אָז קָם הָמָן הָרָשָׁע, מִזֶּרַע עֲמָלֵק, וְגָזַר ״לְהַשְׁמִיד לַהֲרֹג וּלְאַבֵּד אֶת כָּל הַיְּהוּדִים״ בְּי״ג בַּאֲדָר.
🤔 לָמָּה? שִׂמְחַת פּוּרִים מְלַמֶּדֶת שֶׁהַקְּדֻשָּׁה שֶׁל יִשְׂרָאֵל לֹא זָזָה מִמְּקוֹמָהּ. גַּם כְּשֶׁיִּשְׂרָאֵל חוֹטְאִים, הֵם נִקְרָאִים ״בָּנִים לַמָּקוֹם״, וַה׳ מְסַבֵּב אֶת הַדְּבָרִים לְטוֹבָתָם.
(פֶּרֶק ט״ו, הֲלָכָה א׳)

2וְנַהֲפוֹךְ הוּא

מָרְדֳּכַי לֹא הִסְכִּים לִכְרֹעַ וּלְהִשְׁתַּחֲווֹת לְהָמָן. אֲבָל ה׳ כְּבָר הִקְדִּים רְפוּאָה לַמַּכָּה: אֶסְתֵּר הָיְתָה בְּבֵית הַמֶּלֶךְ, וְהִיא וּמָרְדֳּכַי הֵפֵרוּ אֶת עֲצַת הָמָן. וְנַהֲפוֹךְ הוּא – הַיְּהוּדִים גָּבְרוּ עַל שׂוֹנְאֵיהֶם, וְאֶת הָמָן תָּלוּ עַל הָעֵץ שֶׁהֵכִין לְמָרְדֳּכַי. אַחֲרֵי הַנֵּס הִתְעוֹרְרוּ הַיְּהוּדִים לַעֲלוֹת לְאֶרֶץ יִשְׂרָאֵל וְלִבְנוֹת אֶת בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי.
🤔 לָמָּה? ה׳ הוּא ״מְסַבֵּב הַסִּבּוֹת״: עוֹד לִפְנֵי שֶׁהָיְתָה הַגְּזֵרָה, הוּא הֵכִין אֶת הַהַצָּלָה.
💛 גַּם כְּשֶׁלֹּא רוֹאִים אֶת ה׳ – הוּא שָׁם, וְדוֹאֵג לָנוּ.
(פֶּרֶק ט״ו, הֲלָכָה א׳)

3קִיְּמוּ וְקִבְּלוּ

בְּהַר סִינַי קִבְּלוּ יִשְׂרָאֵל אֶת הַתּוֹרָה אַחֲרֵי נִסִּים גְּדוֹלִים, וְכִמְעַט לֹא הָיְתָה לָהֶם בְּרֵרָה – כְּאִלּוּ ״כָּפָה עֲלֵיהֶם הַר כְּגִיגִית״. בִּימֵי אֲחַשְׁוֵרוֹשׁ יָכְלוּ הַיְּהוּדִים לְהִתְבּוֹלֵל וּלְהִנָּצֵל מֵהַגְּזֵרָה – וּבְכָל זֹאת בָּחֲרוּ מֵרָצוֹן לְהִשָּׁאֵר דְּבוּקִים בַּה׳. הֵם חָזְרוּ בִּתְשׁוּבָה, הִתְחַזְּקוּ בַּמִּצְווֹת, וְאַף הוֹסִיפוּ מִצְוָה חֲדָשָׁה – מִצְווֹת הַפּוּרִים.
🤔 לָמָּה? אָמְרוּ חֲכָמִים: ״הֲדוּר קִבְּלוּהָ בִּימֵי אֲחַשְׁוֵרוֹשׁ״, שֶׁנֶּאֱמַר ״קִיְּמוּ וְקִבְּלוּ הַיְּהוּדִים״ – קִיְּמוּ מָה שֶׁקִּבְּלוּ כְּבָר. וְהַקַּבָּלָה הַזֹּאת, מֵאַהֲבָה, גְּדוֹלָה בְּמִדָּה מְסֻיֶּמֶת מִזּוֹ שֶׁבְּסִינַי.
💛 כְּשֶׁאֲנִי בּוֹחֵר בַּתּוֹרָה מֵרָצוֹן, וְלֹא כִּי מַכְרִיחִים אוֹתִי – זוֹ אַהֲבָה אֲמִתִּית.
(פֶּרֶק ט״ו, הֲלָכָה ב׳)

4כִּתְבוּנִי לְדוֹרוֹת

אֶסְתֵּר בִּקְּשָׁה מִן הַחֲכָמִים: ״כִּתְבוּנִי לְדוֹרוֹת״ – שֶׁיִּכְתְּבוּ אֶת הַסִּפּוּר בַּתַּנַ״ךְ, וְ״קִבְעוּנִי לְדוֹרוֹת״ – שֶׁיִּקְבְּעוּ אֶת פּוּרִים לְחַג. בַּסּוֹף כָּתְבוּ אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה אֶת מְגִלַּת אֶסְתֵּר בְּרוּחַ הַקֹּדֶשׁ, וְקָבְעוּ אֶת פּוּרִים לְדוֹרוֹת. בֵּית הַדִּין הַגָּדוֹל הַזֶּה הָיָה בּוֹ מֵאָה וְעֶשְׂרִים זְקֵנִים – וּבֵינֵיהֶם גַּם מָרְדֳּכַי הַיְּהוּדִי וְעֶזְרָא הַסּוֹפֵר.
🤔 לָמָּה? נֵס פּוּרִים הוּא הַנֵּס הָאַחֲרוֹן שֶׁנִּכְתַּב בַּתַּנַ״ךְ – ״אֶסְתֵּר סוֹף כָּל הַנִּסִּים״. פּוּרִים הוּא הַחֻלְיָה שֶׁמְּחַבֶּרֶת בֵּין הַתּוֹרָה שֶׁבִּכְתָב לַתּוֹרָה שֶׁבְּעַל פֶּה.
(פֶּרֶק ט״ו, הֲלָכָה ג׳)

5שֶׁבַע מִצְווֹת בְּפוּרִים

בְּפוּרִים יֵשׁ שֶׁבַע מִצְווֹת. אַרְבַּע מְיֻחָדוֹת לוֹ: קְרִיאַת הַמְּגִלָּה, מִשְׁלוֹחַ מָנוֹת אִישׁ לְרֵעֵהוּ, מַתָּנוֹת לָאֶבְיוֹנִים, וּמִשְׁתֶּה וְשִׂמְחָה. וְעוֹד שָׁלֹשׁ, שֶׁיֵּשׁ גַּם בְּחַגִּים אֲחֵרִים: קְרִיאַת הַתּוֹרָה (לִגְבָרִים), ״עַל הַנִּסִּים״ בַּתְּפִלָּה וּבְבִרְכַּת הַמָּזוֹן, וְאִסּוּר הֶסְפֵּד וְתַעֲנִית.
(פֶּרֶק ט״ו, הֲלָכָה ג׳)

6פְּרָזִים וּמֻקָּפִים

בְּרֹב הַמְּקוֹמוֹת חוֹגְגִים פּוּרִים בְּי״ד בַּאֲדָר. בַּעֲרִים שֶׁהָיוּ מֻקָּפוֹת חוֹמָה מִימוֹת יְהוֹשֻׁעַ בִּן נוּן, וּבְשׁוּשַׁן הַבִּירָה – חוֹגְגִים בְּט״ו בַּאֲדָר. הַיּוֹם רַק בִּירוּשָׁלַיִם חוֹגְגִים בְּט״ו, כִּי רַק עָלֶיהָ יֵשׁ מָסֹרֶת בְּרוּרָה שֶׁהָיְתָה מֻקֶּפֶת חוֹמָה בִּימֵי יְהוֹשֻׁעַ.
🤔 לָמָּה? בְּי״ג בַּאֲדָר נִלְחֲמוּ הַיְּהוּדִים בְּשׂוֹנְאֵיהֶם, וּבְי״ד נָחוּ וְשָׂמְחוּ. בְּשׁוּשַׁן הָיוּ הַרְבֵּה שׂוֹנְאִים, וְהַיְּהוּדִים נִלְחֲמוּ גַּם בְּי״ד – וְנָחוּ רַק בְּט״ו. וְלָמָּה ״מִימוֹת יְהוֹשֻׁעַ״ וְלֹא מִימֵי אֲחַשְׁוֵרוֹשׁ? כִּי אָז אֶרֶץ יִשְׂרָאֵל הָיְתָה חֲרֵבָה, וְלֹא הָיְתָה בָּהּ אַף עִיר מֻקֶּפֶת חוֹמָה. מִפְּנֵי כְּבוֹד אֶרֶץ יִשְׂרָאֵל קָבְעוּ לְפִי יְמֵי יְהוֹשֻׁעַ.
(פֶּרֶק ט״ו, הֲלָכָה ד׳)

7כֻּלָּם שׁוֹמְעִים מְגִלָּה – בְּרֹב עָם

הַכֹּל חַיָּבִים בְּמִקְרָא מְגִלָּה – גְּבָרִים, נָשִׁים וְגֵרִים. מִי שֶׁשּׁוֹמֵעַ אֶת הַקְּרִיאָה – יוֹצֵא יְדֵי חוֹבָה, אִם הוּא שׁוֹמֵעַ מִמִּי שֶׁחַיָּב בַּמִּצְוָה. הַשּׁוֹמֵעַ מִיֶּלֶד קָטָן, שֶׁעוֹד לֹא חַיָּב בְּמִצְווֹת – לֹא יָצָא. מִצְוָה לִשְׁמֹעַ מְגִלָּה ״בְּרֹב עָם״, עִם הַרְבֵּה אֲנָשִׁים, וּלְפָחוֹת בְּמִנְיָן. רַק אִם אֵין שׁוּם אֶפְשָׁרוּת – קוֹרְאִים לְבַד, בִּבְרָכָה.
🤔 לָמָּה? עִקַּר הַמִּצְוָה הוּא לְפַרְסֵם אֶת הַנֵּס – לְגַלּוֹת שֶׁה׳ מַנְהִיג אֶת הָעוֹלָם וּמְכַוֵּן אֶת הַכֹּל לְטוֹבָה. כָּל כָּךְ חָשׁוּב לִשְׁמֹעַ בְּרֹב עָם, עַד שֶׁהַכֹּהֲנִים בַּמִּקְדָּשׁ דָּחוּ אֶת קָרְבַּן הַתָּמִיד כְּדֵי לִשְׁמֹעַ מְגִלָּה עִם הַצִּבּוּר.
💛 כְּשֶׁכָּל בֵּית הַכְּנֶסֶת שׁוֹמֵעַ יַחַד אֶת הַסִּפּוּר – כֻּלָּנוּ מַכְרִיזִים: ה׳ שׁוֹמֵר עָלֵינוּ!
(פֶּרֶק ט״ו, הֲלָכָה ה׳)

8בַּלַּיְלָה וּבַיּוֹם

קוֹרְאִים אֶת הַמְּגִלָּה פַּעֲמַיִם: בַּלַּיְלָה, וְשׁוּב בַּיּוֹם. קְרִיאַת הַלַּיְלָה – מִצֵּאת הַכּוֹכָבִים וְעַד עֲלוֹת הַשַּׁחַר. קְרִיאַת הַיּוֹם – מֵהָנֵץ הַחַמָּה וְעַד הַשְּׁקִיעָה. הַזְּרִיזִים קוֹרְאִים מִיָּד אַחֲרֵי עַרְבִית, וּבַבֹּקֶר – מִיָּד אַחֲרֵי שַׁחֲרִית.
🤔 לָמָּה? כְּמוֹ שֶׁיִּשְׂרָאֵל זָעֲקוּ אֶל ה׳ בְּצָרָתָם בַּיּוֹם וּבַלַּיְלָה – כָּךְ קוֹרְאִים אֶת הַמְּגִלָּה בַּיּוֹם וּבַלַּיְלָה. לְדַעַת כַּמָּה מִגְּדוֹלֵי הָרִאשׁוֹנִים עִקַּר הַמִּצְוָה בַּיּוֹם, וְלָכֵן מִשְׁתַּדְּלִים יוֹתֵר שֶׁקְּרִיאַת הַיּוֹם תִּהְיֶה בְּרֹב עָם.
(פֶּרֶק ט״ו, הֲלָכָה ו׳)

9לִפְנֵי הַמְּגִלָּה

אָסוּר לֶאֱכֹל אוֹ לִישֹׁן לִפְנֵי שֶׁשּׁוֹמְעִים אֶת הַמְּגִלָּה – אֲבָל לִלְמֹד תּוֹרָה מֻתָּר. מִי שֶׁקָּשֶׁה לוֹ לְהַמְשִׁיךְ אֶת הַצּוֹם – מֻתָּר לוֹ לִשְׁתּוֹת, וּמִי שֶׁרָעֵב – מֻתָּר לוֹ לֶאֱכֹל מְעַט: פֵּרוֹת כַּמָּה שֶׁרוֹצֶה, וְלֶחֶם אוֹ עוּגוֹת עַד כַּבֵּיצָה. אֲבָל לֹא אֲרוּחָה. כָּךְ גַּם בַּבֹּקֶר, לִפְנֵי קְרִיאַת הַיּוֹם, וְכָךְ גַּם נָשִׁים וּבָנוֹת.
🤔 לָמָּה? שֶׁלֹּא יִמָּשֵׁךְ בַּאֲרוּחָה אוֹ יֵרָדֵם, וְיִשְׁכַּח לִשְׁמֹעַ מְגִלָּה.
(פֶּרֶק ט״ו, הֲלָכָה ו׳)

10נָשִׁים וּבָנוֹת בְּמִקְרָא מְגִלָּה

נָשִׁים וּבָנוֹת חַיָּבוֹת בְּמִקְרָא מְגִלָּה. אֲבָל נֶחְלְקוּ הָרִאשׁוֹנִים בְּמָה בְּדִיּוּק:
🤔 לָמָּה? בִּגְלַל הַמַּחֲלֹקֶת הוֹרוּ רֹב הָאַחֲרוֹנִים שֶׁאִשָּׁה לֹא תִּקְרָא כְּדֵי לְהוֹצִיא אִישׁ, אֶלָּא בִּשְׁעַת הַדְּחָק, כְּשֶׁאֵין שׁוּם אֶפְשָׁרוּת אַחֶרֶת. וְהָרוֹצוֹת לְהַדֵּר – שׁוֹמְעוֹת אֶת הַמְּגִלָּה בְּבֵית הַכְּנֶסֶת, בְּרֹב עָם.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת רַשִׁ״י וְהָרַמְבַּ״ם נָשִׁים חַיָּבוֹת בִּקְרִיאַת הַמְּגִלָּה כְּמוֹ גְּבָרִים, וְאִשָּׁה יְכוֹלָה לִקְרֹא לִבְנֵי בֵּיתָהּ.
לְדַעַת בַּעַל הֲלָכוֹת גְּדוֹלוֹת נָשִׁים חַיָּבוֹת רַק בִּשְׁמִיעַת הַמְּגִלָּה, וְלָכֵן אִשָּׁה לֹא יְכוֹלָה לִקְרֹא כְּדֵי לְהוֹצִיא אִישׁ.
(פֶּרֶק ט״ו, הֲלָכָה ז׳)

11בַּת שֶׁקּוֹרֵאת מְגִלָּה לְנָשִׁים

אִשָּׁה אוֹ בַּת שֶׁחַיֶּבֶת בְּמִצְווֹת יְכוֹלָה לִקְרֹא מִמְּגִלָּה כְּשֵׁרָה וּלְהוֹצִיא נָשִׁים אֲחֵרוֹת. הַאִם הִיא מְבָרֶכֶת?
🤔 לָמָּה? וּכְשֶׁאִישׁ קוֹרֵא מְגִלָּה לְנָשִׁים: הַמִּנְהָג הָרוֹוֵחַ שֶׁהַקּוֹרֵא מְבָרֵךְ בִּשְׁבִילָן, וְיֵשׁ נוֹהֲגִים שֶׁאִשָּׁה אַחַת מְבָרֶכֶת בִּשְׁבִיל כֻּלָּן. שְׁנֵי הַמִּנְהָגִים כְּשֵׁרִים.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת רֹב הַפּוֹסְקִים מְבָרֶכֶת לִפְנֵי הַקְּרִיאָה כְּמוֹ הַגְּבָרִים, וְאִם הֵן עֶשֶׂר – מְבָרֶכֶת בַּסּוֹף גַּם ״הָרָב אֶת רִיבֵנוּ״.
וְיֵשׁ אוֹמְרִים (בֶּן אִישׁ חַי, כַּף הַחַיִּים) כְּשֶׁהַקְּרִיאָה הִיא בִּשְׁבִיל נָשִׁים – לֹא מְבָרְכִים.
(פֶּרֶק ט״ו, הֲלָכָה ז׳)

12יְלָדִים בַּמְּגִלָּה

מִצְוָה לְחַנֵּךְ בָּנִים וּבָנוֹת לִשְׁמֹעַ מְגִלָּה, מֵהַזְּמַן שֶׁהֵם מְבִינִים וִיכוֹלִים לִשְׁמֹעַ אֶת כֻּלָּהּ כְּמוֹ שֶׁצָּרִיךְ. זֶה לֹא תָּלוּי בְּגִיל אֶלָּא בְּכָל יֶלֶד, וְרַבִּים מַגִּיעִים לְכָךְ אַחֲרֵי גִּיל תֵּשַׁע. מִנְהָג טוֹב לְהָבִיא גַּם בְּנֵי חָמֵשׁ אוֹ שֵׁשׁ, שֶׁמְּבִינִים אֶת עִקַּר הַסִּפּוּר. אֲבָל יְלָדִים קְטַנִּים שֶׁעֲלוּלִים לְהַפְרִיעַ – לֹא מְבִיאִים.
🤔 לָמָּה? אָסוּר לְחַנֵּךְ אֶת הַיְּלָדִים שֶׁלָּנוּ עַל חֶשְׁבּוֹן שְׁאָר הַמִּתְפַּלְּלִים, שֶׁצְּרִיכִים לִשְׁמֹעַ כָּל מִלָּה.
(פֶּרֶק ט״ו, הֲלָכָה ח׳)

13אַרְבָּעָה פְּסוּקִים וְרַעֲשָׁנִים

כְּדֵי לְעוֹרֵר שִׂמְחָה וּלְמַשֵּׁךְ אֶת הַיְּלָדִים, הַקָּהָל אוֹמֵר בְּקוֹל רָם אַרְבָּעָה פְּסוּקִים: ״אִישׁ יְהוּדִי״, ״וּמָרְדֳּכַי יָצָא״, ״לַיְּהוּדִים הָיְתָה אוֹרָה״ וְ״כִּי מָרְדֳּכַי הַיְּהוּדִי״ – וְהַחַזָּן חוֹזֵר וְקוֹרֵא אוֹתָם מִן הַמְּגִלָּה. נוֹהֲגִים גַּם ״לְהַכּוֹת אֶת הָמָן״ בְּרַעֲשָׁנִים. אֲבָל מַרְעִישִׁים רַק כְּשֶׁהַקּוֹרֵא עוֹצֵר וּמְחַכֶּה – אַף פַּעַם לֹא כְּשֶׁהוּא מַמְשִׁיךְ לִקְרֹא.
🤔 לָמָּה? הַהַכָּאָה מְבַטֵּאת שִׂנְאָה לָרְשָׁעִים וְשִׂמְחָה עַל מַפַּלְתָּם, וְהָרַמָּ״א כָּתַב: ״אֵין לְבַטֵּל שׁוּם מִנְהָג״. אֲבָל אָסוּר לַהֲפֹךְ אֶת הַטָּפֵל לָעִקָּר: אִם הָרַעַשׁ גּוֹרֵם שֶׁמִּישֶׁהוּ לֹא שָׁמַע מִלָּה – הוּא לֹא יָצָא.
(פֶּרֶק ט״ו, הֲלָכָה ח׳, הֲלָכָה י״ג)

14מְגִלָּה כְּשֵׁרָה

מְגִלַּת אֶסְתֵּר הִיא אֶחָד מִסִּפְרֵי הַקֹּדֶשׁ. כּוֹתְבִים אוֹתָהּ כְּמוֹ סֵפֶר תּוֹרָה: בְּיָד, לְשֵׁם קְדֻשַּׁת הַמְּגִלָּה, בִּדְיוֹ שָׁחֹר עַל קְלָף, וְתוֹפְרִים אֶת הַיְּרִיעוֹת בְּגִידִים. מְגִלָּה שֶׁנִּכְתְּבָה עַל נְיָר – פְּסוּלָה. אִם חֲסֵרוֹת בַּמְּגִלָּה אוֹתִיּוֹת, אֲבָל עִקָּרָהּ כָּתוּב כַּהֲלָכָה – אֶפְשָׁר לִקְרֹא בָּהּ בִּבְרָכָה, וּמַשְׁלִימִים אֶת הֶחָסֵר מִסֵּפֶר אוֹ בְּעַל פֶּה.
🤔 לָמָּה? בְּסֵפֶר תּוֹרָה שֶׁחֲסֵרָה בּוֹ אֲפִלּוּ אוֹת אַחַת – לֹא מְבָרְכִים. אֲבָל הַמְּגִלָּה נִקְרֵאת גַּם ״אִגֶּרֶת״ – מִכְתָּב שֶׁמְּסַפֵּר סִפּוּר – וְלָכֵן כָּל עוֹד עִקָּרָהּ כָּתוּב, יוֹצְאִים בָּהּ.
(פֶּרֶק ט״ו, הֲלָכָה ט׳)

15כָּל מִלָּה חֲשׁוּבָה

קוֹרְאִים מִתּוֹךְ מְגִלָּה כְּשֵׁרָה. מִי שֶׁאָמַר אֶת הַמְּגִלָּה בְּעַל פֶּה אוֹ קָרָא מִסֵּפֶר מֻדְפָּס – לֹא יָצָא. מִי שֶׁלֹּא שָׁמַע אֲפִלּוּ מִלָּה אַחַת – לֹא יָצָא. זוֹ הַהֲלָכָה הַחֲשׁוּבָה בְּיוֹתֵר בְּהִלְכוֹת הַמְּגִלָּה! מִי שֶׁפִּסְפֵּס מִלָּה – מַשְׁלִים אוֹתָהּ מִיָּד מֵהַסֵּפֶר שֶׁלְּפָנָיו, וּמַמְשִׁיךְ לִקְרֹא לְעַצְמוֹ עַד שֶׁמַּגִּיעַ לַמָּקוֹם שֶׁל הַקּוֹרֵא. אֲבָל מִי שֶׁשּׁוֹמֵעַ מִן הַקּוֹרֵא – לֹא קוֹרֵא יַחַד אִתּוֹ.
🤔 לָמָּה? כְּשֶׁיֵּשׁ הַרְבֵּה רַעַשׁ, וּבְמִיֻחָד אַחֲרֵי הַכָּאַת הָמָן, יֵשׁ שֶׁמְּפַסְפְּסִים מִלָּה – וּמַפְסִידִים אֶת כָּל הַמִּצְוָה. וּמִי שֶׁקּוֹרֵא יַחַד עִם הַחַזָּן מַפְרִיעַ לְעַצְמוֹ וְלַשְּׁכֵנִים לִשְׁמֹעַ.
(פֶּרֶק ט״ו, הֲלָכָה י׳)

16כַּסֵּדֶר וּבְעֵרָנוּת

קוֹרְאִים אֶת הַמְּגִלָּה כַּסֵּדֶר. מִי שֶׁקָּרָא שֶׁלֹּא כַּסֵּדֶר – ״לְמַפְרֵעַ״ – לֹא יָצָא. לָכֵן מִי שֶׁפִּסְפֵּס מִלָּה – מַשְׁלִים מִיָּד, וְלֹא בַּסּוֹף. מִי שֶׁשָּׁמַע כְּשֶׁהוּא מִתְנַמְנֵם – לֹא יָצָא, כִּי בְּוַדַּאי הָיוּ מִלִּים שֶׁלֹּא שָׁמַע. גַּם מִי שֶׁלֹּא מֵבִין עִבְרִית – יוֹצֵא בִּשְׁמִיעָה, אִם הוּא מְכַוֵּן לַמִּצְוָה. בִּשְׁמִיעָה דֶּרֶךְ טֶלֶפוֹן, רַמְקוֹל אוֹ שִׁדּוּר – נוֹהֲגִים לְהַחְמִיר, וְרַק כְּשֶׁאֵין בְּרֵרָה סוֹמְכִים עַל הַמְּקִלִּים.
(פֶּרֶק ט״ו, הֲלָכָה י״א)

17פּוֹרְשִׂים כְּמוֹ אִגֶּרֶת – וְהַבְּרָכוֹת

לִפְנֵי הַקְּרִיאָה בַּצִּבּוּר פּוֹשְׁטִים אֶת הַמְּגִלָּה וְכוֹפְלִים אוֹתָהּ דַּף עַל דַּף, כְּמוֹ מִכְתָּב, וְאַחֲרֵי הַקְּרִיאָה גּוֹלְלִים אוֹתָהּ בַּחֲזָרָה. הַחַזָּן עוֹמֵד, וְהַשּׁוֹמְעִים יְכוֹלִים לָשֶׁבֶת, וּבִשְׁעַת הַבְּרָכוֹת רֹב יִשְׂרָאֵל עוֹמְדִים. לִפְנֵי הַקְּרִיאָה בַּלַּיְלָה מְבָרְכִים שָׁלֹשׁ בְּרָכוֹת: ״עַל מִקְרָא מְגִלָּה״, ״שֶׁעָשָׂה נִסִּים לַאֲבוֹתֵינוּ״ וְ״שֶׁהֶחֱיָנוּ״. אַחֲרֵי הַקְּרִיאָה מְבָרְכִים ״הָרָב אֶת רִיבֵנוּ״ – רַק בְּמִנְיָן. וּבַבֹּקֶר – הַאִם חוֹזְרִים לְבָרֵךְ ״שֶׁהֶחֱיָנוּ״?
🤔 לָמָּה? הַבְּרָכוֹת עוֹזְרוֹת לְכַוֵּן אֶת הַלֵּב לַמִּצְוָה וּלְהָבִין אוֹתָהּ. בְּ״שֶׁהֶחֱיָנוּ״ נָכוֹן לְכַוֵּן גַּם עַל מִשְׁלוֹחַ מָנוֹת, מַתָּנוֹת לָאֶבְיוֹנִים וּסְעוּדָה: הַסְּפָרַדִּים – בַּלַּיְלָה, וְהָאַשְׁכְּנַזִּים – בַּיּוֹם.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הַסְּפָרַדִּים לֹא מְבָרְכִים שׁוּב ״שֶׁהֶחֱיָנוּ״ בַּבֹּקֶר, כִּי הַבְּרָכָה שֶׁל הַלַּיְלָה מוֹעִילָה לְכָל הַיּוֹם.
לְמִנְהַג הָאַשְׁכְּנַזִּים חוֹזְרִים וּמְבָרְכִים ״שֶׁהֶחֱיָנוּ״ גַּם בַּבֹּקֶר, כִּי קְרִיאַת הַיּוֹם הִיא מִצְוָה בִּפְנֵי עַצְמָהּ – וְהִיא הַקְּרִיאָה הַחֲשׁוּבָה.
(פֶּרֶק ט״ו, הֲלָכָה י״א, הֲלָכָה י״ב)

18עֲשֶׂרֶת בְּנֵי הָמָן וְ״אָרוּר הָמָן״

אֶת שְׁמוֹת עֲשֶׂרֶת בְּנֵי הָמָן כּוֹתְבִים בַּמְּגִלָּה בְּשׁוּרוֹת יְשָׁרוֹת: הַשֵּׁמוֹת בְּצַד יָמִין, וְהַמִּלָּה ״וְאֵת״ בְּצַד שְׂמֹאל. אֶת הָאוֹת ו׳ שֶׁל ״וַיְזָתָא״ כּוֹתְבִים זְקוּפָה. קוֹרְאִים אֶת כָּל הַשֵּׁמוֹת בִּנְשִׁימָה אַחַת. וְאַחֲרֵי הַקְּרִיאָה אוֹמְרִים: ״אָרוּר הָמָן, בָּרוּךְ מָרְדֳּכַי, אֲרוּרָה זֶרֶשׁ, בְּרוּכָה אֶסְתֵּר, אֲרוּרִים כָּל הָרְשָׁעִים, בְּרוּכִים כָּל הַצַּדִּיקִים, וְגַם חַרְבוֹנָא זָכוּר לַטּוֹב״.
🤔 לָמָּה? הַנְּשִׁימָה הָאַחַת וְהַו׳ הַזְּקוּפָה מְלַמְּדוֹת שֶׁכֻּלָּם מֵתוּ וְנִתְלוּ יַחַד. יְסוֹד אֱמוּנַת יִשְׂרָאֵל הוּא ה׳ אֶחָד, וּכְשֶׁהָרְשָׁעִים שֶׁנִּלְחֲמוּ בָּהּ נֶעֱנָשׁוּ יַחַד – הִתְגַּלְּתָה הָאַחְדוּת שֶׁל ה׳ בָּעוֹלָם.
(פֶּרֶק ט״ו, הֲלָכָה י״ג)

19״עַל הַנִּסִּים״

בְּפוּרִים מוֹסִיפִים ״עַל הַנִּסִּים״ בִּבְרָכַת הַהוֹדָאָה שֶׁבִּשְׁמוֹנֶה עֶשְׂרֵה, וּבַבְּרָכָה ״נוֹדֶה לְּךָ״ שֶׁבְּבִרְכַּת הַמָּזוֹן. בַּבְּרָכָה מֵעֵין שָׁלֹשׁ לֹא מַזְכִּירִים אֶת פּוּרִים. מִי שֶׁשָּׁכַח – יָצָא. אִם נִזְכַּר לִפְנֵי שֶׁאָמַר אֶת שֵׁם ה׳ בְּסוֹף הַבְּרָכָה – חוֹזֵר. וְאִם כְּבָר אָמַר – מַמְשִׁיךְ, וְטוֹב שֶׁיֹּאמַר אוֹתוֹ בְּסוֹף הַתְּפִלָּה, אוֹ בְּבִרְכַּת הַמָּזוֹן בֵּין הַ״הָרַחֲמָן״. מִי שֶׁהִתְחִיל סְעוּדָה בְּפוּרִים וְהִמְשִׁיךְ עַד הַלַּיְלָה – אוֹמֵר ״עַל הַנִּסִּים״ בְּבִרְכַּת הַמָּזוֹן.
🤔 לָמָּה? הוֹלְכִים אַחַר הַהַתְחָלָה: הַסְּעוּדָה הִתְחִילָה בְּפוּרִים.
(פֶּרֶק ט״ו, הֲלָכָה י״ד)

20קְרִיאַת הַתּוֹרָה – וְלָמָּה אֵין הַלֵּל

בְּפוּרִים קוֹרְאִים בַּתּוֹרָה לִשְׁלוֹשָׁה עוֹלִים אֶת פָּרָשַׁת ״וַיָּבֹא עֲמָלֵק״ – תִּשְׁעָה פְּסוּקִים, כִּי בָּהֶם נִגְמָר הָעִנְיָן. הַאִם קוֹרְאִים אֶת הַפָּסוּק הָאַחֲרוֹן פַּעֲמַיִם?
🤔 לָמָּה? יֵשׁ בַּתִּשְׁעָה פְּסוּקִים רֶמֶז: כָּל עוֹד לֹא נִמְחָה עֲמָלֵק, שְׁמוֹ שֶׁל ה׳ לֹא מִתְגַּלֶּה בִּשְׁלֵמוּת. וְלָמָּה לֹא אוֹמְרִים הַלֵּל בְּפוּרִים? נֶאֶמְרוּ שְׁלוֹשָׁה טְעָמִים: הַנֵּס הָיָה בְּחוּץ לָאָרֶץ; אַחֲרֵי הַנֵּס עֲדַיִן נִשְׁאַרְנוּ מְשֻׁעְבָּדִים לַאֲחַשְׁוֵרוֹשׁ; וּקְרִיאַת הַמְּגִלָּה הִיא בִּמְקוֹם הַהַלֵּל.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְפִי הַשֻּׁלְחָן עָרוּךְ – יֵשׁ נוֹהֲגִים לִקְרֹא אֶת הַפָּסוּק הָאַחֲרוֹן פַּעֲמַיִם, כְּדֵי לְהַשְׁלִים לַעֲשָׂרָה פְּסוּקִים.
לְפִי הָרַמָּ״א קוֹרְאִים אוֹתוֹ פַּעַם אַחַת.
(פֶּרֶק ט״ו, הֲלָכָה י״ד)

21יְמֵי שִׂמְחָה – וּבִגְדֵי שַׁבָּת

בְּי״ד וּבְט״ו בַּאֲדָר אָסוּר לְהִתְעַנּוֹת וּלְהַסְפִּיד – גַּם בְּפוּרִים וְגַם בְּשׁוּשַׁן פּוּרִים. בִּשְׁנֵי הַיָּמִים לֹא אוֹמְרִים תַּחֲנוּן וְלֹא ״לַמְנַצֵּחַ״. נוֹהֲגִים לִלְבֹּשׁ בְּלֵיל פּוּרִים וּבְיוֹמוֹ בִּגְדֵי שַׁבָּת וְיוֹם טוֹב.
💛 כְּשֶׁאֲנִי לוֹבֵשׁ בִּגְדֵי שַׁבָּת בְּפוּרִים – אֲנִי מַרְאֶה שֶׁזֶּה יוֹם שֶׁל כָּבוֹד וְשִׂמְחָה לִפְנֵי ה׳.
(פֶּרֶק ט״ו, הֲלָכָה י״ד)

22מְלָאכָה בְּפוּרִים

בַּהַתְחָלָה פּוּרִים לֹא נִקְבַּע כְּיוֹם שֶׁאָסוּר בִּמְלָאכָה, אֲבָל לִכְבוֹד הַיּוֹם נָהֲגוּ יִשְׂרָאֵל שֶׁלֹּא לַעֲשׂוֹת בּוֹ מְלָאכָה – וְהַמִּנְהָג מְחַיֵּב. לֹא יוֹצְאִים לָעֲבוֹדָה הַקְּבוּעָה, חוּץ מִמָּקוֹם שֶׁיִּגָּרֵם הֶפְסֵד גָּדוֹל, אוֹ עָנִי שֶׁאֵין לוֹ מָה לֶאֱכֹל. מֻתָּר: מְלָאכָה מְשַׂמַּחַת (כְּמוֹ לִשְׁתֹּל עֵץ נוֹי בַּגִּנָּה), מְלָאכָה שֶׁל מִצְוָה (כְּמוֹ לִכְתֹּב חִדּוּשֵׁי תּוֹרָה) וּמְלָאכָה קַלָּה (כְּמוֹ לִכְתֹּב מִכְתָּב אוֹ לִגְזֹז צִפָּרְנַיִם) – בִּתְנַאי שֶׁלֹּא מְבַטְּלִים בִּגְלָלָהּ אֶת שִׂמְחַת פּוּרִים. כְּבִיסָה בַּיָּד, תְּפִירָה וְתִסְפֹּרֶת – רַק לְצֹרֶךְ פּוּרִים. וּמָה בְּלֵיל פּוּרִים?
🤔 לָמָּה? אָמְרוּ: ״כָּל הָעוֹשֶׂה מְלָאכָה בְּפוּרִים – אֵינוֹ רוֹאֶה מִמֶּנָּה סִימַן בְּרָכָה״. וּמִי שֶׁגָּר בִּירוּשָׁלַיִם מֻתָּר לוֹ לַעֲבֹד בְּי״ד, וּמִי שֶׁגָּר בִּשְׁאָר הַמְּקוֹמוֹת – בְּט״ו.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ מֵהָאַחֲרוֹנִים אוֹסְרִים מְלָאכָה גַּם בַּלַּיְלָה.
לְדַעַת הָרַב מֶלָמֵד בַּלַּיְלָה מֻתָּר, כִּי מֵאַחַר שֶׁנֶּחְלְקוּ – אֵין מִנְהָג מֻסְכָּם לֶאֱסֹר.
(פֶּרֶק ט״ו, הֲלָכָה ט״ו)

📝 מָה לָמַדְנוּ

  • בְּפוּרִים ה׳ הָפַךְ אֶת הַגְּזֵרָה לִישׁוּעָה, וְיִשְׂרָאֵל קִבְּלוּ מֵחָדָשׁ אֶת הַתּוֹרָה מֵאַהֲבָה – ״קִיְּמוּ וְקִבְּלוּ״.
  • בְּרֹב הַמְּקוֹמוֹת חוֹגְגִים בְּי״ד בַּאֲדָר, וּבִירוּשָׁלַיִם – בְּט״ו.
  • כֻּלָּם – גַּם נָשִׁים וּבָנוֹת – שׁוֹמְעִים מְגִלָּה בַּלַּיְלָה וּבַיּוֹם, מִמְּגִלָּה כְּשֵׁרָה וּבְרֹב עָם.
  • מִי שֶׁלֹּא שָׁמַע מִלָּה אַחַת – לֹא יָצָא; מַרְעִישִׁים רַק כְּשֶׁהַקּוֹרֵא עוֹצֵר, וּמַשְׁלִימִים מִלָּה שֶׁפִּסְפְּסוּ מִיָּד.
  • מְבָרְכִים שָׁלֹשׁ בְּרָכוֹת לִפְנֵי הַמְּגִלָּה, אוֹמְרִים ״עַל הַנִּסִּים״, לוֹבְשִׁים בִּגְדֵי שַׁבָּת וְלֹא עוֹשִׂים מְלָאכָה בְּיוֹם פּוּרִים.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה בִּירוּשָׁלַיִם חוֹגְגִים פּוּרִים בְּט״ו, וּבִשְׁאָר הַמְּקוֹמוֹת – בְּי״ד?
  2. בְּאֶמְצַע הַמְּגִלָּה הָיָה רַעַשׁ, וּפִסְפַסְתָּ כַּמָּה מִלִּים. מָה תַּעֲשֶׂה – וּמָתַי?
  3. כַּמָּה פְּעָמִים מְבָרְכִים ״שֶׁהֶחֱיָנוּ״ עַל הַמְּגִלָּה – לְמִנְהַג הַסְּפָרַדִּים וּלְמִנְהַג הָאַשְׁכְּנַזִּים? וְלָמָּה?

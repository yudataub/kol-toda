סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת פּוּרִים
מַהֲדוּרַת יְלָדִים

🧱 פְּרָזִים וּמֻקָּפִים

פֶּרֶק י״ז · לַמַּהֲדוּרָה לַמּוֹרֶה
בִּרְחוֹבוֹת פֶּתַח תִּקְוָה כְּבָר נִגְמַר פּוּרִים, וְהַיְּלָדִים מְקַפְּלִים אֶת הַתַּחְפֹּשׂוֹת. אֲבָל בִּירוּשָׁלַיִם – הַחַגִּיגָה רַק מַתְחִילָה! אֵיךְ יִתָּכֵן שֶׁבְּאוֹתָהּ אֶרֶץ חוֹגְגִים בִּשְׁנֵי יָמִים שׁוֹנִים? וּמָה קוֹרֶה לְיֶלֶד יְרוּשַׁלְמִי שֶׁנּוֹסֵעַ לְסַבְתָּא בְּפֶתַח תִּקְוָה? וְאֵיךְ חוֹגְגִים כְּשֶׁפּוּרִים שֶׁל יְרוּשָׁלַיִם חָל בְּשַׁבָּת? בּוֹאוּ נִלְמַד.

1מָה זוֹ עִיר מֻקֶּפֶת חוֹמָה?

בְּכָל הַמְּקוֹמוֹת חוֹגְגִים פּוּרִים בְּי״ד בַּאֲדָר, וּבַעֲרִים שֶׁהָיוּ מֻקָּפוֹת חוֹמָה מִימוֹת יְהוֹשֻׁעַ בִּן נוּן – בְּט״ו בַּאֲדָר. אֵין הֶבְדֵּל אִם הָעִיר בְּאֶרֶץ יִשְׂרָאֵל אוֹ בְּחוּץ לָאָרֶץ, וְגַם אִם הַחוֹמָה נֶחְרְבָה מֵאָז – הָעִיר נֶחְשֶׁבֶת מֻקֶּפֶת. וְשׁוּשַׁן הַבִּירָה, שֶׁבָּהּ הָיָה הַנֵּס, חוֹגֶגֶת בְּט״ו – אַף שֶׁבִּימֵי יְהוֹשֻׁעַ עוֹד לֹא הָיְתָה קַיֶּמֶת.
🤔 לָמָּה? שׁוּשַׁן מְקַבֶּלֶת כָּבוֹד מְיֻחָד, כִּי שָׁם הָיָה הַנֵּס גָּדוֹל יוֹתֵר, וְשָׁם קָרוּ כָּל הַמַּעֲשִׂים שֶׁבַּמְּגִלָּה.
(פֶּרֶק י״ז, הֲלָכָה א׳)

2הַיּוֹם – רַק יְרוּשָׁלַיִם

בִּזְמַן אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה הָיוּ הַרְבֵּה עָרִים שֶׁיָּדְעוּ שֶׁהָיוּ מֻקָּפוֹת חוֹמָה בִּימֵי יְהוֹשֻׁעַ. אֲבָל בְּמֶשֶׁךְ הַשָּׁנִים הֶעָרִים נֶחְרְבוּ, וְהַמָּסֹרֶת עֲלֵיהֶן נִשְׁכְּחָה. לְמָשָׁל, יֵשׁ הַיּוֹם עִיר שֶׁנִּקְרֵאת לוֹד – אֲבָל אֵין אָנוּ יוֹדְעִים בְּוַדָּאוּת אִם הִיא בְּדִיּוּק בַּמָּקוֹם שֶׁל לוֹד הָעַתִּיקָה. וְעַל חֶבְרוֹן אָנוּ יוֹדְעִים שֶׁהָיְתָה קַיֶּמֶת – אֲבָל לֹא אִם הָיְתָה לָהּ אָז חוֹמָה. רַק עַל עִיר אַחַת נִשְׁאֲרָה מָסֹרֶת בְּרוּרָה: יְרוּשָׁלַיִם, עִיר קָדְשֵׁנוּ – וְרַק בָּהּ חוֹגְגִים הַיּוֹם פּוּרִים בְּט״ו.
💛 יְרוּשָׁלַיִם שׁוֹמֶרֶת עַל מָסֹרֶת בַּת אַלְפֵי שָׁנִים – גַּם בְּפוּרִים הִיא מְיֻחֶדֶת.
(פֶּרֶק י״ז, הֲלָכָה א׳)

3יְרוּשָׁלַיִם וְכָל הַשְּׁכוּנוֹת

אָמְרוּ חֲכָמִים: ״כְּרַךְ וְכָל הַסָּמוּךְ לוֹ וְכָל הַנִּרְאֶה עִמּוֹ – נִדּוֹן כִּכְרַךְ״. לָכֵן לֹא רַק בָּעִיר הָעַתִּיקָה חוֹגְגִים בְּט״ו, אֶלָּא בְּכָל הַשְּׁכוּנוֹת שֶׁסְּמוּכוֹת זוֹ לָזוֹ – כֻּלָּן נִגְרָרוֹת אַחֲרֵי יְרוּשָׁלַיִם הָעַתִּיקָה. וּמָה עִם שְׁכוּנוֹת שֶׁנִּבְנוּ בְּהַתְחָלָה רָחוֹק מִשְּׁאָר הָעִיר?
🤔 לָמָּה? וּבְעֶזְרַת ה׳ הַשְּׁאֵלָה הוֹלֶכֶת וְנִפְתֶּרֶת: יְרוּשָׁלַיִם נִבְנֵית וּמִתְרַחֶבֶת, וְהַשְּׁכוּנוֹת הָרְחוֹקוֹת נַעֲשׂוֹת צְמוּדוֹת לָעִיר.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ אוֹמְרִים רַק שְׁכוּנוֹת שֶׁיֵּשׁ בֵּינָן לְבֵין הָעִיר הָעַתִּיקָה רֶצֶף שֶׁל בָּתִּים – חוֹגְגוֹת בְּט״ו. שְׁכוּנָה שֶׁיֵּשׁ לִפְנֶיהָ הֶפְסֵק גָּדוֹל – חוֹגֶגֶת בְּי״ד.
וְיֵשׁ אוֹמְרִים – וְכָךְ הַמִּנְהָג הָרוֹוֵחַ כָּל הַשְּׁכוּנוֹת שֶׁשַּׁיָּכוֹת לָעִיר יְרוּשָׁלַיִם חוֹגְגוֹת בְּט״ו.
💛 ״בּוֹנֵה יְרוּשָׁלַיִם ה׳״ – אֲנַחְנוּ זוֹכִים לִרְאוֹת אֶת יְרוּשָׁלַיִם גְּדֵלָה מוּל עֵינֵינוּ.
(פֶּרֶק י״ז, הֲלָכָה ב׳)

4עָרִים שֶׁל סָפֵק

יֵשׁ עָרִים שֶׁיֵּשׁ עֲלֵיהֶן סָפֵק אִם הָיוּ מֻקָּפוֹת חוֹמָה בִּימֵי יְהוֹשֻׁעַ: טְבֶרְיָה, חֶבְרוֹן, שְׁכֶם, יָפוֹ, לוֹד, עַזָּה, צְפַת, עַכּוֹ וְחֵיפָה – וְיֵשׁ שֶׁמּוֹסִיפִים עוֹד. בְּכֻלָּן קוֹרְאִים אֶת הַמְּגִלָּה בְּי״ד בִּבְרָכָה. הַאִם קוֹרְאִים בָּהֶן גַּם בְּט״ו?
🤔 לָמָּה? לָמָּה בְּי״ד בִּבְרָכָה לְכָל הַדֵּעוֹת? כִּי אֲפִלּוּ יְרוּשַׁלְמִי שֶׁקָּרָא בְּי״ד – יָצָא בְּדִיעֲבַד. לְמַעֲשֶׂה, בְּרֹב הַמְּקוֹמוֹת הַמְּסֻפָּקִים חוֹגְגִים הַיּוֹם רַק בְּי״ד. וּבִמְקוֹמוֹת שֶׁהַסָּפֵק חָזָק יוֹתֵר, כְּמוֹ טְבֶרְיָה וְחֶבְרוֹן, רַבִּים קוֹרְאִים גַּם בְּט״ו, וְיֵשׁ שֶׁמְּקַיְּמִים שָׁם גַּם אֶת שְׁאָר מִצְווֹת הַפּוּרִים.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ אוֹמְרִים חוֹגְגִים רַק בְּי״ד. רַק מִי שֶׁרוֹצֶה לְהַחְמִיר קוֹרֵא גַּם בְּט״ו – בְּלִי בְּרָכָה.
וְיֵשׁ אוֹמְרִים (וְכָךְ בַּשֻּׁלְחָן עָרוּךְ) קוֹרְאִים גַּם בְּט״ו, בְּלִי בְּרָכָה – כְּדֵי שֶׁהַסָּפֵק לֹא יִשָּׁכַח, וְגַם לִכְבוֹד אֶרֶץ יִשְׂרָאֵל.
(פֶּרֶק י״ז, הֲלָכָה ג׳)

5״פָּרוּז בֶּן יוֹמוֹ״

״פְּרָזִים״ הֵם הַמְּקוֹמוֹת שֶׁחוֹגְגִים בְּי״ד, וּ״מֻקָּפִים״ – הַמְּקוֹמוֹת שֶׁחוֹגְגִים בְּט״ו. מָתַי חוֹגֵג מִי שֶׁנּוֹסֵעַ? הַכְּלָל: הַחִיּוּב נִקְבָּע לְפִי הַמָּקוֹם שֶׁבּוֹ הָאָדָם נִמְצָא בְּיוֹם הַפּוּרִים, וְלֹא לְפִי הַמָּקוֹם שֶׁבּוֹ הוּא גָּר. וְהָרֶגַע הַקּוֹבֵעַ הוּא עֲלוֹת הַשַּׁחַר. יְרוּשַׁלְמִי שֶׁבָּא לְעִיר אַחֶרֶת בְּלֵיל י״ד וְנִשְׁאַר שָׁם עַד אַחֲרֵי עֲלוֹת הַשַּׁחַר – חוֹגֵג בְּי״ד, כְּמוֹ אַנְשֵׁי הַמָּקוֹם. אֲבָל אִם הִגִּיעַ רַק בַּבֹּקֶר, אַחֲרֵי עֲלוֹת הַשַּׁחַר, וְחָזַר לִירוּשָׁלַיִם – הוּא חוֹגֵג בְּט״ו, כְּמוֹ כָּל הַיְּרוּשַׁלְמִים.
🤔 לָמָּה? בַּעֲלוֹת הַשַּׁחַר מַתְחִיל זְמַן קְרִיאַת הַמְּגִלָּה שֶׁל הַיּוֹם – וְלָכֵן הַמָּקוֹם שֶׁבּוֹ אָדָם נִמְצָא בָּרֶגַע הַזֶּה קוֹבֵעַ אֶת הַפּוּרִים שֶׁלּוֹ. גַּם מִי שֶׁשּׁוֹהֶה יוֹם אֶחָד בִּלְבַד – נִקְרָא ״פָּרוּז בֶּן יוֹמוֹ״.
(פֶּרֶק י״ז, הֲלָכָה ד׳)

6פּוּרִים פַּעֲמַיִם?

בֶּן פְּרָזִים שֶׁחָגַג פּוּרִים בְּי״ד, וּבָא לִירוּשָׁלַיִם בְּלֵיל ט״ו וְנִשְׁאַר שָׁם עַד אַחֲרֵי עֲלוֹת הַשַּׁחַר – נֶחְשָׁב כִּירוּשַׁלְמִי, וְחוֹגֵג שׁוּב אֶת כָּל מִצְווֹת הַפּוּרִים בְּט״ו. אֲבָל אֶת בִּרְכוֹת הַמְּגִלָּה הוּא מִשְׁתַּדֵּל לִשְׁמֹעַ מֵאָדָם אַחֵר, וְאִם הוּא קוֹרֵא לְעַצְמוֹ – לֹא מְבָרֵךְ.
🤔 לָמָּה? יֵשׁ אוֹמְרִים שֶׁמִּי שֶׁכְּבָר חָגַג פּוּרִים – אֵינוֹ צָרִיךְ לַחְגֹּג שׁוּב. אַף שֶׁהַהֲלָכָה הִיא שֶׁחוֹגֵג שׁוּב, בִּבְרָכוֹת מְחַשְּׁבִים גַּם אֶת דַּעְתָּם.
💛 מִי שֶׁזּוֹכֶה – חוֹגֵג פּוּרִים פַּעֲמַיִם! שִׂמְחָה כְּפוּלָה.
(פֶּרֶק י״ז, הֲלָכָה ד׳)

7פּוּרִים הַמְשֻׁלָּשׁ

י״ד בַּאֲדָר לְעוֹלָם לֹא חָל בְּשַׁבָּת, אֲבָל ט״ו – יָכוֹל לָחוּל בְּשַׁבָּת. אָז הַפּוּרִים שֶׁל יְרוּשָׁלַיִם נִקְרָא ״פּוּרִים הַמְשֻׁלָּשׁ״, כִּי הַמִּצְווֹת שֶׁלּוֹ מִתְחַלְּקוֹת לִשְׁלוֹשָׁה יָמִים. לָמָּה לֹא חוֹגְגִים הַכֹּל בְּשַׁבָּת? חֲכָמִים גָּזְרוּ שֶׁלֹּא לִקְרֹא מְגִלָּה בְּשַׁבָּת, שֶׁמָּא מִישֶׁהוּ יִשָּׂא אֶת הַמְּגִלָּה בִּרְשׁוּת הָרַבִּים. וְגַם אֶת סְעוּדַת הַפּוּרִים לֹא עוֹשִׂים בְּשַׁבָּת, כִּי הַסְּעוּדָה צְרִיכָה לִהְיוֹת בִּגְלַל פּוּרִים – וְלֹא בִּגְלַל שַׁבָּת.
(פֶּרֶק י״ז, הֲלָכָה ה׳)

8שִׁשִּׁי, שַׁבָּת וְרִאשׁוֹן

בְּיוֹם שִׁשִּׁי קוֹרְאִים אֶת הַמְּגִלָּה – וּמִשְׁתַּדְּלִים בְּמִנְיָן – וְנוֹתְנִים מַתָּנוֹת לָאֶבְיוֹנִים. בְּשַׁבָּת אוֹמְרִים ״עַל הַנִּסִּים״ בַּתְּפִלָּה וּבְבִרְכַּת הַמָּזוֹן, וּמוֹצִיאִים שְׁנֵי סִפְרֵי תוֹרָה: בָּרִאשׁוֹן קוֹרְאִים אֶת פָּרָשַׁת הַשָּׁבוּעַ, וּבַשֵּׁנִי – ״וַיָּבֹא עֲמָלֵק״. בְּיוֹם רִאשׁוֹן עוֹשִׂים אֶת סְעוּדַת הַפּוּרִים, וְשׁוֹלְחִים מִשְׁלוֹחַ מָנוֹת. וְטוֹב לְהוֹסִיף בְּשַׁבָּת בָּשָׂר וּמַאֲכָלִים טוֹבִים.
🤔 לָמָּה? אֶת הַמְּגִלָּה מַקְדִּימִים לְיוֹם שִׁשִּׁי, כִּי לֹא מְאַחֲרִים אֶת פִּרְסוּם הַנֵּס – ״וְלֹא יַעֲבוֹר״. וְהַמַּתָּנוֹת לָאֶבְיוֹנִים – יַחַד עִם הַמְּגִלָּה, כִּי ״עֵינֵיהֶם שֶׁל עֲנִיִּים נְשׂוּאוֹת לְמִקְרָא מְגִלָּה״, וְהֵם יִצְטַעֲרוּ אִם לֹא יְקַבְּלוּ. אֶת הַסְּעוּדָה אֶפְשָׁר לַעֲשׂוֹת רַק מִשֶּׁהִגִּיעַ יוֹם הַנֵּס – וּמֵאַחַר שֶׁלֹּא עוֹשִׂים אוֹתָהּ בְּשַׁבָּת, דּוֹחִים לְיוֹם רִאשׁוֹן. וּמִשְׁלוֹחַ מָנוֹת קָשׁוּר לַסְּעוּדָה.
💛 גַּם כְּשֶׁהַפּוּרִים מִתְפַּצֵּל לִשְׁלוֹשָׁה יָמִים – הַשִּׂמְחָה רַק גְּדֵלָה.
(פֶּרֶק י״ז, הֲלָכָה ה׳)

📝 מָה לָמַדְנוּ

  • עִיר שֶׁהָיְתָה מֻקֶּפֶת חוֹמָה מִימוֹת יְהוֹשֻׁעַ בִּן נוּן, וְשׁוּשַׁן – חוֹגְגוֹת פּוּרִים בְּט״ו; וְהַיּוֹם יֵשׁ מָסֹרֶת בְּרוּרָה רַק עַל יְרוּשָׁלַיִם.
  • כָּל הַשְּׁכוּנוֹת הַסְּמוּכוֹת לִירוּשָׁלַיִם חוֹגְגוֹת עִמָּהּ בְּט״ו.
  • בְּעָרִים שֶׁל סָפֵק קוֹרְאִים בְּי״ד בִּבְרָכָה, וְיֵשׁ קוֹרְאִים גַּם בְּט״ו בְּלִי בְּרָכָה.
  • הַמָּקוֹם שֶׁבּוֹ אָדָם נִמְצָא בַּעֲלוֹת הַשַּׁחַר קוֹבֵעַ מָתַי הוּא חוֹגֵג.
  • בְּפוּרִים הַמְשֻׁלָּשׁ: שִׁשִּׁי – מְגִלָּה וּמַתָּנוֹת לָאֶבְיוֹנִים; שַׁבָּת – ״עַל הַנִּסִּים״ וּקְרִיאַת הַתּוֹרָה; רִאשׁוֹן – סְעוּדָה וּמִשְׁלוֹחַ מָנוֹת.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה הַיּוֹם חוֹגְגִים פּוּרִים בְּט״ו רַק בִּירוּשָׁלַיִם, וְלֹא גַּם בְּלוֹד אוֹ בְּחֶבְרוֹן?
  2. יֶלֶד מִירוּשָׁלַיִם יָשַׁן אֵצֶל סַבְתָּא בְּפֶתַח תִּקְוָה בְּלֵיל י״ד, וְנִשְׁאַר שָׁם עַד הַבֹּקֶר. מָתַי הוּא חוֹגֵג פּוּרִים? וְלָמָּה?
  3. בְּפוּרִים הַמְשֻׁלָּשׁ – מָה עוֹשִׂים בְּיוֹם שִׁשִּׁי, מָה בְּשַׁבָּת וּמָה בְּיוֹם רִאשׁוֹן?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״שִׂמְחָה, חֶסֶד וִירוּשָׁלַיִם״

סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת פּוּרִים
מַהֲדוּרַת יְלָדִים

🎭 חֹדֶשׁ אֲדָר

פֶּרֶק י״ד · לַמַּהֲדוּרָה לַמּוֹרֶה
רֹאשׁ חֹדֶשׁ אֲדָר! בַּכִּתָּה תּוֹלִים קִשּׁוּטִים, מַתְחִילִים לְתַכְנֵן תַּחְפֹּשׂוֹת, וְכֻלָּם מְחַיְּכִים. אֲבָל לָמָּה דַּוְקָא בַּחֹדֶשׁ הַזֶּה ״מַרְבִּים בְּשִׂמְחָה״? וּמָה עוֹשִׂים בְּאֲדָר עוֹד לִפְנֵי פּוּרִים – אֵילוּ פָּרָשִׁיּוֹת קוֹרְאִים, מִי זֶה עֲמָלֵק, לָמָּה צָמִים בְּתַעֲנִית אֶסְתֵּר וּמָה זֶה ״מַחֲצִית הַשֶּׁקֶל״? בּוֹאוּ נִתְכּוֹנֵן יַחַד לַחֹדֶשׁ הַשָּׂמֵחַ.

1מִשֶּׁנִּכְנַס אֲדָר מַרְבִּים בְּשִׂמְחָה

אָמְרוּ חֲכָמִים: ״מִשֶּׁנִּכְנַס אָב מְמַעֲטִין בְּשִׂמְחָה״, וּ״מִשֶּׁנִּכְנַס אֲדָר מַרְבִּים בְּשִׂמְחָה״. אֵין מִקְרֶה בָּעוֹלָם, וּלְכָל זְמַן יֵשׁ אֹפִי מִשֶּׁלּוֹ. נֵס פּוּרִים קָרָה בְּאֶמְצַע חֹדֶשׁ אֲדָר – סִימָן שֶׁיֵּשׁ לַחֹדֶשׁ הַזֶּה סְגֻלָּה לַהֲפֹךְ אֶת הָרַע לְטוֹב.
🤔 לָמָּה? שִׂמְחָה רְגִילָה הִיא עַל הַטּוֹב שֶׁבָּעוֹלָם, וְהִיא לֹא שְׁלֵמָה, כִּי יֵשׁ בָּעוֹלָם גַּם צַעַר. אֲבָל כְּשֶׁגַּם הָרַע מִתְהַפֵּךְ לְטוֹב – הַשִּׂמְחָה גְּדוֹלָה וּשְׁלֵמָה. כָּךְ הָיָה בְּפוּרִים.
💛 מִפּוּרִים אֲנַחְנוּ לוֹמְדִים שֶׁכָּל מָה שֶׁה׳ עוֹשֶׂה – גַּם מָה שֶׁנִּרְאֶה רַע – יִתְהַפֵּךְ בַּסּוֹף לְטוֹבָה.
(פֶּרֶק י״ד, הֲלָכָה א׳)

2אַרְבַּע פָּרָשִׁיּוֹת

חֲכָמִים תִּקְּנוּ לִקְרֹא אַרְבַּע פָּרָשִׁיּוֹת מְיֻחָדוֹת: ״שְׁקָלִים״, ״זָכוֹר״, ״פָּרָה״ וְ״הַחֹדֶשׁ״. קוֹרְאִים אוֹתָן בְּשַׁבְּתוֹת חֹדֶשׁ אֲדָר, וְאֶת הָרִאשׁוֹנָה – בְּדֶרֶךְ כְּלָל בַּשַּׁבָּת שֶׁלִּפְנֵי אֲדָר. בַּשַּׁבָּתוֹת הָאֵלֶּה מוֹצִיאִים שְׁנֵי סִפְרֵי תוֹרָה: בָּרִאשׁוֹן קוֹרְאִים אֶת פָּרָשַׁת הַשָּׁבוּעַ, וּבַשֵּׁנִי קוֹרְאִים לַמַּפְטִיר אֶת הַפָּרָשָׁה הַמְּיֻחֶדֶת. גַּם הַהַפְטָרָה עוֹסֶקֶת בַּפָּרָשָׁה הַמְּיֻחֶדֶת.
(פֶּרֶק י״ד, הֲלָכָה ב׳)

3לָמָּה קוֹרְאִים אֶת הַפָּרָשִׁיּוֹת?

״שְׁקָלִים״ – מַזְכִּירָה לָתֵת מַחֲצִית הַשֶּׁקֶל, שֶׁבָּהּ קָנוּ אֶת קָרְבְּנוֹת הַצִּבּוּר. ״זָכוֹר״ – לִזְכֹּר אֶת מָה שֶׁעָשָׂה לָנוּ עֲמָלֵק, לִפְנֵי פּוּרִים. ״פָּרָה״ – לוֹמְדִים אֵיךְ מִטַּהֲרִים, כְּדֵי לְהַקְרִיב קָרְבַּן פֶּסַח. ״הַחֹדֶשׁ״ – עַל קִדּוּשׁ הַחֹדֶשׁ וּמִצְווֹת הַפֶּסַח, כִּי נִיסָן הוּא רִאשׁוֹן לֶחֳדָשִׁים. כְּשֶׁרֹאשׁ חֹדֶשׁ אֲדָר אוֹ נִיסָן חָל בְּשַׁבָּת – מוֹצִיאִים שְׁלוֹשָׁה סִפְרֵי תוֹרָה: פָּרָשַׁת הַשָּׁבוּעַ, רֹאשׁ חֹדֶשׁ, וְהַפָּרָשָׁה הַמְּיֻחֶדֶת.
🤔 לָמָּה? מִנִּיסָן הִתְחִילוּ לִקְנוֹת קָרְבָּנוֹת מֵהַתְּרוּמָה הַחֲדָשָׁה, וְלָכֵן קוֹרְאִים ״שְׁקָלִים״ חֹדֶשׁ לִפְנֵי. אַף שֶׁהַיּוֹם אֵין בֵּית מִקְדָּשׁ, קוֹרְאִים זֵכֶר לַמִּקְדָּשׁ. לְרֹב הַפּוֹסְקִים קְרִיאַת ״זָכוֹר״ הִיא מִצְוָה מִן הַתּוֹרָה, וְלָכֵן מְדַקְדְּקִים בָּהּ יוֹתֵר מִכֻּלָּן.
(פֶּרֶק י״ד, הֲלָכָה ב׳)

4שָׁנָה מְעֻבֶּרֶת

הַחֳדָשִׁים שֶׁלָּנוּ נִקְבָּעִים לְפִי הַלְּבָנָה, וְהַשָּׁנִים – לְפִי הַחַמָּה וְעוֹנוֹת הַשָּׁנָה. כְּדֵי שֶׁפֶּסַח יִהְיֶה תָּמִיד בָּאָבִיב, מוֹסִיפִים לִפְעָמִים חֹדֶשׁ – אֲדָר שֵׁנִי. בְּכָל תְּשַׁע עֶשְׂרֵה שָׁנִים יֵשׁ שֶׁבַע שָׁנִים מְעֻבָּרוֹת. בְּשָׁנָה מְעֻבֶּרֶת חוֹגְגִים פּוּרִים וְקוֹרְאִים אֶת אַרְבַּע הַפָּרָשִׁיּוֹת בַּאֲדָר הַשֵּׁנִי.
🤔 לָמָּה? כָּךְ הַשִּׂמְחָה עַל גְּאֻלַּת פּוּרִים סְמוּכָה לִגְאֻלַּת מִצְרַיִם בְּנִיסָן. וּשְׁלוֹשׁ מֵהַפָּרָשִׁיּוֹת הֵן הֲכָנָה לְנִיסָן, וְ״זָכוֹר״ צְרִיכָה לִהְיוֹת לִפְנֵי פּוּרִים.
(פֶּרֶק י״ד, הֲלָכָה ג׳)

5פּוּרִים קָטָן

גַּם בַּאֲדָר הָרִאשׁוֹן יֵשׁ חֲגִיגִיּוּת: בְּי״ד וּבְט״ו בּוֹ לֹא מִתְעַנִּים, לֹא מַסְפִּידִים וְלֹא נוֹפְלִים אַפַּיִם (לֹא אוֹמְרִים תַּחֲנוּן). נָכוֹן לְהַרְבּוֹת קְצָת בִּסְעוּדָה בְּי״ד בַּאֲדָר הָרִאשׁוֹן – וְהוּא נִקְרָא ״פּוּרִים קָטָן״. וְ״מַרְבִּים בְּשִׂמְחָה״ כְּבָר מֵרֹאשׁ חֹדֶשׁ אֲדָר הָרִאשׁוֹן.
(פֶּרֶק י״ד, הֲלָכָה ג׳)

6שָׁלֹשׁ מִצְווֹת עַל עֲמָלֵק

שָׁלֹשׁ מִצְווֹת בַּתּוֹרָה עוֹסְקוֹת בַּעֲמָלֵק: ״זָכוֹר אֵת אֲשֶׁר עָשָׂה לְךָ עֲמָלֵק״ – מִצְוַת עֲשֵׂה לִזְכֹּר. ״לֹא תִּשְׁכָּח״ – מִצְוַת לֹא תַעֲשֶׂה, לֹא לִשְׁכֹּחַ. וְ״תִּמְחֶה אֶת זֵכֶר עֲמָלֵק״ – מִצְוַת עֲשֵׂה לִמְחוֹת אוֹתוֹ. מִיָּד אַחֲרֵי יְצִיאַת מִצְרַיִם, בְּלִי שׁוּם סִבָּה, בָּא עֲמָלֵק וְתָקַף אֶת הָעֲבָדִים הָעֲיֵפִים שֶׁיָּצְאוּ לְחֵרוּת.
🤔 לָמָּה? עֲמָלֵק שׂוֹנֵא אֶת יִשְׂרָאֵל, וּמִמֵּילָא גַּם אֶת הַתּוֹרָה וְאֶת הַדֶּרֶךְ שֶׁל חֶסֶד וֶאֱמֶת שֶׁה׳ רוֹצֶה לְהָבִיא לָעוֹלָם. הוּא הִתְחִיל אֶת שִׂנְאַת יִשְׂרָאֵל בָּעוֹלָם.
(פֶּרֶק י״ד, הֲלָכָה ד׳)

7לָמָּה צָרִיךְ לִזְכֹּר?

יְהוּדִי הוּא רַחֲמָן וּבַעַל חֶסֶד מִטִּבְעוֹ, וְהַרְבֵּה מִצְווֹת מְחַנְּכוֹת אוֹתוֹ לְכָךְ. מִטִּבְעוֹ הוּא הָיָה נוֹטֶה לִסְלֹחַ גַּם לַעֲמָלֵק. לָכֵן הַתּוֹרָה צִוְּתָה לִזְכֹּר אֶת מַעֲשָׂיו.
🤔 לָמָּה? כְּשֶׁזּוֹכְרִים אֶת עֲמָלֵק, זוֹכְרִים שֶׁיֵּשׁ רֹעַ בָּעוֹלָם וְשֶׁצָּרִיךְ לְהִלָּחֵם בּוֹ. רַק אַחַר כָּךְ אֶפְשָׁר לְתַקֵּן אֶת הָעוֹלָם.
💛 מִי שֶׁאוֹהֵב אֶת ה׳ וְאֶת הַטּוֹב – לֹא נִשְׁאָר אָדִישׁ לָרֹעַ.
(פֶּרֶק י״ד, הֲלָכָה ד׳)

8שָׁאוּל וַאֲגַג

אַחֲרֵי שֶׁיִּשְׂרָאֵל הִתְיַשְּׁבוּ בְּאַרְצָם וּמִנּוּ מֶלֶךְ, צִוָּה שְׁמוּאֵל הַנָּבִיא אֶת שָׁאוּל הַמֶּלֶךְ לְהִלָּחֵם בַּעֲמָלֵק. אֲבָל שָׁאוּל לֹא קִיֵּם אֶת הַמִּצְוָה כְּמוֹ שֶׁצָּרִיךְ: הוּא חָמַל עַל אֲגַג מֶלֶךְ עֲמָלֵק וְעַל מֵיטַב הַצֹּאן וְהַבָּקָר. לָכֵן הֶעֱבִיר ה׳ מִמֶּנּוּ אֶת הַמַּלְכוּת וּנְתָנָהּ לְדָוִד.
🤔 לָמָּה? סִפְּרוּ חֲזַ״ל: בַּזְּמַן שֶׁשָּׁאוּל הִתְמַהְמֵהַּ בַּהֲרִיגַת אֲגַג, נִמְשַׁךְ זַרְעוֹ – וּמִמֶּנּוּ נוֹלַד לְבַסּוֹף הָמָן הָאֲגָגִי, שֶׁרָצָה לְהַשְׁמִיד אֶת כָּל הַיְּהוּדִים.
(פֶּרֶק י״ד, הֲלָכָה ה׳)

9פָּרָשַׁת זָכוֹר

חֲכָמִים תִּקְּנוּ לְקַיֵּם אֶת מִצְוַת הַזְּכִירָה בִּקְרִיאַת פָּרָשַׁת ״זָכוֹר״ פַּעַם אַחַת בְּשָׁנָה – בַּשַּׁבָּת שֶׁלִּפְנֵי פּוּרִים. לֹא צָרִיךְ שֶׁכָּל אֶחָד יִקְרָא בְּעַצְמוֹ: הַשּׁוֹמֵעַ מִן הַקּוֹרֵא בַּתּוֹרָה – יוֹצֵא יְדֵי חוֹבָה. רָאוּי לִקְרֹא מִסֵּפֶר תּוֹרָה מְהֻדָּר, וְהַקּוֹרֵא מְדַקְדֵּק מְאוֹד בַּקְּרִיאָה.
🤔 לָמָּה? אִם עוֹבֶרֶת שָׁנָה שְׁלֵמָה בְּלִי לִזְכֹּר – הַדָּבָר נִשְׁכָּח. זְכִירָה אַחַת בְּשָׁנָה שׁוֹמֶרֶת עַל הַזִּכָּרוֹן. וְקוֹרְאִים לִפְנֵי פּוּרִים, כְּדֵי לְחַבֵּר אֶת זְכִירַת עֲמָלֵק לְמַפַּלַת הָמָן, שֶׁהָיָה מִזַּרְעוֹ.
(פֶּרֶק י״ד, הֲלָכָה ו׳)

10מְכַוְּנִים לַמִּצְוָה

צָרִיךְ לְכַוֵּן בִּשְׁעַת הַקְּרִיאָה לְקַיֵּם אֶת מִצְוַת זְכִירַת עֲמָלֵק, וְטוֹב שֶׁהַגַּבַּאי אוֹ הַקּוֹרֵא יַכְרִיז עַל כָּךְ לִפְנֵי הַקְּרִיאָה. לְכַתְּחִלָּה טוֹב לִשְׁמֹעַ בַּמִּבְטָא וּבַטְּעָמִים שֶׁל הַמִּשְׁפָּחָה, אֲבָל כָּל הַנֻּסָּחִים – סְפָרַדִּי, אַשְׁכְּנַזִּי אוֹ תֵּימָנִי – כְּשֵׁרִים לְכֻלָּם. מִי שֶׁנִּמְצָא בְּמָקוֹם שֶׁאֵין בּוֹ מִנְיָן – קוֹרֵא בְּסֵפֶר תּוֹרָה אִם יֵשׁ, וְאִם אֵין – קוֹרֵא אֶת הַפָּרָשָׁה מִתּוֹךְ חֻמָּשׁ אוֹ סִדּוּר.
🤔 לָמָּה? ״מִצְווֹת צְרִיכוֹת כַּוָּנָה״ – כְּשֶׁמְּכַוְּנִים, הַשְּׁמִיעָה נַעֲשֵׂית מִצְוָה.
(פֶּרֶק י״ד, הֲלָכָה ו׳)

11נָשִׁים וּבָנוֹת וּפָרָשַׁת זָכוֹר

הַאִם נָשִׁים חַיָּבוֹת לִשְׁמֹעַ פָּרָשַׁת ״זָכוֹר״?
🤔 לָמָּה? לְמַעֲשֶׂה נָשִׁים פְּטוּרוֹת, אֲבָל לְכַתְּחִלָּה טוֹב שֶׁיִּשְׁמְעוּ אֶת פָּרָשַׁת ״זָכוֹר״ – וְכָךְ נוֹהֲגוֹת נָשִׁים רַבּוֹת. מִי שֶׁקָּשֶׁה לָהּ לָבוֹא לְבֵית הַכְּנֶסֶת – תִּקְרָא בְּעַצְמָהּ אֶת הַפָּרָשָׁה מִתּוֹךְ חֻמָּשׁ. וְיֵשׁ מְקוֹמוֹת שֶׁקּוֹרְאִים לַנָּשִׁים מִסֵּפֶר תּוֹרָה, גַּם בְּלִי מִנְיָן.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת רֹב הַפּוֹסְקִים נָשִׁים פְּטוּרוֹת, כִּי הַזְּכִירָה קְשׁוּרָה לַמִּלְחָמָה בַּעֲמָלֵק, וְנָשִׁים אֵינָן מְצֻוּוֹת לְהִלָּחֵם.
וְיֵשׁ אוֹמְרִים גַּם נָשִׁים חַיָּבוֹת, כִּי הֵן מְסַיְּעוֹת לַלּוֹחֲמִים, וּמִן הַתּוֹרָה אֵין לַמִּצְוָה זְמַן קָבוּעַ.
(פֶּרֶק י״ד, הֲלָכָה ז׳)

12תַּעֲנִית אֶסְתֵּר

מִימֵי הַגְּאוֹנִים נוֹהֲגִים כָּל יִשְׂרָאֵל לְהִתְעַנּוֹת בְּי״ג בַּאֲדָר. זֶה זֵכֶר לַצּוֹם שֶׁל אֶסְתֵּר לִפְנֵי שֶׁנִּכְנְסָה אֶל הַמֶּלֶךְ כְּדֵי לְבַטֵּל אֶת הַגְּזֵרָה, וְזֵכֶר לַתַּעֲנִית שֶׁל יִשְׂרָאֵל בְּי״ג בַּאֲדָר בְּאוֹתָהּ שָׁנָה – הַיּוֹם שֶׁבּוֹ הַיְּהוּדִים נִלְחֲמוּ בְּשׂוֹנְאֵיהֶם.
🤔 לָמָּה? הַסַּכָּנָה הָיְתָה גְּדוֹלָה, וְהַיְּהוּדִים שֶׁלֹּא הָיוּ צְרִיכִים לְהִלָּחֵם – חָזְרוּ בִּתְשׁוּבָה וְצָמוּ, כִּי בְּעֵת צָרָה מִתְעוֹרְרִים לִתְשׁוּבָה. וְגַם הַיּוֹם יֵשׁ שׂוֹנְאִים שֶׁרוֹצִים לְהַשְׁמִיד אוֹתָנוּ, וַאֲנַחְנוּ צְרִיכִים תְּשׁוּבָה וּתְפִלָּה בְּכָל שָׁנָה מֵחָדָשׁ.
💛 הַצּוֹם עוֹזֵר לָנוּ לְהַרְגִּישׁ פָּחוֹת אֶת הַגּוּף וְיוֹתֵר אֶת הַנְּשָׁמָה – וּלְהִתְקָרֵב לַה׳.
(פֶּרֶק י״ד, הֲלָכָה ט׳)

13דִּינֵי תַּעֲנִית אֶסְתֵּר

תַּעֲנִית אֶסְתֵּר נִקְבְּעָה עַל פִּי מִנְהַג יִשְׂרָאֵל, וְלָכֵן דִּינָהּ קַל מִשְּׁאָר הַצּוֹמוֹת הַקַּלִּים – אֲבָל לְמַעֲשֶׂה כִּמְעַט אֵין הֶבְדֵּל בֵּינֵיהֶם. בַּתְּפִלּוֹת וּבִקְרִיאַת הַתּוֹרָה נוֹהֲגִים כְּמוֹ בִּשְׁאָר הַצּוֹמוֹת. בְּמִנְחָה לֹא אוֹמְרִים תַּחֲנוּן, כִּי זֶה עֶרֶב פּוּרִים, וְהָאַשְׁכְּנַזִּים לֹא אוֹמְרִים אָז ״אָבִינוּ מַלְכֵּנוּ״. כְּשֶׁי״ג בַּאֲדָר חָל בְּשַׁבָּת – מַקְדִּימִים אֶת הַתַּעֲנִית לְיוֹם חֲמִישִׁי, וְאָז מִתְפַּלְּלִים מִנְחָה כְּמוֹ בְּכָל הַצּוֹמוֹת.
(פֶּרֶק י״ד, הֲלָכָה ט׳)

14זֵכֶר לְמַחֲצִית הַשֶּׁקֶל

בִּזְמַן הַמִּקְדָּשׁ נָתַן כָּל אֶחָד בַּאֲדָר מַחֲצִית הַשֶּׁקֶל לְקָרְבְּנוֹת הַצִּבּוּר. הַיּוֹם נוֹהֲגִים לָתֵת צְדָקָה ״זֵכֶר לְמַחֲצִית הַשֶּׁקֶל״. הַזְּמַן הַטּוֹב – לִפְנֵי תְּפִלַּת מִנְחָה שֶׁל תַּעֲנִית אֶסְתֵּר. כַּמָּה נוֹתְנִים?
🤔 לָמָּה? נוֹתְנִים לִפְנֵי מִנְחָה שֶׁל הַתַּעֲנִית, כְּדֵי שֶׁהַצְּדָקָה תִּצְטָרֵף לַתַּעֲנִית. כָּל הַמִּנְהָגִים כְּשֵׁרִים, וּמִי שֶׁמּוֹסִיף בִּצְדָקָה – תָּבוֹא עָלָיו בְּרָכָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ נוֹהֲגִים לָתֵת חֲצִי מִן הַמַּטְבֵּעַ הַמְּקֻבָּל בַּמָּקוֹם – הַיּוֹם: חֲצִי שֶׁקֶל.
וְיֵשׁ נוֹהֲגִים (רַמָּ״א) לָתֵת שְׁלוֹשָׁה חֲצָאִים – כְּנֶגֶד שָׁלֹשׁ פְּעָמִים שֶׁכָּתוּב ״תְּרוּמָה״ בְּפָרָשַׁת שְׁקָלִים.
וְיֵשׁ נוֹהֲגִים לָתֵת אֶת הַשֹּׁוִי שֶׁל מַחֲצִית הַשֶּׁקֶל – כֶּסֶף טָהוֹר בְּמִשְׁקָל שֶׁל קָרוֹב לְעֶשֶׂר גְּרַם.
(פֶּרֶק י״ד, הֲלָכָה י׳)

15בִּשְׁבִיל מִי נוֹתְנִים?

יֵשׁ אוֹמְרִים שֶׁהַמִּנְהָג שַׁיָּךְ רַק לִגְבָרִים מִגִּיל עֶשְׂרִים, שֶׁהָיוּ חַיָּבִים בְּמַחֲצִית הַשֶּׁקֶל. וְיֵשׁ אוֹמְרִים שֶׁגַּם נְעָרִים מִגִּיל שְׁלוֹשׁ עֶשְׂרֵה, וְיֵשׁ אוֹמְרִים שֶׁגַּם בִּשְׁבִיל הַיְּלָדִים הַקְּטַנִּים, וְיֵשׁ אוֹמְרִים שֶׁגַּם נָשִׁים. הַיּוֹם נוֹהֲגִים לָתֵת לְפָחוֹת חֲצִי שֶׁקֶל עֲבוּר כָּל אֶחָד מִבְּנֵי הַבַּיִת. אֶת הַצְּדָקָה הַזֹּאת לֹא נוֹתְנִים מִכַּסְפֵּי מַעֲשֵׂר.
🤔 לָמָּה? אָסוּר לְקַיֵּם מִצְווֹת וּמִנְהָגִים שֶׁאָדָם חַיָּב בָּהֶם מִכֶּסֶף שֶׁכְּבָר שַׁיָּךְ לַצְּדָקָה.
💛 כְּשֶׁכָּל בְּנֵי הַבַּיִת נוֹתְנִים יַחַד – כֻּלָּנוּ שֻׁתָּפִים, כְּמוֹ שֶׁהָיִינוּ כֻּלָּנוּ שֻׁתָּפִים בְּקָרְבְּנוֹת הַמִּקְדָּשׁ.
(פֶּרֶק י״ד, הֲלָכָה י׳)

📝 מָה לָמַדְנוּ

  • ״מִשֶּׁנִּכְנַס אֲדָר מַרְבִּים בְּשִׂמְחָה״ – כִּי בַּחֹדֶשׁ הַזֶּה הָרַע הִתְהַפֵּךְ לְטוֹב.
  • בְּשַׁבְּתוֹת אֲדָר קוֹרְאִים אַרְבַּע פָּרָשִׁיּוֹת: שְׁקָלִים, זָכוֹר, פָּרָה וְהַחֹדֶשׁ; ״זָכוֹר״ – בַּשַּׁבָּת שֶׁלִּפְנֵי פּוּרִים.
  • מִצְוָה לִזְכֹּר אֶת מָה שֶׁעָשָׂה לָנוּ עֲמָלֵק – וּבְשָׁנָה מְעֻבֶּרֶת חוֹגְגִים פּוּרִים בַּאֲדָר הַשֵּׁנִי.
  • בְּי״ג בַּאֲדָר צָמִים תַּעֲנִית אֶסְתֵּר, זֵכֶר לַצּוֹם וְלַתְּשׁוּבָה בִּימֵי הַנֵּס.
  • לִפְנֵי מִנְחָה שֶׁל תַּעֲנִית אֶסְתֵּר נוֹתְנִים צְדָקָה ״זֵכֶר לְמַחֲצִית הַשֶּׁקֶל״ עֲבוּר כָּל אֶחָד מִבְּנֵי הַבַּיִת.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה דַּוְקָא בַּחֹדֶשׁ אֲדָר ״מַרְבִּים בְּשִׂמְחָה״?
  2. מָה הֵן אַרְבַּע הַפָּרָשִׁיּוֹת, וְלָמָּה קוֹרְאִים אֶת פָּרָשַׁת ״זָכוֹר״ דַּוְקָא בַּשַּׁבָּת שֶׁלִּפְנֵי פּוּרִים?
  3. זֵכֶר לְמָה צָמִים בְּתַעֲנִית אֶסְתֵּר, וּמָה נוֹתְנִים לִפְנֵי מִנְחָה שֶׁל הַתַּעֲנִית?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״מִשֶּׁנִּכְנַס אֲדָר מַרְבִּים בְּשִׂמְחָה״

סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת
מַהֲדוּרַת יְלָדִים

🔥 ל״ג בָּעֹמֶר

פֶּרֶק ה׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
עֵצִים, קַרְשִׁים, גַּפְרוּרִים, תַּפּוּחֵי אֲדָמָה בְּנְיָר כֶּסֶף – וּמְדוּרָה גְּדוֹלָה שֶׁמְּאִירָה אֶת הַלַּיְלָה! ל״ג בָּעֹמֶר הוּא אֶחָד הַיָּמִים הַשְּׂמֵחִים בַּשָּׁנָה. אֲבָל לָמָּה שְׂמֵחִים דַּוְקָא בּוֹ? מִי הָיָה רַבִּי שִׁמְעוֹן בַּר יוֹחָאי, וּמָה הַקֶּשֶׁר שֶׁלּוֹ לְרַבִּי עֲקִיבָא? בּוֹאוּ נִלְמַד.

1לָמָּה שְׂמֵחִים בְּל״ג בָּעֹמֶר?

בְּאֶמְצַע יְמֵי הָאֲבֵלוּת שֶׁל הַסְּפִירָה יֵשׁ יוֹם אֶחָד שֶׁל שִׂמְחָה – ל״ג בָּעֹמֶר. לְכָל הַמִּנְהָגִים מֻתָּר בּוֹ לָשִׁיר, לְנַגֵּן וְלִרְקֹד, מִתְּחִלָּתוֹ וְעַד סוֹפוֹ. לֹא אוֹמְרִים בּוֹ תַּחֲנוּן – וְגַם לֹא בְּמִנְחָה שֶׁלְּפָנָיו – וְלֹא מִתְעַנִּים בּוֹ.
🤔 לָמָּה? יֵשׁ כַּמָּה טְעָמִים: בְּל״ג בָּעֹמֶר הִפְסִיקוּ תַּלְמִידֵי רַבִּי עֲקִיבָא לָמוּת. וְיֵשׁ אוֹמְרִים שֶׁבּוֹ הִתְחִיל רַבִּי עֲקִיבָא לְלַמֵּד תַּלְמִידִים חֲדָשִׁים – וּבָהֶם רַבִּי שִׁמְעוֹן בַּר יוֹחָאי – וּמֵהֶם הִתְפַּשְּׁטָה הַתּוֹרָה בְּכָל יִשְׂרָאֵל. וְהוּא גַּם יוֹם הַהִלּוּלָא שֶׁל רַבִּי שִׁמְעוֹן בַּר יוֹחָאי.
💛 דַּוְקָא בְּתוֹךְ יְמֵי הַצַּעַר ה׳ נוֹתֵן לָנוּ יוֹם שֶׁל אוֹר – לְהַזְכִּיר שֶׁהַתּוֹרָה מַמְשִׁיכָה לְהָאִיר.
(פֶּרֶק ה׳, הֲלָכָה א׳)

2מִסְתַּפְּרִים בְּל״ג בָּעֹמֶר?

שִׁירָה וְרִקּוּדִים מֻתָּרִים בְּל״ג בָּעֹמֶר לְכָל הַמִּנְהָגִים. אֲבָל בְּתִסְפֹּרֶת יֵשׁ הֶבְדֵּל:
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג אַשְׁכְּנַז וְחֵלֶק מִקְּהִלּוֹת הַסְּפָרַדִּים מֻתָּר לְהִסְתַּפֵּר בְּל״ג בָּעֹמֶר.
לְמִנְהַג חֵלֶק מִקְּהִלּוֹת הַסְּפָרַדִּים אָסוּר לְהִסְתַּפֵּר בְּל״ג בָּעֹמֶר, וּמְחַכִּים לְבֹקֶר ל״ד. אֲבָל כְּשֶׁל״ג בָּעֹמֶר חָל בְּיוֹם שִׁשִּׁי – מִסְתַּפְּרִים בּוֹ לִכְבוֹד שַׁבָּת.
הַנּוֹהֲגִים לְפִי הָאֲרִ״י לֹא מִסְתַּפְּרִים גַּם בְּל״ג בָּעֹמֶר, עַד עֶרֶב חַג הַשָּׁבוּעוֹת.
(פֶּרֶק ה׳, הֲלָכָה א׳)

3הַהִלּוּלָא בְּמֵירוֹן

רַבִּים נוֹהֲגִים לַעֲלוֹת בְּל״ג בָּעֹמֶר לְמֵירוֹן, לְקִבְרוֹ שֶׁל רַבִּי שִׁמְעוֹן בַּר יוֹחָאי וְשֶׁל בְּנוֹ רַבִּי אֶלְעָזָר. שָׁם עוֹרְכִים שִׂמְחָה גְּדוֹלָה, מַדְלִיקִים מְדוּרָה, רוֹקְדִים וְשָׁרִים.
🤔 לָמָּה? בְּדֶרֶךְ כְּלָל יוֹם פְּטִירָה שֶׁל צַדִּיק הוּא יוֹם עָצוּב. אֲבָל רַבִּי שִׁמְעוֹן בִּקֵּשׁ שֶׁיִּשְׂמְחוּ בְּיוֹם פְּטִירָתוֹ. בְּיוֹם זֶה הוּא גִּלָּה לְתַלְמִידָיו סוֹדוֹת גְּדוֹלִים בַּתּוֹרָה, וְהַזֹּהַר קוֹרֵא לַיּוֹם הַזֶּה ״הִלּוּלָא״ – יוֹם שֶׁל שִׂמְחָה, כִּי בּוֹ נִשְׁמָתוֹ הִתְקָרְבָה לְגַמְרֵי אֶל ה׳.
💛 גַּם מִי שֶׁלֹּא מֵבִין אֶת סוֹדוֹת הַתּוֹרָה, שָׂמֵחַ שֶׁהַתּוֹרָה ״עֲמֻקָּה מִנִּי יָם״ – וְשֶׁיֵּשׁ צַדִּיקִים שֶׁזּוֹכִים לְהִתְקָרֵב לְסוֹדוֹתֶיהָ.
(פֶּרֶק ה׳, הֲלָכָה ב׳)

4רַבִּי שִׁמְעוֹן בַּמְּעָרָה

רַבִּי שִׁמְעוֹן בַּר יוֹחָאי דִּבֵּר נֶגֶד מַלְכוּת רוֹמָא הָרְשָׁעָה, וְהָרוֹמָאִים גָּזְרוּ עָלָיו מָוֶת. הוּא בָּרַח עִם בְּנוֹ, רַבִּי אֶלְעָזָר, וְהִתְחַבֵּא בִּמְעָרָה. נַעֲשָׂה לָהֶם נֵס: צָמַח שָׁם עֵץ חָרוּב וְנָבַע מַעְיָן, וּמֵהֶם הֵם חָיוּ. שְׁתֵּים עֶשְׂרֵה שָׁנָה לָמְדוּ שָׁם תּוֹרָה, עַד שֶׁמֵּת הַקֵּיסָר וְהַגְּזֵרָה בֻּטְּלָה. כְּשֶׁיָּצְאוּ, הֵם הָיוּ כָּל כָּךְ קְדוֹשִׁים שֶׁלֹּא יָכְלוּ לִסְבֹּל אֶת עִסְקֵי הָעוֹלָם הַזֶּה, וְחָזְרוּ לַמְּעָרָה לְעוֹד שָׁנָה – עַד שֶׁלָּמְדוּ לְהַעֲרִיךְ גַּם אֶת הָעוֹלָם הַזֶּה.
💛 רַבִּי שִׁמְעוֹן מָסַר אֶת נַפְשׁוֹ עַל כְּבוֹד יִשְׂרָאֵל וְעַל הַתּוֹרָה. מִמֶּנּוּ לוֹמְדִים מָה זֶה לִדְבֹּק בַּתּוֹרָה בְּכָל הַלֵּב.
(פֶּרֶק ה׳, הֲלָכָה ג׳)

5דַּרְכּוֹ הַמְּיֻחֶדֶת שֶׁל רַבִּי שִׁמְעוֹן

רֹב חַכְמֵי יִשְׂרָאֵל לִמְּדוּ שֶׁכָּל אָדָם צָרִיךְ גַּם לַעֲבֹד וּלְהִתְפַּרְנֵס, וְגַם לִלְמֹד תּוֹרָה. רַבִּי שִׁמְעוֹן בָּחַר בְּדֶרֶךְ אַחֶרֶת: לִלְמֹד תּוֹרָה כָּל הַזְּמַן, וְלִבְטֹחַ שֶׁה׳ יִדְאַג לְכָל הַשְּׁאָר. הַגְּמָרָא אוֹמֶרֶת: ״הַרְבֵּה עָשׂוּ כְּרַבִּי יִשְׁמָעֵאל (שֶׁעָבַד וְלָמַד) – וְעָלְתָה בְּיָדָם; כְּרַבִּי שִׁמְעוֹן – וְלֹא עָלְתָה בְּיָדָם״.
🤔 לָמָּה? הַדֶּרֶךְ שֶׁל רַבִּי שִׁמְעוֹן אֵינָהּ מַתְאִימָה לְכֻלָּם, וְהַהַדְרָכָה לַצִּבּוּר נִקְבַּעַת לְפִי רֹב הַחֲכָמִים. אֲבָל יֵשׁ עֵרֶךְ גָּדוֹל לְצַדִּיק אֶחָד שֶׁחַי כֻּלּוֹ לְמַעַן הַתּוֹרָה – כְּדֵי שֶׁכֻּלָּנוּ נִרְאֶה מָה זוֹ דְּבֵקוּת שְׁלֵמָה בַּתּוֹרָה.
(פֶּרֶק ה׳, הֲלָכָה ג׳)

6רַבִּי עֲקִיבָא – רַבּוֹ שֶׁל רַבִּי שִׁמְעוֹן

בְּל״ג בָּעֹמֶר זוֹכְרִים גַּם אֶת רַבִּי עֲקִיבָא, רַבּוֹ שֶׁל רַבִּי שִׁמְעוֹן. רַבִּי עֲקִיבָא הִתְחִיל לִלְמֹד תּוֹרָה בִּתְנָאִים קָשִׁים מְאוֹד, וּבִזְכוּת הַשְּׁקִידָה וְהָאֱמוּנָה שֶׁלּוֹ נַעֲשָׂה גְּדוֹל הַתּוֹרָה שֶׁבְּעַל פֶּה. אִשְׁתּוֹ רָחֵל וִתְּרָה עַל כָּל עָשְׁרָהּ וְעוֹדְדָה אוֹתוֹ לִלְמֹד, וּכְשֶׁנַּעֲשָׂה גָּדוֹל אָמַר לְתַלְמִידָיו: ״שֶׁלִּי וְשֶׁלָּכֶם – שֶׁלָּהּ הוּא״. גַּם כְּשֶׁרָאָה שׁוּעָל יוֹצֵא מִמְּקוֹם בֵּית הַמִּקְדָּשׁ הֶחָרֵב, הוּא שָׂמַח – כִּי הֶאֱמִין שֶׁכְּמוֹ שֶׁהִתְקַיְּמוּ דִּבְרֵי הַחֻרְבָּן, יִתְקַיְּמוּ גַּם דִּבְרֵי הַנֶּחָמָה.
🤔 לָמָּה? לֹא קָבְעוּ הִלּוּלָא בְּיוֹם פְּטִירָתוֹ שֶׁל רַבִּי עֲקִיבָא, כִּי הוּא נֶהֱרַג בִּידֵי הָרוֹמָאִים. לָכֵן בְּהִלּוּלַת רַבִּי שִׁמְעוֹן כְּלוּלָה גַּם הַהִלּוּלָא שֶׁל רַבּוֹ, וְרָאוּי לִלְמֹד בְּל״ג בָּעֹמֶר עַל תּוֹרָתוֹ וּגְדֻלָּתוֹ.
(פֶּרֶק ה׳, הֲלָכָה ד׳)

7״בְּכָל נַפְשְׁךָ״

כְּשֶׁהָרוֹמָאִים גָּזְרוּ שֶׁלֹּא לְלַמֵּד תּוֹרָה, רַבִּי עֲקִיבָא הִמְשִׁיךְ לְלַמֵּד בָּרַבִּים. הוּא נִתְפַּס וְנִגְזַר עָלָיו מָוֶת. כְּשֶׁהוֹצִיאוּ אוֹתוֹ לְהֵהָרֵג, הָיָה זְמַן קְרִיאַת שְׁמַע, וְהוּא קִבֵּל עָלָיו עֹל מַלְכוּת שָׁמַיִם. אָמַר לְתַלְמִידָיו: ״כָּל יָמַי הָיִיתִי מִצְטַעֵר עַל פָּסוּק זֶה: ׳בְּכָל נַפְשְׁךָ׳ – אֲפִלּוּ נוֹטֵל אֶת נִשְׁמָתְךָ. אָמַרְתִּי: מָתַי יָבוֹא לְיָדִי וַאֲקַיְּמֶנּוּ?״ וְהֶאֱרִיךְ בַּמִּלָּה ״אֶחָד״ עַד שֶׁיָּצְאָה נִשְׁמָתוֹ.
💛 רַבִּי עֲקִיבָא אָהַב אֶת ה׳ ״בְּכָל נַפְשְׁךָ״ – עַד הָרֶגַע הָאַחֲרוֹן. כְּשֶׁאֲנִי אוֹמֵר ״שְׁמַע יִשְׂרָאֵל... ה׳ אֶחָד״, אֲנִי מַמְשִׁיךְ אֶת הָאַהֲבָה שֶׁלּוֹ.
(פֶּרֶק ה׳, הֲלָכָה ד׳)

8הַמְּדוּרָה – אוֹר הַתּוֹרָה

כְּבָר מֵאוֹת שָׁנִים מַדְלִיקִים בְּמֵירוֹן מְדוּרָה לִכְבוֹד הַהִלּוּלָא שֶׁל רַבִּי שִׁמְעוֹן, וְגַם בִּמְקוֹמוֹת אֲחֵרִים נוֹהֲגִים לְהַדְלִיק מְדוּרוֹת. יֵשׁ שֶׁמַּדְלִיקִים נֵרוֹת בְּבֵית הַכְּנֶסֶת. אֲבָל חָשׁוּב לָדַעַת: הַמְּדוּרָה וְהָעֲלִיָּה לְמֵירוֹן הֵן מִנְהָג שֶׁל רְשׁוּת, וְלֹא חוֹבָה. הֵן לֹא כְּתוּבוֹת בָּרַמְבַּ״ם וּבַשֻּׁלְחָן עָרוּךְ, וְרַבִּים מִגְּדוֹלֵי יִשְׂרָאֵל לֹא נָהֲגוּ בָּהֶן.
🤔 לָמָּה? הָאֵשׁ רוֹמֶזֶת לָאוֹר הַגָּדוֹל שֶׁל סוֹדוֹת הַתּוֹרָה שֶׁגִּלָּה רַבִּי שִׁמְעוֹן. בַּזֹּהַר מְסֻפָּר שֶׁבְּיוֹם פְּטִירָתוֹ, כְּשֶׁגִּלָּה סוֹדוֹת, הָיְתָה סְבִיבוֹ אֵשׁ כָּל כָּךְ גְּדוֹלָה, שֶׁתַּלְמִידָיו לֹא יָכְלוּ לְהִתְקָרֵב אֵלָיו.
💛 ״כִּי נֵר מִצְוָה וְתוֹרָה אוֹר״. כְּמוֹ שֶׁהָאֵשׁ יוֹצֵאת מֵעֵצִים קָרִים וּמְאִירָה וּמְחַמֶּמֶת – כָּךְ הַתּוֹרָה וְהַמִּצְווֹת מְאִירוֹת אֶת הָעוֹלָם.
(פֶּרֶק ה׳, הֲלָכָה ה׳)

9תִּסְפֹּרֶת רִאשׁוֹנָה בְּגִיל שָׁלֹשׁ

בְּמִשְׁפָּחוֹת רַבּוֹת לֹא מְסַפְּרִים בֵּן קָטָן עַד גִּיל שָׁלֹשׁ. אָז מְסַפְּרִים אוֹתוֹ וּמַשְׁאִירִים לוֹ פֵּאוֹת – וְזוֹ אַחַת הַמִּצְווֹת הָרִאשׁוֹנוֹת שֶׁמְּחַנְּכִים אוֹתוֹ בָּהּ: ״לֹא תַקִּפוּ פְּאַת רֹאשְׁכֶם״. עוֹשִׂים שִׂמְחָה וּמַזְמִינִים קְרוֹבִים וַחֲבֵרִים. רַבִּים מְסַפְּרִים אֶת הַקְּטַנִּים בְּל״ג בָּעֹמֶר לְיַד קֶבֶר רַבִּי שִׁמְעוֹן בְּמֵירוֹן, וְיֵשׁ שֶׁמְּבַקְּשִׁים מִתַּלְמִיד חָכָם שֶׁיַּתְחִיל לְסַפֵּר.
🤔 לָמָּה? חֲזַ״ל דִּמּוּ אֶת שָׁלֹשׁ הַשָּׁנִים הָרִאשׁוֹנוֹת שֶׁל יֶלֶד לְעֵץ בִּשְׁנוֹת הָ״עָרְלָה״, שֶׁאֵין אוֹכְלִים מִפֵּרוֹתָיו. בַּשָּׁנָה הָרְבִיעִית הַפֵּרוֹת ״קֹדֶשׁ הִלּוּלִים לַה׳״ – וְכָךְ גַּם הַיֶּלֶד מַתְחִיל אָז לְהִתְחַנֵּךְ לְמִצְווֹת. הַפֵּאוֹת מְסַמְּנוֹת שֶׁהוּא יְהוּדִי.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ נוֹהֲגִים לֹא מְסַפְּרִים אֶת הַבֵּן עַד שֶׁמַּגִּיעַ לְגִיל שָׁלֹשׁ, וְאָז עוֹשִׂים ״חַלָּאקָה״ – תִּסְפֹּרֶת רִאשׁוֹנָה עִם פֵּאוֹת.
וְיֵשׁ נוֹהֲגִים לְסַפֵּר אֶת הַיֶּלֶד גַּם לִפְנֵי גִּיל שָׁלֹשׁ. הַמִּנְהָג אֵינוֹ מֻזְכָּר בָּרַמְבַּ״ם וּבַשֻּׁלְחָן עָרוּךְ, וְכָךְ נָהֲגוּ תַּלְמִידֵי חֲכָמִים רַבִּים.
💛 כְּבָר מִן הַתִּסְפֹּרֶת הָרִאשׁוֹנָה הַיֶּלֶד לוֹמֵד לִשְׂמֹחַ בַּמִּצְווֹת.
(פֶּרֶק ה׳, הֲלָכָה ו׳)

10בְּגָדִים לַמְּדוּרָה?

הָיוּ שֶׁנָּהֲגוּ לְהַשְׁלִיךְ לַמְּדוּרָה בְּמֵירוֹן בְּגָדִים יְקָרִים לִכְבוֹד רַבִּי שִׁמְעוֹן. גְּדוֹלִים אֲחֵרִים אָמְרוּ שֶׁאֵין בָּזֶה טַעַם, וְשֶׁזֶּה אָסוּר מִשּׁוּם ״בַּל תַּשְׁחִית״ – אִסּוּר לְקַלְקֵל דָּבָר לְחִנָּם. לְמַעֲשֶׂה: מִי שֶׁאֲבוֹתָיו לֹא נָהֲגוּ כָּךְ – לֹא יַתְחִיל. מִי שֶׁזֶּה מִנְהַג אֲבוֹתָיו – רַשַּׁאי, אֲבָל עָדִיף שֶׁיִּתְרֹם אֶת הַבְּגָדִים לִצְדָקָה.
🤔 לָמָּה? יֵשׁ שֶׁלִּמְּדוּ זְכוּת וְאָמְרוּ שֶׁכְּשֶׁעוֹשִׂים זֹאת לִכְבוֹד צַדִּיק, זֶה לֹא ״לְחִנָּם״. אֲבָל בֶּגֶד שֶׁנִּתָּן לְעָנִי – גַּם מְכַבֵּד אֶת הַצַּדִּיק וְגַם עוֹזֵר לְיָהוּדִי.
(פֶּרֶק ה׳, הֲלָכָה ז׳)

11מִתְפַּלְּלִים רַק אֶל ה׳

כְּשֶׁבָּאִים לְהִתְפַּלֵּל לְיַד קִבְרֵי צַדִּיקִים, נִזְהָרִים לֹא לְהִתְפַּלֵּל אֶל הַצַּדִּיק עַצְמוֹ – רַק אֶל ה׳. מִי שֶׁמִּתְפַּלֵּל אֶל הַצַּדִּיק עוֹבֵר עַל אִסּוּר מִן הַתּוֹרָה. וְהַאִם מֻתָּר לְבַקֵּשׁ מִן הַצַּדִּיק שֶׁיַּמְלִיץ עָלֵינוּ טוֹב לִפְנֵי ה׳?
🤔 לָמָּה? כְּשֶׁאָנוּ מִתְקַשְּׁרִים לַתּוֹרָה וְלַמַּעֲשִׂים הַטּוֹבִים שֶׁל הַצַּדִּיק, אָנוּ נַעֲשִׂים טוֹבִים יוֹתֵר – וּבִזְכוּת זֶה מְבַקְּשִׁים שֶׁה׳ יְקַבֵּל אֶת תְּפִלָּתֵנוּ.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ שֶׁהִתִּירוּ לְבַקֵּשׁ מִן הַצַּדִּיק שֶׁיַּמְלִיץ טוֹב לִפְנֵי ה׳ עַל הַמִּתְפַּלֵּל.
וְיֵשׁ שֶׁאָסְרוּ כָּל הַתְּפִלָּה מְכֻוֶּנֶת רַק אֶל ה׳, בְּלִי שׁוּם שָׁלִיחַ. בְּתוֹךְ הַתְּפִלָּה מֻתָּר לְבַקֵּשׁ מִן ה׳ שֶׁיְּקַבֵּל אוֹתָהּ בִּזְכוּת הַצַּדִּיק.
💛 אֵין שׁוּם מְחִצָּה בֵּינֵינוּ לְבֵין ה׳ – כָּל יְהוּדִי יָכוֹל לְדַבֵּר אִתּוֹ יָשָׁר, בְּכָל מָקוֹם.
(פֶּרֶק ה׳, הֲלָכָה ז׳)

📝 מָה לָמַדְנוּ

  • בְּל״ג בָּעֹמֶר שְׂמֵחִים: שָׁרִים וְרוֹקְדִים, לֹא אוֹמְרִים תַּחֲנוּן וְלֹא מִתְעַנִּים.
  • שְׂמֵחִים כִּי הַתַּלְמִידִים הִפְסִיקוּ לָמוּת, כִּי רַבִּי עֲקִיבָא הֵחֵל לְלַמֵּד תַּלְמִידִים חֲדָשִׁים, וּבְיוֹם הַהִלּוּלָא שֶׁל רַבִּי שִׁמְעוֹן בַּר יוֹחָאי.
  • בְּתִסְפֹּרֶת בְּל״ג בָּעֹמֶר יֵשׁ מִנְהָגִים שׁוֹנִים: אַשְׁכְּנַז וְחֵלֶק מֵהַסְּפָרַדִּים מִסְתַּפְּרִים, חֵלֶק מֵהַסְּפָרַדִּים לֹא.
  • הַמְּדוּרָה, הָעֲלִיָּה לְמֵירוֹן וְהַחַלָּאקָה הֵם מִנְהָגִים יָפִים שֶׁל רְשׁוּת – לֹא חוֹבָה.
  • לְיַד קִבְרֵי צַדִּיקִים מִתְפַּלְּלִים רַק אֶל ה׳.

❓ שְׁאֵלוֹת חֲזָרָה

  1. תֵּן שְׁנֵי טְעָמִים לַשִּׂמְחָה בְּל״ג בָּעֹמֶר.
  2. סַפֵּר עַל רַבִּי שִׁמְעוֹן בַּמְּעָרָה: מִמָּה חָיוּ, כַּמָּה זְמַן, וּמָה קָרָה כְּשֶׁיָּצְאוּ?
  3. אֵיךְ קִיֵּם רַבִּי עֲקִיבָא אֶת ״בְּכָל נַפְשְׁךָ״, וּמָה אֲנַחְנוּ יְכוֹלִים לִלְמֹד מִמֶּנּוּ?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״סְפִירַת הָעֹמֶר וְל״ג בָּעֹמֶר״

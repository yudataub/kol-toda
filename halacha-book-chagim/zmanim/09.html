סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת
מַהֲדוּרַת יְלָדִים

🥚 עֶרֶב תִּשְׁעָה בְּאָב

פֶּרֶק ט׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
אַחַר הַצָּהֳרַיִם שֶׁל עֶרֶב תִּשְׁעָה בְּאָב. אַבָּא מְקַלֵּף בֵּיצָה קָשָׁה, יוֹשֵׁב עַל הָרִצְפָּה בְּפִנַּת הַחֶדֶר, וְאוֹכֵל בְּשֶׁקֶט. אַף אֶחָד לֹא אוֹמֵר ״בְּתֵאָבוֹן״, וְאַף אֶחָד לֹא מְזַמֵּן. לָמָּה הַסְּעוּדָה הַזֹּאת כָּל כָּךְ שׁוֹנָה? וּמָה עוֹשִׂים כְּשֶׁתִּשְׁעָה בְּאָב מַתְחִיל בְּמוֹצָאֵי שַׁבָּת? בּוֹאוּ נִלְמַד.

1הַסְּעוּדָה הַמַּפְסֶקֶת

הַסְּעוּדָה הָאַחֲרוֹנָה לִפְנֵי הַצּוֹם, שֶׁאוֹכְלִים אוֹתָהּ אַחֲרֵי חֲצוֹת הַיּוֹם, נִקְרֵאת ״סְעוּדָה מַפְסֶקֶת״. אוֹכְלִים בָּהּ תַּבְשִׁיל אֶחָד בִּלְבַד, וְלֹא אוֹכְלִים בָּשָׂר וְלֹא שׁוֹתִים יַיִן. הַמִּנְהָג הַמְקֻבָּל: בַּצָּהֳרַיִם אוֹכְלִים אֲרוּחָה רְגִילָה עִם כַּמָּה תַּבְשִׁילִים, וְלִקְרַאת הַצּוֹם – אֶת הַסְּעוּדָה הַמַּפְסֶקֶת. אֲבָל לֹא מַעֲרִימִים: לֹא אוֹכְלִים אֲרוּחָה גְּדוֹלָה סָמוּךְ לַצּוֹם, וּמִיָּד אַחֲרֶיהָ ״סְעוּדָה מַפְסֶקֶת״.
🤔 לָמָּה? בַּשָּׁעָה הַזֹּאת הָאֵבֶל עַל הַחֻרְבָּן גּוֹבֵר, וְלֹא רָאוּי לְהִתְפַּנֵּק בִּשְׁנֵי תַּבְשִׁילִים. וְהָאֲרוּחָה שֶׁל הַצָּהֳרַיִם נוֹתֶנֶת כֹּחַ לַצּוֹם.
💛 תַּבְשִׁיל אֶחָד, בְּלִי חֲשִׁיבוּת וּבְלִי פִּנּוּק – כָּךְ נִכְנָסִים לַצּוֹם בַּעֲנָוָה, כְּמִי שֶׁמַּרְגִּישׁ שֶׁחָסֵר לוֹ דָּבָר יָקָר.
(פֶּרֶק ט׳, הֲלָכָה א׳)

2מָה נֶחְשָׁב תַּבְשִׁיל?

פֵּרוֹת וִירָקוֹת חַיִּים מֻתָּר לֶאֱכֹל בְּלִי הַגְבָּלָה, וְגַם סָלָט יְרָקוֹת. גַּם לֶחֶם, גְּבִינָה, לֶבֶּן וְחֶמְאָה אֵינָם נֶחְשָׁבִים תַּבְשִׁיל. אֲבָל פִּיצָה – כֵּן, בִּגְלַל הַגְּבִינָה שֶׁעָלֶיהָ. גֶּזֶר וְתַפּוּחֵי אֲדָמָה שֶׁבֻּשְּׁלוּ יַחַד – שְׁנֵי תַּבְשִׁילִים. גַּם בֵּיצָה קָשָׁה וּבֵיצָה רַכָּה – שְׁנֵי תַּבְשִׁילִים. אֲבָל אֹרֶז עִם מְעַט בָּצָל, שֶׁהַבָּצָל רַק מוֹסִיף לוֹ טַעַם – תַּבְשִׁיל אֶחָד. נָהֲגוּ שֶׁלֹּא לֶאֱכֹל דָּגִים, וְנָכוֹן שֶׁלֹּא לֶאֱכֹל שׁוֹקוֹלָד, חֲטִיפִים וּמַמְתַּקִּים.
🤔 לָמָּה? דָּגִים נֶחְשָׁבִים מַאֲכָל חָשׁוּב כְּמוֹ בָּשָׂר. מַמְתַּקִּים נוֹעֲדוּ לְפִנּוּק וְלֹא לִשְׂבִיעָה – וְאִם אֵין דָּבָר אַחֵר לֶאֱכֹל, מֻתָּר. וְגַם מִיץ מָתוֹק: לְצֹרֶךְ מֻתָּר, וּלְפִנּוּק – לֹא.
(פֶּרֶק ט׳, הֲלָכָה ב׳)

3בֵּיצָה קָשָׁה – מַאֲכַל אֲבֵלִים

הַתַּלְמוּד מְסַפֵּר שֶׁרַבִּי יְהוּדָה בְּרַבִּי אִלְעַאי אָכַל בַּסְּעוּדָה הַזֹּאת פַּת יְבֵשָׁה בְּמֶלַח, יָשַׁב בְּמָקוֹם בָּזוּי, וְשָׁתָה כַּד מַיִם – כְּמִי שֶׁמֵּתוֹ מֻטָּל לְפָנָיו. אֲבָל הַהַדְרָכָה לְכֻלָּנוּ הִיא לֶאֱכֹל גַּם פֵּרוֹת וִירָקוֹת, כְּדֵי לֶאֱגֹר כֹּחַ לַצּוֹם. רַבִּים אוֹכְלִים אֶת הַתַּבְשִׁיל הָאֶחָד – בֵּיצָה קָשָׁה, וְאֵין אִסּוּר לֶאֱכֹל שְׁתֵּי בֵּיצִים. וְיֵשׁ שֶׁאוֹכְלִים תַּבְשִׁיל עֲדָשִׁים.
🤔 לָמָּה? בֵּיצָה וַעֲדָשִׁים הֵם מַאֲכַל אֲבֵלִים. הַבֵּיצָה עֲגֻלָּה – רֶמֶז לַ״גַּלְגַּל הַחוֹזֵר בָּעוֹלָם״.
💛 כָּל כָּךְ הִרְגִּישׁוּ חֲכָמֵינוּ אֶת הַחֻרְבָּן. גַּם אֲנַחְנוּ לוֹמְדִים לְהַרְגִּישׁ אוֹתוֹ – כִּי מִי שֶׁאוֹהֵב, מִתְגַּעְגֵּעַ.
(פֶּרֶק ט׳, הֲלָכָה ג׳)

4יוֹשְׁבִים עַל הָאָרֶץ – כָּל אֶחָד לְבַד

בַּסְּעוּדָה הַמַּפְסֶקֶת יוֹשְׁבִים עַל הַקַּרְקַע, וְלֹא צָרִיךְ לְהָסִיר אֶת הַנַּעֲלַיִם. כָּל אֶחָד יוֹשֵׁב לְבַד בְּפִנָּתוֹ, כְּמוֹ שֶׁנֶּאֱמַר עַל הָאָבֵל: ״יֵשֵׁב בָּדָד וְיִדֹּם״. וְלָכֵן גַּם כְּשֶׁשְּׁלֹשָׁה אוֹכְלִים בְּאוֹתוֹ מָקוֹם – הֵם לֹא מְזַמְּנִים. מִי שֶׁקָּשֶׁה לוֹ לָשֶׁבֶת עַל הַקַּרְקַע, כְּמוֹ חוֹלֶה אוֹ זָקֵן – רַשַּׁאי לָשֶׁבֶת עַל כִּסֵּא, וְטוֹב שֶׁיְּשַׁנֶּה אֶת מְקוֹמוֹ.
🤔 לָמָּה? הַיְּשִׁיבָה הַנְּמוּכָה מְבַטֵּאת אֶת הַשִּׁפְלוּת שֶׁלָּנוּ מֵאָז הַחֻרְבָּן. יֵשׁ אוֹמְרִים שֶׁעַל פִּי הַקַּבָּלָה נָכוֹן לְהַנִּיחַ בַּד בֵּין הַיּוֹשֵׁב לַקַּרְקַע.
(פֶּרֶק ט׳, הֲלָכָה ג׳)

5אַחֲרֵי הַסְּעוּדָה

מִי שֶׁסִּיֵּם אֶת הַסְּעוּדָה הַמַּפְסֶקֶת וְרוֹצֶה לֶאֱכֹל אוֹ לִשְׁתּוֹת עוֹד – רַשַּׁאי, עַד הַשְּׁקִיעָה. רַק מִי שֶׁקִּבֵּל עַל עַצְמוֹ אֶת הַצּוֹם קֹדֶם – אָסוּר לוֹ. בְּעֶרֶב תִּשְׁעָה בְּאָב, מֵחֲצוֹת הַיּוֹם, לוֹמְדִים לְכַתְּחִלָּה דְּבָרִים עֲצוּבִים שֶׁקְּשׁוּרִים לְתִשְׁעָה בְּאָב אוֹ לְהִלְכוֹת אֲבֵלוּת. וּמִי שֶׁחוֹשֵׁשׁ שֶׁאִם יַגְבִּיל אֶת עַצְמוֹ לַנּוֹשְׂאִים הָאֵלֶּה יִלְמַד פָּחוֹת – מוּטָב שֶׁיִּלְמַד מָה שֶׁלִּבּוֹ חָפֵץ.
🤔 לָמָּה? הַצּוֹם מַתְחִיל בַּשְּׁקִיעָה, וְלֹא בְּסוֹף הַסְּעוּדָה.
(פֶּרֶק ט׳, הֲלָכָה ג׳)

6כְּשֶׁתִּשְׁעָה בְּאָב מַתְחִיל בְּמוֹצָאֵי שַׁבָּת

אֵין אֲבֵלוּת בְּשַׁבָּת. לָכֵן כְּשֶׁתִּשְׁעָה בְּאָב חָל בְּשַׁבָּת – הַצּוֹם נִדְחֶה לְיוֹם רִאשׁוֹן, וּבַשַּׁבָּת אוֹכְלִים בָּשָׂר וְשׁוֹתִים יַיִן ״כִּסְעוּדַת שְׁלֹמֹה בְּשַׁעְתּוֹ״. אוֹכְלִים סְעוּדָה שְׁלִישִׁית כְּבְכָל שַׁבָּת וְשָׁרִים שִׁירֵי שַׁבָּת, אֲבָל מַפְסִיקִים לֶאֱכֹל וְלִשְׁתּוֹת לִפְנֵי הַשְּׁקִיעָה. נִשְׁאָרִים בְּבִגְדֵי שַׁבָּת וּבְנַעֲלַיִם עַד צֵאת הַשַּׁבָּת. אָז אוֹמְרִים ״בָּרוּךְ הַמַּבְדִּיל בֵּין קֹדֶשׁ לְחֹל״, חוֹלְצִים אֶת הַנַּעֲלַיִם, וּמַחְלִיפִים לְבִגְדֵי חֹל שֶׁכְּבָר לָבְשׁוּ – לֹא מְכֻבָּסִים. עַרְבִית מַתְחִילָה כְּרֶבַע שָׁעָה אַחֲרֵי צֵאת הַשַּׁבָּת, כְּדֵי שֶׁכֻּלָּם יַסְפִּיקוּ לְהַחְלִיף.
🤔 לָמָּה? בֵּין הַשְּׁקִיעָה לְצֵאת הַשַּׁבָּת יֵשׁ זְמַן שֶׁהוּא גַּם שַׁבָּת וְגַם צוֹם. בּוֹ לֹא עוֹשִׂים שׁוּם מַעֲשֶׂה שֶׁל אֲבֵלוּת, כִּי זוֹ עֲדַיִן שַׁבָּת – אֲבָל נִמְנָעִים מִדְּבָרִים שֶׁאֵין בָּהֶם צֹרֶךְ לַשַּׁבָּת, כְּמוֹ אֲכִילָה וּשְׁתִיָּה.
💛 גַּם לִפְנֵי הַיּוֹם הֶעָצוּב בְּיוֹתֵר אָנוּ מְכַבְּדִים אֶת הַשַּׁבָּת עַד הָרֶגַע הָאַחֲרוֹן. הַשַּׁבָּת שֶׁל ה׳ שְׁלֵמָה, וְאֵין בָּהּ אֵבֶל.
(פֶּרֶק ט׳, הֲלָכָה ד׳)

7הַבְדָּלָה בְּתִשְׁעָה בְּאָב

כְּשֶׁתִּשְׁעָה בְּאָב מַתְחִיל בְּמוֹצָאֵי שַׁבָּת, אִי אֶפְשָׁר לְהַבְדִּיל עַל הַיַּיִן. לָכֵן: אוֹמְרִים ״בָּרוּךְ הַמַּבְדִּיל בֵּין קֹדֶשׁ לְחֹל״ (וּבִתְפִלַּת עַרְבִית – ״אַתָּה חוֹנַנְתָּנוּ״), וּמֵאָז מֻתָּר לַעֲשׂוֹת מְלָאכָה. עַל הַנֵּר מְבָרְכִים ״בּוֹרֵא מְאוֹרֵי הָאֵשׁ״ אַחֲרֵי עַרְבִית, לִפְנֵי קְרִיאַת אֵיכָה – וְגַם נָשִׁים וּבָנוֹת מְבָרְכוֹת עַל הַנֵּר. בְּסוֹף הַצּוֹם, לִפְנֵי שֶׁאוֹכְלִים, מַבְדִּילִים עַל הַכּוֹס – רַק ״הַגֶּפֶן״ וְ״הַמַּבְדִּיל״, בְּלִי בְּשָׂמִים וּבְלִי נֵר.
🤔 לָמָּה? בִּרְכַּת הַנֵּר הִיא הוֹדָאָה עַל הָאֵשׁ שֶׁה׳ גִּלָּה לָאָדָם בְּמוֹצָאֵי שַׁבָּת – הִיא לֹא תְּלוּיָה בַּכּוֹס. וְיֶלֶד קָטָן שֶׁאוֹכֵל בְּתִשְׁעָה בְּאָב אֵינוֹ צָרִיךְ לְהַבְדִּיל לִפְנֵי שֶׁהוּא אוֹכֵל.
(פֶּרֶק ט׳, הֲלָכָה ה׳)

📝 מָה לָמַדְנוּ

  • בַּסְּעוּדָה הַמַּפְסֶקֶת, אַחֲרֵי חֲצוֹת, אוֹכְלִים תַּבְשִׁיל אֶחָד – בְּלִי בָּשָׂר וּבְלִי יַיִן; רַבִּים אוֹכְלִים בֵּיצָה קָשָׁה.
  • פֵּרוֹת, יְרָקוֹת חַיִּים, לֶחֶם וּגְבִינָה מֻתָּרִים; פִּיצָה נֶחְשֶׁבֶת תַּבְשִׁיל, וְנוֹהֲגִים שֶׁלֹּא לֶאֱכֹל דָּגִים וּמַמְתַּקִּים.
  • יוֹשְׁבִים עַל הַקַּרְקַע, כָּל אֶחָד לְבַד, וְלֹא מְזַמְּנִים.
  • כְּשֶׁתִּשְׁעָה בְּאָב בְּמוֹצָאֵי שַׁבָּת – שַׁבָּת כָּרָגִיל עַד הַסּוֹף, ״בָּרוּךְ הַמַּבְדִּיל״, בְּרָכָה עַל הַנֵּר, וְהַבְדָּלָה עַל הַכּוֹס בְּסוֹף הַצּוֹם.

❓ שְׁאֵלוֹת חֲזָרָה

  1. מָה מֻתָּר וּמָה אָסוּר לֶאֱכֹל בַּסְּעוּדָה הַמַּפְסֶקֶת? תֵּן שְׁתֵּי דֻּגְמָאוֹת לְכָל אֶחָד.
  2. לָמָּה אוֹכְלִים בֵּיצָה קָשָׁה, וְלָמָּה כָּל אֶחָד יוֹשֵׁב לְבַד?
  3. תִּשְׁעָה בְּאָב מַתְחִיל בְּמוֹצָאֵי שַׁבָּת: מָה עוֹשִׂים בְּסוֹף הַשַּׁבָּת, וְאֵיךְ מַבְדִּילִים?

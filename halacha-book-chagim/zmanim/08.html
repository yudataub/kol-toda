סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת
מַהֲדוּרַת יְלָדִים

🏚️ שְׁלֹשֶׁת הַשָּׁבוּעוֹת וְתִשְׁעַת הַיָּמִים

פֶּרֶק ח׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
קַיִץ. חֻפְשָׁה גְּדוֹלָה, חַם בַּחוּץ, וְכֻלָּם רוֹצִים לָצֵאת לְטַיֵּל וְלִשְׂחוֹת. אֲבָל דַּוְקָא בַּשָּׁבוּעוֹת הָאֵלֶּה, מִי״ז בְּתַמּוּז וְעַד תִּשְׁעָה בְּאָב, עַם יִשְׂרָאֵל מִתְאַבֵּל עַל בֵּית הַמִּקְדָּשׁ. אֵיךְ מִתְאַבְּלִים בְּאֶמְצַע הַחֻפְשָׁה? מָה מֻתָּר, מָה אָסוּר, וּמָה מִשְׁתַּנֶּה כְּשֶׁמַּגִּיעַ חֹדֶשׁ אָב? בּוֹאוּ נִלְמַד.

1״בֵּין הַמְּצָרִים״

שְׁלֹשֶׁת הַשָּׁבוּעוֹת – מִלֵּיל י״ז בְּתַמּוּז וְעַד תִּשְׁעָה בְּאָב – הֵם יְמֵי צַעַר. עֲלֵיהֶם נֶאֱמַר בְּמְגִלַּת אֵיכָה: ״כָּל רֹדְפֶיהָ הִשִּׂיגוּהָ בֵּין הַמְּצָרִים״. בְּשָׁלֹשׁ הַשַּׁבָּתוֹת הָאֵלֶּה קוֹרְאִים הַפְטָרוֹת עַל הַפֻּרְעָנוּת, וּבְשֶׁבַע הַשַּׁבָּתוֹת שֶׁאַחֲרֵי תִּשְׁעָה בְּאָב – הַפְטָרוֹת שֶׁל נֶחָמָה. בַּיָּמִים הָאֵלֶּה לֹא רוֹקְדִים, לֹא מְנַגְּנִים, וְלֹא מְבָרְכִים ״שֶׁהֶחֱיָנוּ״.
🤔 לָמָּה? חֲכָמִים אָמְרוּ שֶׁהַיָּמִים הָאֵלֶּה מְסֻגָּלִים לְפֻרְעָנוּת. לָכֵן מִי שֶׁיּוֹצֵא לְטִיּוּל אוֹ לִרְחֹץ בַּיָּם – תָּמִיד צָרִיךְ לִשְׁמֹר עַל הַנֶּפֶשׁ, וּבַיָּמִים הָאֵלֶּה נִזְהָרִים בְּמִשְׁנֶה זְהִירוּת.
💛 אַחֲרֵי שָׁלֹשׁ הַפְטָרוֹת שֶׁל צַעַר בָּאוֹת שֶׁבַע שֶׁל נֶחָמָה – ה׳ לֹא מַשְׁאִיר אוֹתָנוּ בְּלִי תִּקְוָה.
(פֶּרֶק ח׳, הֲלָכָה א׳)

2בְּלִי רִקּוּדִים וּנְגִינָה

מִי״ז בְּתַמּוּז וְעַד תִּשְׁעָה בְּאָב לֹא עוֹרְכִים רִקּוּדִים וּמְחוֹלוֹת, לֹא מְנַגְּנִים וְלֹא שׁוֹמְעִים כְּלֵי זֶמֶר. לָכֵן אֵין מְקַיְּמִים חוּגֵי רִקּוּד, הוֹפָעוֹת וְעַרְבֵי שִׁירָה, וְלֹא מִשְׁתַּתְּפִים בָּהֶם. אֲבָל מֻתָּר לְהַמְשִׁיךְ בְּשִׁעוּרֵי נְגִינָה עַד הַשָּׁבוּעַ שֶׁחָל בּוֹ תִּשְׁעָה בְּאָב, וְטוֹב לְנַגֵּן בָּהֶם נִגּוּנִים עֲצוּבִים.
🤔 לָמָּה? הָאִסּוּר הוּא בִּגְלַל הַשִּׂמְחָה שֶׁבַּנְּגִינָה. בְּשִׁעוּר נְגִינָה – לוֹמְדִים, וְאֵין בּוֹ שִׂמְחָה.
(פֶּרֶק ח׳, הֲלָכָה ב׳)

3שִׂמְחָה שֶׁל מִצְוָה

בִּסְעוּדַת מִצְוָה – בְּרִית מִילָה, פִּדְיוֹן הַבֵּן, וּבַר מִצְוָה אוֹ בַּת מִצְוָה בַּיּוֹם שֶׁנִּכְנָסִים בּוֹ לְמִצְווֹת – מֻתָּר לָשִׁיר שִׁירֵי שִׂמְחָה. בְּמָקוֹם שֶׁתָּמִיד מַזְמִינִים נַגָּנִים לְשִׂמְחָה כָּזוֹ – הָרוֹצֶה לְהָקֵל רַשַּׁאי לְהַזְמִין גַּם עַכְשָׁו. אֲבָל מִשֶּׁנִּכְנַס חֹדֶשׁ אָב לֹא מַזְמִינִים תִּזְמֹרֶת לְשׁוּם שִׂמְחָה, וְלֹא מַשְׁמִיעִים שִׁירֵי שִׂמְחָה מִמַּכְשִׁיר. רַק שָׁרִים בַּפֶּה אֶת שִׁירֵי הַמִּצְוָה, וְאֶפְשָׁר לִרְקֹד מְעַט בְּמַעְגָּל.
🤔 לָמָּה? הָאֲבֵלוּת חָלָה עַל שִׂמְחָה שֶׁל רְשׁוּת. שִׂמְחָה שֶׁל מִצְוָה מַמְשִׁיכָה גַּם בִּימֵי הַצַּעַר.
(פֶּרֶק ח׳, הֲלָכָה ג׳)

4מוּזִיקָה מִמַּכְשִׁיר בַּבַּיִת

שִׁירִים בְּלִי כְּלֵי נְגִינָה – רַבִּים מַתִּירִים לִשְׁמֹעַ. וּמָה עִם מוּזִיקָה עִם כְּלֵי נְגִינָה מִמַּכְשִׁיר בַּבַּיִת?
🤔 לָמָּה? הַמַּחְמִירִים סוֹבְרִים שֶׁאֵין הֶבְדֵּל בֵּין נַגָּן חַי לְמַכְשִׁיר. הַמְּקִלִּים סוֹבְרִים שֶׁהַיּוֹם כֻּלָּם שׁוֹמְעִים מוּזִיקָה כָּל הַזְּמַן וְאֵין בָּזֶה חֲגִיגִיּוּת, וְשֶׁהָאִסּוּר הוּא עַל נִגּוּנֵי שִׂמְחָה. כָּל אֶחָד שׁוֹאֵל בַּבַּיִת מָה הַמִּנְהָג.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת כַּמָּה מִגְּדוֹלֵי הַפּוֹסְקִים (הָרַב מֹשֶׁה פַיינְשְׁטֵיין, הָרַב עוֹבַדְיָה יוֹסֵף) אָסוּר כָּל שְׁלֹשֶׁת הַשָּׁבוּעוֹת, כְּמוֹ נְגִינָה חַיָּה. וְיֵשׁ שֶׁאוֹסְרִים אֲפִלּוּ שִׁירִים בְּלִי כְּלֵי נְגִינָה.
לְפִי הָרַב מֶלָמֵד מְחַלְּקִים לְשָׁלֹשׁ: שִׁירֵי שִׂמְחָה (כְּמוֹ שִׁירֵי חֲתֻנָּה) – אָסוּר מִי״ז בְּתַמּוּז. שִׁירִים רְגִילִים – אָסוּר מֵרֹאשׁ חֹדֶשׁ אָב. שִׁירִים עֲצוּבִים, כְּמוֹ שִׁירִים עַל הַחֻרְבָּן – מֻתָּר. וְלֹא שׁוֹמְעִים בְּקוֹל רָם.
💛 שִׁיר עָצוּב עַל יְרוּשָׁלַיִם יָכוֹל לְעוֹרֵר אֶת הַלֵּב לְהִתְגַּעְגֵּעַ לְבֵית ה׳.
(פֶּרֶק ח׳, הֲלָכָה ד׳, הֲלָכָה ה׳)

5טִיּוּלִים וּבְרֵכָה

עַד סוֹף חֹדֶשׁ תַּמּוּז מֻתָּר לְטַיֵּל, לִשְׂחוֹת וְלָנוּחַ בְּבֵית מָלוֹן – וּמַקְפִּידִים בְּמִשְׁנֶה זְהִירוּת עַל כְּלָלֵי הַבְּטִיחוּת. אֲבָל מִשֶּׁנִּכְנַס חֹדֶשׁ אָב נִמְנָעִים מִטִּיּוּלִים וּמִבִּלּוּיִים שֶׁעִקָּרָם הֲנָאָה וְשִׂמְחָה, וּמִשְּׂחִיָּה בַּבְּרֵכָה אוֹ בַּיָּם לְשֵׁם בִּלּוּי. טִיּוּל לִמּוּדִי אוֹ טִיּוּל לְצֹרֶךְ בְּרִיאוּת – מֻתָּר.
🤔 לָמָּה? יֵשׁ אוֹמְרִים שֶׁצָּרִיךְ לְהִמָּנַע מִטִּיּוּלִים כָּל שְׁלֹשֶׁת הַשָּׁבוּעוֹת. אֲבָל לַהֲלָכָה רַק מֵרֹאשׁ חֹדֶשׁ אָב מְמַעֲטִים בְּשִׂמְחָה. לִפְנֵי כֵן נִמְנָעִים רַק מֵאֵרוּעֵי שִׂמְחָה מְיֻחָדִים, כְּמוֹ מְסִבּוֹת וְהוֹפָעוֹת.
(פֶּרֶק ח׳, הֲלָכָה ו׳)

6לֹא מְבָרְכִים ״שֶׁהֶחֱיָנוּ״

בִּשְׁלֹשֶׁת הַשָּׁבוּעוֹת נִזְהָרִים לֹא לֶאֱכֹל פְּרִי חָדָשׁ שֶׁצָּרִיךְ לְבָרֵךְ עָלָיו ״שֶׁהֶחֱיָנוּ״, וְלֹא קוֹנִים בֶּגֶד חָדָשׁ שֶׁמְּבָרְכִים עָלָיו ״שֶׁהֶחֱיָנוּ״. אֲבָל דְּבָרִים שֶׁאֵין מְבָרְכִים עֲלֵיהֶם ״שֶׁהֶחֱיָנוּ״ – כְּמוֹ גַּרְבַּיִם אוֹ גּוּפִיּוֹת – מֻתָּר לִקְנוֹת עַד סוֹף חֹדֶשׁ תַּמּוּז. וְאֶפְשָׁר גַּם לִקְנוֹת בֶּגֶד עַד סוֹף תַּמּוּז וְלִלְבֹּשׁ אוֹתוֹ לָרִאשׁוֹנָה (וּלְבָרֵךְ) אַחֲרֵי תִּשְׁעָה בְּאָב.
🤔 לָמָּה? אֵיךְ נֹאמַר ״שֶׁהֶחֱיָנוּ וְקִיְּמָנוּ וְהִגִּיעָנוּ לַזְּמַן הַזֶּה״ – בִּזְמַן שֶׁל פֻּרְעָנוּת? יֵשׁ גְּדוֹלִים שֶׁסָּבְרוּ שֶׁאֵין צֹרֶךְ לְהִזָּהֵר בָּזֶה, אֲבָל בְּמֶשֶׁךְ הַדּוֹרוֹת הִתְפַּשֵּׁט הַמִּנְהָג לְהַחְמִיר.
💛 אָנוּ מְחַכִּים לְבָרֵךְ ״שֶׁהֶחֱיָנוּ״ בְּשִׂמְחָה שְׁלֵמָה – בַּיּוֹם שֶׁבּוֹ יִבָּנֶה הַמִּקְדָּשׁ.
(פֶּרֶק ח׳, הֲלָכָה ז׳)

7מָתַי בְּכָל זֹאת מְבָרְכִים?

מִי שֶׁהִזְדַּמְּנָה לוֹ מִצְוָה שֶׁמְּבָרְכִים עָלֶיהָ ״שֶׁהֶחֱיָנוּ״, כְּמוֹ פִּדְיוֹן הַבֵּן – מְבָרֵךְ. מִי שֶׁפָּגַשׁ חָבֵר אָהוּב שֶׁלֹּא רָאָה שְׁלֹשִׁים יוֹם וְשָׂמַח לִקְרָאתוֹ – מְבָרֵךְ. וּבְשַׁבָּת לְדַעַת רֹב הַפּוֹסְקִים מֻתָּר לְבָרֵךְ ״שֶׁהֶחֱיָנוּ״ – אֲפִלּוּ בְּשַׁבַּת חֲזוֹן. לָכֵן מִי שֶׁיֵּשׁ לוֹ פְּרִי חָדָשׁ, יְחַכֶּה לְשַׁבָּת וִיבָרֵךְ עָלָיו. וְעַל בֶּגֶד חָדָשׁ – אֶפְשָׁר לְבָרֵךְ בְּשַׁבָּתוֹת שֶׁעַד סוֹף תַּמּוּז, אֲבָל לֹא בְּשַׁבָּת שֶׁאַחֲרֵי רֹאשׁ חֹדֶשׁ אָב.
🤔 לָמָּה? כְּשֶׁאֲנַחְנוּ לֹא קָבַעְנוּ אֶת זְמַן הַבְּרָכָה, אֶלָּא הִיא בָּאָה אֵלֵינוּ מִן הַשָּׁמַיִם, אוֹ שֶׁאִם לֹא נְבָרֵךְ עַכְשָׁו נַפְסִיד אוֹתָהּ – מְבָרְכִים. וּבְשַׁבָּת אֵין אֲבֵלוּת.
(פֶּרֶק ח׳, הֲלָכָה ח׳)

8תִּסְפֹּרֶת

חֲכָמִים תִּקְּנוּ שֶׁלֹּא לְהִסְתַּפֵּר בַּשָּׁבוּעַ שֶׁחָל בּוֹ תִּשְׁעָה בְּאָב. יֵשׁ שֶׁמַּחְמִירִים יוֹתֵר:
🤔 לָמָּה? וּמָה עִם יְלָדִים? בַּשָּׁבוּעַ שֶׁחָל בּוֹ תִּשְׁעָה בְּאָב לֹא מְסַפְּרִים גַּם קְטַנִּים, כְּדֵי לְחַנְּכָם לְהִתְאַבֵּל עַל הַחֻרְבָּן. לִפְנֵי כֵן, בִּשְׁעַת הַצֹּרֶךְ, אֶפְשָׁר לְהָקֵל לַקְּטַנִּים. לְהִסְתָּרֵק – מֻתָּר. וְיֶלֶד שֶׁתִּהְיֶה לוֹ בַּר מִצְוָה בַּיָּמִים הָאֵלֶּה – רָאוּי שֶׁיִּסְתַּפֵּר לִפְנֵי י״ז בְּתַמּוּז.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג רַבִּים מִיּוֹצְאֵי סְפָרַד לֹא מִסְתַּפְּרִים מִתְּחִלַּת הַשָּׁבוּעַ שֶׁחָל בּוֹ תִּשְׁעָה בְּאָב.
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז וְחֵלֶק מִיּוֹצְאֵי סְפָרַד (כְּמוֹ יוֹצְאֵי מָרוֹקוֹ וְגִ׳רְבָּה, וְהַנּוֹהֲגִים כָּאֲרִ״י) לֹא מִסְתַּפְּרִים בְּכָל שְׁלֹשֶׁת הַשָּׁבוּעוֹת.
לְמִנְהַג עוֹלֵי תּוּנִיס וְאַלְגִּ׳יר לֹא מִסְתַּפְּרִים מֵרֹאשׁ חֹדֶשׁ אָב.
💛 הַשֵּׂעָר שֶׁלֹּא סֻפַּר מְסַפֵּר לְכֻלָּם שֶׁאָנוּ מִתְאַבְּלִים עַל יְרוּשָׁלַיִם – כָּךְ גַּם הַגּוּף שֶׁלָּנוּ זוֹכֵר אֶת הַמִּקְדָּשׁ.
(פֶּרֶק ח׳, הֲלָכָה י׳)

9״מִשֶּׁנִּכְנַס אָב מְמַעֲטִים בְּשִׂמְחָה״

תִּשְׁעַת הַיָּמִים – מֵרֹאשׁ חֹדֶשׁ אָב וְעַד תִּשְׁעָה בְּאָב – הֵם יְמֵי אֵבֶל. בָּהֶם לֹא עוֹרְכִים אֵרוּעִים מְשַׂמְּחִים: לֹא טִיּוּלִים, לֹא נֹפֶשׁ, וַאֲפִלּוּ לֹא מְסִבּוֹת קְטַנּוֹת שֶׁל חֲבֵרִים. אֵרוּעַ שֶׁמַּטְּרָתוֹ חִנּוּכִית – מֻתָּר. אֵין תּוֹפְרִים בְּגָדִים חֲדָשִׁים, וְאֵין סוֹרְגִים בְּגָדִים אוֹ כִּפּוֹת. אֲבָל לְתַקֵּן בֶּגֶד יָשָׁן – מֻתָּר.
🤔 לָמָּה? נָהֲגוּ לֹא לַעֲרֹךְ אֶת ״חוּטֵי הַשְּׁתִי״ – הַחוּטִים שֶׁהֵם הַבָּסִיס שֶׁל הַבַּד – כִּי בַּחֻרְבָּן נִפְגְּמָה ״אֶבֶן הַשְּׁתִיָּה״, שֶׁהִיא הַבָּסִיס שֶׁל הָעוֹלָם.
(פֶּרֶק ח׳, הֲלָכָה י״ב)

10בָּשָׂר וְיַיִן

בְּתִשְׁעַת הַיָּמִים נָהֲגוּ לֹא לֶאֱכֹל בָּשָׂר וְלֹא לִשְׁתּוֹת יַיִן. מֵאֵימָתַי?
🤔 לָמָּה? בָּשָׂר וְיַיִן מְשַׂמְּחִים, וּ״מִשֶּׁנִּכְנַס אָב מְמַעֲטִים בְּשִׂמְחָה״. וְעוֹד: מֵאָז הַחֻרְבָּן לֹא מַעֲלִים בְּשַׂר קָרְבָּנוֹת עַל הַמִּזְבֵּחַ, וְלֹא מְנַסְּכִים עָלָיו יַיִן. גַּם בְּיוֹם י׳ בְּאָב, שֶׁבּוֹ נִשְׂרַף בֵּית הַמִּקְדָּשׁ, מַחְמִירִים – יֵשׁ עַד חֲצוֹת הַיּוֹם, וְיֵשׁ עַד סוֹפוֹ.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז כָּל תִּשְׁעַת הַיָּמִים, כּוֹלֵל רֹאשׁ חֹדֶשׁ אָב.
לְמִנְהַג רֹב יוֹצְאֵי סְפָרַד בְּרֹאשׁ חֹדֶשׁ אָב מֻתָּר, וְהָאִסּוּר מַתְחִיל אַחֲרֵי שֶׁנִּגְמָר יוֹם רֹאשׁ חֹדֶשׁ.
לְמִנְהַג עוֹלֵי תֵּימָן אוֹכְלִים בָּשָׂר וְשׁוֹתִים יַיִן, וְנִמְנָעִים רַק בַּסְּעוּדָה הַמַּפְסֶקֶת שֶׁלִּפְנֵי הַצּוֹם.
💛 כְּשֶׁאָנוּ מְוַתְּרִים עַל בָּשָׂר וְיַיִן, אָנוּ זוֹכְרִים אֶת הַמִּזְבֵּחַ שֶׁחָסֵר לָנוּ – וּמִתְגַּעְגְּעִים אֵלָיו.
(פֶּרֶק ח׳, הֲלָכָה י״ג)

11מָה נִכְלָל בָּאִסּוּר?

כָּל סוּגֵי הַבָּשָׂר אֲסוּרִים – גַּם בְּשַׂר עוֹף, כְּמוֹ שְׁנִיצֶל. דָּגִים – מֻתָּרִים. גַּם תַּבְשִׁיל שֶׁבֻּשַּׁל עִם בָּשָׂר, כְּמוֹ תַּפּוּחֵי אֲדָמָה מִסִּיר שֶׁל בָּשָׂר, לֹא אוֹכְלִים, כִּי מַרְגִּישִׁים בּוֹ טַעַם בָּשָׂר. מִיץ עֲנָבִים – כְּמוֹ יַיִן, אָסוּר. חוֹלֶה, אֲפִלּוּ חוֹלֶה קְצָת, שֶׁהַבָּשָׂר עוֹזֵר לוֹ – רַשַּׁאי לֶאֱכֹל. וּבְמוֹצָאֵי שַׁבַּת חֲזוֹן, מִי שֶׁמַּבְדִּיל עַל יַיִן – מָה עוֹשֶׂה עִם הַיַּיִן?
🤔 לָמָּה? יֶלֶד בַּגִּיל הַזֶּה כְּבָר מְבָרֵךְ עַל מָה שֶׁהוּא אוֹכֵל, אֲבָל עוֹד לֹא מֵבִין מַמָּשׁ אֶת הָאֵבֶל עַל יְרוּשָׁלַיִם.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי סְפָרַד וְחֵלֶק מִיּוֹצְאֵי אַשְׁכְּנַז הַמַּבְדִּיל שׁוֹתֶה בְּעַצְמוֹ. וְעָדִיף לְהַבְדִּיל עַל מִיץ עֲנָבִים, שֶׁאֵינוֹ מְשַׂמֵּחַ כָּל כָּךְ.
יֵשׁ מִיּוֹצְאֵי אַשְׁכְּנַז שֶׁנּוֹהֲגִים אִם יֵשׁ שָׁם יֶלֶד בֵּין שֵׁשׁ לְתֵשַׁע – הוּא שׁוֹתֶה אֶת הַיַּיִן. אִם אֵין יֶלֶד כָּזֶה – שׁוֹתֶה הַמַּבְדִּיל.
(פֶּרֶק ח׳, הֲלָכָה י״ד)

12שַׁבָּת וּסְעוּדוֹת מִצְוָה

בְּשַׁבָּת אוֹכְלִים בָּשָׂר וְשׁוֹתִים יַיִן כְּמוֹ בְּכָל שַׁבָּת – גַּם בְּשַׁבַּת חֲזוֹן, כִּי אֵין אֲבֵלוּת בְּשַׁבָּת. מֻתָּר אֲפִלּוּ לִטְעֹם בְּיוֹם שִׁשִּׁי אֶת תַּבְשִׁילֵי הַבָּשָׂר שֶׁמְּכִינִים לְשַׁבָּת, כְּדֵי לִבְדֹּק אִם הֵם טְעִימִים. גַּם בִּסְעוּדַת מִצְוָה – בְּרִית מִילָה, פִּדְיוֹן הַבֵּן, סִיּוּם מַסֶּכֶת, וּבַר מִצְוָה בְּיוֹם הַכְּנִיסָה לְמִצְווֹת – אוֹכְלִים בָּשָׂר וְשׁוֹתִים יַיִן. אֲבָל לֹא מְתַכְנְנִים סִיּוּם מַסֶּכֶת בְּמִיֻחָד כְּדֵי לְהַתִּיר בָּשָׂר – רַק מִי שֶׁהִגִּיעַ לַסִּיּוּם לְפִי סֵדֶר לִמּוּדוֹ.
🤔 לָמָּה? מִי שֶׁמְּתַכְנֵן סִיּוּם רַק כְּדֵי לֶאֱכֹל בָּשָׂר – מְבַטֵּל בְּכַוָּנָה אֶת הָאֲבֵלוּת עַל הַחֻרְבָּן.
💛 גַּם בְּאֶמְצַע יְמֵי הָאֵבֶל, הַשַּׁבָּת נִשְׁאֶרֶת שְׂמֵחָה וּשְׁלֵמָה – מַתָּנָה שֶׁל ה׳ שֶׁאִישׁ לֹא יָכוֹל לָקַחַת מִמֶּנּוּ.
(פֶּרֶק ח׳, הֲלָכָה ט״ו)

13בְּנִיָּה, צְבִיעָה וּנְטִיעָה

בְּתִשְׁעַת הַיָּמִים לֹא בּוֹנִים בִּנְיָן שֶׁל שִׂמְחָה – כְּמוֹ הַרְחָבַת דִּירָה בְּלִי צֹרֶךְ – וְלֹא צוֹבְעִים אֶת קִירוֹת הַבַּיִת, וְלֹא עוֹשִׂים שִׁפּוּצִים לְנוֹי. לֹא שׁוֹתְלִים עֲצֵי נוֹי וּפְרָחִים, אֲבָל מֻתָּר לְהַשְׁקוֹת אֶת הַגִּנָּה וּלְטַפֵּל בָּהּ כָּרָגִיל, וּמֻתָּר לִנְטֹעַ עֲצֵי פְּרִי. בְּנִיָּה לְצֹרֶךְ מִצְוָה – כְּמוֹ בֵּית כְּנֶסֶת אוֹ בֵּית סֵפֶר – מֻתֶּרֶת. וְגַם מִי שֶׁגָּר בְּדִירָה צְפוּפָה רַשַּׁאי לִבְנוֹת חֶדֶר נוֹסָף.
🤔 לָמָּה? צְבִיעָה וּפְרָחִים הֵם תּוֹסֶפֶת שֶׁל יֹפִי וְשִׂמְחָה – אֶפְשָׁר לִחְיוֹת בְּלִי זֶה. מָה שֶׁנָּחוּץ בֶּאֱמֶת, אוֹ שֶׁהוּא מִצְוָה – מֻתָּר.
(פֶּרֶק ח׳, הֲלָכָה ט״ז)

14קְנִיּוֹת בְּתִשְׁעַת הַיָּמִים

בְּתִשְׁעַת הַיָּמִים לֹא קוֹנִים דְּבָרִים מְשַׂמְּחִים: תַּכְשִׁיטִים, בְּגָדִים, כֵּלִים יָפִים, רָהִיטִים וּמְכוֹנִית – גַּם לֹא בָּאִינְטֶרְנֶט. וְטוֹב לַעֲשׂוֹת אֶת הַקְּנִיּוֹת הַגְּדוֹלוֹת שֶׁל הַבַּיִת לִפְנֵי תִּשְׁעַת הַיָּמִים אוֹ אַחֲרֵיהֶם. אֲבָל דְּבָרִים שֶׁל מִצְוָה מֻתָּר לִקְנוֹת – כְּמוֹ תְּפִלִּין וּסְפָרִים שֶׁל קֹדֶשׁ. וּמִי שֶׁאֵין לוֹ נַעֲלֵי בַּיִת פְּשׁוּטוֹת לְתִשְׁעָה בְּאָב – רַשַּׁאי לִקְנוֹת.
🤔 לָמָּה? חֲכָמִים אָמְרוּ שֶׁבַּיָּמִים הָאֵלֶּה מְמַעֲטִים בְּמַשָּׂא וּמַתָּן שֶׁל שִׂמְחָה. צָרְכֵי מִצְוָה אֵינָם שִׂמְחָה שֶׁל רְשׁוּת.
💛 בְּתִשְׁעַת הַיָּמִים אָנוּ מְמַעֲטִים בְּשִׂמְחָה שֶׁל דְּבָרִים – כְּדֵי לְהַשְׁאִיר מָקוֹם לַגַּעְגּוּעַ לַשִּׂמְחָה הָאֲמִתִּית שֶׁל בֵּית הַמִּקְדָּשׁ.
(פֶּרֶק ח׳, הֲלָכָה י״ח)

15כְּבִיסָה וּבְגָדִים מְכֻבָּסִים

חֲכָמִים אָסְרוּ לְכַבֵּס בַּשָּׁבוּעַ שֶׁחָל בּוֹ תִּשְׁעָה בְּאָב – וְגַם לְגַהֵץ. וְכָמוֹ שֶׁאָסוּר לְכַבֵּס, כָּךְ אָסוּר לִלְבֹּשׁ בְּגָדִים מְכֻבָּסִים, לִפְרֹס סְדִינִים אוֹ מַפָּה מְכֻבָּסִים, וּלְהַתְחִיל מַגֶּבֶת מְכֻבֶּסֶת. מֵאֵימָתַי?
🤔 לָמָּה? אָבֵל מַפְסִיק לְטַפֵּחַ וּלְפַנֵּק אֶת עַצְמוֹ מִתּוֹךְ צַעַר. אֵיךְ מִתְכּוֹנְנִים? לִפְנֵי הַזְּמַן לוֹבְשִׁים כַּמָּה בְּגָדִים, כָּל אֶחָד בְּעֵרֶךְ שָׁעָה, וְאָז הֵם כְּבָר לֹא נֶחְשָׁבִים ״מְכֻבָּסִים״. מִי שֶׁלֹּא הֵכִין – מַנִּיחַ בֶּגֶד מְכֻבָּס עַל הָרִצְפָּה לְזְמַן מָה. וּלְבָנִים וְגַרְבַּיִם, שֶׁמַּחְלִיפִים לְעִתִּים קְרוֹבוֹת – מֻתָּר לְהַחְלִיף.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי סְפָרַד בַּשָּׁבוּעַ שֶׁחָל בּוֹ תִּשְׁעָה בְּאָב.
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז מֵרֹאשׁ חֹדֶשׁ אָב. רַק לִכְבוֹד שַׁבַּת חֲזוֹן לוֹבְשִׁים בִּגְדֵי שַׁבָּת מְכֻבָּסִים.
(פֶּרֶק ח׳, הֲלָכָה י״ט)

16בִּגְדֵי יְלָדִים

בְּגָדִים שֶׁל תִּינוֹקוֹת שֶׁמִּתְלַכְלְכִים כָּל הַזְּמַן – מֻתָּר לְכַבֵּס, וְכֵן סְדִינִים שֶׁל יְלָדִים שֶׁהִרְטִיבוּ בַּלַּיְלָה. רַבִּים מְקִלִּים בִּשְׁעַת הַצֹּרֶךְ לְכַבֵּס גַּם בִּגְדֵי יְלָדִים גְּדוֹלִים יוֹתֵר, שֶׁגַּם הֵם מִתְלַכְלְכִים. אֲבָל לֹא מוֹסִיפִים לַמְּכוֹנָה בְּגָדִים שֶׁל גְּדוֹלִים, וְטוֹב לְיַבֵּשׁ אֶת בִּגְדֵי הַיְּלָדִים בְּתוֹךְ הַבַּיִת וְלֹא בַּחוּץ.
🤔 לָמָּה? בְּכִבּוּס בִּגְדֵי יְלָדִים שֶׁהִתְלַכְלְכוּ אֵין שִׂמְחָה וְאֵין פִּנּוּק – רַק נִקָּיוֹן. וּמְיַבְּשִׁים בַּבַּיִת, כְּדֵי שֶׁלֹּא יֵרָאֶה כְּאִלּוּ לֹא מִתְאַבְּלִים.
(פֶּרֶק ח׳, הֲלָכָה כ׳)

17רְחִיצָה

מֵעִקַּר הַדִּין רְחִיצָה אֲסוּרָה רַק בְּתִשְׁעָה בְּאָב עַצְמוֹ, אֲבָל נָהֲגוּ לְהַחְמִיר גַּם לְפָנָיו:
🤔 לָמָּה? הָרַב מֶלָמֵד מַסְבִּיר שֶׁפַּעַם רְחִיצָה הָיְתָה פִּנּוּק מְיֻחָד, וְהַיּוֹם כֻּלָּם מִתְרַחֲצִים כָּל יוֹם, וּמִי שֶׁלֹּא מִתְרַחֵץ סוֹבֵל. לָכֵן לְפִי הָרַב מֶלָמֵד, מִי שֶׁמִּצְטַעֵר אִם לֹא יִתְרַחֵץ – רַשַּׁאי לְהִתְרַחֵץ בְּמַיִם פּוֹשְׁרִים, עִם סַבּוֹן, לְשֵׁם נִקָּיוֹן וְלֹא לְתַעֲנוּג. וּמִי שֶׁיֵּשׁ לוֹ רֵיחַ זֵעָה – מוּטָב שֶׁיִּתְרַחֵץ, כִּי גָּדוֹל כְּבוֹד הַבְּרִיּוֹת. שְׂחִיָּה בַּבְּרֵכָה לְשֵׁם בִּלּוּי – אֲסוּרָה מֵרֹאשׁ חֹדֶשׁ אָב.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג רַבִּים מִיּוֹצְאֵי סְפָרַד לֹא מִתְרַחֲצִים בְּמַיִם חַמִּים בַּשָּׁבוּעַ שֶׁחָל בּוֹ תִּשְׁעָה בְּאָב.
לְמִנְהַג אַשְׁכְּנַז לֹא מִתְרַחֲצִים כְּלָל בְּתִשְׁעַת הַיָּמִים, אֲפִלּוּ בְּמַיִם קָרִים, וְרַק לִקְרַאת שַׁבַּת חֲזוֹן מִתְרַחֲצִים מְעַט.
(פֶּרֶק ח׳, הֲלָכָה כ״א)

18שַׁבַּת חֲזוֹן

שַׁבַּת חֲזוֹן הִיא הַשַּׁבָּת שֶׁלִּפְנֵי תִּשְׁעָה בְּאָב. קוֹרְאִים בָּהּ אֶת הַהַפְטָרָה ״חֲזוֹן יְשַׁעְיָהוּ״, עִם הַתּוֹכָחוֹת שֶׁנֶּאֶמְרוּ לִפְנֵי הַחֻרְבָּן. לְמִנְהַג יוֹצְאֵי סְפָרַד אֵין בָּהּ שׁוּם סִימָן שֶׁל אֲבֵלוּת. לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז הִיא בְּתוֹךְ תִּשְׁעַת הַיָּמִים, וְהַמִּנְהָג הָרוֹוֵחַ הַיּוֹם – לְהִתְרַחֵץ לִקְרָאתָהּ בְּפוֹשְׁרִים וְלִלְבֹּשׁ בִּגְדֵי שַׁבָּת. יֵשׁ שֶׁמְּשַׁנִּים בֶּגֶד אֶחָד מִבִּגְדֵי הַשַּׁבָּת, כְּדֵי לְבַטֵּא אֶת הַצַּעַר. בַּר מִצְוָה שֶׁעוֹלֶה לַתּוֹרָה בְּשַׁבַּת חֲזוֹן – עוֹרְכִים לוֹ קִדּוּשׁ כָּרָגִיל, וְגַם ״שָׁלוֹם זָכָר״ עוֹרְכִים כָּרָגִיל.
🤔 לָמָּה? אֵין מַרְאִים סִימָנֵי אֲבֵלוּת בְּשַׁבָּת. הַשַּׁבָּת נִשְׁאֶרֶת שַׁבָּת – גַּם כְּשֶׁהִיא בְּתוֹךְ יְמֵי הַצַּעַר.
💛 ״חֲזוֹן יְשַׁעְיָהוּ״ מְסַיֵּם בְּנֶחָמָה: ״צִיּוֹן בְּמִשְׁפָּט תִּפָּדֶה״ – גַּם הַתּוֹכָחָה שֶׁל ה׳ בָּאָה מִתּוֹךְ אַהֲבָה.
(פֶּרֶק ח׳, הֲלָכָה כ״ב)

📝 מָה לָמַדְנוּ

  • בִּשְׁלֹשֶׁת הַשָּׁבוּעוֹת לֹא רוֹקְדִים, לֹא מְנַגְּנִים וְלֹא מְבָרְכִים ״שֶׁהֶחֱיָנוּ״ – אֲבָל שִׂמְחָה שֶׁל מִצְוָה מֻתֶּרֶת.
  • תִּסְפֹּרֶת: לְמִנְהַג אַשְׁכְּנַז כָּל שְׁלֹשֶׁת הַשָּׁבוּעוֹת, לְמִנְהַג רַבִּים מִיּוֹצְאֵי סְפָרַד – בַּשָּׁבוּעַ שֶׁחָל בּוֹ תִּשְׁעָה בְּאָב.
  • ״מִשֶּׁנִּכְנַס אָב מְמַעֲטִים בְּשִׂמְחָה״: לֹא מְטַיְּלִים לְשֵׁם בִּלּוּי, לֹא קוֹנִים דְּבָרִים מְשַׂמְּחִים וְלֹא צוֹבְעִים אֶת הַבַּיִת.
  • בָּשָׂר וְיַיִן: לְמִנְהַג אַשְׁכְּנַז מֵרֹאשׁ חֹדֶשׁ אָב, לְמִנְהַג רֹב יוֹצְאֵי סְפָרַד אַחֲרָיו. דָּגִים מֻתָּרִים, וּבְשַׁבָּת וּבִסְעוּדַת מִצְוָה אוֹכְלִים כָּרָגִיל.
  • כְּבִיסָה: לְמִנְהַג אַשְׁכְּנַז מֵרֹאשׁ חֹדֶשׁ, לְמִנְהַג יוֹצְאֵי סְפָרַד בַּשָּׁבוּעַ שֶׁחָל בּוֹ – וּמְכִינִים בְּגָדִים מֵרֹאשׁ.

❓ שְׁאֵלוֹת חֲזָרָה

  1. מָה הַהֶבְדֵּל בֵּין שְׁלֹשֶׁת הַשָּׁבוּעוֹת לְתִשְׁעַת הַיָּמִים? תֵּן שְׁתֵּי דֻּגְמָאוֹת לְמָה שֶׁמֻּתָּר בַּשְּׁלֹשָׁה שָׁבוּעוֹת וְאָסוּר בְּתִשְׁעַת הַיָּמִים.
  2. לָמָּה לֹא מְבָרְכִים ״שֶׁהֶחֱיָנוּ״ בְּבֵין הַמְּצָרִים, וּמָה עוֹשֶׂה מִי שֶׁקִּבֵּל פְּרִי חָדָשׁ?
  3. אֵיךְ מִתְכּוֹנְנִים מֵרֹאשׁ לַיָּמִים שֶׁבָּהֶם אָסוּר לִלְבֹּשׁ בְּגָדִים מְכֻבָּסִים?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״הַצּוֹמוֹת וּשְׁלֹשֶׁת הַשָּׁבוּעוֹת״

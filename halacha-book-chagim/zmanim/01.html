סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת
מַהֲדוּרַת יְלָדִים

🌙 רֹאשׁ חֹדֶשׁ וּבִרְכַּת הַלְּבָנָה

פֶּרֶק א׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
הִסְתַּכַּלְתֶּם פַּעַם בַּלְּבָנָה כַּמָּה לֵילוֹת בָּרֶצֶף? פַּעַם הִיא קַו דַּק, פַּעַם חֲצִי עִגּוּל, וּפַעַם עִגּוּל שָׁלֵם וּמֵאִיר. הַלְּבָנָה הִיא שָׁעוֹן גָּדוֹל שֶׁה׳ תָּלָה בַּשָּׁמַיִם. בַּפֶּרֶק הַזֶּה נִלְמַד עַל רֹאשׁ חֹדֶשׁ – הַיּוֹם הַקָּטָן וְהַשָּׂמֵחַ שֶׁפּוֹתֵחַ כָּל חֹדֶשׁ – וְעַל הַבְּרָכָה הַמְּיֻחֶדֶת שֶׁאוֹמְרִים עַל הַלְּבָנָה.

1הַלְּבָנָה – שָׁעוֹן בַּשָּׁמַיִם

הַשָּׁנָה נִקְבַּעַת לְפִי הַשֶּׁמֶשׁ, וְהַחֹדֶשׁ – לְפִי הַלְּבָנָה. בִּתְחִלַּת הַחֹדֶשׁ הַלְּבָנָה נִרְאֵית כְּקַו דַּק. הִיא הוֹלֶכֶת וְגָדֵלָה, וּבְאֶמְצַע הַחֹדֶשׁ הִיא עִגּוּל שָׁלֵם. אַחַר כָּךְ הִיא הוֹלֶכֶת וּמִתְמַעֶטֶת, וּבְסוֹף הַחֹדֶשׁ הִיא נֶעֱלֶמֶת לְיוֹם בְּעֵרֶךְ. כְּשֶׁהִיא חוֹזֶרֶת כְּקַו דַּק – מַתְחִיל חֹדֶשׁ חָדָשׁ. מַחֲזוֹר הַלְּבָנָה הוּא כְּעֶשְׂרִים וְתִשְׁעָה יָמִים וָחֵצִי, וְלָכֵן יֵשׁ חֹדֶשׁ שֶׁל 29 יָמִים, ״חֹדֶשׁ חָסֵר״, וְיֵשׁ חֹדֶשׁ שֶׁל 30 יוֹם, ״חֹדֶשׁ מָלֵא״.
🤔 לָמָּה? כָּל הַחַגִּים שֶׁלָּנוּ תְּלוּיִים בַּתַּאֲרִיךְ שֶׁל הַחֹדֶשׁ: פֶּסַח בְּט״ו בְּנִיסָן, יוֹם הַכִּפּוּרִים בְּי׳ בְּתִשְׁרֵי, סֻכּוֹת בְּט״ו בְּתִשְׁרֵי. מִי שֶׁלֹּא יוֹדֵעַ מָתַי מַתְחִיל הַחֹדֶשׁ – לֹא יֵדַע מָתַי חָל הֶחָג.
💛 ״וְהָיוּ לְאֹתֹת וּלְמוֹעֲדִים״ – בְּכָל פַּעַם שֶׁאֲנִי מַבִּיט בַּלְּבָנָה, אֲנִי רוֹאֶה אֵיךְ ה׳ מְסַדֵּר לָנוּ אֶת הַזְּמַן.
(פֶּרֶק א׳, הֲלָכָה א׳)

2יִשְׂרָאֵל מְקַדְּשִׁים אֶת הַחֹדֶשׁ

הַלְּבָנָה לְבַדָּהּ עוֹד לֹא מַתְחִילָה אֶת הַחֹדֶשׁ. בִּימֵי בֵּית הַדִּין הַגָּדוֹל בָּאוּ עֵדִים שֶׁרָאוּ אֶת הַלְּבָנָה הַחֲדָשָׁה, וְעַל פִּי עֵדוּתָם קִדֵּשׁ בֵּית הַדִּין אֶת הַחֹדֶשׁ. זוֹ הַמִּצְוָה הָרִאשׁוֹנָה שֶׁנִּצְטַוּוּ יִשְׂרָאֵל בַּתּוֹרָה: ״הַחֹדֶשׁ הַזֶּה לָכֶם״. הַדָּבָר הָיָה כָּל כָּךְ חָשׁוּב, שֶׁהַתּוֹרָה הִתִּירָה לָעֵדִים לְחַלֵּל שַׁבָּת כְּדֵי לָלֶכֶת לִירוּשָׁלַיִם וּלְהָעִיד.
🤔 לָמָּה? ה׳ רָצָה שֶׁיִּשְׂרָאֵל יִהְיוּ שֻׁתָּפִים בִּקְבִיעַת הַזְּמַן. לָכֵן חוֹתְמִים בִּתְפִלַּת מוּסָף שֶׁל רֹאשׁ חֹדֶשׁ: ״מְקַדֵּשׁ יִשְׂרָאֵל וְרָאשֵׁי חֳדָשִׁים״ – ה׳ מְקַדֵּשׁ אֶת יִשְׂרָאֵל, וְיִשְׂרָאֵל מְקַדְּשִׁים אֶת הַחֳדָשִׁים.
💛 ה׳ סוֹמֵךְ עָלֵינוּ וּמַפְקִיד בְּיָדֵינוּ אֶת הַזְּמַנִּים שֶׁלּוֹ – וְזֶה סִימָן לְאַהֲבָה גְּדוֹלָה.
(פֶּרֶק א׳, הֲלָכָה א׳, הֲלָכָה ב׳)

3הַלּוּחַ שֶׁל הִלֵּל הַנָּשִׂיא

בִּימֵי הָרוֹמָאִים גָּבְרוּ הַגְּזֵרוֹת עַל הַיְּהוּדִים בְּאֶרֶץ יִשְׂרָאֵל, וְהָיָה קָשֶׁה מְאוֹד לְקַדֵּשׁ חֳדָשִׁים בְּבֵית הַדִּין. לָכֵן הִלֵּל הַשֵּׁנִי, הַנָּשִׂיא, וּבֵית דִּינוֹ חִשְּׁבוּ אֶת כָּל הַחֳדָשִׁים וְהַשָּׁנִים, וְקִדְּשׁוּ אוֹתָם עַד סוֹף כָּל הַדּוֹרוֹת. מֵאָז אֲנַחְנוּ מוֹנִים לְפִי הַלּוּחַ הָעִבְרִי שֶׁהוּא הִתְקִין.
🤔 לָמָּה? חֲכָמִים רָאוּ שֶׁבִּגְלַל הַצָּרוֹת בֵּית הַדִּין עָלוּל לְהִתְבַּטֵּל, וְדָאֲגוּ שֶׁכָּל יִשְׂרָאֵל, בְּכָל מָקוֹם, יֵדְעוּ מָתַי חָל כָּל חַג.
💛 אֲנַחְנוּ מִתְפַּלְּלִים לַה׳ שֶׁנִּזְכֶּה בִּמְהֵרָה לַגְּאֻלָּה, וְנַחֲזֹר לְקַדֵּשׁ חֳדָשִׁים בְּבֵית הַדִּין שֶׁבִּירוּשָׁלַיִם.
(פֶּרֶק א׳, הֲלָכָה ג׳)

4יוֹם אֶחָד אוֹ יוֹמַיִם?

כְּשֶׁהַחֹדֶשׁ הַקּוֹדֵם חָסֵר (29 יָמִים), רֹאשׁ חֹדֶשׁ הוּא יוֹם אֶחָד. כְּשֶׁהוּא מָלֵא (30 יוֹם), רֹאשׁ חֹדֶשׁ נִמְשָׁךְ יוֹמַיִם: יוֹם ל׳ שֶׁל הַחֹדֶשׁ הַקּוֹדֵם, וְיוֹם א׳ שֶׁל הַחֹדֶשׁ הֶחָדָשׁ. עִקַּר רֹאשׁ חֹדֶשׁ הוּא הַיּוֹם הַשֵּׁנִי, אֲבָל גַּם בַּיּוֹם הָרִאשׁוֹן נוֹהֲגִים בְּכָל דִּינֵי רֹאשׁ חֹדֶשׁ: מוּסָף, הַלֵּל וְ״יַעֲלֶה וְיָבוֹא״.
(פֶּרֶק א׳, הֲלָכָה ד׳)

5לָמָּה הַלְּבָנָה קְטַנָּה?

רֹאשׁ חֹדֶשׁ נִקְרָא ״מוֹעֵד״. בְּבֵית הַמִּקְדָּשׁ הִקְרִיבוּ בּוֹ קָרְבַּן מוּסָף וְתָקְעוּ בַּחֲצוֹצְרוֹת, וְהָיוּ רְגִילִים לָבוֹא בּוֹ אֶל הָרַב לְהַקְבִּיל אֶת פָּנָיו. חֲזַ״ל מְסַפְּרִים: בַּתְּחִלָּה בָּרָא ה׳ שְׁנֵי מְאוֹרוֹת גְּדוֹלִים. הַלְּבָנָה טָעֲנָה: ״אֵיךְ יִשְׁתַּמְּשׁוּ שְׁנֵי מְלָכִים בְּכֶתֶר אֶחָד?״ אָמַר לָהּ ה׳: ״לְכִי וּמַעֲטִי אֶת עַצְמֵךְ״. לָכֵן בְּרֹאשׁ חֹדֶשׁ הִקְרִיבוּ שָׂעִיר ״לְחַטָּאת לַה׳״.
🤔 לָמָּה? הַלְּבָנָה הַקְּטַנָּה מַזְכִּירָה לָנוּ שֶׁיֵּשׁ בָּעוֹלָם חֶסְרוֹנוֹת וּנְפִילוֹת. אֲבָל כָּל יְרִידָה הִיא לְצֹרֶךְ עֲלִיָּה. לָכֵן רֹאשׁ חֹדֶשׁ הוּא זְמַן טוֹב לְהַתְחָלוֹת חֲדָשׁוֹת וְלִתְשׁוּבָה.
💛 כְּמוֹ הַלְּבָנָה שֶׁנֶּעֱלֶמֶת וְחוֹזֶרֶת – גַּם מִי שֶׁנָּפַל יָכוֹל לָשׁוּב אֶל ה׳ וּלְהַגִּיעַ גָּבוֹהַּ יוֹתֵר.
(פֶּרֶק א׳, הֲלָכָה ה׳)

6סְעוּדַת רֹאשׁ חֹדֶשׁ

רֹאשׁ חֹדֶשׁ הוּא יוֹם טוֹב קָטָן, וְרָאוּי לִשְׂמֹחַ בּוֹ. מִצְוָה לְהוֹסִיף לִכְבוֹדוֹ מַאֲכָל מְיֻחָד – גַּם כְּשֶׁהוּא חָל בְּשַׁבָּת – וּמִצְוָה לֶאֱכֹל בּוֹ פַּת. טוֹב לַעֲרֹךְ אֶת הַשֻּׁלְחָן בְּכָבוֹד, וְיֵשׁ שֶׁלּוֹבְשִׁים בּוֹ בְּגָדִים חֲגִיגִיִּים. כְּשֶׁרֹאשׁ חֹדֶשׁ יוֹמַיִם – מִצְוָה לְהַרְבּוֹת בִּסְעוּדָה בִּשְׁנֵי הַיָּמִים. זוֹ מִצְוָה, אֲבָל לֹא חוֹבָה.
🤔 לָמָּה? מַאֲכָל מְיֻחָד מְסַפֵּר לְכָל הַבַּיִת: הַיּוֹם הַזֶּה אַחֵר – קָדוֹשׁ וְשָׂמֵחַ.
💛 גַּם עוּגָה קְטַנָּה לִכְבוֹד רֹאשׁ חֹדֶשׁ הִיא דֶּרֶךְ לוֹמַר לַה׳: אֲנִי שָׂמֵחַ בַּזְּמַן שֶׁנָּתַתָּ לָנוּ.
(פֶּרֶק א׳, הֲלָכָה ו׳)

7לֹא מִצְטַעֲרִים בְּרֹאשׁ חֹדֶשׁ

בְּרֹאשׁ חֹדֶשׁ אָסוּר לְהִתְעַנּוֹת – אָסוּר לְהִמָּנַע מֵאֲכִילָה כְּדֵי לָצוּם, אֲפִלּוּ לְשָׁעָה אַחַת. וְלֹא עוֹשִׂים בּוֹ דְּבָרִים מְצַעֲרִים: לֹא מַסְפִּידִים, וְנוֹהֲגִים שֶׁלֹּא לָלֶכֶת לְבֵית הַקְּבָרוֹת. לְקִבְרֵי צַדִּיקִים מֻתָּר לָלֶכֶת, כִּי אֵין בָּזֶה צַעַר.
🤔 לָמָּה? יוֹם שֶׁל שִׂמְחָה צָרִיךְ לִשְׁמֹר עַל הַשִּׂמְחָה שֶׁלּוֹ. וְאִם בְּמִקְרֶה לֹא הִזְדַּמֵּן לְךָ לֶאֱכֹל כַּמָּה שָׁעוֹת – אֵין בָּזֶה אִסּוּר. הָאִסּוּר הוּא לִצּוֹם בְּכַוָּנָה.
(פֶּרֶק א׳, הֲלָכָה ו׳)

8הַנָּשִׁים וְרֹאשׁ חֹדֶשׁ

מֻתָּר לַעֲשׂוֹת מְלָאכָה בְּרֹאשׁ חֹדֶשׁ. אֲבָל הַנָּשִׁים נוֹהֲגוֹת לְכַבֵּד אוֹתוֹ בְּמִיּוּחָד: כָּל אִשָּׁה נִמְנַעַת בּוֹ מִקְּצָת מְלָאכוֹת, לְמָשָׁל אֵינָהּ סוֹרֶגֶת, כְּדֵי שֶׁיִּהְיֶה הֶבְדֵּל בֵּין יוֹם רָגִיל לְרֹאשׁ חֹדֶשׁ. הַמְּהַדְּרוֹת גַּם אֵינָן תּוֹפְרוֹת וְאֵינָן מְתַקְּנוֹת דְּבָרִים בַּבַּיִת. לְבַשֵּׁל, לֶאֱפוֹת, לְגַהֵץ וּלְכַבֵּס בִּמְכוֹנָה – מֻתָּר. וְגַם עֲבוֹדָה לְצֹרֶךְ פַּרְנָסָה מֻתֶּרֶת.
🤔 לָמָּה? בַּמִּדְבָּר הַנָּשִׁים לֹא הִסְכִּימוּ לָתֵת אֶת נִזְמֵיהֶן לְחֵטְא הָעֵגֶל. לָכֵן נָתַן לָהֶן ה׳ שָׂכָר: הֵן שׁוֹמְרוֹת רָאשֵׁי חֳדָשִׁים יוֹתֵר מִן הָאֲנָשִׁים, וְהֵן מַרְגִּישׁוֹת יוֹתֵר אֶת קְדֻשַּׁת הַיּוֹם.
💛 הַנֶּאֱמָנוּת שֶׁל הַנָּשִׁים בַּמִּדְבָּר מְאִירָה עַד הַיּוֹם – בְּכָל רֹאשׁ חֹדֶשׁ מֵחָדָשׁ.
(פֶּרֶק א׳, הֲלָכָה ז׳)

9שַׁבַּת מְבָרְכִים

בַּשַּׁבָּת שֶׁלִּפְנֵי רֹאשׁ חֹדֶשׁ מַכְרִיזִים בְּבֵית הַכְּנֶסֶת מָתַי יִהְיֶה רֹאשׁ חֹדֶשׁ, וּמְבָרְכִים שֶׁה׳ יְחַדְּשֵׁהוּ עָלֵינוּ לְטוֹבָה וְלִבְרָכָה. מַזְכִּירִים אֶת זְמַן הַמּוֹלָד, וְעוֹמְדִים – זֵכֶר לְקִדּוּשׁ הַחֹדֶשׁ בְּבֵית הַדִּין, שֶׁהַקָּהָל עָמַד בּוֹ. רַק לִפְנֵי חֹדֶשׁ תִּשְׁרֵי לֹא מְבָרְכִים, כִּי כֻּלָּם יוֹדְעִים שֶׁהוּא רֹאשׁ הַשָּׁנָה.
🤔 לָמָּה? מַכְרִיזִים בְּשַׁבָּת, כִּי אָז כָּל הַקָּהָל בְּבֵית הַכְּנֶסֶת וְכֻלָּם שׁוֹמְעִים. וְעוֹד: כָּל הַיָּמִים מִתְבָּרְכִים מִן הַשַּׁבָּת, וְגַם הַקְּדֻשָּׁה שֶׁל רֹאשׁ חֹדֶשׁ נוֹבַעַת מִן הַשַּׁבָּת שֶׁלְּפָנָיו.
💛 כְּבָר בְּשַׁבַּת מְבָרְכִים מַתְחִילִים לְהַרְגִּישׁ אֶת הַחֲגִיגִיּוּת שֶׁל הַחֹדֶשׁ הֶחָדָשׁ שֶׁה׳ נוֹתֵן לָנוּ.
(פֶּרֶק א׳, הֲלָכָה ח׳)

10״יוֹם כִּפּוּר קָטָן״

רֹאשׁ חֹדֶשׁ הוּא זְמַן כַּפָּרָה. כָּךְ אוֹמְרִים בִּתְפִלַּת מוּסָף: ״זְמַן כַּפָּרָה לְכָל תּוֹלְדוֹתָם״. לָכֵן יֵשׁ שֶׁחוֹזְרִים בִּתְשׁוּבָה לִקְרַאת רֹאשׁ חֹדֶשׁ, וְיֵשׁ שֶׁקּוֹרְאִים לְעֶרֶב רֹאשׁ חֹדֶשׁ ״יוֹם כִּפּוּר קָטָן״ וְאַף צָמִים בּוֹ. הַיּוֹם הַמִּנְהָג לָצוּם אֵינוֹ נָפוֹץ, וְטוֹב לְהַרְבּוֹת בִּמְקוֹמוֹ בְּלִמּוּד תּוֹרָה וּבִצְדָקָה.
🤔 לָמָּה? יוֹם הַכִּפּוּרִים הוּא זְמַן כַּפָּרָה עַל כָּל הַשָּׁנָה, וְעֶרֶב רֹאשׁ חֹדֶשׁ – עַל הַחֹדֶשׁ שֶׁעָבַר.
💛 בְּכָל חֹדֶשׁ ה׳ נוֹתֵן לָנוּ דַּף חָדָשׁ וְנָקִי לְהַתְחִיל בּוֹ.
(פֶּרֶק א׳, הֲלָכָה ט׳)

11״יַעֲלֶה וְיָבוֹא״ בַּתְּפִלָּה

בְּרֹאשׁ חֹדֶשׁ מוֹסִיפִים ״יַעֲלֶה וְיָבוֹא״ בִּתְפִלַּת הָעֲמִידָה, בְּבִרְכַּת ״רְצֵה״. בָּהּ אָנוּ מְבַקְּשִׁים שֶׁה׳ יִזְכְּרֵנוּ לְטוֹבָה בְּיוֹם רֹאשׁ הַחֹדֶשׁ. מָה עוֹשֶׂה מִי שֶׁשָּׁכַח? אִם נִזְכַּר מִיָּד אַחֲרֵי ״רְצֵה״, לִפְנֵי ״מוֹדִים״ – אוֹמֵר שָׁם וּמַמְשִׁיךְ. אִם כְּבָר הִתְחִיל ״מוֹדִים״ וְעוֹד לֹא סִיֵּם אֶת הַתְּפִלָּה – חוֹזֵר לִתְחִלַּת ״רְצֵה״. אִם סִיֵּם אֶת הַתְּפִלָּה – בְּשַׁחֲרִית וּבְמִנְחָה חוֹזֵר וּמִתְפַּלֵּל, וּבְעַרְבִית אֵינוֹ חוֹזֵר.
🤔 לָמָּה? הַתְּפִלּוֹת נִתְקְנוּ כְּנֶגֶד הַקָּרְבָּנוֹת, וּבְרֹאשׁ חֹדֶשׁ הִקְרִיבוּ קָרְבַּן מוּסָף. אוֹמְרִים זֹאת בְּ״רְצֵה״, כִּי שָׁם מְבַקְּשִׁים שֶׁתַּחֲזֹר הָעֲבוֹדָה לְבֵית הַמִּקְדָּשׁ. וּבְעַרְבִית לֹא חוֹזְרִים, כִּי בֵּית הַדִּין לֹא הָיָה מְקַדֵּשׁ אֶת הַחֹדֶשׁ בַּלַּיְלָה.
💛 כְּשֶׁאֵין לָנוּ מִקְדָּשׁ, הַמִּלִּים שֶׁלָּנוּ עוֹלוֹת לִפְנֵי ה׳ בִּמְקוֹם קָרְבַּן הַמּוּסָף.
(פֶּרֶק א׳, הֲלָכָה י׳)

12״יַעֲלֶה וְיָבוֹא״ בְּבִרְכַּת הַמָּזוֹן

גַּם בְּבִרְכַּת הַמָּזוֹן אוֹמְרִים ״יַעֲלֶה וְיָבוֹא״, בְּבִרְכַּת ״רַחֵם״. מִי שֶׁשָּׁכַח – אֵינוֹ חוֹזֵר. וּמִי שֶׁהִתְחִיל סְעוּדָה בְּרֹאשׁ חֹדֶשׁ, אָכַל כְּזַיִת פַּת לִפְנֵי הַשְּׁקִיעָה, וְהִמְשִׁיךְ לֶאֱכֹל עַד אַחֲרֵי צֵאת הַכּוֹכָבִים – אוֹמֵר ״יַעֲלֶה וְיָבוֹא״, כִּי הַסְּעוּדָה הִתְחִילָה בְּרֹאשׁ חֹדֶשׁ.
🤔 לָמָּה? בְּשַׁבָּת וּבְיוֹם טוֹב יֵשׁ חוֹבָה לֶאֱכֹל סְעוּדָה עִם לֶחֶם, וְלָכֵן מִי שֶׁשָּׁכַח לְהַזְכִּיר אֶת הַיּוֹם – חוֹזֵר. בְּרֹאשׁ חֹדֶשׁ אֵין חוֹבָה כָּזוֹ, וְלָכֵן לֹא חוֹזְרִים.
(פֶּרֶק א׳, הֲלָכָה י״א)

13הַבְּרָכָה עַל הַהַלֵּל

מִנְהַג יִשְׂרָאֵל לוֹמַר הַלֵּל בְּרֹאשׁ חֹדֶשׁ. מִצַּד הַדִּין אֵין חוֹבָה, כִּי מֻתָּר לַעֲשׂוֹת בּוֹ מְלָאכָה – אֲבָל יִשְׂרָאֵל נָהֲגוּ לְאָמְרוֹ מִתּוֹךְ קְדֻשַּׁת הַיּוֹם. כְּדֵי שֶׁיִּהְיֶה בָּרוּר שֶׁזֶּה מִנְהָג, מְדַלְּגִים עַל שְׁנֵי קְטָעִים מִן הַהַלֵּל הַשָּׁלֵם. נֶחְלְקוּ הָרִאשׁוֹנִים אִם מְבָרְכִים עַל מִנְהָג, וְכָל אֶחָד מַמְשִׁיךְ בְּמִנְהַג אֲבוֹתָיו:
🤔 לָמָּה? מִי שֶׁמְּבָרֵךְ – כִּי זֶה מִנְהָג חָשׁוּב מְאוֹד. מִי שֶׁאֵינוֹ מְבָרֵךְ – כִּי לֹא מְבָרְכִים עַל מִנְהָג.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז מְבָרְכִים עַל הַהַלֵּל, אֲפִלּוּ מִי שֶׁמִּתְפַּלֵּל לְבַד.
לְמִנְהַג יוֹצְאֵי סְפָרַד שֶׁחָיוּ בְּאֶרֶץ יִשְׂרָאֵל וּסְבִיבוֹתֶיהָ לֹא מְבָרְכִים כְּלָל.
לְמִנְהַג רֹב קְהִלּוֹת צְפוֹן אַפְרִיקָה הַחַזָּן מְבָרֵךְ בְּקוֹל רָם בַּהַתְחָלָה וּבַסּוֹף, וּמוֹצִיא אֶת כֻּלָּם. מִי שֶׁמִּתְפַּלֵּל לְבַד – אֵינוֹ מְבָרֵךְ.
💛 שְׁנֵי הַמִּנְהָגִים יוֹצְאִים מֵאוֹתָהּ אַהֲבָה לַהַלֵּל: זֶה שָׂמֵחַ בַּמִּנְהָג, וְזֶה נִזְהָר בִּכְבוֹד שֵׁם שָׁמַיִם.
(פֶּרֶק א׳, הֲלָכָה י״ב)

14אֵיךְ אוֹמְרִים הַלֵּל?

אוֹמְרִים הַלֵּל מִיָּד אַחֲרֵי תְּפִלַּת שַׁחֲרִית, וּמִשְׁתַּדְּלִים לְאָמְרוֹ עִם הַצִּבּוּר. אוֹמְרִים אוֹתוֹ בַּעֲמִידָה, כַּסֵּדֶר מִתְּחִלָּתוֹ וְעַד סוֹפוֹ, וְלֹא מַפְסִיקִים בְּאֶמְצָעוֹ – אֲפִלּוּ לֹא בִּשְׁתִיקָה. מֻתָּר לְהַפְסִיק רַק כְּדֵי לַעֲנוֹת דְּבָרִים שֶׁבִּקְדֻשָּׁה, אוֹ לְצֹרֶךְ חָשׁוּב, כְּמוֹ כְּדֵי שֶׁלֹּא לְהַעֲלִיב מִישֶׁהוּ. נָכוֹן לוֹמַר אוֹתוֹ בְּנַחַת וּבִנְעִימָה, וְרַבִּים שָׁרִים חֲלָקִים מִמֶּנּוּ.
🤔 לָמָּה? הַהַלֵּל הוּא עֵדוּת עַל שִׁבְחוֹ שֶׁל ה׳, וְעֵדוּת אוֹמְרִים בַּעֲמִידָה.
💛 כְּשֶׁשָּׁרִים אֶת הַהַלֵּל יַחַד, הַתְּפִלָּה נַעֲשֵׂית שִׁיר שֶׁל אַהֲבָה לַה׳.
(פֶּרֶק א׳, הֲלָכָה י״ב, הֲלָכָה י״ג)

15״הוֹדוּ לַה׳ כִּי טוֹב״

הַחַזָּן אוֹמֵר בְּקוֹל רָם אַרְבָּעָה פְּסוּקִים: ״הוֹדוּ לַה׳ כִּי טוֹב״, ״יֹאמַר נָא יִשְׂרָאֵל״, ״יֹאמְרוּ נָא בֵית אַהֲרֹן״ וְ״יֹאמְרוּ נָא יִרְאֵי ה׳״. מִן הַפָּסוּק ״אוֹדְךָ״ וְעַד סוֹף הַהַלֵּל – אוֹמְרִים כָּל פָּסוּק פַּעֲמַיִם. וְאֶת ״אָנָּא ה׳ הוֹשִׁיעָה נָּא, אָנָּא ה׳ הַצְלִיחָה נָּא״ כּוֹפְלִים כָּךְ: קֹדֶם ״אָנָּא ה׳ הוֹשִׁיעָה נָּא״ פַּעֲמַיִם, וְאַחַר כָּךְ ״אָנָּא ה׳ הַצְלִיחָה נָּא״ פַּעֲמַיִם. וּמָה עוֹנֶה הַקָּהָל אַחֲרֵי הַחַזָּן?
🤔 לָמָּה? חֲזַ״ל סִפְּרוּ שֶׁאֶת הַפְּסוּקִים מִ״אוֹדְךָ״ וָהָלְאָה אָמְרוּ דָּוִד, יִשַׁי אָבִיו וְאֶחָיו – וּמִפְּנֵי שֶׁהֵם חֲשׁוּבִים כָּל כָּךְ, אוֹמְרִים אוֹתָם פַּעֲמַיִם.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז אַחֲרֵי כָּל אֶחָד מֵאַרְבַּעַת הַפְּסוּקִים עוֹנִים: ״הוֹדוּ לַה׳ כִּי טוֹב כִּי לְעוֹלָם חַסְדּוֹ״.
לְמִנְהַג יוֹצְאֵי סְפָרַד חוֹזְרִים עַל הַפָּסוּק שֶׁאָמַר הַחַזָּן.
(פֶּרֶק א׳, הֲלָכָה י״ג)

16קְרִיאַת הַתּוֹרָה וּמוּסָף

בְּרֹאשׁ חֹדֶשׁ עוֹלִים לַתּוֹרָה אַרְבָּעָה עוֹלִים, וְקוֹרְאִים עַל קָרְבַּן הַתָּמִיד וְעַל קָרְבְּנוֹת רֹאשׁ חֹדֶשׁ. אַחַר כָּךְ מִתְפַּלְּלִים מוּסָף, וְחוֹתְמִים בּוֹ: ״מְקַדֵּשׁ יִשְׂרָאֵל וְרָאשֵׁי חֳדָשִׁים״. לִפְנֵי מוּסָף, אַחֲרֵי הַקַּדִּישׁ, חוֹלְצִים אֶת הַתְּפִלִּין, וְנָכוֹן לְחַכּוֹת עַד שֶׁכּוֹרְכִים אוֹתָן וּמַכְנִיסִים לַנַּרְתִּיק. בְּשַׁחֲרִית אוֹמְרִים גַּם ״בָּרְכִי נַפְשִׁי״, שֶׁכָּתוּב בּוֹ: ״עָשָׂה יָרֵחַ לְמוֹעֲדִים״.
🤔 לָמָּה? תְּפִלַּת מוּסָף נֶחְשֶׁבֶת ״אוֹת״, כְּמוֹ יוֹם טוֹב – וּכְשֶׁיֵּשׁ אוֹת אֶחָד, לֹא צָרִיךְ עוֹד אוֹת שֶׁל תְּפִלִּין. וּמַתְחִילִים לִקְרֹא בְּקָרְבַּן הַתָּמִיד, כִּי מִתּוֹךְ הַקְּדֻשָּׁה הַקְּבוּעָה שֶׁל כָּל יוֹם צוֹמַחַת הַקְּדֻשָּׁה הַחֲדָשָׁה שֶׁל רֹאשׁ חֹדֶשׁ.
💛 מִי שֶׁמִּתְפַּלֵּל בְּכָל יוֹם, יָכוֹל לְהַרְגִּישׁ בְּרֹאשׁ חֹדֶשׁ אֵיךְ ה׳ מְחַדֵּשׁ אֶת הַקֶּשֶׁר אִתּוֹ.
(פֶּרֶק א׳, הֲלָכָה י״ד)

17בִּרְכַּת הַלְּבָנָה – ״דָּוִד מֶלֶךְ יִשְׂרָאֵל חַי וְקַיָּם״

פַּעַם בְּחֹדֶשׁ מְבָרְכִים עַל הַלְּבָנָה הַמִּתְחַדֶּשֶׁת, וּמוֹדִים לַה׳ שֶׁבָּרָא אוֹתָהּ, וְאָנוּ נֶהֱנִים מֵאוֹרָהּ בַּלַּיְלָה. הַלְּבָנָה דּוֹמָה לְעַם יִשְׂרָאֵל: יֵשׁ לָהּ יְרִידוֹת וַעֲלִיּוֹת, אֲבָל הִיא תָּמִיד חוֹזֶרֶת וְגָדֵלָה. לָכֵן אוֹמְרִים בְּקִדּוּשׁ לְבָנָה: ״דָּוִד מֶלֶךְ יִשְׂרָאֵל חַי וְקַיָּם״.
🤔 לָמָּה? דָּוִד הַמֶּלֶךְ לִמֵּד אוֹתָנוּ לַהֲפֹךְ כָּל נְפִילָה לַעֲלִיָּה, וְלָכֵן מַלְכוּתוֹ נִמְשְׁלָה לַלְּבָנָה. וּבַגְּאֻלָּה הַשְּׁלֵמָה יִהְיֶה אוֹר הַלְּבָנָה כְּאוֹר הַחַמָּה.
💛 עַם יִשְׂרָאֵל חַי וְקַיָּם, כִּי הַקֶּשֶׁר שֶׁלּוֹ לַה׳ נִצְחִי – וְגַם אֲנִי תָּמִיד יָכוֹל לַחֲזֹר וְלִגְדֹּל.
(פֶּרֶק א׳, הֲלָכָה ט״ו)

18מְבָרְכִים בְּשִׂמְחָה

בִּרְכַּת הַלְּבָנָה הִיא כְּמוֹ קַבָּלַת פְּנֵי הַשְּׁכִינָה. לָכֵן אוֹמְרִים אוֹתָהּ בַּעֲמִידָה, וְנוֹהֲגִים לְבָרֵךְ בְּמִנְיָן (וּמִצַּד הַדִּין אֶפְשָׁר גַּם לְבַד). יוֹצְאִים מִן הַבַּיִת אֶל תַּחַת כִּפַּת הַשָּׁמַיִם, וְנוֹהֲגִים לְבָרֵךְ בְּמוֹצָאֵי שַׁבָּת, כְּשֶׁשְּׂמֵחִים וּלְבוּשִׁים בְּגָדִים נָאִים. אַחֲרֵי הַבְּרָכָה שָׁרִים וְרוֹקְדִים. מִי שֶׁחוֹלֶה אוֹ חוֹשֵׁשׁ שֶׁיִּצְטַנֵּן – יָכוֹל לִרְאוֹת אֶת הַלְּבָנָה דֶּרֶךְ הַחַלּוֹן וּלְבָרֵךְ.
🤔 לָמָּה? כְּמוֹ שֶׁיּוֹצְאִים לְקַבֵּל אֶת פְּנֵי הַמֶּלֶךְ – כָּךְ יוֹצְאִים לִקְרַאת הַבְּרָכָה הַזֹּאת. הַשִּׁירָה וְהָרִקּוּד מְבַטְּאִים אֶת הַשִּׂמְחָה עַל הַקֶּשֶׁר הַקָּרוֹב בֵּין יִשְׂרָאֵל לַה׳, שֶׁמִּתְחַדֵּשׁ בְּכָל חֹדֶשׁ.
💛 חֲכָמִים אָמְרוּ: אִלּוּ זָכוּ יִשְׂרָאֵל רַק לְהַקְבִּיל פְּנֵי אֲבִיהֶם שֶׁבַּשָּׁמַיִם פַּעַם אַחַת בְּחֹדֶשׁ – דַּיָּם.
(פֶּרֶק א׳, הֲלָכָה ט״ז)

19מָתַי לֹא מְבָרְכִים?

צָרִיךְ לְבָרֵךְ עַל הַלְּבָנָה בְּשִׂמְחָה. לָכֵן לֹא מְבָרְכִים לִפְנֵי תִּשְׁעָה בְּאָב, בִּגְלַל הָאֵבֶל עַל הַחֻרְבָּן, וְלֹא לִפְנֵי יוֹם הַכִּפּוּרִים, בִּגְלַל הַמֶּתַח שֶׁל יוֹם הַדִּין. בְּמוֹצָאֵי יוֹם הַכִּפּוּרִים מְבָרְכִים מִיָּד, עוֹד לִפְנֵי שֶׁאוֹכְלִים. בְּמוֹצָאֵי תִּשְׁעָה בְּאָב נָכוֹן לְחַכּוֹת עַד שֶׁיֹּאכְלוּ וְיִשְׁתּוּ. וְגַם בְּלֵיל שַׁבָּת נוֹהֲגִים שֶׁלֹּא לְבָרֵךְ.
🤔 לָמָּה? בְּמוֹצָאֵי יוֹם הַכִּפּוּרִים שְׂמֵחִים שֶׁזָּכִינוּ לַעֲמֹד לִפְנֵי ה׳ בִּתְשׁוּבָה. וּבְלֵיל שַׁבָּת לֹא מְבָרְכִים, כְּדֵי שֶׁלֹּא לְעָרֵב אֶת שִׂמְחַת הַשַּׁבָּת בְּשִׂמְחָה אַחֶרֶת.
(פֶּרֶק א׳, הֲלָכָה ט״ז)

20רוֹאִים אֶת הַלְּבָנָה

מְבָרְכִים בַּלַּיְלָה, כְּשֶׁרוֹאִים אֶת הַלְּבָנָה וְנֶהֱנִים מֵאוֹרָהּ – וְלֹא בֵּין הַשְּׁמָשׁוֹת, כְּשֶׁהַשֶּׁמֶשׁ עוֹד מְאִירָה. לִפְנֵי הַבְּרָכָה מִסְתַּכְּלִים מְעַט בַּלְּבָנָה, וּבִשְׁעַת הַבְּרָכָה לֹא נוֹהֲגִים לְהִסְתַּכֵּל בָּהּ. כְּשֶׁעֲנָנִים מְכַסִּים אוֹתָהּ – לֹא מְבָרְכִים. עָנָן דַּק, שֶׁרוֹאִים דַּרְכּוֹ אֶת גְּבוּל הַלְּבָנָה – מֻתָּר. וְאִם בְּאֶמְצַע הַבְּרָכָה הִיא הִתְכַּסְּתָה – מַמְשִׁיכִים.
🤔 לָמָּה? מוֹדִים לַה׳ עַל הָאוֹר שֶׁל הַלְּבָנָה. כְּשֶׁאִי אֶפְשָׁר לֵהָנוֹת מֵאוֹרָהּ – אֵין עַל מָה לְהוֹדוֹת.
(פֶּרֶק א׳, הֲלָכָה י״ז)

21מֵאֵימָתַי וְעַד מָתַי?

נוֹהֲגִים לְבָרֵךְ עַל הַלְּבָנָה בְּמוֹצָאֵי שַׁבָּת שֶׁבִּתְחִלַּת הַחֹדֶשׁ. אֶפְשָׁר לְבָרֵךְ עַד סוֹף לֵיל ט״ו בַּחֹדֶשׁ, כִּי מִלֵּיל ט״ז הַלְּבָנָה מַתְחִילָה לְהִתְמַעֵט. וּמֵאֵיזֶה מוֹצָאֵי שַׁבָּת מַתְחִילִים?
🤔 לָמָּה? רוֹצִים לְחַכּוֹת שֶׁהַלְּבָנָה תִּגְדַּל מְעַט, כְּדֵי שֶׁאֶפְשָׁר יִהְיֶה לֵהָנוֹת מֵאוֹרָהּ. בְּמָקוֹם שֶׁמִּתְפַּלְּלִים יַחַד בְּנֵי עֵדוֹת שׁוֹנוֹת, וּמוֹצָאֵי שַׁבָּת חָל בְּשִׁבְעָה לַחֹדֶשׁ – נָכוֹן שֶׁכֻּלָּם יְבָרְכוּ יַחַד.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג אַשְׁכְּנַז וּמָרוֹקוֹ בְּמוֹצָאֵי שַׁבָּת שֶׁאַחֲרֵי שֶׁעָבְרוּ שְׁלֹשָׁה יָמִים שְׁלֵמִים מִן הַמּוֹלָד.
לְמִנְהַג הַסְּפָרַדִּים וְהַחֲסִידִים בְּמוֹצָאֵי שַׁבָּת שֶׁחָל אַחֲרֵי הַיּוֹם הַשְּׁבִיעִי לַחֹדֶשׁ.
💛 כְּשֶׁכָּל עֵדָה שׁוֹמֶרֶת אֶת מִנְהָגָהּ וּמְכַבֶּדֶת אֶת מִנְהַג חֲבֶרְתָּהּ – מִתְרַבָּה אַהֲבָה בֵּין בָּנָיו שֶׁל ה׳.
(פֶּרֶק א׳, הֲלָכָה י״ח)

📝 מָה לָמַדְנוּ

  • הַחֹדֶשׁ נִקְבָּע לְפִי הַלְּבָנָה, וְיִשְׂרָאֵל הֵם שֶׁמְּקַדְּשִׁים אֶת הַחֳדָשִׁים.
  • רֹאשׁ חֹדֶשׁ הוּא יוֹם שָׂמֵחַ: מוֹסִיפִים מַאֲכָל מְיֻחָד, וְלֹא צָמִים וְלֹא מִצְטַעֲרִים בּוֹ.
  • בְּרֹאשׁ חֹדֶשׁ אוֹמְרִים ״יַעֲלֶה וְיָבוֹא״, חֲצִי הַלֵּל בַּעֲמִידָה, קוֹרְאִים בַּתּוֹרָה וּמִתְפַּלְּלִים מוּסָף.
  • הַנָּשִׁים נוֹהֲגוֹת לְהִמָּנַע בְּרֹאשׁ חֹדֶשׁ מִקְּצָת מְלָאכוֹת – שָׂכָר עַל שֶׁלֹּא חָטְאוּ בָּעֵגֶל.
  • מְבָרְכִים עַל הַלְּבָנָה בַּלַּיְלָה, בַּחוּץ, בַּעֲמִידָה וּבְשִׂמְחָה – בְּדֶרֶךְ כְּלָל בְּמוֹצָאֵי שַׁבָּת, עַד לֵיל ט״ו.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה יֵשׁ חֹדֶשׁ ״חָסֵר״ וְחֹדֶשׁ ״מָלֵא״, וּמָתַי רֹאשׁ חֹדֶשׁ נִמְשָׁךְ יוֹמַיִם?
  2. שָׁכַחְתָּ ״יַעֲלֶה וְיָבוֹא״: מָה עוֹשִׂים בְּשַׁחֲרִית, מָה בְּעַרְבִית וּמָה בְּבִרְכַּת הַמָּזוֹן – וְלָמָּה?
  3. מָה מְלַמֶּדֶת אוֹתָנוּ הַלְּבָנָה, שֶׁנֶּעֱלֶמֶת וְחוֹזֶרֶת, וְאֵיךְ זֶה קָשׁוּר לְ״דָּוִד מֶלֶךְ יִשְׂרָאֵל חַי וְקַיָּם״?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״רֹאשׁ חֹדֶשׁ וְקִדּוּשׁ לְבָנָה״

סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת
מַהֲדוּרַת יְלָדִים

🌾 סְפִירַת הָעֹמֶר

פֶּרֶק ב׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
אֵיךְ מְחַכִּים לְיוֹם הֻלֶּדֶת, לְטִיּוּל אוֹ לְאוֹרֵחַ אָהוּב? סוֹפְרִים אֶת הַיָּמִים: ״עוֹד חֲמִשָּׁה יָמִים... עוֹד אַרְבָּעָה...״. כָּךְ בְּדִיּוּק עַם יִשְׂרָאֵל מְחַכֶּה לְמַתַּן תּוֹרָה. בְּכָל לַיְלָה, מִפֶּסַח וְעַד שָׁבוּעוֹת, אֲנַחְנוּ סוֹפְרִים עוֹד יוֹם. בַּפֶּרֶק הַזֶּה נִלְמַד אֵיךְ סוֹפְרִים, מָתַי, וּמָה עוֹשִׂים אִם שָׁכַחְנוּ.

1אַרְבָּעִים וְתִשְׁעָה יָמִים

מִצְוָה לִסְפֹּר אַרְבָּעִים וְתִשְׁעָה יָמִים, שֶׁהֵם שִׁבְעָה שָׁבוּעוֹת. מַתְחִילִים בְּלֵיל ט״ז בְּנִיסָן, בְּמוֹצָאֵי הַיּוֹם הָרִאשׁוֹן שֶׁל פֶּסַח. בִּימֵי בֵּית הַמִּקְדָּשׁ קָצְרוּ בַּלַּיְלָה הַזֶּה שְׂעוֹרִים, טָחֲנוּ אוֹתָן לְקֶמַח, וּלְמָחֳרָת הִקְרִיבוּ מֵהֶן אֶת קָרְבַּן הָעֹמֶר. רַק אַחֲרֵי הַקָּרְבָּן הַזֶּה הֻתַּר לֶאֱכֹל מִן הַתְּבוּאָה הַחֲדָשָׁה.
🤔 לָמָּה? חַג הַשָּׁבוּעוֹת מְיֻחָד: אֵין לוֹ תַּאֲרִיךְ קָבוּעַ בַּחֹדֶשׁ, כְּמוֹ פֶּסַח אוֹ סֻכּוֹת. הוּא בָּא אַחֲרֵי שֶׁגּוֹמְרִים לִסְפֹּר שִׁבְעָה שָׁבוּעוֹת – וְלָכֵן שְׁמוֹ ״חַג הַשָּׁבוּעוֹת״.
(פֶּרֶק ב׳, הֲלָכָה א׳)

2כָּל אֶחָד סוֹפֵר בְּעַצְמוֹ

הַמִּצְוָה אֵינָהּ רַק שֶׁל בֵּית הַדִּין – הִיא מִצְוָה עַל כָּל אֶחָד מִיִּשְׂרָאֵל. בְּהַרְבֵּה מִצְווֹת שֶׁל דִּבּוּר אֶפְשָׁר לָצֵאת כְּשֶׁשּׁוֹמְעִים מִמִּישֶׁהוּ אַחֵר, וְגַם אֶת הַבְּרָכָה עַל הַסְּפִירָה אֶפְשָׁר לִשְׁמֹעַ מִן הַחַזָּן. אֲבָל אֶת הַסְּפִירָה עַצְמָהּ כָּל אֶחָד צָרִיךְ לוֹמַר בְּפִיו.
🤔 לָמָּה? הַתּוֹרָה אָמְרָה ״וּסְפַרְתֶּם לָכֶם״ – לָכֶם, כָּל אֶחָד לְעַצְמוֹ. יֵשׁ פּוֹסְקִים שֶׁסּוֹבְרִים שֶׁגַּם בַּסְּפִירָה אֶפְשָׁר לָצֵאת בִּשְׁמִיעָה, אֲבָל כְּדֵי לָצֵאת לְדַעַת כֻּלָּם – כָּל אֶחָד סוֹפֵר בְּעַצְמוֹ.
(פֶּרֶק ב׳, הֲלָכָה א׳)

3לָמָּה סוֹפְרִים?

כְּשֶׁיָּצְאוּ יִשְׂרָאֵל מִמִּצְרַיִם, מֹשֶׁה בִּשֵּׂר לָהֶם שֶׁבְּעוֹד חֲמִשִּׁים יוֹם יְקַבְּלוּ אֶת הַתּוֹרָה. מֵרֹב חִבָּה הֵם סָפְרוּ כָּל יוֹם: ״הֲרֵי עָבַר יוֹם רִאשׁוֹן, הֲרֵי עָבַר יוֹם שֵׁנִי...״. בְּמִצְרַיִם הֵם הָיוּ שְׁקוּעִים בְּטֻמְאָה, וּבַשָּׁבוּעוֹת הָאֵלֶּה הֵם הִטַּהֲרוּ וְהִתְכּוֹנְנוּ לְקַבֵּל אֶת הַתּוֹרָה. גַּם אֲנַחְנוּ, בְּכָל שָׁנָה, סוֹפְרִים וּמִתְכּוֹנְנִים.
🤔 לָמָּה? בְּפֶסַח ה׳ בָּחַר בָּנוּ לִהְיוֹת עַמּוֹ, וּבְשָׁבוּעוֹת נָתַן לָנוּ אֶת תּוֹרָתוֹ. הַסְּפִירָה הִיא סֻלָּם שֶׁמְּחַבֵּר בֵּינֵיהֶם – כְּמוֹ שֶׁאוֹמְרִים בְּבִרְכַּת הַתּוֹרָה: ״אֲשֶׁר בָּחַר בָּנוּ מִכָּל הָעַמִּים וְנָתַן לָנוּ אֶת תּוֹרָתוֹ״.
💛 מִי שֶׁסּוֹפֵר אֶת הַיָּמִים עַד שֶׁיְּקַבֵּל מַתָּנָה – מַרְאֶה כַּמָּה הוּא אוֹהֵב אֶת הַנּוֹתֵן. כָּךְ אֲנַחְנוּ מְחַכִּים לַתּוֹרָה שֶׁל ה׳.
(פֶּרֶק ב׳, הֲלָכָה א׳, הֲלָכָה ב׳)

4סוֹפְרִים יָמִים וְשָׁבוּעוֹת

לִפְנֵי הַסְּפִירָה מְבָרְכִים: ״בָּרוּךְ אַתָּה ה׳... אֲשֶׁר קִדְּשָׁנוּ בְּמִצְוֹתָיו וְצִוָּנוּ עַל סְפִירַת הָעֹמֶר״. מְבָרְכִים וְסוֹפְרִים בַּעֲמִידָה, וּמִי שֶׁיָּשַׁב – יָצָא. סוֹפְרִים גַּם אֶת הַיָּמִים וְגַם אֶת הַשָּׁבוּעוֹת. בַּיּוֹם הַשְּׁבִיעִי אוֹמְרִים: ״הַיּוֹם שִׁבְעָה יָמִים, שֶׁהֵם שָׁבוּעַ אֶחָד״. וּבַיּוֹם הָעֲשִׂירִי: ״הַיּוֹם עֲשָׂרָה יָמִים, שֶׁהֵם שָׁבוּעַ אֶחָד וּשְׁלֹשָׁה יָמִים״. יֵשׁ אוֹמְרִים ״לָעֹמֶר״ וְיֵשׁ אוֹמְרִים ״בָּעֹמֶר״ – וּבְכָל הַנֻּסָּחִים יוֹצְאִים.
🤔 לָמָּה? שִׁבְעָה הוּא מִסְפָּר שָׁלֵם, כְּמוֹ שִׁבְעַת יְמֵי הַבְּרִיאָה. כְּדֵי לְהִתְכּוֹנֵן לַתּוֹרָה, שֶׁהִיא גְּבוֹהָה מְאוֹד, לֹא מַסְפִּיק שִׁבְעָה יָמִים – צָרִיךְ שִׁבְעָה שָׁבוּעוֹת, כְּדֵי שֶׁכָּל צַד בָּאֹפִי שֶׁלָּנוּ יִזְדַּכֵּךְ.
(פֶּרֶק ב׳, הֲלָכָה ג׳)

5מְחַכִּים לְבֵית הַמִּקְדָּשׁ

נֶחְלְקוּ הַפּוֹסְקִים: יֵשׁ אוֹמְרִים שֶׁהַיּוֹם, כְּשֶׁאֵין בֵּית מִקְדָּשׁ וְאֵין קָרְבַּן עֹמֶר, הַסְּפִירָה הִיא מִדִּבְרֵי חֲכָמִים, זֵכֶר לַסְּפִירָה שֶׁבַּזְּמַן הַמִּקְדָּשׁ. וְיֵשׁ אוֹמְרִים שֶׁגַּם הַיּוֹם הִיא מִצְוָה מִן הַתּוֹרָה. לָכֵן מִי שֶׁסָּפַר בֵּין הַשְּׁקִיעָה לְצֵאת הַכּוֹכָבִים – יָצָא, אֲבָל נָכוֹן שֶׁיַּחֲזֹר וְיִסְפֹּר אַחֲרֵי צֵאת הַכּוֹכָבִים, בְּלִי בְּרָכָה.
🤔 לָמָּה? בֵּין הַשְּׁקִיעָה לְצֵאת הַכּוֹכָבִים זֶה זְמַן שֶׁל סָפֵק – אוּלַי עוֹד יוֹם וְאוּלַי כְּבָר לַיְלָה. בִּסְפֵק שֶׁל מִצְוָה מִדִּבְרֵי חֲכָמִים מְקִלִּים, וּכְדֵי לָצֵאת גַּם לְדַעַת הָאוֹמְרִים שֶׁהִיא מִן הַתּוֹרָה – סוֹפְרִים שׁוּב.
💛 אַחֲרֵי הַסְּפִירָה נוֹהֲגִים לְהִתְפַּלֵּל שֶׁיִּבָּנֶה בֵּית הַמִּקְדָּשׁ – וְאָז נִסְפֹּר שׁוּב יַחַד עִם קָרְבַּן הָעֹמֶר.
(פֶּרֶק ב׳, הֲלָכָה ד׳)

6סוֹפְרִים בַּלַּיְלָה

סוֹפְרִים בַּלַּיְלָה. מִצְוָה לִסְפֹּר בִּתְחִלַּת הַלַּיְלָה, וּבְמִיּוּחָד בַּלַּיְלָה הָרִאשׁוֹן. אֲבָל מִי שֶׁעוֹמֵד לְהִתְפַּלֵּל עַרְבִית – מִתְפַּלֵּל קֹדֶם, וְרַק אַחַר כָּךְ סוֹפֵר.
🤔 לָמָּה? הַתּוֹרָה אָמְרָה ״שֶׁבַע שַׁבָּתוֹת תְּמִימֹת״ – שְׁלֵמוֹת. הַיְּמָמָה מַתְחִילָה בַּלַּיְלָה, וּכְדֵי שֶׁהַסְּפִירָה תִּהְיֶה שְׁלֵמָה וְתִכְלֹל אֶת כָּל הַשָּׁעוֹת – סוֹפְרִים בִּתְחִלַּת הַלַּיְלָה. וְעַרְבִית קוֹדֶמֶת, כִּי מִצְוָה שֶׁעוֹשִׂים כָּל הַשָּׁנָה (״תְּדִירָה״) קוֹדֶמֶת לְמִצְוָה שֶׁאֵינָהּ תְּדִירָה.
💛 כָּל לַיְלָה בַּסְּפִירָה הוּא עוֹד שְׁלַב בַּסֻּלָּם. מִי שֶׁסּוֹפֵר בִּזְמַנּוֹ אוֹמֵר לַה׳: אֲנִי מְחַכֶּה לַתּוֹרָה שֶׁלְּךָ.
(פֶּרֶק ב׳, הֲלָכָה ה׳)

7מִתְפַּלְּלִים מְאֻחָר?

מִי שֶׁרָגִיל לְהִתְפַּלֵּל עַרְבִית בְּכָל הַשָּׁנָה בְּשָׁעָה מְאֻחֶרֶת – מוּטָב שֶׁיִּסְפֹּר אַחֲרֵי הַתְּפִלָּה הַקְּבוּעָה שֶׁלּוֹ, עִם הַצִּבּוּר. אֲבָל מִי שֶׁיִּתְפַּלֵּל מְאֻחָר לְבַד – מוּטָב שֶׁיִּסְפֹּר מִיָּד אַחֲרֵי צֵאת הַכּוֹכָבִים.
🤔 לָמָּה? מִי שֶׁמְּשַׁנֶּה מִן הַסֵּדֶר הָרָגִיל שֶׁלּוֹ עָלוּל לִטְעוֹת אוֹ לִשְׁכֹּחַ, וְיֵשׁ מַעֲלָה לְקַיֵּם מִצְוָה בַּצִּבּוּר. וּמִי שֶׁמִּתְפַּלֵּל לְבַד – עָלוּל לִשְׁכֹּחַ לִסְפֹּר אַחֲרֵי הַתְּפִלָּה, וְלָכֵן עָדִיף שֶׁיִּזְדָּרֵז.
(פֶּרֶק ב׳, הֲלָכָה ו׳)

8שָׁכַחְתִּי לִסְפֹּר בַּלַּיְלָה

מִי שֶׁשָּׁכַח לִסְפֹּר בַּלַּיְלָה וְנִזְכַּר בַּיּוֹם – סוֹפֵר בַּיּוֹם בְּלִי בְּרָכָה. וּמִן הַלַּיְלָה הַבָּא הוּא מַמְשִׁיךְ לִסְפֹּר כָּרָגִיל, בִּבְרָכָה.
🤔 לָמָּה? נֶחְלְקוּ הָרִאשׁוֹנִים אִם אֶפְשָׁר לְהַשְׁלִים אֶת הַסְּפִירָה בַּיּוֹם. כְּדֵי לָצֵאת לְדַעַת הָאוֹמְרִים שֶׁאֶפְשָׁר – סוֹפְרִים. וּכְדֵי שֶׁלֹּא לְבָרֵךְ בְּרָכָה לְבַטָּלָה – סוֹפְרִים בְּלִי בְּרָכָה.
(פֶּרֶק ב׳, הֲלָכָה ו׳)

9שָׁכַחְתִּי יוֹם שָׁלֵם

מִי שֶׁשָּׁכַח לִסְפֹּר יוֹם שָׁלֵם – גַּם בַּלַּיְלָה וְגַם בַּיּוֹם שֶׁאַחֲרָיו – מַמְשִׁיךְ לִסְפֹּר בְּכָל הַיָּמִים הַבָּאִים, אֲבָל בְּלִי בְּרָכָה. כְּדֵי שֶׁלֹּא יַפְסִיד אֶת הַבְּרָכָה, יִשְׁמַע אוֹתָהּ מִן הַחַזָּן וְיִתְכַּוֵּן לָצֵאת בָּהּ, וְאַחַר כָּךְ יִסְפֹּר בְּעַצְמוֹ.
🤔 לָמָּה? לְדַעַת רֹב הַפּוֹסְקִים, כָּל יוֹם הוּא מִצְוָה בִּפְנֵי עַצְמָהּ – וְלָכֵן מַמְשִׁיכִים לִסְפֹּר. אֲבָל יֵשׁ אוֹמְרִים שֶׁכָּל הַסְּפִירָה הִיא מִצְוָה אַחַת אֲרֻכָּה, ״שֶׁבַע שַׁבָּתוֹת תְּמִימֹת״, וּמִי שֶׁשָּׁכַח יוֹם הִפְסִיד אוֹתָהּ – וְלָכֵן לֹא מְבָרְכִים.
💛 הַסְּפִירָה הִיא שַׁרְשֶׁרֶת שֶׁמְּחַבֶּרֶת אֶת פֶּסַח לְשָׁבוּעוֹת. כָּל יוֹם שֶׁאֲנִי זוֹכֵר לִסְפֹּר מוֹסִיף עוֹד חֻלְיָה לַשַּׁרְשֶׁרֶת.
(פֶּרֶק ב׳, הֲלָכָה ז׳)

10סָפֵק – וּבַר מִצְוָה בְּאֶמְצַע הַסְּפִירָה

מִי שֶׁלֹּא בָּטוּחַ אִם שָׁכַח לִסְפֹּר יוֹם – מַמְשִׁיךְ לִסְפֹּר בִּבְרָכָה. רַק כְּשֶׁבָּרוּר שֶׁשָּׁכַח יוֹם שָׁלֵם – מַמְשִׁיךְ בְּלִי בְּרָכָה. וּמָה עִם נַעַר שֶׁנַּעֲשָׂה בַּר מִצְוָה בְּאֶמְצַע הַסְּפִירָה, וְסָפַר בְּכָל הַיָּמִים?
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת רֹב הַפּוֹסְקִים, וְכֵן הַמִּנְהָג הָרוֹוֵחַ מַמְשִׁיךְ לִסְפֹּר בִּבְרָכָה, כִּי הַסְּפִירָה שֶׁלּוֹ שְׁלֵמָה מִן הַהַתְחָלָה.
לְדַעַת כַּמָּה פּוֹסְקִים מַמְשִׁיךְ בְּלִי בְּרָכָה, כִּי מָה שֶׁסָּפַר כְּשֶׁהָיָה קָטָן אֵינוֹ מִתְחַבֵּר לַסְּפִירָה שֶׁל גָּדוֹל.
💛 הַסְּפִירָה שֶׁל יֶלֶד אֵינָהּ מִשְׂחָק. כְּשֶׁהוּא סוֹפֵר בְּכָל יוֹם, הִיא מַמְשִׁיכָה אִתּוֹ גַּם כְּשֶׁהוּא נַעֲשֶׂה גָּדוֹל.
(פֶּרֶק ב׳, הֲלָכָה ח׳)

11נָשִׁים וּבָנוֹת בִּסְפִירַת הָעֹמֶר

נָשִׁים וּבָנוֹת פְּטוּרוֹת מִסְּפִירַת הָעֹמֶר, כִּי זוֹ מִצְוַת עֲשֵׂה שֶׁהַזְּמַן גְּרָמָהּ – מִצְוָה שֶׁתְּלוּיָה בִּזְמַן מְסֻיָּם. אֲבָל מִי שֶׁרוֹצָה לִסְפֹּר – יֵשׁ לָהּ מִצְוָה, כְּמוֹ שֶׁנָּשִׁים שׁוֹמְעוֹת שׁוֹפָר, נוֹטְלוֹת לוּלָב וְיוֹשְׁבוֹת בַּסֻּכָּה. וְהַאִם הִיא מְבָרֶכֶת?
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג רֹב יוֹצְאֵי סְפָרַד סוֹפֶרֶת בְּלִי בְּרָכָה, כִּי נָשִׁים אֵינָן מְבָרְכוֹת עַל מִצְווֹת שֶׁהַזְּמַן גְּרָמָן.
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז מִי שֶׁיּוֹדַעַת שֶׁתּוּכַל לִסְפֹּר אֶת כָּל הַסְּפִירָה, וְאִם תִּשְׁכַּח יוֹם – תֵּדַע לְהַמְשִׁיךְ בְּלִי בְּרָכָה, רַשָּׁאִית לִסְפֹּר בִּבְרָכָה. וְיֵשׁ מִפּוֹסְקֵי אַשְׁכְּנַז שֶׁכָּתְבוּ שֶׁמּוּטָב שֶׁלֹּא תְּבָרֵךְ, כִּי קַל יוֹתֵר לִשְׁכֹּחַ כְּשֶׁאֵין הוֹלְכִים לְבֵית הַכְּנֶסֶת.
💛 גַּם מִי שֶׁפְּטוּרָה יְכוֹלָה לִבְחֹר לְהִצְטָרֵף לַסְּפִירָה – וּלְהַרְאוֹת שֶׁגַּם הִיא מְחַכָּה לַתּוֹרָה בְּאַהֲבָה.
(פֶּרֶק ב׳, הֲלָכָה ט׳)

12״כַּמָּה הַיּוֹם לָעֹמֶר?״

חָבֵר שׁוֹאֵל אוֹתְךָ בָּעֶרֶב כַּמָּה הַיּוֹם לָעֹמֶר, וְעוֹד לֹא סָפַרְתָּ? אַל תַּעֲנֶה ״הַיּוֹם כָּךְ וְכָךְ״ – כִּי אָז כְּבָר יָצָאתָ יְדֵי הַסְּפִירָה, וְלֹא תּוּכַל לְבָרֵךְ. עֲנֵה: ״אֶתְמוֹל הָיָה כָּךְ וְכָךְ״. לִפְנֵי שֶׁמְּבָרְכִים חוֹשְׁבִים כַּמָּה הַיּוֹם. וּמִי שֶׁמִּסְתַּפֵּק אִם הַיּוֹם תִּשְׁעָה אוֹ עֲשָׂרָה, וְאֵין אֶת מִי לִשְׁאֹל – אוֹמֵר אֶת שְׁתֵּי הַסְּפִירוֹת, בְּלִי בְּרָכָה.
🤔 לָמָּה? כְּשֶׁאוֹמְרִים ״הַיּוֹם כָּךְ וְכָךְ״ – זוֹ כְּבָר סְפִירָה. וּמִי שֶׁסָּפַר, אֵינוֹ יָכוֹל לְבָרֵךְ עַל מָה שֶׁכְּבָר עָשָׂה.
💛 הַדִּקְדּוּק הַקָּטָן הַזֶּה מַרְאֶה כַּמָּה אֲנַחְנוּ אוֹהֲבִים אֶת הַמִּצְוָה: רוֹצִים לְקַיֵּם אוֹתָהּ כְּמוֹ שֶׁצָּרִיךְ – עִם בְּרָכָה.
(פֶּרֶק ב׳, הֲלָכָה י׳)

📝 מָה לָמַדְנוּ

  • סוֹפְרִים 49 יָמִים, שִׁבְעָה שָׁבוּעוֹת, מִלֵּיל ט״ז בְּנִיסָן – וְאָז מַגִּיעַ חַג הַשָּׁבוּעוֹת.
  • כָּל אֶחָד סוֹפֵר בְּפִיו, בַּלַּיְלָה, בַּעֲמִידָה, וּמַזְכִּיר יָמִים וְשָׁבוּעוֹת.
  • שָׁכַח בַּלַּיְלָה – סוֹפֵר בַּיּוֹם בְּלִי בְּרָכָה. שָׁכַח יוֹם שָׁלֵם – מַמְשִׁיךְ בְּלִי בְּרָכָה.
  • נָשִׁים וּבָנוֹת פְּטוּרוֹת, וְהָרוֹצָה סוֹפֶרֶת: לְמִנְהַג אַשְׁכְּנַז אֶפְשָׁר בִּבְרָכָה, לְמִנְהַג רֹב יוֹצְאֵי סְפָרַד – בְּלִי.
  • לִפְנֵי שֶׁסָּפַרְתָּ – עוֹנִים לְחָבֵר ״אֶתְמוֹל הָיָה כָּךְ וְכָךְ״.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה לְחַג הַשָּׁבוּעוֹת אֵין תַּאֲרִיךְ קָבוּעַ, וְלָמָּה הוּא נִקְרָא כָּךְ?
  2. מָה הַהֶבְדֵּל בֵּין מִי שֶׁשָּׁכַח לִסְפֹּר בַּלַּיְלָה וְנִזְכַּר בַּיּוֹם, לְבֵין מִי שֶׁשָּׁכַח יוֹם שָׁלֵם?
  3. חָבֵר שׁוֹאֵל אוֹתְךָ בָּעֶרֶב ״כַּמָּה הַיּוֹם לָעֹמֶר?״ – מָה תַּעֲנֶה, וְלָמָּה?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״סְפִירַת הָעֹמֶר וְל״ג בָּעֹמֶר״

סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת
מַהֲדוּרַת יְלָדִים

🧱 צוֹמוֹת הַחֻרְבָּן

פֶּרֶק ו׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
בְּלוּחַ הַשָּׁנָה יֵשׁ אַרְבָּעָה יָמִים שֶׁכָּתוּב לְיָדָם ״צוֹם״. מָה קָרָה בַּיָּמִים הָאֵלֶּה? לָמָּה צָמִים דַּוְקָא בָּהֶם? וְהַאִם הֵם יִשָּׁאֲרוּ יְמֵי צוֹם לְתָמִיד? בַּפֶּרֶק הַזֶּה נִפְתַּח אֶת סֵפֶר הַהִיסְטוֹרְיָה שֶׁל עַם יִשְׂרָאֵל, וְנִלְמַד עַל כָּל צוֹם – וְעַל הַהַבְטָחָה הַגְּדוֹלָה שֶׁל הַנָּבִיא.

1אַרְבָּעָה צוֹמוֹת

אַחֲרֵי חֻרְבַּן בֵּית הַמִּקְדָּשׁ הָרִאשׁוֹן תִּקְּנוּ הַנְּבִיאִים אַרְבָּעָה צוֹמוֹת: בַּעֲשָׂרָה בְּטֵבֵת – הִתְחִיל הַמָּצוֹר עַל יְרוּשָׁלַיִם. בְּתַמּוּז – הֻבְקְעָה חוֹמַת הָעִיר. בְּתִשְׁעָה בְּאָב – נֶחְרַב בֵּית הַמִּקְדָּשׁ. וּבִשְׁלֹשָׁה בְּתִשְׁרֵי – נֶהֱרַג גְּדַלְיָה בֶּן אֲחִיקָם.
🤔 לָמָּה? הַצּוֹמוֹת מְעוֹרְרִים אוֹתָנוּ לְהִצְטַעֵר עַל הַחֻרְבָּן וְעַל הַגָּלוּת, וּמִתּוֹךְ כָּךְ לַחֲזֹר בִּתְשׁוּבָה וּלְתַקֵּן אֶת הַמַּעֲשִׂים הָרָעִים שֶׁגָּרְמוּ לְכָל הַצָּרוֹת.
💛 הַצּוֹם אֵינוֹ רַק צַעַר. הוּא קְרִיאָה לָשׁוּב אֶל ה׳ – כְּמוֹ יֶלֶד שֶׁמִּתְגַּעְגֵּעַ לַבַּיִת שֶׁל אָבִיו.
(פֶּרֶק ו׳, הֲלָכָה א׳)

2״לְשָׂשׂוֹן וּלְשִׂמְחָה״

כְּשֶׁחָזְרוּ יִשְׂרָאֵל מִבָּבֶל וּבָנוּ אֶת בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי, אָמַר זְכַרְיָה הַנָּבִיא: ״צוֹם הָרְבִיעִי וְצוֹם הַחֲמִישִׁי וְצוֹם הַשְּׁבִיעִי וְצוֹם הָעֲשִׂירִי יִהְיֶה לְבֵית יְהוּדָה לְשָׂשׂוֹן וּלְשִׂמְחָה וּלְמֹעֲדִים טוֹבִים, וְהָאֱמֶת וְהַשָּׁלוֹם אֱהָבוּ״. וְכָךְ, בִּימֵי הַבַּיִת הַשֵּׁנִי, יְמֵי הַצּוֹם הָפְכוּ לְיָמִים שְׂמֵחִים. כְּשֶׁנֶּחְרַב הַבַּיִת הַשֵּׁנִי – חָזְרוּ לָצוּם.
🤔 לָמָּה? בַּחֻרְבָּן הָרִאשׁוֹן הֻבְקְעָה הַחוֹמָה בְּט׳ בְּתַמּוּז, וּבַחֻרְבָּן הַשֵּׁנִי – בְּי״ז בְּתַמּוּז. הַצַּעַר שֶׁל הַחֻרְבָּן הַשֵּׁנִי קָרוֹב אֵלֵינוּ יוֹתֵר, וְלָכֵן הַיּוֹם צָמִים בְּי״ז בְּתַמּוּז. וְתִשְׁעָה בְּאָב לֹא הִשְׁתַּנָּה, כִּי שְׁנֵי בָּתֵּי הַמִּקְדָּשׁ נֶחְרְבוּ בְּאוֹתוֹ יוֹם.
💛 ״וְהָאֱמֶת וְהַשָּׁלוֹם אֱהָבוּ״ – כָּל מַעֲשֶׂה שֶׁל אֱמֶת וְשֶׁל שָׁלוֹם מְקָרֵב אֶת הַיּוֹם שֶׁבּוֹ הַצּוֹמוֹת יֵהָפְכוּ לְחַגִּים.
(פֶּרֶק ו׳, הֲלָכָה א׳)

3עֲשָׂרָה בְּטֵבֵת

בַּעֲשָׂרָה בְּטֵבֵת בָּא נְבוּכַדְנֶאצַּר, מֶלֶךְ בָּבֶל, עִם צְבָאוֹ וְצָר עַל יְרוּשָׁלַיִם – וּמִשָּׁם הִתְחִיל הַחֻרְבָּן. לַצּוֹם הַזֶּה הִצְטָרְפוּ עוֹד שְׁנֵי מְאֹרָעוֹת עֲצוּבִים שֶׁקָּרוּ סָמוּךְ לוֹ: בְּח׳ בְּטֵבֵת תִּרְגְּמוּ אֶת הַתּוֹרָה לִיוָנִית בְּגְזֵרַת מַלְכוּת יָוָן, וְיָרַד חֹשֶׁךְ לָעוֹלָם שְׁלֹשָׁה יָמִים. וּבְט׳ בְּטֵבֵת מֵת עֶזְרָא הַסּוֹפֵר. הָרַבָּנוּת הָרָאשִׁית קָבְעָה אֶת עֲשָׂרָה בְּטֵבֵת לְ״יוֹם הַקַּדִּישׁ הַכְּלָלִי״ לְכָל קְדוֹשֵׁי הַשּׁוֹאָה שֶׁלֹּא נוֹדַע יוֹם פְּטִירָתָם.
🤔 לָמָּה? הַתּוֹרָה שַׁיֶּכֶת לְעַם יִשְׂרָאֵל. כְּשֶׁתִּרְגְּמוּ אוֹתָהּ לִיוָנִית בְּכֹחַ, הִיא הִתְחִילָה לְהֵרָאוֹת כְּמוֹ סֵפֶר רָגִיל שֶׁכָּל אֶחָד יָכוֹל לְהַחְלִיט מָה כָּתוּב בּוֹ. חֲכָמִים אָמְרוּ שֶׁהַיּוֹם הַזֶּה הָיָה קָשֶׁה לְיִשְׂרָאֵל כְּמוֹ הַיּוֹם שֶׁבּוֹ נַעֲשָׂה הָעֵגֶל.
(פֶּרֶק ו׳, הֲלָכָה ב׳)

4עֶזְרָא הַסּוֹפֵר – וּמָה מְתַקְּנִים

חֲכָמִים אָמְרוּ שֶׁעֶזְרָא הַסּוֹפֵר הָיָה רָאוּי שֶׁהַתּוֹרָה תִּנָּתֵן עַל יָדוֹ, אִלּוּלֵא קְדָמוֹ מֹשֶׁה רַבֵּנוּ. הוּא הֶעֱלָה אֶת הָעָם מִבָּבֶל, בָּנָה אֶת בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי, וְתִקֵּן עֶשֶׂר תַּקָּנוֹת חֲשׁוּבוֹת. הוּא הָיָה הָאַחֲרוֹן שֶׁל הַנְּבִיאִים וְהָרִאשׁוֹן שֶׁל חַכְמֵי הַתּוֹרָה שֶׁבְּעַל פֶּה. הָרַב צְבִי יְהוּדָה קוּק אָמַר שֶׁבַּעֲשָׂרָה בְּטֵבֵת צָרִיךְ לְתַקֵּן שְׁלֹשָׁה דְּבָרִים: כְּנֶגֶד הַמָּצוֹר – לִבְנוֹת אֶת יְרוּשָׁלַיִם וְאֶת הָאָרֶץ. כְּנֶגֶד מוֹת עֶזְרָא – לְהַגְדִּיל תּוֹרָה. וּכְנֶגֶד תַּרְגּוּם הַתּוֹרָה לִיוָנִית – לְחַזֵּק אֶת הָרוּחַ הַיִּשְׂרְאֵלִית הַמְּקוֹרִית.
💛 כָּךְ הַצַּעַר נַעֲשֶׂה כֹּחַ לִבְנוֹת: עַל כָּל דָּבָר שֶׁנֶּחְרַב, אֲנַחְנוּ יְכוֹלִים לִבְנוֹת מַשֶּׁהוּ טוֹב לִפְנֵי ה׳.
(פֶּרֶק ו׳, הֲלָכָה ב׳)

5שִׁבְעָה עָשָׂר בְּתַמּוּז

חֲמִשָּׁה דְּבָרִים רָעִים קָרוּ בְּי״ז בְּתַמּוּז: מֹשֶׁה רַבֵּנוּ שָׁבַר אֶת הַלּוּחוֹת כְּשֶׁרָאָה אֶת הָעֵגֶל, בָּטַל קָרְבַּן הַתָּמִיד בְּבֵית הַמִּקְדָּשׁ, הֻבְקְעָה חוֹמַת יְרוּשָׁלַיִם, אַפּוֹסְטְמוֹס שָׂרַף אֶת הַתּוֹרָה, וְהֶעֱמִידוּ צֶלֶם בַּהֵיכָל. הַצּוֹם נִקְבַּע בְּעִקָּר עַל הַבְקָעַת הַחוֹמָה. כְּשָׁלֹשׁ שָׁנִים צָרוּ הָרוֹמָאִים עַל יְרוּשָׁלַיִם וְלֹא יָכְלוּ לָהּ. אֲבָל בִּגְלַל שִׂנְאַת חִנָּם וּמִלְחֶמֶת אַחִים נֶחְלְשׁוּ הַמְּגִנִּים, וּבְי״ז בְּתַמּוּז הָרוֹמָאִים בָּקְעוּ אֶת הַחוֹמָה. אַחֲרֵי שְׁלֹשָׁה שָׁבוּעוֹת שֶׁל קְרָבוֹת, בְּתִשְׁעָה בְּאָב, נִשְׂרַף בֵּית הַמִּקְדָּשׁ.
🤔 לָמָּה? בְּכָל חֲמֵשֶׁת הַדְּבָרִים יֵשׁ ״סֶדֶק״ בַּשָּׁרָשִׁים הָרוּחָנִיִּים – בָּאֱמוּנָה, בַּתּוֹרָה וּבָעֲבוֹדָה. סֶדֶק שֶׁלֹּא מְתַקְּנִים בִּזְמַן הוֹלֵךְ וּמִתְרַחֵב, עַד שֶׁהוּא גּוֹרֵם לְחֻרְבָּן גָּמוּר.
💛 אִם שִׂנְאַת חִנָּם הֶחְרִיבָה – אַהֲבַת חִנָּם תִּבְנֶה. כָּל מִלָּה טוֹבָה לְחָבֵר הִיא עוֹד אֶבֶן בְּבִנְיַן יְרוּשָׁלַיִם.
(פֶּרֶק ו׳, הֲלָכָה ג׳)

6תִּשְׁעָה בְּאָב

חֲמִשָּׁה דְּבָרִים קָרוּ בְּתִשְׁעָה בְּאָב: נִגְזַר עַל דּוֹר הַמִּדְבָּר שֶׁלֹּא יִכָּנְסוּ לָאָרֶץ, חָרַב בֵּית הַמִּקְדָּשׁ הָרִאשׁוֹן, חָרַב בֵּית הַמִּקְדָּשׁ הַשֵּׁנִי, נִלְכְּדָה הָעִיר בֵּיתָר (בְּמֶרֶד בַּר כּוֹכְבָא), וְנֶחְרְשָׁה הָעִיר יְרוּשָׁלַיִם.
🤔 לָמָּה? כָּל הַמְּאֹרָעוֹת הָאֵלֶּה פָּגְעוּ בַּשְּׁלִיחוּת שֶׁל עַם יִשְׂרָאֵל – לְגַלּוֹת אֶת הַשְּׁכִינָה בָּעוֹלָם, בְּאֶרֶץ יִשְׂרָאֵל. עַל כָּךְ מִתְאַבְּלִים וְצָמִים.
(פֶּרֶק ו׳, הֲלָכָה ד׳)

7חֵטְא הַמְּרַגְּלִים

הַכֹּל הִתְחִיל בַּמִּדְבָּר. מֹשֶׁה שָׁלַח שְׁנֵים עָשָׂר מְרַגְּלִים לָתוּר אֶת הָאָרֶץ. עֲשָׂרָה מֵהֶם חָזְרוּ וְאָמְרוּ שֶׁאִי אֶפְשָׁר לִכְבֹּשׁ אוֹתָהּ, וְהָעָם בָּכָה כָּל הַלַּיְלָה וְרָצָה לַחֲזֹר לְמִצְרַיִם. רַק יְהוֹשֻׁעַ וְכָלֵב אָמְרוּ: ״טוֹבָה הָאָרֶץ מְאֹד מְאֹד... וַה׳ אִתָּנוּ, אַל תִּירָאֻם״. הַלַּיְלָה הַהוּא הָיָה לֵיל תִּשְׁעָה בְּאָב. אָמַר ה׳: ״אַתֶּם בְּכִיתֶם בְּכִיָּה שֶׁל חִנָּם, וַאֲנִי אֶקְבַּע לָכֶם בְּכִיָּה לְדוֹרוֹת״.
🤔 לָמָּה? בְּחֵטְא הָעֵגֶל עוֹד הֶאֱמִינוּ בַּה׳, רַק טָעוּ. אֲבָל הַמְּרַגְּלִים לֹא הֶאֱמִינוּ שֶׁה׳ יָכוֹל לַעֲזֹר לָהֶם לִכְבֹּשׁ אֶת הָאָרֶץ, וּמָאֲסוּ בָּאָרֶץ הַטּוֹבָה שֶׁה׳ נָתַן לָהֶם. לָכֵן הַחֵטְא הַזֶּה חָמוּר כָּל כָּךְ.
💛 הַתִּקּוּן לְחֵטְא הַמְּרַגְּלִים הוּא לֶאֱהֹב אֶת אֶרֶץ יִשְׂרָאֵל וּלְהַאֲמִין שֶׁה׳ אִתָּנוּ – כְּמוֹ יְהוֹשֻׁעַ וְכָלֵב.
(פֶּרֶק ו׳, הֲלָכָה ד׳)

8צוֹם גְּדַלְיָה

אַחֲרֵי חֻרְבַּן הַבַּיִת הָרִאשׁוֹן, מֶלֶךְ בָּבֶל מִנָּה אֶת גְּדַלְיָה בֶּן אֲחִיקָם לְמוֹשֵׁל עַל הַיְּהוּדִים שֶׁנִּשְׁאֲרוּ בָּאָרֶץ. הָיְתָה תִּקְוָה שֶׁהֵם יִתְבַּסְּסוּ וְיִבְנוּ שׁוּב. אֲבָל יִשְׁמָעֵאל בֶּן נְתַנְיָה, שֶׁרָצָה לִשְׁלֹט בִּמְקוֹמוֹ, רָצַח אוֹתוֹ. גְּדַלְיָה הֻזְהַר מֵרֹאשׁ, אֲבָל לֹא הֶאֱמִין לָאַזְהָרָה. בָּרֶצַח הַזֶּה כָּבְתָה הַגַּחֶלֶת הָאַחֲרוֹנָה שֶׁל שִׁלְטוֹן יִשְׂרָאֵל בָּאָרֶץ. צָמִים עַל כָּךְ בִּשְׁלֹשָׁה בְּתִשְׁרֵי.
🤔 לָמָּה? חֲכָמִים לָמְדוּ מִכָּאן: ״שְׁקוּלָה מִיתָתָן שֶׁל צַדִּיקִים כִּשְׂרֵפַת בֵּית אֱלֹהֵינוּ״. יֵשׁ מִן הָרִאשׁוֹנִים שֶׁכָּתְבוּ שֶׁגְּדַלְיָה נֶהֱרַג בְּרֹאשׁ הַשָּׁנָה, וְדָחוּ אֶת הַצּוֹם לִשְׁלֹשָׁה בְּתִשְׁרֵי.
💛 בְּכָל דּוֹר יֵשׁ צַדִּיקִים שֶׁמַּחְזִיקִים אֶת הָעָם. הַצּוֹם מְלַמֵּד אוֹתָנוּ כַּמָּה יָקָר כָּל צַדִּיק לִפְנֵי ה׳.
(פֶּרֶק ו׳, הֲלָכָה ה׳)

📝 מָה לָמַדְנוּ

  • הַנְּבִיאִים תִּקְּנוּ אַרְבָּעָה צוֹמוֹת עַל הַחֻרְבָּן: עֲשָׂרָה בְּטֵבֵת, י״ז בְּתַמּוּז, תִּשְׁעָה בְּאָב וְצוֹם גְּדַלְיָה.
  • בַּעֲשָׂרָה בְּטֵבֵת הִתְחִיל הַמָּצוֹר, בְּי״ז בְּתַמּוּז הֻבְקְעָה הַחוֹמָה, וּבְתִשְׁעָה בְּאָב נֶחְרְבוּ שְׁנֵי בָּתֵּי הַמִּקְדָּשׁ.
  • הַבְּכִיָּה שֶׁל חִנָּם בְּחֵטְא הַמְּרַגְּלִים הָיְתָה בְּלֵיל תִּשְׁעָה בְּאָב, וְשִׂנְאַת חִנָּם הֵבִיאָה לְהַבְקָעַת הַחוֹמָה.
  • זְכַרְיָה הַנָּבִיא הִבְטִיחַ: כְּשֶׁיִּבָּנֶה הַמִּקְדָּשׁ, הַצּוֹמוֹת יֵהָפְכוּ ״לְשָׂשׂוֹן וּלְשִׂמְחָה״.

❓ שְׁאֵלוֹת חֲזָרָה

  1. מָה קָרָה בְּכָל אֶחָד מֵאַרְבַּעַת הַצּוֹמוֹת?
  2. לָמָּה צָמִים הַיּוֹם בְּי״ז בְּתַמּוּז וְלֹא בְּט׳ בְּתַמּוּז?
  3. מָה הָיָה חֵטְא הַמְּרַגְּלִים, וְאֵיךְ אֲנַחְנוּ יְכוֹלִים לְתַקֵּן אוֹתוֹ?

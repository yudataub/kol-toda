סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת
מַהֲדוּרַת יְלָדִים

✂️ מִנְהֲגֵי אֲבֵלוּת בִּסְפִירַת הָעֹמֶר

פֶּרֶק ג׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
שַׂמְתֶּם לֵב שֶׁבֵּין פֶּסַח לְשָׁבוּעוֹת הַרְבֵּה יְלָדִים מִסְתּוֹבְבִים עִם שֵׂעָר אָרֹךְ? וְשֶׁבַּחֲתֻנּוֹת וּבִמְסִבּוֹת פִּתְאוֹם יֵשׁ הַפְסָקָה? זֶה לֹא בְּמִקְרֶה. הַיָּמִים הָאֵלֶּה הֵם יְמֵי סְפִירָה וְצִפִּיָּה לַתּוֹרָה – אֲבָל גַּם יָמִים שֶׁל צַעַר. בַּפֶּרֶק הַזֶּה נִלְמַד עַל מָה מִצְטַעֲרִים, מִמָּה נִמְנָעִים, וְעַד מָתַי.

1יְמֵי צַעַר

בֵּין פֶּסַח לְשָׁבוּעוֹת מֵתוּ עֶשְׂרִים וְאַרְבָּעָה אֶלֶף תַּלְמִידֵי רַבִּי עֲקִיבָא, ״מִפְּנֵי שֶׁלֹּא נָהֲגוּ כָּבוֹד זֶה לָזֶה״. לָכֵן נוֹהֲגִים בַּיָּמִים הָאֵלֶּה מִקְצָת מִנְהֲגֵי אֲבֵלוּת: לֹא מִסְתַּפְּרִים, לֹא עוֹרְכִים רִקּוּדִים, וְלֹא עוֹרְכִים חֲתֻנּוֹת. זֶה מִנְהָג שֶׁנָּהֲגוּ יִשְׂרָאֵל, וְלָכֵן יֵשׁ הֶבְדֵּלִים בֵּין הָעֵדוֹת.
🤔 לָמָּה? גַּם אַחַר כָּךְ, בִּימֵי מַסְעֵי הַצְּלָב וּבִגְזֵרוֹת ת״ח וְת״ט, נִרְצְחוּ רְבָבוֹת יְהוּדִים בְּאַשְׁכְּנַז וּבְמִזְרַח אֵירוֹפָּה – וְרֹב הַפְּרָעוֹת הָיוּ בִּימֵי הַסְּפִירָה. לָכֵן בְּאַשְׁכְּנַז הֶחְמִירוּ יוֹתֵר בַּמִּנְהָגִים הָאֵלֶּה.
(פֶּרֶק ג׳, הֲלָכָה א׳)

2רַבִּי עֲקִיבָא מַתְחִיל מֵחָדָשׁ

אַחֲרֵי שֶׁמֵּתוּ תַּלְמִידָיו, הָעוֹלָם הָיָה שָׁמֵם. אֲבָל רַבִּי עֲקִיבָא לֹא הִתְיָאֵשׁ. הוּא לִמֵּד תַּלְמִידִים חֲדָשִׁים – רַבִּי מֵאִיר, רַבִּי יְהוּדָה, רַבִּי יוֹסֵי, רַבִּי שִׁמְעוֹן וְרַבִּי אֶלְעָזָר בֶּן שַׁמּוּעַ – וְהֵם הֶעֱמִידוּ אֶת הַתּוֹרָה. הוּא אָמַר לָהֶם: ״בָּנַי, הָרִאשׁוֹנִים לֹא מֵתוּ אֶלָּא שֶׁהָיְתָה עֵינָם צָרָה אֵלּוּ לָאֵלּוּ – תְּנוּ דַּעְתְּכֶם שֶׁלֹּא תַּעֲשׂוּ כְּמַעֲשֵׂיהֶם״.
🤔 לָמָּה? לָכֵן בַּיָּמִים הָאֵלֶּה לֹא רַק מִתְאַבְּלִים – גַּם מִשְׁתַּדְּלִים לְתַקֵּן אֶת הַיְּחָסִים בֵּין אָדָם לַחֲבֵרוֹ, וּבְמִיּוּחָד בֵּין הַלּוֹמְדִים.
💛 מִי שֶׁמְּכַבֵּד אֶת חֲבֵרוֹ וְשָׂמֵחַ בְּהַצְלָחָתוֹ – מְכַבֵּד אֶת ה׳ שֶׁבָּרָא אוֹתוֹ.
(פֶּרֶק ג׳, הֲלָכָה א׳)

3עַד מָתַי נִמְשֶׁכֶת הָאֲבֵלוּת?

יֵשׁ כַּמָּה מִנְהָגִים, וְכָל עֵדָה מַמְשִׁיכָה בְּמִנְהָגָהּ:
🤔 לָמָּה? לְמִנְהַג הַסְּפָרַדִּים, הַתַּלְמִידִים מֵתוּ עַד ״פְּרוֹס הָעֲצֶרֶת״ – חֲצִי חֹדֶשׁ לִפְנֵי שָׁבוּעוֹת, כְּלוֹמַר עַד יוֹם ל״ד. וּמַסְפִּיק מְעַט מִן הַבֹּקֶר שֶׁל ל״ד – ״מִקְצָת הַיּוֹם כְּכֻלּוֹ״. לְפִי מָסֹרֶת אַחֶרֶת, בְּל״ג בָּעֹמֶר הִפְסִיקוּ הַתַּלְמִידִים לָמוּת. וּבְאַשְׁכְּנַז מַמְשִׁיכִים קְצָת אַחֲרָיו, כִּי יֵשׁ תַּלְמִידִים שֶׁחָלוּ לִפְנֵי ל״ג וּמֵתוּ אַחֲרָיו, וּבִגְלַל הַגְּזֵרוֹת שֶׁהָיוּ בְּסוֹף הַסְּפִירָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג רַבִּים מִיּוֹצְאֵי סְפָרַד מִתְּחִלַּת הַסְּפִירָה וְעַד הַבֹּקֶר שֶׁל ל״ד בָּעֹמֶר. בְּל״ג בָּעֹמֶר מֻתָּר לָשִׁיר וְלִרְקֹד לִכְבוֹד רַבִּי שִׁמְעוֹן בַּר יוֹחָאי, אֲבָל שְׁאָר הַמִּנְהָגִים נִשְׁאָרִים, וּבְלֵיל ל״ד חוֹזְרִים לֹא לְנַגֵּן.
לְמִנְהַג קְהִלּוֹת רַבּוֹת מִיּוֹצְאֵי סְפָרַד (כְּמוֹ טוּרְקִיָּה וּמִצְרַיִם) כָּל מִנְהֲגֵי הָאֲבֵלוּת מִסְתַּיְּמִים בְּל״ג בָּעֹמֶר.
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז עִקַּר הָאֲבֵלוּת – לֹא מִסְתַּפְּרִים וְלֹא מְנַגְּנִים – עַד ל״ג בָּעֹמֶר. אַחֲרָיו, עַד שָׁבוּעוֹת, נִמְנָעִים רַק מִשְּׂמָחוֹת גְּדוֹלוֹת.
💛 כָּל עֵדָה שָׁמְרָה עַל הַמָּסֹרֶת שֶׁלָּהּ בִּמְסִירוּת – וְכֻלָּן זוֹכְרוֹת יַחַד אֶת הַצַּעַר, וְיוֹדְעוֹת לִשְׂמֹחַ בַּזְּמַן הַנָּכוֹן.
(פֶּרֶק ג׳, הֲלָכָה ב׳, הֲלָכָה ג׳, הֲלָכָה ד׳)

4לֹא מִסְתַּפְּרִים

בִּימֵי הָאֲבֵלוּת שֶׁבַּסְּפִירָה לֹא מִסְתַּפְּרִים. מָתַי מֻתָּר שׁוּב?
🤔 לָמָּה? תִּסְפֹּרֶת יֵשׁ בָּהּ שִׂמְחָה. לָכֵן מִי שֶׁהַשֵּׂעָר גּוֹרֵם לוֹ כְּאֵב רֹאשׁ אוֹ פְּצָעִים – מֻתָּר לוֹ לְהִסְתַּפֵּר. וְלִכְבוֹד בְּרִית מִילָה מֻתָּר לְבַעֲלֵי הַשִּׂמְחָה – אֲבִי הַבֵּן, הַסַּנְדָּק וְהַמּוֹהֵל – לְהִסְתַּפֵּר.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג רַבִּים מִיּוֹצְאֵי סְפָרַד בַּבֹּקֶר שֶׁל ל״ד בָּעֹמֶר. בְּל״ג בָּעֹמֶר עֲדַיִן לֹא מִסְתַּפְּרִים.
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז וּקְהִלּוֹת רַבּוֹת מִיּוֹצְאֵי סְפָרַד בַּבֹּקֶר שֶׁל ל״ג בָּעֹמֶר. יֵשׁ מְקִלִּים כְּבָר מִלֵּיל ל״ג, וּבִשְׁעַת הַצֹּרֶךְ אֶפְשָׁר לִסְמֹךְ עֲלֵיהֶם.
הַנּוֹהֲגִים לְפִי הָאֲרִ״י לֹא מִסְתַּפְּרִים כָּל יְמֵי הַסְּפִירָה, עַד עֶרֶב חַג הַשָּׁבוּעוֹת – וְאָז מִסְתַּפְּרִים לִכְבוֹד הֶחָג.
💛 הַשֵּׂעָר הָאָרֹךְ מַזְכִּיר לָנוּ בְּכָל יוֹם שֶׁאֵלֶּה יְמֵי צַעַר – וְשֶׁכְּדַאי לְהוֹסִיף בָּהֶם אַהֲבַת חֲבֵרִים.
(פֶּרֶק ג׳, הֲלָכָה ו׳)

5גַּם בָּנוֹת, וְגַם יְלָדִים קְטַנִּים

גַּם לְנָשִׁים וּלְבָנוֹת אָסוּר לְהִסְתַּפֵּר בִּימֵי הָאֲבֵלוּת. וְגַם אֶת הַיְּלָדִים הַקְּטַנִּים לֹא מְסַפְּרִים. אֲבָל לְצֹרֶךְ גָּדוֹל, כְּדֵי שֶׁהַיֶּלֶד לֹא יִצְטַעֵר – מֻתָּר.
🤔 לָמָּה? הָאֲבֵלוּת הִיא שֶׁל כָּל עַם יִשְׂרָאֵל – גְּדוֹלִים וּקְטַנִּים, בָּנִים וּבָנוֹת. אֲבָל מִנְהַג אֲבֵלוּת לֹא בָּא לְצַעֵר יְלָדִים, וְלָכֵן בְּצֹרֶךְ גָּדוֹל מְקִלִּים.
(פֶּרֶק ג׳, הֲלָכָה ו׳)

6בְּלִי רִקּוּדִים וּכְלֵי נְגִינָה

בִּימֵי הָאֲבֵלוּת אֵין עוֹרְכִים רִקּוּדִים וּמְחוֹלוֹת שֶׁל רְשׁוּת, וְלֹא מְנַגְּנִים וְלֹא שׁוֹמְעִים כְּלֵי זֶמֶר. מֻתָּר לִלְמֹד לְנַגֵּן וּלְלַמֵּד נְגִינָה, כִּי אֵין בָּזֶה שִׂמְחָה. וּמָתַי חוֹזְרִים לְנַגֵּן?
🤔 לָמָּה? בִּימֵי צַעַר לֹא מַרְבִּים בִּשְׂמָחוֹת יְתֵרוֹת. וּבְל״ג בָּעֹמֶר שָׁרִים וְרוֹקְדִים לִכְבוֹד הַהִלּוּלָא שֶׁל רַבִּי שִׁמְעוֹן בַּר יוֹחָאי.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג רַבִּים מִיּוֹצְאֵי סְפָרַד בְּל״ג בָּעֹמֶר מֻתָּר לְנַגֵּן וְלִרְקֹד, בְּלֵיל ל״ד שׁוּב אָסוּר, וּמִן הַבֹּקֶר שֶׁל ל״ד – הַכֹּל מֻתָּר.
לְמִנְהַג קְהִלּוֹת רַבּוֹת מִיּוֹצְאֵי סְפָרַד מִיּוֹם ל״ג בָּעֹמֶר וָהָלְאָה – הַכֹּל מֻתָּר.
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז מִלֵּיל ל״ג בָּעֹמֶר מֻתָּר לְנַגֵּן וְלִרְקֹד. אַחֲרָיו רֹב יוֹצְאֵי אַשְׁכְּנַז נִמְנָעִים עַד שָׁבוּעוֹת מִשְּׂמָחוֹת גְּדוֹלוֹת – כְּמוֹ מְסִבַּת סִיּוּם שָׁנָה חֲגִיגִית אוֹ עֶרֶב רִקּוּדִים – אֲבָל מֻתָּר לְנַגֵּן וְלִשְׁמֹעַ נְגִינָה.
(פֶּרֶק ג׳, הֲלָכָה ח׳)

7חֹל הַמּוֹעֵד פֶּסַח

חֹל הַמּוֹעֵד שֶׁל פֶּסַח נִמְצָא בְּתוֹךְ יְמֵי הַסְּפִירָה, אֲבָל אֵין בּוֹ שׁוּם מִנְהַג אֲבֵלוּת: מְנַגְּנִים וְרוֹקְדִים, כִּי מִצְוָה לִשְׂמֹחַ בּוֹ. וּבְכָל זֹאת לֹא מִסְתַּפְּרִים בְּחֹל הַמּוֹעֵד וְלֹא עוֹרְכִים בּוֹ חֲתֻנּוֹת – מִסִּבּוֹת אֲחֵרוֹת.
🤔 לָמָּה? חֲכָמִים אָסְרוּ לְהִסְתַּפֵּר בְּחֹל הַמּוֹעֵד כְּדֵי שֶׁיִּסְתַּפְּרוּ לִפְנֵי הֶחָג וְיִכָּנְסוּ אֵלָיו מְסֻדָּרִים. וַחֲתֻנּוֹת לֹא עוֹרְכִים בּוֹ, כִּי ״אֵין מְעָרְבִין שִׂמְחָה בְּשִׂמְחָה״.
💛 גַּם בְּתוֹךְ יְמֵי הַצַּעַר, שִׂמְחַת הַחַג שֶׁה׳ נָתַן לָנוּ נִשְׁאֶרֶת שְׁלֵמָה.
(פֶּרֶק ג׳, הֲלָכָה ח׳, הֲלָכָה י״ב)

8שִׂמְחָה שֶׁל מִצְוָה

בִּסְעוּדַת מִצְוָה מֻתָּר לָשִׁיר וְלִרְקֹד, כְּמוֹ בְּכָל הַשָּׁנָה: בִּבְרִית מִילָה, בְּפִדְיוֹן הַבֵּן וּבְסִיּוּם מַסֶּכֶת. מַכְנִיסִים סֵפֶר תּוֹרָה לְבֵית הַכְּנֶסֶת בְּשִׁירָה וּבְרִקּוּדִים. וְכָךְ גַּם בִּסְעוּדַת בַּר מִצְוָה אוֹ בַּת מִצְוָה שֶׁנַּעֲרֶכֶת בַּיּוֹם שֶׁנִּכְנָסִים בּוֹ לְמִצְווֹת. אִם הַמְּסִבָּה בְּיוֹם אַחֵר – עוֹרְכִים אוֹתָהּ בְּלִי מוּזִיקָה, אֶלָּא אִם כֵּן בַּר הַמִּצְוָה אוֹ בַּת הַמִּצְוָה מְסַיְּמִים בָּהּ מַסֶּכֶת אוֹ סֵפֶר חָשׁוּב.
🤔 לָמָּה? הַמִּנְהָג הוּא לְהִמָּנַע מִשִּׂמְחָה שֶׁל רְשׁוּת. שִׂמְחָה שֶׁל מִצְוָה אֵינָהּ נִדְחֵית.
💛 כְּשֶׁאָנוּ שְׂמֵחִים יַחַד בַּתּוֹרָה וּבַמִּצְווֹת, אָנוּ מְתַקְּנִים בְּדִיּוּק אֶת מָה שֶׁהֶחְסִירוּ הַתַּלְמִידִים – אַחְדוּת וְכָבוֹד סְבִיב הַתּוֹרָה.
(פֶּרֶק ג׳, הֲלָכָה ט׳)

9מוּזִיקָה מִמַּכְשִׁיר בַּבַּיִת

שִׁירִים בְּלִי כְּלֵי נְגִינָה (בְּקוֹל בִּלְבַד) – רַבִּים מַתִּירִים לִשְׁמֹעַ. וּמָה עִם שִׁירִים עִם כְּלֵי נְגִינָה, מִמַּכְשִׁיר שֶׁבַּבַּיִת?
🤔 לָמָּה? הַמַּחְמִירִים סוֹבְרִים שֶׁאֵין הֶבְדֵּל בֵּין נַגָּן חַי לְמַכְשִׁיר. הַמְּקִלִּים סוֹבְרִים שֶׁהַיּוֹם כֻּלָּם שׁוֹמְעִים מוּזִיקָה בְּמַכְשִׁירִים כָּל הַזְּמַן, וְאֵין בָּזֶה חֲגִיגִיּוּת כְּמוֹ בִּנְגִינָה חַיָּה. בְּשְׁאֵלָה כָּזוֹ שׁוֹאֲלִים בַּבַּיִת מָה הַמִּנְהָג שֶׁלָּנוּ.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת רַבִּים מִן הַפּוֹסְקִים (הָרַב מֹשֶׁה פַיינְשְׁטֵיין, הָרַב עוֹבַדְיָה יוֹסֵף) אָסוּר, כְּמוֹ נְגִינָה חַיָּה. שִׁירִים בְּלִי כְּלֵי נְגִינָה – מֻתָּר.
וְיֵשׁ מַחְמִירִים יוֹתֵר גַּם שִׁירִים בְּלִי כְּלֵי נְגִינָה אָסוּר לִשְׁמֹעַ מִמַּכְשִׁיר, כִּי הַמַּכְשִׁיר עַצְמוֹ כְּמוֹ כְּלִי זֶמֶר.
לְפִי הָרַב מֶלָמֵד הָרוֹצֶה לְהָקֵל רַשַּׁאי לִשְׁמֹעַ שִׁירִים רְגִילִים אוֹ עֲצוּבִים – לֹא שִׁירֵי שִׂמְחָה – וְלֹא בְּקוֹל רָם.
(פֶּרֶק ג׳, הֲלָכָה י׳)

10שֶׁהֶחֱיָנוּ, בְּגָדִים חֲדָשִׁים וְטִיּוּלִים

בִּימֵי הַסְּפִירָה מֻתָּר לְבָרֵךְ ״שֶׁהֶחֱיָנוּ״ עַל פְּרִי חָדָשׁ, וְגַם עַל בֶּגֶד חָדָשׁ אוֹ רָהִיט חָדָשׁ. מִי שֶׁרוֹצֶה לְהַחְמִיר וְלֹא לִקְנוֹת בְּגָדִים חֲדָשִׁים – תָּבוֹא עָלָיו בְּרָכָה, וְיִלְבַּשׁ אוֹתָם לָרִאשׁוֹנָה בְּשַׁבָּת אוֹ בְּרֹאשׁ חֹדֶשׁ. מֻתָּר לְהַזְמִין חֲבֵרִים לִסְעוּדָה (בְּלִי כְּלֵי נְגִינָה), וּמֻתָּר לְטַיֵּל. אֲבָל טִיּוּל שֶׁל בֵּית סֵפֶר, שֶׁיֵּשׁ בּוֹ שִׂמְחָה גְּדוֹלָה, מוּטָב לְקַיֵּם אַחֲרֵי ל״ג בָּעֹמֶר. טִיּוּל חִנּוּכִי – מֻתָּר.
🤔 לָמָּה? בִּימֵי הַסְּפִירָה נִמְנָעִים מִשִּׂמְחָה מְיֻחֶדֶת, וְלֹא מִכָּל דָּבָר נָעִים. אֵין לְהַשְׁווֹת אוֹתָם לִימֵי ״בֵּין הַמְּצָרִים״, שֶׁבָּהֶם לֹא מְבָרְכִים ״שֶׁהֶחֱיָנוּ״.
💛 הַתּוֹרָה אֵינָהּ רוֹצָה שֶׁנִּהְיֶה עֲצוּבִים – רַק שֶׁנִּזְכֹּר, וְנַמְשִׁיךְ לְהוֹדוֹת לַה׳ עַל הַטּוֹב.
(פֶּרֶק ג׳, הֲלָכָה י״א)

11יָמִים מְיֻחָדִים בְּתוֹךְ הַסְּפִירָה

בְּרֹאשׁ חֹדֶשׁ אִיָּר נוֹהֲגִים לֹא לְהִסְתַּפֵּר, אַף שֶׁהוּא יוֹם שֶׁל שִׂמְחָה. כְּשֶׁל״ג בָּעֹמֶר חָל בְּיוֹם שִׁשִּׁי – גַּם הַסְּפָרַדִּים הַמַּחְמִירִים מִסְתַּפְּרִים בּוֹ, לִכְבוֹד שַׁבָּת. וּמָה כְּשֶׁל״ג בָּעֹמֶר חָל בְּיוֹם רִאשׁוֹן, אוֹ כְּשֶׁרֹאשׁ חֹדֶשׁ אִיָּר חָל בְּשַׁבָּת – הַאִם מִסְתַּפְּרִים בְּיוֹם שִׁשִּׁי?
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז מֻתָּר לְהִסְתַּפֵּר בְּיוֹם שִׁשִּׁי, לִכְבוֹד שַׁבָּת.
לְמִנְהַג יוֹצְאֵי סְפָרַד כְּשֶׁל״ג בָּעֹמֶר חָל בְּיוֹם רִאשׁוֹן – אָסוּר. כְּשֶׁרֹאשׁ חֹדֶשׁ אִיָּר חָל בְּשַׁבָּת – הַמִּנְהָג הָרוֹוֵחַ לְהָקֵל רַק בִּשְׁעַת הַדְּחָק.
(פֶּרֶק ג׳, הֲלָכָה י״ב)

📝 מָה לָמַדְנוּ

  • בֵּין פֶּסַח לְשָׁבוּעוֹת מִתְאַבְּלִים עַל תַּלְמִידֵי רַבִּי עֲקִיבָא, שֶׁלֹּא נָהֲגוּ כָּבוֹד זֶה לָזֶה – וּמְתַקְּנִים אֶת הַיְּחָסִים עִם חֲבֵרִים.
  • לֹא מִסְתַּפְּרִים (גַּם בָּנוֹת וּקְטַנִּים), לֹא רוֹקְדִים וְלֹא מְנַגְּנִים – אֲבָל בְּשִׂמְחָה שֶׁל מִצְוָה מֻתָּר.
  • לְמִנְהַג אַשְׁכְּנַז הָאֲבֵלוּת עַד ל״ג בָּעֹמֶר; לְמִנְהַג רַבִּים מִיּוֹצְאֵי סְפָרַד עַד הַבֹּקֶר שֶׁל ל״ד; הַנּוֹהֲגִים כָּאֲרִ״י לֹא מִסְתַּפְּרִים עַד עֶרֶב שָׁבוּעוֹת.
  • בְּחֹל הַמּוֹעֵד פֶּסַח אֵין אֲבֵלוּת, וּמֻתָּר לְבָרֵךְ ״שֶׁהֶחֱיָנוּ״ וּלְטַיֵּל בִּימֵי הַסְּפִירָה.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה נוֹהֲגִים אֲבֵלוּת בֵּין פֶּסַח לְשָׁבוּעוֹת, וּמָה אֶפְשָׁר לְתַקֵּן בַּיָּמִים הָאֵלֶּה?
  2. עַד מָתַי לֹא מִסְתַּפְּרִים – לְמִנְהַג אַשְׁכְּנַז, לְמִנְהַג רַבִּים מִיּוֹצְאֵי סְפָרַד, וְלַנּוֹהֲגִים כָּאֲרִ״י?
  3. בְּאֵילּוּ שְׂמָחוֹת מֻתָּר לָשִׁיר וְלִרְקֹד גַּם בִּימֵי הָאֲבֵלוּת, וְלָמָּה?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״סְפִירַת הָעֹמֶר וְל״ג בָּעֹמֶר״

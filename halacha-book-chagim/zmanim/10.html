סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת
מַהֲדוּרַת יְלָדִים

🕯️ תִּשְׁעָה בְּאָב

פֶּרֶק י׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
בֵּית הַכְּנֶסֶת חָשׁוּךְ. רַק כַּמָּה נֵרוֹת דּוֹלְקִים. הַפָּרֹכֶת הוּסְרָה מִן הָאָרוֹן, כֻּלָּם יוֹשְׁבִים עַל הָרִצְפָּה, וְהַחַזָּן מַתְחִיל בְּקוֹל עָצוּב: ״אֵיכָה יָשְׁבָה בָדָד...״. זֶה תִּשְׁעָה בְּאָב – הַיּוֹם הֶעָצוּב בְּיוֹתֵר בַּשָּׁנָה, הַיּוֹם שֶׁבּוֹ נֶחְרְבוּ שְׁנֵי בָּתֵּי הַמִּקְדָּשׁ. בַּפֶּרֶק הַזֶּה נִלְמַד אֵיךְ מִתְאַבְּלִים עַל יְרוּשָׁלַיִם – וְאֵיךְ מִתּוֹךְ הָאֵבֶל מִתְחִילָה הַנֶּחָמָה.

1הַצּוֹם הֶחָמוּר

תִּשְׁעָה בְּאָב הוּא הַחָמוּר מִכָּל צוֹמוֹת הַחֻרְבָּן. הוּא שׁוֹנֶה מִן הַצּוֹמוֹת הַקַּלִּים בִּשְׁלֹשָׁה דְּבָרִים: הוּא נִמְשָׁךְ יְמָמָה שְׁלֵמָה – מִשְּׁקִיעַת הַחַמָּה וְעַד צֵאת הַכּוֹכָבִים. אָסוּר בּוֹ בַּחֲמִשָּׁה דְּבָרִים, כְּמוֹ בְּיוֹם הַכִּפּוּרִים: אֲכִילָה וּשְׁתִיָּה, רְחִיצָה, סִיכָה, נְעִילַת הַסַּנְדָּל וְתַשְׁמִישׁ הַמִּטָּה. וְרַק חוֹלִים פְּטוּרִים מִמֶּנּוּ.
🤔 לָמָּה? בְּתִשְׁעָה בְּאָב נִכְפְּלוּ הַצָּרוֹת – שְׁנֵי בָּתֵּי הַמִּקְדָּשׁ נֶחְרְבוּ בּוֹ. לָכֵן כָּל עוֹד בֵּית הַמִּקְדָּשׁ חָרֵב, חַיָּבִים לָצוּם בּוֹ בְּכָל מַצָּב, כְּמוֹ שֶׁתִּקְּנוּ הַנְּבִיאִים.
(פֶּרֶק י׳, הֲלָכָה א׳, הֲלָכָה ב׳)

2כְּמוֹ יוֹם הַכִּפּוּרִים – וְשׁוֹנֶה מִמֶּנּוּ

תִּשְׁעָה בְּאָב קַל מִיּוֹם הַכִּפּוּרִים בִּשְׁנֵי דְּבָרִים: חוֹלֶה פָּטוּר מִמֶּנּוּ, וּבְכָל סָפֵק – מְקִלִּים, כִּי הוּא מִדִּבְרֵי חֲכָמִים. אֲבָל מִצַּד שֵׁנִי הוּא חָמוּר יוֹתֵר, כִּי הוּא יוֹם אֵבֶל: יוֹשְׁבִים עַל הָאָרֶץ, לֹא אוֹמְרִים שָׁלוֹם, מַחְשִׁיכִים אֶת הָאוֹר וְלֹא לוֹמְדִים תּוֹרָה מְשַׂמַּחַת. בְּיוֹם הַכִּפּוּרִים, לְעֻמַּת זֹאת, לוֹבְשִׁים בְּגָדִים נָאִים, יוֹשְׁבִים עַל סַפְסָלִים, שָׁרִים וְלוֹמְדִים תּוֹרָה.
🤔 לָמָּה? יוֹם הַכִּפּוּרִים הוּא יוֹם שֶׁל תַּעֲנִית וְשֶׁל חַג. תִּשְׁעָה בְּאָב הוּא יוֹם שֶׁל תַּעֲנִית וְשֶׁל אֵבֶל.
💛 בְּיוֹם הַכִּפּוּרִים אָנוּ שְׂמֵחִים בַּכַּפָּרָה, וּבְתִשְׁעָה בְּאָב – מִתְגַּעְגְּעִים לְבֵית אָבִינוּ. שְׁנֵי הַיָּמִים מְקָרְבִים אוֹתָנוּ לַה׳.
(פֶּרֶק י׳, הֲלָכָה ב׳)

3חוֹלִים

חוֹלֶה פָּטוּר לְגַמְרֵי מִצּוֹם תִּשְׁעָה בְּאָב. הוּא אֵינוֹ צָרִיךְ לֶאֱכֹל ״פָּחוֹת פָּחוֹת מִכַּשִּׁעוּר״ כְּמוֹ בְּיוֹם הַכִּפּוּרִים, וְגַם אֵינוֹ צָרִיךְ לָצוּם כַּמָּה שָׁעוֹת. יֵשׁ אוֹמְרִים שֶׁאִם אֶפְשָׁר, נָכוֹן שֶׁהַחוֹלֶה לֹא יֹאכַל בַּלַּיְלָה, כְּדֵי לְהִשְׁתַּתֵּף בְּצַעַר הַצִּבּוּר, וּבַבֹּקֶר יֹאכַל כְּפִי צָרְכּוֹ. הוּא אוֹכֵל דְּבָרִים פְּשׁוּטִים, וְלֹא מַעֲדַנִּים וּפִנּוּקִים.
🤔 לָמָּה? יוֹם הַכִּפּוּרִים הוּא מִן הַתּוֹרָה, וְלָכֵן גַּם חוֹלִים חַיָּבִים בּוֹ (אִם אֵין סַכָּנָה). תִּשְׁעָה בְּאָב הוּא מִדִּבְרֵי חֲכָמִים, וַחֲכָמִים לֹא גָּזְרוּ עַל הַחוֹלִים.
(פֶּרֶק י׳, הֲלָכָה ג׳)

4בְּלִי רְחִיצָה

בְּתִשְׁעָה בְּאָב אָסוּר לִרְחֹץ לְתַעֲנוּג – לֹא בְּמַיִם חַמִּים וְלֹא בְּקָרִים, וַאֲפִלּוּ לְהַכְנִיס אֶצְבַּע קְטַנָּה לַמַּיִם. אֲבָל מִי שֶׁהִתְלַכְלֵךְ בְּבֹץ, אוֹ שֶׁיָּצָא לוֹ דָּם מִן הָאַף – רוֹחֵץ אֶת הַמָּקוֹם הַמְלֻכְלָךְ. מִי שֶׁהִזִּיעַ מְאוֹד וְסוֹבֵל – רוֹחֵץ אֶת מְקוֹם הַזֵּעָה. מִי שֶׁמַּכִּין אֹכֶל לִילָדִים אוֹ לְסוֹף הַצּוֹם – רַשַּׁאי לִשְׁטֹף מַאֲכָלִים וְכֵלִים. לִשְׁטֹף אֶת הַפֶּה – אָסוּר. רַק מִי שֶׁיִּצְטַעֵר מְאוֹד – שׁוֹטֵף, וּמְצַחְצֵחַ שִׁנַּיִם בְּלִי מִשְׁחָה.
🤔 לָמָּה? אָסוּר לִרְחֹץ לְתַעֲנוּג. מִי שֶׁרוֹחֵץ כְּדֵי לְהָסִיר לִכְלוּךְ – אֵינוֹ מִתְכַּוֵּן לְתַעֲנוּג.
💛 בַּיּוֹם הַזֶּה אָנוּ מְוַתְּרִים עַל הַנְּעִימוּת שֶׁל הַגּוּף, כְּדֵי שֶׁהַלֵּב יַרְגִּישׁ אֶת הַצַּעַר עַל בֵּית ה׳.
(פֶּרֶק י׳, הֲלָכָה ה׳)

5נְטִילַת יָדַיִם

בַּבֹּקֶר נוֹטְלִים יָדַיִם שָׁלֹשׁ פְּעָמִים לְסֵרוּגִין – אֲבָל רַק עַד סוֹף קִשְׁרֵי הָאֶצְבָּעוֹת, וְלֹא אֶת כָּל כַּף הַיָּד. כָּךְ גַּם אַחֲרֵי הַשֵּׁרוּתִים. לִפְנֵי שְׁאָר הַתְּפִלּוֹת לֹא נוֹטְלִים יָדַיִם, כִּי אֵין בָּזֶה חִיּוּב. וּמִי שֶׁנָּגַע בִּמְקוֹמוֹת מְכֻסִּים בַּגּוּף וְרוֹצֶה לוֹמַר דְּבָרִים שֶׁבִּקְדֻשָּׁה – נוֹטֵל יָדַיִם. הַכֹּהֲנִים נוֹטְלִים אֶת כָּל כַּף הַיָּד לִפְנֵי בִּרְכַּת כֹּהֲנִים.
🤔 לָמָּה? נְטִילָה שֶׁל מִצְוָה אֵינָהּ רְחִיצָה שֶׁל תַּעֲנוּג. וְעַד סוֹף קִשְׁרֵי הָאֶצְבָּעוֹת – זֶה מַסְפִּיק מִן הַדִּין.
(פֶּרֶק י׳, הֲלָכָה ו׳)

6סִיכָה וּבְשָׂמִים

אָסוּר לָסוּךְ אֲפִלּוּ מָקוֹם קָטָן בַּגּוּף בְּשֶׁמֶן אוֹ בִּקְרֵם לְתַעֲנוּג אוֹ לְרֵיחַ טוֹב. אֲבָל לִרְפוּאָה – מֻתָּר: וָזֶלִין לִשְׂפָתַיִם סְדוּקוֹת, מִשְׁחָה נֶגֶד גֵּרוּד, וּמִשְׁחָה נֶגֶד יַתּוּשִׁים. וְלֹא מְרִיחִים בְּשָׂמִים בְּתִשְׁעָה בְּאָב.
🤔 לָמָּה? בַּיּוֹם שֶׁבּוֹ נֶחְרַב בֵּית מִקְדָּשֵׁנוּ, אָדָם מְמַעֵט בְּתַעֲנוּגוֹת. יֵשׁ שֶׁהֵקֵלּוּ בִּבְשָׂמִים, כִּי הֵם לֹא אֶחָד מֵחֲמֵשֶׁת הָעִנּוּיִים – אֲבָל רֹב הַפּוֹסְקִים מַחְמִירִים.
(פֶּרֶק י׳, הֲלָכָה ז׳)

7נַעֲלַיִם בְּתִשְׁעָה בְּאָב

בְּתִשְׁעָה בְּאָב אָסוּר לִנְעֹל נַעֲלֵי עוֹר. נַעֲלֵי בַּיִת פְּשׁוּטוֹת מִבַּד אוֹ מִגּוּמִי, שֶׁלֹּא הוֹלְכִים בָּהֶן בַּחוּץ – מֻתָּרוֹת לְכָל הַדֵּעוֹת. וּמָה עִם נַעֲלַיִם וְסַנְדָּלִים טוֹבִים שֶׁאֵינָם מֵעוֹר, כְּמוֹ נַעֲלֵי סְפּוֹרְט?
🤔 לָמָּה? חוֹלֶה שֶׁעָלוּל לְהִצְטַנֵּן, מִי שֶׁהוֹלֵךְ בְּמָקוֹם שֶׁיֵּשׁ בּוֹ עַקְרַבִּים אוֹ בֹּץ, וְחַיָּל בְּשֵׁרוּת מִבְצָעִי – רַשָּׁאִים לִנְעֹל נַעֲלַיִם. הָאִסּוּר הוּא לִנְעֹל כְּדֵי לָלֶכֶת בְּנוֹחוּת, וְלֹא כְּשֶׁיֵּשׁ סִבָּה אַחֶרֶת.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְפִי רֹב הַפּוֹסְקִים מֻתָּר, כִּי הָאִסּוּר הוּא רַק עַל נַעֲלֵי עוֹר.
לְפִי הָרַב מֶלָמֵד הַיּוֹם, כְּשֶׁעוֹשִׂים נַעֲלַיִם טוֹבוֹת מִכָּל חֹמֶר, כָּל נַעַל שֶׁהוֹלְכִים בָּהּ בַּחוּץ עַל אֲבָנִים – אֲסוּרָה. אֲבָל אֵין לִמְחוֹת בְּיַד מִי שֶׁסּוֹמֵךְ עַל הַמַּתִּירִים.
💛 כְּשֶׁמַּרְגִּישִׁים אֶת הָאֲבָנִים מִתַּחַת לָרַגְלַיִם, זוֹכְרִים אֶת אַבְנֵי יְרוּשָׁלַיִם הַחֲרֵבָה – וּמִתְפַּלְּלִים שֶׁתִּבָּנֶה.
(פֶּרֶק י׳, הֲלָכָה ח׳)

8מָה לוֹמְדִים בְּתִשְׁעָה בְּאָב?

כְּמוֹ שֶׁאָבֵל אֵינוֹ לוֹמֵד תּוֹרָה, כָּךְ אָסוּר לִלְמֹד תּוֹרָה בְּתִשְׁעָה בְּאָב – אֲפִלּוּ בְּמַחְשָׁבָה. אֲבָל מִצְוָה לִלְמֹד דְּבָרִים עֲצוּבִים: מְגִלַּת אֵיכָה, פִּרְקֵי הַחֻרְבָּן בְּסֵפֶר מְלָכִים, נְבוּאוֹת הַחֻרְבָּן בְּיִרְמְיָהוּ, סֵפֶר אִיּוֹב (בְּלִי הַסּוֹף), הַתּוֹכָחוֹת שֶׁבַּתּוֹרָה, אַגָּדוֹת הַחֻרְבָּן בְּמַסֶּכֶת גִּטִּין, הִלְכוֹת תִּשְׁעָה בְּאָב וְהִלְכוֹת אֲבֵלוּת, וְסִפְרֵי מוּסָר. לוֹמְדִים בְּפַשְׁטוּת, לֹא בְּעִיּוּן. אָסוּר לִקְרֹא סְפָרִים מְרַתְּקִים אוֹ עִתּוֹנִים, אֲבָל רָאוּי לִקְרֹא סִפְרֵי הִיסְטוֹרְיָה עַל הַחֻרְבָּן וְעַל צָרוֹת יִשְׂרָאֵל.
🤔 לָמָּה? ״פִּקּוּדֵי ה׳ יְשָׁרִים מְשַׂמְּחֵי לֵב״ – דִּבְרֵי תּוֹרָה מְשַׂמְּחִים, וּבְיוֹם אֵבֶל לֹא לוֹמְדִים אוֹתָם. וְכָךְ לֹא מַסִּיחִים אֶת הַדַּעַת מִן הָאֵבֶל.
💛 דַּוְקָא בַּיּוֹם שֶׁאָסוּר לִלְמֹד, מַרְגִּישִׁים כַּמָּה הַתּוֹרָה מְשַׂמַּחַת אוֹתָנוּ – וְכַמָּה אָנוּ מְחַכִּים לַחֲזֹר אֵלֶיהָ.
(פֶּרֶק י׳, הֲלָכָה י׳)

9בְּלִי ״שָׁלוֹם״

כְּמוֹ אָבֵל, גַּם בְּתִשְׁעָה בְּאָב לֹא אוֹמְרִים ״שָׁלוֹם״ אִישׁ לְרֵעֵהוּ. מִי שֶׁאֵינוֹ יוֹדֵעַ אֶת הַהֲלָכָה וְאוֹמֵר לָנוּ שָׁלוֹם – עוֹנִים לוֹ בְּשָׂפָה רָפָה וּבְכֹבֶד רֹאשׁ, וְנִזְהָרִים מְאוֹד שֶׁלֹּא לִפְגֹּעַ בּוֹ. מֻתָּר לְבָרֵךְ ״מַזָּל טוֹב״ וְלִלְחֹץ יָד. לֹא שׁוֹלְחִים מַתָּנוֹת – אֲבָל צְדָקָה מֻתֶּרֶת וַאֲפִלּוּ מִצְוָה, כְּמוֹ לִשְׁלֹחַ אֹכֶל לַעֲנִיִּים לְסוֹף הַצּוֹם. וּמָה עִם ״בֹּקֶר טוֹב״?
🤔 לָמָּה? בְּיוֹם שֶׁאָנוּ מִתְאַבְּלִים עַל בֵּית הַמִּקְדָּשׁ – אֵינֶנּוּ שְׁרוּיִים בְּשָׁלוֹם.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת רֹב הַפּוֹסְקִים גַּם ״בֹּקֶר טוֹב״ וְ״עֶרֶב טוֹב״ אָסוּר.
וְיֵשׁ אוֹמְרִים רַק ״שָׁלוֹם״ אָסוּר, וּ״בֹקֶר טוֹב״ מֻתָּר. לָכֵן כְּשֶׁצָּרִיךְ לַעֲנוֹת כְּדֵי לֹא לִפְגֹּעַ – עָדִיף לוֹמַר ״בֹּקֶר טוֹב״ מֵאֲשֶׁר ״שָׁלוֹם״.
💛 גַּם כְּשֶׁאָנוּ לֹא אוֹמְרִים ״שָׁלוֹם״, אָנוּ נִזְהָרִים מְאוֹד בִּכְבוֹד הֶחָבֵר. הָאֵבֶל עַל הַמִּקְדָּשׁ אֵינוֹ בָּא עַל חֶשְׁבּוֹן אַהֲבַת יִשְׂרָאֵל.
(פֶּרֶק י׳, הֲלָכָה י״א)

10לֹא מְטַיְּלִים – וְהוֹלְכִים לַכֹּתֶל

בְּתִשְׁעָה בְּאָב לֹא מְטַיְּלִים, וְלֹא מְשׂוֹחֲחִים שִׂיחוֹת רֵעִים, כְּדֵי שֶׁלֹּא לְהַגִּיעַ לִצְחוֹק וּלְקַלּוּת רֹאשׁ. טוֹב לְשׂוֹחֵחַ עַל הַחֻרְבָּן וְעַל תְּשׁוּבָה. אֲבָל לָלֶכֶת לַכֹּתֶל הַמַּעֲרָבִי וּלְהִתְפַּלֵּל עַל בִּנְיַן בֵּית הַמִּקְדָּשׁ – זֶה תִּקּוּן גָּדוֹל לְתִשְׁעָה בְּאָב. כְּשֶׁפּוֹגְשִׁים שָׁם חֲבֵרִים – לֹא אוֹמְרִים שָׁלוֹם, אֶפְשָׁר לִלְחֹץ יָד בְּאַהֲבָה, וּמִתְפַּלְּלִים יַחַד.
🤔 לָמָּה? הָרַב מֶלָמֵד מֵבִיא מֵאָבִיו: אֵין תִּקּוּן גָּדוֹל יוֹתֵר לְתִשְׁעָה בְּאָב מֵאֲשֶׁר לָלֶכֶת לַכֹּתֶל וּלְהִתְפַּלֵּל. וּכְשֶׁבָּאִים רַבִּים – כֹּחַ הַתְּפִלָּה גָּדֵל.
💛 הָאֵבֶל עַל הַמִּקְדָּשׁ אֵינוֹ יֵאוּשׁ. בַּכֹּתֶל אָנוּ עוֹמְדִים מוּל מְקוֹם הַמִּקְדָּשׁ וּמְבַקְּשִׁים שֶׁיִּבָּנֶה בִּמְהֵרָה בְּיָמֵינוּ.
(פֶּרֶק י׳, הֲלָכָה י״ב)

11יוֹשְׁבִים וִישֵׁנִים נָמוּךְ

נוֹהֲגִים לָשֶׁבֶת עַל הָאָרֶץ כְּמוֹ אֲבֵלִים. מֻתָּר לָשֶׁבֶת עַל כָּרִית נְמוּכָה, וִישִׁיבָה עַל מַדְרֵגוֹת נֶחְשֶׁבֶת כִּישִׁיבָה עַל הַקַּרְקַע. מִי שֶׁקָּשֶׁה לוֹ – יָשֵׁב עַל כִּסֵּא נָמוּךְ, וְזָקֵן, חוֹלֶה אוֹ מִי שֶׁכּוֹאֵב לוֹ הַגַּב – עַל כִּסֵּא רָגִיל. גַּם בַּשֵּׁינָה יֵשׁ מִנְהֲגֵי אֵבֶל: רַבִּים מוֹרִידִים אֶת הַמִּזְרָן לָרִצְפָּה, וּמִי שֶׁקָּשֶׁה לוֹ – יָשֵׁן כָּרָגִיל. עַד מָתַי יוֹשְׁבִים עַל הָאָרֶץ?
🤔 לָמָּה? מִן הַדִּין אֵין חוֹבָה לָשֶׁבֶת וְלִישֹׁן עַל הָאָרֶץ – זֶה מִנְהָג, וְלָכֵן דִּינָיו קַלִּים יוֹתֵר. יֵשׁ נוֹהֲגִים, עַל פִּי הַקַּבָּלָה, לְהַנִּיחַ בַּד אוֹ עֵץ בֵּינָם לְבֵין הָאֲדָמָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז וְחֵלֶק מִיּוֹצְאֵי סְפָרַד עַד חֲצוֹת הַיּוֹם.
יֵשׁ מִיּוֹצְאֵי סְפָרַד שֶׁמַּחְמִירִים עַד תְּפִלַּת מִנְחָה.
💛 כְּשֶׁיּוֹשְׁבִים נָמוּךְ, הַלֵּב מַרְגִּישׁ שָׁפָל וְעָצוּב – וּמִתּוֹךְ זֶה אָנוּ מִתְפַּלְּלִים שֶׁה׳ יְרוֹמֵם אֶת יְרוּשָׁלַיִם.
(פֶּרֶק י׳, הֲלָכָה י״ג)

12מְלָאכָה

נָהֲגוּ כָּל יִשְׂרָאֵל שֶׁלֹּא לַעֲשׂוֹת מְלָאכָה בְּתִשְׁעָה בְּאָב עַד חֲצוֹת הַיּוֹם, וְגַם אַחֲרֵי חֲצוֹת נָכוֹן לְהִמָּנַע מִמֶּנָּה. אֲסוּרוֹת מְלָאכוֹת שֶׁלּוֹקְחוֹת זְמַן וְטוֹרְדוֹת אֶת הַמַּחְשָׁבָה: תְּפִירָה, תִּקּוּן בְּגָדִים, רָהִיטִים וּמַכְשִׁירִים, וּכְתִיבָה (אֲבָל מֻתָּר לְהַעְתִּיק דְּבָרִים עַל תִּשְׁעָה בְּאָב). מֻתָּרוֹת מְלָאכוֹת קְצָרוֹת: הַדְלָקָה וְכִבּוּי שֶׁל אוֹר, קְשִׁירָה וְהַתָּרָה. אַחֲרֵי חֲצוֹת מֻתָּר לְהָכִין אֶת הָאֲרוּחָה שֶׁל סוֹף הַצּוֹם.
🤔 לָמָּה? חֲכָמִים אָמְרוּ: ״כָּל הָעוֹשֶׂה מְלָאכָה בְּתִשְׁעָה בְּאָב אֵינוֹ רוֹאֶה מִמֶּנָּה סִימַן בְּרָכָה לְעוֹלָם״, כִּי הוּא מַסִּיחַ אֶת דַּעְתּוֹ מִן הָאֵבֶל. יֵשׁ נָשִׁים שֶׁנּוֹהֲגוֹת לְנַקּוֹת אֶת הַבַּיִת אַחֲרֵי חֲצוֹת, מִתּוֹךְ צִפִּיָּה לַמָּשִׁיחַ, שֶׁנּוֹלַד בְּתִשְׁעָה בְּאָב.
(פֶּרֶק י׳, הֲלָכָה י״ד)

13אֵיכָה בַּחֹשֶׁךְ

בְּלֵיל תִּשְׁעָה בְּאָב, אַחֲרֵי עַרְבִית, קוֹרְאִים מְגִלַּת אֵיכָה. מַחְשִׁיכִים אֶת בֵּית הַכְּנֶסֶת וּמַשְׁאִירִים רַק מְעַט אוֹר, וְגַם בַּבַּיִת רָאוּי לְמַעֵט אֶת הָאוֹרוֹת. עִקַּר הַקְּרִיאָה בַּלַּיְלָה – ״בָּכוֹ תִבְכֶּה בַּלַּיְלָה״, וְרַבִּים קוֹרְאִים אוֹתָהּ שׁוּב בַּיּוֹם, אַחֲרֵי הַקִּינוֹת. הַאִם מְבָרְכִים עַל קְרִיאַת אֵיכָה?
🤔 לָמָּה? ״בְּמַחֲשַׁכִּים הוֹשִׁיבַנִי״. חֲכָמִים סִפְּרוּ: כְּשֶׁנֶּחְרַב הַבַּיִת שָׁאַל ה׳ אֶת מַלְאֲכֵי הַשָּׁרֵת: ״מֶלֶךְ בָּשָׂר וָדָם כְּשֶׁהוּא אָבֵל, מָה הוּא עוֹשֶׂה?״ אָמְרוּ לוֹ: ״מְכַבֶּה אֶת הַפָּנָסִים שֶׁלּוֹ״. אָמַר: ״אֲנִי אֶעֱשֶׂה כֵּן״.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי סְפָרַד וְרַבִּים מִיּוֹצְאֵי אַשְׁכְּנַז (וּבָהֶם כָּל הַחֲסִידִים) קוֹרְאִים בְּלִי בְּרָכָה.
לְמִנְהַג חֵלֶק מִיּוֹצְאֵי אַשְׁכְּנַז (בְּמִיֻחָד הַנּוֹהֲגִים כַּגְּרָ״א) קוֹרְאִים מִתּוֹךְ מְגִלַּת קְלָף כְּשֵׁרָה, בִּבְרָכָה.
💛 גַּם ה׳ כִּבְיָכוֹל מִתְאַבֵּל אִתָּנוּ – וְהוּא גַּם יְנַחֵם אוֹתָנוּ.
(פֶּרֶק י׳, הֲלָכָה ט״ו)

14עֲנֵנוּ, נַחֵם וְתַחֲנוּן

בְּתִשְׁעָה בְּאָב מִתְפַּלְּלִים כְּמוֹ אֲבֵלִים: לְאַט, בְּקוֹל חַלָּשׁ וּבְלִי חַזָּנוּת. לֹא אוֹמְרִים תַּחֲנוּן – וְגַם לֹא בְּמִנְחָה שֶׁל עֶרֶב תִּשְׁעָה בְּאָב. מוֹסִיפִים ״נַחֵם״ בְּבִרְכַּת ״בּוֹנֵה יְרוּשָׁלַיִם״. וּבְאֵילוּ תְּפִלּוֹת לַחַשׁ אוֹמְרִים ״עֲנֵנוּ״?
🤔 לָמָּה? לֹא אוֹמְרִים תַּחֲנוּן, כִּי תִּשְׁעָה בְּאָב נִקְרָא ״מוֹעֵד״: ״קָרָא עָלַי מוֹעֵד לִשְׁבֹּר בַּחוּרָי״. מוֹעֵד הוּא זְמַן מְיֻחָד: לֹא זָכִינוּ – נַעֲשָׂה יוֹם אֵבֶל; כְּשֶׁנִּזְכֶּה – יֵהָפֵךְ לְיוֹם טוֹב.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי סְפָרַד בְּכָל שָׁלֹשׁ הַתְּפִלּוֹת שֶׁל הַצּוֹם – עַרְבִית, שַׁחֲרִית וּמִנְחָה.
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז בְּמִנְחָה בִּלְבַד. וְגַם ״נַחֵם״ – בְּמִנְחָה בִּלְבַד.
💛 ״מְנַחֵם צִיּוֹן וּבוֹנֵה יְרוּשָׁלַיִם״ – כְּבָר בְּתוֹךְ הַיּוֹם הֶעָצוּב אָנוּ מְבָרְכִים אֶת ה׳ עַל הַנֶּחָמָה שֶׁתָּבוֹא.
(פֶּרֶק י׳, הֲלָכָה ט״ז)

15פָּרֹכֶת, טַלִּית וּתְפִלִּין

לִפְנֵי עַרְבִית מְסִירִים אֶת הַפָּרֹכֶת מִן הָאָרוֹן, וּמַחֲזִירִים אוֹתָהּ לִפְנֵי מִנְחָה. בְּשַׁחֲרִית רֹב הַקְּהִלּוֹת – כָּל קְהִלּוֹת אַשְׁכְּנַז וְהַרְבֵּה קְהִלּוֹת סְפָרַדִּיּוֹת – אֵינָן מִתְעַטְּפוֹת בְּטַלִּית וְאֵינָן מַנִּיחוֹת תְּפִלִּין, אֶלָּא בְּמִנְחָה. וְיֵשׁ קְהִלּוֹת סְפָרַדִּיּוֹת שֶׁמִּתְעַטְּפִים וּמַנִּיחִים בְּשַׁחֲרִית – וְכָל קְהִלָּה מַמְשִׁיכָה בְּמִנְהָגָהּ. גַּם בְּבִרְכַּת כֹּהֲנִים יֵשׁ מִנְהָגִים: לְמִנְהַג אַשְׁכְּנַז וְחֵלֶק מִן הַסְּפָרַדִּים הַכֹּהֲנִים לֹא מְבָרְכִים בְּשַׁחֲרִית, וּלְמִנְהַג חֵלֶק מִן הַסְּפָרַדִּים – מְבָרְכִים. בְּמִנְחָה – כֻּלָּם מְבָרְכִים.
🤔 לָמָּה? ״בִּצַּע אֶמְרָתוֹ״ – חֲזַ״ל אָמְרוּ שֶׁכִּבְיָכוֹל ה׳ קָרַע אֶת גְּלִימָתוֹ, וְ״הִשְׁלִיךְ מִשָּׁמַיִם אֶרֶץ תִּפְאֶרֶת יִשְׂרָאֵל״ – אֵלּוּ הַתְּפִלִּין כִּבְיָכוֹל. בְּשַׁחֲרִית, כְּשֶׁאוֹמְרִים קִינוֹת, הָאֵבֶל בְּשִׂיאוֹ. בְּמִנְחָה כְּבָר מַתְחִילָה קְצָת נֶחָמָה.
(פֶּרֶק י׳, הֲלָכָה ט״ז, הֲלָכָה י״ז)

16״שֶׁעָשָׂה לִי כָּל צָרְכִּי״

בְּבִרְכוֹת הַשַּׁחַר יֵשׁ בְּרָכָה ״שֶׁעָשָׂה לִי כָּל צָרְכִּי״ – הוֹדָאָה עַל הַנַּעֲלַיִם. הַאִם אוֹמְרִים אוֹתָהּ בְּתִשְׁעָה בְּאָב, כְּשֶׁאֵין נוֹעֲלִים נַעֲלַיִם?
🤔 לָמָּה? פִּרְקֵי הַתְּפִלָּה – גַּם אֵלֶּה שֶׁיֵּשׁ בָּהֶם פְּסוּקִים וּמִשְׁנָיוֹת – רַבִּים אוֹמְרִים כָּרָגִיל, כִּי הֵם חֵלֶק מִן הַתְּפִלָּה. אֶת הַתְּהִלִּים הַיּוֹמִי עָדִיף לֹא לוֹמַר. וּקְרִיאַת שְׁמַע שֶׁעַל הַמִּטָּה אוֹמְרִים כֻּלָּהּ.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז וּמִקְצָת מִיּוֹצְאֵי סְפָרַד אוֹמְרִים, כִּי זוֹ הוֹדָאָה כְּלָלִית עַל מִנְהָגוֹ שֶׁל עוֹלָם, וְלֹא עַל הַנַּעֲלַיִם שֶׁל הַיּוֹם.
לְמִנְהַג רֹב יוֹצְאֵי סְפָרַד (לְפִי הָאֲרִ״י) לֹא אוֹמְרִים אוֹתָהּ בְּתִשְׁעָה בְּאָב וּבְיוֹם הַכִּפּוּרִים.
(פֶּרֶק י׳, הֲלָכָה י״ח)

17עֲשִׂירִי בְּאָב

בְּתִשְׁעָה בְּאָב לִפְנוֹת עֶרֶב הִצִּיתוּ הַבַּבְלִים אֶת בֵּית הַמִּקְדָּשׁ, וְהוּא נִשְׂרַף כָּל יוֹם י׳ בְּאָב. לָכֵן נָהֲגוּ לֹא לֶאֱכֹל בָּשָׂר וְלֹא לִשְׁתּוֹת יַיִן גַּם בַּעֲשִׂירִי בְּאָב. לְדַעַת רֹב הָאַחֲרוֹנִים, עַד חֲצוֹת גַּם לֹא מְכַבְּסִים, לֹא מִסְתַּפְּרִים, לֹא שׁוֹמְעִים מוּזִיקָה שְׂמֵחָה וְלֹא מִתְרַחֲצִים בְּחַמִּים, וְלֹא מְבָרְכִים ״שֶׁהֶחֱיָנוּ״. וְעַד מָתַי לֹא אוֹכְלִים בָּשָׂר?
🤔 לָמָּה? רַבִּי יוֹחָנָן אָמַר שֶׁהָיָה קוֹבֵעַ אֶת הַצּוֹם בַּעֲשִׂירִי, שֶׁבּוֹ נִשְׂרַף רֹב הַהֵיכָל. אֲבָל חֲכָמִים קָבְעוּ אוֹתוֹ בַּתְּשִׁיעִי, כִּי הַכֹּל הוֹלֵךְ אַחַר הַהַתְחָלָה. וְאֶת בִּרְכַּת הַלְּבָנָה דּוֹחִים לְאַחַר הַצּוֹם – נָכוֹן לְבָרֵךְ שָׁעָה אוֹ שְׁעָתַיִם אַחֲרָיו, אַחֲרֵי שֶׁאָכְלוּ וְהִתְרַחֲצוּ מְעַט, כְּדֵי לְבָרֵךְ בְּשִׂמְחָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז וְחֵלֶק מִיּוֹצְאֵי סְפָרַד עַד חֲצוֹת הַיּוֹם.
לְמִנְהַג חֵלֶק מִיּוֹצְאֵי סְפָרַד כָּל הַיּוֹם.
(פֶּרֶק י׳, הֲלָכָה י״ט)

18כְּשֶׁתִּשְׁעָה בְּאָב חָל בְּשַׁבָּת

כְּשֶׁתִּשְׁעָה בְּאָב חָל בְּשַׁבָּת, הַצּוֹם נִדְחֶה לְיוֹם רִאשׁוֹן. בַּשַּׁבָּת אֵין שׁוּם סִימַן אֲבֵלוּת – אוֹכְלִים וְשׁוֹתִים כְּבְכָל שַׁבָּת. וּבְמוֹצָאֵי הַצּוֹם הַנִּדְחֶה, מֻתָּר מִיָּד לְהִסְתַּפֵּר, לְכַבֵּס וּלְהִתְרַחֵץ בְּחַמִּים. וּמָה עִם בָּשָׂר וְיַיִן בְּאוֹתוֹ עֶרֶב?
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת רַבִּים נִמְנָעִים מִבָּשָׂר וְיַיִן בְּאוֹתוֹ עֶרֶב, כִּי לֹא רָאוּי לִשְׂמֹחַ בְּבָשָׂר וְיַיִן מִיָּד אַחֲרֵי הַצּוֹם.
וְיֵשׁ מְקִלִּים לֶאֱכֹל בָּשָׂר וְלִשְׁתּוֹת יַיִן מִיָּד אַחֲרֵי הַצּוֹם.
(פֶּרֶק י׳, הֲלָכָה כ׳)

19יְלָדִים בְּתִשְׁעָה בְּאָב

יְלָדִים אֵינָם צָמִים כָּל הַיּוֹם. מִגִּיל תֵּשַׁע נוֹהֲגִים לְחַנְּכָם לָצוּם כַּמָּה שָׁעוֹת לְפִי כֹּחָם, וְרַבִּים מְהַדְּרִים שֶׁיְּלָדִים מִגִּיל שֵׁשׁ לֹא יֹאכְלוּ וְלֹא יִשְׁתּוּ בַּלַּיְלָה. כְּשֶׁיְּלָדִים אוֹכְלִים – נוֹתְנִים לָהֶם מַאֲכָלִים פְּשׁוּטִים. מִגִּיל שֵׁשׁ בְּעֵרֶךְ, כְּשֶׁהַיֶּלֶד מַתְחִיל לְהָבִין מַהוּ הַחֻרְבָּן, מְחַנְּכִים אוֹתוֹ שֶׁלֹּא לִנְעֹל נַעֲלַיִם שֶׁל חוּץ, וְשֶׁלֹּא לִרְחֹץ וְלָסוּךְ לְתַעֲנוּג – וְיֵשׁ שֶׁמַּקְפִּידִים עַל הַנַּעֲלַיִם כְּבָר מִגִּיל שָׁלֹשׁ. יֶלֶד רַשַּׁאי לִלְמֹד בְּעַצְמוֹ אֶת מָה שֶׁמֻּתָּר לַגְּדוֹלִים.
🤔 לָמָּה? מִצְוָה לְחַנֵּךְ אֶת הַיְּלָדִים לְמִצְווֹת תִּשְׁעָה בְּאָב וְלָאֵבֶל עַל הַחֻרְבָּן. אֲבָל יְלָדִים חַלָּשִׁים מִלָּצוּם יוֹם שָׁלֵם, וְלָכֵן מְחַנְּכִים אוֹתָם לְפִי כֹּחָם. נֶחְלְקוּ הַפּוֹסְקִים אִם מֻתָּר לִגְדוֹלִים לְלַמֵּד יְלָדִים דְּבָרִים עַל הַחֻרְבָּן, אוֹ רַק לְסַפֵּר לָהֶם – וְכָל אֶחָד רַשַּׁאי לִבְחֹר.
💛 גַּם הַיְּלָדִים שֻׁתָּפִים בָּאֵבֶל עַל יְרוּשָׁלַיִם – וְהֵם יִהְיוּ שֻׁתָּפִים גַּם בַּשִּׂמְחָה הַגְּדוֹלָה כְּשֶׁתִּבָּנֶה.
(פֶּרֶק י׳, הֲלָכָה כ״א)

📝 מָה לָמַדְנוּ

  • תִּשְׁעָה בְּאָב – צוֹם שֶׁל יְמָמָה, עִם חֲמִשָּׁה עִנּוּיִים, וְרַק חוֹלִים פְּטוּרִים.
  • הוּא יוֹם אֵבֶל: יוֹשְׁבִים נָמוּךְ, לֹא אוֹמְרִים שָׁלוֹם, לוֹמְדִים רַק דְּבָרִים עֲצוּבִים, וְלֹא עוֹשִׂים מְלָאכָה עַד חֲצוֹת.
  • נַעֲלֵי עוֹר אֲסוּרוֹת; בְּנַעֲלַיִם שֶׁאֵינָן מֵעוֹר רֹב הַפּוֹסְקִים מַתִּירִים, וְהָרַב מֶלָמֵד אוֹסֵר נַעֲלַיִם שֶׁל חוּץ.
  • קוֹרְאִים אֵיכָה בַּחֹשֶׁךְ, בְּשַׁחֲרִית אוֹמְרִים קִינוֹת, וּבְמִנְחָה מַנִּיחִים טַלִּית וּתְפִלִּין וְאוֹמְרִים ״נַחֵם״.
  • יְלָדִים צָמִים לְפִי כֹּחָם, וּמִגִּיל שֵׁשׁ מִתְחַנְּכִים לֹא לִנְעֹל נַעֲלַיִם שֶׁל חוּץ וְלֹא לִרְחֹץ לְתַעֲנוּג.

❓ שְׁאֵלוֹת חֲזָרָה

  1. בְּמָה תִּשְׁעָה בְּאָב קַל מִיּוֹם הַכִּפּוּרִים, וּבְמָה הוּא חָמוּר מִמֶּנּוּ?
  2. נָפַלְתָּ בַּבֹּץ בְּתִשְׁעָה בְּאָב, וּפָגַשְׁתָּ חָבֵר שֶׁאָמַר לְךָ ״שָׁלוֹם״. מָה תַּעֲשֶׂה בְּכָל מִקְרֶה, וְלָמָּה?
  3. לָמָּה לֹא אוֹמְרִים תַּחֲנוּן בְּתִשְׁעָה בְּאָב, וּמָה זֶה מְלַמֵּד אוֹתָנוּ עַל הָעָתִיד?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״עֶרֶב תִּשְׁעָה בְּאָב וְתִשְׁעָה בְּאָב״

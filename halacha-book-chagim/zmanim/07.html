סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ חֹדֶשׁ, סְפִירַת הָעֹמֶר וְהַצּוֹמוֹת
מַהֲדוּרַת יְלָדִים

⏰ דִּינֵי הַצּוֹמוֹת הַקַּלִּים

פֶּרֶק ז׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
בַּבֹּקֶר שֶׁל צוֹם הַבַּיִת נִרְאֶה קְצָת אַחֶרֶת: אַבָּא לֹא שׁוֹתֶה קָפֶה, אֵין אֲרוּחַת בֹּקֶר גְּדוֹלָה, וְכֻלָּם מְדַבְּרִים עַל ״עֲנֵנוּ״. מָתַי מַתְחִיל הַצּוֹם וּמָתַי הוּא נִגְמָר? מָה אָסוּר בּוֹ? וּמָה עוֹשִׂים יְלָדִים, שֶׁעוֹד לֹא חַיָּבִים לָצוּם? בַּפֶּרֶק הַזֶּה נִלְמַד אֶת דִּינֵי שְׁלֹשֶׁת הַצּוֹמוֹת הַקַּלִּים: עֲשָׂרָה בְּטֵבֵת, י״ז בְּתַמּוּז וְצוֹם גְּדַלְיָה.

1צָמִים עַד שֶׁיִּבָּנֶה הַמִּקְדָּשׁ

דִּין הַצּוֹמוֹת תָּלוּי בַּמַּצָּב שֶׁל עַם יִשְׂרָאֵל. כְּשֶׁיֵּשׁ גְּזֵרוֹת קָשׁוֹת – חַיָּבִים לָצוּם. כְּשֶׁבֵּית הַמִּקְדָּשׁ בָּנוּי – הַצּוֹמוֹת הוֹפְכִים לְיָמִים שְׂמֵחִים. וּבְמַצָּב שֶׁל ״בֵּינַיִם״, כְּשֶׁאֵין גְּזֵרוֹת אֲבָל הַמִּקְדָּשׁ עֲדַיִן חָרֵב – הַצּוֹמוֹת הַקַּלִּים תְּלוּיִים בִּרְצוֹן יִשְׂרָאֵל. וְיִשְׂרָאֵל קִבְּלוּ עַל עַצְמָם לָצוּם בָּהֶם, וְלָכֵן הַיּוֹם כָּל אֶחָד חַיָּב לָצוּם – עַד שֶׁיִּבָּנֶה בֵּית הַמִּקְדָּשׁ.
🤔 לָמָּה? בְּתִשְׁעָה בְּאָב חַיָּבִים לָצוּם בְּכָל מַצָּב, כִּי בּוֹ נִכְפְּלוּ הַצָּרוֹת – שְׁנֵי בָּתֵּי הַמִּקְדָּשׁ נֶחְרְבוּ בּוֹ.
💛 עַם יִשְׂרָאֵל בָּחַר בְּעַצְמוֹ לְהַמְשִׁיךְ לָצוּם – כִּי הוּא לֹא מוּכָן לִשְׁכֹּחַ אֶת בֵּית ה׳.
(פֶּרֶק ז׳, הֲלָכָה א׳)

2מָה הַהֶבְדֵּל בֵּין צוֹם קַל לְתִשְׁעָה בְּאָב?

תִּשְׁעָה בְּאָב הוּא צוֹם חָמוּר, כְּמוֹ יוֹם הַכִּפּוּרִים: הוּא מַתְחִיל כְּבָר בָּעֶרֶב, וְאָסוּר בּוֹ גַּם לִרְחֹץ, לִסּוֹךְ וְלִנְעֹל נַעֲלֵי עוֹר. הַצּוֹמוֹת הַקַּלִּים קַלִּים יוֹתֵר: צָמִים רַק בַּיּוֹם, וְאָסוּר רַק לֶאֱכֹל וְלִשְׁתּוֹת. רְחִיצָה וּנְעִילַת נַעֲלַיִם – מֻתָּרוֹת. בְּכָל זֹאת טוֹב לֹא לְהִתְרַחֵץ בְּמַיִם חַמִּים (לְנִקָּיוֹן מֻתָּר בְּפוֹשְׁרִים), וְנָכוֹן לֹא לְהִסְתַּפֵּר, לֹא לִשְׁמֹעַ מוּזִיקָה שְׂמֵחָה וְלֹא לִקְנוֹת דְּבָרִים מְשַׂמְּחִים בִּזְמַן הַצּוֹם.
🤔 לָמָּה? כְּשֶׁיִּשְׂרָאֵל קִבְּלוּ עַל עַצְמָם לָצוּם בַּצּוֹמוֹת הַקַּלִּים, הֵם לֹא קִבְּלוּ אֶת כָּל הַחֻמְרוֹת שֶׁל יוֹם הַכִּפּוּרִים. בְּתִשְׁעָה בְּאָב נִשְׁאַר הַדִּין כְּמוֹ שֶׁתִּקְּנוּ אוֹתוֹ הַנְּבִיאִים.
(פֶּרֶק ז׳, הֲלָכָה ב׳)

3מֵעֲלוֹת הַשַּׁחַר וְעַד צֵאת הַכּוֹכָבִים

הַצּוֹמוֹת הַקַּלִּים מַתְחִילִים בַּעֲלוֹת הַשַּׁחַר – כְּשֶׁהָאוֹר הָרִאשׁוֹן נִרְאֶה בַּמִּזְרָח, וְנִגְמָרִים בְּצֵאת הַכּוֹכָבִים – כְּשֶׁרוֹאִים שְׁלֹשָׁה כּוֹכָבִים בֵּינוֹנִיִּים בַּשָּׁמַיִם. הַזְּמַנִּים מִשְׁתַּנִּים לְפִי עוֹנַת הַשָּׁנָה וּלְפִי הַמָּקוֹם, וְלָכֵן בּוֹדְקִים בְּלוּחַ מְדֻיָּק. כְּשֶׁעֲשָׂרָה בְּטֵבֵת חָל בְּיוֹם שִׁשִּׁי – צָמִים עַד צֵאת הַכּוֹכָבִים, אַף שֶׁהַשַּׁבָּת כְּבָר נִכְנְסָה.
🤔 לָמָּה? יֵשׁ דֵּעוֹת שׁוֹנוֹת מָתַי בְּדִיּוּק עֲלוֹת הַשַּׁחַר וְצֵאת הַכּוֹכָבִים. מִן הַדִּין אֶפְשָׁר לָלֶכֶת לְפִי הַמְּקִלִּים, אֲבָל אַחֲרֵי שֶׁצָּמִים כָּל הַיּוֹם – כְּדַאי לְהוֹסִיף עוֹד כַּמָּה דַּקּוֹת וְלָצֵאת לְדַעַת כֻּלָּם.
(פֶּרֶק ז׳, הֲלָכָה ג׳)

4קַמְתִּי לִפְנֵי עֲלוֹת הַשַּׁחַר

מִי שֶׁהָלַךְ לִישֹׁן לִפְנֵי הַצּוֹם וְקָם לִפְנֵי עֲלוֹת הַשַּׁחַר – אָסוּר לוֹ לֶאֱכֹל, כִּי כְּשֶׁהָלַךְ לִישֹׁן כְּבָר הֵסִיחַ דַּעְתּוֹ מִלֶּאֱכֹל, וְזֶה כְּאִלּוּ קִבֵּל עָלָיו אֶת הַצּוֹם. אֲבָל אִם לִפְנֵי הַשֵּׁינָה הִתְנָה בְּלִבּוֹ: ״אִם אָקוּם לִפְנֵי עֲלוֹת הַשַּׁחַר – אֹכַל״, מֻתָּר לוֹ. וּמָה עִם שְׁתִיָּה, אִם לֹא הִתְנָה?
🤔 לָמָּה? לְמַעֲשֶׂה, מִי שֶׁרוֹצֶה לִשְׁתּוֹת אִם יִתְעוֹרֵר – יַתְנֶה זֹאת בְּלִבּוֹ לִפְנֵי הַשֵּׁינָה. וּמִי שֶׁהִתְעוֹרֵר צָמֵא לִפְנֵי עֲלוֹת הַשַּׁחַר וְלֹא הִתְנָה – בְּדִיעֲבַד רַשַּׁאי לִשְׁתּוֹת.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְפִי הָרַמָּ״א מֻתָּר לִשְׁתּוֹת גַּם בְּלִי תְּנַאי, כִּי רַבִּים רְגִילִים לִשְׁתּוֹת מַיִם כְּשֶׁקָּמִים מִשֵּׁינָה.
לְפִי הַשֻּׁלְחָן עָרוּךְ אֵין הֶבְדֵּל בֵּין אֲכִילָה לִשְׁתִיָּה – מֻתָּר לִשְׁתּוֹת רַק אִם הִתְנָה.
💛 הַהֲכָנָה לַצּוֹם הִיא חֵלֶק מִן הַצּוֹם. מִי שֶׁחוֹשֵׁב מֵרֹאשׁ, שׁוֹמֵר גַּם עַל הַהֲלָכָה וְגַם עַל עַצְמוֹ.
(פֶּרֶק ז׳, הֲלָכָה ד׳)

5צִחְצוּחַ שִׁנַּיִם בַּצּוֹם

מִי שֶׁחוֹשֵׁשׁ מֵרֵיחַ רַע מִן הַפֶּה – הַאִם מֻתָּר לוֹ לִשְׁטֹף אֶת הַפֶּה וּלְצַחְצֵחַ שִׁנַּיִם בְּצוֹם קַל?
🤔 לָמָּה? בְּתִשְׁעָה בְּאָב, שֶׁהוּא צוֹם חָמוּר, מַחְמִירִים יוֹתֵר: רַק מִי שֶׁיִּצְטַעֵר מְאוֹד שׁוֹטֵף אֶת פִּיו, וּמְצַחְצֵחַ בְּלִי מִשְׁחָה. וּבְיוֹם הַכִּפּוּרִים, שֶׁהַצּוֹם בּוֹ מִן הַתּוֹרָה – לֹא מְקִלִּים בָּזֶה כְּלָל.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְפִי הָרַב מֶלָמֵד מֻתָּר לִשְׁטֹף אֶת הַפֶּה וְאַף לְצַחְצֵחַ שִׁנַּיִם עִם מִשְׁחָה, כִּי אֵין כַּוָּנָה לִשְׁתּוֹת – וּמִשְׁתַּדְּלִים מְאוֹד שֶׁלֹּא לִבְלֹעַ טִפּוֹת מַיִם.
לְפִי הַמַּחְמִירִים לֹא שׁוֹטְפִים אֶת הַפֶּה בְּתַעֲנִית צִבּוּר, מֵחֲשָׁשׁ שֶׁמָּא יִבְלַע.
(פֶּרֶק ז׳, הֲלָכָה ה׳)

6שָׁכַחְתִּי וְאָכַלְתִּי

מִי שֶׁשָּׁכַח שֶׁהַיּוֹם צוֹם, וְאָכַל אוֹ שָׁתָה – חַיָּב לְהַמְשִׁיךְ לָצוּם. הוּא אֵינוֹ צָרִיךְ לָצוּם יוֹם אַחֵר בִּמְקוֹמוֹ, כִּי הַצּוֹם קָשׁוּר לַיּוֹם שֶׁבּוֹ קָרוּ הַצָּרוֹת. וְטוֹב שֶׁיּוֹסִיף צְדָקָה וְלִמּוּד תּוֹרָה. וּמָה עִם מִי שֶׁשָּׁכַח, בֵּרַךְ ״שֶׁהַכֹּל״ עַל מַיִם, וְנִזְכַּר לִפְנֵי שֶׁשָּׁתָה?
🤔 לָמָּה? מִי שֶׁעָשָׂה טָעוּת אַחַת, אֵין לוֹ הֶתֵּר לַעֲשׂוֹת עוֹד טָעֻיּוֹת. מַמְשִׁיכִים מֵאוֹתוֹ רֶגַע כָּרָאוּי.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ אוֹמְרִים יִשְׁתֶּה מְעַט, כְּדֵי שֶׁהַבְּרָכָה לֹא תִּהְיֶה לְבַטָּלָה.
וְיֵשׁ אוֹמְרִים, וְכָךְ נִרְאֶה לְמַעֲשֶׂה לֹא יִשְׁתֶּה כְּלָל, כִּי אֵין מְתַקְּנִים עֲבֵרָה אַחַת בַּעֲבֵרָה אַחֶרֶת.
💛 מִי שֶׁנִּכְשַׁל אֵינוֹ מְוַתֵּר – הוּא מַמְשִׁיךְ מִן הָרֶגַע הַזֶּה בְּכָל הַכֹּחַ, וְה׳ רוֹאֶה אֶת הָרָצוֹן הַטּוֹב.
(פֶּרֶק ז׳, הֲלָכָה ו׳)

7חוֹלֶה פָּטוּר

חוֹלֶה פָּטוּר מִן הַצּוֹמוֹת הַקַּלִּים, גַּם כְּשֶׁאֵין בּוֹ סַכָּנָה. מִי נֶחְשָׁב חוֹלֶה? מִי שֶׁבִּגְלַל הַכְּאֵבִים אוֹ הַחֻלְשָׁה צָרִיךְ לִשְׁכַּב בַּמִּטָּה – לְמָשָׁל, מִי שֶׁיֵּשׁ לוֹ שַׁפַּעַת אוֹ חֹם גָּבוֹהַּ. גַּם מִי שֶׁיּוֹדֵעַ שֶׁאִם יָצוּם, הוּא עָלוּל לַחֲלוֹת – פָּטוּר. הַחוֹלֶה אוֹכֵל וְשׁוֹתֶה כְּפִי צָרְכּוֹ, אֲבָל לֹא מַאַכְלֵי פִּנּוּק. וּמִי שֶׁצָּרִיךְ תְּרוּפָה – בּוֹלֵעַ אוֹתָהּ, וְאִם אֶפְשָׁר – בְּלִי מַיִם. בְּכָל סָפֵק שׁוֹאֲלִים רוֹפֵא יְרֵא שָׁמַיִם.
🤔 לָמָּה? כְּשֶׁתִּקְּנוּ הַנְּבִיאִים וְהַחֲכָמִים אֶת הַצּוֹמוֹת, הֵם תִּקְּנוּ אוֹתָם לַבְּרִיאִים. אֲבָל כְּאֵב רֹאשׁ וְחֻלְשָׁה שֶׁל צוֹם אֵינָם מַחֲלָה – זוֹ הַרְגָּשָׁה טִבְעִית שֶׁעוֹבֶרֶת אַחֲרֵי הַצּוֹם.
💛 הַתּוֹרָה דּוֹאֶגֶת לַחוֹלִים בְּרַחֲמִים – ה׳ רוֹצֶה שֶׁנִּשְׁמֹר עַל עַצְמֵנוּ.
(פֶּרֶק ז׳, הֲלָכָה ז׳)

8יְלָדִים בַּצּוֹם

יְלָדִים שֶׁלֹּא הִגִּיעוּ לְגִיל מִצְווֹת פְּטוּרִים מִן הַצּוֹמוֹת שֶׁתִּקְּנוּ חֲכָמִים (בַּת – מִגִּיל שְׁתֵּים עֶשְׂרֵה, וּבֵן – מִגִּיל שְׁלֹשׁ עֶשְׂרֵה). חֲכָמִים לֹא חִיְּבוּ לְחַנֵּךְ אוֹתָם לָצוּם, אֲבָל רַבִּים נוֹהֲגִים לְחַנֵּךְ אוֹתָם לָצוּם כַּמָּה שָׁעוֹת לְפִי כֹּחָם – וְלֹא כָּל הַיּוֹם. כְּשֶׁיְּלָדִים אוֹכְלִים בְּיוֹם הַצּוֹם, נוֹתְנִים לָהֶם מַאֲכָלִים פְּשׁוּטִים בִּלְבַד.
🤔 לָמָּה? רַק בְּיוֹם הַכִּפּוּרִים, שֶׁהַצּוֹם בּוֹ מִן הַתּוֹרָה, תִּקְּנוּ לְחַנֵּךְ יְלָדִים לָצוּם לִפְנֵי גִּיל מִצְווֹת. וְהַמַּאֲכָלִים הַפְּשׁוּטִים מְחַנְּכִים אֶת הַיְּלָדִים לְהִתְאַבֵּל יַחַד עִם כָּל הַצִּבּוּר.
💛 גַּם יֶלֶד שֶׁאוֹכֵל יָכוֹל לְהִשְׁתַּתֵּף בְּצַעַר הַצִּבּוּר: לְוַתֵּר עַל מַמְתָּק, וּלְהִתְפַּלֵּל שֶׁבֵּית הַמִּקְדָּשׁ יִבָּנֶה.
(פֶּרֶק ז׳, הֲלָכָה ט׳)

9בְּרִית מִילָה וְחַיָּלִים בְּיוֹם צוֹם

גַּם בַּעֲלֵי הַבְּרִית – אֲבִי הַבֵּן, הַסַּנְדָּק וְהַמּוֹהֵל – חַיָּבִים לָצוּם. לָכֵן נוֹהֲגִים לַעֲרֹךְ אֶת הַבְּרִית לִקְרַאת סוֹף הַצּוֹם, וְאֶת הַסְּעוּדָה – אַחֲרֵי צֵאת הַכּוֹכָבִים. וְכָךְ גַּם בְּפִדְיוֹן הַבֵּן. חַיָּלִים שֶׁעוֹסְקִים בִּפְעִילוּת בִּטְחוֹנִית, שֶׁהַצּוֹם עָלוּל לִפְגֹּעַ בָּהּ – אוֹכְלִים וְשׁוֹתִים כָּרָגִיל. אֲבָל חַיָּלִים בְּאִמּוּנִים – חַיָּבִים לָצוּם.
🤔 לָמָּה? צַעַר הָרַבִּים עַל הַחֻרְבָּן קוֹדֵם לְשִׂמְחָה פְּרָטִית. וּשְׁמִירָה עַל עַם יִשְׂרָאֵל – קוֹדֶמֶת לַצּוֹם.
(פֶּרֶק ז׳, הֲלָכָה ט׳)

10״עֲנֵנוּ״

בְּיוֹם צוֹם מוֹסִיפִים בַּתְּפִלָּה ״עֲנֵנוּ״ – בַּקָּשָׁה שֶׁה׳ יַעֲנֶה לָנוּ בְּיוֹם צוֹם תַּעֲנִיתֵנוּ. בַּחֲזָרַת הַשַּׁ״ץ אוֹמֵר אוֹתָהּ הַחַזָּן כִּבְרָכָה בִּפְנֵי עַצְמָהּ, בֵּין ״גּוֹאֵל יִשְׂרָאֵל״ לְ״רְפָאֵנוּ״, וְרַק כְּשֶׁיֵּשׁ לְפָחוֹת שִׁשָּׁה מִתְעַנִּים. בִּתְפִלַּת הַלַּחַשׁ אוֹמְרִים אוֹתָהּ בְּתוֹךְ בִּרְכַּת ״שׁוֹמֵעַ תְּפִלָּה״. מִי שֶׁאָכַל אוֹ שָׁתָה כַּשִּׁעוּר וְשָׁבַר אֶת הַצּוֹם – כְּבָר אֵינוֹ אוֹמֵר ״עֲנֵנוּ״ (אֲבָל מַמְשִׁיךְ לָצוּם). וּבְאֵילוּ תְּפִלּוֹת אוֹמְרִים אוֹתָהּ?
🤔 לָמָּה? הָאַשְׁכְּנַזִּים חוֹשְׁשִׁים שֶׁמִּי שֶׁיֹּאמַר ״עֲנֵנוּ״ בְּשַׁחֲרִית יֵחָלֵשׁ בְּמֶשֶׁךְ הַיּוֹם וְיִשְׁבֹּר אֶת הַצּוֹם – וְאָז יִמָּצֵא שֶׁאָמַר דְּבַר שֶׁקֶר: ״בְּיוֹם צוֹם תַּעֲנִיתֵנוּ״. מִי שֶׁצָּם עַד מִנְחָה – מִן הַסְּתָם יְסַיֵּם. וְכָל אֶחָד מַמְשִׁיךְ בְּמִנְהַג אֲבוֹתָיו.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג עוֹלֵי תֵּימָן וְחֵלֶק מִקְּהִלּוֹת הַסְּפָרַדִּים בְּכָל שָׁלֹשׁ הַתְּפִלּוֹת שֶׁל יוֹם הַצּוֹם, גַּם בְּעַרְבִית שֶׁלִּפְנֵי הַצּוֹם.
לְמִנְהַג חֵלֶק מִקְּהִלּוֹת הַסְּפָרַדִּים בְּשַׁחֲרִית וּבְמִנְחָה, כְּשֶׁכְּבָר צָמִים (וּבְתִשְׁעָה בְּאָב גַּם בְּעַרְבִית).
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז בְּמִנְחָה בִּלְבַד.
💛 גַּם בְּיוֹם שֶׁל צַעַר, ה׳ שׁוֹמֵעַ – ״כִּי אַתָּה ה׳ הָעוֹנֶה בְּעֵת צָרָה״.
(פֶּרֶק ז׳, הֲלָכָה י׳)

11קְרִיאַת ״וַיְחַל״

בְּתַעֲנִית צִבּוּר קוֹרְאִים בַּתּוֹרָה, בְּשַׁחֲרִית וּבְמִנְחָה, אֶת הַפָּרָשָׁה ״וַיְחַל מֹשֶׁה״ – עַל כַּפָּרַת חֵטְא הָעֵגֶל. קוֹרְאִים רַק כְּשֶׁיֵּשׁ לְפָחוֹת שִׁשָּׁה מִתְעַנִּים, וְלֹא מַעֲלִים לַתּוֹרָה מִי שֶׁאֵינוֹ צָם.
🤔 לָמָּה? כְּמוֹ שֶׁה׳ כִּפֵּר עַל חֵטְא הָעֵגֶל וְנָתַן לָנוּ לוּחוֹת שְׁנִיִּים – כָּךְ נְבַקֵּשׁ שֶׁיְּכַפֵּר עַל חֲטָאֵינוּ וְיִבְנֶה אֶת בֵּית הַמִּקְדָּשׁ בִּמְהֵרָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז מַפְטִירִים בְּמִנְחָה ״דִּרְשׁוּ ה׳ בְּהִמָּצְאוֹ״, וְאוֹמְרִים ״אָבִינוּ מַלְכֵּנוּ״ אַחֲרֵי שַׁחֲרִית וּמִנְחָה.
לְמִנְהַג רֹב הַסְּפָרַדִּים לֹא מַפְטִירִים בְּמִנְחָה, וְלֹא אוֹמְרִים ״אָבִינוּ מַלְכֵּנוּ״.
(פֶּרֶק ז׳, הֲלָכָה י״א)

12בִּרְכַּת כֹּהֲנִים בְּמִנְחָה

בְּכָל הַשָּׁנָה הַכֹּהֲנִים לֹא מְבָרְכִים אֶת הָעָם בִּתְפִלַּת מִנְחָה. אֲבָל בְּתַעֲנִית צִבּוּר, אִם מִתְפַּלְּלִים מִנְחָה קָרוֹב לַשְּׁקִיעָה – הַכֹּהֲנִים נוֹשְׂאִים כַּפַּיִם. לָכֵן טוֹב לִקְבֹּעַ אֶת מִנְחָה שֶׁל תַּעֲנִית בַּחֲצִי הַשָּׁעָה שֶׁלִּפְנֵי הַשְּׁקִיעָה (וְאֶפְשָׁר גַּם אַחֲרֵי ״פְּלַג הַמִּנְחָה״), כְּדֵי לִזְכּוֹת בְּבִרְכַּת כֹּהֲנִים. כֹּהֵן שֶׁאֵינוֹ צָם – לֹא עוֹלֶה לַדּוּכָן.
🤔 לָמָּה? בְּדֶרֶךְ כְּלָל מִנְחָה הִיא אַחֲרֵי הַסְּעוּדָה, וְחָשְׁשׁוּ שֶׁמָּא הַכֹּהֲנִים שָׁתוּ יַיִן. בְּיוֹם צוֹם אֵין חֲשָׁשׁ כָּזֶה, וּמִנְחָה סָמוּךְ לַשְּׁקִיעָה דּוֹמָה לִתְפִלַּת נְעִילָה – שֶׁבָּהּ מְבָרְכִים הַכֹּהֲנִים.
💛 דַּוְקָא בְּיוֹם שֶׁל צוֹם וְצַעַר, ה׳ שׁוֹלֵחַ לָנוּ בְּרָכָה בְּאַהֲבָה – ״לְבָרֵךְ אֶת עַמּוֹ יִשְׂרָאֵל בְּאַהֲבָה״.
(פֶּרֶק ז׳, הֲלָכָה י״ב)

📝 מָה לָמַדְנוּ

  • יִשְׂרָאֵל קִבְּלוּ לָצוּם בַּצּוֹמוֹת הַקַּלִּים עַד שֶׁיִּבָּנֶה הַמִּקְדָּשׁ.
  • בְּצוֹם קַל צָמִים מֵעֲלוֹת הַשַּׁחַר עַד צֵאת הַכּוֹכָבִים, וְאָסוּר רַק לֶאֱכֹל וְלִשְׁתּוֹת.
  • מִי שֶׁשָּׁכַח וְאָכַל – מַמְשִׁיךְ לָצוּם. חוֹלֶה פָּטוּר. יְלָדִים פְּטוּרִים, וְרַבִּים מִתְרַגְּלִים לָצוּם כַּמָּה שָׁעוֹת וְאוֹכְלִים אֹכֶל פָּשׁוּט.
  • אוֹמְרִים ״עֲנֵנוּ״ (לְכָל עֵדָה לְפִי מִנְהָגָהּ), וְקוֹרְאִים בַּתּוֹרָה ״וַיְחַל מֹשֶׁה״.

❓ שְׁאֵלוֹת חֲזָרָה

  1. מָה שְׁנֵי הַהֶבְדֵּלִים הָעִקָּרִיִּים בֵּין צוֹם קַל לְתִשְׁעָה בְּאָב?
  2. קַמְתָּ צָמֵא בְּשָׁעָה אַרְבַּע לִפְנוֹת בֹּקֶר בְּיוֹם צוֹם קַל, לִפְנֵי עֲלוֹת הַשַּׁחַר. מָה הַדִּין, וּמָה הָיָה עָדִיף לַעֲשׂוֹת בָּעֶרֶב?
  3. בְּאֵילוּ תְּפִלּוֹת אוֹמְרִים ״עֲנֵנוּ״ לְפִי הַמִּנְהָגִים הַשּׁוֹנִים, וְלָמָּה הָאַשְׁכְּנַזִּים אוֹמְרִים רַק בְּמִנְחָה?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״הַצּוֹמוֹת וּשְׁלֹשֶׁת הַשָּׁבוּעוֹת״

סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים
מַהֲדוּרַת יְלָדִים

🏛️ עֲבוֹדַת יוֹם הַכִּפֻּרִים

פֶּרֶק י׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
דַּמְיֵן שֶׁאַתָּה עוֹמֵד בַּעֲזָרָה שֶׁל בֵּית הַמִּקְדָּשׁ, בְּיוֹם הַכִּפֻּרִים. אֲלָפִים עוֹמְדִים בְּשֶׁקֶט. הַכֹּהֵן הַגָּדוֹל, לָבוּשׁ לָבָן, נִכְנָס לְבַדּוֹ אֶל הַמָּקוֹם הֲכִי קָדוֹשׁ בָּעוֹלָם – בְּשֵׁם כֻּלָּנוּ. בַּפֶּרֶק הַזֶּה נִלְמַד מָה עָשָׂה הַכֹּהֵן הַגָּדוֹל בְּיוֹם הַכִּפֻּרִים, וְאֵיךְ אֲנַחְנוּ מַמְשִׁיכִים אֶת עֲבוֹדָתוֹ הַיּוֹם, בִּתְפִלַּת מוּסָף.

1קֹדֶשׁ הַקֳּדָשִׁים

בְּבֵית הַמִּקְדָּשׁ, מֵאֲחוֹרֵי הַפָּרֹכֶת, הָיָה הַמָּקוֹם הֲכִי קָדוֹשׁ בָּעוֹלָם – ״קֹדֶשׁ הַקֳּדָשִׁים״. שָׁם עָמַד אֲרוֹן הַבְּרִית וּבוֹ הַלּוּחוֹת, וְעַל הָאָרוֹן הַכַּפֹּרֶת וּשְׁנֵי הַכְּרוּבִים, שֶׁמְּבַטְּאִים אֶת קֶשֶׁר הַבְּרִית וְהָאַהֲבָה שֶׁבֵּין ה׳ לְיִשְׂרָאֵל. הַמָּקוֹם הָיָה עַל ״אֶבֶן הַשְּׁתִיָּה״, שֶׁחֲכָמִים אָמְרוּ שֶׁמִּמֶּנָּה הוּשְׁתַת הָעוֹלָם. אַף אָדָם לֹא נִכְנַס לְשָׁם – רַק הַכֹּהֵן הַגָּדוֹל, בְּיוֹם הַכִּפֻּרִים, בְּשֵׁם כָּל יִשְׂרָאֵל, אַרְבַּע פְּעָמִים.
🤔 לָמָּה? ״כִּי בֶּעָנָן אֵרָאֶה עַל הַכַּפֹּרֶת״. הַמָּקוֹם קָדוֹשׁ כָּל כָּךְ, שֶׁרַק בַּיּוֹם הַקָּדוֹשׁ בְּיוֹתֵר, וְרַק הָאָדָם הַקָּדוֹשׁ בְּיוֹתֵר, נִכְנַס אֵלָיו.
💛 גַּם אַחֲרֵי הַחֻרְבָּן הַשְּׁכִינָה לֹא זָזָה מִן הַמָּקוֹם הַזֶּה. הַגַּעְגּוּעִים שֶׁלָּנוּ לְבֵית הַמִּקְדָּשׁ מְקָרְבִים אֶת הַגְּאֻלָּה.
(פֶּרֶק י׳, הֲלָכָה א׳)

2הַכֹּהֵן הַגָּדוֹל

הַכֹּהֵן הַגָּדוֹל הוּא הַמֻּבְחָר שֶׁבַּכֹּהֲנִים, וְהוּא מַמְשִׁיךְ אֶת דַּרְכּוֹ שֶׁל אַהֲרֹן: ״אוֹהֵב שָׁלוֹם וְרוֹדֵף שָׁלוֹם, אוֹהֵב אֶת הַבְּרִיּוֹת וּמְקָרְבָן לַתּוֹרָה״. הוּא לָבַשׁ שְׁמוֹנָה בְּגָדִים. עַל מִצְחוֹ הָיָה הַצִּיץ, וְעָלָיו חָקוּק ״קֹדֶשׁ לַה׳״, וְעַל לִבּוֹ הַחֹשֶׁן, וְעָלָיו שְׁמוֹת שִׁבְטֵי יִשְׂרָאֵל. כָּל בֶּגֶד כִּפֵּר עַל חֵטְא אַחֵר: הַמְּעִיל – עַל לָשׁוֹן הָרָע, הַמִּצְנֶפֶת – עַל גַּאֲוָה, וְהַחֹשֶׁן – עַל דִּינִים.
🤔 לָמָּה? הַצִּיץ עַל הַמֵּצַח מְבַטֵּא אֶת הַדְּבֵקוּת שֶׁלּוֹ בַּה׳, וְהַחֹשֶׁן עַל הַלֵּב מְבַטֵּא אֶת הָאַהֲבָה וְהָאַחֲרָיוּת שֶׁלּוֹ לְכָל יִשְׂרָאֵל.
💛 בְּבַיִת רִאשׁוֹן שִׁמְּשׁוּ כֹּהֲנִים גְּדוֹלִים צַדִּיקִים, וְזָכוּ לַאֲרִיכוּת יָמִים: ״יִרְאַת ה׳ תּוֹסִיף יָמִים״.
(פֶּרֶק י׳, הֲלָכָה ג׳)

3יוֹם אֶחָד – כֹּהֵן אֶחָד

בְּכָל יְמוֹת הַשָּׁנָה כָּל הַכֹּהֲנִים עוֹבְדִים בַּמִּקְדָּשׁ. אֲבָל בְּיוֹם הַכִּפֻּרִים, מֵרֹב קְדֻשַּׁת הַיּוֹם, רַק הַכֹּהֵן הַגָּדוֹל עָשָׂה אֶת כָּל הָעֲבוֹדוֹת: אֶת הַקָּרְבָּנוֹת שֶׁל כָּל יוֹם, אֶת קָרְבְּנוֹת הַמּוּסָף, וְאֶת הַקָּרְבָּנוֹת הַמְּיֻחָדִים לְיוֹם הַכִּפֻּרִים – פַּר לְחַטָּאת וְאַיִל לְעוֹלָה שֶׁקָּנָה מִכַּסְפּוֹ, וּשְׁנֵי שְׂעִירֵי עִזִּים. הֵכִינוּ גַּם כֹּהֵן נוֹסָף, שֶׁיַּחְלִיף אוֹתוֹ אִם יִקְרֶה לוֹ מַשֶּׁהוּ.
🤔 לָמָּה? כְּשֶׁכָּל הָעֲבוֹדָה נַעֲשֵׂית עַל יְדֵי אָדָם אֶחָד, הַקָּדוֹשׁ בְּיוֹתֵר, הַכֹּל מִתְרוֹמֵם וּמִתְקַשֵּׁר אֶל שֹׁרֶשׁ הַקְּדֻשָּׁה שֶׁבְּקֹדֶשׁ הַקֳּדָשִׁים.
(פֶּרֶק י׳, הֲלָכָה ד׳, הֲלָכָה י״ד)

4בִּגְדֵי לָבָן

אֶת הָעֲבוֹדוֹת הָרְגִילוֹת עָשָׂה הַכֹּהֵן הַגָּדוֹל בִּשְׁמוֹנַת בִּגְדֵי הַזָּהָב, הַצִּבְעוֹנִיִּים וְהַמְפֹאָרִים. אֲבָל אֶת הָעֲבוֹדוֹת שֶׁבְּקֹדֶשׁ הַקֳּדָשִׁים עָשָׂה בְּאַרְבָּעָה בִּגְדֵי לָבָן פְּשׁוּטִים, מִפִּשְׁתָּן: כֻּתֹּנֶת, מִכְנָסַיִם, אַבְנֵט וּמִצְנֶפֶת. אִם עָשָׂה אוֹתָן בְּבִגְדֵי זָהָב – עֲבוֹדָתוֹ פְּסוּלָה.
🤔 לָמָּה? חֲכָמִים אָמְרוּ: ״אֵין קָטֵגוֹר נַעֲשֶׂה סָנֵגוֹר״. הַזָּהָב מַזְכִּיר אֶת חֵטְא עֵגֶל הַזָּהָב, וְאִי אֶפְשָׁר לְבַקֵּשׁ סְלִיחָה כְּשֶׁלּוֹבְשִׁים אֶת מָה שֶׁמַּזְכִּיר אֶת הַחֵטְא. וְהַלָּבָן הַפָּשׁוּט מְבַטֵּא הִתְעַלּוּת אֶל מֵעֵבֶר לְכָל הַגְּוָנִים שֶׁל הָעוֹלָם הַזֶּה.
💛 גַּם אֲנַחְנוּ לוֹבְשִׁים לָבָן בְּיוֹם הַכִּפֻּרִים – כְּמוֹ הַכֹּהֵן הַגָּדוֹל, כְּמוֹ הַמַּלְאָכִים.
(פֶּרֶק י׳, הֲלָכָה ה׳)

5חָמֵשׁ טְבִילוֹת וַעֲשָׂרָה קִדּוּשִׁין

כָּל כֹּהֵן שֶׁנִּכְנַס לַעֲבֹד בַּמִּקְדָּשׁ טוֹבֵל, וּמְקַדֵּשׁ אֶת יָדָיו וְרַגְלָיו בְּמַיִם מִן הַכִּיּוֹר. בְּיוֹם הַכִּפֻּרִים טָבַל הַכֹּהֵן הַגָּדוֹל בְּכָל פַּעַם שֶׁהֶחְלִיף בְּגָדִים – מִבִּגְדֵי זָהָב לְבִגְדֵי לָבָן, וּמִבִּגְדֵי לָבָן לְבִגְדֵי זָהָב – וְקִדֵּשׁ אֶת יָדָיו וְרַגְלָיו פַּעֲמַיִם בְּכָל הַחְלָפָה. בְּיַחַד: חָמֵשׁ טְבִילוֹת וַעֲשָׂרָה קִדּוּשִׁין בְּיוֹם אֶחָד. לִכְבוֹדוֹ הֵבִיאוּ לוֹ אֶת הַמַּיִם בְּקִיתוֹן שֶׁל זָהָב, כְּדֵי שֶׁלֹּא יִטְרַח לָלֶכֶת שׁוּב וָשׁוּב לַכִּיּוֹר.
🤔 לָמָּה? הַטְּבִילָה הִיא מַעֲבָר מִמַּדְרֵגָה לְמַדְרֵגָה. וְקִדּוּשׁ הַיָּדַיִם וְהָרַגְלַיִם מְקַדֵּשׁ אֶת כָּל הַכֹּחוֹת שֶׁל הָאָדָם – הַיָּדַיִם שֶׁעוֹשׂוֹת וְהָרַגְלַיִם שֶׁהוֹלְכוֹת – לַעֲבוֹדַת ה׳.
(פֶּרֶק י׳, הֲלָכָה ו׳)

6לַיְלָה שֶׁל תּוֹרָה

בְּלֵיל יוֹם הַכִּפֻּרִים הַכֹּהֵן הַגָּדוֹל לֹא יָשַׁן, כְּדֵי שֶׁלֹּא יִטָּמֵא בְּשֵׁנָתוֹ. כָּל הַלַּיְלָה לָמַד תּוֹרָה עִם גְּדוֹלֵי הַתּוֹרָה שֶׁל יְרוּשָׁלַיִם, וְשָׁרוּ לְפָנָיו שִׁירֵי שֶׁבַח לַה׳. אִם הִתְחִיל לְהִתְנַמְנֵם – פִּרְחֵי הַכְּהֻנָּה, הַכֹּהֲנִים הַצְּעִירִים, הָיוּ מַקִּישִׁים לְפָנָיו בָּאֶצְבַּע כְּדֵי לְעוֹרֵר אוֹתוֹ, וְהָיוּ מַצִּיעִים לוֹ לַעֲמֹד עַל הָרִצְפָּה הַקָּרָה.
💛 מִי שֶׁעוֹמֵד לִפְנֵי ה׳ בְּשֵׁם כָּל הָעָם רוֹצֶה לָבוֹא מוּכָן – וְהַהֲכָנָה הַטּוֹבָה בְּיוֹתֵר הִיא תּוֹרָה.
(פֶּרֶק י׳, הֲלָכָה י״ד)

7קֹדֶם – עַל עַצְמוֹ

לִפְנֵי שֶׁהַכֹּהֵן הַגָּדוֹל כִּפֵּר עַל כָּל יִשְׂרָאֵל, הוּא כִּפֵּר עַל עַצְמוֹ. הוּא הֵבִיא פַּר שֶׁקָּנָה מִכַּסְפּוֹ, סָמַךְ עָלָיו אֶת יָדָיו וְהִתְוַדָּה עַל עֲווֹנוֹתָיו וַעֲווֹנוֹת בֵּיתוֹ. אַחַר כָּךְ הִתְוַדָּה עַל הַפָּר שׁוּב – עַל עֲווֹנוֹת אֶחָיו הַכֹּהֲנִים. וְרַק אָז עָבַר לְכַפֵּר עַל כָּל הָעָם.
🤔 לָמָּה? ״מוּטָב יָבוֹא זַכַּאי וִיכַפֵּר עַל הַחַיָּב״. הַכֹּהֲנִים אַחְרָאִים עַל הַקֶּשֶׁר בֵּין יִשְׂרָאֵל לַה׳, וְלָכֵן הֵם צְרִיכִים כַּפָּרָה קֹדֶם.
💛 גַּם אֲנִי, לִפְנֵי שֶׁאֲנִי מְבַקֵּשׁ שֶׁאֲחֵרִים יִשְׁתַּפְּרוּ, יָכוֹל לְהַתְחִיל בִּתְשׁוּבָה שֶׁלִּי.
(פֶּרֶק י׳, הֲלָכָה ט׳, הֲלָכָה י״ד)

8שְׁנֵי הַשְּׂעִירִים וְהַגּוֹרָל

לְכַפָּרַת יִשְׂרָאֵל לָקְחוּ שְׁנֵי שְׂעִירֵי עִזִּים שָׁוִים – בְּמַרְאֶה, בְּגֹדֶל וּבִמְחִיר. הַכֹּהֵן הַגָּדוֹל הִטִּיל עֲלֵיהֶם גּוֹרָל: ״גּוֹרָל אֶחָד לַה׳ וְגוֹרָל אֶחָד לַעֲזָאזֵל״. הוּא נִכְנַס לְקֹדֶשׁ הַקֳּדָשִׁים עִם קְטֹרֶת, וְאַחַר כָּךְ הִזָּה שָׁם מִדַּם הַפָּר וּמִדַּם הַשָּׂעִיר שֶׁעָלָה לַה׳. עַל הַשָּׂעִיר הַשֵּׁנִי סָמַךְ אֶת שְׁתֵּי יָדָיו, הִתְוַדָּה עַל כָּל עֲווֹנוֹת יִשְׂרָאֵל, וְשָׁלַח אוֹתוֹ אֶל הַמִּדְבָּר: ״וְנָשָׂא הַשָּׂעִיר עָלָיו אֶת כָּל עֲוֹנֹתָם״.
🤔 לָמָּה? קֹדֶם מְטַהֲרִים אֶת הָאֱמוּנָה – אֶת הַקֶּשֶׁר שֶׁלָּנוּ עִם ה׳. וְאָז מִתְבָּרֵר שֶׁהַחֲטָאִים אֵינָם שַׁיָּכִים לָנוּ בֶּאֱמֶת: הֵם בָּאוּ מִבַּחוּץ, וְאֶפְשָׁר לִשְׁלֹחַ אוֹתָם לַמִּדְבָּר.
💛 הַגּוֹרָל מַזְכִּיר שֶׁה׳ מַשְׁגִּיחַ עַל הַכֹּל. וְהַשָּׂעִיר שֶׁנִּשְׁלָח מְלַמֵּד: הָרַע אֵינוֹ חֵלֶק מִמְּךָ, וְאֶפְשָׁר לְהִפָּרֵד מִמֶּנּוּ.
(פֶּרֶק י׳, הֲלָכָה ט׳, הֲלָכָה י״ב)

9יוֹם טוֹב לַכֹּהֵן הַגָּדוֹל

כְּשֶׁהוֹדִיעוּ לַכֹּהֵן הַגָּדוֹל שֶׁהַשָּׂעִיר הִגִּיעַ לַמִּדְבָּר, קָרָא בַּתּוֹרָה לִפְנֵי הָעָם וּבֵרַךְ בְּרָכוֹת: עַל הָעֲבוֹדָה, עַל מְחִילַת הֶעָווֹן, עַל הַמִּקְדָּשׁ, עַל יִשְׂרָאֵל וְעַל הַכֹּהֲנִים. בַּפַּעַם הָאַחֲרוֹנָה נִכְנַס לְקֹדֶשׁ הַקֳּדָשִׁים כְּדֵי לְהוֹצִיא אֶת כְּלֵי הַקְּטֹרֶת. בְּסוֹף הַיּוֹם לָבַשׁ אֶת בְּגָדָיו שֶׁלּוֹ וְיָצָא לְבֵיתוֹ. כָּל הָעָם לִוּוּ אוֹתוֹ, וְהוּא עָשָׂה יוֹם טוֹב עַל שֶׁיָּצָא בְּשָׁלוֹם מִן הַקֹּדֶשׁ.
💛 כָּל הָעָם שָׂמַח יַחַד עִם הַכֹּהֵן הַגָּדוֹל, כִּי הַכַּפָּרָה שֶׁלּוֹ הָיְתָה הַכַּפָּרָה שֶׁל כֻּלָּם.
(פֶּרֶק י׳, הֲלָכָה י״ז)

10סֵדֶר הָעֲבוֹדָה בִּתְפִלַּת מוּסָף

הַיּוֹם אֵין לָנוּ בֵּית מִקְדָּשׁ. בִּתְפִלַּת מוּסָף הַחַזָּן מְסַפֵּר, וְכָל הַקָּהָל אִתּוֹ, אֶת ״סֵדֶר הָעֲבוֹדָה״ שֶׁל הַכֹּהֵן הַגָּדוֹל. צָרִיךְ לוֹמַר אוֹתוֹ בְּדִיּוּק, כִּי אֲמִירַת סֵדֶר הַקָּרְבָּנוֹת נֶחְשֶׁבֶת כְּמוֹ הַקְרָבָתָם. ה׳ אָמַר לְאַבְרָהָם: ״כָּל זְמַן שֶׁקּוֹרְאִים בָּהֶם, מַעֲלֶה אֲנִי עֲלֵיהֶם כְּאִלּוּ מַקְרִיבִים לְפָנַי קָרְבָּן, וּמוֹחֵל אֲנִי עַל כָּל עֲווֹנוֹתֵיהֶם״.
🤔 לָמָּה? לְכָל מַעֲשֶׂה יֵשׁ נְשָׁמָה, וְהַנְּשָׁמָה שֶׁל הַמִּצְווֹת הֵן דִּבְרֵי הַתּוֹרָה עֲלֵיהֶן. כְּשֶׁאִי אֶפְשָׁר לְהַקְרִיב, הָאֲמִירָה – וּבִמְיֻחָד בַּצִּבּוּר – נֶחְשֶׁבֶת כְּמוֹ הַקָּרְבָּן.
💛 אֲנַחְנוּ מִתְפַּלְּלִים שֶׁיִּבָּנֶה בֵּית הַמִּקְדָּשׁ בִּמְהֵרָה, וְנִזְכֶּה לִרְאוֹת בְּעֵינֵינוּ אֶת הַכֹּהֵן הַגָּדוֹל יוֹצֵא בְּשָׁלוֹם מִן הַקֹּדֶשׁ.
(פֶּרֶק י׳, הֲלָכָה י״ח)

📝 מָה לָמַדְנוּ

  • בְּיוֹם הַכִּפֻּרִים רַק הַכֹּהֵן הַגָּדוֹל עָשָׂה אֶת כָּל הָעֲבוֹדוֹת, וְנִכְנַס לְקֹדֶשׁ הַקֳּדָשִׁים בְּשֵׁם כָּל יִשְׂרָאֵל.
  • בְּקֹדֶשׁ הַקֳּדָשִׁים עָבַד בְּאַרְבָּעָה בִּגְדֵי לָבָן, וְלֹא בְּבִגְדֵי זָהָב – ״אֵין קָטֵגוֹר נַעֲשֶׂה סָנֵגוֹר״.
  • קֹדֶם כִּפֵּר עַל עַצְמוֹ, אַחַר כָּךְ עַל הַכֹּהֲנִים, וּבַסּוֹף עַל כָּל הָעָם.
  • בְּגוֹרָל נִבְחַר שָׂעִיר אֶחָד לַה׳ וְשָׂעִיר אֶחָד לַעֲזָאזֵל, שֶׁנָּשָׂא אֶת הָעֲווֹנוֹת לַמִּדְבָּר.
  • הַיּוֹם אוֹמְרִים אֶת ״סֵדֶר הָעֲבוֹדָה״ בְּמוּסָף, וְהָאֲמִירָה נֶחְשֶׁבֶת כְּמוֹ הַקָּרְבָּן.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה נִכְנַס הַכֹּהֵן הַגָּדוֹל לְקֹדֶשׁ הַקֳּדָשִׁים בְּבִגְדֵי לָבָן וְלֹא בְּבִגְדֵי זָהָב?
  2. סַפֵּר מָה עָשׂוּ בִּשְׁנֵי הַשְּׂעִירִים. מָה מְלַמֵּד הַשָּׂעִיר שֶׁנִּשְׁלַח לַמִּדְבָּר?
  3. אֵיךְ אֲנַחְנוּ מַמְשִׁיכִים הַיּוֹם אֶת עֲבוֹדַת הַכֹּהֵן הַגָּדוֹל, וְלָמָּה זֶה נֶחְשָׁב כְּמוֹ קָרְבָּן?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״יוֹם הַכִּפּוּרִים״

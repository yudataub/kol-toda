סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים
מַהֲדוּרַת יְלָדִים

📯 מִצְוַת הַשּׁוֹפָר

פֶּרֶק ד׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
בֵּית הַכְּנֶסֶת מָלֵא, וּפִתְאוֹם – שֶׁקֶט. אַף אֶחָד לֹא זָז. הַתּוֹקֵעַ מְכַסֶּה אֶת רֹאשׁוֹ בַּטַּלִּית, מֵרִים אֶת הַשּׁוֹפָר, וְהָרַב קוֹרֵא: ״תְּקִיעָה!״. הַקּוֹל עוֹבֵר בְּכָל הַגּוּף. מָה יֵשׁ בַּקּוֹל הַזֶּה שֶׁמְּעוֹרֵר אֶת הַלֵּב? בַּפֶּרֶק הַזֶּה נִלְמַד אֵיךְ מְקַיְּמִים אֶת מִצְוַת הַשּׁוֹפָר – וּמָה הִיא אוֹמֶרֶת לָנוּ.

1מִצְוָה לִשְׁמֹעַ

מִצְוַת עֲשֵׂה לִשְׁמֹעַ קוֹל שׁוֹפָר בְּרֹאשׁ הַשָּׁנָה: ״יוֹם תְּרוּעָה יִהְיֶה לָכֶם״. שִׂים לֵב: הַמִּצְוָה הִיא לִשְׁמֹעַ, וְלֹא לִתְקֹעַ. לָכֵן צָרִיךְ לִשְׁמֹעַ אֶת קוֹל הַשּׁוֹפָר עַצְמוֹ. מִי שֶׁשָּׁמַע רַק הֵד, אוֹ קוֹל מְעֹרָב בְּהֵד – לֹא יָצָא.
🤔 לָמָּה? בַּבְּרָכָה כָּתוּב ״לִשְׁמֹעַ קוֹל שׁוֹפָר״, וְלֹא ״לִתְקֹעַ״. תּוֹקֵעַ אֶחָד תּוֹקֵעַ, וְכָל הַקָּהָל מְקַיֵּם אֶת הַמִּצְוָה בִּשְׁמִיעָה.
(פֶּרֶק ד׳, הֲלָכָה א׳, הֲלָכָה ה׳)

2תְּקִיעָה, תְּרוּעָה, תְּקִיעָה

מִן הַתּוֹרָה צָרִיךְ לִשְׁמֹעַ בְּרֹאשׁ הַשָּׁנָה שָׁלֹשׁ תְּרוּעוֹת, וְלִפְנֵי כָּל תְּרוּעָה וְאַחֲרֶיהָ – תְּקִיעָה. כָּל ״יְחִידָה״ כָּזוֹ מְסַפֶּרֶת סִפּוּר. הַתְּקִיעָה הָרִאשׁוֹנָה – הַנְּשָׁמָה הַיְּשָׁרָה וְהַטְּהוֹרָה, כְּמוֹ יֶלֶד רַךְ שֶׁעוֹד לֹא חָטָא. הַתְּרוּעָה – הַקְּשָׁיִים וְהַנְּפִילוֹת שֶׁבַּחַיִּים, כַּאֲנָחָה וּכְבֶכִי. וְהַתְּקִיעָה הָאַחֲרוֹנָה – הַיֹּשֶׁר שֶׁחוֹזֵר אַחֲרֵי הַתְּשׁוּבָה.
🤔 לָמָּה? הַשּׁוֹפָר מַחְרִיד אֶת הַשּׁוֹמֵעַ, וְגַם מְעוֹרֵר אוֹתוֹ לַחֲזֹר אֶל הַטּוֹב שֶׁבְּתוֹכוֹ. הַקּוֹל שֶׁלּוֹ טִבְעִי וּפָשׁוּט – קוֹל שֶׁל הַלֵּב, עוֹד לִפְנֵי הַמִּלִּים.
💛 בְּסוֹף כָּל הַתְּקִיעוֹת נוֹהֲגִים לִתְקֹעַ תְּקִיעָה גְּדוֹלָה וַאֲרֻכָּה – סִימָן שֶׁבַּסּוֹף, אַחֲרֵי כָּל הַקְּשָׁיִים, הַכֹּל מִתְתַּקֵּן.
(פֶּרֶק ד׳, הֲלָכָה א׳)

3שְׁלֹשִׁים תְּקִיעוֹת

הַתְּרוּעָה הִיא קוֹל שֶׁל צַעַר, וּבְמֶשֶׁךְ הַשָּׁנִים נוֹצַר סָפֵק אֵיךְ הִיא צְרִיכָה לְהִשָּׁמַע: כְּמוֹ אֲנָחוֹת (שְׁבָרִים), כְּמוֹ בֶּכִי (תְּרוּעָה), אוֹ שְׁנֵיהֶם יַחַד. רַבִּי אַבָּהוּ תִּקֵּן בְּעִירוֹ קֵיסָרִי לִתְקֹעַ אֶת שְׁלָשְׁתָּם, וְכָל הַקְּהִלּוֹת קִבְּלוּ אֶת תַּקָּנָתוֹ. הַסֵּדֶר: שָׁלֹשׁ פְּעָמִים תשר״ת (תְּקִיעָה, שְׁבָרִים, תְּרוּעָה, תְּקִיעָה), שָׁלֹשׁ פְּעָמִים תש״ת וְשָׁלֹשׁ פְּעָמִים תר״ת. מִן הַתּוֹרָה – תֵּשַׁע תְּקִיעוֹת, וְהַיּוֹם חַיָּבִים שְׁלֹשִׁים.
🤔 לָמָּה? כָּל הַקְּהִלּוֹת יָצְאוּ יְדֵי חוֹבָה גַּם קֹדֶם, אֲבָל זֶה נִרְאָה כְּאִלּוּ יִשְׂרָאֵל חֲלוּקִים בַּמִּצְוָה. וְעוֹד: לְכָל סוּג שֶׁל תְּרוּעָה יֵשׁ עֵרֶךְ מִשֶּׁלּוֹ.
(פֶּרֶק ד׳, הֲלָכָה ב׳)

4אֵיךְ נִשְׁמָעִים הַקּוֹלוֹת?

תְּרוּעָה – קוֹלוֹת קְצָרִים וּרְצוּפִים, כְּמוֹ בֶּכִי. כָּל קוֹל קָצָר כָּזֶה נִקְרָא ״טְרוּמִיט״, וְתוֹקְעִים לְפָחוֹת תִּשְׁעָה. שְׁבָרִים – שְׁלֹשָׁה קוֹלוֹת בֵּינוֹנִיִּים, כְּמוֹ אֲנָחָה, וְכָל שֶׁבֶר אָרֹךְ כְּמוֹ שְׁלֹשָׁה טְרוּמִיטִין. תְּקִיעָה – קוֹל אָרֹךְ וּפָשׁוּט, וְהִיא צְרִיכָה לִהְיוֹת אֲרֻכָּה לְפָחוֹת כְּמוֹ הַתְּרוּעָה שֶׁבְּאוֹתָהּ יְחִידָה. מִי שֶׁרוֹצֶה לָצֵאת לְכָל הַדֵּעוֹת – מַאֲרִיךְ בַּתְּקִיעוֹת שֶׁל תשר״ת חָמֵשׁ שְׁנִיּוֹת, וּבַתְּקִיעוֹת שֶׁל תש״ת וְתר״ת – שְׁתֵּי שְׁנִיּוֹת וָחֵצִי. גַּם כְּשֶׁהַקּוֹל מִשְׁתַּנֶּה בְּאֶמְצַע הַתְּקִיעָה, מֵעָבֶה לְדַק אוֹ מִצָּלוּל לְצָרוּד – הַתְּקִיעָה כְּשֵׁרָה.
💛 ״כָּל הַקּוֹלוֹת כְּשֵׁרִים בַּשּׁוֹפָר״ – גַּם קוֹל צָרוּד וְלֹא מֻשְׁלָם עוֹלֶה לִפְנֵי ה׳, כְּשֶׁהוּא יוֹצֵא מִן הַלֵּב.
(פֶּרֶק ד׳, הֲלָכָה י״א, הֲלָכָה י״ב)

5נְשִׁימָה בֵּין הַקּוֹלוֹת

בְּכָל יְחִידָה מַפְסִיקִים כְּדֵי נְשִׁימָה בֵּין הַתְּקִיעָה הָרִאשׁוֹנָה לַשְּׁבָרִים אוֹ לַתְּרוּעָה, וְשׁוּב לִפְנֵי הַתְּקִיעָה הָאַחֲרוֹנָה. אֲבָל בְּאֶמְצַע הַשְּׁבָרִים אוֹ בְּאֶמְצַע הַתְּרוּעָה אָסוּר לְהַפְסִיק לִנְשִׁימָה. וּבֵין הַשְּׁבָרִים לַתְּרוּעָה שֶׁל תשר״ת: בַּתְּקִיעוֹת שֶׁלִּפְנֵי מוּסָף לֹא נוֹשְׁמִים, וּבַתְּקִיעוֹת שֶׁבְּתוֹךְ מוּסָף נוֹשְׁמִים – כְּדֵי לָצֵאת לְכָל הַשִּׁיטוֹת.
🤔 לָמָּה? הַתְּקִיעָה הִיא קוֹל שֶׁל שִׂמְחָה וְשָׁלוֹם, וְהַשְּׁבָרִים וְהַתְּרוּעָה הֵם קוֹל שֶׁל בֶּכִי וַאֲנָחָה. לֹא מְחַבְּרִים בֵּינֵיהֶם.
(פֶּרֶק ד׳, הֲלָכָה י״ג)

6תְּקִיעוֹת דִּמְיֻשָּׁב

חֲכָמִים תִּקְּנוּ לִתְקֹעַ בְּתוֹךְ תְּפִלַּת מוּסָף, עַל סֵדֶר הַבְּרָכוֹת מַלְכֻיּוֹת, זִכְרוֹנוֹת וְשׁוֹפָרוֹת. וְעוֹד תִּקְּנוּ לִתְקֹעַ לִפְנֵי מוּסָף, וְהִתִּירוּ לַשּׁוֹמְעִים לָשֶׁבֶת בַּתְּקִיעוֹת הָאֵלֶּה. לָכֵן הֵן נִקְרָאוֹת ״תְּקִיעוֹת דִּמְיֻשָּׁב״.
🤔 לָמָּה? לָמָּה תּוֹקְעִים פַּעֲמַיִם? ״כְּדֵי לְעַרְבֵּב אֶת הַשָּׂטָן״: כְּשֶׁהוּא רוֹאֶה שֶׁיִּשְׂרָאֵל מְחַבְּבִים אֶת הַמִּצְווֹת וּמוֹסִיפִים עַל מָה שֶׁצֻּוּוּ – נִסְתָּמוֹת טַעֲנוֹתָיו.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הַסְּפָרַדִּים וְהַתֵּימָנִים יוֹשְׁבִים בַּתְּקִיעוֹת שֶׁלִּפְנֵי מוּסָף, כְּדֵי לְהַרְאוֹת שֶׁהַתְּקִיעוֹת שֶׁבְּתוֹךְ מוּסָף הֵן הָעִקָּר.
לְמִנְהַג הָאַשְׁכְּנַזִּים עוֹמְדִים גַּם בַּתְּקִיעוֹת שֶׁלִּפְנֵי מוּסָף, כִּי בָּהֶן כְּבָר יוֹצְאִים יְדֵי חוֹבָה.
(פֶּרֶק ד׳, הֲלָכָה ג׳)

7מֵאָה תְּקִיעוֹת

כִּמְעַט בְּכָל הַקְּהִלּוֹת תּוֹקְעִים הַיּוֹם מֵאָה תְּקִיעוֹת, בְּעִקְבוֹת הָאֲרִ״י הַקָּדוֹשׁ: שְׁלֹשִׁים לִפְנֵי מוּסָף, שְׁלֹשִׁים בַּחֲזָרַת הַשַּׁ״ץ שֶׁל מוּסָף – עֶשֶׂר אַחֲרֵי כָּל אַחַת מִשָּׁלֹשׁ הַבְּרָכוֹת – וְעוֹד אַרְבָּעִים. בַּתְּקִיעוֹת שֶׁל תְּפִלַּת הַלַּחַשׁ יֵשׁ שְׁנֵי מִנְהָגִים:
🤔 לָמָּה? מִן הַתּוֹרָה חַיָּבִים לִשְׁמֹעַ תֵּשַׁע תְּקִיעוֹת, וּבִזְכוּת תַּקָּנַת רַבִּי אַבָּהוּ – שְׁלֹשִׁים. מֵאָה הוּא מִנְהַג כָּל יִשְׂרָאֵל, וְהָאֲרִ״י סִדֵּר לָהֶן כַּוָּנוֹת עַל פִּי הַסּוֹד.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הַסְּפָרַדִּים וְהַחֲסִידִים תּוֹקְעִים גַּם בִּתְפִלַּת הַלַּחַשׁ שֶׁל מוּסָף, וְהַמִּתְפַּלְּלִים מִשְׁתַּדְּלִים לְהִתְפַּלֵּל בַּקֶּצֶב שֶׁל הַתּוֹקֵעַ. אֶת עֶשֶׂר הַתְּקִיעוֹת הָאַחֲרוֹנוֹת תּוֹקְעִים בַּקַּדִּישׁ.
לְמִנְהַג הָאַשְׁכְּנַזִּים תּוֹקְעִים רַק בַּחֲזָרַת הַשַּׁ״ץ, וְאֶת אַרְבָּעִים הַתְּקִיעוֹת הַחֲסֵרוֹת מַשְׁלִימִים אַחֲרֵי הַתְּפִלָּה: שְׁלֹשִׁים אַחֲרֵי ״עָלֵינוּ לְשַׁבֵּחַ״, וְעֶשֶׂר אַחֲרֵי ״אַנְעִים זְמִירוֹת״.
(פֶּרֶק ד׳, הֲלָכָה ד׳)

8מִצְווֹת צְרִיכוֹת כַּוָּנָה

מִי שֶׁמִּתְאַמֵּן לִתְקֹעַ, וְיָצְאוּ לוֹ תְּקִיעוֹת כְּשֵׁרוֹת – לֹא יָצָא, כִּי לֹא הִתְכַּוֵּן לְקַיֵּם אֶת הַמִּצְוָה. וּמִי שֶׁשָּׁמַע מִבֵּיתוֹ אֶת הַתְּקִיעוֹת שֶׁל בֵּית הַכְּנֶסֶת וְלֹא כִּוֵּן לָצֵאת בָּהֶן – לֹא יָצָא. הַתּוֹקֵעַ מְכַוֵּן לְהוֹצִיא אֶת כָּל הַשּׁוֹמְעִים, גַּם מִי שֶׁעוֹמֵד בַּחוּץ, וְהַשּׁוֹמֵעַ מְכַוֵּן לָצֵאת. בְּדִיעֲבַד מַסְפִּיקָה ״כַּוָּנָה רְדוּמָה״: אִם יִשְׁאֲלוּ אֶת הַשּׁוֹמֵעַ לָמָּה בָּא, יַעֲנֶה – כְּדֵי לְקַיֵּם אֶת הַמִּצְוָה.
🤔 לָמָּה? מִצְוָה אֵינָהּ רַק מַעֲשֶׂה. הִיא מַעֲשֶׂה שֶׁעוֹשִׂים כְּדֵי לַעֲשׂוֹת אֶת רְצוֹן ה׳.
💛 כְּשֶׁאֲנִי עוֹמֵד וּמְכַוֵּן לִשְׁמֹעַ אֶת הַשּׁוֹפָר, אֲנִי אוֹמֵר לַה׳ בְּלִי מִלִּים: ״בָּאתִי בִּשְׁבִילְךָ״.
(פֶּרֶק ד׳, הֲלָכָה ה׳)

9הַבְּרָכוֹת – וְ״שֶׁהֶחֱיָנוּ״ בַּיּוֹם הַשֵּׁנִי

לִפְנֵי הַתְּקִיעוֹת מְבָרְכִים שְׁתֵּי בְּרָכוֹת: ״לִשְׁמֹעַ קוֹל שׁוֹפָר״ וְ״שֶׁהֶחֱיָנוּ״. בַּיּוֹם הַשֵּׁנִי יֵשׁ שְׁנֵי מִנְהָגִים בְּבִרְכַּת ״שֶׁהֶחֱיָנוּ״:
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הַסְּפָרַדִּים לֹא מְבָרְכִים ״שֶׁהֶחֱיָנוּ״ עַל הַשּׁוֹפָר בַּיּוֹם הַשֵּׁנִי.
לְמִנְהַג הָאַשְׁכְּנַזִּים מְבָרְכִים ״שֶׁהֶחֱיָנוּ״ גַּם בַּיּוֹם הַשֵּׁנִי, וְטוֹב שֶׁהַתּוֹקֵעַ יִלְבַּשׁ בֶּגֶד חָדָשׁ וִיכַוֵּן גַּם עָלָיו.
(פֶּרֶק ד׳, הֲלָכָה ה׳)

10מָתַי תּוֹקְעִים?

תּוֹקְעִים בַּיּוֹם, כִּי כָּתוּב ״יוֹם תְּרוּעָה״. הַזְּמַן מַתְחִיל בְּהָנֵץ הַחַמָּה, וּמִי שֶׁתָּקַע כְּבָר מֵעֲלוֹת הַשַּׁחַר – יָצָא. מִי שֶׁלֹּא שָׁמַע עַד הַשְּׁקִיעָה – יִשְׁמַע בֵּין הַשְּׁמָשׁוֹת, בְּלִי בְּרָכָה. וּמִי שֶׁאֵינוֹ יָכוֹל לִשְׁמֹעַ עִם הַצִּבּוּר, רָאוּי שֶׁיִּשְׁמַע אַחֲרֵי שָׁלֹשׁ הַשָּׁעוֹת הָרִאשׁוֹנוֹת שֶׁל הַיּוֹם.
🤔 לָמָּה? בְּשָׁלֹשׁ הַשָּׁעוֹת הָרִאשׁוֹנוֹת הַדִּין עוֹד מָתוּחַ, וְאַחֲרֵיהֶן הוּא מִתְמַתֵּק. אֲבָל תְּפִלַּת הַצִּבּוּר מִתְקַבֶּלֶת בְּכָל זְמַן.
(פֶּרֶק ד׳, הֲלָכָה ו׳)

11נָשִׁים וּבָנוֹת וְהַשּׁוֹפָר

נָשִׁים פְּטוּרוֹת מִמִּצְוַת הַשּׁוֹפָר, כִּי הִיא מִצְוַת עֲשֵׂה שֶׁהַזְּמַן גְּרָמָהּ. אֲבָל מִי שֶׁשּׁוֹמַעַת – מְקַיֶּמֶת מִצְוָה וּמְקַבֶּלֶת שָׂכָר, וְרֹב נְשׁוֹת יִשְׂרָאֵל נוֹהֲגוֹת לִשְׁמֹעַ שׁוֹפָר. בַּבְּרָכָה יֵשׁ שְׁנֵי מִנְהָגִים:
🤔 לָמָּה? גַּם מִצְוָה שֶׁאֵין חוֹבָה לְקַיֵּם – מִי שֶׁמִּתְנַדֶּבֶת לְקַיְּמָהּ עוֹשָׂה אֶת רְצוֹן ה׳ וּמְקַבֶּלֶת עָלֶיהָ שָׂכָר.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הַסְּפָרַדִּים אִשָּׁה אֵינָהּ מְבָרֶכֶת עַל הַשּׁוֹפָר, וְגַם גֶּבֶר שֶׁתּוֹקֵעַ לְנָשִׁים אֵינוֹ מְבָרֵךְ.
לְמִנְהַג הָאַשְׁכְּנַזִּים אִשָּׁה מְבָרֶכֶת. כְּשֶׁגֶּבֶר תּוֹקֵעַ לְנָשִׁים, אַחַת מֵהֶן מְבָרֶכֶת בִּשְׁבִיל כֻּלָּן.
(פֶּרֶק ד׳, הֲלָכָה ו׳)

12יְלָדִים וְשׁוֹפָר

יֶלֶד שֶׁהִגִּיעַ לְגִיל חִנּוּךְ – כְּשֶׁהוּא מֵבִין שֶׁמִּצְוָה לִשְׁמֹעַ תשר״ת, תש״ת וְתר״ת – מִצְוָה לְחַנְּכוֹ לִשְׁמֹעַ שׁוֹפָר. גַּם יֶלֶד קָטָן יוֹתֵר שֶׁמַּצְלִיחַ לַעֲמֹד בְּשֶׁקֶט – טוֹב לְהָבִיא אוֹתוֹ. אֲבָל יֶלֶד שֶׁאֵינוֹ מַצְלִיחַ לִשְׁתֹּק – אָסוּר לְהָבִיא אוֹתוֹ לְבֵית הַכְּנֶסֶת בִּזְמַן הַתְּקִיעוֹת וְהַתְּפִלָּה. וְאַחֲרֵי הַתְּקִיעוֹת? גְּדוֹלִים לֹא תּוֹקְעִים בְּלִי צֹרֶךְ, אֲבָל אֶת הַיְּלָדִים מְעוֹדְדִים לִתְקֹעַ כָּל הַיּוֹם – רַק לֹא כְּשֶׁאֲנָשִׁים יְשֵׁנִים.
🤔 לָמָּה? יֶלֶד שֶׁשּׁוֹמֵעַ שׁוֹפָר סוֹפֵג זִכְרוֹנוֹת שֶׁל קְדֻשָּׁה. וִילָדִים שֶׁמִּתְאַמְּנִים הַיּוֹם יֵדְעוּ לִתְקֹעַ כְּשֶׁיִּגְדְּלוּ.
💛 אוּלַי מִבֵּין הַיְּלָדִים שֶׁמִּתְאַמְּנִים הַיּוֹם בֶּחָצֵר יֵצֵא הַתּוֹקֵעַ שֶׁיְּעוֹרֵר אֶת כָּל הַקְּהִלָּה בְּעוֹד כַּמָּה שָׁנִים.
(פֶּרֶק ד׳, הֲלָכָה ו׳)

13שֶׁקֶט גָּמוּר

צָרִיךְ לִשְׁמֹעַ כָּל תְּקִיעָה מִתְּחִלָּתָהּ וְעַד סוֹפָהּ, וּמִי שֶׁלֹּא שָׁמַע אֶת הַהַתְחָלָה אוֹ אֶת הַסּוֹף – לֹא יָצָא. לָכֵן כָּל הַקָּהָל שׁוֹמֵר עַל שֶׁקֶט גָּמוּר, וּמִי שֶׁצָּרִיךְ לְהִשְׁתַּעֵל – מִתְאַפֵּק עַד סוֹף הַתְּקִיעוֹת. וּמִכֵּיוָן שֶׁהַבְּרָכָה שֶׁלִּפְנֵי הַתְּקִיעוֹת חָלָה גַּם עַל הַתְּקִיעוֹת שֶׁבְּמוּסָף, לֹא מְדַבְּרִים עַד שֶׁמִּסְתַּיְּמוֹת הַתְּקִיעוֹת שֶׁל מוּסָף.
🤔 לָמָּה? שִׁעוּל אוֹ דִּבּוּר בְּאֶמְצַע תְּקִיעָה יָכוֹל לִגְרֹם שֶׁמִּישֶׁהוּ לֹא יִשְׁמַע אוֹתָהּ כֻּלָּהּ, וְלֹא יְקַיֵּם אֶת הַמִּצְוָה.
(פֶּרֶק ד׳, הֲלָכָה ז׳)

14הַתּוֹקֵעַ וּמִנְהֲגֵי הַתְּקִיעוֹת

נוֹהֲגִים לְמַנּוֹת לְתוֹקֵעַ אָדָם צַדִּיק, בֶּן תּוֹרָה וּבַעַל מַעֲשִׂים טוֹבִים, אֲבָל לֹא עוֹשִׂים עַל כָּךְ מַחֲלֹקֶת. מַחֲזִיקִים אֶת הַשּׁוֹפָר כְּשֶׁהַצַּד הָרָחָב פּוֹנֶה לְמַעְלָה: ״עָלָה אֱלֹהִים בִּתְרוּעָה״. רַב אוֹ תַּלְמִיד חָכָם מַקְרִיא לַתּוֹקֵעַ אֶת הַתְּקִיעוֹת מִלָּה בְּמִלָּה, כְּדֵי שֶׁלֹּא יִטְעֶה. רַבִּים מֵהַתּוֹקְעִים לוֹקְחִים שְׁנֵי שׁוֹפָרוֹת, לְמִקְרֶה שֶׁיִּתְקַשּׁוּ לִתְקֹעַ בְּאֶחָד. אָדָם אֶחָד תּוֹקֵעַ אֶת כָּל הַתְּקִיעוֹת, וְאִם נֶחְלַשׁ – אַחֵר מַמְשִׁיךְ בִּמְקוֹמוֹ.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הָאַשְׁכְּנַזִּים מַאֲרִיכִים בַּתְּקִיעָה הָאַחֲרוֹנָה שֶׁל הַשְּׁלֹשִׁים הָרִאשׁוֹנוֹת וְשֶׁל הַמֵּאָה – ״תְּקִיעָה גְּדוֹלָה״, שֶׁרוֹמֶזֶת לֶאֱמוּנָה גְּדוֹלָה.
לְמִנְהַג הַסְּפָרַדִּים בְּסוֹף הַתְּפִלָּה מְרִיעִים עוֹד תְּרוּעָה אֲרֻכָּה, ״לְעַרְבֵּב אֶת הַשָּׂטָן״ – לִרְמֹז שֶׁגַּם הַמַּשְׁבְּרִים מִתְהַפְּכִים לְטוֹבָה.
(פֶּרֶק ד׳, הֲלָכָה ז׳)

15קִדּוּשׁ לִפְנֵי הַתְּקִיעוֹת

יֵשׁ נוֹהֲגִים, בְּעִקָּר בַּיְשִׁיבוֹת שֶׁמַּאֲרִיכִים בַּתְּפִלָּה, לְקַדֵּשׁ וְלִטְעֹם מַשֶּׁהוּ אַחֲרֵי שַׁחֲרִית, כְּדֵי לְהִתְחַזֵּק לִקְרַאת מוּסָף. אָסוּר לִקְבֹּעַ סְעוּדָה לִפְנֵי מִצְוָה, אֲבָל טְעִימָה קְטַנָּה מֻתֶּרֶת: לְכַתְּחִלָּה אוֹכְלִים מִזּוֹנוֹת כְּבֵיצָה וְלֹא יוֹתֵר. פֵּרוֹת וּמַאֲכָלִים קַלִּים אֶפְשָׁר מְעַט יוֹתֵר, אֲבָל לֹא הַרְבֵּה, כְּדֵי לֹא לְהִתְעַיֵּף. בְּמָקוֹם שֶׁלֹּא מַאֲרִיכִים בַּתְּפִלָּה – עָדִיף שֶׁלֹּא לְהַפְסִיק לְקִדּוּשׁ.
🤔 לָמָּה? חֲכָמִים חָשְׁשׁוּ שֶׁמִּי שֶׁיּוֹשֵׁב לִסְעוּדָה יִשְׁכַּח אֶת הַמִּצְוָה. בִּטְעִימָה קְטַנָּה אֵין חֲשָׁשׁ כָּזֶה, וּבִמְיֻחָד כְּשֶׁכֻּלָּם מְחַכִּים לַשּׁוֹפָר וּמַזְכִּירִים זֶה לָזֶה.
(פֶּרֶק ד׳, הֲלָכָה ח׳)

16רֹאשׁ הַשָּׁנָה שֶׁחָל בְּשַׁבָּת

מִן הַתּוֹרָה תּוֹקְעִים גַּם בְּשַׁבָּת, אֲבָל חֲכָמִים אָסְרוּ לִתְקֹעַ בְּרֹאשׁ הַשָּׁנָה שֶׁחָל בְּשַׁבָּת. בְּבֵית הַמִּקְדָּשׁ תָּקְעוּ גַּם בְּשַׁבָּת. אַחֲרֵי הַחֻרְבָּן תִּקֵּן רַבָּן יוֹחָנָן בֶּן זַכַּאי לִתְקֹעַ בַּמָּקוֹם שֶׁל בֵּית הַדִּין הַגָּדוֹל. הַיּוֹם אֵין בֵּית דִּין כָּזֶה, וְלָכֵן לֹא תּוֹקְעִים בְּשׁוּם מָקוֹם.
🤔 לָמָּה? כֻּלָּם חַיָּבִים בַּשּׁוֹפָר, וְלֹא כֻּלָּם יוֹדְעִים הֵיטֵב אֶת אִסּוּר הַטִּלְטוּל. חֲכָמִים חָשְׁשׁוּ שֶׁמִּישֶׁהוּ יִשָּׂא אֶת הַשּׁוֹפָר בִּרְשׁוּת הָרַבִּים אֶל מִי שֶׁיְּלַמֵּד אוֹתוֹ לִתְקֹעַ, וְיַעֲבֹר עַל אִסּוּר שַׁבָּת.
💛 כְּשֶׁרֹאשׁ הַשָּׁנָה חָל בְּשַׁבָּת, הוּא נִקְרָא ״זִכְרוֹן תְּרוּעָה״: זוֹכְרִים אֶת הַתְּרוּעָה וְלֹא מְרִיעִים. חַכְמֵי הַסּוֹד אָמְרוּ שֶׁקְּדֻשַּׁת הַשַּׁבָּת פּוֹעֶלֶת אָז כִּמְעַט אֶת כָּל מָה שֶׁהַתְּקִיעוֹת פּוֹעֲלוֹת.
(פֶּרֶק ד׳, הֲלָכָה ט׳, הֲלָכָה י׳)

17טָעוּת בַּתְּקִיעוֹת

״יְחִידָה״ הִיא תשר״ת אֶחָד, תש״ת אֶחָד אוֹ תר״ת אֶחָד. תּוֹקֵעַ שֶׁהִתְבַּלְבֵּל וְתָקַע בְּכַוָּנָה קוֹל זָר בְּאֶמְצַע הַיְּחִידָה – חוֹזֵר לַתְּקִיעָה הָרִאשׁוֹנָה שֶׁל אוֹתָהּ יְחִידָה, אֲבָל לְעוֹלָם אֵינוֹ חוֹזֵר לִתְחִלַּת כָּל הַסֵּדֶר. וּמִי שֶׁנִּסָּה לִתְקֹעַ כָּרָאוּי, וְיָצָא לוֹ בְּלִי כַּוָּנָה קוֹל לֹא נָכוֹן – נוֹשֵׁם וְתוֹקֵעַ שׁוּב רַק אֶת הַקּוֹל שֶׁהָיָה צָרִיךְ, וְאֵינוֹ חוֹזֵר לִתְחִלַּת הַיְּחִידָה.
🤔 לָמָּה? כָּל יְחִידָה עוֹמֶדֶת בִּפְנֵי עַצְמָהּ. וְקוֹל שֶׁיָּצָא בְּלִי כַּוָּנָה אֵינוֹ נֶחְשָׁב חֵלֶק מֵהַיְּחִידָה.
(פֶּרֶק ד׳, הֲלָכָה י״ג, הֲלָכָה י״ד)

18אֵיזֶה שׁוֹפָר?

שׁוֹפָר הוּא קֶרֶן חֲלוּלָה שֶׁגְּדֵלָה עַל רֹאשׁ שֶׁל בְּהֵמָה אוֹ חַיָּה. קֶרֶן שֶׁל פָּרָה פְּסוּלָה, כִּי הִיא נִקְרֵאת ״קֶרֶן״ וְלֹא ״שׁוֹפָר״. גַּם קַרְנֵי אַיָּלִים פְּסוּלוֹת, כִּי הֵן עֶצֶם מָלֵא וְאֵין בָּהֶן חָלָל. מִצְוָה שֶׁהַשּׁוֹפָר יִהְיֶה כָּפוּף, וּמִצְוָה מִן הַמֻּבְחָר – שׁוֹפָר כָּפוּף שֶׁל אַיִל. אִם צָרִיךְ לִבְחֹר בֵּין שׁוֹפָר כָּפוּף שֶׁאֵינוֹ שֶׁל אַיִל לְבֵין שׁוֹפָר יָשָׁר שֶׁל אַיִל – עָדִיף הַכָּפוּף. שׁוֹפָר שֶׁ״הָפְכוּ״ אוֹתוֹ בְּחִמּוּם – פָּסוּל, וְלָכֵן קוֹנִים שׁוֹפָר מֵאָדָם נֶאֱמָן.
🤔 לָמָּה? רַבִּי אַבָּהוּ אָמַר: ה׳ אָמַר ״תִּקְעוּ לְפָנַי בְּשׁוֹפָר שֶׁל אַיִל, כְּדֵי שֶׁאֶזְכֹּר לָכֶם עֲקֵדַת יִצְחָק בֶּן אַבְרָהָם״. וְהַשּׁוֹפָר הַכָּפוּף מַזְכִּיר לָנוּ לָכֹף אֶת הַלֵּב לִפְנֵי ה׳.
💛 ״וּמַעֲלֶה אֲנִי עֲלֵיכֶם כְּאִלּוּ עֲקַדְתֶּם עַצְמְכֶם לְפָנַי״. כָּל תְּקִיעָה מַזְכִּירָה אֶת הַמְּסִירוּת שֶׁל אַבְרָהָם וְיִצְחָק – וְאֶת הָאַהֲבָה שֶׁלָּנוּ.
(פֶּרֶק ד׳, הֲלָכָה ט״ו)

19שׁוֹפָר כָּשֵׁר וְשׁוֹפָר פָּסוּל

הַשּׁוֹפָר צָרִיךְ לִהְיוֹת אָרֹךְ לְפָחוֹת טֶפַח, כְּדֵי שֶׁיֵּרָאֶה מִשְּׁנֵי צִדֵּי הַיָּד שֶׁאוֹחֶזֶת בּוֹ. קוֹל גָּבוֹהַּ מְאֹד, נָמוּךְ מְאֹד אוֹ צָרוּד – כָּשֵׁר. שׁוֹפָר שֶׁהִדְבִּיקוּ לוֹ חֲתִיכָה, אֲפִלּוּ קְטַנָּה, אוֹ שֶׁהֶאֱרִיכוּ אוֹתוֹ בְּמַתֶּכֶת אוֹ בִּפְלַסְטִיק – פָּסוּל. מֻתָּר לַחֲקֹק עַל הַשּׁוֹפָר צוּרוֹת, אֲבָל לֹא מְצַיְּרִים עָלָיו, שֶׁמָּא הַצֶּבַע יְשַׁנֶּה אֶת קוֹלוֹ. וְאִם קִצְּרוּ אוֹתוֹ וְנִשְׁאַר טֶפַח – כָּשֵׁר.
🤔 לָמָּה? הַשּׁוֹפָר צָרִיךְ לִהְיוֹת אֶחָד, כְּמוֹ שֶׁבָּרָא אוֹתוֹ ה׳, בְּלִי שׁוּם תּוֹסֶפֶת, וְהַקּוֹל צָרִיךְ לָצֵאת מִן הַשּׁוֹפָר עַצְמוֹ.
(פֶּרֶק ד׳, הֲלָכָה ט״ז)

📝 מָה לָמַדְנוּ

  • הַמִּצְוָה הִיא לִשְׁמֹעַ קוֹל שׁוֹפָר, וְצָרִיךְ לְכַוֵּן לְקַיֵּם אוֹתָהּ.
  • בִּזְכוּת רַבִּי אַבָּהוּ תּוֹקְעִים שְׁלֹשִׁים: תשר״ת, תש״ת וְתר״ת, וְהַיּוֹם נוֹהֲגִים לִתְקֹעַ מֵאָה.
  • שׁוֹמְרִים עַל שֶׁקֶט גָּמוּר, כִּי צָרִיךְ לִשְׁמֹעַ כָּל תְּקִיעָה מִתְּחִלָּתָהּ וְעַד סוֹפָהּ.
  • נָשִׁים פְּטוּרוֹת וְרֹבָּן שׁוֹמְעוֹת; יְלָדִים מְחַנְּכִים לִשְׁמֹעַ, וּמְעוֹדְדִים אוֹתָם לְהִתְאַמֵּן אַחֲרֵי הַתְּקִיעוֹת.
  • הַשּׁוֹפָר הַמֻּבְחָר הוּא קֶרֶן כְּפוּפָה שֶׁל אַיִל – זֵכֶר לַעֲקֵדַת יִצְחָק.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה תִּקֵּן רַבִּי אַבָּהוּ לִתְקֹעַ שְׁלֹשִׁים תְּקִיעוֹת, וּמָה הַסֵּדֶר שֶׁלָּהֶן?
  2. דָּנִי שָׁמַע מִבֵּיתוֹ אֶת הַתְּקִיעוֹת שֶׁל בֵּית הַכְּנֶסֶת, אֲבָל לֹא חָשַׁב עַל הַמִּצְוָה. הַאִם יָצָא? לָמָּה?
  3. מָה עֲדִיף: שׁוֹפָר יָשָׁר שֶׁל אַיִל אוֹ שׁוֹפָר כָּפוּף שֶׁאֵינוֹ שֶׁל אַיִל? וְלָמָּה מְבַקְּשִׁים שׁוֹפָר שֶׁל אַיִל?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״קוֹל הַשּׁוֹפָר״

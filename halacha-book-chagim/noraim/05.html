סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים
מַהֲדוּרַת יְלָדִים

🔟 עֲשֶׂרֶת יְמֵי תְּשׁוּבָה

פֶּרֶק ה׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
לִפְנֵי שָׁבוּעַ רָבַתָּ עִם חָבֵר בַּהַפְסָקָה. אֲמַרְתֶּם מִלִּים לֹא נְעִימוֹת, וּמֵאָז אַתֶּם לֹא מְדַבְּרִים. בְּעוֹד כַּמָּה יָמִים יוֹם הַכִּפֻּרִים... הַאִם מַסְפִּיק לְהִתְפַּלֵּל חָזָק, אוֹ שֶׁצָּרִיךְ קֹדֶם לָלֶכֶת אֵלָיו? בַּפֶּרֶק הַזֶּה נִלְמַד מָה עוֹשִׂים בַּיָּמִים הַמְּיֻחָדִים שֶׁבֵּין רֹאשׁ הַשָּׁנָה לְיוֹם הַכִּפֻּרִים.

1ה׳ קָרוֹב – חֶשְׁבּוֹן נֶפֶשׁ

בַּעֲשֶׂרֶת הַיָּמִים שֶׁבֵּין רֹאשׁ הַשָּׁנָה לְיוֹם הַכִּפֻּרִים ה׳ קָרוֹב לְכָל קוֹרְאָיו, וְהַתְּשׁוּבָה מִתְקַבֶּלֶת מִיָּד. אֲפִלּוּ יָחִיד שֶׁעוֹשֶׂה תְּשׁוּבָה – מְקַבְּלִים אוֹתָהּ כְּמוֹ תְּשׁוּבָה שֶׁל צִבּוּר. לָכֵן רָאוּי לְפַשְׁפֵּשׁ בַּמַּעֲשִׂים וּלְתַקֵּן, וּבִמְיֻחָד בַּדְּבָרִים שֶׁבֵּין אָדָם לַחֲבֵרוֹ. מִנְהַג חֲסִידִים לַעֲשׂוֹת חֶשְׁבּוֹן נֶפֶשׁ בְּכָל לַיְלָה לִפְנֵי הַשֵּׁנָה, וּבַיָּמִים הָאֵלֶּה רָאוּי לְכָל אָדָם לִנְהֹג כָּךְ. נִזְהָרִים מִכָּל אִסּוּר, וּמַרְבִּים בְּתוֹרָה, בִּתְפִלָּה, בִּצְדָקָה וּבְמַעֲשִׂים טוֹבִים.
🤔 לָמָּה? הָאֲרִ״י אָמַר שֶׁשִּׁבְעַת הַיָּמִים שֶׁבֵּין רֹאשׁ הַשָּׁנָה לְיוֹם הַכִּפֻּרִים הֵם כְּנֶגֶד שִׁבְעַת יְמֵי הַשָּׁבוּעַ: בְּיוֹם רִאשׁוֹן מְתַקְּנִים אֶת מָה שֶׁפָּגַמְנוּ בְּכָל יְמֵי הָרִאשׁוֹן בַּשָּׁנָה, בְּיוֹם שֵׁנִי – אֶת יְמֵי הַשֵּׁנִי, וְכָךְ הָלְאָה.
💛 חֶשְׁבּוֹן נֶפֶשׁ לִפְנֵי הַשֵּׁנָה הוּא כְּמוֹ שִׂיחָה קְטַנָּה עִם אַבָּא בְּסוֹף הַיּוֹם: מָה הָיָה טוֹב, וּמָה אֲנִי רוֹצֶה לְתַקֵּן מָחָר.
(פֶּרֶק ה׳, הֲלָכָה א׳)

2חֶצְיוֹ זַכַּאי וְחֶצְיוֹ חַיָּב

חֲכָמִים אָמְרוּ: יִרְאֶה אָדָם אֶת עַצְמוֹ כְּאִלּוּ חֶצְיוֹ זַכַּאי וְחֶצְיוֹ חַיָּב, וְגַם אֶת כָּל הָעוֹלָם כֻּלּוֹ. עָשָׂה מִצְוָה אַחַת – אַשְׁרָיו, שֶׁהִכְרִיעַ אֶת עַצְמוֹ וְאֶת כָּל הָעוֹלָם לְכַף זְכוּת. וּבַיָּמִים הָאֵלֶּה יֵשׁ נוֹהֲגִים לְדַקְדֵּק יוֹתֵר בַּהֲלָכָה, כְּמוֹ לֶאֱכֹל בָּשָׂר ״גְּלַאט״ וּלְהַקְפִּיד עַל ״מַיִם אַחֲרוֹנִים״.
🤔 לָמָּה? כְּמוֹ מֹאזְנַיִם שֶׁשְּׁתֵּי הַכַּפּוֹת שֶׁלָּהֶם שָׁווֹת – כָּל מַעֲשֶׂה קָטָן יָכוֹל לְהַכְרִיעַ. אֵין מִצְוָה ״קְטַנָּה מִדַּי״.
💛 הַמִּצְוָה הַבָּאָה שֶׁלְּךָ יְכוֹלָה לְהַכְרִיעַ אֶת כָּל הָעוֹלָם לְטוֹבָה. כַּמָּה אֵמוּן ה׳ נוֹתֵן בְּךָ!
(פֶּרֶק ה׳, הֲלָכָה א׳)

3״הַמֶּלֶךְ הַקָּדוֹשׁ״ וְ״הַמֶּלֶךְ הַמִּשְׁפָּט״

בַּעֲשֶׂרֶת יְמֵי תְּשׁוּבָה אוֹמְרִים בָּעֲמִידָה ״הַמֶּלֶךְ הַקָּדוֹשׁ״ בִּמְקוֹם ״הָאֵל הַקָּדוֹשׁ״, וּבְבִרְכַּת ״הָשִׁיבָה שׁוֹפְטֵינוּ״ – ״הַמֶּלֶךְ הַמִּשְׁפָּט״. מִי שֶׁאָמַר ״הָאֵל הַקָּדוֹשׁ״ וְלֹא תִּקֵּן מִיָּד – חוֹזֵר לְרֹאשׁ הַתְּפִלָּה. מִי שֶׁאֵינוֹ זוֹכֵר מָה אָמַר – מִן הַסְּתָם אָמַר כְּמוֹ שֶׁהוּא רָגִיל, וְחוֹזֵר; אֲבָל אִם זָכַר לוֹמַר ״מִי כָמוֹךָ״, מִן הַסְּתָם אָמַר גַּם ״הַמֶּלֶךְ״ כָּרָאוּי. וּמִי שֶׁטָּעָה וְאָמַר ״מֶלֶךְ אוֹהֵב צְדָקָה וּמִשְׁפָּט״ – יֵשׁ בָּזֶה שְׁנֵי מִנְהָגִים:
🤔 לָמָּה? הָעוֹמֵד בִּתְפִלָּה צָרִיךְ לִרְאוֹת אֶת עַצְמוֹ כְּעוֹמֵד לִפְנֵי הַמֶּלֶךְ. בַּיָּמִים שֶׁבָּהֶם ה׳ דָּן אֶת עוֹלָמוֹ, מִי שֶׁלֹּא מַזְכִּיר אֶת מַלְכוּתוֹ – כְּאִלּוּ מִתְפַּלֵּל רַק מִן הַשָּׂפָה וְלַחוּץ.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג חֵלֶק מֵהַסְּפָרַדִּים לֹא יָצָא. אִם עוֹד לֹא סִיֵּם אֶת הַתְּפִלָּה – חוֹזֵר לְ״הָשִׁיבָה שׁוֹפְטֵינוּ״, וְאִם כְּבָר סִיֵּם – מִתְפַּלֵּל שׁוּב בִּתְנַאי: אִם אֵינוֹ חַיָּב, הֲרֵי זוֹ תְּפִלַּת נְדָבָה.
לְמִנְהַג הָאַשְׁכְּנַזִּים וְחֵלֶק מֵהַסְּפָרַדִּים יָצָא, כִּי גַּם בַּנֻּסָּח הָרָגִיל מַזְכִּירִים ״מֶלֶךְ״. כָּךְ מִנְהַג רֹב יִשְׂרָאֵל, וְכָךְ יִנְהַג מִי שֶׁאֵינוֹ יוֹדֵעַ מָה מִנְהַג אֲבוֹתָיו.
(פֶּרֶק ה׳, הֲלָכָה ב׳)

4אַרְבַּע תּוֹסָפוֹת

בַּעֲשֶׂרֶת יְמֵי תְּשׁוּבָה מוֹסִיפִים בָּעֲמִידָה אַרְבַּע תְּפִלּוֹת קְצָרוֹת: ״זָכְרֵנוּ לְחַיִּים״ בַּבְּרָכָה הָרִאשׁוֹנָה, ״מִי כָמוֹךָ״ בַּשְּׁנִיָּה, ״וּכְתֹב לְחַיִּים״ בְּ״מוֹדִים״, וּ״בְסֵפֶר חַיִּים״ בַּבְּרָכָה הָאַחֲרוֹנָה. מִי שֶׁשָּׁכַח אַחַת מֵהֶן – אֵינוֹ חוֹזֵר.
💛 בְּכָל תְּפִלָּה בַּיָּמִים הָאֵלֶּה אֲנַחְנוּ מְבַקְּשִׁים חַיִּים – ״לְמַעַנְךָ אֱלֹהִים חַיִּים״: חַיִּים שֶׁיִּהְיוּ בִּשְׁבִיל ה׳.
(פֶּרֶק ה׳, הֲלָכָה ב׳)

5תּוֹרָה וּמוּסָר

לִמּוּד תּוֹרָה הוּא יְסוֹד גָּדוֹל בִּתְשׁוּבָה, כִּי ״הַתַּלְמוּד מֵבִיא לִידֵי מַעֲשֶׂה״. רָאוּי לִלְמֹד בְּכָל יוֹם מְעַט בְּסֵפֶר מוּסָר, וּבִמְיֻחָד בְּחֹדֶשׁ אֱלוּל וּבַעֲשֶׂרֶת יְמֵי תְּשׁוּבָה – וּלְקַבֵּל, בְּלִי נֶדֶר, לְהַמְשִׁיךְ כָּךְ כָּל הַשָּׁנָה. לְלִמּוּד הַתּוֹרָה יֵשׁ כֹּחַ גָּדוֹל לְכַפֵּר עַל עֲווֹנוֹת: מִי שֶׁהָיָה רָגִיל לִלְמֹד דַּף אֶחָד – יִלְמַד שְׁנַיִם, וּמִי שֶׁהָיָה רָגִיל לִלְמֹד פֶּרֶק אֶחָד – יִלְמַד שְׁנַיִם.
🤔 לָמָּה? כְּמוֹ שֶׁמִּי שֶׁטּוֹבֵל בְּנַחַל עוֹלֶה מִטֻּמְאָה לְטָהֳרָה, כָּךְ מִי שֶׁנִּכְנָס לְאָהֳלָהּ שֶׁל תּוֹרָה וְלוֹמֵד בִּשְׁקִידָה – עוֹלֶה מִכַּף חוֹבָה לְכַף זְכוּת.
💛 כְּשֶׁאֲנִי לוֹמֵד תּוֹרָה, הַשֵּׂכֶל שֶׁלִּי מִתְחַבֵּר לָרָצוֹן שֶׁל ה׳. זֶה הַקֶּשֶׁר הַגָּבוֹהַּ בְּיוֹתֵר.
(פֶּרֶק ה׳, הֲלָכָה ג׳)

6״וּצְדָקָה תַּצִּיל מִמָּוֶת״

נָהֲגוּ יִשְׂרָאֵל לְהַרְבּוֹת בִּצְדָקָה בַּעֲשֶׂרֶת יְמֵי תְּשׁוּבָה, כְּמוֹ שֶׁכָּתוּב: ״וּצְדָקָה תַּצִּיל מִמָּוֶת״. רָאוּי לַעֲשׂוֹת חֶשְׁבּוֹן נֶפֶשׁ גַּם עַל הַחֶסֶד וְהַצְּדָקָה שֶׁלָּנוּ, וּלְהִתְחַזֵּק בְּמַעֲשַׂר כְּסָפִים. וּמִי שֶׁאֵינוֹ יוֹדֵעַ לִלְמֹד – יַעֲסֹק בִּצְדָקָה, וְיִחְיֶה.
🤔 לָמָּה? חֲכָמִים סִפְּרוּ עַל בְּנֵי בֵּית עֵלִי הַכֹּהֵן, שֶׁנִּגְזַר עֲלֵיהֶם שֶׁיָּמוּתוּ צְעִירִים. אֲבָל אִם יַעַסְקוּ בַּתּוֹרָה – יִחְיוּ, וְאִם יוֹסִיפוּ גַּם גְּמִילוּת חֲסָדִים – יַאֲרִיכוּ יָמִים יוֹתֵר.
💛 גַּם מַטְבֵּעַ קָטָן שֶׁיֶּלֶד נוֹתֵן לִצְדָקָה הוּא חֶסֶד אֲמִתִּי. מִי שֶׁמְּרַחֵם עַל אֲחֵרִים – ה׳ מְרַחֵם עָלָיו.
(פֶּרֶק ה׳, הֲלָכָה ג׳)

7בֵּין אָדָם לַחֲבֵרוֹ

חֲכָמִים אָמְרוּ: ״עֲבֵרוֹת שֶׁבֵּין אָדָם לַמָּקוֹם – יוֹם הַכִּפֻּרִים מְכַפֵּר. עֲבֵרוֹת שֶׁבֵּין אָדָם לַחֲבֵרוֹ – אֵין יוֹם הַכִּפֻּרִים מְכַפֵּר, עַד שֶׁיְּרַצֶּה אֶת חֲבֵרוֹ״. מִי שֶׁפָּגַע בְּחָבֵר – צָרִיךְ לְפַיֵּס אוֹתוֹ מַהֵר, וּלְפָחוֹת לִפְנֵי יוֹם הַכִּפֻּרִים. קֹדֶם מְשַׁלְּמִים לֶחָבֵר עַל מָה שֶׁהִזִּיקוּ לוֹ וּמְבַקְּשִׁים מִמֶּנּוּ סְלִיחָה, וְרַק אַחַר כָּךְ מִתְוַדִּים לִפְנֵי ה׳.
🤔 לָמָּה? מִי שֶׁפּוֹגֵעַ בְּחָבֵר חוֹטֵא גַּם לַחָבֵר וְגַם לַה׳. כָּל עוֹד הֶחָבֵר נִשְׁאָר בְּצַעֲרוֹ, הַחֵטְא גָּדֵל. אֲפִלּוּ כָּל הַקָּרְבָּנוֹת וְהַתְּפִלּוֹת שֶׁבָּעוֹלָם לֹא יוֹעִילוּ בְּלִי לְפַיֵּס אֶת הֶחָבֵר.
💛 ה׳ אוֹהֵב כָּל אֶחָד מִבָּנָיו. כְּשֶׁאֲנִי מְפַיֵּס אֶת חֲבֵרִי, אֲנִי מְשַׂמֵּחַ גַּם אֶת אָבִינוּ שֶׁבַּשָּׁמַיִם.
(פֶּרֶק ה׳, הֲלָכָה ד׳)

8לְתַקֵּן לָשׁוֹן הָרָע

מִי שֶׁדִּבֵּר בִּגְנוּת חֲבֵרוֹ – צָרִיךְ לְדַבֵּר בְּשִׁבְחוֹ לִפְנֵי כָּל מִי שֶׁשָּׁמַע אֶת הַגְּנוּת, וּלְהַסְבִּיר שֶׁלֹּא הִכִּיר אֶת כָּל הַתְּמוּנָה. מִי שֶׁהֶעֱלִיב חָבֵר לִפְנֵי רַבִּים – נָכוֹן שֶׁיְּבַקֵּשׁ מִמֶּנּוּ סְלִיחָה לִפְנֵי רַבִּים. כְּשֶׁמְּבַקְּשִׁים סְלִיחָה, מְפָרְטִים אֶת הַחֵטְא. אֲבָל אִם הַפֵּרוּט יְצַעֵר אוֹ יְבַיֵּשׁ אֶת הֶחָבֵר – עָדִיף שֶׁלֹּא לְפָרֵט.
🤔 לָמָּה? הַמִּלִּים הָרָעוֹת מַמְשִׁיכוֹת לְהַשְׁפִּיעַ גַּם אַחֲרֵי שֶׁנֶּאֶמְרוּ. רַק מִלִּים טוֹבוֹת יְכוֹלוֹת לְבַטֵּל אֶת הַהַשְׁפָּעָה שֶׁלָּהֶן.
(פֶּרֶק ה׳, הֲלָכָה ד׳, הֲלָכָה ה׳)

9אֵיךְ מְבַקְּשִׁים סְלִיחָה?

לְכַתְּחִלָּה מְבַקְּשִׁים סְלִיחָה בְּעַצְמֵנוּ. מִי שֶׁמִּתְבַּיֵּשׁ, אוֹ חוֹשֵׁב שֶׁדַּוְקָא עַל יְדֵי שָׁלִיחַ חֲבֵרוֹ יִתְרַצֶּה יוֹתֵר – יָכוֹל לִשְׁלֹחַ שָׁלִיחַ. אִם הֶחָבֵר לֹא סָלַח, הוֹלְכִים אֵלָיו עִם שְׁלֹשָׁה אֲנָשִׁים, עַד שָׁלֹשׁ פְּעָמִים. אִם גַּם אָז לֹא סָלַח – שׁוּב אֵין צֹרֶךְ לְהִתְאַמֵּץ, וּמִי שֶׁלֹּא מָחַל הוּא הַחוֹטֵא. וּמִי שֶׁפָּגַע בְּרַבּוֹ – הוֹלֵךְ אֵלָיו אֲפִלּוּ אֶלֶף פְּעָמִים, עַד שֶׁיִּמְחַל.
🤔 לָמָּה? יֵשׁ שֶׁמְּבַקְּשִׁים לִפְנֵי יוֹם הַכִּפֻּרִים סְלִיחָה כְּלָלִית מִכָּל הַמַּכִּירִים – וְזֶה כִּמְעַט לֹא מוֹעִיל. הָעִקָּר הוּא לַעֲשׂוֹת חֶשְׁבּוֹן נֶפֶשׁ, לְהִזָּכֵר בְּמִי בֶּאֱמֶת פָּגַעְנוּ, וּמִמֶּנּוּ לְבַקֵּשׁ סְלִיחָה.
(פֶּרֶק ה׳, הֲלָכָה ה׳)

10לִסְלֹחַ בְּלֵב שָׁלֵם

רָאוּי לְמִי שֶׁנִּפְגַּע שֶׁלֹּא יִהְיֶה אַכְזָרִי, וְיִסְלַח לְמִי שֶׁמְּבַקֵּשׁ מִמֶּנּוּ סְלִיחָה. לִפְעָמִים אֶפְשָׁר לְחַכּוֹת מְעַט לִפְנֵי שֶׁסּוֹלְחִים, כְּדֵי שֶׁהַפּוֹגֵעַ יִלְמַד לְהִזָּהֵר בַּפַּעַם הַבָּאָה. אֲבָל גַּם אָז, מֵרֶגַע שֶׁבִּקֵּשׁ סְלִיחָה, מוֹצִיאִים מֵהַלֵּב אֶת הַשִּׂנְאָה וְהַכַּעַס.
🤔 לָמָּה? חֲכָמִים אָמְרוּ: ״כָּל הַמַּעֲבִיר עַל מִדּוֹתָיו – מַעֲבִירִים לוֹ עַל כָּל פְּשָׁעָיו״. מִי שֶׁסּוֹלֵחַ לַאֲחֵרִים – גַּם בַּשָּׁמַיִם סוֹלְחִים לוֹ.
💛 כְּשֶׁאֲנִי סוֹלֵחַ, אֲנִי הוֹלֵךְ בִּדְרָכָיו שֶׁל ה׳ – ״אֵל רַחוּם וְחַנּוּן״.
(פֶּרֶק ה׳, הֲלָכָה ה׳)

11אָבִינוּ מַלְכֵּנוּ

בַּעֲשֶׂרֶת יְמֵי תְּשׁוּבָה אוֹמְרִים ״אָבִינוּ מַלְכֵּנוּ״ בְּשַׁחֲרִית וּבְמִנְחָה, אַחֲרֵי חֲזָרַת הַשַּׁ״ץ. אוֹמְרִים אוֹתָהּ בַּעֲמִידָה, וְרַבִּים פּוֹתְחִים אֶת אֲרוֹן הַקֹּדֶשׁ. גַּם מִי שֶׁמִּתְפַּלֵּל לְבַד יָכוֹל לוֹמַר ״אָבִינוּ מַלְכֵּנוּ״.
🤔 לָמָּה? בַּתְּפִלָּה הַזֹּאת פּוֹנִים לַה׳ בִּשְׁתֵּי דְּרָכִים: כְּבָנִים אֶל אֲבִיהֶם וְכַעֲבָדִים אֶל מַלְכָּם. פַּעַם לֹא יָרְדוּ גְּשָׁמִים, וְרַק כְּשֶׁרַבִּי עֲקִיבָא עָמַד לִפְנֵי הַתֵּבָה וְאָמַר: ״אָבִינוּ מַלְכֵּנוּ, אֵין לָנוּ מֶלֶךְ אֶלָּא אָתָּה; אָבִינוּ מַלְכֵּנוּ, לְמַעַנְךָ רַחֵם עָלֵינוּ״ – יָרְדוּ גְּשָׁמִים.
💛 לְאַבָּא שֶׁהוּא גַּם מֶלֶךְ אֶפְשָׁר לְבַקֵּשׁ הַכֹּל – כִּי הוּא גַּם אוֹהֵב וְגַם יָכוֹל.
(פֶּרֶק ה׳, הֲלָכָה ו׳)

12שַׁבַּת שׁוּבָה

הַשַּׁבָּת שֶׁבְּתוֹךְ עֲשֶׂרֶת יְמֵי תְּשׁוּבָה נִקְרֵאת ״שַׁבַּת שׁוּבָה״, עַל שֵׁם הַהַפְטָרָה: ״שׁוּבָה יִשְׂרָאֵל עַד ה׳ אֱלֹהֶיךָ״. בַּשַּׁבָּת הַזֹּאת רַב הַמָּקוֹם דּוֹרֵשׁ וּמְעוֹרֵר אֶת הָעָם לִתְשׁוּבָה, לְתוֹרָה וְלִצְדָקָה. לַדְּרָשָׁה בָּאִים כֻּלָּם – גַּם גְּדוֹלִים שֶׁיּוֹדְעִים אֶת כָּל הַתּוֹרָה, וְגַם קְטַנִּים שֶׁעוֹד לֹא מְבִינִים אֶת דִּבְרֵי הָרַב. בְּ״אָבִינוּ מַלְכֵּנוּ״ בַּשַּׁבָּת יֵשׁ שְׁנֵי מִנְהָגִים:
🤔 לָמָּה? הַשַּׁבָּת הִיא מְקוֹר הַבְּרָכָה שֶׁל כָּל הַשָּׁבוּעַ, וְלָכֵן שַׁבַּת שׁוּבָה הִיא כְּמוֹ הַשֹּׁרֶשׁ שֶׁל יוֹם הַכִּפֻּרִים.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הָאַשְׁכְּנַזִּים, הַתֵּימָנִים וְחֵלֶק מֵהַסְּפָרַדִּים לֹא אוֹמְרִים ״אָבִינוּ מַלְכֵּנוּ״ בְּשַׁבָּת וּבְמִנְחָה שֶׁל עֶרֶב שַׁבָּת, כִּי לֹא מְבַקְּשִׁים צְרָכִים בַּשַּׁבָּת.
לְמִנְהַג רַבִּים מִמִּתְפַּלְּלֵי נֻסַּח סְפָרַד אוֹמְרִים ״אָבִינוּ מַלְכֵּנוּ״ גַּם בְּשַׁבַּת שׁוּבָה, כִּי הַיָּמִים הָאֵלֶּה מְיֻחָדִים לִתְשׁוּבָה וְלִתְפִלָּה.
(פֶּרֶק ה׳, הֲלָכָה ז׳)

13קִדּוּשׁ לְבָנָה

אֶת בִּרְכַּת הַלְּבָנָה מְבָרְכִים בְּשִׂמְחָה. בְּחֹדֶשׁ תִּשְׁרֵי יֵשׁ שְׁנֵי מִנְהָגִים מָתַי לְבָרֵךְ אוֹתָהּ, וְכָל קְהִלָּה מַמְשִׁיכָה בְּמִנְהָגָהּ.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
רַבִּים נוֹהֲגִים לִדְחוֹת אוֹתָהּ לְמוֹצָאֵי יוֹם הַכִּפֻּרִים, כִּי רַק אָז הַמֶּתַח שֶׁל יְמֵי הַדִּין מִתְפּוֹגֵג, וְיוֹצְאִים מִבֵּית הַכְּנֶסֶת בְּשִׂמְחָה.
וְיֵשׁ אוֹמְרִים שֶׁאַדְּרַבָּה, טוֹב לְבָרֵךְ אוֹתָהּ בְּתוֹךְ עֲשֶׂרֶת יְמֵי תְּשׁוּבָה, כְּדֵי שֶׁהַמִּצְוָה הַזֹּאת תַּכְרִיעַ לְכַף זְכוּת.
(פֶּרֶק ה׳, הֲלָכָה ז׳)

14כַּפָּרוֹת

יֵשׁ נוֹהֲגִים בְּעֶרֶב יוֹם הַכִּפֻּרִים לָקַחַת תַּרְנְגוֹל לְכָל אֶחָד מִבְּנֵי הַבַּיִת – תַּרְנְגוֹל לְכָל בֵּן וְתַרְנְגֹלֶת לְכָל בַּת – לְסוֹבֵב אוֹתוֹ מֵעַל הָרֹאשׁ וְלוֹמַר: ״זֶה חֲלִיפָתִי, זֶה תְּמוּרָתִי, זֶה כַּפָּרָתִי״, וְלִשְׁחֹט אוֹתוֹ. אֶת הַבָּשָׂר אוֹ אֶת שָׁוְיוֹ נוֹתְנִים לָעֲנִיִּים. רַבִּים עוֹשִׂים הַיּוֹם כַּפָּרוֹת בְּכֶסֶף לִצְדָקָה, בְּשׁוֹוִי שֶׁל תַּרְנְגוֹל לְכָל אֶחָד מִבְּנֵי הַבַּיִת. כָּל אֶחָד רַשַּׁאי לִבְחֹר אֶת מִנְהָגוֹ, וְטוֹב לְהַמְשִׁיךְ בְּמִנְהַג הַמִּשְׁפָּחָה. וּבְכָל הַמִּנְהָגִים – מַרְבִּים בִּצְדָקָה לִקְרַאת יוֹם הַכִּפֻּרִים.
🤔 לָמָּה? מִי שֶׁנּוֹהֵג בְּכַפָּרוֹת חוֹשֵׁב: מָה שֶׁהָיָה רָאוּי לָבוֹא עָלַי – בִּזְכוּת הַתְּשׁוּבָה ה׳ מְסַלֵּק מִמֶּנִּי. וְהַבָּשָׂר שֶׁנִּתָּן לָעֲנִיִּים נוֹתֵן לָהֶם מָה לֶאֱכֹל לִפְנֵי הַצּוֹם וְאַחֲרָיו.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְפִי הַשֻּׁלְחָן עָרוּךְ אֵין לִנְהֹג בְּכַפָּרוֹת, כִּי זֶה דּוֹמֶה לְמִנְהַג גּוֹיִם. הַנּוֹהֲגִים כָּךְ נוֹתְנִים צְדָקָה, בְּלִי לְסוֹבֵב אֶת הַכֶּסֶף מֵעַל הָרֹאשׁ.
לְפִי הָרַמָּ״א וְהָאֲרִ״י זֶה מִנְהַג וָתִיקִין, וְאֵין לְבַטְּלוֹ. בְּעִקְבוֹת הָאֲרִ״י נוֹהֲגִים בּוֹ רַבִּים מִיּוֹצְאֵי סְפָרַד וְאַשְׁכְּנַז – בְּתַרְנְגוֹל אוֹ בְּכֶסֶף.
(פֶּרֶק ה׳, הֲלָכָה ח׳)

15אוֹכְלִים בְּעֶרֶב יוֹם הַכִּפֻּרִים

מִצְוָה לְהַרְבּוֹת בַּאֲכִילָה וּבִשְׁתִיָּה בְּעֶרֶב יוֹם הַכִּפֻּרִים, יוֹתֵר מִבְּכָל יוֹם. חֲכָמִים אָמְרוּ: ״כָּל הָאוֹכֵל וְשׁוֹתֶה בַּתְּשִׁיעִי – מַעֲלֶה עָלָיו הַכָּתוּב כְּאִלּוּ הִתְעַנָּה תְּשִׁיעִי וַעֲשִׂירִי״. לְכַתְּחִלָּה קוֹבְעִים לְפָחוֹת סְעוּדָה אַחַת עַל לֶחֶם, וְרַבִּים קוֹבְעִים שְׁתַּיִם. אוֹכְלִים מַאֲכָלִים קַלִּים לְעִכּוּל, כְּדֵי לְהִתְפַּלֵּל בְּיוֹם הַכִּפֻּרִים בְּדַעַת צְלוּלָה. גַּם מִי שֶׁפָּטוּר מֵהַצּוֹם – אוֹכֵל בְּעֶרֶב יוֹם הַכִּפֻּרִים לִכְבוֹד הַמִּצְוָה.
🤔 לָמָּה? כְּדֵי שֶׁיִּהְיֶה לָנוּ כֹּחַ לָצוּם וּלְהִתְפַּלֵּל. וְעוֹד: עַל הַתְּשׁוּבָה רָאוּי לִשְׂמֹחַ, אֲבָל בְּיוֹם הַכִּפֻּרִים עַצְמוֹ אֵין אוֹכְלִים, וְלָכֵן אֶת הַשִּׂמְחָה הַזֹּאת מְבַטְּאִים בַּסְּעוּדָה שֶׁלִּפְנֵי הַצּוֹם.
💛 ה׳ צִוָּה לָצוּם רַק יוֹם אֶחָד בַּשָּׁנָה, וּמֵרֹב דְּאָגָתוֹ לָנוּ צִוָּה גַּם לֶאֱכֹל לְפָנָיו – כְּדֵי שֶׁלֹּא יִקְרֶה לָנוּ שׁוּם נֵזֶק.
(פֶּרֶק ה׳, הֲלָכָה ט׳)

16מִנְהֲגֵי עֶרֶב יוֹם הַכִּפֻּרִים

בְּעֶרֶב יוֹם הַכִּפֻּרִים לֹא אוֹמְרִים ״תַּחֲנוּן״, כִּי הוּא קְצָת יוֹם טוֹב. רַבִּים טוֹבְלִים בְּמִקְוֶה, בְּלִי בְּרָכָה. מִתְפַּלְּלִים מִנְחָה לִפְנֵי הַסְּעוּדָה הַמַּפְסֶקֶת, וְאוֹמְרִים וִדּוּי בְּסוֹף הָעֲמִידָה. וּבְמִשְׁפָּחוֹת רַבּוֹת הַהוֹרִים מְבָרְכִים אֶת יַלְדֵיהֶם לִפְנֵי שֶׁהוֹלְכִים לְבֵית הַכְּנֶסֶת. בְּ״אָבִינוּ מַלְכֵּנוּ״ יֵשׁ שְׁנֵי מִנְהָגִים:
🤔 לָמָּה? חֲכָמִים חָשְׁשׁוּ שֶׁמָּא יִקְרֶה לְאָדָם מַשֶּׁהוּ בַּסְּעוּדָה, וְלֹא יוּכַל לְהִתְוַדּוֹת בְּלֵיל יוֹם הַכִּפֻּרִים. לָכֵן מַקְדִּימִים וִדּוּי כְּבָר בְּמִנְחָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הָאַשְׁכְּנַזִּים לֹא אוֹמְרִים ״אָבִינוּ מַלְכֵּנוּ״ בְּעֶרֶב יוֹם הַכִּפֻּרִים. רַק כְּשֶׁיּוֹם הַכִּפֻּרִים חָל בְּשַׁבָּת, וְלָכֵן לֹא אוֹמְרִים אוֹתוֹ בּוֹ, אוֹמְרִים ״אָבִינוּ מַלְכֵּנוּ״ בְּשַׁחֲרִית שֶׁל הָעֶרֶב.
לְמִנְהַג הַסְּפָרַדִּים אוֹמְרִים ״אָבִינוּ מַלְכֵּנוּ״ בְּעֶרֶב יוֹם הַכִּפֻּרִים בְּכָל מִקְרֶה.
💛 בִּרְכַּת אַבָּא וְאִמָּא לִפְנֵי יוֹם הַכִּפֻּרִים מַזְכִּירָה לַיֶּלֶד: אַתָּה נִכְנָס לַיּוֹם הַקָּדוֹשׁ מְבֹרָךְ וְאָהוּב.
(פֶּרֶק ה׳, הֲלָכָה י׳)

17הַתָּרַת נְדָרִים – וּ״בְלִי נֶדֶר״

בְּעֶרֶב רֹאשׁ הַשָּׁנָה נוֹהֲגִים לְהַתִּיר נְדָרִים: עוֹמְדִים לִפְנֵי שְׁלֹשָׁה גְּבָרִים, וְהֵם אוֹמְרִים שָׁלֹשׁ פְּעָמִים ״מֻתָּר לָךְ״ אוֹ ״מָחוּל לָךְ״. אֶפְשָׁר שֶׁהַשְּׁלֹשָׁה יִהְיוּ קְרוֹבִים – אֲפִלּוּ שְׁלֹשָׁה אַחִים. מִנְהָג טוֹב שֶׁאָדָם נָהַג שָׁלֹשׁ פְּעָמִים בְּלִי לוֹמַר ״בְּלִי נֶדֶר״ – נֶחְשָׁב כְּנֶדֶר, וְהַהַתָּרָה מוֹעִילָה לוֹ. אֲבָל נֶדֶר שֶׁאָדָם זוֹכֵר – צָרִיךְ לְפָרֵט לִפְנֵי שְׁלֹשָׁה. נָשִׁים וּבָנוֹת יוֹצְאוֹת יְדֵי הַתָּרַת נְדָרִים בְּ״כָל נִדְרֵי״, וְלָכֵן נָהֲגוּ לְהַקְפִּיד לָבוֹא לְבֵית הַכְּנֶסֶת לְ״כָל נִדְרֵי״.
🤔 לָמָּה? רוֹצִים לְהִכָּנֵס לַיָּמִים הַנּוֹרָאִים נְקִיִּים מִן הָעָווֹן הֶחָמוּר שֶׁל נֶדֶר שֶׁלֹּא קִיַּמְנוּ. וּבַהַתָּרָה מוֹסְרִים גַּם ״מוֹדָעָה״ – שֶׁהַקַּבָּלוֹת הַטּוֹבוֹת שֶׁנְּקַבֵּל בַּשָּׁנָה לֹא יֵחָשְׁבוּ כִּנְדָרִים.
💛 כְּשֶׁאֲנִי מְקַבֵּל עַל עַצְמִי מַשֶּׁהוּ טוֹב, אֲנִי אוֹמֵר ״בְּלִי נֶדֶר״. כָּךְ אֲנִי מוֹסִיף מִצְווֹת מִתּוֹךְ שִׂמְחָה, בְּלִי לְהִסְתַּבֵּךְ.
(פֶּרֶק ה׳, הֲלָכָה י״א, הֲלָכָה י״ב)

📝 מָה לָמַדְנוּ

  • בַּעֲשֶׂרֶת יְמֵי תְּשׁוּבָה ה׳ קָרוֹב, וְעוֹשִׂים חֶשְׁבּוֹן נֶפֶשׁ, לוֹמְדִים תּוֹרָה וּמַרְבִּים בִּצְדָקָה.
  • אוֹמְרִים ״הַמֶּלֶךְ הַקָּדוֹשׁ״ וְ״הַמֶּלֶךְ הַמִּשְׁפָּט״, וּמוֹסִיפִים ״זָכְרֵנוּ לְחַיִּים״ וְעוֹד שָׁלֹשׁ תּוֹסָפוֹת.
  • יוֹם הַכִּפֻּרִים לֹא מְכַפֵּר עַל עֲבֵרוֹת שֶׁבֵּין אָדָם לַחֲבֵרוֹ, עַד שֶׁמְּבַקְּשִׁים סְלִיחָה וּמְפַיְּסִים.
  • מִי שֶׁמְּבַקְּשִׁים מִמֶּנּוּ סְלִיחָה – סוֹלֵחַ בְּלֵב שָׁלֵם: ״כָּל הַמַּעֲבִיר עַל מִדּוֹתָיו, מַעֲבִירִים לוֹ עַל כָּל פְּשָׁעָיו״.
  • בְּעֶרֶב יוֹם הַכִּפֻּרִים מִצְוָה לֶאֱכֹל, וְיֵשׁ בּוֹ מִנְהָגִים: כַּפָּרוֹת, וִדּוּי בְּמִנְחָה וּבִרְכַּת הַיְּלָדִים.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה יוֹם הַכִּפֻּרִים לֹא מְכַפֵּר עַל עֲבֵרוֹת שֶׁבֵּין אָדָם לַחֲבֵרוֹ, וּמָה עוֹשִׂים קֹדֶם – מְבַקְּשִׁים סְלִיחָה מֵהֶחָבֵר אוֹ מִתְוַדִּים לִפְנֵי ה׳?
  2. מָה עוֹשֶׂה מִי שֶׁבִּקֵּשׁ סְלִיחָה וְהֶחָבֵר לֹא סָלַח לוֹ?
  3. מָה פֵּרוּשׁ ״חֶצְיוֹ זַכַּאי וְחֶצְיוֹ חַיָּב״, וּמָה זֶה מְלַמֵּד אוֹתָנוּ עַל כָּל מִצְוָה שֶׁאֲנַחְנוּ עוֹשִׂים?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״עֲשֶׂרֶת יְמֵי תְּשׁוּבָה״

סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים
מַהֲדוּרַת יְלָדִים

🍎 רֹאשׁ הַשָּׁנָה

פֶּרֶק ג׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
הַשֻּׁלְחָן עָרוּךְ בְּמַפָּה לְבָנָה. יֵשׁ עָלָיו תַּפּוּחַ וּצְלוֹחִית דְּבַשׁ, רִמּוֹן אָדֹם, תְּמָרִים וְחַלָּה עֲגֻלָּה. כֻּלָּם לְבוּשִׁים יָפֶה וּמְחַיְּכִים, וּבְכָל זֹאת יֵשׁ בָּאֲוִיר מַשֶּׁהוּ רְצִינִי. אֵיךְ יָכוֹל יוֹם לִהְיוֹת גַּם חַג שָׂמֵחַ וְגַם יוֹם דִּין? בַּפֶּרֶק הַזֶּה נִלְמַד אֵיךְ מַמְלִיכִים אֶת ה׳ בְּרֹאשׁ הַשָּׁנָה – בַּתְּפִלָּה, בַּסְּעוּדָה וּבַהִתְנַהֲגוּת שֶׁלָּנוּ.

1יוֹם הַזִּכָּרוֹן

בְּרֹאשׁ הַשָּׁנָה ה׳ בּוֹרֵא אֶת הַשָּׁנָה הַחֲדָשָׁה וְנוֹתֵן חַיִּים חֲדָשִׁים לְכָל בְּרוּאָיו. לָכֵן בַּתְּפִלָּה וּבַקִּדּוּשׁ הוּא נִקְרָא ״יוֹם הַזִּכָּרוֹן״: ה׳ זוֹכֵר אֶת כָּל הַבְּרוּאִים וְאֶת כָּל מַעֲשֵׂיהֶם, וְיוֹשֵׁב עַל כִּסֵּא הַדִּין. וּבַיּוֹם שֶׁבּוֹ ה׳ זוֹכֵר אוֹתָנוּ, גַּם אֲנַחְנוּ צְרִיכִים לִזְכֹּר: לִזְכֹּר שֶׁה׳ בָּרָא אֶת הָעוֹלָם, וּלְקַבֵּל עָלֵינוּ אֶת מַלְכוּתוֹ.
🤔 לָמָּה? רֹאשׁ הַשָּׁנָה הוּא הֶחָג הַיָּחִיד שֶׁחָל בְּרֹאשׁ חֹדֶשׁ, כְּשֶׁהַלְּבָנָה עוֹד מְכֻסָּה. לָכֵן קוֹרְאִים לוֹ גַּם ״כֶּסֶה״ – יוֹם נִסְתָּר, שֶׁהַסּוֹדוֹת שֶׁלּוֹ מִתְגַּלִּים לְאַט לְאַט.
💛 כְּשֶׁה׳ ״זוֹכֵר״ אוֹתְךָ, הוּא פּוֹקֵד אוֹתְךָ וְנוֹתֵן לְךָ חַיִּים. אַתָּה חָשׁוּב לוֹ.
(פֶּרֶק ג׳, הֲלָכָה א׳)

2יוֹם תְּרוּעָה

הַתּוֹרָה קוֹרֵאת לְרֹאשׁ הַשָּׁנָה ״יוֹם תְּרוּעָה״. הַתְּקִיעָה הִיא קוֹל שֶׁל שִׂמְחָה, וְהַתְּרוּעָה הִיא קוֹל שֶׁל בֶּכִי וְשֶׁבֶר. כָּךְ הָיָה גַּם בַּמִּדְבָּר: כְּשֶׁהִתְאַסְּפוּ יַחַד – תָּקְעוּ בַּחֲצוֹצְרוֹת, וּכְשֶׁיָּצְאוּ לְמִלְחָמָה אוֹ לְמַסָּע – הֵרִיעוּ. וּבְכָל זֹאת, לִפְנֵי כָּל תְּרוּעָה וְאַחֲרֶיהָ תּוֹקְעִים תְּקִיעָה.
🤔 לָמָּה? בְּרֹאשׁ הַשָּׁנָה הַשָּׁנָה הַקּוֹדֶמֶת נִגְמְרָה, וְהַחַיִּים שֶׁל הַשָּׁנָה הַחֲדָשָׁה עוֹד לֹא נִקְבְּעוּ – וְלָכֵן הַלֵּב רוֹעֵד. אֲבָל הַתְּקִיעוֹת שֶׁמַּקִּיפוֹת אֶת הַתְּרוּעָה מְלַמְּדוֹת שֶׁגַּם הַדִּין מְכֻוָּן לְטוֹבָה.
💛 יֵשׁ בֶּכִי שֶׁל עֶצֶב, וְיֵשׁ בֶּכִי שֶׁל גַּעְגּוּעִים. בְּרֹאשׁ הַשָּׁנָה הַלֵּב ״בּוֹכֶה״ מִגַּעְגּוּעִים לִהְיוֹת קָרוֹב יוֹתֵר לַה׳.
(פֶּרֶק ג׳, הֲלָכָה ב׳)

3רֹאשׁ לְכָל הַשָּׁנָה

חֲכָמִים נֶחְלְקוּ בְּאֵיזֶה יוֹם נִבְרָא הָעוֹלָם: לְדַעַת רַבִּי יְהוֹשֻׁעַ – בְּא׳ בְּנִיסָן, וּלְדַעַת רַבִּי אֱלִיעֶזֶר – בְּא׳ בְּתִשְׁרֵי. אָנוּ נוֹהֲגִים כְּרַבִּי אֱלִיעֶזֶר, וְלָכֵן אוֹמְרִים בַּתְּפִלָּה: ״זֶה הַיּוֹם תְּחִלַּת מַעֲשֶׂיךָ״. וְהַכֹּל מַסְכִּימִים שֶׁבְּא׳ בְּתִשְׁרֵי ה׳ דָּן אֶת עוֹלָמוֹ. בְּרֹאשׁ הַשָּׁנָה עוֹבְרִים גַּם לַשָּׁנָה הַבָּאָה בִּסְפִירַת הַשָּׁנִים, בְּדִינֵי שְׁמִטָּה וּבְמַעַשְׂרוֹת.
🤔 לָמָּה? הוּא נִקְרָא ״רֹאשׁ״ הַשָּׁנָה, כִּי כָּל מָה שֶׁקּוֹרֶה בְּמֶשֶׁךְ הַשָּׁנָה נִגְרָר אַחֲרָיו, כְּמוֹ שֶׁהַגּוּף הוֹלֵךְ אַחֲרֵי הָרֹאשׁ.
(פֶּרֶק ג׳, הֲלָכָה ג׳)

4חַג – וְגַם יוֹם דִּין

רֹאשׁ הַשָּׁנָה הוּא חַג, ״מִקְרָא קֹדֶשׁ״. מִצְוָה לַעֲרֹךְ בּוֹ שְׁתֵּי סְעוּדוֹת חֲגִיגִיּוֹת, בַּלַּיְלָה וּבַיּוֹם, וְלִשְׂמֹחַ בָּהֶן בְּבָשָׂר וְיַיִן. אֲבָל לֹא אוֹכְלִים עַד הַשֹּׂבַע, כְּדֵי שֶׁיִּרְאַת ה׳ תִּהְיֶה עַל פָּנֵינוּ. וּמִשּׁוּם שֶׁזֶּה יוֹם דִּין, לֹא אוֹמְרִים בּוֹ ״מוֹעֲדִים לְשִׂמְחָה״ כְּמוֹ בִּשְׁאָר הַחַגִּים, וְגַם לֹא אוֹמְרִים הַלֵּל.
🤔 לָמָּה? מַלְאֲכֵי הַשָּׁרֵת שָׁאֲלוּ: לָמָּה יִשְׂרָאֵל לֹא אוֹמְרִים שִׁירָה בְּרֹאשׁ הַשָּׁנָה? עָנָה לָהֶם ה׳: ״אֶפְשָׁר – מֶלֶךְ יוֹשֵׁב עַל כִּסֵּא דִּין, וְסִפְרֵי חַיִּים וְסִפְרֵי מֵתִים פְּתוּחִים לְפָנָיו, וְיִשְׂרָאֵל אוֹמְרִים שִׁירָה?!״
(פֶּרֶק ג׳, הֲלָכָה ד׳)

5בְּגָדִים נָאִים

לוֹבְשִׁים בְּרֹאשׁ הַשָּׁנָה בְּגָדִים מְכֻבָּדִים, נָאִים וּנְקִיִּים. אֲבָל לֹא לוֹבְשִׁים אֶת הַבְּגָדִים הַיָּפִים בְּיוֹתֵר, כְּמוֹ בִּשְׁאָר הַחַגִּים, מִפְּנֵי מוֹרָא הַדִּין. יֵשׁ נוֹהֲגִים לִלְבֹּשׁ בְּגָדִים לְבָנִים. מִי שֶׁשְּׂעָרוֹ אָרֹךְ מִדַּי – מִסְתַּפֵּר לִכְבוֹד הַחַג.
🤔 לָמָּה? הַבְּגָדִים מְבַטְּאִים אֶת שְׁנֵי הַצְּדָדִים שֶׁל הַיּוֹם: נָאִים – כִּי זֶה חַג, וְלֹא הַיָּפִים בְּיוֹתֵר – כִּי זֶה יוֹם דִּין.
(פֶּרֶק ג׳, הֲלָכָה ד׳)

6שְׂמֵחִים בְּיוֹם הַדִּין

בִּימֵי עֶזְרָא וּנְחֶמְיָה הִתְאַסֵּף כָּל הָעָם בְּרֹאשׁ הַשָּׁנָה, וְעֶזְרָא קָרָא לִפְנֵיהֶם בַּתּוֹרָה. כְּשֶׁהֵבִינוּ שֶׁנִּכְשְׁלוּ בַּעֲבֵרוֹת, הִתְחִילוּ לִבְכּוֹת. אָמְרוּ לָהֶם עֶזְרָא וּנְחֶמְיָה: ״הַיּוֹם קָדֹשׁ הוּא לַה׳ אֱלֹהֵיכֶם, אַל תִּתְאַבְּלוּ וְאַל תִּבְכּוּ... לְכוּ אִכְלוּ מַשְׁמַנִּים וּשְׁתוּ מַמְתַקִּים, וְשִׁלְחוּ מָנוֹת לְאֵין נָכוֹן לוֹ... כִּי חֶדְוַת ה׳ הִיא מָעֻזְּכֶם״. לָכֵן אָסוּר לָצוּם בְּרֹאשׁ הַשָּׁנָה.
🤔 לָמָּה? אָדָם רָגִיל שֶׁיֵּשׁ לוֹ מִשְׁפָּט – לוֹבֵשׁ שְׁחֹרִים וּמִתְדָּאֵג. אֲבָל יִשְׂרָאֵל לוֹבְשִׁים לְבָנִים, אוֹכְלִים וְשׂוֹמְחִים בְּרֹאשׁ הַשָּׁנָה, כִּי הֵם יוֹדְעִים שֶׁה׳ רוֹצֶה לְהֵיטִיב לָהֶם. מִי שֶׁמְּקַבֵּל עָלָיו אֶת מַלְכוּת ה׳ וּמִתְעוֹרֵר לְהִתְעַלּוֹת – ה׳ דָּן אוֹתוֹ לְטוֹבָה.
💛 הָעָם הֵבִין שֶׁה׳ אֵינוֹ רוֹצֶה לְהַעֲנִישׁ אוֹתָם, אֶלָּא שָׂמֵחַ בַּחֲזָרָתָם. כָּךְ גַּם אָבִינוּ שָׂמֵחַ כְּשֶׁאֲנַחְנוּ חוֹזְרִים אֵלָיו.
(פֶּרֶק ג׳, הֲלָכָה ד׳, הֲלָכָה ה׳)

7״הַמֶּלֶךְ הַקָּדוֹשׁ״

עִקַּר הַתְּפִלּוֹת שֶׁל רֹאשׁ הַשָּׁנָה הוּא לְהַמְלִיךְ אֶת ה׳ עָלֵינוּ. לָכֵן בְּסוֹף הַבְּרָכָה הַשְּׁלִישִׁית שֶׁל הָעֲמִידָה אוֹמְרִים ״הַמֶּלֶךְ הַקָּדוֹשׁ״ בִּמְקוֹם ״הָאֵל הַקָּדוֹשׁ״, וְכָךְ מַמְשִׁיכִים בְּכָל עֲשֶׂרֶת יְמֵי תְּשׁוּבָה. מִי שֶׁטָּעָה וְלֹא אָמַר ״הַמֶּלֶךְ הַקָּדוֹשׁ״ – לֹא יָצָא, וְחוֹזֵר לְהִתְפַּלֵּל. וּבְכָל הַתְּפִלּוֹת שֶׁל רֹאשׁ הַשָּׁנָה חוֹתְמִים אֶת בִּרְכַּת הֶחָג: ״מֶלֶךְ עַל כָּל הָאָרֶץ, מְקַדֵּשׁ יִשְׂרָאֵל וְיוֹם הַזִּכָּרוֹן״.
🤔 לָמָּה? הָיִינוּ יְכוֹלִים לְהַקְדִּישׁ אֶת כָּל יוֹם הַדִּין לְבַקָּשׁוֹת עַל פַּרְנָסָה וּבְרִיאוּת. אֲבָל הָרָצוֹן הָעָמֹק בְּיוֹתֵר שֶׁל יִשְׂרָאֵל הוּא שֶׁמַּלְכוּת ה׳ תִּתְגַּלֶּה וְהָעוֹלָם כֻּלּוֹ יִתָּקֵן. וּכְשֶׁיִּשְׂרָאֵל עוֹסְקִים בִּכְבוֹד ה׳, נִסְתָּמוֹת טַעֲנוֹת הַשָּׂטָן.
💛 ה׳ אוֹמֵר לַמַּלְאָכִים: ״רְאוּ בָּנַי חֲבִיבַי, שֶׁמַּנִּיחִים צַעֲרָם וְעוֹסְקִים בִּכְבוֹדִי״.
(פֶּרֶק ג׳, הֲלָכָה ו׳)

8שְׁנֵי יָמִים – יוֹם אָרֹךְ אֶחָד

לְפִי הַתּוֹרָה רֹאשׁ הַשָּׁנָה הוּא יוֹם אֶחָד, אֲבָל אֲפִלּוּ בְּאֶרֶץ יִשְׂרָאֵל שׁוֹמְרִים אוֹתוֹ יוֹמַיִם. פַּעַם קִדְּשׁוּ אֶת הַחֹדֶשׁ לְפִי עֵדִים שֶׁרָאוּ אֶת הַיָּרֵחַ הֶחָדָשׁ, וְלֹא יָדְעוּ מֵרֹאשׁ בְּאֵיזֶה יוֹם יָחוּל הַחַג. רֹאשׁ הַשָּׁנָה הוּא הֶחָג הַיָּחִיד שֶׁחָל בְּרֹאשׁ חֹדֶשׁ, וְלָכֵן שָׁמְרוּ אוֹתוֹ יוֹמַיִם בְּכָל מָקוֹם. גַּם הַיּוֹם, כְּשֶׁיֵּשׁ לָנוּ לוּחַ קָבוּעַ, חֲכָמִים קָבְעוּ לְהַמְשִׁיךְ בְּכָךְ. הַיּוֹם הָרִאשׁוֹן הוּא מִן הַתּוֹרָה וְהַשֵּׁנִי מִדִּבְרֵי חֲכָמִים, וּשְׁנֵיהֶם נִקְרָאִים ״יוֹמָא אֲרִיכְתָּא״ – יוֹם אָרֹךְ אֶחָד. אָסוּר לְהָכִין מִן הַיּוֹם הָרִאשׁוֹן לַשֵּׁנִי: לֹא מְבַשְּׁלִים, לֹא עוֹרְכִים אֶת הַשֻּׁלְחָן וְלֹא שׁוֹטְפִים כֵּלִים בִּשְׁבִילוֹ.
🤔 לָמָּה? בַּזֹּהַר מְבֹאָר שֶׁבִּגְלַל חֻמְרַת הַדִּין רָאוּ חֲכָמִים צֹרֶךְ לְהוֹסִיף עוֹד יוֹם, כְּדֵי שֶׁהַדִּין יִהְיֶה מָלֵא גַּם בְּרַחֲמִים.
(פֶּרֶק ג׳, הֲלָכָה ז׳, הֲלָכָה ח׳, הֲלָכָה ט׳)

9״שֶׁהֶחֱיָנוּ״ בַּלַּיְלָה הַשֵּׁנִי

רֹב הַפּוֹסְקִים אוֹמְרִים שֶׁמְּבָרְכִים ״שֶׁהֶחֱיָנוּ״ גַּם בַּקִּדּוּשׁ שֶׁל הַלַּיְלָה הַשֵּׁנִי, וְכָךְ נוֹהֲגִים. אֲבָל יֵשׁ אוֹמְרִים שֶׁשְּׁנֵי הַיָּמִים הֵם כְּמוֹ יוֹם אֶחָד, וְלֹא מְבָרְכִים שׁוּב. לָכֵן, כְּדֵי לְבָרֵךְ לְכָל הַדֵּעוֹת, מְבִיאִים לַשֻּׁלְחָן פְּרִי חָדָשׁ אוֹ לוֹבְשִׁים בֶּגֶד חָדָשׁ, וּמְכַוְּנִים עֲלֵיהֶם בַּ״שֶּׁהֶחֱיָנוּ״. אַחַר כָּךְ, כְּשֶׁאוֹכְלִים אֶת הַפְּרִי, לֹא מְבָרְכִים עָלָיו שׁוּב ״שֶׁהֶחֱיָנוּ״.
(פֶּרֶק ג׳, הֲלָכָה ט׳)

10מַלְכֻיּוֹת, זִכְרוֹנוֹת וְשׁוֹפָרוֹת

בִּתְפִלַּת מוּסָף, הַתְּפִלָּה הַחֲשׁוּבָה בְּיוֹתֵר שֶׁל רֹאשׁ הַשָּׁנָה, תִּקְּנוּ חֲכָמִים שָׁלֹשׁ בְּרָכוֹת מְיֻחָדוֹת. ה׳ אָמַר: ״אִמְרוּ לְפָנַי בְּרֹאשׁ הַשָּׁנָה מַלְכֻיּוֹת – כְּדֵי שֶׁתַּמְלִיכוּנִי עֲלֵיכֶם, זִכְרוֹנוֹת – כְּדֵי שֶׁיַּעֲלֶה זִכְרוֹנְכֶם לְפָנַי לְטוֹבָה, וּבַמָּה? בַּשּׁוֹפָר״. בְּכָל בְּרָכָה אוֹמְרִים עֲשָׂרָה פְּסוּקִים: שְׁלֹשָׁה מִן הַתּוֹרָה, שְׁלֹשָׁה מִן הַכְּתוּבִים, שְׁלֹשָׁה מִן הַנְּבִיאִים, וּפָסוּק אַחֲרוֹן מִן הַתּוֹרָה.
🤔 לָמָּה? עֲשָׂרָה פְּסוּקִים – כְּנֶגֶד עֲשָׂרָה מַאֲמָרוֹת שֶׁבָּהֶם נִבְרָא הָעוֹלָם, וּכְנֶגֶד עֲשֶׂרֶת הַדִּבְּרוֹת. שָׁלֹשׁ הַבְּרָכוֹת הֵן יְחִידָה אַחַת, וְגַם הַסֵּדֶר שֶׁלָּהֶן קָבוּעַ: מִי שֶׁאָמַר אוֹתָן בְּסֵדֶר אַחֵר – לֹא יָצָא.
💛 בְּבִרְכַּת זִכְרוֹנוֹת אֲנַחְנוּ מְבַקְּשִׁים: ״וַעֲקֵדַת יִצְחָק לְזַרְעוֹ הַיּוֹם בְּרַחֲמִים תִּזְכֹּר״. ה׳ זוֹכֵר אֶת הָאַהֲבָה שֶׁל אֲבוֹתֵינוּ וּמֵיטִיב לַבָּנִים.
(פֶּרֶק ג׳, הֲלָכָה י׳)

11בְּלִי וִדּוּי

בְּרֹאשׁ הַשָּׁנָה לֹא אוֹמְרִים וִדּוּי – לֹא ״אָשַׁמְנוּ״ וְלֹא וִדּוּי אִישִׁי. הַיּוֹם מֻקְדָּשׁ לְהַמְלָכַת ה׳ עַל כָּל הָעוֹלָם, וְאֶת הַתִּקּוּן הָאִישִׁי מַשְׁאִירִים לִשְׁאָר יְמֵי הַתְּשׁוּבָה. בִּתְפִלַּת ״אָבִינוּ מַלְכֵּנוּ״ יֵשׁ מִשְׁפָּטִים שֶׁמַּזְכִּירִים חֵטְא, כְּמוֹ ״אָבִינוּ מַלְכֵּנוּ, חָטָאנוּ לְפָנֶיךָ״, וּבָהֶם יֵשׁ שְׁנֵי מִנְהָגִים.
🤔 לָמָּה? לֹא רָאוּי לְהַזְכִּיר חֲטָאִים בַּיּוֹם שֶׁבּוֹ מַכְתִּירִים אֶת הַמֶּלֶךְ, וְיֵשׁ חֲשָׁשׁ שֶׁהַזְכָּרָתָם תְּחַזֵּק אֶת הַמְקַטְרְגִים.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ סוֹבְרִים (הַבֵּית יוֹסֵף וְהָאֲרִ״י) מַשְׁמִיטִים בְּרֹאשׁ הַשָּׁנָה אֶת הַמִּשְׁפָּטִים שֶׁמַּזְכִּירִים חֵטְא.
וְיֵשׁ נוֹהֲגִים (כְּדַעַת הָרַמָּ״א) אוֹמְרִים אֶת כָּל הַנֻּסָּח, כִּי זֶה תֵּאוּר שֶׁל מַצָּב כְּלָלִי וְלֹא וִדּוּי.
(פֶּרֶק ג׳, הֲלָכָה י״א)

12קְרִיאַת הַתּוֹרָה

בַּיּוֹם הָרִאשׁוֹן קוֹרְאִים בַּתּוֹרָה: ״וַה׳ פָּקַד אֶת שָׂרָה״ – עַל הֻלֶּדֶת יִצְחָק, וּמַפְטִירִים בְּסִפּוּר חַנָּה שֶׁיָּלְדָה אֶת שְׁמוּאֵל הַנָּבִיא. בַּיּוֹם הַשֵּׁנִי קוֹרְאִים אֶת עֲקֵדַת יִצְחָק. בִּשְׁנֵי הַיָּמִים עוֹלִים לַתּוֹרָה חֲמִשָּׁה עוֹלִים, כְּמוֹ בְּכָל חַג, וּמוֹצִיאִים סֵפֶר תּוֹרָה שֵׁנִי לַמַּפְטִיר, וְקוֹרְאִים בּוֹ עַל קָרְבְּנוֹת הַיּוֹם.
🤔 לָמָּה? בְּרֹאשׁ הַשָּׁנָה נִפְקְדוּ שָׁלֹשׁ נָשִׁים צַדְקָנִיּוֹת: שָׂרָה אִמֵּנוּ, רָחֵל אִמֵּנוּ וְחַנָּה. וְאֶת הָעֲקֵדָה קוֹרְאִים כְּדֵי לְהַזְכִּיר אֶת זְכוּת הָאָבוֹת.
(פֶּרֶק ג׳, הֲלָכָה י״א)

13סִימָנִים לְשָׁנָה טוֹבָה

כָּל מָה שֶׁעוֹשִׂים בְּרֹאשׁ הַשָּׁנָה הוּא סִימָן לְכָל הַשָּׁנָה. לָכֵן אוֹכְלִים מַאֲכָלִים שֶׁל סִימָן טוֹב: תַּפּוּחַ בִּדְבַשׁ – לְשָׁנָה טוֹבָה וּמְתוּקָה; רִמּוֹן וְרוּבְיָא (לוּבְיָה) – שֶׁיִּרְבּוּ זְכֻיּוֹתֵינוּ; תְּמָרִים – שֶׁיִּתַּמּוּ שׂוֹנְאֵינוּ; סֶלֶק – שֶׁיִּסְתַּלְּקוּ שׂוֹנְאֵינוּ; כְּרֵתִי (כְּרֵשָׁה) – שֶׁיִּכָּרְתוּ שׂוֹנְאֵינוּ; וְרֹאשׁ שֶׁל דָּג אוֹ כֶּבֶשׂ – שֶׁנִּהְיֶה לְרֹאשׁ וְלֹא לְזָנָב. עַל כָּל סִימָן אוֹמְרִים תְּפִלָּה קְצָרָה: ״יְהִי רָצוֹן...״. כָּל עֵדָה נוֹהֶגֶת לְפִי הַמַּאֲכָלִים וְהַשָּׂפָה שֶׁלָּהּ, וְטוֹב לְהַמְשִׁיךְ בַּמָּסֹרֶת.
🤔 לָמָּה? ״סִימָנָא מִלְּתָא הִיא״ – לַסִּימָנִים יֵשׁ מַשְׁמָעוּת. אֲבָל הָעִקָּר אֵינוֹ הָאֹכֶל עַצְמוֹ, אֶלָּא לְהִתְעוֹרֵר מִתּוֹךְ הָאֲכִילָה לִתְשׁוּבָה וְלִתְפִלָּה עַל שָׁנָה טוֹבָה.
💛 אֲפִלּוּ הַסְּעוּדָה שֶׁל רֹאשׁ הַשָּׁנָה הוֹפֶכֶת לִתְפִלָּה. כָּל בִּיס הוּא בַּקָּשָׁה מֵאַבָּא.
(פֶּרֶק ג׳, הֲלָכָה י״ב)

14חַלָּה בִּדְבַשׁ

רַבִּים טוֹבְלִים בְּרֹאשׁ הַשָּׁנָה אֶת הַחַלָּה שֶׁל ״הַמּוֹצִיא״ בִּדְבַשׁ אוֹ בְּמֵי סֻכָּר, לְסִימָן שֶׁתִּהְיֶה הַשָּׁנָה טוֹבָה וּמְתוּקָה, וְיֵשׁ מַמְשִׁיכִים כָּךְ בְּכָל הַשַּׁבָּתוֹת וְהַחַגִּים עַד שִׂמְחַת תּוֹרָה. כָּל הַמִּנְהָגִים כְּשֵׁרִים, וְרָאוּי לְכָל אֶחָד לְהַמְשִׁיךְ בַּמָּסֹרֶת שֶׁל אֲבוֹתָיו.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ נוֹהֲגִים לִטְבֹּל אֶת הַחַלָּה קֹדֶם בִּמְעַט מֶלַח, וְאַחַר כָּךְ בִּדְבַשׁ.
וְיֵשׁ נוֹהֲגִים לִטְבֹּל רַק בִּדְבַשׁ, וּלְהַנִּיחַ אֶת הַמֶּלַח עַל הַשֻּׁלְחָן בְּלִי לִטְבֹּל בּוֹ.
וְיֵשׁ שֶׁלֹּא נָהֲגוּ לִטְבֹּל אֶת הַחַלָּה בִּדְבַשׁ כְּלָל, אֲפִלּוּ בְּרֹאשׁ הַשָּׁנָה.
(פֶּרֶק ג׳, הֲלָכָה י״ב)

15מָה לֹא אוֹכְלִים?

בְּרֹאשׁ הַשָּׁנָה אוֹכְלִים מַאֲכָלִים טוֹבִים וּמְשֻׁבָּחִים, לְסִימָן טוֹב. רַבִּים נִמְנָעִים מִלְּבַשֵּׁל מַאֲכָלִים חֲמוּצִים, מְלוּחִים וּמָרִים, וּמִלֶּאֱכֹל פֵּרוֹת שֶׁלֹּא הִבְשִׁילוּ. כָּךְ נוֹהֲגִים כָּל יוֹצְאֵי אַשְׁכְּנַז וְרַבִּים מִיּוֹצְאֵי סְפָרַד. מִמַּאֲכָלִים חֲרִיפִים רַבִּים לֹא נִמְנְעוּ. וְיוֹצְאֵי אַשְׁכְּנַז נָהֲגוּ גַּם שֶׁלֹּא לֶאֱכֹל אֱגוֹזִים בְּרֹאשׁ הַשָּׁנָה.
🤔 לָמָּה? רוֹצִים שֶׁהַשָּׁנָה תִּהְיֶה מְתוּקָה וְלֹא מָרָה. הָאֱגוֹז רוֹמֵז לְחֵטְא, וְהוּא גַּם מַרְבֶּה לֵחָה בַּגָּרוֹן וְעָלוּל לְהַפְרִיעַ בַּתְּפִלָּה.
(פֶּרֶק ג׳, הֲלָכָה י״ב)

16סֵדֶר הַסְּעוּדָה

אַחֲרֵי הַקִּדּוּשׁ נוֹטְלִים יָדַיִם, מְבָרְכִים ״הַמּוֹצִיא״ עַל הַחַלָּה, וְרַק אַחַר כָּךְ אוֹכְלִים אֶת הַסִּימָנִים. עַל פֵּרוֹת הָעֵץ מְבָרְכִים ״בּוֹרֵא פְּרִי הָעֵץ״, וְנָכוֹן לְבָרֵךְ עַל הַתָּמָר וְלִפְטֹר בּוֹ אֶת שְׁאָר הַפֵּרוֹת. אוֹכְלִים מְעַט מִן הַתָּמָר, וְרַק אַחַר כָּךְ אוֹמְרִים ״יְהִי רָצוֹן״. עַל הַסִּימָנִים הַמְּבֻשָּׁלִים שֶׁבִּרְכָתָם ״הָאֲדָמָה״ לֹא מְבָרְכִים, כִּי הֵם חֵלֶק מֵהַסְּעוּדָה.
🤔 לָמָּה? הַלֶּחֶם חָשׁוּב מִכָּל הַמַּאֲכָלִים, וְלָכֵן מַקְדִּימִים אוֹתוֹ. הַתָּמָר הוּא מִשִּׁבְעַת הַמִּינִים, וְהוּא חָשׁוּב מֵהָרִמּוֹן. וְאֶת הַ״יְּהִי רָצוֹן״ אוֹמְרִים אַחֲרֵי שֶׁטָּעֲמוּ, כְּדֵי לֹא לְהַפְסִיק בֵּין הַבְּרָכָה לָאֲכִילָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ נוֹהֲגִים לֶאֱכֹל אֶת הַסִּימָנִים רַק בַּלַּיְלָה הָרִאשׁוֹן.
וְרֹב יִשְׂרָאֵל נוֹהֲגִים לֶאֱכֹל אֶת הַסִּימָנִים בִּשְׁנֵי הַלֵּילוֹת, וְיֵשׁ מְהַדְּרִים גַּם בַּסְּעוּדוֹת שֶׁל הַיּוֹם.
(פֶּרֶק ג׳, הֲלָכָה י״ג)

17סִימָן טוֹב בַּהִתְנַהֲגוּת

לֹא רַק הָאֹכֶל הוּא סִימָן. בְּרֹאשׁ הַשָּׁנָה לֹא כּוֹעֲסִים, לֹא מִתְקוֹטְטִים וְלֹא מְדַבְּרִים רָעוֹת עַל אֲחֵרִים – גַּם כִּי זֶה אָסוּר, וְגַם כִּי זֶה סִימָן רַע לַשָּׁנָה. מַרְבִּים בְּאַהֲבָה, דָּנִים כָּל אָדָם לְכַף זְכוּת, וּבַסְּעוּדָה מִשְׁתַּדְּלִים לְשַׂמֵּחַ אֶת כָּל בְּנֵי הַמִּשְׁפָּחָה וְהָאוֹרְחִים.
🤔 לָמָּה? כְּמוֹ שֶׁהַיּוֹם הָרִאשׁוֹן שֶׁל הַשָּׁנָה מַשְׁפִּיעַ עַל כֻּלָּהּ, כָּךְ חִיּוּךְ וּמִלָּה טוֹבָה בְּרֹאשׁ הַשָּׁנָה הֵם הַתְחָלָה טוֹבָה לְשָׁנָה שְׁלֵמָה.
💛 מִי שֶׁמְּשַׂמֵּחַ אֶת בְּנֵי מִשְׁפַּחְתּוֹ בְּרֹאשׁ הַשָּׁנָה, מַמְלִיךְ אֶת ה׳ גַּם בְּתוֹךְ הַבַּיִת שֶׁלּוֹ.
(פֶּרֶק ג׳, הֲלָכָה י״ד)

18לֹא מְבַזְבְּזִים אֶת הַיּוֹם

חֲכָמִים אָמְרוּ: ״כָּל הַיָּשֵׁן בְּרֹאשׁ הַשָּׁנָה – מַזָּלוֹ יָשֵׁן״. לָכֵן לֹא מַרְבִּים בְּשֵׁנָה, וּמִלְּבַד הַתְּפִלּוֹת וְהַסְּעוּדוֹת מַרְבִּים בְּלִמּוּד תּוֹרָה. מִי שֶׁיּוֹשֵׁב בְּבַטָּלָה, אַף שֶׁהוּא עֵר – נֶחְשָׁב כְּיָשֵׁן. וּמִי שֶׁעָיֵף אַחֲרֵי הַסְּעוּדָה – טוֹב שֶׁיִּישַׁן, כְּדֵי שֶׁיּוּכַל לִלְמֹד כָּרָאוּי.
🤔 לָמָּה? הַשֵּׁנָה עַצְמָהּ אֵינָהּ הַבְּעָיָה, אֶלָּא בִּזְבּוּז הַזְּמַן הַיָּקָר. הָעִקָּר שֶׁהַתְּפִלּוֹת וְהַלִּמּוּד יְמַלְּאוּ לְפָחוֹת חֲצִי מֵהַיּוֹם.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ נוֹהֲגִים שֶׁלֹּא לִישֹׁן כָּל הַיּוֹם. רֹב הָאַחֲרוֹנִים כָּתְבוּ שֶׁהָעִקָּר הוּא שֶׁלֹּא לִישֹׁן לִפְנֵי חֲצוֹת הַיּוֹם.
וְיֵשׁ מֵהַגְּדוֹלִים שֶׁנָּהֲגוּ לִישֹׁן בְּרֹאשׁ הַשָּׁנָה כְּמוֹ בְּכָל הַחַגִּים.
(פֶּרֶק ג׳, הֲלָכָה י״ד)

19״לְשָׁנָה טוֹבָה תִּכָּתֵב וְתֵחָתֵם״

אַחֲרֵי תְּפִלַּת עַרְבִית מְבָרְכִים זֶה אֶת זֶה: ״לְשָׁנָה טוֹבָה תִּכָּתֵב וְתֵחָתֵם״, וּלְאִשָּׁה אוֹ לְבַת: ״תִּכָּתְבִי וְתֵחָתֵמִי״. יֵשׁ אוֹמְרִים (הַגְּרָ״א) שֶׁבְּרֹאשׁ הַשָּׁנָה אוֹמְרִים רַק ״לְשָׁנָה טוֹבָה תִּכָּתֵב״, כִּי הַחֲתִימָה הִיא בְּיוֹם הַכִּפֻּרִים. לְמָחֳרָת מְבָרְכִים כָּךְ עַד חֲצוֹת הַיּוֹם, וְאַחַר כָּךְ אוֹמְרִים רַק ״שָׁנָה טוֹבָה״. מִי שֶׁמְּבָרְכִים אוֹתוֹ – מֵשִׁיב בְּאוֹתָן בְּרָכוֹת, אוֹ אוֹמֵר: ״וְכֵן לְמַר״ – כְּלוֹמַר, כָּל הַבְּרָכוֹת יָחוּלוּ גַּם עָלֶיךָ.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ נוֹהֲגִים שֶׁלֹּא לְבָרֵךְ כָּךְ בַּיּוֹם הַשֵּׁנִי, כִּי עִקַּר הַדִּין בָּרִאשׁוֹן.
וְרֹב הַצִּבּוּר נוֹהֵג לְבָרֵךְ גַּם בַּיּוֹם הַשֵּׁנִי, כִּי גַּם בּוֹ יֵשׁ כְּתִיבַת דִּין.
💛 בְּרָכָה טוֹבָה לַחֲבֵר הִיא תְּפִלָּה קְטַנָּה עָלָיו, וְכָךְ כָּל עַם יִשְׂרָאֵל מִתְפַּלֵּל זֶה עַל זֶה.
(פֶּרֶק ג׳, הֲלָכָה י״ד)

20תַּשְׁלִיךְ

רַבִּים נוֹהֲגִים לָלֶכֶת בַּיּוֹם הָרִאשׁוֹן לִמְקוֹם מַיִם – יָם, נָהָר אוֹ מַעְיָן – וְלוֹמַר פְּסוּקִים, וּבָהֶם: ״וְתַשְׁלִיךְ בִּמְצֻלוֹת יָם כָּל חַטֹּאתָם״. יֵשׁ נוֹהֲגִים לְנַעֵר אֶת שׁוּלֵי הַבְּגָדִים. כְּשֶׁאֵין מַיִם קְרוֹבִים, אוֹמְרִים לְיַד בּוֹר מַיִם אוֹ מִקְוֶה. זֶה מִנְהָג וְלֹא חוֹבָה: יֵשׁ מִגְּדוֹלֵי יִשְׂרָאֵל שֶׁלֹּא נָהֲגוּ בּוֹ, כְּמוֹ הַגְּרָ״א, וְכֵן רֹב יוֹצְאֵי תֵּימָן. רֹב הַנָּשִׁים לֹא נָהֲגוּ לוֹמַר תַּשְׁלִיךְ, וּמִי שֶׁרוֹצָה – רַשָּׁאִית.
🤔 לָמָּה? הַמַּיִם רוֹמְזִים לְטָהֳרָה וּלְחַיִּים, וּמְבַקְּשִׁים שֶׁה׳ יַשְׁלִיךְ אֶת הַקִּטְרוּגִים לְמָקוֹם שֶׁלֹּא יִזָּכְרוּ עוֹד. וְנִעוּר הַבְּגָדִים מַרְאֶה שֶׁהַחֲטָאִים אֵינָם חֵלֶק מֵאִתָּנוּ – הֵם דְּבוּקִים רַק מִבַּחוּץ, וַאֲנַחְנוּ מִתְנַעֲרִים מֵהֶם.
(פֶּרֶק ג׳, הֲלָכָה ט״ו)

21רֹאשׁ הַשָּׁנָה שֶׁחָל בְּשַׁבָּת

כְּשֶׁרֹאשׁ הַשָּׁנָה חָל בְּשַׁבָּת, לֹא תּוֹקְעִים בַּשּׁוֹפָר (נִלְמַד עַל כָּךְ בַּפֶּרֶק הַבָּא). רַבִּים מֵהַנּוֹהֲגִים לוֹמַר תַּשְׁלִיךְ דּוֹחִים אוֹתוֹ לַיּוֹם הַשֵּׁנִי. וּבַ״אָבִינוּ מַלְכֵּנוּ״ יֵשׁ שְׁנֵי מִנְהָגִים.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הָאַשְׁכְּנַזִּים, הַתֵּימָנִים וְחֵלֶק מֵהַסְּפָרַדִּים לֹא אוֹמְרִים ״אָבִינוּ מַלְכֵּנוּ״ בְּשַׁבָּת, כִּי לֹא מְבַקְּשִׁים בַּשַּׁבָּת עַל צְרָכִים.
לְמִנְהַג רֹב הַסְּפָרַדִּים אוֹמְרִים ״אָבִינוּ מַלְכֵּנוּ״ גַּם בְּשַׁבָּת, וְכָךְ הִסְכִּים גַּם הָאֲרִ״י.
(פֶּרֶק ג׳, הֲלָכָה ט״ז)

📝 מָה לָמַדְנוּ

  • רֹאשׁ הַשָּׁנָה הוּא יוֹם הַזִּכָּרוֹן וְיוֹם הַתְּרוּעָה, וְעִקַּר הַתְּפִלָּה בּוֹ – לְהַמְלִיךְ אֶת ה׳: ״הַמֶּלֶךְ הַקָּדוֹשׁ״.
  • זֶה חַג שֶׁל סְעוּדוֹת וּבְגָדִים נָאִים, אֲבָל גַּם יוֹם דִּין: לֹא עַד הַשֹּׂבַע, לֹא הַבְּגָדִים הַיָּפִים בְּיוֹתֵר, בְּלִי הַלֵּל וּבְלִי וִדּוּי.
  • שׁוֹמְרִים שְׁנֵי יָמִים גַּם בְּאֶרֶץ יִשְׂרָאֵל, וּמְבִיאִים פְּרִי חָדָשׁ לַ״שֶּׁהֶחֱיָנוּ״ שֶׁל הַלַּיְלָה הַשֵּׁנִי.
  • אוֹכְלִים סִימָנִים לְשָׁנָה טוֹבָה, אַחֲרֵי ״הַמּוֹצִיא״, וְאוֹמְרִים עֲלֵיהֶם ״יְהִי רָצוֹן״.
  • גַּם הַהִתְנַהֲגוּת הִיא סִימָן: לֹא כּוֹעֲסִים, מְשַׂמְּחִים, לוֹמְדִים וּמְבָרְכִים ״לְשָׁנָה טוֹבָה תִּכָּתֵב וְתֵחָתֵם״.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה לֹא אוֹמְרִים הַלֵּל בְּרֹאשׁ הַשָּׁנָה, אַף שֶׁהוּא חַג?
  2. בְּאֵיזֶה סֵדֶר אוֹכְלִים בְּלֵיל רֹאשׁ הַשָּׁנָה: קִדּוּשׁ, ״הַמּוֹצִיא״, סִימָנִים – וְעַל אֵיזֶה פְּרִי נָכוֹן לְבָרֵךְ ״הָעֵץ״?
  3. מָה עוֹשִׂים כְּדֵי לְבָרֵךְ ״שֶׁהֶחֱיָנוּ״ לְכָל הַדֵּעוֹת בַּלַּיְלָה הַשֵּׁנִי?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״רֹאשׁ הַשָּׁנָה – מַמְלִיכִים אֶת ה׳״

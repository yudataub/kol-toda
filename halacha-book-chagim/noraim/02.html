סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים
מַהֲדוּרַת יְלָדִים

🌙 סְלִיחוֹת וּתְפִלּוֹת

פֶּרֶק ב׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
עוֹד חֹשֶׁךְ בַּחוּץ. הָרְחוֹב שָׁקֵט, וּבְכָל זֹאת מִבֵּית הַכְּנֶסֶת בּוֹקַעַת מַנְגִּינָה: ״אָנָּא, סְלַח נָא!״. אוּלַי גַּם אַתָּה קַמְתָּ פַּעַם מֻקְדָּם עִם אַבָּא וְהִצְטָרַפְתָּ. בַּפֶּרֶק הַזֶּה נִלְמַד עַל חֹדֶשׁ אֱלוּל, עַל הַשּׁוֹפָר וְהַסְּלִיחוֹת, וְעַל הַהֲכָנוֹת לְרֹאשׁ הַשָּׁנָה.

1אֱלוּל – יְמֵי רָצוֹן

אַחֲרֵי חֵטְא הָעֵגֶל ה׳ כָּעַס מְאֹד עַל יִשְׂרָאֵל, וְהַלּוּחוֹת הָרִאשׁוֹנִים נִשְׁבְּרוּ. בְּרֹאשׁ חֹדֶשׁ אֱלוּל עָלָה מֹשֶׁה רַבֵּנוּ שׁוּב לְהַר סִינַי, לְבַקֵּשׁ סְלִיחָה וְרַחֲמִים עַל יִשְׂרָאֵל. בְּיוֹם הַכִּפֻּרִים קִבֵּל ה׳ אֶת תְּשׁוּבָתָם, וּמֹשֶׁה יָרַד עִם הַלּוּחוֹת הַשְּׁנִיִּים. לָכֵן אַרְבָּעִים הַיָּמִים שֶׁמִּתְּחִלַּת אֱלוּל וְעַד יוֹם הַכִּפֻּרִים הֵם הַיָּמִים הַמְּסֻגָּלִים בְּיוֹתֵר לִתְשׁוּבָה.
🤔 לָמָּה? הַיָּמִים שֶׁבָּהֶם קָרוּ דְּבָרִים כָּל כָּךְ חֲשׁוּבִים אֵינָם מִקְרִיִּים. מִי שֶׁשָּׁב בִּתְשׁוּבָה בְּאוֹתָם יָמִים, הוֹלֵךְ בַּדֶּרֶךְ שֶׁכְּבָר נִפְתְּחָה לְפָנָיו.
💛 גַּם אַחֲרֵי חֵטְא גָּדוֹל, ה׳ הֶחֱזִיר אֶת יִשְׂרָאֵל אֵלָיו וְצִוָּה לְהָקִים מִשְׁכָּן, כְּדֵי לִשְׁכֹּן בְּתוֹכָם.
(פֶּרֶק ב׳, הֲלָכָה א׳)

2שׁוֹפָר בְּחֹדֶשׁ אֱלוּל

כְּשֶׁעָלָה מֹשֶׁה לָהָר, הֶעֱבִירוּ שׁוֹפָר בְּכָל הַמַּחֲנֶה, כְּדֵי שֶׁלֹּא יִטְעוּ שׁוּב אַחֲרֵי עֲבוֹדָה זָרָה. לָכֵן תִּקְּנוּ חֲכָמִים לִתְקֹעַ בַּשּׁוֹפָר בְּחֹדֶשׁ אֱלוּל. זוֹ אֵינָהּ חוֹבָה, אֲבָל רָאוּי לַצִּבּוּר לְהִשְׁתַּדֵּל לְקַיֵּם אֶת הַמִּנְהָג. יָחִיד שֶׁלֹּא שָׁמַע שׁוֹפָר בֶּאֱלוּל אֵינוֹ צָרִיךְ לְחַפֵּשׂ מִי שֶׁיִּתְקַע לוֹ.
🤔 לָמָּה? לְקוֹל הַשּׁוֹפָר יֵשׁ כֹּחַ לְהַזְהִיר אֶת הָעָם מִלַּחֲטֹא וּלְעוֹרֵר אֶת הָרַבִּים לִתְשׁוּבָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז תּוֹקְעִים בְּכָל יוֹם בְּסוֹף תְּפִלַּת שַׁחֲרִית.
לְמִנְהַג יוֹצְאֵי סְפָרַד תּוֹקְעִים בַּקַּדִּישׁ שֶׁבְּסוֹף הַסְּלִיחוֹת, וְרַבִּים תּוֹקְעִים גַּם כְּשֶׁאוֹמְרִים י״ג מִדּוֹת רַחֲמִים.
💛 קוֹל הַשּׁוֹפָר בֶּאֱלוּל הוּא כְּמוֹ קְרִיאָה שֶׁל אַבָּא: ״בּוֹא הַבַּיְתָה, מְחַכִּים לְךָ!״
(פֶּרֶק ב׳, הֲלָכָה א׳)

3עַל מָה מִתְפַּלְּלִים בַּסְּלִיחוֹת?

מִימֵי הַגְּאוֹנִים נוֹהֲגִים רַבִּים מִיִּשְׂרָאֵל לְהַשְׁכִּים לִסְלִיחוֹת. מְבַקְּשִׁים סְלִיחָה וּמְחִילָה, וּמִתְחַנְּנִים שֶׁה׳ יִזְכֹּר אֶת הַבְּרִית שֶׁכָּרַת עִם אֲבוֹתֵינוּ. מִתְפַּלְּלִים לֹא רַק עַל עַצְמֵנוּ, אֶלָּא עַל כָּל עַם יִשְׂרָאֵל: עַל קִבּוּץ הַגָּלֻיּוֹת, עַל בִּנְיַן הָאָרֶץ וִירוּשָׁלַיִם וְעַל בֵּית הַמִּקְדָּשׁ. יַחַד עִם הַסְּלִיחוֹת לוֹמְדִים בְּסִפְרֵי מוּסָר וְשׁוֹמְעִים דְּרָשׁוֹת שֶׁמְּעוֹרְרוֹת לִתְשׁוּבָה.
🤔 לָמָּה? מִי שֶׁמִּתְפַּלֵּל עַל כְּלַל יִשְׂרָאֵל – דַּוְקָא מִתּוֹךְ כָּךְ מִתְקַבְּלוֹת גַּם הַתְּפִלּוֹת הָאִישִׁיּוֹת שֶׁלּוֹ. וְגַם הַיּוֹם, כְּשֶׁה׳ הִתְחִיל לְקַבֵּץ אֶת הַגָּלֻיּוֹת וְלִבְנוֹת אֶת הָאָרֶץ, צָרִיךְ לְהִתְפַּלֵּל שֶׁיַּמְשִׁיךְ בְּחַסְדּוֹ.
💛 כְּשֶׁאֲנִי אוֹמֵר סְלִיחוֹת, אֲנִי מְחֻבָּר לְכָל הַיְּהוּדִים בְּכָל הַדּוֹרוֹת, וְכֻלָּנוּ יַחַד עוֹמְדִים לִפְנֵי אָבִינוּ.
(פֶּרֶק ב׳, הֲלָכָה ב׳, הֲלָכָה ג׳)

4הַסֵּדֶר שֶׁל הַסְּלִיחוֹת

חֲכָמִים לֹא קָבְעוּ לַסְּלִיחוֹת נֻסָּח אֶחָד, וְכָל עֵדָה הוֹסִיפָה פִּיּוּטִים מִשֶּׁלָּהּ. אֲבָל יֵשׁ מִסְגֶּרֶת מְשֻׁתֶּפֶת לְכֻלָּם: פּוֹתְחִים בְּ״אַשְׁרֵי״ וַחֲצִי קַדִּישׁ, אוֹמְרִים פְּסוּקֵי בַּקָּשָׁה, י״ג מִדּוֹת רַחֲמִים וּוִדּוּי, ״עֲנֵנוּ״ וּנְפִילַת אַפַּיִם, וְחוֹתְמִים בְּקַדִּישׁ. כְּשֶׁאֵין זְמַן, מְדַלְּגִים עַל חֵלֶק מֵהַפִּיּוּטִים, וְאוֹמְרִים אֶת אֵלֶּה שֶׁמְּעוֹרְרִים יוֹתֵר לִתְשׁוּבָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג סְפָרַד אוֹמְרִים אֶת אוֹתוֹ נֻסָּח בְּכָל הַיָּמִים.
לְמִנְהַג אַשְׁכְּנַז בְּכָל יוֹם מוֹסִיפִים פִּיּוּטִים אֲחֵרִים.
(פֶּרֶק ב׳, הֲלָכָה ד׳)

5מָתַי מַתְחִילִים לוֹמַר סְלִיחוֹת?

בִּימֵי הַגְּאוֹנִים אָמְרוּ סְלִיחוֹת בְּעִקָּר בַּעֲשֶׂרֶת יְמֵי תְּשׁוּבָה. הַיּוֹם יֵשׁ שְׁנֵי מִנְהָגִים. וּכְכָל שֶׁמִּתְקָרְבִים לְרֹאשׁ הַשָּׁנָה, רַבִּים יוֹתֵר מַקְפִּידִים לָקוּם לִסְלִיחוֹת.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הַסְּפָרַדִּים אוֹמְרִים סְלִיחוֹת בְּכָל חֹדֶשׁ אֱלוּל וּבַעֲשֶׂרֶת יְמֵי תְּשׁוּבָה, אֲבָל בְּיוֹם רֹאשׁ חֹדֶשׁ אֱלוּל עַצְמוֹ אֵין אוֹמְרִים.
לְמִנְהַג הָאַשְׁכְּנַזִּים מַתְחִילִים בְּמוֹצָאֵי הַשַּׁבָּת שֶׁלִּפְנֵי רֹאשׁ הַשָּׁנָה, וּבִתְנַאי שֶׁיִּהְיוּ לְפָחוֹת אַרְבָּעָה יְמֵי סְלִיחוֹת. לָכֵן אִם רֹאשׁ הַשָּׁנָה חָל בְּיוֹם שֵׁנִי אוֹ שְׁלִישִׁי, מַתְחִילִים בְּמוֹצָאֵי הַשַּׁבָּת שֶׁלִּפְנֵי כֵן.
(פֶּרֶק ב׳, הֲלָכָה ה׳)

6חוֹבָה אוֹ מִנְהָג?

חֲכָמִים לֹא תִּקְּנוּ סְלִיחוֹת כְּחוֹבָה, אֲבָל כָּךְ הוּא מִנְהַג יִשְׂרָאֵל. מִי שֶׁקָּשֶׁה לוֹ לָקוּם אֵינוֹ חַיָּב לָקוּם לִסְלִיחוֹת בְּחֹדֶשׁ אֱלוּל, וּבַעֲשֶׂרֶת יְמֵי תְּשׁוּבָה יִשְׁתַּדֵּל יוֹתֵר. מִי שֶׁהַקִּימָה תַּעֲשֶׂה אוֹתוֹ עָיֵף כָּל כָּךְ שֶׁלֹּא יוּכַל לַעֲבֹד אוֹ לִלְמֹד כָּרָאוּי – עָדִיף שֶׁלֹּא יָקוּם, וְיֹאמַר בִּמְקוֹם זֶה פִּרְקֵי תְּהִלִּים. גַּם תַּלְמִידֵי חֲכָמִים קָמִים לִסְלִיחוֹת, וְכָךְ נוֹהֲגִים בְּכָל הַיְשִׁיבוֹת.
🤔 לָמָּה? ה׳ רוֹצֶה תְּפִלָּה מִן הַלֵּב, וְלֹא תְּפִלָּה שֶׁמְּקַלְקֶלֶת אֶת שְׁאָר הַיּוֹם. גַּם הַשְּׁמִירָה עַל הַכֹּחַ לַעֲבֹד וְלִלְמֹד הִיא חֵלֶק מֵעֲבוֹדַת ה׳.
(פֶּרֶק ב׳, הֲלָכָה ה׳)

7בְּאַשְׁמֹרֶת הַבֹּקֶר

הַזְּמַן הַטּוֹב בְּיוֹתֵר לִסְלִיחוֹת הוּא ״אַשְׁמֹרֶת הַבֹּקֶר״ – לִקְרַאת סוֹף הַלַּיְלָה, וּכְבָר מֵחֲצוֹת הַלַּיְלָה הַזְּמַן מַתְאִים. הַיּוֹם רֹב הָאֲנָשִׁים הוֹלְכִים לִישֹׁן מְאֻחָר, וְלָכֵן רַבִּים קָמִים לִסְלִיחוֹת כַּחֲצִי שָׁעָה אוֹ שָׁעָה לִפְנֵי תְּפִלַּת שַׁחֲרִית. גַּם אָז הַזְּמַן עֲדַיִן רָאוּי. בַּסְּלִיחוֹת יֵשׁ פִּיּוּטִים שֶׁאוֹמְרִים בָּהֶם ״קַמְנוּ בְּאַשְׁמֹרֶת״.
🤔 לָמָּה? בְּסוֹף הַלַּיְלָה הָעוֹלָם שָׁקֵט, כֻּלָּם יְשֵׁנִים וּמְחַכִּים לָאוֹר. זוֹ שְׁעַת רַחֲמִים וְרָצוֹן, וְהַתְּפִלָּה יוֹצֵאת מֵעֹמֶק הַלֵּב.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ סוֹבְרִים מִי שֶׁאוֹמֵר סְלִיחוֹת אַחֲרֵי עֲלוֹת הַשַּׁחַר יְדַלֵּג עַל ״קַמְנוּ בְּאַשְׁמֹרֶת״, כְּדֵי שֶׁלֹּא לוֹמַר דָּבָר שֶׁאֵינוֹ נָכוֹן.
וְהַמִּנְהָג אוֹמְרִים אוֹתוֹ, כִּי הַנֻּסָּח נִתְקַן לְכָל יִשְׂרָאֵל, וּבְכָל יוֹם יֵשׁ יְהוּדִים שֶׁקָּמִים בְּאַשְׁמֹרֶת.
(פֶּרֶק ב׳, הֲלָכָה ו׳)

8סְלִיחוֹת בְּמִנְיָן

אוֹמְרִים סְלִיחוֹת בְּמִנְיָן, כִּי י״ג מִדּוֹת רַחֲמִים הֵן ״דָּבָר שֶׁבִּקְדֻשָּׁה״, וְגַם הַקַּדִּישִׁים צְרִיכִים מִנְיָן. אִם הַמִּנְיָן עוֹד לֹא הִתְאַסֵּף, מַתְחִילִים בְּ״אַשְׁרֵי״ וּבַפִּיּוּטִים, וּמְדַלְּגִים עַל י״ג מִדּוֹת עַד שֶׁיַּגִּיעוּ עֲשָׂרָה. מִי שֶׁאֵין לוֹ מִנְיָן יָכוֹל לוֹמַר סְלִיחוֹת לְבַד: הוּא מְדַלֵּג עַל י״ג מִדּוֹת, אוֹ קוֹרֵא אוֹתָן בַּטְּעָמִים, כְּמוֹ שֶׁקּוֹרְאִים בַּתּוֹרָה.
🤔 לָמָּה? בְּי״ג מִדּוֹת רַחֲמִים מִתְגַּלָּה הַמַּעֲלָה שֶׁל כְּלַל יִשְׂרָאֵל, וְלָכֵן אוֹמְרִים אוֹתָן כִּתְפִלָּה רַק כְּשֶׁיֵּשׁ צִבּוּר. מִי שֶׁקּוֹרֵא אוֹתָן בַּטְּעָמִים, קוֹרֵא אוֹתָן כִּפְסוּקִים מִן הַתּוֹרָה.
(פֶּרֶק ב׳, הֲלָכָה ז׳)

9י״ג מִדּוֹת רַחֲמִים

אַחֲרֵי שֶׁמָּחַל ה׳ עַל חֵטְא הָעֵגֶל, בִּקֵּשׁ מֹשֶׁה: ״הַרְאֵנִי נָא אֶת כְּבֹדֶךָ״. וְה׳ גִּלָּה לוֹ אֶת הַמִּדּוֹת שֶׁבָּהֶן הוּא מַנְהִיג אֶת יִשְׂרָאֵל: ״ה׳ ה׳ אֵל רַחוּם וְחַנּוּן, אֶרֶךְ אַפַּיִם וְרַב חֶסֶד וֶאֱמֶת...״. חֲכָמִים אָמְרוּ שֶׁה׳ הֶרְאָה לְמֹשֶׁה סֵדֶר תְּפִלָּה וְאָמַר: ״כָּל זְמַן שֶׁיִּשְׂרָאֵל חוֹטְאִים – יַעֲשׂוּ לְפָנַי כַּסֵּדֶר הַזֶּה, וַאֲנִי מוֹחֵל לָהֶם״. לָכֵן חוֹזְרִים עַל י״ג הַמִּדּוֹת שׁוּב וָשׁוּב בַּסְּלִיחוֹת.
🤔 לָמָּה? דַּוְקָא אַחֲרֵי הַחֵטְא הַנּוֹרָא הִתְבָּרֵר שֶׁהַקֶּשֶׁר בֵּין ה׳ לְיִשְׂרָאֵל הוּא נִצְחִי, וְשׁוּם חֵטְא לֹא מְבַטֵּל אוֹתוֹ. לָכֵן תָּמִיד אֶפְשָׁר לַחֲזֹר בִּתְשׁוּבָה.
💛 כְּשֶׁאֲנַחְנוּ אוֹמְרִים ״ה׳ ה׳ אֵל רַחוּם וְחַנּוּן״, אֲנַחְנוּ מַזְכִּירִים לְעַצְמֵנוּ כַּמָּה אָבִינוּ רוֹצֶה לִסְלֹחַ לָנוּ.
(פֶּרֶק ב׳, הֲלָכָה ח׳)

10עוֹמְדִים אוֹ יוֹשְׁבִים?

בַּסְּלִיחוֹת יֵשׁ שְׁלֹשָׁה מִנְהָגִים. בְּכֻלָּם, מִי שֶׁקָּשֶׁה לוֹ לַעֲמֹד רַשַּׁאי לָשֶׁבֶת, וְיִשְׁתַּדֵּל לַעֲמֹד לְפָחוֹת בַּוִּדּוּי, בְּי״ג מִדּוֹת וּכְשֶׁפּוֹתְחִים אֶת הָאָרוֹן. זְקֵנִים, חַלָּשִׁים וְחוֹלִים שֶׁגַּם זֶה קָשֶׁה לָהֶם – רַשָּׁאִים לָשֶׁבֶת בְּמֶשֶׁךְ כָּל הַסְּלִיחוֹת.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הָאַשְׁכְּנַזִּים אוֹמְרִים אֶת כָּל הַסְּלִיחוֹת בַּעֲמִידָה.
לְמִנְהַג הַסְּפָרַדִּים אוֹמְרִים חֵלֶק מֵהַסְּלִיחוֹת בִּישִׁיבָה וְחֵלֶק בַּעֲמִידָה.
לְמִנְהַג הַתֵּימָנִים אוֹמְרִים אֶת רֹב הַסְּלִיחוֹת בִּישִׁיבָה.
(פֶּרֶק ב׳, הֲלָכָה ז׳)

11מְשַׁל הַמֶּלֶךְ וְהַמַּס

מְדִינָה אַחַת הָיְתָה חַיֶּבֶת לַמֶּלֶךְ מַס גָּדוֹל, וְלֹא הָיָה לָהּ מִמָּה לְשַׁלֵּם. יָצָא הַמֶּלֶךְ עִם צְבָאוֹ לִגְבּוֹת אֶת הַחוֹב. יָצְאוּ לִקְרָאתוֹ גְּדוֹלֵי הַמְּדִינָה וּבִקְּשׁוּ רַחֲמִים, וְהַמֶּלֶךְ וִתֵּר לָהֶם עַל שְׁלִישׁ. אַחַר כָּךְ יָצְאוּ הַבֵּינוֹנִיִּים, וְהוּא וִתֵּר עַל עוֹד שְׁלִישׁ. וּכְשֶׁיָּצְאוּ כֻּלָּם, גְּדוֹלִים וּקְטַנִּים – וִתֵּר לָהֶם עַל הַכֹּל.
🤔 לָמָּה? הַמֶּלֶךְ הוּא ה׳. בְּעֶרֶב רֹאשׁ הַשָּׁנָה גְּדוֹלֵי הַדּוֹר מִתְעַנִּים, וְה׳ מְוַתֵּר עַל שְׁלִישׁ מֵהָעֲווֹנוֹת. בַּעֲשֶׂרֶת יְמֵי תְּשׁוּבָה יְחִידִים מִתְעַנִּים, וְה׳ מְוַתֵּר עַל עוֹד שְׁלִישׁ. וּבְיוֹם הַכִּפֻּרִים כָּל יִשְׂרָאֵל מִתְעַנִּים – אֲנָשִׁים, נָשִׁים וָטַף – וְה׳ מוֹחֵל לָהֶם עַל הַכֹּל.
💛 כְּשֶׁכָּל עַם יִשְׂרָאֵל יוֹצֵא יַחַד אֶל ה׳, גְּדוֹלִים וּקְטַנִּים, גַּם הַתְּפִלָּה שֶׁל יֶלֶד קָטָן חֲשׁוּבָה לוֹ.
(פֶּרֶק ב׳, הֲלָכָה ט׳)

12צָמִים בְּעֶרֶב רֹאשׁ הַשָּׁנָה?

בִּימֵי הָרִאשׁוֹנִים רֹב יִשְׂרָאֵל צָמוּ בְּעֶרֶב רֹאשׁ הַשָּׁנָה, אֲבָל לֹא עַד הָעֶרֶב, כְּדֵי שֶׁלֹּא יִכָּנְסוּ לַחַג כְּשֶׁהֵם מְעֻנִּים. הַיּוֹם רֹב יִשְׂרָאֵל אֵינָם צָמִים, וּמְעַטִּים מְהַדְּרִים לָצוּם עַד חֲצוֹת הַיּוֹם. מִי שֶׁרוֹצֶה לְקַיֵּם אֶת הַמִּנְהָג וְהַצּוֹם קָשֶׁה לוֹ – טוֹב שֶׁיִּפְדֶּה אֶת הַתַּעֲנִית בִּצְדָקָה.
🤔 לָמָּה? גְּדוֹלֵי הַחֲסִידוּת אָמְרוּ שֶׁבַּדּוֹרוֹת הָאַחֲרוֹנִים עִקַּר הָעֲבוֹדָה צְרִיכָה לִהְיוֹת בְּשִׂמְחָה. לָכֵן הַיּוֹם לֹא מַרְבִּים בְּצוֹמוֹת, וּמַרְבִּים בִּצְדָקָה.
(פֶּרֶק ב׳, הֲלָכָה ט׳, הֲלָכָה י׳)

13מִתְכּוֹנְנִים לַחַג

רֹאשׁ הַשָּׁנָה הוּא ״מִקְרָא קֹדֶשׁ״, וּמִתְכּוֹנְנִים אֵלָיו כְּמוֹ לְשַׁבָּת וּלְחַג: מְנַקִּים אֶת הַבַּיִת, מְכַבְּסִים בְּגָדִים, מִתְרַחֲצִים, מְכִינִים מַאֲכָלִים טוֹבִים וְעוֹרְכִים אֶת הַשֻּׁלְחָן. מִי שֶׁצָּרִיךְ לְהִסְתַּפֵּר – מִסְתַּפֵּר לִכְבוֹד הַחַג. יֵשׁ נוֹהֲגִים לִטְבֹּל בְּמִקְוֶה. בְּעֶרֶב רֹאשׁ הַשָּׁנָה גַּם מַתִּירִים נְדָרִים (נִלְמַד עַל כָּךְ בְּפֶרֶק ה׳). וְלֹא תּוֹקְעִים בַּשּׁוֹפָר. מִי שֶׁרוֹצֶה לְהִתְאַמֵּן לִקְרַאת הַתְּקִיעוֹת – תּוֹקֵעַ בְּחֶדֶר סָגוּר.
🤔 לָמָּה? בֶּאֱלוּל הַתְּקִיעוֹת הֵן מִנְהָג, וּבְרֹאשׁ הַשָּׁנָה הֵן מִצְוָה מִן הַתּוֹרָה. הַהַפְסָקָה בְּעֶרֶב רֹאשׁ הַשָּׁנָה מַבְדִּילָה בֵּינֵיהֶן. וְהַמַּאֲכָלִים הַטּוֹבִים הֵם גַּם סִימָן טוֹב לְשָׁנָה טוֹבָה.
(פֶּרֶק ב׳, הֲלָכָה י׳)

14הַחַזָּן שֶׁל הַיָּמִים הַנּוֹרָאִים

בַּיָּמִים הַנּוֹרָאִים מַקְפִּידִים שֶׁהַחַזָּן – ״שְׁלִיחַ הַצִּבּוּר״ – יִהְיֶה רָאוּי לְתַפְקִידוֹ: אָדָם הָגוּן, עָנָו וְאָהוּב עַל הַצִּבּוּר, שֶׁיּוֹדֵעַ הֵיטֵב אֶת הַתְּפִלָּה וְרָגִיל לִלְמֹד תּוֹרָה, וְקוֹלוֹ נָעִים. רָאוּי שֶׁיִּהְיֶה נָשׂוּי וּבֶן שְׁלֹשִׁים לְפָחוֹת. הוּא שָׁר לִכְבוֹד ה׳ וּכְדֵי לְעוֹרֵר אֶת הַצִּבּוּר, וְלֹא כְּדֵי לְהִתְגָּאוֹת בְּקוֹלוֹ. וְאִם בָּחֲרוּ חַזָּן שֶׁאֵינוֹ מַתְאִים – לֹא עוֹשִׂים מַחֲלֹקֶת.
🤔 לָמָּה? הַחַזָּן עוֹמֵד לִפְנֵי ה׳ בְּשֵׁם כָּל הַקָּהָל, וּבַיָּמִים הָאֵלֶּה מְבַקְּשִׁים עַל הַחַיִּים. וּמַחֲלֹקֶת הִיא עֲבֵרָה חֲמוּרָה. חוּץ מִזֶּה, הַיּוֹם כָּל אֶחָד מִתְפַּלֵּל לְעַצְמוֹ מִתּוֹךְ הַמַּחֲזוֹר שֶׁלּוֹ.
💛 חַזָּן שֶׁשָּׁר בְּקוֹל עָרֵב לִכְבוֹד שָׁמַיִם – תָּבוֹא עָלָיו בְּרָכָה, וּשְׂכָרוֹ גָּדוֹל מְאֹד.
(פֶּרֶק ב׳, הֲלָכָה י״א)

15בְּלַחַשׁ – וּבַעֲמִידָה לִכְבוֹד הַתּוֹרָה

הָיוּ שֶׁנָּהֲגוּ לְהִתְפַּלֵּל עֲמִידָה בְּקוֹל בַּיָּמִים הַנּוֹרָאִים, כְּדֵי לְכַוֵּן. אֲבָל טוֹב יוֹתֵר לְהִתְפַּלֵּל בְּלַחַשׁ. וּבְמָקוֹם שֶׁכִּמְעַט כֻּלָּם מִתְפַּלְּלִים בְּלַחַשׁ, כְּמוֹ בְּרֹב הַקְּהִלּוֹת הַיּוֹם – אָסוּר לְהִתְפַּלֵּל בְּקוֹל. כְּשֶׁפּוֹתְחִים אֶת אֲרוֹן הַקֹּדֶשׁ נוֹהֲגִים כֻּלָּם לַעֲמֹד. זְקֵנִים וַחֲלָשִׁים רַשָּׁאִים לָשֶׁבֶת, וְיִתְאַמְּצוּ לַעֲמֹד כְּשֶׁמּוֹלִיכִים אֶת סֵפֶר הַתּוֹרָה.
🤔 לָמָּה? מִי שֶׁמִּתְפַּלֵּל בְּקוֹל מַפְרִיעַ לַכַּוָּנָה שֶׁל הַמִּתְפַּלְּלִים שֶׁסְּבִיבוֹ. וְעוֹמְדִים כְּשֶׁהָאָרוֹן פָּתוּחַ לִכְבוֹד הַתּוֹרָה שֶׁנִּגְלֵית לַצִּבּוּר.
(פֶּרֶק ב׳, הֲלָכָה י״ב)

📝 מָה לָמַדְנוּ

  • אַרְבָּעִים הַיָּמִים מֵרֹאשׁ חֹדֶשׁ אֱלוּל וְעַד יוֹם הַכִּפֻּרִים הֵם הַמְּסֻגָּלִים בְּיוֹתֵר לִתְשׁוּבָה, וְתוֹקְעִים בָּהֶם בַּשּׁוֹפָר.
  • הַסְּפָרַדִּים אוֹמְרִים סְלִיחוֹת בְּכָל אֱלוּל, וְהָאַשְׁכְּנַזִּים מַתְחִילִים בְּמוֹצָאֵי הַשַּׁבָּת שֶׁלִּפְנֵי רֹאשׁ הַשָּׁנָה.
  • הַזְּמַן הַטּוֹב לִסְלִיחוֹת הוּא אַשְׁמֹרֶת הַבֹּקֶר, וְאוֹמְרִים אוֹתָן בְּמִנְיָן בִּגְלַל י״ג מִדּוֹת רַחֲמִים.
  • מִתְכּוֹנְנִים לְרֹאשׁ הַשָּׁנָה כְּמוֹ לְשַׁבָּת, וּבְעֶרֶב רֹאשׁ הַשָּׁנָה לֹא תּוֹקְעִים בַּשּׁוֹפָר.
  • בַּיָּמִים הַנּוֹרָאִים מַקְפִּידִים עַל חַזָּן רָאוּי, וּמִתְפַּלְּלִים עֲמִידָה בְּלַחַשׁ.

❓ שְׁאֵלוֹת חֲזָרָה

  1. מָה קָרָה בְּרֹאשׁ חֹדֶשׁ אֱלוּל וּבְיוֹם הַכִּפֻּרִים אַחֲרֵי חֵטְא הָעֵגֶל, וּמָה לָמַדְנוּ מִזֶּה?
  2. מָה עוֹשֶׂה מִי שֶׁאוֹמֵר סְלִיחוֹת בְּלִי מִנְיָן, וְלָמָּה?
  3. סַפֵּר אֶת מְשַׁל הַמֶּלֶךְ וְהַמַּס, וְהַסְבֵּר אֶת הַנִּמְשָׁל.

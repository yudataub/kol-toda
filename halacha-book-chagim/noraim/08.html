סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים
מַהֲדוּרַת יְלָדִים

💧 דִּינֵי הַתַּעֲנִית

פֶּרֶק ח׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
צָהֳרֵי יוֹם הַכִּפֻּרִים. סָבָא יוֹשֵׁב עַל הַסַּפָּה, חִוֵּר וְחַלָּשׁ, וְהָרוֹפֵא אָמַר לוֹ לִשְׁתּוֹת. ״אֲבָל הַיּוֹם צוֹם!״, אוֹמֵר אָחִיךָ הַקָּטָן. מִי צוֹדֵק? בַּפֶּרֶק הַזֶּה נִלְמַד מָה הִיא מִצְוַת הַתַּעֲנִית, וּמָתַי דַּוְקָא הַמִּצְוָה הִיא לֶאֱכֹל וְלִשְׁתּוֹת.

1אֲפִלּוּ טִפָּה

מִצְוַת עֲשֵׂה לְהִתְעַנּוֹת בְּיוֹם הַכִּפֻּרִים: לֹא אוֹכְלִים אֲפִלּוּ מְעַט, וְלֹא שׁוֹתִים אֲפִלּוּ טִפָּה אַחַת שֶׁל מַיִם. מִי שֶׁאָכַל אוֹ שָׁתָה אֲפִלּוּ מְעַט – עָבַר עַל אִסּוּר תּוֹרָה. הָעֹנֶשׁ הֶחָמוּר (כָּרֵת) הוּא עַל מִי שֶׁאָכַל בְּמֵזִיד כְּמוֹ ״כּוֹתֶבֶת״ – תָּמָר גָּדוֹל – אוֹ שָׁתָה ״מְלֹא לֻגְמָיו״. וְגַם דָּבָר שֶׁאֵינוֹ רָאוּי לַאֲכִילָה – עָלִים, תַּבְלִין חָרִיף מְאֹד אוֹ אֹכֶל מְקֻלְקָל – אָסוּר מִדִּבְרֵי חֲכָמִים.
🤔 לָמָּה? כַּמּוּת כְּמוֹ ״כּוֹתֶבֶת״ מְיַשֶּׁבֶת אֶת הַדַּעַת, וּמִי שֶׁאָכַל אוֹתָהּ כְּבָר אֵינוֹ מְעֻנֶּה. וּמִי שֶׁהֶחְלִיט לֶאֱכֹל בַּצּוֹם דָּבָר מוּזָר – הֶרְאָה שֶׁבִּשְׁבִילוֹ זֶה אֹכֶל.
(פֶּרֶק ח׳, הֲלָכָה א׳)

2חוֹלֶה שֶׁאֵין בּוֹ סַכָּנָה

גַּם חוֹלֶה שֶׁסּוֹבֵל, כָּל עוֹד אֵין סַכָּנָה לְחַיָּיו, אָסוּר לוֹ לֶאֱכֹל וְלִשְׁתּוֹת אֲפִלּוּ מְעַט. לָכֵן חוֹלֶה בְּשַׁפַּעַת אוֹ בְּדַלֶּקֶת גָּרוֹן – חַיָּב לָצוּם. עָדִיף שֶׁיִּשְׁכַּב כָּל הַיּוֹם בְּמִטָּתוֹ וְלֹא יֵלֵךְ לְבֵית הַכְּנֶסֶת, וּבִלְבַד שֶׁלֹּא יִשְׁתֶּה. וְיִשְׁתַּדֵּל לְהִתְפַּלֵּל בַּמִּטָּה כְּפִי יְכָלְתּוֹ; וְאִם קָשֶׁה לוֹ לִקְרֹא בַּמַּחֲזוֹר – יִתְפַּלֵּל תְּפִלּוֹת אִישִׁיּוֹת בְּלִבּוֹ וּבִשְׂפָתָיו.
🤔 לָמָּה? צוֹם יוֹם הַכִּפֻּרִים הוּא מִן הַתּוֹרָה, וְרַק פִּקּוּחַ נֶפֶשׁ דּוֹחֶה אוֹתוֹ. הַצּוֹם חָשׁוּב יוֹתֵר מֵהַתְּפִלָּה בְּבֵית הַכְּנֶסֶת, כִּי הוּא עִקַּר הַמִּצְוָה שֶׁל הַיּוֹם.
(פֶּרֶק ח׳, הֲלָכָה ב׳)

3כַּדּוּרִים בְּלִי מַיִם

חוֹלֶה שֶׁמִּצְטַעֵר בְּגַלַל מַחֲלָתוֹ, אוֹ מִי שֶׁהַצּוֹם גּוֹרֵם לוֹ צַעַר גָּדוֹל – כְּמוֹ כְּאֵב רֹאשׁ חָזָק – רַשַּׁאי לִבְלֹעַ כַּדּוּרִים, בִּתְנַאי שֶׁאֵין לָהֶם טַעַם טוֹב, וּבְלִי מַיִם.
🤔 לָמָּה? כַּדּוּר בְּלִי טַעַם טוֹב, שֶׁבּוֹלְעִים בְּלִי מַיִם, אֵינוֹ כְּמוֹ אֲכִילָה רְגִילָה. כָּךְ אֶפְשָׁר לְהַמְשִׁיךְ לָצוּם בְּלִי צַעַר גָּדוֹל.
(פֶּרֶק ח׳, הֲלָכָה ב׳)

4״וָחַי בָּהֶם״

חוֹלֶה שֶׁהַצּוֹם עָלוּל לְסַכֵּן אֶת חַיָּיו – מִצְוָה שֶׁיֹּאכַל וְיִשְׁתֶּה כְּפִי צָרְכּוֹ. וּמִי שֶׁנִּמְצָא בְּסָפֵק סַכָּנָה וּמַחְמִיר עַל עַצְמוֹ שֶׁלֹּא לֶאֱכֹל – חוֹטֵא. אֲבָל לֹא מַגְזִימִים בַּחֲשָׁשׁוֹת רְחוֹקִים: סַכָּנָה הִיא מַצָּב שֶׁאֲנָשִׁים מְמַהֲרִים לְטַפֵּל בּוֹ, כְּמוֹ לְהָבִיא אֶת הַחוֹלֶה בִּדְחִיפוּת לְבֵית חוֹלִים.
🤔 לָמָּה? ״וּשְׁמַרְתֶּם אֶת חֻקֹּתַי... אֲשֶׁר יַעֲשֶׂה אֹתָם הָאָדָם וָחַי בָּהֶם״ – וָחַי בָּהֶם, וְלֹא שֶׁיָּמוּת בָּהֶם. הַמִּצְווֹת נִתְּנוּ כְּדֵי לִחְיוֹת.
💛 מִי שֶׁהָרוֹפֵא אָמַר לוֹ לֶאֱכֹל – אוֹכֵל בְּשִׂמְחָה, כִּי כְּשֶׁהוּא שׁוֹמֵר עַל חַיָּיו הוּא מְקַיֵּם אֶת רְצוֹן בּוֹרְאוֹ.
(פֶּרֶק ח׳, הֲלָכָה ג׳)

5מִי מַחְלִיט?

אֶת הַשְּׁאֵלָה אִם יֵשׁ סַכָּנָה שׁוֹאֲלִים רוֹפֵא יְרֵא שָׁמַיִם – רוֹפֵא יָשָׁר וְאַחֲרַאי, שֶׁחֲשׁוּבִים לוֹ גַּם קְדֻשַּׁת הַצּוֹם וְגַם חַיֵּי אָדָם. גַּם הַחוֹלֶה צָרִיךְ לְסַפֵּר לָרוֹפֵא אֶת הָאֱמֶת, בְּלִי לְהַגְזִים. וְאִם הַחוֹלֶה עַצְמוֹ מַרְגִּישׁ שֶׁהוּא בְּסַכָּנָה – אוֹכֵל וְשׁוֹתֶה, אֲפִלּוּ אִם הָרוֹפֵא חוֹשֵׁב שֶׁאֵין סַכָּנָה: ״לֵב יוֹדֵעַ מָרַּת נַפְשׁוֹ״.
🤔 לָמָּה? לִפְעָמִים אָדָם מַרְגִּישׁ אֶת חֻמְרַת מַצָּבוֹ יוֹתֵר מִן הָרוֹפְאִים, וּבְכָל סָפֵק שֶׁל סַכָּנַת נְפָשׁוֹת – אוֹכְלִים.
(פֶּרֶק ח׳, הֲלָכָה ד׳)

6לֹא מְפַחֲדִים – וְלֹא מַקִּילִים

לִפְעָמִים אָדָם נֶחְלָשׁ מְאֹד בַּצּוֹם, וּמְפַחֵד שֶׁהוּא בְּסַכָּנָה. אִם הוּא מְבַקֵּשׁ לֶאֱכֹל – נוֹתְנִים לוֹ. בַּהַתְחָלָה נוֹתְנִים לוֹ לִטְעֹם מְעַט, וּפְעָמִים שֶׁבְּזֶה הוּא נִרְגָּע. אִם עֲדַיִן לֹא הִתְאוֹשֵׁשׁ – נוֹתְנִים לוֹ לֶאֱכֹל וְלִשְׁתּוֹת בְּכַמֻּיּוֹת קְטַנּוֹת, ״לְשִׁעוּרִים״, וְאִם גַּם זֶה לֹא עוֹזֵר – כְּפִי צָרְכּוֹ.
🤔 לָמָּה? בַּתַּלְמוּד הַיְרוּשַׁלְמִי מְסֻפָּר עַל רַבִּי חַגַּי, שֶׁנֶּחְלַשׁ מְאֹד בַּצּוֹם. כְּשֶׁרַבִּי מָנָא הוֹרָה לוֹ לִשְׁתּוֹת – הוּא הִרְגִּישׁ שֶׁהוּא מְסֻגָּל לְהִתְאַפֵּק, וְסִיֵּם אֶת הַצּוֹם.
💛 גְּדוֹלֵי יִשְׂרָאֵל הָיוּ הוֹלְכִים בְּיוֹם הַכִּפֻּרִים אֶל חוֹלֶה שֶׁחָשְׁשׁוּ שֶׁיַּחְמִיר עַל עַצְמוֹ, כְּדֵי לְשַׁכְנֵעַ אוֹתוֹ לֶאֱכֹל וְלִשְׁתּוֹת. גַּם זוֹ אַהֲבַת ה׳ וְאַהֲבַת הָאָדָם.
(פֶּרֶק ח׳, הֲלָכָה ה׳, הֲלָכָה ו׳)

7יְלָדִים שֶׁאוֹכְלִים בְּיוֹם הַכִּפֻּרִים

יְלָדִים שֶׁאוֹכְלִים בְּיוֹם הַכִּפֻּרִים אוֹכְלִים וְשׁוֹתִים כָּרָגִיל. הֵם מְבָרְכִים לִפְנֵי הָאֲכִילָה, וּמְבָרְכִים בְּרָכָה אַחֲרוֹנָה. מִי שֶׁאוֹכֵל לֶחֶם נוֹטֵל יָדַיִם רַק עַד סוֹף קִשְׁרֵי הָאֶצְבָּעוֹת, פַּעֲמַיִם עַל כָּל יָד. בְּבִרְכַּת הַמָּזוֹן מוֹסִיפִים ״יַעֲלֶה וְיָבוֹא״ וּמַזְכִּירִים אֶת יוֹם הַכִּפֻּרִים, וְאִם שָׁכַח – אֵינוֹ חוֹזֵר.
🤔 לָמָּה? גַּם בְּיוֹם הַצּוֹם, הָאֹכֶל שֶׁל מִי שֶׁמֻּתָּר לוֹ לֶאֱכֹל הוּא אֹכֶל כָּשֵׁר וְטוֹב, וּמְבָרְכִים עָלָיו כְּמוֹ תָּמִיד. וְאֶת הַנְּטִילָה מְצַמְצְמִים, כִּי בְּיוֹם הַכִּפֻּרִים יֵשׁ אִסּוּר רְחִיצָה.
(פֶּרֶק ח׳, הֲלָכָה ז׳)

📝 מָה לָמַדְנוּ

  • בְּיוֹם הַכִּפֻּרִים אָסוּר לֶאֱכֹל וְלִשְׁתּוֹת אֲפִלּוּ מְעַט – זֶה אִסּוּר תּוֹרָה.
  • חוֹלֶה שֶׁאֵין בּוֹ סַכָּנָה צָם, וְעָדִיף שֶׁיִּשְׁכַּב בַּמִּטָּה וְלֹא יֵלֵךְ לְבֵית הַכְּנֶסֶת.
  • חוֹלֶה שֶׁיֵּשׁ בּוֹ סַכָּנָה – מִצְוָה שֶׁיֹּאכַל וְיִשְׁתֶּה: ״וָחַי בָּהֶם, וְלֹא שֶׁיָּמוּת בָּהֶם״.
  • שׁוֹאֲלִים רוֹפֵא יְרֵא שָׁמַיִם, וְאִם הַחוֹלֶה מַרְגִּישׁ סַכָּנָה – אוֹכֵל.
  • יְלָדִים שֶׁאוֹכְלִים מְבָרְכִים כָּרָגִיל, וּמַזְכִּירִים אֶת יוֹם הַכִּפֻּרִים בְּבִרְכַּת הַמָּזוֹן.

❓ שְׁאֵלוֹת חֲזָרָה

  1. חוֹלֶה בְּשַׁפַּעַת בְּיוֹם הַכִּפֻּרִים: הַאִם הוּא צָם? וְהַאִם עָדִיף שֶׁיֵּלֵךְ לְבֵית הַכְּנֶסֶת?
  2. מָה פֵּרוּשׁ ״וָחַי בָּהֶם״, וּמָה דִּינוֹ שֶׁל מִי שֶׁבְּסָפֵק סַכָּנָה וּמַחְמִיר לָצוּם?
  3. סַפֵּר עַל רַבִּי חַגַּי וְרַבִּי מָנָא. מָה לוֹמְדִים מֵהַסִּפּוּר עַל חוֹלֶה שֶׁמְּפַחֵד?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״יוֹם הַכִּפּוּרִים״

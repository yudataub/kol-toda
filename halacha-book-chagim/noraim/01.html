סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים
מַהֲדוּרַת יְלָדִים

⚖️ הַדִּין, הַשָּׂכָר וְהָעֹנֶשׁ

פֶּרֶק א׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
מָחָר יֵשׁ לְךָ מִבְחָן. אַתָּה יוֹשֵׁב וְחוֹזֵר עַל הַחֹמֶר, קְצָת מִתְרַגֵּשׁ וּקְצָת מְקַוֶּה שֶׁיִּהְיֶה טוֹב. עַכְשָׁו תְּאַר לְעַצְמְךָ שֶׁפַּעַם בְּשָׁנָה כָּל הָעוֹלָם כֻּלּוֹ עוֹמֵד לִפְנֵי ה׳... זֶה רֹאשׁ הַשָּׁנָה. בַּפֶּרֶק הַזֶּה נִלְמַד מָה זֶה ״יוֹם הַדִּין״, וְלָמָּה אֲנַחְנוּ נִגָּשִׁים אֵלָיו גַּם בְּיִרְאָה וְגַם בְּשִׂמְחָה.

1חַיִּים חֲדָשִׁים – וְגַם דִּין

בְּכָל שָׁנָה ה׳ נוֹתֵן חַיִּים חֲדָשִׁים לְכָל בְּרִיָּה וּבְרִיָּה בָּעוֹלָם. וּבְאוֹתוֹ זְמַן הוּא גַּם דָּן אֶת כֻּלָּם: לַטּוֹבִים הוּא נוֹתֵן שֶׁפַע וּבְרָכָה, וּמֵהָרָעִים הוּא מְמַעֵט אֶת הַבְּרָכָה. לָכֵן רֹאשׁ הַשָּׁנָה, הַיּוֹם שֶׁבּוֹ נִבְרֵאת הַשָּׁנָה הַחֲדָשָׁה, הוּא יוֹם הַדִּין שֶׁל כָּל הָעוֹלָם: ״כָּל בָּאֵי עוֹלָם עוֹבְרִים לְפָנָיו״.
🤔 לָמָּה? אִם גַּם הָרְשָׁעִים הָיוּ מְקַבְּלִים אֶת כָּל הַבְּרָכָה, הֵם הָיוּ מִתְחַזְּקִים בָּרִשְׁעוּת שֶׁלָּהֶם וּמַזִּיקִים לָעוֹלָם כֻּלּוֹ. הַדִּין שׁוֹמֵר שֶׁהַטּוֹב שֶׁל ה׳ יַגִּיעַ לַמָּקוֹם הַנָּכוֹן.
💛 ה׳ לֹא שׁוֹכֵחַ אַף אֶחָד. גַּם אוֹתְךָ הוּא זוֹכֵר, וּבְכָל שָׁנָה הוּא בּוֹחֵר מֵחָדָשׁ לָתֵת לְךָ חַיִּים.
(פֶּרֶק א׳, הֲלָכָה א׳, הֲלָכָה ג׳)

2נִכְתָּבִים וְנֶחְתָּמִים

רַבִּי מֵאִיר אָמַר: ״הַכֹּל נִדּוֹנִים בְּרֹאשׁ הַשָּׁנָה, וּגְזַר דִּינָם נֶחְתָּם בְּיוֹם הַכִּפֻּרִים״. בְּרֹאשׁ הַשָּׁנָה הַדִּין נִכְתָּב, וּבְיוֹם הַכִּפֻּרִים הוּא נֶחְתָּם. הַיָּמִים שֶׁבֵּינֵיהֶם הֵם זְמַן שֶׁל תְּשׁוּבָה וּתְפִלָּה, כְּדֵי לְהֵיטִיב אֶת הַדִּין. וּבְמִקְרִים מְיֻחָדִים אֶפְשָׁר לְשַׁנּוֹת אוֹתוֹ אֲפִלּוּ עַד הוֹשַׁעְנָא רַבָּה וּשְׁמִינִי עֲצֶרֶת.
🤔 לָמָּה? כְּמוֹ מִכְתָּב שֶׁכְּבָר נִכְתַּב אֲבָל עוֹד לֹא נֶחְתַּם: עוֹד אֶפְשָׁר לְתַקֵּן בּוֹ. ה׳ נוֹתֵן לָנוּ זְמַן לְשַׁפֵּר אֶת הַ״מִּכְתָּב״ שֶׁל הַשָּׁנָה שֶׁלָּנוּ.
(פֶּרֶק א׳, הֲלָכָה ג׳)

3״דִּרְשׁוּ ה׳ בְּהִמָּצְאוֹ״

כָּל יְמוֹת הַשָּׁנָה טוֹבִים לִתְשׁוּבָה. אֲבָל בַּעֲשֶׂרֶת הַיָּמִים שֶׁמֵּרֹאשׁ הַשָּׁנָה וְעַד יוֹם הַכִּפֻּרִים, הַתְּשׁוּבָה מִתְקַבֶּלֶת יוֹתֵר, וְלָכֵן הֵם נִקְרָאִים ״עֲשֶׂרֶת יְמֵי תְּשׁוּבָה״. כָּךְ אָמַר הַנָּבִיא יְשַׁעְיָהוּ: ״דִּרְשׁוּ ה׳ בְּהִמָּצְאוֹ, קְרָאֻהוּ בִּהְיוֹתוֹ קָרוֹב״.
🤔 לָמָּה? בַּיָּמִים שֶׁבָּהֶם ה׳ נוֹתֵן חַיִּים חֲדָשִׁים, הוּא מִתְקָרֵב אֶל הַבְּרוּאִים שֶׁלּוֹ. וּכְשֶׁהַמֶּלֶךְ קָרוֹב, קַל יוֹתֵר לְדַבֵּר אִתּוֹ.
💛 בַּיָּמִים הָאֵלֶּה ה׳ כְּאִלּוּ יוֹצֵא לִקְרָאתֵנוּ. כָּל מָה שֶׁנִּשְׁאַר לָנוּ הוּא לָלֶכֶת אֵלָיו.
(פֶּרֶק א׳, הֲלָכָה א׳)

4מַקְדִּימִים תְּרוּפָה לַמַּכָּה

הַדִּין עַצְמוֹ מַתְחִיל בְּרֹאשׁ הַשָּׁנָה. אֲבָל עָדִיף לְהִתְעוֹרֵר לִתְשׁוּבָה עוֹד לִפְנֵי כֵן, כְּדֵי שֶׁכְּשֶׁיַּגִּיעוּ עֲשֶׂרֶת יְמֵי תְּשׁוּבָה נוּכַל לָשׁוּב אֶל ה׳ בֶּאֱמֶת. לָכֵן נָהֲגוּ יִשְׂרָאֵל לְהִתְעוֹרֵר לִתְשׁוּבָה כְּבָר בְּחֹדֶשׁ אֱלוּל.
🤔 לָמָּה? כָּךְ זֶה גַּם בְּבֵית מִשְׁפָּט: לִפְנֵי שֶׁהַמִּשְׁפָּט מַתְחִיל, קַל יוֹתֵר לְהִתְחָרֵט וּלְתַקֵּן. אֲבָל כְּשֶׁהַמְקַטְרֵג כְּבָר עוֹמֵד וְטוֹעֵן אֶת טַעֲנוֹתָיו, קָשֶׁה יוֹתֵר לְבַטֵּל אֶת הַקִּטְרוּג.
(פֶּרֶק א׳, הֲלָכָה א׳)

5בְּיִרְאָה וּבְשִׂמְחָה

בְּכָל שָׁנָה אֲנַחְנוּ נִגָּשִׁים לִימֵי הַתְּשׁוּבָה בְּיִרְאָה וּבְשִׂמְחָה. יִרְאָה – כִּי אֵינֶנּוּ יוֹדְעִים מָה יִגָּזֵר עָלֵינוּ. וְשִׂמְחָה – עַל הַהִזְדַּמְּנוּת לַחֲזֹר אֶל ה׳, לַעֲמֹד לְפָנָיו בִּתְפִלָּה, לְהִתְנַקּוֹת מִן הָרַע וּלְהִתְחַבֵּר מֵחָדָשׁ לְכָל הַטּוֹב.
🤔 לָמָּה? בְּלִי חֶשְׁבּוֹן נֶפֶשׁ שֶׁל פַּעַם בְּשָׁנָה, הָיִינוּ שׁוֹכְחִים אֶת כָּל הַדְּבָרִים הַטּוֹבִים שֶׁרָצִינוּ לִלְמֹד וְלַעֲשׂוֹת. הַיָּמִים הַנּוֹרָאִים מַזְכִּירִים לָנוּ אוֹתָם, וְכָךְ אֲנַחְנוּ עוֹלִים מִשָּׁנָה לְשָׁנָה.
💛 הַשִּׂמְחָה שֶׁל יְמֵי הַתְּשׁוּבָה הִיא שִׂמְחָה שֶׁל יֶלֶד שֶׁחוֹזֵר הַבַּיְתָה וְיוֹדֵעַ שֶׁמְּחַכִּים לוֹ.
(פֶּרֶק א׳, הֲלָכָה א׳)

6מַתָּנָה שֶׁל בְּחִירָה

ה׳ נָתַן לָאָדָם בְּחִירָה חָפְשִׁית. מִי שֶׁבּוֹחֵר בַּטּוֹב מִתְקָרֵב לַה׳, מְקוֹר הַחַיִּים, וְזוֹכֶה לִבְרָכָה. וּמִי שֶׁמִּתְרַחֵק מִמֶּנּוּ, מִתְרַחֵק גַּם מִן הַבְּרָכָה. עַל כָּל מִצְוָה מְקַבְּלִים שָׂכָר. וְהָעֹנֶשׁ אֵינוֹ נְקָמָה, אֶלָּא תִּקּוּן: ״כִּי כַּאֲשֶׁר יְיַסֵּר אִישׁ אֶת בְּנוֹ, ה׳ אֱלֹהֶיךָ מְיַסְּרֶךָּ״.
🤔 לָמָּה? אִם ה׳ הָיָה נוֹתֵן לָנוּ הַכֹּל בְּמַתָּנָה, לֹא הָיְתָה לָנוּ שִׂמְחָה אֲמִתִּית. כְּשֶׁאֲנַחְנוּ בּוֹחֲרִים בַּטּוֹב, הַטּוֹב שֶׁאֲנַחְנוּ מְקַבְּלִים הוּא בֶּאֱמֶת שֶׁלָּנוּ.
💛 ״טוֹב ה׳ לַכֹּל וְרַחֲמָיו עַל כָּל מַעֲשָׂיו״. גַּם כְּשֶׁה׳ מְחַנֵּךְ אוֹתָנוּ, הוּא עוֹשֶׂה זֹאת כְּמוֹ אַבָּא אוֹהֵב.
(פֶּרֶק א׳, הֲלָכָה ב׳)

7דִּין גַּם בָּרְגָלִים

בְּרֹאשׁ הַשָּׁנָה יֵשׁ דִּין כְּלָלִי עַל כָּל הַשָּׁנָה. וּבִשְׁלֹשֶׁת הָרְגָלִים יֵשׁ דִּין מְיֻחָד: בְּפֶסַח נִדּוֹנִים עַל הַתְּבוּאָה, בְּשָׁבוּעוֹת עַל פֵּרוֹת הָאִילָן, וּבְסֻכּוֹת עַל הַמַּיִם.
🤔 לָמָּה? הַחַגִּים מַתְאִימִים לַטֶּבַע: סֻכּוֹת בָּא לִקְרַאת הַחֹרֶף וְהַגְּשָׁמִים, בְּפֶסַח הַתְּבוּאָה גְּדֵלָה, וּבְשָׁבוּעוֹת הַפֵּרוֹת מַתְחִילִים לְהַבְשִׁיל. הַבְּרָכָה מַגִּיעָה לָעוֹלָם דֶּרֶךְ הַיָּמִים הַקְּדוֹשִׁים, וְיַחַד עִמָּהּ גַּם הַדִּין.
(פֶּרֶק א׳, הֲלָכָה ג׳)

8גַּם בְּכָל יוֹם

הַבְּרָכָה שֶׁנִּקְבְּעָה בְּרֹאשׁ הַשָּׁנָה יוֹרֶדֶת לָעוֹלָם לְאַט לְאַט: דֶּרֶךְ רָאשֵׁי הַחֳדָשִׁים, הַשַּׁבָּתוֹת וְכָל יְמוֹת הַשָּׁבוּעַ. לָכֵן גַּם רֹאשׁ חֹדֶשׁ הוּא זְמַן טוֹב לִתְשׁוּבָה, וְהַמְּהַדְּרִים עוֹשִׂים תְּשׁוּבָה בְּעֶרֶב רֹאשׁ חֹדֶשׁ – ״יוֹם כִּפּוּר קָטָן״. חֲכָמִים אָמְרוּ: ״אָדָם נִדּוֹן בְּכָל יוֹם״, וְלָכֵן אֲנַחְנוּ מִתְפַּלְּלִים בְּכָל יוֹם שַׁחֲרִית, מִנְחָה וְעַרְבִית.
🤔 לָמָּה? חֲכָמִים סִפְּרוּ: אִם בְּרֹאשׁ הַשָּׁנָה נִגְזְרוּ עַל יִשְׂרָאֵל מְעַט גְּשָׁמִים, וּבְמֶשֶׁךְ הַשָּׁנָה הֵם חָזְרוּ בִּתְשׁוּבָה – ה׳ מוֹרִיד אֶת הַגְּשָׁמִים הַמְּעַטִּים בַּזְּמַן הַנָּכוֹן וּבַמָּקוֹם שֶׁצָּרִיךְ אוֹתָם, וְהֵם מְבִיאִים בְּרָכָה גְּדוֹלָה. כָּךְ הַמַּעֲשִׂים שֶׁלָּנוּ בְּכָל יוֹם מְשַׁנִּים אֶת הַדֶּרֶךְ שֶׁבָּהּ הַבְּרָכָה מַגִּיעָה.
💛 כָּל יוֹם הוּא הִזְדַּמְּנוּת חֲדָשָׁה לְהִתְקָרֵב לַה׳. לֹא צָרִיךְ לְחַכּוֹת לְרֹאשׁ הַשָּׁנָה הַבָּא.
(פֶּרֶק א׳, הֲלָכָה ד׳)

9יִשְׂרָאֵל – הַלֵּב שֶׁל הָעוֹלָם

יִשְׂרָאֵל בֵּין הָעַמִּים הֵם כְּמוֹ הַלֵּב בֵּין אֵיבְרֵי הַגּוּף. הָעוֹלָם כֻּלּוֹ צָרִיךְ אֶת אוֹר הַתּוֹרָה שֶׁיִּשְׂרָאֵל מְגַלִּים בּוֹ. לָכֵן בְּרֹאשׁ הַשָּׁנָה ה׳ דָּן קֹדֶם אֶת יִשְׂרָאֵל, וְאַחַר כָּךְ אֶת שְׁאָר הָעַמִּים: ״כִּי חֹק לְיִשְׂרָאֵל הוּא, מִשְׁפָּט לֵאלֹהֵי יַעֲקֹב״.
🤔 לָמָּה? מִי שֶׁיֵּשׁ לוֹ אַחֲרָיוּת גְּדוֹלָה, גַּם הַשָּׂכָר שֶׁלּוֹ גָּדוֹל. כְּשֶׁיִּשְׂרָאֵל בּוֹחֲרִים בַּטּוֹב, הֵם מְבִיאִים בְּרָכָה לְכָל הָעוֹלָם.
💛 ה׳ כָּרַת בְּרִית עִם עַמּוֹ: ״כִּי לֹא יִטֹּשׁ ה׳ עַמּוֹ וְנַחֲלָתוֹ לֹא יַעֲזֹב״. הַדִּין קוֹבֵעַ אֵיךְ תַּעֲבֹר עָלֵינוּ הַשָּׁנָה, אֲבָל הָאַהֲבָה שֶׁל ה׳ לְעַמּוֹ אֵינָהּ מִשְׁתַּנָּה.
(פֶּרֶק א׳, הֲלָכָה ט׳)

📝 מָה לָמַדְנוּ

  • בְּרֹאשׁ הַשָּׁנָה ה׳ נוֹתֵן חַיִּים חֲדָשִׁים לָעוֹלָם וְדָן אֶת כָּל הַבְּרוּאִים.
  • בְּרֹאשׁ הַשָּׁנָה הַדִּין נִכְתָּב, וּבְיוֹם הַכִּפֻּרִים הוּא נֶחְתָּם. בֵּינֵיהֶם – עֲשֶׂרֶת יְמֵי תְּשׁוּבָה.
  • מִתְעוֹרְרִים לִתְשׁוּבָה כְּבָר בֶּאֱלוּל, וְנִגָּשִׁים לַדִּין בְּיִרְאָה וּבְשִׂמְחָה.
  • הָעֹנֶשׁ שֶׁל ה׳ הוּא תִּקּוּן, כְּמוֹ אַבָּא שֶׁמְּחַנֵּךְ אֶת בְּנוֹ.
  • יֵשׁ דִּין גַּם בָּרְגָלִים וְגַם בְּכָל יוֹם, וְכָל יוֹם הוּא הִזְדַּמְּנוּת לְהִתְקָרֵב.

❓ שְׁאֵלוֹת חֲזָרָה

  1. מָה הַהֶבְדֵּל בֵּין מָה שֶׁקּוֹרֶה בְּרֹאשׁ הַשָּׁנָה לְמָה שֶׁקּוֹרֶה בְּיוֹם הַכִּפֻּרִים?
  2. לָמָּה כְּדַאי לְהִתְחִיל בִּתְשׁוּבָה כְּבָר בְּחֹדֶשׁ אֱלוּל? הַסְבֵּר בְּעֶזְרַת הַמָּשָׁל שֶׁל בֵּית הַמִּשְׁפָּט.
  3. עַל מָה נִדּוֹנִים בְּפֶסַח, בְּשָׁבוּעוֹת וּבְסֻכּוֹת, וְלָמָּה דַּוְקָא בַּחַגִּים הָאֵלֶּה?

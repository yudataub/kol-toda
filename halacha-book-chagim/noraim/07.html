סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים
מַהֲדוּרַת יְלָדִים

🕯️ הִלְכוֹת יוֹם הַכִּפֻּרִים

פֶּרֶק ז׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
הַנֵּרוֹת דּוֹלְקִים, אַבָּא לָבוּשׁ לָבָן, וְכֻלָּם הוֹלְכִים לְבֵית הַכְּנֶסֶת לְ״כָל נִדְרֵי״. מָחָר בָּעֶרֶב, אַחֲרֵי יוֹם שָׁלֵם שֶׁל תְּפִלּוֹת, יִשָּׁמַע הַשּׁוֹפָר. מָה קוֹרֶה בְּתוֹךְ הַיּוֹם הֲכִי קָדוֹשׁ בַּשָּׁנָה? בַּפֶּרֶק הַזֶּה נַעֲבֹר עַל יוֹם הַכִּפֻּרִים מֵהַהַתְחָלָה וְעַד הַסּוֹף: הַבְּגָדִים, הַוִּדּוּי, הַתְּפִלּוֹת וְהַהַבְדָּלָה.

1בְּגָדִים לְבָנִים

בְּשַׁבָּת וּבְחַג מְכַבְּדִים אֶת הַיּוֹם בְּמַאֲכָל, בְּמַשְׁקֶה וּבִבְגָדִים נְקִיִּים. בְּיוֹם הַכִּפֻּרִים, שֶׁבּוֹ לֹא אוֹכְלִים, מְכַבְּדִים אוֹתוֹ בִּבְגָדִים נְקִיִּים. רַבִּים לוֹבְשִׁים לָבָן, וְגַם נָשִׁים רַבּוֹת לוֹבְשׁוֹת לָבָן. גַּם מִי שֶׁאֵינָהּ לוֹבֶשֶׁת לָבָן – אֵינָהּ מִתְקַשֶּׁטֶת בְּתַכְשִׁיטִים, מִפְּנֵי אֵימַת יוֹם הַדִּין. גְּבָרִים רַבִּים מִיּוֹצְאֵי אַשְׁכְּנַז לוֹבְשִׁים ״קִיטְל״ – בֶּגֶד לָבָן מְיֻחָד לִתְפִלָּה.
🤔 לָמָּה? בְּיוֹם הַכִּפֻּרִים אֲנַחְנוּ לֹא נִגְרָרִים אַחֲרֵי תַּאֲווֹת הַגּוּף, וַאֲנַחְנוּ נְקִיִּים כְּמוֹ מַלְאֲכֵי הַשָּׁרֵת. הַקִּיטְל מַזְכִּיר גַּם אֶת הַמַּלְאָכִים, וְגַם אֶת הַתַּכְרִיכִים – וְכָךְ הַלֵּב נִכְנָע וּמִתְעוֹרֵר לִתְשׁוּבָה.
💛 הַבֶּגֶד הַלָּבָן אוֹמֵר: הַיּוֹם אֲנִי רוֹצֶה לִהְיוֹת נָקִי, כְּמוֹ מַלְאָךְ שֶׁעוֹמֵד לִפְנֵי ה׳.
(פֶּרֶק ז׳, הֲלָכָה א׳)

2בַּיִת וּבֵית כְּנֶסֶת מְכֻבָּדִים

מְנַקִּים אֶת הַבַּיִת לִכְבוֹד יוֹם הַכִּפֻּרִים, פּוֹרְשִׂים עַל הַשֻּׁלְחָן מַפָּה נָאָה כְּמוֹ בְּשַׁבָּת, וּמִתְרַחֲצִים לִפְנֵי הַחַג. מְנַקִּים וּמְכִינִים גַּם אֶת בֵּית הַכְּנֶסֶת, וּמַדְלִיקִים בּוֹ אֶת כָּל הָאוֹרוֹת: ״בָּאֻרִים כַּבְּדוּ ה׳״. וּבְיוֹם הַכִּפֻּרִים מֻתָּר לְהָרִיחַ בְּשָׂמִים, וּמְבָרְכִים עֲלֵיהֶם – שֶׁלֹּא כְּמוֹ בְּתִשְׁעָה בְּאָב.
🤔 לָמָּה? יוֹם הַכִּפֻּרִים הוּא יוֹם טוֹב, וְכָל מָה שֶׁמְּכַבֵּד אֶת הַיּוֹם וְאֵינוֹ אָסוּר בִּגְלַל הַתַּעֲנִית – מִצְוָה לַעֲשׂוֹתוֹ. תִּשְׁעָה בְּאָב הוּא יוֹם אֵבֶל עַל הַחֻרְבָּן, וְיוֹם הַכִּפֻּרִים הוּא יוֹם שֶׁל טָהֳרָה.
(פֶּרֶק ז׳, הֲלָכָה א׳)

3שַׁבַּת שַׁבָּתוֹן

בְּיוֹם הַכִּפֻּרִים אֲסוּרוֹת כָּל הַמְּלָאכוֹת שֶׁאֲסוּרוֹת בְּשַׁבָּת: ״שַׁבַּת שַׁבָּתוֹן הוּא לָכֶם״. וּכְמוֹ בְּשַׁבָּת, לֹא עוֹשִׂים אוֹתוֹ כְּיוֹם חֹל: לֹא פּוֹתְחִים חֲנוּת, לֹא סוֹחֲבִים מַשָּׂאוֹת לִקְרַאת יְמֵי הַחֹל, וְהַהִלּוּךְ וְהַדִּבּוּר שֶׁלָּנוּ שׁוֹנִים מִבְּיוֹם חֹל.
🤔 לָמָּה? כָּל דִּינֵי שַׁבָּת חָלִים בְּיוֹם הַכִּפֻּרִים, וְנוֹסֶפֶת עֲלֵיהֶם הַתַּעֲנִית. לָכֵן יֵשׁ בּוֹ עוֹד יוֹתֵר שְׁבִיתָה וּפְרִישָׁה מֵעִנְיְנֵי הָעוֹלָם.
(פֶּרֶק ז׳, הֲלָכָה ב׳)

4מִצְוַת הַוִּדּוּי

מִצְוָה עַל כָּל אָדָם לָשׁוּב בִּתְשׁוּבָה וּלְהִתְוַדּוֹת עַל חֲטָאָיו בְּיוֹם הַכִּפֻּרִים. הַוִּדּוּי הוּא לוֹמַר בַּפֶּה אֶת מָה שֶׁבַּלֵּב. גַּם בְּכָל הַשָּׁנָה, מִי שֶׁחָטָא בְּטָעוּת אוֹמֵר לִפְנֵי ה׳: ״חָטָאתִי״, מִי שֶׁחָטָא בְּכַוָּנָה – ״עָוִיתִי״, וּמִי שֶׁחָטָא בְּמֶרֶד – ״פָּשַׁעְתִּי״, וּבְכָךְ הוּא מְקַיֵּם אֶת מִצְוַת הַוִּדּוּי.
🤔 לָמָּה? כְּשֶׁמּוֹצִיאִים אֶת הַחֲרָטָה מִן הַלֵּב אֶל הַפֶּה, הִיא נַעֲשֵׂית בְּרוּרָה וַחֲזָקָה, וְהַהַחְלָטָה שֶׁלֹּא לַחֲטֹא עוֹד מִתְחַזֶּקֶת. בִּזְכוּת הַוִּדּוּי הַתְּשׁוּבָה נִגְמֶרֶת.
💛 לוֹמַר ״טָעִיתִי״ אֵינוֹ חֻלְשָׁה – זֶה אֹמֶץ. וְה׳ מְחַכֶּה לַמִּלָּה הַזֹּאת.
(פֶּרֶק ז׳, הֲלָכָה ג׳)

5״אָשַׁמְנוּ״ – בִּלְשׁוֹן רַבִּים

הַנֻּסָּח שֶׁחַיָּבִים לוֹמַר בְּיוֹם הַכִּפֻּרִים הוּא: ״אֲבָל אֲנַחְנוּ חָטָאנוּ, עָוִינוּ, פָּשַׁעְנוּ״. נוֹהֲגִים לְהוֹסִיף ״אָשַׁמְנוּ, בָּגַדְנוּ...״ לְפִי סֵדֶר הָאוֹתִיּוֹת, וְ״עַל חֵטְא שֶׁחָטָאנוּ לְפָנֶיךָ״. מִתְוַדִּים בַּעֲמִידָה, כּוֹפְפִים מְעַט אֶת הָרֹאשׁ, וּמַכִּים בָּאֶגְרוֹף עַל הֶחָזֶה – כְּאוֹמֵר: ״הַלֵּב שֶׁלִּי גָּרַם לַחֵטְא״.
🤔 לָמָּה? הַוִּדּוּי הוּא בִּלְשׁוֹן רַבִּים, כִּי כָּל יִשְׂרָאֵל עֲרֵבִים זֶה לָזֶה, כְּמוֹ גּוּף אֶחָד. כָּל אֶחָד מִתְוַדֶּה בְּשֵׁם כָּל הָעָם, כְּמוֹ שֶׁהַכֹּהֵן הַגָּדוֹל הִתְוַדָּה בְּשֵׁם כָּל יִשְׂרָאֵל. לָכֵן גַּם צַדִּיק אוֹמֵר ״גָּזַלְנוּ״.
💛 כְּשֶׁאֲנִי אוֹמֵר ״חָטָאנוּ״, אֲנִי לֹא לְבַד: כָּל עַם יִשְׂרָאֵל עוֹמֵד אִתִּי לִפְנֵי ה׳.
(פֶּרֶק ז׳, הֲלָכָה ד׳)

6בְּסֵתֶר אוֹ בְּגָלוּי?

אֶפְשָׁר לָצֵאת יְדֵי חוֹבַת הַוִּדּוּי גַּם בְּלִי לְפָרֵט אֶת הַחֵטְא, בַּנֻּסָּח שֶׁבַּמַּחֲזוֹר. מִי שֶׁיּוֹדֵעַ שֶׁהַפֵּרוּט יַעֲמִיק אֶת הַתְּשׁוּבָה שֶׁלּוֹ – טוֹב שֶׁיְּפָרֵט בְּלַחַשׁ. עַל חֲטָאִים שֶׁבֵּין אָדָם לַמָּקוֹם מִתְוַדִּים בְּדֶרֶךְ כְּלָל בְּסֵתֶר. אֲבָל עַל פְּגִיעָה בְּחָבֵר – עָדִיף בְּדֶרֶךְ כְּלָל לְהוֹדוֹת לִפְנֵי רַבִּים, כְּדֵי שֶׁכֻּלָּם יִרְאוּ שֶׁכְּבוֹד הֶחָבֵר חָשׁוּב, אֶלָּא אִם כֵּן זֶה יְבַיֵּשׁ אוֹתוֹ.
🤔 לָמָּה? מִי שֶׁפָּגַע בְּחָבֵר לִפְנֵי כֻּלָּם וּמְבַקֵּשׁ סְלִיחָה רַק בְּסֵתֶר – הַתְּשׁוּבָה שֶׁלּוֹ אֵינָהּ שְׁלֵמָה. וְהַבּוּשָׁה מִן הַחֵטְא הִיא חֵלֶק חָשׁוּב מִן הַתְּשׁוּבָה.
(פֶּרֶק ז׳, הֲלָכָה ו׳, הֲלָכָה ז׳)

7עֲשָׂרָה וִדּוּיִים

אוֹמְרִים וִדּוּי בְּמִנְחָה שֶׁלִּפְנֵי הַסְּעוּדָה הַמַּפְסֶקֶת, בְּעַרְבִית, וּבְשַׁחֲרִית, מוּסָף, מִנְחָה וּנְעִילָה – בְּכָל אַחַת פַּעַם בְּלַחַשׁ וּפַעַם עִם הַחַזָּן. בְּיַחַד – עֲשָׂרָה וִדּוּיִים. נָשִׁים וְחוֹלִים שֶׁאֵינָם בָּאִים לְבֵית הַכְּנֶסֶת אֵינָם חַיָּבִים בַּעֲשָׂרָה וִדּוּיִים: אוֹמְרִים וִדּוּי בְּכָל תְּפִלָּה שֶׁמִּתְפַּלְּלִים, וּמִי שֶׁאֵינוֹ מִתְפַּלֵּל – אוֹמֵר לְפָחוֹת וִדּוּי אֶחָד, וְנָכוֹן לוֹמַר אֶחָד בִּכְנִיסַת הַיּוֹם וְאֶחָד בִּשְׁעַת הַנְּעִילָה.
🤔 לָמָּה? עֲשָׂרָה וִדּוּיִים – כְּנֶגֶד עֶשֶׂר הַפְּעָמִים שֶׁהַכֹּהֵן הַגָּדוֹל הִזְכִּיר אֶת הַשֵּׁם הַמְּפֹרָשׁ בְּיוֹם הַכִּפֻּרִים. וּמִצְוַת הַיּוֹם הִיא תְּשׁוּבָה, וְלָכֵן בְּכָל פַּעַם שֶׁעוֹמְדִים לִפְנֵי ה׳ – מִתְוַדִּים.
(פֶּרֶק ז׳, הֲלָכָה ה׳)

8נִכְנָסִים לַיּוֹם בִּתְשׁוּבָה

הָרַמְבַּ״ן כָּתַב שֶׁצָּרִיךְ לְהִכָּנֵס לְיוֹם הַכִּפֻּרִים מִתּוֹךְ וִדּוּי, וְכָךְ רָאוּי לִנְהֹג. לָכֵן לִפְנֵי כְּנִיסַת הַיּוֹם אוֹמְרִים תְּפִלָּה שֶׁיֵּשׁ בָּהּ וִדּוּי:
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הָאַשְׁכְּנַזִּים אוֹמְרִים ״תְּפִלָּה זַכָּה״.
לְמִנְהַג הַסְּפָרַדִּים אוֹמְרִים אֶת הַפִּיּוּט ״לְךָ אֵלִי תְּשׁוּקָתִי״.
💛 מִי שֶׁנִּכְנָס לַיּוֹם הַקָּדוֹשׁ כְּשֶׁהוּא כְּבָר מִתְחָרֵט – נִכְנָס כְּמוֹ בֵּן שֶׁכְּבָר בַּדֶּרֶךְ הַבַּיְתָה.
(פֶּרֶק ז׳, הֲלָכָה ה׳)

9מוֹסִיפִים מֵחֹל עַל הַקֹּדֶשׁ

כְּמוֹ כָּל יוֹם בַּתּוֹרָה, יוֹם הַכִּפֻּרִים מַתְחִיל בַּלַּיְלָה: הוּא נִכְנָס בִּשְׁקִיעַת הַחַמָּה וְיוֹצֵא בְּצֵאת הַכּוֹכָבִים. וְיֵשׁ מִצְוָה לְהוֹסִיף מֵחֹל עַל הַקֹּדֶשׁ: לְקַבֵּל אֶת קְדֻשַּׁת הַיּוֹם כַּמָּה דַּקּוֹת לִפְנֵי הַשְּׁקִיעָה, וּלְהִפָּרֵד מִמֶּנּוּ כַּמָּה דַּקּוֹת אַחֲרֵי צֵאת הַכּוֹכָבִים. מְקַבְּלִים אֶת הַיּוֹם בְּדִבּוּר: נָשִׁים בְּהַדְלָקַת הַנֵּרוֹת וּבַבְּרָכָה, וּגְבָרִים בְּ״שֶׁהֶחֱיָנוּ״ אוֹ בְּדִבּוּר מְפֹרָשׁ. עַד שֶׁמְּקַבְּלִים אֶת הַיּוֹם – עוֹד מֻתָּר לֶאֱכֹל.
🤔 לָמָּה? כְּשֶׁמּוֹסִיפִים זְמַן לַיּוֹם הַקָּדוֹשׁ, מַרְאִים כַּמָּה הוּא יָקָר לָנוּ: מְמַהֲרִים לְקַבֵּל אוֹתוֹ, וּמְאַחֲרִים לְהִפָּרֵד מִמֶּנּוּ.
💛 דַּקּוֹת שֶׁל חֹל שֶׁהוֹפְכוֹת לְקֹדֶשׁ – כָּךְ גַּם הַחַיִּים הָרְגִילִים שֶׁלָּנוּ יְכוֹלִים לְהִתְקַדֵּשׁ.
(פֶּרֶק ז׳, הֲלָכָה ח׳)

10הַדְלָקַת נֵרוֹת

מַדְלִיקִים נֵרוֹת לִכְבוֹד יוֹם הַכִּפֻּרִים, כְּמוֹ בְּשַׁבָּת, וּמְבָרְכִים: ״לְהַדְלִיק נֵר שֶׁל יוֹם הַכִּפֻּרִים״. כְּשֶׁיּוֹם הַכִּפֻּרִים חָל בְּשַׁבָּת: ״לְהַדְלִיק נֵר שֶׁל שַׁבָּת וְשֶׁל יוֹם הַכִּפֻּרִים״. הַנָּשִׁים מְבָרְכוֹת אַחֲרֵי הַהַדְלָקָה ״שֶׁהֶחֱיָנוּ״, וּמֵאָז הֵן מְקַבְּלוֹת אֶת הַיּוֹם וַאֲסוּרוֹת בְּכָל אִסּוּרָיו. אִשָּׁה שֶׁצְּרִיכָה לִנְסֹעַ לְבֵית הַכְּנֶסֶת אַחֲרֵי הַהַדְלָקָה – מְכַוֶּנֶת שֶׁלֹּא לְקַבֵּל אֶת הַיּוֹם בְּהַדְלָקָה, וּמְקַבֶּלֶת אוֹתוֹ יַחַד עִם הַצִּבּוּר.
🤔 לָמָּה? גַּם כְּשֶׁאֵין סְעוּדָה, הַנֵּרוֹת מְכַבְּדִים אֶת הַיּוֹם וּמְבִיאִים שְׁלוֹם בַּיִת: בְּנֵי הַבַּיִת רוֹאִים אֶת הַחֲפָצִים וְלֹא נִתְקָלִים בָּהֶם.
(פֶּרֶק ז׳, הֲלָכָה ט׳)

11״כָּל נִדְרֵי״

בִּפְתִיחַת הַיּוֹם מוֹצִיאִים סִפְרֵי תּוֹרָה, וְהַחַזָּן אוֹמֵר: ״עַל דַּעַת הַמָּקוֹם וְעַל דַּעַת הַקָּהָל... אָנוּ מַתִּירִין לְהִתְפַּלֵּל עִם הָעֲבַרְיָנִים״. אַחַר כָּךְ אוֹמְרִים ״כָּל נִדְרֵי״, פְּסוּקֵי סְלִיחָה וּ״שֶׁהֶחֱיָנוּ״. כְּשֶׁמּוֹלִיכִים אֶת סִפְרֵי הַתּוֹרָה, רַבִּים מְנַשְּׁקִים אוֹתָם וּמְבַקְּשִׁים סְלִיחָה עַל מָה שֶׁפָּגְמוּ בִּכְבוֹד הַתּוֹרָה. מִי שֶׁרָגִיל לְהִתְעַטֵּף בְּטַלִּית – מִתְעַטֵּף גַּם בְּלֵיל יוֹם הַכִּפֻּרִים, לִפְנֵי הַשְּׁקִיעָה.
🤔 לָמָּה? בְּיוֹם הַכִּפֻּרִים מִתְגַּלָּה הַנְּשָׁמָה הַקְּדוֹשָׁה שֶׁבְּכָל יְהוּדִי, וְלָכֵן כֻּלָּם מֻזְמָנִים לְהִתְפַּלֵּל יַחַד. חֲכָמִים אָמְרוּ: ״כָּל תַּעֲנִית שֶׁאֵין בָּהּ מִפּוֹשְׁעֵי יִשְׂרָאֵל – אֵינָהּ תַּעֲנִית״, כְּמוֹ הַחֶלְבְּנָה שֶׁרֵיחָהּ רַע, וּבְכָל זֹאת הִיא חֵלֶק מִן הַקְּטֹרֶת.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ נוֹהֲגִים לְסַיֵּם אֶת ״כָּל נִדְרֵי״ לִפְנֵי הַשְּׁקִיעָה, וּלְקַבֵּל אֶת הַיּוֹם בְּ״שֶׁהֶחֱיָנוּ״.
וּבְרֹב הַקְּהִלּוֹת מְסַיְּמִים ״כָּל נִדְרֵי״ אַחֲרֵי הַשְּׁקִיעָה, וְלָכֵן הַגַּבַּאי מַכְרִיז לִפְנֵי הַשְּׁקִיעָה: ״הִנְנוּ מְקַבְּלִים עָלֵינוּ אֶת קְדֻשַּׁת יוֹם הַכִּפֻּרִים״.
💛 כְּשֶׁאֲנַחְנוּ מַבִּיטִים עַל כָּל יְהוּדִי בְּעַיִן טוֹבָה, גַּם ה׳ מַבִּיט עָלֵינוּ בְּאַהֲבָה.
(פֶּרֶק ז׳, הֲלָכָה י׳, הֲלָכָה י״א)

12״בָּרוּךְ שֵׁם״ בְּקוֹל רָם

בְּכָל הַשָּׁנָה אוֹמְרִים אַחֲרֵי ״שְׁמַע יִשְׂרָאֵל״ בְּלַחַשׁ: ״בָּרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ לְעוֹלָם וָעֶד״. בְּיוֹם הַכִּפֻּרִים אוֹמְרִים אוֹתוֹ בְּקוֹל רָם. חֲכָמִים סִפְּרוּ שֶׁמֹּשֶׁה רַבֵּנוּ שָׁמַע בַּמָּרוֹם אֶת מַלְאֲכֵי הַשָּׁרֵת מְשַׁבְּחִים כָּךְ אֶת ה׳, וְלִמֵּד אֶת יִשְׂרָאֵל לוֹמַר זֹאת – אֲבָל בְּלַחַשׁ. רַק בְּיוֹם הַכִּפֻּרִים, כְּשֶׁיִּשְׂרָאֵל נְקִיִּים כְּמַלְאֲכֵי הַשָּׁרֵת, אוֹמְרִים אוֹתוֹ בְּקוֹל.
🤔 לָמָּה? בְּיוֹם הַכִּפֻּרִים אֲנַחְנוּ צָמִים וְשׁוֹבְתִים מִצָּרְכֵי הַגּוּף, וְדוֹמִים לַמַּלְאָכִים. אָז אֶפְשָׁר לוֹמַר בְּגָלוּי אֶת הַשֶּׁבַח שֶׁהֵם אוֹמְרִים.
💛 פַּעַם בְּשָׁנָה כָּל עַם יִשְׂרָאֵל אוֹמֵר בְּקוֹל אֶת שִׁירַת הַמַּלְאָכִים – וְגַם הַקּוֹל שֶׁלְּךָ נִשְׁמָע בְּתוֹךְ הַמַּקְהֵלָה.
(פֶּרֶק ז׳, הֲלָכָה י״ב)

13לוֹמְדִים תּוֹרָה בַּלַּיְלָה

בְּיוֹם הַכִּפֻּרִים כִּמְעַט כָּל הַיּוֹם מֻקְדָּשׁ לִתְפִלָּה, וְלָכֵן הַזְּמַן הַמַּתְאִים בְּיוֹתֵר לִלְמֹד תּוֹרָה הוּא אַחֲרֵי תְּפִלַּת עַרְבִית. בִּקְהִלּוֹת רַבּוֹת הָרַב אוֹמֵר לִפְנֵי עַרְבִית דִּבְרֵי הִתְעוֹרְרוּת עַל הַיּוֹם וְעַל הַתְּשׁוּבָה.
🤔 לָמָּה? אִם בְּכָל יוֹם קָדוֹשׁ יֵשׁ מִצְוָה מְיֻחֶדֶת לִלְמֹד תּוֹרָה – קַל וָחֹמֶר בְּיוֹם הַכִּפֻּרִים.
(פֶּרֶק ז׳, הֲלָכָה י״א)

14קְרִיאַת הַתּוֹרָה וְסֵפֶר יוֹנָה

בְּשַׁחֲרִית קוֹרְאִים בַּתּוֹרָה עַל עֲבוֹדַת הַכֹּהֵן הַגָּדוֹל בְּיוֹם הַכִּפֻּרִים, וְעוֹלִים שִׁשָּׁה עוֹלִים – יוֹתֵר מִבְּחַג, וּפָחוֹת מִבְּשַׁבָּת. כְּשֶׁיּוֹם הַכִּפֻּרִים חָל בְּשַׁבָּת – שִׁבְעָה. בַּהַפְטָרָה הַנָּבִיא יְשַׁעְיָהוּ מְעוֹרֵר לִתְשׁוּבָה, וּבְמִנְחָה מַפְטִירִים בְּסֵפֶר יוֹנָה. נוֹהֲגִים לִתְרֹם בְּיוֹם הַכִּפֻּרִים צְדָקָה לְעִלּוּי נִשְׁמַת הוֹרִים וּבְנֵי מִשְׁפָּחָה שֶׁנִּפְטְרוּ.
🤔 לָמָּה? בְּסֵפֶר יוֹנָה לוֹמְדִים שֶׁאֵין דָּבָר נִסְתָּר מִן ה׳, וְעַל כֹּחָהּ הַגָּדוֹל שֶׁל הַתְּשׁוּבָה – שֶׁמּוֹעִילָה גַּם כְּשֶׁהִיא אֵינָהּ שְׁלֵמָה. וְהַצְּדָקָה מוֹעִילָה גַּם לַנִּפְטָרִים: כְּשֶׁיַּלְדֵיהֶם עוֹשִׂים מַעֲשִׂים טוֹבִים בִּזְכוּתָם, מִתְבָּרֵר שֶׁהַהַשְׁפָּעָה הַטּוֹבָה שֶׁלָּהֶם נִמְשֶׁכֶת. לָכֵן הַיּוֹם נִקְרָא ״כִּפֻּרִים״ – בִּלְשׁוֹן רַבִּים, לַחַיִּים וְלַמֵּתִים.
(פֶּרֶק ז׳, הֲלָכָה י״ג)

15כּוֹרְעִים בְּסֵדֶר הָעֲבוֹדָה

בִּתְפִלַּת מוּסָף מְסַפְּרִים אֶת ״סֵדֶר הָעֲבוֹדָה״ שֶׁל הַכֹּהֵן הַגָּדוֹל, וְכוֹרְעִים עַל הַבִּרְכַּיִם עַד שֶׁהַפָּנִים מַגִּיעוֹת לָאָרֶץ – כְּמוֹ שֶׁהַכֹּהֲנִים וְהָעָם בָּעֲזָרָה כָּרְעוּ וְהִשְׁתַּחֲווּ כְּשֶׁשָּׁמְעוּ אֶת הַשֵּׁם הַמְּפֹרָשׁ. כּוֹרְעִים אַרְבַּע פְּעָמִים: שָׁלֹשׁ בִּשְׁלֹשֶׁת הַוִּדּוּיִים שֶׁל הַכֹּהֵן הַגָּדוֹל, וְאֶת הָרְבִיעִית – לְפִי הַמִּנְהָג. פּוֹרְשִׂים מַשֶּׁהוּ עַל הָרִצְפָּה, כִּי מִחוּץ לַמִּקְדָּשׁ אָסוּר לְהִשְׁתַּחֲווֹת עַל רִצְפַּת אֲבָנִים.
🤔 לָמָּה? עוֹבְדֵי הָאֱלִילִים הָיוּ מִשְׁתַּחֲוִים לָאֲבָנִים, וְלָכֵן הַתּוֹרָה אָסְרָה לְהִשְׁתַּחֲווֹת עַל אֶבֶן מִחוּץ לַמִּקְדָּשׁ. בְּכָל הַשָּׁנָה כִּמְעַט לֹא כּוֹרְעִים כָּךְ, אֲבָל בְּיוֹם הַכִּפֻּרִים, יוֹם שֶׁל תְּשׁוּבָה וּמְסִירוּת נֶפֶשׁ – כּוֹרְעִים לִפְנֵי ה׳.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הַסְּפָרַדִּים כּוֹרְעִים בַּפַּעַם הָרְבִיעִית כְּשֶׁמַּזְכִּירִים אֶת הַשָּׂעִיר שֶׁעָלָה עָלָיו הַגּוֹרָל לַה׳.
לְמִנְהַג הָאַשְׁכְּנַזִּים כּוֹרְעִים בַּפַּעַם הָרְבִיעִית בְּ״וַאֲנַחְנוּ כּוֹרְעִים״ שֶׁבְּ״עָלֵינוּ לְשַׁבֵּחַ״, לִפְנֵי סֵדֶר הָעֲבוֹדָה.
לְמִנְהַג הַתֵּימָנִים לֹא כּוֹרְעִים בִּתְפִלַּת מוּסָף.
💛 חֲכָמִים אָמְרוּ: ״הַכֹּל בִּזְכוּת הִשְׁתַּחֲוָיָה״ – אַבְרָהָם חָזַר מֵהָעֲקֵדָה, יִשְׂרָאֵל נִגְאֲלוּ וְהַתּוֹרָה נִתְּנָה. כְּשֶׁאֲנִי כּוֹרֵעַ, אֲנִי אוֹמֵר לַה׳: הַכֹּל שֶׁלְּךָ.
(פֶּרֶק ז׳, הֲלָכָה י״ד, הֲלָכָה ט״ו)

16בִּרְכַּת כֹּהֲנִים בִּנְעִילָה

בְּכָל הַשָּׁנָה הַכֹּהֲנִים לֹא נוֹשְׂאִים כַּפַּיִם בְּמִנְחָה, כִּי הִיא אַחֲרֵי הָאֲרוּחָה, וְחָשְׁשׁוּ שֶׁמָּא שָׁתוּ יַיִן. אֲבָל בְּיוֹם הַכִּפֻּרִים, כְּשֶׁכֻּלָּם צָמִים, הַכֹּהֲנִים מְבָרְכִים אֶת הָעָם בִּתְפִלַּת נְעִילָה. מִשְׁתַּדְּלִים שֶׁבִּרְכַּת כֹּהֲנִים תִּהְיֶה לִפְנֵי הַשְּׁקִיעָה, וְלָכֵן מְקַצְּרִים בַּפִּיּוּטִים.
🤔 לָמָּה? בִּרְכַּת כֹּהֲנִים הִיא מִצְוָה מִן הַתּוֹרָה, וּזְמַנָּהּ בַּיּוֹם – כְּמוֹ עֲבוֹדַת הַקָּרְבָּנוֹת.
(פֶּרֶק ז׳, הֲלָכָה ט״ז)

17תְּפִלַּת נְעִילָה

לִקְרַאת סוֹף הַיּוֹם מִתְפַּלְּלִים תְּפִלָּה נוֹסֶפֶת – ״נְעִילָה״, כְּנֶגֶד נְעִילַת שַׁעֲרֵי הַהֵיכָל בַּמִּקְדָּשׁ וּנְעִילַת שַׁעֲרֵי שָׁמַיִם. מַתְחִילִים כְּשֶׁהַשֶּׁמֶשׁ בְּרֹאשׁ הָאִילָנוֹת, כְּאַרְבָּעִים דַּקּוֹת לִפְנֵי הַשְּׁקִיעָה. בְּכָל מָקוֹם שֶׁבִּקַּשְׁנוּ ״כָּתְבֵנוּ״, בִּנְעִילָה מְבַקְּשִׁים ״חָתְמֵנוּ״, וּמִי שֶׁטָּעָה וְאָמַר ״כָּתְבֵנוּ״ – יָצָא. נוֹהֲגִים לִפְתֹּחַ אֶת אֲרוֹן הַקֹּדֶשׁ בְּכָל חֲזָרַת הַשַּׁ״ץ.
🤔 לָמָּה? ״הַכֹּל הוֹלֵךְ אַחַר הַחִתּוּם״: תַּכְלִית עֲשֶׂרֶת יְמֵי תְּשׁוּבָה הִיא יוֹם הַכִּפֻּרִים, וְתַכְלִית יוֹם הַכִּפֻּרִים הִיא תְּפִלַּת נְעִילָה. לָכֵן גַּם מִי שֶׁנֶּחְלַשׁ מִן הַצּוֹם – מִתְאַמֵּץ מְאֹד לְהִתְפַּלֵּל בְּכַוָּנָה.
💛 ״אִם לֹא עַכְשָׁו – אֵימָתַי?״ בָּרְגָעִים הָאֵלֶּה כָּל מִלָּה מִן הַלֵּב עוֹלָה יָשָׁר לִפְנֵי ה׳.
(פֶּרֶק ז׳, הֲלָכָה י״ז)

18״ה׳ הוּא הָאֱלֹהִים״ וּתְקִיעַת שׁוֹפָר

בְּסוֹף נְעִילָה, לִפְנֵי שֶׁסּוֹגְרִים אֶת אֲרוֹן הַקֹּדֶשׁ, כָּל הַקָּהָל מְקַבֵּל יַחַד עֹל מַלְכוּת שָׁמַיִם: אוֹמְרִים ״שְׁמַע יִשְׂרָאֵל״ וּ״בָרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ לְעוֹלָם וָעֶד״, וְשֶׁבַע פְּעָמִים ״ה׳ הוּא הָאֱלֹהִים״. אַחַר כָּךְ, עִם צֵאת הַכּוֹכָבִים, תּוֹקְעִים בַּשּׁוֹפָר. בִּקְהִלּוֹת רַבּוֹת רוֹקְדִים וְשָׁרִים: ״לְשָׁנָה הַבָּאָה בִּירוּשָׁלַיִם הַבְּנוּיָה״.
🤔 לָמָּה? ״ה׳ הוּא הָאֱלֹהִים״ שֶׁבַע פְּעָמִים – כְּנֶגֶד הַשְּׁכִינָה שֶׁעוֹלָה לִמְקוֹמָהּ מֵעַל שִׁבְעָה רְקִיעִים, אַחֲרֵי שֶׁהִתְקָרְבָה אֵלֵינוּ בַּיָּמִים הַנּוֹרָאִים. וְהַשּׁוֹפָר הוּא סִימָן לְסוֹף הַיּוֹם – ״עָלָה אֱלֹהִים בִּתְרוּעָה״ – וְזֵכֶר לַשּׁוֹפָר שֶׁל הַיּוֹבֵל.
💛 אַחֲרֵי הַתְּקִיעָה כָּל יִשְׂרָאֵל יוֹדְעִים שֶׁה׳ אוֹהֵב אוֹתָם וְקִבֵּל אֶת תְּשׁוּבָתָם, וְהַלֵּב מִתְמַלֵּא שִׂמְחָה.
(פֶּרֶק ז׳, הֲלָכָה י״ח)

19סוֹף הַצּוֹם וּבִרְכַּת הַלְּבָנָה

בְּאֶרֶץ יִשְׂרָאֵל, כַּחֲצִי שָׁעָה אַחֲרֵי הַשְּׁקִיעָה, כְּבָר עָבַר זְמַן צֵאת הַכּוֹכָבִים וְעוֹד כַּמָּה דַּקּוֹת שֶׁל תּוֹסֶפֶת, וְאֶפְשָׁר לְהַבְדִּיל וְלֶאֱכֹל. רַבִּים מְבָרְכִים אֶת בִּרְכַּת הַלְּבָנָה בְּמוֹצָאֵי יוֹם הַכִּפֻּרִים, מִתּוֹךְ שִׂמְחָה. יֵשׁ שֶׁהוֹלְכִים קֹדֶם לֶאֱכֹל, וּמִתְאַסְּפִים אַחַר כָּךְ לְבָרֵךְ.
🤔 לָמָּה? אֶת בִּרְכַּת הַלְּבָנָה אוֹמְרִים בְּשִׂמְחָה, וְאַחֲרֵי יְמֵי הַדִּין הַשִּׂמְחָה מַגִּיעָה לְשִׂיאָהּ.
(פֶּרֶק ז׳, הֲלָכָה י״ט)

20הַבְדָּלָה – וְאוֹכְלִים בְּשִׂמְחָה

גַּם אַחֲרֵי צֵאת הַכּוֹכָבִים לֹא אוֹכְלִים עַד שֶׁשּׁוֹמְעִים הַבְדָּלָה עַל כּוֹס יַיִן. מְבָרְכִים עַל הַגֶּפֶן, עַל הָאוֹר וְעַל הַהַבְדָּלָה – בְּלִי בְּשָׂמִים. אֶת בִּרְכַּת הָאֵשׁ מְבָרְכִים עַל נֵר שֶׁדָּלַק כָּל יוֹם הַכִּפֻּרִים, ״נֵר שֶׁשָּׁבַת״, וְלָכֵן מַדְלִיקִים לִפְנֵי הַצּוֹם נֵר נְשָׁמָה. אַחֲרֵי הַהַבְדָּלָה אוֹכְלִים בְּשִׂמְחָה, וַחֲסִידִים מְמַהֲרִים לְהַתְחִיל לִבְנוֹת אֶת הַסֻּכָּה – לָצֵאת מִמִּצְוָה לְמִצְוָה.
🤔 לָמָּה? בְּמוֹצָאֵי שַׁבָּת מוֹדִים עַל הָאֵשׁ שֶׁהִתְגַּלְּתָה לְאָדָם הָרִאשׁוֹן, וּמְבָרְכִים עַל אֵשׁ חֲדָשָׁה. בְּמוֹצָאֵי יוֹם הַכִּפֻּרִים מוֹדִים עַל כָּךְ שֶׁשּׁוּב מֻתָּר לְהִשְׁתַּמֵּשׁ בָּאֵשׁ שֶׁ״שָּׁבְתָה״ כָּל הַיּוֹם. וּבְשָׂמִים לֹא צְרִיכִים, כִּי אַחֲרֵי הַצּוֹם אֵין צַעַר, אֶלָּא שִׂמְחָה עַל הַכַּפָּרָה.
💛 בְּמוֹצָאֵי יוֹם הַכִּפֻּרִים יוֹצֵאת בַּת קוֹל וְאוֹמֶרֶת: ״לֵךְ אֱכֹל בְּשִׂמְחָה לַחְמֶךָ... כִּי כְבָר רָצָה הָאֱלֹהִים אֶת מַעֲשֶׂיךָ״.
(פֶּרֶק ז׳, הֲלָכָה כ׳)

📝 מָה לָמַדְנוּ

  • מְכַבְּדִים אֶת יוֹם הַכִּפֻּרִים בִּבְגָדִים לְבָנִים וּנְקִיִּים, וְאָסוּר בּוֹ כָּל מָה שֶׁאָסוּר בְּשַׁבָּת.
  • מִתְוַדִּים עֶשֶׂר פְּעָמִים, בַּעֲמִידָה וּבִלְשׁוֹן רַבִּים, כִּי כָּל יִשְׂרָאֵל עֲרֵבִים זֶה לָזֶה.
  • מוֹסִיפִים מֵחֹל עַל הַקֹּדֶשׁ, מַדְלִיקִים נֵרוֹת וּפוֹתְחִים בְּ״כָל נִדְרֵי״ – יַחַד עִם כָּל יְהוּדִי.
  • בְּיוֹם הַכִּפֻּרִים אוֹמְרִים ״בָּרוּךְ שֵׁם״ בְּקוֹל, וְכוֹרְעִים בְּסֵדֶר הָעֲבוֹדָה.
  • בִּנְעִילָה מְבַקְּשִׁים ״חָתְמֵנוּ״, תּוֹקְעִים בַּשּׁוֹפָר, וּמַבְדִּילִים עַל נֵר שֶׁדָּלַק כָּל הַיּוֹם.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה הַוִּדּוּי בְּיוֹם הַכִּפֻּרִים הוּא בִּלְשׁוֹן רַבִּים (״חָטָאנוּ״), גַּם כְּשֶׁאֲנִי לֹא עָשִׂיתִי אֶת הַחֵטְא?
  2. לָמָּה בְּיוֹם הַכִּפֻּרִים אוֹמְרִים ״בָּרוּךְ שֵׁם כְּבוֹד מַלְכוּתוֹ״ בְּקוֹל רָם?
  3. עַל אֵיזֶה נֵר מְבָרְכִים ״בּוֹרֵא מְאוֹרֵי הָאֵשׁ״ בְּמוֹצָאֵי יוֹם הַכִּפֻּרִים, וְלָמָּה לֹא מְבָרְכִים עַל בְּשָׂמִים?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״יוֹם הַכִּפּוּרִים״

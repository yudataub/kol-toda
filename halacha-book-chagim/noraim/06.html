סֵפֶר הַהֲלָכוֹת ‹ רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים
מַהֲדוּרַת יְלָדִים

🤍 יוֹם הַכִּפֻּרִים

פֶּרֶק ו׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
שָׂמְתָּ לֵב שֶׁבְּיוֹם הַכִּפֻּרִים הַכְּבִישִׁים רֵיקִים, וְכֻלָּם לְבוּשִׁים לָבָן? יֵשׁ בָּאֲוִיר שֶׁקֶט מְיֻחָד, כְּאִלּוּ כָּל הָעוֹלָם עוֹצֵר. מָה יֵשׁ בַּיּוֹם הַזֶּה שֶׁאֵין בְּשׁוּם יוֹם אַחֵר? בַּפֶּרֶק הַזֶּה נִלְמַד לָמָּה ה׳ נָתַן לָנוּ אֶת יוֹם הַכִּפֻּרִים – וְלָמָּה הוּא שָׂמֵחַ בּוֹ.

1יוֹם שֶׁל טָהֳרָה

מִתּוֹךְ אַהֲבָתוֹ הַגְּדוֹלָה לְעַמּוֹ, קָבַע ה׳ לָנוּ יוֹם מְיֻחָד בַּשָּׁנָה לְכַפֵּר עַל עֲווֹנוֹתֵינוּ: ״כִּי בַיּוֹם הַזֶּה יְכַפֵּר עֲלֵיכֶם לְטַהֵר אֶתְכֶם מִכֹּל חַטֹּאתֵיכֶם, לִפְנֵי ה׳ תִּטְהָרוּ״. בְּלִי יוֹם הַכִּפֻּרִים, אָדָם שֶׁמִּתְחָרֵט הָיָה יָכוֹל לְתַקֵּן אֶת דְּרָכָיו מִכָּאן וָהָלְאָה, אֲבָל לֹא לִמְחֹק אֶת מָה שֶׁכְּבָר עָשָׂה. בְּיוֹם הַכִּפֻּרִים שַׁעֲרֵי שָׁמַיִם נִפְתָּחִים, וְאוֹר ה׳ מְטַהֵר אֶת נִשְׁמוֹת יִשְׂרָאֵל.
🤔 לָמָּה? רַבִּי עֲקִיבָא אָמַר: ״אַשְׁרֵיכֶם יִשְׂרָאֵל, לִפְנֵי מִי אַתֶּם מִטַּהֲרִין, וּמִי מְטַהֵר אֶתְכֶם? אֲבִיכֶם שֶׁבַּשָּׁמַיִם... מָה מִקְוֶה מְטַהֵר אֶת הַטְּמֵאִים – אַף הַקָּדוֹשׁ בָּרוּךְ הוּא מְטַהֵר אֶת יִשְׂרָאֵל״.
💛 כְּמוֹ שֶׁיּוֹצְאִים מִן הַמִּקְוֶה נְקִיִּים, כָּךְ יוֹצְאִים מִיּוֹם הַכִּפֻּרִים. וּמִי שֶׁמְּטַהֵר אוֹתָנוּ הוּא אָבִינוּ.
(פֶּרֶק ו׳, הֲלָכָה א׳)

2מִצְווֹת הַיּוֹם

בְּיוֹם הַכִּפֻּרִים יֵשׁ שָׁלֹשׁ מִצְווֹת שֶׁיֵּשׁ גַּם בְּיָמִים קְדוֹשִׁים אֲחֵרִים: לַעֲשׂוֹת אוֹתוֹ ״מִקְרָא קֹדֶשׁ״ וּלְכַבֵּד אוֹתוֹ, לִשְׁבֹּת בּוֹ מִכָּל מְלָאכָה כְּמוֹ בְּשַׁבָּת, וּלְהַקְרִיב קָרְבְּנוֹת מוּסָף. וְעוֹד שָׁלֹשׁ מִצְווֹת מְיֻחָדוֹת לוֹ: לְהִתְעַנּוֹת; לָשׁוּב בִּתְשׁוּבָה וּלְהִתְוַדּוֹת; וְסֵדֶר הַקָּרְבָּנוֹת הַמְּיֻחָד, שֶׁבּוֹ הַכֹּהֵן הַגָּדוֹל נִכְנַס לְקֹדֶשׁ הַקֳּדָשִׁים. הַיּוֹם, כְּשֶׁבֵּית הַמִּקְדָּשׁ חָרֵב, הַיּוֹם עַצְמוֹ, הַתַּעֲנִית וְהַתְּשׁוּבָה מְכַפְּרִים, וְהַתְּפִלּוֹת – וּבִמְיֻחָד תְּפִלַּת מוּסָף – הֵן כְּמוֹ תַּחֲלִיף לַקָּרְבָּנוֹת.
🤔 לָמָּה? הַתַּעֲנִית חֲשׁוּבָה כָּל כָּךְ, שֶׁגַּם כְּשֶׁיּוֹם הַכִּפֻּרִים חָל בְּשַׁבָּת צָמִים – אַף שֶׁבְּכָל שַׁבָּת מִצְוָה לְעַנֵּג אוֹתָהּ בַּאֲכִילָה.
(פֶּרֶק ו׳, הֲלָכָה ב׳)

3ה׳ שָׂמֵחַ לִמְחֹל

חֲכָמִים מָשְׁלוּ מָשָׁל: מֶלֶךְ שֶׁעֲבָדָיו וּבְנֵי בֵּיתוֹ הָיוּ מוֹצִיאִים אֶת הָאַשְׁפָּה מִן הָעִיר, וּמַשְׁלִיכִים אוֹתָהּ מוּל הַשַּׁעַר שֶׁל הַמֶּלֶךְ. כְּשֶׁהַמֶּלֶךְ יוֹצֵא וְרוֹאֶה אֶת הָאַשְׁפָּה – הוּא שָׂמֵחַ שִׂמְחָה גְּדוֹלָה, כִּי הָעִיר נְקִיָּה. כָּךְ בְּיוֹם הַכִּפֻּרִים: כְּשֶׁה׳ מוֹחֵל לְיִשְׂרָאֵל, הוּא אֵינוֹ עָצוּב אֶלָּא שָׂמֵחַ, וְקוֹרֵא לֶהָרִים וְלַגְּבָעוֹת: ״בּוֹאוּ וְשִׂמְחוּ עִמִּי שִׂמְחָה גְּדוֹלָה, שֶׁאֲנִי מוֹחֵל לַעֲווֹנוֹתֵיהֶם שֶׁל יִשְׂרָאֵל״.
🤔 לָמָּה? לַמִּלָּה ״כִּפּוּר״ יֵשׁ כַּמָּה מַשְׁמָעֻיּוֹת: כִּסּוּי (כְּמוֹ הַכַּפֹּרֶת שֶׁכִּסְּתָה אֶת הָאָרוֹן), נִקּוּי, בִּטּוּל וּפִיּוּס. בְּיוֹם הַכִּפֻּרִים ה׳ מְכַסֶּה, מְנַקֶּה וּמְבַטֵּל אֶת הַחֲטָאִים.
💛 ה׳ לֹא סוֹלֵחַ לָנוּ ״בְּלֵית בְּרֵירָה״. הוּא שָׂמֵחַ לִסְלֹחַ, כְּמוֹ אַבָּא שֶׁשָּׂמֵחַ שֶׁבְּנוֹ חָזַר.
(פֶּרֶק ו׳, הֲלָכָה ג׳)

4הַיּוֹם שֶׁהַשָּׂטָן שׁוֹתֵק

חֲכָמִים אָמְרוּ שֶׁהַמִּלָּה ״הַשָּׂטָן״ עוֹלָה בְּגִימַטְרִיָּה 364, וּבַשָּׁנָה יֵשׁ 365 יָמִים. כְּלוֹמַר: 364 יָמִים בַּשָּׁנָה יֵשׁ לַשָּׂטָן רְשׁוּת לְקַטְרֵג עַל יִשְׂרָאֵל, וּבְיוֹם אֶחָד – יוֹם הַכִּפֻּרִים – אֵין לוֹ רְשׁוּת.
🤔 לָמָּה? בְּיוֹם הַכִּפֻּרִים מִתְגַּלֶּה הַשֹּׁרֶשׁ הַקָּדוֹשׁ שֶׁל יִשְׂרָאֵל, שֶׁקָּשׁוּר לַה׳. וְאִם לֹא הָיָה יוֹם שֶׁל כַּפָּרָה, הָעֲווֹנוֹת הָיוּ מִצְטַבְּרִים שָׁנָה אַחַר שָׁנָה.
(פֶּרֶק ו׳, הֲלָכָה ג׳)

5עַד הָרֶגַע הָאַחֲרוֹן

כָּל יוֹם הַכִּפֻּרִים מְכַפֵּר, מִתְּחִלָּתוֹ וְעַד סוֹפוֹ. אֲבָל הַכַּפָּרָה נִשְׁלֶמֶת בְּצֵאת הַיּוֹם, כְּשֶׁהַצּוֹם מַגִּיעַ לְשִׂיאוֹ וְכָל הַתְּפִלּוֹת שֶׁל הַיּוֹם מִצְטָרְפוֹת יַחַד. לָכֵן מִתְאַמְּצִים בִּמְיֻחָד בִּתְפִלַּת נְעִילָה.
💛 גַּם מִי שֶׁהִתְעַיֵּף בְּמֶשֶׁךְ הַיּוֹם – הָרְגָעִים הָאַחֲרוֹנִים שֶׁלּוֹ יְקָרִים מְאֹד לִפְנֵי ה׳.
(פֶּרֶק ו׳, הֲלָכָה ג׳)

6לָמָּה צָמִים?

לִכְאוֹרָה הָיָה עָדִיף לֶאֱכֹל מְעַט בְּיוֹם הַכִּפֻּרִים, כְּדֵי לְהִתְרַכֵּז בַּתְּפִלָּה. אֲבָל בְּכָל הַשָּׁנָה הַנְּשָׁמָה מְכֻסָּה בְּתַאֲווֹת הַגּוּף, שֶׁגּוֹרְמוֹת לָאָדָם לִשְׁכֹּחַ מָה הוּא רוֹצֶה בֶּאֱמֶת. כְּשֶׁצָּמִים, הַנְּשָׁמָה מִשְׁתַּחְרֶרֶת מְעַט מִכַּבְלֵי הַגּוּף, וּמִתְבָּרֵר לָנוּ שֶׁהָרָצוֹן הָאֲמִתִּי שֶׁלָּנוּ הוּא לְהִדָּבֵק בַּה׳. מִי שֶׁצָּרִיךְ לִשְׁכַּב בְּמִטָּתוֹ כְּדֵי לְהַמְשִׁיךְ לָצוּם – אַל יִפֹּל לִבּוֹ: גַּם בַּמִּטָּה הוּא קוֹלֵט אֶת עִקַּר הַיּוֹם, וְיָכוֹל לַחְשֹׁב מַחֲשָׁבוֹת טוֹבוֹת וּלְהַחְלִיט לְהוֹסִיף בְּתוֹרָה וּבְמִצְווֹת.
🤔 לָמָּה? הַתַּעֲנִית הִיא כְּמוֹ קָרְבָּן. פַּעַם הִקְרִיבוּ בְּהֵמָה עַל הַמִּזְבֵּחַ, וּבְיוֹם הַכִּפֻּרִים כָּל אֶחָד מִיִּשְׂרָאֵל כְּאִלּוּ מַקְרִיב אֶת עַצְמוֹ לִפְנֵי ה׳.
💛 הַגּוּף רָעֵב, אֲבָל הַנְּשָׁמָה שְׂבֵעָה: הִיא קְרוֹבָה לַה׳ יוֹתֵר מִבְּכָל הַשָּׁנָה.
(פֶּרֶק ו׳, הֲלָכָה ו׳)

7מָה יוֹם הַכִּפֻּרִים לֹא מְכַפֵּר?

״עֲבֵרוֹת שֶׁבֵּין אָדָם לַחֲבֵרוֹ – אֵין יוֹם הַכִּפֻּרִים מְכַפֵּר, עַד שֶׁיְּרַצֶּה אֶת חֲבֵרוֹ״. כָּל עוֹד הֶחָבֵר לֹא הִתְפַּיֵּס, הַפְּגָם נִשְׁאָר. וְעוֹד אָמְרוּ חֲכָמִים: ״הָאוֹמֵר: אֶחֱטָא וְיוֹם הַכִּפֻּרִים מְכַפֵּר – אֵין יוֹם הַכִּפֻּרִים מְכַפֵּר״.
🤔 לָמָּה? זִכְרוֹן יוֹם הַכִּפֻּרִים צָרִיךְ לִשְׁמֹר עָלֵינוּ מִלַּחֲטֹא. מִי שֶׁחוֹטֵא דַּוְקָא כִּי ״יֵשׁ יוֹם כִּפּוּרִים״ – לֹא הֵבִין מַהוּ הַיּוֹם הַקָּדוֹשׁ. וְרַק אִם יִתְאַמֵּץ מְאֹד לָשׁוּב, תְּשׁוּבָתוֹ תִּתְקַבֵּל.
(פֶּרֶק ו׳, הֲלָכָה ז׳)

8יוֹם שֶׁל חֵרוּת

בִּזְמַן שֶׁיִּשְׂרָאֵל יָשְׁבוּ בְּאַרְצָם, בְּכָל חֲמִשִּׁים שָׁנָה הָיְתָה שְׁנַת ״יוֹבֵל״: הָעֲבָדִים יָצְאוּ לְחֵרוּת, וְהַשָּׂדוֹת חָזְרוּ לִבְעָלֵיהֶם. וּמָתַי? בְּיוֹם הַכִּפֻּרִים, כְּשֶׁבֵּית הַדִּין תָּקַע בַּשּׁוֹפָר. זֵכֶר לְכָךְ תּוֹקְעִים בַּשּׁוֹפָר בְּסוֹף יוֹם הַכִּפֻּרִים.
🤔 לָמָּה? בְּכָל יוֹם כִּפּוּרִים אֲנַחְנוּ יוֹצְאִים לְחֵרוּת מֵהַשִּׁעְבּוּד לַיֵּצֶר הָרַע, כְּמוֹ הָעֶבֶד שֶׁיּוֹצֵא לְחָפְשִׁי. וְהַגּוּף חוֹזֵר אֶל הַנְּשָׁמָה, כְּמוֹ הַשָּׂדֶה שֶׁחוֹזֵר לִבְעָלָיו.
💛 ״אֵין לְךָ בֶּן חוֹרִין אֶלָּא מִי שֶׁעוֹסֵק בַּתּוֹרָה״. הַתְּשׁוּבָה וְהַתּוֹרָה מְשַׁחְרְרוֹת אוֹתִי לִהְיוֹת מִי שֶׁאֲנִי בֶּאֱמֶת רוֹצֶה לִהְיוֹת.
(פֶּרֶק ו׳, הֲלָכָה י״א)

📝 מָה לָמַדְנוּ

  • יוֹם הַכִּפֻּרִים הוּא מַתָּנָה שֶׁל אַהֲבָה: ה׳ מְטַהֵר אוֹתָנוּ כְּמוֹ מִקְוֶה.
  • יֵשׁ בּוֹ שֵׁשׁ מִצְווֹת, וּמֵהֶן הַמְּיֻחָדוֹת: תַּעֲנִית, תְּשׁוּבָה וּוִדּוּי, וַעֲבוֹדַת הַכֹּהֵן הַגָּדוֹל.
  • ה׳ שָׂמֵחַ לִמְחֹל, וּבְיוֹם הַכִּפֻּרִים אֵין לַשָּׂטָן רְשׁוּת לְקַטְרֵג.
  • צָמִים כְּדֵי שֶׁהַנְּשָׁמָה תִּשְׁתַּחְרֵר מֵהַגּוּף וְנִזְכֹּר מָה אֲנַחְנוּ רוֹצִים בֶּאֱמֶת.
  • יוֹם הַכִּפֻּרִים לֹא מְכַפֵּר עַל עֲבֵרוֹת שֶׁבֵּין אָדָם לַחֲבֵרוֹ, עַד שֶׁמְּפַיְּסִים אֶת הֶחָבֵר.

❓ שְׁאֵלוֹת חֲזָרָה

  1. סַפֵּר אֶת מְשַׁל הַמֶּלֶךְ וְהָאַשְׁפָּה. מָה הוּא מְלַמֵּד עַל הַדֶּרֶךְ שֶׁבָּהּ ה׳ מוֹחֵל?
  2. מָה הַקֶּשֶׁר בֵּין הַמִּלָּה ״הַשָּׂטָן״ לְיוֹם הַכִּפֻּרִים?
  3. לָמָּה תּוֹקְעִים בַּשּׁוֹפָר בְּסוֹף יוֹם הַכִּפֻּרִים, וּמָה הַקֶּשֶׁר לִשְׁנַת הַיּוֹבֵל?

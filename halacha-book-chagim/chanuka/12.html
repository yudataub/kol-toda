סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת חֲנֻכָּה
מַהֲדוּרַת יְלָדִים

🕯️ הַדְלָקַת נֵרוֹת חֲנֻכָּה

פֶּרֶק י״ב · לַמַּהֲדוּרָה לַמּוֹרֶה
אַבָּא מוֹצִיא אֶת הַחֲנֻכִּיָּה, אַתָּה מֵבִיא אֶת קֻפְסַת הַגַּפְרוּרִים, וְכָל הַמִּשְׁפָּחָה מִתְאַסֶּפֶת לְיַד הַחַלּוֹן. אֲבָל רֶגַע – מִי מַדְלִיק? כַּמָּה נֵרוֹת? מֵאֵיזֶה צַד מַתְחִילִים? וּמָה עוֹשִׂים אִם נֵר כָּבָה בָּאֶמְצַע? בַּפֶּרֶק הַזֶּה נִלְמַד אֶת כָּל מָה שֶׁצָּרִיךְ לָדַעַת כְּדֵי לְהַדְלִיק נֵרוֹת חֲנֻכָּה – וּלְהַדְלִיק אוֹתָם יָפֶה.

1מִצְוָה שֶׁל חֲכָמִים – ״וְצִוָּנוּ״

חֲכָמִים תִּקְּנוּ לְהַדְלִיק נֵרוֹת בִּשְׁמוֹנַת יְמֵי הַחֲנֻכָּה, כְּדֵי לִזְכֹּר וּלְפַרְסֵם אֶת הַנֵּס. וְאַף שֶׁזּוֹ מִצְוָה מִדִּבְרֵי חֲכָמִים, מְבָרְכִים עָלֶיהָ: ״אֲשֶׁר קִדְּשָׁנוּ בְּמִצְוֹתָיו וְצִוָּנוּ לְהַדְלִיק נֵר שֶׁל חֲנֻכָּה״.
🤔 לָמָּה? אֵיךְ אוֹמְרִים ״וְצִוָּנוּ״, אִם הַמִּצְוָה לֹא כְּתוּבָה בַּתּוֹרָה? כִּי הַתּוֹרָה עַצְמָהּ נָתְנָה לַחֲכָמִים סַמְכוּת לְתַקֵּן מִצְווֹת, כְּמוֹ שֶׁנֶּאֱמַר: ״לֹא תָסוּר מִן הַדָּבָר אֲשֶׁר יַגִּידוּ לְךָ״.
💛 כְּשֶׁאֲנִי שׁוֹמֵעַ בְּקוֹל הַחֲכָמִים – אֲנִי שׁוֹמֵעַ בְּקוֹל ה׳.
(פֶּרֶק י״ב, הֲלָכָה א׳)

2גַּם נָשִׁים וּבָנוֹת חַיָּבוֹת

נָשִׁים חַיָּבוֹת בְּמִצְוַת נֵר חֲנֻכָּה, כְּמוֹ גְּבָרִים. בְּדֶרֶךְ כְּלָל נָשִׁים פְּטוּרוֹת מִמִּצְווֹת שֶׁתְּלוּיוֹת בִּזְמַן – אֲבָל בַּמִּצְוָה הַזֹּאת הֵן חַיָּבוֹת. אִשָּׁה נְשׂוּאָה נוֹהֶגֶת לָצֵאת יְדֵי חוֹבָה בְּהַדְלָקַת בַּעְלָהּ.
🤔 לָמָּה? גַּם הַנָּשִׁים הָיוּ שֻׁתָּפוֹת בַּנֵּס.
(פֶּרֶק י״ב, הֲלָכָה א׳)

3פִּרְסוּם הַנֵּס – וְגַם הֶעָנִי מַדְלִיק

כָּל הַדִּינִים שֶׁל מְקוֹם הַהַדְלָקָה וּזְמַנָּהּ נוֹעֲדוּ לְפַרְסֵם אֶת הַנֵּס: מַדְלִיקִים לְיַד פֶּתַח אוֹ חַלּוֹן שֶׁפּוֹנֶה לָרְחוֹב, בִּשְׁעַת עֶרֶב שֶׁבָּהּ עוֹבְרִים שָׁם אֲנָשִׁים. אֲבָל גַּם יְהוּדִי שֶׁגָּר לְבַד בְּמָקוֹם שֶׁאַף אֶחָד לֹא עוֹבֵר בּוֹ – מַדְלִיק, כְּדֵי לִזְכֹּר בְּעַצְמוֹ אֶת הַנֵּס. וַאֲפִלּוּ עָנִי שֶׁאֵין לוֹ כֶּסֶף לְנֵרוֹת – צָרִיךְ לְבַקֵּשׁ מֵאֲחֵרִים, אוֹ לִמְכֹּר אֶת בִּגְדּוֹ, כְּדֵי לִקְנוֹת נֵרוֹת. מַסְפִּיק שֶׁיַּדְלִיק נֵר אֶחָד בְּכָל לַיְלָה.
🤔 לָמָּה? בְּמִצְווֹת אֲחֵרוֹת לֹא צָרִיךְ לִמְכֹּר בְּגָדִים. אֲבָל בְּנֵר חֲנֻכָּה יֵשׁ פִּרְסוּם הַנֵּס – וְלָכֵן הַחִיּוּב גָּדוֹל יוֹתֵר.
(פֶּרֶק י״ב, הֲלָכָה א׳)

4שָׁלֹשׁ מַדְרֵגוֹת

בְּנֵר חֲנֻכָּה יֵשׁ שָׁלֹשׁ מַדְרֵגוֹת. הַחוֹבָה: נֵר אֶחָד בְּכָל לַיְלָה לְכָל הַבַּיִת. ״מְהַדְּרִין״: נֵר לְכָל אֶחָד מִבְּנֵי הַבַּיִת. ״מְהַדְּרִין מִן הַמְּהַדְּרִין״: לְפִי מִסְפַּר הַיָּמִים. בֵּית שַׁמַּאי אָמְרוּ: מַתְחִילִים בִּשְׁמוֹנָה נֵרוֹת וּפוֹחֲתִים. בֵּית הִלֵּל אָמְרוּ: מַתְחִילִים בְּנֵר אֶחָד וּמוֹסִיפִים בְּכָל לַיְלָה. כָּל יִשְׂרָאֵל נוֹהֲגִים כְּ״מְהַדְּרִין מִן הַמְּהַדְּרִין״, כְּבֵית הִלֵּל – וּבְסַךְ הַכֹּל מַדְלִיקִים בַּחֲנֻכָּה ל״ו נֵרוֹת.
🤔 לָמָּה? בְּכָל יוֹם שֶׁהַמְּנוֹרָה דָּלְקָה מִפַּךְ הַשֶּׁמֶן הַקָּטָן – הַנֵּס גָּדַל. כְּשֶׁמּוֹסִיפִים נֵר בְּכָל לַיְלָה, רוֹאִים אֵיךְ הַנֵּס הוֹלֵךְ וְגָדֵל, וְעוֹלִים בַּקֹּדֶשׁ.
💛 בְּרֹב הַמִּצְווֹת יֵשׁ חוֹבָה וְהִדּוּר. בַּחֲנֻכָּה יֵשׁ גַּם ״הִדּוּר שֶׁל הִדּוּר״ – וְכָל עַם יִשְׂרָאֵל בּוֹחֵר בּוֹ, מֵאַהֲבָה.
(פֶּרֶק י״ב, הֲלָכָה ב׳)

5מִי מַדְלִיק? שְׁנֵי מִנְהָגִים

כֻּלָּם מַדְלִיקִים לְפִי מִסְפַּר הַיָּמִים. אֲבָל מִי מַדְלִיק בַּבַּיִת?
🤔 לָמָּה? שְׁנֵי הַמִּנְהָגִים רוֹצִים אֶת אוֹתוֹ דָּבָר: שֶׁיִּרְאוּ אֶת מִסְפַּר הַנֵּרוֹת וְיִתְפַּרְסֵם הַנֵּס. הַסְּפָרַדִּים שׁוֹמְרִים עַל כָּךְ בְּכָךְ שֶׁרַק אֶחָד מַדְלִיק, וְהָאַשְׁכְּנַזִּים – בְּכָךְ שֶׁמַּרְחִיקִים אֶת הַחֲנֻכִּיּוֹת.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הַסְּפָרַדִּים רַק אֶחָד מַדְלִיק בִּשְׁבִיל כָּל בְּנֵי הַבַּיִת – רָאוּי שֶׁיִּהְיֶה זֶה גְּדוֹל הַבַּיִת. כָּךְ רוֹאִים בְּדִיּוּק כַּמָּה יָמִים נִמְשַׁךְ הַנֵּס, וְהַנֵּרוֹת שֶׁל כֻּלָּם לֹא מִתְעַרְבְּבִים.
לְמִנְהַג הָאַשְׁכְּנַזִּים כָּל אֶחָד מִבְּנֵי הַבַּיִת מַדְלִיק בִּבְרָכָה חֲנֻכִּיָּה מִשֶּׁלּוֹ, וּמַרְחִיקִים אֶת הַחֲנֻכִּיּוֹת זוֹ מִזּוֹ, כְּדֵי שֶׁיִּרְאוּ אֶת הַמִּסְפָּר שֶׁל כָּל אַחַת. גַּם יְלָדִים שֶׁהִגִּיעוּ לְגִיל חִנּוּךְ – בְּעֵרֶךְ מִגִּיל שֵׁשׁ, כְּשֶׁהֵם מְבִינִים אֶת הַנֵּס – מַדְלִיקִים בִּבְרָכָה.
(פֶּרֶק י״ב, הֲלָכָה ג׳, הֲלָכָה ד׳)

6יֶלֶד בְּבַיִת סְפָרַדִּי שֶׁרוֹצֶה לְהַדְלִיק

בְּבַיִת סְפָרַדִּי, אִם הַיְּלָדִים רוֹצִים לְהַדְלִיק – אֶפְשָׁר לָתֵת לָהֶם חֲנֻכִּיָּה מִשֶּׁלָּהֶם, וּמַקְפִּידִים לְהַרְחִיק אוֹתָהּ מֵהַחֲנֻכִּיָּה שֶׁל אַבָּא. הַאִם הַיֶּלֶד מְבָרֵךְ?
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג רַבִּים מֵהַסְּפָרַדִּים מַדְלִיק בְּלִי בְּרָכָה.
לְהוֹרָאַת הָרַב מָרְדְּכַי אֵלִיָּהוּ זַצַ״ל יְלָדִים עַד גִּיל בַּר מִצְוָה רַשָּׁאִים לְבָרֵךְ.
(פֶּרֶק י״ב, הֲלָכָה ג׳)

7בָּנוֹת וְנֵר חֲנֻכָּה

בְּהַרְבֵּה בָּתִּים שֶׁנּוֹהֲגִים כְּמִנְהַג אַשְׁכְּנַז, בָּנוֹת לֹא נָהֲגוּ לְהַדְלִיק. אֲבָל בַּת שֶׁרוֹצָה – רַשָּׁאִית לְהַדְלִיק בִּבְרָכָה. וְטוֹב לְעוֹדֵד בָּנוֹת שֶׁהִגִּיעוּ לְגִיל חִנּוּךְ לְהַדְלִיק, לְפָחוֹת עַד בַּת מִצְוָה. וְאִם יִרְצוּ לְהַדְלִיק גַּם אַחַר כָּךְ – תָּבוֹא עֲלֵיהֶן בְּרָכָה.
🤔 לָמָּה? הַהַדְלָקָה מְקַשֶּׁרֶת אֶת הַבָּנוֹת לַתּוֹרָה וְלַמִּצְווֹת.
💛 הָאוֹר שֶׁל חֲנֻכָּה שַׁיָּךְ לְכֻלָּנוּ – בָּנִים וּבָנוֹת – כִּי כֻּלָּנוּ הָיִינוּ בַּנֵּס.
(פֶּרֶק י״ב, הֲלָכָה ד׳)

8הַבְּרָכוֹת

לִפְנֵי הַהַדְלָקָה מְבָרְכִים שְׁתֵּי בְּרָכוֹת. הָרִאשׁוֹנָה – עַל הַמִּצְוָה. הַשְּׁנִיָּה – ״שֶׁעָשָׂה נִסִּים לַאֲבוֹתֵינוּ בַּיָּמִים הָהֵם בַּזְּמַן הַזֶּה״. בַּלַּיְלָה הָרִאשׁוֹן מוֹסִיפִים ״שֶׁהֶחֱיָנוּ״. מִיָּד אַחֲרֵי הַבְּרָכוֹת מַדְלִיקִים, וְלֹא מְדַבְּרִים בָּאֶמְצַע. אֵיךְ מְסַיְּמִים אֶת הַבְּרָכָה הָרִאשׁוֹנָה?
🤔 לָמָּה? שְׁתֵּי הַבְּרָכוֹת מְכַוְּנוֹת אוֹתָנוּ לִשְׁנֵי הַדְּבָרִים שֶׁבַּמִּצְוָה: לְקַיֵּם אֶת הַמִּצְוָה, וּלְהוֹדוֹת לַה׳ עַל הַנִּסִּים. וּבְ״שֶׁהֶחֱיָנוּ״ מוֹדִים שֶׁזָּכִינוּ לְהַגִּיעַ שׁוּב לַחֲנֻכָּה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
בְּנֻסַּח סְפָרַד ״...וְצִוָּנוּ לְהַדְלִיק נֵר חֲנֻכָּה״.
לְמִנְהַג הָאַשְׁכְּנַזִּים ״...וְצִוָּנוּ לְהַדְלִיק נֵר שֶׁל חֲנֻכָּה״.
(פֶּרֶק י״ב, הֲלָכָה ה׳)

9״הַנֵּרוֹת הַלָּלוּ״ – וּמַדְלִיקִים הֵיטֵב

אַחֲרֵי שֶׁמַּדְלִיקִים אֶת הַנֵּר הָרִאשׁוֹן, נוֹהֲגִים לְהַתְחִיל לוֹמַר ״הַנֵּרוֹת הַלָּלוּ״, וּמַמְשִׁיכִים לְהַדְלִיק אֶת שְׁאָר הַנֵּרוֹת תּוֹךְ כְּדֵי. מִי שֶׁקָּשֶׁה לוֹ – אוֹמֵר אוֹתוֹ אַחֲרֵי שֶׁסִּיֵּם לְהַדְלִיק. מַדְלִיקִים כָּל נֵר הֵיטֵב, וּמְחַכִּים שֶׁהָאֵשׁ תֹּאחַז בְּרֹב הַפְּתִילָה – לֹא מְמַהֲרִים לַנֵּר הַבָּא.
🤔 לָמָּה? ״הַנֵּרוֹת הַלָּלוּ״ מַסְבִּיר אֶת עִנְיַן הַמִּצְוָה, וְלָכֵן הוּא לֹא נֶחְשָׁב הֶפְסֵק. וְהַחוֹבָה מִתְקַיֶּמֶת כְּבָר בַּנֵּר הָרִאשׁוֹן – שְׁאָר הַנֵּרוֹת הֵם הִדּוּר.
(פֶּרֶק י״ב, הֲלָכָה ה׳)

10כָּל הַמִּשְׁפָּחָה מִתְאַסֶּפֶת

מִשְׁתַּדְּלִים שֶׁכָּל בְּנֵי הַבַּיִת יִתְאַסְּפוּ לְהַדְלָקַת הַנֵּרוֹת: יִשְׁמְעוּ אֶת הַבְּרָכוֹת, יַעֲנוּ ״אָמֵן״ וְיִרְאוּ אֶת הַנֵּרוֹת. מִי שֶׁלֹּא מַדְלִיק בְּעַצְמוֹ – כְּמוֹ יְלָדִים שֶׁיּוֹצְאִים בְּהַדְלָקַת אַבָּא – צָרִיךְ לִשְׁמֹעַ אֶת הַבְּרָכוֹת. וּמִי שֶׁלֹּא יָכוֹל לִהְיוֹת בַּבַּיִת – יִשְׁתַּדֵּל לִשְׁמֹעַ בְּרָכוֹת בְּבַיִת אַחֵר אוֹ בְּבֵית הַכְּנֶסֶת.
🤔 לָמָּה? זֶה כָּבוֹד לַמִּצְוָה וּפִרְסוּם לַנֵּס. וּמִי שֶׁשּׁוֹמֵעַ אֶת הַבְּרָכוֹת – מִשְׁתַּתֵּף בַּתּוֹדָה לַה׳ עַל הַנִּסִּים.
💛 כְּשֶׁכָּל הַמִּשְׁפָּחָה עוֹמֶדֶת יַחַד מוּל הַנֵּרוֹת וְעוֹנָה ״אָמֵן״ – כֻּלָּנוּ אוֹמְרִים תּוֹדָה לַה׳ בְּקוֹל אֶחָד.
(פֶּרֶק י״ב, הֲלָכָה ו׳)

11שֶׁמֶן וְנֵרוֹת

כָּל הַשְּׁמָנִים וְהַפְּתִילוֹת כְּשֵׁרִים לְנֵר חֲנֻכָּה – אֲפִלּוּ כָּאֵלֶּה שֶׁפְּסוּלִים לְנֵר שַׁבָּת. כְּכָל שֶׁהַנֵּר מֵאִיר יָפֶה יוֹתֵר – כָּךְ הוּא מְהֻדָּר יוֹתֵר. רַבִּים מַדְלִיקִים נֵרוֹת שַׁעֲוָה אוֹ פָּרָפִין, וְעוֹד יוֹתֵר מְהֻדָּר שֶׁמֶן זַיִת: הָאוֹר שֶׁלּוֹ צָלוּל, וְהוּא מַזְכִּיר אֶת נֵס פַּךְ הַשֶּׁמֶן. צָרִיךְ שֶׁיִּהְיֶה בַּנֵּר מַסְפִּיק כְּדֵי לִדְלֹק חֲצִי שָׁעָה. נֵר קָטָן שֶׁיִּדְלַק פָּחוֹת – מַדְלִיקִים בְּלִי בְּרָכָה.
🤔 לָמָּה? נֵר שַׁבָּת נוֹעַד לְהָאִיר, וְאִם לֹא יִדְלַק יָפֶה, יֵשׁ חֲשָׁשׁ שֶׁיַּטֶּה אוֹתוֹ כְּדֵי לְתַקֵּן וִיחַלֵּל שַׁבָּת. בְּנֵר חֲנֻכָּה אָסוּר לְהִשְׁתַּמֵּשׁ – וְלָכֵן אֵין חֲשָׁשׁ כָּזֶה.
(פֶּרֶק י״ב, הֲלָכָה ז׳)

12נוּרוֹת חַשְׁמַל

לְדַעַת רֹב הַפּוֹסְקִים לֹא יוֹצְאִים יְדֵי חוֹבָה בְּנוּרוֹת חַשְׁמַל. מִי שֶׁאֵין לוֹ שׁוּם נֵר כָּשֵׁר – יַדְלִיק נוּרוֹת חַשְׁמַל בְּלִי בְּרָכָה, כְּדֵי לְהַזְכִּיר אֶת הַנֵּס. חֲנֻכִּיּוֹת חַשְׁמַל גְּדוֹלוֹת בִּמְקוֹמוֹת צִבּוּרִיִּים – לֹא מְקַיְּמִים בָּהֶן אֶת הַמִּצְוָה, אֲבָל יֵשׁ בָּהֶן מַעֲלָה: הֵן מַזְכִּירוֹת לָרַבִּים אֶת הַנֵּס.
🤔 לָמָּה? בְּנוּרָה אֵין פְּתִילָה וְשֶׁמֶן, וְאוֹרָהּ חָזָק כְּמוֹ אֲבוּקָה. נֵר חֲנֻכָּה נוֹעַד לְהַזְכִּיר אֶת הַנֵּס, וְלָכֵן הוּא צָרִיךְ לִהְיוֹת דּוֹמֶה לְנֵרוֹת הַמִּקְדָּשׁ.
(פֶּרֶק י״ב, הֲלָכָה ח׳)

13הַהַדְלָקָה הִיא הַמִּצְוָה

הַמִּצְוָה הִיא לְהַדְלִיק – כְּמוֹ שֶׁאוֹמְרִים בַּבְּרָכָה: ״לְהַדְלִיק נֵר שֶׁל חֲנֻכָּה״. לָכֵן, אִם הַנֵּרוֹת יָכְלוּ לִדְלֹק חֲצִי שָׁעָה, וּמִישֶׁהוּ הִפִּיל אוֹתָם בְּטָעוּת וְהֵם כָּבוּ – כְּבָר יָצָא יְדֵי חוֹבָה. אֲבָל רָאוּי לְהַחְמִיר וּלְהַדְלִיק אוֹתָם שׁוּב, כְּדֵי שֶׁיִּדְלְקוּ חֲצִי שָׁעָה.
(פֶּרֶק י״ב, הֲלָכָה ט׳)

14הַנֵּרוֹת כָּבוּ בָּרוּחַ

מִי שֶׁהִדְלִיק בְּמָקוֹם שֶׁיֵּשׁ בּוֹ רוּחַ, וְהָרוּחַ כִּבְּתָה אֶת הַנֵּרוֹת לִפְנֵי חֲצִי שָׁעָה – לֹא יָצָא, כִּי כְּבָר כְּשֶׁהִדְלִיק, הַנֵּרוֹת לֹא הָיוּ יְכוֹלִים לִדְלֹק חֲצִי שָׁעָה. הוּא צָרִיךְ לְהַדְלִיק שׁוּב, בְּמָקוֹם מוּגָן. הַאִם יְבָרֵךְ שׁוּב?
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת רֹב הַפּוֹסְקִים מַדְלִיק שׁוּב בִּבְרָכָה.
לְפִי הָרַב מֶלָמֵד מַדְלִיק שׁוּב בְּלִי בְּרָכָה, כִּי יֵשׁ בָּזֶה סָפֵק – ״סְפֵק בְּרָכוֹת לְהָקֵל״.
(פֶּרֶק י״ב, הֲלָכָה ט׳)

15מַדְלִיקִים לְשֵׁם מִצְוָה – וּבַמָּקוֹם

נֵר שֶׁדָּלַק עוֹד מִלִּפְנֵי כֵן, לֹא לְשֵׁם מִצְוָה – לֹא יוֹצְאִים בּוֹ, גַּם אִם מְרִימִים אוֹתוֹ וּמַנִּיחִים שׁוּב. צָרִיךְ לְכַבּוֹת אוֹתוֹ וּלְהַדְלִיק מֵחָדָשׁ לְשֵׁם מִצְוָה. וּמַדְלִיקִים אֶת הַנֵּרוֹת בַּמָּקוֹם שֶׁבּוֹ הֵם יַעַמְדוּ – לֹא מַדְלִיקִים בְּמָקוֹם אֶחָד וּמַעֲבִירִים. מִי שֶׁשָּׁכַח לְהַצְמִיד אֶת הַחֲנֻכִּיָּה לַחַלּוֹן – מֻתָּר לְקָרֵב אוֹתָהּ מְעַט אַחֲרֵי הַהַדְלָקָה, כְּדֵי שֶׁיִּרְאוּ אוֹתָהּ טוֹב מֵהָרְחוֹב.
(פֶּרֶק י״ב, הֲלָכָה ט׳)

16לֹא מִשְׁתַּמְּשִׁים בָּאוֹר

אָסוּר לְהִשְׁתַּמֵּשׁ בְּאוֹר נֵרוֹת חֲנֻכָּה – לֹא לִסְפֹּר כֶּסֶף לְאוֹרָם, וַאֲפִלּוּ לֹא לִלְמֹד בְּסֵפֶר. נוֹהֲגִים לֹא לֵהָנוֹת מֵהֶם גַּם אַחֲרֵי חֲצִי שָׁעָה. שֶׁמֶן וּפְתִילוֹת שֶׁנִּשְׁאֲרוּ אַחֲרֵי הַהַדְלָקָה – לֹא מִשְׁתַּמְּשִׁים בָּהֶם לְדָבָר אַחֵר: אֶפְשָׁר לְהַדְלִיק בָּהֶם בַּלֵּילוֹת הַבָּאִים, אוֹ לִשְׂרֹף אוֹתָם, אוֹ לִשְׁפֹּךְ לַכִּיּוֹר וְלִזְרֹק לַפַּח. מָה שֶׁנִּשְׁאַר בַּחֲבִילָה אוֹ בַּבַּקְבּוּק – מֻתָּר לְכָל דָּבָר. וְאִם הַשַּׁמָּשׁ כָּבָה בְּאֶמְצַע – מַדְלִיקִים אוֹתוֹ בְּגַפְרוּר, וְלֹא מִנֵּר שֶׁל מִצְוָה.
🤔 לָמָּה? הַנֵּרוֹת מְקֻדָּשִׁים לַמִּצְוָה, כְּמוֹ הַנֵּרוֹת בִּמְנוֹרַת הַמִּקְדָּשׁ. וְהֵם נוֹעֲדוּ לְפִרְסוּם הַנֵּס – מִי שֶׁמִּשְׁתַּמֵּשׁ בָּהֶם נִרְאֶה כְּמִי שֶׁהִדְלִיק כְּדֵי לְהָאִיר. וְהַשַּׁמָּשׁ הוּא נֵר שֶׁל חֹל, וְלֹא מַדְלִיקִים נֵר שֶׁל חֹל מִנֵּר שֶׁל מִצְוָה (רַק אִם אֵין שׁוּם גַּפְרוּר – מֻתָּר).
(פֶּרֶק י״ב, הֲלָכָה י׳)

17הַחֲנֻכִּיָּה וְהַשַּׁמָּשׁ

רָאוּי לְהַדְלִיק בַּחֲנֻכִּיָּה נָאָה, כָּל אֶחָד לְפִי יְכָלְתּוֹ. מִי שֶׁאֵין לוֹ חֲנֻכִּיָּה – יָכוֹל לְהַדְבִּיק נֵרוֹת עַל מַשְׁטָח יָשָׁר, אוֹ לְהַנִּיחַ שֶׁמֶן וּפְתִילָה בְּכוֹסִיּוֹת. מַשְׁאִירִים רֶוַח בֵּין הַנֵּרוֹת, וּלְכַתְּחִלָּה מַעֲמִידִים אוֹתָם בְּשׁוּרָה שָׁוָה. וּמַדְלִיקִים גַּם שַׁמָּשׁ – נֵר שֶׁמֻּתָּר לְהִשְׁתַּמֵּשׁ לְאוֹרוֹ. מַנִּיחִים אוֹתוֹ בְּגֹבַהּ אַחֵר אוֹ בְּרִחוּק מֵהַנֵּרוֹת, וְנָהֲגוּ לְהַנִּיחַ אוֹתוֹ גָּבוֹהַּ יוֹתֵר.
🤔 לָמָּה? כָּל נֵר צָרִיךְ לְהֵרָאוֹת לְבַד, כְּדֵי שֶׁיִּרְאוּ אֶת מִסְפַּר הַנֵּרוֹת; וְאִם הַלֶּהָבוֹת יִתְאַחֲדוּ – זֶה כְּמוֹ מְדוּרָה, וְלֹא יוֹצְאִים בָּזֶה. וְאֶת הַשַּׁמָּשׁ מַרְחִיקִים כְּדֵי שֶׁלֹּא יִטְעוּ בְּמִסְפַּר הַנֵּרוֹת.
(פֶּרֶק י״ב, הֲלָכָה י״א)

18מֵאֵיפֹה מַתְחִילִים?

בַּלַּיְלָה הָרִאשׁוֹן קוֹבְעִים אֶת הַנֵּר בַּצַּד הַיְּמָנִי שֶׁל הַחֲנֻכִּיָּה – יָמִין שֶׁל הַמַּדְלִיק, שֶׁעוֹמֵד מוּלָהּ. בְּכָל לַיְלָה מוֹסִיפִים נֵר מִשְּׂמֹאל, מַדְלִיקִים קֹדֶם אֶת הַנֵּר הֶחָדָשׁ, וְאַחַר כָּךְ פּוֹנִים יָמִינָה וּמַדְלִיקִים אֶת הַשְּׁאָר. טוֹב לַעֲמֹד בִּזְמַן הַבְּרָכוֹת לְצַד שְׂמֹאל שֶׁל הַחֲנֻכִּיָּה, כָּךְ שֶׁהַנֵּר הַקָּרוֹב הוּא הַנֵּר שֶׁמַּדְלִיקִים רִאשׁוֹן. בְּכָל דֶּרֶךְ שֶׁיַּדְלִיק – קִיֵּם אֶת הַמִּצְוָה.
🤔 לָמָּה? תָּמִיד מַעֲדִיפִים אֶת צַד יָמִין. הַנֵּר הֶחָדָשׁ מְבַטֵּא אֶת הַנֵּס שֶׁהוֹלֵךְ וְגָדֵל, וְלָכֵן מַתְחִילִים בּוֹ. וְאַחַר כָּךְ פּוֹנִים יָמִינָה – כִּי אָמְרוּ חֲכָמִים שֶׁבְּכָל פְּנִיָּה עָדִיף לִפְנוֹת דֶּרֶךְ יָמִין.
(פֶּרֶק י״ב, הֲלָכָה י״ב)

19לִפְנֵי הַהַדְלָקָה

בַּחֲצִי הַשָּׁעָה שֶׁלִּפְנֵי זְמַן הַהַדְלָקָה אָסוּר לְהַתְחִיל אֲרוּחָה, אֲפִלּוּ קַלָּה. אֲבָל פֵּרוֹת וִירָקוֹת מֻתָּר לֶאֱכֹל בְּלִי הַגְבָּלָה, וְלֶחֶם אוֹ עוּגוֹת – עַד כַּבֵּיצָה. גַּם לֹא הוֹלְכִים לִישֹׁן וְלֹא מַתְחִילִים עֲבוֹדָה אֲרֻכָּה. כְּשֶׁמַּגִּיעַ הַזְּמַן – מִזְדָּרְזִים לְהַדְלִיק, וַאֲפִלּוּ לִלְמֹד תּוֹרָה אָסוּר אָז. אֲבָל שִׁעוּר תּוֹרָה קָבוּעַ שֶׁמִּתְקַיֵּם בְּאוֹתָהּ שָׁעָה – מְקַיְּמִים, וּבְסוֹפוֹ מַזְכִּירִים לְכֻלָּם לְהַדְלִיק.
🤔 לָמָּה? שֶׁמָּא יִמָּשֵׁךְ בַּאֲרוּחָה אוֹ בָּעֲבוֹדָה, וְיִשְׁכַּח לְהַדְלִיק. מִי שֶׁבִּקֵּשׁ מֵחָבֵר שֶׁיַּזְכִּיר לוֹ – מֻתָּר לוֹ לְהַתְחִיל.
(פֶּרֶק י״ב, הֲלָכָה י״ג)

20בְּבֵית הַכְּנֶסֶת

נָהֲגוּ לְהַדְלִיק נֵרוֹת גַּם בְּבֵית הַכְּנֶסֶת, בְּכָל הַבְּרָכוֹת. מַנִּיחִים אֶת הַחֲנֻכִּיָּה לְיַד הַכֹּתֶל הַדְּרוֹמִי, כְּמוֹ מְקוֹם הַמְּנוֹרָה בַּמִּקְדָּשׁ, וּמַדְלִיקִים לִפְנֵי תְּפִלַּת עַרְבִית, כְּשֶׁיֵּשׁ שָׁם הַרְבֵּה אֲנָשִׁים. צְרִיכִים לִהְיוֹת נוֹכְחִים לְפָחוֹת עֲשָׂרָה. בְּעֶרֶב שַׁבָּת – בְּדֶרֶךְ כְּלָל מַדְלִיקִים לִפְנֵי מִנְחָה, וּבְמוֹצָאֵי שַׁבָּת – אַחֲרֵי עַרְבִית. הַהַדְלָקָה בְּבֵית הַכְּנֶסֶת לֹא פּוֹטֶרֶת: גַּם מִי שֶׁהִדְלִיק שָׁם – חוֹזֵר וּמַדְלִיק בְּבֵיתוֹ בִּבְרָכָה.
🤔 לָמָּה? חֲכָמִים תִּקְּנוּ לְהַדְלִיק בַּבַּיִת, וְנָהֲגוּ לְהַדְלִיק גַּם בְּבֵית הַכְּנֶסֶת כְּדֵי לְפַרְסֵם אֶת הַנֵּס יוֹתֵר.
(פֶּרֶק י״ב, הֲלָכָה י״ד)

21חֲנֻכִּיָּה בִּמְסִבָּה

רַבִּים מַדְלִיקִים חֲנֻכִּיָּה בְּכָל מָקוֹם שֶׁמִּתְאַסְּפִים בּוֹ אֲנָשִׁים – בַּחֲתֻנָּה, בְּבַר מִצְוָה אוֹ בִּמְסִבַּת חֲנֻכָּה. הַאִם מְבָרְכִים עַל הַהַדְלָקָה?
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת רַבִּים מֵרַבָּנֵי דּוֹרֵנוּ מַדְלִיקִים בְּלִי בְּרָכָה, כִּי רַק בְּבֵית הַכְּנֶסֶת נָהֲגוּ לְבָרֵךְ.
לְדַעַת כַּמָּה פּוֹסְקִים מֻתָּר לְבָרֵךְ, כִּי הַמַּטָּרָה הִיא פִּרְסוּם הַנֵּס. וְאִם יֵשׁ שָׁם אֲנָשִׁים שֶׁאוּלַי לֹא הִדְלִיקוּ בְּבֵיתָם – חָשׁוּב מְאוֹד לְהַדְלִיק בִּבְרָכָה, כְּדֵי שֶׁיִּשְׁמְעוּ וְיִלְמְדוּ.
💛 כְּדַאי לְכַבֵּד בַּהַדְלָקָה יְהוּדִי שֶׁאֵינוֹ רָגִיל בְּמִצְווֹת – כְּדֵי שֶׁיִּהְיֶה בָּרוּר שֶׁהַמִּצְווֹת שַׁיָּכוֹת לְכָל עַם יִשְׂרָאֵל.
(פֶּרֶק י״ב, הֲלָכָה ט״ו)

📝 מָה לָמַדְנוּ

  • כָּל יִשְׂרָאֵל מַדְלִיקִים כְּ״מְהַדְּרִין מִן הַמְּהַדְּרִין״: נֵר אֶחָד בַּלַּיְלָה הָרִאשׁוֹן, וּמוֹסִיפִים עַד שְׁמוֹנָה.
  • לְמִנְהַג הַסְּפָרַדִּים גְּדוֹל הַבַּיִת מַדְלִיק לְכֻלָּם; לְמִנְהַג הָאַשְׁכְּנַזִּים כָּל אֶחָד מַדְלִיק חֲנֻכִּיָּה מִשֶּׁלּוֹ.
  • מְבָרְכִים ״לְהַדְלִיק נֵר (שֶׁל) חֲנֻכָּה״ וְ״שֶׁעָשָׂה נִסִּים״, וּבַלַּיְלָה הָרִאשׁוֹן גַּם ״שֶׁהֶחֱיָנוּ״ – וְכָל הַמִּשְׁפָּחָה שׁוֹמַעַת וְעוֹנָה ״אָמֵן״.
  • הַמִּצְוָה הִיא בַּהַדְלָקָה, בְּנֵר שֶׁיָּכוֹל לִדְלֹק חֲצִי שָׁעָה; וְאָסוּר לְהִשְׁתַּמֵּשׁ בְּאוֹר הַנֵּרוֹת – לְשֵׁם כָּךְ יֵשׁ שַׁמָּשׁ.
  • בַּלַּיְלָה הַשֵּׁנִי וְהָלְאָה מַדְלִיקִים קֹדֶם אֶת הַנֵּר הֶחָדָשׁ שֶׁבַּשְּׂמֹאל, וּפוֹנִים יָמִינָה.

❓ שְׁאֵלוֹת חֲזָרָה

  1. מָה הֵן שָׁלֹשׁ הַמַּדְרֵגוֹת שֶׁל נֵר חֲנֻכָּה, וְאֵיךְ נוֹהֲגִים כָּל יִשְׂרָאֵל?
  2. הַחֲנֻכִּיָּה נָפְלָה בְּטָעוּת אַחֲרֵי חָמֵשׁ דַּקּוֹת, וְהַנֵּרוֹת כָּבוּ. הַאִם יָצָאנוּ? וּמָה הַהֶבְדֵּל אִם הָרוּחַ כִּבְּתָה אוֹתָם?
  3. לָמָּה אָסוּר לִקְרֹא סֵפֶר לְאוֹר נֵרוֹת הַחֲנֻכָּה, וּבִשְׁבִיל מָה צָרִיךְ שַׁמָּשׁ?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״מַדְלִיקִים נֵר חֲנֻכָּה״

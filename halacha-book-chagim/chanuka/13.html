סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת חֲנֻכָּה
מַהֲדוּרַת יְלָדִים

🪟 דִּינֵי הַמָּקוֹם וְהַזְּמַן

פֶּרֶק י״ג · לַמַּהֲדוּרָה לַמּוֹרֶה
אַתֶּם הוֹלְכִים בָּרְחוֹב בְּעֶרֶב חֲנֻכָּה וְסוֹפְרִים חַלּוֹנוֹת: כָּאן שְׁלֹשָׁה נֵרוֹת, שָׁם שְׁלֹשָׁה, וּבַחַלּוֹן הַגָּבוֹהַּ – עוֹד חֲנֻכִּיָּה! כָּל נֵר כָּזֶה ״מְסַפֵּר״ לָעוֹבְרִים: הָיָה נֵס! אֲבָל אֵיפֹה בְּדִיּוּק שָׂמִים אֶת הַחֲנֻכִּיָּה? בְּאֵיזוֹ שָׁעָה מַדְלִיקִים? וּמָה עוֹשִׂים כְּשֶׁאֲנַחְנוּ אֵצֶל סַבָּא, בְּמָלוֹן אוֹ בְּטִיּוּל? בַּפֶּרֶק הַזֶּה נִלְמַד אֵיךְ מְפַרְסְמִים אֶת הַנֵּס – בַּמָּקוֹם הַנָּכוֹן וּבַזְּמַן הַנָּכוֹן.

1עַל פֶּתַח הַבַּיִת מִבַּחוּץ

אָמְרוּ חֲכָמִים: ״נֵר חֲנֻכָּה – מִצְוָה לְהַנִּיחוֹ עַל פֶּתַח בֵּיתוֹ מִבַּחוּץ״. מַדְלִיקִים בְּצַד שְׂמֹאל שֶׁל הַפֶּתַח, מוּל הַמְּזוּזָה. מִי שֶׁגָּר בַּקּוֹמָה הַשְּׁנִיָּה, וְאֵין לוֹ פֶּתַח מִשֶּׁלּוֹ לָרְחוֹב – מַנִּיחַ בַּחַלּוֹן שֶׁפּוֹנֶה לָרְחוֹב. וּבִזְמַן סַכָּנָה, כְּשֶׁפָּחֲדוּ מֵהַגּוֹיִים – הִנִּיחוּ אֶת הַנֵּר עַל הַשֻּׁלְחָן בְּתוֹךְ הַבַּיִת.
🤔 לָמָּה? בַּפֶּתַח רוֹאִים אֶת הַנֵּר כָּל הָעוֹבְרִים בָּרְחוֹב, וְהַנֵּס מִתְפַּרְסֵם. וּבְצַד שְׂמֹאל – כְּדֵי שֶׁמִּי שֶׁנִּכְנָס יִהְיֶה מֻקָּף בְּמִצְווֹת: הַמְּזוּזָה מִיָּמִין, וְנֵר חֲנֻכָּה מִשְּׂמֹאל.
💛 כְּשֶׁאֲנִי נִכְנָס הַבַּיְתָה בֵּין הַמְּזוּזָה לְבֵין נֵר חֲנֻכָּה – אֲנִי מַרְגִּישׁ שֶׁהַמִּצְווֹת שֶׁל ה׳ עוֹטְפוֹת אוֹתִי.
(פֶּרֶק י״ג, הֲלָכָה א׳)

2בְּאֵיזֶה גֹּבַהּ?

לְכַתְּחִלָּה מַנִּיחִים אֶת הַחֲנֻכִּיָּה כָּךְ שֶׁהַלֶּהָבָה תִּהְיֶה בֵּין שְׁלֹשָׁה לַעֲשָׂרָה טְפָחִים – בְּעֵרֶךְ בֵּין 23 לְ־76 ס״מ מֵהָרִצְפָּה. בְּדִיעֲבַד, מִי שֶׁהִדְלִיק נָמוּךְ אוֹ גָּבוֹהַּ יוֹתֵר – יָצָא. אֲבָל מִי שֶׁהִדְלִיק בַּפֶּתַח לְמַעְלָה מֵעֶשְׂרִים אַמָּה (כְּתִשְׁעָה מֶטֶר) – לֹא יָצָא.
🤔 לָמָּה? אִם הַנֵּר גָּבוֹהַּ מִדַּי – יַחְשְׁבוּ שֶׁהוּא נוֹעַד לְהָאִיר אֶת הַכְּנִיסָה. אִם הוּא עַל הָרִצְפָּה – יֵרָאֶה כְּאִלּוּ הִנִּיחוּ אוֹתוֹ שָׁם רַק לְרֶגַע. וּלְמַעְלָה מֵעֶשְׂרִים אַמָּה – אֲנָשִׁים בִּכְלָל לֹא מִסְתַּכְּלִים כָּל כָּךְ גָּבוֹהַּ.
(פֶּרֶק י״ג, הֲלָכָה א׳)

3בְּבַיִת פְּרָטִי

בִּימֵי חֲכָמִים הַבָּתִים הָיוּ צְפוּפִים, וְהָרוּחַ לֹא כִּבְּתָה אֶת הַנֵּרוֹת. הַיּוֹם, נֵרוֹת שֶׁמַּדְלִיקִים מִחוּץ לַפֶּתַח יִכְבּוּ בָּרוּחַ. מִי שֶׁרוֹצֶה לְהַדֵּר בְּכָל הַהִדּוּרִים – קוֹנֶה תֵּבַת זְכוּכִית וּמַדְלִיק בָּהּ מִחוּץ לַפֶּתַח, בְּצַד שְׂמֹאל. מִי שֶׁלֹּא – מַדְלִיק בְּתוֹךְ הַבַּיִת. כְּשֶׁכַּמָּה בְּנֵי מִשְׁפָּחָה מַדְלִיקִים, שָׂמִים לֵב שֶׁכָּל חֲנֻכִּיָּה תֵּרָאֶה לְבַד, וְאֶפְשָׁר לְהַדְלִיק בְּכַמָּה חַלּוֹנוֹת. וּכְשֶׁהַפֶּתַח מֻסְתָּר מֵהָרְחוֹב, וְאֶת הַחַלּוֹן רוֹאִים יוֹתֵר – אֵיפֹה עָדִיף?
🤔 לָמָּה? חֲכָמִים לֹא חִיְּבוּ לִקְנוֹת תֵּבַת זְכוּכִית, וְלָכֵן אֶפְשָׁר לְהַדְלִיק בִּפְנִים. אֲבָל מִי שֶׁמַּדְלִיק בַּחַלּוֹן שֶׁפּוֹנֶה לָרְחוֹב – מְהַדֵּר בְּפִרְסוּם הַנֵּס.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ אוֹמְרִים עָדִיף בַּפֶּתַח, כְּמוֹ שֶׁתִּקְּנוּ חֲכָמִים בַּהַתְחָלָה – כְּדֵי שֶׁהַנִּכְנָס יִהְיֶה מֻקָּף בְּמִצְווֹת.
וְיֵשׁ אוֹמְרִים – וְכָךְ נִרְאֶה לָהָרַב מֶלָמֵד עָדִיף בַּחַלּוֹן, כִּי עִקַּר הַתַּקָּנָה הוּא פִּרְסוּם הַנֵּס, וּמֵהַחַלּוֹן יוֹתֵר אֲנָשִׁים רוֹאִים. וְגַם לְמִי שֶׁמַּדְלִיק בַּפֶּתַח יֵשׁ מַעֲלָה.
(פֶּרֶק י״ג, הֲלָכָה ב׳)

4בְּבִנְיַן דִּירוֹת

רֹב הָאֲנָשִׁים גָּרִים הַיּוֹם בְּבִנְיְנֵי קוֹמוֹת. הַמָּקוֹם הַטּוֹב בְּיוֹתֵר הוּא חַלּוֹן אוֹ מִרְפֶּסֶת שֶׁפּוֹנִים לָרְחוֹב, כָּךְ שֶׁגַּם בְּנֵי הַבַּיִת וְגַם הָעוֹבְרִים בָּרְחוֹב יִרְאוּ אֶת הַנֵּרוֹת. לָכֵן לֹא לוֹקְחִים חֲנֻכִּיָּה עִם גַּב, כִּי הַגַּב מַסְתִּיר אֶת הַנֵּרוֹת מִצַּד אֶחָד. מַדְלִיקִים בַּחַלּוֹן גַּם אִם הוּא גָּבוֹהַּ מֵעֲשָׂרָה טְפָחִים, וְגַם בְּקוֹמָה גְּבוֹהָה – אִם רוֹאִים אֶת הַנֵּרוֹת מֵהָרְחוֹב. וּמִי שֶׁמַּדְלִיק בְּתוֹךְ הַבַּיִת – גַּם יָצָא. אֲבָל מָה עָדִיף: הַחַלּוֹן, אוֹ פֶּתַח הַדִּירָה שֶׁפּוֹנֶה לַחֲדַר הַמַּדְרֵגוֹת?
🤔 לָמָּה? יוֹתֵר חָשׁוּב שֶׁהַרְבֵּה אֲנָשִׁים יִרְאוּ אֶת הַנֵּרוֹת מֵאֲשֶׁר לְדַקְדֵּק בַּגֹּבַהּ שֶׁל הַחֲנֻכִּיָּה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ סוֹבְרִים עָדִיף עַל פֶּתַח הַדִּירָה, כְּדֵי שֶׁהַנִּכְנָס יִהְיֶה מֻקָּף בְּמִצְווֹת – מְזוּזָה מִיָּמִין וְנֵר מִשְּׂמֹאל.
לְפִי הָרַב מֶלָמֵד טוֹב יוֹתֵר בַּחַלּוֹן שֶׁפּוֹנֶה לָרְחוֹב, כִּי שָׁם הַנֵּס מִתְפַּרְסֵם יוֹתֵר. וְגַם בַּפֶּתַח יוֹצְאִים יְדֵי חוֹבָה.
(פֶּרֶק י״ג, הֲלָכָה ג׳)

5מָתַי מַדְלִיקִים?

חֲכָמִים קָבְעוּ לְהַדְלִיק ״מִשֶּׁתִּשְׁקַע הַחַמָּה וְעַד שֶׁתִּכְלֶה רֶגֶל מִן הַשּׁוּק״ – כְּשֶׁכְּבָר מַתְחִיל לְהַחְשִׁיךְ, וְהָרְחוֹבוֹת עוֹד מְלֵאִים אֲנָשִׁים שֶׁחוֹזְרִים הַבַּיְתָה. נֶחְלְקוּ הָרִאשׁוֹנִים מָה הַכַּוָּנָה:
🤔 לָמָּה? מִי שֶׁרָגִיל לְהִתְפַּלֵּל עַרְבִית מִיָּד בְּצֵאת הַכּוֹכָבִים – מִתְפַּלֵּל קֹדֶם, כִּי עַרְבִית הִיא תְּפִלָּה שֶׁל כָּל יוֹם וְיֵשׁ בָּהּ קְרִיאַת שְׁמַע, וְאַחַר כָּךְ מְמַהֵר הַבַּיְתָה לְהַדְלִיק. וּמִי שֶׁמִּתְפַּלֵּל מְאֻחָר יוֹתֵר – מַדְלִיק קֹדֶם, וְאַחַר כָּךְ מִתְפַּלֵּל.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ אוֹמְרִים מִתְּחִלַּת הַשְּׁקִיעָה, כְּשֶׁהַשֶּׁמֶשׁ נֶעֱלֶמֶת – כִּי אָז יֵשׁ יוֹתֵר אֲנָשִׁים בָּרְחוֹב.
הַמִּנְהָג הָרוֹוֵחַ בְּצֵאת הַכּוֹכָבִים, כְּעֶשְׂרִים דַּקּוֹת אַחֲרֵי הַשְּׁקִיעָה – כִּי אָז כְּבָר חָשׁוּךְ, וְהַנֵּרוֹת נִרְאִים הֵיטֵב.
(פֶּרֶק י״ג, הֲלָכָה ד׳)

6נֵרוֹת שֶׁדּוֹלְקִים הַרְבֵּה זְמַן

מַסְפִּיק נֵר שֶׁדּוֹלֵק חֲצִי שָׁעָה. אֲבָל כְּשֶׁהַנֵּרוֹת נִרְאִים לָעוֹבְרִים בָּרְחוֹב, טוֹב לְהַדֵּר וּלְהָכִין נֵרוֹת שֶׁיִּדְלְקוּ שְׁעָתַיִם וְאַף יוֹתֵר.
🤔 לָמָּה? הַיּוֹם אֲנָשִׁים מִסְתּוֹבְבִים בָּרְחוֹב עַד מְאֻחָר – וְטוֹב שֶׁכֻּלָּם יִרְאוּ אֶת הַנֵּרוֹת, וְהַנֵּס יִתְפַּרְסֵם.
(פֶּרֶק י״ג, הֲלָכָה ד׳)

7עֶרֶב שַׁבַּת חֲנֻכָּה

בְּעֶרֶב שַׁבָּת אִי אֶפְשָׁר לְהַדְלִיק אַחֲרֵי הַשְּׁקִיעָה, כִּי הַשַּׁבָּת כְּבָר נִכְנְסָה. לָכֵן מַדְלִיקִים נֵרוֹת חֲנֻכָּה לִפְנֵי נֵרוֹת שַׁבָּת. מַקְפִּידִים שֶׁיִּהְיֶה בַּנֵּרוֹת מַסְפִּיק שֶׁמֶן, כְּדֵי שֶׁיִּדְלְקוּ עַד חֲצִי שָׁעָה אַחֲרֵי צֵאת הַכּוֹכָבִים. וְטוֹב לְהִתְפַּלֵּל קֹדֶם מִנְחָה בְּצִבּוּר.
🤔 לָמָּה? אָסוּר לְהַדְלִיק אֵשׁ בְּשַׁבָּת. וְעִקַּר פִּרְסוּם הַנֵּס הוּא בַּלַּיְלָה – לָכֵן צָרִיךְ שֶׁהַנֵּרוֹת יַמְשִׁיכוּ לִדְלֹק כְּשֶׁכְּבָר חָשׁוּךְ. וּמִנְחָה שַׁיֶּכֶת לַיּוֹם שֶׁעוֹבֵר, וְהַנֵּרוֹת – לַלַּיְלָה שֶׁבָּא.
(פֶּרֶק י״ג, הֲלָכָה ה׳)

8מוֹצָאֵי שַׁבַּת חֲנֻכָּה

בְּמוֹצָאֵי שַׁבָּת מִתְפַּלְּלִים קֹדֶם עַרְבִית, וְאַחַר כָּךְ מַדְלִיקִים נֵרוֹת חֲנֻכָּה. וּמָה עוֹשִׂים קֹדֶם – הַבְדָּלָה אוֹ נֵר חֲנֻכָּה?
🤔 לָמָּה? לִשְׁנֵי הַמִּנְהָגִים יֵשׁ מָקוֹם בַּהֲלָכָה, וְכָל אֶחָד רַשַּׁאי לִבְחֹר אֶת מִנְהָגוֹ.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
רַבִּים נוֹהֲגִים לְהַבְדִּיל קֹדֶם, כִּי הַהַבְדָּלָה מְסַיֶּמֶת אֶת הַשַּׁבָּת, וְנֵרוֹת הַחֲנֻכָּה שַׁיָּכִים כְּבָר לַיּוֹם הַבָּא.
וְיֵשׁ נוֹהֲגִים לְהַדְלִיק קֹדֶם נֵר חֲנֻכָּה – כְּדֵי לְהַדְלִיק קָרוֹב כְּכָל הָאֶפְשָׁר לְצֵאת הַכּוֹכָבִים, וּלְהַאֲרִיךְ קְצָת אֶת קְדֻשַּׁת הַשַּׁבָּת. מִי שֶׁמַּדְלִיק לִפְנֵי הַהַבְדָּלָה צָרִיךְ לוֹמַר קֹדֶם ״אַתָּה חוֹנַנְתָּנוּ״ בַּתְּפִלָּה, אוֹ ״בָּרוּךְ הַמַּבְדִּיל בֵּין קֹדֶשׁ לְחֹל״.
(פֶּרֶק י״ג, הֲלָכָה ה׳)

9מִי שֶׁחוֹזֵר מְאֻחָר

הַזְּמַן הַמְּהֻדָּר הוּא מִצֵּאת הַכּוֹכָבִים וְעַד חֲצִי שָׁעָה אַחֲרָיו. מִי שֶׁקָּשֶׁה לוֹ לַחֲזֹר הַבַּיְתָה בַּזְּמַן הַזֶּה – רַשַּׁאי לְהַדְלִיק מְאֻחָר יוֹתֵר, וּמִשְׁתַּדֵּל לַחֲזֹר מַהֵר כְּכָל שֶׁאֶפְשָׁר, וְלֹא לְאַחֵר אַחֲרֵי הַשָּׁעָה תֵּשַׁע. עַד שֶׁמַּדְלִיק – לֹא אוֹכֵל אֲרוּחָה. וּבְבֵית כְּנֶסֶת שֶׁיֵּשׁ בּוֹ שִׁעוּר תּוֹרָה קָבוּעַ אַחֲרֵי עַרְבִית – עָדִיף לְקַיֵּם אֶת הַשִּׁעוּר, וּבְסוֹפוֹ לְהַזְכִּיר לְכֻלָּם לְהַדְלִיק.
🤔 לָמָּה? עַד הַשָּׁעָה תֵּשַׁע עוֹד מְקֻבָּל הַיּוֹם לַחֲזֹר מֵהָעֲבוֹדָה, וְהָרְחוֹבוֹת עֲדַיִן מְלֵאִים. וּמִצְוַת לִמּוּד תּוֹרָה עֲדִיפָה עַל הַהִדּוּר לְהַדְלִיק בַּזְּמַן הַמְּהֻדָּר.
(פֶּרֶק י״ג, הֲלָכָה ו׳)

10מְחַכִּים לְבְנֵי הַמִּשְׁפָּחָה

לִפְעָמִים אַבָּא אוֹ אִמָּא חוֹזְרִים מְאֻחָר. מִצַּד הַדִּין, מִי שֶׁבַּבַּיִת יָכוֹל לְהַדְלִיק בַּזְּמַן, וְכֻלָּם יוֹצְאִים. אֲבָל בְּהַרְבֵּה מִקְרִים עָדִיף לְחַכּוֹת לְמִי שֶׁמְּאַחֵר: אִם לֹא יוּכַל לִשְׁמֹעַ אֶת הַבְּרָכוֹת בְּמָקוֹם אַחֵר, אִם יִצְטַעֵר שֶׁהִדְלִיקוּ בְּלְעָדָיו, אוֹ אִם הוּא מְאַחֵר כָּל יוֹם וְהַקֶּשֶׁר שֶׁלּוֹ לַמִּצְוָה עָלוּל לְהֵחָלֵשׁ. בְּבַיִת שֶׁנּוֹהֵג כְּמִנְהַג רֹב הַסְּפָרַדִּים, שֶׁרַק אֶחָד מַדְלִיק – אִם הַבָּנִים וְהַבָּנוֹת חוֹזְרִים מְאֻחָר, מוּטָב לְחַכּוֹת לָהֶם.
🤔 לָמָּה? חָשׁוּב שֶׁכָּל אֶחָד בַּמִּשְׁפָּחָה יַרְגִּישׁ שֶׁהוּא חֵלֶק מֵהַמִּצְוָה, וְשֶׁיִּהְיֶה שָׁלוֹם בַּבַּיִת. גַּם כְּשֶׁמְּחַכִּים – לֹא מְאַחֲרִים אַחֲרֵי הַשָּׁעָה תֵּשַׁע, וְלֹא אוֹכְלִים אֲרוּחָה עַד הַהַדְלָקָה.
💛 נֵר חֲנֻכָּה שֶׁכָּל הַמִּשְׁפָּחָה עוֹמֶדֶת סְבִיבוֹ – מֵאִיר יוֹתֵר.
(פֶּרֶק י״ג, הֲלָכָה ז׳)

11מְאֻחָר בַּלַּיְלָה

מִי שֶׁלֹּא הִדְלִיק עַד בְּעֵרֶךְ הַשָּׁעָה תֵּשַׁע – מַדְלִיק כָּל הַלַּיְלָה, עַד עֲלוֹת הַשַּׁחַר. הוּא מְבָרֵךְ רַק אִם סָבִיר שֶׁמִּישֶׁהוּ יִרְאֶה אֶת הַנֵּרוֹת – אֶחָד מִבְּנֵי הַבַּיִת אוֹ אָדָם בָּרְחוֹב. אִם כֻּלָּם יְשֵׁנִים, וּבָרְחוֹב אֵין אַף אֶחָד – מַדְלִיק בְּלִי בְּרָכָה. מִי שֶׁלֹּא הִדְלִיק כָּל הַלַּיְלָה – הִפְסִיד אֶת הַמִּצְוָה שֶׁל אוֹתוֹ לַיְלָה, אֲבָל לְמָחֳרָת מַדְלִיק כְּמוֹ כָּל הָאֲנָשִׁים.
🤔 לָמָּה? הַנֵּרוֹת נוֹעֲדוּ לְפַרְסֵם אֶת הַנֵּס – וּכְשֶׁאֵין מִי שֶׁיִּרְאֶה, לֹא מְבָרְכִים.
(פֶּרֶק י״ג, הֲלָכָה ח׳)

12לִפְנֵי הַשְּׁקִיעָה – רַק בִּשְׁעַת הַדְּחָק

בְּיוֹם רָגִיל לֹא מַדְלִיקִים לִפְנֵי הַשְּׁקִיעָה. רַק מִי שֶׁאֵין לוֹ בְּרֵרָה – לְמָשָׁל, הוּא חַיָּב לָצֵאת לִפְנֵי הַשְּׁקִיעָה וְיַחֲזֹר מְאֹד מְאֻחָר, וְאֵין בַּבַּיִת מִי שֶׁיַּדְלִיק – יָכוֹל לְהַדְלִיק מִ״פְּלַג הַמִּנְחָה״, כְּאַרְבָּעִים דַּקּוֹת לִפְנֵי הַשְּׁקִיעָה, בִּבְרָכָה, וְיַקְפִּיד שֶׁהַנֵּרוֹת יִדְלְקוּ עַד חֲצִי שָׁעָה אַחֲרֵי צֵאת הַכּוֹכָבִים.
(פֶּרֶק י״ג, הֲלָכָה ח׳)

13אֲרוּחַת עֶרֶב אֵצֶל קְרוֹבִים

מִשְׁפָּחָה שֶׁאוֹכֶלֶת אֲרוּחַת עֶרֶב אֵצֶל קְרוֹבִים אוֹ חֲבֵרִים, וְחוֹזֶרֶת לִישֹׁן בְּבֵיתָהּ – הַבַּיִת שֶׁל הַמְּאָרְחִים אֵינוֹ הַבַּיִת שֶׁלָּהּ, וְהִיא לֹא יוֹצֵאת שָׁם יְדֵי חוֹבָה. טוֹב שֶׁאֶחָד מִבְּנֵי הַמִּשְׁפָּחָה יֵלֵךְ הַבַּיְתָה לְהַדְלִיק בְּצֵאת הַכּוֹכָבִים, וְהַשְּׁאָר יִשְׁמְעוּ בְּרָכוֹת וְיִרְאוּ נֵרוֹת אֵצֶל הַמְּאָרְחִים. וְאִם הֵם רְחוֹקִים – יַדְלִיקוּ בִּבְרָכָה כְּשֶׁיַּחְזְרוּ, וְיִזְכְּרוּ לְהַזְכִּיר זֶה לָזֶה.
🤔 לָמָּה? חֲכָמִים תִּקְּנוּ לְהַדְלִיק בַּבַּיִת שֶׁבּוֹ גָּרִים – הַמָּקוֹם שֶׁבּוֹ יְשֵׁנִים.
(פֶּרֶק י״ג, הֲלָכָה ט׳)

14יְשֵׁנִים אֵצֶל הַמְּאָרְחִים

מִשְׁפָּחָה שֶׁגַּם יְשֵׁנָה אֵצֶל הַמְּאָרְחִים – זֶה הַבַּיִת שֶׁלָּהּ בְּאוֹתוֹ לַיְלָה. וְכָךְ גַּם מִשְׁפָּחָה שֶׁמִּתְאָרַחַת כָּל הַשַּׁבָּת אֵצֶל סַבָּא – הַבַּיִת שֶׁל סַבָּא הוּא הַבַּיִת שֶׁלָּהּ בְּאוֹתָהּ שַׁבָּת. אֵיךְ הָאוֹרְחִים מְקַיְּמִים אֶת הַמִּצְוָה?
🤔 לָמָּה? וְאִם נָתְנוּ לָהֶם דִּירָה נִפְרֶדֶת – לְכָל הַמִּנְהָגִים מַדְלִיקִים שָׁם בִּבְרָכָה, וּמִשְׁתַּדְּלִים שֶׁיִּרְאוּ אֶת הַנֵּרוֹת מֵהָרְחוֹב.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הַסְּפָרַדִּים נוֹתְנִים לְבַעַל הַבַּיִת פְּרוּטָה (מַטְבֵּעַ קָטָן) אוֹ שֶׁהוּא נוֹתֵן לָהֶם חֵלֶק בַּנֵּר בְּמַתָּנָה – וְכָךְ הֵם שֻׁתָּפִים בַּנֵּרוֹת וְיוֹצְאִים בְּהַדְלָקָתוֹ.
לְמִנְהַג הָאַשְׁכְּנַזִּים כָּל אֶחָד מַדְלִיק, וְלָכֵן גַּם הָאוֹרְחִים מַדְלִיקִים שָׁם בִּבְרָכָה.
(פֶּרֶק י״ג, הֲלָכָה ט׳, הֲלָכָה י׳)

15מוֹצָאֵי שַׁבָּת אֵצֶל סַבָּא

בְּמוֹצָאֵי שַׁבָּת: מִי שֶׁחוֹזֵר מַהֵר הַבַּיְתָה – מַדְלִיק בְּבֵיתוֹ. מִי שֶׁיַּחֲזֹר מְאֻחָר מְאוֹד, כְּשֶׁכְּבָר אֵין אֲנָשִׁים בָּרְחוֹב – מַדְלִיק אֵצֶל הַמְּאָרְחִים, כְּמוֹ בְּעֶרֶב שַׁבָּת. וּמִי שֶׁחוֹזֵר לֹא מַהֵר וְלֹא מְאֻחָר – רַשַּׁאי לִבְחֹר.
🤔 לָמָּה? מִצַּד הַיּוֹם שֶׁעָבַר הֵם עֲדַיִן שַׁיָּכִים לְבֵית הַמְּאָרְחִים, וּמִצַּד הַיּוֹם הַבָּא – לְבֵיתָם.
(פֶּרֶק י״ג, הֲלָכָה י׳)

16בַּחוּרֵי יְשִׁיבָה וּבְנוֹת מִכְלָלָה

בַּחוּרֵי יְשִׁיבָה שֶׁיְּשֵׁנִים בַּחֶדֶר וְאוֹכְלִים בַּחֲדַר הָאֹכֶל – מַדְלִיקִים בַּחֶדֶר, שֶׁהוּא מְיֻחָד לָהֶם. אִם יֵשׁ חַלּוֹן לָרְחוֹב – לְיָדוֹ, וְאִם לֹא – טוֹב בְּצַד שְׂמֹאל שֶׁל הַדֶּלֶת. וְכָךְ גַּם חַיָּלִים וּסְטוּדֶנְטִים, וְכָךְ בָּנוֹת שֶׁלּוֹמְדוֹת בְּמִכְלָלָה וְגָרוֹת בִּפְנִימִיָּה אוֹ בְּדִירָה שְׂכוּרָה. הַאִם הֵם מַדְלִיקִים בִּבְרָכָה?
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הָאַשְׁכְּנַזִּים וְחֵלֶק מֵהַסְּפָרַדִּים מַדְלִיקִים בַּחֶדֶר בִּבְרָכָה.
לְמִנְהַג רַבִּים מֵהַסְּפָרַדִּים סוֹמְכִים עַל הַהַדְלָקָה שֶׁל הַהוֹרִים בַּבַּיִת, כִּי בֵּית הַהוֹרִים הוּא עֲדַיִן בֵּיתָם. וְאִם אֵין שָׁם מִי שֶׁמַּדְלִיק – מַדְלִיקִים בְּלִי בְּרָכָה.
(פֶּרֶק י״ג, הֲלָכָה י״ג)

17בְּמָלוֹן

הַמִּתְאָרְחִים בְּמָלוֹן צְרִיכִים לְהַדְלִיק. מִשְׁפָּחָה שֶׁנּוֹסַעַת יַחַד – מַדְלִיקָה בִּבְרָכָה. הַנָּכוֹן לְהַדְלִיק בַּחֶדֶר, שֶׁהוּא מְיֻחָד לָהֶם, וְאִם יֵשׁ חַלּוֹן לָרְחוֹב – לְיָדוֹ. נִשְׁאָרִים בַּחֶדֶר לְפָחוֹת חֲצִי שָׁעָה, וּכְשֶׁיּוֹצְאִים – מְכַבִּים אֶת הַנֵּרוֹת. אִם הַמָּלוֹן לֹא מַרְשֶׁה לְהַדְלִיק בַּחֲדָרִים – מַדְלִיקִים בַּחֲדַר הָאֹכֶל אוֹ בַּלּוֹבִּי. וּמִי שֶׁהוֹרָיו מַדְלִיקִים בִּשְׁבִילוֹ בַּבַּיִת?
🤔 לָמָּה? מְכַבִּים כְּשֶׁיּוֹצְאִים מֵהַחֶדֶר – כְּדֵי שֶׁלֹּא תִּפְרֹץ שְׂרֵפָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הַסְּפָרַדִּים מַדְלִיק בְּלִי בְּרָכָה.
לְמִנְהַג הָאַשְׁכְּנַזִּים מַדְלִיק בִּבְרָכָה.
(פֶּרֶק י״ג, הֲלָכָה י״ד)

18בְּבֵית חוֹלִים

גַּם מִי שֶׁמְּאֻשְׁפָּז בְּבֵית חוֹלִים חַיָּב בְּנֵרוֹת חֲנֻכָּה. אֲבָל יֶלֶד שֶׁהוֹרָיו מַדְלִיקִים בַּבַּיִת – יוֹצֵא בְּהַדְלָקָתָם. הַאִם יַדְלִיק גַּם בְּבֵית הַחוֹלִים?
🤔 לָמָּה? מִי שֶׁלֹּא מַדְלִיק בְּעַצְמוֹ – מִשְׁתַּדֵּל לִרְאוֹת נֵרוֹת וְלִשְׁמֹעַ אֶת הַבְּרָכוֹת מִמִּי שֶׁמַּדְלִיק, כְּדֵי לְהִשְׁתַּתֵּף בַּתּוֹדָה עַל הַנֵּס.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הָאַשְׁכְּנַזִּים אִם הוּא רוֹצֶה לְהַדֵּר – מַדְלִיק בִּבְרָכָה.
לְמִנְהַג רֹב הַסְּפָרַדִּים לֹא מַדְלִיק.
💛 גַּם בְּמִטַּת בֵּית הַחוֹלִים – הָאוֹר שֶׁל חֲנֻכָּה מַגִּיעַ אֵלֶיךָ, וַה׳ אִתְּךָ.
(פֶּרֶק י״ג, הֲלָכָה ט״ו)

19בְּטִיּוּל, בְּאֹהֶל וּבָרַכֶּבֶת

נֶחְלְקוּ הַפּוֹסְקִים אִם מִי שֶׁאֵין לוֹ בַּיִת חַיָּב לְהַדְלִיק. לָכֵן מְטַיֵּל שֶׁיָּשֵׁן בַּשָּׂדֶה, אוֹ בְּאֹהֶל קָטָן מְאוֹד – מַדְלִיק בְּלִי בְּרָכָה. אֲבָל אֹהֶל גָּדוֹל נֶחְשָׁב כְּבַיִת, וּמַדְלִיקִים בִּבְרָכָה בְּפֶתַח הָאֹהֶל. מִי שֶׁנּוֹסֵעַ כָּל הַלַּיְלָה בְּרַכֶּבֶת, בְּמָטוֹס אוֹ בְּאֳנִיָּה – הַקָּרוֹן נֶחְשָׁב כְּבַיִת, וּמַדְלִיק בִּבְרָכָה, אִם מַרְשִׁים לוֹ. וְאִם מַרְשִׁים רַק לְאֶחָד – הוּא מַדְלִיק נֵר אֶחָד בִּשְׁבִיל כֻּלָּם.
🤔 לָמָּה? כְּשֶׁיֵּשׁ סָפֵק אִם צָרִיךְ לְבָרֵךְ – לֹא מְבָרְכִים.
💛 גַּם בְּאֶמְצַע הַשָּׂדֶה, תַּחַת הַכּוֹכָבִים – יְהוּדִי מַדְלִיק נֵר, כִּי הַנֵּס הוֹלֵךְ אִתּוֹ לְכָל מָקוֹם.
(פֶּרֶק י״ג, הֲלָכָה ט״ז)

📝 מָה לָמַדְנוּ

  • מְפַרְסְמִים אֶת הַנֵּס: מַדְלִיקִים בַּפֶּתַח מוּל הַמְּזוּזָה אוֹ בַּחַלּוֹן שֶׁפּוֹנֶה לָרְחוֹב – בְּבִנְיַן דִּירוֹת, הֲכִי טוֹב בַּחַלּוֹן אוֹ בַּמִּרְפֶּסֶת.
  • הַמִּנְהָג הָרוֹוֵחַ לְהַדְלִיק בְּצֵאת הַכּוֹכָבִים, וּמִשְׁתַּדְּלִים לֹא לְאַחֵר אַחֲרֵי הַשָּׁעָה תֵּשַׁע.
  • בְּעֶרֶב שַׁבָּת מַדְלִיקִים נֵרוֹת חֲנֻכָּה לִפְנֵי נֵרוֹת שַׁבָּת; בְּמוֹצָאֵי שַׁבָּת – אַחֲרֵי עַרְבִית.
  • אוֹרְחִים שֶׁיְּשֵׁנִים אֵצֶל הַמְּאָרְחִים – הַבַּיִת שֶׁלָּהֶם שָׁם; מִי שֶׁרַק אוֹכֵל שָׁם – מַדְלִיק בְּבֵיתוֹ.
  • בְּמָלוֹן, בִּישִׁיבָה, בְּבֵית חוֹלִים וּבְטִיּוּל – יֵשׁ לְכָל מָקוֹם דִּין, וּפְעָמִים רַבּוֹת סְפָרַדִּים וְאַשְׁכְּנַזִּים נוֹהֲגִים אַחֶרֶת.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה מַדְלִיקִים בְּצַד שְׂמֹאל שֶׁל הַפֶּתַח, וְלָמָּה לֹא כְּדַאי לָקַחַת לַחַלּוֹן חֲנֻכִּיָּה עִם גַּב?
  2. בְּמוֹצָאֵי שַׁבַּת חֲנֻכָּה – מָה עוֹשִׂים קֹדֶם, הַבְדָּלָה אוֹ נֵר חֲנֻכָּה? הַסְבֵּר אֶת שְׁנֵי הַמִּנְהָגִים.
  3. הַמִּשְׁפָּחָה שֶׁלְּךָ אוֹכֶלֶת אֲרוּחַת עֶרֶב אֵצֶל הַדּוֹדִים וְחוֹזֶרֶת לִישֹׁן בַּבַּיִת. אֵיפֹה וּמָתַי תַּדְלִיקוּ?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״אֵיפֹה וּמָתַי מַדְלִיקִים״

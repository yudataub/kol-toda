סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת סֻכּוֹת
מַהֲדוּרַת יְלָדִים

🍋 אַרְבַּעַת הַמִּינִים

פֶּרֶק ד׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
עֶרֶב סֻכּוֹת בַּשּׁוּק: שׁוּלְחָנוֹת מְלֵאִים קֻפְסָאוֹת שֶׁל אֶתְרוֹגִים, עֲרֵמוֹת שֶׁל לוּלָבִים, וַאֲנָשִׁים עִם זְכוּכִית מַגְדֶּלֶת שֶׁבּוֹדְקִים כָּל נְקֻדָּה. מָה בֶּאֱמֶת חָשׁוּב לִבְדֹּק? וּמָה מְלַמְּדִים אוֹתָנוּ אַרְבָּעָה צְמָחִים שֶׁאוֹחֲזִים יַחַד בַּיָּד? בַּפֶּרֶק הַזֶּה נִלְמַד לִבְחֹר אַרְבָּעָה מִינִים כְּשֵׁרִים וִיפִים – וּבְלִי לְחַץ.

1מָה הֵם אַרְבַּעַת הַמִּינִים?

הַתּוֹרָה אוֹמֶרֶת: ״וּלְקַחְתֶּם לָכֶם בַּיּוֹם הָרִאשׁוֹן פְּרִי עֵץ הָדָר, כַּפֹּת תְּמָרִים, וַעֲנַף עֵץ עָבֹת, וְעַרְבֵי נָחַל״. ״פְּרִי עֵץ הָדָר״ – אֶתְרוֹג; ״כַּפֹּת תְּמָרִים״ – לוּלָב; ״עֲנַף עֵץ עָבֹת״ – הֲדַס; ״עַרְבֵי נָחַל״ – עֲרָבָה. נוֹטְלִים אֶתְרוֹג אֶחָד, לוּלָב אֶחָד, שְׁלֹשָׁה הֲדַסִּים וּשְׁתֵּי עֲרָבוֹת. וּמְבָרְכִים ״עַל נְטִילַת לוּלָב״.
🤔 לָמָּה? הַלּוּלָב הוּא הַגָּבוֹהַּ שֶׁבְּאַרְבַּעַת הַמִּינִים, וְלָכֵן הַמִּצְוָה כֻּלָּהּ נִקְרֵאת עַל שְׁמוֹ.
(פֶּרֶק ד׳, הֲלָכָה א׳)

2שִׁבְעָה יָמִים – זֵכֶר לַמִּקְדָּשׁ

מִן הַתּוֹרָה נוֹטְלִים לוּלָב רַק בַּיּוֹם הָרִאשׁוֹן, וְרַק בַּמִּקְדָּשׁ נוֹטְלִים שִׁבְעָה יָמִים. אַחֲרֵי שֶׁבֵּית הַמִּקְדָּשׁ נֶחֱרַב, תִּקֵּן רַבָּן יוֹחָנָן בֶּן זַכַּאי שֶׁבְּכָל מָקוֹם יִטְּלוּ לוּלָב שִׁבְעָה יָמִים – זֵכֶר לַמִּקְדָּשׁ.
🤔 לָמָּה? חֲכָמִים אָמְרוּ שֶׁכְּשֶׁזּוֹכְרִים אֶת הַמִּקְדָּשׁ וּמַמְשִׁיכִים אֶת הַמִּצְווֹת שֶׁעָשׂוּ בּוֹ, מְרַפְּאִים אֶת הַכְּאֵב שֶׁל הַחֻרְבָּן וּמְקָרְבִים אֶת הַגְּאֻלָּה.
💛 כָּל נִעְנוּעַ בְּחֹל הַמּוֹעֵד הוּא כְּמוֹ גַּעְגּוּעַ קָטָן לְבֵית ה׳ – וּתְפִלָּה שֶׁנַּחֲזֹר אֵלָיו.
(פֶּרֶק ד׳, הֲלָכָה א׳)

3כֻּלָּם יַחַד

אַרְבַּעַת הַמִּינִים מְעַכְּבִים זֶה אֶת זֶה: אִם חָסֵר מִין אֶחָד – אֵין מְקַיְּמִים אֶת הַמִּצְוָה בִּשְׁלֹשֶׁת הָאֲחֵרִים. לְכַתְּחִלָּה אוֹגְדִים אֶת הַלּוּלָב עִם הַהֲדַסִּים וְהָעֲרָבוֹת, וְנוֹטְלִים אֶת כֻּלָּם יַחַד. חֲכָמִים אָמְרוּ שֶׁהֵם כְּמוֹ אַרְבָּעָה סוּגִים שֶׁל יְהוּדִים: הָאֶתְרוֹג – יֵשׁ בּוֹ טַעַם וְרֵיחַ, כְּמוֹ מִי שֶׁיֵּשׁ בּוֹ תּוֹרָה וּמַעֲשִׂים טוֹבִים. הַלּוּלָב – טַעַם בְּלִי רֵיחַ, כְּמוֹ מִי שֶׁיֵּשׁ בּוֹ תּוֹרָה. הַהֲדַס – רֵיחַ בְּלִי טַעַם, כְּמוֹ מִי שֶׁיֵּשׁ בּוֹ מַעֲשִׂים טוֹבִים. וְהָעֲרָבָה – לֹא טַעַם וְלֹא רֵיחַ.
🤔 לָמָּה? מָה עוֹשֶׂה ה׳ עִם הַיְּהוּדִים הַפְּשׁוּטִים? ״לְאַבְּדָם אִי אֶפְשָׁר! אֶלָּא יִקָּשְׁרוּ כֻּלָּם אֲגֻדָּה אַחַת, וְהֵם מְכַפְּרִים אֵלּוּ עַל אֵלּוּ״. בְּלִי הָעֲרָבָה הַפְּשׁוּטָה – גַּם הָאֶתְרוֹג הַמְּהֻדָּר אֵינוֹ כָּשֵׁר.
💛 ה׳ רוֹצֶה אֶת כֻּלָּנוּ יַחַד בַּאֲגֻדָּה אַחַת – אַף אֶחָד לֹא מְיֻתָּר אֶצְלוֹ.
(פֶּרֶק ד׳, הֲלָכָה ב׳)

4הַלּוּלָב – לֵב אֶחָד

הַלּוּלָב רוֹמֵז לַתּוֹרָה וּלְתַלְמִידֵי הַחֲכָמִים. עָלָיו צוֹמְחִים לִשְׁנֵי צְדָדִים, אֲבָל כֻּלָּם צְמוּדִים לְשִׁדְרָה אַחַת, וְהוּא יָשָׁר וּמְכֻוָּן לְמַטָּרָה אַחַת. אָמְרוּ חֲכָמִים: ״מָה תָּמָר זֶה אֵין לוֹ אֶלָּא לֵב אֶחָד, אַף יִשְׂרָאֵל אֵין לָהֶם אֶלָּא לֵב אֶחָד לַאֲבִיהֶם שֶׁבַּשָּׁמַיִם״.
🤔 לָמָּה? בַּתּוֹרָה יֵשׁ דֵּעוֹת שׁוֹנוֹת וּמַחֲלֹקוֹת, אֲבָל הַשֹּׁרֶשׁ שֶׁל כֻּלָּן אֶחָד. כְּמוֹ עָלֵי הַלּוּלָב – פּוֹנִים לְכִוּוּנִים שׁוֹנִים, וּמְחֻבָּרִים לְשִׁדְרָה אַחַת.
(פֶּרֶק ד׳, הֲלָכָה ג׳)

5הֲדַס וַעֲרָבָה

הַהֲדַס רוֹמֵז לְמִצְווֹת וּמַעֲשִׂים טוֹבִים: כְּמוֹ שֶׁהָרֵיחַ הַטּוֹב מִתְפַּשֵּׁט סָבִיב – כָּךְ הַהַשְׁפָּעָה שֶׁל מַעֲשֶׂה טוֹב. הָעֲרָבָה – אֵין בָּהּ טַעַם וְרֵיחַ, אֲבָל יֵשׁ בָּהּ כֹּחַ צְמִיחָה עָצוּם וְחִיּוּת. מִן הַחִיּוּת שֶׁל הַיְּהוּדִים הַפְּשׁוּטִים מִתְחַזְּקִים גַּם הַחֲכָמִים, וְהַרְבֵּה פְּעָמִים דַּוְקָא מִמִּשְׁפָּחוֹת פְּשׁוּטוֹת צָמְחוּ גְּדוֹלֵי יִשְׂרָאֵל.
🤔 לָמָּה? רַק כְּשֶׁכָּל הַכֹּחוֹת מִתְאַחֲדִים – עַם יִשְׂרָאֵל יָכוֹל לְתַקֵּן אֶת הָעוֹלָם וּלְהֵיטִיב לְכָל הַבְּרוּאִים.
(פֶּרֶק ד׳, הֲלָכָה ג׳)

6לֹא צָרִיךְ לְהִלָּחֵץ

מִין נִפְסָל רַק כְּשֶׁיֵּשׁ בּוֹ שִׁנּוּי מַשְׁמָעוּתִי: כְּשֶׁהוּא מִין אַחֵר (לִימוֹן בִּמְקוֹם אֶתְרוֹג), כְּשֶׁצּוּרָתוֹ הִשְׁתַּנְּתָה מְאוֹד, כְּשֶׁהוּא קָטָן מִדַּי, כְּשֶׁהוּא יָבֵשׁ לְגַמְרֵי, אוֹ כְּשֶׁחָסֵר מִמֶּנּוּ חֵלֶק. רֹב הַמִּינִים בַּשּׁוּק כְּשֵׁרִים! נוֹהֲגִים לְהַדֵּר וְלִבְחֹר מִינִים יָפִים – אֲבָל אֵין צֹרֶךְ לִהְיוֹת בְּמֶתַח. וְגַם מִין גָּזוּל פָּסוּל.
🤔 לָמָּה? חִסָּרוֹן קָטָן פּוֹסֵל רַק בַּיּוֹם הָרִאשׁוֹן, שֶׁבּוֹ הַמִּינִים צְרִיכִים לִהְיוֹת שְׁלֵמִים. וּלְרֹב הַפּוֹסְקִים גַּם מִין יָבֵשׁ, שֶׁאֵין בּוֹ ״הָדָר״, פָּסוּל רַק בַּיּוֹם הָרִאשׁוֹן.
💛 ה׳ שָׂמֵחַ בַּמִּצְוָה שֶׁלְּךָ – לֹא בִּזְכוּכִית הַמַּגְדֶּלֶת.
(פֶּרֶק ד׳, הֲלָכָה ד׳)

7לוּלָב כָּשֵׁר

עָלֵי הַלּוּלָב צוֹמְחִים מִשְּׁנֵי צְדָדָיו וּמְכַסִּים אֶת הַשִּׁדְרָה. לְכַתְּחִלָּה הֵם צְמוּדִים וּסְגוּרִים, כָּךְ שֶׁאִם מַנִּיחִים אֶת הַלּוּלָב עַל הַשֻּׁלְחָן – הֵם נִשְׁאָרִים צְמוּדִים. לוּלָב שֶׁהִתְעַקֵּם מְאוֹד כְּמוֹ מַגָּל – פָּסוּל; לוּלָב שֶׁרֹב עָלָיו הִתְיַבְּשׁוּ וְהִלְבִּינוּ לְגַמְרֵי – פָּסוּל. אֹרֶךְ הַשִּׁדְרָה – לְפָחוֹת אַרְבָּעָה טְפָחִים (כְּ־32 ס״מ). וּמֵהִדּוּרֵי הַלּוּלָב – שֶׁיִּהְיֶה יָשָׁר וְגָבוֹהַּ.
🤔 לָמָּה? לוּלָב שֶׁעָלָיו נִפְתְּחוּ וְהִתְקַשּׁוּ כְּבָר אֵינוֹ לוּלָב – הוּא נַעֲשָׂה ״חֲרָיָה״, עֲנַף דֶּקֶל פָּתוּחַ כְּמוֹ מְנִיפָה.
(פֶּרֶק ד׳, הֲלָכָה ה׳, הֲלָכָה ז׳)

8הַתְּיוֹמֶת – הֶעָלֶה הָאֶמְצָעִי

כָּל עָלֶה בַּלּוּלָב בָּנוּי מִשְּׁנֵי עָלִים שֶׁמְּחֻבָּרִים בַּגַּב, כְּמוֹ תְּאוֹמִים – וְזוֹ ״הַתְּיוֹמֶת״. אִם נִפְתְּחָה רֹב הַתְּיוֹמֶת שֶׁל הֶעָלֶה הָאֶמְצָעִי הָעֶלְיוֹן – לֹא נוֹטְלִים אֶת הַלּוּלָב בַּיּוֹם הָרִאשׁוֹן, אֲבָל בִּשְׁאָר הַיָּמִים הוּא כָּשֵׁר. כִּמְעַט כָּל הַלּוּלָבִים כְּשֵׁרִים גַּם לַיּוֹם הָרִאשׁוֹן. מִי שֶׁחוֹשֵׁשׁ שֶׁהֶעָלֶה יַמְשִׁיךְ לְהִפָּתַח – יָכוֹל לְהַדְבִּיק אוֹתוֹ.
🤔 לָמָּה? בַּיּוֹם הָרִאשׁוֹן הַמִּינִים צְרִיכִים לִהְיוֹת שְׁלֵמִים, וְהַתְּיוֹמֶת הַפְּתוּחָה הִיא כְּמוֹ חִסָּרוֹן. וְעָדִיף לָקַחַת לוּלָב גָּדוֹל וְיָפֶה שֶׁהֶעָלֶה הָעֶלְיוֹן שֶׁלּוֹ פָּתוּחַ קְצָת, מֵאֲשֶׁר לוּלָב קָטָן וְצָמוּק שֶׁהֶעָלֶה שֶׁלּוֹ סָגוּר.
(פֶּרֶק ד׳, הֲלָכָה ו׳)

9לוּלָב קָנָרִי

יֵשׁ זַן מְיֻחָד שֶׁל דֶּקֶל שֶׁהוּבָא מֵהָאִיִּים הַקָּנָרִיִּים. הָעָלִים שֶׁלּוֹ קְצָרִים, צְפוּפִים וְרַכִּים, הַשִּׁדְרָה שֶׁלּוֹ רַכָּה וְנוֹטָה, וְהַתְּמָרִים שֶׁלּוֹ אֵינָם טְעִימִים. לְמַעֲשֶׂה, כִּי הַדָּבָר שָׁנוּי בְּמַחֲלֹקֶת, נָכוֹן לְהִמָּנַע מִלּוּלָב קָנָרִי, וּבִשְׁעַת הַדְּחָק אֶפְשָׁר לִסְמֹךְ עַל הַמַּכְשִׁירִים וּלְבָרֵךְ עָלָיו.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ מַכְשִׁירִים כִּי הוּא דֶּקֶל שֶׁמַּצְמִיחַ תְּמָרִים, וְכָל הַשִּׁנּוּיִים אֵינָם מוֹצִיאִים אוֹתוֹ מִן הַמִּין.
וְיֵשׁ פּוֹסְלִים כִּי הַפֵּרוֹת שֶׁלּוֹ כִּמְעַט אֵינָם רְאוּיִים לַאֲכִילָה, וְיֵשׁ בּוֹ הַרְבֵּה שִׁנּוּיִים מִדֶּקֶל רָגִיל – וְאֵין זֶה ״כַּפֹּת תְּמָרִים״.
(פֶּרֶק ד׳, הֲלָכָה ז׳)

10הֲדַס – שְׁלֹשָׁה עָלִים בְּשׁוּרָה

״עֲנַף עֵץ עָבֹת״ הוּא הֲדַס שֶׁעָלָיו גְּדֵלִים בִּשְׁלָשׁוֹת – שְׁלֹשָׁה עָלִים בְּאוֹתוֹ גֹּבַהּ, כְּמוֹ קְלִיעָה. אִם שְׁנַיִם יוֹצְאִים יַחַד וְהַשְּׁלִישִׁי לְמַעְלָה אוֹ לְמַטָּה – זֶה ״הֲדַס שׁוֹטֶה״, וְהוּא פָּסוּל. אֹרֶךְ כָּל הֲדַס – לְפָחוֹת שְׁלֹשָׁה טְפָחִים (כְּ־24 ס״מ). הֲדַס שֶׁעָלָיו הִתְיַבְּשׁוּ לְגַמְרֵי, עַד שֶׁהֵם נִפְרָכִים בַּצִּפֹּרֶן וּלְבָנִים – פָּסוּל. וְאִם יֵשׁ עָלָיו גַּרְגְּרִים אֲדֻמִּים אוֹ שְׁחֹרִים, יוֹתֵר מִן הֶעָלִים – פָּסוּל, עַד שֶׁמּוֹרִידִים אוֹתָם (אֲבָל לֹא בְּיוֹם טוֹב).
🤔 לָמָּה? אֵין צֹרֶךְ לְדַקְדֵּק בְּמִלִּימֶטְרִים: כָּל שֶׁבְּעֵינֵי בְּנֵי אָדָם הָעָלִים נִרְאִים כִּשְׁלִישִׁיָּה בְּגֹבַהּ אֶחָד – הֵם מְשֻׁלָּשִׁים.
(פֶּרֶק ד׳, הֲלָכָה ח׳)

11עֲרָבָה – וְלֹא צַפְצָפָה

לָעֲרָבָה שְׁלֹשָׁה סִימָנִים: הֶעָלֶה מָשׁוּךְ לְאָרְכּוֹ כְּמוֹ נַחַל, שְׂפַת הֶעָלֶה חֲלָקָה, וְהֶעָנָף אָדֹם קְצָת. יֵשׁ צֶמַח דּוֹמֶה – צַפְצָפָה – שֶׁבִּשְׂפַת עָלֶיהָ יֵשׁ שִׁנַּיִם כְּמוֹ מַסּוֹר, וְהָעָנָף שֶׁלָּהּ יָרֹק – וְהִיא פְּסוּלָה. אֹרֶךְ כָּל עֲרָבָה – לְפָחוֹת שְׁלֹשָׁה טְפָחִים. עֲרָבָה שֶׁרֹב עָלֶיהָ נָשְׁרוּ, אוֹ שֶׁהִתְיַבְּשׁוּ וְהִלְבִּינוּ, אוֹ שֶׁנִּקְטַם הָרֹאשׁ שֶׁלָּהּ – פְּסוּלָה.
🤔 לָמָּה? הָעֲרָבָה מְלֵאָה חִיּוּת וּגְדֵלָה לְיַד מַיִם – אֲבָל בְּלִי מַיִם הִיא מִתְיַבֶּשֶׁת מַהֵר. לָכֵן שׁוֹמְרִים עָלֶיהָ, וְנִזְהָרִים כְּשֶׁמַּכְנִיסִים אוֹתָהּ לָאֲגֻדָּה שֶׁלֹּא יִתָּלְשׁוּ הֶעָלִים.
(פֶּרֶק ד׳, הֲלָכָה ט׳)

12אֶתְרוֹג – לֹא לִימוֹן וְלֹא מֻרְכָּב

״פְּרִי עֵץ הָדָר״ הוּא אֶתְרוֹג, וְזֶה עָבַר בַּמָּסֹרֶת מִדּוֹר לְדוֹר. יֵשׁ אֶתְרוֹגִים גְּדוֹלִים וּקְטַנִּים, צְהֻבִּים וִירַקְרַקִּים – וְכֻלָּם כְּשֵׁרִים. אֲבָל אֶתְרוֹג שֶׁגָּדַל עַל עֵץ מֻרְכָּב – עַל לִימוֹן אוֹ עַל עֵץ אַחֵר – פָּסוּל. הַיּוֹם הַמְּגַדְּלִים שׁוֹתְלִים עֵצִים שֶׁאֵינָם מֻרְכָּבִים, וְאֶפְשָׁר לִסְמֹךְ עַל הַמּוֹכְרִים. וְאֶתְרוֹג בְּלִי פִּטָּם – כָּשֵׁר לְכַתְּחִלָּה, כִּי אֵצֶל רֹב הָאֶתְרוֹגִים הַפִּטָּם נוֹשֵׁר כְּשֶׁהֵם עוֹד קְטַנִּים.
🤔 לָמָּה? אֶתְרוֹג מֻרְכָּב הוּא כְּמוֹ פְּרִי חָדָשׁ, תַּעֲרֹבֶת שֶׁל שְׁנֵי פֵּרוֹת – וְהַתּוֹרָה צִוְּתָה לִטֹּל אֶתְרוֹג.
(פֶּרֶק ד׳, הֲלָכָה י׳)

13כְּתָמִים וְחוֹרִים

אֶתְרוֹג שֶׁנִּקַּב וְנֶחְסַר מִמֶּנּוּ חֵלֶק – פָּסוּל בַּיּוֹם הָרִאשׁוֹן, וְכָשֵׁר בִּשְׁאָר הַיָּמִים. כְּתָמִים רְגִילִים, צְהֻבִּים, אֲפֹרִים אוֹ חוּמִים – אֵינָם פּוֹסְלִים, כִּי כָּךְ טִבְעָם שֶׁל אֶתְרוֹגִים. רַק כֶּתֶם שָׁחֹר אוֹ לָבָן מְשֻׁנֶּה, שֶׁמְּכַסֶּה אֶת רֹב הָאֶתְרוֹג, אוֹ כֶּתֶם בּוֹלֵט עַל הַ״חֹטֶם״ – הַחֵלֶק הָעֶלְיוֹן הַמְּשֻׁפָּע – פּוֹסֵל, וְזֶה נָדִיר מְאוֹד. שׁוֹמְרִים אֶת הָאֶתְרוֹג בְּפִשְׁתָּן אוֹ בְּשַׁרְווּל סְפוֹגִי, כִּי מַכָּה קַלָּה יְכוֹלָה לַעֲשׂוֹת בּוֹ כֶּתֶם.
🤔 לָמָּה? ״וּלְקַחְתֶּם״ – לְקִיחָה תַּמָּה, שְׁלֵמָה. בַּיּוֹם הָרִאשׁוֹן הָאֶתְרוֹג צָרִיךְ לִהְיוֹת שָׁלֵם. אֲבָל כְּכָל שֶׁיֵּשׁ יוֹתֵר כְּתָמִים, הָאֶתְרוֹג פָּחוֹת מְהֻדָּר.
(פֶּרֶק ד׳, הֲלָכָה י״א)

14אֶתְרוֹג שָׁחֹר, קָטָן אוֹ עָגֹל

אֶתְרוֹג שָׁחֹר – פָּסוּל. אֶתְרוֹג יָרֹק־שָׁחֹר שֶׁעוֹד לֹא הִבְשִׁיל – פָּסוּל, וְאֶתְרוֹג שֶׁהִצְהִיב עַד שֶׁנַּעֲשָׂה כָּתֹם – כָּשֵׁר. אֶתְרוֹג קָטָן מִנֶּפַח בֵּיצָה – פָּסוּל, וּלְכַתְּחִלָּה טוֹב שֶׁיִּהְיֶה בְּגֹדֶל רָגִיל. אֶתְרוֹג יָבֵשׁ לְגַמְרֵי – פָּסוּל. וְגַם אֶתְרוֹג שֶׁצּוּרָתוֹ מְשֻׁנָּה לְגַמְרֵי – עָגֹל כְּמוֹ כַּדּוּר, אוֹ שֶׁגִּדְּלוּ אוֹתוֹ בְּתַבְנִית מְרֻבַּעַת – פָּסוּל. אֲבָל אֶתְרוֹג עִם שְׁנֵי רָאשִׁים, כְּמוֹ תְּאוֹמִים – כָּשֵׁר.
🤔 לָמָּה? הָאֶתְרוֹג צָרִיךְ לִהְיוֹת פְּרִי שֶׁנִּגְמַר וְרָאוּי לַאֲכִילָה, וּבַצּוּרָה שֶׁל אֶתְרוֹג. לָכֵן גַּם אֶתְרוֹג שֶׁל עָרְלָה אוֹ שֶׁלֹּא הִפְרִישׁוּ מִמֶּנּוּ תְּרוּמוֹת וּמַעַשְׂרוֹת – פָּסוּל.
(פֶּרֶק ד׳, הֲלָכָה י״ב)

15״לָכֶם״ – בַּיּוֹם הָרִאשׁוֹן

״וּלְקַחְתֶּם לָכֶם בַּיּוֹם הָרִאשׁוֹן״ – בַּיּוֹם הָרִאשׁוֹן הַלּוּלָב צָרִיךְ לִהְיוֹת שֶׁל מִי שֶׁמְּקַיֵּם בּוֹ אֶת הַמִּצְוָה, וְלֹא שָׁאוּל. לָכֵן בַּעַל הַלּוּלָב נוֹתֵן אוֹתוֹ לַחֲבֵרוֹ ״בְּמַתָּנָה עַל מְנָת לְהַחֲזִיר״. בִּשְׁאָר הַיָּמִים אֶפְשָׁר לְקַיֵּם אֶת הַמִּצְוָה גַּם בְּלוּלָב שָׁאוּל. יֶלֶד שֶׁעוֹד לֹא הִגִּיעַ לְמִצְווֹת יָכוֹל לְקַבֵּל מַתָּנָה, אֲבָל לֹא לָתֵת מַתָּנָה – לָכֵן בַּיּוֹם הָרִאשׁוֹן נוֹתְנִים לוֹ אֶת הַלּוּלָב רַק אַחֲרֵי שֶׁכָּל הַגְּדוֹלִים קִיְּמוּ בּוֹ אֶת הַמִּצְוָה.
🤔 לָמָּה? אִם הַיֶּלֶד יְקַבֵּל אֶת הַלּוּלָב בְּמַתָּנָה, הוּא לֹא יוּכַל לְהַחֲזִיר אוֹתוֹ בְּמַתָּנָה – וְהַגְּדוֹלִים לֹא יוּכְלוּ לָצֵאת בּוֹ. וְאִם מִי שֶׁקִּבֵּל לֹא מַחְזִיר – הַמַּתָּנָה בְּטֵלָה מִן הַהַתְחָלָה.
(פֶּרֶק ד׳, הֲלָכָה י״ג)

16לוּלָב גָּזוּל – וְלוּלָב בְּבֵית הַכְּנֶסֶת

מִין גָּזוּל אוֹ גָּנוּב – פָּסוּל כָּל שִׁבְעַת הַיָּמִים, אֲפִלּוּ אִם הוּא יָפֶה וּמְהֻדָּר. מִי שֶׁאֵין לוֹ לוּלָב וְרָאָה לוּלָב בְּבֵית הַכְּנֶסֶת – מְבַקֵּשׁ רְשׁוּת מִבַּעַל הַלּוּלָב. וְאִם בַּעַל הַלּוּלָב אֵינוֹ שָׁם וְאִי אֶפְשָׁר לִשְׁאֹל אוֹתוֹ – מֻתָּר לִטֹּל אֶת הַלּוּלָב מִן הַיּוֹם הַשֵּׁנִי וָהָלְאָה, בְּלִי לְהוֹצִיא אוֹתוֹ מִמְּקוֹמוֹ, וְשׁוֹמְרִים עָלָיו מְאוֹד. אֲבָל אִם יָדוּעַ שֶׁבַּעַל הַלּוּלָב מַקְפִּיד עַל חֲפָצָיו – אָסוּר.
🤔 לָמָּה? מִצְוָה שֶׁבָּאָה מִתּוֹךְ עֲבֵרָה – אֵינָהּ מִצְוָה. אֲבָל חֲכָמִים הֶעֱרִיכוּ שֶׁסְּתָם אָדָם שָׂמֵחַ שֶׁמְּקַיְּמִים מִצְוָה בַּחֲפָצִים שֶׁלּוֹ.
(פֶּרֶק ד׳, הֲלָכָה י״ג)

17הִדּוּר מִצְוָה – עַד שְׁלִישׁ

״זֶה אֵלִי וְאַנְוֵהוּ״ – מִצְוָה לְהַדֵּר וּלְהוֹסִיף עַד שְׁלִישׁ מִמְּחִיר הַמִּצְוָה. לְמָשָׁל: אִם יֵשׁ סֶט בְּשִׁשִּׁים שְׁקָלִים, סֶט יָפֶה יוֹתֵר בִּשְׁמוֹנִים, וְסֶט יָפֶה עוֹד יוֹתֵר בְּמֵאָה – מִצְוַת הַהִדּוּר הִיא לִקְנוֹת אֶת הַסֶּט שֶׁל שְׁמוֹנִים. וּמִי שֶׁמּוֹסִיף יוֹתֵר – ה׳ יְשַׁלֵּם שְׂכָרוֹ, אִם זֶה לֹא עַל חֶשְׁבּוֹן מִצְווֹת חֲשׁוּבוֹת יוֹתֵר אוֹ צָרְכֵי הַבַּיִת. מִי שֶׁמַּצָּבוֹ דָּחוּק – אֵין עָלָיו מִצְוָה לְהוֹסִיף שְׁלִישׁ.
🤔 לָמָּה? ה׳ קִבֵּל אֶת הַקָּרְבָּן שֶׁל הֶבֶל, שֶׁהֵבִיא מִן הַמֻּבְחָר, וְלֹא אֶת הַקָּרְבָּן שֶׁל קַיִן, שֶׁקִּמֵּץ. וּמִי שֶׁמּוֹסִיף בְּבִגְדֵי חֹל, רָאוּי שֶׁיּוֹסִיף לְפָחוֹת כָּךְ גַּם בַּמִּצְווֹת.
💛 כְּשֶׁאֲנִי בּוֹחֵר אֶתְרוֹג יָפֶה, אֲנִי אוֹמֵר לַה׳: בִּשְׁבִילְךָ – אֶת הַטּוֹב בְּיוֹתֵר.
(פֶּרֶק ד׳, הֲלָכָה י״ד)

📝 מָה לָמַדְנוּ

  • נוֹטְלִים אֶתְרוֹג אֶחָד, לוּלָב אֶחָד, שְׁלֹשָׁה הֲדַסִּים וּשְׁתֵּי עֲרָבוֹת – וְכֻלָּם יַחַד, כְּמוֹ כָּל עַם יִשְׂרָאֵל.
  • מִין נִפְסָל רַק בְּשִׁנּוּי מַשְׁמָעוּתִי, וְרֹב הַמִּינִים בַּשּׁוּק כְּשֵׁרִים.
  • הֲדַס – שְׁלֹשָׁה עָלִים בְּגֹבַהּ אֶחָד; עֲרָבָה – עָלֶה חָלָק וְעָנָף אֲדַמְדַּם; אֶתְרוֹג – לֹא מֻרְכָּב וְלֹא קָטָן מִבֵּיצָה.
  • בַּיּוֹם הָרִאשׁוֹן הַמִּינִים צְרִיכִים לִהְיוֹת שְׁלֵמִים וְשֶׁלָּנוּ – ״מַתָּנָה עַל מְנָת לְהַחֲזִיר״.
  • מְהַדְּרִים בַּמִּצְוָה וּמוֹסִיפִים עַד שְׁלִישׁ מִן הַמְּחִיר.

❓ שְׁאֵלוֹת חֲזָרָה

  1. אֵיזֶה סוּג שֶׁל יְהוּדִי רוֹמֵז כָּל אֶחָד מֵאַרְבַּעַת הַמִּינִים, וְלָמָּה אוֹגְדִים אוֹתָם יַחַד?
  2. מָה הַהֶבְדֵּל בֵּין הֲדַס כָּשֵׁר לַ״הֲדַס שׁוֹטֶה״, וּבֵין עֲרָבָה לְצַפְצָפָה?
  3. אֵיךְ אֶפְשָׁר לָתֵת לְחָבֵר לְקַיֵּם מִצְוָה בַּלּוּלָב שֶׁלִּי בַּיּוֹם הָרִאשׁוֹן, וְלָמָּה נוֹתְנִים לַיֶּלֶד רַק בַּסּוֹף?

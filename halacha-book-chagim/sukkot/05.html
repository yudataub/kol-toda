סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת סֻכּוֹת
מַהֲדוּרַת יְלָדִים

🌴 נְטִילַת לוּלָב

פֶּרֶק ה׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
בֹּקֶר שֶׁל חַג בְּבֵית הַכְּנֶסֶת. מִכָּל צַד נִשְׁמָע רַחַשׁ שֶׁל עָלִים, וְרָאשֵׁי לוּלָבִים יְרֻקִּים עוֹלִים וְיוֹרְדִים כְּמוֹ יַעַר קָטָן שֶׁרוֹקֵד. בְּאֵיזוֹ יָד מַחְזִיקִים אֶת הָאֶתְרוֹג? לְאָן מְנַעְנְעִים קֹדֶם? וְלָמָּה מַקִּיפִים אֶת הַבִּימָה? בַּפֶּרֶק הַזֶּה נִלְמַד אֵיךְ נוֹטְלִים לוּלָב – צַעַד אַחַר צַעַד.

1מָתַי נוֹטְלִים?

נוֹטְלִים לוּלָב בַּיּוֹם וְלֹא בַּלַּיְלָה. נוֹהֲגִים לִטֹּל אוֹתוֹ בִּתְפִלַּת שַׁחֲרִית, וְלֹא לִפְנֵי הָנֵץ הַחַמָּה – הָרֶגַע שֶׁהַשֶּׁמֶשׁ מַתְחִילָה לְהֵרָאוֹת. מִי שֶׁלֹּא נָטַל בַּבֹּקֶר – נוֹטֵל אַחַר כָּךְ בְּמֶשֶׁךְ הַיּוֹם. אִם כְּבָר שָׁקְעָה הַחַמָּה – נוֹטֵל בְּלִי בְּרָכָה, וְאִם יָצְאוּ הַכּוֹכָבִים – הִפְסִיד אֶת הַמִּצְוָה. וּמִי שֶׁיּוֹצֵא לַדֶּרֶךְ מֻקְדָּם מְאוֹד – נוֹטֵל מִשֶּׁמַּתְחִיל הָאוֹר הָרִאשׁוֹן בַּמִּזְרָח.
(פֶּרֶק ה׳, הֲלָכָה א׳)

2יוֹם רִאשׁוֹן שֶׁחָל בְּשַׁבָּת

כְּשֶׁהַיּוֹם הָרִאשׁוֹן שֶׁל סֻכּוֹת חָל בְּשַׁבָּת – לֹא נוֹטְלִים לוּלָב, אֲפִלּוּ שֶׁזּוֹ מִצְוָה מִן הַתּוֹרָה. נוֹטְלִים רַק מִן הַיּוֹם הַשֵּׁנִי.
🤔 לָמָּה? חֲכָמִים חָשְׁשׁוּ שֶׁמָּא מִישֶׁהוּ יִשְׁכַּח וִיטַלְטֵל אֶת הַלּוּלָב בִּרְשׁוּת הָרַבִּים וִיחַלֵּל שַׁבָּת. וּכְדֵי שֶׁכָּל יִשְׂרָאֵל יִנְהֲגוּ מִנְהָג אֶחָד – גַּם בְּאֶרֶץ יִשְׂרָאֵל לֹא נוֹטְלִים. הַשַּׁבָּת הִיא יְסוֹד הַחַיִּים שֶׁל עַם יִשְׂרָאֵל, וּקְדֻשָּׁתָהּ בָּאָה בִּמְקוֹם הַלּוּלָב.
(פֶּרֶק ה׳, הֲלָכָה א׳)

3אוֹגְדִים אֶת הַלּוּלָב

מִצְוָה לֶאֱגֹד יַחַד אֶת הַלּוּלָב, הַהֲדַסִּים וְהָעֲרָבוֹת – וְהָאֶתְרוֹג נִשְׁאָר לְבַד. אֶפְשָׁר לֶאֱגֹד בְּכָל חוּט אוֹ רְצוּעָה, וְרַבִּים אוֹגְדִים בְּעָלֵי לוּלָב. שִׁדְרַת הַלּוּלָב צְרִיכָה לִהְיוֹת גְּבוֹהָה טֶפַח מֵעַל הַהֲדַסִּים וְהָעֲרָבוֹת. לֹא נוֹטְלִים יוֹתֵר מִלּוּלָב אֶחָד וְאֶתְרוֹג אֶחָד, וְאָסוּר לְהוֹסִיף מִין חֲמִישִׁי – זֶה אִסּוּר ״בַּל תּוֹסִיף״. יֵשׁ מְהַדְּרִים שֶׁהַהֲדַסִּים יִהְיוּ גְּבוֹהִים מְעַט מִן הָעֲרָבוֹת.
🤔 לָמָּה? בַּפָּסוּק כָּתוּב: ״פְּרִי עֵץ הָדָר, כַּפֹּת תְּמָרִים וַעֲנַף עֵץ עָבֹת וְעַרְבֵי נָחַל״ – הָאוֹת ו׳ מְחַבֶּרֶת אֶת שְׁלֹשֶׁת הַמִּינִים, וְהָאֶתְרוֹג עוֹמֵד לְבַד.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ נוֹהֲגִים לִקְשֹׁר אֶת שְׁלֹשֶׁת הַהֲדַסִּים מִיָּמִין לַלּוּלָב, וְאֶת הָעֲרָבוֹת מִשְּׂמֹאל.
וְיֵשׁ נוֹהֲגִים, עַל פִּי הָאֲרִ״י לִקְשֹׁר הֲדַס אֶחָד מִיָּמִין, אֶחָד מִשְּׂמֹאל וְאֶחָד בָּאֶמְצַע, וַעֲרָבָה אַחַת מִיָּמִין וְאַחַת מִשְּׂמֹאל. וּשְׁנֵי הָאֳפָנִים מְהֻדָּרִים.
(פֶּרֶק ה׳, הֲלָכָה ב׳)

4אֵיךְ אוֹחֲזִים?

בְּיָד יָמִין אוֹחֲזִים אֶת הַלּוּלָב עִם הַהֲדַסִּים וְהָעֲרָבוֹת, וּבְיָד שְׂמֹאל – אֶת הָאֶתְרוֹג, וּמַצְמִידִים אוֹתָם זֶה לָזֶה. אוֹחֲזִים אֶת הַמִּינִים כְּדֶרֶךְ גְּדִילָתָם: הַחֵלֶק הַתַּחְתּוֹן לְמַטָּה וְהָרֹאשׁ לְמַעְלָה; וְהָאֶתְרוֹג – הָעֹקֶץ לְמַטָּה וְהַחֹטֶם לְמַעְלָה. שִׁדְרַת הַלּוּלָב פּוֹנָה אֶל הַפָּנִים שֶׁל הַנּוֹטֵל. מִי שֶׁהֶחֱלִיף אֶת הַיָּדַיִם – יָצָא. אֲבָל מִי שֶׁהָפַךְ אֶת אֶחָד הַמִּינִים – לֹא יָצָא. נוֹטְלִים בַּיָּדַיִם עַצְמָן, וְלֹא בְּתוֹךְ כְּלִי. טַבַּעַת עַל הָאֶצְבַּע אֵינָהּ מַפְרִיעָה.
🤔 לָמָּה? שְׁלֹשֶׁת הַמִּינִים הָאֲגוּדִים חֲשׁוּבִים יוֹתֵר מִן הָאֶתְרוֹג, וְלָכֵן נוֹטְלִים אוֹתָם בְּיָד יָמִין, הַחֲשׁוּבָה וְהַחֲזָקָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְפִי הַשֻּׁלְחָן עָרוּךְ גַּם אִטֵּר יָד – מִי שֶׁעוֹשֶׂה הַכֹּל בְּיָד שְׂמֹאל – נוֹטֵל אֶת הַלּוּלָב בְּיָמִין, כְּמוֹ כֻּלָּם, כִּי הַיָּמִין רוֹמֶזֶת לְמִדַּת הַחֶסֶד. וְכָךְ נוֹהֲגִים עַל פִּי הַקַּבָּלָה.
לְפִי הָרַמָּ״א אִטֵּר יָד נוֹטֵל אֶת הַלּוּלָב בְּיָד שְׂמֹאל – הַיָּד הַחֲזָקָה וְהַחֲשׁוּבָה שֶׁלּוֹ – וְאֶת הָאֶתְרוֹג בְּיָמִין.
(פֶּרֶק ה׳, הֲלָכָה ג׳)

5הַבְּרָכָה

מְבָרְכִים: ״בָּרוּךְ אַתָּה ה׳ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם אֲשֶׁר קִדְּשָׁנוּ בְּמִצְוֹתָיו וְצִוָּנוּ עַל נְטִילַת לוּלָב״. בַּיּוֹם הָרִאשׁוֹן מוֹסִיפִים ״שֶׁהֶחֱיָנוּ״. כְּדֵי שֶׁהַבְּרָכָה תִּהְיֶה לִפְנֵי הַמִּצְוָה, אוֹחֲזִים בִּזְמַן הַבְּרָכָה אֶת הָאֶתְרוֹג הָפוּךְ – הָעֹקֶץ לְמַעְלָה – וּמִיָּד אַחֲרֶיהָ הוֹפְכִים אוֹתוֹ וּמְנַעְנְעִים. וְיֵשׁ נוֹהֲגִים לְהַחְזִיק בִּזְמַן הַבְּרָכָה רַק אֶת הַלּוּלָב, וְלָקַחַת אֶת הָאֶתְרוֹג מִיָּד אַחֲרֶיהָ. עוֹמְדִים בִּזְמַן הַבְּרָכָה. נוֹהֲגִים לְבָרֵךְ לִפְנֵי הַהַלֵּל, וְיֵשׁ מְהַדְּרִים לְבָרֵךְ בַּסֻּכָּה לִפְנֵי שֶׁהוֹלְכִים לְבֵית הַכְּנֶסֶת.
🤔 לָמָּה? בָּרֶגַע שֶׁמַּחְזִיקִים אֶת כָּל הַמִּינִים כְּדֶרֶךְ גְּדִילָתָם – כְּבָר מְקַיְּמִים אֶת הַמִּצְוָה. וּמְבָרְכִים עַל מִצְוָה לִפְנֵי שֶׁעוֹשִׂים אוֹתָהּ. וּמִי שֶׁמְּבָרֵךְ בַּסֻּכָּה מְחַבֵּר אֶת מִצְוַת הַלּוּלָב עִם מִצְוַת הַסֻּכָּה.
(פֶּרֶק ה׳, הֲלָכָה ג׳)

6הַנִּעְנוּעִים

בְּעֶצֶם זֶה שֶׁמַּגְבִּיהִים אֶת הַמִּינִים – כְּבָר מְקַיְּמִים אֶת הַמִּצְוָה. וַחֲכָמִים תִּקְּנוּ לְנַעְנֵעַ, כְּמוֹ שֶׁהֵנִיפוּ קָרְבָּנוֹת בַּמִּקְדָּשׁ. מַצְמִידִים אֶת הָאֶתְרוֹג לַלּוּלָב, אוֹחֲזִים אוֹתָם קָרוֹב לַגּוּף, מוֹלִיכִים אוֹתָם אֶל הַכִּוּוּן – וּמַחֲזִירִים אֶל הַגּוּף. כָּךְ שָׁלֹשׁ פְּעָמִים לְכָל אֶחָד מִשִּׁשָּׁה כִּוּוּנִים: אַרְבַּע רוּחוֹת, לְמַעְלָה וּלְמַטָּה. גַּם כְּשֶׁמְּנַעְנְעִים לְמַטָּה – רֹאשׁ הַלּוּלָב נִשְׁאָר לְמַעְלָה.
🤔 לָמָּה? ״מוֹלִיךְ וּמֵבִיא – לְמִי שֶׁאַרְבַּע הָרוּחוֹת שֶׁלּוֹ; מַעֲלֶה וּמוֹרִיד – לְמִי שֶׁהַשָּׁמַיִם וְהָאָרֶץ שֶׁלּוֹ״. וְעוֹד: כְּדֵי לַעֲצֹר רוּחוֹת רָעוֹת וּטְלָלִים רָעִים. וְגַם – דֶּגֶל שֶׁל נִצָּחוֹן אַחֲרֵי יְמֵי הַדִּין.
💛 כְּשֶׁאֲנִי מְנַעְנֵעַ לְכָל הַכִּוּוּנִים, אֲנִי אוֹמֵר בְּלִי מִלִּים: ה׳, אַתָּה נִמְצָא בְּכָל מָקוֹם, וְאֵין מָקוֹם שֶׁאֵינְךָ אִתִּי.
(פֶּרֶק ה׳, הֲלָכָה ד׳)

7לְאָן מְנַעְנְעִים קֹדֶם?

יֵשׁ כַּמָּה מִנְהָגִים בְּסֵדֶר הַכִּוּוּנִים. יֵשׁ שֶׁפּוֹנִים בְּכָל נִעְנוּעַ אֶל הַצַּד שֶׁאֵלָיו מְנַעְנְעִים, וְיֵשׁ שֶׁעוֹמְדִים כָּל הַזְּמַן מוּל הַמִּזְרָח וּמַטִּים אֶת הַלּוּלָב. כָּל הַמִּנְהָגִים טוֹבִים, וְנָכוֹן שֶׁכָּל אֶחָד יַמְשִׁיךְ בְּמִנְהַג אֲבוֹתָיו. וּמִי שֶׁאֵינוֹ יוֹדֵעַ מָה מִנְהַג אֲבוֹתָיו – נוֹהֵג כְּפִי שֶׁהוּא רוֹצֶה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְפִי הַשֻּׁלְחָן עָרוּךְ מַתְחִילִים בְּמִזְרָח, וּמִשָּׁם פּוֹנִים יָמִינָה: דָּרוֹם, מַעֲרָב וְצָפוֹן, וְאַחַר כָּךְ לְמַעְלָה וּלְמַטָּה.
לְפִי הָאֲרִ״י דָּרוֹם, צָפוֹן, מִזְרָח, לְמַעְלָה, לְמַטָּה, וּבַסּוֹף – מַעֲרָב.
(פֶּרֶק ה׳, הֲלָכָה ד׳)

8מְכַסְכְּסִים בַּלּוּלָב?

״לְכַסְכֵּס״ פֵּרוּשׁוֹ לְהַשְׁמִיעַ רַחַשׁ קַל בֶּעָלִים שֶׁבְּרֹאשׁ הַלּוּלָב בִּזְמַן הַנִּעְנוּעִים.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג אַשְׁכְּנַז מִשְׁתַּדְּלִים לְכַסְכֵּס בַּלּוּלָב, וְלָכֵן מַשְׁאִירִים אֶת הַחֵלֶק הָעֶלְיוֹן שֶׁלּוֹ בְּלִי קְשִׁירָה.
לְמִנְהַג הַסְּפָרַדִּים אֵין נוֹהֲגִים לְכַסְכֵּס בַּלּוּלָב.
(פֶּרֶק ה׳, הֲלָכָה ב׳, הֲלָכָה ד׳)

9נִעְנוּעִים בַּהַלֵּל

אוֹחֲזִים אֶת הַלּוּלָב בִּזְמַן אֲמִירַת הַהַלֵּל, וּמְנַעְנְעִים בַּפְּסוּקִים ״הוֹדוּ לַה׳ כִּי טוֹב כִּי לְעוֹלָם חַסְדּוֹ״ וְ״אָנָּא ה׳ הוֹשִׁיעָה נָּא״. נוֹהֲגִים לְחַלֵּק אֶת הַנִּעְנוּעִים לְפִי הַמִּלִּים: בְּ״הוֹדוּ״ – כִּוּוּן אֶחָד לְכָל מִלָּה, וּבְ״אָנָּא״ – שְׁנֵי כִּוּוּנִים לְכָל מִלָּה. וּכְשֶׁאוֹמְרִים אֶת שֵׁם ה׳ – עוֹצְרִים וּמְכַוְּנִים.
🤔 לָמָּה? כָּתוּב: ״אָז יְרַנְּנוּ עֲצֵי הַיָּעַר... הוֹדוּ לַה׳ כִּי טוֹב... וְאִמְרוּ הוֹשִׁיעֵנוּ... וְקַבְּצֵנוּ״ – הָעֵצִים מְרַנְּנִים כְּשֶׁהֵם זָזִים. בַּנִּעְנוּעִים יֵשׁ גַּם תְּפִלָּה לְקִבּוּץ גָּלֻיּוֹת מֵאַרְבַּע כַּנְפוֹת הָאָרֶץ, וְתוֹדָה עַל הַגְּשָׁמִים שֶׁל הַשָּׁנָה שֶׁעָבְרָה וּבַקָּשָׁה לְגִשְׁמֵי בְּרָכָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הַסְּפָרַדִּים וְחֵלֶק מֵהַחֲסִידִים עַל פִּי הַמְקֻבָּלִים מְנַעְנְעִים חָמֵשׁ פְּעָמִים: אַחֲרֵי הַבְּרָכָה, בְּ״הוֹדוּ״ הָרִאשׁוֹן, בִּשְׁנֵי הַ״אָנָּא ה׳ הוֹשִׁיעָה נָּא״, וּבְ״הוֹדוּ״ שֶׁבְּסוֹף הַהַלֵּל – הַחַזָּן וְהַקָּהָל כְּאֶחָד. וְעוֹלֵי תֵּימָן מְנַעְנְעִים אַרְבַּע פְּעָמִים.
לְמִנְהַג אַשְׁכְּנַז הַקָּהָל מְנַעְנֵעַ תֵּשַׁע פְּעָמִים וְהַחַזָּן שֶׁבַע, כִּי מְנַעְנְעִים בְּכָל פַּעַם שֶׁאוֹמְרִים ״הוֹדוּ לַה׳״, גַּם כְּשֶׁהַקָּהָל עוֹנֶה אַחֲרֵי הַחַזָּן.
(פֶּרֶק ה׳, הֲלָכָה ה׳)

10נָשִׁים וּבָנוֹת בַּלּוּלָב

נָשִׁים פְּטוּרוֹת מִנְּטִילַת לוּלָב, כִּי זוֹ מִצְוָה שֶׁיֵּשׁ לָהּ זְמַן. אֲבָל אִשָּׁה שֶׁנּוֹטֶלֶת – מְקַיֶּמֶת מִצְוָה וְיֵשׁ לָהּ שָׂכָר. וְטוֹב לְעוֹדֵד בָּנוֹת קְטַנּוֹת לְנַעְנֵעַ בַּלּוּלָב בְּכָל יוֹם, כְּדֵי לְחַנֵּךְ אוֹתָן לְחַבֵּב מִצְווֹת.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאוֹת אַשְׁכְּנַז נָשִׁים מְבָרְכוֹת, כִּי הֵן מְקַיְּמוֹת מִצְוָה. וּבַבְּרָכָה אוֹמְרִים ״וְצִוָּנוּ״ – אֶת כָּל יִשְׂרָאֵל, וְגַם הַנָּשִׁים בִּכְלָל זֶה.
לְמִנְהַג רֹב יוֹצְאוֹת סְפָרַד נָשִׁים אֵינָן מְבָרְכוֹת עַל מִצְווֹת שֶׁהַזְּמַן גְּרָמָן – אֲבָל עַל הַלּוּלָב נָהֲגוּ רַבּוֹת מֵהֶן לְבָרֵךְ, וְיֵשׁ לְכָךְ טְעָמִים עַל פִּי הַקַּבָּלָה.
(פֶּרֶק ה׳, הֲלָכָה ו׳)

11יְלָדִים וְלוּלָב

יֶלֶד שֶׁיּוֹדֵעַ לְנַעְנֵעַ כַּדִּין – לְהוֹלִיךְ וּלְהָבִיא, לְהַעֲלוֹת וּלְהוֹרִיד – אָבִיו חַיָּב לְחַנֵּךְ אוֹתוֹ לַמִּצְוָה. אִם הוּא כְּבָר בָּא לְבֵית הַכְּנֶסֶת לְהִתְפַּלֵּל, נָכוֹן שֶׁאָבִיו יִקְנֶה לוֹ אַרְבָּעָה מִינִים מִשֶּׁלּוֹ, כְּדֵי שֶׁיְּנַעְנֵעַ בַּהַלֵּל. וְאִם אִי אֶפְשָׁר – לְפָחוֹת יִתֵּן לוֹ בְּכָל יוֹם אֶת הַלּוּלָב שֶׁלּוֹ לְקַיֵּם בּוֹ אֶת הַמִּצְוָה.
💛 כְּשֶׁאֲנִי מַחְזִיק לוּלָב מִשֶּׁלִּי לְיַד אַבָּא, אֲנִי מַרְגִּישׁ שֶׁגַּם אֲנִי חֵלֶק מִן הַשִּׂמְחָה הַגְּדוֹלָה שֶׁל עַם יִשְׂרָאֵל לִפְנֵי ה׳.
(פֶּרֶק ה׳, הֲלָכָה ו׳)

12שׁוֹמְרִים עַל הַמִּינִים רַעֲנַנִּים

כְּדֵי שֶׁהַהֲדַסִּים וְהָעֲרָבוֹת יִשָּׁאֲרוּ רַעֲנַנִּים, שׁוֹמְרִים אוֹתָם בְּמַיִם, בְּמַגֶּבֶת לַחָה אוֹ בְּנַרְתִּיק אָטוּם. בְּיוֹם טוֹב מֻתָּר לְהַחֲזִיר אֶת הַלּוּלָב לַכְּלִי שֶׁהָיָה בּוֹ, וַאֲפִלּוּ לְהוֹסִיף מַיִם – אֲבָל אָסוּר לְהַחֲלִיף אֶת הַמַּיִם אוֹ לְמַלֵּא כְּלִי חָדָשׁ. בְּחֹל הַמּוֹעֵד מֻתָּר לְהַחֲלִיף אֶת הַמַּיִם. יֵשׁ מְהַדְּרִים לְהַחְלִיף אֶת הָעֲרָבוֹת בְּכָל יוֹם. וּכְשֶׁמַּכְנִיסִים עֲרָבוֹת חֲדָשׁוֹת לָאֲגֻדָּה – לֹא מִסְתַּפְּקִים בִּתְחִיבָה, אֶלָּא קוֹשְׁרִים מֵחָדָשׁ אוֹ מוֹסִיפִים קֶשֶׁר.
🤔 לָמָּה? הַחְלָפַת מַיִם בְּיוֹם טוֹב נִרְאֵית כְּמוֹ ״תִּקּוּן״ שֶׁל הַמִּינִים. וְהַקֶּשֶׁר הֶחָדָשׁ מְקַיֵּם שׁוּב אֶת מִצְוַת הָאִגּוּד.
(פֶּרֶק ה׳, הֲלָכָה ז׳)

13מֻקְצֶה לְמִצְוָתָם

מִן הָרֶגַע שֶׁנּוֹטְלִים אֶת הַמִּינִים בַּיּוֹם הָרִאשׁוֹן, הֵם ״מֻקְצִים לְמִצְוָתָם״ עַד סוֹף הֶחָג: אָסוּר לֶאֱכֹל אֶת הָאֶתְרוֹג, וְאָסוּר לְהָרִיחַ בַּהֲדַס – אֲפִלּוּ בְּהַבְדָּלָה שֶׁל מוֹצָאֵי שַׁבָּת. אֲבָל לְהָרִיחַ בָּאֶתְרוֹג – מֻתָּר, וּמְבָרְכִים ״הַנּוֹתֵן רֵיחַ טוֹב בַּפֵּרוֹת״. בִּשְׁעַת הַמִּצְוָה מְכַוְּנִים שֶׁנּוֹטְלִים לְשֵׁם מִצְוָה, וְלֹא כְּדֵי לֵהָנוֹת מִן הָרֵיחַ. אַחֲרֵי הֶחָג מֻתָּר לְהִשְׁתַּמֵּשׁ בַּמִּינִים, אֲבָל לֹא מְבַזִּים אוֹתָם.
🤔 לָמָּה? הָאֶתְרוֹג נוֹעַד לַאֲכִילָה – וּמִן הָאֲכִילָה הוּא הֻקְצָה. אֲבָל הָרֵיחַ שֶׁלּוֹ אֵינוֹ הַשִּׁמּוּשׁ הָעִקָּרִי שֶׁלּוֹ, וּמִמֶּנּוּ הוּא לֹא הֻקְצָה. וּמִי שֶׁהִתְנָה לִפְנֵי הַנְּטִילָה שֶׁהַמִּינִים לֹא יִהְיוּ מֻקְצִים – מֻתָּר לוֹ לְהִשְׁתַּמֵּשׁ בָּהֶם.
(פֶּרֶק ה׳, הֲלָכָה ח׳)

14הַקָּפוֹת בַּמִּקְדָּשׁ – כְּמוֹ בִּירִיחוֹ

בְּכָל יוֹם שֶׁל סֻכּוֹת הִקִּיפוּ הַכֹּהֲנִים אֶת הַמִּזְבֵּחַ פַּעַם אַחַת, וּבַיּוֹם הַשְּׁבִיעִי – שֶׁבַע פְּעָמִים, וְאָמְרוּ: ״אָנָּא ה׳ הוֹשִׁיעָה נָּא, אָנָּא ה׳ הַצְלִיחָה נָּא״. חֲכָמִים אָמְרוּ שֶׁזֶּה זֵכֶר לַהַקָּפוֹת שֶׁל יְרִיחוֹ בִּימֵי יְהוֹשֻׁעַ: שִׁשָּׁה יָמִים הִקִּיפוּ אֶת הָעִיר פַּעַם אַחַת בְּכָל יוֹם, וּבַיּוֹם הַשְּׁבִיעִי – שֶׁבַע פְּעָמִים. הַכֹּהֲנִים תָּקְעוּ, הָעָם הֵרִיעַ, וְהַחוֹמָה נָפְלָה.
🤔 לָמָּה? חוֹמַת יְרִיחוֹ הָיְתָה כְּמוֹ מְחִיצָה שֶׁעָצְרָה אֶת הָאוֹר שֶׁל ה׳. כְּשֶׁמַּקִּיפִים בְּסֻכּוֹת, מַפִּילִים כְּאִלּוּ אֶת הַחוֹמוֹת שֶׁמַּסְתִּירוֹת אֶת הַקְּדֻשָּׁה, וּמְגַלִּים שֶׁה׳ נִמְצָא גַּם בְּתוֹךְ הָעוֹלָם הָרָגִיל.
(פֶּרֶק ה׳, הֲלָכָה ט׳)

15הַקָּפוֹת בְּבֵית הַכְּנֶסֶת

אַחֲרֵי הַחֻרְבָּן, זֵכֶר לַמִּקְדָּשׁ, מַנִּיחִים סֵפֶר תּוֹרָה עַל הַבִּימָה, וְכָל הַקָּהָל מַקִּיף אוֹתָהּ עִם אַרְבַּעַת הַמִּינִים וְאוֹמֵר ״הוֹשַׁע נָא״ – פַּעַם אַחַת בְּכָל יוֹם, וּבַיּוֹם הַשְּׁבִיעִי שֶׁבַע פְּעָמִים. נוֹהֲגִים שֶׁאֶחָד מַחְזִיק אֶת סֵפֶר הַתּוֹרָה. מַקִּיפִים דֶּרֶךְ יָמִין, וַאֲרוֹן הַקֹּדֶשׁ נִשְׁאָר פָּתוּחַ. מִי שֶׁאֵין לוֹ לוּלָב – אֵינוֹ מַקִּיף. וּבְשַׁבָּת אֵין מַקִּיפִים.
🤔 לָמָּה? לְאַחַר שֶׁנֶּחֱרַב בֵּית הַמִּקְדָּשׁ, לִמּוּד הַתּוֹרָה בָּא בִּמְקוֹם הַמִּזְבֵּחַ – וְלָכֵן מַקִּיפִים אֶת סֵפֶר הַתּוֹרָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הַסְּפָרַדִּים וְהַחֲסִידִים מַקִּיפִים אַחֲרֵי אֲמִירַת הַהַלֵּל, לִפְנֵי קַדִּישׁ ״תִּתְקַבַּל״.
לְמִנְהַג מִקְצָת הָאַשְׁכְּנַזִּים מַקִּיפִים אַחֲרֵי תְּפִלַּת מוּסָף.
(פֶּרֶק ה׳, הֲלָכָה י׳)

📝 מָה לָמַדְנוּ

  • נוֹטְלִים לוּלָב בַּיּוֹם, בִּתְפִלַּת שַׁחֲרִית; וּכְשֶׁהַיּוֹם הָרִאשׁוֹן חָל בְּשַׁבָּת – לֹא נוֹטְלִים.
  • לוּלָב, הֲדַסִּים וַעֲרָבוֹת בְּיָמִין, אֶתְרוֹג בִּשְׂמֹאל – כֻּלָּם כְּדֶרֶךְ גְּדִילָתָם.
  • מְנַעְנְעִים לְשִׁשָּׁה כִּוּוּנִים, שָׁלֹשׁ פְּעָמִים לְכָל כִּוּוּן, וְרֹאשׁ הַלּוּלָב תָּמִיד לְמַעְלָה.
  • נָשִׁים וּבָנוֹת שֶׁנּוֹטְלוֹת מְקַיְּמוֹת מִצְוָה, וְאַבָּא מְחַנֵּךְ אֶת הַיֶּלֶד לְנַעְנֵעַ.
  • מַקִּיפִים אֶת הַבִּימָה בְּכָל יוֹם, וּבְהוֹשַׁעְנָא רַבָּה שֶׁבַע פְּעָמִים – זֵכֶר לַמִּקְדָּשׁ.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה מַחְזִיקִים אֶת הָאֶתְרוֹג הָפוּךְ בִּזְמַן הַבְּרָכָה?
  2. מָה אוֹמְרִים הַנִּעְנוּעִים – ״מוֹלִיךְ וּמֵבִיא, מַעֲלֶה וּמוֹרִיד״?
  3. מָה הַקֶּשֶׁר בֵּין הַהַקָּפוֹת בְּבֵית הַכְּנֶסֶת לַהַקָּפוֹת סְבִיב יְרִיחוֹ?

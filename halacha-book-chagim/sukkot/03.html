סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת סֻכּוֹת
מַהֲדוּרַת יְלָדִים

🏡 יוֹשְׁבִים בַּסֻּכָּה

פֶּרֶק ג׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
הַסֻּכָּה עוֹמֶדֶת, הַקִּשּׁוּטִים תְּלוּיִים, וְהַשֻּׁלְחָן עָרוּךְ. עַכְשָׁו מַתְחִיל הַחֵלֶק הֲכִי כֵּיף: לָגוּר בַּסֻּכָּה! אֲבָל מָה בְּדִיּוּק עוֹשִׂים בָּהּ? מֻתָּר לֶאֱכֹל תַּפּוּחַ בַּמִּטְבָּח? מָה עוֹשִׂים כְּשֶׁמַּתְחִיל גֶּשֶׁם? וּמָה עִם טִיּוּל בְּחֹל הַמּוֹעֵד? בַּפֶּרֶק הַזֶּה נִלְמַד אֵיךְ גָּרִים בַּסֻּכָּה – כְּמוֹ בַּבַּיִת.

1״תֵּשְׁבוּ – כְּעֵין תָּדוּרוּ״

מִצְוָה לָגוּר בַּסֻּכָּה שִׁבְעָה יָמִים, כְּמוֹ שֶׁגָּרִים בַּבַּיִת כָּל הַשָּׁנָה. יֵשׁ אַרְבָּעָה סוּגִים שֶׁל דְּבָרִים: א. מָה שֶׁעוֹשִׂים דַּוְקָא בַּבַּיִת – סְעוּדוֹת וְשֵׁינָה – חוֹבָה לַעֲשׂוֹת בַּסֻּכָּה. ב. מָה שֶׁעוֹשִׂים לִפְעָמִים בַּבַּיִת וְלִפְעָמִים בַּחוּץ – נִשְׁנוּשׁ, לִמּוּד תּוֹרָה, קְרִיאָה בְּסֵפֶר וְשִׂיחָה עִם חֲבֵרִים – מִצְוָה לַעֲשׂוֹת בַּסֻּכָּה. ג. מָה שֶׁעוֹשִׂים תָּמִיד מִחוּץ לַבַּיִת – תְּפִלָּה בְּמִנְיָן וְשִׁעוּרֵי תּוֹרָה – עוֹשִׂים בְּבֵית הַכְּנֶסֶת. ד. דְּבָרִים לֹא מְכֻבָּדִים – אָסוּר לַעֲשׂוֹת בַּסֻּכָּה.
🤔 לָמָּה? מִי שֶׁיּוֹצֵא מִן הַסֻּכָּה בְּלִי שׁוּם סִבָּה, כְּדֵי לַעֲשׂוֹת בַּבַּיִת דְּבָרִים שֶׁאֶפְשָׁר לַעֲשׂוֹת בָּהּ, מְזַלְזֵל בַּמִּצְוָה. כָּל רֶגַע שֶׁנִּמְצָאִים בַּסֻּכָּה – יֵשׁ בּוֹ מִצְוָה.
(פֶּרֶק ג׳, הֲלָכָה א׳)

2נָשִׁים וּבָנוֹת בַּסֻּכָּה

נָשִׁים פְּטוּרוֹת מִמִּצְוַת הַסֻּכָּה, כִּי הִיא ״מִצְוַת עֲשֵׂה שֶׁהַזְּמַן גְּרָמָהּ״ – מִצְוָה שֶׁיֵּשׁ לָהּ זְמַן קָבוּעַ. אֲבָל אִשָּׁה אוֹ בַּת שֶׁיּוֹשֶׁבֶת בַּסֻּכָּה – מְקַיֶּמֶת מִצְוָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג אַשְׁכְּנַז וְחֵלֶק מִיּוֹצְאוֹת סְפָרַד נָשִׁים וּבָנוֹת מְבָרְכוֹת ״לֵישֵׁב בַּסֻּכָּה״, כְּמוֹ הַגְּבָרִים.
לְמִנְהַג רֹב יוֹצְאוֹת סְפָרַד אֵינָן מְבָרְכוֹת עַל הַיְּשִׁיבָה בַּסֻּכָּה, כִּי אֵינָן חַיָּבוֹת לָשֶׁבֶת בָּהּ.
(פֶּרֶק ג׳, הֲלָכָה א׳)

3הַסֻּכָּה – הַבַּיִת שֶׁלָּנוּ

כָּל שִׁבְעַת הַיָּמִים הַסֻּכָּה הִיא הַבַּיִת הַקָּבוּעַ, וְהַבַּיִת – אַרְעַי. לָכֵן מַעֲבִירִים לַסֻּכָּה אֶת הַשֻּׁלְחָן וְהַכִּסְאוֹת הַיָּפִים, וְאֶת הַמִּטָּה וְהַמַּצָּעִים הַטּוֹבִים, וְהַבַּיִת מְשַׁמֵּשׁ רַק כְּמוֹ מִטְבָּח וּמַחְסָן שֶׁעוֹזְרִים לַסֻּכָּה. לִמּוּד תּוֹרָה רָגִיל – טוֹב לִלְמֹד בַּסֻּכָּה. לִמּוּד עָמֹק שֶׁצָּרִיךְ בּוֹ רִכּוּז, אוֹ לִמּוּד עִם הַרְבֵּה סְפָרִים – אֶפְשָׁר לִלְמֹד בַּבַּיִת אוֹ בְּבֵית הַמִּדְרָשׁ.
🤔 לָמָּה? לֹא מַסְפִּיק לֶאֱכֹל וְלִישֹׁן בַּסֻּכָּה – צָרִיךְ שֶׁעִקַּר הַחַיִּים שֶׁלָּנוּ יִהְיֶה בָּהּ.
(פֶּרֶק ג׳, הֲלָכָה ב׳)

4כְּבוֹד הַסֻּכָּה

נוֹהֲגִים בַּסֻּכָּה כְּמוֹ בַּחֶדֶר הַיָּפֶה וְהַמְכֻבָּד שֶׁבַּבַּיִת. לָכֵן לֹא שׁוֹטְפִים בָּהּ כֵּלִים, לֹא מַחְלִיפִים בָּהּ חִתּוּל לְתִינוֹק, לֹא מַנִּיחִים בָּהּ דְּלִי אוֹ פַּח לִשְׁיָרֵי אֹכֶל, וְלֹא בְּגָדִים שֶׁצְּרִיכִים כְּבִיסָה. אַחֲרֵי הָאֲרוּחָה מְפַנִּים מַהֵר אֶת הַצַּלָּחוֹת וְאֶת הַסַּכּוּ״ם. אֶת כּוֹסוֹת הַשְּׁתִיָּה מֻתָּר לְהַשְׁאִיר, וּמֻתָּר לְהַנִּיחַ פַּח לִנְיָרוֹת. וְלִפְנֵי הַשֵּׁינָה מֻתָּר לְהַנִּיחַ אֶת הַבְּגָדִים עַל כִּסֵּא וְלַחְלֹץ נַעֲלַיִם וְגַרְבַּיִם בַּסֻּכָּה, כְּמוֹ בַּבַּיִת.
🤔 לָמָּה? הַסֻּכָּה הִיא מָקוֹם שֶׁל מִצְוָה. כְּמוֹ שֶׁלֹּא מַשְׁאִירִים צַלָּחוֹת מְלֻכְלָכוֹת בְּסָלוֹן שֶׁמְּקַבְּלִים בּוֹ אוֹרְחִים – כָּךְ גַּם בַּסֻּכָּה.
(פֶּרֶק ג׳, הֲלָכָה ב׳)

5מְשׂוֹחֲחִים וּמְשַׂחֲקִים – בַּסֻּכָּה

אֵין שׁוּם פְּגָם בְּשִׂיחָה רְגִילָה בַּסֻּכָּה. מִי שֶׁרוֹצֶה לְשׂוֹחֵחַ עִם חָבֵר, אוֹ לְדַבֵּר בַּטֶּלֶפוֹן – טוֹב שֶׁיַּעֲשֶׂה זֹאת בַּסֻּכָּה. וְגַם מִי שֶׁרוֹצֶה לְשַׂחֵק שַׁחְמָט אוֹ מוֹנוֹפּוֹל – טוֹב שֶׁיְּשַׂחֵק בַּסֻּכָּה. יֵשׁ מְהַדְּרִים שֶׁלֹּא לְדַבֵּר בַּסֻּכָּה דִּבְרֵי חֹל, אֲבָל אִם בִּגְלַל זֶה הֵם נִמְצָאִים בָּהּ פָּחוֹת – אֵין בְּכָךְ הִדּוּר.
🤔 לָמָּה? כָּל אֵימַת שֶׁנִּמְצָאִים בַּסֻּכָּה – מְקַיְּמִים מִצְוָה. גַּם מִשְׂחָק עִם הָאַחִים הוֹפֵךְ לְמִצְוָה כְּשֶׁהוּא בַּסֻּכָּה.
💛 אֲפִלּוּ מִשְׂחָק שֶׁל שַׁחְמָט בַּסֻּכָּה הוּא זְמַן שֶׁאֲנִי יוֹשֵׁב בְּצִלּוֹ שֶׁל ה׳.
(פֶּרֶק ג׳, הֲלָכָה ב׳)

6הַלַּיְלָה הָרִאשׁוֹן

בְּכָל יְמֵי הֶחָג, מִי שֶׁמִּסְתַּפֵּק בְּנִשְׁנוּשִׁים יָכוֹל לֶאֱכֹל אוֹתָם מִחוּץ לַסֻּכָּה. אֲבָל בַּלַּיְלָה הָרִאשׁוֹן – חוֹבָה לֶאֱכֹל לֶחֶם בַּסֻּכָּה, כְּמוֹ שֶׁבְּלֵיל הַסֵּדֶר חוֹבָה לֶאֱכֹל מַצָּה. אוֹכְלִים אַחֲרֵי צֵאת הַכּוֹכָבִים, וּלְכַתְּחִלָּה לִפְנֵי חֲצוֹת הַלַּיְלָה. מִי שֶׁלֹּא הִסְפִּיק – אוֹכֵל עַד עֲלוֹת הַשַּׁחַר. וְלִפְנֵי שֶׁאוֹכְלִים מְכַוְּנִים: אֲנִי מְקַיֵּם מִצְוַת ה׳, זֵכֶר לִיצִיאַת מִצְרַיִם וּלְעַנְנֵי הַכָּבוֹד.
🤔 לָמָּה? הַלַּיְלָה הָרִאשׁוֹן הוּא הַיְּסוֹד שֶׁל כָּל הֶחָג. וּכְדֵי לֶאֱכֹל בְּתֵאָבוֹן, לֹא אוֹכְלִים דְּבָרִים מַשְׂבִּיעִים כְּשָׁלֹשׁ שָׁעוֹת לִפְנֵי הַשְּׁקִיעָה בְּעֶרֶב הֶחָג.
💛 הָאֲרוּחָה הָרִאשׁוֹנָה בַּסֻּכָּה הִיא כְּמוֹ כְּנִיסָה לְבֵיתוֹ שֶׁל ה׳ – וְהוּא מְחַכֶּה לָנוּ שָׁם.
(פֶּרֶק ג׳, הֲלָכָה ג׳)

7כַּמָּה לֶחֶם בַּלַּיְלָה הָרִאשׁוֹן?

בַּלַּיְלָה הָרִאשׁוֹן מְדַקְדְּקִים כְּמוֹ בַּמַּצָּה שֶׁל לֵיל הַסֵּדֶר: אוֹכְלִים לְפָחוֹת כַּחֲצִי בֵּיצָה לֶחֶם, וְטוֹב לֶאֱכֹל יוֹתֵר מִבֵּיצָה שְׁלֵמָה. אֵין צֹרֶךְ לִמְעֹךְ אֶת הַחַלָּה כְּדֵי לִמְדֹּד – מַעֲרִיכִים אוֹתָהּ כְּמוֹ שֶׁהִיא. אוֹכְלִים בְּנַחַת וּבִרְצִיפוּת, בְּלִי לְהַפְסִיק יוֹתֵר מִשֵּׁשׁ אוֹ שֶׁבַע דַּקּוֹת.
(פֶּרֶק ג׳, הֲלָכָה ג׳)

8גֶּשֶׁם בַּלַּיְלָה הָרִאשׁוֹן

יֵשׁ אוֹמְרִים שֶׁגַּם בַּלַּיְלָה הָרִאשׁוֹן, מִי שֶׁמִּצְטַעֵר בִּגְלַל הַגֶּשֶׁם פָּטוּר, וְיֵשׁ אוֹמְרִים שֶׁבַּלַּיְלָה הָרִאשׁוֹן הוּא חַיָּב. לְמַעֲשֶׂה חוֹשְׁשִׁים לִשְׁתֵּי הַדֵּעוֹת: מַמְתִּינִים שָׁעָה אוֹ שְׁעָתַיִם – אוּלַי הַגֶּשֶׁם יִפָּסֵק. אִם הוּא מַמְשִׁיךְ, מְקַדְּשִׁים בַּסֻּכָּה, מְבָרְכִים ״שֶׁהֶחֱיָנוּ״ וְאוֹכְלִים כְּזַיִת לֶחֶם – אֲבָל לֹא מְבָרְכִים ״לֵישֵׁב בַּסֻּכָּה״. וְאִם הַגֶּשֶׁם נִפְסָק לִפְנֵי חֲצוֹת – חוֹזְרִים לַסֻּכָּה, אוֹכְלִים לֶחֶם וּמְבָרְכִים ״לֵישֵׁב בַּסֻּכָּה״.
🤔 לָמָּה? כָּךְ מְקַיְּמִים אֶת הַמִּצְוָה לְפִי הַדֵּעָה שֶׁחַיָּבִים, וְלֹא מְבָרְכִים בְּרָכָה שֶׁאוּלַי אֵינָהּ צְרִיכָה, לְפִי הַדֵּעָה שֶׁפְּטוּרִים.
(פֶּרֶק ג׳, הֲלָכָה ג׳)

9סְעוּדָה אוֹ נִשְׁנוּשׁ?

״אֲכִילַת קֶבַע״ – אֲכִילָה שֶׁאוֹכְלִים כְּדֵי לִשְׂבֹּעַ – חוֹבָה בַּסֻּכָּה. ״אֲכִילַת עֲרַאי״ – כְּדֵי לִטְעֹם אוֹ לְהַרְגִּיעַ מְעַט אֶת הָרָעָב – מֻתֶּרֶת גַּם בַּחוּץ. לֶחֶם, עוּגוֹת, פְּתִיתִים, אִטְרִיּוֹת וְדַיְסָה – יוֹתֵר מִנֶּפַח שֶׁל בֵּיצָה – אוֹכְלִים רַק בַּסֻּכָּה. פֵּרוֹת, מַיִם וּמִיצִים – מֻתָּר לֶאֱכֹל וְלִשְׁתּוֹת בַּחוּץ כַּמָּה שֶׁרוֹצִים. מְעַט בָּשָׂר, דָּגִים אוֹ גְּבִינָה – מֻתָּר בַּחוּץ, אֲבָל אֲרוּחָה שֶׁשְּׂבֵעִים מִמֶּנָּה – רַק בַּסֻּכָּה.
🤔 לָמָּה? הַדָּגָן הוּא עִקַּר הַמָּזוֹן שֶׁל הָאָדָם, וּמִמֶּנּוּ רְגִילִים לִשְׂבֹּעַ. אֶת מָה שֶׁאוֹכְלִים לִפְעָמִים גַּם מִחוּץ לַבַּיִת – מֻתָּר לֶאֱכֹל גַּם מִחוּץ לַסֻּכָּה. וְהַמְהַדְּרִים אוֹכְלִים בַּסֻּכָּה אֲפִלּוּ נִשְׁנוּשִׁים, וְאֵינָם שׁוֹתִים בַּחוּץ אֲפִלּוּ מַיִם – אֲבָל אֵין זוֹ חוֹבָה.
(פֶּרֶק ג׳, הֲלָכָה ד׳)

10בְּאֶמְצַע הַסְּעוּדָה

בִּזְמַן הַסְּעוּדָה, כָּל מָה שֶׁאוֹכְלִים הוּא חֵלֶק מִן הַסְּעוּדָה, וְצָרִיךְ לֶאֱכֹל אוֹתוֹ בַּסֻּכָּה. לָכֵן מִי שֶׁיּוֹצֵא בְּאֶמְצַע הָאֲרוּחָה לַבַּיִת כְּדֵי לְהָבִיא מֶלַח אוֹ צַלַּחַת – לֹא שׁוֹתֶה שָׁם אֲפִלּוּ מְעַט מַיִם, לֹא טוֹעֵם כְּלוּם, וְאֵינוֹ מַמְשִׁיךְ לִבְלֹעַ שָׁם אֶת מָה שֶׁהִתְחִיל לֶאֱכֹל בַּסֻּכָּה.
(פֶּרֶק ג׳, הֲלָכָה ד׳)

11מָתַי מְבָרְכִים ״לֵישֵׁב בַּסֻּכָּה״?

חֲכָמִים תִּקְּנוּ לְבָרֵךְ: ״בָּרוּךְ אַתָּה ה׳ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם אֲשֶׁר קִדְּשָׁנוּ בְּמִצְוֹתָיו וְצִוָּנוּ לֵישֵׁב בַּסֻּכָּה״. עוֹלֵי תֵּימָן מְבָרְכִים בְּכָל פַּעַם שֶׁנִּכְנָסִים לַסֻּכָּה לִשְׁהוֹת בָּהּ – בַּעֲמִידָה, וְאָז מִתְיַשְּׁבִים. שְׁאָר הָעֵדוֹת מְבָרְכִים עַל הָאֲכִילָה, כִּי הִיא הַחֵלֶק הָעִקָּרִי, וְהַבְּרָכָה עָלֶיהָ פּוֹטֶרֶת אֶת כָּל מָה שֶׁעוֹשִׂים בַּסֻּכָּה. וְהַיּוֹם, גַּם מִי שֶׁאוֹכֵל אֲרוּחָה מְלֵאָה בְּלִי לֶחֶם – מָרָק, בָּשָׂר, אֹרֶז וְתַפּוּחֵי אֲדָמָה – צָרִיךְ לְבָרֵךְ, כִּי זוֹ סְעוּדָה חֲשׁוּבָה. וּכְדֵי לָצֵאת מִסָּפֵק, עָדִיף לֶאֱכֹל בָּהּ לֶחֶם.
🤔 לָמָּה? עַל הַשֵּׁינָה לֹא מְבָרְכִים, כִּי אוּלַי לֹא יֵרָדֵם. אֲבָל הָאֲכִילָה תְּלוּיָה בְּהַחְלָטָה שֶׁלָּנוּ – וְעָלֶיהָ רָאוּי לְבָרֵךְ.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג אַשְׁכְּנַז מְבָרְכִים עַל לֶחֶם אוֹ מְזוֹנוֹת יוֹתֵר מִכְּבֵיצָה. וּמִי שֶׁלֹּא יֹאכַל כָּל כָּךְ בְּכָל הַזְּמַן שֶׁהוּא בַּסֻּכָּה – מְבָרֵךְ גַּם עַל טְעִימָה שֶׁל מְזוֹנוֹת, יַיִן אוֹ מַאֲכָל חָשׁוּב אַחֵר.
לְמִנְהַג סְפָרַד מְבָרְכִים עַל לֶחֶם יוֹתֵר מִכְּבֵיצָה. עַל עוּגוֹת וּשְׁאָר מְזוֹנוֹת – רַק כְּשֶׁאוֹכְלִים כְּמוֹ אֲרוּחָה מְשַׂבַּעַת, בְּנֶפַח שֶׁל אַרְבַּע בֵּיצִים.
(פֶּרֶק ג׳, הֲלָכָה ה׳)

12אֵיזוֹ בְּרָכָה קוֹדֶמֶת?

מְבָרְכִים ״לֵישֵׁב בַּסֻּכָּה״ לִפְנֵי הָאֲכִילָה, וְיַחַד עִם הַבְּרָכָה עַל הָאֹכֶל. אֵיזוֹ מֵהֶן קוֹדֶמֶת? יֵשׁ בָּזֶה שְׁנֵי מִנְהָגִים, וְנָכוֹן שֶׁכָּל אֶחָד יַמְשִׁיךְ בְּמִנְהַג אֲבוֹתָיו.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הָאַשְׁכְּנַזִּים וְחֵלֶק מֵהַסְּפָרַדִּים מְבָרְכִים קֹדֶם עַל הָאֹכֶל – לְמָשָׁל ״הַמּוֹצִיא״ – וְאַחַר כָּךְ ״לֵישֵׁב בַּסֻּכָּה״, וְאֵין צֹרֶךְ לַעֲמֹד, כִּי הָאֲכִילָה הִיא שֶׁמְּחַיֶּבֶת לָשֶׁבֶת בַּסֻּכָּה.
לְמִנְהַג חֵלֶק מֵהַסְּפָרַדִּים מְבָרְכִים קֹדֶם ״לֵישֵׁב בַּסֻּכָּה״ בַּעֲמִידָה, וְאַחַר כָּךְ מִתְיַשְּׁבִים וּמְבָרְכִים עַל הָאֹכֶל.
(פֶּרֶק ג׳, הֲלָכָה ו׳)

13שָׁכַחְתִּי לְבָרֵךְ? יָצָאתִי וְחָזַרְתִּי?

מִי שֶׁשָּׁכַח לְבָרֵךְ ״לֵישֵׁב בַּסֻּכָּה״ בִּתְחִלַּת הַסְּעוּדָה – מְבָרֵךְ בְּאֶמְצָעָהּ וּמַמְשִׁיךְ לֶאֱכֹל. הַבְּרָכָה מוֹעִילָה כָּל זְמַן שֶׁנִּשְׁאָרִים בַּסֻּכָּה, גַּם לִסְעוּדָה נוֹסֶפֶת. מִי שֶׁיָּצָא לְרֶגַע – לְהָבִיא מַשֶּׁהוּ אוֹ לְדַבֵּר עִם חָבֵר – אֵינוֹ מְבָרֵךְ שׁוּב כְּשֶׁהוּא חוֹזֵר. אֲבָל מִי שֶׁיָּצָא לְדָבָר חָשׁוּב, כְּמוֹ לְהִתְפַּלֵּל, אוֹ שֶׁיָּצָא לְיוֹתֵר מִשָּׁעָה – מְבָרֵךְ שׁוּב כְּשֶׁהוּא חוֹזֵר וְאוֹכֵל.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת רֹב הַפּוֹסְקִים מִי שֶׁנִּזְכַּר רַק אַחֲרֵי שֶׁסִּיֵּם אֶת הַסְּעוּדָה – מְבָרֵךְ ״לֵישֵׁב בַּסֻּכָּה״, אַף שֶׁאֵינוֹ מִתְכַּוֵּן לְהַמְשִׁיךְ לֶאֱכֹל.
לְמִנְהַג הַסְּפָרַדִּים מִי שֶׁנִּזְכַּר רַק אַחֲרֵי הַסְּעוּדָה – אֵינוֹ מְבָרֵךְ. (וְאִם עוֹד אֶפְשָׁר לֶאֱכֹל אוֹ לִשְׁתּוֹת מַשֶּׁהוּ לִפְנֵי בִּרְכַּת הַמָּזוֹן – לְכָל הַדֵּעוֹת מְבָרֵךְ וְאוֹכֵל מְעַט.)
(פֶּרֶק ג׳, הֲלָכָה ו׳)

14יְשֵׁנִים בַּסֻּכָּה

חוֹבָה לִישֹׁן בַּסֻּכָּה – גַּם שֵׁינָה אֲרֻכָּה בַּלַּיְלָה, וְגַם שֵׁינָה קְצָרָה בַּצָּהֳרַיִם. מִי שֶׁמַּנִּיחַ אֶת רֹאשׁוֹ עַל הַשֻּׁלְחָן כְּדֵי לִישֹׁן מְעַט – יִישַׁן בַּסֻּכָּה. אֲבָל מִי שֶׁנִּרְדָּם בְּלִי לִרְצוֹת, בְּאֶמְצַע נְסִיעָה אוֹ שִׁעוּר, וּמִתְעוֹרֵר שׁוּב וָשׁוּב – אֵינוֹ עוֹבֵר עֲבֵרָה.
🤔 לָמָּה? נִשְׁנוּשׁ אוֹכְלִים לִפְעָמִים גַּם מִחוּץ לַבַּיִת, וְלָכֵן הוּא מֻתָּר מִחוּץ לַסֻּכָּה. אֲבָל לִישֹׁן – אֲפִלּוּ קְצָת – רֹב הָאֲנָשִׁים לֹא רְגִילִים בַּחוּץ, וְגַם שֵׁינָה קְצָרָה חֲשׁוּבָה וּמַרְגִּיעָה אֶת הָעֲיֵפוּת.
(פֶּרֶק ג׳, הֲלָכָה ז׳)

15גֶּשֶׁם – מִצְטַעֵר פָּטוּר

מִי שֶׁמִּצְטַעֵר בַּסֻּכָּה פָּטוּר מִמֶּנָּה. הַמִּקְרֶה הַשָּׁכִיחַ הוּא גֶּשֶׁם. כַּמָּה טִפּוֹת בִּלְבַד אֵינָן פּוֹטְרוֹת, אֲבָל גֶּשֶׁם שֶׁיָּכוֹל לְקַלְקֵל אֶת הָאֹכֶל – פּוֹטֵר. מִי שֶׁיָּצָא בִּגְלַל הַגֶּשֶׁם, הִתְחִיל לֶאֱכֹל בַּבַּיִת, וְהַגֶּשֶׁם פָּסַק – לֹא צָרִיךְ לַחֲזֹר, וּמְסַיֵּם בַּבַּיִת. וְאִם הָלַךְ לִישֹׁן בַּבַּיִת בִּגְלַל הַגֶּשֶׁם – יִישַׁן שָׁם עַד הַבֹּקֶר. וּמִי שֶׁמִּתְעַקֵּשׁ לֶאֱכֹל בַּסֻּכָּה כְּשֶׁהַגֶּשֶׁם יוֹרֵד עָלָיו – אֵין בְּיָדוֹ מִצְוָה.
🤔 לָמָּה? ״תֵּשְׁבוּ כְּעֵין תָּדוּרוּ״: אָדָם אֵינוֹ גָּר בְּמָקוֹם שֶׁהוּא סוֹבֵל בּוֹ. וּבַחַג צָרִיךְ לְכַבֵּד אֶת הַיּוֹם וְלֹא לְצַעֵר אֶת עַצְמוֹ. אֲבָל מִי שֶׁהַגֶּשֶׁם פָּסַק וְהוּא טָרַח וְחָזַר לַסֻּכָּה – יֵשׁ לוֹ שָׂכָר.
💛 ה׳ לֹא רוֹצֶה שֶׁנִּסְבֹּל בַּחַג – הוּא רוֹצֶה שֶׁנָּגוּר אִתּוֹ בְּשִׂמְחָה.
(פֶּרֶק ג׳, הֲלָכָה ח׳, הֲלָכָה ט׳)

16עוֹד מִצְטַעֲרִים

גַּם יַתּוּשִׁים רַבִּים, רֵיחַ רַע מְאוֹד, חֹם כָּבֵד בַּצָּהֳרַיִם אוֹ קֹר חָזָק בַּלַּיְלָה – פּוֹטְרִים מִן הַסֻּכָּה, כְּשֶׁהַסֵּבֶל כָּל כָּךְ גָּדוֹל שֶׁבִּגְלָלוֹ אָדָם הָיָה עוֹבֵר מִבֵּיתוֹ לְדִירָה אַחֶרֶת. וְזֶה רַק כְּשֶׁבַּבַּיִת הַצַּעַר נֶעֱלָם; אִם גַּם בַּבַּיִת יֵשׁ יַתּוּשִׁים – נִשְׁאָרִים בַּסֻּכָּה. כָּבָה הָאוֹר בַּסֻּכָּה בְּלֵיל שַׁבָּת: אִם אֶפְשָׁר לֶאֱכֹל בַּסֻּכָּה שֶׁל שָׁכֵן בְּלִי טֹרַח רַב – הוֹלְכִים אֵלָיו, וְאִם לֹא – אוֹכְלִים בַּבַּיִת.
🤔 לָמָּה? כְּשֶׁרַק אֲנָשִׁים רְגִישִׁים מִצְטַעֲרִים, כְּמוֹ מֵעָלִים שֶׁנּוֹשְׁרִים עַל הַשֻּׁלְחָן – הָרְגִישִׁים פְּטוּרִים, וְהַשְּׁאָר חַיָּבִים. וּמִי שֶׁהִתְרַשֵּׁל וּבָנָה סֻכָּה רְעוּעָה – אֵינוֹ יָכוֹל לִטְעֹן אַחַר כָּךְ שֶׁהוּא מִצְטַעֵר בָּהּ, אֶלָּא צָרִיךְ לְחַזֵּק אוֹתָהּ מִיָּד.
(פֶּרֶק ג׳, הֲלָכָה ט׳)

17קָשֶׁה לִי לִישֹׁן בַּסֻּכָּה

מִי שֶׁמִּצְטַנֵּן בַּסֻּכָּה בַּלֵּילוֹת הַקָּרִים, גַּם עִם שְׂמִיכָה חַמָּה – פָּטוּר מִשֵּׁינָה בַּסֻּכָּה. אֲבָל בַּלֵּילוֹת שֶׁאֵין חֲשָׁשׁ שֶׁיִּצְטַנֵּן, וּבְשֵׁינַת הַצָּהֳרַיִם – חַיָּב לִישֹׁן בַּסֻּכָּה. גַּם מִי שֶׁפָּשׁוּט אֵינוֹ מַצְלִיחַ לְהֵרָדֵם בַּסֻּכָּה, כִּי הַמָּקוֹם שׁוֹנֶה וְהוּא מָתוּחַ – פָּטוּר מִשֵּׁינָה בָּהּ. וּמִי שֶׁרוֹצֶה לִישֹׁן בַּצָּהֳרַיִם, וּלְיַד הַסֻּכָּה יְלָדִים מַרְעִישִׁים וְאֵינָם מַצְלִיחִים לִשְׁמֹר עַל שֶׁקֶט – מֻתָּר לוֹ לִישֹׁן בַּבַּיִת.
🤔 לָמָּה? מִי שֶׁאֵינוֹ יָשֵׁן בְּפֹעַל – מִצְטַעֵר. וְהַתּוֹרָה לֹא צִוְּתָה לִבְנוֹת קִירוֹת שֶׁל בִּנְיָן כְּדֵי לִישֹׁן בַּסֻּכָּה – אֲבָל מִי שֶׁבּוֹנֶה סֻכָּה נוֹחָה, שֶׁאֶפְשָׁר לִישֹׁן בָּהּ, תָּבוֹא עָלָיו בְּרָכָה.
(פֶּרֶק ג׳, הֲלָכָה י׳)

18חוֹלֶה פָּטוּר

חוֹלֶה פָּטוּר מִן הַסֻּכָּה – לֹא רַק חוֹלֶה מְסֻכָּן, אֶלָּא גַּם מִי שֶׁכּוֹאֵב לוֹ הָרֹאשׁ וְקָשֶׁה לוֹ לָשֶׁבֶת בַּסֻּכָּה. וְאִם הוּא צָרִיךְ מִישֶׁהוּ שֶׁיַּעֲזֹר לוֹ – גַּם הָעוֹזֵר פָּטוּר.
💛 מִי שֶׁמַּגִּישׁ כּוֹס תֵּה לְאַבָּא חוֹלֶה בַּבַּיִת, עוֹסֵק בְּמִצְוָה – וְה׳ אוֹהֵב אֶת הַחֶסֶד הַזֶּה.
(פֶּרֶק ג׳, הֲלָכָה ח׳)

19יְלָדִים בַּסֻּכָּה

יֶלֶד שֶׁמֵּבִין שֶׁצָּרִיךְ לֶאֱכֹל וְלִישֹׁן בַּסֻּכָּה – בְּדֶרֶךְ כְּלָל בְּגִיל חָמֵשׁ עַד שֵׁשׁ – חַיָּב בַּסֻּכָּה מִשּׁוּם מִצְוַת חִנּוּךְ, וְהַגְּדוֹלִים לֹא נוֹתְנִים לוֹ לֶאֱכֹל סְעוּדָה אוֹ לִישֹׁן מִחוּץ לַסֻּכָּה. אֲבָל יְלָדִים רְגִישִׁים יוֹתֵר לְקֹר, וְלִפְעָמִים מְפַחֲדִים – וְאָז הֵם כְּמוֹ מִצְטַעֲרִים, וּפְטוּרִים מִשֵּׁינָה בַּסֻּכָּה. וְאִם קָשֶׁה לַיֶּלֶד לְהֵרָדֵם כְּשֶׁהַגְּדוֹלִים לוֹמְדִים וּמְשׂוֹחֲחִים בַּסֻּכָּה – מַשְׁכִּיבִים אוֹתוֹ בַּבַּיִת, וְאַחַר כָּךְ מַעֲבִירִים אוֹתוֹ לַסֻּכָּה.
(פֶּרֶק ג׳, הֲלָכָה י״ב)

20חַיָּלִים וּסְעוּדוֹת שֶׁל מִצְוָה

״הָעוֹסֵק בְּמִצְוָה פָּטוּר מִן הַמִּצְוָה״: מִי שֶׁמְּלַוֶּה חוֹלֶה בְּבֵית הַחוֹלִים, אוֹ חַיָּל שֶׁעוֹסֵק בִּשְׁמִירָה וְאֵין לוֹ זְמַן פָּנוּי – פָּטוּר מִן הַסֻּכָּה. אֲבָל אִם יֵשׁ סֻכָּה קְרוֹבָה וְנוֹחָה – בַּשָּׁעוֹת הַפְּנוּיוֹת אוֹכְלִים וִישֵׁנִים בָּהּ. וְהַמְּפַקְּדִים צְרִיכִים לִדְאֹג לְסֻכָּה נוֹחָה לַחַיָּלִים. וּסְעוּדוֹת שֶׁל מִצְוָה – בְּרִית מִילָה, פִּדְיוֹן הַבֵּן, בַּר מִצְוָה וְסִיּוּם מַסֶּכֶת – עוֹרְכִים בַּסֻּכָּה.
🤔 לָמָּה? הַטִּרְחָה לְחַפֵּשׂ סֻכָּה עֲלוּלָה לִפְגֹּעַ בַּמִּצְוָה שֶׁהוּא עָסוּק בָּהּ.
(פֶּרֶק ג׳, הֲלָכָה י״ב, הֲלָכָה י״ג)

21טִיּוּל בְּחֹל הַמּוֹעֵד

מִי שֶׁרוֹצֶה לָצֵאת עִם מִשְׁפַּחְתּוֹ לְטִיּוּל, מְתַכְנֵן אוֹתוֹ כָּךְ שֶׁיֹּאכְלוּ אֶת הָאֲרוּחוֹת בַּסֻּכָּה. וּבִכְלָל, לֹא מְבַזְבְּזִים אֶת כָּל יְמֵי חֹל הַמּוֹעֵד עַל טִיּוּלִים – הֵם נוֹעֲדוּ גַּם לְתוֹרָה, לִתְפִלָּה וְלִסְעוּדוֹת שֶׁל מִצְוָה. וּמִי שֶׁנּוֹסֵעַ לִנְסִיעָה שֶׁל מִצְוָה – כְּמוֹ לְהִתְפַּלֵּל בַּכֹּתֶל הַמַּעֲרָבִי אוֹ לְבַקֵּר אֶת רַבּוֹ – וְקָשֶׁה לוֹ לִמְצֹא סֻכָּה, רַשַּׁאי לֶאֱכֹל אֲרוּחָה מִחוּץ לַסֻּכָּה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ פּוֹסְקִים שֶׁמַּתִּירִים לְמִי שֶׁיּוֹצֵא לְטִיּוּל לֶאֱכֹל אֲרוּחַת קֶבַע בְּלִי סֻכָּה, כְּמוֹ שֶׁבְּכָל הַשָּׁנָה מְטַיְּלִים אֵינָם מַקְפִּידִים לֶאֱכֹל בְּתוֹךְ בַּיִת.
לְפִי הָרַב מֶלָמֵד לְמַעֲשֶׂה אֵין לְהָקֵל, כִּי מִי שֶׁיּוֹצֵא לְטִיּוּל לַהֲנָאָתוֹ מְבַטֵּל אֶת עַצְמוֹ מִן הַמִּצְוָה. אוֹכְלִים אֶת הָאֲרוּחוֹת בַּסֻּכָּה, וּבַטִּיּוּל מִסְתַּפְּקִים בְּפֵרוֹת, יְרָקוֹת וּמְעַט מְזוֹנוֹת.
(פֶּרֶק ג׳, הֲלָכָה י״ד)

📝 מָה לָמַדְנוּ

  • גָּרִים בַּסֻּכָּה כְּמוֹ בַּבַּיִת: סְעוּדוֹת וְשֵׁינָה – חוֹבָה, לִמּוּד, שִׂיחָה וּמִשְׂחָק – מִצְוָה.
  • בַּלַּיְלָה הָרִאשׁוֹן חוֹבָה לֶאֱכֹל לֶחֶם בַּסֻּכָּה, כְּמוֹ מַצָּה בְּלֵיל הַסֵּדֶר.
  • לֶחֶם וּמְזוֹנוֹת יוֹתֵר מִכְּבֵיצָה – רַק בַּסֻּכָּה; פֵּרוֹת וּמַיִם – מֻתָּר גַּם בַּחוּץ.
  • מְבָרְכִים ״לֵישֵׁב בַּסֻּכָּה״ עַל הָאֲכִילָה, וְכָל עֵדָה לְפִי מִנְהָגָהּ.
  • מִצְטַעֵר, חוֹלֶה וְעוֹסֵק בְּמִצְוָה – פְּטוּרִים מִן הַסֻּכָּה.

❓ שְׁאֵלוֹת חֲזָרָה

  1. מָה הַהֶבְדֵּל בֵּין ״אֲכִילַת קֶבַע״ לַ״אֲכִילַת עֲרַאי״? תֵּן דֻּגְמָה לְכָל אַחַת.
  2. יָרַד גֶּשֶׁם, נִכְנַסְנוּ לֶאֱכֹל בַּבַּיִת, וְהַגֶּשֶׁם פָּסַק. מָה עוֹשִׂים, וְלָמָּה?
  3. מָה עוֹשִׂים כְּשֶׁיּוֹרֵד גֶּשֶׁם בַּלַּיְלָה הָרִאשׁוֹן שֶׁל סֻכּוֹת?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״יוֹשְׁבִים בַּסֻּכָּה״

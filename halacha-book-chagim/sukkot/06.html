סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת סֻכּוֹת
מַהֲדוּרַת יְלָדִים

🌾 הוֹשַׁעְנָא רַבָּה

פֶּרֶק ו׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
הַיּוֹם הַשְּׁבִיעִי שֶׁל סֻכּוֹת. בְּבֵית הַכְּנֶסֶת מַקִּיפִים אֶת הַבִּימָה שׁוּב וָשׁוּב – שֶׁבַע פְּעָמִים! – וְאַחַר כָּךְ כֻּלָּם מַחְזִיקִים צְרוֹר קָטָן שֶׁל עֲרָבוֹת וְחוֹבְטִים בּוֹ עַל הָאָרֶץ, וְעָלִים יְרֻקִּים עָפִים לְכָל עֵבֶר. מָה הַסּוֹד שֶׁל הַיּוֹם הַזֶּה? וְאֵיךְ נִפְרָדִים מִן הַסֻּכָּה? בַּפֶּרֶק הַזֶּה נִלְמַד עַל הוֹשַׁעְנָא רַבָּה.

1יוֹם שֶׁל ״הוֹשַׁע נָא״

הַיּוֹם הַשְּׁבִיעִי שֶׁל סֻכּוֹת נִקְרָא ״הוֹשַׁעְנָא רַבָּה״. בְּחַג הַסֻּכּוֹת נִדּוֹנִים עַל הַמַּיִם – כַּמָּה גֶּשֶׁם יֵרֵד בַּשָּׁנָה הַבָּאָה – וּבַיּוֹם הָאַחֲרוֹן שֶׁל הֶחָג הַדִּין נֶחְתָּם. לָכֵן מַרְבִּים בּוֹ בִּתְפִלּוֹת ״הוֹשַׁע נָא״, וְלוֹקְחִים בּוֹ עֲרָבוֹת.
🤔 לָמָּה? מִן הַמַּיִם חַיִּים הַצְּמָחִים, הַבְּהֵמוֹת וּבְנֵי הָאָדָם. וְהָעֲרָבוֹת צְרִיכוֹת מַיִם יוֹתֵר מִכֻּלָּם – כְּשֶׁאֵין לָהֶן מַיִם, הֵן מִתְיַבְּשׁוֹת מִיָּד.
(פֶּרֶק ו׳, הֲלָכָה א׳)

2הַשָּׁלָב הָאַחֲרוֹן שֶׁל הַדִּין

״אֵין מַיִם אֶלָּא תּוֹרָה״ – כְּמוֹ שֶׁהַמַּיִם נוֹתְנִים חַיִּים לַגּוּף, הַתּוֹרָה נוֹתֶנֶת חַיִּים לַנְּשָׁמָה. לָכֵן הַדִּין עַל הַמַּיִם כּוֹלֵל אֶת כָּל הַחַיִּים. חֲכָמִים אָמְרוּ שֶׁיֵּשׁ שְׁלֹשָׁה זְמַנִּים לַדִּין: רֹאשׁ הַשָּׁנָה, יוֹם הַכִּפּוּרִים וְהוֹשַׁעְנָא רַבָּה. גְּזַר הַדִּין נֶחְתָּם בְּיוֹם הַכִּפּוּרִים, וּבְהוֹשַׁעְנָא רַבָּה נִכְתָּבִים הַ״פְּתָקִים״ וְנִמְסָרִים לַשְּׁלִיחִים. עַד אָז עוֹד אֶפְשָׁר לְשַׁנּוֹת לְטוֹבָה – וְלָכֵן נָכוֹן לַחֲזֹר בִּתְשׁוּבָה בַּיּוֹם הַזֶּה.
🤔 לָמָּה? כְּמוֹ בֵּית דִּין שֶׁכְּבָר הֶחְלִיט, אֲבָל כָּל עוֹד הַפְּסַק לֹא נִכְתַּב וְלֹא נִשְׁלַח – עוֹד אֶפְשָׁר לְבַקֵּשׁ. וְגַם אַחֲרֵי הוֹשַׁעְנָא רַבָּה, עַד סוֹף שְׁמִינִי עֲצֶרֶת, הַתְּשׁוּבָה עוֹד מוֹעִילָה.
💛 ה׳ נוֹתֵן לָנוּ עוֹד הִזְדַּמְּנוּת וְעוֹד הִזְדַּמְּנוּת – כִּי הוּא רוֹצֶה לְהֵיטִיב לָנוּ.
(פֶּרֶק ו׳, הֲלָכָה א׳)

3לֵיל הוֹשַׁעְנָא רַבָּה

הַמִּנְהָג הָרוֹוֵחַ הוּא לִלְמֹד תּוֹרָה בְּלֵיל הוֹשַׁעְנָא רַבָּה, מִתּוֹךְ שִׂמְחַת הֶחָג – אֲבָל בְּלִי רִקּוּדִים עִם תִּזְמֹרֶת. מִנְהַג חֲסִידוּת לְהִשָּׁאֵר עֵרִים כָּל הַלַּיְלָה וְלַעֲסֹק בַּתּוֹרָה. עַל פִּי הָאֲרִ״י סֻדַּר ״תִּקּוּן״ – קְרִיאַת סֵפֶר דְּבָרִים וְסֵפֶר תְּהִלִּים. וּבַיּוֹם עוֹרְכִים סְעוּדָה חֲשׁוּבָה, מִפְּנֵי קְדֻשַּׁת הַיּוֹם.
🤔 לָמָּה? זֶה הַלַּיְלָה לִפְנֵי גְּמַר הַדִּין, וְהַלִּמּוּד מְזַכֵּךְ אֶת הַנֶּפֶשׁ וּמְתַקֵּן אֶת מָה שֶׁצָּרִיךְ תִּקּוּן.
(פֶּרֶק ו׳, הֲלָכָה א׳)

4שֶׁבַע הַקָּפוֹת

בְּכָל יוֹם שֶׁל סֻכּוֹת מַקִּיפִים אֶת הַבִּימָה פַּעַם אַחַת עִם אַרְבַּעַת הַמִּינִים, וּבְהוֹשַׁעְנָא רַבָּה – שֶׁבַע פְּעָמִים, וּמַרְבִּים בִּתְחִנּוֹת אַחֲרֵי הַהַקָּפָה הַשְּׁבִיעִית.
(פֶּרֶק ו׳, הֲלָכָה א׳)

5הָעֲרָבָה בַּמִּקְדָּשׁ

חוּץ מִן הָעֲרָבוֹת שֶׁבַּלּוּלָב, הָיְתָה בַּמִּקְדָּשׁ מִצְוָה מְיֻחֶדֶת – ״הֲלָכָה לְמֹשֶׁה מִסִּינַי״: הֵבִיאוּ עֲרָבוֹת אֲרֻכּוֹת וְזָקְפוּ אוֹתָן בְּצִדֵּי הַמִּזְבֵּחַ, וְרָאשֵׁיהֶן כְּפוּפִים עַל גַּבֵּי הַמִּזְבֵּחַ, וְהַכֹּהֲנִים תָּקְעוּ. הַנְּבִיאִים הִנְהִיגוּ שֶׁבְּכָל מָקוֹם יִטְּלוּ עֲרָבוֹת וְיַחְבְּטוּ בָּהֶן. וְאַחֲרֵי הַחֻרְבָּן מַמְשִׁיכִים לִטֹּל עֲרָבוֹת בַּיּוֹם הַשְּׁבִיעִי – זֵכֶר לַמִּקְדָּשׁ.
🤔 לָמָּה? בַּמִּקְדָּשׁ לָקְחוּ עֲרָבָה בְּכָל יְמֵי הֶחָג, אֲבָל עִקַּר מִצְוַת הָעֲרָבָה – בַּיּוֹם הַשְּׁבִיעִי, שֶׁבּוֹ הִקִּיפוּ אֶת הַמִּזְבֵּחַ שֶׁבַע פְּעָמִים.
(פֶּרֶק ו׳, הֲלָכָה ב׳)

6לְעוֹלָם לֹא בְּשַׁבָּת

מִנְהַג הָעֲרָבָה כָּל כָּךְ חָשׁוּב, שֶׁהַלּוּחַ הָעִבְרִי נִקְבַּע כָּךְ שֶׁהוֹשַׁעְנָא רַבָּה לְעוֹלָם לֹא יָחוּל בְּשַׁבָּת.
🤔 לָמָּה? כְּדֵי שֶׁמִּנְהַג הָעֲרָבוֹת לֹא יִתְבַּטֵּל, וּכְדֵי שֶׁנּוּכַל לְהִתְחַנֵּן וְלִזְעֹק עַל הַמַּיִם.
(פֶּרֶק ו׳, הֲלָכָה א׳, הֲלָכָה ב׳)

7הָעֲרָבָה – עֲנָוָה וּתְשׁוּבָה

הָעֲרָבָה, שֶׁאֵין בָּהּ טַעַם וְרֵיחַ, רוֹמֶזֶת לְבַעַל תְּשׁוּבָה: מִי שֶׁמַּכִּיר בְּחֶסְרוֹנוֹתָיו, וְיוֹדֵעַ שֶׁרַק ה׳ יָכוֹל לְהוֹשִׁיעַ אוֹתוֹ. לָכֵן הָעֲרָבוֹת נִקְרָאוֹת ״הוֹשַׁעְנוֹת״. וְגַם הַחֲבָטָה מְבַטֵּאת אֶת הָרָצוֹן שֶׁלָּנוּ ״לַחְבֹּט״ בַּיֵּצֶר הָרַע וְלַעֲבֹד אֶת ה׳ בְּלֵב שָׁלֵם.
🤔 לָמָּה? חֲכָמִים אָמְרוּ: ״בִּמְקוֹם שֶׁבַּעֲלֵי תְּשׁוּבָה עוֹמְדִים – צַדִּיקִים גְּמוּרִים אֵינָם עוֹמְדִים״. מִי שֶׁיּוֹדֵעַ שֶׁהוּא תָּלוּי לְגַמְרֵי בַּה׳ – קָשׁוּר אֵלָיו מְאוֹד.
💛 גַּם כְּשֶׁאֲנִי מַרְגִּישׁ כְּמוֹ עֲרָבָה פְּשׁוּטָה, ה׳ פּוֹתֵחַ לִי יָד וּמְחַכֶּה שֶׁאֶחֱזֹר אֵלָיו.
(פֶּרֶק ו׳, הֲלָכָה ב׳)

8חִבּוּט הָעֲרָבָה

חִבּוּט הָעֲרָבָה הוּא מִנְהָג שֶׁל הַנְּבִיאִים, וְלָכֵן לֹא מְבָרְכִים עָלָיו. אֹרֶךְ כָּל עֲרָבָה – שְׁלֹשָׁה טְפָחִים, כְּמוֹ הָעֲרָבָה שֶׁבַּלּוּלָב. מִן הַדִּין מַסְפִּיקָה עֲרָבָה אַחַת, וְנוֹהֲגִים עַל פִּי הָאֲרִ״י לִטֹּל חָמֵשׁ עֲרָבוֹת קְשׁוּרוֹת יַחַד. מִן הַדִּין מַסְפִּיק שֶׁיִּשָּׁאֵר עָלֶה אֶחָד, אֲבָל נָכוֹן לְהַדֵּר בַּעֲרָבוֹת יָפוֹת. אוֹתָן עֲרָבוֹת יְכוֹלוֹת לְשַׁמֵּשׁ גַּם אֲנָשִׁים נוֹסָפִים, אֲבָל לְכַתְּחִלָּה לוֹקְחִים עֲרָבוֹת חֲדָשׁוֹת.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְפִי הַשֻּׁלְחָן עָרוּךְ חוֹבְטִים בָּעֲרָבָה פַּעֲמַיִם אוֹ שָׁלֹשׁ עַל הַקַּרְקַע אוֹ עַל כְּלִי.
עַל פִּי הָאֲרִ״י חוֹבְטִים חָמֵשׁ פְּעָמִים עַל הַקַּרְקַע. יֵשׁ שֶׁחוֹבְטִים קֹדֶם עַל הַקַּרְקַע וְאַחַר כָּךְ גַּם עַל סַפְסָל, כְּדֵי שֶׁיִּנְשְׁרוּ עָלִים.
(פֶּרֶק ו׳, הֲלָכָה ג׳)

9סֵדֶר הַיּוֹם בְּבֵית הַכְּנֶסֶת

נוֹהֲגִים לְהַקִּיף אֶת שֶׁבַע הַהַקָּפוֹת עִם אַרְבַּעַת הַמִּינִים בִּלְבַד. אַחַר כָּךְ מַנִּיחִים אֶת אַרְבַּעַת הַמִּינִים, לוֹקְחִים אֶת הָעֲרָבוֹת, אוֹמְרִים עוֹד תְּפִלּוֹת – וְחוֹבְטִים. וְעוֹלֵי תֵּימָן אוֹחֲזִים בַּהַקָּפוֹת גַּם אֶת הַלּוּלָב וְגַם אֶת הָעֲרָבוֹת. אֶת הָעֲרָבָה שֶׁקְּשׁוּרָה בַּלּוּלָב אֵין חוֹבְטִים – אֲבָל אִם מוֹצִיאִים אוֹתָהּ מִן הָאֲגֻדָּה אַחֲרֵי הַנְּטִילָה, אֶפְשָׁר לַחְבֹּט בָּהּ.
(פֶּרֶק ו׳, הֲלָכָה ג׳)

10אַחֲרֵי הֶחָג – אַרְבָּעָה מִינִים וַעֲרָבוֹת

כָּל שִׁבְעַת הַיָּמִים הַמִּינִים מֻקְצִים, וְגַם בְּהוֹשַׁעְנָא רַבָּה אָסוּר לֶאֱכֹל אֶת הָאֶתְרוֹג. מִשֶּׁנִּגְמָר הַיּוֹם הַשְּׁבִיעִי – מֻתָּר לֶאֱכֹל אוֹתוֹ וּלְהִשְׁתַּמֵּשׁ בַּמִּינִים. הֵם ״תַּשְׁמִישֵׁי מִצְוָה״ וְאֵינָם צְרִיכִים גְּנִיזָה, אֲבָל אָסוּר לְבַזּוֹת אוֹתָם: מַנִּיחִים אוֹתָם בְּצַד הֶחָצֵר, בַּמָּקוֹם שֶׁמַּנִּיחִים עֲנָפִים גְּזוּמִים – וְלֹא בְּאַשְׁפָּה מְטֻנֶּפֶת וְלֹא בְּמָקוֹם שֶׁדּוֹרְכִים עֲלֵיהֶם. יֵשׁ נוֹהֲגִים לִשְׁמֹר אֶת הָעֲרָבוֹת עַד פֶּסַח, כְּדֵי לִשְׂרֹף בָּהֶן אֶת הֶחָמֵץ.
🤔 לָמָּה? עָשִׂינוּ בָּהֶם מִצְוָה – וְלָכֵן נוֹהֲגִים בָּהֶם כָּבוֹד גַּם אַחֲרֶיהָ.
(פֶּרֶק ו׳, הֲלָכָה ד׳)

11נִפְרָדִים מִן הַסֻּכָּה

לֹא מְפָרְקִים אֶת הַסֻּכָּה עַד סוֹף הֶחָג – גַּם לֹא אַחֲרֵי הַסְּעוּדָה שֶׁל הוֹשַׁעְנָא רַבָּה. כְּשָׁעָתַיִם וָחֵצִי לִפְנֵי הַשְּׁקִיעָה מֻתָּר לְפַנּוֹת אֶת הָרָהִיטִים לִקְרַאת שְׁמִינִי עֲצֶרֶת. רָאוּי לָשֶׁבֶת בַּסֻּכָּה בַּשָּׁעוֹת הָאַחֲרוֹנוֹת, וְיֵשׁ נוֹהֲגִים לְנַשֵּׁק אֶת הַסֻּכָּה כְּשֶׁנִּפְרָדִים מִמֶּנָּה, וְלוֹמַר תְּפִלַּת פְּרֵדָה מִן הַסִּדּוּר. הַסְּכָךְ, הַדְּפָנוֹת וְהַקִּשּׁוּטִים מֻקְצִים עַד סוֹף שְׁמִינִי עֲצֶרֶת, וְגַם אַחֲרֵי כֵן לֹא מְבַזִּים אוֹתָם.
🤔 לָמָּה? עַד הָעֶרֶב עוֹד יֵשׁ מִצְוָה לָשֶׁבֶת בַּסֻּכָּה – וְכָל שִׂיחָה אוֹ לִמּוּד בָּהּ הֵם מִצְוָה. וּמִי שֶׁעוֹמֵד לְהִפָּרֵד מִן הַמִּצְוָה לְשָׁנָה שְׁלֵמָה – רוֹצֶה לִהְיוֹת בְּצִלָּהּ עוֹד רֶגַע.
💛 הַנְּשִׁיקָה לַסֻּכָּה הִיא תּוֹדָה עַל שִׁבְעָה יָמִים שֶׁגַּרְנוּ בְּצֵל הָאֱמוּנָה.
(פֶּרֶק ו׳, הֲלָכָה ה׳)

12בִּשְׁמִינִי עֲצֶרֶת – לֹא בַּסֻּכָּה

בְּאֶרֶץ יִשְׂרָאֵל אָסוּר לֶאֱכֹל בַּסֻּכָּה בִּשְׁמִינִי עֲצֶרֶת. מִי שֶׁאֵין לוֹ מָקוֹם אַחֵר לֶאֱכֹל – מֵסִיר מִן הַסְּכָךְ לִפְנֵי הֶחָג חֵלֶק שֶׁל אַרְבָּעָה טְפָחִים עַל אַרְבָּעָה טְפָחִים, וְכָךְ רוֹאִים שֶׁהוּא אֵינוֹ יוֹשֵׁב בָּהּ לְשֵׁם מִצְוָה. וּבְחוּץ לָאָרֶץ אוֹכְלִים בַּסֻּכָּה בִּשְׁמִינִי עֲצֶרֶת בְּלִי בְּרָכָה, וּבַיּוֹם הַתְּשִׁיעִי – שִׂמְחַת תּוֹרָה – אֵין אוֹכְלִים בָּהּ.
🤔 לָמָּה? ״לֹא תֹסֵף עָלָיו וְלֹא תִגְרַע מִמֶּנּוּ״ – מִי שֶׁיּוֹשֵׁב בַּסֻּכָּה לְשֵׁם מִצְוָה בְּיוֹם שֶׁאֵינוֹ מִשִּׁבְעַת יְמֵי הֶחָג, מוֹסִיף עַל מִצְוַת ה׳.
(פֶּרֶק ו׳, הֲלָכָה ה׳)

📝 מָה לָמַדְנוּ

  • בְּהוֹשַׁעְנָא רַבָּה נֶחְתָּם הַדִּין עַל הַמַּיִם, וְעוֹד אֶפְשָׁר לְשַׁנּוֹת לְטוֹבָה בִּתְשׁוּבָה.
  • מַקִּיפִים אֶת הַבִּימָה שֶׁבַע פְּעָמִים, וְאַחַר כָּךְ חוֹבְטִים בַּעֲרָבוֹת – מִנְהַג נְבִיאִים, בְּלִי בְּרָכָה.
  • הוֹשַׁעְנָא רַבָּה לְעוֹלָם אֵינוֹ חָל בְּשַׁבָּת, כְּדֵי שֶׁמִּנְהַג הָעֲרָבָה לֹא יִתְבַּטֵּל.
  • לֹא מְפָרְקִים אֶת הַסֻּכָּה עַד סוֹף הֶחָג, וְנִפְרָדִים מִמֶּנָּה בְּכָבוֹד.
  • בִּשְׁמִינִי עֲצֶרֶת בְּאֶרֶץ יִשְׂרָאֵל לֹא אוֹכְלִים בַּסֻּכָּה – ״בַּל תּוֹסִיף״.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה דַּוְקָא בְּהוֹשַׁעְנָא רַבָּה לוֹקְחִים עֲרָבוֹת? תֵּן שְׁנֵי טְעָמִים.
  2. מָה עוֹשִׂים עִם הָעֲרָבוֹת וְהָאֶתְרוֹג אַחֲרֵי הֶחָג, וְאֵיפֹה אָסוּר לְהַנִּיחַ אוֹתָם?
  3. לָמָּה אָסוּר לֶאֱכֹל בַּסֻּכָּה בִּשְׁמִינִי עֲצֶרֶת בְּאֶרֶץ יִשְׂרָאֵל?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״הוֹשַׁעְנָא רַבָּה וְשִׂמְחַת תּוֹרָה״

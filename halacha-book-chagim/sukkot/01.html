סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת סֻכּוֹת
מַהֲדוּרַת יְלָדִים

🌿 חַג הַסֻּכּוֹת – זְמַן שֶׁל שִׂמְחָה

פֶּרֶק א׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
הָאֲוִיר מִתְקָרֵר, רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים כְּבָר מֵאֲחוֹרֵינוּ, וּפִתְאֹם בְּכָל חָצֵר וּבְכָל מִרְפֶּסֶת צוֹמְחִים בָּתִּים קְטַנִּים עִם גַּג שֶׁל עֲנָפִים. לָמָּה יוֹצְאִים מִן הַבַּיִת הֶחָם וְהַנּוֹחַ לִצְרִיף שֶׁל שִׁבְעָה יָמִים? וְלָמָּה דַּוְקָא בְּסֻכּוֹת הַתּוֹרָה מַזְכִּירָה שִׂמְחָה שָׁלֹשׁ פְּעָמִים? בַּפֶּרֶק הַזֶּה נְגַלֶּה אֶת הַסּוֹדוֹת שֶׁל חַג הַסֻּכּוֹת.

1שְׁלֹשָׁה שֵׁמוֹת לֶחָג

לֶחָג יֵשׁ שְׁלֹשָׁה שֵׁמוֹת: ״חַג הַסֻּכּוֹת״ – עַל שֵׁם מִצְוַת הַסֻּכָּה. ״חַג הָאָסִיף״ – כִּי בּוֹ מְסַיְּמִים לֶאֱסֹף אֶת הַתְּבוּאָה וְהַפֵּרוֹת מִן הַשָּׂדוֹת אֶל הַבָּתִּים וְהַמַּחְסָנִים. וְלִפְעָמִים הוּא נִקְרָא פָּשׁוּט ״הֶחָג״ – כְּמוֹ שֶׁכָּתוּב: ״וַיַּעַשׂ שְׁלֹמֹה אֶת הֶחָג״.
🤔 לָמָּה? הוּא הַשָּׂמֵחַ וְהֶחָגִיגִי שֶׁבְּכָל הַחַגִּים. בַּמִּקְדָּשׁ הָיוּ בּוֹ רִקּוּדִים שֶׁל שִׂמְחַת בֵּית הַשּׁוֹאֵבָה, וְהִקְרִיבוּ בּוֹ יוֹתֵר קָרְבָּנוֹת מִבְּכָל חַג אַחֵר.
(פֶּרֶק א׳, הֲלָכָה א׳, הֲלָכָה ב׳)

2שְׁלֹשָׁה דְּבָרִים בְּחַג אֶחָד

בְּחַג הַסֻּכּוֹת יֵשׁ שְׁלֹשָׁה דְּבָרִים גְּדוֹלִים, וְכֻלָּם קְשׁוּרִים זֶה בָּזֶה: קְדֻשַּׁת הַיָּמִים – הַיּוֹם הָרִאשׁוֹן וּשְׁמִינִי עֲצֶרֶת הֵם יָמִים טוֹבִים, וּבְחֹל הַמּוֹעֵד נִמְנָעִים מֵחֵלֶק מִן הַמְּלָאכוֹת; מִצְוַת אַרְבַּעַת הַמִּינִים; וּמִצְוַת הַסֻּכָּה, שֶׁעַל שְׁמָהּ נִקְרָא הֶחָג.
🤔 לָמָּה? כָּל אֶחָד מֵהֶם מוֹסִיף לַשִּׂמְחָה: מוֹדִים לַה׳ עַל הַיְּבוּל, שְׂמֵחִים עַל הַתְּשׁוּבָה וְהַכַּפָּרָה שֶׁל הַיָּמִים הַנּוֹרָאִים, וְזוֹכְרִים אֵיךְ ה׳ הִשְׁגִּיחַ עַל עַמּוֹ כְּשֶׁיָּצָא מִמִּצְרַיִם.
(פֶּרֶק א׳, הֲלָכָה א׳)

3שִׂמְחָה שְׁלֵמָה

בְּחַג הַפֶּסַח הַתּוֹרָה לֹא הִזְכִּירָה שִׂמְחָה, בְּחַג הַשָּׁבוּעוֹת הִיא הִזְכִּירָה שִׂמְחָה פַּעַם אַחַת, וּבְחַג הַסֻּכּוֹת – שָׁלֹשׁ פְּעָמִים! עַכְשָׁו כְּבָר אָסַפְנוּ אֶת כָּל הַיְּבוּל שֶׁל הַשָּׁנָה, וְגַם עָבַרְנוּ אֶת רֹאשׁ הַשָּׁנָה וְיוֹם הַכִּפּוּרִים וְנִקִּינוּ אֶת הַלֵּב בִּתְשׁוּבָה – וְהַשִּׂמְחָה שְׁלֵמָה.
🤔 לָמָּה? בְּפֶסַח עוֹד לֹא יוֹדְעִים אִם הַתְּבוּאָה תַּצְלִיחַ, וּבְשָׁבוּעוֹת עוֹד דּוֹאֲגִים לַפֵּרוֹת. בְּסֻכּוֹת הַכֹּל כְּבָר אָסוּף: גַּם הַפֵּרוֹת שֶׁבַּשָּׂדֶה, וְגַם הַתּוֹרָה וְהַמַּעֲשִׂים הַטּוֹבִים שֶׁאָסַפְנוּ כָּל הַשָּׁנָה.
💛 הַשִּׂמְחָה שֶׁל סֻכּוֹת הִיא שִׂמְחָה שֶׁל מִי שֶׁחָזַר אֶל ה׳ – וְיוֹדֵעַ שֶׁה׳ קִבֵּל אוֹתוֹ בְּאַהֲבָה.
(פֶּרֶק א׳, הֲלָכָה ב׳, הֲלָכָה ח׳)

4אַרְבָּעָה מִינִים – תּוֹדָה וְנִצָּחוֹן

כְּשֶׁאָדָם אוֹסֵף יְבוּל רַב, הוּא שָׂמֵחַ מְאוֹד. כְּדֵי שֶׁהַשִּׂמְחָה תִּהְיֶה קְדוֹשָׁה, הַתּוֹרָה צִוְּתָה לִטֹּל אַרְבָּעָה מִינִים – תּוֹדָה לְמִי שֶׁבָּרָא אֶת הָעוֹלָם וּמְקַיֵּם אוֹתוֹ. וּמְנַעְנְעִים אוֹתָם לְאַרְבַּע הָרוּחוֹת, לְמַעְלָה וּלְמַטָּה – כִּי הַשָּׁמַיִם וְהָאָרֶץ וְכָל הָרוּחוֹת שַׁיָּכִים לַה׳. וְיֵשׁ בְּכָךְ גַּם תְּפִלָּה: שֶׁהַיְּבוּל שֶׁל הַשָּׁנָה הַבָּאָה יַצְלִיחַ, וְשֶׁה׳ יַצִּיל אוֹתָנוּ מֵרוּחוֹת רָעוֹת.
🤔 לָמָּה? חֲכָמִים אָמְרוּ מָשָׁל: שְׁנַיִם נִכְנְסוּ לְמִשְׁפָּט, וְלֹא יָדְעוּ מִי נִצַּח – עַד שֶׁאֶחָד יָצָא וְהֵנִיף אֶת הָרֹמַח שֶׁלּוֹ. כָּךְ יִשְׂרָאֵל: אַחֲרֵי יְמֵי הַדִּין, כְּשֶׁהֵם יוֹצְאִים וְהַלּוּלָבִים וְהָאֶתְרוֹגִים בִּידֵיהֶם – כֻּלָּם רוֹאִים שֶׁהֵם נִצְּחוּ בַּדִּין.
💛 הַלּוּלָב בַּיָּד שֶׁלְּךָ הוּא כְּמוֹ דֶּגֶל שֶׁאוֹמֵר: חָזַרְתִּי אֶל ה׳, וַאֲנִי שֶׁלּוֹ.
(פֶּרֶק א׳, הֲלָכָה ג׳)

5זֵכֶר לְעַנְנֵי הַכָּבוֹד

הַתּוֹרָה אוֹמֶרֶת: ״לְמַעַן יֵדְעוּ דֹרֹתֵיכֶם כִּי בַסֻּכּוֹת הוֹשַׁבְתִּי אֶת בְּנֵי יִשְׂרָאֵל בְּהוֹצִיאִי אוֹתָם מֵאֶרֶץ מִצְרָיִם״. לְדַעַת רַבִּי אֱלִיעֶזֶר – הַכַּוָּנָה לְעַנְנֵי הַכָּבוֹד שֶׁהֵגֵנּוּ עַל יִשְׂרָאֵל בַּמִּדְבָּר. לְדַעַת רַבִּי עֲקִיבָא – לַסֻּכּוֹת מַמָּשׁ שֶׁעָשׂוּ יִשְׂרָאֵל כְּשֶׁיָּצְאוּ מִמִּצְרַיִם. לְמַעֲשֶׂה, כְּשֶׁיּוֹשְׁבִים בַּסֻּכָּה זוֹכְרִים שֶׁהִיא זֵכֶר לְעַנְנֵי הַכָּבוֹד וְגַם זֵכֶר לִיצִיאַת מִצְרַיִם. וּמִי שֶׁשָּׁכַח לְכַוֵּן לָזֶה – יָצָא, אִם יָדַע שֶׁהוּא מְקַיֵּם מִצְוַת ה׳.
🤔 לָמָּה? חֲכָמִים אָמְרוּ שֶׁשִּׁבְעָה עַנְנֵי כָּבוֹד הָיוּ עִם יִשְׂרָאֵל: אַרְבָּעָה מֵאַרְבַּע הָרוּחוֹת, אֶחָד מִלְמַעְלָה, אֶחָד מִלְמַטָּה, וְאֶחָד שֶׁהָלַךְ לִפְנֵיהֶם וְתִקֵּן לָהֶם אֶת הַדֶּרֶךְ. וְזָכוּ לָזֶה כִּי הָלְכוּ אַחֲרֵי ה׳ בַּמִּדְבָּר, ״בְּאֶרֶץ לֹא זְרוּעָה״.
💛 ה׳ לֹא רַק נָתַן לָנוּ מָן וּמַיִם בַּמִּדְבָּר – הוּא עָטַף אוֹתָנוּ בַּעֲנָנָיו, כְּמוֹ אַבָּא שֶׁמְּחַבֵּק אֶת יְלָדָיו.
(פֶּרֶק א׳, הֲלָכָה ד׳, הֲלָכָה ה׳)

6לָמָּה דַּוְקָא בַּסְּתָו?

יָצָאנוּ מִמִּצְרַיִם בָּאָבִיב, בְּחֹדֶשׁ נִיסָן. אָז לָמָּה יוֹשְׁבִים בַּסֻּכָּה בַּסְּתָו? כִּי בָּאָבִיב, כְּשֶׁנָּעִים בַּחוּץ, הַרְבֵּה אֲנָשִׁים בּוֹנִים סֻכּוֹת לַהֲנָאָתָם, וְלֹא הָיוּ רוֹאִים שֶׁאֲנַחְנוּ יוֹשְׁבִים בַּסֻּכָּה בִּשְׁבִיל הַמִּצְוָה.
🤔 לָמָּה? בַּסְּתָו, כְּשֶׁמַּתְחִיל לְהִתְקָרֵר וְכֻלָּם נִכְנָסִים הַבַּיְתָה, כָּל אֶחָד רוֹאֶה שֶׁיּוֹצְאִים לַסֻּכָּה רַק כְּדֵי לְקַיֵּם אֶת מִצְוַת ה׳.
(פֶּרֶק א׳, הֲלָכָה ה׳)

7דִּירַת אַרְעַי – מִי שׁוֹמֵר עָלֵינוּ?

דַּוְקָא כְּשֶׁאָסַפְנוּ אֶת כָּל הַיְּבוּל וְהַמַּחְסָנִים מְלֵאִים, יוֹצְאִים לַסֻּכָּה – בַּיִת קָטָן וַאֲרָעִי. הַתּוֹרָה הִזְהִירָה שֶׁכְּשֶׁיִּהְיוּ לָנוּ בָּתִּים טוֹבִים וְשֶׁפַע, לֹא נֹאמַר: ״כֹּחִי וְעֹצֶם יָדִי עָשָׂה לִי אֶת הַחַיִל הַזֶּה״. בַּסֻּכָּה אֲנַחְנוּ זוֹכְרִים שֶׁלֹּא הַקִּירוֹת הַחֲזָקִים שׁוֹמְרִים עָלֵינוּ – ה׳ שׁוֹמֵר עָלֵינוּ.
🤔 לָמָּה? לָכֵן הַסֻּכָּה נִקְרֵאת ״צֵל הָאֱמוּנָה״. וְהִיא בָּאָה בְּדִיּוּק לִפְנֵי הַחֹרֶף, כְּשֶׁאֲנַחְנוּ עוֹמְדִים לְהִסָּגֵר בַּבַּיִת מִפְּנֵי הַקֹּר וְהַגֶּשֶׁם: ״אִם ה׳ לֹא יִבְנֶה בַיִת – שָׁוְא עָמְלוּ בוֹנָיו בּוֹ״.
💛 בַּסֻּכָּה הַקְּטַנָּה אֲנִי מַרְגִּישׁ שֶׁה׳ הוּא הַגַּג הָאֲמִתִּי שֶׁמֵּעָלַי.
(פֶּרֶק א׳, הֲלָכָה ו׳)

8אוֹכְלִים, יְשֵׁנִים – וּמְקַיְּמִים מִצְוָה

מִצְוַת הַסֻּכָּה מְיֻחֶדֶת: כָּל מָה שֶׁעוֹשִׂים בָּהּ – אוֹכְלִים, שׁוֹתִים, מְשׂוֹחֲחִים וִישֵׁנִים – הוֹפֵךְ לְמִצְוָה. חַכְמֵי הַקַּבָּלָה אָמְרוּ שֶׁהַסֻּכָּה הִיא ״אוֹר מַקִּיף״ – אוֹר שֶׁעוֹטֵף אוֹתָנוּ מִכָּל הַצְּדָדִים, גָּדוֹל יוֹתֵר מִמָּה שֶׁאֲנַחְנוּ יְכוֹלִים לְהָבִין.
🤔 לָמָּה? הַתַּפְקִיד שֶׁל עַם יִשְׂרָאֵל הוּא לְגַלּוֹת שֶׁגַּם הַחַיִּים הָרְגִילִים קְשׁוּרִים לַקְּדֻשָּׁה. בַּסֻּכָּה רוֹאִים אֶת זֶה בָּעֵינַיִם: אֲפִלּוּ אֲרוּחַת עֶרֶב הִיא מִצְוָה.
💛 בַּסֻּכָּה אֲנִי לוֹמֵד שֶׁאֶפְשָׁר לִהְיוֹת קָרוֹב לַה׳ גַּם כְּשֶׁאֲנִי סְתָם אוֹכֵל אֲרוּחַת צָהֳרַיִם.
(פֶּרֶק א׳, הֲלָכָה ז׳)

9כָּל יִשְׂרָאֵל בְּסֻכָּה אַחַת

אָמְרוּ חֲכָמִים: ״רְאוּיִים כָּל יִשְׂרָאֵל לֵישֵׁב בְּסֻכָּה אַחַת״. בְּחַג הָאָסִיף נֶאֱסָפִים יַחַד כָּל הַדְּבָרִים הַטּוֹבִים שֶׁל כָּל הַיְּהוּדִים – וְכָךְ מִתְגַּלָּה הָאַחְדוּת. וְעוֹד אָמְרוּ: מִי שֶׁמְּקַיֵּם מִצְוַת סֻכָּה בָּעוֹלָם הַזֶּה – הַקָּדוֹשׁ בָּרוּךְ הוּא יָגֵן עָלָיו לֶעָתִיד לָבוֹא.
🤔 לָמָּה? כָּל עוֹד כָּל אֶחָד עוֹמֵד לְבַד עִם הַמַּעֲלָה שֶׁלּוֹ, אֵין אַחְדוּת. כְּשֶׁכֻּלָּם נֶאֱסָפִים יַחַד, כָּל אֶחָד מַשְׁלִים אֶת הַשֵּׁנִי, וְה׳ פּוֹרֵשׂ עָלֵינוּ ״סֻכַּת שָׁלוֹם״.
(פֶּרֶק א׳, הֲלָכָה ח׳)

10מְגִלַּת קֹהֶלֶת

רַבִּים נוֹהֲגִים לִקְרֹא בְּסֻכּוֹת אֶת מְגִלַּת קֹהֶלֶת. הַמְּגִלָּה מְלַמֶּדֶת שֶׁעֹשֶׁר, בָּתִּים גְּדוֹלִים וּבְגָדִים יָפִים הֵם ״הֲבֵל הֲבָלִים״ – וְרַק דָּבָר אֶחָד חָשׁוּב בֶּאֱמֶת: ״אֶת הָאֱלֹהִים יְרָא וְאֶת מִצְוֹתָיו שְׁמוֹר, כִּי זֶה כָּל הָאָדָם״. וְהִיא מְלַמֶּדֶת אֵיךְ לִשְׂמֹחַ בֶּאֱמֶת – בְּשִׂמְחָה שֶׁל מִצְוָה.
🤔 לָמָּה? מָשָׁל יָפֶה: ״הֶבֶל״ הוּא כְּמוֹ הַסִּפְרָה אֶפֶס, וְיִרְאַת ה׳ הִיא הַסִּפְרָה אַחַת. כְּשֶׁיִּרְאַת ה׳ בָּרֹאשׁ, כָּל מַעֲלָה שֶׁמִּצְטָרֶפֶת – חָכְמָה, עֹשֶׁר, יֹפִי – מוֹסִיפָה אֶפֶס, וְהָאַחַת הוֹפֶכֶת לְעֶשֶׂר, לְמֵאָה וּלְאֶלֶף. אֲבָל בְּלִי הָאַחַת בָּרֹאשׁ – הַכֹּל רַק אֲפָסִים.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג אַשְׁכְּנַז קוֹרְאִים אֶת קֹהֶלֶת בְּשַׁבָּת שֶׁחָלָה בְּחַג הַסֻּכּוֹת, בִּתְפִלַּת שַׁחֲרִית, לִפְנֵי קְרִיאַת הַתּוֹרָה. רֹב הָאַשְׁכְּנַזִּים אֵינָם מְבָרְכִים עַל הַקְּרִיאָה, וְתַלְמִידֵי הַגְּרָ״א קוֹרְאִים מִתּוֹךְ מְגִלַּת קְלָף וּמְבָרְכִים.
לְמִנְהַג רֹב הַסְּפָרַדִּים לֹא נָהֲגוּ לִקְרֹא קֹהֶלֶת בְּסֻכּוֹת. וְהַתֵּימָנִים קוֹרְאִים חֵלֶק מִמֶּנָּה סָמוּךְ לְמִנְחָה שֶׁל שַׁבָּת, וְחֵלֶק בְּיוֹם טוֹב אַחֲרוֹן.
(פֶּרֶק א׳, הֲלָכָה ט׳)

11שִׂמְחַת בֵּית הַשּׁוֹאֵבָה בַּמִּקְדָּשׁ

בְּכָל הַשָּׁנָה נִסְּכוּ עַל הַמִּזְבֵּחַ יַיִן, וְרַק בְּסֻכּוֹת נִסְּכוּ גַּם מַיִם – מַיִם שֶׁשָּׁאֲבוּ מִמַּעְיַן הַשִּׁילוֹחַ. עַל הַשְּׁאִיבָה הַזּוֹ הָיְתָה בַּמִּקְדָּשׁ שִׂמְחָה גְּדוֹלָה כָּל הַלַּיְלָה: מְנוֹרוֹת זָהָב עֲנָקִיּוֹת הֵאִירוּ אֶת כָּל חַצְרוֹת יְרוּשָׁלַיִם, הַלְוִיִּם נִגְּנוּ בַּחֲלִילִים וּבְכִנּוֹרוֹת, וַחֲכָמִים וַחֲסִידִים רָקְדוּ עִם אֲבוּקוֹת שֶׁל אֵשׁ. רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל הָיָה זוֹרֵק שְׁמוֹנֶה אֲבוּקוֹת וְתוֹפֵס אוֹתָן. כָּל הָעָם, אֲנָשִׁים וְנָשִׁים, בָּאוּ לִרְאוֹת וְלִשְׁמֹעַ, וּכְדֵי שֶׁלֹּא תִּהְיֶה קַלּוּת רֹאשׁ, בָּנוּ מִרְפָּסוֹת: הַנָּשִׁים עָמְדוּ לְמַעְלָה וְהַגְּבָרִים לְמַטָּה. אָמְרוּ חֲכָמִים: ״מִי שֶׁלֹּא רָאָה שִׂמְחַת בֵּית הַשּׁוֹאֵבָה – לֹא רָאָה שִׂמְחָה מִיָּמָיו״.
🤔 לָמָּה? בְּכָל הַשָּׁנָה רַק הַיַּיִן, שֶׁהוּא מַשְׁקֶה מְיֻחָד, הָיָה עוֹלֶה עַל הַמִּזְבֵּחַ. בְּסֻכּוֹת, אַחֲרֵי כָּל הַחַגִּים וְהַתְּשׁוּבָה, גַּם הַמַּיִם הַפְּשׁוּטִים, שֶׁכָּל הַחַיִּים תְּלוּיִים בָּהֶם, נַעֲשִׂים קְדוֹשִׁים. וּבְסֻכּוֹת נִדּוֹנִים עַל הַמַּיִם – וְנִסּוּךְ הַמַּיִם מֵבִיא בְּרָכָה לְגִשְׁמֵי הַשָּׁנָה.
💛 חֲכָמִים אָמְרוּ שֶׁמִּתּוֹךְ הַשִּׂמְחָה הַזּוֹ זָכוּ גְּדוֹלֵי יִשְׂרָאֵל לְרוּחַ הַקֹּדֶשׁ – שִׂמְחָה שֶׁל מִצְוָה מְקָרֶבֶת אֶת הָאָדָם אֶל ה׳.
(פֶּרֶק א׳, הֲלָכָה י׳, הֲלָכָה י״א)

12שִׂמְחַת בֵּית הַשּׁוֹאֵבָה הַיּוֹם

גַּם הַיּוֹם עוֹרְכִים בְּלֵילוֹת חֹל הַמּוֹעֵד שִׂמְחַת בֵּית הַשּׁוֹאֵבָה, זֵכֶר לַמִּקְדָּשׁ, עִם נְגִינָה וְרִקּוּדִים. אֵין חוֹבָה לַעֲשׂוֹת אוֹתָהּ, אֲבָל יֵשׁ בָּהּ מִצְוָה: גַּם ״וְשָׂמַחְתָּ בְּחַגֶּךָ״, וְגַם זִכָּרוֹן לַמִּקְדָּשׁ, שֶׁמְּקָרֵב אֶת בִּנְיָנוֹ. יֵשׁ נוֹהֲגִים לוֹמַר בִּתְחִלַּת הַשִּׂמְחָה אֶת חֲמֵשׁ עֶשְׂרֵה הַמִּזְמוֹרִים שֶׁל ״שִׁיר הַמַּעֲלוֹת״ שֶׁבִּתְהִלִּים.
🤔 לָמָּה? הַמִּנְהָג הִתְחַזֵּק כְּשֶׁיְּהוּדִים הִתְחִילוּ לַחֲזֹר לְאֶרֶץ יִשְׂרָאֵל: בִּירוּשָׁלַיִם וּבִצְפַת רָקְדוּ בְּמַעְגָּל עִם תֻּפִּים וַאֲבוּקוֹת, וּמֵאֶרֶץ יִשְׂרָאֵל הִתְפַּשֵּׁט הַמִּנְהָג לְכָל הַגּוֹלָה.
(פֶּרֶק א׳, הֲלָכָה י״ב)

13שִׁבְעִים פָּרִים

בְּשִׁבְעַת יְמֵי הַחַג הִקְרִיבוּ בַּמִּקְדָּשׁ שִׁבְעִים פָּרִים, כְּנֶגֶד שִׁבְעִים אֻמּוֹת הָעוֹלָם: בַּיּוֹם הָרִאשׁוֹן שְׁלֹשָׁה עָשָׂר, בַּשֵּׁנִי שְׁנֵים עָשָׂר, וְכָךְ פּוֹחֲתִים עַד שִׁבְעָה בַּיּוֹם הַשְּׁבִיעִי. וּבִשְׁמִינִי עֲצֶרֶת – פַּר אֶחָד בִּלְבַד, כְּנֶגֶד הָאֻמָּה הַיְּחִידָה – יִשְׂרָאֵל.
🤔 לָמָּה? מִתּוֹךְ אַהֲבָה הִקְרִיבוּ יִשְׂרָאֵל עֲבוּר כָּל הָעַמִּים, כְּדֵי לְהוֹסִיף לָהֶם שֶׁפַע וּבְרָכָה. וְהַנָּבִיא זְכַרְיָה אוֹמֵר שֶׁלֶּעָתִיד לָבוֹא גַּם הַגּוֹיִים יַעֲלוּ לִירוּשָׁלַיִם לָחֹג אֶת חַג הַסֻּכּוֹת.
💛 כְּמוֹ מֶלֶךְ שֶׁאַחֲרֵי סְעוּדָה גְּדוֹלָה לְכֻלָּם אוֹמֵר לְאוֹהֲבוֹ: ״עֲשֵׂה לִי סְעוּדָה קְטַנָּה, כְּדֵי שֶׁאֵהָנֶה מִמְּךָ״ – כָּךְ ה׳ אוֹהֵב אוֹתָנוּ.
(פֶּרֶק א׳, הֲלָכָה י״ג)

14אוּשְׁפִּיזִין – אוֹרְחִים בַּסֻּכָּה

מִצְוָה מִן הַתּוֹרָה לְשַׁתֵּף בְּשִׂמְחַת הַחַג אֶת הָעֲנִיִּים, הַיְּתוֹמִים וְהַבּוֹדְדִים, וּלְהַזְמִין אוֹתָם לַסֻּכָּה. וּבַזֹּהַר נֶאֱמַר שֶׁמַּזְמִינִים לַסֻּכָּה גַּם ״אוּשְׁפִּיזִין עִלָּאִין״ – אוֹרְחִים עֶלְיוֹנִים: שִׁבְעָה צַדִּיקִים, מֵאַבְרָהָם אָבִינוּ וְעַד דָּוִד הַמֶּלֶךְ. בְּכָל יוֹם אֶחָד מֵהֶם נִכְנָס בָּרֹאשׁ. בְּסִדּוּרִים רַבִּים יֵשׁ נֻסָּח לְהַזְמָנַת הָאוּשְׁפִּיזִין.
🤔 לָמָּה? הַזֹּהַר אוֹמֵר שֶׁהַצַּדִּיקִים בָּאִים רַק לְמִי שֶׁנּוֹתֵן גַּם לָעֲנִיִּים אֶת חֶלְקָם, וְקָמִים מִשֻּׁלְחָנוֹ שֶׁל מִי שֶׁשּׁוֹכֵחַ אוֹתָם. וּבַדּוֹר שֶׁלָּנוּ, שֶׁיֵּשׁ פָּחוֹת רְעֵבִים וְיוֹתֵר אֲנָשִׁים עֲצוּבִים וּבוֹדְדִים, מִצְוָה גְּדוֹלָה לְהַזְמִין גַּם אוֹתָם.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג הַסְּפָרַדִּים וְהַחֲסִידִים סֵדֶר הָאוּשְׁפִּיזִין: אַבְרָהָם, יִצְחָק, יַעֲקֹב, מֹשֶׁה, אַהֲרֹן, יוֹסֵף וְדָוִד.
לְמִנְהַג אַשְׁכְּנַז סֵדֶר הָאוּשְׁפִּיזִין: אַבְרָהָם, יִצְחָק, יַעֲקֹב, יוֹסֵף, מֹשֶׁה, אַהֲרֹן וְדָוִד.
💛 כְּשֶׁאֲנִי מְשַׂמֵּחַ אוֹרֵחַ בּוֹדֵד בַּסֻּכָּה שֶׁלִּי, אֲנִי עוֹשֶׂה מָה שֶׁאַבְרָהָם אָבִינוּ עָשָׂה כָּל יָמָיו – וְה׳ שָׂמֵחַ אִתִּי.
(פֶּרֶק א׳, הֲלָכָה י״ד)

📝 מָה לָמַדְנוּ

  • לֶחָג שְׁלֹשָׁה שֵׁמוֹת: חַג הַסֻּכּוֹת, חַג הָאָסִיף, וְ״הֶחָג״ – הַשָּׂמֵחַ שֶׁבַּחַגִּים.
  • הַסֻּכָּה הִיא זֵכֶר לְעַנְנֵי הַכָּבוֹד וְלִיצִיאַת מִצְרַיִם, וּמַזְכִּירָה לָנוּ שֶׁה׳ שׁוֹמֵר עָלֵינוּ.
  • אַרְבַּעַת הַמִּינִים הֵם תּוֹדָה עַל הַיְּבוּל וְ״דֶגֶל נִצָּחוֹן״ אַחֲרֵי יְמֵי הַדִּין.
  • בַּמִּקְדָּשׁ שָׂמְחוּ כָּל הַלַּיְלָה בְּשִׂמְחַת בֵּית הַשּׁוֹאֵבָה, וְגַם הַיּוֹם עוֹשִׂים אוֹתָהּ זֵכֶר לַמִּקְדָּשׁ.
  • מַזְמִינִים לַסֻּכָּה אוֹרְחִים בּוֹדְדִים וַעֲנִיִּים – וְאָז בָּאִים גַּם הָאוּשְׁפִּיזִין, שִׁבְעַת הַצַּדִּיקִים.

❓ שְׁאֵלוֹת חֲזָרָה

  1. מָה הֵם שְׁלֹשֶׁת הַשֵּׁמוֹת שֶׁל הֶחָג, וְלָמָּה נִקְרָא כָּל אֶחָד מֵהֶם כָּךְ?
  2. לָמָּה יוֹשְׁבִים בַּסֻּכָּה בַּסְּתָו וְלֹא בָּאָבִיב, אַף שֶׁיָּצָאנוּ מִמִּצְרַיִם בָּאָבִיב?
  3. מָה נִסְּכוּ עַל הַמִּזְבֵּחַ רַק בְּסֻכּוֹת, וְאֵיךְ שָׂמְחוּ עַל כָּךְ בַּמִּקְדָּשׁ?

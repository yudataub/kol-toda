סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת סֻכּוֹת
מַהֲדוּרַת יְלָדִים

🔨 בּוֹנִים סֻכָּה כְּשֵׁרָה

פֶּרֶק ב׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
מוֹצָאֵי יוֹם הַכִּפּוּרִים. הַצּוֹם נִגְמַר, וּבֶחָצֵר כְּבָר נִשְׁמָעִים דְּפִיקוֹת שֶׁל פַּטִּישׁ: אַבָּא וְהַיְּלָדִים מַתְחִילִים לִבְנוֹת אֶת הַסֻּכָּה! אֲבָל רֶגַע – מִמָּה עוֹשִׂים סְכָךְ? כַּמָּה קִירוֹת צָרִיךְ? וּמָה קוֹרֶה אִם הַסֻּכָּה עוֹמֶדֶת מִתַּחַת לָעֵץ? בַּפֶּרֶק הַזֶּה נִלְמַד אֵיךְ בּוֹנִים סֻכָּה כְּשֵׁרָה וִיפָה.

1לֹא קְטַנָּה מִדַּי וְלֹא גְּבוֹהָה מִדַּי

הַסֻּכָּה הִיא ״דִּירַת אַרְעַי״ – בַּיִת לִזְמַן קָצָר. כְּדֵי שֶׁתִּהְיֶה דִּירָה, גָּבְהָהּ צָרִיךְ לִהְיוֹת לְפָחוֹת עֲשָׂרָה טְפָחִים (כְּ־80 ס״מ), וְרָחְבָּהּ לְפָחוֹת שִׁבְעָה טְפָחִים (כְּ־56 ס״מ). סֻכָּה צָרָה מִזֶּה פְּסוּלָה, גַּם אִם הִיא אֲרֻכָּה מְאוֹד. וְאִם הַסֻּכָּה גְּבוֹהָה מֵעֶשְׂרִים אַמָּה (כְּ־9 מֶטֶר) – הִיא פְּסוּלָה.
🤔 לָמָּה? בְּסֻכָּה קְטַנָּה כָּל כָּךְ אֲפִלּוּ אָדָם אֶחָד אֵינוֹ יָכוֹל לָשֶׁבֶת וְלֶאֱכֹל, וְהִיא אֵינָהּ דִּירָה. וּסְכָךְ בְּגֹבַהּ כָּל כָּךְ גָּדוֹל צָרִיךְ בִּנְיָן חָזָק וְקָבוּעַ שֶׁיַּחְזִיק אוֹתוֹ – וְאָז זוֹ כְּבָר לֹא דִּירַת אַרְעַי.
(פֶּרֶק ב׳, הֲלָכָה א׳)

2סֻכָּה עַל עֲגָלָה

מִפְּנֵי שֶׁהַסֻּכָּה הִיא דִּירַת אַרְעַי, אֶפְשָׁר לִבְנוֹת אוֹתָהּ גַּם עַל עֲגָלָה, עַל מְכוֹנִית אוֹ עַל סְפִינָה – וְהִיא כְּשֵׁרָה גַּם בִּזְמַן הַנְּסִיעָה, אִם הַדְּפָנוֹת וְהַסְּכָךְ עוֹמְדִים בְּרוּחַ רְגִילָה. וְהַסֻּכָּה פְּטוּרָה מִמְּזוּזָה. אֲבָל סֻכָּה שֶׁדְּפָנוֹתֶיהָ רְעוּעוֹת וְנוֹפְלוֹת בְּרוּחַ רְגִילָה, אוֹ שֶׁהַסְּכָךְ שֶׁלָּהּ עָשׂוּי עָלִים שֶׁיִּתְיַבְּשׁוּ וְיִנְשְׁרוּ בְּתוֹךְ הֶחָג – פְּסוּלָה.
🤔 לָמָּה? הָאַרְעִיּוּת צְרִיכָה לִהְיוֹת בַּסְּכָךְ. הַדְּפָנוֹת יְכוֹלוֹת לִהְיוֹת חֲזָקוֹת וּקְבוּעוֹת, וְלָכֵן מֻתָּר אֲפִלּוּ לִפְתֹּחַ אֶת הַתִּקְרָה שֶׁל חֶדֶר בַּבַּיִת וּלְהַנִּיחַ בִּמְקוֹמָהּ סְכָךְ.
(פֶּרֶק ב׳, הֲלָכָה א׳)

3שְׁלֹשָׁה תְּנָאִים לַסְּכָךְ

הַסְּכָךְ הוּא עִקַּר הַסֻּכָּה, וְעַל שְׁמוֹ הִיא נִקְרֵאת ״סֻכָּה״. יֵשׁ לוֹ שְׁלֹשָׁה תְּנָאִים: א. שֶׁיִּגְדַּל מִן הַקַּרְקַע – כְּמוֹ עֵצִים, עֲנָפִים וְקָנִים. ב. שֶׁיִּהְיֶה תָּלוּשׁ – לֹא עֲנָפִים שֶׁעוֹד מְחֻבָּרִים לָעֵץ, וְלֹא גֶּפֶן מְטַפֶּסֶת. ג. שֶׁלֹּא יִהְיֶה כְּלִי – לֹא כִּסְאוֹת, לֹא מִטּוֹת וְלֹא שׁוּם כְּלִי שֶׁעָשׂוּ מִן הָעֵץ.
🤔 לָמָּה? קַרְשִׁים פְּשׁוּטִים וַעֲנָפִים אֵינָם מְקַבְּלִים טֻמְאָה. אֲבָל כְּשֶׁעוֹשִׂים מֵהֶם כְּלִי שֶׁבְּנֵי אָדָם מִשְׁתַּמְּשִׁים בּוֹ, הוּא יָכוֹל לְקַבֵּל טֻמְאָה – וְדָבָר כָּזֶה פָּסוּל לַסְּכָךְ.
(פֶּרֶק ב׳, הֲלָכָה ב׳)

4מָה פָּסוּל לַסְּכָךְ?

בַּרְזֶל, פְּלַסְטִיק וְעָפָר – פְּסוּלִים, כִּי אֵינָם צוֹמְחִים. גַּם עוֹרוֹת שֶׁל בַּעֲלֵי חַיִּים פְּסוּלִים. פֵּרוֹת וִירָקוֹת שֶׁבְּנֵי אָדָם אוֹכְלִים – פְּסוּלִים. שִׁבְרֵי כֵּלִים וּמִטּוֹת – אָסוּר לְסַכֵּךְ בָּהֶם. וְגַם נְיָר, קַרְטוֹן וְצֶמֶר גֶּפֶן פְּסוּלִים, כִּי הֵם עָבְרוּ עִבּוּד וְשִׁנּוּ אֶת צוּרָתָם. מַחְצֶלֶת שֶׁל קָנִים: אִם עָשׂוּ אוֹתָהּ כְּדֵי לִשְׁכַּב עָלֶיהָ – פְּסוּלָה, וְאִם עָשׂוּ אוֹתָהּ כְּדֵי לְסַכֵּךְ בָּהּ – כְּשֵׁרָה.
🤔 לָמָּה? שִׁבְרֵי כֵּלִים כְּבָר אֵינָם מְקַבְּלִים טֻמְאָה, אֲבָל חֲכָמִים אָסְרוּ אוֹתָם – שֶׁמָּא יִטְעוּ וִיסַכְּכוּ גַּם בְּכֵלִים שְׁלֵמִים.
(פֶּרֶק ב׳, הֲלָכָה ב׳)

5צִלָּתָהּ מְרֻבָּה מֵחַמָּתָהּ

הַסְּכָךְ צָרִיךְ לְהָגֵן מִן הַשֶּׁמֶשׁ: לְמַעְלָה, בַּסְּכָךְ עַצְמוֹ, צָרִיךְ שֶׁיִּהְיֶה יוֹתֵר צֵל מִשֶּׁמֶשׁ. מוֹדְדִים לְמַעְלָה, וְלֹא לְפִי כִּתְמֵי הָאוֹר עַל הָרִצְפָּה. לְכַתְּחִלָּה טוֹב שֶׁהַצֵּל יִהְיֶה רַב, כְּדֵי שֶׁיִּהְיֶה נָעִים לָשֶׁבֶת, וְעִם זֹאת שֶׁיִּרְאוּ דֶּרֶךְ הַסְּכָךְ כּוֹכָבִים בַּלַּיְלָה. סְכָךְ אָטוּם כָּל כָּךְ שֶׁגַּם גֶּשֶׁם אֵינוֹ עוֹבֵר בּוֹ – יֵשׁ פּוֹסְלִים אוֹתוֹ, וּכְדַאי לְדַלֵּל אוֹתוֹ.
🤔 לָמָּה? קַרְנֵי הַשֶּׁמֶשׁ מִתְרַחֲבוֹת בַּדֶּרֶךְ לְמַטָּה, וְלָכֵן עַל הָרִצְפָּה נִדְמֶה שֶׁיֵּשׁ הַרְבֵּה אוֹר – אֲבָל בֶּאֱמֶת הַצֵּל רַב. וּסְכָךְ אָטוּם לְגַמְרֵי נִרְאֶה כְּמוֹ גַּג שֶׁל בַּיִת קָבוּעַ.
💛 כְּמוֹ שֶׁהַסְּכָךְ מֵגֵן מִן הַשֶּׁמֶשׁ וּמַשְׁאִיר לָנוּ לִרְאוֹת כּוֹכָבִים – כָּךְ ה׳ שׁוֹמֵר עָלֵינוּ וּמַרְאֶה לָנוּ אוֹר בְּמִדָּה שֶׁאֲנַחְנוּ יְכוֹלִים לְקַבֵּל.
(פֶּרֶק ב׳, הֲלָכָה ג׳)

6נְסָרִים וּ״סְכָךְ לָנֶצַח״

תִּקְרַת בַּיִת מִקְּרָשִׁים פְּסוּלָה לַסֻּכָּה, כִּי הִיא קְבוּעָה. לָכֵן גָּזְרוּ חֲכָמִים שֶׁלֹּא לְסַכֵּךְ בִּנְסָרִים רְחָבִים, שֶׁנִּרְאִים כְּמוֹ תִּקְרָה, וְהַיּוֹם נוֹהֲגִים לְהַחְמִיר גַּם בִּנְסָרִים צָרִים יוֹתֵר. נְסָרִים שֶׁרָחְבָּם פָּחוֹת מִטֶּפַח (כְּ־7.5 ס״מ) – כְּשֵׁרִים לְכָל הַדֵּעוֹת, אִם לֹא קוֹבְעִים אוֹתָם בְּמַסְמְרִים אוֹ בְּדֶבֶק. וּמֻתָּר לִצְבֹּעַ אוֹתָם. ״סְכָךְ לָנֶצַח״ – קָנִים אוֹ נְסָרִים דַּקִּים שֶׁמְּחֻבָּרִים בְּחוּטִים כְּמוֹ מַחְצֶלֶת – כָּשֵׁר, וְכָךְ נוֹהֲגִים רַבִּים.
🤔 לָמָּה? יֵשׁ שֶׁחָשְׁשׁוּ שֶׁהַחוּטִים הוֹפְכִים אֶת הַקָּנִים לְנֶסֶר אֶחָד רָחָב. אֲבָל הַחִבּוּר רָפוּי וְגָמִישׁ, וְאֵינוֹ דּוֹמֶה כְּלָל לְתִקְרָה שֶׁל בַּיִת.
(פֶּרֶק ב׳, הֲלָכָה ד׳)

7עַל מָה מַנִּיחִים אֶת הַסְּכָךְ?

לִפְעָמִים בּוֹנִים אֶת הַסֻּכָּה עַל מִסְגֶּרֶת שֶׁל בַּרְזֶל. הַאִם מֻתָּר לְהַנִּיחַ אֶת הַסְּכָךְ יָשָׁר עַל הַבַּרְזֶל? בְּדָבָר הַזֶּה יֵשׁ שְׁתֵּי דֵּעוֹת. מִי שֶׁרוֹצֶה לְהַחְמִיר וְחוֹשֵׁשׁ שֶׁהָרוּחַ תָּעִיף אֶת הַסְּכָךְ – מַנִּיחַ עָלָיו קְרָשִׁים כְּבֵדִים, שֶׁהֵם כְּשֵׁרִים לַסְּכָךְ.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת רֹב הַפּוֹסְקִים מֻתָּר לְהַנִּיחַ אֶת הַסְּכָךְ עַל בַּרְזֶל אוֹ עַל פְּלַסְטִיק, כִּי רַק הַסְּכָךְ עַצְמוֹ צָרִיךְ לִהְיוֹת כָּשֵׁר. הַסֻּכָּה כְּשֵׁרָה, וּמְבָרְכִים בָּהּ.
יֵשׁ מַחְמִירִים גַּם מָה שֶׁמַּחְזִיק אֶת הַסְּכָךְ צָרִיךְ לִהְיוֹת כָּשֵׁר. לָכֵן מַנִּיחִים עַל הַבַּרְזֶל קוֹרוֹת עֵץ, וַעֲלֵיהֶן אֶת הַסְּכָךְ. הָרַב מֶלָמֵד כּוֹתֵב שֶׁלְּכַתְּחִלָּה טוֹב לָחוּשׁ לְדַעְתָּם.
(פֶּרֶק ב׳, הֲלָכָה ה׳)

8דְּפָנוֹת הַסֻּכָּה

כָּל חֹמֶר כָּשֵׁר לַדְּפָנוֹת – עֵץ, אֶבֶן, זְכוּכִית, פְּלַסְטִיק וַאֲפִלּוּ גֶּדֶר רֶשֶׁת – בִּתְנַאי שֶׁהֵן עוֹמְדוֹת בְּרוּחַ רְגִילָה. אֲפִלּוּ קִירוֹת אֶבֶן שֶׁל בַּיִת כְּשֵׁרִים. לְכַתְּחִלָּה בּוֹנִים סֻכָּה מְהֻדֶּרֶת, עִם אַרְבַּע דְּפָנוֹת שְׁלֵמוֹת וּפֶתַח שֶׁאֶפְשָׁר לִסְגֹּר. מִי שֶׁרוֹצֶה לַחְסֹךְ – בּוֹנֶה שְׁתֵּי דְּפָנוֹת שְׁלֵמוֹת, וְדֹפֶן שְׁלִישִׁית שֶׁרָחְבָּהּ לְפָחוֹת שִׁבְעָה טְפָחִים (כְּ־56 ס״מ).
🤔 לָמָּה? רַק הַסְּכָךְ צָרִיךְ לִהְיוֹת אַרְעַי. הַדְּפָנוֹת אֵינָן צְרִיכוֹת לַעֲצֹר אֶת הָרוּחַ וְהַשֶּׁמֶשׁ – הֵן רַק סוֹגְרוֹת אֶת הַסֻּכָּה וְהוֹפְכוֹת אוֹתָהּ לְדִירָה.
(פֶּרֶק ב׳, הֲלָכָה ו׳)

9גֹּבַהּ הַדְּפָנוֹת וְ״לָבוּד״

הַדֹּפֶן צְרִיכָה לִהְיוֹת גְּבוֹהָה לְפָחוֹת עֲשָׂרָה טְפָחִים, וּלְהַתְחִיל קָרוֹב לָרִצְפָּה: אִם יֵשׁ רֶוַח שֶׁל שְׁלֹשָׁה טְפָחִים (כְּ־22 ס״מ) בֵּין הָרִצְפָּה לַדֹּפֶן – הִיא פְּסוּלָה. אֲבָל בֵּין הַדֹּפֶן לַסְּכָךְ מֻתָּר שֶׁיִּהְיֶה רֶוַח. אֶפְשָׁר לַעֲשׂוֹת דֹּפֶן גַּם מִמּוֹטוֹת אוֹ מֵחוּטִים חֲזָקִים, אִם בֵּין כָּל אֶחָד לַשֵּׁנִי יֵשׁ פָּחוֹת מִשְּׁלֹשָׁה טְפָחִים.
🤔 לָמָּה? זֶה דִּין ״לָבוּד״: רֶוַח קָטָן מִשְּׁלֹשָׁה טְפָחִים נֶחְשָׁב בַּהֲלָכָה כְּאִלּוּ הוּא סָגוּר וּמְחֻבָּר – גַּם אִם הָרוּחַ וְהַשֶּׁמֶשׁ נִכְנָסִים דַּרְכּוֹ.
(פֶּרֶק ב׳, הֲלָכָה ז׳)

10״סֻכָּה לָנֶצַח״ – דְּפָנוֹת שֶׁל בַּד

הַרְבֵּה אֲנָשִׁים בּוֹנִים הַיּוֹם ״סֻכָּה לָנֶצַח״ – מִסְגֶּרֶת שֶׁל בַּרְזֶל, וּדְפָנוֹת שֶׁל יְרִיעוֹת בַּד עָבֶה. הִיא זוֹלָה, קַל לִבְנוֹת וּלְפָרֵק אוֹתָהּ, וְנוֹחַ לְאַחְסֵן אוֹתָהּ. אֲבָל יְרִיעָה שֶׁאֵינָהּ קְשׁוּרָה לְמַטָּה, וְהָרוּחַ מְרִימָה אוֹתָהּ – פְּסוּלָה, כִּי הִיא עוֹלָה מֵעַל שְׁלֹשָׁה טְפָחִים מִן הָרִצְפָּה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ מִפּוֹסְקֵי זְמַנֵּנוּ פִּקְפְּקוּ בְּ״סֻכָּה לָנֶצַח״, כִּי לְדַעְתָּם הַדְּפָנוֹת צְרִיכוֹת לִהְיוֹת יַצִּיבוֹת, וּכְשֶׁהָרוּחַ מְנִיעָה אֶת הַבַּד הֵנָּה וָהֵנָּה – הֵן פְּסוּלוֹת.
לְפִי הָרַב מֶלָמֵד הַסֻּכָּה כְּשֵׁרָה, וּמְבָרְכִים בָּהּ, כִּי הַיְּרִיעוֹת קְשׁוּרוֹת הֵיטֵב מִכָּל הַצְּדָדִים. וְהָרוֹצִים לְהַדֵּר מוֹסִיפִים בָּהּ מוֹטוֹת, שֶׁיַּעֲשׂוּ דֹּפֶן שֶׁל ״לָבוּד״.
(פֶּרֶק ב׳, הֲלָכָה ח׳)

11תַּחַת כִּפַּת הַשָּׁמַיִם

הַסֻּכָּה צְרִיכָה לַעֲמֹד תַּחַת הַשָּׁמַיִם, כְּדֵי שֶׁרַק הַסְּכָךְ יְסַכֵּךְ עַל הַיּוֹשְׁבִים. סֻכָּה תַּחַת עֵץ, תַּחַת תִּקְרָה אוֹ תַּחַת מִרְפֶּסֶת שֶׁל הַשָּׁכֵן – פְּסוּלָה. אֲבָל מֻתָּר לִבְנוֹת סֻכָּה לְיַד בִּנְיָן גָּבוֹהַּ שֶׁמַּסְתִּיר אֶת הַשֶּׁמֶשׁ, וְגַם מִתַּחַת לְחַבְלֵי כְּבִיסָה אוֹ לְחוּטֵי חַשְׁמַל דַּקִּים.
🤔 לָמָּה? רַק מָה שֶׁנִּמְצָא בֵּין הַסְּכָךְ לַשָּׁמַיִם פּוֹסֵל. בִּנְיָן שֶׁעוֹמֵד בַּצַּד אֵינוֹ מַפְרִיד בֵּינֵיהֶם. וְחוּטִים דַּקִּים עוֹשִׂים מְעַט מְאוֹד צֵל, וְגַם לֹא נוֹעֲדוּ לְצֵל.
(פֶּרֶק ב׳, הֲלָכָה ט׳)

12בְּצֵל הַסְּכָךְ – קִשּׁוּטִים, סָדִין וְכוֹבַע

הַמִּצְוָה הִיא לָשֶׁבֶת בְּצֵל הַסְּכָךְ הַכָּשֵׁר. מִי שֶׁפּוֹרֵשׂ סָדִין מִתַּחַת לַסְּכָךְ כְּדֵי לְהַרְבּוֹת צֵל – פָּסַל אֶת הַסֻּכָּה. אֲבָל מֻתָּר לָשֶׁבֶת בַּסֻּכָּה עִם כּוֹבַע גָּדוֹל עַל הָרֹאשׁ. וּמֻתָּר לִתְלוֹת מִתַּחַת לַסְּכָךְ פֵּרוֹת וְקִשּׁוּטִים – אִם הֵם קְרוֹבִים לַסְּכָךְ, בְּתוֹךְ אַרְבָּעָה טְפָחִים (כְּ־30 ס״מ), אֲפִלּוּ אִם הֵם מְכַסִּים אֶת כֻּלּוֹ.
🤔 לָמָּה? הַכּוֹבַע הוּא חֵלֶק מִן הַבְּגָדִים שֶׁל הָאָדָם, וְאֵינוֹ מַפְרִיד בֵּינוֹ לַסְּכָךְ. וְהַקִּשּׁוּטִים הַקְּרוֹבִים לַסְּכָךְ ״בְּטֵלִים״ אֵלָיו – הֵם כְּמוֹ חֵלֶק מִמֶּנּוּ.
(פֶּרֶק ב׳, הֲלָכָה י׳)

13רֶוַח בַּסְּכָךְ וּמִטַּת קוֹמָתַיִם

אִם יֵשׁ בַּסְּכָךְ רֶוַח שֶׁל אֲוִיר שֶׁרָחְבּוֹ שְׁלֹשָׁה טְפָחִים (כְּ־22 ס״מ) – אָסוּר לָשֶׁבֶת תַּחְתָּיו. רֶוַח קָטָן מִזֶּה – בָּטֵל לַסֻּכָּה, וּבִלְבַד שֶׁרֹב הָרֹאשׁ וְרֹב הַגּוּף לֹא יִהְיוּ תַּחְתָּיו. מִטַּת קוֹמָתַיִם: אִם בֵּין הַמִּטָּה הַתַּחְתּוֹנָה לָעֶלְיוֹנָה יֵשׁ עֲשָׂרָה טְפָחִים (כְּ־76 ס״מ) – הַיָּשֵׁן לְמַטָּה לֹא קִיֵּם מִצְוַת סֻכָּה. אִם יֵשׁ פָּחוֹת – קִיֵּם. וְכָךְ גַּם מִי שֶׁיָּשֵׁן מִתַּחַת לַשֻּׁלְחָן אוֹ בְּתוֹךְ כִּלָּה גְּבוֹהָה.
🤔 לָמָּה? מָקוֹם בְּגֹבַהּ עֲשָׂרָה טְפָחִים נֶחְשָׁב כְּמוֹ ״אֹהֶל״ בִּפְנֵי עַצְמוֹ, וְהוּא מַפְרִיד בֵּין הַיָּשֵׁן לַסְּכָךְ.
(פֶּרֶק ב׳, הֲלָכָה י׳, הֲלָכָה י״א)

14מִמִּצְוָה לְמִצְוָה

מִצְוָה לַעֲסֹק בְּהָכָנַת הַסֻּכָּה. חֲסִידִים וְאַנְשֵׁי מַעֲשֶׂה מַתְחִילִים לִבְנוֹת אוֹתָהּ מִיָּד בְּמוֹצָאֵי יוֹם הַכִּפּוּרִים, וּמִשְׁתַּדְּלִים לְסַיֵּם עַד לְמָחֳרָת. סֻכָּה שֶׁנִּשְׁאֲרָה מִשָּׁנָה שֶׁעָבְרָה, אוֹ סֻכָּה שֶׁעָשׂוּ רוֹעִים לְצֵל – כְּשֵׁרָה. אֲבָל לְכַתְּחִלָּה מְחַדְּשִׁים בָּהּ דָּבָר לִכְבוֹד הֶחָג: מוֹסִיפִים סְכָךְ טֶפַח עַל טֶפַח, אוֹ קָנֶה דַּק לְכָל אָרְכָּהּ.
🤔 לָמָּה? ״מִצְוָה הַבָּאָה לְיָדְךָ – אַל תַּחְמִיצֶנָּה״. הַבְּרָכָה ״לֵישֵׁב בַּסֻּכָּה״ נֶאֱמֶרֶת כְּשֶׁיּוֹשְׁבִים בָּהּ, וְלֹא כְּשֶׁבּוֹנִים אוֹתָהּ – וּבְכָל זֹאת גַּם לַבְּנִיָּה יֵשׁ צַד שֶׁל מִצְוָה.
💛 כָּל מַסְמֵר שֶׁאֲנִי דּוֹפֵק בַּסֻּכָּה הוּא הֲכָנָה לְהִתְאָרֵחַ אֵצֶל ה׳ שִׁבְעָה יָמִים.
(פֶּרֶק ב׳, הֲלָכָה י״ב)

15קֹדֶם דְּפָנוֹת – אַחַר כָּךְ סְכָךְ

״תַּעֲשֶׂה – וְלֹא מִן הֶעָשׂוּי״: אֶת הַסֻּכָּה עוֹשִׂים עַל יְדֵי הֲנָחַת הַסְּכָךְ. לָכֵן בּוֹנִים קֹדֶם אֶת הַדְּפָנוֹת, וְרַק אַחַר כָּךְ מַנִּיחִים אֶת הַסְּכָךְ. מִי שֶׁהֵנִיחַ קֹדֶם סְכָךְ וְאַחַר כָּךְ בָּנָה דְּפָנוֹת – לְדַעַת פּוֹסְקִים רַבִּים הַסֻּכָּה פְּסוּלָה. מֻתָּר לְהַתְקִין גַּגּוֹן נִפְתָּח לְשָׁעוֹת הַגֶּשֶׁם: כְּשֶׁהוּא פָּרוּשׂ – הַסֻּכָּה פְּסוּלָה, וּכְשֶׁמְּסִירִים אוֹתוֹ – הִיא חוֹזֶרֶת לִהְיוֹת כְּשֵׁרָה.
🤔 לָמָּה? אִם הַסְּכָךְ הוּנַח קֹדֶם, הַסֻּכָּה ״נַעֲשֵׂית״ כְּשֶׁבּוֹנִים אֶת הַדְּפָנוֹת, וְלֹא עַל יְדֵי הַסְּכָךְ. וְהַגַּגּוֹן מַפְרִיד בֵּין הַסְּכָךְ לַשָּׁמַיִם רַק כָּל עוֹד הוּא פָּרוּשׂ.
(פֶּרֶק ב׳, הֲלָכָה י״ג)

16סֻכָּה שְׁאוּלָה – וְלֹא גְּזוּלָה

מֻתָּר לְקַיֵּם אֶת הַמִּצְוָה בְּסֻכָּה שֶׁל חָבֵר שֶׁנָּתַן רְשׁוּת. וְגַם כְּשֶׁבַּעַל הַסֻּכָּה אֵינוֹ נִמְצָא וְאִי אֶפְשָׁר לִשְׁאֹל אוֹתוֹ – מֻתָּר לָשֶׁבֶת בָּהּ. אֲבָל אִם יָדוּעַ שֶׁהוּא מַקְפִּיד, אוֹ שֶׁהוּא עָלוּל לַחֲזֹר וּלְהִתְבַּיֵּשׁ לְהִכָּנֵס לַסֻּכָּה שֶׁלּוֹ כְּשֶׁזָּרִים יוֹשְׁבִים בָּהּ – אָסוּר בְּלִי רְשׁוּת. וּמִי שֶׁבָּנָה סֻכָּה בְּשֶׁטַח שֶׁל אַחֵר בְּלִי רְשׁוּת, אוֹ בְּשֶׁטַח צִבּוּרִי כְּשֶׁהַצִּבּוּר מִתְנַגֵּד – לֹא יְבָרֵךְ בָּהּ.
🤔 לָמָּה? חֲכָמִים הֶעֱרִיכוּ שֶׁנּוֹחַ לִיהוּדִי שֶׁעוֹשִׂים מִצְוָה בְּמָמוֹנוֹ. אֲבָל מִצְוָה שֶׁנַּעֲשֵׂית בְּגֶזֶל – הַבְּרָכָה עָלֶיהָ אֵינָהּ בְּרָכָה אֶלָּא כַּעַס לִפְנֵי ה׳.
(פֶּרֶק ב׳, הֲלָכָה י״ג)

17סֻכָּה נָאָה

״זֶה אֵלִי וְאַנְוֵהוּ״ – מִצְוָה לִבְנוֹת סֻכָּה נָאָה, מְרֻוַּחַת וּנְעִימָה, וּלְקַשֵּׁט אוֹתָהּ בְּשַׁרְשְׁרָאוֹת, בִּפְרָחִים, בִּתְמוּנוֹת יָפוֹת וּבְנוּרוֹת, לִפְרֹשׂ מַפָּה יָפָה עַל הַשֻּׁלְחָן וּלְהָבִיא כֵּלִים נָאִים. לֹא מְסַכְּכִים בִּצְמָחִים שֶׁרֵיחָם רַע אוֹ שֶׁעָלֵיהֶם נוֹשְׁרִים. וְנִזְהָרִים מִשְּׂרֵפָה: מַרְחִיקִים אֶת הַנּוּרוֹת מִן הַסְּכָךְ, וְלֹא מַשְׁאִירִים נֵרוֹת אוֹ חִבּוּרֵי חַשְׁמַל לֹא בְּטוּחִים.
🤔 לָמָּה? חֲכָמִים אָמְרוּ: ״הִתְנָאֵה לְפָנָיו בְּמִצְווֹת – עֲשֵׂה לְפָנָיו סֻכָּה נָאָה״. וְאִם יֵשׁ בַּסֻּכָּה רֵיחַ רַע אוֹ עָלִים נוֹשְׁרִים, עָלוּל הָאָדָם לַעֲזֹב אוֹתָהּ וְלָלֶכֶת הַבַּיְתָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ אוֹסְרִים לְקַשֵּׁט אֶת הַסֻּכָּה בִּפְסוּקִים, כְּמוֹ ״בַּסֻּכֹּת תֵּשְׁבוּ שִׁבְעַת יָמִים״, כִּי מֻתָּר לִכְתֹּב חֲלָקִים מִן הַתּוֹרָה רַק לְצֹרֶךְ גָּדוֹל שֶׁל לִמּוּד.
וְיֵשׁ מַתִּירִים כִּי יֵשׁ בָּזֶה תּוֹעֶלֶת שֶׁל חִנּוּךְ. הָרַב מֶלָמֵד כּוֹתֵב שֶׁאֶפְשָׁר לְהָקֵל, וּבִלְבַד שֶׁלֹּא יִכְתְּבוּ אֶת הַפְּסוּקִים בַּכְּתָב שֶׁל סֵפֶר תּוֹרָה.
💛 כְּשֶׁאֲנִי מְקַשֵּׁט אֶת הַסֻּכָּה, אֲנִי מֵכִין בַּיִת יָפֶה לְאוֹרֵחַ הֲכִי חָשׁוּב – לַה׳.
(פֶּרֶק ב׳, הֲלָכָה ט״ו)

18קְדֻשַּׁת הַסֻּכָּה וְקִשּׁוּטֶיהָ

הַסֻּכָּה וְקִשּׁוּטֶיהָ קְדוֹשִׁים כָּל הֶחָג, וְעַד סוֹף שְׁמִינִי עֲצֶרֶת. אָסוּר לָקַחַת מֵהֶם שׁוּם דָּבָר לְשִׁמּוּשׁ אַחֵר – אֲפִלּוּ קֵיסָם לַשִּׁנַּיִם, וַאֲפִלּוּ אִם הַסֻּכָּה נָפְלָה. אֲבָל מֻתָּר לְהִשָּׁעֵן עַל הַדֹּפֶן וְלִתְלוֹת עָלֶיהָ חֲפָצִים, כְּמוֹ עַל קִיר שֶׁל בַּיִת. קִשּׁוּט שֶׁנִּתְקַלְקֵל – מֻתָּר לְהָסִיר וְלִזְרֹק בְּכָבוֹד. כְּשֶׁמַּתְחִיל גֶּשֶׁם – מֻתָּר לְהוֹרִיד אֶת הַקִּשּׁוּטִים כְּדֵי לִתְלוֹת אוֹתָם שׁוּב. וְאַחֲרֵי הֶחָג לֹא מְבַזִּים אֶת עֲצֵי הַסֻּכָּה וְהַקִּשּׁוּטִים.
🤔 לָמָּה? ״חַג הַסֻּכּוֹת שִׁבְעַת יָמִים לַה׳״ – הַסֻּכָּה מְקֻדֶּשֶׁת לַמִּצְוָה. וְגַם בַּקִּשּׁוּטִים יֵשׁ מִצְוָה, וְלָכֵן גַּם הֵם ״מֻקְצִים״ לְמִצְוָתָם. מִי שֶׁרוֹצֶה לֵהָנוֹת מִן הַקִּשּׁוּטִים, יָכוֹל לְהַתְנוֹת עֲלֵיהֶם לִפְנֵי הֶחָג.
(פֶּרֶק ב׳, הֲלָכָה ט״ז)

19סֻכָּה מִתַּחַת לְפֶּרְגּוֹלָה

פֶּרְגּוֹלָה הִיא מִבְנֶה עֵץ קָבוּעַ בֶּחָצֵר, שֶׁעוֹשֶׂה צֵל. יֵשׁ מַתִּירִים לְהִשְׁתַּמֵּשׁ בָּהּ כְּסֻכָּה, כִּי הַגֶּשֶׁם עוֹבֵר דַּרְכָּהּ וְהִיא לֹא נוֹעֲדָה לְדִירָה. אֲבָל לְמַעֲשֶׂה מַחְמִירִים, כִּי הִיא מִבְנֶה חָזָק וְקָבוּעַ: אִם רֹב הַגַּג שֶׁלָּהּ מְכֻסֶּה קְרָשִׁים קְבוּעִים – מְמַעֲטִים אוֹתָם עַד שֶׁרֹב הַגַּג פָּתוּחַ, וּמַנִּיחִים מֵעָלֶיהָ סְכָךְ כָּשֵׁר שֶׁעוֹשֶׂה לְבַדּוֹ יוֹתֵר צֵל מִשֶּׁמֶשׁ. אֶפְשָׁר גַּם לְהָסִיר אֶת הַקְּרָשִׁים וּלְהַחֲזִיר אוֹתָם בְּלִי מַסְמְרִים וּבְלִי דֶּבֶק.
🤔 לָמָּה? כְּשֶׁיֵּשׁ סָפֵק בְּדִין שֶׁל תּוֹרָה – מַחְמִירִים.
(פֶּרֶק ב׳, הֲלָכָה י״ז)

📝 מָה לָמַדְנוּ

  • הַסֻּכָּה הִיא דִּירַת אַרְעַי: לְפָחוֹת עֲשָׂרָה טְפָחִים גֹּבַהּ וְשִׁבְעָה טְפָחִים רֹחַב, וְלֹא יוֹתֵר מֵעֶשְׂרִים אַמָּה.
  • הַסְּכָךְ צוֹמֵחַ מִן הָאֲדָמָה, תָּלוּשׁ וְאֵינוֹ כְּלִי, וְעוֹשֶׂה יוֹתֵר צֵל מִשֶּׁמֶשׁ.
  • כָּל חֹמֶר כָּשֵׁר לַדְּפָנוֹת, אִם הֵן עוֹמְדוֹת בָּרוּחַ; בּוֹנִים אוֹתָן לִפְנֵי שֶׁמַּנִּיחִים אֶת הַסְּכָךְ.
  • הַסֻּכָּה עוֹמֶדֶת תַּחַת הַשָּׁמַיִם, וְהַקִּשּׁוּטִים תְּלוּיִים קָרוֹב לַסְּכָךְ.
  • הַסֻּכָּה וְקִשּׁוּטֶיהָ קְדוֹשִׁים כָּל הֶחָג – לֹא לוֹקְחִים מֵהֶם אֲפִלּוּ קֵיסָם.

❓ שְׁאֵלוֹת חֲזָרָה

  1. מָה הֵם שְׁלֹשֶׁת הַתְּנָאִים שֶׁל סְכָךְ כָּשֵׁר? תֵּן דֻּגְמָה לִסְכָךְ פָּסוּל.
  2. לָמָּה מֻתָּר לִבְנוֹת סֻכָּה מִתַּחַת לְחַבְלֵי כְּבִיסָה, אֲבָל לֹא מִתַּחַת לָעֵץ?
  3. בְּאֵיזֶה סֵדֶר בּוֹנִים אֶת הַסֻּכָּה, וְלָמָּה?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״בּוֹנִים סֻכָּה כְּשֵׁרָה״

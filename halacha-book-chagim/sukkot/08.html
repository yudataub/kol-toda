סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת סֻכּוֹת
מַהֲדוּרַת יְלָדִים

👑 מִצְוַת הַקְהֵל

פֶּרֶק ח׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
דַּמְיֵן: חֲצוֹצְרוֹת כֶּסֶף נִשְׁמָעוֹת בְּכָל רְחוֹבוֹת יְרוּשָׁלַיִם. מִכָּל הַכִּוּוּנִים זוֹרְמִים אֲנָשִׁים – אַבָּאִים, אִמָּהוֹת, יְלָדִים וַאֲפִלּוּ תִּינוֹקוֹת. בְּאֶמְצַע עוֹמֶדֶת בִּימָה גְּדוֹלָה שֶׁל עֵץ, וְעָלֶיהָ – הַמֶּלֶךְ, עִם סֵפֶר תּוֹרָה בְּיָדוֹ. זֶה קוֹרֶה רַק פַּעַם בְּשֶׁבַע שָׁנִים! בַּפֶּרֶק הַזֶּה נִלְמַד עַל מִצְוַת הַקְהֵל.

1פַּעַם בְּשֶׁבַע שָׁנִים

״מִקֵּץ שֶׁבַע שָׁנִים בְּמֹעֵד שְׁנַת הַשְּׁמִטָּה בְּחַג הַסֻּכּוֹת... הַקְהֵל אֶת הָעָם הָאֲנָשִׁים וְהַנָּשִׁים וְהַטַּף״. מִצְוָה לְהַקְהִיל אֶת כָּל יִשְׂרָאֵל בַּמִּקְדָּשׁ, בְּחַג הַסֻּכּוֹת שֶׁאַחֲרֵי שְׁנַת הַשְּׁמִטָּה, וְלִקְרֹא לִפְנֵיהֶם פָּרָשִׁיּוֹת מִן הַתּוֹרָה שֶׁמְּעוֹרְרוֹת לֶאֱמוּנָה, לְיִרְאַת ה׳, לְלִמּוּד תּוֹרָה וּלְקִיּוּם הַמִּצְווֹת.
🤔 לָמָּה? ״לְמַעַן יִשְׁמְעוּ וּלְמַעַן יִלְמְדוּ, וְיָרְאוּ אֶת ה׳ אֱלֹהֵיכֶם וְשָׁמְרוּ לַעֲשׂוֹת אֶת כָּל דִּבְרֵי הַתּוֹרָה הַזֹּאת״.
(פֶּרֶק ח׳, הֲלָכָה א׳)

2לָמָּה כָּל הַכִּנּוּס הַגָּדוֹל הַזֶּה?

לֹא הָיָה מַעֲמָד מְכֻבָּד וּמַרְשִׁים מִמַּעֲמַד הַקְהֵל: כָּל יִשְׂרָאֵל יַחַד, וְהַנִּכְבָּד שֶׁבְּכֻלָּם – הַמֶּלֶךְ – קוֹרֵא בַּתּוֹרָה. כָּל אֶחָד שָׁאַל אֶת עַצְמוֹ: ״לָמָּה כֻּלָּנוּ כָּאן?״ – וְהַתְּשׁוּבָה הָיְתָה בְּרוּרָה: כְּדֵי לִשְׁמֹעַ אֶת דִּבְרֵי הַתּוֹרָה, ״שֶׁהִיא כָּל עִקָּרֵנוּ וְהוֹדֵנוּ וְתִפְאַרְתֵּנוּ״.
🤔 לָמָּה? כָּל אֶחָד קִבֵּל מִשֶּׁהוּ: מִי שֶׁיָּכוֹל לִלְמֹד – הִתְעוֹרֵר לִלְמֹד יוֹתֵר. הַיְּלָדִים קִבְּלוּ חֵשֶׁק גָּדוֹל לִלְמֹד תּוֹרָה וּלְקַיֵּם מִצְווֹת. וַאֲפִלּוּ הַתִּינוֹקוֹת סָפְגוּ בַּנְּשָׁמָה שֶׁאֵין דָּבָר חָשׁוּב מִן הַתּוֹרָה – אִם כֻּלָּם בָּאִים לִשְׁמֹעַ אוֹתָהּ. וְהַהוֹרִים נִזְכְּרוּ בַּשְּׁלִיחוּת שֶׁלָּהֶם: לְחַנֵּךְ אֶת יַלְדֵיהֶם לַתּוֹרָה.
💛 כְּשֶׁכָּל הָעָם מִתְאַסֵּף רַק כְּדֵי לִשְׁמֹעַ אֶת דְּבַר ה׳ – כָּל יֶלֶד מֵבִין כַּמָּה ה׳ וְהַתּוֹרָה שֶׁלּוֹ יְקָרִים לָנוּ.
(פֶּרֶק ח׳, הֲלָכָה א׳)

3כְּמוֹ בְּהַר סִינַי

גַּם חֲכָמִים גְּדוֹלִים שֶׁיָּדְעוּ אֶת כָּל הַתּוֹרָה, וְגַם מִי שֶׁלֹּא הֵבִין עִבְרִית – הָיוּ חַיָּבִים לָבוֹא. וּמִי שֶׁעָמַד רָחוֹק וְלֹא שָׁמַע הֵיטֵב – הָיָה צָרִיךְ לְכַוֵּן אֶת לִבּוֹ לַקְּרִיאָה.
🤔 לָמָּה? מַעֲמַד הַקְהֵל הוּא כְּמוֹ מַעֲמַד הַר סִינַי: כָּל יִשְׂרָאֵל יַחַד, וְכָל אֶחָד מַרְגִּישׁ כְּאִלּוּ עַכְשָׁו הוּא מְקַבֵּל אֶת הַתּוֹרָה מִפִּי ה׳. הַמֶּלֶךְ הוּא רַק שָׁלִיחַ – לְהַשְׁמִיעַ אֶת דִּבְרֵי ה׳.
(פֶּרֶק ח׳, הֲלָכָה א׳, הֲלָכָה ג׳)

4מִי קוֹרֵא?

חֲכָמִים תִּקְּנוּ שֶׁהַמֶּלֶךְ יִקְרָא, כְּדֵי לְכַבֵּד אֶת הַמַּעֲמָד כַּמָּה שֶׁיּוֹתֵר. וּכְשֶׁאֵין מֶלֶךְ, אוֹ שֶׁקּוֹלוֹ חַלָּשׁ, אוֹ שֶׁהוּא עוֹד קָטָן – הַמִּצְוָה אֵינָהּ מִתְבַּטֶּלֶת, וְקוֹרֵא הַגָּדוֹל שֶׁבְּיִשְׂרָאֵל: נָשִׂיא, כֹּהֵן גָּדוֹל אוֹ גְּדוֹל הַדּוֹר.
(פֶּרֶק ח׳, הֲלָכָה א׳)

5מָה קָרָא הַמֶּלֶךְ?

הַקְּרִיאָה הָיְתָה בִּלְשׁוֹן הַקֹּדֶשׁ, כְּמוֹ שֶׁהַתּוֹרָה כְּתוּבָה – וְגַם מִי שֶׁלֹּא הֵבִין הָיָה חַיָּב לִשְׁמֹעַ. הַמֶּלֶךְ קָרָא מִתְּחִלַּת סֵפֶר דְּבָרִים וְעַד סוֹף פָּרָשַׁת ״שְׁמַע״, וְאַחַר כָּךְ ״וְהָיָה אִם שָׁמֹעַ״, שְׁתֵּי פָּרָשִׁיּוֹת עַל הַמַּעַשְׂרוֹת, פָּרָשַׁת הַמֶּלֶךְ, וְהַבְּרָכוֹת וְהַקְּלָלוֹת.
🤔 לָמָּה? בַּפָּרָשִׁיּוֹת הָאֵלֶּה יֵשׁ אֶת כָּל הַיְּסוֹדוֹת: הַכְּנִיסָה לְאֶרֶץ יִשְׂרָאֵל, מַעֲמַד הַר סִינַי, ״שְׁמַע יִשְׂרָאֵל״ וְאַהֲבַת ה׳, שְׂכַר הַמִּצְווֹת, וְחִנּוּךְ הַבָּנִים לַתּוֹרָה.
(פֶּרֶק ח׳, הֲלָכָה ב׳)

6מָתַי הָיָה הַהַקְהֵל?

הַהַקְהֵל הִתְקַיֵּם בְּחַג הַסֻּכּוֹת, כְּשֶׁכָּל יִשְׂרָאֵל עָלוּ לָרֶגֶל. אֲבָל לֹא בַּיּוֹם הָרִאשׁוֹן עַצְמוֹ – כִּי בְּיוֹם טוֹב אִי אֶפְשָׁר לִבְנוֹת אֶת הַבִּימָה – אֶלָּא בְּמוֹצָאֵי הַיּוֹם הָרִאשׁוֹן. יֵשׁ אוֹמְרִים שֶׁהַמַּעֲמָד הָיָה בַּלַּיְלָה שֶׁל מוֹצָאֵי הַיּוֹם הָרִאשׁוֹן, וְיֵשׁ אוֹמְרִים שֶׁהָיָה לְמָחֳרָת, בַּיּוֹם הָרִאשׁוֹן שֶׁל חֹל הַמּוֹעֵד.
🤔 לָמָּה? הַהַקְהֵל הָיָה כְּמוֹ חוֹתָם שֶׁל קְדֻשָּׁה עַל שֶׁבַע הַשָּׁנִים שֶׁעָבְרוּ: מָה שֶׁקָּשׁוּר לַתּוֹרָה – נִשְׁאָר לָנֶצַח. וּמִמֶּנּוּ קִבֵּל עַם יִשְׂרָאֵל כֹּחַ לְהַמְשִׁיךְ בְּדֶרֶךְ הַתּוֹרָה בְּשֶׁבַע הַשָּׁנִים הַבָּאוֹת.
(פֶּרֶק ח׳, הֲלָכָה ג׳)

7גַּם הַנָּשִׁים וְגַם הַטַּף

מִי שֶׁפָּטוּר מִן הָעֲלִיָּה לָרֶגֶל – כְּמוֹ חוֹלֶה אוֹ זָקֵן שֶׁאֵינוֹ יָכוֹל לַעֲלוֹת בְּרַגְלָיו – פָּטוּר גַּם מִן הַהַקְהֵל. אֲבָל הַנָּשִׁים, אַף שֶׁאֵינָן חַיָּבוֹת בַּעֲלִיָּה לָרֶגֶל, חַיָּבוֹת בְּהַקְהֵל. וְהַהוֹרִים מְצֻוִּים לְהָבִיא אֶת כָּל יַלְדֵיהֶם הַקְּטַנִּים לַמַּעֲמָד.
🤔 לָמָּה? הַתּוֹרָה אָמְרָה בִּמְפֹרָשׁ: ״הָאֲנָשִׁים וְהַנָּשִׁים וְהַטַּף״ – כָּל הָעָם צָרִיךְ לִהְיוֹת שָׁם, כְּמוֹ בְּהַר סִינַי.
💛 בַּמַּעֲמָד הֲכִי חָשׁוּב שֶׁל עַם יִשְׂרָאֵל – גַּם הַיְּלָדִים הֲכִי קְטַנִּים הָיוּ בִּפְנִים.
(פֶּרֶק ח׳, הֲלָכָה ג׳)

8סֵדֶר הַמַּעֲמָד

הַכֹּהֲנִים הָיוּ מְסַבְּבִים בְּכָל יְרוּשָׁלַיִם וְתוֹקְעִים בַּחֲצוֹצְרוֹת, כְּדֵי לְהַקְהִיל אֶת הָעָם. בְּאֶמְצַע עֶזְרַת הַנָּשִׁים הֶעֱמִידוּ בִּימָה גְּדוֹלָה שֶׁל עֵץ, וְהַמֶּלֶךְ עָלָה עָלֶיהָ, כְּדֵי שֶׁכֻּלָּם יִשְׁמְעוּ וְיִרְאוּ אוֹתוֹ. סֵפֶר הַתּוֹרָה עָבַר מִיָּד לְיָד: חַזַּן הַכְּנֶסֶת נָתַן אוֹתוֹ לְרֹאשׁ הַכְּנֶסֶת, רֹאשׁ הַכְּנֶסֶת לַסְּגָן, הַסְּגָן לַכֹּהֵן הַגָּדוֹל, וְהַכֹּהֵן הַגָּדוֹל לַמֶּלֶךְ – וְהַמֶּלֶךְ קִבֵּל אוֹתוֹ בַּעֲמִידָה. הַמֶּלֶךְ יָכוֹל הָיָה לָשֶׁבֶת, אֲבָל אִם עָמַד לִכְבוֹד הַתּוֹרָה – הֲרֵי זֶה מְשֻׁבָּח.
🤔 לָמָּה? הַהַעֲבָרָה מִגָּדוֹל לְגָדוֹל יוֹתֵר הָיְתָה כָּבוֹד לַתּוֹרָה וְלַמֶּלֶךְ.
(פֶּרֶק ח׳, הֲלָכָה ד׳)

9הַבְּרָכוֹת שֶׁל הַמֶּלֶךְ

לִפְנֵי הַקְּרִיאָה וְאַחֲרֶיהָ בֵּרַךְ הַמֶּלֶךְ כְּמוֹ כָּל מִי שֶׁעוֹלֶה לַתּוֹרָה. וְאַחַר כָּךְ הוֹסִיף עוֹד שֶׁבַע בְּרָכוֹת – וּבָהֶן תְּפִלָּה עַל בֵּית הַמִּקְדָּשׁ שֶׁיַּעֲמֹד, עַל עַם יִשְׂרָאֵל, עַל הַכֹּהֲנִים, וּתְפִלָּה אִישִׁית שֶׁסִּיֵּם בַּמִּלִּים: ״הוֹשַׁע ה׳ אֶת עַמְּךָ יִשְׂרָאֵל, שֶׁעַמְּךָ צְרִיכִין לְהִוָּשֵׁעַ. בָּרוּךְ אַתָּה ה׳ שׁוֹמֵעַ תְּפִלָּה״.
💛 אֲפִלּוּ הַמֶּלֶךְ, הַחָשׁוּב בְּיוֹתֵר בָּעָם, עוֹמֵד לִפְנֵי ה׳ וּמְבַקֵּשׁ: הוֹשִׁיעֵנוּ – כִּי כֻּלָּנוּ צְרִיכִים אוֹתוֹ.
(פֶּרֶק ח׳, הֲלָכָה ד׳)

10זֵכֶר לְהַקְהֵל – בַּכֹּתֶל

הַיּוֹם אֵין מִצְוָה לְקַיֵּם הַקְהֵל, כִּי הִיא קְשׁוּרָה לָעֲלִיָּה לָרֶגֶל לְבֵית הַמִּקְדָּשׁ. אֲבָל גְּדוֹלֵי הָרַבָּנִים עוֹרְרוּ לַעֲשׂוֹת ״זֵכֶר לְהַקְהֵל״. בִּשְׁנַת תש״ו, אַחֲרֵי הַשּׁוֹאָה וְלִפְנֵי קוּם הַמְּדִינָה, הִתְקַיֵּם לָרִאשׁוֹנָה מַעֲמָד כָּזֶה, וּמֵאָז, בְּכָל מוֹצָאֵי שְׁמִטָּה, בַּיּוֹם הָרִאשׁוֹן שֶׁל חֹל הַמּוֹעֵד סֻכּוֹת – מִתְקַיֵּם בַּכֹּתֶל הַמַּעֲרָבִי מַעֲמָד ״זֵכֶר לְהַקְהֵל״, בְּהַשְׁתַּתְּפוּת הָרַבָּנִים הָרָאשִׁיִּים וְהָמוֹן רַב.
🤔 לָמָּה? ״צִיּוֹן הִיא, דֹּרֵשׁ אֵין לָהּ״ – מִכָּאן לָמְדוּ חֲכָמִים שֶׁצָּרִיךְ לִדְרֹשׁ אֶת צִיּוֹן וּלְהַזְכִּיר אוֹתָהּ, וְכָךְ מְרַפְּאִים אֶת הַחֻרְבָּן. וּבִמְיֻחָד בַּדּוֹרוֹת שֶׁבָּהֶם עַם יִשְׂרָאֵל חוֹזֵר לְאַרְצוֹ.
💛 כְּשֶׁאֲנַחְנוּ עוֹמְדִים יַחַד בַּכֹּתֶל וְשׁוֹמְעִים תּוֹרָה, אֲנַחְנוּ אוֹמְרִים לַה׳: אֲנַחְנוּ מְחַכִּים לַיּוֹם שֶׁנַּעֲמֹד שׁוּב לְפָנֶיךָ בְּבֵית הַמִּקְדָּשׁ.
(פֶּרֶק ח׳, הֲלָכָה ה׳)

📝 מָה לָמַדְנוּ

  • פַּעַם בְּשֶׁבַע שָׁנִים, בְּסֻכּוֹת שֶׁאַחֲרֵי הַשְּׁמִטָּה, הִקְהִילוּ אֶת כָּל יִשְׂרָאֵל בַּמִּקְדָּשׁ לִשְׁמֹעַ תּוֹרָה.
  • הַמֶּלֶךְ קָרָא בִּלְשׁוֹן הַקֹּדֶשׁ מִסֵּפֶר דְּבָרִים, וְכָל אֶחָד הִרְגִּישׁ כְּאִלּוּ הוּא עוֹמֵד בְּהַר סִינַי.
  • גַּם הַנָּשִׁים וְהַטַּף חַיָּבִים בְּהַקְהֵל, וְגַם חֲכָמִים גְּדוֹלִים.
  • הַיּוֹם עוֹשִׂים ״זֵכֶר לְהַקְהֵל״ בַּכֹּתֶל, בְּמוֹצָאֵי כָּל שְׁמִטָּה.

❓ שְׁאֵלוֹת חֲזָרָה

  1. מָתַי הִתְקַיֵּם מַעֲמַד הַקְהֵל, וּמִי הָיָה צָרִיךְ לָבוֹא אֵלָיו?
  2. לָמָּה גַּם חֲכָמִים שֶׁיָּדְעוּ אֶת כָּל הַתּוֹרָה הָיוּ צְרִיכִים לָבוֹא לִשְׁמֹעַ?
  3. מָה עוֹשִׂים הַיּוֹם זֵכֶר לְהַקְהֵל, וְלָמָּה?

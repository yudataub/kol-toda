סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת סֻכּוֹת
מַהֲדוּרַת יְלָדִים

📜 שְׁמִינִי עֲצֶרֶת וְשִׂמְחַת תּוֹרָה

פֶּרֶק ז׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
הַסֻּכָּה כְּבָר רֵיקָה, הַלּוּלָב מֻנָּח בַּצַּד – וּבְכָל זֹאת זֶה הַיּוֹם הֲכִי שָׂמֵחַ! כָּל סִפְרֵי הַתּוֹרָה יוֹצְאִים מִן הָאָרוֹן, כֻּלָּם רוֹקְדִים בְּמַעְגָּלִים, וְהַיְּלָדִים עוֹלִים לַתּוֹרָה מִתַּחַת לְטַלִּית גְּדוֹלָה. אֵין סֻכָּה וְאֵין לוּלָב – אָז עַל מָה כָּל הַשִּׂמְחָה? בַּפֶּרֶק הַזֶּה נִלְמַד עַל שְׁמִינִי עֲצֶרֶת וְשִׂמְחַת תּוֹרָה.

1חַג בִּפְנֵי עַצְמוֹ

שְׁמִינִי עֲצֶרֶת בָּא מִיָּד אַחֲרֵי שִׁבְעַת יְמֵי סֻכּוֹת, אֲבָל הוּא חַג בִּפְנֵי עַצְמוֹ: אֵין בּוֹ סֻכָּה וְאֵין בּוֹ לוּלָב, וּשְׁמוֹ שׁוֹנֶה – בַּתְּפִלָּה, בַּקִּדּוּשׁ וּבְבִרְכַּת הַמָּזוֹן מַזְכִּירִים ״שְׁמִינִי עֲצֶרֶת״ וְלֹא ״חַג הַסֻּכּוֹת״. גַּם הַקָּרְבָּנוֹת שֶׁלּוֹ שׁוֹנִים: בִּמְקוֹם שִׁשָּׁה פָּרִים – פַּר אֶחָד. לָכֵן מְבָרְכִים עָלָיו ״שֶׁהֶחֱיָנוּ״ בַּקִּדּוּשׁ שֶׁל הַלַּיְלָה.
🤔 לָמָּה? אִם שְׁמִינִי עֲצֶרֶת הָיָה הֶמְשֵׁךְ שֶׁל סֻכּוֹת, הָיוּ מַקְרִיבִים בּוֹ שִׁשָּׁה פָּרִים – אֶחָד פָּחוֹת מִן הַיּוֹם הַשְּׁבִיעִי. אֲבָל הַתּוֹרָה צִוְּתָה פַּר אֶחָד, וּמִכָּאן שֶׁהוּא חַג חָדָשׁ.
(פֶּרֶק ז׳, הֲלָכָה א׳)

2״עֲשֵׂה לִי סְעוּדָה קְטַנָּה״

בְּשִׁבְעַת יְמֵי הֶחָג הִקְרִיבוּ שִׁבְעִים פָּרִים כְּנֶגֶד שִׁבְעִים אֻמּוֹת הָעוֹלָם, וּבִשְׁמִינִי עֲצֶרֶת – פַּר אֶחָד, כְּנֶגֶד אֻמָּה אַחַת: יִשְׂרָאֵל. מָשָׁל לְמֶלֶךְ שֶׁאָמַר לַעֲבָדָיו: ״עֲשׂוּ לִי סְעוּדָה גְּדוֹלָה״. בַּיּוֹם הָאַחֲרוֹן אָמַר לְאוֹהֲבוֹ: ״עֲשֵׂה לִי סְעוּדָה קְטַנָּה, כְּדֵי שֶׁאֵהָנֶה מִמְּךָ״.
🤔 לָמָּה? בַּיּוֹם הַזֶּה אֵין שׁוּם מִצְוָה מְיֻחֶדֶת – רַק לִשְׂמֹחַ בַּה׳, בַּתּוֹרָה שֶׁלּוֹ וּבַיְשׁוּעָה שֶׁלּוֹ: ״זֶה הַיּוֹם עָשָׂה ה׳ – נָגִילָה וְנִשְׂמְחָה בוֹ״.
💛 אַחֲרֵי שֶׁכָּל הָאוֹרְחִים הָלְכוּ, ה׳ רוֹצֶה לִהְיוֹת רַק אִתָּנוּ – כְּמוֹ אַבָּא שֶׁיּוֹשֵׁב בְּשֶׁקֶט עִם הַיֶּלֶד הָאָהוּב שֶׁלּוֹ.
(פֶּרֶק ז׳, הֲלָכָה ב׳)

3״קָשֶׁה עָלַי פְּרֵדַתְכֶם״

״עֲצֶרֶת״ – מִלְּשׁוֹן עֲצִירָה וְהִתְאַסְּפוּת. בַּיּוֹם הַזֶּה נִגְמָר מַסְלוּל הַחַגִּים שֶׁל הַשָּׁנָה: פֶּסַח, שָׁבוּעוֹת, רֹאשׁ הַשָּׁנָה, יוֹם הַכִּפּוּרִים וְסֻכּוֹת. מָשָׁל לְמֶלֶךְ שֶׁהִזְמִין אֶת בָּנָיו לִסְעוּדָה כַּמָּה יָמִים. כְּשֶׁהִגִּיעַ זְמַנָּם לָלֶכֶת, אָמַר: ״בָּנַי, בְּבַקָּשָׁה מִכֶּם, עִכְּבוּ עִמִּי עוֹד יוֹם – קָשֶׁה עָלַי פְּרֵדַתְכֶם״.
🤔 לָמָּה? הַשִּׂמְחָה בַּיּוֹם הַזֶּה כְּבָר אֵינָהּ תְּלוּיָה בְּסֻכָּה אוֹ בְּלוּלָב, אֶלָּא בְּכָךְ שֶׁאֲנַחְנוּ בָּנָיו שֶׁל ה׳. וּבַיּוֹם הַזֶּה אֶפְשָׁר ״לֶאֱצֹר״ בַּנְּשָׁמָה אֶת כָּל הָאוֹר שֶׁקִּבַּלְנוּ בַּחַגִּים, וְלָקַחַת אוֹתוֹ אִתָּנוּ לְכָל הַשָּׁנָה. הַגְּרָ״א אָמַר שֶׁזֶּה הַיּוֹם הַשָּׂמֵחַ בְּיוֹתֵר מִכָּל יְמֵי הֶחָג.
(פֶּרֶק ז׳, הֲלָכָה ג׳)

4מַשִּׁיב הָרוּחַ וּמוֹרִיד הַגֶּשֶׁם

בִּימוֹת הַחֹרֶף מַזְכִּירִים בַּבְּרָכָה הַשְּׁנִיָּה שֶׁל הָעֲמִידָה ״מַשִּׁיב הָרוּחַ וּמוֹרִיד הַגֶּשֶׁם״. בְּסֻכּוֹת עוֹד לֹא מַזְכִּירִים, וּמַתְחִילִים בִּתְפִלַּת מוּסָף שֶׁל שְׁמִינִי עֲצֶרֶת. מִי שֶׁשָּׁכַח לְהַזְכִּיר גֶּשֶׁם וְאָמַר ״מוֹרִיד הַטָּל״ – אֵינוֹ צָרִיךְ לַחֲזֹר.
🤔 לָמָּה? גֶּשֶׁם בְּסֻכּוֹת אֵינוֹ סִימָן טוֹב, כִּי הוּא מַפְרִיעַ לָשֶׁבֶת בַּסֻּכָּה. וְדַוְקָא בְּמוּסָף – כִּי אָז כָּל הַצִּבּוּר נִמְצָא בְּבֵית הַכְּנֶסֶת, וְאֶפְשָׁר לְהַכְרִיז לִפְנֵי כֻּלָּם.
(פֶּרֶק ז׳, הֲלָכָה ד׳)

5תְּפִלַּת הַגֶּשֶׁם

יַחַד עִם הַתְּחִלָּה שֶׁל ״מַשִּׁיב הָרוּחַ״ אוֹמְרִים אֶת תְּפִלַּת הַגֶּשֶׁם – בַּקָּשָׁה עַל גִּשְׁמֵי בְּרָכָה בַּשָּׁנָה הַבָּאָה. פּוֹתְחִים אֶת אֲרוֹן הַקֹּדֶשׁ, וְאוֹמְרִים אוֹתָהּ בְּכַוָּנָה וּבְתַחֲנוּנִים. טוֹב לְכַוֵּן גַּם עַל הַ״מַּיִם״ שֶׁל הַנְּשָׁמָה – שֶׁהַשָּׁנָה תִּהְיֶה מְבֹרֶכֶת בְּתוֹרָה.
🤔 לָמָּה? ״מָה מַיִם חַיִּים לָעוֹלָם – כָּךְ דִּבְרֵי תּוֹרָה חַיִּים לָעוֹלָם״.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי סְפָרַד וְחֵלֶק מִיּוֹצְאֵי אַשְׁכְּנַז אוֹמְרִים אֶת תְּפִלַּת הַגֶּשֶׁם לִפְנֵי תְּפִלַּת מוּסָף.
לְמִנְהַג רֹב יוֹצְאֵי אַשְׁכְּנַז אוֹמְרִים אוֹתָהּ בְּתוֹךְ חֲזָרַת הַחַזָּן שֶׁל מוּסָף, וְהַגַּבַּאי מַכְרִיז לִפְנֵי תְּפִלַּת הַלַּחַשׁ ״מַשִּׁיב הָרוּחַ וּמוֹרִיד הַגֶּשֶׁם״. הַחַזָּן לוֹבֵשׁ ״קִיטְל״ – בֶּגֶד לָבָן – וּמְנַגֵּן בְּנִגּוּן דּוֹמֶה לְנִגּוּן הַיָּמִים הַנּוֹרָאִים.
(פֶּרֶק ז׳, הֲלָכָה ד׳)

6״וְתֵן טַל וּמָטָר״ – רַק בְּז׳ בְּחֶשְׁוָן

אֶת הַבַּקָּשָׁה ״וְתֵן טַל וּמָטָר לִבְרָכָה״ בְּבִרְכַּת הַשָּׁנִים מַתְחִילִים בְּאֶרֶץ יִשְׂרָאֵל רַק בְּלֵיל ז׳ בְּחֹדֶשׁ מַרְחֶשְׁוָן – חֲמִשָּׁה עָשָׂר יוֹם אַחֲרֵי הֶחָג.
🤔 לָמָּה? חֲכָמִים דָּאֲגוּ לְעוֹלֵי הָרֶגֶל שֶׁבָּאוּ מֵרָחוֹק, כְּדֵי שֶׁגַּם הָאַחֲרוֹן שֶׁבָּהֶם, שֶׁבָּא מִנְּהַר פְּרָת, יַגִּיעַ לְבֵיתוֹ בְּלִי שֶׁיֵּרֵד עָלָיו גֶּשֶׁם בַּדֶּרֶךְ. וְגַם הַיּוֹם מַמְשִׁיכִים בַּמִּנְהָג הַיָּקָר הַזֶּה, שֶׁמַּזְכִּיר אֶת יְמֵי בֵּית הַמִּקְדָּשׁ.
💛 גַּם בַּבַּקָּשָׁה עַל הַגֶּשֶׁם אֲנַחְנוּ חוֹשְׁבִים קֹדֶם עַל הַיְּהוּדִי הָאַחֲרוֹן שֶׁעוֹד בַּדֶּרֶךְ.
(פֶּרֶק ז׳, הֲלָכָה ד׳)

7מְסַיְּמִים אֶת הַתּוֹרָה

בְּכָל שַׁבָּת קוֹרְאִים פָּרָשָׁה אַחַת בַּתּוֹרָה, וּבִשְׁמִינִי עֲצֶרֶת מְסַיְּמִים אֶת כָּל הַתּוֹרָה – וְלָכֵן הוּא נִקְרָא ״שִׂמְחַת תּוֹרָה״. לִכְבוֹד הַסִּיּוּם מַעֲלִים לַתּוֹרָה אֶת כָּל הַמִּתְפַּלְּלִים: חוֹזְרִים וְקוֹרְאִים אֶת הַפָּרָשָׁה שׁוּב וָשׁוּב, עַד שֶׁכֻּלָּם עוֹלִים. כְּשֶׁיֵּשׁ הַרְבֵּה מִתְפַּלְּלִים, מִתְחַלְּקִים לְכַמָּה מִנְיָנִים, וְאַחַר כָּךְ חוֹזְרִים יַחַד לְסִיּוּם הַתּוֹרָה.
🤔 לָמָּה? כְּשֶׁכֻּלָּם עוֹלִים לַתּוֹרָה, רוֹאִים שֶׁלְּכָל אֶחָד מִיִּשְׂרָאֵל יֵשׁ חֵלֶק בַּתּוֹרָה – זָקֵן וְצָעִיר, חָכָם וְתָם.
💛 גַּם לִי יֵשׁ חֵלֶק בַּתּוֹרָה – וְה׳ שָׂמֵחַ כְּשֶׁאֲנִי עוֹלֶה אֵלֶיהָ.
(פֶּרֶק ז׳, הֲלָכָה ה׳)

8״כָּל הַנְּעָרִים״

בְּשִׂמְחַת תּוֹרָה מַעֲלִים לַתּוֹרָה גַּם יְלָדִים שֶׁעוֹד לֹא הִגִּיעוּ לְמִצְווֹת. יֶלֶד שֶׁיּוֹדֵעַ לְבָרֵךְ עַל הַתּוֹרָה וְלִקְרֹא בְּלַחַשׁ עִם בַּעַל הַקּוֹרֵא – עוֹלֶה לְבַד. וְהַקְּטַנִּים עוֹלִים יַחַד בַּעֲלִיַּת ״כָּל הַנְּעָרִים״, עִם אֶחָד מִנִּכְבַּדֵי הַקָּהָל: הוּא אוֹמֵר אֶת הַבְּרָכוֹת לְאַט וּבְקוֹל, וְהַיְּלָדִים חוֹזְרִים אַחֲרָיו מִלָּה בְּמִלָּה. הָעוֹלֶה הַזֶּה נִקְרָא ״חֲתַן מְעֹנָה״, עַל שֵׁם הַמִּלָּה הָרִאשׁוֹנָה שֶׁל הַקְּרִיאָה.
(פֶּרֶק ז׳, הֲלָכָה ה׳)

9חֲתַן תּוֹרָה וַחֲתַן בְּרֵאשִׁית

הָעֲלִיָּה הַחֲשׁוּבָה בְּיוֹתֵר הִיא זוֹ שֶׁבָּהּ מְסַיְּמִים אֶת הַתּוֹרָה, וְהָעוֹלֶה נִקְרָא ״חֲתַן תּוֹרָה״. מִיָּד אַחֲרֶיהָ מוֹצִיאִים סֵפֶר שֵׁנִי וּמַתְחִילִים לִקְרֹא מִ״בְּרֵאשִׁית״ – וְהָעוֹלֶה נִקְרָא ״חֲתַן בְּרֵאשִׁית״. אַחֲרָיו מוֹצִיאִים סֵפֶר שְׁלִישִׁי, וְקוֹרְאִים בּוֹ לַ״מַּפְטִיר״ עַל קָרְבְּנוֹת הֶחָג. מִי שֶׁעָלָה לַ״חֲתַן תּוֹרָה״ – לֹא יַעֲלֶה מִיָּד גַּם לַ״חֲתַן בְּרֵאשִׁית״.
🤔 לָמָּה? אֲפִלּוּ לְרֶגַע אֵינֶנּוּ נִפְרָדִים מִן הַתּוֹרָה – גּוֹמְרִים וּמִיָּד מַתְחִילִים מֵחָדָשׁ. וּמִי שֶׁעוֹלֶה פַּעֲמַיִם בָּרֶצֶף נִרְאֶה כְּאִלּוּ הַסֵּפֶר הָרִאשׁוֹן הָיָה פָּסוּל.
(פֶּרֶק ז׳, הֲלָכָה ו׳)

10הַהַפְטָרָה – סֵפֶר יְהוֹשֻׁעַ

אַחֲרֵי סִיּוּם הַתּוֹרָה קוֹרְאִים לַהַפְטָרָה אֶת תְּחִלַּת סֵפֶר יְהוֹשֻׁעַ, וּבָהּ הַפְּסוּקִים: ״רַק חֲזַק וֶאֱמַץ מְאֹד לִשְׁמֹר לַעֲשׂוֹת כְּכָל הַתּוֹרָה... לֹא יָמוּשׁ סֵפֶר הַתּוֹרָה הַזֶּה מִפִּיךָ, וְהָגִיתָ בּוֹ יוֹמָם וָלַיְלָה״.
🤔 לָמָּה? כָּל הַתַּכְלִית שֶׁל הַתּוֹרָה הִיא לְקַיֵּם אוֹתָהּ בְּאֶרֶץ יִשְׂרָאֵל – וְסֵפֶר יְהוֹשֻׁעַ מְסַפֵּר עַל הַכְּנִיסָה לָאָרֶץ. וְרָאוּי מִיָּד אַחֲרֵי הַתּוֹרָה לְהַתְחִיל לִלְמֹד נְבִיאִים.
(פֶּרֶק ז׳, הֲלָכָה ו׳)

11הַקָּפוֹת

בְּלֵיל שִׂמְחַת תּוֹרָה וּבַיּוֹם עוֹרְכִים שֶׁבַע הַקָּפוֹת. מוֹצִיאִים אֶת כָּל סִפְרֵי הַתּוֹרָה, וּמַרְבִּים בְּרִקּוּדִים – גַּם מֵעֵבֶר לְשֶׁבַע הַהַקָּפוֹת, לִכְבוֹד הַתּוֹרָה. שָׁרִים פִּיּוּטִים, כָּל עֵדָה לְפִי מִנְהָגָהּ.
🤔 לָמָּה? הַהַקָּפוֹת רוֹמְזוֹת לְ״אוֹר מַקִּיף״ – אוֹר גָּדוֹל שֶׁל הַתּוֹרָה כֻּלָּהּ, שֶׁעוֹטֵף אוֹתָנוּ וּמְרוֹמֵם אוֹתָנוּ, גַּם אִם אֵינֶנּוּ יְכוֹלִים לְהָבִין אוֹתוֹ בַּשֵּׂכֶל.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז נוֹשְׂאִים אֶת כָּל סִפְרֵי הַתּוֹרָה וְרוֹקְדִים אִתָּם בְּכָל הַהַקָּפוֹת.
לְמִנְהַג יוֹצְאֵי סְפָרַד בְּרֹב הַזְּמַן סִפְרֵי הַתּוֹרָה מֻנָּחִים עַל הַבִּימָה, אֶחָד אוֹחֵז סֵפֶר תּוֹרָה, וְהַצִּבּוּר מַקִּיף וְרוֹקֵד סְבִיבָם. וְעַל פִּי הָאֲרִ״י מַקִּיפִים בְּכָל הַקָּפָה פַּעַם אַחַת עִם סֵפֶר תּוֹרָה.
(פֶּרֶק ז׳, הֲלָכָה ז׳)

12עוֹמְדִים לִכְבוֹד הַתּוֹרָה

כְּשֶׁמַּקִּיפִים וְרוֹקְדִים עִם סִפְרֵי הַתּוֹרָה – כָּל הַנִּמְצָאִים בְּבֵית הַכְּנֶסֶת עוֹמְדִים. מִי שֶׁקָּשֶׁה לוֹ לַעֲמֹד – יוֹשֵׁב, וְעוֹמֵד בִּתְחִלַּת כָּל הַקָּפָה. יֵשׁ נוֹהֲגִים לַעֲשׂוֹת גַּם ״הַקָּפוֹת שְׁנִיּוֹת״ בְּמוֹצָאֵי הֶחָג, עִם כְּלֵי נְגִינָה – וְגַם בָּהֶן יֵשׁ מִצְוָה, כִּי הֵן לִכְבוֹד הַתּוֹרָה.
(פֶּרֶק ז׳, הֲלָכָה ז׳)

13רוֹקְדִים וּמוֹחֲאִים כַּפַּיִם

אֵין שִׂמְחָה גְּדוֹלָה מִשִּׂמְחַת הַתּוֹרָה. בְּשַׁבָּת וּבְיוֹם טוֹב חֲכָמִים גָּזְרוּ שֶׁלֹּא לִרְקֹד וְלֹא לִמְחֹא כַּפַּיִם – אֲבָל לִכְבוֹד שִׂמְחַת הַתּוֹרָה הִתִּירוּ. עוֹרְכִים סְעוּדָה לִכְבוֹד סִיּוּם הַתּוֹרָה, כְּמוֹ שֶׁשְּׁלֹמֹה הַמֶּלֶךְ עָשָׂה מִשְׁתֶּה כְּשֶׁזָּכָה לְחָכְמָה. עַל הַגְּרָ״א סִפְּרוּ שֶׁהָיָה רוֹקֵד לִפְנֵי סֵפֶר הַתּוֹרָה בְּכָל כֹּחוֹ, וּפָנָיו מְאִירוֹת כְּמוֹ לַפִּיד.
🤔 לָמָּה? דָּוִד הַמֶּלֶךְ רָקַד בְּכָל עֹז לִפְנֵי אֲרוֹן ה׳, וְאָמַר: לֹא לִכְבוֹדִי רָקַדְתִּי, אֶלָּא ״לִפְנֵי ה׳... וְשִׂחַקְתִּי לִפְנֵי ה׳״. וְאָמְרוּ שֶׁמִּי שֶׁרוֹקֵד וְשָׂמֵחַ בְּכָל כֹּחוֹ לִכְבוֹד הַתּוֹרָה – מֻבְטָח לוֹ שֶׁהַתּוֹרָה לֹא תִּפָּסֵק מִזַּרְעוֹ.
💛 כְּשֶׁאֲנִי רוֹקֵד עִם סֵפֶר הַתּוֹרָה, אֲנִי מַרְאֶה לַה׳ שֶׁהַמַּתָּנָה שֶׁלּוֹ הִיא הַשִּׂמְחָה הֲכִי גְּדוֹלָה שֶׁלִּי.
(פֶּרֶק ז׳, הֲלָכָה ח׳)

14כֻּלָּם בָּאִים לְבֵית הַכְּנֶסֶת

נָהֲגוּ יִשְׂרָאֵל שֶׁגַּם הַנָּשִׁים וְהַטַּף בָּאִים לְבֵית הַכְּנֶסֶת לִכְבוֹד שִׂמְחַת הַתּוֹרָה.
🤔 לָמָּה? יֵשׁ בְּשִׂמְחַת תּוֹרָה דִּמְיוֹן לְמִצְוַת הַקְהֵל: כְּמוֹ שֶׁכָּל יִשְׂרָאֵל – אֲנָשִׁים, נָשִׁים וְטַף – הָיוּ מִתְאַסְּפִים לִשְׁמֹעַ אֶת הַמֶּלֶךְ קוֹרֵא בַּתּוֹרָה, כָּךְ כֻּלָּם מִתְאַסְּפִים לִכְבוֹד הַתּוֹרָה.
(פֶּרֶק ז׳, הֲלָכָה ח׳)

📝 מָה לָמַדְנוּ

  • שְׁמִינִי עֲצֶרֶת הוּא חַג בִּפְנֵי עַצְמוֹ – בְּלִי סֻכָּה וּבְלִי לוּלָב – וּמְבָרְכִים בּוֹ ״שֶׁהֶחֱיָנוּ״.
  • פַּר אֶחָד כְּנֶגֶד יִשְׂרָאֵל: ה׳ אוֹמֵר לָנוּ ״קָשֶׁה עָלַי פְּרֵדַתְכֶם״.
  • בְּמוּסָף שֶׁל שְׁמִינִי עֲצֶרֶת מַתְחִילִים ״מַשִּׁיב הָרוּחַ״, וּבְז׳ בְּחֶשְׁוָן ״וְתֵן טַל וּמָטָר״.
  • מְסַיְּמִים אֶת הַתּוֹרָה וּמַתְחִילִים מִ״בְּרֵאשִׁית״, וְכֻלָּם – גַּם הַיְּלָדִים – עוֹלִים לַתּוֹרָה.
  • רוֹקְדִים וּמוֹחֲאִים כַּפַּיִם בְּשֶׁבַע הַקָּפוֹת, כִּי אֵין שִׂמְחָה כְּשִׂמְחַת הַתּוֹרָה.

❓ שְׁאֵלוֹת חֲזָרָה

  1. מִנַּיִן יוֹדְעִים שֶׁשְּׁמִינִי עֲצֶרֶת הוּא חַג בִּפְנֵי עַצְמוֹ? תֵּן שְׁתֵּי רְאָיוֹת.
  2. לָמָּה מַתְחִילִים לְהַזְכִּיר גֶּשֶׁם רַק בִּשְׁמִינִי עֲצֶרֶת, וּלְבַקֵּשׁ גֶּשֶׁם רַק בְּז׳ בְּחֶשְׁוָן?
  3. לָמָּה מִיָּד אַחֲרֵי שֶׁמְּסַיְּמִים אֶת הַתּוֹרָה מַתְחִילִים לִקְרֹא אוֹתָהּ מֵחָדָשׁ?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״הוֹשַׁעְנָא רַבָּה וְשִׂמְחַת תּוֹרָה״

סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת פֶּסַח
מַהֲדוּרַת יְלָדִים

🔥 בִּטּוּל חָמֵץ וּבִעוּרוֹ

פֶּרֶק ה׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
הַבְּדִיקָה נִגְמְרָה, הַשַּׂקִּית עִם הַפְּתִיתִים סְגוּרָה, וְאַבָּא אוֹמֵר מִשְׁפָּט מוּזָר: ״כָּל חֲמִירָא וַחֲמִיעָא...״. מָה הוּא אוֹמֵר? וְלָמָּה לְמָחֳרָת בַּבֹּקֶר כֻּלָּם עוֹמְדִים לְיַד מְדוּרָה קְטַנָּה וּמְבִיאִים אֶת הַחָמֵץ הָאַחֲרוֹן? בַּפֶּרֶק הַזֶּה נִלְמַד אֵיךְ מְבַטְּלִים אֶת הֶחָמֵץ בַּלֵּב, אֵיךְ שׂוֹרְפִים אוֹתוֹ – וּמָה עוֹשִׂים אִם פִּתְאוֹם מוֹצְאִים עוּגִיָּה בְּאֶמְצַע הֶחָג.

1הַבִּטּוּל הָרִאשׁוֹן – מִיָּד אַחֲרֵי הַבְּדִיקָה

מִיָּד אַחֲרֵי הַבְּדִיקָה מְבַטְּלִים אֶת הֶחָמֵץ – וְכָךְ מְבַעֲרִים אוֹתוֹ בַּמַּחְשָׁבָה. הַנֹּסַח כָּתוּב בַּאֲרָמִית: ״כָּל חֲמִירָא וַחֲמִיעָא דְּאִכָּא בִרְשׁוּתִי, דְּלָא חֲזִיתֵהּ וּדְלָא בִעַרְתֵּהּ, לִבָּטֵל וְלֶהֱוֵי הֶפְקֵר כְּעַפְרָא דְאַרְעָא״. וְאֶפְשָׁר לוֹמַר אוֹתוֹ בְּעִבְרִית: ״כָּל חָמֵץ וּשְׂאוֹר שֶׁיֵּשׁ בִּרְשׁוּתִי, שֶׁלֹּא רְאִיתִיו וְשֶׁלֹּא בִּעַרְתִּיו, יִבָּטֵל וִיהֵא הֶפְקֵר כַּעֲפַר הָאָרֶץ״.
🤔 לָמָּה? הַנֹּסַח נִכְתַּב בַּאֲרָמִית, כִּי בַּתְּקוּפָה שֶׁבָּהּ חִבְּרוּ אוֹתוֹ רַבִּים מֵהָעָם הֵבִינוּ רַק אֲרָמִית.
(פֶּרֶק ה׳, הֲלָכָה א׳)

2מָה לֹא מְבַטְּלִים בַּלַּיְלָה?

בַּלַּיְלָה מְבַטְּלִים רַק אֶת הֶחָמֵץ שֶׁלֹּא מָצָאנוּ – ״שֶׁלֹּא רְאִיתִיו״. אֶת הֶחָמֵץ שֶׁמָּצָאנוּ בַּבְּדִיקָה, וְאֶת הֶחָמֵץ שֶׁשּׁוֹמְרִים לַאֲרוּחַת הָעֶרֶב וַאֲרוּחַת הַבֹּקֶר – לֹא מְבַטְּלִים.
🤔 לָמָּה? הֶחָמֵץ שֶׁאֲנַחְנוּ עוֹמְדִים לֶאֱכֹל חָשׁוּב לָנוּ, וְאִי אֶפְשָׁר לְבַטֵּל אוֹתוֹ בְּלֵב שָׁלֵם. מִי שֶׁ״מְבַטֵּל״ חָמֵץ שֶׁהוּא מִתְכַּנֵּן לֶאֱכֹל – הַבִּטּוּל שֶׁלּוֹ רַק מִן הַשָּׂפָה וְלַחוּץ, וְלֹא מוֹעִיל כְּלָל. וְאֶת הֶחָמֵץ שֶׁמָּצָאנוּ – רוֹצִים לִשְׂרֹף.
(פֶּרֶק ה׳, הֲלָכָה א׳)

3הַבִּטּוּל הַשֵּׁנִי – בַּבֹּקֶר

בְּבֹקֶר י״ד בְּנִיסָן, אַחֲרֵי שֶׁשּׂוֹרְפִים אֶת הֶחָמֵץ, מְבַטְּלִים שׁוּב – הַפַּעַם אֶת כָּל הֶחָמֵץ: ״כָּל חָמֵץ וּשְׂאוֹר שֶׁיֵּשׁ בִּרְשׁוּתִי, שֶׁרְאִיתִיו וְשֶׁלֹּא רְאִיתִיו, שֶׁבִּעַרְתִּיו וְשֶׁלֹּא בִּעַרְתִּיו, יִבָּטֵל וִיהֵא הֶפְקֵר כַּעֲפַר הָאָרֶץ״. מַקְפִּידִים לְבַטֵּל עַד סוֹף הַשָּׁעָה הַחֲמִישִׁית.
🤔 לָמָּה? אוּלַי חָמֵץ מֵאֲרוּחַת הַבֹּקֶר נִשְׁמַט מֵהָעֵינַיִם וְנִשְׁכַּח – וְהַבִּטּוּל הַשֵּׁנִי שׁוֹמֵר שֶׁלֹּא נַעֲבֹר עָלָיו. וּמִתְּחִלַּת הַשָּׁעָה הַשִּׁשִּׁית הֶחָמֵץ אָסוּר בַּהֲנָאָה, וּכְבָר אִי אֶפְשָׁר לְבַטֵּל אוֹתוֹ.
(פֶּרֶק ה׳, הֲלָכָה א׳)

4שְׁנֵי נֻסְחָאוֹת

בְּנֹסַח הַבִּטּוּל יֵשׁ שְׁנֵי מִנְהָגִים, וּשְׁנֵיהֶם כַּהֲלָכָה:
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְנֹסַח סְפָרַד מַזְכִּירִים רַק ״חָמֵץ״, כִּי הוּא כּוֹלֵל גַּם אֶת הַשְּׂאוֹר; וּמַזְכִּירִים רַק ״בִּטּוּל״, כִּי מָה שֶׁמְּבֻטָּל מִמֵּילָא מֻפְקָר.
לְנֹסַח אַשְׁכְּנַז מַזְכִּירִים ״חֲמִירָא וַחֲמִיעָא״ – חָמֵץ וּשְׂאוֹר, וּמַזְכִּירִים גַּם בִּטּוּל וְגַם הֶפְקֵר.
(פֶּרֶק ה׳, הֲלָכָה א׳, הֲלָכָה ב׳)

5מָה עוֹשֶׂה הַבִּטּוּל?

אֵיךְ מִשְׁפָּט אֶחָד גּוֹרֵם שֶׁלֹּא נַעֲבֹר עַל ״לֹא יֵרָאֶה״? יֵשׁ שְׁנֵי פֵּרוּשִׁים:
🤔 לָמָּה? בְּנֹסַח אַשְׁכְּנַז מִתְחַשְּׁבִים בִּשְׁנֵי הַפֵּרוּשִׁים, וְלָכֵן אוֹמְרִים גַּם ״יִבָּטֵל״ וְגַם ״הֶפְקֵר״.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְפִי רַשִׁ״י וְהָרַמְבַּ״ן בְּפֶסַח הַתּוֹרָה אוֹמֶרֶת שֶׁהֶחָמֵץ כְּלוּם, כְּמוֹ עָפָר. כְּשֶׁאָנוּ מְבַטְּלִים, אָנוּ מַסְכִּימִים עִם הַתּוֹרָה – וְהֶחָמֵץ נַעֲשֶׂה אֶצְלֵנוּ כְּמוֹ עָפָר.
לְפִי בַּעֲלֵי הַתּוֹסָפוֹת הַבִּטּוּל מַפְקִיר אֶת הֶחָמֵץ. חָמֵץ שֶׁל הֶפְקֵר אֵינוֹ שֶׁלָּנוּ, וְעָלָיו אֵין אִסּוּר ״לֹא יֵרָאֶה לְךָ״.
(פֶּרֶק ה׳, הֲלָכָה ב׳)

6צָרִיךְ לְהָבִין מָה אוֹמְרִים

אֶפְשָׁר לוֹמַר אֶת הַבִּטּוּל בַּאֲרָמִית, בְּעִבְרִית אוֹ בְּכָל שָׂפָה – הָעִקָּר לְהָבִין. מִי שֶׁאָמַר אֶת הַנֹּסַח בַּאֲרָמִית, וְלֹא הֵבִין שֶׁהוּא מְבַטֵּל חָמֵץ, אֶלָּא חָשַׁב שֶׁזּוֹ תְּפִלָּה לִקְרַאת הֶחָג – לֹא בִּטֵּל אֶת חֲמֵצוֹ.
🤔 לָמָּה? הַבִּטּוּל הוּא הַחְלָטָה שֶׁל הַלֵּב. מִי שֶׁלֹּא יוֹדֵעַ מָה הוּא אוֹמֵר – לֹא הֶחְלִיט כְּלוּם.
💛 גַּם בַּתְּפִלָּה וּבַבְּרָכוֹת – כְּשֶׁאֲנִי מֵבִין אֶת הַמִּלִּים, הֵן יוֹצְאוֹת מֵהַלֵּב וּמַגִּיעוֹת לַה׳.
(פֶּרֶק ה׳, הֲלָכָה ב׳)

7בַּלֵּב וּבַפֶּה

לְפִי רֹב הָרִאשׁוֹנִים, מִן הַדִּין אֶפְשָׁר לְבַטֵּל אֶת הֶחָמֵץ גַּם בַּלֵּב בִּלְבַד. אֲבָל לְכַתְּחִלָּה אוֹמְרִים אֶת הַבִּטּוּל בַּפֶּה, כְּדֵי שֶׁיִּהְיֶה בָּרוּר. לֹא צָרִיךְ לוֹמַר אוֹתוֹ לִפְנֵי אֲנָשִׁים אֲחֵרִים, וְיֵשׁ מְהַדְּרִים לוֹמַר אוֹתוֹ לִפְנֵי בְּנֵי הַמִּשְׁפָּחָה, כְּדֵי לְהַזְכִּיר לָהֶם אֶת הַמִּצְוָה. אֶפְשָׁר לְבַטֵּל עַל יְדֵי שָׁלִיחַ, אֲבָל עָדִיף שֶׁבַּעַל הֶחָמֵץ יְבַטֵּל בְּעַצְמוֹ.
🤔 לָמָּה? יֵשׁ מֵהָרִאשׁוֹנִים שֶׁסּוֹבְרִים שֶׁחַיָּבִים לוֹמַר אֶת הַבִּטּוּל בַּפֶּה, וְיֵשׁ שֶׁסּוֹבְרִים שֶׁרַק בַּעַל הֶחָמֵץ יָכוֹל לְבַטֵּל. לָכֵן מְהַדְּרִים לָצֵאת יְדֵי כֻּלָּם.
(פֶּרֶק ה׳, הֲלָכָה ג׳)

8בְּלֵב שָׁלֵם

הַבִּטּוּל צָרִיךְ לִהְיוֹת בְּלֵב שָׁלֵם: לְהַחְלִיט בֶּאֱמֶת שֶׁהֶחָמֵץ בָּטֵל לְעוֹלָם, וְשֶׁגַּם אַחֲרֵי פֶּסַח לֹא נִשְׁתַּמֵּשׁ בּוֹ. מִי שֶׁמְּבַטֵּל וּבְלִבּוֹ חוֹשֵׁב לֶאֱכֹל אֶת הֶחָמֵץ אַחֲרֵי הֶחָג – הַבִּטּוּל שֶׁלּוֹ אֵינוֹ בִּטּוּל, וְהוּא עוֹבֵר עַל ״לֹא יֵרָאֶה״ וְ״לֹא יִמָּצֵא״.
💛 ה׳ רוֹאֶה אֶת הַלֵּב. הַמִּלִּים חֲשׁוּבוֹת, אֲבָל מָה שֶׁקּוֹבֵעַ הוּא מָה שֶׁאֲנִי בֶּאֱמֶת מַחְלִיט.
(פֶּרֶק ה׳, הֲלָכָה ג׳)

9שָׁכַח לְבַעֵר וְהוּא רָחוֹק מֵהַבַּיִת

מִי שֶׁשָּׁכַח לְבַעֵר אֶת הֶחָמֵץ, וְהִגִּיעַ עֶרֶב פֶּסַח כְּשֶׁהוּא רָחוֹק מֵהַבַּיִת – מְבַטֵּל אֶת הֶחָמֵץ, וּמִיָּד כְּשֶׁהוּא חוֹזֵר הַבַּיְתָה – שׂוֹרֵף אוֹתוֹ. וַאֲפִלּוּ אִם חָזַר אַחֲרֵי פֶּסַח – חַיָּב לְבַעֵר אוֹתוֹ.
🤔 לָמָּה? אִם יֹאכַל אוֹתוֹ אַחֲרֵי פֶּסַח, הוּא יַרְאֶה שֶׁהַבִּטּוּל שֶׁלּוֹ לֹא הָיָה בְּלֵב שָׁלֵם.
(פֶּרֶק ה׳, הֲלָכָה ג׳)

10שְׂרֵפַת הֶחָמֵץ

בְּבֹקֶר עֶרֶב פֶּסַח מְבַעֲרִים אֶת הֶחָמֵץ שֶׁנִּמְצָא בַּבְּדִיקָה (וְאֶת עֲשֶׂרֶת הַפְּתִיתִים) וְאֶת מָה שֶׁנִּשְׁאַר מֵאֲרוּחַת הַבֹּקֶר. מִן הַדִּין אֶפְשָׁר לְבַעֵר בִּדְרָכִים שׁוֹנוֹת: לְפוֹרֵר וּלְפַזֵּר לָרוּחַ, לִזְרֹק לַיָּם אוֹ לַנָּהָר, אוֹ לְפוֹרֵר וּלְהוֹרִיד בַּשֵּׁרוּתִים. אֲבָל יִשְׂרָאֵל קְדוֹשִׁים נָהֲגוּ לְהַדֵּר וְלִשְׂרֹף.
🤔 לָמָּה? אֵין דָּבָר שֶׁמַּעֲלִים אֶת הֶחָמֵץ מִן הָעוֹלָם יוֹתֵר מֵהַשְּׂרֵפָה. וְיֵשׁ מִי שֶׁסּוֹבֵר שֶׁהַמִּצְוָה הִיא דַּוְקָא בִּשְׂרֵפָה.
💛 כְּשֶׁאֲנִי רוֹאֶה אֶת הֶחָמֵץ נִשְׂרָף, אֲנִי יָכוֹל לְבַקֵּשׁ מֵה׳ שֶׁגַּם הַגַּאֲוָה וְהַכַּעַס שֶׁבִּי ״יִשָּׂרְפוּ״, וְאֶכָּנֵס לֶחָג נָקִי.
(פֶּרֶק ה׳, הֲלָכָה ד׳)

11קֹדֶם שׂוֹרְפִים – וְאַחַר כָּךְ מְבַטְּלִים

מִי שֶׁמְּהַדֵּר לִשְׂרֹף, מְבַטֵּל אֶת הֶחָמֵץ רַק אַחֲרֵי הַשְּׂרֵפָה. אַחֲרֵי שֶׁנִּשְׂרַף כַּזַּיִת מֵהֶחָמֵץ, כְּבָר קִיַּמְנוּ אֶת הַהִדּוּר, וְאֶפְשָׁר לוֹמַר אֶת הַבִּטּוּל. צָרִיךְ לְהַקְפִּיד לְהַשְׁאִיר זְמַן לַבִּטּוּל לִפְנֵי סוֹף הַשָּׁעָה הַחֲמִישִׁית. וְיֵשׁ מְדַקְדְּקִים: אִם מַדְלִיקִים אֶת הָאֵשׁ בְּנֵפְט – שׁוֹפְכִים אוֹתוֹ רַק עַל הָעֵצִים וְלֹא עַל הֶחָמֵץ.
🤔 לָמָּה? אִם נְבַטֵּל לִפְנֵי הַשְּׂרֵפָה – הֶחָמֵץ כְּבָר לֹא יִהְיֶה שֶׁלָּנוּ, וְלֹא נוּכַל לְקַיֵּם בּוֹ אֶת מִצְוַת הַבִּעוּר. וְאֶת הַנֵּפְט לֹא שׁוֹפְכִים עַל הֶחָמֵץ, כְּדֵי שֶׁרַק הָאֵשׁ תְּכַלֶּה אוֹתוֹ, וְהוּא לֹא יִפָּסֵל לִפְנֵי כֵן.
(פֶּרֶק ה׳, הֲלָכָה ד׳)

12חָמֵץ בְּפַח הָאַשְׁפָּה

פַּח אַשְׁפָּה שֶׁשַּׁיָּךְ לָנוּ, אוֹ שֶׁעוֹמֵד בַּחֲצַר הַבַּיִת – לְכַתְּחִלָּה שׁוֹפְכִים עַל הֶחָמֵץ שֶׁבּוֹ אֶקוֹנוֹמִיקָה אוֹ חֹמֶר אַחֵר, עַד שֶׁגַּם כֶּלֶב לֹא יֹאכַל אוֹתוֹ. בִּשְׁעַת הַצֹּרֶךְ אֶפְשָׁר לְהָקֵל, כִּי חָמֵץ שֶׁנִּזְרַק לַפַּח כְּבָר נֶחְשָׁב מָאוּס. וּפַח שֶׁל הָעִירִיָּה בָּרְחוֹב – אֵין צֹרֶךְ לְבַעֵר אֶת מָה שֶׁבּוֹ.
🤔 לָמָּה? הַפַּח שֶׁל הָעִירִיָּה אֵינוֹ שֶׁלָּנוּ, וְגַם הָעִירִיָּה לֹא רוֹצָה אֶת הֶחָמֵץ – הִיא רַק מְפַנָּה אוֹתוֹ לַמִּזְבָּלָה.
(פֶּרֶק ה׳, הֲלָכָה ה׳)

13מָצָאתִי חָמֵץ אַחֲרֵי חֲצוֹת

מִי שֶׁמּוֹצֵא חָמֵץ אַחֲרֵי חֲצוֹת יוֹם י״ד אוֹ בְּתוֹךְ הַפֶּסַח – מְבַעֵר אוֹתוֹ מִיָּד. מִצְוָה לִשְׂרֹף אוֹתוֹ, וְאֶפְשָׁר גַּם לְפוֹרֵר וּלְפַזֵּר לָרוּחַ, אוֹ לְפוֹרֵר וּלְהוֹרִיד בַּשֵּׁרוּתִים. אֲבָל לִזְרֹק אוֹתוֹ לְמָקוֹם צִבּוּרִי – לֹא מוֹעִיל. וְגַם לִשְׁפֹּךְ עָלָיו אֶקוֹנוֹמִיקָה – כְּבָר לֹא מוֹעִיל.
🤔 לָמָּה? אֶקוֹנוֹמִיקָה מוֹעִילָה רַק לִפְנֵי זְמַן הָאִסּוּר. אִם בִּזְמַן הָאִסּוּר הֶחָמֵץ הָיָה רָאוּי לַאֲכִילָה, חַיָּבִים לְהַעֲלִים אוֹתוֹ לְגַמְרֵי מִן הָעוֹלָם. וְגַם הָאֵפֶר שֶׁלּוֹ אָסוּר בַּהֲנָאָה.
(פֶּרֶק ה׳, הֲלָכָה ו׳)

14מָצָאתִי חָמֵץ בְּשַׁבָּת אוֹ בְּיוֹם טוֹב

מִי שֶׁמּוֹצֵא חָמֵץ בְּשַׁבָּת אוֹ בְּיוֹם טוֹב – אָסוּר לוֹ לִשְׂרֹף אוֹתוֹ, וְגַם אָסוּר לְטַלְטֵל אוֹתוֹ, כִּי הוּא מֻקְצֶה. לָכֵן מְכַסִּים אוֹתוֹ בִּכְלִי, כְּדֵי שֶׁלֹּא יֹאכְלוּ אוֹתוֹ בְּטָעוּת, וּמִיָּד בְּמוֹצָאֵי הַשַּׁבָּת אוֹ הֶחָג – שׂוֹרְפִים. וּמָה אִם שָׁכַח גַּם לְבַטֵּל אֶת חֲמֵצוֹ לִפְנֵי פֶּסַח?
🤔 לָמָּה? מִי שֶׁלֹּא בִּטֵּל, עוֹבֵר בְּכָל רֶגַע עַל ״לֹא יֵרָאֶה״ וְ״לֹא יִמָּצֵא״ – אִסּוּרֵי תּוֹרָה. אִסּוּר מֻקְצֶה הוּא מִדִּבְרֵי חֲכָמִים, וְלָכֵן רַבִּים הִתִּירוּ לוֹ לְטַלְטֵל.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת פּוֹסְקִים רַבִּים מֻתָּר לָקַחַת אֶת הֶחָמֵץ, לְפוֹרֵר אוֹתוֹ מְעַט וּלְהוֹרִיד בַּשֵּׁרוּתִים – כְּדֵי שֶׁלֹּא יַעֲבֹר עַל אִסּוּרֵי תּוֹרָה.
וְיֵשׁ מַחְמִירִים לֹא נוֹגְעִים בַּמֻּקְצֶה, וּמְחַכִּים עַד מוֹצָאֵי שַׁבָּת – וְאָז מְבַעֲרִים.
(פֶּרֶק ה׳, הֲלָכָה ו׳)

15מְבָרְכִים עַל חָמֵץ שֶׁנִּמְצָא?

מִי שֶׁמּוֹצֵא חָמֵץ בְּפֶסַח – מְבַעֵר אוֹתוֹ בְּלִי בְּרָכָה. אֲבָל אִם הֶחָמֵץ נוֹצַר אוֹ הִגִּיעַ אֵלָיו רַק בְּתוֹךְ הַפֶּסַח – לְמָשָׁל בָּצֵק שֶׁהֶחְמִיץ לוֹ, אוֹ שֶׁקָּנָה חָמֵץ בְּטָעוּת – מְבָרֵךְ ״עַל בִּעוּר חָמֵץ״.
🤔 לָמָּה? הַבְּרָכָה שֶׁבֵּרַךְ לִפְנֵי הַבְּדִיקָה חָלָה עַל כָּל הֶחָמֵץ שֶׁהָיָה לוֹ אָז. חָמֵץ חָדָשׁ – לֹא הָיָה אָז בִּרְשׁוּתוֹ, וְלָכֵן הַבְּרָכָה לֹא חָלָה עָלָיו.
(פֶּרֶק ה׳, הֲלָכָה ו׳)

📝 מָה לָמַדְנוּ

  • אַחֲרֵי הַבְּדִיקָה מְבַטְּלִים אֶת הֶחָמֵץ שֶׁלֹּא נִמְצָא, וּבַבֹּקֶר – אַחֲרֵי הַשְּׂרֵפָה – מְבַטְּלִים אֶת הַכֹּל, עַד סוֹף הַשָּׁעָה הַחֲמִישִׁית.
  • לְנֹסַח סְפָרַד מַזְכִּירִים חָמֵץ וּבִטּוּל; לְנֹסַח אַשְׁכְּנַז – חָמֵץ וּשְׂאוֹר, בִּטּוּל וְהֶפְקֵר.
  • צָרִיךְ לְהָבִין מָה אוֹמְרִים וּלְבַטֵּל בְּלֵב שָׁלֵם.
  • יִשְׂרָאֵל קְדוֹשִׁים נָהֲגוּ לְבַעֵר אֶת הֶחָמֵץ בִּשְׂרֵפָה.
  • חָמֵץ שֶׁנִּמְצָא בְּפֶסַח מְבַעֲרִים מִיָּד, בְּלִי בְּרָכָה; בְּשַׁבָּת וּבְיוֹם טוֹב – מְכַסִּים בִּכְלִי וְשׂוֹרְפִים בְּמוֹצָאֵי הֶחָג.

❓ שְׁאֵלוֹת חֲזָרָה

  1. מָה הַהֶבְדֵּל בֵּין הַבִּטּוּל שֶׁל הַלַּיְלָה לַבִּטּוּל שֶׁל הַבֹּקֶר, וְלָמָּה?
  2. מִי שֶׁאָמַר אֶת הַבִּטּוּל בַּאֲרָמִית וְחָשַׁב שֶׁזּוֹ תְּפִלָּה – הַאִם בִּטֵּל? לָמָּה?
  3. בְּאֶמְצַע הֶחָג מָצָאתָ עוּגִיָּה עַל הַמַּדָּף. מָה עוֹשִׂים בְּיוֹם חֹל הַמּוֹעֵד, וּמָה עוֹשִׂים בְּיוֹם טוֹב?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״מְחַפְּשִׂים וּמְבַעֲרִים״

מַהֲדוּרַת יְלָדִים

🕯️ בְּדִיקַת חָמֵץ

פֶּרֶק ד׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
הַבַּיִת חָשׁוּךְ וְשָׁקֵט. אַבָּא מַדְלִיק נֵר, כֻּלָּם עוֹמְדִים סְבִיבוֹ וְשׁוֹמְעִים בְּרָכָה, וְאָז – יוֹצְאִים לְחַפֵּשׂ! מִתַּחַת לַמִּטָּה, בַּכִּיסִים, בַּמְּכוֹנִית... זֶה אֶחָד הָרְגָעִים הַמְּרַגְּשִׁים בַּשָּׁנָה. בַּפֶּרֶק הַזֶּה נִלְמַד מָתַי בּוֹדְקִים, אֵיפֹה בּוֹדְקִים, בַּמֶּה בּוֹדְקִים – וְאֵיךְ כָּל הַמִּשְׁפָּחָה יְכוֹלָה לְהִשְׁתַּתֵּף.

1בְּלֵיל י״ד בְּנִיסָן

בּוֹדְקִים חָמֵץ בִּתְחִלַּת הַלַּיְלָה שֶׁל י״ד בְּנִיסָן – הַלַּיְלָה שֶׁלִּפְנֵי עֶרֶב פֶּסַח. לִכְאוֹרָה הָיָה מַתְאִים לִבְדֹּק בַּיּוֹם, סָמוּךְ לִזְמַן הַבִּעוּר, אֲבָל חֲכָמִים תִּקְּנוּ לִבְדֹּק בַּלַּיְלָה.
🤔 לָמָּה? בַּיּוֹם אֲנָשִׁים טְרוּדִים בָּעֲבוֹדָה וַעֲלוּלִים לִשְׁכֹּחַ, וּבַלַּיְלָה כֻּלָּם בַּבַּיִת. וְעוֹד: בַּלַּיְלָה אוֹר הַנֵּר בּוֹלֵט וּמֵאִיר הֵיטֵב אֶת הַחוֹרִים וְהַסְּדָקִים, וּבַיּוֹם אוֹר הַשֶּׁמֶשׁ מַחְלִישׁ אוֹתוֹ.
(פֶּרֶק ד׳, הֲלָכָה א׳)

2קֹדֶם עַרְבִית – וְאַחַר כָּךְ מְמַהֲרִים

בִּתְחִלַּת הַלַּיְלָה מִתְפַּלְּלִים קֹדֶם עַרְבִית, וְאַחַר כָּךְ מִזְדָּרְזִים לִבְדֹּק חָמֵץ. מִי שֶׁרָגִיל לְהִתְפַּלֵּל בְּמִנְיָן מְאֻחָר יוֹתֵר – בּוֹדֵק בְּצֵאת הַכּוֹכָבִים, וּמִתְפַּלֵּל בַּשָּׁעָה שֶׁהוּא רָגִיל.
🤔 לָמָּה? עַרְבִית הִיא מִצְוָה שֶׁעוֹשִׂים בְּכָל יוֹם, וּמִצְוָה תְּדִירָה – מַקְדִּימִים.
(פֶּרֶק ד׳, הֲלָכָה א׳)

3מִי בּוֹדֵק?

הַחוֹבָה לִבְדֹּק הִיא עַל בַּעַל הַבַּיִת. אִם קָשֶׁה לוֹ – מִפְּנֵי שֶׁהוּא חַלָּשׁ אוֹ שֶׁאֵינוֹ רוֹאֶה טוֹב – הוּא מְמַנֶּה אֶת אִשְׁתּוֹ אוֹ אֶחָד מִבְּנֵי הַבַּיִת הַגְּדוֹלִים שֶׁיִּבְדְּקוּ בִּמְקוֹמוֹ. בַּמִּצְוָה הַזֹּאת אֵין הֶבְדֵּל בֵּין אִישׁ לְאִשָּׁה – הָעִקָּר לִבְחֹר מִי שֶׁיִּבְדֹּק בְּאַחְרָיוּת. וְאִם בַּעַל הַבַּיִת יַחֲזֹר מְאֻחָר, הוּא מְמַנֶּה מִישֶׁהוּ שֶׁיִּבְדֹּק בַּזְּמַן.
(פֶּרֶק ד׳, הֲלָכָה א׳)

4חֲצִי שָׁעָה לִפְנֵי הַבְּדִיקָה

חֲצִי שָׁעָה לִפְנֵי זְמַן הַבְּדִיקָה לֹא מַתְחִילִים סְעוּדָה אוֹ עֲבוֹדָה. מֻתָּר לֶאֱכֹל פֵּרוֹת אוֹ עוּגָה בְּדֶרֶךְ אֲרָעִי, וּמֻתָּר לַעֲשׂוֹת עֲבוֹדָה קַלָּה וּקְצָרָה.
🤔 לָמָּה? חֲכָמִים חָשְׁשׁוּ שֶׁמִּי שֶׁמַּתְחִיל אֲרוּחָה גְּדוֹלָה אוֹ עֲבוֹדָה אֲרֻכָּה – יִמָּשֵׁךְ אַחֲרֶיהָ וְיִשְׁכַּח לִבְדֹּק.
(פֶּרֶק ד׳, הֲלָכָה ב׳)

5הִגִּיעַ הַזְּמַן – בּוֹדְקִים!

כְּשֶׁמַּגִּיעַ זְמַן הַבְּדִיקָה, לֹא עוֹשִׂים שׁוּם עֲבוֹדָה וְלֹא אוֹכְלִים, אֲפִלּוּ לֹא פְּרִי קָטָן – אֶלָּא מְמַהֲרִים לִבְדֹּק. גַּם לֹא מַתְחִילִים לִלְמֹד תּוֹרָה. שִׁעוּר תּוֹרָה קָבוּעַ לָרַבִּים – לֹא מְבַטְּלִים, וְטוֹב שֶׁבְּסוֹפוֹ יַזְכִּירוּ זֶה לָזֶה לְמַהֵר לִבְדֹּק. וּמָה עִם מִי שֶׁכְּבָר הִתְחִיל לִלְמֹד לִפְנֵי הַזְּמַן?
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ מוֹרִים מִי שֶׁהִתְחִיל לִלְמֹד לִפְנֵי זְמַן הַבְּדִיקָה – יָכוֹל לְהַמְשִׁיךְ.
לְדַעַת הַרְבֵּה פּוֹסְקִים גַּם הוּא מַפְסִיק בְּצֵאת הַכּוֹכָבִים, כְּדֵי לְקַיֵּם אֶת הַבְּדִיקָה בִּזְמַנָּהּ.
💛 ״זְרִיזִין מַקְדִּימִין לְמִצְווֹת״ – כְּשֶׁאֲנִי מְמַהֵר לַמִּצְוָה, אֲנִי מַרְאֶה לַה׳ כַּמָּה הִיא חֲשׁוּבָה לִי.
(פֶּרֶק ד׳, הֲלָכָה ב׳)

6הַבְּרָכָה

לִפְנֵי הַבְּדִיקָה מְבָרְכִים: ״בָּרוּךְ אַתָּה ה׳ אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם, אֲשֶׁר קִדְּשָׁנוּ בְּמִצְווֹתָיו וְצִוָּנוּ עַל בִּעוּר חָמֵץ״. מִי שֶׁיֵּשׁ לוֹ כַּמָּה בָּתִּים – בּוֹדֵק אֶת כֻּלָּם, וּמְבָרֵךְ רַק פַּעַם אַחַת, בַּמָּקוֹם הָרִאשׁוֹן.
🤔 לָמָּה? אֲנַחְנוּ אוֹמְרִים ״עַל בִּעוּר חָמֵץ״ וְלֹא ״עַל בְּדִיקַת חָמֵץ״, כִּי הַבְּדִיקָה הִיא הַהַתְחָלָה שֶׁל כָּל תַּהֲלִיךְ הַבִּעוּר, שֶׁנִּגְמָר לְמָחֳרָת בַּשְּׂרֵפָה וּבַבִּטּוּל.
(פֶּרֶק ד׳, הֲלָכָה ג׳)

7שֶׁקֶט – מְחַפְּשִׂים!

בֵּין הַבְּרָכָה לִתְחִלַּת הַבְּדִיקָה – אָסוּר לְדַבֵּר. מִי שֶׁדִּבֵּר בְּעִנְיָן אַחֵר – הִפְסִיד אֶת הַבְּרָכָה, וְצָרִיךְ לְבָרֵךְ שׁוּב. וְאִם כְּבָר הִתְחִיל לִבְדֹּק וְדִבֵּר – לֹא הִפְסִיד אֶת הַבְּרָכָה. אֲבָל לְכַתְּחִלָּה גַּם בְּמֶשֶׁךְ כָּל הַבְּדִיקָה לֹא מְדַבְּרִים בְּעִנְיָנִים אֲחֵרִים.
🤔 לָמָּה? כָּךְ הַבּוֹדֵק מִתְרַכֵּז בַּבְּדִיקָה. מֻתָּר לְדַבֵּר עַל הַבְּדִיקָה עַצְמָהּ – לְמָשָׁל ״בָּדַקְתָּ אֶת הַמְּגֵרָה?״
(פֶּרֶק ד׳, הֲלָכָה ג׳)

8מָתַי לֹא מְבָרְכִים?

מִי שֶׁנּוֹסֵעַ וּבוֹדֵק בְּאֶחָד הַלֵּילוֹת שֶׁלִּפְנֵי י״ד – הַבְּדִיקָה כְּשֵׁרָה, אֲבָל הוּא לֹא מְבָרֵךְ עָלֶיהָ. וּמִי שֶׁלֹּא הִסְפִּיק לִבְדֹּק בְּלֵיל י״ד, וּבוֹדֵק בְּיוֹם י״ד אוֹ בְּתוֹךְ הַפֶּסַח – מְבָרֵךְ.
🤔 לָמָּה? מְבָרְכִים רַק עַל בְּדִיקָה שֶׁסְּמוּכָה לַבִּעוּר – כְּשֶׁכָּל חָמֵץ שֶׁנִּמְצָא יְבֹעַר מִיָּד אוֹ לְמָחֳרָת.
(פֶּרֶק ד׳, הֲלָכָה ג׳)

9אֵיפֹה בּוֹדְקִים?

בּוֹדְקִים כָּל מָקוֹם שֶׁלִּפְעָמִים נִכְנָסִים אֵלָיו עִם חָמֵץ: הַמִּטְבָּח, הַסָּלוֹן, הַחֲדָרִים, הַמַּחְסָן וְהַמִּרְפֶּסֶת. אֲרוֹנוֹת בְּגָדִים – אִם יֵשׁ בַּבַּיִת יְלָדִים קְטַנִּים, בּוֹדְקִים אוֹתָם, כִּי אוּלַי שִׂחֲקוּ בָּהֶם עִם אֹכֶל אוֹ הֶחְבִּיאוּ בָּהֶם מַשֶּׁהוּ. אֲבָל אֲרוֹנוֹת גְּבוֹהִים שֶׁהַקְּטַנִּים לֹא מַגִּיעִים אֲלֵיהֶם – לֹא צָרִיךְ לִבְדֹּק.
🤔 לָמָּה? בּוֹדְקִים בְּכָל מָקוֹם שֶׁיֵּשׁ חֲשָׁשׁ שֶׁנִּמְצָא בּוֹ כַּזַּיִת חָמֵץ. מָקוֹם שֶׁחָמֵץ לֹא מַגִּיעַ אֵלָיו – אֵין סִבָּה לִבְדֹּק.
(פֶּרֶק ד׳, הֲלָכָה ד׳)

10מְכוֹנִית, תִּיקִים וְכִיסִים

בּוֹדְקִים גַּם אֶת הַמְּכוֹנִית וְאֶת הַתִּיקִים – כְּמוֹ תִּיק בֵּית הַסֵּפֶר – שֶׁלִּפְעָמִים שָׂמִים בָּהֶם אֹכֶל. אֶת הַכִּיסִים שֶׁל הַיְּלָדִים הַקְּטַנִּים בּוֹדְקִים תָּמִיד. מְבֻגָּרִים שֶׁאֵינָם רְגִילִים לָשִׂים אֹכֶל בַּכִּיס – לֹא צְרִיכִים לִבְדֹּק אֶת כִּיסֵיהֶם. וּבְגָדִים שֶׁכֻּבְּסוּ לִפְנֵי פֶּסַח, וְרוֹקְנוּ אֶת כִּיסֵיהֶם לִפְנֵי הַכְּבִיסָה – לֹא צְרִיכִים בְּדִיקָה.
(פֶּרֶק ד׳, הֲלָכָה ד׳)

11חָצֵר, מִרְפֶּסֶת וַחֲדַר מַדְרֵגוֹת

חָצֵר פְּתוּחָה – לֹא צָרִיךְ לִבְדֹּק, כִּי בַּעֲלֵי חַיִּים אוֹכְלִים אֶת מָה שֶׁנִּשְׁאַר שָׁם. אֲבָל אִם יֵשׁ בָּהּ חוֹרִים וּסְדָקִים שֶׁאוּלַי תָּחֲבוּ בָּהֶם חָמֵץ, אוֹ שֶׁיּוֹדְעִים שֶׁיֵּשׁ בָּהּ חָמֵץ – בּוֹדְקִים. מִרְפֶּסֶת מְגֻדֶּרֶת – בּוֹדְקִים. חֲדַר מַדְרֵגוֹת שֶׁל בַּיִת מְשֻׁתָּף – שַׁיָּךְ לְכָל הַשְּׁכֵנִים, וְהֵם מְמַנִּים אֶחָד מֵהֶם שֶׁיִּבְדֹּק. מְקוֹמוֹת צִבּוּרִיִּים שֶׁל הֶפְקֵר – כְּמוֹ גִּנָּה צִבּוּרִית אוֹ תַּחֲנַת אוֹטוֹבּוּס – לֹא צְרִיכִים בְּדִיקָה.
🤔 לָמָּה? אִסּוּר חָמֵץ הוּא רַק עַל מָה שֶׁשַּׁיָּךְ לִיהוּדִי. מָקוֹם שֶׁל הֶפְקֵר אֵינוֹ שֶׁל אַף אֶחָד.
(פֶּרֶק ד׳, הֲלָכָה ד׳)

12נֵר אוֹ פַּנָּס?

חֲכָמִים תִּקְּנוּ לִבְדֹּק לְאוֹר נֵר, כִּי אוֹרוֹ מְמֻקָּד. לֹא בּוֹדְקִים בַּ״אֲבוּקָה״ – נֵר שֶׁיֵּשׁ לוֹ שְׁתֵּי פְּתִילוֹת אוֹ יוֹתֵר – וְגַם לֹא בְּנֵר שֶׁמֶן. נָהֲגוּ לִבְדֹּק בְּנֵר שַׁעֲוָה, שֶׁכִּמְעַט אֵינוֹ מְטַפְטֵף. וּמָה עִם פַּנָּס?
🤔 לָמָּה? בָּאֲבוּקָה הַלֶּהָבָה גְּדוֹלָה, וְהַבּוֹדֵק יִפְחַד לִשְׂרֹף דְּבָרִים וְלֹא יִתְרַכֵּז. וְעִם נֵר שֶׁמֶן הוּא יִפְחַד לִשְׁפֹּךְ שֶׁמֶן וְלֹא יַכְנִיס אוֹתוֹ לִמְקוֹמוֹת צָרִים. לְמַעֲשֶׂה, כָּל אֶחָד רַשַּׁאי לִבְחֹר, וְאֶפְשָׁר גַּם לְהַתְחִיל בְּנֵר וּלְהַמְשִׁיךְ בְּפַנָּס. מִי שֶׁחוֹשֵׁשׁ מִשְּׂרֵפָה, אוֹ שֶׁאֵינוֹ רוֹאֶה טוֹב לְאוֹר הַנֵּר – עָדִיף שֶׁיִּבְדֹּק בְּפַנָּס.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ מְהַדְּרִים לִבְדֹּק דַּוְקָא בְּנֵר, כְּמוֹ שֶׁנָּהֲגוּ יִשְׂרָאֵל מִדּוֹרֵי דּוֹרוֹת, כִּי חֲכָמִים לָמְדוּ מֵהַפְּסוּקִים שֶׁמְּחַפְּשִׂים לְאוֹר נֵר.
וּמִן הַדִּין מֻתָּר לִבְדֹּק גַּם בְּפַנָּס, כִּי גַּם אוֹרוֹ מְמֻקָּד, וְהוּא לֹא שׂוֹרֵף וְלֹא מְלַכְלֵךְ.
(פֶּרֶק ד׳, הֲלָכָה ה׳)

13מָה מְחַפְּשִׂים – כַּזַּיִת אוֹ כָּל פֵּרוּר?

הַבְּדִיקָה נוֹעֲדָה לִמְצֹא חָמֵץ, אֲבָל מָה הַגֹּדֶל שֶׁמְּחַפְּשִׂים? יֵשׁ בָּזֶה שְׁתֵּי שִׁיטוֹת. גַּם לְדַעַת הַמַּחְמִירִים לֹא מְחַפְּשִׂים פֵּרוּרִים קְטַנְטַנִּים אוֹ מְלֻכְלָכִים, וְלֹא מְחַטְּטִים בֵּין הַמַּרְצָפוֹת.
🤔 לָמָּה? וְיֵשׁ מְקוֹמוֹת שֶׁבָּהֶם לְכָל הַדֵּעוֹת לֹא מַשְׁאִירִים אֲפִלּוּ פֵּרוּר: שֻׁלְחַן הָאֹכֶל, הַשַּׁיִשׁ וַאֲרוֹנוֹת הַמִּטְבָּח. הֵם נוֹגְעִים בָּאֹכֶל שֶׁל פֶּסַח, וַאֲפִלּוּ פֵּרוּר חָמֵץ קָטָן אוֹסֵר אוֹתוֹ.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ מַחְמִירִים מְחַפְּשִׂים כָּל פֵּרוּר שֶׁרָאוּי לַאֲכִילָה, כְּדֵי שֶׁלֹּא נָבוֹא לֶאֱכֹל אוֹתוֹ בְּפֶסַח.
לְדַעַת הַמְּקִלִּים, וְכָךְ הֲלָכָה בַּבַּיִת מְחַפְּשִׂים חֲתִיכוֹת חָמֵץ בְּגֹדֶל כַּזַּיִת, כִּי חוֹבַת הַבְּדִיקָה הִיא מִדִּבְרֵי חֲכָמִים.
(פֶּרֶק ד׳, הֲלָכָה ו׳)

14וּמָה עִם הַסְּפָרִים?

הַאִם צָרִיךְ לְדַפְדֵּף בְּכָל הַסְּפָרִים שֶׁלָּמַדְנוּ בָּהֶם בְּמֶשֶׁךְ הַשָּׁנָה, אוּלַי נָפַל לְתוֹכָם פֵּרוּר?
🤔 לָמָּה? אֲבָל לְכָל הַדֵּעוֹת לֹא מַנִּיחִים בְּפֶסַח עַל שֻׁלְחַן הָאֹכֶל סְפָרִים שֶׁלֹּא נִשְׁמְרוּ מֵחָמֵץ בְּמֶשֶׁךְ הַשָּׁנָה – שֶׁמָּא יִפֹּל מֵהֶם פֵּרוּר לָאֹכֶל. לוֹמְדִים בָּהֶם עַל שֻׁלְחָן אַחֵר. וְאִם הִנִּיחוּ אוֹתָם עַל שֻׁלְחַן הָאֹכֶל שֶׁלֹּא בִּזְמַן הָאֲרוּחָה – מְנַקִּים אַחַר כָּךְ אֶת הַשֻּׁלְחָן.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת כַּמָּה אַחֲרוֹנִים (הַחֲזוֹן אִישׁ) צָרִיךְ לִבְדֹּק אֶת הַסְּפָרִים דַּף אַחַר דַּף.
וְהַמִּנְהָג לְהָקֵל לֹא בּוֹדְקִים דַּף אַחַר דַּף. הָרַב מֶלָמֵד כּוֹתֵב שֶׁזּוֹ חֻמְרָה יְתֵרָה, שֶׁעֲלוּלָה לִגְרֹם לְבִטּוּל תּוֹרָה.
💛 הַהֲלָכָה דּוֹאֶגֶת שֶׁהַזְּהִירוּת מֵחָמֵץ לֹא תַּפְרִיעַ לָנוּ לִלְמֹד תּוֹרָה – גַּם בְּעֶרֶב פֶּסַח הָעָמוּס.
(פֶּרֶק ד׳, הֲלָכָה ז׳)

15אֲרוֹן הַסְּפָרִים

אִם אֵין בַּבַּיִת יְלָדִים קְטַנִּים, וְהַגְּדוֹלִים מַקְפִּידִים לֹא לָשִׂים אֹכֶל בָּאָרוֹן – לֹא צָרִיךְ לִבְדֹּק אֶת אֲרוֹן הַסְּפָרִים. אִם יֵשׁ יְלָדִים שֶׁאוּלַי הֶחְבִּיאוּ בּוֹ אֹכֶל – בּוֹדְקִים בֵּין הַסְּפָרִים וּמֵאֲחוֹרֵיהֶם. וְאִם נִקּוּ אוֹתוֹ הֵיטֵב לִפְנֵי פֶּסַח – מַסְפִּיקָה בְּדִיקָה קַלָּה.
(פֶּרֶק ד׳, הֲלָכָה ז׳)

16נִקִּינוּ – וּבְכָל זֹאת בּוֹדְקִים

בְּרֹב הַבָּתִּים מְנַקִּים הֵיטֵב לִקְרַאת פֶּסַח. מָקוֹם שֶׁנֻּקָּה הֵיטֵב, וְנִזְהֲרוּ לֹא לְהַכְנִיס אֵלָיו חָמֵץ אַחַר כָּךְ – מָה עוֹשִׂים בּוֹ בְּלֵיל הַבְּדִיקָה?
🤔 לָמָּה? וּבְכָל זֹאת, חוֹבָה לִבְדֹּק אֶת כָּל הַבַּיִת בִּבְרָכָה: אֶת הַמְּקוֹמוֹת שֶׁאָכְלוּ בָּהֶם בְּוַדַּאי צָרִיךְ לִבְדֹּק, וְאוּלַי שָׁכְחוּ לְנַקּוֹת אָרוֹן, מְגֵרָה אוֹ פִּנָּה. גַּם בְּדִיקָה קַלָּה הִיא בְּדִיקָה אֲמִתִּית: מִסְתַּכְּלִים בְּכָל הַפִּנּוֹת, לְאֹרֶךְ הַקִּירוֹת וּבֵין הָרָהִיטִים, וּפוֹתְחִים כָּל אָרוֹן וּמְגֵרָה. מִי שֶׁלֹּא נִקָּה בְּעַצְמוֹ – שׁוֹאֵל אֶת מִי שֶׁנִּקָּה, אוֹ שֶׁמְּסַמְּנִים בְּמַדְבֵּקָה אֶת הַמְּקוֹמוֹת שֶׁנֻּקּוּ.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ מַחְמִירִים בּוֹדְקִים גַּם אוֹתוֹ בְּכָל הַחוֹרִים וְהַסְּדָקִים, כִּי חֲכָמִים תִּקְּנוּ לִבְדֹּק אֶת כָּל הַבַּיִת לְאוֹר הַנֵּר.
וְהַמִּנְהָג לְהָקֵל עוֹבְרִים עָלָיו בִּבְדִיקָה קַלָּה, כִּי נִקּוּי יְסוֹדִי מוֹעִיל יוֹתֵר מִבְּדִיקָה.
(פֶּרֶק ד׳, הֲלָכָה ח׳)

17עֲשָׂרָה פְּתִיתִים

נוֹהֲגִים לְהַחְבִּיא בַּבַּיִת פְּתִיתֵי חָמֵץ לִפְנֵי הַבְּדִיקָה, כְּדֵי שֶׁהַבּוֹדֵק יִמְצָא אוֹתָם. הָאֲרִ״י נָהַג לְהַנִּיחַ עֲשָׂרָה. בְּבַיִת נָקִי מַחְבִּיאִים אוֹתָם בִּמְקוֹמוֹת שֶׁקַּל לִמְצֹא, וּבִמְקוֹמוֹת שֶׁלֹּא נֻקּוּ הֵיטֵב – מַחְבִּיאִים יוֹתֵר. וְטוֹב שֶׁמִּי שֶׁמַּחְבִּיא יִרְשֹׁם לְעַצְמוֹ אֵיפֹה, כְּדֵי שֶׁאִם לֹא יִמְצְאוּ אוֹתָם – יוֹצִיאוּ אוֹתָם וִיבַעֲרוּ.
🤔 לָמָּה? יֵשׁ שֶׁחָשְׁבוּ שֶׁבְּלִי פְּתִיתִים הַבְּרָכָה תִּהְיֶה לְבַטָּלָה. אֲבָל זֶה לֹא נָכוֹן: גַּם מִי שֶׁלֹּא מָצָא כְּלוּם קִיֵּם אֶת הַמִּצְוָה, כִּי הַבְּדִיקָה וִדְּאָה שֶׁאֵין חָמֵץ, וְהַבְּרָכָה ״עַל בִּעוּר חָמֵץ״ הִיא עַל כָּל הַתַּהֲלִיךְ. וּבְכָל זֹאת לֹא מְבַטְּלִים אֶת מִנְהַג יִשְׂרָאֵל.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ שֶׁכָּתְבוּ שֶׁהַפְּתִיתִים נוֹעֲדוּ כְּדֵי שֶׁיִּשָּׁאֵר חָמֵץ לְבַעֵר, וְכָךְ הַבּוֹדֵק לֹא יִשְׁכַּח לְבַטֵּל.
וְיֵשׁ שֶׁבֵּאֲרוּ שֶׁבִּזְכוּת הַפְּתִיתִים הַבּוֹדֵק לֹא יִתְרַשֵּׁל בַּבְּדִיקָה.
(פֶּרֶק ד׳, הֲלָכָה ט׳)

18כָּל הַמִּשְׁפָּחָה בּוֹדֶקֶת

מִי שֶׁקָּשֶׁה לוֹ לִבְדֹּק לְבַד אֶת כָּל הַבַּיִת – מְבַקֵּשׁ מִבְּנֵי הַבַּיִת וּמִבְּנוֹת הַבַּיִת לַעֲזֹר. כֻּלָּם עוֹמְדִים לְיָדוֹ בִּזְמַן הַבְּרָכָה וְעוֹנִים ״אָמֵן״, וְאַחַר כָּךְ מִתְפַּזְּרִים לִבְדֹּק בְּכָל הַבַּיִת. וְאִם בַּעַל הַבַּיִת אֵינוֹ יָכוֹל לִבְדֹּק כְּלָל – מִישֶׁהוּ אַחֵר בּוֹדֵק בִּמְקוֹמוֹ, וְהוּא מְבָרֵךְ.
💛 כְּשֶׁכָּל הַמִּשְׁפָּחָה מְחַפֶּשֶׂת יַחַד, הַמִּצְוָה הוֹפֶכֶת לַחֲוָיָה מְשֻׁתֶּפֶת – וְכֻלָּנוּ שׁוֹתְפִים בַּהֲכָנַת הַבַּיִת לַה׳.
(פֶּרֶק ד׳, הֲלָכָה ט׳)

19אַחֲרֵי הַבְּדִיקָה

אַחֲרֵי הַבְּדִיקָה מְבַטְּלִים אֶת הֶחָמֵץ שֶׁלֹּא מָצָאנוּ (עַל כָּךְ נִלְמַד בַּפֶּרֶק הַבָּא). אֶת הֶחָמֵץ שֶׁמָּצְאוּ בַּבְּדִיקָה, וְאֶת הֶחָמֵץ שֶׁשּׁוֹמְרִים לַאֲרוּחַת הַבֹּקֶר – שׁוֹמְרִים הֵיטֵב בְּמָקוֹם אֶחָד, כְּדֵי שֶׁלֹּא יִתְפַּזֵּר שׁוּב בַּבַּיִת.
(פֶּרֶק ד׳, הֲלָכָה ט׳)

20נוֹסְעִים לִפְנֵי פֶּסַח

מִשְׁפָּחָה שֶׁנּוֹסַעַת מֵהַבַּיִת לְכָל הַפֶּסַח, וְיוֹצֵאת בִּשְׁלוֹשִׁים הַיָּמִים שֶׁלִּפְנֵי הֶחָג (מִפּוּרִים וָהָלְאָה) – בּוֹדֶקֶת אֶת הַבַּיִת לְאוֹר הַנֵּר בַּלַּיְלָה הָאַחֲרוֹן לִפְנֵי הַיְּצִיאָה, בְּלִי בְּרָכָה. וּמִי שֶׁיּוֹצֵא לִזְמַן קָצָר וְחוֹזֵר לִפְנֵי פֶּסַח – יָכוֹל לְבַקֵּשׁ מֵחָבֵר אוֹ מִקָּרוֹב שֶׁיִּבְדֹּק בִּמְקוֹמוֹ אִם יִתְעַכֵּב.
🤔 לָמָּה? מִפּוּרִים כְּבָר מִתְחִילָה הַהֲכָנָה לְפֶסַח, וְחוֹבַת הַבְּדִיקָה חָלָה. לֹא מְבָרְכִים, כִּי זוֹ בְּדִיקָה לִפְנֵי הַזְּמַן שֶׁתִּקְּנוּ חֲכָמִים.
(פֶּרֶק ד׳, הֲלָכָה י׳)

21שָׁכַחְנוּ לִבְדֹּק!

מִי שֶׁלֹּא בָּדַק בְּלֵיל י״ד – בּוֹדֵק בְּיוֹם י״ד בִּבְרָכָה. וְאִם לֹא בָּדַק לִפְנֵי פֶּסַח – בּוֹדֵק בְּתוֹךְ הַפֶּסַח בִּבְרָכָה, גַּם אִם כְּבָר בִּטֵּל אֶת חֲמֵצוֹ. וְאִם נִזְכַּר רַק אַחֲרֵי פֶּסַח – בּוֹדֵק בְּלִי בְּרָכָה, כְּדֵי שֶׁלֹּא יֹאכַל חָמֵץ שֶׁעָבַר עָלָיו הַפֶּסַח.
🤔 לָמָּה? בְּתוֹךְ הַפֶּסַח בּוֹדְקִים כְּדֵי לְקַיֵּם אֶת תַּקָּנַת חֲכָמִים, וּכְדֵי שֶׁלֹּא נִרְאֶה פִּתְאוֹם עוּגָה, נִשְׁכַּח וְנֹאכַל.
(פֶּרֶק ד׳, הֲלָכָה י״ב)

22בְּמָלוֹן וּבְבֵית חוֹלִים

מִשְׁפָּחָה שֶׁשּׂוֹכֶרֶת חֶדֶר בְּמָלוֹן לְפֶסַח – הַחֶדֶר שֶׁלָּהּ, וְלָכֵן בּוֹדְקִים אוֹתוֹ בְּלֵיל י״ד בִּבְרָכָה, וּמְבַטְּלִים אַחַר כָּךְ. מִי שֶׁהִגִּיעַ לַמָּלוֹן בְּאֶמְצַע הַפֶּסַח – שׁוֹאֵל אִם בָּדְקוּ אֶת הַחֶדֶר, וְאִם לֹא – בּוֹדֵק בְּעַצְמוֹ בִּבְרָכָה. חוֹלֶה שֶׁנִּמְצָא בְּבֵית חוֹלִים – בּוֹדֵק אֶת חַדְרוֹ וְאֶת הָאָרוֹן שֶׁלּוֹ, אֲבָל בְּלִי בְּרָכָה.
🤔 לָמָּה? חֶדֶר בְּמָלוֹן עוֹמֵד לִרְשׁוּת הַשּׂוֹכֵר, וְיֵשׁ לוֹ מַפְתֵּחַ. בְּבֵית חוֹלִים הַחֶדֶר אֵינוֹ שֶׁל הַחוֹלֶה – בְּכָל רֶגַע אֶפְשָׁר לְהַעֲבִיר אוֹתוֹ אוֹ לְהַכְנִיס חוֹלִים נוֹסָפִים.
(פֶּרֶק ד׳, הֲלָכָה י״ב)

23בֵּית כְּנֶסֶת וּפְנִימִיָּה

גַּם בֵּית הַכְּנֶסֶת צָרִיךְ בְּדִיקָה, כִּי לִפְעָמִים אוֹכְלִים בּוֹ, וְלִפְעָמִים נִכְנָסִים אֵלָיו יְלָדִים עִם חָמֵץ. מִי שֶׁבּוֹדֵק אוֹתוֹ – טוֹב שֶׁיִּבְדֹּק קֹדֶם אֶת בֵּיתוֹ, וִיכַוֵּן בַּבְּרָכָה גַּם עַל בֵּית הַכְּנֶסֶת. בַּחוּרִים וּבַחוּרוֹת שֶׁגָּרִים בִּפְנִימִיָּה – בּוֹדְקִים אֶת הַחֶדֶר שֶׁלָּהֶם. אִם הֵם נִשְׁאָרִים שָׁם בְּפֶסַח – בּוֹדְקִים בְּלֵיל י״ד בִּבְרָכָה; וְאִם יוֹצְאִים כַּמָּה יָמִים לִפְנֵי – בּוֹדְקִים בַּלַּיְלָה הָאַחֲרוֹן, בְּלִי בְּרָכָה.
🤔 לָמָּה? בְּבֵית הַכְּנֶסֶת יֵשׁ סָפֵק אִם מְבָרְכִים, וְלָכֵן מְכַוְּנִים לְבֵית הַכְּנֶסֶת בַּבְּרָכָה שֶׁל הַבַּיִת.
(פֶּרֶק ד׳, הֲלָכָה י״ג)

24פְּרוּסָה שֶׁנִּתְקְעָה מֵאֲחוֹרֵי הָאָרוֹן

חָמֵץ שֶׁנִּתְקַע מֵאֲחוֹרֵי אֲרוֹן קִיר, וְאִי אֶפְשָׁר לְהוֹצִיא אוֹתוֹ בְּלִי לְפָרֵק אֶת הָאָרוֹן אוֹ לְהָזִיז אוֹתוֹ – לֹא צָרִיךְ לְהוֹצִיא, וּמַסְפִּיק לְבַטֵּל אוֹתוֹ. וְאִם אֶפְשָׁר לְהוֹצִיא אוֹתוֹ אֲבָל זֶה מְאוֹד מְסֻבָּךְ – אֶפְשָׁר לִשְׁפֹּךְ עָלָיו אֶקוֹנוֹמִיקָה אוֹ מֵי סַבּוֹן, עַד שֶׁאֲפִלּוּ כֶּלֶב לֹא יֹאכַל אוֹתוֹ. וְחָמֵץ שֶׁקָּבוּר תַּחַת יוֹתֵר מִשְּׁלֹשָׁה טְפָחִים (כְּ-24 סֶנְטִימֶטֶר) שֶׁל אֲבָנִים אוֹ עָפָר – נֶחְשָׁב מְבֹעָר, וּמְבַטְּלִים אוֹתוֹ.
🤔 לָמָּה? חָמֵץ שֶׁקָּבוּר עָמֹק – כֶּלֶב לֹא יָכוֹל לְהָרִיחַ אוֹתוֹ וּלְהוֹצִיא אוֹתוֹ, וְלָכֵן הוּא כְּאִלּוּ אֵינֶנּוּ. אֲבָל מְבַטְּלִים, שֶׁמָּא בְּמֶשֶׁךְ הַפֶּסַח יְפַנּוּ קְצָת מֵהָאֲבָנִים.
(פֶּרֶק ד׳, הֲלָכָה י״ד)

25מַחְסָן מָלֵא

מַחְסָן שֶׁמְּמֻלָּא בַּחֲפָצִים, וְלֹא מִתְכַּוְּנִים לְפַנּוֹת אוֹתוֹ עַד אַחֲרֵי פֶּסַח – לֹא צָרִיךְ לְהוֹצִיא אֶת הַכֹּל וְלִבְדֹּק מִתַּחַת. מַסְפִּיק לְבַטֵּל אֶת הֶחָמֵץ שֶׁאוּלַי יֵשׁ שָׁם. אֲבָל אִם מִלְּאוּ אוֹתוֹ בִּשְׁלוֹשִׁים הַיָּמִים שֶׁלִּפְנֵי פֶּסַח – הָיוּ צְרִיכִים לִבְדֹּק אוֹתוֹ לִפְנֵי שֶׁהִכְנִיסוּ אֶת הַחֲפָצִים.
(פֶּרֶק ד׳, הֲלָכָה י״ד)

📝 מָה לָמַדְנוּ

  • בּוֹדְקִים חָמֵץ בִּתְחִלַּת לֵיל י״ד בְּנִיסָן, אַחֲרֵי עַרְבִית, וּמְבָרְכִים ״עַל בִּעוּר חָמֵץ״ – בְּלִי לְדַבֵּר בֵּין הַבְּרָכָה לַבְּדִיקָה.
  • בּוֹדְקִים כָּל מָקוֹם שֶׁנִּכְנָסִים אֵלָיו עִם חָמֵץ – גַּם מְכוֹנִית, תִּיקִים וְכִיסֵי הַקְּטַנִּים.
  • בּוֹדְקִים לְאוֹר נֵר שַׁעֲוָה אוֹ פַּנָּס – יֵשׁ מְהַדְּרִים בְּנֵר, וּמִן הַדִּין מֻתָּר גַּם פַּנָּס.
  • גַּם בַּיִת נָקִי בּוֹדְקִים בִּבְרָכָה; שֻׁלְחַן הָאֹכֶל וְהַשַּׁיִשׁ – בְּלִי שׁוּם פֵּרוּר.
  • נוֹהֲגִים לְהַחְבִּיא עֲשָׂרָה פְּתִיתִים, וְכָל הַמִּשְׁפָּחָה יְכוֹלָה לַעֲזֹר בַּבְּדִיקָה.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה בּוֹדְקִים חָמֵץ דַּוְקָא בַּלַּיְלָה, לְאוֹר נֵר?
  2. אֵילוּ מְקוֹמוֹת צָרִיךְ לִבְדֹּק בְּבַיִת שֶׁיֵּשׁ בּוֹ יְלָדִים קְטַנִּים, וְאֵילוּ לֹא?
  3. הַבַּיִת נֻקָּה הֵיטֵב לְפֶסַח. הַאִם עֲדַיִן צָרִיךְ לִבְדֹּק? אֵיךְ?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״מְחַפְּשִׂים וּמְבַעֲרִים״

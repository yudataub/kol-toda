סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת פֶּסַח
מַהֲדוּרַת יְלָדִים

🕯️ עֶרֶב פֶּסַח שֶׁחָל בְּשַׁבָּת

פֶּרֶק י״ד · לַמַּהֲדוּרָה לַמּוֹרֶה
דִּמְיְנוּ שֶׁעֶרֶב פֶּסַח יוֹצֵא בְּשַׁבָּת. אֵיךְ שׂוֹרְפִים חָמֵץ, אִם בְּשַׁבָּת אָסוּר לְהַדְלִיק אֵשׁ? מָה אוֹכְלִים בִּסְעוּדוֹת שַׁבָּת, אִם חָמֵץ כְּבָר אָסוּר וּמַצָּה עוֹד אָסוּר? וּמָתַי עוֹרְכִים אֶת שֻׁלְחַן הַסֵּדֶר? זֶה קוֹרֶה פַּעַם בְּכַמָּה שָׁנִים – וּכְדַאי לָדַעַת מֵרֹאשׁ מָה עוֹשִׂים.

1בּוֹדְקִים בְּיוֹם חֲמִישִׁי, שׂוֹרְפִים בְּיוֹם שִׁשִּׁי

כְּשֶׁעֶרֶב פֶּסַח (י״ד בְּנִיסָן) חָל בְּשַׁבָּת, מַקְדִּימִים: בּוֹדְקִים חָמֵץ בִּבְרָכָה בְּלֵיל י״ג בְּנִיסָן – בְּלֵיל שִׁשִּׁי – וְאַחֲרֵי הַבְּדִיקָה מְבַטְּלִים אֶת הֶחָמֵץ, כְּמוֹ בְּכָל שָׁנָה. לְמָחֳרָת, בְּיוֹם שִׁשִּׁי, שׂוֹרְפִים אֶת הֶחָמֵץ שֶׁנִּמְצָא, וְטוֹב לִשְׂרֹף אוֹתוֹ עַד הַשָּׁעָה הַשִּׁשִּׁית.
🤔 לָמָּה? בְּשַׁבָּת אִי אֶפְשָׁר לִבְדֹּק חָמֵץ וְלִשְׂרֹף אוֹתוֹ. וְשׂוֹרְפִים בְּיוֹם שִׁשִּׁי בְּאוֹתָהּ שָׁעָה שֶׁל כָּל שָׁנָה – כְּדֵי שֶׁבַּשָּׁנִים הָאֲחֵרוֹת לֹא יִטְעוּ וְיִשְׂרְפוּ מְאֻחָר מִדַּי.
(פֶּרֶק י״ד, הֲלָכָה א׳)

2לֶחֶם לִשְׁתֵּי הַסְּעוּדוֹת

מַשְׁאִירִים לֶחֶם חָמֵץ לִשְׁתֵּי הַסְּעוּדוֹת הָרִאשׁוֹנוֹת שֶׁל שַׁבָּת: בְּלֵיל שַׁבָּת וּבְשַׁבָּת בַּבֹּקֶר. רַבִּים מְבַשְּׁלִים לַשַּׁבָּת תַּבְשִׁילִים כְּשֵׁרִים לְפֶסַח, אוֹכְלִים אֶת הַלֶּחֶם בְּחֶדֶר שֶׁלֹּא אוֹכְלִים בּוֹ בְּפֶסַח, מְנַקִּים אֶת הַיָּדַיִם וְהַבְּגָדִים – וְעוֹבְרִים לֶאֱכֹל אֶת הַתַּבְשִׁילִים בַּמָּקוֹם שֶׁל פֶּסַח. אֶת סְעוּדַת הַבֹּקֶר צָרִיךְ לְסַיֵּם עַד סוֹף הַשָּׁעָה הָרְבִיעִית, וְלָכֵן מַקְדִּימִים בְּאוֹתָהּ שַׁבָּת אֶת תְּפִלַּת שַׁחֲרִית.
🤔 לָמָּה? לָמָּה לֹא לֶאֱכֹל פָּשׁוּט מַצָּה בְּשַׁבָּת? כִּי מַצָּה אֲסוּרָה בְּעֶרֶב פֶּסַח – כְּדֵי שֶׁתִּהְיֶה חֲבִיבָה עָלֵינוּ בַּלַּיְלָה. וְסוֹף הַשָּׁעָה הָרְבִיעִית הוּא סוֹף זְמַן אֲכִילַת חָמֵץ.
(פֶּרֶק י״ד, הֲלָכָה א׳)

3מְבַעֲרִים בְּלִי אֵשׁ

אַחֲרֵי הַסְּעוּדָה הַשְּׁנִיָּה אוֹסְפִים אֶת כָּל שְׁאֵרִיּוֹת הֶחָמֵץ מֵהַקְּעָרוֹת וּמֵהַמַּפָּה. אָסוּר לִשְׂרֹף בְּשַׁבָּת, וְלָכֵן טוֹב לְהַשְׁלִיךְ אוֹתָן לְבֵית הַכִּסֵּא – וְכָךְ הֵן מִתְבַּעֲרוֹת מִן הָעוֹלָם. אַחַר כָּךְ מְבַטְּלִים אֶת הֶחָמֵץ, עַד סוֹף הַשָּׁעָה הַחֲמִישִׁית. וְאַחֲרֵי הַסְּעוּדָה מְצַחְצְחִים הֵיטֵב אֶת הַשִּׁנַּיִם. כְּדַאי לְתַכְנֵן מֵרֹאשׁ אֶת מַאַכְלֵי הַשַּׁבָּת, כָּךְ שֶׁיִּהְיֶה קַל לֶאֱסֹף אֶת הַשְּׁאֵרִיּוֹת.
💛 גַּם כְּשֶׁהַשַּׁבָּת וְהַפֶּסַח נִפְגָּשִׁים, אֶפְשָׁר לִשְׁמֹר עַל שְׁנֵיהֶם – בְּתִכְנוּן טוֹב וּבְאַהֲבָה.
(פֶּרֶק י״ד, הֲלָכָה א׳)

4סְעוּדָה שְׁלִישִׁית

בִּסְעוּדָה שְׁלִישִׁית חָמֵץ כְּבָר אָסוּר, וּמַצָּה אֲסוּרָה כָּל יוֹם י״ד. לָכֵן אוֹכְלִים בָּהּ בָּשָׂר, דָּגִים אוֹ פֵּרוֹת. מֻתָּר גַּם לֶאֱכֹל קְצִיצוֹת מְבֻשָּׁלוֹת מִקֶּמַח מַצָּה, וּלְדַעַת רַבִּים מֻתָּר גַּם לְבַשֵּׁל מַצּוֹת שְׁלֵמוֹת וּלְאָכְלָן. וְלֹא מַרְבִּים בָּהּ בַּאֲכִילָה.
🤔 לָמָּה? כְּדֵי שֶׁבָּעֶרֶב יִהְיֶה כֹּחַ לֶאֱכֹל אֶת הַמַּצָּה לְתֵאָבוֹן.
(פֶּרֶק י״ד, הֲלָכָה א׳)

5תַּעֲנִית בְּכוֹרוֹת – בְּיוֹם חֲמִישִׁי

כְּשֶׁעֶרֶב פֶּסַח חָל בְּשַׁבָּת, מַקְדִּימִים אֶת תַּעֲנִית הַבְּכוֹרוֹת לְיוֹם חֲמִישִׁי, י״ב בְּנִיסָן. וְכַיּוֹם, כְּמוֹ בְּכָל שָׁנָה, נוֹהֲגִים לְהִשְׁתַּתֵּף בְּסִיּוּם מַסֶּכֶת בִּמְקוֹם לָצוּם.
🤔 לָמָּה? בְּשַׁבָּת לֹא מִתְעַנִּים, כִּי שַׁבָּת הִיא יוֹם שֶׁל עֹנֶג.
(פֶּרֶק י״ד, הֲלָכָה ב׳)

6מְכִינִים לַסֵּדֶר – אֲבָל לֹא בְּשַׁבָּת

כְּדֵי שֶׁהַסֵּדֶר לֹא יִתְעַכֵּב, מְכִינִים כְּבָר בְּיוֹם שִׁשִּׁי אֶת כָּל מָה שֶׁצָּרִיךְ: מְבַשְּׁלִים אֶת הַתַּבְשִׁילִים, מְכִינִים אֶת הַחֲרֹסֶת, הַמָּרוֹר וְהַזְּרוֹעַ, וּמַכְנִיסִים לַמְּקָרֵר אוֹ לַמַּקְפִּיא. אֶת שֻׁלְחַן הַסֵּדֶר אָסוּר לַעֲרֹךְ בְּשַׁבָּת – עוֹרְכִים אוֹתוֹ מִיָּד אַחֲרֵי צֵאת הַשַּׁבָּת. וּמְכִינִים לִפְנֵי שַׁבָּת פָּמוֹטִים כְּפוּלִים: לְשַׁבָּת וּלְיוֹם טוֹב, כִּי אָסוּר לְהַדְבִּיק נֵרוֹת בְּיוֹם טוֹב.
🤔 לָמָּה? אָסוּר לְהָכִין מִשַּׁבָּת לְיוֹם טוֹב. הַשַּׁבָּת קְדוֹשָׁה בִּפְנֵי עַצְמָהּ, וְלֹא עוֹבְדִים בָּהּ בִּשְׁבִיל יוֹם אַחֵר.
💛 הַשַּׁבָּת אֵינָהּ ״יוֹם הֲכָנָה״ לַחַג – הִיא מַתָּנָה שְׁלֵמָה בִּפְנֵי עַצְמָהּ.
(פֶּרֶק י״ד, הֲלָכָה ב׳)

📝 מָה לָמַדְנוּ

  • כְּשֶׁעֶרֶב פֶּסַח חָל בְּשַׁבָּת – בּוֹדְקִים חָמֵץ בְּלֵיל שִׁשִּׁי וְשׂוֹרְפִים בְּיוֹם שִׁשִּׁי.
  • מַשְׁאִירִים לֶחֶם לִשְׁתֵּי הַסְּעוּדוֹת הָרִאשׁוֹנוֹת, וְאֶת הַשְּׁאֵרִיּוֹת זוֹרְקִים לְבֵית הַכִּסֵּא וּמְבַטְּלִים.
  • סְעוּדָה שְׁלִישִׁית – בְּבָשָׂר, דָּגִים אוֹ פֵּרוֹת, וְלֹא מַרְבִּים.
  • תַּעֲנִית בְּכוֹרוֹת – בְּיוֹם חֲמִישִׁי; וְאֶת שֻׁלְחַן הַסֵּדֶר עוֹרְכִים רַק אַחֲרֵי צֵאת הַשַּׁבָּת.

❓ שְׁאֵלוֹת חֲזָרָה

  1. מָתַי בּוֹדְקִים וְשׂוֹרְפִים חָמֵץ כְּשֶׁעֶרֶב פֶּסַח חָל בְּשַׁבָּת, וְלָמָּה?
  2. מָה עוֹשִׂים בְּשַׁבָּת עִם שְׁאֵרִיּוֹת הֶחָמֵץ, אִם אָסוּר לִשְׂרֹף?
  3. לָמָּה אָסוּר לַעֲרֹךְ אֶת שֻׁלְחַן הַסֵּדֶר בְּשַׁבָּת אַחַר הַצָּהֳרַיִם?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״מַצָּה שְׁמוּרָה וְעֶרֶב פֶּסַח״

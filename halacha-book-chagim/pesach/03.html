סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת פֶּסַח
מַהֲדוּרַת יְלָדִים

🧹 מִצְוַת הַשְׁבָּתַת הֶחָמֵץ

פֶּרֶק ג׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
הַשָּׁכֵן שֶׁלָּכֶם, שֶׁאֵינוֹ יְהוּדִי, בָּא לְבַקֵּר בְּפֶסַח עִם כָּרִיךְ בַּיָּד. מֻתָּר לוֹ לְהִכָּנֵס? וְאִם חָבֵר הִשְׁאִיר אֶצְלְכֶם שַׂקִּית עִם עוּגִיּוֹת – שֶׁל מִי הֶחָמֵץ, וּמִי צָרִיךְ לְבַעֵר אוֹתוֹ? בַּפֶּרֶק הַזֶּה נִלְמַד אֶת הַמִּצְוָה הָרִאשׁוֹנָה שֶׁל פֶּסַח: לְהַשְׁבִּית אֶת הֶחָמֵץ – בַּמַּחְשָׁבָה וּבַמַּעֲשֶׂה.

1״תַּשְׁבִּיתוּ שְׂאֹר מִבָּתֵּיכֶם״

מִצְוַת עֲשֵׂה מִן הַתּוֹרָה לְבַעֵר אֶת הֶחָמֵץ מִן הַבַּיִת לִקְרַאת הַפֶּסַח. חֲכָמִים לָמְדוּ שֶׁצָּרִיךְ לְסַיֵּם לְבַעֵר עַד חֲצוֹת הַיּוֹם שֶׁל עֶרֶב פֶּסַח, י״ד בְּנִיסָן – הַזְּמַן שֶׁבּוֹ הָיוּ שׁוֹחֲטִים אֶת קָרְבַּן הַפֶּסַח. בְּמִצְוָה הַזֹּאת, כְּמוֹ בִּשְׁאָר מִצְווֹת הַפֶּסַח, אֲנָשִׁים וְנָשִׁים שָׁוִים.
🤔 לָמָּה? הַתּוֹרָה אוֹמֶרֶת: ״לֹא תִשְׁחַט עַל חָמֵץ דַּם זִבְחִי״ – אָסוּר לִשְׁחֹט אֶת קָרְבַּן הַפֶּסַח כְּשֶׁיֵּשׁ עֲדַיִן חָמֵץ בַּבַּיִת.
(פֶּרֶק ג׳, הֲלָכָה א׳)

2כָּל רֶגַע חָשׁוּב

מִי שֶׁלֹּא בִּעֵר אֶת הֶחָמֵץ עַד חֲצוֹת הַיּוֹם – בְּכָל רֶגַע שֶׁהֶחָמֵץ נִשְׁאָר אֶצְלוֹ הוּא מְבַטֵּל מִצְוַת עֲשֵׂה. וּכְשֶׁנִּכְנָס הֶחָג, הוּא עוֹבֵר גַּם עַל ״לֹא יֵרָאֶה״ וְעַל ״לֹא יִמָּצֵא״. נִמְצָא שֶׁמִּי שֶׁמְּבַעֵר אֶת הֶחָמֵץ בִּזְמַן – נִצָּל מִשְּׁנֵי אִסּוּרִים.
(פֶּרֶק ג׳, הֲלָכָה א׳)

3הַמִּצְוָה הָרִאשׁוֹנָה שֶׁל פֶּסַח

הַשְׁבָּתַת הֶחָמֵץ הִיא הַמִּצְוָה שֶׁפּוֹתַחַת אֶת כָּל מִצְווֹת הַפֶּסַח. קֹדֶם מְפַנִּים אֶת הֶחָמֵץ, וְרַק אַחַר כָּךְ אוֹכְלִים מַצָּה וְעוֹרְכִים אֶת הַסֵּדֶר.
🤔 לָמָּה? הֶחָמֵץ בְּפֶסַח רוֹמֵז לַיֵּצֶר הָרָע. כְּדֵי לְקַבֵּל אֶת הַקְּדֻשָּׁה שֶׁל הַמַּצָּה וְשֶׁל קָרְבַּן הַפֶּסַח, צָרִיךְ קֹדֶם לְפַנּוֹת מָקוֹם – בַּבַּיִת וּבַלֵּב.
💛 כְּשֶׁאֲנִי מְנַקֶּה וּמְבַעֵר חָמֵץ, אֲנִי מֵכִין לַה׳ ״מָקוֹם נָקִי״ – בַּבַּיִת שֶׁלִּי וּבַלֵּב שֶׁלִּי.
(פֶּרֶק ג׳, הֲלָכָה א׳)

4רַק הֶחָמֵץ שֶׁלְּךָ

הַתּוֹרָה אוֹמֶרֶת: ״וְלֹא יֵרָאֶה לְךָ חָמֵץ״ – לְךָ, כְּלוֹמַר חָמֵץ שֶׁשַּׁיָּךְ לִיהוּדִי. חָמֵץ שֶׁל גּוֹי אוֹ חָמֵץ שֶׁל הֶפְקֵר – אֵין אִסּוּר שֶׁיִּהְיֶה בִּרְשׁוּתֵנוּ. לָכֵן גּוֹי שֶׁגָּר בַּחֲצַר שֶׁל יְהוּדִי, אֲפִלּוּ אִם הוּא עוֹבֵד אֶצְלוֹ – הַיְּהוּדִי אֵינוֹ צָרִיךְ לְבַעֵר אֶת הֶחָמֵץ שֶׁל הַגּוֹי.
(פֶּרֶק ג׳, הֲלָכָה ב׳)

5חָמֵץ שֶׁל גּוֹי בְּתוֹךְ הַבַּיִת

גּוֹי שֶׁהִשְׁאִיר אֶצְלְךָ חָמֵץ לִפְנֵי פֶּסַח, וְלֹא קִבַּלְתָּ עָלָיו אַחְרָיוּת – אֵינְךָ צָרִיךְ לְבַעֵר אוֹתוֹ. אֲבָל צָרִיךְ לְהַעֲמִיד לְפָנָיו מְחִצָּה גְּבוֹהָה עֲשָׂרָה טְפָחִים, אוֹ לִנְעֹל אוֹתוֹ בְּמַפְתֵּחַ וּלְהַחְבִּיא אֶת הַמַּפְתֵּחַ, אוֹ לִסְגֹּר אוֹתוֹ בָּאָרוֹן וּלְהַדְבִּיק עַל הַדְּלָתוֹת נְיַר דֶּבֶק.
🤔 לָמָּה? כְּשֶׁהֶחָמֵץ פָּתוּחַ לְיָדֵנוּ, אֶפְשָׁר לִשְׁכֹּחַ וְלֶאֱכֹל מִמֶּנּוּ. הַמְּחִצָּה אוֹ נְיַר הַדֶּבֶק מַזְכִּירִים לָנוּ מִיָּד: ״זֶה חָמֵץ!״
(פֶּרֶק ג׳, הֲלָכָה ב׳)

6אוֹרֵחַ עִם כָּרִיךְ

מֻתָּר לְגוֹי לְהִכָּנֵס לְבֵית יְהוּדִי בְּפֶסַח עִם הֶחָמֵץ שֶׁלּוֹ. אֲבָל אָסוּר לֶאֱכֹל אִתּוֹ עַל שֻׁלְחָן אֶחָד – אֲפִלּוּ אִם שָׂמִים דָּבָר שֶׁמַּזְכִּיר לֹא לֶאֱכֹל מֵהֶחָמֵץ שֶׁלּוֹ. אַחֲרֵי שֶׁהוּא גּוֹמֵר לֶאֱכֹל, מְנַקִּים הֵיטֵב אֶת הַשֻּׁלְחָן מִכָּל הַפֵּרוּרִים, וְאָז מֻתָּר לֶאֱכֹל עָלָיו.
🤔 לָמָּה? יֵשׁ שְׁנֵי חֲשָׁשׁוֹת: שֶׁמָּא נִשְׁכַּח וְנֹאכַל מֵהֶחָמֵץ שֶׁלּוֹ, וְשֶׁמָּא יִפֹּל פֵּרוּר מֵהֶחָמֵץ שֶׁלּוֹ לָאֹכֶל שֶׁלָּנוּ.
💛 אֶפְשָׁר לְאָרֵחַ בְּשִׂמְחָה וּבְנִימוּס – וְגַם לִשְׁמֹר עַל הַמִּצְוָה. כָּךְ מְקַדְּשִׁים שֵׁם שָׁמַיִם.
(פֶּרֶק ג׳, הֲלָכָה ב׳)

7קִבַּלְתָּ אַחְרָיוּת? הֶחָמֵץ כְּשֶׁלְּךָ

יְהוּדִי שֶׁקִּבֵּל עַל עַצְמוֹ לִשְׁמֹר חָמֵץ שֶׁל גּוֹי, וְהִתְחַיֵּב לְשַׁלֵּם אִם יִקָּרֶה לוֹ מַשֶּׁהוּ – הֶחָמֵץ נֶחְשָׁב כְּשֶׁלּוֹ. אָסוּר לוֹ לְהַשְׁאִיר אוֹתוֹ אֶצְלוֹ, וְעָלָיו לְהַחֲזִיר אוֹתוֹ לַגּוֹי אוֹ לְבַעֵר אוֹתוֹ.
🤔 לָמָּה? הַתּוֹרָה אָמְרָה גַּם ״לֹא יִמָּצֵא בְּבָתֵּיכֶם״. חֲכָמִים הִסְבִּירוּ: חָמֵץ שֶׁל גּוֹי – מֻתָּר; אֲבָל חָמֵץ שֶׁהַיְּהוּדִי אַחְרַאי עָלָיו – הוּא כְּאִלּוּ שֶׁלּוֹ.
(פֶּרֶק ג׳, הֲלָכָה ג׳)

8חָמֵץ שֶׁל חָבֵר יְהוּדִי

יְהוּדִי שֶׁהִשְׁאִיר חָמֵץ אֵצֶל חֲבֵרוֹ הַיְּהוּדִי כְּדֵי שֶׁיִּשְׁמֹר עָלָיו – שְׁנֵיהֶם חַיָּבִים לְבַעֵר אוֹתוֹ: מִי שֶׁהִשְׁאִיר – כִּי הֶחָמֵץ שֶׁלּוֹ, וּמִי שֶׁשּׁוֹמֵר – כִּי הוּא אַחְרַאי עָלָיו. וַאֲפִלּוּ אִם לֹא הִתְחַיֵּב לִשְׁמֹר – חַיָּב לְבַעֵר.
(פֶּרֶק ג׳, הֲלָכָה ג׳)

9בַּמַּחְשָׁבָה וּבַמַּעֲשֶׂה

מְבַעֲרִים אֶת הֶחָמֵץ בִּשְׁתֵּי דְּרָכִים. בַּמַּחְשָׁבָה – מְבַטְּלִים אֶת הֶחָמֵץ וּמַפְקִירִים אוֹתוֹ, וְחוֹשְׁבִים עָלָיו שֶׁהוּא כְּמוֹ עָפָר. וּבַמַּעֲשֶׂה – בּוֹדְקִים אֶת הַבַּיִת בְּלֵיל י״ד, וּבַבֹּקֶר מְבַעֲרִים אֶת הֶחָמֵץ. מִן הַתּוֹרָה מַסְפִּיקָה דֶּרֶךְ אַחַת, אֲבָל חֲכָמִים תִּקְּנוּ לַעֲשׂוֹת אֶת שְׁתֵּיהֶן.
🤔 לָמָּה? אִם רַק מְבַטְּלִים – אוּלַי מִישֶׁהוּ לֹא יְבַטֵּל בְּלֵב שָׁלֵם, כִּי הוּא רוֹצֶה לֶאֱכֹל אֶת הֶחָמֵץ אַחֲרֵי פֶּסַח; וְאוּלַי יִרְאֶה עוּגָה בְּפֶסַח, יִשְׁכַּח וְיֹאכַל. וְאִם רַק בּוֹדְקִים – אוּלַי לֹא יִמְצְאוּ אֶת כָּל הֶחָמֵץ, וּכְשֶׁיִּמְצְאוּ אוֹתוֹ בְּפֶסַח, יְחַכּוּ רֶגַע לִפְנֵי שֶׁיִּשְׂרְפוּ אוֹתוֹ – וְיַעַבְרוּ עַל הָאִסּוּר. שְׁתֵּי הַדְּרָכִים יַחַד שׁוֹמְרוֹת עָלֵינוּ.
(פֶּרֶק ג׳, הֲלָכָה ד׳)

10לָמָּה מַשְׁאִירִים חָמֵץ לִשְׂרֵפָה?

יֵשׁ שְׁאֵלָה גְּדוֹלָה: מָה עִקַּר הַמִּצְוָה – לְהַשְׁבִּית חָמֵץ בְּפֹעַל, אוֹ שֶׁהַבַּיִת יִהְיֶה נָקִי מֵחָמֵץ? וּלְמַעֲשֶׂה, יִשְׂרָאֵל קְדוֹשִׁים מִשְׁתַּדְּלִים שֶׁיִּהְיֶה לָהֶם חָמֵץ בְּעֶרֶב פֶּסַח, כְּדֵי לְקַיֵּם בּוֹ אֶת הַמִּצְוָה לְפִי כָּל הַדֵּעוֹת – וּמְהַדְּרִים לְבַעֵר אוֹתוֹ דַּוְקָא בִּשְׂרֵפָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת רֹב הָרִאשׁוֹנִים עִקַּר הַמִּצְוָה שֶׁהַבַּיִת יִהְיֶה נָקִי מֵחָמֵץ. גַּם מִי שֶׁאֵין לוֹ חָמֵץ כְּלָל – מְקַיֵּם אֶת הַמִּצְוָה.
לְדַעַת כַּמָּה רִאשׁוֹנִים רַק מִי שֶׁיֵּשׁ לוֹ חָמֵץ חַיָּב בַּמִּצְוָה, וּמְקַיֵּם אוֹתָהּ כְּשֶׁהוּא מְבַעֵר אוֹתוֹ.
💛 אֲנַחְנוּ לֹא מְחַפְּשִׂים אֵיךְ לְהִפָּטֵר מִמִּצְוָה – אֲנַחְנוּ מְחַפְּשִׂים אֵיךְ לְקַיֵּם אוֹתָהּ בְּיֹפִי.
(פֶּרֶק ג׳, הֲלָכָה ה׳)

11עַד מָתַי אוֹכְלִים וְעַד מָתַי מְבַטְּלִים?

בְּאַרְבַּע הַשָּׁעוֹת הָרִאשׁוֹנוֹת שֶׁל עֶרֶב פֶּסַח מֻתָּר לֶאֱכֹל חָמֵץ. בַּשָּׁעָה הַחֲמִישִׁית – אָסוּר לֶאֱכֹל, אֲבָל אֶפְשָׁר לְהַאֲכִיל בּוֹ בְּהֵמָה, לִמְכֹּר אוֹתוֹ לְגוֹי וּלְבַטֵּל אוֹתוֹ. מִתְּחִלַּת הַשָּׁעָה הַשִּׁשִּׁית – הֶחָמֵץ אָסוּר בַּהֲנָאָה, וּכְבָר אִי אֶפְשָׁר לִמְכֹּר אוֹתוֹ אוֹ לְבַטֵּל אוֹתוֹ. אֶפְשָׁר רַק לִשְׂרֹף אוֹתוֹ, אוֹ לְפוֹרֵר וּלְפַזֵּר לָרוּחַ אוֹ לַיָּם.
🤔 לָמָּה? כְּשֶׁהֶחָמֵץ אָסוּר בַּהֲנָאָה, הוּא כְּבָר לֹא נֶחְשָׁב שֶׁלָּנוּ – וְלָכֵן אִי אֶפְשָׁר לִמְכֹּר אוֹתוֹ אוֹ לְהַפְקִיר אוֹתוֹ בְּמַחְשָׁבָה. נִשְׁאָר רַק לְהַעֲלִים אוֹתוֹ מִן הָעוֹלָם.
(פֶּרֶק ג׳, הֲלָכָה ו׳)

12מֵאֵימָתַי סוֹפְרִים אֶת הַשָּׁעוֹת?

יֵשׁ מַחְלֹקֶת מָתַי מַתְחִיל הַיּוֹם לְעִנְיַן חֶשְׁבּוֹן הַשָּׁעוֹת. לָכֵן בַּלּוּחוֹת כְּתוּבִים שְׁנֵי זְמַנִּים: מֻקְדָּם וּמְאֻחָר.
🤔 לָמָּה? הַזְּמַנִּים הָאֵלֶּה שֶׁל עֶרֶב פֶּסַח הֵם מִדִּבְרֵי חֲכָמִים, וּבְסָפֵק בְּדִבְרֵי חֲכָמִים הֲלָכָה כַּמֵּקֵל – אֶפְשָׁר לִסְמֹךְ עַל הַזְּמַן הַמְּאֻחָר. אֲבָל כְּשֶׁאֶפְשָׁר, טוֹב לְהַחְמִיר וְלִגְמֹר עַד הַזְּמַן הַמֻּקְדָּם.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְשִׁיטַת הַמָּגֵן אַבְרָהָם סוֹפְרִים מֵעֲלוֹת הַשַּׁחַר – כְּשֶׁרוֹאִים אֶת הָאוֹר הָרִאשׁוֹן בַּמִּזְרָח. זֶה הַזְּמַן הַמֻּקְדָּם.
לְשִׁיטַת הַגְּרָ״א סוֹפְרִים מֵהָנֵץ הַחַמָּה – כְּשֶׁרוֹאִים אֶת הַשֶּׁמֶשׁ עַצְמָהּ. זֶה הַזְּמַן הַמְּאֻחָר.
(פֶּרֶק ג׳, הֲלָכָה ו׳)

13אַרְבָּעָה שְׁלַבִּים

הַשְׁבָּתַת הֶחָמֵץ נַעֲשֵׂית בְּאַרְבָּעָה שְׁלַבִּים. בְּלֵיל י״ד: בְּדִיקַת חָמֵץ, וּמִיָּד אַחֲרֶיהָ – בִּטּוּל רִאשׁוֹן. בְּבֹקֶר י״ד: בִּעוּר הֶחָמֵץ (נָהֲגוּ לִשְׂרֹף אוֹתוֹ), וְאַחֲרָיו – בִּטּוּל שֵׁנִי. וְיֵשׁ עוֹד שְׁתֵּי דְּרָכִים לְהִפָּטֵר מֵהֶחָמֵץ: לִמְכֹּר אוֹתוֹ לְגוֹי, אוֹ לְהַפְקִיר אוֹתוֹ.
🤔 לָמָּה? בַּבְּדִיקָה, בַּבִּעוּר וּבַבִּטּוּל – אָנוּ ״נִלְחָמִים״ בֶּחָמֵץ וּמְכַלִּים אוֹתוֹ. בַּמְּכִירָה וּבַהֶפְקֵר – אָנוּ רַק מוֹצִיאִים אוֹתוֹ מֵרְשׁוּתֵנוּ. כָּל הַדְּרָכִים הָאֵלֶּה כְּשֵׁרוֹת.
(פֶּרֶק ג׳, הֲלָכָה ז׳)

📝 מָה לָמַדְנוּ

  • מִצְוַת עֲשֵׂה לְבַעֵר אֶת הֶחָמֵץ עַד חֲצוֹת יוֹם י״ד בְּנִיסָן – אֲנָשִׁים וְנָשִׁים שָׁוִים בָּהּ.
  • אָסוּר רַק חָמֵץ שֶׁל יְהוּדִי; חָמֵץ שֶׁל גּוֹי – שָׂמִים לְפָנָיו מְחִצָּה, וְלֹא אוֹכְלִים אִתּוֹ עַל שֻׁלְחָן אֶחָד.
  • חָמֵץ שֶׁקִּבַּלְנוּ עָלָיו אַחְרָיוּת – נֶחְשָׁב כְּשֶׁלָּנוּ, וְצָרִיךְ לְהַחְזִיר אוֹ לְבַעֵר.
  • מְבַעֲרִים בַּמַּחְשָׁבָה (בִּטּוּל) וּבַמַּעֲשֶׂה (בְּדִיקָה וּבִעוּר) – בְּאַרְבָּעָה שְׁלַבִּים.
  • מִתְּחִלַּת הַשָּׁעָה הַשִּׁשִּׁית אִי אֶפְשָׁר עוֹד לִמְכֹּר אוֹ לְבַטֵּל – רַק לְבַעֵר.

❓ שְׁאֵלוֹת חֲזָרָה

  1. שָׁכֵן גּוֹי בָּא לְבַקֵּר בְּפֶסַח עִם כָּרִיךְ. מָה מֻתָּר וּמָה אָסוּר?
  2. לָמָּה חֲכָמִים לֹא הִסְתַּפְּקוּ רַק בְּבִטּוּל אוֹ רַק בִּבְדִיקָה?
  3. מָהֵם אַרְבָּעַת הַשְּׁלַבִּים שֶׁל הַשְׁבָּתַת הֶחָמֵץ, וּמָתַי כָּל אֶחָד מֵהֶם?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״חָמֵץ וּמַצָּה – מָה הַסּוֹד?״

סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת פֶּסַח
מַהֲדוּרַת יְלָדִים

🌕 לֵיל הַסֵּדֶר

פֶּרֶק ט״ז · לַמַּהֲדוּרָה לַמּוֹרֶה
קַדֵּשׁ, וּרְחַץ, כַּרְפַּס, יַחַץ... אֶת הַשִּׁיר הַזֶּה אַתֶּם בֶּטַח מַכִּירִים. בְּלֵיל הַסֵּדֶר הַכֹּל הוֹלֵךְ לְפִי ״סֵדֶר״ – וּבְכָל שָׁלָב יֵשׁ הֲלָכוֹת, טְעָמִים וּמִנְהָגִים. בַּפֶּרֶק הַזֶּה נַעֲבֹר יַחַד עַל כָּל הַלַּיְלָה, צַעַד אַחַר צַעַד: מֵהַהֲכָנוֹת וְהַקְּעָרָה, דֶּרֶךְ אַרְבַּע הַכּוֹסוֹת, הַהַגָּדָה, הַמַּצָּה וְהַמָּרוֹר, וְעַד הָאֲפִיקוֹמָן, הַהַלֵּל וְ״נִרְצָה״. מוּכָנִים? בּוֹאוּ נַתְחִיל!

1מָה עוֹשִׂים בַּלַּיְלָה הַזֶּה?

בְּלֵיל הַסֵּדֶר יֵשׁ שְׁנֵי דְּבָרִים מֶרְכָּזִיִּים: לִזְכֹּר אֵיךְ יָצָאנוּ מֵעַבְדוּת לְחֵרוּת, וּלְהַעֲבִיר אֶת הַסִּפּוּר לַדּוֹר הַבָּא. מִן הַתּוֹרָה צָרִיךְ לֶאֱכֹל בַּלַּיְלָה הַזֶּה פֶּסַח, מַצָּה וּמָרוֹר. הַיּוֹם, כְּשֶׁאֵין בֵּית מִקְדָּשׁ, אֵין קָרְבַּן פֶּסַח, וּבִמְקוֹמוֹ אוֹכְלִים אֶת הָ״אֲפִיקוֹמָן״ – זֵכֶר לְקָרְבַּן הַפֶּסַח. הַמַּצָּה נִשְׁאֲרָה מִצְוָה מִן הַתּוֹרָה, וְאֶת הַמָּרוֹר אוֹכְלִים כִּי חֲכָמִים תִּקְּנוּ. וַחֲכָמִים הוֹסִיפוּ עוֹד: אַרְבַּע כּוֹסוֹת, וַאֲכִילָה וּשְׁתִיָּה בַּהֲסִבָּה – דֶּרֶךְ חֵרוּת.
🤔 לָמָּה? הַפֶּסַח מַזְכִּיר שֶׁה׳ פָּסַח עַל בָּתֵּי יִשְׂרָאֵל וְהִצִּיל אֶת הַבְּכוֹרִים. הַמַּצָּה – זֵכֶר לַמַּצּוֹת שֶׁאָכְלוּ אֲבוֹתֵינוּ כְּשֶׁיָּצְאוּ לְחֵרוּת. הַמָּרוֹר – זֵכֶר לַשִּׁעְבּוּד הַמַּר.
(פֶּרֶק ט״ז, הֲלָכָה א׳)

2מְשַׁנִּים – כְּדֵי שֶׁנִּשְׁאַל

כְּדֵי שֶׁהַיְּלָדִים יִשָּׁאֲרוּ עֵרִים וְסַקְרָנִים, מְשַׁנִּים בְּלֵיל הַסֵּדֶר הַרְבֵּה דְּבָרִים: טוֹבְלִים פַּעֲמַיִם, נוֹטְלִים יָדַיִם פַּעֲמַיִם, וְנִרְאֶה כְּאִלּוּ מַתְחִילִים לֶאֱכֹל – וּפִתְאֹם עוֹצְרִים לְהַגָּדָה. גַּם הַמַּצָּה, אַרְבַּע הַכּוֹסוֹת וְהַהֲסִבָּה מְעוֹרְרִים אֶת הַשְּׁאֵלָה ״מַה נִּשְׁתַּנָּה?״. וַחֲכָמִים אָמְרוּ לְחַלֵּק לַיְּלָדִים בִּתְחִלַּת הַסֵּדֶר אֱגוֹזִים וּמִינֵי מְתִיקָה, וְטוֹב לְחַלֵּק לָהֶם מַמְתַּקִּים קְטַנִּים גַּם בְּמֶשֶׁךְ הַהַגָּדָה.
🤔 לָמָּה? אֶחָד מִשְּׁנֵי הַדְּבָרִים הַמֶּרְכָּזִיִּים שֶׁל הַלַּיְלָה הוּא לְהַעֲבִיר אֶת הַמָּסֹרֶת לַבָּנִים וְלַבָּנוֹת – וְזֶה אֶפְשָׁרִי רַק כְּשֶׁהֵם עֵרִים, שְׂמֵחִים וְשׁוֹאֲלִים.
💛 כָּל מַמְתָּק קָטָן בַּסֵּדֶר אוֹמֵר לְךָ: אַתָּה חָשׁוּב כָּאן, וְהַסִּפּוּר הַזֶּה שֶׁלְּךָ.
(פֶּרֶק ט״ז, הֲלָכָה ב׳)

3בְּגָדִים חֲדָשִׁים וְשֻׁלְחָן מְפֹאָר

לִפְנֵי פֶּסַח מִשְׁתַּדְּלִים לִקְנוֹת לַיְּלָדִים וּלְכָל בְּנֵי הַבַּיִת בְּגָדִים חֲדָשִׁים. בְּכָל חַג מִצְוָה לְשַׂמֵּחַ אֶת בְּנֵי הַבַּיִת: לִקְנוֹת לַנָּשִׁים וְלַבָּנוֹת בְּגָדִים וְתַכְשִׁיטִים, וּלְחַלֵּק לַיְּלָדִים הַקְּטַנִּים מִינֵי מְתִיקָה וֶאֱגוֹזִים. וְעַל הַשֻּׁלְחָן שָׂמִים אֶת הַכֵּלִים הַיָּפִים בְּיוֹתֵר, וּמְכִינִים כִּסְאוֹת נוֹחִים לַהֲסִבָּה.
🤔 לָמָּה? כְּשֶׁלּוֹבְשִׁים בֶּגֶד חָדָשׁ לִפְנֵי לֵיל הַסֵּדֶר, מִתְעוֹרֶרֶת הִתְרַגְּשׁוּת מְיֻחֶדֶת לִקְרַאת הַלַּיְלָה הַקָּדוֹשׁ. וְשֻׁלְחָן מְפֹאָר הוּא עוֹד דֶּרֶךְ לְהַרְגִּישׁ חֵרוּת וְשִׂמְחָה.
(פֶּרֶק ט״ז, הֲלָכָה ב׳)

4מְקַדְּשִׁים מִיָּד – אֲבָל לֹא לִפְנֵי צֵאת הַכּוֹכָבִים

נָכוֹן לַעֲרֹךְ אֶת הַשֻּׁלְחָן, הַקְּעָרָה וְהַצַּלָּחוֹת לִפְנֵי הָעֶרֶב, כְּדֵי שֶׁמִּיָּד כְּשֶׁחוֹזְרִים מִבֵּית הַכְּנֶסֶת אֶפְשָׁר לְהַתְחִיל בַּקִּדּוּשׁ – כְּשֶׁהַקְּטַנִּים עוֹד עֵרִים. אֲבָל לֹא מְקַדְּשִׁים לִפְנֵי צֵאת הַכּוֹכָבִים.
🤔 לָמָּה? כּוֹס הַקִּדּוּשׁ הִיא הַכּוֹס הָרִאשׁוֹנָה מֵאַרְבַּע הַכּוֹסוֹת, וְאֶת אַרְבַּע הַכּוֹסוֹת שׁוֹתִים בַּלַּיְלָה – בְּלֵיל ט״ו בְּנִיסָן, הַזְּמַן שֶׁל אֲכִילַת הַמַּצָּה.
(פֶּרֶק ט״ז, הֲלָכָה ב׳)

5הַקְּעָרָה

בַּקְּעָרָה מַנִּיחִים: שָׁלֹשׁ מַצּוֹת (בְּתוֹךְ מַפִּית); מָרוֹר – חַסָּה אוֹ חֲזֶרֶת; זְרוֹעַ צְלוּיָה – זֵכֶר לְקָרְבַּן פֶּסַח; בֵּיצָה צְלוּיָה אוֹ מְבֻשֶּׁלֶת – זֵכֶר לְקָרְבַּן חֲגִיגָה; כַּרְפַּס, עִם מֵי מֶלַח אוֹ חֹמֶץ לִטְבֹּל בָּהֶם; וַחֲרֹסֶת – זֵכֶר לַטִּיט שֶׁעָשׂוּ אֲבוֹתֵינוּ בְּמִצְרַיִם, וּבָהּ טוֹבְלִים אֶת הַמָּרוֹר. הַיַּיִן לֹא בַּקְּעָרָה, כִּי הוּא מַשְׁקֶה וְלֹא מַאֲכָל.
🤔 לָמָּה? כָּל מַאֲכָל בַּקְּעָרָה מַזְכִּיר רַעְיוֹן. הַזְּרוֹעַ – כִּי נִגְאַלְנוּ ״בִּזְרוֹעַ נְטוּיָה״, וְצוֹלִים אוֹתָהּ כְּמוֹ שֶׁצָּלוּ אֶת הַפֶּסַח. הַבֵּיצָה – כִּי נוֹהֲגִים לְהַגִּישׁ בֵּיצָה לַאֲבֵלִים, וְגַם אָנוּ מִתְנַחֲמִים שֶׁבִּמְהֵרָה יִבָּנֶה בֵּית הַמִּקְדָּשׁ; וּבַאֲרָמִית ״בֵּיעָא״ פֵּרוּשׁוֹ גַּם ״בַּקָּשָׁה״ – שֶׁה׳ יִגְאָלֵנוּ.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי סְפָרַד לַזְּרוֹעַ לוֹקְחִים רֶגֶל קִדְמִית שֶׁל בְּהֵמָה.
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז לַזְּרוֹעַ לוֹקְחִים כָּנָף שֶׁל עוֹף.
(פֶּרֶק ט״ז, הֲלָכָה ג׳)

6אֵיךְ מְסַדְּרִים אֶת הַקְּעָרָה?

אֵין חוֹבָה לְהָנִיחַ קְעָרָה לִפְנֵי כָּל אֶחָד – מַסְפִּיק לִפְנֵי עוֹרֵךְ הַסֵּדֶר. וְיֵשׁ נוֹהֲגִים לְהָנִיחַ קְעָרָה לִפְנֵי כָּל בַּעַל מִשְׁפָּחָה. גַּם בְּסִדּוּר הַמַּאֲכָלִים בַּקְּעָרָה יֵשׁ כַּמָּה מִנְהָגִים, וּבַהַגָּדוֹת רַבּוֹת מֻדְפָּס צִיּוּר שֶׁל הַקְּעָרָה. לְכָל מִנְהָג יֵשׁ מָקוֹם בַּהֲלָכָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְפִי הָרַמָּ״א מָה שֶׁאוֹכְלִים קֹדֶם מֻנָּח קָרוֹב יוֹתֵר לָעוֹרֵךְ: הַכַּרְפַּס וּמֵי הַמֶּלַח, אַחֲרֵיהֶם הַמַּצּוֹת, אַחֲרֵיהֶן הַמָּרוֹר וְהַחֲרֹסֶת, וְהָרְחוֹקִים בְּיוֹתֵר – הַזְּרוֹעַ וְהַבֵּיצָה. כָּךְ לֹא ״מְדַלְּגִים״ עַל מִצְוָה.
לְפִי הָאֲרִ״י סֵדֶר שֶׁרוֹמֵז לְעֶשֶׂר סְפִירוֹת. כָּךְ נוֹהֲגִים רֹב יוֹצְאֵי סְפָרַד וְרֹב מִתְפַּלְּלֵי נֹסַח סְפָרַד, וְגַם יֵשׁ מִמִּתְפַּלְּלֵי נֹסַח אַשְׁכְּנַז.
(פֶּרֶק ט״ז, הֲלָכָה ד׳)

7קַדֵּשׁ

פּוֹתְחִים אֶת הַסֵּדֶר בְּקִדּוּשׁ. בְּלֵיל הַסֵּדֶר מוֹזְגִים כּוֹס לְכָל אֶחָד מֵהַמְּסֻבִּים, וְאַחֲרֵי הַקִּדּוּשׁ כָּל אֶחָד מֵסֵב וְשׁוֹתֶה – זוֹ הַכּוֹס הָרִאשׁוֹנָה מֵאַרְבַּע הַכּוֹסוֹת. בְּסוֹף הַקִּדּוּשׁ מְבָרְכִים ״שֶׁהֶחֱיָנוּ״, וּמִי שֶׁשָּׁכַח – מְבָרֵךְ כְּשֶׁנִּזְכָּר, כָּל עוֹד לֹא נִגְמַר הַחַג. מִי שֶׁאוֹמֵר ״לְשֵׁם יִחוּד״ אוֹ ״הִנְנִי מוּכָן״ – אוֹמֵר לִפְנֵי הַקִּדּוּשׁ, וְלֹא בֵּין הַבְּרָכָה לַשְּׁתִיָּה.
🤔 לָמָּה? בְּכָל קִדּוּשׁ אוֹמְרִים ״זֵכֶר לִיצִיאַת מִצְרַיִם״ – כִּי שָׁם הִתְחִילָה קְדֻשַּׁת יִשְׂרָאֵל. וּבַלַּיְלָה שֶׁבּוֹ יָצָאנוּ, הַקִּדּוּשׁ כָּפוּל וּמְכֻפָּל. וְלָמָּה עַל יַיִן? כְּדֵי לְלַמֵּד שֶׁהַקְּדֻשָּׁה מִתְגַּלָּה גַּם בַּחַיִּים הַגַּשְׁמִיִּים וּבַשִּׂמְחָה – לֹא רַק בַּתְּפִלָּה וּבַלִּמּוּד.
💛 ״שֶׁהֶחֱיָנוּ וְקִיְּמָנוּ וְהִגִּיעָנוּ לַזְּמַן הַזֶּה״ – תּוֹדָה, ה׳, שֶׁזָּכִינוּ לְהַגִּיעַ לַלַּיְלָה הַזֶּה עוֹד פַּעַם.
(פֶּרֶק ט״ז, הֲלָכָה ה׳)

8אַרְבַּע כּוֹסוֹת

חֲכָמִים תִּקְּנוּ לִשְׁתּוֹת בְּלֵיל הַסֵּדֶר אַרְבַּע כּוֹסוֹת, וְכָל חֵלֶק שֶׁל הַלַּיְלָה נֶאֱמָר עַל כּוֹס: הַקִּדּוּשׁ – עַל הַכּוֹס הָרִאשׁוֹנָה; הַהַגָּדָה וַחֲצִי הַהַלֵּל – עַל הַשְּׁנִיָּה; בִּרְכַּת הַמָּזוֹן – עַל הַשְּׁלִישִׁית; וַחֲצִי הַהַלֵּל הַשֵּׁנִי וְהַלֵּל הַגָּדוֹל – עַל הָרְבִיעִית. מִי שֶׁשּׁוֹתֶה אֶת אַרְבַּע הַכּוֹסוֹת בְּרֶצֶף, זוֹ אַחַר זוֹ – נֶחְשָׁב כְּאִלּוּ שָׁתָה כּוֹס אַחַת בִּלְבַד.
🤔 לָמָּה? כְּדֵי לְהַרְבּוֹת בְּשִׂמְחַת הַגְּאֻלָּה וּבְחֵרוּת. אַרְבַּע הַכּוֹסוֹת הֵן כְּנֶגֶד אַרְבַּע לְשׁוֹנוֹת שֶׁל גְּאֻלָּה: ״וְהוֹצֵאתִי, וְהִצַּלְתִּי, וְגָאַלְתִּי, וְלָקַחְתִּי״. וְהַמִּסְפָּר אַרְבַּע רוֹמֵז לְאַרְבַּע רוּחוֹת הָעוֹלָם – יְצִיאַת מִצְרַיִם שִׁנְּתָה אֶת הָעוֹלָם כֻּלּוֹ.
(פֶּרֶק ט״ז, הֲלָכָה ו׳)

9יַיִן אוֹ מִיץ עֲנָבִים?

חֲכָמִים תִּקְּנוּ אַרְבַּע כּוֹסוֹת שֶׁל יַיִן מְשַׂמֵּחַ, כְּדֵי שֶׁנְּסַפֵּר אֶת סִפּוּר יְצִיאַת מִצְרַיִם מִתּוֹךְ שִׂמְחָה וְחֵרוּת. אֲבָל צָרִיךְ לְהִזָּהֵר שֶׁהַיַּיִן לֹא יִהְיֶה חָזָק מִדַּי וְיַפְרִיעַ לְהִתְרַכֵּז בַּהַגָּדָה. מִי שֶׁקָּשֶׁה לוֹ – מְעָרֵב בַּיַּיִן מִיץ עֲנָבִים אוֹ מַיִם, וּבִלְבַד שֶׁהַמַּיִם לֹא יִהְיוּ הָרֹב. יֵשׁ מַעֲלָה בְּיַיִן אָדֹם, אֲבָל יַיִן לָבָן מְשֻׁבָּח עָדִיף עַל אָדֹם פָּחוֹת מְשֻׁבָּח. גַּם נָשִׁים וּבָנוֹת מְצֻוּוֹת בְּאַרְבַּע כּוֹסוֹת, כְּמוֹ בְּכָל מִצְווֹת לֵיל הַסֵּדֶר.
🤔 לָמָּה? בִּימֵי חֲכָמִים לֹא הָיָה אֶפְשָׁר לִשְׁמֹר מִיץ עֲנָבִים מֵהַבָּצִיר עַד פֶּסַח. מְסֻפָּר עַל רַבִּי יְהוּדָה בְּרַבִּי אִלְעַאי שֶׁהָיָה קוֹשֵׁר מִטְפַּחַת עַל רֹאשׁוֹ מִפֶּסַח וְעַד שָׁבוּעוֹת, מֵרֹב כְּאֵב רֹאשׁ מֵאַרְבַּע הַכּוֹסוֹת – וְלֹא וִתֵּר עֲלֵיהֶן.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְפִי רַבִּים מֵהַפּוֹסְקִים אֶפְשָׁר לָצֵאת יְדֵי חוֹבַת אַרְבַּע כּוֹסוֹת גַּם בְּמִיץ עֲנָבִים.
לְפִי הָרַב מֶלָמֵד הַמִּצְוָה כְּפִי שֶׁתִּקְּנוּ חֲכָמִים הִיא בְּיַיִן מְשַׂמֵּחַ. בְּמִיץ עֲנָבִים יוֹצְאִים בְּדִיעֲבַד, וּמִי שֶׁהַיַּיִן מַזִּיק לוֹ – כְּגוֹן שֶׁגּוֹרֵם לוֹ כְּאֵב רֹאשׁ – רַשַּׁאי לָצֵאת בְּמִיץ עֲנָבִים.
(פֶּרֶק ט״ז, הֲלָכָה ז׳)

10כּוֹס שְׁלֵמָה, נְקִיָּה וּמְלֵאָה

בְּכָל כּוֹס צָרִיךְ לִהְיוֹת לְפָחוֹת ״רְבִיעִית״. הַכּוֹס צְרִיכָה לִהְיוֹת שְׁלֵמָה וְלֹא שְׁבוּרָה, נְקִיָּה וּשְׁטוּפָה מִבַּחוּץ וּמִבִּפְנִים, וּמְלֵאָה בִּנְדִיבוּת – לֹא עַד הַשָּׂפָה מַמָּשׁ, שֶׁלֹּא יִשָּׁפֵךְ, אֶלָּא בְּעֵרֶךְ חֲצִי ס״מ מִתַּחַת לַשָּׂפָה. לִפְנֵי הַכּוֹסוֹת הַבָּאוֹת לֹא צָרִיךְ לִשְׁטֹף שׁוּב, כָּל עוֹד הַכּוֹס לֹא הִתְלַכְלְכָה.
🤔 לָמָּה? כּוֹס שֶׁל מִצְוָה צְרִיכָה כַּמּוּת חֲשׁוּבָה. וְכוֹס מְלֵאָה – שֶׁלֹּא נֵרָאֶה כְּמוֹ קַמְצָנִים.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְפִי רֹב הַפּוֹסְקִים (וְכָךְ נוֹהֲגִים יוֹצְאֵי סְפָרַד) רְבִיעִית הִיא כְּ־75 מִילִילִיטֶר.
לְפִי הַחֲזוֹן אִישׁ (מִפּוֹסְקֵי אַשְׁכְּנַז) רְבִיעִית הִיא כְּ־150 מִילִילִיטֶר. אֲבָל בְּאַרְבַּע כּוֹסוֹת, שֶׁהֵן מִדִּבְרֵי חֲכָמִים, אֶפְשָׁר לָצֵאת גַּם בַּשִּׁעוּר הַקָּטָן – וּלְכַתְּחִלָּה מוֹזְגִים כּוֹס נָאָה וּגְדוֹלָה.
(פֶּרֶק ט״ז, הֲלָכָה ח׳)

11כַּמָּה שׁוֹתִים?

לְכַתְּחִלָּה טוֹב לִשְׁתּוֹת אֶת כָּל הַכּוֹס, וּלְפָחוֹת אֶת רֻבָּהּ. הַשִּׁעוּר הַקָּטָן בְּיוֹתֵר: רֹב רְבִיעִית שֶׁהוּא גַּם ״מְלֹא לֻגְמָיו״ – כַּמּוּת שֶׁמְּמַלֵּאת אֶת הַפֶּה כְּשֶׁמְּנַפְּחִים לֶחִי אַחַת. שׁוֹתִים אוֹתוֹ בְּבַת אַחַת, בִּרְצִיפוּת, עִם הַפְסָקוֹת קַלּוֹת לִנְשִׁימָה. וְהַיְּלָדִים? יְלָדִים שֶׁהִגִּיעוּ לְגִיל חִנּוּךְ – בְּדֶרֶךְ כְּלָל בְּגִיל חָמֵשׁ אוֹ שֵׁשׁ, כְּשֶׁהֵם מְבִינִים אֶת הַקִּדּוּשׁ וְהַהַגָּדָה – מְקַבְּלִים אַרְבַּע כּוֹסוֹת, וּמַסְפִּיק שֶׁיִּשְׁתּוּ ״מְלֹא לֻגְמֵיהֶם״.
🤔 לָמָּה? שְׁתִיָּה שֶׁל מְלֹא הַפֶּה מְיַשֶּׁבֶת אֶת הַדַּעַת. וּמִי שֶׁשּׁוֹתֶה חֲצִי עַכְשָׁו וַחֲצִי אַחֲרֵי זְמַן רַב – זוֹ לֹא שְׁתִיָּה אַחַת.
💛 גַּם הַיְּלָדִים שׁוֹתִים אַרְבַּע כּוֹסוֹת – כִּי גַּם הֵם יָצְאוּ מִמִּצְרַיִם!
(פֶּרֶק ט״ז, הֲלָכָה ט׳)

12הֲסִבָּה – לָמָּה?

״בְּכָל דּוֹר וָדוֹר חַיָּב אָדָם לְהַרְאוֹת אֶת עַצְמוֹ כְּאִלּוּ הוּא יָצָא מִמִּצְרַיִם״. לָכֵן תִּקְּנוּ חֲכָמִים לֶאֱכֹל וְלִשְׁתּוֹת בַּהֲסִבָּה. חוֹבָה לְהָסֵב בַּאֲכִילַת כְּזַיִת מַצָּה, כְּזַיִת כּוֹרֵךְ, כְּזַיִת אֲפִיקוֹמָן, וּבִשְׁתִיַּת אַרְבַּע הַכּוֹסוֹת. בִּשְׁאָר הַסְּעוּדָה – מִי שֶׁמֵּסֵב, מְשֻׁבָּח. וְלֹא מְסִבִּים: בַּאֲכִילַת הַמָּרוֹר, בִּזְמַן אֲמִירַת הַהַגָּדָה, וּבְבִרְכַּת הַמָּזוֹן.
🤔 לָמָּה? עֶבֶד יוֹשֵׁב זָקוּף, מוּכָן לָקוּם בְּכָל רֶגַע לָעֲבוֹדָה. בֶּן חוֹרִין יָכוֹל לְהִשָּׁעֵן אָחוֹרָה וְלָנוּחַ. בַּמָּרוֹר לֹא מְסִבִּים, כִּי הוּא זֵכֶר לַשִּׁעְבּוּד. וְאֶת הַהַגָּדָה וּבִרְכַּת הַמָּזוֹן אוֹמְרִים בְּרִכּוּז וּבְיִרְאַת שָׁמַיִם.
💛 בַּלַּיְלָה הַזֶּה אֲנִי מַרְגִּישׁ אֶת זֶה בַּגּוּף: ה׳ שִׁחְרֵר אוֹתִי – וַאֲנִי בֶּן חוֹרִין.
(פֶּרֶק ט״ז, הֲלָכָה י׳)

13אֵיךְ מְסִבִּים?

מוֹשְׁכִים אֶת הַגּוּף קְצָת קָדִימָה עַל הַכִּסֵּא, נִשְׁעָנִים עַל גַּב הַכִּסֵּא, וּמַטִּים אֶת הַגּוּף לְצַד שְׂמֹאל. לְכַתְּחִלָּה טוֹב כִּסֵּא מְרֻפָּד עִם מִשְׁעָנוֹת לַיָּדַיִם, אוֹ כָּרִית. גַּם מִי שֶׁהוּא שְׂמָאלִי – מֵסֵב לְצַד שְׂמֹאל, וּמִשְׁתַּמֵּשׁ בַּלַּיְלָה הַזֶּה בְּיַד יָמִין. מִי שֶׁהֵסֵב לְצַד יָמִין – לֹא יָצָא (אֲבָל שְׂמָאלִי שֶׁהֵסֵב לְיָמִין – יָצָא). וְתַלְמִיד שֶׁיּוֹשֵׁב לִפְנֵי רַבּוֹ – מְבַקֵּשׁ מִמֶּנּוּ רְשׁוּת לְהָסֵב.
🤔 לָמָּה? כְּשֶׁמֵּסֵבִּים לִשְׂמֹאל, יַד יָמִין פְּנוּיָה לָקַחַת אֶת הַמַּצָּה וְהַכּוֹס. וְעוֹד: מִי שֶׁמַּטֶּה לְיָמִין – עָלוּל לְהֵחָנֵק. וּלְפָנֵי הָרַב מְבַקְּשִׁים רְשׁוּת, כִּי כְּבוֹד הַתּוֹרָה קוֹדֵם.
(פֶּרֶק ט״ז, הֲלָכָה י״א)

14שָׁכַחְתִּי לְהָסֵב!

מִי שֶׁאָכַל אֶת הַכְּזַיִת מַצָּה בְּלִי הֲסִבָּה – לֹא יָצָא. הוּא חוֹזֵר וְאוֹכֵל כְּזַיִת מַצָּה בַּהֲסִבָּה, אֲבָל לֹא מְבָרֵךְ שׁוּב ״עַל אֲכִילַת מַצָּה״. מִי שֶׁשָּׁכַח לְהָסֵב בַּכּוֹרֵךְ – לֹא חוֹזֵר. וּבָאֲפִיקוֹמָן – אִם קַל לוֹ, חוֹזֵר וְאוֹכֵל בַּהֲסִבָּה, וְאִם קָשֶׁה לוֹ – לֹא. וּמִי שֶׁקָּשֶׁה לוֹ לִשְׁתּוֹת עוֹד כּוֹס – לֹא חוֹזֵר בְּאַף כּוֹס.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְפִי הַשֻּׁלְחָן עָרוּךְ (מִנְהַג יוֹצְאֵי סְפָרַד) מִי שֶׁשָּׁתָה כּוֹס בְּלִי הֲסִבָּה – לְכַתְּחִלָּה חוֹזֵר וְשׁוֹתֶה אוֹתָהּ בַּהֲסִבָּה.
לְפִי הָרַמָּ״א (מִנְהַג יוֹצְאֵי אַשְׁכְּנַז) חוֹזֵר וְשׁוֹתֶה רַק אִם זוֹ הָיְתָה הַכּוֹס הַשְּׁנִיָּה. בְּכוֹס רִאשׁוֹנָה, שְׁלִישִׁית אוֹ רְבִיעִית – לֹא חוֹזֵר, שֶׁלֹּא יֵרָאֶה כְּמוֹסִיף עַל אַרְבַּע הַכּוֹסוֹת.
(פֶּרֶק ט״ז, הֲלָכָה י״ב)

15נָשִׁים וּבָנוֹת וְהַהֲסִבָּה

נָשִׁים וּבָנוֹת חַיָּבוֹת בַּמַּצָּה וּבְאַרְבַּע הַכּוֹסוֹת כְּמוֹ כֻּלָּם. וּמָה עִם הַהֲסִבָּה? הָרַב מֶלָמֵד כּוֹתֵב שֶׁנָּשִׁים לְכַתְּחִלָּה אוֹכְלוֹת אֶת הַמַּצָּה וְשׁוֹתוֹת אֶת אַרְבַּע הַכּוֹסוֹת בַּהֲסִבָּה, אֲבָל אִם שָׁכְחוּ – אֵינָן צְרִיכוֹת לַחֲזֹר.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְפִי הָרַב מֶלָמֵד נָשִׁים מְסִבּוֹת לְכַתְּחִלָּה, וְאִם שָׁכְחוּ – אֵינָן חוֹזְרוֹת.
לְפִי הָרַמָּ״א רַבּוֹת מִנְּשׁוֹת אַשְׁכְּנַז נוֹהֲגוֹת שֶׁלֹּא לְהָסֵב.
(פֶּרֶק ט״ז, הֲלָכָה י״ב)

16בֵּין כּוֹס רִאשׁוֹנָה לַשְּׁנִיָּה

בֵּין הַכּוֹס הָרִאשׁוֹנָה לַשְּׁנִיָּה טוֹב שֶׁלֹּא לִשְׁתּוֹת עוֹד יַיִן, אֲבָל מֻתָּר לִשְׁתּוֹת מִיץ עֲנָבִים אוֹ מִיץ אַחֵר – וּבְדֶרֶךְ כְּלָל לֹא צָרִיךְ לְבָרֵךְ עָלָיו, כִּי בִּרְכַּת הַיַּיִן פָּטְרָה אוֹתוֹ. בְּכוֹר שֶׁצָּם, אוֹ מִי שֶׁרָעֵב מְאוֹד, יָכוֹל לֶאֱכֹל אַחֲרֵי הַקִּדּוּשׁ מְעַט תַּפּוּחֵי אֲדָמָה, בֵּיצָה אוֹ פֵּרוֹת – אֲבָל רַק כְּשֶׁיֵּשׁ צֹרֶךְ גָּדוֹל. וּמִשֶּׁהִתְחִילוּ לוֹמַר אֶת הַהַגָּדָה – כְּבָר לֹא אוֹכְלִים וְלֹא שׁוֹתִים.
🤔 לָמָּה? יַיִן נוֹסָף עָלוּל לְהַפְרִיעַ לְהִתְרַכֵּז בַּהַגָּדָה. וְאֶת הָאֹכֶל מְמַעֲטִים, כְּדֵי לְהַגִּיעַ לַמַּצָּה רְעֵבִים.
(פֶּרֶק ט״ז, הֲלָכָה י״ג)

17וּרְחַץ

אַחֲרֵי הַקִּדּוּשׁ נוֹטְלִים יָדַיִם – בְּלִי בְּרָכָה – לִפְנֵי אֲכִילַת הַכַּרְפַּס. מִי שֶׁטָּעָה וּבֵרַךְ – אֵין בְּיָדוֹ עָווֹן, כִּי יֵשׁ פּוֹסְקִים שֶׁאוֹמְרִים לְבָרֵךְ. וּבְכָל זֹאת, כְּשֶׁיִּטֹּל לִפְנֵי הַמַּצָּה – יְבָרֵךְ שׁוּב.
🤔 לָמָּה? תִּקְּנוּ לִטְבֹּל אֶת הַיָּרָק בְּמַשְׁקֶה, וְאָז צָרִיךְ לִטֹּל יָדַיִם לְפָנָיו. כָּךְ נוֹטְלִים יָדַיִם פַּעֲמַיִם בַּלַּיְלָה, וְהַיְּלָדִים שׁוֹאֲלִים: מַה נִּשְׁתַּנָּה? וְלֹא מְבָרְכִים, כִּי יֵשׁ אוֹמְרִים שֶׁכַּיּוֹם אֵין צֹרֶךְ בִּנְטִילָה כָּזוֹ – וּבְסָפֵק בְּרָכוֹת לֹא מְבָרְכִים.
(פֶּרֶק ט״ז, הֲלָכָה י״ד)

18כַּרְפַּס

אוֹכְלִים מְעַט יָרָק טָבוּל בְּמֵי מֶלַח אוֹ בְּחֹמֶץ. מְבָרְכִים ״בּוֹרֵא פְּרִי הָאֲדָמָה״, וּמְכַוְּנִים שֶׁהַבְּרָכָה תִּפְטֹר גַּם אֶת הַמָּרוֹר שֶׁנֹּאכַל אַחַר כָּךְ. אוֹכְלִים פָּחוֹת מִכְּזַיִת, וְלֹא מְסִבִּים. כָּל מִשְׁפָּחָה מַמְשִׁיכָה בַּמִּנְהָג שֶׁלָּהּ.
🤔 לָמָּה? יָרָק לִפְנֵי הַסְּעוּדָה הוּא שִׁנּוּי שֶׁמְּעוֹרֵר שְׁאֵלוֹת, וְגַם דֶּרֶךְ שֶׁל בְּנֵי חוֹרִין – לִפְתֹּחַ סְעוּדָה חֲשׁוּבָה בְּמַאֲכָל שֶׁמְּעוֹרֵר תֵּאָבוֹן. וְהַשֵּׁם ״כַּרְפַּס״ רוֹמֵז: ״ס׳ פֶּרֶךְ״ – שִׁשִּׁים רִבּוֹא (שֵׁשׁ מֵאוֹת אֶלֶף) אִישׁ שֶׁעָבְדוּ בַּעֲבוֹדַת פֶּרֶךְ. וְלֹא מְסִבִּים, כִּי יֵשׁ אוֹמְרִים שֶׁהַכַּרְפַּס רוֹמֵז לְצַעַר הַשִּׁעְבּוּד.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי סְפָרַד לוֹקְחִים סֶלֶרִי.
לְמִנְהַג רֹב יוֹצְאֵי אַשְׁכְּנַז לוֹקְחִים תַּפּוּחֵי אֲדָמָה מְבֻשָּׁלִים. וּמִקְצָת יוֹצְאֵי אַשְׁכְּנַז לוֹקְחִים פֶּטְרוֹזִילְיָה.
(פֶּרֶק ט״ז, הֲלָכָה ט״ו)

19יַחַץ

בּוֹצְעִים אֶת הַמַּצָּה הָאֶמְצָעִית לִשְׁנַיִם. אֶת הַחֵלֶק הַקָּטָן מַשְׁאִירִים בֵּין שְׁתֵּי הַמַּצּוֹת הַשְּׁלֵמוֹת, וְאֶת הַחֵלֶק הַגָּדוֹל שׁוֹמְרִים לָ״אֲפִיקוֹמָן״. נוֹהֲגִים לַעֲטֹף אוֹתוֹ בְּמַפִּית, וְיֵשׁ נוֹהֲגִים לָשִׂים אוֹתוֹ רֶגַע עַל הַכָּתֵף – זֵכֶר לַפָּסוּק ״מִשְׁאֲרֹתָם צְרֻרֹת בְּשִׂמְלֹתָם עַל שִׁכְמָם״. אַחַר כָּךְ מַחְבִּיאִים אֶת הָאֲפִיקוֹמָן לְסוֹף הַסְּעוּדָה. יֵשׁ מִשְׁפָּחוֹת שֶׁהַיְּלָדִים ״חוֹטְפִים״ אֶת הָאֲפִיקוֹמָן, וּמַחְזִירִים אוֹתוֹ בְּסוֹף הַסְּעוּדָה תְּמוּרַת מַתָּנָה. וּבְבֵית הָרַב מֶלָמֵד נוֹהֲגִים לָתֵת מַתָּנוֹת לְכָל הַיְּלָדִים שֶׁנִּשְׁאָרִים עֵרִים עַד אֲכִילַת הָאֲפִיקוֹמָן.
🤔 לָמָּה? הַמַּצָּה הִיא ״לֶחֶם עֹנִי״, וְעָנִי אוֹכֵל חֲצִי לֶחֶם – אֵין לוֹ כֶּסֶף לְכִכָּר שְׁלֵמָה. וּבוֹצְעִים לִפְנֵי הַהַגָּדָה, כִּי אוֹמְרִים אֶת הַהַגָּדָה כְּשֶׁהַמַּצָּה – כְּמוֹ שֶׁנֹּאכַל אוֹתָהּ – מֻנַּחַת לְפָנֵינוּ. וְהָעֶלְיוֹנָה וְהַתַּחְתּוֹנָה נִשְׁאָרוֹת שְׁלֵמוֹת, כְּדֵי שֶׁיִּהְיֶה ״לֶחֶם מִשְׁנֶה״ כְּמוֹ בְּכָל שַׁבָּת וְחַג.
💛 גַּם הַ״חֲטִיפָה״ שֶׁל הָאֲפִיקוֹמָן הִיא חֵלֶק מֵהַמִּצְוָה: הִיא עוֹזֶרֶת לָכֶם לְהִשָּׁאֵר עֵרִים עִם הַסִּפּוּר עַד הַסּוֹף.
(פֶּרֶק ט״ז, הֲלָכָה ט״ז)

20הָא לַחְמָא עַנְיָא – וְהַקְּעָרָה נֶעֱלֶמֶת

עוֹרֵךְ הַסֵּדֶר מֵסִיר אֶת הַמַּפִּית, מַגְבִּיהַּ אֶת הַמַּצּוֹת, מַרְאֶה אוֹתָן לַמְּסֻבִּים וְאוֹמֵר ״הָא לַחְמָא עַנְיָא״. אַחַר כָּךְ מְסִירִים אֶת הַקְּעָרָה מֵהַשֻּׁלְחָן – כְּאִלּוּ הַסֵּדֶר נִגְמַר! – וּמוֹזְגִים כּוֹס שְׁנִיָּה. אָז הַיְּלָדִים שׁוֹאֲלִים ״מַה נִּשְׁתַּנָּה״. אַחַר כָּךְ מַחְזִירִים אֶת הַקְּעָרָה, וּמַתְחִילִים לְסַפֵּר. לַיְּלָדִים הַקְּטַנִּים עָדִיף לִמְזֹג אֶת הַכּוֹס הַשְּׁנִיָּה רַק סָמוּךְ לַשְּׁתִיָּה. וְלֹא מַאֲרִיכִים בַּהַגָּדָה עַד שֶׁכֻּלָּם מִתְעַיְּפִים – הַנֹּסַח עַצְמוֹ מַסְפִּיק כְּדֵי לְקַיֵּם אֶת הַמִּצְוָה לַמְּהַדְּרִין.
🤔 לָמָּה? כְּשֶׁמְּסִירִים אֶת הַקְּעָרָה, הַיְּלָדִים תְּמֵהִים: עוֹד לֹא אָכַלְנוּ! וְיֵשׁ נוֹהֲגִים, אִם הַיְּלָדִים לֹא שָׁאֲלוּ, לְהָסִיר עוֹד כֵּלִים – עַד שֶׁיִּשְׁאֲלוּ. וְאֶת הַכּוֹס לַקְּטַנִּים מוֹזְגִים מְאֻחָר, כְּדֵי שֶׁלֹּא תִּשָּׁפֵךְ עַל הַמַּפָּה בְּמֶשֶׁךְ הַהַגָּדָה הָאֲרֻכָּה.
(פֶּרֶק ט״ז, הֲלָכָה י״ז)

21אוֹמְרִים בַּפֶּה – וְכֻלָּם שׁוֹמְעִים

מִי שֶׁרַק מְהַרְהֵר אֶת הַהַגָּדָה בַּלֵּב – לֹא יָצָא, כִּי כָּתוּב ״וְהִגַּדְתָּ״ – צָרִיךְ לְהַגִּיד בַּפֶּה. אֲבָל לֹא כָּל אֶחָד חַיָּב לוֹמַר בְּעַצְמוֹ: מַסְפִּיק שֶׁעוֹרֵךְ הַסֵּדֶר אוֹ מִישֶׁהוּ אַחֵר אוֹמֵר בְּקוֹל, וְכֻלָּם שׁוֹמְעִים. כַּיּוֹם נוֹהֲגִים שֶׁהָעוֹרֵךְ אוֹמֵר בְּקוֹל רָם וְהַשְּׁאָר מִצְטָרְפִים בְּשֶׁקֶט. וְיֵשׁ נוֹהֲגִים שֶׁכָּל קֶטַע קוֹרֵא מִישֶׁהוּ אַחֵר – אֲבָל יוֹצְאִים יְדֵי חוֹבָה רַק בִּקְרִיאָה שֶׁל מִי שֶׁהִגִּיעַ לְגִיל מִצְווֹת. הוֹרִים שֶׁרוֹאִים שֶׁהַיְּלָדִים עוֹמְדִים לְהֵרָדֵם – אוֹמְרִים לָהֶם קֹדֶם אֶת הָעִקָּר: פֶּסַח, מַצָּה וּמָרוֹר, וּמָה הֵם אוֹמְרִים.
🤔 לָמָּה? כָּךְ הִיא דַּרְכּוֹ שֶׁל סִפּוּר: אֶחָד מְסַפֵּר וְהַשְּׁאָר מַקְשִׁיבִים. וּפֶסַח, מַצָּה וּמָרוֹר הֵם הַמִּינִימוּם שֶׁל הַסִּפּוּר – בָּהֶם מִתְבָּאֵר שֶׁהָיִינוּ עֲבָדִים וְה׳ גְּאָלָנוּ.
(פֶּרֶק ט״ז, הֲלָכָה י״ח)

22מְגַלִּים, מְכַסִּים, מַגְבִּיהִים וּמַטִּיפִים

אֶת הַהַגָּדָה אוֹמְרִים בְּלִי הֲסִבָּה, בִּרְצִינוּת – וּבְשִׂמְחָה. כָּל זְמַן הַהַגָּדָה הַמַּצּוֹת מְגֻלּוֹת. אֲבָל כְּשֶׁמַּגְבִּיהִים אֶת הַכּוֹס – בְּ״וְהִיא שֶׁעָמְדָה״, בְּ״לְפִיכָךְ״ וּבְבִרְכַּת הַגְּאֻלָּה – מְכַסִּים אֶת הַמַּצּוֹת. בְּ״מַצָּה זוֹ״ מַגְבִּיהִים אֶת הַמַּצָּה, וּבְ״מָרוֹר זֶה״ – אֶת הַמָּרוֹר. אֲבָל בְּ״פֶסַח זֶה״ לֹא מַגְבִּיהִים אֶת הַזְּרוֹעַ. וּבַאֲמִירַת ״דָּם וָאֵשׁ וְתִימְרוֹת עָשָׁן״, עֶשֶׂר הַמַּכּוֹת וּ״דְּצַ״ךְ עֲדַ״שׁ בְּאַחַ״ב״ מַטִּיפִים מְעַט יַיִן מֵהַכּוֹס – בְּסַךְ הַכֹּל שֵׁשׁ עֶשְׂרֵה פְּעָמִים.
🤔 לָמָּה? הַמַּצָּה הִיא ״לֶחֶם עֹנִי״ – לֶחֶם שֶׁ״עוֹנִים״ עָלָיו דְּבָרִים הַרְבֵּה, וְלָכֵן הִיא מְגֻלָּה. אֲבָל הַמַּצָּה חֲשׁוּבָה מֵהַיַּיִן, וּכְשֶׁמַּגְבִּיהִים אֶת הַכּוֹס – מְכַסִּים אוֹתָהּ, שֶׁלֹּא נֵרָאֶה כְּמִי שֶׁמְּכַבֵּד אֶת הַיַּיִן יוֹתֵר. וְאֶת הַזְּרוֹעַ לֹא מַגְבִּיהִים, כִּי הִיא רַק זֵכֶר – וְשֶׁלֹּא נֵרָאֶה כְּמִי שֶׁמַּקְרִיב קָרְבָּן מִחוּץ לַמִּקְדָּשׁ.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ נוֹהֲגִים לְהַטִּיף אֶת הַיַּיִן בָּאֶצְבַּע.
וְיֵשׁ נוֹהֲגִים לִשְׁפֹּךְ מְעַט מֵהַכּוֹס לִכְלִי שָׁבוּר.
(פֶּרֶק ט״ז, הֲלָכָה י״ט)

23הַלֵּל – שִׁירָה שֶׁל לַיְלָה

בְּלֵיל הַסֵּדֶר אוֹמְרִים הַלֵּל, אֲבָל חוֹצִים אוֹתוֹ: חֶצְיוֹ לִפְנֵי הַסְּעוּדָה וְחֶצְיוֹ אַחֲרֶיהָ. הַמִּנְהָג שֶׁלֹּא לְבָרֵךְ עַל הַהַלֵּל שֶׁל לֵיל הַסֵּדֶר. בְּכָל הַשָּׁנָה אוֹמְרִים הַלֵּל בַּעֲמִידָה, אֲבָל בַּלַּיְלָה הַזֶּה – בִּישִׁיבָה, כִּי הַכֹּל דֶּרֶךְ חֵרוּת. וְנוֹהֲגִים לוֹמַר אוֹתוֹ בְּקוֹל רָם, בְּשִׁירָה וּבִנְעִימָה.
🤔 לָמָּה? בְּכָל הַחַגִּים הַהַלֵּל הוּא שֶׁבַח וְהוֹדָיָה. אֲבָל בַּלַּיְלָה הַזֶּה כָּל יְהוּדִי רוֹאֶה אֶת עַצְמוֹ כְּאִלּוּ הוּא עַצְמוֹ יָצָא מִמִּצְרַיִם – וּמִי שֶׁנִּגְאַל, שָׁר! וְחוֹצִים אוֹתוֹ, כְּדֵי שֶׁהָאֲכִילָה – כְּמוֹ קָרְבַּן הַפֶּסַח פַּעַם – תִּהְיֶה בְּתוֹךְ הַשִּׁירָה, וְכָךְ גַּם שׁוֹתִים אֶת כָּל אַרְבַּע הַכּוֹסוֹת תּוֹךְ כְּדֵי שִׁירָה.
💛 ״וַאֲנַחְנוּ נְבָרֵךְ יָהּ״ – הַלַּיְלָה אֲנִי לֹא רַק אוֹמֵר תּוֹדָה, אֲנִי שָׁר אוֹתָהּ.
(פֶּרֶק ט״ז, הֲלָכָה כ׳)

24בְּרָכָה עַל כָּל כּוֹס?

זוֹ הַמַּחֲלֹקֶת הַמַּעֲשִׂית הָעִקָּרִית בֵּין הָעֵדוֹת בְּלֵיל הַסֵּדֶר: הַאִם מְבָרְכִים ״בּוֹרֵא פְּרִי הַגֶּפֶן״ עַל הַכּוֹס הַשְּׁנִיָּה וְהָרְבִיעִית. וּבְרָכָה אַחֲרוֹנָה – כֻּלָּם שָׁוִים: בִּרְכַּת הַמָּזוֹן פּוֹטֶרֶת אֶת שְׁתֵּי הַכּוֹסוֹת הָרִאשׁוֹנוֹת, וְאַחֲרֵי הַכּוֹס הָרְבִיעִית מְבָרְכִים ״עַל הַגֶּפֶן״.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז (וְתֵימָן, הַנּוֹהֲגִים כָּרַמְבַּ״ם) מְבָרְכִים ״בּוֹרֵא פְּרִי הַגֶּפֶן״ עַל כָּל אַחַת מֵאַרְבַּע הַכּוֹסוֹת, כִּי כָּל כּוֹס הִיא מִצְוָה בִּפְנֵי עַצְמָהּ.
לְמִנְהַג יוֹצְאֵי סְפָרַד מְבָרְכִים רַק עַל הַכּוֹס הָרִאשׁוֹנָה וְעַל הַשְּׁלִישִׁית; הַבְּרָכָה פּוֹטֶרֶת גַּם אֶת הַכּוֹס שֶׁאַחֲרֶיהָ.
(פֶּרֶק ט״ז, הֲלָכָה כ״א)

25מוֹצִיא מַצָּה

נוֹטְלִים יָדַיִם בִּבְרָכָה. עוֹרֵךְ הַסֵּדֶר מַגְבִּיהַּ אֶת שָׁלֹשׁ הַמַּצּוֹת וּמְבָרֵךְ ״הַמּוֹצִיא לֶחֶם מִן הָאָרֶץ״. אַחַר כָּךְ מַשְׁמִיט אֶת הַמַּצָּה הַתַּחְתּוֹנָה מִיָּדוֹ, וּמְבָרֵךְ ״עַל אֲכִילַת מַצָּה״ עַל הָעֶלְיוֹנָה הַשְּׁלֵמָה וְעַל הָאֶמְצָעִית הַשְּׁבוּרָה. לוֹקְחִים כְּזַיִת מִכָּל אַחַת מֵהֶן וְאוֹכְלִים יַחַד, בַּהֲסִבָּה. הָעוֹרֵךְ טוֹעֵם קְצָת מִיָּד אַחֲרֵי הַבְּרָכָה, מְחַלֵּק לְכֻלָּם, וְכֻלָּם אוֹכְלִים מִבִּרְכָתוֹ. עֲדִיף שֶׁכָּל בַּעַל מִשְׁפָּחָה יְחַלֵּק לִבְנֵי מִשְׁפַּחְתּוֹ. הַמַּצָּה צְרִיכָה לִהְיוֹת שְׁמוּרָה, וְאוֹכְלִים בְּכַוָּנָה לְקַיֵּם מִצְוַת הַתּוֹרָה – וְזוֹכְרִים שֶׁהִיא זֵכֶר לַמַּצּוֹת שֶׁאָכְלוּ אֲבוֹתֵינוּ כְּשֶׁיָּצְאוּ לְחֵרוּת.
🤔 לָמָּה? שְׁתֵּי הַמַּצּוֹת הַשְּׁלֵמוֹת הֵן ״לֶחֶם מִשְׁנֶה״ לְ״הַמּוֹצִיא״, וְהַמַּצָּה הַשְּׁבוּרָה – הַ״לֶּחֶם עֹנִי״ – מַתְאִימָה לִבְרָכַת ״עַל אֲכִילַת מַצָּה״.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי סְפָרַד הָעוֹרֵךְ טוֹבֵל אֶת הַמַּצָּה בְּמֶלַח, כְּמוֹ בְּכָל הַשָּׁנָה.
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז לֹא טוֹבְלִים אֶת הַמַּצָּה בְּמֶלַח, כְּדֵי שֶׁתֵּרָאֶה יוֹתֵר ״לֶחֶם עֹנִי״.
💛 זֶה הָרֶגַע שֶׁל מִצְוָה מִן הַתּוֹרָה: בְּכָל נְגִיסָה אֲנִי אוֹכֵל אֶת הַטַּעַם שֶׁל הַחֵרוּת שֶׁה׳ נָתַן לָנוּ.
(פֶּרֶק ט״ז, הֲלָכָה כ״ב, הֲלָכָה כ״ו)

26כַּמָּה זֶה כְּזַיִת?

מִן הַתּוֹרָה צָרִיךְ לֶאֱכֹל כְּזַיִת מַצָּה, וַחֲכָמִים הוֹסִיפוּ: כְּזַיִת אֶחָד מֵהָעֶלְיוֹנָה וּכְזַיִת אֶחָד מֵהָאֶמְצָעִית בַּאֲכִילָה הָרִאשׁוֹנָה, עוֹד כְּזַיִת בַּכּוֹרֵךְ, וְעוֹד כְּזַיִת בָּאֲפִיקוֹמָן. כְּזַיִת מַצָּה הוּא בְּעֵרֶךְ שְׁלִישׁ מַצַּת מְכוֹנָה (וּכְגֹדֶל זֶה בְּמַצַּת יָד). לָכֵן מִיָּד אַחֲרֵי הַבְּרָכוֹת אוֹכְלִים כִּשְׁנֵי שְׁלִישִׁים שֶׁל מַצָּה, וּבַכּוֹרֵךְ וּבָאֲפִיקוֹמָן – כִּשְׁלִישׁ בְּכָל פַּעַם. בִּשְׁנֵי הַשְּׁלִישִׁים שֶׁאוֹכְלִים בַּהַתְחָלָה יֵשׁ כְּזַיִת גַּם לְפִי הַשִּׁעוּר הַגָּדוֹל שֶׁל הַ״חֲזוֹן אִישׁ״. וּמִי שֶׁקָּשֶׁה לוֹ – אוֹכֵל שְׁלִישׁ מַצָּה וּמְבָרֵךְ עָלָיו.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְפִי נֶפַח – כָּךְ מוֹרֶה הָרַב מֶלָמֵד לְכָל הָעֵדוֹת כְּזַיִת הוּא כִּשְׁלִישׁ מַצָּה. כָּךְ מֵעִקַּר הַדִּין, וְכָךְ הוֹרוּ גַּם גְּדוֹלֵי רַבָּנִים סְפָרַדִּים, כְּמוֹ הָרַב בֶּן צִיּוֹן אַבָּא שָׁאוּל וְהָרַב שָׁלוֹם מְשַׁאשׁ.
לְמִנְהַג רַבִּים מִיּוֹצְאֵי סְפָרַד – לְפִי מִשְׁקָל כְּזַיִת הוּא כְּ־25 גְּרָם – קָרוֹב לְמַצָּה שְׁלֵמָה – וְאוֹכְלִים בַּסֵּדֶר אַרְבַּע מַצּוֹת: שְׁתַּיִם אַחֲרֵי הַבְּרָכוֹת, אַחַת לַכּוֹרֵךְ וְאַחַת לָאֲפִיקוֹמָן.
(פֶּרֶק ט״ז, הֲלָכָה כ״ב, הֲלָכָה כ״ג, הֲלָכָה כ״ד)

27אוֹכְלִים בִּרְצִיפוּת – בְּלִי שָׁעוֹן

אֶת הַכְּזַיִת צָרִיךְ לֶאֱכֹל בִּרְצִיפוּת, בְּתוֹךְ ״כְּדֵי אֲכִילַת פְּרָס״ – כַּמָּה דַּקּוֹת. מִי שֶׁאוֹכֵל חֲצִי כְּזַיִת, וְאַחֲרֵי עֶשֶׂר דַּקּוֹת עוֹד חֲצִי – כְּאִלּוּ אָכַל רַק חֲצִי כְּזַיִת. אֲבָל אֵין צֹרֶךְ לְהִסְתַּכֵּל בַּשָּׁעוֹן: כָּל מִי שֶׁאוֹכֵל בְּנַחַת, בְּלִי לְהַפְסִיק בִּדְבָרִים אֲחֵרִים – וַדַּאי יָצָא, אֲפִלּוּ אִם הוּא אוֹכֵל לְאַט.
🤔 לָמָּה? ״פְּרָס״ פֵּרוּשׁוֹ חֲצִי – חֲצִי כִּכָּר, כַּמָּה שֶׁאָדָם הָיָה אוֹכֵל בְּאַרוּחָה אַחַת. זֶה פִּי שְׁמוֹנָה אוֹ תֵּשַׁע מִכְּזַיִת – וְלָכֵן מִי שֶׁאוֹכֵל כְּזַיִת בְּלִי הַפְסָקָה, בָּטוּחַ מַסְפִּיק בַּזְּמַן.
(פֶּרֶק ט״ז, הֲלָכָה כ״ה)

28מָרוֹר

טוֹבְלִים אֶת הַמָּרוֹר בַּחֲרֹסֶת – רַק נְגִיעָה קְטַנָּה – וּמְנַעֲרִים מִמֶּנּוּ אֶת הַחֲרֹסֶת שֶׁנִּדְבְּקָה. מְבָרְכִים ״עַל אֲכִילַת מָרוֹר״, וְאוֹכְלִים כְּזַיִת, בְּלִי הֲסִבָּה. הַמִּין הַמְּהֻדָּר בְּיוֹתֵר הוּא חַסָּה, וְאֶפְשָׁר גַּם חֲזֶרֶת (חֲרֵיין). וְיֵשׁ שֶׁמְּעָרְבִים עִם הַחַסָּה מְעַט חֲרֵיין, כְּדֵי לְהַרְגִּישׁ קְצָת מְרִירוּת. בְּעֶרֶב פֶּסַח בּוֹדְקִים וּמְנַקִּים הֵיטֵב אֶת הַחַסָּה מִתּוֹלָעִים, וְטוֹב לִקְנוֹת חַסָּה שֶׁגָּדְלָה בְּלִי תּוֹלָעִים.
🤔 לָמָּה? הַמָּרוֹר זֵכֶר לַשִּׁעְבּוּד הַמַּר, וְהַחֲרֹסֶת ״מְבַטֶּלֶת״ אֶת הַחֲרִיפוּת – אֲבָל לֹא אוֹכְלִים מָרוֹר מָתוֹק. וְלָמָּה חַסָּה? שְׁמָהּ רוֹמֵז שֶׁה׳ ״חָס״ עָלֵינוּ. וְהִיא רַכָּה וּמְתוּקָה בַּהַתְחָלָה, וּבַסּוֹף נַעֲשֵׂית קָשָׁה וּמָרָה – כְּמוֹ הַשִּׁעְבּוּד בְּמִצְרַיִם: הִתְחִיל בְּ״פֶה רַךְ״ וּבְתַשְׁלוּם, וְנַעֲשָׂה קָשֶׁה וּמַר.
💛 גַּם בְּתוֹךְ הַמָּרוֹר מֻסְתֶּרֶת ״חַסָּה״ – ה׳ חָס עָלֵינוּ גַּם בַּזְּמַנִּים הַקָּשִׁים.
(פֶּרֶק ט״ז, הֲלָכָה כ״ז)

29כּוֹרֵךְ

כּוֹרְכִים כְּזַיִת מָרוֹר בְּתוֹךְ כְּזַיִת מַצָּה, טוֹבְלִים בַּחֲרֹסֶת, אוֹמְרִים ״זֵכֶר לַמִּקְדָּשׁ כְּהִלֵּל...״, וְאוֹכְלִים בַּהֲסִבָּה. מֵהַבְּרָכָה עַל הַמַּצָּה וְעַד סוֹף הַכּוֹרֵךְ נִזְהָרִים שֶׁלֹּא לְדַבֵּר, חוּץ מִדְּבָרִים שֶׁל הַמִּצְוָה.
🤔 לָמָּה? בִּזְמַן בֵּית הַמִּקְדָּשׁ, הִלֵּל הַזָּקֵן אָכַל אֶת הַמַּצָּה וְהַמָּרוֹר יַחַד, כְּמוֹ שֶׁכָּתוּב: ״עַל מַצּוֹת וּמְרֹרִים יֹאכְלֻהוּ״, וַחֲכָמִים אָכְלוּ כָּל אֶחָד לְחוּד. הַיּוֹם הַמַּצָּה מִן הַתּוֹרָה וְהַמָּרוֹר מִדִּבְרֵי חֲכָמִים, וְאִם נֹאכַל אוֹתָם יַחַד, הַמָּרוֹר ״יִפְגַּע״ בַּמַּצָּה. לָכֵן קֹדֶם מַצָּה לְבַד, אַחַר כָּךְ מָרוֹר לְבַד – וּבַסּוֹף כּוֹרְכִים, זֵכֶר לְהִלֵּל. כָּךְ מְקַיְּמִים גַּם אֶת דַּעַת חֲכָמִים וְגַם אֶת דַּעַת הִלֵּל.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ נוֹהֲגִים לְנַעֵר גַּם בַּכּוֹרֵךְ אֶת הַחֲרֹסֶת מֵהַמָּרוֹר.
וְיֵשׁ נוֹהֲגִים לֹא לְהָסִיר אֶת הַחֲרֹסֶת מֵהַמָּרוֹר שֶׁל הַכּוֹרֵךְ.
(פֶּרֶק ט״ז, הֲלָכָה כ״ח)

30מִי שֶׁקָּשֶׁה לוֹ לֶאֱכֹל

מִי שֶׁקָּשֶׁה לוֹ לֶאֱכֹל אֶת כָּל הַשִּׁעוּרִים, יִשְׁתַּדֵּל לֶאֱכֹל בַּהַתְחָלָה שְׁנֵי שְׁלִישֵׁי מַצָּה, וּבַכּוֹרֵךְ וּבָאֲפִיקוֹמָן יָכוֹל לְהִסְתַּפֵּק בַּחֲמִישִׁית מַצָּה. וְאִם גַּם זֶה קָשֶׁה – יֹאכַל לְפָחוֹת שְׁלִישׁ מַצָּה לַמִּצְוָה, וִיבָרֵךְ. מִי שֶׁקָּשֶׁה לוֹ לִלְעֹס – יָכוֹל לְפוֹרֵר אֶת הַמַּצָּה, וְאִם צָרִיךְ – לְהַשְׁרוֹת אוֹתָהּ בְּמַיִם. אֲבָל מַצָּה מְבֻשֶּׁלֶת, אוֹ שֶׁהֻשְׁרְתָה עַד שֶׁנִּמּוֹחָה – אֵין בָּהּ טַעַם מַצָּה, וְאֵין יוֹצְאִים בָּהּ. וְחוֹלֵי צֶלְיָאק: רֻבָּם חַיָּבִים לֶאֱכֹל כְּזַיִת מַצָּה, וּמִי שֶׁהַמַּצָּה גּוֹרֶמֶת לוֹ תְּגוּבָה קָשָׁה – פָּטוּר. כַּיּוֹם יֵשׁ מַצּוֹת מִשִּׁבֹּלֶת שׁוּעָל, שֶׁטּוֹבוֹת יוֹתֵר לְחוֹלֵי צֶלְיָאק.
💛 ה׳ לֹא מְבַקֵּשׁ מִמֶּנּוּ יוֹתֵר מִמָּה שֶׁאֲנַחְנוּ יְכוֹלִים – כָּל אֶחָד מְקַיֵּם אֶת הַמִּצְוָה כְּפִי כֹּחוֹ.
(פֶּרֶק ט״ז, הֲלָכָה כ״ט, הֲלָכָה ל׳)

31עַד חֲצוֹת

אֶת הַמַּצָּה וְאֶת הַמָּרוֹר אוֹכְלִים לִפְנֵי חֲצוֹת הַלַּיְלָה, וּלְכַתְּחִלָּה גַּם אֶת הָאֲפִיקוֹמָן, וּמְסַיְּמִים גַּם אֶת הַהַלֵּל לִפְנֵי חֲצוֹת. מִי שֶׁלֹּא הִסְפִּיק לֶאֱכֹל מַצָּה וּמָרוֹר לִפְנֵי חֲצוֹת – יֹאכַל אַחֲרֵי חֲצוֹת, אֲבָל בְּלִי הַבְּרָכוֹת ״עַל אֲכִילַת מַצָּה״ וְ״עַל אֲכִילַת מָרוֹר״.
🤔 לָמָּה? כְּדֵי לָצֵאת לְכָל הַדֵּעוֹת, מַקְדִּימִים לִפְנֵי חֲצוֹת. וּמִי שֶׁאוֹכֵל אַחַר כָּךְ – לֹא מְבָרֵךְ, כְּדֵי שֶׁלֹּא לְהִכָּנֵס לְסָפֵק שֶׁל בְּרָכָה לְבַטָּלָה.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת רַבִּי אֶלְעָזָר בֶּן עֲזַרְיָה אֶת קָרְבַּן הַפֶּסַח – וְאִתּוֹ אֶת הַמַּצָּה וְהַמָּרוֹר – אוֹכְלִים עַד חֲצוֹת הַלַּיְלָה, כִּי בַּחֲצוֹת הֻכּוּ בְּכוֹרֵי מִצְרַיִם.
לְדַעַת רַבִּי עֲקִיבָא אוֹכְלִים כָּל הַלַּיְלָה, עַד עֲלוֹת הַשַּׁחַר – כִּי בַּבֹּקֶר יָצָאנוּ לְחֵרוּת.
(פֶּרֶק ט״ז, הֲלָכָה ל״א)

32שֻׁלְחָן עוֹרֵךְ

מְכִינִים סְעוּדָה מְכֻבֶּדֶת, בְּכֵלִים יָפִים וּבְמַאֲכָלִים טְעִימִים. מֻתָּר לִשְׁתּוֹת יַיִן בַּסְּעוּדָה, וְזֶה לֹא נֶחְשָׁב תּוֹסֶפֶת עַל אַרְבַּע הַכּוֹסוֹת. הַנּוֹהֲגִים שֶׁלֹּא לֶאֱכֹל צָלִי – לֹא אוֹכְלִים שׁוּם בָּשָׂר צָלוּי, גַּם לֹא עוֹף, וְגַם לֹא בָּשָׂר שֶׁבִּשְּׁלוּ וְאַחַר כָּךְ צָלוּ. אֲבָל דָּגִים וּבֵיצִים צְלוּיִים – מֻתָּרִים, כִּי אֵינָם צְרִיכִים שְׁחִיטָה. יֵשׁ נוֹהֲגִים לֶאֱכֹל בַּסְּעוּדָה בֵּיצִים, וְהַמִּנְהָג הָרוֹוֵחַ לֶאֱכֹל אֶת הַבֵּיצָה שֶׁבַּקְּעָרָה. וְלֹא מַרְבִּים בַּאֲכִילָה – כְּדֵי שֶׁיִּהְיֶה כֹּחַ לֶאֱכֹל אֶת הָאֲפִיקוֹמָן, לְהַמְשִׁיךְ בַּהַלֵּל וּלְהוֹסִיף לְסַפֵּר.
🤔 לָמָּה? הַבֵּיצִים – יֵשׁ אוֹמְרִים שֶׁהֵן סִימַן אֲבֵלוּת עַל הַחֻרְבָּן, וְהַגְּרָ״א בֵּאֵר שֶׁהֵן זֵכֶר לְקָרְבַּן חֲגִיגָה שֶׁהָיוּ אוֹכְלִים לִפְנֵי הַפֶּסַח.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג כָּל עוֹלֵי אַשְׁכְּנַז וְרֹב עוֹלֵי סְפָרַד לֹא אוֹכְלִים בָּשָׂר צָלוּי בְּלֵיל הַסֵּדֶר, שֶׁלֹּא יֵרָאֶה כְּקָרְבַּן פֶּסַח.
לְמִנְהַג עוֹלֵי תֵּימָן אוֹכְלִים צָלִי בְּלֵיל הַסֵּדֶר.
(פֶּרֶק ט״ז, הֲלָכָה ל״ב)

33צָפוּן – הָאֲפִיקוֹמָן

בְּסוֹף הַסְּעוּדָה אוֹכְלִים כְּזַיִת מֵהַמַּצָּה שֶׁהֻחְבְּאָה – הָ״אֲפִיקוֹמָן״ – זֵכֶר לְקָרְבַּן הַפֶּסַח. אוֹכְלִים אוֹתוֹ בַּהֲסִבָּה, ״עַל הַשֹּׂבַע״, בְּמָקוֹם אֶחָד, וּלְכַתְּחִלָּה לִפְנֵי חֲצוֹת. אִם אֵין בָּאֲפִיקוֹמָן מַסְפִּיק לְכֻלָּם, אוֹ שֶׁהוּא אָבַד – לוֹקְחִים מַצָּה שְׁמוּרָה אַחֶרֶת. אַחֲרֵי הָאֲפִיקוֹמָן לֹא אוֹכְלִים כְּלוּם עַד הַשֵּׁנָה, אֲבָל מַיִם מֻתָּר לִשְׁתּוֹת. וְיֵשׁ נוֹהֲגִים לֶאֱכֹל שְׁנֵי זֵיתִים לָאֲפִיקוֹמָן, אֲבָל זֶה לֹא חוֹבָה.
🤔 לָמָּה? ״עַל הַשֹּׂבַע״ – כְּשֶׁכְּבָר שְׂבֵעִים, אֲבָל עוֹד רוֹצִים לֶאֱכֹל. מִי שֶׁמַּכְרִיחַ אֶת עַצְמוֹ לֶאֱכֹל כְּשֶׁהוּא שָׂבֵעַ לְגַמְרֵי – זוֹ לֹא אֲכִילָה שֶׁל מִצְוָה. וְלֹא אוֹכְלִים אַחֲרָיו, כְּדֵי שֶׁטַּעַם הַמַּצָּה יִשָּׁאֵר בַּפֶּה, כְּמוֹ שֶׁהָיָה עִם קָרְבַּן הַפֶּסַח. וּבְמָקוֹם אֶחָד – כִּי עַל הַפֶּסַח נֶאֱמַר: ״בְּבַיִת אֶחָד יֵאָכֵל״.
💛 הַטַּעַם הָאַחֲרוֹן שֶׁל הַסְּעוּדָה הוּא טַעַם שֶׁל מִצְוָה – וְהוּא נִשְׁאָר אִתִּי עַד שֶׁאֲנִי נִרְדָּם.
(פֶּרֶק ט״ז, הֲלָכָה ל״ג, הֲלָכָה ל״ד)

34בָּרֵךְ – וְכוֹס שֶׁל אֵלִיָּהוּ

אַחֲרֵי הָאֲפִיקוֹמָן אוֹמְרִים בִּרְכַּת הַמָּזוֹן עַל הַכּוֹס הַשְּׁלִישִׁית, וְשׁוֹתִים אוֹתָהּ. אַחַר כָּךְ מוֹזְגִים לְכֻלָּם אֶת הַכּוֹס הָרְבִיעִית, וּמוֹזְגִים עוֹד כּוֹס – ״כּוֹס שֶׁל אֵלִיָּהוּ״. אוֹתָהּ לֹא שׁוֹתִים: שׁוֹמְרִים אוֹתָהּ מְכֻסָּה עַד הַבֹּקֶר, מַחְזִירִים אֶת הַיַּיִן לַבַּקְבּוּק, וּמִמֶּנּוּ מְקַדְּשִׁים בַּיּוֹם.
🤔 לָמָּה? אַחֲרֵי אַרְבַּע לְשׁוֹנוֹת הַגְּאֻלָּה יֵשׁ לָשׁוֹן חֲמִישִׁית: ״וְהֵבֵאתִי אֶתְכֶם אֶל הָאָרֶץ״ – הַגְּאֻלָּה הַשְּׁלֵמָה. נֶחְלְקוּ אִם לִשְׁתּוֹת כּוֹס חֲמִישִׁית, וְהַגְּרָ״א בֵּאֵר: עַל כָּל סָפֵק אוֹמְרִים שֶׁאֵלִיָּהוּ הַנָּבִיא יָבוֹא וְיַכְרִיעַ. לָכֵן מוֹזְגִים אֶת הַכּוֹס לִכְבוֹדוֹ, וּכְשֶׁיָּבוֹא – יַגִּיד לָנוּ אִם לִשְׁתּוֹת אוֹתָהּ.
💛 הַכּוֹס הַמְּלֵאָה שֶׁל אֵלִיָּהוּ אוֹמֶרֶת: אֲנַחְנוּ מְחַכִּים לַגְּאֻלָּה הַשְּׁלֵמָה – וּמַאֲמִינִים שֶׁהִיא בַּדֶּרֶךְ.
(פֶּרֶק ט״ז, הֲלָכָה ל״ה, הֲלָכָה ל״ו)

35״שְׁפֹךְ חֲמָתְךָ״ – פּוֹתְחִים אֶת הַדֶּלֶת

לִפְנֵי הַהַלֵּל אוֹמְרִים ״שְׁפֹךְ חֲמָתְךָ אֶל הַגּוֹיִם אֲשֶׁר לֹא יְדָעוּךָ...״. יֵשׁ נוֹהֲגִים לִפְתֹּחַ אָז אֶת הַדֶּלֶת, וְיֵשׁ נוֹהֲגִים לַעֲמֹד. אַחֲרֵי ״שְׁפֹךְ חֲמָתְךָ״ סוֹגְרִים אֶת הַדֶּלֶת.
🤔 לָמָּה? פּוֹתְחִים אֶת הַדֶּלֶת כְּדֵי לְהַרְאוֹת שֶׁהַלַּיְלָה הַזֶּה ״מְשֻׁמָּר״ מִן הַמַּזִּיקִים, וְאֵינֶנּוּ מְפַחֲדִים מִשּׂוֹנְאֵינוּ. וּבִזְכוּת הָאֱמוּנָה הַזֹּאת יָבוֹא הַמָּשִׁיחַ.
💛 דֶּלֶת פְּתוּחָה בְּאֶמְצַע הַלַּיְלָה – כִּי אֲנִי סוֹמֵךְ עַל ה׳ שֶׁשּׁוֹמֵר עָלַי.
(פֶּרֶק ט״ז, הֲלָכָה ל״ה)

36הַלֵּל, הַלֵּל הַגָּדוֹל – וְנִרְצָה

מַמְשִׁיכִים בַּחֲצִי הַשֵּׁנִי שֶׁל הַהַלֵּל. אֶת הַפְּסוּקִים ״הוֹדוּ לַה׳ כִּי טוֹב״, ״יֹאמַר נָא יִשְׂרָאֵל״, ״יֹאמְרוּ נָא בֵית אַהֲרֹן״, ״יֹאמְרוּ נָא יִרְאֵי ה׳״, ״אָנָּא ה׳ הוֹשִׁיעָה נָּא״ וְ״אָנָּא ה׳ הַצְלִיחָה נָּא״ – גְּדוֹל הַבַּיִת אוֹמֵר, וְכֻלָּם עוֹנִים אַחֲרָיו. לְכַתְּחִלָּה טוֹב שֶׁיִּהְיוּ בַּסֵּדֶר שְׁלֹשָׁה מְבֻגָּרִים לְכָךְ. אַחַר כָּךְ אוֹמְרִים הַלֵּל הַגָּדוֹל (תְּהִלִּים קל״ו) וְ״נִשְׁמַת״, וְחוֹתְמִים בִּבְרָכָה. אָז שׁוֹתִים אֶת הַכּוֹס הָרְבִיעִית – וְיֵשׁ שֶׁשּׁוֹתִים אוֹתָהּ רַק אַחֲרֵי כַּמָּה פִּיּוּטִים; כָּל מִשְׁפָּחָה כְּמִנְהָגָהּ – וּמְבָרְכִים ״עַל הַגֶּפֶן״. וְאַחַר כָּךְ שָׁרִים פִּיּוּטִים, עַד ״נִרְצָה״.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי סְפָרַד חוֹתְמִים בַּבְּרָכָה ״יְהַלְלוּךָ״, שֶׁבְּסוֹף הַהַלֵּל.
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז חוֹתְמִים בַּבְּרָכָה ״יִשְׁתַּבַּח״, שֶׁבְּסוֹף פְּסוּקֵי דְּזִמְרָה.
💛 ״הוֹדוּ לַה׳ כִּי טוֹב, כִּי לְעוֹלָם חַסְדּוֹ״ – כָּל הַמִּשְׁפָּחָה עוֹנָה יַחַד, כְּמוֹ מַקְהֵלָה שֶׁל עַם שָׁלֵם.
(פֶּרֶק ט״ז, הֲלָכָה ל״ה)

37אַחֲרֵי הַסֵּדֶר – מָה שׁוֹתִים?

אַחֲרֵי הַסֵּדֶר לֹא שׁוֹתִים יַיִן. עַל קָפֶה אוֹ מִיץ – נֶחְלְקוּ. לְמַעֲשֶׂה, מִי שֶׁרוֹצֶה – רַשַּׁאי לִשְׁתּוֹת קָפֶה אוֹ מִיץ, וּלְכַתְּחִלָּה טוֹב לִשְׁתּוֹת רַק מַיִם. וּמִי שֶׁרוֹצֶה קָפֶה כְּדֵי שֶׁיּוּכַל לְהַמְשִׁיךְ לִלְמֹד הִלְכוֹת פֶּסַח וּלְסַפֵּר בִּיצִיאַת מִצְרַיִם – לְכַתְּחִלָּה יִשְׁתֶּה קָפֶה.
🤔 לָמָּה? יַיִן לֹא שׁוֹתִים, כְּדֵי שֶׁנּוּכַל לְהַמְשִׁיךְ לַעֲסֹק בְּסִפּוּר יְצִיאַת מִצְרַיִם עַד שֶׁהַשֵּׁנָה תַּחְטֹף אוֹתָנוּ – וְשֶׁלֹּא יֵרָאֶה כְּמוֹסִיף עַל אַרְבַּע הַכּוֹסוֹת.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת רֹב הַפּוֹסְקִים מֻתָּר לִשְׁתּוֹת קָפֶה אוֹ מִיץ, כִּי רַק אֹכֶל מְבַטֵּל אֶת טַעַם הָאֲפִיקוֹמָן.
לְדַעַת כַּמָּה פּוֹסְקִים שׁוֹתִים רַק מַיִם, כִּי גַּם מַשְׁקֶה בַּעַל טַעַם מְבַטֵּל אֶת טַעַם הַמַּצָּה.
(פֶּרֶק ט״ז, הֲלָכָה ל״ז)

38מְסַפְּרִים עַד שֶׁהַשֵּׁנָה חוֹטֶפֶת

גַּם אַחֲרֵי שֶׁסִּיַּמְנוּ אֶת הַהַגָּדָה, מִצְוָה לְהַרְבּוֹת כָּל הַלַּיְלָה בְּסִפּוּר יְצִיאַת מִצְרַיִם וּבַנִּסִּים שֶׁעָשָׂה ה׳ לַאֲבוֹתֵינוּ – עַד שֶׁהַשֵּׁנָה חוֹטֶפֶת. גַּם לִמּוּד הִלְכוֹת פֶּסַח הוּא חֵלֶק מֵהַמִּצְוָה. וּמִי שֶׁחוֹשֵׁשׁ שֶׁלֹּא יוּכַל לְהִתְפַּלֵּל כָּרָאוּי בַּבֹּקֶר – יִשְׁתַּדֵּל לְהַמְשִׁיךְ לְפָחוֹת עַד אַחֲרֵי חֲצוֹת. יֵשׁ נוֹהֲגִים לִקְרֹא אַחֲרֵי הַהַגָּדָה אֶת שִׁיר הַשִּׁירִים. וְלִפְנֵי הַשֵּׁנָה אוֹמְרִים רַק קְרִיאַת שְׁמַע וּבִרְכַּת ״הַמַּפִּיל״.
🤔 לָמָּה? שִׁיר הַשִּׁירִים רוֹמֵז לָאַהֲבָה שֶׁבֵּין הַקָּדוֹשׁ בָּרוּךְ הוּא וְיִשְׂרָאֵל. וּבִשְׁאָר הַלֵּילוֹת מוֹסִיפִים פְּסוּקִים לִשְׁמִירָה מִן הַמַּזִּיקִים – אֲבָל הַלַּיְלָה הַזֶּה הוּא ״לֵיל שִׁמֻּרִים״, שָׁמוּר מִן הַמַּזִּיקִים וּמוּכָן לַגְּאֻלָּה.
💛 ״לֵיל שִׁמֻּרִים״ – אֲנִי הוֹלֵךְ לִישֹׁן בַּלַּיְלָה הַזֶּה כְּשֶׁאֲנִי יוֹדֵעַ שֶׁה׳ שׁוֹמֵר עָלַי.
(פֶּרֶק ט״ז, הֲלָכָה ל״ח)

📝 מָה לָמַדְנוּ

  • בְּלֵיל הַסֵּדֶר מְקַיְּמִים מִצְווֹת מִן הַתּוֹרָה וּמִדִּבְרֵי חֲכָמִים: מַצָּה, מָרוֹר, אַרְבַּע כּוֹסוֹת, הֲסִבָּה, סִפּוּר וְהַלֵּל – וְגַם נָשִׁים וּבָנוֹת חַיָּבוֹת בָּהֶן.
  • הַכֹּל בָּנוּי כְּדֵי שֶׁהַיְּלָדִים יִשְׁאֲלוּ: אֱגוֹזִים, טִבּוּל פַּעֲמַיִם, הֲסָרַת הַקְּעָרָה וּ״מַה נִּשְׁתַּנָּה״.
  • אוֹכְלִים כְּזַיִת מַצָּה (כִּשְׁלִישׁ מַצָּה) בַּהֲסִבָּה לִשְׂמֹאל, וּמָרוֹר טָבוּל בַּחֲרֹסֶת – לִפְנֵי חֲצוֹת.
  • יֵשׁ מִנְהָגִים שׁוֹנִים – בְּרָכָה עַל הַכּוֹסוֹת, כַּרְפַּס, מֶלַח, זְרוֹעַ וְשִׁעוּר הַכְּזַיִת – וְכֻלָּם כַּהֲלָכָה.
  • מְסַיְּמִים בָּאֲפִיקוֹמָן, בְּהַלֵּל בְּשִׁירָה וּבְכוֹס שֶׁל אֵלִיָּהוּ – וּמַמְשִׁיכִים לְסַפֵּר עַד שֶׁהַשֵּׁנָה חוֹטֶפֶת.

❓ שְׁאֵלוֹת חֲזָרָה

  1. בְּאֵילוּ מִצְווֹת חוֹבָה לְהָסֵב, וּבְאֵילוּ לֹא מְסִבִּים? לְאֵיזֶה צַד מְסִבִּים, וְלָמָּה?
  2. לָמָּה מְסִירִים אֶת הַקְּעָרָה מֵהַשֻּׁלְחָן לִפְנֵי ״מַה נִּשְׁתַּנָּה״, וְלָמָּה מְכַסִּים אֶת הַמַּצּוֹת כְּשֶׁמַּגְבִּיהִים אֶת הַכּוֹס?
  3. מָה הַהֶבְדֵּל בֵּין מִנְהַג יוֹצְאֵי סְפָרַד לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז בִּבְרָכָה עַל אַרְבַּע הַכּוֹסוֹת?

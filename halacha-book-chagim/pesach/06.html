סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת פֶּסַח
מַהֲדוּרַת יְלָדִים

📝 מְכִירַת חָמֵץ

פֶּרֶק ו׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
בַּמִּטְבָּח יֵשׁ אָרוֹן סָגוּר בִּנְיַר דֶּבֶק, וְעָלָיו כָּתוּב: ״חָמֵץ שֶׁנִּמְכַּר״. רֶגַע... מָכַרְנוּ אֶת הֶחָמֵץ? לְמִי? וְאִם הוּא נִמְכַּר – לָמָּה הוּא עֲדַיִן אֶצְלֵנוּ בַּבַּיִת? וּמָתַי מֻתָּר לִפְתֹּחַ אֶת הָאָרוֹן שׁוּב? בַּפֶּרֶק הַזֶּה נִלְמַד עַל מְכִירַת חָמֵץ לְגוֹי – מְכִירָה אֲמִתִּית, שֶׁכְּדַאי לְהָבִין אוֹתָהּ.

1לֹא רוֹצִים לְהַפְסִיד

עַד חֲצוֹת יוֹם י״ד צָרִיךְ לְהַשְׁבִּית אֶת כָּל הֶחָמֵץ. פַּעַם הָיוּ יְהוּדִים מְתַכְנְנִים אֶת הַקְּנִיּוֹת כָּךְ שֶׁעַד פֶּסַח יִגְמְרוּ אֶת הֶחָמֵץ, וְהִשְׁאִירוּ רַק מְעַט לִשְׂרֵפָה. מִי שֶׁנִּשְׁאַר לוֹ הַרְבֵּה חָמֵץ יָכוֹל לִשְׂרֹף אוֹתוֹ, לָתֵת אוֹתוֹ בְּמַתָּנָה לְגוֹי הָגוּן – אוֹ לִמְכֹּר אוֹתוֹ לְגוֹי לִפְנֵי פֶּסַח, וְלִקְנוֹת אוֹתוֹ בַּחֲזָרָה אַחֲרֵי הֶחָג.
🤔 לָמָּה? כָּל עוֹד לֹא הִגִּיעָה הַשָּׁעָה הַשִּׁשִּׁית שֶׁל עֶרֶב פֶּסַח, מֻתָּר לֵהָנוֹת מֵהֶחָמֵץ – וְגַם לִמְכֹּר אוֹתוֹ בִּמְחִיר מָלֵא.
(פֶּרֶק ו׳, הֲלָכָה א׳)

2אֵיךְ הִתְחִיל הַמִּנְהָג?

לִפְנֵי כְּאַרְבַּע מֵאוֹת שָׁנָה, יְהוּדִים רַבִּים בְּאֵירוֹפָּה הִתְפַּרְנְסוּ מִמַּשְׁקֶה שֶׁעָשׂוּ מִשְּׂעוֹרִים וְחִטִּים – חָמֵץ גָּמוּר. לֹא הָיָה אֶפְשָׁר לְאַבֵּד כָּל שָׁנָה סְחוֹרָה כָּל כָּךְ יְקָרָה, וְלָכֵן מָכְרוּ אוֹתָהּ לְגוֹי לִפְנֵי פֶּסַח, וְקָנוּ בַּחֲזָרָה אַחֲרָיו. גְּדוֹלֵי יִשְׂרָאֵל רָאוּ שֶׁלִּפְעָמִים הַמְּכִירָה נַעֲשֵׂית שֶׁלֹּא כַּדִּין, וְלָכֵן הָרַבָּנִים הִתְחִילוּ לְסַדֵּר אוֹתָהּ בְּעַצְמָם.
🤔 לָמָּה? אִם הַמְּכִירָה לֹא נַעֲשֵׂית כַּדִּין – הֶחָמֵץ נִשְׁאָר שֶׁל הַיְּהוּדִי. הוּא עוֹבֵר עָלָיו עַל ״לֹא יֵרָאֶה״ וְ״לֹא יִמָּצֵא״, וְגַם אַחֲרֵי פֶּסַח הֶחָמֵץ אָסוּר. כְּשֶׁהָרַבָּנִים מְסַדְּרִים – הַכֹּל נַעֲשֶׂה כַּהֲלָכָה, וְכָךְ הִצְטָרְפוּ לַמְּכִירָה עוֹד וְעוֹד יְהוּדִים.
(פֶּרֶק ו׳, הֲלָכָה ב׳)

3מְכִירָה אֲמִתִּית

כַּמָּה מִגְּדוֹלֵי יִשְׂרָאֵל טָעֲנוּ שֶׁהַמְּכִירָה הִיא רַק ״הַעֲרָמָה״, כִּי בָּרוּר שֶׁהֶחָמֵץ יַחֲזֹר לַיְּהוּדִי. אֲבָל דַּעַת הַפּוֹסְקִים, כִּמְעַט כֻּלָּם, שֶׁסּוֹמְכִים עַל הַמְּכִירָה: הִיא מְכִירָה מִשְׁפָּטִית אֲמִתִּית, וְאַחֲרֵי פֶּסַח הַגּוֹי רַשַּׁאי לְהַחְלִיט שֶׁהוּא לֹא מוֹכֵר אֶת הֶחָמֵץ בַּחֲזָרָה. לָכֵן מְדַקְדְּקִים בְּכָל הַפְּרָטִים: עוֹשִׂים אוֹתָהּ בְּכָל סוּגֵי הַקִּנְיָנִים, גַּם לְפִי חֻקֵּי הַמְּדִינָה, וּמַשְׂכִּירִים לַגּוֹי אֶת הַמָּקוֹם שֶׁבּוֹ מֻנָּח הֶחָמֵץ.
🤔 לָמָּה? כְּשֶׁהַגּוֹי שׂוֹכֵר אֶת הַמָּקוֹם, הֶחָמֵץ נִמְצָא בִּרְשׁוּתוֹ – כְּמוֹ בְּכָל מְכִירָה רְגִילָה, שֶׁהַקּוֹנֶה לוֹקֵחַ אֶת מָה שֶׁקָּנָה. בְּאֶרֶץ יִשְׂרָאֵל לֹא מוֹכְרִים קַרְקַע לְגוֹי, וְלָכֵן רַק מַשְׂכִּירִים לוֹ אֶת הַמָּקוֹם.
(פֶּרֶק ו׳, הֲלָכָה ב׳)

4לְמִי נוֹעֲדָה הַמְּכִירָה הַיּוֹם?

הַיּוֹם יֵשׁ לְמִפְעָלֵי מָזוֹן, לְרִשְׁתוֹת שִׁוּוּק וְלַחֲנֻיּוֹת מְלַאי גָּדוֹל שֶׁל חָמֵץ. הֵם מוֹכְרִים אוֹתוֹ לְגוֹי לִפְנֵי פֶּסַח, וּמִיָּד אַחֲרָיו קוֹנִים אוֹתוֹ בַּחֲזָרָה. בִּזְכוּת זֶה, מִיָּד אַחֲרֵי הֶחָג אֶפְשָׁר לִקְנוֹת שׁוּב לֶחֶם וְעוּגִיּוֹת בַּחֲנוּת.
🤔 לָמָּה? בְּלִי הַמְּכִירָה, הַמִּפְעָלִים הָיוּ מַפְסִידִים הוֹן רַב, וְהָיוּ עוֹבְרִים יָמִים וְשָׁבוּעוֹת עַד שֶׁיַּחְזְרוּ לְיַצֵּר – וְגַם הַקּוֹנִים הָיוּ מִצְטַעֲרִים.
(פֶּרֶק ו׳, הֲלָכָה ג׳)

5גַּם מְעַט חָמֵץ?

מִן הַדִּין כָּל אֶחָד יָכוֹל לִמְכֹּר אֶת חֲמֵצוֹ בַּמְּכִירָה שֶׁמְּסַדְּרִים הָרַבָּנִים, אֲפִלּוּ חֲבִילַת אִטְרִיּוֹת קְטַנָּה. וְיֵשׁ מַחְמִירִים לִמְכֹּר רַק כְּשֶׁיֵּשׁ הֶפְסֵד גָּדוֹל. לְמַעֲשֶׂה, מַמְלִיצִים לְכֻלָּם לְהִשְׁתַּתֵּף בִּמְכִירַת חָמֵץ, בִּשְׁבִיל מוּצָרִים וּתְרוּפוֹת טְעִימוֹת שֶׁיֵּשׁ סָפֵק אִם יֵשׁ בָּהֶם חָמֵץ. אֲבָל חָמֵץ וַדַּאי בְּכַמּוּת קְטַנָּה – מַמְלִיצִים לֹא לִמְכֹּר, אֶלָּא לְבַעֵר. וְכָל אֶחָד קוֹבֵעַ לְעַצְמוֹ מָה נֶחְשָׁב אֶצְלוֹ הֶפְסֵד קָטָן וּמָה גָּדוֹל.
🤔 לָמָּה? אֶת הַמּוּצָרִים שֶׁיֵּשׁ בָּהֶם סָפֵק לֹא רָאוּי לְאַבֵּד, וְגַם אִי אֶפְשָׁר לְהַשְׁאִיר אוֹתָם סְתָם בַּבַּיִת – וְלָכֵן מוֹכְרִים אוֹתָם וְיוֹצְאִים מִן הַסָּפֵק. וְחָמֵץ וַדַּאי מוּעָט – עָדִיף לֹא לְהִשְׁתַּמֵּשׁ בְּהֶתֵּר הַמְּכִירָה לְצֹרֶךְ קָטָן.
(פֶּרֶק ו׳, הֲלָכָה ג׳)

6הֶחָמֵץ הַמָּכוּר – בָּאָרוֹן הַסָּגוּר

הֶחָמֵץ שֶׁנִּמְכַּר נִשְׁאָר בַּבַּיִת, וְלָכֵן דּוֹאֲגִים שֶׁלֹּא נִשְׁכַּח וְנֹאכַל מִמֶּנּוּ: מַעֲמִידִים לְפָנָיו מְחִצָּה גְּבוֹהָה עֲשָׂרָה טְפָחִים (כְּ-80 סֶנְטִימֶטֶר), אוֹ נוֹעֲלִים אוֹתוֹ בָּאָרוֹן וּמַצְנִיעִים אֶת הַמַּפְתֵּחַ, אוֹ מַדְבִּיקִים עַל דַּלְתוֹת הָאָרוֹן נְיַר דֶּבֶק. טוֹב לִכְתֹּב עַל נְיַר הַדֶּבֶק: ״חָמֵץ שֶׁנִּמְכַּר״.
🤔 לָמָּה? כְּשֶׁהַיָּד נִתְקֶלֶת בַּנְּיָר הַדָּבִיק – נִזְכָּרִים מִיָּד, וְלֹא פּוֹתְחִים בְּטָעוּת.
(פֶּרֶק ו׳, הֲלָכָה ג׳)

7שְׁטַר הַמְּכִירָה

רָאוּי שֶׁכָּל מִי שֶׁמּוֹכֵר חָמֵץ יִקְרָא אֶת הַשְּׁטָר שֶׁהוּא חוֹתֵם עָלָיו, כְּדֵי שֶׁיָּבִין שֶׁהוּא מְמַנֶּה אֶת הָרַבָּנוּת לִמְכֹּר אֶת חֲמֵצוֹ, וְשֶׁהַמְּכִירָה אֲמִתִּית. כּוֹתְבִים בַּשְּׁטָר אֶת שֵׁם הַמּוֹכֵר, אֶת הַכְּתֹבֶת שֶׁלּוֹ וְאֶת מִסְפַּר הַטֶּלֶפוֹן – כְּדֵי שֶׁהַגּוֹי יוּכַל לָבוֹא וְלָקַחַת אֶת חֲמֵצוֹ.
🤔 לָמָּה? זֶה הַיְּסוֹד הֶחָשׁוּב בְּיוֹתֵר: אַחֲרֵי הַמְּכִירָה הֶחָמֵץ בֶּאֱמֶת שַׁיָּךְ לַגּוֹי, וְאִם יָבוֹא – נוֹתְנִים לוֹ לְהִכָּנֵס וְלָקַחַת אוֹתוֹ.
💛 יְהוּדִי שֶׁמְּקַיֵּם אֶת מִלָּתוֹ גַּם כְּלַפֵּי גּוֹי – מְקַדֵּשׁ שֵׁם שָׁמַיִם.
(פֶּרֶק ו׳, הֲלָכָה ד׳)

8אֵיפֹה הֶחָמֵץ?

נָכוֹן לִכְתֹּב בַּשְּׁטָר אֵיפֹה בְּדִיּוּק נִמְצָא הֶחָמֵץ, לְמָשָׁל: ״בָּאָרוֹן הָעֶלְיוֹן בַּמִּטְבָּח, מִשְּׂמֹאל״, אוֹ ״בַּחֶדֶר הַיְּמָנִי, בָּאַרְגָּז הַמְּסֻמָּן״. מְרַכְּזִים אֶת הֶחָמֵץ בְּמָקוֹם אֶחָד וּמְסַמְּנִים אוֹתוֹ. כָּל מָה שֶׁמַּכְנִיסִים לַמָּקוֹם הַזֶּה עַד כִּשְׁעָתַיִם לִפְנֵי זְמַן אִסּוּר הַהֲנָאָה – נִכְלָל בַּמְּכִירָה.
(פֶּרֶק ו׳, הֲלָכָה ד׳)

9מוֹכְרִים גַּם בָּאִינְטֶרְנֶט

אֶפְשָׁר לִמְכֹּר אֶת הֶחָמֵץ עַל יְדֵי שָׁלִיחַ שֶׁכּוֹתֵב אֶת כָּל הַפְּרָטִים, וּבִשְׁעַת הַצֹּרֶךְ – גַּם בַּטֶּלֶפוֹן אוֹ בָּאִינְטֶרְנֶט. אֲבָל לֹא מוֹכְרִים ״חָמֵץ בָּלוּעַ בַּכֵּלִים״.
🤔 לָמָּה? הָעִקָּר הוּא הַקִּנְיָן שֶׁהָרַבָּנוּת עוֹשָׂה עִם הַגּוֹי, וְהוּא חָל עַל כָּל מִי שֶׁבִּקֵּשׁ לִמְכֹּר. וְחָמֵץ בָּלוּעַ בַּכֵּלִים – אֵין לוֹ שׁוּם שׁוֹוִי, וְאַף גּוֹי לֹא יִרְצֶה לִקְנוֹת אוֹתוֹ. כְּשֶׁכּוֹתְבִים אוֹתוֹ בַּשְּׁטָר, הַמְּכִירָה נִרְאֵית כְּמוֹ צְחוֹק.
(פֶּרֶק ו׳, הֲלָכָה ד׳)

10עַד מָתַי מוֹכְרִים?

אֶת הַמְּכִירָה עוֹשִׂים לִפְנֵי שֶׁמַּתְחִילָה הַשָּׁעָה הַשִּׁשִּׁית שֶׁל עֶרֶב פֶּסַח. הָרַבָּנִים עוֹשִׂים אוֹתָהּ בְּיוֹם י״ד, סָמוּךְ לְסוֹף הַזְּמַן – כְּדֵי שֶׁכֻּלָּם יוּכְלוּ לְהִצְטָרֵף עַד הָרֶגַע הָאַחֲרוֹן.
🤔 לָמָּה? מֵהַשָּׁעָה הַשִּׁשִּׁית הֶחָמֵץ אָסוּר בַּהֲנָאָה, וְאִי אֶפְשָׁר לִמְכֹּר אוֹתוֹ – רַק לְבַעֵר.
(פֶּרֶק ו׳, הֲלָכָה ה׳)

11בְּחוּץ לָאָרֶץ בְּפֶסַח

אִסּוּר חָמֵץ חָל לְפִי הַמָּקוֹם שֶׁבּוֹ הָאָדָם נִמְצָא. לָכֵן תּוֹשָׁב אֲמֶרִיקָה שֶׁנִּמְצָא בְּאֶרֶץ יִשְׂרָאֵל – מוֹכֵר אֶת חֲמֵצוֹ בְּאֶרֶץ יִשְׂרָאֵל. וּמִשְׁפָּחָה יִשְׂרְאֵלִית שֶׁנָּסְעָה לְחוּץ לָאָרֶץ – מִן הַדִּין יְכוֹלָה לִמְכֹּר שָׁם, אֲבָל לְכַתְּחִלָּה מוֹכֶרֶת בְּאֶרֶץ יִשְׂרָאֵל.
🤔 לָמָּה? בְּאֶרֶץ יִשְׂרָאֵל הַשָּׁעָה הַשִּׁשִּׁית מַגִּיעָה שֶׁבַע שָׁעוֹת לִפְנֵי אֲמֶרִיקָה. מִי שֶׁנִּמְצָא כָּאן וּמוֹכֵר שָׁם – הַמְּכִירָה תִּהְיֶה מְאֻחֶרֶת מִדַּי. וּמִי שֶׁנִּמְצָא שָׁם – יֵשׁ פּוֹסְקִים שֶׁהוֹלְכִים לְפִי מְקוֹם הֶחָמֵץ, וְלָכֵן לְכַתְּחִלָּה מוֹכְרִים כָּאן.
(פֶּרֶק ו׳, הֲלָכָה ה׳)

12אַחֲרֵי פֶּסַח

אַחֲרֵי פֶּסַח טוֹב לְהַמְתִּין מְעַט, עַד שֶׁאֶפְשָׁר לְשַׁעֵר שֶׁהָרַבָּנוּת כְּבָר קָנְתָה בַּחֲזָרָה אֶת הֶחָמֵץ שֶׁל כֻּלָּם – וְאָז פּוֹתְחִים אֶת הָאָרוֹן. בִּשְׁעַת הַצֹּרֶךְ אֶפְשָׁר לָקַחַת מִיָּד, אִם מוּכָנִים לְשַׁלֵּם לַגּוֹי אֶת הַמְּחִיר הַמָּלֵא אִם יְבַקֵּשׁ.
(פֶּרֶק ו׳, הֲלָכָה ו׳)

13אוֹכְלִים מֵחָמֵץ שֶׁנִּמְכַּר?

יֵשׁ שְׁנֵי מִנְהָגִים:
🤔 לָמָּה? אִסּוּר חָמֵץ שֶׁעָבַר עָלָיו הַפֶּסַח הוּא מִדִּבְרֵי חֲכָמִים, וּבְסָפֵק בְּדִבְרֵי חֲכָמִים – הֲלָכָה כַּמֵּקֵל. וְכָאן רֹב גָּדוֹל שֶׁל הַפּוֹסְקִים מַתִּירִים.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ נוֹהֲגִים לְהַחְמִיר לֹא אוֹכְלִים מֵחָמֵץ שֶׁנִּמְכַּר, כִּי לְדַעַת מְעַטִּים הַמְּכִירָה אֵינָהּ כַּדִּין, וְאָז הֶחָמֵץ הוּא ״חָמֵץ שֶׁעָבַר עָלָיו הַפֶּסַח״.
רֹב הַפּוֹסְקִים מַתִּירִים, וְכָךְ פּוֹסֵק הָרַב מֶלָמֵד מֻתָּר לֶאֱכֹל מִמֶּנּוּ אַחֲרֵי פֶּסַח, כִּי הַמְּכִירָה כְּשֵׁרָה. וְיֵשׁ גְּדוֹלִים שֶׁהִקְפִּידוּ לֶאֱכֹל מִמֶּנּוּ, כְּדֵי לְהַרְאוֹת שֶׁהַמְּכִירָה נַעֲשְׂתָה כַּדִּין.
(פֶּרֶק ו׳, הֲלָכָה ו׳)

14קוֹנִים חָמֵץ בַּחֲנוּת אַחֲרֵי פֶּסַח

מִי שֶׁקּוֹנֶה חָמֵץ בַּחֲנוּת אַחֲרֵי פֶּסַח – בּוֹדֵק שֶׁיֵּשׁ לַמּוֹכֵר תְּעוּדָה שֶׁמָּכַר אֶת חֲמֵצוֹ כַּדִּין. אִם הַמּוֹכֵר מָכַר כַּדִּין, וְנִזְהַר לֹא לִמְכֹּר מֵהֶחָמֵץ הַמָּכוּר בְּפֶסַח – מֻתָּר לִקְנוֹת אֶצְלוֹ מִיָּד. אֲבָל אִם הוּא הִמְשִׁיךְ לִמְכֹּר חָמֵץ בְּפֶסַח – נָכוֹן לְחַכּוֹת עַד שֶׁתַּגִּיעַ לַחֲנוּת סְחוֹרָה חֲדָשָׁה, שֶׁיֻּצְּרָה אַחֲרֵי פֶּסַח.
🤔 לָמָּה? כְּדֵי שֶׁלֹּא נִקְנֶה חָמֵץ שֶׁל יְהוּדִי שֶׁעָבַר עָלָיו הַפֶּסַח, שֶׁאָסוּר בַּאֲכִילָה.
(פֶּרֶק ו׳, הֲלָכָה ו׳)

📝 מָה לָמַדְנוּ

  • מִי שֶׁלֹּא רוֹצֶה לְהַפְסִיד אֶת חֲמֵצוֹ – מוֹכֵר אוֹתוֹ לְגוֹי דֶּרֶךְ הָרַבָּנוּת, לִפְנֵי הַשָּׁעָה הַשִּׁשִּׁית.
  • זוֹ מְכִירָה אֲמִתִּית: הַגּוֹי רַשַּׁאי לָבוֹא וְלָקַחַת אֶת חֲמֵצוֹ, וְלָכֵן כּוֹתְבִים שֵׁם, כְּתֹבֶת, טֶלֶפוֹן וּמְקוֹם הֶחָמֵץ.
  • הֶחָמֵץ הַמָּכוּר נִשְׁאָר בָּאָרוֹן סָגוּר בִּנְיַר דֶּבֶק, אוֹ נָעוּל, אוֹ מֵאֲחוֹרֵי מְחִצָּה.
  • מַמְלִיצִים לְכֻלָּם לִמְכֹּר מוּצָרִים שֶׁיֵּשׁ בָּהֶם סָפֵק; חָמֵץ וַדַּאי מוּעָט – מְבַעֲרִים.
  • אַחֲרֵי פֶּסַח מַמְתִּינִים מְעַט; רֹב הַפּוֹסְקִים מַתִּירִים לֶאֱכֹל מֵחָמֵץ שֶׁנִּמְכַּר, וְיֵשׁ מַחְמִירִים.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה אוֹמְרִים שֶׁמְּכִירַת חָמֵץ הִיא ״מְכִירָה אֲמִתִּית״ וְלֹא הַעֲרָמָה?
  2. אֵיךְ שׁוֹמְרִים עַל הֶחָמֵץ הַמָּכוּר בַּבַּיִת, וְלָמָּה?
  3. מָה צָרִיךְ לִבְדֹּק כְּשֶׁקּוֹנִים לֶחֶם בַּחֲנוּת מִיָּד אַחֲרֵי פֶּסַח?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״מְכִירַת חָמֵץ, כַּשְׁרוּת וְקִטְנִיּוֹת״

סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת פֶּסַח
מַהֲדוּרַת יְלָדִים

⏳ עֶרֶב פֶּסַח וּמִנְהָגָיו

פֶּרֶק י״ג · לַמַּהֲדוּרָה לַמּוֹרֶה
עֶרֶב פֶּסַח. הַבַּיִת מַבְרִיק, הַסִּירִים עַל הָאֵשׁ, וְכֻלָּם מִתְרוֹצְצִים. אֶפְשָׁר לִשְׁאֹל: מָה עוֹד יָכוֹל לִהְיוֹת מְיֻחָד בַּיּוֹם הַזֶּה, חוּץ מֵהַהֲכָנוֹת? הִנֵּה הַפְתָּעָה: לְעֶרֶב פֶּסַח יֵשׁ הֲלָכוֹת מִשֶּׁלּוֹ! יֵשׁ בּוֹ צוֹם מְיֻחָד, יֵשׁ מַאֲכָל שֶׁאָסוּר לֶאֱכֹל דַּוְקָא בּוֹ, וּמֵהַצָּהֳרַיִם הוּא נִהְיֶה כִּמְעַט כְּמוֹ חַג.

1מֵחֲצוֹת הַיּוֹם – כִּמְעַט חַג

מֵחֲצוֹת הַיּוֹם שֶׁל עֶרֶב פֶּסַח (י״ד בְּנִיסָן) אָסוּר לַעֲשׂוֹת מְלָאכָה חֲשׁוּבָה. הָאִסּוּר הַזֶּה חָמוּר אֲפִלּוּ יוֹתֵר מֵעֶרֶב שַׁבָּת: בְּעֶרֶב שַׁבָּת הוּא מַתְחִיל רַק כְּשָׁעָתַיִם וָחֵצִי לִפְנֵי הַשְּׁקִיעָה, וּבְעֶרֶב פֶּסַח – כְּבָר מֵחֲצוֹת הַיּוֹם.
🤔 לָמָּה? בִּזְמַן בֵּית הַמִּקְדָּשׁ, כָּל יְהוּדִי הִקְרִיב בְּעֶרֶב פֶּסַח אַחַר חֲצוֹת אֶת קָרְבַּן הַפֶּסַח. מִי שֶׁקָּרְבָּנוֹ עוֹלֶה עַל הַמִּזְבֵּחַ – הַיּוֹם הַהוּא הוּא לוֹ כְּמוֹ יוֹם טוֹב. וְעוֹד: כְּדֵי שֶׁלֹּא נִהְיֶה עֲסוּקִים בַּעֲבוֹדָה, וְנִשְׁכַּח לְבַעֵר אֶת הֶחָמֵץ וּלְהָכִין אֶת הַמַּצּוֹת וְכָל מָה שֶׁצָּרִיךְ לַסֵּדֶר.
💛 אֲפִלּוּ בְּלִי בֵּית מִקְדָּשׁ, אֲנַחְנוּ מַשְׁאִירִים לַה׳ ״מָקוֹם פָּנוּי״ בַּיּוֹם – כְּאִלּוּ הַקָּרְבָּן שֶׁלָּנוּ מְחַכֶּה.
(פֶּרֶק י״ג, הֲלָכָה א׳)

2מָה אָסוּר וּמָה מֻתָּר אַחַר חֲצוֹת?

אָסוּר לַעֲשׂוֹת מְלָאכָה גְּמוּרָה, כְּמוֹ תְּפִירַת בְּגָדִים, בְּנִיַּת רָהִיטִים, נְטִיעַת שְׁתִילִים וְתִסְפֹּרֶת. אֲבָל מֻתָּר לְבַשֵּׁל, לְנַקּוֹת אֶת הַבַּיִת וְלִנְסֹעַ לִקְרַאת הַחַג. מֻתָּר גַּם לְגַהֵץ, לְצַחְצֵחַ נַעֲלַיִם, לִגְזֹז צִפָּרְנַיִם, לְהִתְגַּלֵּחַ בִּמְכוֹנַת גִּלּוּחַ, לִתְפֹּר כַּפְתּוֹר שֶׁנָּפַל, וְלִכְתֹּב לְעַצְמְךָ סִכּוּמִים שֶׁל מָה שֶׁלָּמַדְתָּ. וְיֵשׁ מְהַדְּרִים לְהִתְגַּלֵּחַ וְלִגְזֹז צִפָּרְנַיִם לִפְנֵי חֲצוֹת.
🤔 לָמָּה? אָסְרוּ מְלָאכוֹת שֶׁאֲנָשִׁים מִתְפַּרְנְסִים מֵהֶן, שֶׁלּוֹקְחוֹת אֶת הָרֹאשׁ לְעִנְיָנִים אֲחֵרִים. אֲבָל כָּל מָה שֶׁמֵּכִין אוֹתָנוּ לַחַג – מֻתָּר וַאֲפִלּוּ רָצוּי. כְּלָל טוֹב לִזְכֹּר: מָה שֶׁמֻּתָּר לַעֲשׂוֹת בְּחֹל הַמּוֹעֵד – מֻתָּר גַּם בְּעֶרֶב פֶּסַח אַחַר חֲצוֹת.
(פֶּרֶק י״ג, הֲלָכָה ב׳)

3תַּעֲנִית בְּכוֹרוֹת – לָמָּה?

הַבְּכוֹרִים נוֹהֲגִים לְהִתְעַנּוֹת בְּעֶרֶב פֶּסַח, זֵכֶר לַנֵּס שֶׁהָיָה בְּמִצְרַיִם: כָּל בְּכוֹרֵי מִצְרַיִם מֵתוּ, וּבְכוֹרֵי יִשְׂרָאֵל נִצְּלוּ.
🤔 לָמָּה? בְּכָל בְּכוֹר יֵשׁ ״רִאשׁוֹנִיּוּת״ – אִתּוֹ מַתְחִיל הַדּוֹר הֶחָדָשׁ, וְלָכֵן יֵשׁ עָלָיו אַחְרָיוּת גְּדוֹלָה: אַחֲרָיו הוֹלְכִים הָאַחִים. פַּרְעֹה, שֶׁהָיָה בְּכוֹר, הִתְגָּאָה וְכָפַר בַּה׳. וּבְכוֹרֵי יִשְׂרָאֵל שָׁחֲטוּ אֶת הַשֶּׂה – הָאֱלִיל שֶׁל מִצְרַיִם – וְשָׂמוּ מִדָּמוֹ עַל הַמְּזוּזוֹת בִּמְסִירוּת נֶפֶשׁ, וְנִצְּלוּ. לָכֵן לִפְנֵי לֵיל הַסֵּדֶר הַבְּכוֹרִים שׁוֹאֲלִים אֶת עַצְמָם: הַאִם אֲנַחְנוּ מְגַלִּים אֶת שֵׁם ה׳ כָּרָאוּי? – וְחוֹזְרִים בִּתְשׁוּבָה.
💛 לִהְיוֹת ״רִאשׁוֹן״ זֶה לֹא לְהִתְגָּאוֹת – זֶה לְהַרְאוֹת לַאֲחֵרִים אֶת הַדֶּרֶךְ אֶל ה׳.
(פֶּרֶק י״ג, הֲלָכָה ג׳)

4מִי צָם?

צָם בֵּן שֶׁהוּא בְּכוֹר לְאָבִיו, וְגַם בֵּן שֶׁהוּא בְּכוֹר לְאִמּוֹ. גַּם בְּכוֹרִים שֶׁל כֹּהֲנִים וּלְוִיִּם צָמִים. אַבָּא שֶׁנּוֹלַד לוֹ בֵּן בְּכוֹר – צָם בִּמְקוֹם בְּנוֹ, עַד שֶׁהַבֵּן יִגְדַּל. וּבַת בְּכוֹרָה? יֵשׁ אוֹמְרִים שֶׁגַּם בְּכוֹרוֹת בָּנוֹת מֵתוּ בְּמִצְרַיִם, אֲבָל הַמִּנְהָג הָרוֹוֵחַ שֶׁבַּת בְּכוֹרָה אֵינָהּ מִתְעַנָּה. וּכְשֶׁעֶרֶב פֶּסַח חָל בְּשַׁבָּת – מַקְדִּימִים אֶת הַתַּעֲנִית לְיוֹם חֲמִישִׁי.
🤔 לָמָּה? בְּמִצְרַיִם מֵתוּ כָּל סוּגֵי הַבְּכוֹרִים – גַּם בְּכוֹרִים לָאָב וְגַם בְּכוֹרִים לָאֵם – וְלָכֵן כֻּלָּם בַּמִּנְהָג.
(פֶּרֶק י״ג, הֲלָכָה ד׳)

5צוֹם קַל

תַּעֲנִית בְּכוֹרוֹת קַלָּה מִשְּׁאָר הַצּוֹמוֹת. מִי שֶׁכּוֹאֵב לוֹ הָרֹאשׁ אוֹ הָעֵינַיִם – פָּטוּר מִמֶּנָּה, אַף שֶׁבִּשְׁאָר הַצּוֹמוֹת הָיָה צָם. וּמִי שֶׁחוֹשֵׁשׁ שֶׁאִם יָצוּם לֹא יוּכַל לְקַיֵּם כָּרָאוּי בַּלַּיְלָה אֶת מִצְווֹת הַסֵּדֶר – מוּטָב שֶׁלֹּא יָצוּם.
🤔 לָמָּה? אֶת שְׁאָר הַצּוֹמוֹת תִּקְּנוּ חֲכָמִים, וְאִלּוּ תַּעֲנִית בְּכוֹרוֹת הִיא מִנְהָג שֶׁהַבְּכוֹרִים קִבְּלוּ עַל עַצְמָם. וְהָעִקָּר הוּא לְהַגִּיעַ לְלֵיל הַסֵּדֶר עִם כֹּחַ וְשִׂמְחָה.
(פֶּרֶק י״ג, הֲלָכָה ג׳)

6סִיּוּם מַסֶּכֶת

רֹב הַבְּכוֹרִים בָּאִים בְּעֶרֶב פֶּסַח, אַחֲרֵי תְּפִלַּת שַׁחֲרִית, לְסִיּוּם מַסֶּכֶת. אַחֲרֵי הַסִּיּוּם מַגִּישִׁים כִּבּוּד, וְכֻלָּם טוֹעֲמִים. זוֹ סְעוּדַת מִצְוָה שֶׁל שִׂמְחַת הַתּוֹרָה, וּמִי שֶׁאוֹכֵל בָּהּ – נִפְטָר מֵהַצּוֹם. סִיּוּם שֶׁפּוֹטֵר הוּא סִיּוּם שֶׁל מַסֶּכֶת גְּמָרָא, סֵדֶר מִשְׁנָה אוֹ סֵפֶר חָשׁוּב, שֶׁלָּמְדוּ אוֹתוֹ בַּהֲבָנָה טוֹבָה. וּמִי שֶׁיּוֹדֵעַ שֶׁהַצּוֹם לֹא יַפְרִיעַ לוֹ בַּהֲכָנוֹת וּבַסֵּדֶר – יָכוֹל לָצוּם. כָּךְ נָהֲגוּ הָרַב קוּק וּבְנוֹ הָרַב צְבִי יְהוּדָה, שֶׁהָיוּ בְּכוֹרִים.
🤔 לָמָּה? כְּשֶׁמְּסַיְּמִים מַסֶּכֶת – רָאוּי לִשְׂמֹחַ, וְזוֹ סְעוּדַת מִצְוָה. וְעוֹד: אִם הַבְּכוֹרִים יָצוּמוּ, קָרוֹב לְוַדַּאי שֶׁהַצּוֹם יַחֲלִישׁ אוֹתָם, וּבְלֵיל הַסֵּדֶר הֵם יְמַהֲרוּ לְסַיֵּם אֶת הַהַגָּדָה כְּדֵי לֶאֱכֹל.
💛 אֵיזוֹ דֶּרֶךְ יָפָה לְהִתְכּוֹנֵן לְלֵיל הַסֵּדֶר: לִשְׂמֹחַ בְּסִיּוּם שֶׁל לִמּוּד תּוֹרָה.
(פֶּרֶק י״ג, הֲלָכָה ה׳)

7בְּעֶרֶב פֶּסַח – בְּלִי מַצָּה!

חֲכָמִים אָסְרוּ לֶאֱכֹל מַצָּה בְּעֶרֶב פֶּסַח. הָאִסּוּר מַתְחִיל בַּעֲלוֹת הַשַּׁחַר. הוּא חָל גַּם עַל יְלָדִים שֶׁכְּבָר מְבִינִים שֶׁהַמַּצָּה הִיא זֵכֶר לִיצִיאַת מִצְרַיִם. אֲבָל יְלָדִים קְטַנִּים שֶׁעוֹד לֹא מְבִינִים זֹאת – מֻתָּר לָתֵת לָהֶם מַצָּה.
🤔 לָמָּה? כְּדֵי שֶׁנֹּאכַל אֶת הַמַּצָּה בַּלַּיְלָה בְּתֵאָבוֹן וּבְחִבָּה, וּכְדֵי שֶׁיִּהְיֶה הֶבְדֵּל בָּרוּר בֵּין סְתָם אֲכִילָה לִפְנֵי פֶּסַח לְבֵין אֲכִילַת מַצַּת מִצְוָה בְּלֵיל הַסֵּדֶר.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
מִצַּד הַדִּין נִמְנָעִים מִמַּצָּה רַק בְּיוֹם י״ד בְּנִיסָן – עֶרֶב פֶּסַח.
יֵשׁ שֶׁנָּהֲגוּ לְהִמָּנַע מִמַּצָּה כְּבָר מֵרֹאשׁ חֹדֶשׁ נִיסָן, וְיֵשׁ – שְׁלֹשִׁים יוֹם לִפְנֵי הַפֶּסַח.
💛 הַגַּעְגּוּעַ לַמַּצָּה כָּל הַיּוֹם הוּא חֵלֶק מֵהַהֲכָנָה – כְּמוֹ שֶׁמְּחַכִּים לְאוֹרֵחַ אָהוּב.
(פֶּרֶק י״ג, הֲלָכָה ו׳)

8מָה כֵּן אוֹכְלִים בְּעֶרֶב פֶּסַח?

חָמֵץ מֻתָּר לֶאֱכֹל עַד סוֹף הַשָּׁעָה הָרְבִיעִית שֶׁל הַיּוֹם (הַזְּמַן מֻדְפָּס בְּלוּחוֹת הַזְּמַנִּים). אַחַר כָּךְ – לֹא חָמֵץ וְלֹא מַצָּה. קְצִיצוֹת מְבֻשָּׁלוֹת מִקֶּמַח מַצָּה (״קְנֵיידְלַךְ״) מֻתָּר לֶאֱכֹל לְכָל הַדֵּעוֹת, כִּי אַחֲרֵי הַבִּשּׁוּל הֵן כְּבָר לֹא נֶחְשָׁבוֹת מַצָּה. וְעַל עוּגוֹת וְעוּגִיּוֹת שֶׁנֶּאֱפוּ מִקֶּמַח מַצָּה – נֶחְלְקוּ.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ מְקִלִּים מֻתָּר לֶאֱכֹל עוּגוֹת מִקֶּמַח מַצָּה בְּעֶרֶב פֶּסַח, כִּי כְּבָר אֵין לָהֶן צוּרַת מַצָּה. וּלְמַעֲשֶׂה אֶפְשָׁר לְהָקֵל.
וְיֵשׁ מַחְמִירִים גַּם עוּגוֹת כָּאֵלֶּה אֲסוּרוֹת, כִּי עֲדַיִן יֵשׁ עֲלֵיהֶן שֵׁם מַצָּה. וְהַמַּחְמִיר – תָּבוֹא עָלָיו בְּרָכָה.
(פֶּרֶק י״ג, הֲלָכָה ו׳, הֲלָכָה ז׳)

9מַצָּה עֲשִׁירָה

״מַצָּה עֲשִׁירָה״ הִיא מַצָּה שֶׁהַקֶּמַח שֶׁלָּהּ נִלּוֹשׁ בְּמִיץ פֵּרוֹת – כְּמוֹ יַיִן, שֶׁמֶן אוֹ מִיצִים אֲחֵרִים – וְלֹא בְּמַיִם. כַּיּוֹם יֵשׁ סְפֵקוֹת גְּדוֹלִים עַל הַדֶּרֶךְ שֶׁבָּהּ מְיַצְּרִים מַצָּה עֲשִׁירָה, וְלָכֵן רַבִּים מֵהַפּוֹסְקִים אוֹמְרִים שֶׁגַּם יוֹצְאֵי סְפָרַד צְרִיכִים לְהִמָּנַע מִמֶּנָּה. וּמִי שֶׁנּוֹהֵג לְאָכְלָהּ – אוֹכֵל רַק עַד סוֹף הַשָּׁעָה הַתְּשִׁיעִית שֶׁל הַיּוֹם.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי סְפָרַד מֻתָּר לֶאֱכֹל מַצָּה עֲשִׁירָה בְּפֶסַח וּבְעֶרֶב פֶּסַח.
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז אָסוּר לֶאֱכֹל מַצָּה עֲשִׁירָה.
(פֶּרֶק י״ג, הֲלָכָה ז׳)

10שׁוֹמְרִים אֶת הַתֵּאָבוֹן

מִתְּחִלַּת הַשָּׁעָה הָעֲשִׂירִית – כְּשָׁלֹשׁ שָׁעוֹת לִפְנֵי הַחַג – לֹא אוֹכְלִים מִינֵי מְזוֹנוֹת. מִי שֶׁרָעֵב יָכוֹל לֶאֱכֹל מְעַט פֵּרוֹת, יְרָקוֹת, בָּשָׂר אוֹ דָּגִים – אֲבָל רַק מְעַט, כְּדֵי שֶׁיַּגִּיעַ לָעֶרֶב כְּשֶׁהוּא רָעֵב. וּמִי שֶׁיּוֹדֵעַ שֶׁאִם יֹאכַל בַּצָּהֳרַיִם הוּא לֹא יִהְיֶה רָעֵב בַּלַּיְלָה – יְתַכְנֵן אֶת הָאֹכֶל שֶׁלּוֹ בַּיּוֹם הַזֶּה מֵרֹאשׁ.
🤔 לָמָּה? אֶת הַמַּצָּה בְּלֵיל הַסֵּדֶר צְרִיכִים לֶאֱכֹל לְתֵאָבוֹן, בְּחֵשֶׁק וּבְשִׂמְחָה – לֹא כְּשֶׁהַבֶּטֶן כְּבָר מְלֵאָה.
(פֶּרֶק י״ג, הֲלָכָה ז׳)

11סֵדֶר קָרְבַּן פֶּסַח

נוֹהֲגִים לוֹמַר אַחֲרֵי תְּפִלַּת מִנְחָה שֶׁל עֶרֶב פֶּסַח אֶת ״סֵדֶר קָרְבַּן פֶּסַח״ – פְּסוּקִים עַל הַקָּרְבָּן, וְאֵיךְ הָיוּ מַקְרִיבִים אוֹתוֹ. וְעוֹד מִנְהָג: יֵשׁ מְהַדְּרִים לֶאֱפוֹת אֶת מַצּוֹת הַמִּצְוָה דַּוְקָא אַחַר חֲצוֹת שֶׁל עֶרֶב פֶּסַח, בִּזְמַן שֶׁל קָרְבַּן הַפֶּסַח; וְרַבִּים – כְּמוֹ הָרַב קוּק – לֹא אָפוּ אָז, כִּי זוֹ טִרְחָה גְּדוֹלָה בְּשָׁעָה שֶׁכֻּלָּם עֲסוּקִים בַּהֲכָנוֹת.
🤔 לָמָּה? חֲכָמִים אָמְרוּ שֶׁכְּשֶׁאֵין בֵּית מִקְדָּשׁ, הַלִּמּוּד עַל הַקָּרְבָּן הוּא כְּמוֹ תַּחֲלִיף לַקָּרְבָּן עַצְמוֹ. וְהַמַּהֲרַ״ל לִמֵּד שֶׁקָּרְבַּן הַפֶּסַח מְבַטֵּא אַחְדּוּת: אוֹכְלִים אוֹתוֹ בַּחֲבוּרָה, לֹא עוֹבְרִים מֵחֲבוּרָה לַחֲבוּרָה, צוֹלִים אוֹתוֹ – כִּי הַצְּלִיָּה מְלַכֶּדֶת אֶת הַבָּשָׂר – וְלֹא שׁוֹבְרִים בּוֹ עֶצֶם, כִּי שְׁבִירָה הִיא פֵּרוּד.
💛 ״יְהִי רָצוֹן שֶׁיִּבָּנֶה בֵּית הַמִּקְדָּשׁ בִּמְהֵרָה בְיָמֵינוּ״ – וְנַקְרִיב כֻּלָּנוּ יַחַד אֶת קָרְבַּן הַפֶּסַח.
(פֶּרֶק י״ג, הֲלָכָה ח׳)

📝 מָה לָמַדְנוּ

  • מֵחֲצוֹת הַיּוֹם שֶׁל עֶרֶב פֶּסַח לֹא עוֹשִׂים מְלָאכָה חֲשׁוּבָה, כִּי זֶה זְמַן קָרְבַּן הַפֶּסַח.
  • הַבְּכוֹרִים צָמִים זֵכֶר לְהַצָּלַת בְּכוֹרֵי יִשְׂרָאֵל, וְרֻבָּם נִפְטָרִים בְּסִיּוּם מַסֶּכֶת.
  • בְּעֶרֶב פֶּסַח לֹא אוֹכְלִים מַצָּה, כְּדֵי לֶאֱכֹל אוֹתָהּ בַּלַּיְלָה בְּחִבָּה וּבְתֵאָבוֹן.
  • חָמֵץ – עַד סוֹף הַשָּׁעָה הָרְבִיעִית; וּמֵהַשָּׁעָה הָעֲשִׂירִית אוֹכְלִים רַק מְעַט, כְּדֵי לְהַגִּיעַ רְעֵבִים לַסֵּדֶר.
  • אַחֲרֵי מִנְחָה אוֹמְרִים ״סֵדֶר קָרְבַּן פֶּסַח״.

❓ שְׁאֵלוֹת חֲזָרָה

  1. לָמָּה אָסוּר לַעֲשׂוֹת מְלָאכָה חֲשׁוּבָה בְּעֶרֶב פֶּסַח אַחַר חֲצוֹת? תֵּן דֻּגְמָה לִמְלָאכָה אֲסוּרָה וְלִמְלָאכָה מֻתֶּרֶת.
  2. מִי צָם בְּתַעֲנִית בְּכוֹרוֹת, וּמָה עוֹשִׂים רֹב הַבְּכוֹרִים כְּדֵי לְהִפָּטֵר מִמֶּנָּה?
  3. לָמָּה אָסוּר לֶאֱכֹל מַצָּה בְּעֶרֶב פֶּסַח, וּמָה מֻתָּר לֶאֱכֹל מִקֶּמַח מַצָּה לְכָל הַדֵּעוֹת?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״מַצָּה שְׁמוּרָה וְעֶרֶב פֶּסַח״

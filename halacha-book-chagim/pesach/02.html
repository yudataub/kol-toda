סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת פֶּסַח
מַהֲדוּרַת יְלָדִים

🍞 כְּלָלֵי אִסּוּר חָמֵץ

פֶּרֶק ב׳ · לַמַּהֲדוּרָה לַמּוֹרֶה
לֶחֶם, פִּתָּה, עוּגָה וּמַצָּה – כֻּלָּם עֲשׂוּיִים מִקֶּמַח וּמַיִם. אָז לָמָּה אֶת הַמַּצָּה אוֹכְלִים בְּפֶסַח, וְאֶת הַלֶּחֶם אָסוּר אֲפִלּוּ לְהַשְׁאִיר בַּבַּיִת? מָה קוֹרֶה לַבָּצֵק בְּתוֹךְ שְׁמוֹנֶה עֶשְׂרֵה דַּקּוֹת? וְעַד אֵיזוֹ שָׁעָה מֻתָּר לֶאֱכֹל אֶת הַפִּתָּה שֶׁל הַבֹּקֶר בְּעֶרֶב פֶּסַח? בַּפֶּרֶק הַזֶּה נִלְמַד מָה זֶה בְּעֶצֶם חָמֵץ.

1אַרְבַּע מִצְווֹת בְּחָמֵץ

בַּתּוֹרָה יֵשׁ אַרְבַּע מִצְווֹת עַל חָמֵץ בְּפֶסַח. שָׁלֹשׁ מִצְווֹת ״לֹא תַעֲשֶׂה״: לֹא לֶאֱכֹל חָמֵץ (וְגַם לֹא לֵהָנוֹת מִמֶּנּוּ), ״לֹא יִמָּצֵא״ – שֶׁלֹּא יִמָּצֵא חָמֵץ שֶׁלָּנוּ בַּבַּיִת, וְ״לֹא יֵרָאֶה״ – שֶׁלֹּא יֵרָאֶה חָמֵץ שֶׁלָּנוּ. וּמִצְוַת עֲשֵׂה אַחַת: ״תַּשְׁבִּיתוּ שְׂאֹר מִבָּתֵּיכֶם״ – לְהַשְׁבִּית אֶת הֶחָמֵץ לִקְרַאת הַפֶּסַח.
🤔 לָמָּה? הַתּוֹרָה הֶחְמִירָה מְאוֹד בְּאִסּוּר חָמֵץ. עַל רֹב הַמַּאֲכָלִים הָאֲסוּרִים הָעֹנֶשׁ הוּא מַלְקוֹת, אֲבָל מִי שֶׁאוֹכֵל חָמֵץ בְּפֶסַח בְּמֵזִיד – עָנְשׁוֹ כָּרֵת.
(פֶּרֶק ב׳, הֲלָכָה א׳)

2כַּמָּה חָמֵץ זֶה ״חָמֵץ״?

עַל ״לֹא יֵרָאֶה״ וְ״לֹא יִמָּצֵא״ עוֹבֵר רַק מִי שֶׁיֵּשׁ לוֹ בְּפֶסַח חָמֵץ בְּגֹדֶל שֶׁל כַּזַּיִת לְפָחוֹת, בְּמָקוֹם אֶחָד. חָמֵץ קָטָן מִכַּזַּיִת – לֹא עוֹבְרִים עָלָיו עַל הָאִסּוּרִים הָאֵלֶּה. אֲבָל לֶאֱכֹל חָמֵץ אָסוּר אֲפִלּוּ בְּפָחוֹת מִכַּזַּיִת.
🤔 לָמָּה? ״כַּזַּיִת״ הוּא שִׁעוּר שֶׁל אֹכֶל שֶׁיֵּשׁ לוֹ חֲשִׁיבוּת. פֵּרוּר קָטָן אֵינוֹ נֶחְשָׁב כְּמוֹ חֲתִיכַת אֹכֶל שֶׁיֵּשׁ לָנוּ בַּבַּיִת. אֲבָל לְתוֹךְ הַפֶּה – שׁוּם חָמֵץ לֹא נִכְנָס.
(פֶּרֶק ב׳, הֲלָכָה א׳)

3מָתַי מַתְחִיל הָאִסּוּר?

הֶחָג מַתְחִיל בְּט״ו בְּנִיסָן, אֲבָל אִסּוּר חָמֵץ מַתְחִיל כְּבָר בַּחֲצוֹת הַיּוֹם שֶׁל עֶרֶב פֶּסַח, י״ד בְּנִיסָן. בַּזְּמַן הַזֶּה הָיוּ מַקְרִיבִים אֶת קָרְבַּן הַפֶּסַח, וְהַתּוֹרָה אָמְרָה: ״לֹא תֹאכַל עָלָיו חָמֵץ״. חֲכָמִים הוֹסִיפוּ עוֹד שָׁעָה לְאִסּוּר הַהֲנָאָה, וְעוֹד שְׁתֵּי שָׁעוֹת לְאִסּוּר הָאֲכִילָה.
🤔 לָמָּה? חֲכָמִים רָצוּ לְהַרְחִיק אוֹתָנוּ מֵהָעֲבֵרָה. בְּיוֹם מְעֻנָּן, כְּשֶׁאֵין שָׁעוֹן, אֶפְשָׁר לִטְעוֹת בְּהַעֲרָכַת הַשָּׁעָה עַד שְׁתֵּי שָׁעוֹת. הַתּוֹסֶפֶת שׁוֹמֶרֶת עָלֵינוּ שֶׁלֹּא נִכָּשֵׁל.
(פֶּרֶק ב׳, הֲלָכָה ב׳)

4שָׁעוֹת זְמַנִּיּוֹת

מְחַלְּקִים אֶת הַיּוֹם לִשְׁתֵּים עֶשְׂרֵה ״שָׁעוֹת זְמַנִּיּוֹת״ שָׁווֹת. בְּאַרְבַּע הַשָּׁעוֹת הָרִאשׁוֹנוֹת שֶׁל עֶרֶב פֶּסַח מֻתָּר לֶאֱכֹל חָמֵץ. בַּשָּׁעָה הַחֲמִישִׁית אָסוּר לֶאֱכֹל, אֲבָל מֻתָּר לֵהָנוֹת מִמֶּנּוּ: לְהַאֲכִיל בּוֹ בְּהֵמָה אוֹ לִמְכֹּר אוֹתוֹ לְגוֹי. מֵהַשָּׁעָה הַשִּׁשִּׁית – אָסוּר גַּם לֵהָנוֹת, וּמִי שֶׁשָּׁכַח לִמְכֹּר אוֹתוֹ צָרִיךְ לְאַבֵּד אוֹתוֹ. וּמֵחֲצוֹת הַיּוֹם – הָאִסּוּר הוּא מִן הַתּוֹרָה.
🤔 לָמָּה? ״שָׁעָה זְמַנִּית״ הִיא חֵלֶק אֶחָד מִשְּׁתֵּים עֶשְׂרֵה שֶׁל הַיּוֹם. בְּיוֹם אָרֹךְ הִיא אֲרֻכָּה מִשִּׁשִּׁים דַּקּוֹת, וּבְיוֹם קָצָר – קְצָרָה יוֹתֵר. לָכֵן בּוֹדְקִים בַּלּוּחַ אֶת הַשָּׁעָה הַמְּדֻיֶּקֶת בְּכָל שָׁנָה.
(פֶּרֶק ב׳, הֲלָכָה ב׳)

5כְּשֶׁנִּכְנָס הֶחָג

מֵחֲצוֹת יוֹם י״ד מִצְוָה לְהַשְׁבִּית אֶת הֶחָמֵץ, וְכָל שָׁעָה שֶׁלֹּא מַשְׁבִּיתִים אוֹתוֹ – מְבַטְּלִים מִצְוַת עֲשֵׂה. וּכְשֶׁנִּכְנָס הֶחָג, נוֹסָפִים עוֹד שְׁנֵי אִסּוּרִים: ״לֹא יֵרָאֶה״ וְ״לֹא יִמָּצֵא״. גַּם אִסּוּר הָאֲכִילָה נַעֲשֶׂה חָמוּר יוֹתֵר: מִי שֶׁאוֹכֵל חָמֵץ בְּמֵזִיד אַחֲרֵי חֲצוֹת יוֹם י״ד עָנְשׁוֹ מַלְקוֹת, וּמִי שֶׁאוֹכֵל בְּתוֹךְ הֶחָג – עָנְשׁוֹ כָּרֵת.
(פֶּרֶק ב׳, הֲלָכָה ב׳)

6וְאַחֲרֵי פֶּסַח?

כְּשֶׁהַפֶּסַח נִגְמָר, הֶחָמֵץ חוֹזֵר לִהְיוֹת מֻתָּר. אֲבָל חָמֵץ שֶׁל יְהוּדִי שֶׁנִּשְׁאַר אֶצְלוֹ בְּפֶסַח – חֲכָמִים אָסְרוּ אוֹתוֹ בַּאֲכִילָה וּבַהֲנָאָה גַּם אַחֲרֵי הֶחָג. לְעֻמַּת זֹאת, חָמֵץ שֶׁל גּוֹי שֶׁעָבַר עָלָיו הַפֶּסַח – מֻתָּר לִקְנוֹת וְלֶאֱכֹל.
🤔 לָמָּה? הַיְּהוּדִי שֶׁהִשְׁאִיר אֶת הֶחָמֵץ שֶׁלּוֹ בְּפֶסַח עָבַר עַל ״לֹא יֵרָאֶה״ וְ״לֹא יִמָּצֵא״, וְלָכֵן חֲכָמִים קָנְסוּ וְאָסְרוּ אֶת הֶחָמֵץ הַזֶּה. הַגּוֹי לֹא הָיָה מְצֻוֶּה בְּכָךְ.
(פֶּרֶק ב׳, הֲלָכָה ב׳)

7מָה זֶה חָמֵץ?

חָמֵץ נוֹצָר מֵאֶחָד מֵחֲמֵשֶׁת מִינֵי דָּגָן – חִטָּה, שְׂעוֹרָה, שִׁבֹּלֶת שׁוּעָל, שִׁיפוֹן וְכֻסְּמִין – שֶׁנָּגְעוּ בְּמַיִם וְהֶחְמִיצוּ. מֵהֶם אוֹפִים לֶחֶם, עוּגוֹת וּפִתּוֹת. וְדַוְקָא מֵהֶם עוֹשִׂים גַּם אֶת הַמַּצָּה: לָשִׁים מַהֵר וּמַכְנִיסִים מִיָּד לַתַּנּוּר, לִפְנֵי שֶׁהַבָּצֵק מַסְפִּיק לְהַחְמִיץ. כָּךְ כָּתוּב: ״וַיִּשָּׂא הָעָם אֶת בְּצֵקוֹ טֶרֶם יֶחְמָץ״.
🤔 לָמָּה? מֵחֲמֵשֶׁת מִינֵי הַדָּגָן עוֹשִׂים לֶחֶם – הַמַּאֲכָל הֶחָשׁוּב שֶׁל הָאָדָם, שֶׁמְּבָרְכִים עָלָיו ״הַמּוֹצִיא״ וּ״בִרְכַּת הַמָּזוֹן״. אֹרֶז וְדֹחַן תּוֹפְחִים קְצָת, אֲבָל אֵין בָּהֶם חִמּוּץ אֲמִתִּי, וְלָכֵן הֵם אֵינָם חָמֵץ (וְגַם אִי אֶפְשָׁר לָצֵאת בָּהֶם מִצְוַת מַצָּה).
💛 ה׳ נָתַן לָנוּ אֶת הַלֶּחֶם כָּל הַשָּׁנָה, וּבְפֶסַח אֶת הַמַּצָּה – וּשְׁנֵיהֶם מֵאוֹתוֹ קֶמַח, מֵאוֹתָהּ יָד טוֹבָה.
(פֶּרֶק ב׳, הֲלָכָה ג׳)

8חָמֵץ וּשְׂאוֹר

יֵשׁ שְׁנֵי סוּגִים שֶׁל חִמּוּץ. ״חָמֵץ״ הוּא בָּצֵק שֶׁהֶחְמִיץ, וּמִמֶּנּוּ אוֹפִים לֶחֶם וְעוּגוֹת. ״שְׂאוֹר״ הוּא בָּצֵק שֶׁהִשְׁאִירוּ זְמַן רַב, עַד שֶׁהֶחְמִיץ מְאוֹד וְנַעֲשָׂה חָמוּץ מְאוֹד – כְּמוֹ שְׁמָרִים. אֶת הַשְּׂאוֹר לֹא אוֹכְלִים, אֶלָּא מוֹסִיפִים אוֹתוֹ לְבָצֵק אַחֵר כְּדֵי שֶׁיַּחְמִיץ מַהֵר וְטוֹב. הַתּוֹרָה אָסְרָה אֶת שְׁנֵיהֶם בְּאוֹתָהּ מִדָּה.
(פֶּרֶק ב׳, הֲלָכָה ג׳)

9כֻּסְּמִין אוֹ כֻּסֶּמֶת?

שִׂימוּ לֵב לִשְׁנֵי שֵׁמוֹת דּוֹמִים: ״כֻּסְּמִין״ הוּא אֶחָד מֵחֲמֵשֶׁת מִינֵי הַדָּגָן, וְהוּא יָכוֹל לְהַחְמִיץ. ״כֻּסֶּמֶת״ הִיא מִין קִטְנִיּוֹת. מִי שֶׁאוֹכֵל קִטְנִיּוֹת בְּפֶסַח רַשַּׁאי לֶאֱכֹל כֻּסֶּמֶת, וְגַם לְמִי שֶׁלֹּא אוֹכֵל קִטְנִיּוֹת – הִיא מֻתֶּרֶת לְחוֹלִים.
🤔 לָמָּה? יֵשׁ אֲנָשִׁים שֶׁמַּחֲלִיפִים בֵּין הַשֵּׁמוֹת, וְקוֹרְאִים לַדָּגָן ״כֻּסֶּמֶת״. לָכֵן צָרִיךְ לִבְדֹּק הֵיטֵב מָה בֶּאֱמֶת כָּתוּב עַל הַשַּׂקִּית.
(פֶּרֶק ב׳, הֲלָכָה ג׳)

10שְׁמוֹנֶה עֶשְׂרֵה דַּקּוֹת

כָּל זְמַן שֶׁלָּשִׁים אֶת הַבָּצֵק, הוּא לֹא מַחְמִיץ – אֲפִלּוּ אִם הַלִּישָׁה נִמְשֶׁכֶת יוֹם שָׁלֵם. אֲבָל בָּצֵק שֶׁל קֶמַח וּמַיִם שֶׁעוֹמֵד 18 דַּקּוֹת בְּלִי תְּנוּעָה – מַתְחִיל לְהַחְמִיץ, וְכָל אִסּוּרֵי חָמֵץ חָלִים עָלָיו. בְּמָקוֹם חַם הוּא מַחְמִיץ עוֹד יוֹתֵר מַהֵר.
🤔 לָמָּה? תְּנוּעַת הַלִּישָׁה עוֹצֶרֶת אֶת הַתְּסִיסָה בַּבָּצֵק. לָכֵן כְּשֶׁאוֹפִים מַצּוֹת – מִזְדָּרְזִים מְאוֹד, וְלֹא מַשְׁאִירִים אֶת הַבָּצֵק לְרֶגַע בְּלִי עֲבוֹדָה.
💛 הַזְּרִיזוּת בַּאֲפִיַּת הַמַּצּוֹת מַזְכִּירָה לָנוּ אֶת הַזְּרִיזוּת שֶׁל יְצִיאַת מִצְרַיִם – כְּשֶׁה׳ קוֹרֵא, לֹא מְחַכִּים.
(פֶּרֶק ב׳, הֲלָכָה ד׳)

11בָּצֵק שֶׁנִּסְדַּק וּבָצֵק שֶׁהִכְסִיף

אִם רוֹאִים סְדָקִים בַּבָּצֵק – הוּא כְּבָר הֶחְמִיץ, גַּם אִם לֹא עָבְרוּ 18 דַּקּוֹת. אֲפִלּוּ אִם הַסְּדָקִים מְעַטִּים, בְּחֵלֶק קָטָן שֶׁל הַבָּצֵק – כָּל הַבָּצֵק חָמֵץ. וְאִם לֹא נִרְאוּ סְדָקִים, רַק הַבָּצֵק ״הִכְסִיף״ – כְּלוֹמַר, פָּנָיו נַעֲשׂוּ חִוְּרִים – הוּא ״חָמֵץ נֻקְשֶׁה״, שֶׁחֲכָמִים אָסְרוּ אוֹתוֹ.
🤔 לָמָּה? כְּנִרְאֶה הַמָּקוֹם הָיָה חַם, אוֹ שֶׁחֵלֶק מֵהַבָּצֵק לֹא נִלַּשׁ הֵיטֵב – וְשָׁם הִתְחִיל הַחִמּוּץ.
(פֶּרֶק ב׳, הֲלָכָה ד׳, הֲלָכָה ה׳)

12חָמֵץ נֻקְשֶׁה

״חָמֵץ נֻקְשֶׁה״ הוּא דָּבָר שֶׁהִתְחִיל לְהַחְמִיץ וְלֹא גָּמַר, וְאֶפְשָׁר לֶאֱכֹל אוֹתוֹ רַק בִּשְׁעַת דֹּחַק. לְמָשָׁל: דֶּבֶק שֶׁפַּעַם הָיוּ מְכִינִים מִקֶּמַח וּמַיִם. לְפִי רֹב הַפּוֹסְקִים אֵין בּוֹ אִסּוּר תּוֹרָה, אֲבָל חֲכָמִים אָסְרוּ לֶאֱכֹל אוֹתוֹ וּלְהַשְׁאִיר אוֹתוֹ בְּפֶסַח, כְּדֵי שֶׁלֹּא יָבוֹאוּ לְהִתְבַּלְבֵּל עִם חָמֵץ אֲמִתִּי. וְאִם הִדְבִּיקוּ בַּדֶּבֶק הַזֶּה נְיָרוֹת, וְהוּא שִׁנָּה אֶת צוּרָתוֹ – יֵשׁ שְׁתֵּי דֵּעוֹת:
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְפִי הָעִקָּר מֻתָּר לְהַשְׁאִיר אֶת הַנְּיָרוֹת הַמְּדֻבָּקִים בְּפֶסַח, כִּי הַדֶּבֶק כְּבָר שִׁנָּה אֶת צוּרָתוֹ.
יֵשׁ מַחְמִירִים (הָרָמָ״א) אִם הַדֶּבֶק מְבַצְבֵּץ מִבֵּין הַנְּיָרוֹת – צוּרָתוֹ עוֹמֶדֶת, וְאָסוּר לְהַשְׁאִיר אוֹתוֹ בְּפֶסַח.
(פֶּרֶק ב׳, הֲלָכָה ה׳)

13חָמֵץ שֶׁהִתְקַלְקֵל

לֶחֶם שֶׁהִתְעַפֵּשׁ, וְאָדָם כְּבָר לֹא יֹאכַל אוֹתוֹ – עֲדַיִן חָמֵץ גָּמוּר. רַק חָמֵץ שֶׁהִתְקַלְקֵל כָּל כָּךְ עַד שֶׁאֲפִלּוּ כֶּלֶב לֹא יֹאכַל אוֹתוֹ – כְּבָר אֵינוֹ נֶחְשָׁב אֹכֶל, וְאֵין עָלָיו דִּין חָמֵץ. אֲבָל זֶה רַק אִם הוּא נִפְסַל לִפְנֵי שֶׁהִגִּיעַ זְמַן הָאִסּוּר. אִם בִּזְמַן הָאִסּוּר הוּא עוֹד הָיָה רָאוּי לְכֶלֶב – חוֹבָה לְבַעֵר אוֹתוֹ.
🤔 לָמָּה? מַאֲכָלִים אֲסוּרִים אֲחֵרִים מְאַבְּדִים אֶת אִסּוּרָם כְּשֶׁהֵם מִתְקַלְקְלִים. אֲבָל חָמֵץ מְקֻלְקָל עֲדַיִן יָכוֹל לְהַחְמִיץ בָּצֵק אַחֵר, כְּמוֹ שְׂאוֹר – וְלָכֵן הוּא נִשְׁאָר חָמֵץ.
(פֶּרֶק ב׳, הֲלָכָה ו׳)

14הָאֹכֶל שֶׁל הַכֶּלֶב וְהֶחָתוּל

אֹכֶל לִכְלָבִים וּלְחָתוּלִים שֶׁיֵּשׁ בּוֹ חָמֵץ – חוֹבָה לְבַעֵר לִפְנֵי פֶּסַח, וְאָסוּר לְהַאֲכִיל בּוֹ אֶת בַּעֲלֵי הַחַיִּים בְּפֶסַח. לְבַעֲלֵי חַיִּים שֶׁבַּבַּיִת קוֹנִים לְפֶסַח אֹכֶל שֶׁאֵין בּוֹ חָמֵץ.
🤔 לָמָּה? הֶחָמֵץ שֶׁבְּאֹכֶל בַּעֲלֵי הַחַיִּים הָיָה מִתְּחִלָּתוֹ רָאוּי לְמַאֲכַל אָדָם – קֶמַח, לֶחֶם אוֹ דָּגָן. לָכֵן הוּא חָמֵץ לְכָל דָּבָר.
(פֶּרֶק ב׳, הֲלָכָה ו׳)

📝 מָה לָמַדְנוּ

  • בְּחָמֵץ יֵשׁ אַרְבַּע מִצְווֹת: לֹא לֶאֱכֹל, ״לֹא יֵרָאֶה״, ״לֹא יִמָּצֵא״, וּלְהַשְׁבִּית אֶת הֶחָמֵץ.
  • בְּעֶרֶב פֶּסַח אוֹכְלִים חָמֵץ עַד סוֹף הַשָּׁעָה הָרְבִיעִית, נֶהֱנִים מִמֶּנּוּ עַד סוֹף הַחֲמִישִׁית, וּמֵחֲצוֹת הַיּוֹם – אִסּוּר תּוֹרָה.
  • חָמֵץ הוּא אֶחָד מֵחֲמֵשֶׁת מִינֵי דָּגָן שֶׁנָּגַע בְּמַיִם וְהֶחְמִיץ; בָּצֵק שֶׁעוֹמֵד 18 דַּקּוֹת בְּלִי לִישָׁה – מַחְמִיץ.
  • חָמֵץ שֶׁל יְהוּדִי שֶׁעָבַר עָלָיו הַפֶּסַח – אָסוּר גַּם אַחֲרֵי הֶחָג.
  • רַק חָמֵץ שֶׁאֲפִלּוּ כֶּלֶב לֹא יֹאכַל אוֹתוֹ אֵינוֹ חָמֵץ; אֹכֶל לְבַעֲלֵי חַיִּים עִם חָמֵץ – מְבַעֲרִים.

❓ שְׁאֵלוֹת חֲזָרָה

  1. מָה הֵן אַרְבַּע הַמִּצְווֹת שֶׁל חָמֵץ, וְאֵיזוֹ מֵהֶן הִיא מִצְוַת עֲשֵׂה?
  2. מָה מֻתָּר וּמָה אָסוּר לַעֲשׂוֹת בֶּחָמֵץ בַּשָּׁעָה הַחֲמִישִׁית שֶׁל עֶרֶב פֶּסַח?
  3. לֶחֶם שֶׁהִתְעַפֵּשׁ – הַאִם הוּא עֲדַיִן חָמֵץ? הַסְבֵּר לָמָּה.
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״חָמֵץ וּמַצָּה – מָה הַסּוֹד?״

סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת פֶּסַח
מַהֲדוּרַת יְלָדִים

📖 הַהַגָּדָה

פֶּרֶק ט״ו · לַמַּהֲדוּרָה לַמּוֹרֶה
יֵשׁ סִפּוּר אֶחָד שֶׁכָּל אַבָּא וְכָל אִמָּא בְּיִשְׂרָאֵל מְסַפְּרִים לַיְּלָדִים – בְּאוֹתוֹ לַיְלָה, כְּבָר יוֹתֵר מִשְּׁלֹשֶׁת אֲלָפִים שָׁנָה. סָבָא שֶׁל סָבָא שֶׁלְּךָ שָׁמַע אוֹתוֹ, וְגַם הַנְּכָדִים שֶׁלְּךָ יִשְׁמְעוּ אוֹתוֹ. זֶה סִפּוּר יְצִיאַת מִצְרַיִם. בַּפֶּרֶק הַזֶּה נִלְמַד לָמָּה דַּוְקָא אַתֶּם, הַיְּלָדִים, נִמְצָאִים בַּמֶּרְכָּז שֶׁל לֵיל הַסֵּדֶר – וְלָמָּה הַשְּׁאֵלוֹת שֶׁלָּכֶם כָּל כָּךְ חֲשׁוּבוֹת.

1״וְהִגַּדְתָּ לְבִנְךָ״

מִצְוַת עֲשֵׂה מִן הַתּוֹרָה לְסַפֵּר בִּיצִיאַת מִצְרַיִם בְּלֵיל ט״ו בְּנִיסָן. עִקַּר הַמִּצְוָה – לְסַפֵּר לַיְּלָדִים, כְּמוֹ שֶׁכָּתוּב: ״וְהִגַּדְתָּ לְבִנְךָ בַּיּוֹם הַהוּא לֵאמֹר, בַּעֲבוּר זֶה עָשָׂה ה׳ לִי בְּצֵאתִי מִמִּצְרָיִם״. גַּם מִי שֶׁאֵין לוֹ יְלָדִים מְצֻוֶּה לִזְכֹּר וּלְסַפֵּר. וְכָל הַמַּרְבֶּה לְסַפֵּר בִּיצִיאַת מִצְרַיִם – עַל הַחֲסָדִים, הַנִּסִּים וְהַנִּפְלָאוֹת, וּלְהוֹדוֹת לַה׳ – הֲרֵי זֶה מְשֻׁבָּח.
💛 ״בַּעֲבוּר זֶה עָשָׂה ה׳ לִי״ – לֹא רַק לְאַבְרָהָם אוֹ לְמֹשֶׁה, אֶלָּא גַּם לִי.
(פֶּרֶק ט״ו, הֲלָכָה א׳)

2לָמָּה דַּוְקָא לַיְּלָדִים?

בִּשְׁתֵּי מִצְווֹת הַתּוֹרָה מְצַוָּה אֶת הַהוֹרִים לְחַנֵּךְ אֶת הַיְּלָדִים: לְלַמֵּד אוֹתָם תּוֹרָה, וּלְסַפֵּר לָהֶם בִּיצִיאַת מִצְרַיִם בְּלֵיל הַסֵּדֶר. בְּלֵיל הַסֵּדֶר מַעֲבִירִים לַיְּלָדִים אֶת הַיְּסוֹדוֹת שֶׁל הָאֱמוּנָה: אֵיךְ נוֹצַר עַם יִשְׂרָאֵל, וְשֶׁה׳ בָּחַר בָּנוּ לִהְיוֹת עַם סְגֻלָּה, לְקַבֵּל אֶת הַתּוֹרָה וּלְתַקֵּן אֶת הָעוֹלָם.
🤔 לָמָּה? הַהוֹרִים לֹא יִחְיוּ לָעַד, וְהַיְּלָדִים הֵם שֶׁיַּמְשִׁיכוּ לָשֵׂאת אֶת ״לַפִּיד הַמָּסֹרֶת״ – אֶת הַתַּפְקִיד הַגָּדוֹל שֶׁל עַם יִשְׂרָאֵל. וּכְדֵי לְסַפֵּר טוֹב, הַהוֹרִים עַצְמָם לוֹמְדִים וּמִתְעַמְּקִים – וּמְשַׁמְּשִׁים דֻּגְמָה.
💛 בְּלֵיל הַסֵּדֶר אַתָּה לֹא רַק שׁוֹמֵעַ סִפּוּר – אַתָּה מְקַבֵּל אֶת הַלַּפִּיד לְיָדֶיךָ.
(פֶּרֶק ט״ו, הֲלָכָה א׳)

3כָּל יוֹם זוֹכְרִים – הַלַּיְלָה מְסַפְּרִים

בְּכָל יוֹם וּבְכָל לַיְלָה אָנוּ זוֹכְרִים אֶת יְצִיאַת מִצְרַיִם בִּקְרִיאַת שְׁמַע, בַּפָּרָשָׁה ״וַיֹּאמֶר״. אֲבָל בְּלֵיל הַסֵּדֶר הַמִּצְוָה שׁוֹנָה: מְסַפְּרִים בַּאֲרִיכוּת, וְלֹא רַק מַזְכִּירִים; מְסַפְּרִים לַיְּלָדִים; מְסַפְּרִים כְּשֶׁהַמַּצָּה וְהַמָּרוֹר מֻנָּחִים לְפָנֵינוּ; וּמְסַפְּרִים בְּדֶרֶךְ שֶׁל שְׁאֵלָה וּתְשׁוּבָה. וְעוֹד הֶבְדֵּל: מִמִּצְוַת הַזִּכָּרוֹן שֶׁבְּכָל יוֹם נָשִׁים פְּטוּרוֹת, אֲבָל בְּמִצְוַת הַסִּפּוּר בְּלֵיל הַסֵּדֶר – נָשִׁים וּבָנוֹת חַיָּבוֹת.
🤔 לָמָּה? יְצִיאַת מִצְרַיִם הִיא הַיְּסוֹד שֶׁל הָאֱמוּנָה: אָז רָאָה עַם שָׁלֵם אֶת הַשְׁגָּחַת ה׳. לָכֵן כָּל הַשַּׁבָּתוֹת וְהַחַגִּים הֵם ״זֵכֶר לִיצִיאַת מִצְרַיִם״ – וְהַסִּפּוּר הַגָּדוֹל שֶׁל לֵיל הַסֵּדֶר נוֹתֵן מַשְׁמָעוּת לְכָל הַזְּכִירוֹת הַקְּטַנּוֹת בְּמֶשֶׁךְ הַשָּׁנָה.
(פֶּרֶק ט״ו, הֲלָכָה ב׳)

4פּוֹתְחִים בִּשְׁאֵלָה

מְסַפְּרִים בִּיצִיאַת מִצְרַיִם בְּדֶרֶךְ שֶׁל שְׁאֵלָה וּתְשׁוּבָה, כְּמוֹ שֶׁכָּתוּב: ״כִּי יִשְׁאָלְךָ בִנְךָ מָחָר...״. לָכֵן חֲכָמִים תִּקְּנוּ הַרְבֵּה שִׁנּוּיִים בַּסֵּדֶר: בַּהַתְחָלָה מְחַלְּקִים לַיְּלָדִים אֱגוֹזִים וּמִינֵי מְתִיקָה; אַחֲרֵי הַקִּדּוּשׁ נוֹטְלִים יָדַיִם וְאוֹכְלִים כַּרְפַּס טָבוּל; וְאַחַר כָּךְ, בִּמְקוֹם לְהַתְחִיל לֶאֱכֹל, מוֹזְגִים כּוֹס שְׁנִיָּה וּמְפַנִּים אֶת הַקְּעָרָה מֵהַשֻּׁלְחָן.
🤔 לָמָּה? שְׁאֵלָה פּוֹתַחַת אֶת הַלֵּב וְאֶת הַמֹּחַ לְקַבֵּל אֶת הַתְּשׁוּבָה. כְּשֶׁהַיְּלָדִים תְּמֵהִים בֶּאֱמֶת, הֵם שׁוֹאֲלִים ״מַה נִּשְׁתַּנָּה״ מֵעֹמֶק הַלֵּב – וּמַקְשִׁיבִים לַתְּשׁוּבָה.
💛 שְׁאֵלָה טוֹבָה הִיא כְּמוֹ דֶּלֶת – וּמֵאֲחוֹרֶיהָ מְחַכָּה הַהִכָּרוּת עִם ה׳.
(פֶּרֶק ט״ו, הֲלָכָה ג׳)

5מַה נִּשְׁתַּנָּה

חֲכָמִים קָבְעוּ נֹסַח קָבוּעַ לַשְּׁאֵלָה: ״מַה נִּשְׁתַּנָּה הַלַּיְלָה הַזֶּה מִכָּל הַלֵּילוֹת״. שׁוֹאֲלִים עַל מִצְווֹת הָאֲכִילָה שֶׁל הַלַּיְלָה – הַמַּצָּה וְהַמָּרוֹר – וְעַל הַטִּבּוּל פַּעֲמַיִם. אַחֲרֵי חֻרְבַּן הַמִּקְדָּשׁ כְּבָר לֹא שׁוֹאֲלִים עַל קָרְבַּן הַפֶּסַח, וּבִמְקוֹמוֹ הוֹסִיפוּ שְׁאֵלָה עַל הַהֲסִבָּה. כְּשֶׁאֵין יְלָדִים – שׁוֹאֵל הַצָּעִיר שֶׁבַּמְּסֻבִּים. גַּם כְּשֶׁכֻּלָּם חֲכָמִים – אֶחָד מֵהֶם שׁוֹאֵל. וַאֲפִלּוּ מִי שֶׁעוֹשֶׂה סֵדֶר לְבַד – שׁוֹאֵל אֶת עַצְמוֹ. וּמִשֶּׁאֶחָד שָׁאַל, הַשְּׁאָר לֹא צְרִיכִים לַחֲזֹר עַל הַשְּׁאֵלוֹת, וּמִיָּד מַתְחִילִים ״עֲבָדִים הָיִינוּ״.
🤔 לָמָּה? כָּךְ הוּא הַסֵּדֶר: פּוֹתְחִים בִּשְׁאֵלָה, וְאָז הַסִּפּוּר שָׁלֵם יוֹתֵר.
(פֶּרֶק ט״ו, הֲלָכָה ד׳)

6אַרְבָּעָה בָּנִים

בַּתּוֹרָה כָּתוּב אַרְבַּע פְּעָמִים שֶׁצָּרִיךְ לְסַפֵּר לַבֵּן עַל יְצִיאַת מִצְרַיִם – וּבְכָל פַּעַם בְּלָשׁוֹן אַחֶרֶת. מִכָּאן לָמַדְנוּ לְהַסְבִּיר לְכָל יֶלֶד לְפִי הַכִּשְׁרוֹנוֹת וְהַתְּכוּנוֹת שֶׁלּוֹ. הֶחָכָם שׁוֹאֵל בְּפֵרוּט – וְעוֹנִים לוֹ בַּאֲרִיכוּת. הָרָשָׁע אוֹמֵר ״מָה הָעֲבֹדָה הַזֹּאת לָכֶם״, כִּי הוּא לֹא מַרְגִּישׁ שֻׁתָּף – וּבְכָל זֹאת מַסְבִּירִים לוֹ אֶת סְגֻלַּת יִשְׂרָאֵל. הַתָּם שׁוֹאֵל ״מַה זֹּאת?״ – וּמְסַפְּרִים לוֹ עַל הַמַּכּוֹת הַגְּדוֹלוֹת וְעַל לֵב פַּרְעֹה הַקָּשֶׁה. וְלַבֵּן שֶׁאֵינוֹ יוֹדֵעַ לִשְׁאֹל – מְעוֹרְרִים אוֹתוֹ בִּדְבָרִים מוּחָשִׁיִּים: ״בַּעֲבוּר זֶה״ – בִּזְכוּת הַמַּצָּה וְהַמָּרוֹר שֶׁלְּפָנֵינוּ.
🤔 לָמָּה? גַּם לָרָשָׁע מְסַפְּרִים, כִּי תָּמִיד צָרִיךְ לְהַאֲמִין שֶׁאוּלַי הַדְּבָרִים יִכָּנְסוּ לְלִבּוֹ. וּבִשְׁבִיל הַבֵּן שֶׁאֵינוֹ יוֹדֵעַ לִשְׁאֹל מַנִּיחִים אֶת הַקְּעָרָה עַל הַשֻּׁלְחָן – שֶׁכָּל מַאֲכָל יַמְחִישׁ לוֹ אֶת הַסִּפּוּר.
💛 לְכָל יֶלֶד יֵשׁ דֶּרֶךְ מִשֶּׁלּוֹ אֶל ה׳, וְאַף אֶחָד לֹא נִשְׁאָר בַּחוּץ.
(פֶּרֶק ט״ו, הֲלָכָה ה׳)

7הַתְּשׁוּבָה לֶחָכָם – הַמֶּסֶר שֶׁל הַלַּיְלָה

בַּתְּשׁוּבָה לַבֵּן הֶחָכָם מֻסְתָּר הַמֶּסֶר הַמֶּרְכָּזִי שֶׁל לֵיל הַסֵּדֶר: ״עֲבָדִים הָיִינוּ לְפַרְעֹה בְּמִצְרָיִם, וַיּוֹצִיאֵנוּ ה׳ מִמִּצְרַיִם בְּיָד חֲזָקָה... וְאוֹתָנוּ הוֹצִיא מִשָּׁם לְמַעַן הָבִיא אֹתָנוּ לָתֶת לָנוּ אֶת הָאָרֶץ... וַיְצַוֵּנוּ ה׳ לַעֲשׂוֹת אֶת כָּל הַחֻקִּים הָאֵלֶּה״. הַמַּטָּרָה הִיא שֶׁמִּתּוֹךְ הַסִּפּוּר יִהְיֶה בְּלֵב הַבָּנִים וְהַבָּנוֹת חֵשֶׁק לִהְיוֹת חֵלֶק מֵעַם יִשְׂרָאֵל, לָגוּר בָּאָרֶץ הַמֻּבְטַחַת, לְהִדָּבֵק בַּה׳ וּלְקַיֵּם אֶת מִצְווֹתָיו.
🤔 לָמָּה? כְּדֵי שֶׁלֹּא נַשְׁמִיט שׁוּם חֵלֶק חָשׁוּב מֵהַסִּפּוּר, אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה – בִּתְחִלַּת יְמֵי בַּיִת שֵׁנִי – תִּקְּנוּ אֶת נֹסַח הַהַגָּדָה. הַתַּנָּאִים, הָאָמוֹרָאִים וְהַגְּאוֹנִים הוֹסִיפוּ עָלָיו, וְלִפְנֵי כִּשְׁמוֹנֶה מֵאוֹת שָׁנָה הִתְקַבֵּל נֹסַח אֶחָד בְּכָל קְהִלּוֹת יִשְׂרָאֵל.
💛 הַתְּפִלָּה שֶׁלָּנוּ הִיא שֶׁכָּל הַבָּנִים יִגְדְּלוּ וְיִתְעַלּוּ – עַד שֶׁכֻּלָּם יִהְיוּ חֲכָמִים.
(פֶּרֶק ט״ו, הֲלָכָה ו׳)

8מַתְחִילִים בִּגְנוּת – וּמְסַיְּמִים בְּשֶׁבַח

אֶת סִפּוּר יְצִיאַת מִצְרַיִם פּוֹתְחִים בִּגְנוּת – בְּמָה שֶׁהָיָה לֹא טוֹב – וּמְסַיְּמִים בְּשֶׁבַח. וּלְמַעֲשֶׂה אוֹמְרִים בַּהַגָּדָה אֶת שְׁתֵּי הַ״גְּנוּיוֹת״: גַּם ״עֲבָדִים הָיִינוּ״, וְגַם ״מִתְּחִלָּה עוֹבְדֵי עֲבוֹדָה זָרָה הָיוּ אֲבוֹתֵינוּ״ – וּמְסַפְּרִים אֵיךְ יָצָאנוּ מֵעַבְדוּת לְחֵרוּת, וּמֵעֲבוֹדָה זָרָה לֶאֱמוּנָה שְׁלֵמָה.
🤔 לָמָּה? מִתּוֹךְ הַחֹשֶׁךְ רוֹאִים טוֹב יוֹתֵר אֶת הָאוֹר: כְּכָל שֶׁנָּבִין כַּמָּה קָשֶׁה הָיָה, נָבִין כַּמָּה גְּדוֹלָה הַגְּאֻלָּה. וְעוֹד: ה׳ בָּחַר בָּנוּ גַּם כְּשֶׁהָיִינוּ עֲבָדִים – לֹא בִּגְלַל הַהַצְלָחוֹת שֶׁלָּנוּ. וְגַם: מִי שֶׁזּוֹכֵר שֶׁהָיָה עֶבֶד, מְרַחֵם יוֹתֵר עַל הַגֵּר וְעַל כָּל מִי שֶׁסּוֹבֵל.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
יֵשׁ אוֹמְרִים (שְׁמוּאֵל) הַגְּנוּת הִיא שֶׁל הַגּוּף: ״עֲבָדִים הָיִינוּ לְפַרְעֹה בְּמִצְרָיִם״.
וְיֵשׁ אוֹמְרִים (רַב) הַגְּנוּת הִיא שֶׁל הָרוּחַ: ״מִתְּחִלָּה עוֹבְדֵי עֲבוֹדָה זָרָה הָיוּ אֲבוֹתֵינוּ״.
💛 ה׳ אוֹהֵב אוֹתָנוּ לֹא בִּגְלַל שֶׁאֲנַחְנוּ מֻשְׁלָמִים – הוּא בָּחַר בָּנוּ, וְהוּא מוֹצִיא אוֹתָנוּ מִכָּל חֹשֶׁךְ אֶל הָאוֹר.
(פֶּרֶק ט״ו, הֲלָכָה ז׳)

9עֶשֶׂר הַמַּכּוֹת

הַתּוֹרָה מְתָאֶרֶת אֶת עֶשֶׂר הַמַּכּוֹת בְּפֵרוּט, בְּלִי לְדַלֵּג עַל אַף אַחַת מֵהֶן – וְגַם אֲנַחְנוּ מוֹנִים אוֹתָן בַּהַגָּדָה. פַּרְעֹה אָמַר: ״מִי ה׳ אֲשֶׁר אֶשְׁמַע בְּקֹלוֹ?״, וְהִקְשָׁה אֶת לִבּוֹ שׁוּב וָשׁוּב. אֲבָל אַף מֶלֶךְ – גַּם לֹא שֶׁל הַמַּמְלָכָה הֶחָזָקָה בָּעוֹלָם – לֹא יָכוֹל לַעֲמֹד מוּל דְּבַר ה׳. מַכָּה אַחַר מַכָּה נִשְׁבַּר כֹּחַ מִצְרַיִם, וְיִשְׂרָאֵל יָצְאוּ לְחֵרוּת.
🤔 לָמָּה? הַלֶּקַח הַפָּשׁוּט: יֵשׁ דִּין וְיֵשׁ דַּיָּן בָּעוֹלָם, וּבְסוֹף הָרָשָׁע מְקַבֵּל אֶת עָנְשׁוֹ. וְיֵשׁ גַּם רֶמֶז עָמֹק: הָעוֹלָם נִבְרָא בַּעֲשָׂרָה מַאֲמָרוֹת. אוֹתָם עֲשָׂרָה כֹּחוֹת הִתְגַּלּוּ בְּמִצְרַיִם כְּעֶשֶׂר מַכּוֹת, וּבְהַר סִינַי – כַּעֲשֶׂרֶת הַדִּבְּרוֹת.
(פֶּרֶק ט״ו, הֲלָכָה ח׳)

10פֶּסַח, מַצָּה וּמָרוֹר

רַבָּן גַּמְלִיאֵל הָיָה אוֹמֵר: ״כָּל שֶׁלֹּא אָמַר שְׁלֹשָׁה דְּבָרִים אֵלּוּ בַּפֶּסַח – לֹא יָצָא יְדֵי חוֹבָתוֹ, וְאֵלּוּ הֵן: פֶּסַח, מַצָּה וּמָרוֹר״. גַּם מִי שֶׁלֹּא יָכוֹל לוֹמַר אֶת כָּל הַהַגָּדָה – צָרִיךְ לְפָחוֹת לְהַסְבִּיר אֶת שְׁלֹשֶׁת הַדְּבָרִים הָאֵלֶּה. לָכֵן שׁוֹאֲלִים בַּהַגָּדָה: ״פֶּסַח זֶה... עַל שׁוּם מָה?״, ״מַצָּה זוֹ... עַל שׁוּם מָה?״, ״מָרוֹר זֶה... עַל שׁוּם מָה?״. הַפֶּסַח מְבַטֵּא אֶת סְגֻלַּת יִשְׂרָאֵל – ה׳ פָּסַח עַל בָּתֵּינוּ; הַמַּצָּה – אֶת הַחֵרוּת; וְהַמָּרוֹר – אֶת מְרִירוּת הַשִּׁעְבּוּד.
🤔 לָמָּה? הַמַּצָּה הִיא מִצְוָה מִן הַתּוֹרָה גַּם הַיּוֹם, כִּי הַחֵרוּת שֶׁקִּבַּלְנוּ בִּיצִיאַת מִצְרַיִם נִשְׁאֲרָה לָנֶצַח: ״אֵין לְךָ בֶּן חוֹרִין אֶלָּא מִי שֶׁעוֹסֵק בַּתּוֹרָה״. גַּם כְּשֶׁהַגּוּף מְשֻׁעְבָּד – הָרוּחַ שֶׁלָּנוּ חָפְשִׁית.
💛 גַּם בַּזְּמַנִּים הַקָּשִׁים – כְּמוֹ הַמָּרוֹר – אָנוּ מַאֲמִינִים שֶׁה׳ מוֹבִיל אוֹתָנוּ לְטוֹבָה.
(פֶּרֶק ט״ו, הֲלָכָה ט׳)

📝 מָה לָמַדְנוּ

  • מִצְוָה מִן הַתּוֹרָה לְסַפֵּר בִּיצִיאַת מִצְרַיִם בְּלֵיל הַסֵּדֶר, וְעִקָּרָהּ – לְסַפֵּר לַיְּלָדִים. גַּם נָשִׁים וּבָנוֹת חַיָּבוֹת.
  • מְסַפְּרִים בְּדֶרֶךְ שֶׁל שְׁאֵלָה וּתְשׁוּבָה, וּמְשַׁנִּים דְּבָרִים כְּדֵי שֶׁהַיְּלָדִים יִשְׁאֲלוּ ״מַה נִּשְׁתַּנָּה״.
  • מַסְבִּירִים לְכָל בֵּן לְפִי דַּרְכּוֹ – חָכָם, רָשָׁע, תָּם וְשֶׁאֵינוֹ יוֹדֵעַ לִשְׁאֹל.
  • פּוֹתְחִים בִּגְנוּת וּמְסַיְּמִים בְּשֶׁבַח, וּמַסְבִּירִים פֶּסַח, מַצָּה וּמָרוֹר.

❓ שְׁאֵלוֹת חֲזָרָה

  1. בְּמָה שׁוֹנָה מִצְוַת הַסִּפּוּר בְּלֵיל הַסֵּדֶר מֵהַזִּכָּרוֹן שֶׁל יְצִיאַת מִצְרַיִם בְּכָל יוֹם? מְנֵה לְפָחוֹת שְׁלֹשָׁה הֶבְדֵּלִים.
  2. לָמָּה מְחַלְּקִים לַיְּלָדִים אֱגוֹזִים וּמְפַנִּים אֶת הַקְּעָרָה? וּמִי שׁוֹאֵל ״מַה נִּשְׁתַּנָּה״ כְּשֶׁאֵין יְלָדִים?
  3. מָה אָמַר רַבָּן גַּמְלִיאֵל, וּמָה מְבַטֵּא כָּל אֶחָד מִשְּׁלֹשֶׁת הַדְּבָרִים?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״וְהִגַּדְתָּ לְבִנְךָ״

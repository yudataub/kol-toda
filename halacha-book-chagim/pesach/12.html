סֵפֶר הַהֲלָכוֹת ‹ הִלְכוֹת פֶּסַח
מַהֲדוּרַת יְלָדִים

🫓 הִלְכוֹת מַצָּה

פֶּרֶק י״ב · לַמַּהֲדוּרָה לַמּוֹרֶה
קֶמַח וּמַיִם – זֶה הַכֹּל. מָה יָכוֹל לִהְיוֹת פָּשׁוּט יוֹתֵר מִמַּצָּה? אֲבָל דַּעְתֶּם שֶׁעַל הַמַּצָּה שׁוֹמְרִים כְּבָר מֵהַשָּׂדֶה, כְּשֶׁהַחִטָּה עוֹד עוֹמֶדֶת בַּשִּׁבֳּלִים? בַּפֶּרֶק הַזֶּה נֵצֵא לְמַסָּע: מֵהַקְּצִירָה, דֶּרֶךְ הַמַּיִם וְהַלִּישָׁה, וְעַד הַתַּנּוּר הַלּוֹהֵט. נִלְמַד אֵיךְ נוֹלֶדֶת מַצָּה כְּשֵׁרָה – וְלָמָּה שׁוֹמְרִים עָלֶיהָ כָּל כָּךְ.

1מִצְוָה מִן הַתּוֹרָה: לֶאֱכֹל מַצָּה

בְּלֵיל ט״ו בְּנִיסָן – לֵיל הַסֵּדֶר – יֵשׁ מִצְוָה מִן הַתּוֹרָה לֶאֱכֹל מַצָּה, כְּמוֹ שֶׁכָּתוּב: ״בָּעֶרֶב תֹּאכְלוּ מַצֹּת״. בִּשְׁאָר יְמֵי הַפֶּסַח אֵין חוֹבָה לֶאֱכֹל מַצָּה, אֲבָל הִיא הַמַּאֲכָל הָעִקָּרִי שֶׁלָּנוּ בִּמְקוֹם הַלֶּחֶם. וְהַרְבֵּה מִגְּדוֹלֵי יִשְׂרָאֵל כָּתְבוּ: מִי שֶׁאוֹכֵל מַצָּה גַּם בִּשְׁאָר יְמֵי הַפֶּסַח – מְקַיֵּם מִצְוָה, אַף שֶׁאֵין זוֹ חוֹבָה.
🤔 לָמָּה? בְּלֵיל הַסֵּדֶר הַמַּצָּה הִיא חוֹבָה, כִּי בַּלַּיְלָה הַזֶּה יָצָאנוּ מִמִּצְרַיִם. וּבִשְׁאָר הַיָּמִים – כָּל אֶחָד יָכוֹל לְהַחְלִיט שֶׁהוּא מוֹסִיף מִצְוָה מֵרְצוֹנוֹ.
💛 כָּל כְּזַיִת מַצָּה בְּפֶסַח יָכוֹל לִהְיוֹת מַתָּנָה קְטַנָּה שֶׁאֲנִי נוֹתֵן לַה׳ מֵרְצוֹנִי הַטּוֹב.
(פֶּרֶק י״ב, הֲלָכָה א׳)

2מַצָּה שְׁמוּרָה

הַתּוֹרָה אוֹמֶרֶת: ״וּשְׁמַרְתֶּם אֶת הַמַּצּוֹת״. אֶת מַצּוֹת הַמִּצְוָה שֶׁל לֵיל הַסֵּדֶר שׁוֹמְרִים שְׁמִירָה מְיֻחֶדֶת, שֶׁלֹּא יַגִּיעוּ אֲלֵיהֶן מַיִם וְלֹא יַחְמִיצוּ. לְכַתְּחִלָּה שׁוֹמְרִים אוֹתָן כְּבָר מִשְּׁעַת הַקְּצִירָה: קוֹצְרִים אֶת הַחִטִּים כְּשֶׁהֵן עוֹד קְצָת לַחוֹת, וּמְאַחְסְנִים אוֹתָן בְּמָקוֹם שֶׁאֵין בּוֹ מַיִם. מִצַּד הַדִּין אֶפְשָׁר לָצֵאת גַּם בְּמַצָּה שֶׁנִּשְׁמְרָה מִשְּׁעַת הַטְּחִינָה, אֲבָל כַּיּוֹם כֻּלָּם מְהַדְּרִים, וְקוֹנִים לְלֵיל הַסֵּדֶר מַצּוֹת שְׁמוּרוֹת מִשְּׁעַת קְצִירָה. וְעוֹד דָּבָר: קֶמַח רָגִיל מֵהַחֲנוּת, בְּלִי הַשְׁגָּחָה לְפֶסַח – לֹא קוֹנִים לְפֶסַח.
🤔 לָמָּה? חִטָּה שֶׁנִּרְטְבָה עֲלוּלָה לְהַחְמִיץ. חִטִּים שֶׁהִתְיַבְּשׁוּ לְגַמְרֵי בַּשָּׂדֶה וְיָרַד עֲלֵיהֶן הַרְבֵּה גֶּשֶׁם – עֲלוּלוֹת לְהַחְמִיץ. וְלִפְנֵי שֶׁטּוֹחֲנִים קֶמַח רָגִיל, פְּעָמִים רַבּוֹת מַשְׁרִים אֶת הַחִטִּים בְּמַיִם.
(פֶּרֶק י״ב, הֲלָכָה ב׳)

3״לְשֵׁם מַצַּת מִצְוָה״

אֶת מַצּוֹת הַמִּצְוָה לָשִׁים וְאוֹפִים יְהוּדִים בְּנֵי מִצְוָה, שֶׁמְּכַוְּנִים שֶׁהַכֹּל נַעֲשֶׂה לְשֵׁם מַצַּת מִצְוָה. בִּתְחִלַּת הָעֲבוֹדָה אוֹמְרִים בַּפֶּה: ״לְשֵׁם מַצַּת מִצְוָה״, וּמִי שֶׁרַק חָשַׁב כָּךְ בַּלֵּב – בְּדִיעֲבַד יָצָא. אֶת הַקְּצִירָה וְאֶת הַטְּחִינָה מֻתָּר לַעֲשׂוֹת גַּם עַל יְדֵי גּוֹיִים, בִּתְנַאי שֶׁיְּהוּדִי עוֹמֵד לְיָדָם וּמַשְׁגִּיחַ שֶׁלֹּא יִהְיֶה חִמּוּץ.
🤔 לָמָּה? לַמִּלָּה ״וּשְׁמַרְתֶּם״ יֵשׁ שְׁתֵּי מַשְׁמָעֻיּוֹת: לִשְׁמֹר שֶׁלֹּא יִהְיֶה חִמּוּץ, וּלְכַוֵּן שֶׁהַמַּצָּה נַעֲשֵׂית בִּשְׁבִיל הַמִּצְוָה. לָכֵן אֶת הַלִּישָׁה וְאֶת הָאֲפִיָּה עוֹשִׂים גְּדוֹלִים שֶׁכְּבָר חַיָּבִים בְּמִצְווֹת, וְיוֹדְעִים לְכַוֵּן.
💛 גַּם פְּעֻלָּה פְּשׁוּטָה כְּמוֹ לִישַׁת בָּצֵק הוֹפֶכֶת לְמִצְוָה – כְּשֶׁאוֹמְרִים לְפָנֶיהָ: ״לְשֵׁם מִצְוָה״.
(פֶּרֶק י״ב, הֲלָכָה ג׳)

4מַצּוֹת יָד וּמַצּוֹת מְכוֹנָה

מַצּוֹת מְכוֹנָה כְּשֵׁרוֹת לְפֶסַח כְּשֶׁיֵּשׁ עֲלֵיהֶן הַשְׁגָּחָה טוֹבָה, וְגַם צַדִּיקִים וִירֵאִים אוֹכְלִים אוֹתָן בְּפֶסַח. אֲבָל הַאִם יוֹצְאִים בָּהֶן אֶת מִצְוַת אֲכִילַת מַצָּה בְּלֵיל הַסֵּדֶר? עַל זֶה נֶחְלְקוּ. לְמַעֲשֶׂה אֶפְשָׁר לָצֵאת בָּהֶן, וְרַבִּים מְהַדְּרִים לֶאֱכֹל בְּלֵיל הַסֵּדֶר מַצּוֹת יָד. וְדַי בְּכָךְ שֶׁהַכְּזֵיתִים שֶׁל הַמִּצְוָה יִהְיוּ מִמַּצּוֹת יָד – לֹא צָרִיךְ שֶׁכָּל הַסְּעוּדָה תִּהְיֶה מֵהֶן.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְדַעַת רֹב הַפּוֹסְקִים יוֹצְאִים בְּמַצַּת מְכוֹנָה, כִּי הָאָדָם מַפְעִיל אֶת הַמְּכוֹנָה לְשֵׁם מַצַּת מִצְוָה.
יֵשׁ אוֹמְרִים אֵין יוֹצְאִים בָּהּ בְּלֵיל הַסֵּדֶר, כִּי מְכוֹנָה לֹא יְכוֹלָה לְכַוֵּן.
(פֶּרֶק י״ב, הֲלָכָה ד׳)

5וּבְכָל יְמֵי הַפֶּסַח?

בִּשְׁאָר יְמֵי הַפֶּסַח אֶפְשָׁר לֶאֱכֹל מַצּוֹת רְגִילוֹת שֶׁכְּשֵׁרוֹת לְפֶסַח. הֵן שְׁמוּרוֹת מִשְּׁעַת הַטְּחִינָה, וּכְשֵׁרוֹת לְכַתְּחִלָּה. הַמְּהַדְּרִים אוֹכְלִים כָּל הַפֶּסַח מַצּוֹת שְׁמוּרוֹת מִשְּׁעַת קְצִירָה, כִּי בָּהֶן מְדַקְדְּקִים יוֹתֵר: עוֹצְרִים אֶת הַמְּכוֹנוֹת כָּל שְׁמוֹנֶה עֶשְׂרֵה דַּקּוֹת, וּמְנַקִּים אוֹתָן הֵיטֵב.
🤔 לָמָּה? מִכָּל הַמַּאֲכָלִים שֶׁלָּנוּ בְּפֶסַח, דַּוְקָא הַמַּצָּה הִיא הָעֲלוּלָה בְּיוֹתֵר לְהַחְמִיץ – הֲרֵי הִיא עֲשׂוּיָה מִקֶּמַח וּמַיִם. לָכֵן יֵשׁ מָקוֹם לְהִדּוּר.
(פֶּרֶק י״ב, הֲלָכָה ה׳)

6מַיִם שֶׁלָּנוּ

לָשִׁים אֶת הַמַּצּוֹת בְּ״מַיִם שֶׁלָּנוּ״: מַיִם שֶׁשּׁוֹאֲבִים בַּיּוֹם, וּמַשְׁאִירִים אוֹתָם לָלוּן כָּל הַלַּיְלָה בְּמָקוֹם קָרִיר. לֹא לָשִׁים מַצּוֹת בְּמַיִם פּוֹשְׁרִים. בַּאֲרָצוֹת חַמּוֹת, אִם הַמַּיִם הִתְחַמְּמוּ קְצָת – מַכְנִיסִים אוֹתָם לַמְּקָרֵר. וּבְמִפְעֲלֵי הַמַּצּוֹת לוֹקְחִים מַיִם מֵהַבֶּרֶז, מְסַנְּנִים אוֹתָם הֵיטֵב, וּמַשְׁאִירִים אוֹתָם לָלוּן בְּמָקוֹם צוֹנֵן.
🤔 לָמָּה? ״שֶׁלָּנוּ״ – מִלְּשׁוֹן ״לָלוּן״: מַיִם שֶׁלָּנוּ, כְּלוֹמַר שֶׁ״יָשְׁנוּ״ לַיְלָה שָׁלֵם. הַחֹם מְזָרֵז אֶת הַחִמּוּץ, וַאֲפִלּוּ מַיִם מִבּוֹר אוֹ מִמַּעְיָן עֲלוּלִים לִהְיוֹת חַמִּים מְעַט מֵחֹם הָאֲדָמָה. לָכֵן נוֹתְנִים לָהֶם לְהִצְטַנֵּן לַיְלָה שָׁלֵם.
(פֶּרֶק י״ב, הֲלָכָה ו׳)

7לָשִׁים בִּזְרִיזוּת וּבְקֹר

כְּשֶׁלָּשִׁים בָּצֵק לְמַצּוֹת, נִזְהָרִים מִכָּל דָּבָר שֶׁמְּחַמֵּם אוֹ מְעַכֵּב: לֹא שָׂמִים בָּעִסָּה מֶלַח אוֹ פִּלְפֵּל; לֹא לָשִׁים בַּשֶּׁמֶשׁ אוֹ לְיַד הַתַּנּוּר; וְלֹא מַשְׁאִירִים אֶת הָעִסָּה בְּלִי עֵסֶק אֲפִלּוּ רֶגַע. כְּשֶׁהַיָּדַיִם שֶׁל הַלָּשׁ מִתְחַמְּמוֹת – הוּא מְצַנֵּן אוֹתָן בְּמַיִם קָרִים. וְעוֹד: טוֹחֲנִים אֶת הַקֶּמַח לְפָחוֹת יוֹם אֶחָד לִפְנֵי הַלִּישָׁה, וְלֹא לָשִׁים בְּבַת אַחַת יוֹתֵר מִשִּׁעוּר חַלָּה – בְּעֵרֶךְ קִילוֹ וָחֵצִי.
🤔 לָמָּה? בָּצֵק שֶׁמִּתְחַמֵּם, אוֹ שֶׁעוֹמֵד בְּלִי שֶׁעוֹסְקִים בּוֹ, מַתְחִיל לְהַחְמִיץ. גַּם הַטְּחִינָה מְחַמֶּמֶת קְצָת אֶת הַקֶּמַח. וְאֶת הַכַּמּוּת מַגְבִּילִים, כִּי קָשֶׁה לְאָדָם אֶחָד לָלוּשׁ הֵיטֵב וּמַהֵר עִסָּה גְּדוֹלָה.
(פֶּרֶק י״ב, הֲלָכָה ז׳)

8בַּתַּנּוּר הַלּוֹהֵט

מְחַמְּמִים אֶת הַתַּנּוּר הֵיטֵב, כְּדֵי שֶׁהַבָּצֵק יַתְחִיל לְהֵאָפוֹת מִיָּד. מֻתָּר לֶאֱפוֹת גַּם עַל פְּלָטָה רוֹתַחַת שֶׁהָאֵשׁ בּוֹעֶרֶת מִתַּחְתֶּיהָ, וְגַם בְּתַנּוּר חַשְׁמַלִּי. אֲבָל מַצָּה שֶׁנֶּאֶפְתָה בְּחֹם הַשֶּׁמֶשׁ – אֲפִלּוּ אִם בָּרוּר שֶׁלֹּא הֶחְמִיצָה – אֵין יוֹצְאִים בָּהּ יְדֵי חוֹבָה. לֹא מְקַשְּׁטִים אֶת הַמַּצָּה בְּצִיּוּרִים, וְלֹא עוֹשִׂים מַצָּה עָבָה טֶפַח (כְּ־7.6 ס״מ).
🤔 לָמָּה? תַּנּוּר חַלָּשׁ אוֹ צִיּוּרִים מְעַכְּבִים אֶת הָאֲפִיָּה, וּבֵינְתַיִם הַבָּצֵק עָלוּל לְהַחְמִיץ. בְּמַצָּה עָבָה מְאוֹד הָאֵשׁ לֹא שׁוֹלֶטֶת הֵיטֵב בִּפְנִים. וּמַצָּה שֶׁנֶּאֶפְתָה בַּשֶּׁמֶשׁ אֵינָהּ נִקְרֵאת ״לֶחֶם״ – וְהַמַּצָּה שֶׁלָּנוּ הִיא ״לֶחֶם עֹנִי״.
🔀 שְׁנֵי מִנְהָגִים – וּשְׁנֵיהֶם כַּהֲלָכָה
לְמִנְהַג יוֹצְאֵי אַשְׁכְּנַז עוֹשִׂים מַצּוֹת דַּקּוֹת וְקָשׁוֹת, שֶׁהָאֵשׁ שׁוֹלֶטֶת בָּהֶן הֵיטֵב.
אֵצֶל יוֹצְאֵי סְפָרַד יֵשׁ שֶׁנָּהֲגוּ לַעֲשׂוֹת מַצָּה בְּעֹבִי אֶצְבַּע, וְיֵשׁ שֶׁנָּהֲגוּ לַעֲשׂוֹתָהּ דַּקָּה כְּמוֹ יוֹצְאֵי אַשְׁכְּנַז.
(פֶּרֶק י״ב, הֲלָכָה ח׳)

9מַצָּה שֶׁשַּׁיֶּכֶת לִי בֶּאֱמֶת

אֵין יוֹצְאִים יְדֵי חוֹבָה בְּמַצָּה גְּנוּבָה אוֹ גְּזוּלָה. לָכֵן כְּשֶׁקּוֹנִים מַצּוֹת: אִם הַמּוֹכֵר רוֹצֶה אֶת הַכֶּסֶף מִיָּד – מְשַׁלְּמִים לוֹ מִיָּד. וּמִי שֶׁהַמּוֹכֵר רוֹדֵף אַחֲרָיו שֶׁיְּשַׁלֵּם, וְהוּא דּוֹחֶה אוֹתוֹ בְּ״לֵךְ וָשׁוּב״ – הַמַּצּוֹת אֵינָן בֶּאֱמֶת שֶׁלּוֹ, וְאֵינוֹ יוֹצֵא בָּהֶן יְדֵי חוֹבָה.
🤔 לָמָּה? הַמִּצְוָה צְרִיכָה לִהְיוֹת נְקִיָּה – לֹא רַק מֵחִמּוּץ, אֶלָּא גַּם מִכָּל עָוֶל כְּלַפֵּי בְּנֵי אָדָם.
💛 ה׳ אוֹהֵב מִצְוָה יְשָׁרָה: מַצָּה שֶׁשִּׁלַּמְתִּי עָלֶיהָ בַּזְּמַן הִיא מַתָּנָה שְׁלֵמָה לוֹ.
(פֶּרֶק י״ב, הֲלָכָה ח׳)

📝 מָה לָמַדְנוּ

  • בְּלֵיל הַסֵּדֶר מִצְוָה מִן הַתּוֹרָה לֶאֱכֹל מַצָּה, וּבִשְׁאָר הַפֶּסַח הִיא הַמַּאֲכָל הָעִקָּרִי בִּמְקוֹם הַלֶּחֶם.
  • מַצַּת מִצְוָה שְׁמוּרָה מֵחִמּוּץ – לְכַתְּחִלָּה מִשְּׁעַת קְצִירָה – וְנַעֲשֵׂית ״לְשֵׁם מַצַּת מִצְוָה״.
  • לָשִׁים בְּמַיִם שֶׁלָּנוּ, בְּמָקוֹם קָרִיר וּבִזְרִיזוּת, וְאוֹפִים בְּתַנּוּר חַם מְאוֹד – לֹא בַּשֶּׁמֶשׁ.
  • יוֹצְאִים רַק בְּמַצָּה שֶׁשַּׁיֶּכֶת לָנוּ: מְשַׁלְּמִים לַמּוֹכֵר כְּמוֹ שֶׁבִּקֵּשׁ.

❓ שְׁאֵלוֹת חֲזָרָה

  1. מָה הֵם ״מַיִם שֶׁלָּנוּ״, וְלָמָּה לָשִׁים בָּהֶם אֶת הַמַּצּוֹת?
  2. לָמָּה אֵין יוֹצְאִים בְּמַצָּה שֶׁנֶּאֶפְתָה בַּשֶּׁמֶשׁ, גַּם אִם הִיא לֹא הֶחְמִיצָה?
  3. אֵיזוֹ מַחֲלֹקֶת יֵשׁ עַל מַצּוֹת מְכוֹנָה בְּלֵיל הַסֵּדֶר, וּמָה דַּעַת רֹב הַפּוֹסְקִים?
🎯 רוֹצִים לִבְדֹּק אֶת עַצְמְכֶם? לַחִידוֹן ״מַצָּה שְׁמוּרָה וְעֶרֶב פֶּסַח״

מְצוּדַת דָּוִד: הַפַּחַד הִפְנָה אֶת הָעָם לַמִּזְבֵּחַ, לְבַקֵּשׁ עֶזְרָה מֵה׳.

רַשִׁ״י: הַשִּׁירָה בְּיוֹם הַיְּסוֹד הָיְתָה מִמִּזְמוֹרֵי דָּוִד – הַקָּשֶׁר לַבַּיִת הָרִאשׁוֹן נִשְׁאַר.

הַלָּשׁוֹן ׳וַיַּעֲנוּ בְהַלֵּל׳ – רַשִׁ״י מְבָאֵר: ׳לְשׁוֹן עֲנִיַּת זֶמֶר, כְּמוֹ וַתַּעַן לָהֶם מִרְיָם׳ (שְׁמוֹת ט״ו). וּמְצוּדַת דָּוִד: הֵם אָמְרוּ ׳כִּי טוֹב לְהוֹדוֹת לַה׳, כִּי לְעוֹלָם מוֹשֵׁךְ חַסְדּוֹ עַל יִשְׂרָאֵל׳. כְּשֶׁאֶחָד אוֹמֵר וְכֻלָּם עוֹנִים – זֶה הַלֵּל.

רַשִׁ״י: ׳וַיַּעֲנוּ׳ – אֶחָד שָׁר וְהַקָּהָל עוֹנֶה, כְּמוֹ בְּשִׁירַת הַיָּם.

👴 מְצוּדַת דָּוִד: לָמָּה בָּכוּ הַזְּקֵנִים?

מְצוּדַת דָּוִד: הַזְּקֵנִים ׳הָיוּ בּוֹכִים עַל כִּי הָיָה קָטָן הַרְבֵּה מִן הַבַּיִת הָרִאשׁוֹן׳. וְרַשִׁ״י מוֹסִיף: הֵם בָּכוּ ׳מִתּוֹךְ שֶׁהָיוּ זוֹכְרִים אוֹתוֹ בִּנְיָן גָּדוֹל׳. אֵלֶּה שֶׁלֹּא רָאוּ אֶת הָרִאשׁוֹן ׳שְׂמֵחִים מֵרֹב שִׂמְחָתָם שֶׁיָּצְאוּ מִגָּלוּתָם׳.

מְצוּדַת דָּוִד וְרַשִׁ״י: הַזְּקֵנִים בָּכוּ מִגֹּדֶל הָאַהֲבָה לַבַּיִת שֶׁזָּכְרוּ, וְהַצְּעִירִים שָׂמְחוּ עַל מָה שֶׁיֵּשׁ.

🔊 רַשִׁ״י: קוֹל הַבֶּכִי נִשְׁמָע

׳וְאֵין הָעָם מַכִּירִים קוֹל תְּרוּעַת הַשִּׂמְחָה לְקוֹל בְּכִי הָעָם׳. רַשִׁ״י: ׳הַשּׁוֹמְעִים לֹא הָיוּ מַכִּירִים קוֹל תְּרוּעָה שֶׁל שִׂמְחָה מִפְּנֵי קוֹל בְּכִי הָעָם׳, ׳וְקוֹל הַבֶּכִי נִשְׁמַע יוֹתֵר וְיוֹתֵר עַד לְמֵרָחוֹק׳. הַשֵּׁנִי לֹא בִּטֵּל אֶת הָרִאשׁוֹן – שְׁנֵיהֶם נִשְׁמְעוּ.

רַשִׁ״י: מְעוֹרָבִים שְׁנֵי קוֹלוֹת עַד שֶׁאִי אֶפְשָׁר לְהַפְרִיד – כָּךְ זֶה כְּשֶׁאוֹהֲבִים.

רָמָה ב׳ – מָה אוֹמְרִים הַמְּפָרְשִׁים

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מַסְבִּירִים לָמָּה הַמִּזְבֵּחַ בָּא קֹדֶם, מָה הָיָה בַּשִּׁירָה, וְלָמָּה בָּכוּ הַזְּקֵנִים. ׳בְּכִי׳ אֵינוֹ הִפּוּךְ מֵאַהֲבָה, אֶלָּא אַהֲבָה שֶׁזּוֹכֶרֶת. ה׳ אוֹהֵב אֶת הַלֵּב שֶׁבּוֹכֶה וְאֶת הַלֵּב שֶׁשָּׂמֵחַ.

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • הָעוֹלִים בָּנוּ מִזְבֵּחַ לִפְנֵי שֶׁהָיָה בַּיִת, כִּי לֹא רָצוּ לְחַכּוֹת. עַל אֵיזֶה דָּבָר טוֹב אֲנִי מִתְחִיל עַכְשָׁו, גַּם אִם הוּא קָטָן, בִּמְקוֹם לְחַכּוֹת שֶׁיִּהְיֶה מֻשְׁלָם?
  • הַזְּקֵנִים בָּכוּ וְהַצְּעִירִים שָׂמְחוּ – וְה׳ קִבֵּל אֶת שְׁנֵיהֶם. מָתַי הִרְגַּשְׁתִּי בְּאוֹתוֹ זְמַן גַּם שִׂמְחָה וְגַם עֶצֶב, וְאֵיךְ אֲנִי מוֹדֶה לַה׳ עַל שְׁנֵיהֶם?
  • הַפָּסוּק ׳כִּי טוֹב כִּי לְעוֹלָם חַסְדּוֹ׳ נִשְׁמַע בַּיּוֹם הַהוּא וְנִשְׁמַע גַּם בַּהַלֵּל שֶׁלָּנוּ. מָתַי אֲנִי אוֹמֵר אוֹתוֹ וּמָה אֲנִי חוֹשֵׁב בִּזְמַן שֶׁאֲנִי אוֹמְרוֹ?

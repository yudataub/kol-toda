רַשִׁ״י: ׳הָיוּ אוֹמְרִים כָּךְ לְמַעַן תִּתְבַּטֵּל מְלֶאכֶת הַבַּיִת עַל יְדֵיהֶם׳. וּמְצוּדַת דָּוִד: ׳כַּוָּנָתָם הָיְתָה לְהִתְחַבֵּר עִמָּהֶם לְדַעַת מַצְפּוּנֵי לִבָּם לִמְצֹא עֲלִילָה מָה לְהַשְׁבִּית אֶת הַמְּלָאכָה׳. כְּלוֹמַר: הַצָּעַת עֶזְרָה שֶׁנּוֹלְדָה מִכַּוָּנָה לְהַפְרִיעַ.

רַשִׁ״י וּמְצוּדַת דָּוִד: הַהַצָּעָה ׳נִבְנֶה עִמָּכֶם׳ הָיְתָה תַּחְבּוּלָה לְבַטֵּל אֶת הַבִּנְיָן.

📜 הַפָּסוּקוַיֹּאמֶר לָהֶם זְרֻבָּבֶל וְיֵשׁוּעַ וּשְׁאָר רָאשֵׁי הָאָבוֹת לְיִשְׂרָאֵל לֹא לָכֶם וָלָנוּ לִבְנוֹת בַּיִת לֵאלֹהֵינוּ כִּי אֲנַחְנוּ יַחַד נִבְנֶה לַיהוָה אֱלֹהֵי יִשְׂרָאֵל

רָמָה ב׳ – עִיּוּן

🚪 מְצוּדַת דָּוִד: ׳לֹא לָכֶם וְלָנוּ׳

מְצוּדַת דָּוִד: ׳לֹא לִשְׁנֵינוּ יַחַד אֲנַחְנוּ וְאַתֶּם לִבְנוֹת הַבַּיִת, כִּי מָה לָכֶם לִבְנוֹת הַבַּיִת לַה׳, רַק אֲנַחְנוּ בְּנֵי יִשְׂרָאֵל יַחַד נִבְנֶה׳. וְעוֹד: ׳כַּאֲשֶׁר צִוָּנוּ הַמֶּלֶךְ׳ – ׳אוֹתָנוּ צִוָּה וְלֹא אֶתְכֶם׳. הַתְּשׁוּבָה נִמְסְרָה בְּנִימוּס, עִם נִמּוּק מִן הַמֶּלֶךְ.

מְצוּדַת דָּוִד: הַתְּשׁוּבָה הָיְתָה בְּכָבוֹד, וְעָמְדָה עַל הַהוֹרָאָה שֶׁל הַמֶּלֶךְ.

2

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹלֵחַ לִי עֶזְרָה גַּם דֶּרֶךְ אֲנָשִׁים שֶׁעוֹמְדִים לְיָדִי 💛

📖 עֶזְרָא, פֶּרֶק ה׳, פָּסוּק ב׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן

🔥 רַשִׁ״י: חַגַּי וּזְכַרְיָה

רַשִׁ״י: ׳בִּשְׁנַת שְׁתַּיִם לְדָרְיָוֶשׁ נִתְנַבְּאוּ חַגַּי וּזְכַרְיָה... לִבְנוֹת בִּנְיַן הַבַּיִת׳, וּבִפְסוּק ב׳: ׳הִתְחִילוּ לִבְנוֹת בֵּיתוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא וְעִמָּהֶם הַנְּבִיאִים עוֹזְרִים וּמְסַיְּעִים׳. הַנָּבִיא לֹא רַק אוֹמֵר – הוּא עוֹמֵד אִתָּם בַּעֲבוֹדָה.

רַשִׁ״י: הַנְּבִיאִים לֹא רַק דִּבְּרוּ – הֵם עָמְדוּ בַּבִּנְיָן וְעָזְרוּ.

📜 הַפָּסוּקהַעֵת לָכֶם אַתֶּם לָשֶׁבֶת בְּבָתֵּיכֶם סְפוּנִים וְהַבַּיִת הַזֶּה חָרֵב

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹמֵר לִי: שִׂים לִבְּךָ – וְאֲנִי אוֹהֵב אוֹתְךָ עוֹד יוֹתֵר 💛

📖 עֶזְרָא, פֶּרֶק ה׳, פָּסוּק א׳ (חַגַּי א׳ ד׳, רַשִׁ״י וּמְצוּדַת דָּוִד)

🏠 חַגַּי: ׳הַעֵת לָכֶם... סְפוּנִים׳

חַגַּי מוֹכִיחַ אֶת הָעָם: ׳הַעֵת לָכֶם אַתֶּם לָשֶׁבֶת בְּבָתֵּיכֶם סְפוּנִים וְהַבַּיִת הַזֶּה חָרֵב׳. רַשִׁ״י: ׳סְפוּנִים – מְכֻסִּים בְּנִסְרֵי אֲרָזִים׳. וּמְצוּדַת דָּוִד: ׳וְכִי רַק לָכֶם בָּא הָעֵת... וְהַבַּיִת הַזֶּה יִשָּׁאֵר חָרֵב׳. הוּא שׁוֹאֵל שְׁאֵלָה שֶׁל לֵב: מָה חָשׁוּב יוֹתֵר?

חַגַּי שׁוֹאֵל: אֲנִי גָּר בְּבַיִת יָפֶה וְהַבַּיִת שֶׁל ה׳ חָרֵב?

📜 הַפָּסוּקעֲלוּ הָהָר וַהֲבֵאתֶם עֵץ וּבְנוּ הַבָּיִת וְאֶרְצֶה בּוֹ וְאֶכָּבְדָה

💛 אַהֲבַת הַשֵּׁם: ה׳ שָׂמֵחַ בַּבַּיִת שֶׁלּוֹ גַּם כְּשֶׁהוּא פָּשׁוּט 💛

🗝️ רַשִׁ״י: ׳וְאֶכָּבֵדָה׳ חָסֵר ה׳

חַגַּי אוֹמֵר: ׳עֲלוּ הָהָר וַהֲבֵאתֶם עֵץ וּבְנוּ הַבַּיִת וְאֶרְצֶה בּוֹ וְאֶכָּבְדָה׳. רַשִׁ״י מַסְבִּיר שֶׁ׳וְאֶכָּבְדָה׳ כָּתוּב חָסֵר ה׳ ׳כְּנֶגֶד חֲמִשָּׁה דְּבָרִים שֶׁהָיוּ בַּמִּקְדָּשׁ הָרִאשׁוֹן וְלֹא הָיוּ בַּשֵּׁנִי׳. וּמְצוּדַת דָּוִד: ׳עִם כִּי יִהְיֶה קָטָן, אֶהְיֶה מְרֻצֶּה בּוֹ׳ – גַּם בְּבַיִת פָּשׁוּט ה׳ מְרֻצֶּה.

גַּם בַּבַּיִת הַקָּטָן ה׳ אוֹמֵר: ׳אֶרְצֶה בּוֹ׳ – אֲנִי מְרֻצֶּה.

📜 הַפָּסוּקוְשָׂבֵי יְהוּדָיֵא בָּנַיִן וּמַצְלְחִין בִּנְבוּאַת חַגַּי נְבִיָּא וּזְכַרְיָה בַּר עִדּוֹא וּבְנוֹ וְשַׁכְלִלוּ מִן טַעַם אֱלָהּ יִשְׂרָאֵל וּמִטְּעֵם כּוֹרֶשׁ וְדָרְיָוֶשׁ

💛 אַהֲבַת הַשֵּׁם: ה׳ הוּא הַמְצַוֶּה הָרִאשׁוֹן, וְהַמְּלָכִים הֵם שְׁלוּחָיו 💛

3️⃣ רַשִׁ״י: שְׁלוֹשָׁה טְעָמִים

הָעֵדוּת בַּסּוֹף: ׳וּבְנוֹ וְשַׁכְלִלוּ מִן טַעַם אֱלָהּ יִשְׂרָאֵל וּמִטַּעַם כּוֹרֶשׁ וְדָרְיָוֶשׁ׳. רַשִׁ״י: ׳מִצִּוּוּי שֶׁל אֱלֹהֵי יִשְׂרָאֵל וּמֵרְשׁוּת כּוֹרֶשׁ... וּמֵרְשׁוּת דָּרְיָוֶשׁ׳. עוֹד רַשִׁ״י: ׳בִּנְיַן הַבַּיִת הָיָה אַרְבַּע שָׁנִים׳ – מִשְּׁנַת שְׁתַּיִם עַד שְׁנַת שֵׁשׁ.

רַשִׁ״י: הַבַּיִת נִבְנָה בִּשְׁלוֹשָׁה צְוָאוֹת – שֶׁל ה׳, שֶׁל כּוֹרֶשׁ וְשֶׁל דָּרְיָוֶשׁ.

׳כִּי שִׂמְּחָם ה׳ וְהֵסֵב לֵב מֶלֶךְ אַשּׁוּר עֲלֵיהֶם לְחַזֵּק יְדֵיהֶם׳. רַשִׁ״י: ׳וְהַקָּדוֹשׁ בָּרוּךְ הוּא הֵסֵב לֵב מֶלֶךְ אַשּׁוּר עַל יִשְׂרָאֵל... לְהַחֲזִיקָם וּלְסַיֵּעַ בִּמְלֶאכֶת הַבִּנְיָן׳. וּמְצוּדַת דָּוִד: מֶלֶךְ אַשּׁוּר הוּא דָּרְיָוֶשׁ, וְהוּא נָתַן ׳הוֹצָאוֹת הַבִּנְיָן וְהָעֵזֶר מֵהָאוֹיֵב׳. הָאוֹיֵב שֶׁעָצַר אֶת הַבִּנְיָן הִפְסִיד.

רַשִׁ״י: ה׳ הֵסֵב אֶת לֵב הַמֶּלֶךְ מֵעוֹצֵר לְעוֹזֵר.

מָה לוֹקְחִים מִרָמָה ב׳?

הַמְּפָרְשִׁים מַרְאִים: ׳הַצָּעַת עֶזְרָה׳ הִיא לִפְעָמִים מַלְכֹּדֶת, הַנְּבִיאִים מֵעִירִים אֶת הַלֵּב, וְהַבַּיִת בָּא מִשְּׁלוֹשָׁה טְעָמִים. ה׳ אוֹהֵב אֶת הָעָם שֶׁלּוֹ וּמְחַכֶּה שֶׁנַּחְזֹר לְהִשְׁתַּתֵּף בָּעֲבוֹדָה.

🤔 חוֹשְׁבִים עַל זֶה

  • זְרֻבָּבֶל אָמַר לַשְּׁכֵנִים ׳לֹא לָכֶם וְלָנוּ׳ – וּבְכָל זֹאת נִפְסַק הַבִּנְיָן. מָתַי עָשִׂיתִי דָּבָר נָכוֹן וְהָיָה קָשֶׁה, וְאֵיךְ ה׳ עָזַר לִי אַחַר כָּךְ?
  • חַגַּי שָׁאַל: ׳הַעֵת לָכֶם... לָשֶׁבֶת בְּבָתֵּיכֶם סְפוּנִים וְהַבַּיִת הַזֶּה חָרֵב׳. עַל אֵיזֶה דָּבָר חָשׁוּב אֲנִי כָּל הַזְּמַן דּוֹחֶה, וּמָה אֶפְשָׁר לְהַתְחִיל כְּבָר הַיּוֹם?
  • הַבִּנְיָן נִגְמַר ׳בְּחֶדְוָה׳. מָה הַדָּבָר הַשָּׂמֵחַ שֶׁזָּכִיתִי לְסַיֵּם, וְאֵיךְ אֲנִי מוֹדֶה עָלָיו לַה׳?

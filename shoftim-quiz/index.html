⚔️ חִידוֹנֵי סֵפֶר שׁוֹפְטִים

חִידוֹן לְכָל פֶּרֶק, לְפִי תָּכְנִית הַלִּמּוּדִים שֶׁל מִשְׂרַד הַחִנּוּךְ (תַּנַ״ךְ חֶמֶ״ד). בְּכָל חִידוֹן כַּרְטִיסֵי לִמּוּד עִם אַהֲבַת ה׳ 💛 וְשָׁלוֹשׁ רָמוֹת: תְּמוּנוֹת, שְׁאֵלוֹת לַחֲכָמִים, וּבָרָמָה הַשְּׁלִישִׁית מִשְׂחָקִים – בְּכָל פַּעַם מִשְׂחָק אַחֵר. לְכִתּוֹת ד׳–ו׳

🏕️ פְּתִיחָה – אַחֲרֵי מוֹת יְהוֹשֻׁעַ

1. פְּתִיחָה וְהַשְׁלָמַת הַהִתְנַחֲלוּתפֶּרֶק א׳
2. הַבֹּכִים, מוֹת יְהוֹשֻׁעַ וּמַעְגַּל הַשּׁוֹפְטִיםפֶּרֶק ב׳

⚔️ עָתְנִיאֵל, אֵהוּד, דְּבוֹרָה וּבָרָק

3. הַנִּשְׁאָרִים לְנַסּוֹת וְעָתְנִיאֵל בֶּן קְנַזפֶּרֶק ג׳, פְּסוּקִים א׳–י״א
4. אֵהוּד בֶּן גֵּרָא וְעֶגְלוֹן מֶלֶךְ מוֹאָבפֶּרֶק ג׳, פְּסוּקִים י״ב–ל״א
5. דְּבוֹרָה וּבָרָק – הַמִּלְחָמָה בְּסִיסְרָאפֶּרֶק ד׳
6. שִׁירַת דְּבוֹרָהפֶּרֶק ה׳

🏺 גִּדְעוֹן וַאֲבִימֶלֶךְ

7. גִּדְעוֹן – הַמִּנּוּי וְהָאוֹתפֶּרֶק ו׳, פְּסוּקִים א׳–כ״ד
8. מִזְבַּח הַבַּעַל וְהַגִּזָּהפֶּרֶק ו׳, פְּסוּקִים כ״ה–מ׳
9. שְׁלֹשׁ מֵאוֹת הַמְלַקְקִים וְהַכַּדִּיםפֶּרֶק ז׳
10. הַמִּרְדָּף, סֻכּוֹת וּפְנוּאֵל – ׳ה׳ יִמְשֹׁל בָּכֶם׳פֶּרֶק ח׳
11. אֲבִימֶלֶךְ וּמְשַׁל יוֹתָםפֶּרֶק ט׳

🛡️ יִפְתָּח

12. תּוֹלָע, יָאִיר וְהַתְּשׁוּבָה לִפְנֵי יִפְתָּחפֶּרֶק י׳
13. יִפְתָּח הַגִּלְעָדִיפֶּרֶק י״א
14. שִׁבֹּלֶת – הַמַּאֲבָק בִּבְנֵי אֶפְרַיִםפֶּרֶק י״ב

🦁 שִׁמְשׁוֹן

15. לֵידַת שִׁמְשׁוֹן וְהַנְּזִירוּתפֶּרֶק י״ג
16. הָאַרְיֵה וְהַחִידָהפֶּרֶק י״ד
17. הַשּׁוּעָלִים וּלְחִי הַחֲמוֹרפֶּרֶק ט״ו
18. מוֹת שִׁמְשׁוֹןפֶּרֶק ט״ז

📜 סוֹף הַסֵּפֶר – ׳אֵין מֶלֶךְ בְּיִשְׂרָאֵל׳

19. פֶּסֶל מִיכָהפֶּרֶק י״ז
20. בְּנֵי דָן וְלַיִשׁפֶּרֶק י״ח
21. הַמַּעֲשֶׂה בַּגִּבְעָה – ׳כְּאִישׁ אֶחָד חֲבֵרִים׳פְּרָקִים י״ט–כ׳
22. ״לֹא יִמָּחֶה שֵׁבֶט מִיִּשְׂרָאֵל״פֶּרֶק כ״א

🏆 חִידוֹן סִכּוּם

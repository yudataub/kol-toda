שטר אירוסין – על פי ״נחלת שבעה״
בְּעֶזְרַת הַשֵּׁם יִתְבָּרַךְ
בְּסִימָן טוֹב וּבְמַזָּל טוֹב
יַעֲלֶה וְיִצְמַח כְּגַן רָטֹב
וְהַשֵּׁם הַטּוֹב יֹאמַר לַדֶּבֶק טוֹב
מָצָא אִשָּׁה מָצָא טוֹב
וְיָפֵק רָצוֹן מֵהַשֵּׁם הַטּוֹב

1פתיחת השטר – הצדדים

הַמַּגִּיד מֵרֵאשִׁית אַחֲרִית הוּא יִתֵּן שֵׁם טוֹב וּשְׁאֵרִית לְאֵלֶּה דִּבְרֵי הַתְּנָאִים וְהַבְּרִית שֶׁנִּדְבְּרוּ וְהוּתְנוּ בֵּין הָנֵי תְּרֵי צְדָדִים, הֲלֹא הוּא   הָעוֹמֵד מִצַּד הֶחָתָן הַבָּחוּר הַמֻפְלָג   וּבֵין   הָעוֹמֵד מִצַּד הַכַּלָּה הַבְּתוּלָה הַמְהֻלָּלָה   תחי', וּבִשְׁאֵלַת פִּיהֶם וְאָמְרוּ הֵן.

2עיקר התנאים – החופה והקידושין

רֵאשִׁית דָּבָר, הֲלֹא הוּא הַבָּחוּר הַמֻּפְלָג   נ"י יִשָּׁא לְמַזָּל טוֹב אֶת בַּת גִּילוֹ הַבְּתוּלָה הַמְהֻלָּלָה   תחי' בְּחוּפָּה וְקִדּוּשִׁין כְּדָת מֹשֶׁה וְיִשְׂרָאֵל, וְאַל יַבְרִיחוּ וְאַל יַעְלִימוּ לֹא זֶה מִזּוֹ וְלֹא זוֹ מִזֶּה, רַק יִשְׁלְטוּ בְּנִכְסֵיהוֹן שָׁוֶה בְּשָׁוֶה, וְיָדוּרוּ בְּאַהֲבָה וְאַחֲוָה בְּשָׁלוֹם וְרֵיעוּת.

3התחייבויות הצדדים

  הָעוֹמֵד מִצַּד הֶחָתָן מִתְחַיֵּיב   וְגַם לְהַלְבִּישׁ אֶת הֶחָתָן נ"י מַלְבּוּשֵׁי כָבוֹד, בִּגְדֵי חוֹל שַׁבָּת וְיוֹם טוֹב כְּמִנְהַג בַּעֲלֵי בָתִּים חֲשׁוּבִים פֹּה.   הָעוֹמֵד מִצַּד הַכַּלָּה תחי' מִתְחַיֵּב   וְגַם לְהַלְבִּישׁ אֶת הַכַּלָּה תחי' מַלְבּוּשֵׁי כָבוֹד, בִּגְדֵי חוֹל שַׁבָּת וְיוֹם טוֹב כְּמִנְהַג בַּעֲלֵי בָתִּים חֲשׁוּבִים פֹּה.

4זמן החתונה

זְמַן הַחֲתוּנָה לְמַזָּל טוֹב וּבְשָׁעָה טוֹבָה וּמוּצְלַחַת בְּעֶזְרַת הַשֵּׁם יִתְבָּרַךְ, בְּחֹדֶשׁ   שְׁנַת   הבעל"ט, וּבְתוֹךְ הַזְּמָן מָתַי שֶׁיִּתְרַצּו שְׁנֵי הַצְּדָדִים, עַל הוֹצָאוֹת  .

5הערבים הקבלנים

עָרֵב קַבְּלָן מִצַּד הֶחָתָן נ"י הֲלֹא הוּא  . עָרֵב קַבְּלָן מִצַּד הַכַּלָּה תחי' הֲלֹא הוּא  . הַצְּדָדִים מִתְחַיְּבִים לְפַצּוֹת הָעֲרֵבִים הַקַבְּלָנִים שֶׁלֹּא יַגִּיעַ לָהֶם שׁוּם הֶיזֵּק וּגְרַם הֶיזֵּק כְּלָל.

6הקניין והחתימה

וּקְנֵינָא אֲנַן סָהֲדֵי מֵהֶחָתָן וּמֵהַכַּלָּה וּמֵהַמְחוּתָּנִים וּמֵהָעֲרֵבִים קַבְּלָנִים כְּכָל מַה דְּכָתוּב וּמְפוֹרָשׁ לְעֵיל בְּקִנְיָן גָּמוּר אֲגַב סוּדָר בְּמָנָא דְּכָשַׁר לְמִקְנְיָא בֵיהּ, דְּלָא כְּאַסְמַכְתָּא וּדְלָא כְּטוּפְסֵי דִּשְׁטָרֵי. נַעֲשָׂה הַיּוֹם יוֹם   בְּשַׁבָּת   לְחֹדֶשׁ   שְׁנַת חֲמֵשֶׁת אֲלָפִים   לִבְרִיאַת עוֹלָם, כָּאן עִיר  . הַכֹּל שָׁרִיר וְקַיָּם.

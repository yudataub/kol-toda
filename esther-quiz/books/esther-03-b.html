רַשִׁ״י: אֶסְתֵּר וּמָרְדֳּכַי הֵם הָרְפוּאָה שֶׁה׳ הֵכִין לִפְנֵי הַמַּכָּה.

🛡️ רַשִׁ״י: לָמָּה לֹא כָּרַע?

רַשִׁ״י מַסְבִּיר מַדּוּעַ מָרְדֳּכַי לֹא כָּרַע: הָמָן ׳עָשָׂה עַצְמוֹ אֱלוֹהַּ׳, וְלָכֵן – ׳וּמָרְדְּכַי לֹא יִכְרַע וְלֹא יִשְׁתַּחֲוֶה׳. יְהוּדִי כּוֹרֵעַ רַק לַה׳. זֶה לֹא עִנְיָן שֶׁל גַּאֲוָה, זֶה עִנְיָן שֶׁל אֱמוּנָה.

רַשִׁ״י: הָמָן עָשָׂה עַצְמוֹ אֱלוֹהַּ, וְלָכֵן מָרְדֳּכַי לֹא כָּרַע.

🗣️

📜 הַפָּסוּקוַיְהִי כְּאָמְרָם אֵלָיו יוֹם וָיוֹם וְלֹא שָׁמַע אֲלֵיהֶם וַיַּגִּידוּ לְהָמָן לִרְאוֹת הֲיַעַמְדוּ דִּבְרֵי מָרְדֳּכַי כִּי הִגִּיד לָהֶם אֲשֶׁר הוּא יְהוּדִי

💛 אַהֲבַת הַשֵּׁם: אֲנִי לֹא מִתְבַּיֵּשׁ בְּמִצְווֹת שֶׁלִּי, כִּי הֵן הַחָכְמָה שֶׁה׳ נָתַן 💛

📖 אֶסְתֵּר, פֶּרֶק ג׳, פָּסוּק ד׳ (רַשִׁ״י)

רָמָה ב׳ – רַשִׁ״י עַל הַפֶּרֶק

🗣️ רַשִׁ״י: ׳הֲיַעַמְדוּ דִּבְרֵי מָרְדֳּכַי׳

הָעֲבָדִים הִגִּידוּ לְהָמָן ׳לִרְאוֹת הֲיַעַמְדוּ דִּבְרֵי מָרְדֳּכַי׳. רַשִׁ״י מְפָרֵשׁ: ׳הָאוֹמֵר שֶׁלֹּא יִשְׁתַּחֲוֶה עוֹלָמִית, כִּי הוּא יְהוּדִי, וְהוּזְהַר עַל עֲבוֹדַת אֱלִילִים׳. מָרְדֳּכַי לֹא אָמַר ׳אֲנִי כּוֹעֵס׳, אֶלָּא ׳אֲנִי יְהוּדִי׳. הַתּוֹרָה אָסְרָה עָלָיו.

רַשִׁ״י: מָרְדֳּכַי הִסְבִּיר שֶׁהוּא יְהוּדִי וְהוּזְהַר שֶׁלֹּא לְהִשְׁתַּחֲוֹת.

🎲

📜 הַפָּסוּקבַּחֹדֶשׁ הָרִאשׁוֹן הוּא חֹדֶשׁ נִיסָן בִּשְׁנַת שְׁתֵּים עֶשְׂרֵה לַמֶּלֶךְ אֲחַשְׁוֵרוֹשׁ הִפִּיל פּוּר הוּא הַגּוֹרָל לִפְנֵי הָמָן מִיּוֹם לְיוֹם וּמֵחֹדֶשׁ לְחֹדֶשׁ שְׁנֵים עָשָׂר הוּא חֹדֶשׁ אֲדָר

💛 אַהֲבַת הַשֵּׁם: הַגּוֹרָל נִרְאֶה כְּמוֹ מִקְרֶה, אֲבָל ה׳ מַכְרִיעַ 💛

📖 אֶסְתֵּר, פֶּרֶק ג׳, פָּסוּק ז׳ (רַשִׁ״י)

רָמָה ב׳ – רַשִׁ״י עַל הַפֶּרֶק

🎲 רַשִׁ״י: ׳הִפִּיל פּוּר׳

הַפָּסוּק אוֹמֵר ׳הִפִּיל פּוּר׳ וְלֹא אוֹמֵר מִי. רַשִׁ״י: ׳הִפִּיל מִי שֶׁהִפִּיל, וְלֹא פֵּרֵשׁ מִי. וּמִקְרָא קָצָר הוּא׳. וְאָז: ׳הַכָּתוּב מְפָרֵשׁ וּמַהוּ הַפּוּר – הוּא הַגּוֹרָל. הִפִּיל הַגּוֹרָל בְּאֵיזֶה חֹדֶשׁ יַצְלִיחַ, וּבְאֵיזֶה יוֹם בַּחֹדֶשׁ׳.

רַשִׁ״י: פּוּר הוּא גּוֹרָל, וְהוּא נִפַּל כְּדֵי לִבְחֹר מָתַי יַצְלִיחַ הָמָן.

׳וַיָּסַר הַמֶּלֶךְ אֶת טַבַּעְתּוֹ׳ – רַשִׁ״י מְבָאֵר: ׳הוּא מַתַּן כָּל דָּבָר גָּדוֹל שֶׁיִּשְׁאֲלוּ מֵאֵת הַמֶּלֶךְ, לִהְיוֹת מִי שֶׁהַטַּבַּעַת בְּיָדוֹ שַׁלִּיט בְּכָל דְּבַר הַמֶּלֶךְ׳. כָּל כְּתָב שֶׁנֶּחְתַּם בַּטַּבַּעַת הוּא חֹק שֶׁאִי אֶפְשָׁר לְשַׁנּוֹת. לָכֵן הַמְּצָב כָּל כָּךְ מְסֻכָּן.

רַשִׁ״י: מִי שֶׁמַּחֲזִיק אֶת הַטַּבַּעַת שַׁלִּיט בְּכָל דְּבַר הַמֶּלֶךְ.

🪙

📜 הַפָּסוּק… וְאֶת דָּתֵי הַמֶּלֶךְ אֵינָם עֹשִׂים וְלַמֶּלֶךְ אֵין שֹׁוֶה לְהַנִּיחָם

💛 אַהֲבַת הַשֵּׁם: שֶׁקֶר הוּא דִּבּוּר שֶׁל שְׂנֵאָה; ה׳ אוֹהֵב אֶת הַיְּהוּדִים מֵאַהֲבָה נֶאֱמָנָה 💛

הָמָן אָמַר ׳וְלַמֶּלֶךְ אֵין שֹׁוֶה לְהַנִּיחָם׳. רַשִׁ״י מְבָאֵר: ׳אֵין שֹׁוֶה – אֵין חֲשַׁשׁ. כְּלוֹמַר אֵין בֶּצַע׳, וְגַם ׳וְאֶת דָּתֵי הַמֶּלֶךְ – לָתֵת מַס לַעֲבוֹדַת הַמֶּלֶךְ׳. הָמָן הִצִּיג אֶת הַיְּהוּדִים כְּעַם שֶׁלֹּא כְּדַאי לִשְׁמֹר עָלָיו.

רַשִׁ״י: הָמָן טָעַן שֶׁאֵין לַמֶּלֶךְ תּוֹעֶלֶת בַּיְּהוּדִים.

  • מָרְדֳּכַי לֹא כָּרַע, אֲפִלּוּ שֶׁהַכֹּל כָּרְעוּ. מָתַי הָיִיתִי צָרִיךְ לַעֲמֹד זָקוּף לְבַד?
  • הָמָן הִפִּיל פּוּר וְהַגּוֹרָל נָפַל עַל אֲדָר. אֵיךְ ה׳ הָפַךְ דָּבָר שֶׁנִּרְאָה רַע לְטוֹבָה בְּחַיַּי?
  • רַשִׁ״י אוֹמֵר שֶׁה׳ בּוֹרֵא רְפוּאָה לִפְנֵי הַמַּכָּה. מַה הָיְתָה הָרְפוּאָה שֶׁקִּבַּלְתִּי לִפְנֵי שֶׁהִתְקַשִּׁיתִי?

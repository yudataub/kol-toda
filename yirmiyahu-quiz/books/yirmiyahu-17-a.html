🪵 מוֹסֵרוֹת וּמוֹטוֹת

ה׳ אָמַר לְיִרְמְיָהוּ: ׳עֲשֵׂה לְךָ מוֹסֵרוֹת וּמֹטוֹת וּנְתַתָּם עַל צַוָּארֶךָ׳. רַשִׁ״י מְבָאֵר: ׳מוֹסֵרוֹת – הֵם רְצוּעוֹת הָעֹל׳. יִרְמְיָהוּ הָלַךְ בָּעִיר עִם עֹל עַל הַצַּוָּאר, כְּדֵי שֶׁכֻּלָּם יִרְאוּ וְלֹא רַק יִשְׁמְעוּ.

יִרְמְיָהוּ שָׂם עֹל עַל צַוָּארוֹ כְּדֵי שֶׁהָעָם יִרְאוּ אֶת הַנְּבוּאָה.

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב אֶת כָּל בְּרִיּוֹתָיו, כִּי אֶת כֻּלָּם עָשָׂה 💛

📖 יִרְמְיָהוּ, פֶּרֶק כ״ז, פְּסוּקִים ג׳–ה׳

רָמָה א׳ – הַמּוֹטוֹת עַל הַצַּוָּאר

👑 שִׁלּוּחַ לַחֲמִשָּׁה מְלָכִים

יִרְמְיָהוּ שָׁלַח אֶת הַמּוֹטוֹת ׳אֶל מֶלֶךְ אֱדוֹם וְאֶל מֶלֶךְ מוֹאָב וְאֶל מֶלֶךְ בְּנֵי עַמּוֹן וְאֶל מֶלֶךְ צֹר וְאֶל מֶלֶךְ צִידוֹן׳, בְּיַד שְׁלִיחִים שֶׁבָּאוּ לִירוּשָׁלַיִם אֶל צִדְקִיָּהוּ. ה׳ אָמַר לָהֶם: ׳אָנֹכִי עָשִׂיתִי אֶת הָאָרֶץ אֶת הָאָדָם וְאֶת הַבְּהֵמָה אֲשֶׁר עַל פְּנֵי הָאָרֶץ בְּכֹחִי הַגָּדוֹל וּבִזְרוֹעִי הַנְּטוּיָה׳.

הַמּוֹטוֹת נִשְׁלְחוּ לַחֲמִשָּׁה מְלָכִים, וְה׳ הִזְכִּיר לָהֶם שֶׁהוּא עָשָׂה אֶת הָעוֹלָם.

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁנִּחְיֶה בְּשָׁלוֹם עַל הָאֲדָמָה שֶׁלָּנוּ 💛

📖 יִרְמְיָהוּ, פֶּרֶק כ״ז, פְּסוּקִים ו׳, י״א (רַשִׁ״י)

רָמָה א׳ – הַמּוֹטוֹת עַל הַצַּוָּאר

🌾 מִי שֶׁמְּקַבֵּל – נִשְׁאָר בְּאַדְמָתוֹ

ה׳ אָמַר: ׳נָתַתִּי אֶת כָּל הָאֲרָצוֹת הָאֵלֶּה בְּיַד נְבוּכַדְנֶאצַּר מֶלֶךְ בָּבֶל עַבְדִּי׳. רַשִׁ״י מְבָאֵר: ׳עַבְדִּי – שֶׁיַּעֲשֶׂה רְצוֹנִי׳. וְהַהַבְטָחָה הַיָּפָה: ׳וְהַגּוֹי אֲשֶׁר יָבִיא אֶת צַוָּארוֹ בְּעֹל מֶלֶךְ בָּבֶל וַעֲבָדוֹ וְהִנַּחְתִּיו עַל אַדְמָתוֹ... וַעֲבָדָהּ וְיָשַׁב בָּהּ׳.

מִי שֶׁמְּקַבֵּל אֶת הָעֹל – ה׳ ׳מַנִּיחַ׳ אוֹתוֹ עַל אַדְמָתוֹ.

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹמֵר לָנוּ ׳וִחְיוּ׳ – הוּא רוֹצֶה בְּחַיֵּינוּ 💛

📖 יִרְמְיָהוּ, פֶּרֶק כ״ז, פְּסוּקִים י״ב, י״ז (מְצוּדַת דָּוִד)

🗣️ ׳וִחְיוּ׳

אֶל צִדְקִיָּהוּ מֶלֶךְ יְהוּדָה אָמַר יִרְמְיָהוּ אֶת אוֹתָהּ נְבוּאָה: ׳הָבִיאוּ אֶת צַוְּארֵיכֶם בְּעֹל מֶלֶךְ בָּבֶל וְעִבְדוּ אֹתוֹ וְעַמּוֹ וִחְיוּ׳. מְצוּדַת דָּוִד מְבָאֵר: ׳וִחְיוּ – תִּשָּׁאֲרוּ חַיִּים׳. ׳לָמָּה תִהְיֶה הָעִיר הַזֹּאת חָרְבָּה?׳

יִרְמְיָהוּ בִּקֵּשׁ מִצִּדְקִיָּהוּ: ׳וִחְיוּ׳ – כִּי ה׳ רוֹצֶה שֶׁהָעִיר תַּעֲמֹד.

💛 אַהֲבַת הַשֵּׁם: ה׳ שׁוֹמֵר עָלֵינוּ מִדְּבָרִים יָפִים שֶׁאֵינָם אֱמֶת 💛

📖 יִרְמְיָהוּ, פֶּרֶק כ״ז, פְּסוּקִים י״ד–ט״ז

רָמָה א׳ – הַמּוֹטוֹת עַל הַצַּוָּאר

🚫 נְבִיאֵי הַשֶּׁקֶר

הָיוּ מִי שֶׁאָמְרוּ לָעָם: ׳לֹא תַעַבְדוּ אֶת מֶלֶךְ בָּבֶל׳. ה׳ אָמַר: ׳אַל תִּשְׁמְעוּ אֶל דִּבְרֵי הַנְּבִאִים... כִּי שֶׁקֶר הֵם נִבְּאִים לָכֶם׳, ׳כִּי לֹא שְׁלַחְתִּים׳. הֵם גַּם אָמְרוּ: ׳הִנֵּה כְלֵי בֵית ה׳ מוּשָׁבִים מִבָּבֶלָה עַתָּה מְהֵרָה׳ – אֲבָל זֶה לֹא נָכוֹן.

נְבִיאֵי הַשֶּׁקֶר הִבְטִיחוּ חֲזָרָה מְהִירָה, אֲבָל ה׳ לֹא שְׁלָחָם.

5

💛 אַהֲבַת הַשֵּׁם: ה׳ לֹא שׁוֹכֵחַ – הוּא תָּמִיד מַחֲזִיר מָה שֶׁאָהוּב עָלָיו 💛

📖 יִרְמְיָהוּ, פֶּרֶק כ״ז, פְּסוּקִים י״ט–כ״ב

רָמָה א׳ – הַמּוֹטוֹת עַל הַצַּוָּאר

🏺 כְּלֵי בֵּית ה׳ יָשׁוּבוּ

יִרְמְיָהוּ אָמַר: גַּם הַכֵּלִים הַנּוֹתָרִים – ׳הָעַמֻּדִים וְהַיָּם וְהַמְּכֹנוֹת׳ – ׳בָּבֶלָה יוּבָאוּ וְשָׁמָּה יִהְיוּ עַד יוֹם פָּקְדִי אֹתָם נְאֻם ה׳ וְהַעֲלִיתִים וַהֲשִׁיבֹתִים אֶל הַמָּקוֹם הַזֶּה׳. ה׳ זוֹכֵר אֶת כֵּלָיו, וְיַחֲזִיר אוֹתָם.

הַכֵּלִים יֵלְכוּ לְבָבֶל, אֲבָל ה׳ מַבְטִיחַ: ׳וְהַעֲלִיתִים וַהֲשִׁיבֹתִים׳.

💛 אַהֲבַת הַשֵּׁם: הַבְּשׂוֹרָה שֶׁל חֲנַנְיָה נִשְׁמַעַת נֶחְמָדָה, אֲבָל ה׳ רוֹצֶה אֱמֶת 💛

📖 יִרְמְיָהוּ, פֶּרֶק כ״ח, פְּסוּקִים א׳–ד׳, י׳–י״א

רָמָה א׳ – הַמּוֹטוֹת עַל הַצַּוָּאר

🔨 חֲנַנְיָה שׁוֹבֵר אֶת הַמּוֹטָה

בַּשָּׁנָה הָרְבִעִית לְצִדְקִיָּהוּ אָמַר חֲנַנְיָה בֶן עַזּוּר, נָבִיא מִגִּבְעוֹן, בְּבֵית ה׳: ׳שָׁבַרְתִּי אֶת עֹל מֶלֶךְ בָּבֶל. בְּעוֹד שְׁנָתַיִם יָמִים אֲנִי מֵשִׁיב אֶל הַמָּקוֹם הַזֶּה אֶת כָּל כְּלֵי בֵּית ה׳׳. אַחַר כָּךְ ׳וַיִּקַּח חֲנַנְיָה הַנָּבִיא אֶת הַמּוֹטָה מֵעַל צַוַּאר יִרְמְיָה הַנָּבִיא וַיִּשְׁבְּרֵהוּ׳, ׳וַיֵּלֶךְ יִרְמְיָה הַנָּבִיא לְדַרְכּוֹ׳.

חֲנַנְיָה הִבְטִיחַ חֲזָרָה בְּעוֹד שְׁנָתַיִם, וְשָׁבַר אֶת הַמּוֹטָה בִּפְנֵי הָעָם.

📖 יִרְמְיָהוּ, פֶּרֶק כ״ח, פְּסוּקִים ו׳, י״ג, ט״ו, י״ז

רָמָה א׳ – הַמּוֹטוֹת עַל הַצַּוָּאר

⚒️ ׳אָמֵן׳ וְעֹל בַּרְזֶל

יִרְמְיָהוּ עָנָה: ׳אָמֵן כֵּן יַעֲשֶׂה ה׳׳, כִּי גַּם הוּא רוֹצֶה שֶׁהַכֵּלִים וְהַגּוֹלָה יַחְזְרוּ. אַחַר כָּךְ בָּא דְּבַר ה׳: ׳מוֹטֹת עֵץ שָׁבָרְתָּ וְעָשִׂיתָ תַחְתֵּיהֶן מֹטוֹת בַּרְזֶל׳. וְלַחֲנַנְיָה אָמַר: ׳לֹא שְׁלָחֲךָ ה׳ וְאַתָּה הִבְטַחְתָּ אֶת הָעָם הַזֶּה עַל שָׁקֶר׳. וְכָתוּב: ׳וַיָּמָת חֲנַנְיָה הַנָּבִיא בַּשָּׁנָה הַהִיא׳.

יִרְמְיָהוּ אָמַר ׳אָמֵן׳ מֵאַהֲבָה, וְאַחַר כָּךְ גִּלָּה שֶׁחֲנַנְיָה לֹא נִשְׁלַח: ׳מוֹטוֹת בַּרְזֶל׳ הֵם הָאֱמֶת.

מָה לוֹקְחִים מִרָמָה א׳?

ה׳ אָמַר לַמְּלָכִים: ׳אָנֹכִי עָשִׂיתִי אֶת הָאָרֶץ׳, וְהוּא הַמַּבְטִיחַ ׳וְהַעֲלִיתִים וַהֲשִׁיבֹתִים׳. גַּם כְּשֶׁיֵּשׁ עֹל, ה׳ אִתָּנוּ וְזוֹכֵר אוֹתָנוּ, וְיַחֲזִיר אוֹתָנוּ הַבַּיְתָה.

🤔 חוֹשְׁבִים עַל זֶה

  • ה׳ מַבְטִיחַ ׳וְהַעֲלִיתִים וַהֲשִׁיבֹתִים׳ – הוּא לֹא שׁוֹכֵחַ אֶת מָה שֶׁשַּׁיָּךְ לוֹ. מָה אֲנִי יוֹדֵעַ שֶׁה׳ לֹא שׁוֹכֵחַ אוֹתִי?
  • חֲנַנְיָה אָמַר דְּבָרִים נְעִימִים, וְיִרְמְיָהוּ אָמַר אֱמֶת קָשָׁה. אֵיךְ אֲנִי מַבְחִין בֵּין מִי שֶׁרַק מַנְעִים לִי לְבֵין מִי שֶׁבֶּאֱמֶת אוֹהֵב אוֹתִי?
  • יִרְמְיָהוּ אָמַר ׳אָמֵן׳ לַבְּשׂוֹרָה הַטּוֹבָה שֶׁל חֲנַנְיָה, כִּי גַּם הוּא רָצָה שֶׁכֵּן יִהְיֶה. מָתַי אֲנִי שָׂמֵחַ בְּשִׂמְחָה שֶׁל אַחֵר, גַּם כְּשֶׁאֲנִי חוֹשֵׁב אַחֶרֶת?

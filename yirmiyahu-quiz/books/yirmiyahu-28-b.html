רָמָה ב׳ – עִיּוּן

🧭 נוֹשֵׂא הַיְּחִידָה: ׳שָׁמַע׳

בַּפְּרָקִים מ״ב–מ״ד יֵשׁ שְׁאֵלָה, תְּשׁוּבָה מֵה׳, סֵרוּב וְאַזְהָרָה. הַמִּלָּה הַחוֹזֶרֶת הִיא ׳שָׁמַע׳: ׳נִשְׁמָע בְּקוֹל ה׳׳ (מ״ב ו׳), ׳וְלֹא שָׁמַע יוֹחָנָן בֶּן קָרֵחַ... בְּקוֹל ה׳׳ (מ״ג ד׳), ׳וְלֹא שָׁמְעוּ׳ (מ״ד ה׳). שׁוֹאֲלִים אֶת ה׳ וְגַם צָרִיךְ לִשְׁמֹעַ.

מִלָּה מַנְחָה: ׳שָׁמַע׳. הַבַּקָּשָׁה: ׳נִשְׁמָע׳. הַתּוֹצָאָה: ׳וְלֹא שָׁמְעוּ׳.

📖 יִרְמְיָהוּ, פֶּרֶק מ״ב, פְּסוּקִים ה׳–ו׳ (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן

📖 מְצוּדַת דָּוִד: ׳לְעֵד אֱמֶת וְנֶאֱמָן׳

הָעָם אָמַר: ׳יְהִי ה׳ בָּנוּ לְעֵד אֱמֶת וְנֶאֱמָן אִם לֹא כְּכָל הַדָּבָר... כֵּן נַעֲשֶׂה׳. מְצוּדַת דָּוִד מְבָאֵר: ׳הַמָּקוֹם יִהְיֶה בָּנוּ לְעֵד אֱמֶת שֶׁקִּבַּלְנוּ עַל עַצְמֵנוּ לַעֲשׂוֹת כְּפִי שֶׁתֹּאמַר לָנוּ בִּשְׁמוֹ׳. עַל ׳אִם טוֹב וְאִם רָע׳: ׳בֵּין שֶׁיִּהְיֶה הַדָּבָר טוֹב בְּעֵינֵינוּ בֵּין שֶׁיִּהְיֶה רַע בְּעֵינֵינוּ׳.

הָעָם קִבֵּל עַל עַצְמוֹ לִשְׁמֹעַ בְּקוֹל ה׳, בֵּין אִם הַדָּבָר טוֹב בְּעֵינָיו וּבֵין אִם לֹא.

🌱

💛 אַהֲבַת הַשֵּׁם: ה׳ שָׂמֵחַ לִבְנוֹת מֵחָדָשׁ כְּשֶׁאֲנִי חוֹזֵר אֵלָיו 💛

📖 יִרְמְיָהוּ, פֶּרֶק מ״ב, פָּסוּק י׳ (מְצוּדַת דָּוִד וּמְצוּדַת צִיּוֹן)

רָמָה ב׳ – עִיּוּן

🌱 ׳כִּי נִחַמְתִּי׳ – ה׳ רוֹצֶה לִבְנוֹת

ה׳ אָמַר: ׳כִּי נִחַמְתִּי אֶל הָרָעָה אֲשֶׁר עָשִׂיתִי לָכֶם׳. מְצוּדַת צִיּוֹן: ׳נִחַמְתִּי – עִנְיַן הֲפוֹךְ מַחֲשָׁבָה׳. מְצוּדַת דָּוִד מְבָאֵר: ׳חָשַׁבְתִּי מַחֲשָׁבָה אַחֶרֶת עַל הָרָעָה שֶׁעָשִׂיתִי לָהֶם כִּי לֹא אוֹסִיף עוֹד לְהָרַע לָהֶם׳. וְעַל ׳וּבָנִיתִי אֶתְכֶם׳: ׳יִפְרוּ וְיִרְבּוּ בָהּ וְלֹא יִגְלוּ עוֹד מִמֶּנָּה׳.

ה׳ מוּכָן לְשַׁנּוֹת אֶת הַמַּחֲשָׁבָה כְּשֶׁהָעָם שָׁב: ׳וּבָנִיתִי׳ בִּמְקוֹם ׳אֶהֱרֹס׳.

🛡️

💛 אַהֲבַת הַשֵּׁם: ה׳ מֵגֵן עָלַי וְשׁוֹמֵר גַּם עַל מִי שֶׁרָחוֹק מִמֶּנִּי 💛

📖 יִרְמְיָהוּ, פֶּרֶק מ״ב, פְּסוּקִים י״א–י״ב (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן

🛡️ מְצוּדַת דָּוִד: ׳לְבַל יוּכַל לְהָרַע עִמָּכֶם׳

הָעָם פָּחַד מִמֶּלֶךְ בָּבֶל. מְצוּדַת דָּוִד מְבָאֵר אֶת ׳אֲשֶׁר אַתֶּם יְרֵאִים מִפָּנָיו׳: ׳הַדָּבָר אֲשֶׁר אַתֶּם יְרֵאִים מִפָּנָיו, וְהוּא מֵהֲרִיגַת גְּדַלְיָה׳. ה׳ מַרְגִּיעַ: ׳וּלְהַצִּיל אֶתְכֶם׳ – ׳לְבַל יוּכַל לְהָרַע עִמָּכֶם׳. וְעַל ׳וְהֵשִׁיב אֶתְכֶם׳: ׳הַגּוֹלָה מֵעִמָּכֶם לְבָבֶל יְשִׁיבָם מִרְצוֹנוֹ אֶל אַרְצָם׳.

ה׳ הִבְטִיחַ לְהַצִּיל וְלָתֵת רַחֲמִים, וְגַם לְהָשִׁיב אֶת הַגּוֹלִים לְבָבֶל לְאַרְצָם.

💛 אַהֲבַת הַשֵּׁם: כְּשֶׁאֲנִי בּוֹטֵחַ בְּה׳, אֲנִי לֹא צָרִיךְ לִבְרֹחַ 💛

📖 יִרְמְיָהוּ, פֶּרֶק מ״ב, פְּסוּקִים י״א, י״ד (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן

⚖️ הַשְׁוָאָה: לִבְרֹחַ אוֹ לִבְטֹחַ

הָעָם אָמַר: ׳לֹא נִרְאֶה מִלְחָמָה וְקוֹל שׁוֹפָר לֹא נִשְׁמָע׳. רַשִׁ״י מְבָאֵר: ׳וְקוֹל שׁוֹפָר – שֶׁתּוֹקְעִין הַצּוֹפִים לְהַזְהִיר אֶת הָעָם׳. הָעָם רָצָה לְהִתְרַחֵק מִקּוֹל הַמִּלְחָמָה, וְה׳ אָמַר: ׳אַל תִּירְאוּ... כִּי אִתְּכֶם אָנִי׳. שְׁתֵּי דְּרָכִים: לִבְרֹחַ מֵהַפַּחַד, אוֹ לִבְטֹחַ בְּה׳.

הָעָם בָּרַח מֵהַפַּחַד לְמִצְרַיִם, וְה׳ אָמַר שֶׁדַּוְקָא אִתּוֹ הַמָּקוֹם הַבָּטוּחַ.

📖 יִרְמְיָהוּ, פֶּרֶק מ״ב, פְּסוּקִים ב׳, י״ט–כ׳ (מְצוּדַת צִיּוֹן)

רָמָה ב׳ – עִיּוּן

🔤 מְצוּדַת צִיּוֹן: מִלִּים בַּפָּרָשָׁה

׳תִּפָּל נָא תְחִנָּתֵנוּ׳ – מְצוּדַת צִיּוֹן: ׳תִּפֹּל – עִנְיַן נְטִיָּה׳, ׳נָא – עַתָּה׳. ׳הַעִידֹתִי בָכֶם׳ – ׳עִנְיַן הַתְרָאָה׳. ׳הִתְעֵיתֶם בְּנַפְשׁוֹתֵיכֶם׳ – ׳מִלְּשׁוֹן תּוֹעֶה וְטָעוּת׳, וּמְצוּדַת דָּוִד: ׳עֲשִׂיתֶם טָעוּת גָּדוֹל בְּנַפְשׁוֹתֵיכֶם׳.

הַתְרָאָה הִיא אַזְהָרָה מֵאַהֲבָה: ׳הַעִידֹתִי בָכֶם הַיּוֹם׳.

📖 יִרְמְיָהוּ, פֶּרֶק מ״ד, פְּסוּקִים ד׳, י׳ (מְצוּדַת דָּוִד, רַשִׁ״י)

רָמָה ב׳ – עִיּוּן

🌅 ׳הַשְׁכֵּם וְשָׁלֹחַ׳ – ה׳ מְחַכֶּה שֶׁנָּשׁוּב

ה׳ אָמַר: ׳וָאֶשְׁלַח אֲלֵיכֶם אֶת כָּל עֲבָדַי הַנְּבִיאִים הַשְׁכֵּם וְשָׁלֹחַ׳. מְצוּדַת דָּוִד: ׳הָיִיתִי מַשְׁכִּים וּמְשַׁלֵּחַ בִּזְרִיזוּת רַב׳. עַל ׳אַל נָא תַעֲשׂוּ׳: ׳כְּאוֹמֵר אִם עַד הֵנָּה עֲשִׂיתֶם הָרַע אַל תַּעֲשׂוּ עַתָּה׳. וְרַשִׁ״י עַל ׳לֹא דֻכְּאוּ׳: ׳לִבָּם מֵרָעָתָם׳.

ה׳ שָׁלַח נְבִיאִים בְּזְרִיזוּת, וְחִכָּה שֶׁהַלֵּב שֶׁלָּהֶם יִתְרַכֵּךְ וְיָשׁוּבוּ.

⛓️ ׳עוֹדֶנּוּ עָצוּר׳

גַּם בִּהְיוֹתוֹ בַּכֶּלֶא, ה׳ מְדַבֵּר אֶל יִרְמְיָהוּ פַּעַם שְׁנִיָּה: ׳וַיְהִי דְבַר ה׳ אֶל יִרְמְיָהוּ שֵׁנִית וְהוּא עוֹדֶנּוּ עָצוּר בַּחֲצַר הַמַּטָּרָה׳. מְצוּדַת דָּוִד: ׳שֵׁנִית – בִּדְבַר הַגְּאֻלָּה הָעֲתִידָה׳. הַנָּבִיא בַּכֶּלֶא, אֲבָל ה׳ אִתּוֹ.

ה׳ דִּבֵּר אֶל יִרְמְיָהוּ שֵׁנִית, גַּם כְּשֶׁהוּא ׳עוֹדֶנּוּ עָצוּר׳ בַּחֲצַר הַמַּטָּרָה.

📞 ׳קְרָא אֵלַי וְאֶעֱנֶךָּ׳

ה׳ אוֹמֵר: ׳עֹשָׂהּ ה׳ יוֹצֵר אוֹתָהּ לַהֲכִינָהּ ה׳ שְׁמוֹ׳, וְאַחַר כָּךְ מַזְמִין: ׳קְרָא אֵלַי וְאֶעֱנֶךָּ וְאַגִּידָה לְּךָ גְּדֹלוֹת וּבְצֻרוֹת לֹא יְדַעְתָּם׳. ה׳ מַזְמִין אֶת יִרְמְיָהוּ לִקְרֹא אֵלָיו, וּמַבְטִיחַ לַעֲנוֹת וּלְגַלּוֹת לוֹ דְּבָרִים טוֹבִים שֶׁעוֹד לֹא יָדַע.

׳קְרָא אֵלַי וְאֶעֱנֶךָּ׳ – ה׳ מַזְמִין אוֹתָנוּ לִקְרֹא אֵלָיו, וְהוּא עוֹנֶה.

📖 יִרְמְיָהוּ, פֶּרֶק ל״ג, פְּסוּקִים ו׳–ז׳

רָמָה א׳ – קְרָא אֵלַי

🩺 ׳אֲרֻכָה וּמַרְפֵּא׳

הָעִיר חֲרֵבָה, אֲבָל ה׳ אוֹמֵר: ׳הִנְנִי מַעֲלֶה לָּהּ אֲרֻכָה וּמַרְפֵּא וּרְפָאתִים וְגִלֵּיתִי לָהֶם עֲתֶרֶת שָׁלוֹם וֶאֱמֶת׳. מְצוּדַת דָּוִד: ׳הָעִיר תִּבָּנֶה וְאַנְשֶׁיהָ יָשׁוּבוּ לָהּ׳. וְעוֹד: ׳וַהֲשִׁבֹתִי אֶת שְׁבוּת יְהוּדָה וְאֵת שְׁבוּת יִשְׂרָאֵל וּבְנִתִים כְּבָרִאשֹׁנָה׳.

ה׳ מַבְטִיחַ רְפוּאָה, שָׁלוֹם וֶאֱמֶת, וְשֶׁיָּשִׁיב אֶת הַגּוֹלִים ׳כְּבָרִאשֹׁנָה׳.

💛 אַהֲבַת הַשֵּׁם: ה׳ סוֹלֵחַ לְמִי שֶׁשָּׁב אֵלָיו, וְהוּא מַתְחִיל אִתּוֹ מֵחָדָשׁ 💛

📖 יִרְמְיָהוּ, פֶּרֶק ל״ג, פְּסוּקִים ח׳–ט׳

רָמָה א׳ – קְרָא אֵלַי

🧼 ׳וְטִהַרְתִּים וְסָלַחְתִּי׳

ה׳ מוֹסִיף: ׳וְטִהַרְתִּים מִכָּל עֲוֺנָם אֲשֶׁר חָטְאוּ לִי וְסָלַחְתִּי לְכָל עֲוֺנוֹתֵיהֶם׳. הַסְּלִיחָה שֶׁל ה׳ שְׁלֵמָה. אַחֲרֵי הַסְּלִיחָה הָעִיר תִּהְיֶה ׳לְשֵׁם שָׂשׂוֹן לִתְהִלָּה וּלְתִפְאֶרֶת לְכֹל גּוֹיֵי הָאָרֶץ׳.

ה׳ מְטַהֵר וְסוֹלֵחַ לְכָל עֲוֺנוֹתֵיהֶם, וְהָעִיר תִּהְיֶה ׳לְשֵׁם שָׂשׂוֹן לִתְהִלָּה׳.

רָמָה א׳ – קְרָא אֵלַי

🎶 ׳עוֹד יִשָּׁמַע׳

אֲנָשִׁים אָמְרוּ: ׳חָרֵב הוּא מֵאֵין אָדָם וּמֵאֵין בְּהֵמָה׳. אֲבָל ה׳ אוֹמֵר: ׳עוֹד יִשָּׁמַע בַּמָּקוֹם הַזֶּה... בְּעָרֵי יְהוּדָה וּבְחֻצוֹת יְרוּשָׁלַיִם... קוֹל שָׂשׂוֹן וְקוֹל שִׂמְחָה... קוֹל אֹמְרִים הוֹדוּ אֶת ה׳ צְבָאוֹת כִּי טוֹב ה׳ כִּי לְעוֹלָם חַסְדּוֹ מְבִאִים תּוֹדָה בֵּית ה׳׳.

בִּירוּשָׁלַיִם הֶחָרֵבָה ׳עוֹד יִשָּׁמַע׳ קוֹל שִׂמְחָה וְקוֹל אֹמְרִים: ׳הוֹדוּ אֶת ה׳ צְבָאוֹת כִּי טוֹב ה׳ כִּי לְעוֹלָם חַסְדּוֹ׳.

🐑 ׳נְוֵה רֹעִים׳

הַמָּקוֹם הֶחָרֵב יִהְיֶה שׁוּב מָלֵא חַיִּים: ׳עוֹד יִהְיֶה בַּמָּקוֹם הַזֶּה הֶחָרֵב... נְוֵה רֹעִים מַרְבִּצִים צֹאן׳, ׳עֹד תַּעֲבֹרְנָה הַצֹּאן עַל יְדֵי מוֹנֶה׳. מְצוּדַת דָּוִד: ׳כָּל כָּךְ יִתְרַבּוּ הַצֹּאן עַד שֶׁלֹּא יַסְפִּיקוּ בְּעָלֶיהָ לִמְנוֹתָם בְּעַצְמָם׳.

הַמָּקוֹם הֶחָרֵב יִהְיֶה שׁוּב ׳נְוֵה רֹעִים מַרְבִּצִים צֹאן׳, וְהַצֹּאן תִּתְרַבֶּה.

רָמָה א׳ – קְרָא אֵלַי

🌿 ׳צֶמַח צְדָקָה׳

ה׳ אוֹמֵר: ׳הִנֵּה יָמִים בָּאִים נְאֻם ה׳ וַהֲקִמֹתִי אֶת הַדָּבָר הַטּוֹב אֲשֶׁר דִּבַּרְתִּי׳. ׳אַצְמִיחַ לְדָוִד צֶמַח צְדָקָה וְעָשָׂה מִשְׁפָּט וּצְדָקָה בָּאָרֶץ׳. מְצוּדַת דָּוִד: ׳זֶהוּ מֶלֶךְ הַמָּשִׁיחַ׳. וּבְיָמִים הָהֵם ׳תִּוָּשַׁע יְהוּדָה וִירוּשָׁלַיִם תִּשְׁכּוֹן לָבֶטַח׳, וְיִקְרְאוּ לָהּ ׳ה׳ צִדְקֵנוּ׳.

ה׳ יַצְמִיחַ לְדָוִד ׳צֶמַח צְדָקָה׳ – מֶלֶךְ הַמָּשִׁיחַ – וִירוּשָׁלַיִם תִּשְׁכּוֹן לָבֶטַח.

🌗 ׳בְּרִיתִי יוֹמָם וָלָיְלָה׳

ה׳ אוֹמֵר שֶׁהַבְטָחָתוֹ בְּטוּחָה כְּמוֹ הַיּוֹם וְהַלַּיְלָה: ׳אִם תָּפֵרוּ אֶת בְּרִיתִי הַיּוֹם וְאֶת בְּרִיתִי הַלָּיְלָה... גַּם בְּרִיתִי תֻפַר אֶת דָּוִד עַבְדִּי׳. וְעוֹד: ׳אִם לֹא בְרִיתִי יוֹמָם וָלָיְלָה חֻקּוֹת שָׁמַיִם וָאָרֶץ לֹא שָׂמְתִּי גַּם זֶרַע יַעֲקוֹב וְדָוִד עַבְדִּי אֶמְאַס... כִּי אָשִׁיב אֶת שְׁבוּתָם וְרִחַמְתִּים׳.

הַבְּרִית שֶׁל ה׳ עִם עַמּוֹ בְּטוּחָה כְּמוֹ הַיּוֹם וְהַלַּיְלָה: ׳כִּי אָשִׁיב אֶת שְׁבוּתָם וְרִחַמְתִּים׳.

מָה לוֹקְחִים מִרָמָה א׳?

מִתּוֹךְ הַכֶּלֶא, ה׳ אָמַר לְיִרְמְיָהוּ: ׳קְרָא אֵלַי וְאֶעֱנֶךָּ׳, ׳עוֹד יִשָּׁמַע׳ שִׂמְחָה וְהוֹדָיָה, וּבְרִיתִי בְּטוּחָה כְּמוֹ הַיּוֹם וְהַלַּיְלָה. כָּךְ לָמַדְנוּ שֶׁה׳ שׁוֹמֵעַ אוֹתָנוּ, מְרַפֵּא אֶת הַלֵּב וְאוֹהֵב אוֹתָנוּ תָּמִיד.

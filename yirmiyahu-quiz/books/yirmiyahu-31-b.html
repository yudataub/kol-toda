💭 ׳מֵעַל פָּנָיו׳

׳כִּי עַל אַף ה׳ הָיְתָה בִּירוּשָׁלִַם וִיהוּדָה עַד הַשְׁלִיכוֹ אוֹתָם מֵעַל פָּנָיו׳. רַשִׁ״י מְבָאֵר: ׳הִיא גָּרְמָה לוֹ מֶרֶד צִדְקִיָּהוּ בְּמֶלֶךְ בָּבֶל שֶׁהוּא תְּחִלַּת הַחֻרְבָּן׳. שִׂימוּ לֵב לַמִּלִּים ׳מֵעַל פָּנָיו׳: הַגָּלוּת הִיא הַרְחָקָה מִקֵּרוּב. מִי שֶׁקָּרוֹב אֶל ה׳ נִמְצָא לְפָנָיו, וְלָכֵן הַגַּעְגּוּעַ גָּדוֹל כָּל כָּךְ.

גָּלוּת פֵּרוּשָׁהּ לִהְיוֹת רָחוֹק מִפְּנֵי ה׳, וְלָכֵן אָנוּ מִתְגַּעְגְּעִים לַחֲזֹר.

🔢 עֲשֶׂרֶת אֲלָפִים גּוֹלִים

׳זֶה הָעָם אֲשֶׁר הֶגְלָה נְבוּכַדְרֶאצַּר בִּשְׁנַת שֶׁבַע יְהוּדִים שְׁלֹשֶׁת אֲלָפִים וְעֶשְׂרִים וּשְׁלֹשָׁה׳. רַשִׁ״י מְבָאֵר שֶׁזּוֹ גָּלוּת יְכָנְיָה, וּמוֹסִיף שֶׁבְּסֵפֶר מְלָכִים ב׳ יֵשׁ פָּסוּק אֶחָד שֶׁאוֹמֵר עֲשֶׂרֶת אֲלָפִים וּפָסוּק אַחֵר שֶׁאוֹמֵר שִׁבְעַת אֲלָפִים: ׳יְהוּדִים שְׁלֹשֶׁת אֲלָפִים, וּמִשְּׁאָר שְׁבָטִים שִׁבְעַת אֲלָפִים, הֲרֵי עֲשֶׂרֶת אֲלָפִים׳. כָּךְ הַפָּסוּק כָּאן ׳הִכְרִיעַ בֵּין שְׁנֵי מִקְרָאוֹת׳.

גָּלוּת יְכָנְיָה כְּלְלָה שְׁלֹשֶׁת אֲלָפִים מִיהוּדָה וְעוֹד שִׁבְעַת אֲלָפִים מִשְּׁאָר הַשְּׁבָטִים.

🔤 הַמִּלָּה ׳נָשָׂא׳

׳נָשָׂא אֱוִיל מְרֹדַךְ... אֶת רֹאשׁ יְהוֹיָכִין׳. רַשִׁ״י מְבָאֵר: ׳נְשָׂאוֹ וְגִדְּלוֹ׳. וּבִלְשׁוֹן אַחֶרֶת: ׳מְנָאוֹ עִם שְׁאָר הַמְּלָכִים׳. מְצוּדַת דָּוִד מַסְבִּיר: ׳הֵרִים וְגִדֵּל אוֹתוֹ׳. ׳נָשָׂא אֶת הָרֹאשׁ׳ הוּא גַּם לְשׁוֹן מִנְיָן וְגַם לְשׁוֹן הֲרָמָה. הַמֶּלֶךְ שֶׁהָיָה בְּבֵית הַכֶּלֶא קָם, הָרֹאשׁ שֶׁלּוֹ הוּרַם.

׳נָשָׂא אֶת הָרֹאשׁ׳ פֵּרוּשׁוֹ לְהָרִים וּלְגַדֵּל, וְגַם לִמְנוֹת בֵּין הַחֲשׁוּבִים.

📅 שְׁנֵי תַּאֲרִיכִים לְאוֹתוֹ יוֹם

בְּיִרְמְיָהוּ כָּתוּב: ׳בְּעֶשְׂרִים וַחֲמִשָּׁה לַחֹדֶשׁ׳, וּבְסֵפֶר מְלָכִים ב׳ (כ״ה, כ״ז) כָּתוּב: ׳בְּעֶשְׂרִים וְשִׁבְעָה לַחֹדֶשׁ׳. רַשִׁ״י מְיַשֵּׁב: בְּעֶשְׂרִים וַחֲמִשָּׁה מֵת נְבוּכַדְנֶאצַּר, וּבְעֶשְׂרִים וְשִׁבְעָה מָלַךְ אֱוִיל מְרֹדַךְ וְנָשָׂא אֶת רֹאשׁ יְהוֹיָכִין. אַרְבַּעַת הַפְּסוּקִים הָאַחֲרוֹנִים שֶׁל שְׁנֵי הַסְּפָרִים כִּמְעַט זֵהִים, כִּי שְׁנֵיהֶם מְסַיְּמִים בְּאוֹתָהּ תִּקְוָה.

שְׁנֵי הַסְּפָרִים, יִרְמְיָהוּ וּמְלָכִים, מְסַיְּמִים בְּאוֹתוֹ סִפּוּר שֶׁל תִּקְוָה.

׳וַיְדַבֵּר אִתּוֹ טֹבוֹת וַיִּתֵּן אֶת כִּסְאוֹ מִמַּעַל לְכִסֵּא הַמְּלָכִים אֲשֶׁר אִתּוֹ בְּבָבֶל׳. ׳וְשִׁנָּה אֵת בִּגְדֵי כִלְאוֹ וְאָכַל לֶחֶם לְפָנָיו תָּמִיד כָּל יְמֵי חַיָּו׳. וְגַם: ׳וַאֲרֻחָתוֹ אֲרֻחַת תָּמִיד נִתְּנָה לּוֹ... דְּבַר יוֹם בְּיוֹמוֹ׳. מְצוּדַת דָּוִד מְבָאֵר: ׳הָיָה מְשַׁנֶּה אֶת בְּגָדָיו אֲשֶׁר יָשַׁב בָּהֶן בְּבֵית הַכֶּלֶא לְהַחֲלִיפָם בַּאֲחֵרִים׳.

יְהוֹיָכִין קִבֵּל כִּסֵּא מְכֻבָּד, בְּגָדִים חֲדָשִׁים וְאֲרוּחָה קְבוּעָה כָּל יוֹם.

🕍 ׳בֵּית הַגָּדוֹל׳

הַכָּתוּב אוֹמֵר: ׳וַיִּשְׂרֹף אֶת בֵּית ה׳ וְאֶת בֵּית הַמֶּלֶךְ... וְאֶת כָּל בֵּית הַגָּדוֹל שָׂרַף בָּאֵשׁ׳. רַשִׁ״י מְבָאֵר: ׳בֵּית הַגָּדוֹל – בָּתֵּי כְנֵסִיּוֹת שֶׁמְּגַדְּלִין בָּהֶם תְּפִלָּה, וְיֵשׁ אוֹמְרִים בָּתֵּי מִדְרָשׁוֹת שֶׁמְּגַדְּלִין בָּהֶם אֶת הַתּוֹרָה׳. כְּלוֹמַר: גַּם הַמְּקוֹמוֹת שֶׁגִּדְּלוּ אֶת הַתְּפִלָּה וְאֶת הַתּוֹרָה הָיוּ יְקָרִים.

בָּתֵּי הַתְּפִלָּה וּבָתֵּי הַמִּדְרָשׁ ׳מְגַדְּלִים׳ אֶת הַתְּפִלָּה וְאֶת הַתּוֹרָה.

🍇 כְּרָמִים וְיֹגְבִים

׳וּמִדַּלּוֹת הָאָרֶץ הִשְׁאִיר נְבוּזַרְאֲדָן רַב טַבָּחִים לְכֹרְמִים וּלְיֹגְבִים׳. רַשִׁ״י מְבָאֵר: ׳יֹגְבִים – חוֹפְרִין בָּאָרֶץ׳. הָאָרֶץ לֹא נִשְׁאֲרָה רֵיקָה: הָעֲנִיִּים נִשְׁאֲרוּ לַעֲבֹד אוֹתָהּ. וְיִרְמְיָהוּ כְּבָר הִבְטִיחַ בְּפֶרֶק ל״ב: ׳עוֹד יִקָּנוּ בָתִּים וְשָׂדוֹת וּכְרָמִים בָּאָרֶץ הַזֹּאת׳.

הָאָרֶץ לֹא נִשְׁאֲרָה רֵיקָה, וְהַנְּבוּאָה אוֹמֶרֶת שֶׁעוֹד יִקְנוּ בָּהּ שָׂדוֹת וּכְרָמִים.

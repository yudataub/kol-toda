🏛️ ה׳ שׁוֹלֵחַ אֶת יִרְמְיָהוּ

׳בְּרֵאשִׁית מַמְלְכוּת יְהוֹיָקִים בֶּן יֹאשִׁיָּהוּ׳ אָמַר ה׳ לְיִרְמְיָהוּ: ׳עֲמֹד בַּחֲצַר בֵּית ה׳׳. שָׁם בָּאוּ אֲנָשִׁים מִכָּל עָרֵי יְהוּדָה לְהִשְׁתַּחֲווֹת, וְיִרְמְיָהוּ צָרִיךְ לְדַבֵּר אֲלֵיהֶם: ׳אַל תִּגְרַע דָּבָר׳.

ה׳ שָׁלַח אֶת יִרְמְיָהוּ לַחֲצַר בֵּית ה׳ לְדַבֵּר אֶל כָּל הַבָּאִים, בְּלִי לִגְרֹעַ אֲפִלּוּ דָּבָר אֶחָד.

💛 ׳אוּלַי יִשְׁמְעוּ׳

ה׳ אָמַר: ׳אוּלַי יִשְׁמְעוּ וְיָשֻׁבוּ אִישׁ מִדַּרְכּוֹ הָרָעָה וְנִחַמְתִּי אֶל הָרָעָה אֲשֶׁר אָנֹכִי חֹשֵׁב לַעֲשׂוֹת לָהֶם׳. הַמִּלָּה ׳אוּלַי׳ מַרְאָה דָּבָר יָפֶה: ה׳ מְקַוֶּה שֶׁהָעָם יָשׁוּבוּ. וְאִם יָשׁוּבוּ – הָרָעָה לֹא תָּבוֹא.

׳אוּלַי יִשְׁמְעוּ׳ – ה׳ מְקַוֶּה שֶׁנָּשׁוּב, וְאָז הָרָעָה לֹא תָּבוֹא.

💛 אַהֲבַת הַשֵּׁם: ה׳ שָׁלַח אֵלַי תּוֹרָה וּנְבִיאִים כִּי הוּא רוֹצֶה שֶׁאֶהְיֶה קָרוֹב אֵלָיו 💛

📜 הַתּוֹרָה וְהַנְּבִיאִים

׳אִם לֹא תִשְׁמְעוּ אֵלַי לָלֶכֶת בְּתוֹרָתִי אֲשֶׁר נָתַתִּי לִפְנֵיכֶם׳. ה׳ נָתַן לָעָם תּוֹרָה, וְגַם שָׁלַח אֶת ׳עֲבָדַי הַנְּבִאִים׳ – ׳הַשְׁכֵּם וְשָׁלֹחַ׳, שׁוּב וָשׁוּב, מֻקְדָּם בַּבֹּקֶר.

ה׳ לֹא הִסְתַּפֵּק בַּתּוֹרָה – הוּא שָׁלַח נְבִיאִים שׁוּב וָשׁוּב, ׳הַשְׁכֵּם וְשָׁלֹחַ׳.

🏚️ ׳כְּשִׁלֹה׳

אִם הָעָם לֹא יִשְׁמְעוּ – ׳וְנָתַתִּי אֶת הַבַּיִת הַזֶּה כְּשִׁלֹה׳. מְצוּדַת דָּוִד מְבָאֵר: כְּמוֹ שֶׁמִּשְׁכַּן שִׁלֹה חָרַב בִּימֵי עֵלִי. זוֹ אַזְהָרָה, וְלֹא גְּזֵרָה סוֹפִית: כָּל עוֹד ׳אוּלַי יִשְׁמְעוּ׳, אֶפְשָׁר לְשַׁנּוֹת.

הָאַזְהָרָה: ׳כְּשִׁלֹה׳ – אֲבָל עוֹד אֶפְשָׁר לָשׁוּב.

📖 יִרְמְיָהוּ, פֶּרֶק כ״ו, פְּסוּקִים ז׳–ט׳

😠 תּוֹפְשִׂים אֶת הַנָּבִיא

כְּשֶׁיִּרְמְיָהוּ סִיֵּם לְדַבֵּר, ׳וַיִּתְפְּשׂוּ אֹתוֹ הַכֹּהֲנִים וְהַנְּבִאִים וְכָל הָעָם׳. הֵם כָּעֲסוּ: ׳מַדּוּעַ נִבֵּיתָ בְשֵׁם ה׳?׳ ׳וַיִּקָּהֵל כָּל הָעָם אֶל יִרְמְיָהוּ בְּבֵית ה׳׳, וְהֵם דָּרְשׁוּ לְהַעֲנִישׁ אוֹתוֹ בְּחֻמְרָה.

הַכֹּהֲנִים, הַנְּבִיאִים וְהָעָם כָּעֲסוּ עַל יִרְמְיָהוּ וְתָפְשׂוּ אוֹתוֹ בְּבֵית ה׳.

💛 אַהֲבַת הַשֵּׁם: ה׳ מַשְׁגִּיחַ עַל שְׁלִיחוֹ גַּם בְּרֶגַע קָשֶׁה 💛

⚖️ הַשָּׂרִים בַּשַּׁעַר הֶחָדָשׁ

כְּשֶׁשָּׁמְעוּ שָׂרֵי יְהוּדָה מָה קָרָה, ׳וַיַּעֲלוּ מִבֵּית הַמֶּלֶךְ בֵּית ה׳ וַיֵּשְׁבוּ בְּפֶתַח שַׁעַר ה׳ הֶחָדָשׁ׳. שָׁם נִפְתַּח הַמִּשְׁפָּט: הַכֹּהֲנִים וְהַנְּבִיאִים טָעֲנוּ נֶגֶד יִרְמְיָהוּ, וְהַשָּׂרִים וְהָעָם הִקְשִׁיבוּ.

הַמִּשְׁפָּט נִפְתַּח בְּפֶתַח שַׁעַר ה׳ הֶחָדָשׁ, לִפְנֵי הַשָּׂרִים וְכָל הָעָם.

רָמָה א׳ – הַנְּבוּאָה וְהַמִּשְׁפָּט

🗣️ ׳הֵיטִיבוּ דַרְכֵיכֶם׳

יִרְמְיָהוּ לֹא פָּחַד. הוּא אָמַר: ׳ה׳ שְׁלָחַנִי׳, וְאַחַר כָּךְ קָרָא לָעָם: ׳וְעַתָּה הֵיטִיבוּ דַרְכֵיכֶם וּמַעַלְלֵיכֶם וְשִׁמְעוּ בְּקוֹל ה׳ אֱלֹהֵיכֶם וְיִנָּחֵם ה׳ אֶל הָרָעָה׳. עַל עַצְמוֹ אָמַר רַק: ׳וַאֲנִי הִנְנִי בְיֶדְכֶם׳.

גַּם כְּשֶׁתָּפְשׂוּ אוֹתוֹ, יִרְמְיָהוּ בִּקֵּשׁ מֵהֶם דָּבָר אֶחָד: לָשׁוּב לַה׳.

רָמָה א׳ – הַנְּבוּאָה וְהַמִּשְׁפָּט

🌟 ׳בְּשֵׁם ה׳ אֱלֹהֵינוּ דִּבֶּר אֵלֵינוּ׳

הַשָּׂרִים וְכָל הָעָם אָמְרוּ לַכֹּהֲנִים וְלַנְּבִיאִים: אֵין לְהַעֲנִישׁ אֶת יִרְמְיָהוּ, ׳כִּי בְּשֵׁם ה׳ אֱלֹהֵינוּ דִּבֶּר אֵלֵינוּ׳. וּבַסּוֹף: ׳אַךְ יַד אֲחִיקָם בֶּן שָׁפָן הָיְתָה אֶת יִרְמְיָהוּ לְבִלְתִּי תֵּת אֹתוֹ בְּיַד הָעָם׳. ה׳ שָׁמַר עַל יִרְמְיָהוּ.

הָעָם הֵבִינוּ שֶׁיִּרְמְיָהוּ דִּבֶּר בְּשֵׁם ה׳, וַאֲחִיקָם בֶּן שָׁפָן הֵגֵן עָלָיו.

מָה לוֹקְחִים מִרָמָה א׳?

ה׳ אָמַר ׳אוּלַי יִשְׁמְעוּ׳ – הוּא מְקַוֶּה לָנוּ, שׁוֹלֵחַ לָנוּ נְבִיאִים בְּכָל בֹּקֶר, וּמוּכָן ׳לְהִנָּחֵם׳ עַל הָרָעָה. כָּךְ לָמַדְנוּ כַּמָּה ה׳ אוֹהֵב אוֹתָנוּ וּמְחַכֶּה שֶׁנָּשׁוּב אֵלָיו.

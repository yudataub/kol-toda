וְהַעֲלִיתִים וַהֲשִׁיבֹתִים אֶל הַמָּקוֹם הַזֶּה

📜 סֵפֶר יִרְמְיָהוּ · הַמּוֹטוֹת וְהָעֹל · רָמָה ב׳

הַמּוֹטוֹת וְהָעֹל

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

יִרְמְיָהוּ הוֹלֵךְ עִם עֹל עַל צַוָּארוֹ, חֲנַנְיָה שׁוֹבֵר אֶת הַמּוֹטָה, וְה׳ מַבְטִיחַ שֶׁכְּלֵי בֵּית ה׳ יַחְזְרוּ – כִּי הוּא לֹא שׁוֹכֵחַ אֶת עַמּוֹ

📖 פְּרָקִים כ״ז–כ״ח

🎯 לַחִידוֹן

💛 אַהֲבַת הַשֵּׁם: ה׳ מַבְטִיחַ אֱמֶת, וְלֹא רַק דְּבָרִים נְעִימִים 💛

📖 יִרְמְיָהוּ, פֶּרֶק כ״ז, פְּסוּקִים י׳, י״ד, ט״ז; פֶּרֶק כ״ח, פָּסוּק ט״ו

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

🧭 נוֹשֵׂא הַיְּחִידָה

בְּפֶרֶק כ״ז יֵשׁ מַעֲשֶׂה סִמְלִי – הַמּוֹטוֹת – וְשִׁלּוּחַ לַחֲמִשָּׁה מְלָכִים, וּפְנִיָּה לְצִדְקִיָּהוּ וְלַכֹּהֲנִים. בְּפֶרֶק כ״ח יֵשׁ עִמּוּת עִם חֲנַנְיָה. הַמִּלָּה הַמַּנְחָה: ׳שֶׁקֶר׳ – ׳שֶׁקֶר הֵם נִבְּאִים׳ (כ״ז י׳, י״ד, ט״ז) וְ׳הִבְטַחְתָּ... עַל שָׁקֶר׳ (כ״ח ט״ו). מוּלָהּ הַהַבְטָחָה הָאֲמִתִּית: ׳וְהַעֲלִיתִים וַהֲשִׁיבֹתִים׳.

מִלָּה מַנְחָה: ׳שֶׁקֶר׳ – מוּל הַהַבְטָחָה הָאֲמִתִּית ׳וַהֲשִׁיבֹתִים׳.

1

💛 אַהֲבַת הַשֵּׁם: ה׳ מַזְהִיר מֵרֹאשׁ, עוֹד לִפְנֵי שֶׁהַדְּבָרִים קוֹרִים 💛

📖 יִרְמְיָהוּ, פֶּרֶק כ״ז, פְּסוּקִים א׳, ג׳ (רַשִׁ״י, מְצוּדַת דָּוִד)

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

📖 רַשִׁ״י: הָרֶקַע

רַשִׁ״י מְבָאֵר אֶת הָרֶקַע: ׳בִּימֵי צִדְקִיָּהוּ בַּשָּׁנָה הָרְבִיעִית לְמָלְכוֹ, שֶׁעֲשָׂהוּ נְבוּכַדְנֶאצַּר שַׁלִּיט עַל חֲמִשָּׁה מְלָכִים׳. וְעַל ׳בְּרֵאשִׁית מַמְלֶכֶת יְהוֹיָקִם׳ מְצוּדַת דָּוִד מְבָאֵר: בִּימֵי יְהוֹיָקִים אָמַר לוֹ ה׳ מֵרֹאשׁ שֶׁכְּשֶׁיִּמְלֹךְ צִדְקִיָּהוּ יִשְׁלַח אֶת הַמּוֹסֵרוֹת וְהַמּוֹטוֹת אֶל הַמְּלָכִים.

ה׳ הִזְהִיר מֵרֹאשׁ, שָׁנִים לִפְנֵי שֶׁהַדְּבָרִים הִתְרַחֲשׁוּ.

2

💛 אַהֲבַת הַשֵּׁם: ה׳ מַנְהִיג אֶת הָעוֹלָם, וְלַצָּרוֹת יֵשׁ סוֹף 💛

📖 יִרְמְיָהוּ, פֶּרֶק כ״ז, פָּסוּק ז׳ (רַשִׁ״י)

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

🕰️ ׳עַד בֹּא עֵת אַרְצוֹ׳

׳וְעָבְדוּ אֹתוֹ כָּל הַגּוֹיִם וְאֶת בְּנוֹ וְאֶת בֶּן בְּנוֹ עַד בֹּא עֵת אַרְצוֹ גַּם הוּא׳. רַשִׁ״י מְבָאֵר: ׳וְאֶת בְּנוֹ – אֱוִיל מְרֹדַךְ, וְאֶת בֶּן בְּנוֹ – בֵּלְשַׁאצַּר, עַד בֹּא עֵת אַרְצוֹ – שֶׁיָּבוֹאוּ דָּרְיָוֶשׁ הַמָּדִי וְכוֹרֶשׁ׳. גַּם לַשַּׁלִּיט הַגָּדוֹל יֵשׁ סוֹף, וְה׳ הוּא הַמּוֹשֵׁל בָּעוֹלָם. הָעֹל זְמַנִּי.

הָעֹל זְמַנִּי: ׳עַד בֹּא עֵת אַרְצוֹ גַּם הוּא׳ – בִּימֵי דָּרְיָוֶשׁ וְכוֹרֶשׁ.

3

💛 אַהֲבַת הַשֵּׁם: מִי שֶׁאוֹהֵב אוֹתִי אוֹמֵר לִי אֶת הָאֱמֶת שֶׁתַּצִּיל אוֹתִי 💛

📖 יִרְמְיָהוּ, פֶּרֶק כ״ז, פְּסוּקִים י׳, ט״ו (מְצוּדַת דָּוִד)

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

❓ לָמָּה אָמְרוּ כָּךְ נְבִיאֵי הַשֶּׁקֶר?

מְצוּדַת דָּוִד מְבָאֵר: ׳כְּשֶׁתְּקַבְּלוּ עֲלֵיכֶם אֶת הַשִּׁעְבּוּד תֵּשְׁבוּ בְּאַרְצְכֶם וְתַעֲלוּ הַמַּס, אֲבָל כְּשֶׁתִּלָּחֲמוּ עִמּוֹ וְיִגְבַּר עֲלֵיכֶם – יוֹלִיךְ אֶתְכֶם בַּשֶּׁבִי אֶל אֶרֶץ מֵרָחוֹק׳. כְּלוֹמַר, הַבְּשׂוֹרָה הַיָּפָה שֶׁל הַנְּבִיאִים הַשַּׁקְרָנִים תָּבִיא לְגָלוּת, וְהָאֱמֶת הַקָּשָׁה שֶׁל יִרְמְיָהוּ תַּשְׁאִיר אֶתְכֶם עַל אַדְמַתְכֶם.

דְּבָרִים נְעִימִים שֶׁאֵינָם אֱמֶת יְכוֹלִים לְהַזִּיק; אֱמֶת קָשָׁה מַצִּילָה.

4

💛 אַהֲבַת הַשֵּׁם: ה׳ זוֹכֵר אוֹתָנוּ, גַּם כְּשֶׁנִּדְמֶה לָנוּ שֶׁהוּא רָחוֹק 💛

📖 יִרְמְיָהוּ, פֶּרֶק כ״ז, פָּסוּק כ״ב (רַשִׁ״י, מְצוּדַת דָּוִד)

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

🔔 ׳עַד יוֹם פָּקְדִי אֹתָם׳

הַכֵּלִים יֵלְכוּ לְבָבֶל ׳עַד יוֹם פָּקְדִי אֹתָם׳. רַשִׁ״י מְבָאֵר: ׳בִּימֵי כוֹרֶשׁ, כְּמוֹ שֶׁמְּפֹרָשׁ בְּסֵפֶר עֶזְרָא׳. וּמְצוּדַת דָּוִד: ׳עַד הַזְּמַן שֶׁאֶזְכֹּר אֲנִי בָּהֶם... וְאָז אַעֲלֶה אוֹתָם מִשָּׁם וְאָשִׁיב אוֹתָם אֶל הַמָּקוֹם הַזֶּה׳. ׳פְּקִידָה׳ הִיא זִכָּרוֹן: ה׳ ׳זוֹכֵר׳ אֶת כֵּלָיו וְאֶת עַמּוֹ בַּיּוֹם שֶׁקָּבַע.

׳יוֹם פָּקְדִי׳ – ה׳ זוֹכֵר בַּיּוֹם שֶׁקָּבַע, בִּימֵי כוֹרֶשׁ, וּמַחֲזִיר אֶת הַכֹּל.

5

💛 אַהֲבַת הַשֵּׁם: נָבִיא שֶׁאוֹהֵב אֶת עַמּוֹ מִתְאַוֶּה לִבְשׂוֹרוֹת טוֹבוֹת 💛

📖 יִרְמְיָהוּ, פֶּרֶק כ״ח, פְּסוּקִים ו׳–ז׳, ט׳ (מְצוּדַת דָּוִד)

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

🤲 יִרְמְיָהוּ אוֹמֵר ׳אָמֵן׳

מְצוּדַת דָּוִד מְבָאֵר: ׳אָמֵן – יֵאָמֵן דְּבָרֶיךָ׳, ׳מִי יִתֵּן וְיָקֵם ה׳׳. וּבְפָסוּק ז׳: ׳מַה שֶּׁאָמַרְתִּי אָמֵן הוּא בִּגְלַל שֶׁגַּם אֲנִי מִתְאַוֶּה שֶׁכֵּן יִהְיֶה כִּי טוֹב הוּא׳. יִרְמְיָהוּ לֹא נֶהֱנֶה מִבְּשׂוֹרוֹת רָעוֹת. אַחַר כָּךְ הוּא נוֹתֵן מִבְחָן: ׳הַנָּבִיא אֲשֶׁר יִנָּבֵא לְשָׁלוֹם בְּבֹא דְּבַר הַנָּבִיא יִוָּדַע הַנָּבִיא אֲשֶׁר שְׁלָחוֹ ה׳ בֶּאֱמֶת׳.

יִרְמְיָהוּ רָצָה שֶׁהַבְּשׂוֹרָה תִּתְקַיֵּם, וְנָתַן מִבְחָן: ׳בְּבֹא דְּבַר הַנָּבִיא יִוָּדַע׳.

6

💛 אַהֲבַת הַשֵּׁם: ה׳ מַזְהִיר אוֹתִי כְּדֵי שֶׁלֹּא יִהְיֶה לִי כָּבֵד 💛

📖 יִרְמְיָהוּ, פֶּרֶק כ״ח, פְּסוּקִים י״ג–י״ד (מְצוּדַת דָּוִד)

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

⚒️ עֵץ וּבַרְזֶל

מְצוּדַת דָּוִד מְבָאֵר: ׳עַל יְדֵי הַשְּׁבִירָה וְהַכְחָשַׁת דְּבַר ה׳ יִגְדַּל הַקֶּצֶף וְיֶחֱזַק עֻלּוֹ עוֹד יוֹתֵר׳. כְּשֶׁשּׁוֹבְרִים סֵמֶל, הַמְּצִיאוּת לֹא מִשְׁתַּנָּה. עֹל עֵץ אֶפְשָׁר לְשַׁבֵּר, אֲבָל כְּשֶׁלֹּא שׁוֹמְעִים לְה׳, הָעֹל נַעֲשֶׂה כָּבֵד כְּמוֹ בַּרְזֶל. לְעֻמַּת זֹאת, מִי שֶׁשּׁוֹמֵעַ – ׳וְיָשַׁב בָּהּ׳.

מוֹטוֹת עֵץ שֶׁנִּשְׁבְּרוּ הָפְכוּ ל׳מוֹטוֹת בַּרְזֶל׳: מִי שֶׁלֹּא שׁוֹמֵעַ מַכְבִּיד עַל עַצְמוֹ.

7
💡

רָמָה ב׳ – שְׁאֵלוֹת לַחֲכָמִים

מָה לוֹקְחִים מִרָמָה ב׳?

ה׳ שׁוֹלֵחַ סֵמֶל וְלֹא רַק מִלִּים: מוֹטוֹת עַל הַצַּוָּאר, וּכְלֵי בֵּית ה׳ שֶׁיָּשׁוּבוּ. וּבְתוֹךְ הָעֹל כָּל הַזְּמַן הַהַבְטָחָה ׳וְהַעֲלִיתִים וַהֲשִׁיבֹתִים׳. מִי שֶׁשּׁוֹמֵעַ אֶת הָאֱמֶת מַרְגִּישׁ שֶׁה׳ אוֹהֵב אוֹתוֹ וְלֹא עוֹזֵב אוֹתוֹ.

8

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • ה׳ מַבְטִיחַ ׳וְהַעֲלִיתִים וַהֲשִׁיבֹתִים׳ – הוּא לֹא שׁוֹכֵחַ אֶת מָה שֶׁשַּׁיָּךְ לוֹ. מָה אֲנִי יוֹדֵעַ שֶׁה׳ לֹא שׁוֹכֵחַ אוֹתִי?
  • חֲנַנְיָה אָמַר דְּבָרִים נְעִימִים, וְיִרְמְיָהוּ אָמַר אֱמֶת קָשָׁה. אֵיךְ אֲנִי מַבְחִין בֵּין מִי שֶׁרַק מַנְעִים לִי לְבֵין מִי שֶׁבֶּאֱמֶת אוֹהֵב אוֹתִי?
  • יִרְמְיָהוּ אָמַר ׳אָמֵן׳ לַבְּשׂוֹרָה הַטּוֹבָה שֶׁל חֲנַנְיָה, כִּי גַּם הוּא רָצָה שֶׁכֵּן יִהְיֶה. מָתַי אֲנִי שָׂמֵחַ בְּשִׂמְחָה שֶׁל אַחֵר, גַּם כְּשֶׁאֲנִי חוֹשֵׁב אַחֶרֶת?

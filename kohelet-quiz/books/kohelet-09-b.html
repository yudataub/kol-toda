וּזְכֹר אֶת בּוֹרְאֶיךָ בִּימֵי בְּחוּרֹתֶיךָ

⏳ מְגִלַּת קֹהֶלֶת · וּזְכֹר אֶת בּוֹרְאֶיךָ · רָמָה ב׳

וּזְכֹר אֶת בּוֹרְאֶיךָ

רָמָה ב׳ – עִיּוּן בַּפְּסוּקִים

שְׂמַח בַּנְּעוּרִים, וּזְכֹר אֶת בּוֹרְאֶיךָ כְּבָר עַכְשָׁיו – לִפְנֵי שֶׁיָּבוֹאוּ יְמֵי הָרָעָה, כְּשֶׁהָאוֹר עוֹד בָּהִיר

📖 פְּרָקִים י״א–י״ב

🎯 לַחִידוֹן

📜 הַפָּסוּקשְׂמַח בָּחוּר בְּיַלְדוּתֶיךָ וִיטִיבְךָ לִבְּךָ בִּימֵי בְחוּרוֹתֶךָ וְהַלֵּךְ בְּדַרְכֵי לִבְּךָ וּבְמַרְאֵי עֵינֶיךָ וְדָע כִּי עַל כָּל אֵלֶּה יְבִיאֲךָ הָאֱלֹהִים בַּמִּשְׁפָּט

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁאֶשְׂמַח, וְגַם שֶׁאֶהְיֶה אַחְרָאִי 💛

📖 קֹהֶלֶת, פֶּרֶק י״א, פָּסוּק ט׳ (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּפְּסוּקִים

🧭 מְצוּדַת דָּוִד: ׳בְּדַרְכֵי לִבְּךָ׳

מְצוּדַת דָּוִד מְפָרֵשׁ אֶת ׳שְׂמַח בָּחוּר׳: ׳אַתָּה בָּחוּר עַתָּה, שְׂמַח בַּטּוֹבָה שֶׁהָיְתָה לְךָ בִּימֵי יַלְדוּתְךָ, וְגַם עַתָּה בִּימֵי בְחוּרוֹתֶיךָ עֲשֵׂה טוֹב לְלִבְּךָ וְתִהְיֶה שָׂמֵחַ... אֲבָל דַּע כִּי עַל כָּל אֵלֶּה הַמַּעֲשִׂים יְבִיאֲךָ הָאֱלֹהִים בַּמִּשְׁפָּט׳. הָעֵצָה: שִׂמְחָה כֵּן, אֲבָל בְּאַחֲרָיוּת.

מְצוּדַת דָּוִד: שִׂמְחָה בַּנְּעוּרִים – עִם הַכָּרָה ש׳עַל כָּל אֵלֶּה יְבִיאֲךָ הָאֱלֹהִים בַּמִּשְׁפָּט׳.

1

📜 הַפָּסוּקוְהָסֵר כַּעַס מִלִּבֶּךָ וְהַעֲבֵר רָעָה מִבְּשָׂרֶךָ כִּי הַיַּלְדוּת וְהַשַּׁחֲרוּת הָבֶל

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לֵב רַךְ שֶׁמְּקָרֵב אֶת ה׳ וְלֹא מַכְעִיס אוֹתוֹ 💛

📖 קֹהֶלֶת, פֶּרֶק י״א, פָּסוּק י׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּפְּסוּקִים

🌿 רַשִׁ״י: ׳הָסֵר כַּעַס׳

רַשִׁ״י מְפָרֵשׁ כָּל מִלָּה בְּפָסוּק י׳: ׳וְהָסֵר כַּעַס – דְּבָרִים הַמַּכְעִיסִים אֶת הַמָּקוֹם׳. ׳וְהַעֲבֵר רָעָה – יֵצֶר הָרָע׳. ׳מִבְּשָׂרֶךָ – שֶׁיִּהְיֶה לְךָ לֵב בָּשָׂר׳ (לֵב רַךְ). ׳הַשַּׁחֲרוּת – נַעֲרוּת, שֶׁרֹאשׁ אָדָם שָׁחוֹר בִּימֵי עֲלוּמָיו׳. כְּלוֹמַר, ׳כַּעַס׳ הוּא מַה שֶּׁמַּכְעִיס אֶת ה׳, וּמוֹסִיפִים לֵב רַךְ וְרָגִישׁ.

רַשִׁ״י: ׳הָסֵר כַּעַס׳ הוּא לְהָסִיר דְּבָרִים הַמַּכְעִיסִים אֶת ה׳, וְלִהְיוֹת ׳לֵב בָּשָׂר׳.

2

📜 הַפָּסוּקוּזְכֹר אֶת בּוֹרְאֶיךָ בִּימֵי בְּחוּרֹתֶיךָ עַד

💛 אַהֲבַת הַשֵּׁם: שְׁלוֹשָׁה דְּבָרִים שֶׁאֲנִי זוֹכֵר – וְכָךְ אֲנִי שׁוֹמֵר עַל עַצְמִי מֵעֲבֵרָה 💛

📖 קֹהֶלֶת, פֶּרֶק י״ב, פָּסוּק א׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּפְּסוּקִים

3️⃣ רַשִׁ״י: שְׁלוֹשָׁה דְּבָרִים

רַשִׁ״י מֵבִיא מִמִּשְׁנָה (אָבוֹת ג׳, א׳): ׳עֲקַבְיָא בֶּן מַהֲלַלְאֵל אוֹמֵר: הִסְתַּכֵּל בִּשְׁלוֹשָׁה דְּבָרִים וְאֵין אַתָּה בָּא לִידֵי עֲבֵרָה׳. וְהוּא דָּרַשׁ מִן הַפָּסוּק ׳וּזְכֹר אֶת בּוֹרְאֶיךָ׳ אֶת הַשְּׁלוֹשָׁה: ׳דַּע מֵאַיִן בָּאתָ, וּלְאָן אַתָּה הוֹלֵךְ, וְלִפְנֵי מִי אַתָּה עָתִיד לִתֵּן דִּין וְחֶשְׁבּוֹן׳. שְׁלוֹשָׁה דְּבָרִים לִזְכֹּר: מֵאֵיפֹה בָּאתִי, לְאָן אֲנִי הוֹלֵךְ, וְלִפְנֵי מִי אֲנִי נוֹתֵן דִּין וְחֶשְׁבּוֹן.

רַשִׁ״י: ׳וּזְכֹר אֶת בּוֹרְאֶיךָ׳ – דַּע מֵאַיִן בָּאתָ, וּלְאָן אַתָּה הוֹלֵךְ, וְלִפְנֵי מִי אַתָּה נוֹתֵן דִּין וְחֶשְׁבּוֹן.

3

📜 הַפָּסוּקאֲשֶׁר לֹא יָבֹאוּ יְמֵי הָרָעָה וְהִגִּיעוּ שָׁנִים אֲשֶׁר

💛 אַהֲבַת הַשֵּׁם: הַקֵּרוּב אֶל ה׳ שֶׁאֲנִי בּוֹנֶה עַכְשָׁיו הוּא עוֹגֶן לְכָל חַיַּי 💛

📖 קֹהֶלֶת, פֶּרֶק י״ב, פָּסוּק א׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּפְּסוּקִים

📆 רַשִׁ״י: ׳יְמֵי הָרָעָה׳

רַשִׁ״י מְבָאֵר בְּכַמָּה אֳפָנִים אֶת ׳יְמֵי הָרָעָה׳: ׳יְמֵי הַזִּקְנָה וְהַחַלָּשׁוּת׳, וּבְפֵרוּשׁ אַחֵר: ׳אֵלּוּ יְמֵי הַגּוֹלָה׳. אֶפְשָׁר לְהָבִין: הַזְּמַן הַנָּכוֹן לְהִתְקָרֵב לה׳ הוּא קֹדֶם שֶׁיָּבוֹאוּ יָמִים קָשִׁים, אִישִׁיִּים אוֹ לְאֻמִּיִּים. מִי שֶׁבָּנָה קֵרוּב לה׳ בַּיָּמִים הַטּוֹבִים, יִהְיֶה לוֹ עוֹגֶן בַּיָּמִים הַקָּשִׁים.

רַשִׁ״י: ״יְמֵי הָרָעָה״ הֵם יְמֵי הַזִּקְנָה (וּבְפֵרוּשׁ אַחֵר – יְמֵי הַגּוֹלָה); בּוֹנִים קֵרוּב לה׳ לִפְנֵי שֶׁהֵם בָּאִים.

4

📜 הַפָּסוּקעַד אֲשֶׁר לֹא תֶחְשַׁךְ הַשֶּׁמֶשׁ וְהָאוֹר וְהַיָּרֵחַ וְהַכּוֹכָבִים וְשָׁבוּ הֶעָבִים אַחַר הַגָּשֶׁם

💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה וְהָרַבָּנִים הֵם אוֹרוֹת שֶׁמְּאִירִים לִי אֶת הַדֶּרֶךְ 💛

📖 קֹהֶלֶת, פֶּרֶק י״ב, פָּסוּק ב׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּפְּסוּקִים

🌞 רַשִׁ״י: ׳הַשֶּׁמֶשׁ וְהָאוֹר׳

רַשִׁ״י מַסְבִּיר אֶת ׳עַד אֲשֶׁר לֹא תֶחְשַׁךְ הַשֶּׁמֶשׁ׳ בְּדֶרֶךְ דְּרַשׁ: ׳זוֹ מַלְכוּת בֵּית דָּוִד, שֶׁנֶּאֱמַר: וְכִסְאוֹ כַשֶּׁמֶשׁ נֶגְדִּי. וְהָאוֹר – זוֹ תּוֹרָה, שֶׁנֶּאֱמַר: כִּי נֵר מִצְוָה וְתוֹרָה אוֹר. וְהַיָּרֵחַ – זוֹ סַנְהֶדְרִין. וְהַכּוֹכָבִים – אֵלּוּ הָרַבָּנִים, שֶׁנֶּאֱמַר: וּמַצְדִּיקֵי הָרַבִּים כַּכּוֹכָבִים׳. הַשֶּׁמֶשׁ, הָאוֹר, הַיָּרֵחַ וְהַכּוֹכָבִים הֵם מַלְכוּת, תּוֹרָה, סַנְהֶדְרִין וְרַבָּנִים.

רַשִׁ״י: הַשֶּׁמֶשׁ – מַלְכוּת בֵּית דָּוִד; הָאוֹר – תּוֹרָה; הַיָּרֵחַ – סַנְהֶדְרִין; הַכּוֹכָבִים – הָרַבָּנִים.

5

📜 הַפָּסוּקעַד אֲשֶׁר יֵרָתֵק חֶבֶל הַכֶּסֶף וְתָרֻץ גֻּלַּת הַזָּהָב

💛 אַהֲבַת הַשֵּׁם: אֲנִי חוּט בְּשַׁלְשֶׁלֶת הַמָּסֹרֶת, וְרוֹצֶה לִשְׁמֹר עָלֶיהָ 💛

📖 קֹהֶלֶת, פֶּרֶק י״ב, פָּסוּק ו׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּפְּסוּקִים

📜 רַשִׁ״י: ׳וְיִסְתַּבֵּל׳ – שְׁרֶשֶׁת יוֹחֲסִין

רַשִׁ״י בְּפֵרוּשׁ שֶׁל מִדְרַשׁ אֵיכָה מוֹסִיף: ׳עַד אֲשֶׁר לֹא יֵרָתֵק חֶבֶל הַכֶּסֶף – זוֹ שַׁלְשֶׁלֶת יוֹחֲסִין׳. ׳וְתָרֻץ גֻּלַּת הַזָּהָב – אֵלּוּ דִבְרֵי תוֹרָה, שֶׁנֶּאֱמַר: הַנֶּחֱמָדִים מִזָּהָב׳. כְּלוֹמַר, בַּפֵּרוּשׁ הַמִּדְרָשִׁי, הַחֶבֶל הוּא הַמָּסֹרֶת שֶׁעוֹבֶרֶת מֵאָב לְבֵן, וְהַזָּהָב הוּא הַתּוֹרָה. מִי שֶׁשּׁוֹמֵר עַל הַמָּסֹרֶת – הַחֶבֶל לֹא נִקְרָע.

רַשִׁ״י: ׳חֶבֶל הַכֶּסֶף׳ – שַׁלְשֶׁלֶת הַיִּחוּס; ׳גֻּלַּת הַזָּהָב׳ – דִּבְרֵי תּוֹרָה.

6

📜 הַפָּסוּקוְיָשֹׁב הֶעָפָר עַל הָאָרֶץ כְּשֶׁהָיָה וְהָרוּחַ תָּשׁוּב אֶל הָאֱלֹהִים אֲשֶׁר נְתָנָהּ

💛 אַהֲבַת הַשֵּׁם: הַנְּשָׁמָה שֶׁלִּי יָצְאָה מֵה׳ וְתָשׁוּב אֵלָיו בְּאַהֲבָה 💛

📖 קֹהֶלֶת, פֶּרֶק י״ב, פָּסוּק ז׳ (מְצוּדַת דָּוִד)

רָמָה ב׳ – עִיּוּן בַּפְּסוּקִים

🕊️ רַשִׁ״י: ׳וְהָרוּחַ תָּשׁוּב׳

רַשִׁ״י מְפָרֵשׁ בְּפָסוּק ז׳: ׳וְהָרוּחַ תָּשׁוּב – זוֹ רוּחַ הַקֹּדֶשׁ, שֶׁכֵּיוָן שֶׁנִּסְתַּלְּקָה רוּחַ הַקֹּדֶשׁ, גָּלוּ׳ (בִּדְרָשָׁה עַל הַגָּלוּת). וּמְצוּדַת דָּוִד בְּפֵרוּשׁ הַפְּשָׁט: ׳הַנְּשָׁמָה תָּשׁוּב אֶל הַמָּקוֹר אֲשֶׁר נֶחְצְבָה מִמֶּנּוּ לְמַעְלָה׳. כָּךְ בִּשְׁנֵי הַפֵּרוּשִׁים: הָרוּחַ – מַה שֶּׁמֵּאִיר אֶת הַחַיִּים – שׁוֹבָה אֶל ה׳.

מְצוּדַת דָּוִד: ״הַנְּשָׁמָה תָּשׁוּב אֶל הַמָּקוֹר אֲשֶׁר נֶחְצְבָה מִמֶּנּוּ״.

מָה לוֹקְחִים מִרָמָה ב׳?

רַשִׁ״י מַרְחִיב: ׳בּוֹרְאֶיךָ׳ הוּא גַּם בֹּרְךָ וּבְאֵרְךָ, ׳יְמֵי הָרָעָה׳ הֵם הַזִּקְנָה (אוֹ הַגּוֹלָה), וְהַשֶּׁמֶשׁ וְהָאוֹר הֵם מַלְכוּת וְתוֹרָה. מְצוּדַת דָּוִד: שִׂמְחָה עִם אַחְרָיוּת.

וּזְכֹר אֶת בּוֹרְאֶיךָ בִּימֵי בְּחוּרֹתֶיךָ

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • ׳וּזְכֹר אֶת בּוֹרְאֶיךָ בִּימֵי בְּחוּרֹתֶיךָ׳ – מָה אֲנִי עוֹשֶׂה בַּיּוֹם כְּדֵי לִזְכֹּר אֶת ה׳, וּמָה אֶפְשָׁר לְהוֹסִיף?
  • הָרַב לִמֵּד שֶׁגִּיל שְׁלוֹשׁ עֶשְׂרֵה הוּא גִּיל הָאַהֲבָה. מָה אֲנִי מַרְגִּישׁ כְּלַפֵּי ה׳, וְאֵיךְ אֲנִי מְבַטֵּא אֶת הָאַהֲבָה הַזֹּאת?
  • רַשִׁ״י הֵבִיא שְׁלוֹשָׁה דְּבָרִים לִזְכֹּר: מֵאֵיפֹה בָּאתִי, לְאָן אֲנִי הוֹלֵךְ, לִפְנֵי מִי אֲנִי נוֹתֵן דִּין וְחֶשְׁבּוֹן. אֵיךְ זֶה עוֹזֵר לִי לִבְחֹר נָכוֹן?

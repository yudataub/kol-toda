וּזְכֹר אֶת בּוֹרְאֶיךָ

רָמָה א׳ – הַנְּעוּרִים וְהַבּוֹרֵא

שְׂמַח בַּנְּעוּרִים, וּזְכֹר אֶת בּוֹרְאֶיךָ כְּבָר עַכְשָׁיו – לִפְנֵי שֶׁיָּבוֹאוּ יְמֵי הָרָעָה, כְּשֶׁהָאוֹר עוֹד בָּהִיר

📖 פְּרָקִים י״א–י״ב

🎯 לַחִידוֹן

💛 אַהֲבַת הַשֵּׁם: ה׳ רוֹצֶה שֶׁאֶשְׂמַח בַּנְּעוּרִים, וְשֶׁהַשִּׂמְחָה תִּהְיֶה שִׂמְחָה שֶׁל טוֹב 💛

רָמָה א׳ – הַנְּעוּרִים וְהַבּוֹרֵא

🌟 ׳שְׂמַח בָּחוּר בְּיַלְדוּתֶיךָ׳

שְׁלֹמֹה פּוֹנֶה אֶל הַצְּעִירִים: ׳שְׂמַח בָּחוּר בְּיַלְדוּתֶיךָ וִיטִיבְךָ לִבְּךָ בִּימֵי בְחוּרוֹתֶךָ׳. שִׂמְחוּ בַּנְּעוּרִים! זֶה זְמַן שֶׁל כֹּחַ וְשֶׁל חֲלוֹמוֹת. אֲבָל הוּא מוֹסִיף: ׳וְדָע כִּי עַל כָּל אֵלֶּה יְבִיאֲךָ הָאֱלֹהִים בַּמִּשְׁפָּט׳. כָּל מַעֲשֶׂה חָשׁוּב, וְהַשִּׂמְחָה הַנְּכוֹנָה הִיא שִׂמְחָה שֶׁל מַעֲשִׂים טוֹבִים.

שִׂמְחוּ בַּנְּעוּרִים – וְזִכְרוּ שֶׁכָּל מַעֲשֶׂה חָשׁוּב לִפְנֵי ה׳.

🌿 ׳הָסֵר כַּעַס׳

׳וְהָסֵר כַּעַס מִלִּבֶּךָ וְהַעֲבֵר רָעָה מִבְּשָׂרֶךָ כִּי הַיַּלְדוּת וְהַשַּׁחֲרוּת הָבֶל׳. כְּשֶׁאָדָם צָעִיר, יֵשׁ לוֹ כֹּחוֹת וְגַם רְגָשׁוֹת חֲזָקִים. שְׁלֹמֹה מַזְהִיר: אַל תִּתֵּן לַכַּעַס וְלַתַּאֲווֹת לְנַהֵל אוֹתְךָ. הַנְּעוּרִים עוֹבְרִים מַהֵר – כְּדַאי לְנַצֵּל אוֹתָם לְטוֹב.

הַנְּעוּרִים עוֹבְרִים מַהֵר – לָכֵן מְסִירִים כַּעַס וּבוֹחֲרִים בַּטּוֹב.

🧒 ׳וּזְכֹר אֶת בּוֹרְאֶיךָ׳

הַמִּשְׁפָּט הַמֶּרְכָּזִי: ׳וּזְכֹר אֶת בּוֹרְאֶיךָ בִּימֵי בְּחוּרֹתֶיךָ׳. זוֹכְרִים אֶת ה׳ דַּוְקָא עַכְשָׁיו, כְּשֶׁאֲנִי צָעִיר וְחָזָק, וְלֹא מְחַכִּים לַזִּקְנָה. כְּשֶׁמַּתְחִילִים מֻקְדָּם, הַקֵּרוּב אֶל ה׳ גָּדֵל יַחַד עִם הָאָדָם.

זוֹכְרִים אֶת ה׳ כְּבָר בַּנְּעוּרִים – לֹא מְחַכִּים לְמָחָר.

💛 אַהֲבַת הַשֵּׁם: ה׳ נָתַן לִי כֹּחַ וְחֵפֶץ, וַאֲנִי מְנַצֵּל אוֹתָם לִלְמֹד וּלְהִתְקָרֵב 💛

⏳ ׳עַד אֲשֶׁר לֹא יָבֹאוּ יְמֵי הָרָעָה׳

הַפָּסוּק אוֹמֵר: זְכֹר ׳עַד אֲשֶׁר לֹא יָבֹאוּ יְמֵי הָרָעָה וְהִגִּיעוּ שָׁנִים אֲשֶׁר תֹּאמַר אֵין לִי בָהֶם חֵפֶץ׳. כְּשֶׁזּוֹקְנִים, קָשֶׁה יוֹתֵר לְהַתְחִיל דְּבָרִים חֲדָשִׁים. הַזְּמַן הַטּוֹב לִלְמֹד וְלִהְתְקָרֵב הוּא עַכְשָׁיו, כְּשֶׁהַכֹּחוֹת וְהַחֵפֶץ בָּאִים בְּקַלּוּת.

הַזְּמַן הֲכִי טוֹב לְהִתְקָרֵב אֶל ה׳ הוּא עַכְשָׁיו, בַּנְּעוּרִים.

׳עַד אֲשֶׁר לֹא תֶחְשַׁךְ הַשֶּׁמֶשׁ וְהָאוֹר וְהַיָּרֵחַ וְהַכּוֹכָבִים וְשָׁבוּ הֶעָבִים אַחַר הַגָּשֶׁם׳. שְׁלֹמֹה מְצַיֵּר תְּמוּנָה שֶׁל אוֹר שֶׁמִּתְעַמְעֵם. הַכַּוָּנָה: כְּדַאי לְנַצֵּל אֶת הָאוֹר כְּשֶׁהוּא בָּהִיר. אֲנִי בְּאוֹר, וַאֲנִי רוֹצֶה לִלְמֹד וְלַעֲשׂוֹת טוֹב כְּשֶׁהַכֹּל בָּהִיר.

כְּשֶׁהָאוֹר בָּהִיר – מְנַצְּלִים אוֹתוֹ לִלְמֹד וְלַעֲשׂוֹת טוֹב.

🏺 הַחֶבֶל וְהַכַּד

הַפֶּרֶק מַמְשִׁיךְ בְּדִמּוּיִים: ׳עַד אֲשֶׁר לֹא יֵרָתֵק חֶבֶל הַכֶּסֶף וְתָרֻץ גֻּלַּת הַזָּהָב וְתִשָּׁבֵר כַּד עַל הַמַּבּוּעַ׳. הַחַיִּים הֵם כְּמוֹ בְּאֵר: חֶבֶל, כַּד, מַיִם. הַכֹּל שָׁבִיר וְיָקָר. לָכֵן מְטַפְּלִים בְּכָל דָּבָר כְּאִלּוּ הוּא אוֹצָר.

הַחַיִּים שְׁבִירִים וִיקָרִים – מְטַפְּלִים בָּהֶם בִּזְהִירוּת וּבְהוֹדָיָה.

💛 אַהֲבַת הַשֵּׁם: הַנְּשָׁמָה שֶׁלִּי מַתָּנָה מֵה׳ וַאֲנִי מוֹדֶה עָלֶיהָ בְּכָל בֹּקֶר 💛

🌍 ׳וְיָשֹׁב הֶעָפָר׳

׳וְיָשֹׁב הֶעָפָר עַל הָאָרֶץ כְּשֶׁהָיָה וְהָרוּחַ תָּשׁוּב אֶל הָאֱלֹהִים אֲשֶׁר נְתָנָהּ׳. הַגּוּף בָּא מֵעָפָר, וְהַנְּשָׁמָה בָּאָה מֵה׳ וְחוֹזֶרֶת אֵלָיו. זֶה לֹא פַּחַד – זֶה מָקוֹר שֶׁל אֲהֲבָה: הַנְּשָׁמָה שֶׁלִּי נִתְּנָה לִי מֵה׳ וְהִיא שֶׁלּוֹ.

הַנְּשָׁמָה שֶׁלִּי נִתְּנָה לִי מֵה׳ וְחוֹזֶרֶת אֵלָיו.

💛 אַהֲבַת הַשֵּׁם: כָּל יוֹם אֲנִי עוֹשֶׂה צַעַד קָטָן אֶל ה׳ 💛

🎯 מָה לַעֲשׂוֹת עַכְשָׁיו

מָה אֲנִי עוֹשֶׂה? זוֹכֵר אֶת בּוֹרְאִי בְּכָל בֹּקֶר: ״מוֹדֶה אֲנִי״. מִתְפַּלֵּל בְּכַוָּנָה. לוֹמֵד כָּל יוֹם קְצָת תּוֹרָה. עוֹזֵר לְמִישֶׁהוּ. אֵלֶּה צְעָדִים קְטַנִּים שֶׁמְּקָרְבִים אוֹתִי אֶל ה׳, וְהֵם נִבְנִים כְּשֶׁאֲנִי צָעִיר.

צְעָדִים קְטַנִּים כָּל יוֹם בַּנְּעוּרִים בּוֹנִים קֵרוּב גָּדוֹל אֶל ה׳.

מָה לוֹקְחִים מִרָמָה א׳?

שְׁלֹמֹה פּוֹנֶה אֶל הַצְּעִירִים: שִׂמְחוּ, וְזִכְרוּ אֶת בּוֹרַאֲכֶם עַכְשָׁיו. הַנְּעוּרִים הֵם הַזְּמַן הֲכִי טוֹב לְהִתְקָרֵב אֶל ה׳. ה׳ אוֹהֵב כָּל נַעַר וְנַעֲרָה, וּמַמְתִּין לַקֵּרוּב שֶׁלָּנוּ.

🤔 חוֹשְׁבִים עַל זֶה

  • ׳וּזְכֹר אֶת בּוֹרְאֶיךָ בִּימֵי בְּחוּרֹתֶיךָ׳ – מָה אֲנִי עוֹשֶׂה בַּיּוֹם כְּדֵי לִזְכֹּר אֶת ה׳, וּמָה אֶפְשָׁר לְהוֹסִיף?
  • הָרַב לִמֵּד שֶׁגִּיל שְׁלוֹשׁ עֶשְׂרֵה הוּא גִּיל הָאַהֲבָה. מָה אֲנִי מַרְגִּישׁ כְּלַפֵּי ה׳, וְאֵיךְ אֲנִי מְבַטֵּא אֶת הָאַהֲבָה הַזֹּאת?
  • רַשִׁ״י הֵבִיא שְׁלוֹשָׁה דְּבָרִים לִזְכֹּר: מֵאֵיפֹה בָּאתִי, לְאָן אֲנִי הוֹלֵךְ, לִפְנֵי מִי אֲנִי נוֹתֵן דִּין וְחֶשְׁבּוֹן. אֵיךְ זֶה עוֹזֵר לִי לִבְחֹר נָכוֹן?

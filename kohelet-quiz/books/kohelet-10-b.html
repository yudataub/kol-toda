סוֹף דָּבָר הַכֹּל נִשְׁמָע

רָמָה ב׳ – עִיּוּן בַּפְּסוּקִים

אַחֲרֵי כָּל הַחֲקִירוֹת שְׁלֹמֹה מְסַכֵּם: ׳אֶת הָאֱלֹהִים יְרָא וְאֶת מִצְוֹתָיו שְׁמוֹר כִּי זֶה כָּל הָאָדָם׳

📖 פֶּרֶק י״ב, פְּסוּקִים ט׳–י״ד

🎯 לַחִידוֹן
👂

📜 הַפָּסוּקוְיֹתֵר שֶׁהָיָה קֹהֶלֶת חָכָם עוֹד לִמַּד דַּעַת אֶת הָעָם וְאִזֵּן וְחִקֵּר תִּקֵּן מְשָׁלִים הַרְבֵּה

💛 אַהֲבַת הַשֵּׁם: תַּקָּנוֹת הַחֲכָמִים עוֹזְרוֹת לִי לֶאֱחֹז בַּתּוֹרָה בְּאַהֲבָה 💛

📖 קֹהֶלֶת, פֶּרֶק י״ב, פָּסוּק ט׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּפְּסוּקִים

👂 רַשִׁ״י: ׳וְאִזֵּן׳ – אָזְנַיִם לַתּוֹרָה

רַשִׁ״י מְבָאֵר אֶת ׳וְאִזֵּן׳ בְּמָשָׁל: ׳עָשָׂה אָזְנַיִם לַתּוֹרָה, כְּקֻפָּה זוֹ שֶׁאֵין לָהּ אָזְנַיִם לֶאֱחֹז בָּהּ, וּבָא וְעָשָׂה לָהּ אָזְנַיִם, שֶׁתִּקֵּן עֵרוּבִין לְסַיָּג שְׁמִירַת שַׁבָּת, וּנְטִילַת יָדַיִם סַיָּג לְטָהֳרָה׳. שְׁלֹמֹה תִּקֵּן תַּקָּנוֹת שֶׁעוֹזְרוֹת לֶאֱחֹז בַּתּוֹרָה, כְּמוֹ ״אָזְנַיִם״ לְסַל. עֵרוּב וּנְטִילַת יָדַיִם שׁוֹמְרִים עַל הַשַּׁבָּת וְעַל הַטָּהֳרָה.

רַשִׁ״י: שְׁלֹמֹה ׳עָשָׂה אָזְנַיִם לַתּוֹרָה׳ – תִּקֵּן עֵרוּבִין וּנְטִילַת יָדַיִם, שֶׁעוֹזְרִים לֶאֱחֹז בָּהּ.

💛 אַהֲבַת הַשֵּׁם: יֵשׁ לִי גַּם תּוֹרָה שֶׁבִּכְתָב וְגַם תּוֹרָה שֶׁבְּעַל פֶּה, וַאֲנִי אוֹהֵב אוֹתָן 💛

📖 קֹהֶלֶת, פֶּרֶק י״ב, פָּסוּק י׳ (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּפְּסוּקִים

📜 רַשִׁ״י: ׳דִּבְרֵי חֵפֶץ׳

רַשִׁ״י מְפָרֵשׁ אֶת ׳בִּקֵּשׁ קֹהֶלֶת לִמְצֹא דִּבְרֵי חֵפֶץ׳: ׳דִּבְרֵי חֵפֶץ – הֲלָכָה לְמֹשֶׁה מִסִּינַי׳. וְ׳כָתוּב יֹשֶׁר – זֶה תּוֹרָה שֶׁבִּכְתָב וְהַנְּבִיאִים׳. כְּלוֹמַר, שְׁלֹמֹה חִפֵּשׂ אֶת הַמָּסֹרֶת שֶׁבְּעַל פֶּה (הֲלָכָה לְמֹשֶׁה מִסִּינַי) וְאֶת הַכְּתָב (תּוֹרָה וּנְבִיאִים). שְׁנֵי הַחֲלָקִים שֶׁל הַתּוֹרָה.

רַשִׁ״י: ׳דִּבְרֵי חֵפֶץ׳ הֵם תּוֹרָה שֶׁבְּעַל פֶּה; ׳כָּתוּב יֹשֶׁר׳ הֵם תּוֹרָה שֶׁבִּכְתָב וּנְבִיאִים.

📜 הַפָּסוּקדִּבְרֵי חֲכָמִים כַּדָּרְבֹנוֹת וּכְמַשְׂמְרוֹת נְטוּעִים

💛 אַהֲבַת הַשֵּׁם: דִּבְרֵי חֲכָמִים מְכַוְּנִים אוֹתִי, קְבוּעִים, וּמַפְרִים בְּחַיַּי 💛

רָמָה ב׳ – עִיּוּן בַּפְּסוּקִים

🧭 רַשִׁ״י: ׳דָּרְבָן׳ וְ׳מַסְמֵר׳

רַשִׁ״י מְבָאֵר אֶת הַדִּמּוּיִים: ׳מַה דָּרְבָן זֶה מְכַוֵּן אֶת הַפָּרָה לְתָלְמֶיהָ, כָּךְ דִּבְרֵיהֶם מְכַוְּנִים אֶת הָאָדָם לְדַרְכֵי חַיִּים׳. ׳מַה מַּסְמֵר זֶה קָבוּעַ, אַף דִּבְרֵיהֶם קְבוּעִים, וּמַה נְטִיעָה פָּרָה וְרָבָה, אַף דִּבְרֵיהֶם פָּרִים וְרָבִים לִמְצֹא בָהֶם טַעַם׳. הַחֲכָמִים הֵם כֹּחַ מְכַוֵּן, קָבוּעַ, וּמֵבִיא פֵּרוֹת.

רַשִׁ״י: דִּבְרֵי חֲכָמִים ׳מְכַוְּנִים׳ (דָּרְבָן), ׳קְבוּעִים׳ (מַסְמֵר) ו׳פָּרִים וְרָבִים׳ (נְטִיעָה).

📜 הַפָּסוּקבַּעֲלֵי אֲסֻפּוֹת נִתְּנוּ מֵרֹעֶה אֶחָד

💛 אַהֲבַת הַשֵּׁם: הַתּוֹרָה נִתְּנָה לָנוּ מֵה׳ דֶּרֶךְ משֶׁה, וַאֲנִי מוֹדֶה עַל כָּךְ 💛

📖 קֹהֶלֶת, פֶּרֶק י״ב, פָּסוּק י״א (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּפְּסוּקִים

🐑 רַשִׁ״י: ׳מֹשֶׁה מִפִּי הַגְּבוּרָה׳

רַשִׁ״י מְפָרֵשׁ אֶת ׳נִתְּנוּ מֵרֹעֶה אֶחָד׳: ׳כָּל דִּבְרֵיהֶם דִּבְרֵי אֱלֹהִים חַיִּים. הוּא אֲמָרָן, וְרוֹעֶה אֶחָד נְתָנָן – מֹשֶׁה מִפִּי הַגְּבוּרָה׳. כְּלוֹמַר, ׳הָרוֹעֶה הָאֶחָד׳ הוּא מֹשֶׁה שֶׁקִּבֵּל מִפִּי ה׳. וְעוֹד: ׳וּכְמַשְׂמְרוֹת – כְּתִיב בְּשִׂי״ן, שֶׁהַתּוֹרָה בְּעֶשְׂרִים וְאַרְבָּעָה סְפָרִים, וּכְמִנְיַן מִשְׁמְרוֹת כְּהֻנָּה וּלְוִיָּה׳.

רַשִׁ״י: ׳רוֹעֶה אֶחָד׳ הוּא מֹשֶׁה, שֶׁנָּתַן אֶת דִּבְרֵי אֱלֹהִים חַיִּים מִפִּי הַגְּבוּרָה.

📖 קֹהֶלֶת, פֶּרֶק י״ב, פָּסוּק י״ב (רַשִׁ״י)

רָמָה ב׳ – עִיּוּן בַּפְּסוּקִים

📚 רַשִׁ״י: אִם בָּאנוּ לִכְתֹּב

רַשִׁ״י מַסְבִּיר אֶת הַפָּסוּק ׳עֲשׂוֹת סְפָרִים הַרְבֵּה אֵין קֵץ׳: ׳אִם בָּאנוּ לִכְתֹּב, לֹא הִסְפַּקְנוּ׳. כְּלוֹמַר, לֹא הַכֹּל אֶפְשָׁר לִכְתֹּב. וְעַל ׳וְלַהַג הַרְבֵּה׳: ׳וְאִם בָּא לָתֵת לֵב לְגִרְסָא חֲבִילוֹת יוֹתֵר מִמַּה שֶּׁאֵין הַלֵּב מַשִּׂיג, יְגִיעָה הִיא לַבְּרִיּוֹת... וְלֹא יֹאמַר: הוֹאִיל וְלֹא אוּכַל לִגְמֹר הַמְּלָאכָה, לָמָּה אַתְחִיל?׳. עֲדַיִן – לֹא מְוַתְּרִים: לוֹמְדִים כְּפִי הַיְּכֹלֶת.

רַשִׁ״י: אַל תֹּאמַר ״הוֹאִיל וְלֹא אוּכַל לִגְמֹר – לָמָּה אַתְחִיל?״ – לוֹמְדִים כְּפִי הַיְּכֹלֶת.

רָמָה ב׳ – עִיּוּן בַּפְּסוּקִים

👁️ רַשִׁ״י: ׳עַל כָּל נֶעְלָם׳

רַשִׁ״י מְבָאֵר אֶת הַפָּסוּק הָאַחֲרוֹן: ׳כִּי אֶת כָּל מַעֲשֶׂה אֲשֶׁר אָדָם עוֹשֶׂה, יָבִיא הָאֱלֹהִים בְּמִשְׁפָּט׳. ׳עַל כָּל נֶעְלָם – אֲפִלּוּ עַל הַשּׁוֹגֵג׳. ׳אִם טוֹב וְאִם רָע – אֲפִלּוּ נִכְשַׁל בְּמִצְוָה, כְּגוֹן נוֹתֵן צְדָקָה לְעָנִי בְּפַרְהֶסְיָא׳. כְּלוֹמַר, ה׳ שׁוֹקֵל אֶת הַכַּוָּנָה: מִצְוָה צְרִיכָה לְהֵעָשׂוֹת בִּצְנִיעוּת, כְּדֵי לֹא לְבַיֵּשׁ אֶת הֶעָנִי.

רַשִׁ״י: ׳עַל כָּל נֶעְלָם׳ – אֲפִלּוּ עַל שׁוֹגֵג, וְגַם עַל מִצְוָה שֶׁנַּעֲשֵׂית שֶׁלֹּא כַּדִּין, כְּמוֹ צְדָקָה בְּפַרְהֶסְיָא.

רָמָה ב׳ – עִיּוּן בַּפְּסוּקִים

💪 רַשִׁ״י: ׳מַה שֶּׁתּוּכַל עֲשֵׂה׳

רַשִׁ״י מְפָרֵשׁ אֶת הַפָּסוּק הָאַחֲרוֹן: ׳סוֹף דָּבָר הַכֹּל נִשְׁמָע אֶת הָאֱלֹהִים יְרָא – מַה שֶּׁתּוּכַל עֲשֵׂה, וְלִבְּךָ לַשָּׁמָיִם׳. וְעוֹד: ׳וְאֶת מִצְוֹתָיו שְׁמוֹר כִּי זֶה כָּל הָאָדָם – כִּי לַדָּבָר הַזֶּה נִבְרָא כָּל הָאָדָם׳. עוֹשִׂים מַה שֶּׁיָּכוֹל – וְהַלֵּב לַשָּׁמַיִם. וְכָל אָדָם נִבְרָא לָזֶה.

רַשִׁ״י: ׳מַה שֶּׁתּוּכַל עֲשֵׂה, וְלִבְּךָ לַשָּׁמָיִם׳ – כָּל אָדָם נִבְרָא לָזֶה.

מָה לוֹקְחִים מִרָמָה ב׳?

רַשִׁ״י מְגַלֶּה: שְׁלֹמֹה עָשָׂה אָזְנַיִם לַתּוֹרָה, דִּבְרֵי חֵפֶץ הֵם תּוֹרָה שֶׁבְּעַל פֶּה, דִּבְרֵי חֲכָמִים מְכַוְּנִים כְּדָרְבָן, וְרוֹעֶה אֶחָד – מֹשֶׁה. וְהַפָּסוּק הָאַחֲרוֹן: ׳מַה שֶּׁתּוּכַל עֲשֵׂה, וְלִבְּךָ לַשָּׁמָיִם׳.

סוֹף דָּבָר הַכֹּל נִשְׁמָע אֶת הָאֱלֹהִים יְרָא וְאֶת מִצְוֹתָיו שְׁמוֹר כִּי זֶה כָּל הָאָדָם

סוֹף רָמָה ב׳

🤔 חוֹשְׁבִים עַל זֶה

  • שְׁלֹמֹה סִיֵּם בְּ׳סוֹף דָּבָר׳. אִם הָיִיתִי צָרִיךְ לְסַכֵּם אֶת מַה שֶּׁלָּמַדְתִּי בְּקֹהֶלֶת בְּמִשְׁפָּט אֶחָד, מָה הָיִיתִי אוֹמֵר?
  • רַשִׁ״י לִמֵּד ׳מַה שֶּׁתּוּכַל עֲשֵׂה, וְלִבְּךָ לַשָּׁמָיִם׳. מָה אֲנִי יָכוֹל לַעֲשׂוֹת הַיּוֹם בְּשִׂמְחָה, וּמָה הַלֵּב שֶׁלִּי רוֹצֶה?
  • חֲזַ״ל אָמְרוּ שֶׁכָּל הָעוֹלָם נִבְרָא בִּשְׁבִיל מִי שֶׁיָּרֵא שָׁמַיִם. אֵיךְ זֶה גּוֹרֵם לִי לְהַרְגִּישׁ, וְעַל מָה אֲנִי מוֹדֶה לה׳?

מָה נִשְׁאָר אַחֲרֵי הָאָדָם? שֵׁם טוֹב – נִבְנֶה מִמַּעֲשִׂים קְטַנִּים, מִתּוֹכֵחָה בְּאַהֲבָה וּמִסַּבְלָנוּת, וְעוֹלֶה לִפְנֵי ה׳

🫒 ׳טוֹב שֵׁם מִשֶּׁמֶן טוֹב׳

׳טוֹב שֵׁם מִשֶּׁמֶן טוֹב׳. שֶׁמֶן טוֹב נוֹתֵן רֵיחַ נָעִים שֶׁמִּתְפַּשֵּׁט, אֲבָל שֵׁם טוֹב הוּא יוֹתֵר: כְּשֶׁאָדָם נוֹהֵג יָפֶה, אֲנָשִׁים מְסַפְּרִים עָלָיו דְּבָרִים טוֹבִים, וְהַשֵּׁם שֶׁלּוֹ נוֹסֵעַ לְמֶרְחַקִּים. הוּא רֵיחַ שֶׁנִּשְׁאָר גַּם אַחֲרֵי שֶׁהוּא כְּבָר לֹא פֹּה.

שֵׁם טוֹב הוּא רֵיחַ נָעִים שֶׁנִּשְׁאָר – גַּם כְּשֶׁהָאָדָם כְּבָר לֹא פֹּה.

🏅

💛 אַהֲבַת הַשֵּׁם: כָּל מַעֲשֶׂה טוֹב שֶׁלִּי מְקַדֵּשׁ אֶת שֵׁם ה׳ בָּעוֹלָם 💛

רָמָה א׳ – הַשֵּׁם הַטּוֹב

🏅 אֵיךְ בּוֹנִים שֵׁם טוֹב

שֵׁם טוֹב לֹא קוֹנִים בְּכֶסֶף. בּוֹנִים אוֹתוֹ בִּמְעַט מַעֲשִׂים קְטַנִּים בְּכָל יוֹם: לוֹמַר תּוֹדָה, לַעֲזֹר, לְדַבֵּר יָפֶה, לְהִתְנַצֵּל. מִי שֶׁעוֹשֶׂה אֶת זֶה, הַשֵּׁם הַטּוֹב שֶׁלּוֹ גָּדֵל כְּמוֹ אִילָן.

שֵׁם טוֹב נִבְנֶה מִמַּעֲשִׂים קְטַנִּים וְטוֹבִים כָּל יוֹם.

2
🌹

💛 אַהֲבַת הַשֵּׁם: גְּמִילוּת חֶסֶד עִם אָבֵל מְקָרֶבֶת אוֹתִי לה׳ הַמְנַחֵם 💛

📖 קֹהֶלֶת, פֶּרֶק ז׳, פָּסוּק ב׳

רָמָה א׳ – הַשֵּׁם הַטּוֹב

🌹 בֵּית אֵבֶל

׳טוֹב לָלֶכֶת אֶל בֵּית אֵבֶל מִלֶּכֶת אֶל בֵּית מִשְׁתֶּה בַּאֲשֶׁר הוּא סוֹף כָּל הָאָדָם וְהַחַי יִתֵּן אֶל לִבּוֹ׳. כְּשֶׁהוֹלְכִים לְנַחֵם אָבֵל, הַלֵּב נִפְתַּח, וַאֲנִי מַרְגִּישׁ אֶת הַחֶסֶד. הוּא שְׁעַת הַשְׁרָאָה: מָה חָשׁוּב בֶּאֱמֶת, וּמָה אֲנִי רוֹצֶה שֶׁיִּזְכְּרוּ עָלַי.

הַהֲלִיכָה לְנַחֵם אָבֵל פּוֹתַחַת אֶת הַלֵּב לְמַה שֶּׁחָשׁוּב בֶּאֱמֶת.

3
😠

💛 אַהֲבַת הַשֵּׁם: ה׳ מוֹכִיחַ אוֹתִי מִתּוֹךְ אַהֲבָה, כְּדֵי שֶׁאֶשְׁתַּפֵּר 💛

📖 קֹהֶלֶת, פֶּרֶק ז׳, פָּסוּק ג׳

רָמָה א׳ – הַשֵּׁם הַטּוֹב

😠 ׳טוֹב כַּעַס מִשְּׂחֹק׳

׳טוֹב כַּעַס מִשְּׂחֹק כִּי בְרֹעַ פָּנִים יִיטַב לֵב׳. לִפְעָמִים הוֹרֶה אוֹ מוֹרֶה מַרְאֶה פָּנִים חֲמוּרוֹת כְּדֵי שֶׁנָּבִין שֶׁטָּעִינוּ, וְזֶה טוֹב יוֹתֵר מִצְּחוֹק שֶׁמַּשְׁאִיר אוֹתָנוּ בַּטָּעוּת. הַקֶּפִידָה בָּאָה מֵאַהֲבָה.

פָּנִים רְצִינִיּוֹת מִתּוֹךְ אַהֲבָה עוֹזְרוֹת לַלֵּב לְהִשְׁתַּפֵּר.

4
🧠

💛 אַהֲבַת הַשֵּׁם: אֲנִי שָׂמֵחַ לִשְׁמֹעַ תּוֹכָחָה, כִּי הִיא מְקָרֶבֶת אוֹתִי לַטּוֹב 💛

📖 קֹהֶלֶת, פֶּרֶק ז׳, פָּסוּק ה׳

רָמָה א׳ – הַשֵּׁם הַטּוֹב

🧠 ׳גַּעֲרַת חָכָם׳

׳טוֹב לִשְׁמֹעַ גַּעֲרַת חָכָם מֵאִישׁ שֹׁמֵעַ שִׁיר כְּסִילִים׳. כְּשֶׁחָכָם מַעֲיר לִי, זֶה לֹא נָעִים, אֲבָל זֶה טוֹב לִי. לְעֻמַּת זֹאת, ׳שִׁיר כְּסִילִים׳ שׁוֹמְעִים בְּנָחַת וְלֹא לוֹמְדִים מִמֶּנּוּ כְּלוּם.

הֶעָרָה שֶׁל חָכָם מְתַקֶּנֶת; מַחְמָאוֹת בְּלִי תוֹכֶן לֹא.

5
🏁

💛 אַהֲבַת הַשֵּׁם: אֲנִי רוֹצֶה לְסַיֵּם כָּל דָּבָר טוֹב בְּסַבְלָנוּת וּבְעֲנָוָה 💛

📖 קֹהֶלֶת, פֶּרֶק ז׳, פָּסוּק ח׳

רָמָה א׳ – הַשֵּׁם הַטּוֹב

🏁 ׳טוֹב אַחֲרִית דָּבָר׳

׳טוֹב אַחֲרִית דָּבָר מֵרֵאשִׁיתוֹ טוֹב אֶרֶךְ רוּחַ מִגְּבַהּ רוּחַ׳. מָה שֶׁחָשׁוּב הוּא לֹא רַק הַהַתְחָלָה, אֶלָּא הַסּוֹף. מִי שֶׁמַּתְחִיל וּמְסַיֵּם בְּטוֹב – הַדָּבָר גָּמוּר. וְ׳אֶרֶךְ רוּחַ׳, סַבְלָנוּת, טוֹב מִגֹּבַהּ רוּחַ וְגַאֲוָה.

סְיּוּם טוֹב חָשׁוּב מֵהַתְחָלָה, וְסַבְלָנוּת טוֹבָה מִגַּאֲוָה.

6
🌋

💛 אַהֲבַת הַשֵּׁם: אֲנִי לוֹמֵד מֵה׳ שֶׁהוּא אֶרֶךְ אַפַּיִם, וְרוֹצֶה לִהְיוֹת סַבְלָן 💛

רָמָה א׳ – הַשֵּׁם הַטּוֹב

🌋 ׳אַל תְּבַהֵל בְּרוּחֲךָ לִכְעוֹס׳

׳אַל תְּבַהֵל בְּרוּחֲךָ לִכְעוֹס כִּי כַעַס בְּחֵיק כְּסִילִים יָנוּחַ׳. כְּשֶׁעוֹלֶה כַּעַס, לֹא מְמַהֲרִים. סוֹפְרִים עַד עֶשֶׂר, נוֹשְׁמִים עָמֹק, וְחוֹשְׁבִים. הַכַּעַס מוֹצֵא לוֹ מָקוֹם אֵצֶל הַכְּסִילִים; הֶחָכָם לֹא נוֹתֵן לוֹ לָנוּחַ.

לֹא מְמַהֲרִים לִכְעֹס – עוֹצְרִים וְנוֹשְׁמִים לִפְנֵי שֶׁמְּדַבְּרִים.

🌟

💛 אַהֲבַת הַשֵּׁם: ה׳ אוֹהֵב כָּל אָדָם שֶׁעוֹשֶׂה שֵׁם טוֹב בָּעוֹלָם 💛

🌟 מָה נִשְׁאָר אַחֲרַי

הַפֶּרֶק נוֹתֵן לָנוּ מַבָּט מֵרָחוֹק: מָה יַגִּידוּ עָלַי כְּשֶׁאֶהְיֶה זָקֵן? מָה יֹאמְרוּ הַנְּכָדִים? שֵׁם טוֹב נוֹתֵן שִׂמְחָה גַּם עַכְשָׁיו, וְגַם בָּעֲתִיד. מִי שֶׁחַי כָּךְ, מַרְגִּישׁ שֶׁחַיָּיו טוֹבִים.

מִי שֶׁחַי בְּשֵׁם טוֹב, חַיָּיו בַּעֲלֵי מַשְׁמָעוּת.

מָה לוֹקְחִים מִרָמָה א׳?

שְׁלֹמֹה מְלַמֵּד שֶׁשֵּׁם טוֹב הוּא יוֹתֵר מִשֶּׁמֶן טוֹב: הוּא נִבְנֶה מִמַּעֲשִׂים קְטַנִּים, מִנִּחוּם אֲבֵלִים, מִתּוֹכֵחָה וּמִסַּבְלָנוּת. ה׳ אוֹהֵב כָּל אָדָם שֶׁעוֹשֶׂה שֵׁם טוֹב בָּעוֹלָם.

  • שֵׁם טוֹב נִבְנֶה מִמַּעֲשִׂים קְטַנִּים. אֵיזֶה מַעֲשֶׂה טוֹב קָטָן אֲנִי יָכוֹל לַעֲשׂוֹת הַיּוֹם שֶׁיּוֹסִיף לַשֵּׁם הַטּוֹב שֶׁלִּי?
  • ׳טוֹב לִשְׁמֹעַ גַּעֲרַת חָכָם׳ – מָתַי הִתְקַשִּׁיתִי לִשְׁמֹעַ תּוֹכָחָה, וּמָה אֶפְשָׁר לַעֲשׂוֹת כְּדֵי לְקַבֵּל אוֹתָהּ בְּאַהֲבָה?
